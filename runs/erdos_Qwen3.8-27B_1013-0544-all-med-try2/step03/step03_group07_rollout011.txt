    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)

            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

    return val, grad

    grad = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        if s == 0:
            grad[:] += (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] += (-h[s:])
        else:
            t = -s
            grad[:n-t] += (1.0 - h[t:])
            grad[t:] += (-h[:n-t])
    return grad / top_k * dx, corr

def optimize_topk(h, max_iters, lr0, decay, top_k, momentum=0.9):
    """Optimize using top-k subgradient with momentum."""
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    v = np.zeros(n)  # velocity for momentum
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        grad, corr = topk_subgradient(x, top_k, 2.0/n)
        v = momentum * v + (1 - momentum) * grad
        x_new = project_feasible(x - lr_t * v, target)
        val = compute_c5(x_new)
        if val < best_v:

            best_x = x_new.copy()
        x = x_new
    
    return best_x, best_v

def slsqp_refine(h, maxiter, beta):
    global _n_g, _beta_g
    _n_g = len(h)
    _beta_g = beta
    target = _n_g / 2.0
    bounds = Bounds(np.zeros(_n_g), np.ones(_n_g))
    
    def obj(x):

        return v
    def grad(x):
        _, g = smoothmax_val_grad(x, _beta_g)
        return g
    def con_f(x):
        return x.sum() - target
    def con_j(x):
        return np.ones(_n_g)
    
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

_n_g = 0
_beta_g = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-14:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -50, 0))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(max(sw, 1e-300)) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-13:
            continue
        s = k - (n - 1)
        if s == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] += wk * (-h[s:])
        else:
            t = -s
            grad[:n-t] += wk * (1.0 - h[t:])
            grad[t:] += wk * (-h[:n-t])
    grad *= dx
    return val, grad

def topk_grad(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        if s == 0:
            grad[:] += (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] += (-h[s:])
        else:
            t = -s
            grad[:n-t] += (1.0 - h[t:])
            grad[t:] += (-h[:n-t])
    return grad / top_k * dx, corr

def slsqp_refine(h, maxiter, beta):
    global _n_g, _beta_g
    _n_g = len(h)
    _beta_g = beta
    target = _n_g / 2.0
    bounds = Bounds(np.zeros(_n_g), np.ones(_n_g))
    one_vec = np.ones(_n_g)
    def obj(x):
        v, _ = smoothmax_val_grad(x, _beta_g)
        return v
    def gr(x):
        _, g = smoothmax_val_grad(x, _beta_g)
        return g
    def cf(x):
        return x.sum() - target
    def cj(x):
        return one_vec
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, jac=gr, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def momentum_topk(h, max_iters, lr0, decay, top_k, mu):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    v = np.zeros(n)
    best_x = x.copy()
    best_v = compute_c5(x)
    stall = 0
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        g, _ = topk_grad(x, top_k)
        v = mu * v + (1.0 - mu) * g
        x_new = project_feasible(x - lr_t * v, target)
        val = compute_c5(x_new)
        if val < best_v:
            best_v = val
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1
        x = x_new
        if stall > 500:
            break
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def structured_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    style = rng.randint(4)
    if style == 0:
        f = rng.randint(1, n // 4 + 1)
        t = np.arange(n)
        h = 0.5 + 0.3 * np.sin(2.0 * np.pi * f * t / n)
    elif style == 1:
        f = rng.randint(1, n // 4 + 1)
        t = np.arange(n)
        h = 0.5 + 0.3 * np.sign(np.sin(2.0 * np.pi * f * t / n))
    elif style == 2:
        block = max(2, n // rng.randint(4, 12))
        nblocks = (n + block - 1) // block
        vals = rng.uniform(0.2, 0.8, nblocks)
        h = np.repeat(vals, block)[:n] + rng.randn(n) * 0.05
    else:
        h = 0.5 + rng.randn(n) * 0.15
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)
    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: SLSQP warm-up at current resolution
    for beta in [80, 200, 500]:
        if tl() < 35:
            break
        h_new = slsqp_refine(h_best, 250, beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: Momentum top-k with adaptive k
    if tl() > 50:
        for tk in [4, 8, 16, 32, 64]:
            if tl() < 25:
                break
            h_sg, c_sg = momentum_topk(h_best, 4000, 0.04, 0.99992, tk, 0.88)
            if c_sg < best_c5:
                h_best, best_c5 = h_sg, c_sg
            if tl() > 20:
                h_new = slsqp_refine(h_best, 150, 400)
                c = compute_c5(h_new)
                if c < best_c5:
                    h_best, best_c5 = h_new, c

    # Phase 3: Multi-scale (go down, optimize, come back)
    for n_low in [100, 120, 150, 200]:
        if tl() < 80:
            break
        h_low = interp_to_n(h_best, n_low)
        h_low = slsqp_refine(h_low, 200, 200)
        h_back = interp_to_n(h_low, n)
        c_back = compute_c5(h_back)
        if c_back < best_c5:
            h_best, best_c5 = h_back, c_back
        h_new = slsqp_refine(h_back, 200, 600)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 4: Diverse starts at current n
    for trial in range(12):
        if tl() < 30:
            break
        s = seed + 100 + trial
        h_try = structured_init(n, s)
        # Quick subgradient warm
        h_try, _ = momentum_topk(h_try, 500, 0.06, 0.999, 8, 0.8)
        # SLSQP polish
        h_new = slsqp_refine(h_try, 250, 500)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 5: Higher resolution
    for n_try in [500, 600, 800]:
        if tl() < 100:
            break
        h_interp = interp_to_n(h_best, n_try)
        for beta in [100, 400, 1000]:
            if tl() < 35:
                break
            h_new = slsqp_refine(h_interp, 200, beta)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
                h_interp = h_best.copy()
        if tl() > 40:
            h_sg, c_sg = momentum_topk(h_best, 3000, 0.025, 0.99995, 16, 0.9)
            if c_sg < best_c5:
                h_best, best_c5 = h_sg, c_sg
            if tl() > 20:
                h_new = slsqp_refine(h_best, 200, 1000)
                c = compute_c5(h_new)
                if c < best_c5:
                    h_best, best_c5 = h_new, c
        n = len(h_best)

    # Phase 6: Final top-k push
    if tl() > 40:
        for tk in [6, 12, 24]:
            if tl() < 25:
                break
            h_sg, c_sg = momentum_topk(h_best, 5000, 0.015, 0.99997, tk, 0.92)
            if c_sg < best_c5:
                h_best, best_c5 = h_sg, c_sg

    # Phase 7: Final SLSQP polish
    if tl() > 20:
        for beta in [800, 2000, 5000]:
            if tl() < 12:
                break
            h_final = slsqp_refine(h_best, 350, beta)
            c = compute_c5(h_final)
            if c < best_c5:
                h_best, best_c5 = h_final, c

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
