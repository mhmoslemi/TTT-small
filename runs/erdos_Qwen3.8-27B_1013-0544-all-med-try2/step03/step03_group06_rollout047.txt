def smooth_max_obj_grad(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    shifted = np.clip(beta * (vals - m), -500, 500)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    weights = exp_shifted / wsum
    val = m + np.sum(weights * (vals - m))
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx] * dx
        if w < 1e-14:
            continue
        lag = idx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            v = n - lag
            grad[:v] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:v]
        else:
            ka = -lag
            v = n - ka
            grad[ka:] += w * (1.0 - h[:v])
            grad[:v] -= w * h[ka:]
    return val, grad


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def make_obj_func(dx, beta):
    def obj(x):
        v, _ = smooth_max_obj_grad(x, beta, dx)
        return v
    return obj


def make_jac_func(dx, beta):
    def jac(x):
        _, g = smooth_max_obj_grad(x, beta, dx)
        return g
    return jac


def make_con_func(target):
    def con(x):
        return x.sum() - target
    return con


def make_con_jac(n):
    def j(x):
        return np.ones(n)
    return j


def slsqp_optimize(h0, n_pts, target, dx, beta, maxiter=3000, ftol=1e-15):
    bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    obj = make_obj_func(dx, beta)
    jac = make_jac_func(dx, beta)
    con = make_con_func(target)
    cj = make_con_jac(n_pts)
    constraints = [{'type': 'eq', 'fun': con, 'jac': cj}]
    res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
    return project_h(res.x, target)


def subgrad_step(h, top_lags, target):
    n = len(h)
    grad = np.zeros(n)
    for lag in top_lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            v = n - lag
            grad[:v] += (1.0 - h[lag:])
            grad[lag:] -= h[:v]
        else:
            ka = -lag
            v = n - ka
            grad[ka:] += (1.0 - h[:v])
            grad[:v] -= h[ka:]
    gn = np.linalg.norm(grad)
    if gn < 1e-12:
        return None
    grad /= gn
    return grad


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tleft():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)

    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # Phase 1: Multi-resolution SLSQP with analytical gradient
    resolutions = [200, 250, 300, 400, 500, 600, 500, 400, 300, 200]
    for n_pts in resolutions:
        if tleft() < 50:
            break
        target = n_pts / 2.0
        dx = 2.0 / n_pts

        x_old = np.linspace(0, 2, len(cur_h))
        x_new = np.linspace(0, 2, n_pts)
        h_cur = project_h(np.interp(x_new, x_old, cur_h), target)
        c5_cur = compute_c5(h_cur)

        for beta in [15, 40, 100, 250, 600, 1500, 4000, 10000]:
            if tleft() < 12:
                break
            try:
                h_new = slsqp_optimize(h_cur, n_pts, target, dx, beta)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-15:
                    h_cur = h_new
                    c5_cur = c5_new
            except Exception:
                pass

        if c5_cur < best_c5:
            best_h = h_cur.copy()
            best_c5 = c5_cur

    # Phase 2: Subgradient descent
    n_final = len(best_h)
    target_final = n_final / 2.0
    no_imp = 0

    for it in range(40000):
        if tleft() < 15:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:5]
        top_lags = [int(i) - (n_final - 1) for i in top_idx]
        grad = subgrad_step(best_h, top_lags, target_final)
        if grad is None:
            break

        found = False
        for step in [0.06, 0.03, 0.015, 0.0075, 0.004, 0.002, 0.001]:
            ht = project_h(best_h - step * grad, target_final)
            c5_t = compute_c5(ht)
            if c5_t < best_c5:
                best_h = ht
                best_c5 = c5_t
                found = True
                no_imp = 0
                break

        if not found:
            no_imp += 1
            if no_imp > 150 and it % 100 == 99:
                pert = np.random.randn(n_final) * 0.012
                ht = project_h(best_h + pert, target_final)
                if compute_c5(ht) < best_c5:
                    best_h = ht
                    best_c5 = compute_c5(ht)
                    no_imp = 0
            if no_imp > 3000:
                break

    # Phase 3: Pairwise moves on critical lags
    for it in range(60000):
        if tleft() < 4:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:4]
        improved = False
        for idx in top_idx:
            lag = int(idx) - (n_final - 1)
            abs_lag = abs(lag)
            if abs_lag >= n_final or abs_lag == 0:
                continue
            nv = n_final - abs_lag
            for _ in range(4):
                i = np.random.randint(nv)
                j = i + abs_lag if lag > 0 else i - lag
                for d in [0.035, -0.035, 0.018, -0.018, 0.009, -0.009]:
                    ht = best_h.copy()
                    ht[i] += d
                    ht[j] -= d
                    if 0 <= ht[i] <= 1 and 0 <= ht[j] <= 1:
                        ht = project_h(ht, target_final)
                        c5_t = compute_c5(ht)
                        if c5_t < best_c5:
                            best_h = ht
                            best_c5 = c5_t
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        if not improved and it % 400 == 399:
            pert = np.random.randn(n_final) * 0.006
            ht = project_h(best_h + pert, target_final)
            if compute_c5(ht) < best_c5:
                best_h = ht
                best_c5 = compute_c5(ht)

    # Phase 4: Final high-precision SLSQP
    if tleft() > 12:
        n_p = len(best_h)
        tp = n_p / 2.0
        dxp = 2.0 / n_p
        try:
            for beta in [5000, 15000, 40000]:
                if tleft() < 6:
                    break
                h_new = slsqp_optimize(best_h, n_p, tp, dxp, beta, maxiter=5000, ftol=1e-16)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
        except Exception:
            pass

    # Phase 5: Final subgradient sweep
    for it in range(8000):
        if tleft() < 1:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:3]
        top_lags = [int(i) - (n_final - 1) for i in top_idx]
        grad = subgrad_step(best_h, top_lags, target_final)
        if grad is None:
            break
        for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            ht = project_h(best_h - step * grad, target_final)
            c5_t = compute_c5(ht)
            if c5_t < best_c5:
                best_h = ht
                best_c5 = c5_t
                break

    return best_h, best_c5, len(best_h)
