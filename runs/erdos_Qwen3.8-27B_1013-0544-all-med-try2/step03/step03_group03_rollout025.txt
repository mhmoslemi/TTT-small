    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_box_hyperplane(h, target_sum):
    """Project h onto {x : 0<=x[i]<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection on lambda
    lo, hi = -max(h) - 1, max(1 - h) + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        m = n - lag
        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + lag])

    else:

        m = n - la
        idx = np.arange(m)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

            h_new = project_box_hyperplane(y - lr * grad, target_sum)

        h = h_new
        t = t_new

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_bh(h, target_sum):
    n = len(h)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction for any tiny residual
    residual = target_sum - result.sum()
    if abs(residual) > 1e-15:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if mask.sum() > 0:
            result[mask] += residual / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def subgrad_lag(h, lag, n):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        m = n - lag
        grad[:m] += (1.0 - h[lag:])
        grad[lag:] -= h[:m]
    else:
        la = -lag
        m = n - la
        grad[la:] += (1.0 - h[:m])
        grad[:m] -= h[la:]
    return grad

def topk_subgrad(h, corr, n, k):
    top_idx = np.argsort(corr)[-k:]
    w = np.exp(6.0 * (corr[top_idx] - corr[top_idx].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for j, idx in enumerate(top_idx):
        lag = int(idx) - (n - 1)
        grad += w[j] * subgrad_lag(h, lag, n)
    return grad

    best_h = h.copy()
    stall = 0

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        mu = (t_param - 1.0) / t_new
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        corr = np.correlate(y, 1.0 - y, mode='full')
        # Adaptive k: use more lags when solution is good
        kk = min(k_top + it // 2000, 12)
        grad = topk_subgrad(y, corr, n, kk)
        
        h_new = project_bh(y - lr * grad, target_sum)
        c_new = compute_c5(h_new)
        
        if c_new > best_c5 + 0.003:
            lr *= 0.8
            h_new = project_bh(y - lr * grad, target_sum)
            c_new = compute_c5(h_new)
        
        h_prev = h
        h = h_new
        t_param = t_new

            best_c5 = c_new

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_bh(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - result.sum()
    if abs(residual) > 1e-14:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        if mask.sum() > 0:
            result[mask] += residual / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def subgrad_lag(h, lag, n):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        m = n - lag
        if m > 0:
            grad[lag:] += (1.0 - h[:m])
            grad[:m] -= h[lag:]
    else:
        p = -lag
        m = n - p
        if m > 0:
            grad[:m] += (1.0 - h[p:])
            grad[p:] -= h[:m]
    return grad

def topk_subgrad(h, corr, n, k):
    top_idx = np.argsort(corr)[-k:]
    w = np.exp(6.0 * (corr[top_idx] - corr[top_idx].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for j, idx in enumerate(top_idx):
        lag = int(idx) - (n - 1)
        grad += w[j] * subgrad_lag(h, lag, n)
    return grad

def accel_grad(h0, n, target_sum, max_iter, time_limit, t_start, k_top=4, lr_init=0.015):
    h = h0.copy()
    h_prev = h.copy()
    t_param = 1.0
    lr = lr_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        mu = (t_param - 1.0) / t_new
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode='full')
        kk = min(k_top + it // 3000, 10)
        grad = topk_subgrad(y, corr, n, kk)
        h_new = project_bh(y - lr * grad, target_sum)
        c_new = compute_c5(h_new)
        if c_new > best_c5 + 0.004:
            lr *= 0.75
            h_new = project_bh(y - lr * grad, target_sum)
            c_new = compute_c5(h_new)
        h_prev = h
        h = h_new
        t_param = t_new
        if c_new < best_c5:
            best_h = h.copy()
            best_c5 = c_new
            stall = 0
            lr = lr_init / (1.0 + 0.00005 * it)
        else:
            stall += 1
            if stall > 2000:
                h_prev = h.copy()
                t_param = 1.0
                stall = 0
    
    return best_h, best_c5

def slsqp_obj(x, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_val(corr, beta) * dx

def slsqp_eq(x, ts):
    return x.sum() - ts

def slsqp_eq_jac(x, ts):
    return np.ones(len(x))

                best_h = h_new.copy()
