def lbfgsb_optimize(h, beta, target, maxiter=500, top_k=20, penalty=500.0):
    """L-BFGS-B with penalty for equality constraint."""
    n = len(h)
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta, top_k)
        return v + penalty * (x.sum() - target) ** 2
    
    def objg(x):
        _, g = smoothmax_val_grad(x, beta, top_k)
        g = g + 2.0 * penalty * (x.sum() - target)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):

    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad


def pg_optimize(h, beta, target, maxiter=3000, top_k=10, lr=0.03):
    n = len(h)
    h = project_feasible(h.copy(), target)
    val, grad = smoothmax_val_grad(h, beta, top_k)
    for it in range(maxiter):
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-11:
            break
        d = -grad
        alpha = lr
        improved = False
        for _ in range(15):
            h_new = project_feasible(h + alpha * d, target)
            val_new, grad_new = smoothmax_val_grad(h_new, beta, top_k)
            if val_new <= val - 1e-5 * alpha * np.dot(d, h_new - h):
                h = h_new
                val = val_new
                grad = grad_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            h_new = project_feasible(h + alpha * d, target)
            val_new, grad_new = smoothmax_val_grad(h_new, beta, top_k)
            if val_new < val:
                h = h_new
                val = val_new
                grad = grad_new
            else:
                break
    return project_feasible(h, target)


def lbfgsb_optimize(h, beta, target, maxiter=400, top_k=20, penalty=1000.0):
    n = len(h)

    def obj(x):
        v, _ = smoothmax_val_grad(x, beta, top_k)
        return v + penalty * (x.sum() - target) ** 2

    def objg(x):
        _, g = smoothmax_val_grad(x, beta, top_k)
        g = g + 2.0 * penalty * (x.sum() - target)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-11})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def slsqp_optimize(h, beta, target, maxiter=400, top_k=15):
    n = len(h)

    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]

    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]

    from scipy.optimize import Bounds
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)

    return project_feasible(h_new, n_new / 2.0)


def coord_refine(h, target, max_rounds=50, rng=None):
    n = len(h)
    h = project_feasible(h.copy(), target)
    best = compute_c5(h)
    for r in range(max_rounds):
        improved = False
        order = rng.permutation(n) if rng else np.arange(n)
        for i in order:
            for delta in [0.02, -0.02, 0.04, -0.04, 0.01, -0.01]:
                h_trial = h.copy()
                h_trial[i] = min(1.0, max(0.0, h_trial[i] + delta))
                h_trial = project_feasible(h_trial, target)
                c = compute_c5(h_trial)
                if c < best - 1e-12:
                    h = h_trial
                    best = c
                    improved = True
                    break
        if not improved:
            break
    return h, best
