import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.ndimage import uniform_filter1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def sum_eq(x_in, target):
    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def thue_morse(n):
    tm = np.zeros(n)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

def golden_pattern(n, alpha):
    return (np.floor(np.arange(n) * alpha) % 2).astype(float)

def make_balanced_binary(n, pattern, rng):
    h = pattern.astype(float).copy()
    target = n // 2
    n_ones = int(h.sum())
    if n_ones == target:
        return h
    if n_ones < target:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        for i in range(target - n_ones):
            h[zeros[i % len(zeros)]] = 1.0
    else:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        for i in range(n_ones - target):
            h[ones[i % len(ones)]] = 0.0
    return h

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

            break
        i = np.random.randint(n)
        j = np.random.randint(n)

        cur[i], cur[j] = cur[j], cur[i]
        c_try = compute_c5(cur)

            best_c5 = c_try

            stall = 0
