def active_set_lp(h_cur, n, top_k=5):
    """Solve LP that linearizes the top-k overlap constraints at h_cur"""
    dx = 2.0 / n
    corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
    overlaps = corr * dx  # these are the actual overlap values
    # Find top k lags
    top_idx = np.argsort(overlaps)[::-1][:top_k]
    max_val = overlaps[top_idx[0]]
    
    # Linearize: overlap_k(h) ≈ overlap_k(h_cur) + g_k^T (h - h_cur)
    # = (overlap_k(h_cur) - g_k^T h_cur) + g_k^T h
    # We want: for all top lags k, (overlap_k - g_k^T h_cur) + g_k^T h <= t
    # i.e., g_k^T h <= t - (overlap_k - g_k^T h_cur)
    # = t - overlap_k + g_k^T h_cur
    
    constraints_A = []
    constraints_b = []
    for idx in top_idx:
        k = int(idx) - (n - 1)  # actual lag
        g = compute_gradient_overlap(h_cur, k, n)
        overlap_val = overlaps[idx]
        # g^T h <= t - overlap_val + g^T h_cur
        # Rearranging: g^T h - t <= -overlap_val + g^T h_cur
        # In linprog form: A @ [h, t] <= b
        row = np.append(g, -1.0)
        rhs = -overlap_val + np.dot(g, h_cur)
        constraints_A.append(row)
        constraints_b.append(rhs)
    
    A_ub = np.array(constraints_A)
    b_ub = np.array(constraints_b)
    
    # Also: 0 <= h <= 1, sum(h) = n/2
    # Bounds for h: [0,1], for t: [0, inf)
    bounds = [(0, 1)] * n + [(0, None)]
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Minimize t
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[n]
    return None, None

import numpy as np
from scipy.optimize import minimize, differential_evolution, linprog, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        step = diff / free.sum()
        # Don't overshoot
        max_step_pos = (1.0 - h[free]).min() if step > 0 else 0
        max_step_neg = h[free].min() if step < 0 else 0
        if step > 0:
            step = min(step, max_step_pos + 1e-12)
        else:
            step = max(step, -max_step_neg + 1e-12)
        h[free] += step
        h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - h.mean()))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def compute_gradient_overlap(h, k, n):
    g = np.zeros(n)
    for i in range(n):
        j = i + k
        if 0 <= j < n:
            g[i] += (1.0 - h[j])
            g[j] += (-h[i])
    return g

def active_set_lp_iterate(h_cur, n, top_k, max_iter=200):
    dx = 2.0 / n
    target = n / 2.0
    h = h_cur.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        top_idx = np.argsort(overlaps)[::-1][:top_k]
        
        A_ub_list = []
        b_ub_list = []
        for idx in top_idx:
            k = int(idx) - (n - 1)
            g = compute_gradient_overlap(h, k, n)
            overlap_val = overlaps[idx]
            row = np.append(g, -1.0)
            rhs = -overlap_val + np.dot(g, h)
            A_ub_list.append(row)
            b_ub_list.append(rhs)
        
        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        bounds = [(0, 1)] * n + [(0, None)]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        c_vec = np.zeros(n + 1)
        c_vec[n] = 1.0
        
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            break
        
        h_new = project_h(res.x[:n], target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        if np.abs(c5_new - compute_c5(h)) < 1e-10 and c5_new <= compute_c5(h) + 1e-10:
            h = h_new
            break
        h = h_new
    
    return best_h, best_c5

    np.random.seed(seed)

def project_h(h, target_sum):

    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:

            order = ones[np.argsort(h_interp[ones])]
            n_clear = int(np.ceil(-deficit))
            if n_clear <= len(order):
                hb[order[:n_clear]] = 0.0
        if abs(hb.sum() - n_bin / 2.0) < 1e-6:
            c5_val = compute_c5(hb)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_hb = hb.copy()
    return best_hb, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_parent)
    else:
        n_base = 280
        h_parent = np.full(n_base, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Active-set LP cutting plane at base resolution
    # Multiple passes with increasing top_k
    for top_k_init in [3, 5, 8, 12, 20, 30, 50]:
        if elapsed() > budget_s * 0.5:
            break
        tk_max = min(top_k_init + 10, 100)
        h_lp, c5_lp = active_set_lp(best_h, n_base, top_k_init, tk_max,
                                     max_iter=300, time_limit=budget_s * 0.5, t0=t0)
        if c5_lp < best_c5:
            best_h, best_c5 = h_lp, c5_lp

    # Phase 2: SLSQP refinement at base resolution
    target_base = float(n_base) / 2.0
    for beta in [500, 1500, 4000, 10000]:
        if elapsed() > budget_s * 0.7:
            break
        bounds_s = Bounds(np.zeros(n_base), np.ones(n_base))
        constr = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac,
                   'args': (target_base,)}]
        try:
            def obj_wrapper(x):
                return _slsqp_obj(x, beta, n_base)
            res = minimize(obj_wrapper, best_h, method='SLSQP', bounds=bounds_s,
                           constraints=constr,
                           options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_h(res.x, target_base)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    # Phase 3: Repeat LP after SLSQP (SLSQP may have changed active set)
    if elapsed() < budget_s * 0.8:
        for top_k_init in [5, 10, 20]:
            if elapsed() > budget_s * 0.8:
                break
            h_lp, c5_lp = active_set_lp(best_h, n_base, top_k_init,
                                         top_k_init + 20, max_iter=200,
                                         time_limit=budget_s * 0.8, t0=t0)
            if c5_lp < best_c5:
                best_h, best_c5 = h_lp, c5_lp

    # Phase 4: Upsample and refine at higher resolution
    for mult in [2, 3]:
        n_high = n_base * mult
        if n_high > 1600 or elapsed() > budget_s * 0.9:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_high)
        h_high = project_h(np.interp(x_new, x_old, best_h), float(n_high) / 2.0)
        c5_high = compute_c5(h_high)
        for top_k_init in [5, 10, 15]:
            if elapsed() > budget_s * 0.9:
                break
            h_lp, c5_lp = active_set_lp(h_high, n_high, top_k_init,
                                         top_k_init + 15, max_iter=150,
                                         time_limit=budget_s * 0.9, t0=t0)
            if c5_lp < c5_high:
                h_high, c5_high = h_lp, c5_lp
        if c5_high < best_c5:
            best_h, best_c5 = h_high, c5_high

    # Phase 5: Binary search at multiple resolutions
    for n_bin in [n_base, n_base * 2]:
        if elapsed() > budget_s * 0.96:
            break
        hb, cb = try_binary_round(best_h, n_bin)
        if hb is not None and cb < best_c5:
            best_h, best_c5 = hb, cb

    # Phase 6: Final LP polish
    if elapsed() < budget_s - 5:
        n_final = len(best_h)
        h_lp, c5_lp = active_set_lp(best_h, n_final, 10, 50,
                                     max_iter=200,
                                     time_limit=budget_s - 3, t0=t0)
        if c5_lp < best_c5:
            best_h, best_c5 = h_lp, c5_lp

    # Ensure exact constraint
    best_h = project_h(best_h, float(len(best_h)) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, int(len(best_h))
