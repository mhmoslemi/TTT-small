    """Minimize corr at lag k, subject to constraints."""
    from scipy.optimize import minimize, Bounds
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        if k == 0:
            return np.sum(x * (1 - x)) * dx
        elif k > 0:
            return np.sum(x[:n-k] * (1 - x[k:])) * dx
        else:
            return np.sum(x[-k:] * (1 - x[:n+k])) * dx

def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:

        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad
