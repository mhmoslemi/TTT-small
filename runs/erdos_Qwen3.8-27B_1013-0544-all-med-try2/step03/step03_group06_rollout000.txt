import numpy as np
from scipy.optimize import minimize, Bounds, differential_evolution
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    x_shift = np.clip(beta * (x - np.max(x)), -500, 500)
    return np.max(x) + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_subgrad(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def objective_factory(dx, beta):
    def obj(h_in):
        corr = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smooth_max(corr, beta) * dx
    return obj


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:

        return np.sum(x) - target1

        return np.ones(len(x))

            if c5_new < compute_c5(h_cur) - 1e-15:
                h_cur = h_new
            if c5_new < best_c5:

                best_c5 = c5_new

    c5_sub = best_c5

                c5_sub = c5_trial
