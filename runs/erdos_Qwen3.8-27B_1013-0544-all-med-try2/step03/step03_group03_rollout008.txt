    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break

        if diff > 0:

        else:
            mask = h > 1e-10

            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] -= h[lag:lag+m]

        m = n - la

            top_idx = np.argsort(corr)[-4:]
            grad = np.zeros(n)
            for k in top_idx:
                lag = k - (n - 1)
                grad += grad_for_lag(h, lag, n)
            grad /= len(top_idx)
            lr = 0.0015 / (1.0 + 0.002 * it)
            velocity = 0.4 * velocity + (1 - 0.4) * (-lr * grad)
            h_new = project_feasible(h + velocity, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c_new

    # Phase 6: Final high-precision SLSQP
    if time_left(t0, budget_s) > 8:
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for beta in [200000, 500000]:
            if time_left(t0, budget_s) < 4:
                break
            try:
                res = minimize(obj_full, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 12000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
