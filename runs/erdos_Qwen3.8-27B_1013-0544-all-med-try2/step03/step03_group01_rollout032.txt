import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def lse_grad(h, n, alpha):
    """Compute LSE value and gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    exp_c = np.exp(c - m)
    Z = exp_c.sum()
    vals = exp_c / Z
    f = m + np.log(Z) / alpha
    
    # Gradient
    shifts = np.arange(-(n-1), n)
    # A[j] = sum_k vals[k] * h[j+shifts[k]] (valid indices)
    # B[j] = sum_k vals[k] * h[j-shifts[k]] (valid indices)
    
    # Efficient computation using convolution
    # A[j] = sum_s vals[s+n-1] * h[j+s] for s in [-(n-1), n-1] and 0<=j+s<n
    # This is a cross-correlation of h with vals (reversed)
    
    # Using FFT for efficiency (though for n=100, direct is fine too)
    A = np.zeros(n)
    B = np.zeros(n)
    
    for k in range(2*n-1):
        s = shifts[k]
        w = vals[k]
        if w < 1e-15:
            continue
        if s >= 0:
            # A[j] += w * h[j+s] for j=0,...,n-1-s
            A[:n-s] += w * h[s:]
            # B[j] += w * h[j-s] for j=s,...,n-1
            B[s:] += w * h[:n-s]
        else:
            ss = -s
            # A[j] += w * h[j-s] = w * h[j+ss] for j=0,...,n-1-ss
            A[:n-ss] += w * h[ss:]
            # B[j] += w * h[j+ss] for j=0,...,n-1-ss  
            # Wait, B[j] = sum_k vals[k]*h[j-shifts[k]] = sum_k vals[k]*h[j-s_k]
            # For s_k = -ss: B[j] += w * h[j-(-ss)] = w * h[j+ss] for j=0,...,n-1-ss
            B[:n-ss] += w * h[ss:]
    
    grad = (dx / alpha) * (1.0 - A - B)
    return f, grad

def fista(h0, n, alpha, lr, n_iter, deadline):
    """FISTA with LSE objective."""
    h = h0.copy()
    y = h.copy()
    tk = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        f, grad = lse_grad(y, n, alpha)
        
        # Gradient step
        z = y - lr * grad
        
        # Project
        z = project_feasible(z, n)
        
        # Every 200 iters, check actual C5
        if it % 200 == 0:
            c = compute_c5(z, n)
            if c < best_c:
                best_c = c
                best_h = z.copy()
        
        # FISTA update
        tk_new = (1 + np.sqrt(1 + 4 * tk * tk)) / 2
        y = z + ((tk - 1) / tk_new) * (z - y)
        tk = tk_new
    
    # Final check
    c = compute_c5(z, n)
    if c < best_c:
        best_c = c
        best_h = z.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def lse_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    exp_c = np.exp(c - m)
    Z = exp_c.sum()
    vals = exp_c / Z
    f = m + np.log(Z) / alpha
    shifts = np.arange(-(n - 1), n)
    A = np.zeros(n)
    B = np.zeros(n)
    for k in range(2 * n - 1):
        w = vals[k]
        if w < 1e-16:
            continue
        s = shifts[k]
        if s >= 0:
            A[:n - s] += w * h[s:]
            B[s:] += w * h[:n - s]
        else:
            ss = -s
            A[ss:] += w * h[:n - ss]
            B[:n - ss] += w * h[ss:]
    grad = (dx / alpha) * (1.0 - A - B)
    return f, grad

def fista(h0, n, alpha, lr, n_iter, deadline):
    h = h0.copy()
    y = h.copy()
    tk = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    z = h.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        _, grad = lse_and_grad(y, n, alpha)
        z = y - lr * grad
        z = project_feasible(z, n)
        if it % 200 == 0:
            c = compute_c5(z, n)
            if c < best_c:
                best_c = c
                best_h = z.copy()
        tk_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y = z + ((tk - 1.0) / tk_new) * (z - y)
        tk = tk_new
    c = compute_c5(z, n)
    if c < best_c:
        best_c = c
        best_h = z.copy()
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: float(x.sum() - n / 2.0),
             'jac': lambda x, n=n: np.ones(n)}]
    def obj(x, n=n, alpha=alpha):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_local(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for sweep in range(3):
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    for s in (1.0, -1.0):
                        hi = h[i] + s * d
                        hj = h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i] = hi
                            h[j] = hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= s * d
                                h[j] += s * d
            if not improved:
                break
    return h, best_c

def targeted_moves(h, n, deadline, d):
    best_c = compute_c5(h, n)
    for sweep in range(200):
        if time.time() > deadline:
            break
        c = get_correlations(h, n)
        c_max = c.max()
        thresh = c_max - 0.002 * (2.0 / n)
        worst = np.where(c >= thresh)[0]
        if len(worst) == 0:
            break
        improved_this_sweep = False
        for k in worst[:10]:
            s = k - (n - 1)
            if time.time() > deadline:
                return h, best_c
            for i in range(n):
                idx = i + s
                if idx < 0 or idx >= n or idx == i:
                    continue
                hi_old = h[i]
                hj_old = h[idx]
                for frac in (1.0, 0.5):
                    dd = d * frac
                    h[i] = hi_old - dd
                    h[idx] = hj_old + dd
                    if h[i] < -1e-14 or h[idx] > 1 + 1e-14:
                        h[i] = hi_old
                        h[idx] = hj_old
                        continue
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved_this_sweep = True
                    else:
                        h[i] = hi_old
                        h[idx] = hj_old
                    h[i] = hi_old
                    h[idx] = hj_old
        if not improved_this_sweep and sweep > 20:
            break
    return h, best_c

                best_h = h_try

    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
