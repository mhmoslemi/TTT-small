def subgradient_step(h, active_lags, n, step):
    """Compute one subgradient step for the min-max problem."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active_lags:
        if k == 0:
            grad += (1 - 2*h)
        elif k > 0:
            grad[k:] += (1 - h[:-k])
            grad[:n-k] -= h[k:]
        else:
            m = -k
            grad[:n-m] += (1 - h[m:])
            grad[m:] -= h[:n-m]
    grad /= len(active_lags)
    # Project: subtract mean to preserve sum, then clip
    grad -= grad.mean()  # This preserves sum(h) to first order
    h_new = h - step * grad * dx
    # Project onto [0,1] with sum constraint
    h_new = project_h(h_new, n/2.0)
    return h_new

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_subgrad(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    grad /= len(lags)
    return grad


def make_slsqp_obj(dx, beta):
    def obj(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smooth_max(c, beta) * dx
    return obj


def make_eq_con(target_sum):
    def con(x_in):
        return np.sum(x_in) - target_sum
    return con


def make_eq_jac(shape):
    def jac(x_in):
        return np.ones(x_in.shape)
    return jac


def slsqp_optimize(h, n_points, target_sum, dx, betas, max_iter, time_limit):
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    con = make_eq_con(target_sum)
    jac = make_eq_jac((n_points,))
    constraints = [{'type': 'eq', 'fun': con, 'jac': jac}]
    best = h.copy()
    best_c5 = compute_c5(best)
    start = time.time()
    for beta in betas:
        if time.time() - start > time_limit:
            break
        obj = make_slsqp_obj(dx, beta)
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': max_iter, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new.copy()
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5


def subgradient_optimize(h, target_sum, n_iters, time_limit):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    momentum = np.zeros(n)
    beta_m = 0.85
    no_improve = 0
    start = time.time()
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        if time.time() - start > time_limit - 2:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        M = np.max(corr)
        # Get all lags within 0.001 of max for stable gradient
        threshold = M - 0.001
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(corr)[::-1][:3]
        top_lags = [int(i) - (n - 1) for i in active[:15]]
        grad = compute_subgrad(best, top_lags)
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        momentum = beta_m * momentum - 0.01 * (grad / gnorm)
        mnorm = np.linalg.norm(momentum)
        if mnorm < 1e-14:
            break
        direction = -momentum / mnorm
        found = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_trial = project_h(best + step * direction, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best = h_trial
                best_c5 = c5_trial
                found = True
                no_improve = 0
                break
        if not found:
            no_improve += 1
            if no_improve % 50 == 0:
                perturb = np.random.randn(n) * 0.003
                h_trial = project_h(best + perturb, target_sum)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    best = h_trial
                    best_c5 = c5_trial
                    no_improve = 0
            if no_improve > 300:
                break
    return best, best_c5


def active_leveling(h, target_sum, n_iters, time_limit):
    """Try to equalize the top active lags by targeted moves."""
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    start = time.time()
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        M = np.max(corr)
        active_idx = np.where(corr >= M - 0.0005)[0]
        if len(active_idx) < 2:
            active_idx = np.argsort(corr)[::-1][:4]
        top_lags = [int(i) - (n - 1) for i in active_idx[:8]]
        # Compute subgradient only on active set
        grad = compute_subgrad(best, top_lags)
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        # Try several step sizes with the subgradient direction
        direction = -grad / gnorm
        found = False
        for step in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
            h_trial = project_h(best + step * direction, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best = h_trial
                best_c5 = c5_trial
                found = True
                break
        if not found and it % 200 == 199:
            perturb = np.random.randn(n) * 0.002
            h_trial = project_h(best + perturb, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best = h_trial
                best_c5 = c5_trial
    return best, best_c5
