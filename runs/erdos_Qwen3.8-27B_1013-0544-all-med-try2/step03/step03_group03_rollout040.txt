import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _corr_weights(x, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(beta * (corr - cmax), -700.0, 700.0)
    exps = np.exp(shifted)
    Z = exps.sum()
    return cmax, exps / Z

def _grad_from_w(x, w):
    n = len(x)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                grad[:m] += wk * (1.0 - x[lag:lag + m])
                grad[lag:] -= wk * x[:m]
        else:
            L = -lag
            m = n - L
            if m > 0:
                grad[L:] += wk * (1.0 - x[:m])
                grad[:m] -= wk * x[L:L + m]
    return grad

def _smooth_val(cmax, w, beta, dx):
    Z = w.sum()
    return (cmax + np.log(Z) / beta) * dx

def obj_fn(x, dx, beta):
    cmax, w = _corr_weights(x, beta)
    return _smooth_val(cmax, w, beta, dx)

def jac_fn(x, dx, beta):
    cmax, w = _corr_weights(x, beta)
    return _grad_from_w(x, w) * dx

def con_fun(x, ts):
    return x.sum() - ts

def con_jac_fn(x, ts):
    return np.ones_like(x)

def nesterov_loop(h, dx, ts, t0, budget, min_time, beta0, lr0):
    n = len(h)
    x = h.copy()
    x_prev = x.copy()
    t_n = 1.0
    beta = beta0
    lr = lr0
    best_c = compute_c5(x)
    stagnant = 0
    for it in range(200000):
        rem = budget - (time.time() - t0)
        if rem < min_time:
            break
        y = x + (t_n - 1.0) / (t_n + 2.0) * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        cmax, w = _corr_weights(y, beta)
        gy = _grad_from_w(y, w)
        x_new = project_feasible(y - lr * gy * (2.0 / n), ts)
        x_prev = x.copy()
        x = x_new
        t_n = (1.0 + np.sqrt(1.0 + 4.0 * t_n * t_n)) / 2.0
        if (it + 1) % 300 == 0:
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                h[:] = x
                stagnant = 0
            else:
                stagnant += 1
                if stagnant > 8:
                    beta *= 1.7
                    lr *= 0.5
                    t_n = 1.0
                    x_prev = x.copy()
                    stagnant = 0
    return x, best_c
