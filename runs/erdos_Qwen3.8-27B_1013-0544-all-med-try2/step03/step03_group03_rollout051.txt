```python
    weights = exp_s / sum_exp
    
    obj = sm_val * dx + mu * (h.sum() - target_sum) ** 2

    for lag in range(-(n-1), n):

        if w < 1e-14:

        if lag >= 0:
            cnt = n - lag
            if cnt > 0:
                idx = np.arange(cnt)
                grad[idx] += w * (1.0 - h[idx + lag])
                grad[idx + lag] -= w * h[idx]
        else:
            la = -lag
            cnt = n - la
            if cnt > 0:
                idx = np.arange(cnt)
                grad[idx + la] += w * (1.0 - h[idx])
                grad[idx] -= w * h[idx + la]
    
    grad *= dx
    grad += 2.0 * mu * (h.sum() - target_sum)
    
    return obj, grad

                   method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8,
                            'iprint': -1, 'maxfun': 5000})
    return res.x
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(h, n, dx, beta, target_sum, mu):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.where(shifted > -500.0, shifted, -500.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    sm_val = m + np.log(sum_exp) / beta
    weights = exp_s / sum_exp
    
    obj = sm_val * dx + mu * (h.sum() - target_sum) ** 2
    
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        w = weights[lag + n - 1]
        if w < 1e-14:
            continue
        if lag >= 0:
            cnt = n - lag
            grad[:cnt] += w * (1.0 - h[lag:lag+cnt])
            grad[lag:lag+cnt] -= w * h[:cnt]
        else:
            la = -lag
            cnt = n - la
            grad[la:la+cnt] += w * (1.0 - h[:cnt])
            grad[:cnt] -= w * h[la:la+cnt]
    
    grad *= dx
    grad += 2.0 * mu * (h.sum() - target_sum)
    
    return obj, grad

def run_lbgfsb(x0, n, dx, target_sum, beta, mu, maxiter, t0, budget_s, time_limit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, x0, args=(n, dx, beta, target_sum, mu),
                       method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8,
                                'iprint': -1, 'maxfun': maxiter * 3})
        h_new = project_feasible(res.x, target_sum)
        return h_new
    except Exception:
        return x0

def lbgfsb_continuation(h, n, t0, budget_s, time_frac=0.5):
    dx = 2.0 / n
    target_sum = n / 2.0
    time_limit = time_frac * (budget_s - (time.time() - t0))
    t_start = time.time()
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    betas = [10, 30, 100, 300, 1000, 3000, 10000, 30000]
    mus = [5, 20, 100, 500, 2000]
    
    # Pair up beta and mu values
    # Strategy: for each beta, use a sufficient mu
    for i, beta in enumerate(betas):
        elapsed = time.time() - t_start
        if elapsed > time_limit:
            break
        mu = mus[min(i, len(mus)-1)]
        maxiter = max(50, int(200 * (0.5 + 0.5 * (1.0 - elapsed/time_limit))))
        
        h_new = run_lbgfsb(best_h, n, dx, target_sum, beta, mu, maxiter, t0, budget_s, time_limit)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
    
    return best_h, best_c5

def pair_swap_cd(h, n, target_sum, t0, budget_s, n_sweeps=50):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    np.random.seed(int(time.time()) % 10000)
    
    for sweep in range(n_sweeps):
        if time.time() - t0 > budget_s - 30:
            break
        # Random pairs
        pairs = []
        for _ in range(3 * n):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j:
                pairs.append((i, j))
        
        for (i, j) in pairs:
            old_i = h[i]
            old_j = h[j]
            # Possible deltas: h[i] -= d, h[j] += d
            # Constraint: 0 <= old_i - d <= 1 and 0 <= old_j + d <= 1
            d_min = max(-old_i, -(1.0 - old_j))
            d_max = min(1.0 - old_i, old_j)
            if d_max - d_min < 1e-6:
                continue
            
            # Try a few delta values
            candidates = [d_min + 0.25*(d_max-d_min), d_min + 0.5*(d_max-d_min), 
                         d_min + 0.75*(d_max-d_min), 0.05, -0.05, 0.1, -0.1]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(h, n, dx, beta, target_sum, mu):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    sm_val = m + np.log(sum_exp) / beta
    weights = exp_s / sum_exp
    obj = sm_val * dx + mu * (h.sum() - target_sum) ** 2
    grad = np.zeros(n)
    for lag in range(-(n - 1), n):
        w = weights[lag + n - 1]
        if w < 1e-14:
            continue
        if lag >= 0:
            cnt = n - lag
            grad[:cnt] += w * (1.0 - h[lag:lag + cnt])
            grad[lag:lag + cnt] -= w * h[:cnt]
        else:
            la = -lag
            cnt = n - la
            grad[la:la + cnt] += w * (1.0 - h[:cnt])
            grad[:cnt] -= w * h[la:la + cnt]
    grad *= dx
    grad += 2.0 * mu * (h.sum() - target_sum)
    return obj, grad

def lbgfsb_round(h, n, t0, budget_s, time_budget):
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = h.copy()

                           method='L-BFGS-B', jac=True, bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-13, 'gtol': 1e-9,
                                    'iprint': -1, 'maxfun': maxiter * 3})
```
