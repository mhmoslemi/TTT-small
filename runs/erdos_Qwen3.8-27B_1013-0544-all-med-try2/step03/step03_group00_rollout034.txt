    """Find min_{x in [lo,hi]} max_k (a_k * x + b_k)"""
    # f(x) = max_k (a_k * x + b_k) is convex
    # Use bisection: f is decreasing where its right-derivative < 0
    # and increasing where its left-derivative > 0
    
    for _ in range(50):

        # Compute f at mid-eps and mid+eps to determine slope direction

        f_mid = np.max(a * mid + b)

        min_slope = a[active].min()
        max_slope = a[active].max()
        if min_slope > 1e-14:

        elif max_slope < -1e-14:
            hi = mid  # f is decreasing everywhere, go right
        else:

def project_feasible(h, target_sum):

        vals = a * mid + b

def smoothmax_obj_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = c_sel.max()
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        k = top_idx[idx]
        wk = w[idx]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        j_lo = max(0, -s)
        j_hi = min(n - 1, n - 1 - s)
        # dc_k/dh[i] = (1-h[i+s])*[i in valid] - h[i-s]*[i-s in valid]
        active_i = np.arange(j_lo, j_hi + 1)
        grad[active_i] += wk * (1.0 - h[active_i + s])
        active_j = np.arange(j_lo, j_hi + 1)
        grad[active_j + s] -= wk * h[active_j]
    grad *= dx
    return val, grad

def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return smoothmax_obj_grad(x, beta, top_k)[0]
    
    def objg(x):
        return smoothmax_obj_grad(x, beta, top_k)[1]
    
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq',

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def min_convex_pl(a, b):
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        f_mid = vals.max()
        active = vals >= f_mid - 1e-13
        if active.sum() == 0:
            active = vals >= f_mid - 1e-10
        if a[active].min() > 1e-12:
            hi = mid
        elif a[active].max() < -1e-12:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, max_sweeps=50, tol=1e-13, seed=0):
    n = len(h)
    target = n / 2.0
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        max_change = 0.0
        order = rng.permutation(n) if sweep > 0 else np.arange(n)
        for i in order:
            old = h[i]
            h[i] = 0.0
            g0 = np.correlate(h, 1.0 - h, mode='full').astype(float)
            s_offset = n - 1
            a = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                s = k - s_offset
                j_lo = max(0, -s)
                j_hi = min(n - 1, n - 1 - s)
                a_k = 0.0
                if j_lo <= i <= j_hi:
                    a_k += (1.0 - h[i + s])
                if j_lo <= i - s <= j_hi:
                    a_k -= h[i - s]
                a[k] = a_k
            x_opt = min_convex_pl(a, g0)
            h[i] = x_opt
            max_change = max(max_change, abs(x_opt - old))
        h = project_feasible(h, target)
        if max_change < tol:
            break
    return h

def smoothmax_obj_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = c_sel.max()
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        k = top_idx[idx]
        wk = w[idx]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        j_lo = max(0, -s)
        j_hi = min(n - 1, n - 1 - s)
        active_i = np.arange(j_lo, j_hi + 1)
        grad[active_i] += wk * (1.0 - h[active_i + s])
        active_j = np.arange(j_lo, j_hi + 1)
        grad[active_j + s] -= wk * h[active_j]
    grad *= dx
    return val, grad

def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0
    def obj(x):
        return smoothmax_obj_grad(x, beta, top_k)[0]
    def objg(x):
        return smoothmax_obj_grad(x, beta, top_k)[1]
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def basin_hop(h_best, best_c5, n_hops, maxiter, noise_base, noise_decay, seed, t0, budget_s):
    rng = np.random.RandomState(seed)
    n = len(h_best)
    target = n / 2.0
    for hop in range(n_hops):
        if budget_s - (time.time() - t0) < 50:
            break
        noise = noise_base * (noise_decay ** hop)
        h_pert = h_best + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_local = coord_descent(h_pert, max_sweeps=15, tol=1e-11, seed=seed+hop)
        for beta in [200, 500]:
            if budget_s - (time.time() - t0) < 30:
                break
            h_local = slsqp_solve(h_local, beta, maxiter, top_k=20)
            h_local = coord_descent(h_local, max_sweeps=10, tol=1e-12)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
    return h_best, best_c5

def make_binary_initial(n, seed):

    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)

        h[:half] = 1.0
        h[half:] = 0.0
        rng.shuffle(h)
        h = project_feasible(h, half)
        break
    return h
