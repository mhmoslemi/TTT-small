```python
from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_g_k(h, lag, n):
    """Gradient of g_k(h) = sum_i h[i]*(1-h[i+k]) for the given lag."""
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return grad
        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + lag])
        h[idx + lag]  # h[idx+lag] * (-1) contribution to h[idx]
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m <= 0:
            return grad
        idx = np.arange(m)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

def smooth_obj_power(h, p):
    """Sum of powers smooth approximation to max."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (np.sum(corr**p) ** (1.0/p)) * dx

def smooth_obj_logsumexp(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def weighted_subgrad(h, corr, n, threshold, power=2):
    """Weighted subgradient based on how much each lag exceeds threshold."""
    excess = np.maximum(corr - threshold, 0)
    if excess.max() < 1e-15:
        # If all below threshold, use uniform weights on top
        top_k = np.argsort(corr)[-5:]
        w = np.zeros(len(corr))
        w[top_k] = 1.0
        w /= w.sum()
    else:
        w = excess ** power
        if w.sum() < 1e-15:
            top_k = np.argsort(corr)[-5:]
            w = np.zeros(len(corr))
            w[top_k] = 1.0
        w /= w.sum()
    
    grad = np.zeros(n)

        if w[k_idx] < 1e-10:
            continue
        lag = k_idx - (n - 1)
        grad += w[k_idx] * grad_g_k(h, lag, n)
    return grad

    """Create a periodic initial condition."""

def fista_subgrad(h, n, max_iter, lr0, decay, threshold=0.0, power=2, 
                   time_limit=None, t0=None, best_c5=0, best_h=None):
    """FISTA-accelerated subgradient method."""
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h.copy()
    y = h.copy()
    t_momentum = 1.0
    best_local = compute_c5(h)
    best_local_h = h.copy()
    
    for it in range(max_iter):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        
        corr = np.correlate(y, 1.0 - y, mode='full')
        grad = weighted_subgrad(y, corr, n, threshold, power)
        
        lr = lr0 / (1.0 + decay * it)
        y_new = y - lr * grad
        h_new = project_feasible(y_new, target_sum)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_momentum**2)) / 2.0
        y_new2 = h_new + ((t_momentum - 1.0) / t_new) * (h_new - h)
        y_new2 = np.clip(y_new2, 0.0, 1.0)
        
        c = compute_c5(h_new)
        if c < best_local:
            best_local = c
            best_local_h = h_new.copy()
            if best_c5 is not None and c < best_c5:
                best_c5 = c
                best_h = h_new.copy()
        
        h = h_new
        y = y_new2
        t_momentum = t_new
    
    return best_local_h, best_local, best_c5, best_h

def slsqp_smooth(h, n, beta, time_limit=None, t0=None):
    """SLSQP optimization with smooth max."""
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        return smooth_obj_logsumexp(x, beta)
    
    def con_fun(x):
        return x.sum() - target_sum
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, float('inf')
```

```python
    # Subgradient = {A[k] : B[k] = f0}
    active0 = np.where(B >= f0 - 1e-10)[0]
    if A[active0].min() >= 0:
        return 0.0  # f is non-decreasing from 0, so t=0 is optimal
    
    # Check subgradient at t=1: active lags have A[k]+B[k] = f1
    active1 = np.where(A + B >= f1 - 1e-10)[0]
    if A[active1].max() <= 0:
        return 1.0  # f is non-increasing up to 1, so t=1 is optimal
    
    # Binary search for where subgradient contains 0
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        fmid = np.max(A * mid + B)
        active = np.where(A * mid + B >= fmid - 1e-10)[0]
        if A[active].min() >= 0:
            lo = mid  # need to go right
        elif A[active].max() <= 0:
            hi = mid  # need to go left
        else:
            break  # 0 is in subgradient, we're at the optimum
        mid = (lo + hi) / 2
    
    t_opt = (lo + hi) / 2
    return t_opt
```

This is O(n * n) per coordinate (for computing A and the binary search), and O(n) coordinates per sweep. So O(n³) per sweep. For n=210, that's about 9.3M per sweep. With 50 sweeps, that's 465M operations. In Python, this might be too slow (each operation is interpreted).

```python
def compute_A_for_coord(h, j, n):
    """Compute A vector for coordinate j."""
    k = np.arange(2*n - 1)
    l = k - (n-1)
    
    # Case (a): 0 <= j+l < n
    mask_a = (j + l >= 0) & (j + l < n)
    # Case (b): 0 <= j-l < n
    mask_b = (j - l >= 0) & (j - l < n)
    
    A = np.zeros(2*n - 1)
    A[mask_a] += (1.0 - h[j + l[mask_a]])
    A[mask_b] -= h[j - l[mask_b]]
    return A
```

```python
    for _ in range(50):

        h = np.clip(h, 0.0, 1.0)

            mask_b = (j - l_arr >= 0) & (j - l_arr < n)

                        else:

        if c < best_local:
```
