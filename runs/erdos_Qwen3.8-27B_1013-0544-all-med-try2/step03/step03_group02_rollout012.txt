import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_to(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new), np.linspace(0.0, 2.0, len(h)), h)

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(c)
    shifted = beta * (c - m)
    shifted = np.clip(shifted, -500.0, 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def sum_fn(x, t):
    return np.sum(x) - t

def sum_fn_jac(x, t):
    return np.ones(x.shape)

def slsqp(h0, n, target, beta, bounds, maxiter, t0, budget):
    if budget - (time.time() - t0) < 5:
        return h0
    cons = [{'type': 'eq', 'fun': sum_fn, 'jac': sum_fn_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        out = project_h(r.x, target)
        if compute_c5(out) < 10.0:
            return out
    except Exception:
        pass
    return h0

def find_active(h, k=6):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:k]
    return idx, corr

def focused_sml(h_in, idx, beta, dx, n):
    """Smooth max over only active lags for sharper focus."""
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    vals = c[idx]
    m = np.max(vals)
    sh = beta * (vals - m)
    sh = np.clip(sh, -500.0, 0.0)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * dx

def active_slqP(h0, n, target, bounds, t0, budget, n_active=5):
    cur = h0.copy()
    cur_c5 = compute_c5(cur)
    for it in range(15):
        if budget - (time.time() - t0) < 10:

        idx, _ = find_active(cur, n_active)
        beta = 100 + it * 80

        new_c5 = compute_c5(cur)
        if new_c5 >= cur_c5 - 1e-15:
            break
        cur_c5 = new_c5
    return cur, cur_c5

def fourier_perturb(h, n, scale, n_modes, rng):
    x = np.linspace(0, 2, n)
    p = np.zeros(n)
    for k in range(1, n_modes + 1):
        a = rng.normal(0, 1)
        b = rng.normal(0, 1)
        p += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    p /= np.linalg.norm(p) + 1e-15
    return project_h(h + scale * p, n / 2.0)

def line_search_coord(cur, target, n, t0, budget, max_sweeps=100, init_step=0.08):
    best_c5 = compute_c5(cur)
    steps = np.full(n, init_step)
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 3:
            break
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if budget - (time.time() - t0) < 1:
                break
            for d in [1, -1]:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d * steps[j], 0, 1)
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    cur = trial
                    best_c5 = c_try
                    steps[j] = min(steps[j] * 1.8, 0.4)
                    improved = True
                    break
            else:
                steps[j] = max(steps[j] * 0.6, 0.003)
        if not improved and sw > 8:
            break
    return cur, best_c5

def block_template(n, n_blocks=4):
    """Create a block-structured initial guess."""
    h = np.zeros(n)
    block_size = n // n_blocks
    for i in range(n_blocks):
        start = i * block_size
        end = start + block_size if i < n_blocks - 1 else n
        val = 1.0 if i % 2 == 1 else 0.05
        h[start:end] = val
    return project_h(h, n / 2.0)

def symmetric_refine(h, n, target, t0, budget, bounds):
    """Enforce h[i] = h[n-1-i] to reduce search space."""
    half = n // 2
    h_sym = np.zeros(n)
    h_sym[:half] = (h[:half] + h[n-1:half-1:-1]) / 2.0
    h_sym[half:] = h_sym[n-1:half-1:-1]
    if n % 2 == 1:
        h_sym[half] = h[half]
    h_sym = project_h(h_sym, target)
    h_sym = slsqp(h_sym, n, target, 500, bounds, 3000, t0, budget)
    return h_sym
