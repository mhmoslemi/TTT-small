```python
def compute_delta(h, a, b, n):
    """Compute change in corr vector after swapping h[a] 1->0, h[b] 0->1.
    h is the state BEFORE the swap."""
    s_vals = np.arange(-(n-1), n)
    K = 2*n - 1  # number of lags
    delta = np.zeros(K)
    
    for idx, s in enumerate(s_vals):
        d = 0
        # Term 1: m=a
        if 0 <= a + s < n:
            if a + s == b:
                d += 0 - 1  # old = 1*(1-0) = 1, new = 0
            elif a + s == a:  # s=0
                d += 0  # old = 1*(1-1) = 0, new = 0
            else:
                d += 0 - (1 - h[a+s])
        
        # Term 2: m=b
        if 0 <= b + s < n:
            if b + s == a:
                d += 1 - 0  # new = 1*(1-0) = 1, old = 0
            elif b + s == b:  # s=0
                d += 0  # new = 1*(1-1) = 0, old = 0
            else:
                d += (1 - h[b+s]) - 0
        
        # Term 3: m=a-s
        if 0 <= a - s < n:
            if a - s == b:
                d += 1 - 0  # new = h'[b]*1 = 1, old = h[b]*0 = 0
            elif a - s == a:  # s=0
                d += 0  # new = h'[a]*1 = 0, old = 0
            else:
                d += h[a-s] - 0
        
        # Term 4: m=b-s
        if 0 <= b - s < n:
            if b - s == a:
                d += 0 - 1  # new = h'[a]*0 = 0, old = h[a]*1 = 1
            elif b - s == b:  # s=0
                d += 0  # new = 0, old = h[b]*1 = 0
            else:
                d += 0 - h[b-s]
        
        delta[idx] = d
    
    return delta
```

        # Also from original
        h_r2, c_r2 = grad_descend(h_cur, maxiter=3000, top_k=5, step_init=0.03, t0=t0, budget_s=budget_s)
        if c_r2 < best_c5:
            best_c5 = c_r2
            h_best = h_r2.copy()

    # Phase 3: SLSQP with smooth-max continuation
    if rem() > 200:
        for beta in [20, 50, 100, 200, 400]:
            if rem() < 80:
                break
                pass
            h_s = slsqp_refine(h_best, beta, 400, 10, t0, budget_s)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                best_c5 = c_s
                h_best = h_s.copy()

    # Phase 4: Basin hopping with diverse perturbations
    if rem() > 150:
        rng = np.random.RandomState(seed + 200)
        for hop in range(25):
            if rem() < 50:
                break
            noise = 0.02 * (0.82 ** hop)
            h_pert = h_best + rng.randn(n) * noise
            h_pert = project_feasible(h_pert, n / 2.0)
            h_res, c_res = grad_descend(h_pert, maxiter=800, top_k=4, step_init=0.02, t0=t0, budget_s=budget_s)
            if c_res < best_c5:
                best_c5 = c_res
                h_best = h_res.copy()
            # Also try structured perturbation: shift
            if hop % 5 == 3:
                shift = rng.randint(1, n // 4)
                h_shift = np.roll(h_best, shift)
                h_shift = project_feasible(h_shift, n / 2.0)
                c_sh = compute_c5(h_shift)
                if c_sh < best_c5:
                    best_c5 = c_sh
                    h_best = h_shift.copy()
                    h_res2, c_res2 = grad_descend(h_best, maxiter=500, top_k=4, step_init=0.02, t0=t0, budget_s=budget_s)
                    if c_res2 < best_c5:
                        best_c5 = c_res2
                        h_best = h_res2.copy()

    # Phase 5: Multi-resolution
    for n_try in [150, 300]:
        if rem() < 100:
            break
        h_interp = interp_to_n(h_best, n_try)
        h_mr, c_mr = grad_descend(h_interp, maxiter=3000, top_k=6, step_init=0.025, t0=t0, budget_s=budget_s)
        if c_mr < best_c5:
            best_c5 = c_mr
            h_best = h_mr.copy()
        if rem() > 80:
            h_s = slsqp_refine(h_best, 150, 300, 8, t0, budget_s)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                best_c5 = c_s
                h_best = h_s.copy()

    # Phase 6: Final high-beta polish
    if rem() > 30:
        h_fin = slsqp_refine(h_best, 500, 500, 5, t0, budget_s)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin
        h_fin2, c_fin2 = grad_descend(h_best, maxiter=2000, top_k=3, step_init=0.01, t0=t0, budget_s=budget_s)
        if c_fin2 < best_c5:
            best_c5 = c_fin2
            h_best = h_fin2

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    final_c5 = compute_c5(h_best)
    return h_best, final_c5, len(h_best)
