from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx), c * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h = np.clip(h + (diff / total) * room, 0, 1)
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h = np.clip(h + (diff / total) * h, 0, 1)
    return h

def compute_D(h, j, n):
    """Compute D_j(l) for all lags l = -(n-1), ..., n-1"""
    n_lags = 2 * n - 1
    D = np.zeros(n_lags)
    for m in range(n_lags):
        l = m - (n - 1)
        dval = 0.0
        if 0 <= j - l < n:
            dval += (1.0 - h[j - l])
        if 0 <= j + l < n:
            dval -= h[j + l]
        D[m] = dval
    return D

def coord_step(h, j, c_arr, n, dx):
    """Find optimal h[j] given all other h[i] fixed. Returns new v and new c_arr."""
    D = compute_D(h, j, n)
    # c_l(v) = c_l - h[j]*D_l*dx + v*D_l*dx = (c_l - h[j]*D_l*dx) + v*(D_l*dx)
    a = c_arr - h[j] * D * dx
    b = D * dx
    
    f = lambda v: np.max(a + b * v)
    
    # Ternary search on [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    
    v = (lo + hi) / 2
    # Also check boundaries
    best_v = v
    best_f = f(v)
    for test_v in [0.0, 1.0, h[j]]:
        fv = f(test_v)
        if fv < best_f:
            best_f = fv
            best_v = test_v
    
    # Update c_arr
    delta = best_v - h[j]
    new_c = c_arr + delta * D * dx
    return best_v, new_c, best_f

def coordinate_descent(h, n, max_sweeps=200, tol=1e-12, deadline=None):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    c_val, c_arr = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            v, new_c_arr, new_f = coord_step(h, j, c_arr, n, dx)
            if abs(v - h[j]) > 1e-14:
                h[j] = v
                h = project_feasible(h, n)
                # Recompute c_arr after projection (projection changed other values)
                c_val, c_arr = compute_c5(h, n)
                improved = True
            else:
                c_arr = new_c_arr
        if not improved:
            break
    
    c_val, _ = compute_c5(h, n)
    return h, c_val

def smooth_max_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    # Handle overflow
    if shifted.max() > 500:
        return m + (shifted.max() + np.log(np.sum(np.exp(shifted - shifted.max())))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_step(h, n, alpha, maxiter=2000, ftol=1e-16):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    
    def con(x):
        return x.sum() - target_sum
    
    def con_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_local(h, n, deadline, worst_lag=None):
    """Targeted pair moves, optionally biased toward the worst lag."""
    dx = 2.0 / n
    best_c, _ = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for d in deltas:
        if time.time() > deadline:
            break
        # Get current worst lag
        _, c_arr = compute_c5(h, n)
        worst_idx = np.argmax(c_arr)
        
        for i in range(n):
            if time.time() > deadline:
                break
            # Only try pairs that affect the worst lag
            l = worst_idx - (n - 1)
            j_candidate = i + l
            if 0 <= j_candidate < n and j_candidate != i:
                j = j_candidate
            else:
                continue
            
            for s in (1.0, -1.0):
                hi = h[i] + s * d
                hj = h[j] - s * d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i] = hi
                    h[j] = hj
                    c = compute_c5(h, n)[0]
                    if c < best_c:
                        best_c = c
                    else:
                        h[i] -= s * d
                        h[j] += s * d
    
    return h, best_c
