                                   options={'maxiter': 2000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        n = n_try
                        target_sum = ts_t
                except Exception:
                    pass
            if budget_s - (time.time() - t0) > 15:
                h_t2 = best_h.copy()
                h_t2 = coord_descent_pass(h_t2, len(h_t2), max_passes=3)
                h_t2 = project_feasible(h_t2, len(h_t2) / 2.0)
                c2 = compute_c5(h_t2)
                if c2 < best_c5:
                    best_h = h_t2.copy()
                    best_c5 = c2
                    n = len(best_h)
                    target_sum = n / 2.0

    # Phase 4: Annealing - perturb and re-optimize
    if budget_s - (time.time() - t0) > 40:
        for pert in [0.05, 0.03, 0.015, 0.008]:
            for trial in range(4):
                if budget_s - (time.time() - t0) < 15:
                    break
                x0 = best_h.copy()
                x0 += np.random.randn(len(x0)) * pert
                x0 = project_feasible(x0, target_sum)
                x0 = coord_descent_pass(x0, len(x0), max_passes=3)
                x0 = project_feasible(x0, target_sum)
                c0 = compute_c5(x0)
                if c0 < best_c5:
                    best_h = x0.copy()
                    best_c5 = c0
                if budget_s - (time.time() - t0) > 15:
                    bnds = Bounds(np.zeros(n), np.ones(n))
                    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
                    try:
                        res = minimize(obj_smooth, x0, args=(10000,), method='SLSQP',
                                       bounds=bnds, constraints=cons,
                                       options={'maxiter': 2000, 'ftol': 1e-15})
                        h_new = project_feasible(res.x, target_sum)
                        c5 = compute_c5(h_new)
                        if c5 < best_c5:
                            best_h = h_new.copy()
                            best_c5 = c5
                    except Exception:
                        pass

    # Phase 5: Final coordinate descent + high-beta SLSQP
    if budget_s - (time.time() - t0) > 10:
        h_final = best_h.copy()
        h_final = coord_descent_pass(h_final, n, max_passes=20)
        h_final = project_feasible(h_final, target_sum)
        c_final = compute_c5(h_final)
        if c_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c_final

    if budget_s - (time.time() - t0) > 12:
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
        for beta in [50000, 200000, 500000]:
            if budget_s - (time.time() - t0) < 7:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(beta,), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass
        if budget_s - (time.time() - t0) > 5:
            h_final = best_h.copy()
            h_final = coord_descent_pass(h_final, n, max_passes=10)
            h_final = project_feasible(h_final, target_sum)
            c_final = compute_c5(h_final)
            if c_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c_final

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
