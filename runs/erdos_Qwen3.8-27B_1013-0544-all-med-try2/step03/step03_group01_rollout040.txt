def project_feasible(h):
    n = len(h)
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

            j1 = i - offset
            j2 = i + offset

            if 0 <= j1 < n:
                val += (1.0 - h0[j1])
            if 0 <= j2 < n:
                val -= h0[j2]
            grad[k, i] = dx * val

    # s.t. grad[k] · (h - h0) + c0[k] ≤ t  for all k
    #      0 ≤ h ≤ 1
    #      sum(h) = n/2
    #
    # Variables: h[0..n-1], t
    # Total: n+1 variables
    #
    # Objective: minimize t => c_obj = [0, 0, ..., 0, 1]
    #
    # Constraints:
    # For each k: grad[k,0]*(h[0]-h0[0]) + ... + grad[k,n-1]*(h[n-1]-h0[n-1]) - t ≤ -c0[k]
    #   => grad[k,0]*h[0] + ... + grad[k,n-1]*h[n-1] - t ≤ -c0[k] + grad[k]·h0
    #
    # Equality: sum(h) = n/2
    # Bounds: 0 ≤ h[i] ≤ 1, t ≥ 0 (or t ≥ -inf)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each k: [grad[k,0], ..., grad[k,n-1], -1] @ [h, t] ≤ -c0[k] + grad[k]·h0
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    for k in range(2*n-1):
        A_ub[k, :n] = grad[k, :]
        A_ub[k, n] = -1.0
        b_ub[k] = -c0[k] + np.dot(grad[k, :], h0)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]  # t is free
    
    # Solve LP
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        h_new = result.x[:n]
        t_new = result.x[n]
        return h_new, t_new
    return h0, c0.max()

```python
import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if h_proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_grad(h, alpha, n):
    """Compute gradient of the smooth max approximation."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_scaled = alpha * c
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    p = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for m in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            offset = k - (n - 1)
            j1 = m - offset
            j2 = m + offset
            if 0 <= j1 < n:
                s += p[k] * (1.0 - h[j1])
            if 0 <= j2 < n:
                s -= p[k] * h[j2]
        grad[m] = dx * s
    return grad

def hard_subgrad(h, n, tol=0.01):
    """Compute subgradient of the hard max."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    active = c >= c_max - tol
    n_active = active.sum()
    
    grad = np.zeros(n)
    for m in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            if not active[k]:
                continue
            offset = k - (n - 1)
            j1 = m - offset
            j2 = m + offset
            if 0 <= j1 < n:
                s += (1.0 - h[j1])
            if 0 <= j2 < n:
                s -= h[j2]
        grad[m] = dx * s / n_active
    return grad, c, c_max

def gp_step(h, n, alpha, lr):
    """One gradient step with projection."""
    g = smooth_grad(h, alpha, n)
    h_new = project_feasible(h - lr * g)
    return h_new

def lp_step(h, n):
    """One step of linearization method using LP."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Compute gradient matrix
    # grad[k, m] = dc[k]/dh[m]
    grad = np.zeros((2 * n - 1, n))
    for m in range(n):
        for k in range(2 * n - 1):
            offset = k - (n - 1)
            j1 = m - offset
            j2 = m + offset
            val = 0.0
            if 0 <= j1 < n:
                val += (1.0 - h[j1])
            if 0 <= j2 < n:
                val -= h[j2]
            grad[k, m] = dx * val
    
    # LP: min t
    # s.t. grad[k,:] · h' - t ≤ -c[k] + grad[k,:] · h  for all k
    #      sum(h') = n/2
    #      0 ≤ h' ≤ 1
    #
    # Variables: [h'(0), ..., h'(n-1), t]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        A_ub[k, :n] = grad[k, :]
        A_ub[k, n] = -1.0
        b_ub[k] = -c[k] + np.dot(grad[k, :], h)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            h_new = res.x[:n]
            return project_feasible(h_new)
    except Exception:
        pass
    return h.copy()

def local_search(h, n, deadline, deltas):
    """Pairwise local search."""
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                for s in (1.0, -1.0):
                    hi = h[i] + s * d
                    hj = h[j] - s * d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i] = hi
                        h[j] = hj
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                        else:
                            h[i] -= s * d
                            h[j] += s * d
    return h, best_c
