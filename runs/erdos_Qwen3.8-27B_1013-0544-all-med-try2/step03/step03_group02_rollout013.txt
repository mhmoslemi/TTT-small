    return float(np.max(corr) * dx)

def get_max_lag(h):

def grad_at_lag(h, lag_idx):
    """Compute gradient of correlation at a specific lag."""
    n = len(h)
    k = lag_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] -= h[:n-k]
    else:
        ka = -k
        grad[ka:] += (1.0 - h[:n-ka])

def compute_c5(h):

        g[lag:] -= h[:n-lag]
    else:
        la = -lag
        g[la:] += (1.0 - h[:n-la])
        g[:n-la] -= h[la:]
    return g

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13: break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0: free = np.ones(len(h), dtype=bool)
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta*(x-m), -500, 500)
    return m + np.log(np.sum(np.exp(xs))/len(x))/beta

def _sml_obj(h, beta, dx):
    return _smooth_max(np.correlate(h, 1.0-h, mode="full"), beta) * dx

def _sum_fun(x, tgt): return x.sum() - tgt
def _sum_jac(x, tgt): return np.ones(x.shape)

def resample(h, n_new):
    if len(h) == n_new: return h.copy()
    return np.interp(np.linspace(0, 2.0, n_new), np.linspace(0, 2.0, len(h)), h)

    try:

        return project_h(r.x, tgt)

def _subgrad_bt(h0, tgt, n, niter, t0, bud):
    h = h0.copy()
    best = compute_c5(h)
    stall = 0
    for it in range(niter):
        if bud - (time.time() - t0) < 2:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top3 = np.argsort(corr)[::-1][:3]
        g = np.zeros(n)
        for idx in top3:
            g += _grad_lag(h, int(idx) - (n - 1))
        g /= (np.linalg.norm(g) + 1e-15)
        g = -g
        alpha = 0.08

            ct = compute_c5(t)

                stall = 0
                break

def _coord(h, tgt, n, sweeps, t0, bud):
