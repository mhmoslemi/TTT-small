def active_set_qp(h_old, active_indices, n_lags):
    """Solve a QP to minimize the sum of squared active overlaps (linearized)."""
    n = len(h_old)
    dx = 2.0 / n
    corr = np.correlate(h_old, 1.0 - h_old, mode="full")
    overlaps = corr * dx
    
    # Build the linearized objective
    # For each active shift k (index m in corr):
    # linearized_overlap_m = overlap_m + sum_j (h[j] - h_old[j]) * grad_m[j]
    # = (overlap_m - h_old @ grad_m) + h @ grad_m
    # Let b_m = overlap_m - h_old @ grad_m (constant term)
    # Let A_m = grad_m (row of the linearized matrix)
    # Objective: sum_m (b_m + h @ A_m)^2
    
    # This is a QP: minimize (Ah + b)^T (Ah + b) = h^T A^T A h + 2 b^T A h + b^T b
    # P = A^T A, q = A^T b
    
    # Compute gradients for active shifts
    A_rows = []
    b_vals = []
    for m in active_indices:
        k = m - (n - 1)
        grad = np.zeros(n)
        # h[j] appears as h[i] when i=j: contribution (1-h[j+k]) if valid
        j_lo = max(0, -k)
        j_hi = min(n, n - k)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += (1.0 - h_old[j_lo+k:j_hi+k])
        # h[j] appears as h[i+k] when i=j-k: contribution -h[j-k] if valid
        j_lo2 = max(0, k)
        j_hi2 = min(n, n + k)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= h_old[j_lo2-k:j_hi2-k]
        A_rows.append(grad)
        b_vals.append(overlaps[m] - h_old @ grad)
    
    A_mat = np.array(A_rows)  # shape (n_active, n)
    b_vec = np.array(b_vals)  # shape (n_active,)
    
    # QP: minimize h^T P h + q^T h
    # P = A^T A + regularization
    # q = 2 * A^T b
    P = A_mat.T @ A_mat + 1e-6 * np.eye(n)  # small regularization for numerical stability
    q = 2.0 * A_mat.T @ b_vec
    
    # Solve with SLSQP or cvxpy
    h_var = cp.Variable(n)
    obj = cp.quad_form(h_var, cp.psd_wrap(P)) + q @ h_var
    constraints = [cp.sum(h_var) == n/2.0, h_var >= 0, h_var <= 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)
    prob.solve(solver=cp.SCS)  # or cvxopt
    
    if h_var.value is not None:
        return project_feasible(h_var.value, n/2.0)
    return h_old

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)

        else:
            h[free] += diff / free.sum()

def compute_grads(h):
    """Compute gradient matrix for all shifts."""
    n = len(h)
    A = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        j_lo = max(0, -k)
        j_hi = min(n, n - k)
        if j_hi > j_lo:

        j_lo2 = max(0, k)
        j_hi2 = min(n, n + k)
        if j_hi2 > j_lo2:
            A[m, j_lo2:j_hi2] -= h[j_lo2-k:j_hi2-k]

def lp_linearize_solve(h_old, n_iters=30):
    n = len(h_old)
    dx = 2.0 / n
    h = h_old.copy()
    c_old = compute_c5(h)
    for _ in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        A = compute_grad_matrix(h)
        rhs = -overlaps + A @ h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            A @ h_var - t_var <= rhs,
            cp.sum(h_var) == n / 2.0,
            h_var >= 0, h_var <= 1
        ]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        if h_var.value is None or np.any(np.isnan(h_var.value)):
            break
        h_new = project_feasible(h_var.value, n / 2.0)
        c_new = compute_c5(h_new)
        if c_new < c_old:
            h = h_new
            c_old = c_new
        else:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = np.full(200, 0.5)

    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()

    # Phase 1: Active set QP
    h_cur = active_set_optimize(h_cur, [5, 8, 12, 20, 30], max_rounds=25, t0=t0, budget_s=budget_s)
    c = compute_c5(h_cur)
    if c < best_c5:
        best_c5 = c
        h_best = h_cur.copy()

    # Phase 2: LP linearization
    if budget_s - (time.time() - t0) > 80:
        h_lp = lp_linearize_solve(h_best, n_iters=25)
        c_lp = compute_c5(h_lp)
        if c_lp < best_c5:
            best_c5 = c_lp
            h_best = h_lp.copy()

    # Phase 3: SLSQP continuation
    if budget_s - (time.time() - t0) > 100:
        for beta in [50, 100, 200, 400]:
            if budget_s - (time.time() - t0) < 30:
                break
            h_new = slsqp_solve(h_best, beta, 350, top_k=15)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                h_best = h_new.copy()

    # Phase 4: Basin hopping round 1
    h_best, best_c5 = basin_hop(h_best, best_c5, n_hops=14, seed=seed + 1, t0=t0, budget_s=budget_s)

    # Phase 5: Active set refinement after hopping
    if budget_s - (time.time() - t0) > 60:
        h_ref = active_set_optimize(h_best, [3, 5, 8, 12], max_rounds=20, t0=t0, budget_s=budget_s)
        c_ref = compute_c5(h_ref)
        if c_ref < best_c5:
            best_c5 = c_ref
            h_best = h_ref.copy()

    # Phase 6: Try higher resolutions
    for n_try in [300, 400]:
        if budget_s - (time.time() - t0) < 150:
            break
        h_interp = interp_to_n(h_best, n_try)
        h_c = active_set_optimize(h_interp, [5, 10, 20], max_rounds=15, t0=t0, budget_s=budget_s)
        if budget_s - (time.time() - t0) > 50:
            h_c = slsqp_solve(h_c, 300, 300, top_k=15)
        c_try = compute_c5(h_c)
        if c_try < best_c5:
            best_c5 = c_try
            h_best = h_c.copy()

    # Phase 7: Basin hopping round 2 (finer)
    h_best, best_c5 = basin_hop(h_best, best_c5, n_hops=12, seed=seed + 200, t0=t0, budget_s=budget_s)

    # Phase 8: Final active set + LP
    if budget_s - (time.time() - t0) > 50:
        h_final = active_set_optimize(h_best, [3, 5, 8, 15], max_rounds=20, t0=t0, budget_s=budget_s)
        c_final = compute_c5(h_final)
        if c_final < best_c5:
            best_c5 = c_final
            h_best = h_final.copy()

    # Phase 9: Very final SLSQP squeeze
    if budget_s - (time.time() - t0) > 35:
        h_fin = slsqp_solve(h_best, 600, 600, top_k=10)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin.copy()

    # Phase 10: One last QP pass
    if budget_s - (time.time() - t0) > 20:
        h_last = qp_active_solve(h_best, 8)
        c_last = compute_c5(h_last)
        if c_last < best_c5:
            best_c5 = c_last
            h_best = h_last.copy()

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
