def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    cmax = c.max()
    shifted = alpha * (c - cmax)
    e = np.exp(shifted)
    val = cmax + np.log(e.sum()) / alpha
    w = e / e.sum()
    return val, w

    """Compute gradient of soft-max objective w.r.t. h."""
    grad = np.zeros(n)
    nn1 = n - 1
    for idx in range(2 * n - 1):

        j2 = min(n, n - lag)

        # -h[j-lag] contribution
        j3 = max(0, lag)
        j4 = min(n, n + lag)
        if j4 > j3:
            grad[j3:j4] -= wk * h[j3 - lag:j4 - lag]
    grad *= dx
    return grad

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_eval(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    cmax = c.max()
    shifted = alpha * (c - cmax)
    e = np.exp(shifted)
    s = e.sum()
    val = cmax + np.log(s) / alpha
    w = e / s
    return val, w

def compute_grad(h, w, n, dx, top_k=None):

    nn1 = n - 1
    if top_k is not None and top_k < 2 * nn1 + 1:

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    # Phase 1: Binary local search
    p1 = t0 + 0.06 * total
    hb, cb = binary_swap_search(h0, n, p1, max_sweeps=150, rng=rng)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 2: PGD with increasing alpha (main engine)
    alphas = [10, 25, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 25000, 50000, 100000, 250000, 500000, 1000000]
    p2 = t0 + 0.65 * total
    h = best_h.copy()
    for alpha in alphas:
        rem = p2 - time.time()
        if rem < 3:
            break
        tb = min(18.0, rem)
        h, c = pgd_optimize(h, n, alpha, t0 + time.time() - t0 + tb, max_iter=25000)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 3: Subgradient polish
    p3 = t0 + 0.75 * total
    hs, cs = subgrad_optimize(best_h, n, p3, max_iter=30000)
    if cs < best_c:
        best_c, best_h = cs, hs.copy()
    # Phase 4: Binary search on best
    p4 = t0 + 0.84 * total
    hb2, cb2 = binary_swap_search(best_h, n, p4, max_sweeps=200, rng=rng)
    if cb2 < best_c:
        best_c, best_h = cb2, hb2.copy()
    # Phase 5: High-alpha PGD final polish
    p5 = t0 + 0.92 * total
    h = best_h.copy()
    for alpha in [200000, 1000000, 5000000]:
        if time.time() > p5 - 3:
            break
        h, c = pgd_optimize(h, n, alpha, p5 - 3, max_iter=20000)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 6: Perturbation restarts
    p6 = dl
    scales = [0.001, 0.003, 0.008, 0.02, 0.05]
    for trial in range(5):
        if time.time() > p6 - 8:
            break
        sc = scales[trial % len(scales)]
        ht = best_h + sc * rng.randn(n)
        ht = project_feasible(ht, n)
        for alpha in [5000, 50000, 500000]:
            if time.time() > p6 - 4:
                break
            ht, c = pgd_optimize(ht, n, alpha, p6 - 4, max_iter=10000)
            if c < best_c:
                best_c, best_h = c, ht.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
