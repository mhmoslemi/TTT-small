def binary_push(h, n):
    target = n / 2.0
    counts = np.diff(np.histogram(h, bins=100, range=(0, 1))[0]).astype(float)
    h_bin = (h > 0.5).astype(float)
    deficit = target - h_bin.sum()
    if deficit > 0:
        order = np.argsort(h_bin + (0.5 - np.abs(h - 0.5)) * 0.1)
        for idx in order:
            if deficit <= 0:
                break
            if h_bin[idx] < 1.0:
                h_bin[idx] = 1.0
                deficit -= 1
    elif deficit < 0:
        order = np.argsort(h_bin - (0.5 - np.abs(h - 0.5)) * 0.1)
        for idx in order:
            if deficit >= 0:
                break
            if h_bin[idx] > 0.0:
                h_bin[idx] = 0.0
                deficit += 1
    return project_feasible(h_bin, n)

def perturb(h, n, rng, scale):
    return project_feasible(h + rng.randn(n) * scale, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 650
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Focused L-BFGS (top-10 to top-30 lags)
    t1 = min(t0 + 250, deadline - 500)
    best_h, best_c = lbfgs_focused(
        best_h, n, t1,
        alpha_seq=[3000, 10000, 30000, 100000],
        top_m_seq=[10, 15, 20, 30],
        mu=50.0, max_iters=500)

    # Phase 2: Binary push + re-optimize
    if time.time() < deadline - 400:
        h_bin = binary_push(best_h, n)
        c_bin = compute_c5(h_bin, n)
        if c_bin < best_c:
            best_h, best_c = h_bin, c_bin
        t2 = min(time.time() + 80, deadline - 350)
        h_t, c_t = lbfgs_focused(
            best_h, n, t2,
            alpha_seq=[5000, 20000, 80000],
            top_m_seq=[10, 15, 25],
            mu=100.0, max_iters=400)
        if c_t < best_c:
            best_c, best_h = c_t, h_t

    # Phase 3: Full smoothmax L-BFGS (all lags)
    if time.time() < deadline - 300:
        t3 = min(time.time() + 120, deadline - 250)
        h_t, c_t = lbfgs_full(
            best_h, n, t3,
            alpha_seq=[50000, 200000, 1000000],
            mu=200.0, max_iters=400)
        if c_t < best_c:
            best_c, best_h = c_t, h_t

    # Phase 4: Coordinate search (gradient-directed)
    if time.time() < deadline - 200:
        t4 = min(time.time() + 80, deadline - 150)
        best_h, best_c = coord_search(best_h, n, t4, k_top=30)

    # Phase 5: Restart from perturbed version
    if time.time() < deadline - 120:
        h_p = perturb(best_h, n, rng, 0.015)
        t5 = min(time.time() + 50, deadline - 80)
        h_t, c_t = lbfgs_focused(
            h_p, n, t5,
            alpha_seq=[10000, 50000],
            top_m_seq=[10, 20],
            mu=100.0, max_iters=300)
        if c_t < best_c:
            best_c, best_h = c_t, h_t

    # Phase 6: Try n=500 (coarser grid, different optimum)
    if time.time() < deadline - 100:
        n_c = 500
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_c, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n_c)
        t6 = min(time.time() + 50, deadline - 60)
        h_t, c_t = lbfgs_focused(
            h_int, n_c, t6,
            alpha_seq=[5000, 30000, 100000],
            top_m_seq=[10, 20, 30],
            mu=50.0, max_iters=350)
        if c_t < best_c:
            best_c, best_h, n = c_t, h_t, n_c

    # Phase 7: Final coordinate polish
    if time.time() < deadline - 10:
        best_h, best_c = coord_search(best_h, n, deadline - 5, k_top=25,
                                      steps=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5])

    # Phase 8: Final focused L-BFGS squeeze
    if time.time() < deadline - 2:
        h_t, c_t = lbfgs_focused(
            best_h, n, deadline - 1,
            alpha_seq=[200000, 500000],
            top_m_seq=[8, 15],
            mu=500.0, max_iters=200)
        if c_t < best_c:
            best_c, best_h = c_t, h_t

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
