    n = len(h)

    g = corr * dx

    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    if rng is None:
        rng = np.random.RandomState(0)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        order = rng.permutation(n) if rng is not None else range(n)
        for i in order:
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            m1 = (k1 >= 0) & (k1 < n)
            B[m1] += (1.0 - h[k1[m1]])
            k2 = i + m_range
            m2 = (k2 >= 0) & (k2 < n)
            B[m2] -= h[k2[m2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(50):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sh -= sh.max()
        return m + np.log(np.exp(sh).sum() + 1e-300) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def compute_grad(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx
    g_max = g.max()
    sh = alpha * (g - g_max)
    sh -= sh.max()
    ew = np.exp(sh)
    w = ew / ew.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        wk = w[k] * dx
        j1s = max(0, -s)
        j1e = min(n - 1, n - 1 - s)
        if j1s <= j1e:
            grad[j1s:j1e+1] += wk * (1 - h[j1s+s:j1e+1+s])
        j2s = max(0, s)
        j2e = min(n - 1, n - 1 + s)
        if j2s <= j2e:
            grad[j2s:j2e+1] -= wk * h[j2s-s:j2e+1-s]
    return grad

def lse_obj(h, alpha, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    g_max = corr.max()
    sh = alpha * (corr - g_max)
    sh -= sh.max()
    return g_max + np.log(np.exp(sh).sum() + 1e-300) / alpha

def projected_grad(h, n, alpha_start, alpha_end, deadline, max_iter=50000):
    dx = 2.0 / n
    h = project_box_sum(h.copy(), n / 2.0)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = alpha_start
    lr = 1.0 / (16.0 * alpha / n + 4.0 / n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = compute_grad(h, alpha, dx)
        f_cur = lse_obj(h, alpha, dx)
        lr_try = lr
        h_new = h
        for _ in range(40):
            h_try = project_box_sum(h - lr_try * grad, n / 2.0)
            f_try = lse_obj(h_try, alpha, dx)
            if f_try <= f_cur + 1e-5 * lr_try * np.dot(grad, h - h_try):
                h_new = h_try
                break
            lr_try *= 0.5
        h = h_new
        if lr_try > 0:
            lr = min(lr_try * 1.05, 2.0 / (16.0 * alpha / n + 4.0 / n))
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if it % 50 == 0:
            frac = it / max_iter
            alpha = alpha_start * (alpha_end / alpha_start) ** frac
            lr = 1.0 / (16.0 * alpha / n + 4.0 / n)
    return best_h, best_c

def binary_anneal(h_init, n, deadline, max_iter=500000):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    h = np.zeros(n)
    h[np.argsort(h_init)[::-1][:n//2]] = 1.0
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    T = 0.005
    rng = np.random.RandomState(123)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 5000 == 0:
            T *= 0.98
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        delta = np.zeros(2*n-1)
        for idx, sign in [(i, 1.0), (j, -1.0)]:
            k1 = idx - m_range
            m1 = (k1 >= 0) & (k1 < n)
            delta[m1] += sign * (1.0 - h[k1[m1]])
            k2 = idx + m_range
            m2 = (k2 >= 0) & (k2 < n)
            delta[m2] -= sign * h[k2[m2]]
        new_c = (corr + delta).max() * dx
        if new_c < cur_c or rng.rand() < np.exp((cur_c - new_c) / max(T, 1e-10)):
            h[i], h[j] = 1.0, 0.0
            corr = corr + delta
            cur_c = new_c
            ones.remove(j); zeros.remove(i)
            ones.append(i); zeros.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 20
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Binary anneal for alternate start
    p1 = t0 + 0.04 * total
    hb, cb = binary_anneal(h0, n, p1, max_iter=300000)
    alt_h = None
    if cb < best_c * 1.01:
        alt_h = hb.copy()

    # Phase 2: Coordinate descent from initial
    p2 = t0 + 0.12 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=300, deadline=p2, rng=rng)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 3: SLSQP main
    p3 = t0 + 0.45 * total
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p3 - 5:
            break
        h = slsqp_smooth(h, n, alpha, 2500, 1e-15)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if time.time() < p3 - 20:
            hc2, _ = coord_descent(h, n, max_sweeps=50, deadline=p3-15, rng=rng)
            hc2 = project_feasible(hc2, n)
            c2 = compute_c5(hc2, n)
            if c2 < best_c:
                best_c, best_h = c2, hc2.copy()
            h = best_h.copy()

    # Phase 4: Projected gradient
    p4 = t0 + 0.60 * total
    hg, cg = projected_grad(best_h, n, 500, 200000, p4, max_iter=80000)
    hc3, _ = coord_descent(hg, n, max_sweeps=80, deadline=p4, rng=rng)
    hc3 = project_feasible(hc3, n)
    c3 = compute_c5(hc3, n)
    if c3 < best_c:
        best_c, best_h = c3, hc3.copy()

    # Phase 5: High-alpha SLSQP + coord descent
    p5 = t0 + 0.78 * total
    for alpha in [10000000, 50000000, 200000000]:
        if time.time() > p5 - 5:
            break
        h = slsqp_smooth(best_h, n, alpha, 4000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        hc4, _ = coord_descent(best_h, n, max_sweeps=150, deadline=p5-3, rng=rng)
        hc4 = project_feasible(hc4, n)
        c4 = compute_c5(hc4, n)
        if c4 < best_c:
            best_c, best_h = c4, hc4.copy()

    # Phase 6: Try alternate binary start
    if alt_h is not None and time.time() < dl - 40:
        p6 = dl - 35
        ha, _ = coord_descent(alt_h, n, max_sweeps=200, deadline=p6, rng=rng)
        ha = project_feasible(ha, n)
        ca = compute_c5(ha, n)
        if ca < best_c:
            best_c, best_h = ca, ha.copy()
        ha = slsqp_smooth(ha, n, 100000, 2000, 1e-15)
        ca = compute_c5(ha, n)
        if ca < best_c:
            best_c, best_h = ca, ha.copy()

    # Phase 7: Different n values
    p7 = dl - 15
    for nh in [300, 500, 384, 420, 350]:
        if time.time() > p7 - 8:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, _ = coord_descent(hh, nh, max_sweeps=80, deadline=p7-8, rng=rng)
        hh = project_feasible(hh, nh)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh
        if time.time() < p7 - 5:
            hh2 = slsqp_smooth(hh, nh, 5000000, 1500, 1e-15)
            ch2 = compute_c5(hh2, nh)
            if ch2 < best_c:
                best_c, best_h, n = ch2, hh2.copy(), nh

    # Final
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
