            if tot < 1e-14: break

            tot = h.sum()
            if tot < 1e-14: break

            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lag_focused_search(h, n, deadline, top_k=15):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    stall = 0
    while time.time() < deadline and stall < 50:
        c = compute_c5_vec(h, n)
        top_lags = np.argsort(c)[-top_k:][::-1]
        g = np.zeros(n)
        for lag in top_lags:
            g += grad_c5_at_lag(h, n, int(lag))
        g += 10.0 * (h.sum() - n / 2.0) * np.ones(n)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g /= g_norm
        improved = False
        for lr in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return h, best_c

def lbfgs_refine(h, n, deadline, max_iters=300):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alpha = 100000
    mu = 200.0
    def obj(x):
        xc = np.clip(x, 0, 1)
        c = compute_c5_vec(xc, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
        val += mu * (xc.sum() - n / 2.0) ** 2
        return float(val)
    def grad(x):
        xc = np.clip(x, 0, 1)
        g = smooth_max_grad(xc, n, alpha)
        g += 2.0 * mu * (xc.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_iters, 'ftol': 1e-17, 'gtol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h, best_c

def fourier_perturb(h, n, rng, n_harm=5, amp=0.08):
    t = np.arange(n) * (2.0 / n)
    pert = np.zeros(n)
    for k in range(1, n_harm + 1):
        a = rng.uniform(-amp, amp)
        b = rng.uniform(-amp, amp)
        pert += a * np.sin(np.pi * k * t) + b * np.cos(np.pi * k * t)
    return project_feasible(h + pert, n)

def noise_perturb(h, n, rng, sigma=0.03):
    noise = rng.normal(0, sigma, n)
    return project_feasible(h + noise, n)

def coordinate_descent(h, n, deadline, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    for d in step_sizes:
        while time.time() < deadline:
            improved = False
            c = compute_c5_vec(h, n)
            top_lags = np.argsort(c)[-20:]
            top_set = set(int(x) for x in top_lags)
            for m_idx in top_set:
                if time.time() > deadline:
                    return h, best_c
                k = m_idx - (n - 1)
                if k >= 0:
                    indices_a = list(range(0, n - k))
                    indices_b = list(range(k, n))
                else:
                    kk = -k
                    indices_a = list(range(kk, n))
                    indices_b = list(range(0, n - kk))
                pairs = list(zip(indices_a, indices_b))
                rng_local = np.random.RandomState(m_idx + 7)
                rng_local.shuffle(pairs)
                for ia, ib in pairs[:30]:
                    if time.time() > deadline:
                        return h, best_c
                    for sign in [1, -1]:
                        delta = sign * d
                        if h[ia] + delta < 0 or h[ia] + delta > 1:
                            continue
                        if h[ib] - delta < 0 or h[ib] - delta > 1:
                            continue
                        h[ia] += delta
                        h[ib] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[ia] -= delta
                            h[ib] += delta
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS refinement
    if time.time() < deadline - 250:
        best_h, best_c = lbfgs_refine(best_h, n, deadline - 250, max_iters=400)

    # Phase 2: Adam with adaptive alpha
    if time.time() < deadline - 180:
        t_alloc = min(120, deadline - time.time() - 80)
        h_a, c_a = adam_optimize(best_h, n, time.time() + t_alloc, lr=0.004, alpha=200, mu=30.0)
        if c_a < best_c:
            best_c, best_h = c_a, h_a

    # Phase 3: Lag-focused search
    if time.time() < deadline - 120:
        t_alloc = min(60, deadline - time.time() - 50)
        best_h, best_c = lag_focused_search(best_h, n, time.time() + t_alloc, top_k=20)

    # Phase 4: Lag-focused coordinate descent (targeted pair moves on top lags)
    if time.time() < deadline - 80:
        t_alloc = min(80, deadline - time.time() - 30)
        best_h, best_c = coordinate_descent(best_h, n, time.time() + t_alloc)

    # Phase 5: Multi-start exploration
    if time.time() < deadline - 60:
        n_tries = 6
        for trial in range(n_tries):
            if time.time() > deadline - 40:
                break
            perm = trial % 2
            if perm == 0:
                h_try = fourier_perturb(best_h, n, rng, n_harm=rng.randint(2, 8), amp=rng.uniform(0.03, 0.12))
            else:
                h_try = noise_perturb(best_h, n, rng, sigma=rng.uniform(0.02, 0.06))
            t_alloc = min(50, deadline - time.time() - 20)
            if t_alloc < 15:
                break
            h_t, c_t = adam_optimize(h_try, n, time.time() + t_alloc * 0.4, lr=0.003, alpha=500, mu=50.0)
            if time.time() < deadline - 15:
                h_t, c_t = lag_focused_search(h_t, n, time.time() + (deadline - 10 - time.time()) * 0.5, top_k=10)
            if c_t < best_c:
                best_c, best_h = c_t, h_t

    # Phase 6: Final coordinated refinement
    if time.time() < deadline - 3:
        best_h, best_c = lag_focused_search(best_h, n, deadline - 1, top_k=10)
        best_h, best_c = coordinate_descent(best_h, n, deadline - 0.5, step_sizes=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
