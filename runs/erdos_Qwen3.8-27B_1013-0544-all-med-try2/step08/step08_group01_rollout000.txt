import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _obj_smooth_max(x, n, alpha, active_idx=None):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    if active_idx is not None:
        c = c[active_idx]
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def _jac_smooth_max(x, n, alpha, active_idx=None):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    if active_idx is not None:
        c_sub = c[active_idx]
        all_idx = active_idx
        c_use = c_sub
    else:
        all_idx = np.arange(len(c))
        c_use = c
    m = c_use.max()
    sh = alpha * (c_use - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for ii, m in enumerate(all_idx):
        if w[ii] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[ii] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def lbfgs_optimize(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        tr = deadline - time.time() - 5
        if tr < 5:
            break
        try:
            res = minimize(_obj_smooth_max, h, jac=_jac_smooth_max,
                          args=(n, alpha, None), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': 10000, 'maxls': 50})
            h_new = project_feasible(res.x, n)

            if c_new < best_c:

    return h, best_c

            improved = False
            for idx in range(n):
                if time.time() > deadline:
                    return h, best_c
                i = order[idx]

                    if i == j:

def fourier_param_optimize(h, n, deadline, n_freq=25):
    """Optimize in Fourier parameter space."""
    x = np.arange(n)
    # Initialize Fourier coefficients from current h
    hc = h - h.mean()
    coeffs = np.zeros(2 * n_freq)
    for k in range(1, n_freq + 1):
        coeffs[2*(k-1)] = 2.0/n * np.sum(hc * np.cos(2*np.pi*k*x/n))
        coeffs[2*(k-1)+1] = 2.0/n * np.sum(hc * np.sin(2*np.pi*k*x/n))
    
    def fourier_to_h(params):
        a = params[:n_freq]
        b = params[n_freq:]
        hh = np.full(n, h.mean())
        for k in range(1, n_freq + 1):
            hh += a[k-1] * np.cos(2*np.pi*k*x/n)
            hh += b[k-1] * np.sin(2*np.pi*k*x/n)
        return project_feasible(hh, n)
    
    best_c = compute_c5(h, n)
