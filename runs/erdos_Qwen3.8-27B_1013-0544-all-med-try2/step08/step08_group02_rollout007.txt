```
    e=np.exp(sh-smax)
    s=e.sum()
    if s <=0: w zeros

    grad=np.zeros(n)
    off=n-1
```

```python
import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(120):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            room = 1.0 - h
            tot = room[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += diff * room[mask] / tot
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += diff * h[mask] / tot
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if diff != 0.0:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] -= diff / float(mask.sum())
            h = np.clip(h, 0.0, 1.0)
    return h


def fix_binary_count(x, n):
    x = x.astype(np.int8, copy=False)
    target = n // 2
    cnt = int(x.sum())
    if cnt > target:
        idx = np.where(x == 1)[0]
        x[idx[cnt - target:]] = 0
    elif cnt < target:
        idx = np.where(x == 0)[0]
        x[idx[:target - cnt]] = 1
    return x


def random_binary(n, rng):
    x = np.zeros(n, dtype=np.int8)
    idx = rng.choice(n, size=n // 2, replace=False)
    x[idx] = 1
    return x


def sturmian_binary(n, rng):
    theta = 0.6180339887 + rng.uniform(-0.08, 0.08)
    beta = rng.uniform(0.0, 1.0)
    t = theta * np.arange(n, dtype=float) + beta
    f = t - np.floor(t)
    x = (f < 0.5).astype(np.int8)
    target = n // 2
    cnt = int(x.sum())
    if cnt > target:
        idx = np.where(x == 1)[0]
        order = idx[np.argsort(f[idx])[-(cnt - target):]]
        x[order] = 0
    elif cnt < target:
        idx = np.where(x == 0)[0]
        order = idx[np.argsort(f[idx])[:target - cnt]]
        x[order] = 1
    return x


def random_run_lengths(m, r, rng):
    lengths = np.ones(r, dtype=np.int64)
    rem = m - r
    if rem > 0:
        lengths += rng.multinomial(rem, np.ones(r, dtype=float) / float(r))
    return lengths


def run_binary(n, rng):
    m = n // 2
    min_r = max(4, m // 15)
    max_r = min(m - 1, m // 3)
    if max_r < min_r:
        min_r = max(1, max_r)
    if max_r <= 0:
        return random_binary(n, rng)
    r = int(rng.randint(min_r, max_r + 1))
    a = random_run_lengths(m, r, rng)
    b = random_run_lengths(m, r, rng)
    x = np.empty(n, dtype=np.int8)
    pos = 0
    for ai, bi in zip(a, b):
        x[pos:pos + ai] = 1
        pos += ai
        x[pos:pos + bi] = 0
        pos += bi
    if pos < n:
        x[pos:] = 0
    elif pos > n:
        x = x[:n]
    return fix_binary_count(x, n)


def binary_from_float(h, n):
    x = (h > 0.5).astype(np.int8)
    target = n // 2
    cnt = int(x.sum())
    if cnt > target:
        idx = np.where(x == 1)[0]
        order = idx[np.argsort(h[idx])[:cnt - target]]
        x[order] = 0
    elif cnt < target:
        idx = np.where(x == 0)[0]
        order = idx[np.argsort(h[idx])[-(target - cnt):]]
        x[order] = 1
    return x


def init_state(x, n):
    x = fix_binary_count(x, n).astype(np.int8, copy=True)
    om = 1 - x
    x32 = x.astype(np.int32)
    om32 = om.astype(np.int32)
    coeffs = np.correlate(x32, om32, mode="full").astype(np.int16)
    ccur = int(coeffs.max())
    half = n // 2
    ones = np.where(x == 1)[0].astype(np.int32)
    zeros = np.where(x == 0)[0].astype(np.int32)
    pos_one = np.full(n, -1, dtype=np.int32)
    pos_zero = np.full(n, -1, dtype=np.int32)
    pos_one[ones] = np.arange(len(ones), dtype=np.int32)
    pos_zero[zeros] = np.arange(len(zeros), dtype=np.int32)
    return x, om, coeffs, ones, zeros, pos_one, pos_zero, ccur


def eval_swap_into(x, om, i, j, delta, coeffs, tmp, n):
    off = n - 1
    delta.fill(0)

    r = i
    if r:
        delta[off - r:off] -= om[:r]
    m = n - r - 1
    if m:
        delta[off + 1:off + 1 + m] -= om[r + 1:]
        delta[off - m:off] += x[r + 1:][::-1]
    if r:
        delta[off + 1:off + r + 1] += x[:r][::-1]

    r = j
    if r:
        delta[off - r:off] += om[:r]
        delta[off + 1:off + r + 1] -= x[:r][::-1]
    m = n - r - 1
    if m:
        delta[off + 1:off + 1 + m] += om[r + 1:]
        delta[off - m:off] -= x[r + 1:][::-1]

    delta[off + j - i] += 1
    delta[off + i - j] += 1
    delta[off] = 0
    np.add(coeffs, delta, out=tmp)
    return int(tmp.max())


def update_lists(ones, zeros, pos_one, pos_zero, i, j, half):
    io = pos_one[i]
    last = ones[half - 1]
    ones[io] = last
    pos_one[last] = io
    pos_one[i] = -1

    jz = pos_zero[j]
    lastz = zeros[half - 1]
    zeros[jz] = lastz
    pos_zero[lastz] = jz
    pos_zero[j] = -1

    zeros[half - 1] = i
    pos_zero[i] = half - 1
    ones[half - 1] = j
    pos_one[j] = half - 1


def local_improve_state(x, om, coeffs, ones, zeros, pos_one, pos_zero, ccur, n, deadline, rng):
    off = n - 1
    length = 2 * n - 1
    half = n // 2
    delta = np.zeros(length, dtype=np.int16)
    tmp = np.zeros(length, dtype=np.int16)
    stall = 0
    max_eval = 90 + n // 10

    while time.time() < deadline:
        found = False
        order = np.argsort(coeffs)[::-1]
        seen = set()
        used = 0

        for idx in order:
            if used >= max_eval:
                break
            if idx == off:
                continue
            clag = int(coeffs[idx])
            if clag <= 0:
                continue
            k = int(idx) - off
            if k > 0:
                pairs = np.flatnonzero((x[:n - k] == 1) & (om[k:] == 1))
                if pairs.size == 0:
                    continue
                if pairs.size > 32:
                    sel = rng.choice(pairs, size=32, replace=False)
                else:
                    sel = pairs
                for p in sel:
                    q = p + k
                    code = p * n + q
                    if code in seen:
                        continue
                    seen.add(code)
                    used += 1
                    cn = eval_swap_into(x, om, p, q, delta, coeffs, tmp, n)
                    if cn < ccur:
                        x[p] = 0
                        x[q] = 1
                        om[p] = 1
                        om[q] = 0
                        coeffs += delta
                        update_lists(ones, zeros, pos_one, pos_zero, p, q, half)
                        ccur = cn
                        found = True
                        break
            else:
                kk = -k
                base = np.flatnonzero((x[kk:] == 1) & (om[:n - kk] == 1))
                if base.size == 0:
                    continue
                if base.size > 32:
                    base = rng.choice(base, size=32, replace=False)
                for p in base + kk:
                    q = p - kk
                    code = p * n + q
                    if code in seen:
                        continue
                    seen.add(code)
                    used += 1
                    cn = eval_swap_into(x, om, p, q, delta, coeffs, tmp, n)
                    if cn < ccur:
                        x[p] = 0
                        x[q] = 1
                        om[p] = 1
                        om[q] = 0
                        coeffs += delta
                        update_lists(ones, zeros, pos_one, pos_zero, p, q, half)
                        ccur = cn
                        found = True
                        break
            if found:
                break

        if found:
            stall = 0
            continue

        for _ in range(25):
            i = ones[rng.randint(half)]
            j = zeros[rng.randint(half)]
            cn = eval_swap_into(x, om, i, j, delta, coeffs, tmp, n)
            if cn < ccur:
                x[i] = 0
                x[j] = 1
                om[i] = 1
                om[j] = 0
                coeffs += delta
                update_lists(ones, zeros, pos_one, pos_zero, i, j, half)
                ccur = cn
                found = True
                break

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 4:
                break

    return ccur


def binary_ils(x0, n, deadline, rng):
    x, om, coeffs, ones, zeros, pos_one, pos_zero, ccur = init_state(x0, n)
    best_x = x.copy()
    best_coeffs = coeffs.copy()
    best_c = ccur

    local_deadline = min(time.time() + 120.0, deadline - 4.0)
    ccur = local_improve_state(x, om, coeffs, ones, zeros, pos_one, pos_zero, ccur, n, local_deadline, rng)
    if ccur < best_c:
        best_x = x.copy()
        best_coeffs = coeffs.copy()
        best_c = ccur

    kick_sizes = (3, 5, 8, 12, 20, 10, 15, 25)
    ki = 0
    half = n // 2

    while time.time() < deadline:
        p = kick_sizes[ki % len(kick_sizes)]
        ki += 1
        x = best_x.copy()
        om = 1 - x
        coeffs = best_coeffs.copy()
        ccur = best_c
        ones = np.where(x == 1)[0].astype(np.int32)
        zeros = np.where(x == 0)[0].astype(np.int32)
        pos_one = np.full(n, -1, dtype=np.int32)
        pos_zero = np.full(n, -1, dtype=np.int32)
        pos_one[ones] = np.arange(len(ones), dtype=np.int32)
        pos_zero[zeros] = np.arange(len(zeros), dtype=np.int32)

        delta = np.zeros(2 * n - 1, dtype=np.int16)
        tmp = np.zeros(2 * n - 1, dtype=np.int16)
        step = max(3, n // 100 + 2)
        cap = best_c + max(8, n // 50 + p)

        done = 0
        while done < p and time.time() < deadline - 2.0:
            ok = False
            for _ in range(8):
                i = ones[rng.randint(half)]
                j = zeros[rng.randint(half)]
                cn = eval_swap_into(x, om, i, j, delta, coeffs, tmp, n)
                if cn <= ccur + step and cn <= cap:
                    x[i] = 0
                    x[j] = 1
                    om[i] = 1
                    om[j] = 0
                    coeffs += delta
                    update_lists(ones, zeros, pos_one, pos_zero, i, j, half)
                    ccur = cn
                    done += 1
                    ok = True
                    break
            if not ok:
                break

        left = max(1.0, deadline - time.time())
        local_deadline = min(time.time() + max(20.0, 0.25 * left), deadline - 1.5)
        ccur = local_improve_state(x, om, coeffs, ones, zeros, pos_one, pos_zero, ccur, n, local_deadline, rng)
        if ccur < best_c:
            best_x = x.copy()
            best_coeffs = coeffs.copy()
            best_c = ccur

    return best_x, best_c


def start_score(x, n):
    om = 1 - x
    x32 = x.astype(np.int32)
    om32 = om.astype(np.int32)
    return int(np.max(np.correlate(x32, om32, mode="full")))


def smooth_val(x, n, alpha, mu):
    om = 1.0 - x
    c = np.correlate(x, om, mode="full")
    m = float(c.max())
    sh = alpha * (c - m)
    smax = float(sh.max())
    val = m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha
    if mu:
        s = x.sum() - 0.5 * n
        val += mu * s * s
    return float(val)


def smooth_grad(x, n, alpha, mu):
    om = 1.0 - x
    c = np.correlate(x, om, mode="full")
    m = float(c.max())
    sh = alpha * (c - m)
    smax = float(sh.max())
    e = np.exp(sh - smax)
    s = float(e.sum())
    w = e / s
    grad = np.zeros(n, dtype=float)
    off = n - 1
    for idx in range(2 * n - 1):
        wk = float(w[idx])
        if wk < 1e-12:
            continue
        k = idx - off
        if k >= 0:
            grad[:n - k] += wk * om[k:]
            grad[k:] -= wk * x[:n - k]
        else:
            kk = -k
            grad[kk:] += wk * om[:n - kk]
            grad[:n - kk] -= wk * x[kk:]
    if mu:
        s = x.sum() - 0.5 * n
        grad += 2.0 * mu * s
    return grad


def lbfgs_refine(h, n, deadline):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    max_iter = 18 if n > 800 else 32
    max_fun = 80 if n > 800 else 160
    params = [(100, 100), (500, 500), (2000, 2000), (10000, 10000), (50000, 30000)]

    for alpha, mu in params:
        if time.time() > deadline - 4.0:
            break
        try:
            res = minimize(
                smooth_val,
                h,
                jac=smooth_grad,
                args=(n, alpha, mu),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-8, "maxfun": max_fun},
            )
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c + 0.005:
                h = h2
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2
        except Exception:
            pass

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(1, int(budget_s)) - 5
    rng = np.random.RandomState(seed)

    h_arr = None
    try:
        h_arr = np.asarray(initial_h_values, dtype=float)
        if h_arr.size == 0:
            h_arr = None
    except Exception:
        h_arr = None

    if h_arr is None:
        n0 = 500
        h_arr = random_binary(n0, rng).astype(float)
    else:
        n_init = len(h_arr)
        n0 = n_init if n_init % 2 == 0 else n_init + 1
        if n0 != n_init:
            x_old = np.linspace(0.0, 2.0, n_init, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n0, endpoint=False)
            h_arr = np.interp(x_new, x_old, h_arr)
    h_arr = project_feasible(h_arr, n0)

    best_h = h_arr.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    n_list = []
    raw_ns = [n0, 2 * n0, 1000, 750, 500, 600, 300]
    for nn in raw_ns:
        nn = int(nn)
        if nn % 2:
            nn += 1
        if 200 <= nn <= 1000 and nn not in n_list:
            n_list.append(nn)

    weights = []
    for nn in n_list:
        if nn == n0:
            weights.append(3.0)
        elif nn == 1000 or (n0 > 0 and nn == 2 * n0):
            weights.append(2.5)
        elif nn == 750:
            weights.append(2.0)
        elif nn == 500:
            weights.append(1.5)
        else:
            weights.append(1.0)

    reserve = max(85, int(budget_s * 0.14))
    total_w = float(sum(weights))
    idx = 0

    while idx < len(n_list) and time.time() < deadline - reserve:
        avail = deadline - time.time() - reserve
        if avail < 45:
            break
        slot = avail * (weights[idx] / total_w)
        if slot < 22:
            total_w -= weights[idx]
            idx += 1
            continue

        n_try = n_list[idx]
        ndl = min(time.time() + slot, deadline - reserve)

        if best_n != n_try:
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            if n_try % best_n == 0 and n_try >= best_n:
                x_par = np.repeat(binary_from_float(best_h, best_n), n_try // best_n)
            else:
                h_int = np.interp(x_new, x_old, best_h)
                x_par = binary_from_float(h_int, n_try)
        else:
            x_par = binary_from_float(best_h, n_try)

        starts = [x_par, random_binary(n_try, rng), run_binary(n_try, rng)]
        scored = []
        for xs in starts:
            xs = fix_binary_count(xs, n_try)
            scored.append((start_score(xs, n_try), xs.astype(np.int8, copy=True)))
        scored.sort()

        if len(scored) == 1 or slot < 70:
            allocs = [slot * 0.9]
            to_run = scored[:1]
        else:
            allocs = [slot * 0.62, slot * 0.38]
            to_run = scored[:2]

        for sc, xs in zip(to_run, allocs):
            cur_dl = min(time.time() + allocs[scored.index((sc, xs))] if False else 0, ndl, deadline - reserve)

        for si, (sc, xs) in enumerate(to_run):
            alloc = allocs[si] if si < len(allocs) else 0.0
            cur_dl = min(time.time() + alloc, ndl, deadline - reserve)
            if cur_dl - time.time() < 12:
                continue
            bx, bc = binary_ils(xs, n_try, cur_dl, rng)
            cf = float(2.0 * bc) / float(n_try)
            if cf < best_c:
                best_h = bx.astype(float)
                best_c = cf
                best_n = n_try

        total_w -= weights[idx]
        idx += 1

    if time.time() < deadline - 25:
        t_pol = min(time.time() + max(35, deadline - time.time() - 25), deadline - 20)
        hp, cp = lbfgs_refine(best_h, best_n, t_pol)
        if cp < best_c:
            best_h = hp
            best_c = cp
            best_n = len(hp)

    xb = binary_from_float(best_h, best_n)
    cb = float(2.0 * start_score(xb, best_n)) / float(best_n)
    if cb < best_c:
        best_h = xb.astype(float)
        best_c = cb

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
```
