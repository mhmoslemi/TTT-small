```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax_val(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smax_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag in range(2 * n - 1):
        if w[lag] < 1e-14:
            continue
        k = lag - (n - 1)
        wm = w[lag] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def penalty_obj(x, n, alpha, mu):
    return smax_val(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def penalty_grad(x, n, alpha, mu):
    return smax_grad(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0)

def lbfgs_opt(h, n, deadline, alphas, mu=50.0, maxit=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        t_rem = deadline - time.time() - 5
        maxiter = min(maxit, max(50, int(t_rem * 2)))
        try:
            res = minimize(penalty_obj, h, jac=penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxiter * 10,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_opt(h, n, deadline, alpha, maxit=1500):
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x, n_):
        return x.sum() - n_ / 2.0
    def cons_jac(x, n_):
        return np.ones(n_)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (n,)}]
    try:
        res = minimize(smax_val, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_pair_search(h, n, deadline, alpha=500, steps=None):
    """Gradient-guided pair search."""
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in steps:
        while time.time() < deadline:
            improved = False
            g = smax_grad(h, n, alpha)
            # Sort indices by gradient value
            order = np.argsort(g)
            # Try moves from high gradient to low gradient (decrease high, increase low)
            for idx_hi in range(len(order) - 1, -1, -1):
                if time.time() > deadline:
                    return h, best_c
                j = order[idx_hi]
                if h[j] < step:
                    continue
                for idx_lo in range(len(order)):
                    if time.time() > deadline:
                        return h, best_c
                    i = order[idx_lo]
                    if i == j:
                        continue
                    if h[i] + step > 1.0:
                        continue
                    # Try moving: h[i] += step, h[j] -= step
                    h[i] += step
                    h[j] -= step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        break
                    else:
                        h[i] -= step
                        h[j] += step
                if improved:
                    break
            if not improved:
                break
    return h, best_c
```

  where i ranges so that 0 ≤ i < n and 0 ≤ i+k < n.
  
  Now, if we change h[i0] by δ:
  - h[i0] appears in c[k] when i+k = i0, i.e., i = i0-k
    - Valid when 0 ≤ i0-k < n and 0 ≤ i0 < n
    - Change: δ * (1-h[i0-k]) for each valid k
  
  - h[i0] also appears in c[k] through (1-h[i]) when i = i0
    - c[k] = sum_i h[i+k] * (1-h[i])
    - The term with i = i0 is h[i0+k] * (1-h[i0])
    - Change: h[i0+k] * (-δ) for each valid k (where 0 ≤ i0+k < n)
  
  So total change in c[k] from h[i0] += δ:
  Δc[k] = δ * (1-h[i0-k]) (if 0 ≤ i0-k < n) + h[i0+k] * (-δ) (if 0 ≤ i0+k < n)
  
  This is O(n) per update (since there are at most 2n-1 lags affected).
  
  Similarly for h[j0] -= δ:
  Δc[k] = -δ * (1-h[j0-k]) (if 0 ≤ j0-k < n) + h[j0-k] * δ (if 0 ≤ j0+k < n)
  
  Wait, let me redo this. If h[j0] changes by -δ:
  - h[j0] appears in c[k] as h[i+k] when i+k = j0, i = j0-k
    - Change: (-δ) * (1-h[j0-k])
  - h[j0] appears in c[k] as (1-h[i]) when i = j0
    - The term is h[j0+k] * (1-h[j0])
    - Change: h[j0+k] * (-(-δ)) = h[j0+k] * δ
  
  So for a pair move (h[i0] += δ, h[j0] -= δ):
  Δc[k] = δ*(1-h[i0-k]) - δ*h[i0+k] - δ*(1-h[j0-k]) + δ*h[j0+k]
  
  (with appropriate validity checks)
  
  This means we can update the correlation in O(n) time for each pair move, instead of O(n log n) for FFT or O(n²) for direct computation.
  
  For n=300, O(n) = 300 operations vs O(n²) = 90000. That's a 300x speedup!
  
  This is the key insight for making the pair search much faster!

```python
def make_corr_updater(n):
    """Create a class that maintains the correlation and supports O(n) updates."""
    class CorrUpdater:
        def __init__(self, h):
            self.n = n
            self.h = h.copy()
            self.c = np.correlate(self.h, 1 - self.h, mode='full')
            self.max_c = self.c.max()
        
        def apply_move(self, i, j, delta):
            """Apply h[i] += delta, h[j] -= delta and update correlation."""
            n = self.n
            h = self.h
            c = self.c
            # For h[i0] += delta:
            # Δc[k_index] = delta*(1-h[i0-k]) - delta*h[i0+k]
            # where k = k_index - (n-1)
            # So for k_index range:
            # Term 1: 1-h[i0-k] valid when 0 <= i0-k < n, i.e., k in [i0-n+1, i0]
            #   k_index = k + (n-1) in [i0-1, i0+n-1]
            # Term 2: -h[i0+k] valid when 0 <= i0+k < n, i.e., k in [-i0, n-1-i0]
            #   k_index in [n-1-i0, n-1+i0]
            
            i0 = i
            j0 = j
            d = delta
            
            # Term from h[i0] in the "a" position: d * (1-h[i0-k])
            # k range: max(-i0, -(n-1)) to min(n-1-i0, n-1)
            # k_index = k + (n-1)
            k_lo = max(-i0, -(n-1))
            k_hi = min(n-1-i0, n-1)
            for k_idx in range(k_lo + (n-1), k_hi + (n-1) + 1):
                k = k_idx - (n-1)
                i_val = i0 - k
                if 0 <= i_val < n:
                    c[k_idx] += d * (1.0 - h[i_val])
            
            # Term from h[i0] in the "v" position: -d * h[i0+k]
            # k range: max(-i0, 0)... wait
            # h[i0+k] valid when 0 <= i0+k < n, k in [-i0, n-1-i0]
            k_lo2 = max(-i0, -(n-1))
            k_hi2 = min(n-1-i0, n-1)
            for k_idx in range(k_lo2 + (n-1), k_hi2 + (n-1) + 1):
                k = k_idx - (n-1)
                j_val = i0 + k
                if 0 <= j_val < n:
                    c[k_idx] -= d * h[j_val]
            
            # For h[j0] -= delta (i.e., change = -delta):
            # Term from h[j0] in "a" position: -d * (1-h[j0-k])
            k_lo3 = max(-j0, -(n-1))
            k_hi3 = min(n-1-j0, n-1)
            for k_idx in range(k_lo3 + (n-1), k_hi3 + (n-1) + 1):
                k = k_idx - (n-1)
                i_val = j0 - k
                if 0 <= i_val < n:
                    c[k_idx] -= d * (1.0 - h[i_val])
            
            # Term from h[j0] in "v" position: +d * h[j0+k]
            k_lo4 = max(-j0, -(n-1))
            k_hi4 = min(n-1-j0, n-1)
            for k_idx in range(k_lo4 + (n-1), k_hi4 + (n-1) + 1):
                k = k_idx - (n-1)
                j_val = j0 + k
                if 0 <= j_val < n:
                    c[k_idx] += d * h[j_val]
            
            # Update h
            h[i0] += d
            h[j0] -= d
            self.max_c = c.max()
        
        def undo_move(self, i, j, delta):
            """Undo h[i] += delta, h[j] -= delta."""
            self.apply_move(i, j, -delta)
        
        def get_max(self):
            return self.max_c
    return CorrUpdater
```

```python
def level_method(h, n, deadline, max_rounds=50):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    h = project_feasible(h, n)
    
    for round_idx in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        c_max = c.max()
        # Find active lags (within 1% of max)
        threshold = c_max * 0.99
        active = np.where(c >= threshold)[0]
        
        # Objective: minimize sum of c[k] for k in active
        # c[k] = sum_i h[i+k'] * (1-h[i]) for appropriate indices
        # This is linear in h (for fixed active set) minus quadratic
        # Actually c[k] = sum_i h[i] * (1-h[i-k]) which is bilinear
        
        # Use L-BFGS to minimize sum of active c[k]
        def active_obj(x, active_set, n_):
            cx = np.correlate(x, 1 - x, mode='full')
            return cx[active_set].sum()
        
        def active_grad(x, active_set, n_):
            cx = np.correlate(x, 1 - x, mode='full')
            # Gradient of c[k] w.r.t. x
            grad = np.zeros(n_)
            for k_idx in active_set:
                k = k_idx - (n_ - 1)
                if k >= 0:
                    nv = n_ - k
                    grad[:nv] += (1.0 - x[k:])
                    grad[k:] += (-x[:nv])
                else:
                    kk = -k
                    nv = n_ - kk
                    grad[kk:] += (1.0 - x[:nv])
                    grad[:nv] += (-x[kk:])
            return grad * dx
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(active_obj, h, jac=active_grad,
                          args=(active, n), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 100, 'ftol': 1e-16})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new

            pass
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    val = m / (2.0/n) + np.log(np.sum(np.exp(sh))) / alpha
    # Wait, I need to be careful with dx

    # C5 value = c * dx
    # Let me just work with the raw values and multiply by dx at the end
    return c.max() + np.log(np.sum(np.exp(sh))) / alpha + mu * (x.sum() - n/2.0)**2
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    return m + np.log(np.sum(np.exp(sh))) / alpha + mu * (x.sum() - n / 2.0) ** 2

def smax_grad(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for lag in range(2 * n - 1):
        if w[lag] < 1e-14:
            continue
        k = lag - (n - 1)
        if k >= 0:
            nv = n - k
            grad[:nv] += w[lag] * (1.0 - x[k:])
            grad[k:] += w[lag] * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += w[lag] * (1.0 - x[:nv])
            grad[:nv] += w[lag] * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

            break

                          method='L-BFGS-B', bounds=bounds,
```
