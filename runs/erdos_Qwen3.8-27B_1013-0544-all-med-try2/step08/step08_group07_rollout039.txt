                h[i] = x_opt
    return h, corr

def greedy_binary(h_init, n, deadline, max_sweeps):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary greedy
    p1 = t0 + 0.07 * total
    hb, cb = greedy_binary(h0, n, p1, max_sweeps=300)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Coordinate descent
    p2 = t0 + 0.15 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=250, deadline=p2)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 3: L-BFGS-B with increasing alpha (main optimization)
    p3 = t0 + 0.55 * total
    alphas_l = [100, 500, 2000, 10000, 50000, 200000, 1e6, 5e6, 2e7, 1e8]
    mu_val = 1e12
    h = best_h.copy()
    for alpha in alphas_l:
        if time.time() > p3 - 15:
            break
        h = lbfgs_refine(h, n, alpha, maxiter=20000, ftol=1e-25, mu=mu_val)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: SLSQP polish (exact constraint)
    p4 = t0 + 0.75 * total
    for alpha in [1e7, 5e7, 2e8]:
        if time.time() > p4 - 10:
            break
        h = slsqp_refine(best_h, n, alpha, maxiter=8000, ftol=1e-22)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: Coord descent after SLSQP
    p5 = t0 + 0.83 * total
    hc2, _ = coord_descent(best_h, n, max_sweeps=200, deadline=p5)
    hc2 = project_feasible(hc2, n)
    cc2 = compute_c5(hc2, n)
    if cc2 < best_c:
        best_c, best_h = cc2, hc2.copy()

    # Phase 6: Binarize, greedy, then re-optimize (escape local opt)
    p6 = t0 + 0.92 * total
    hb2, cb2 = greedy_binary(best_h, n, p6 - 5, max_sweeps=200)
    if cb2 < best_c:
        best_c, best_h = cb2, hb2.copy()
    h2 = project_feasible(hb2 + 0.5 * (best_h - hb2), n)
    for alpha in [1e6, 1e7, 1e8]:
        if time.time() > p6 - 2:
            break
        h2 = lbfgs_refine(h2, n, alpha, maxiter=10000, ftol=1e-25, mu=mu_val)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 7: Perturbation restarts
    for trial in range(4):
        if time.time() > deadline - 15:
            break
        h = best_h + 0.004 * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [5e6, 5e7]:
            if time.time() > deadline - 8:
                break
            h = lbfgs_refine(h, n, alpha, maxiter=8000, ftol=1e-25, mu=mu_val)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Final SLSQP polish
    if time.time() < deadline - 5:
        h = slsqp_refine(best_h, n, 5e7, maxiter=5000, ftol=1e-22)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
