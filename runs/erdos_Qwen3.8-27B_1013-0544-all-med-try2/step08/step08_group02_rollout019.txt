import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def correlate_vec(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.where(h < 1.0 - 1e-12, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 1e-12
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] -= (abs(diff) / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return float(m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] -= wm * x[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] -= wm * x[kk:]
    return grad

def pen_obj(x, n, alpha, mu):
    return smooth_max_val(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def pnorm_obj(x, n, p, mu):
    c = np.correlate(x, 1 - x, mode='full')
    return float(np.sum(np.abs(c) ** p) ** (1.0 / p)) + mu * (x.sum() - n / 2.0) ** 2

def pnorm_grad(x, n, p, mu):
    c = np.correlate(x, 1 - x, mode='full')
    pnorm = np.sum(np.abs(c) ** p) ** (1.0 / p)
    if pnorm < 1e-15:
        g = np.zeros(n)
    else:
        wc = (np.abs(c) ** (p - 1)) / (pnorm ** (p - 1))
        g = np.zeros(n)
        for m_idx in range(2 * n - 1):
            if wc[m_idx] < 1e-14:
                continue
            k = m_idx - (n - 1)
            wm = wc[m_idx]
            if k >= 0:
                nv = n - k
                g[:nv] += wm * (1.0 - x[k:])
                g[k:] -= wm * x[:nv]
            else:
                kk = -k
                nv = n - kk
                g[kk:] += wm * (1.0 - x[:nv])
                g[:nv] -= wm * x[kk:]
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_refine(h, n, deadline, max_iters=300, alpha_start=500, alpha_end=2000000):
    """Multi-alpha L-BFGS-B refinement"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    # Ramp alpha from low to high
    n_steps = 7
    for i in range(n_steps):
        if time.time() > deadline - 3:
            break
        frac = i / max(n_steps - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        mu = 10.0
        try:
            res = minimize(pen_obj, best_h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': max_iters * 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def subgradient_step(h, n, step_size, deadline):
    """Take a step in the negative subgradient direction of the max correlation"""
    c = correlate_vec(h, n)
    c_max = c.max()
    # Active set: correlations within 1% of max
    active = c >= c_max * 0.99
    
    # Compute subgradient (average gradient over active set)
    grad = np.zeros(n)
    n_active = active.sum()
    for m_idx in range(2 * n - 1):
        if not active[m_idx]:
            continue
        k = m_idx - (n - 1)
        wm = 1.0 / n_active
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] -= wm * h[kk:]
    
    # Take step and project
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

    temp = temp0

    stuck = 0

            if stuck > 3 and temp < 0.001:

        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]

        else:

            else:

                h_bin[j] = 0

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def correlate_vec(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.where(h < 1.0 - 1e-12, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 1e-12
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] -= (abs(diff) / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return float(m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha)

            h_bin[j] = 1
            c_new = compute_c5(h_bin, n)
            h_bin[i] = 1
            h_bin[j] = 0
            if c_new < best_move_c - 1e-15:
                best_move_c = c_new
                best_i, best_j = i, j
        if best_i >= 0:
            h_bin[best_i] = 0
            h_bin[best_j] = 1
            best_c = best_move_c
            best_h = h_bin.copy()
        else:
            break
    return best_h, best_c

def binary_sa(h, n, deadline, rng, max_flips=3000000, temp0=0.012):
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]
        rem = rng.choice(idx, size=cnt - target, replace=False)
        h_bin[rem] = 0
    elif cnt < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)
        h_bin[add] = 1
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    temp = temp0
    flips = 0
    stuck = 0
    while time.time() < deadline and flips < max_flips:
        if flips % 30000 == 0 and flips > 0:
            temp *= 0.88
            if stuck > 5 and temp < 0.0005:
                break
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        h_bin[i] = 0
        h_bin[j] = 1
        c_new = compute_c5(h_bin, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_bin.copy()
            stuck = 0
        else:
            if rng.random() < np.exp(-(c_new - best_c) / max(temp, 1e-12)):
                stuck += 1
            else:
                h_bin[i] = 1
                h_bin[j] = 0
                stuck += 1
        flips += 1
    return best_h, best_c

def subgradient_descent(h, n, deadline, n_iter=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = correlate_vec(best_h, n)
        c_max = c.max()
        tolerance = max(c_max * 0.02, 0.01)
        active = c >= (c_max - tolerance)
        n_act = active.sum()
        grad = np.zeros(n)
        for m_idx in range(2 * n - 1):
            if not active[m_idx]:
                continue
            k = m_idx - (n - 1)
            wm = 1.0 / n_act
            if k >= 0:
                nv = n - k
                grad[:nv] += wm * (1.0 - best_h[k:])
                grad[k:] -= wm * best_h[:nv]
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += wm * (1.0 - best_h[:nv])
                grad[:nv] -= wm * best_h[kk:]
        step = 0.05 / (1.0 + it * 0.05)
        h_new = project_feasible(best_h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 500
        h_init = np.zeros(n_init)
        half = n_init // 2
        idx = rng.choice(n_init, size=half, replace=False)
        h_init[idx] = 1.0
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init
    # Phase 1: Multi-alpha L-BFGS-B (200s)
    t1 = min(t0 + 180, deadline - 380)
    best_h, best_c = lbfgs_refine(best_h, best_n, t1)
    # Phase 2: p-norm refinement (80s)
    if time.time() < deadline - 320:
        t2 = min(time.time() + 70, deadline - 280)
        h_p, c_p = pnorm_refine(best_h, best_n, t2)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p
    # Phase 3: Informed binary + refine (70s)
    if time.time() < deadline - 230:
        t3 = min(time.time() + 35, deadline - 200)
        h_b, c_b = informed_binary(best_h, best_n, t3, rng)
        if c_b < best_c + 0.002:
            t3b = min(time.time() + 30, deadline - 160)
            h_r, c_r = lbfgs_refine(h_b, best_n, t3b)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
    # Phase 4: Binary SA + refine (140s)
    if time.time() < deadline - 180:
        t4 = min(time.time() + 70, deadline - 150)
        h_sa, c_sa = binary_sa(best_h, best_n, t4, rng)
        if c_sa < best_c + 0.003:
            t4b = min(time.time() + 35, deadline - 110)
            h_r, c_r = lbfgs_refine(h_sa, best_n, t4b)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
    # Phase 5: Subgradient descent (50s)
    if time.time() < deadline - 100:
        t5 = min(time.time() + 40, deadline - 80)
        h_sg, c_sg = subgradient_descent(best_h, best_n, t5)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
    # Phase 6: Final p-norm polish (40s)
    if time.time() < deadline - 30:
        t6 = min(time.time() + 25, deadline - 20)
        h_p, c_p = pnorm_refine(best_h, best_n, t6)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p
    # Phase 7: Final L-BFGS-B high-alpha (remaining)
    if time.time() < deadline - 5:
        t7 = deadline - 3
        h_f, c_f = lbfgs_refine(best_h, best_n, t7, max_iters=200)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
