import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_max_weights(c, alpha, top_k):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    if top_k < len(w):
        idx = np.argsort(w)[-top_k:]
        wt = np.zeros_like(w)
        wt[idx] = w[idx]
        wt /= wt.sum()
        return wt
    return w

def obj_lse(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def grad_lse(x, n, alpha, top_k=60):
    c = compute_c5_vec(x, n)
    w = smooth_max_weights(c, alpha, top_k)
    dx = 2.0 / n
    grad = np.zeros(n)
    top_idx = np.argsort(w)[-top_k:]
    for i in top_idx:
        if w[i] < 1e-12:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    return grad

def nesterov_opt(h0, n, deadline, alpha_init=80.0, alpha_rate=1.15,
                 alpha_max=5000.0, top_k=60, eta_init=0.03, momentum=0.85):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    alpha = alpha_init
    eta = eta_init
    stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        y = np.clip(h + momentum * v, 0.0, 1.0)
        g = grad_lse(y, n, alpha, top_k)
        v = momentum * v - eta * g
        h_new = project_feasible(h + v, n)
        if it % 40 == 0 and alpha < alpha_max:
            alpha *= alpha_rate
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        if stall > 50:
            eta *= 0.5
            if eta < 0.001:
                rng = np.random.RandomState(it * 13 + 7)
                h_new = project_feasible(h_new + 0.02 * (2.0 * rng.random(n) - 1.0), n)
                v = np.zeros(n)
                eta = eta_init
                stall = 0
        h = h_new
    return best_h, best_c

def lbfgs_wrap(x, n, alpha, top_k=80):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 30.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def lbfgs_jac(x, n, alpha, top_k=80):
    c = compute_c5_vec(x, n)
    w = smooth_max_weights(c, alpha, top_k)
    dx = 2.0 / n
    grad = np.zeros(len(x))
    top_idx = np.argsort(w)[-top_k:]
    for i in top_idx:
        if w[i] < 1e-12:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    s = x.sum() - n / 2.0
    grad += 60.0 * s
    return grad

def lbfgs_run(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_wrap, h, jac=lbfgs_jac, args=(n, alpha),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def golden_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(half) * phi).astype(int) % n
    h[positions] = 1.0
    return project_feasible(h, n)

def thue_morse_start(n):
