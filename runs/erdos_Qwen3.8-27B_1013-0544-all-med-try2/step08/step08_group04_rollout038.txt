                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def projected_gradient(h, n, deadline, topk=5, alpha=100000, lr=0.005):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for _ in range(20000):
        if time.time() > deadline:
            break
        _, g = topk_obj_grad(h, n, topk, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        step = lr
        improved = False
        for _ in range(40):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.6
            if lr < 1e-14:
                break
    return best_h, best_c

def directed_pairs(h, n, deadline, topk=5, alpha=50000):
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for _ in range(10000):
        if time.time() > deadline:
            break
        _, g = topk_obj_grad(h, n, topk, alpha)
        order = np.argsort(g)
        improved = False
        for a_idx in range(min(8, n)):
            i = order[n - 1 - a_idx]
            if not improved:
                for b_idx in range(min(8, n)):
                    j = order[b_idx]
                    if i == j:
                        continue
                    for d in deltas:
                        if h[i] - d >= -1e-12 and h[j] + d <= 1 + 1e-12:
                            h[i] -= d; h[j] += d
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new; improved = True
                                break
                            else:
                                h[i] += d; h[j] -= d
                        if improved:
                            break
                    if improved:
                        break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def local_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - 40)
                hi = min(n, i + 41)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] - d >= -1e-12 and h[j] + d <= 1 + 1e-12:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
                    if h[i] + d <= 1 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS top-k with increasing alpha
    t1 = min(t0 + 220, deadline - 500)
    best_h, best_c = lbfgs_topk(best_h, best_n, t1, topk=5,
        alpha_list=[50, 200, 1000, 5000, 50000, 200000, 1000000, 5000000], iters_per=80)

    # Phase 2: Projected gradient at high alpha
    if time.time() < deadline - 350:
        t2 = min(time.time() + 70, deadline - 350)
        h2, c2 = projected_gradient(best_h, best_n, t2, topk=4, alpha=500000, lr=0.003)
        if c2 < best_c:
            best_c = c2; best_h = h2

    # Phase 3: Directed pair search
    if time.time() < deadline - 260:
        t3 = min(time.time() + 90, deadline - 260)
        best_h, best_c = directed_pairs(best_h, best_n, t3, topk=4, alpha=200000)

    # Phase 4: Second L-BFGS with smaller topk
    if time.time() < deadline - 180:
        t4 = min(time.time() + 90, deadline - 180)
        h4, c4 = lbfgs_topk(best_h, best_n, t4, topk=3,
            alpha_list=[500000, 2000000, 10000000], iters_per=200)
        if c4 < best_c:
            best_c = c4; best_h = h4

    # Phase 5: More directed pairs + local refine
    if time.time() < deadline - 120:
        t5 = min(time.time() + 50, deadline - 120)
        best_h, best_c = directed_pairs(best_h, best_n, t5, topk=3, alpha=500000)
    if time.time() < deadline - 60:
        t6 = min(time.time() + 50, deadline - 60)
        best_h, best_c = local_refine(best_h, best_n, t6)

    # Phase 6: Try other n
    if time.time() < deadline - 140:
        for n_try in [200, 400, 500]:
            if time.time() > deadline - 70:
                break
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h, left=best_h[0], right=best_h[-1])
            h_interp = project_feasible(h_interp, n_try)
            alloc = max(35, min(70, deadline - time.time() - 35))
            if alloc < 25:
                break
            h_t, c_t = lbfgs_topk(h_interp, n_try, time.time() + alloc*0.6,
                topk=5, alpha_list=[200, 2000, 20000, 200000, 2000000], iters_per=60)
            if time.time() < deadline - 25:
                h_t, c_t = directed_pairs(h_t, n_try,
                    min(time.time()+25, deadline-15), topk=4, alpha=200000)
            if c_t < best_c:
                best_c = c_t; best_h = h_t; best_n = n_try

    # Phase 7: Final polish
    if time.time() < deadline - 4:
        best_h, best_c = local_refine(best_h, best_n, deadline - 2)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
