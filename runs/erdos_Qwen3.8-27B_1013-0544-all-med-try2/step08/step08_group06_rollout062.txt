```python
    return m + logsum / alpha

def smooth_max_grad(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    g = corr * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    w = exp_s / exp_s.sum()  # softmax weights
    # grad[i] = dx * sum_k w[k] * d_corr[k]/dx[i]
    # d_corr[k]/dx[i] = (1-x[i-o_k])*[0<=i-o_k<n] - x[i+o_k]*[0<=i+o_k<n]
    # where o_k = k - (n-1)
    grad = np.zeros(n)
    offsets = np.arange(2*n-1) - (n-1)
    for idx in range(2*n-1):
        o = offsets[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        # (1-x[i-o]) for i in [max(0,o), min(n, o+n))
        i0 = max(0, o)
        i1 = min(n, o + n)
        if i1 > i0:
            j0 = i0 - o
            j1 = i1 - o
            grad[i0:i1] += wi * (1.0 - x[j0:j1])
        # -x[i+o] for i in [max(0,-o), min(n, n-o))
        i0 = max(0, -o)
        i1 = min(n, n - o)
        if i1 > i0:
            j0 = i0 + o
            j1 = i1 + o
            grad[i0:i1] -= wi * x[j0:j1]
    grad *= dx
    return grad

def projected_gradient(h, n, alpha, n_iters, step_size, momentum, deadline):
    """Project gradient descent with momentum on the smooth max."""
    h = h.copy()
    m = 0.0  # momentum
    vel = np.zeros(n)
    best_h = h.copy()
    best_obj = smooth_max_obj(h, alpha, n)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        obj = smooth_max_obj(h, alpha, n)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
        grad = smooth_max_grad(h, alpha, n)
        vel = momentum * vel - step_size * grad
        h_new = h + vel
        h = project_feasible(h_new, n)
    
    return h, best_h

def greed_binary(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    m_range = np.arange(-(n-1), n)
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-12:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
```

```python
def smooth_max_grad_fast(x, alpha, n, threshold=1e-4):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    g = corr * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    w = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    offsets = np.arange(2*n-1) - (n-1)
    sig = w > threshold
    for idx in np.where(sig)[0]:
        o = offsets[idx]
        wi = w[idx]
        i0 = max(0, o)
        i1 = min(n, o + n)
        if i1 > i0:
            j0 = i0 - o
            j1 = i1 - o
            grad[i0:i1] += wi * (1.0 - x[j0:j1])
        i0 = max(0, -o)
        i1 = min(n, n - o)
        if i1 > i0:
            j0 = i0 + o
            j1 = i1 + o
            grad[i0:i1] -= wi * x[j0:j1]
    grad *= dx
    return grad
```

With threshold=1e-4 and large alpha, typically only 5-20 weights are significant. So the loop becomes O(20*400) = 8000 operations, which is ~10μs. Much faster!

```python
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        T *= cool
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Projected gradient (fast, many iterations)
    p1 = t0 + 0.25 * budget_s
    alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    h, c = proj_grad(best_h, n, alphas, p1)
    if c < best_c:
        best_c, best_h = c, h

    # Phase 2: SLSQP with high alpha
    p2 = t0 + 0.50 * budget_s
    for alpha in [500000, 5000000, 50000000, 500000000]:
        if time.time() > p2 - 15:
            break
        h = slsqp_refine(best_h, n, alpha, maxiter=800, ftol=1e-17)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 3: Binary 2-opt refinement
    p3 = t0 + 0.65 * budget_s
    hb, cb = binary_2opt(best_h, n, p3, max_sweeps=300)
    if cb < best_c:
        best_c, best_h = cb, hb

    # Phase 4: Annealing for better binary
    p4 = t0 + 0.78 * budget_s
    ha, ca = anneal_binary(best_h, n, p4, seed=seed, n_iters=2000000)
    if ca < best_c:
        best_c, best_h = ca, ha
    # Re-smooth from annealed
    if time.time() < p4 + 0.05 * budget_s:
        for alpha in [5000000, 50000000, 500000000]:
            if time.time() > t0 + 0.83 * budget_s:
                break
            h = slsqp_refine(best_h, n, alpha, maxiter=500, ftol=1e-17)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h

    # Phase 5: Multi-resolution
    p5 = t0 + 0.92 * budget_s
    for nh in [200, 300, 500, 600]:
        if time.time() > p5 - 20:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, ch = proj_grad(hh, nh, [10000, 100000, 500000, 5000000], p5 - 15)
        for alpha in [5000000, 50000000]:
            if time.time() > p5 - 10:
                break
            hh = slsqp_refine(hh, nh, alpha, maxiter=500, ftol=1e-17)
            ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh

    # Phase 6: Final high-alpha SLSQP polish
    if time.time() < dl - 3:
        for alpha in [500000000, 5000000000]:
            if time.time() > dl - 3:
                break
            h = slsqp_refine(best_h, n, alpha, maxiter=2000, ftol=1e-18)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h

    # Phase 7: Noise restart + polish
    if time.time() < dl - 2:
        for trial in range(4):
            if time.time() > dl - 2:
                break
            h = best_h + 0.003 * rng.randn(n)
            h = project_feasible(h, n)
            h, c = proj_grad(h, n, [1000000, 10000000], dl - 2, momentum=0.9)
            if c < best_c:
                best_c, best_h = c, h

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
