            h_t = project_feasible(res.x, n_try)
            c_t = compute_c5(h_t, n_try)
        except Exception:
            h_t, c_t = h_interp, compute_c5(h_interp, n_try)

        if time.time() < deadline - 50:
            t_q = min(time.time() + 30, deadline - 30)
            h_t, c_t = fast_local_search(h_t, n_try, t_q, alpha=1000, top_k=40, rng=rng)

        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Phase 4: Final intensive local search
    if time.time() < deadline - 5:
        t4 = deadline - 2
        best_h, best_c = fast_local_search(best_h, best_n, t4, alpha=2000, top_k=60, rng=rng)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
