```python
def compute_A(h, n):
    # Create aToeplitz-like matrix
    # A[p, k] = h[p + (k-(n-1))] + h[p - (k-(n-1))]
    
    # First term: h[p + u] where u = k-(n-1)
    # This is like a Toeplitz matrix where row p, col k gives h[p+k-(n-1)]
    
    # Use scipy or manual
    n_k = 2*n - 1
    A = np.zeros((n, n_k))
    
    for k in range(n_k):
        u = k - (n-1)
        # h[p+u] for p = 0..n-1
        # where 0 <= p+u < n, i.e., -u <= p < n-u
        lo = max(0, -u)
        hi = min(n, n - u)
        if lo < hi:
            A[lo:hi, k] = h[lo+u : hi+u]
        # h[p-u] for p = 0..n-1
        # where 0 <= p-u < n, i.e., u <= p < n+u
        lo2 = max(0, u)
        hi2 = min(n, n + u)
        if lo2 < hi2:
            A[lo2:hi2, k] += h[lo2-u : hi2-u]
    
    return A
```

```python
        # = max_k (W[j_idx] - D[i_idx])

        # D[:, None, :] has shape (n_zeros, 1, 2n-1)
        # Diff = W[None,:,:] - D[:,None,:] has shape (n_zeros, n_ones, 2n-1)
        # Value = Diff.max(axis=2) has shape (n_zeros, n_ones)
        
        # For n=250: n_zeros=125, n_ones=125, 2n-1=499
        # Diff shape: (125, 125, 499) = 7.8M floats = 62MB. Might be tight but OK.
        
        Diff = W[None, :, :] - D[:, None, :]
        Value = Diff.max(axis=2)
        
        # Find the (i, j) that minimizes Value
        min_val = Value.min()
        min_pos = np.unravel_index(Value.argmin(), Value.shape)
        
        i_idx, j_idx = min_pos
        i = zero_indices[i_idx]
        j = one_indices[j_idx]
        
        if min_val * dx < cur_c - 1e-15:
            # Do the swap
            h[i] = 1.0
            h[j] = 0.0
            corr += A[j, :] - A[i, :]
            cur_c = corr.max() * dx
            
            # Update A: rows i and j change
            # A[i, :] was the old row (h[i] was 0, now 1)
            # The change in A[p, k] when h[i] changes from 0 to 1:
            # A[p, k] = h[p+u] + h[p-u]
            # If p+u == i: A[p,k] increases by 1
            # If p-u == i: A[p,k] increases by 1
            # p+u = i => p = i-u = i-k+(n-1)
            # p-u = i => p = i+u = i+k-(n-1)
            
            # Similarly for h[j] changing from 1 to 0: A decreases by 1 at those positions.
            # This is O(n) per row update.
            
            for k in range(2*n-1):
                u = k - (n-1)
                # h[i] went 0->1: A[p=IU, k] += 1 where i+u = p, i.e., p = i+u
                p1 = i + u
                if 0 <= p1 < n:
                    A[p1, k] += 1
                p2 = i - u
                if 0 <= p2 < n:
                    A[p2, k] += 1
                # h[j] went 1->0: A[p, k] -= 1
                p3 = j + u
                if 0 <= p3 < n:
                    A[p3, k] -= 1
                p4 = j - u
                if 0 <= p4 < n:
                    A[p4, k] -= 1
            
            # Update sets
            zero_indices = np.setdiff1d(zero_indices, [i])
            one_indices = np.setdiff1d(one_indices, [j])
            one_indices = np.append(one_indices, i)
            zero_indices = np.append(zero_indices, j)
            one_indices = np.sort(one_indices)
            zero_indices = np.sort(zero_indices)
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            break  # No improvement possible
        
        sweeps += 1
    
    return best_h, best_c
```

```python
    best_h = h.copy()

        c_max = corr.max()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_A(h, n):
    """A[p, k] = h[p+u] + h[p-u] where u = k-(n-1)"""
    A = np.zeros((n, 2*n-1))
    for k in range(2*n-1):
        u = k - (n-1)
        # h[p+u] for p=0..n-1
        lo1 = max(0, -u)
        hi1 = min(n, n - u)
        if lo1 < hi1:
            A[lo1:hi1, k] = h[lo1+u:hi1+u]
        # h[p-u] for p=0..n-1
        lo2 = max(0, u)
        hi2 = min(n, n + u)
        if lo2 < hi2:
            A[lo2:hi2, k] += h[lo2-u:hi2-u]
    return A

    target = n // 2
    
    h = np.zeros(n)

    h[idx[:target]] = 1.0

    cur_c = corr.max() * dx
    
    best_h = h.copy()
    best_c = cur_c
    A = compute_A(h, n)
    
    ones = np.sort(np.where(h == 1)[0])
    zeros = np.sort(np.where(h == 0)[0])
    
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        # W shape: (n_ones, 2n-1), D shape: (n_zeros, 2n-1)
        W = corr[None, :] + A[ones, :]
        D = A[zeros, :]
        
        # Value[i_idx, j_idx] = max_k (W[j_idx, k] - D[i_idx, k])
        # = max_k (corr[k] + A[ones[j_idx], k] - A[zeros[i_idx], k])
        Diff = W[None, :, :] - D[:, None, :]
        Value = Diff.max(axis=2)
        
        min_val_idx = Value.argmin()
        i_idx = min_val_idx // n_ones
        j_idx = min_val_idx % n_ones
        min_val = Value[i_idx, j_idx]
        
        if min_val * dx < cur_c - 1e-14:
            i = zeros[i_idx]
            j = ones[j_idx]
            
            # Do the swap
            h[j] = 0.0
            h[i] = 1.0
            corr += A[j, :] - A[i, :]

                p3 = j + u

                p4 = j - u

                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3:
                break
    
    return best_h, best_c
```

```python
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.01
    
    for it in range(n_iter):
        if time.time() > deadline:
            break

        c_max = corr.max()
        tol = max(0.1, 0.02 * c_max)

            active = np.array([np.argmax(corr)])

        us = active - (n-1)  # displacements for active lags
        grad = np.zeros(n)
        for i in range(n):
            # h[i + u] for u in us, with boundary = 0
            idx1 = i + us
            valid1 = (idx1 >= 0) & (idx1 < n)
            term1 = h[idx1[valid1]].mean() if valid1.any() else 0.0
            
            idx2 = i - us
            valid2 = (idx2 >= 0) & (idx2 < n)

            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.1, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
```

This is a "bottleneck" problem. One approach: sort the values of f_j[k] - A[i,k] for all k, and the max is the largest. I want to minimize this largest value.

But here's a trick: I can use a "blocking" approach. If I know that the current max is at position k*, then for the swap to be improving, I need (M[k*] + A[j,k*] - A[i,k*]) < C. This gives a necessary condition: A[j,k*] - A[i,k*] < 0, i.e., A[j,k*] < A[i,k*].

This filters out many candidates quickly! I can first find candidates that satisfy this necessary condition, then check the full max.

Let me implement this optimization:

```python
one_aks = A[ones, k_star]
```

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_A(h, n):
    A = np.zeros((n, 2*n-1))
    for k in range(2*n-1):
        u = k - (n-1)
        lo1 = max(0, -u)
        hi1 = min(n, n - u)
        if lo1 < hi1:
            A[lo1:hi1, k] = h[lo1+u:hi1+u]
        lo2 = max(0, u)
        hi2 = min(n, n + u)
        if lo2 < hi2:
            A[lo2:hi2, k] += h[lo2-u:hi2-u]
    return A

def binary_swap(h_init, n, deadline, seed=42):
    dx = 2.0 / n
    target = n // 2
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1.0-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    A = compute_A(h, n)
    ones = np.sort(np.where(h == 1)[0])
    zeros = np.sort(np.where(h == 0)[0])
    
    while time.time() < deadline:
        if len(ones) < 1 or len(zeros) < 1:
            break
        no_ = len(ones)
        nz = len(zeros)
        W = corr[None,:] + A[ones, :]
        D = A[zeros, :]
        Diff = W[None,:,:] - D[:,None,:]
        Value = Diff.max(axis=2)
        flat_idx = Value.argmin()
        ii = flat_idx // no_
        jj = flat_idx % no_
        i = zeros[ii]
        j = ones[jj]
        if Value[ii,jj] * dx < cur_c - 1e-14:
            h[j] = 0.0; h[i] = 1.0
            corr += A[j,:] - A[i,:]
            cur_c = corr.max() * dx
            # Update A
            for k in range(2*n-1):
                u = k - (n-1)
                p = i + u
                if 0 <= p < n: A[p,k] += 1
                p = i - u
                if 0 <= p < n: A[p,k] += 1
                p = j + u
                if 0 <= p < n: A[p,k] -= 1
                p = j - u
                if 0 <= p < n: A[p,k] -= 1
            ones = np.sort(np.delete(np.delete(ones, jj), 0))
            zeros = np.sort(np.delete(np.delete(zeros, ii), 0))
            ones = np.sort(np.append(ones, i))
            zeros = np.sort(np.append(zeros, j))
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            break
    return best_h, best_c
```

```python
def grad_refine(h_init, n, deadline, n_iter=3000):
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.01
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1-h, mode='full')
        c_max = corr.max()
        tol = max(0.5, 0.03 * c_max)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        us = active.astype(float) - (n-1)
        grad = np.zeros(n)
        for i in range(n):
            idx1 = i + us
            v1 = (idx1 >= 0) & (idx1 < n)
            t1 = h[idx1.astype(int)[v1]].mean() if v1.any() else 0.0
            idx2 = i - us
            v2 = (idx2 >= 0) & (idx2 < n)
            t2 = h[idx2.astype(int)[v2]].mean() if v2.any() else 0.0
            grad[i] = 1.0 - t1 - t2
        
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.8
            if lr < 1e-7:
                break
    return best_h, best_c
```
