import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary_sorted(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def compute_F(h, n):
    """F[i, k_idx] = h[i+k] + h[i-k] where k = k_idx - (n-1), out of bounds = 0"""
    m = 2 * n - 1
    mid = n - 1
    F = np.zeros((n, m))
    for k_idx in range(m):
        k = k_idx - mid
        if k >= 0:
            F[:n-k, k_idx] += h[k:]       # h[i+k] for i = 0..n-1-k
            F[k:, k_idx] += h[:n-k]       # h[i-k] for i = k..n-1
        else:
            F[-k:, k_idx] += h[:n+k]      # h[i+k] for i = -k..n-1
            F[:n+k, k_idx] += h[-k:]      # h[i-k] for i = 0..n-1+k
    return F

def binary_local_search(h, n, deadline, max_sweeps=10000, seed=42, anneal_T0=0.0, anneal_decay=0.95):
    """Fast binary local search with optional simulated annealing."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    m = 2 * n - 1
    
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = c.max()
    best_max = cur_max * dx
    best_h = h.copy()
    
    T = anneal_T0
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        F = compute_F(h, n)
        A = np.where(h == 1)[0]
        B = np.where(h == 0)[0]
        
        if len(A) == 0 or len(B) == 0:
            break
        
        best_score = float('inf')
        best_a = -1
        best_b = -1
        
        for a in A:
            g = c + F[a]
            scores = (g[None, :] - F[B, :]).max(axis=1)
            idx = np.argmin(scores)
            if scores[idx] < best_score:
                best_score = scores[idx]
                best_a = a
                best_b = B[idx]
        
        if best_a < 0:
            break
        
        delta = best_score - cur_max
        if delta < -1e-15:
            accept = True
        elif T > 0:
            accept = rng.rand() < np.exp(-delta / T) if delta > 0 else True
        else:
            accept = False
            if abs(delta) < 1e-12:
                # Local optimum
                no_improve += 1
                if no_improve >= 3:
                    break
            continue
        
        if accept:
            h[best_a] = 0.0
            h[best_b] = 1.0
            c = c + F[best_a] - F[best_b]
            cur_max = c.max()
            if cur_max * dx < best_max:
                best_max = cur_max * dx
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            T = anneal_decay * T
            if no_improve > 50:
