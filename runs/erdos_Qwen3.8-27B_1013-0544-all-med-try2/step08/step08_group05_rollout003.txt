def compute_c(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return c, float(np.max(c)) * 2.0 / n

def project_feasible(h, target):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(3):
        s = float(h.sum())
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            room = 1.0 - h
            R = float(room.sum())
            if R > 1e-15:
                h = h + (d / R) * room
        else:
            if s > 1e-15:
                h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    d = target - float(h.sum())
    if abs(d) > 1e-13:
        if d > 0.0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + d)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + d)

def default_initial(n, seed):

def build_coef(h):

    L = 2 * n - 1

        p2 = ar(cen - x, cen - x + n)
        coef[x, p2] -= h[p2 + x - cen]
        coef[x, cen] = 0.0
    return coef

def eval_pair(c, tmp, buf, ci, cj, cen, p1, p2, a, b, d):

    bj = b - d
    tmp[cen] = c[cen] + t * (1.0 - t) + bj * (1.0 - bj) - a * (1.0 - a) - b * (1.0 - b)
    dd = d * d

        tmp[p1] += dd
    if p2 != cen and p2 != p1:

    return float(tmp.max())
