```python
def project_to_polytope(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)
```

```python
    """Compute all overlap values ov_k for k in [-(n-1), n-1]"""

def compute_grad(h, n, alpha):
    """Compute gradient of soft max of overlaps"""
    c = np.correlate(h, 1-h, mode='full')
    # Soft max weights
    c_shifted = alpha * (c - c.max())
    w = np.exp(c_shifted)
    w /= w.sum()
    
    # Now compute gradient
    # grad[j] = sum_k w_k * d(ov_k)/dh[j]
    # Using the formula derived above
    grad = np.zeros(n)
    
    # For efficiency, let's compute directly
    # For shift k (corresponding to index m = k + n - 1 in c):
    # d(ov_k)/dh[j] = (1-h[j+k])*[j < n-k] - [j >= k]  for k >= 0
    # d(ov_k)/dh[j] = (1-h[j-kk])*[j >= kk] - [j < n-kk]  for k = -kk < 0
    
    # Let's compute using convolution-like operations
    # For k >= 0:
    #   First part: sum_{k=0}^{n-1} w[k+n-1] * (1-h[j+k]) * [j < n-k]
    #   = sum_{k=0}^{n-1-j} w[k+n-1] * (1-h[j+k])
    #   Second part: -sum_{k=0}^{n-1} w[k+n-1] * [k <= j]
    #   = -sum_{k=0}^{j} w[k+n-1]
    
    # For k < 0 (k = -kk, kk = 1..n-1):
    #   First part: sum_{kk=1}^{n-1} w[n-1-kk] * (1-h[j-kk]) * [j >= kk]
    #   = sum_{kk=1}^{j} w[n-1-kk] * (1-h[j-kk])
    #   Second part: -sum_{kk=1}^{n-1} w[n-1-kk] * [j < n-kk]
    #   = -sum_{kk=1}^{n-1-j} w[n-1-kk]
    
    # This is still O(n^2) but let's just do it directly for n=500
    
    for j in range(n):
        # Positive shifts k=0 to n-1-j
        s1 = 0.0
        for k in range(0, n - j):
            s1 += w[k + n - 1] * (1 - h[j + k])
        # Negative shifts kk=1 to j
        s2 = 0.0
        for kk in range(1, j + 1):
            s2 += w[n - 1 - kk] * (1 - h[j - kk])
        # Penalty terms
        s3 = 0.0
        for k in range(0, j + 1):
            s3 += w[k + n - 1]
        s4 = 0.0
        for kk in range(1, n - j):
            s4 += w[n - 1 - kk]
        
        grad[j] = s1 + s2 - s3 - s4
    
    return grad
```

```python
def pgd_optimize(h, n, alpha, max_iters, deadline):
    h = project_to_polytope(h, n).copy()
    val, _ = soft_max_value_grad(h, n, alpha)
    best_val = val
    best_h = h.copy()
    lr = 0.5
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        _, grad = soft_max_value_grad(h, n, alpha)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break

        for _ in range(20):

        val = val_try
        lr = min(lr * 1.05, 5.0)  # slight increase if we made progress

            best_val = val
            best_h = h.copy()
```
