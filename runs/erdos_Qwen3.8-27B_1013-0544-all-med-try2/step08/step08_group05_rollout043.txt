def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h

    idx = np.argsort(h)[::-1]

    """Compute S[p, l] = b[p+l] + b[p-l] for valid indices, using lag index (0 to 2n-2)"""
    # lag = l - (n-1), ranging from -(n-1) to n-1

            if 0 <= p + lag < n:
                val += b[p + lag]
            if 0 <= p - lag < n:
                val += b[p - lag]
            S[p, l] = val

def compute_S(b, n):
    S = np.zeros((n, 2*n-1))
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    # For each p, S[p, l] = b[p+lag] + b[p-lag]
    # p + lag: for p=0,...,n-1 and lag=-(n-1),...,n-1
    # We can use advanced indexing
    p_idx = np.arange(n)[:, np.newaxis]  # (n, 1)
    lag_idx = lags[np.newaxis, :]  # (1, 2n-1)
    
    # b[p + lag]: valid when 0 <= p+lag < n
    pl = (p_idx + lag_idx).astype(int)
    mask1 = (pl >= 0) & (pl < n)
    S += np.where(mask1, b[pl], 0.0)
    
    # b[p - lag]: valid when 0 <= p-lag < n
    pm = (p_idx - lag_idx).astype(int)
    mask2 = (pm >= 0) & (pm < n)
    S += np.where(mask2, b[pm], 0.0)
    
    return S

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)

def compute_S(b, n):
    S = np.zeros((n, 2 * n - 1))
    p = np.arange(n)
    for lag in range(-(n - 1), n):
        l = lag + (n - 1)
        v1 = (p + lag >= 0) & (p + lag < n)
        if v1.any():
            S[v1, l] += b[p[v1] + lag]
        v2 = (p - lag >= 0) & (p - lag < n)
        if v2.any():
            S[v2, l] += b[p[v2] - lag]
    return S
