    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlation(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_shift(h, n, k):
    """Gradient of I_k(h) = sum h[i]*(1-h[i+k]) with respect to h"""
    dx = 2.0 / n
    grad = np.zeros(n)
    m = k + (n - 1)  # index in full correlation array
    if k >= 0:
        nv = n - k
        grad[:nv] += (1.0 - h[k:])
        grad[k:] += (-h[:nv])
    else:
        kk = -k
        nv = n - kk
        grad[kk:] += (1.0 - h[:nv])
        grad[:nv] += (-h[kk:])
    return grad * dx

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    s = h.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

        return h_new, compute_c5(h_new, n)

                for i, j in pairs:
                    if time.time() > deadline:

                    if h[i] > d and h[j] < 1.0:

                        h[j] += d

                        if c_new < best_c:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

                grad[:nv] += (1.0 - h[k:])

                nv = n - kk
                grad[kk:] += (1.0 - h[:nv])
                grad[:nv] += (-h[kk:])

        if c_new < best_c:

def project(h, n):
    target = n / 2.0

    for _ in range(5000):
        d = target - h.sum()
        if abs(d) < 1e-14:

        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14: break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-14: break
            h += (d / t) * h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14:
                break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def smax_f(h, n, a, mu):
    c = get_corr(h, n)
    m = c.max()
    s = a * (c - m)
    mx = s.max()
    v = m + (mx + np.log(np.sum(np.exp(s - mx)))) / a
    v += mu * (h.sum() - n / 2.0) ** 2
    return float(v)

def smax_g(h, n, a, mu):
    c = get_corr(h, n)
    m = c.max()
    s = a * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(2 * n - 1):
        if w[i] < 1e-10:
            continue
        k = i - (n - 1)
        wm = w[i] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])
    g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def lbfgs_pass(h, n, a, mu, deadline, nit=400):
    if time.time() > deadline - 3:
        return h.copy(), compute_c5(h, n)
    bnd = [(0.0, 1.0)] * n
    try:
        r = minimize(smax_f, h, jac=smax_g, args=(n, a, mu),
                     method='L-BFGS-B', bounds=bnd,
                     options={'maxiter': nit, 'ftol': 1e-18,
                              'maxfun': nit * 50, 'maxls': 100})
        hn = project(r.x, n)
        return hn, compute_c5(hn, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def exact_subgrad(h, n, deadline, iters=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.03
    stuck = 0
    for it in range(iters):
        if time.time() > deadline:
            break
        c = get_corr(h, n)
        cmax = c.max()
        act = np.where(c >= cmax - 1e-9)[0]
        g = np.zeros(n)
        for i in act:
            k = i - (n - 1)
            if k >= 0:
                nv = n - k
                g[:nv] += (1.0 - h[k:])
                g[k:] += -h[:nv]
            else:
                kk = -k
                nv = n - kk
                g[kk:] += (1.0 - h[:nv])
                g[:nv] += -h[kk:]
        g *= dx / max(len(act), 1)
        nrm = np.linalg.norm(g)
        if nrm < 1e-14:
            break
        h_new = project(h - lr * g / nrm, n)
        cn = compute_c5(h_new, n)
        if cn < best_c - 1e-15:
            best_c = cn
            best_h = h_new.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 5000:
                lr *= 0.9
                stuck = 0
        h = h_new
        lr *= 0.99998
    return best_h, best_c

    best_c = compute_c5(h, n)

    cmax = c.max()
    worst = np.argsort(c)[-10:]

            grad_k = np.zeros(n)

            # We want to decrease I_k: move mass from high grad to low grad
