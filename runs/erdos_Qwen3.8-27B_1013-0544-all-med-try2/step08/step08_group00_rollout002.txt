from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_alpha(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w = w / w.sum()
    return m + np.log(w.sum()) / alpha, w

def obj_alpha(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def grad_obj(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def eq_fun(x, n):
    return float(x.sum() - n / 2.0)

def eq_jac(x, n):
    return np.ones(n)

def obj_slsqp(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def lbfgs_opt(h, n, deadline, max_iters=500):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas_mus = [(200, 1.0), (1000, 5.0), (5000, 20.0),
                  (20000, 100.0), (100000, 500.0),
                  (500000, 2000.0), (2000000, 10000.0)]
    for alpha, mu in alphas_mus:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(obj_alpha, h, jac=grad_obj,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-20,
                                  'gtol': 1e-18, 'maxfun': max_iters * 50,
                                  'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_opt(h, n, deadline, alpha=2000000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    try:
        res = minimize(obj_slsqp, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new

from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj_alpha(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad_smooth(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _grad_obj(x, n, alpha, mu):
    g = _grad_smooth(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def _eq_fun(x, n):
    return float(x.sum() - n / 2.0)

def _eq_jac(x, n):
    return np.ones(n)

def _obj_slsqp(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def lbfgs_pass(h, n, deadline, max_iters=400):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(200, 1.0), (1000, 5.0), (5000, 25.0),
               (25000, 100.0), (100000, 500.0),
               (500000, 2000.0), (2000000, 8000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_obj_alpha, h, jac=_grad_obj,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-20,
                                  'gtol': 1e-18, 'maxfun': max_iters * 40,
                                  'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_pass(h, n, deadline, alpha=2000000, maxiter=4000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_slsqp, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_guided_search(h, n, deadline, k_top=30):
    best_c = compute_c5(h, n)
    alpha = 500000
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    for d in steps:
        while time.time() < deadline:
            g = _grad_smooth(h, n, alpha)
            idx = np.argsort(np.abs(g))[::-1]
            candidates = idx[:k_top]
            improved = False
            for ii in range(len(candidates)):
                for jj in range(ii + 1, len(candidates)):
                    if time.time() > deadline:
                        return h, best_c
                    i, j = candidates[ii], candidates[jj]
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[j] + d <= 1.0 and h[i] - d >= 0.0:
                        h[j] += d; h[i] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[j] -= d; h[i] += d
            if not improved:
                break
    return h, best_c

def coord_descent(h, n, deadline):
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
    for d in steps:
        while time.time() < deadline:
            c = compute_c5_vec(h, n)
            active = np.where(c > best_c - 1e-6)[0]
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if h[i] + d > 1.0 or h[i] - d < 0.0:
                    continue
                for j in active:
                    k = j - (n - 1)
                    if k >= 0 and i < n - k:
                        partner = i + k
                    elif k < 0 and i >= -k:
                        partner = i + k
                    else:
                        continue
                    if partner == i:
                        continue
                    if h[i] + d <= 1.0 and h[partner] - d >= 0.0:
                        h[i] += d; h[partner] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-16:
                            best_c = c_new; improved = True
                            break
                        else:
                            h[i] -= d; h[partner] += d
                    if h[partner] + d <= 1.0 and h[i] - d >= 0.0:
                        h[partner] += d; h[i] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-16:
                            best_c = c_new; improved = True
                            break
                        else:
                            h[partner] -= d; h[i] += d
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break
    return h, best_c

def multi_start_search(h, n, deadline, rng):
    best_c = compute_c5(h, n)
    starts = []
    for scale in [0.05, 0.1, 0.15]:
        perturbed = h + rng.randn(n) * scale
        starts.append(project_feasible(perturbed, n))
    for s in starts:
        if time.time() > deadline - 10:
            break
        alloc = min(25, (deadline - time.time() - 5) * 0.4)
        if alloc < 10:
            break
        h_s, c_s = lbfgs_pass(s, n, time.time() + alloc, max_iters=200)
        if c_s < best_c:
            best_c = c_s
            h = h_s
    return h, best_c
