    return h

def set_state(n, alpha, pen):
    global G_N, G_DX, G_ALPHA, G_TARGET, G_PEN
    G_N = int(n)
    G_DX = 2.0 / G_N
    G_ALPHA = float(alpha)
    G_TARGET = 0.5 * G_N
    G_PEN = float(pen)

def smooth_obj(z):
    g = np.correlate(z, 1.0 - z, mode='full') * G_DX
    m = float(np.max(g))
    a = G_ALPHA
    if a < 1.0:
        return m
    e = np.exp(a * np.minimum(g - m, 0.0))
    return m + float(np.log(float(np.sum(e))) / a)

def smooth_jac(z):
    n = int(G_N)
    g = np.correlate(z, 1.0 - z, mode='full') * G_DX
    m = float(np.max(g))
    e = np.exp(G_ALPHA * np.minimum(g - m, 0.0))
    s = float(np.sum(e))
    if s <= 0.0:
        return np.zeros(n)
    w = e / s
    gr = np.zeros(n)
    wl = float(w[n - 1])
    if wl > 0.0:
        gr += wl * (1.0 - 2.0 * z)
    for t in range(2 * n - 1):
        if t == n - 1:
            continue
        wl = float(w[t])
        if wl <= 1e-12:
            continue
        l = t - (n - 1)
        if l > 0:
            L = l
            m1 = n - L
            gr[:m1] += wl * (1.0 - z[L:])
            gr[L:] -= wl * z[:m1]
        else:
            L = -l
            m1 = n - L
            gr[L:] += wl * (1.0 - z[:m1])
            gr[:m1] -= wl * z[L:]
    return gr

def cont_obj(z):
    return smooth_obj(z) + 0.5 * G_PEN * (float(z.sum()) - G_TARGET) ** 2

def cont_jac(z):
    j = smooth_jac(z)
    e = float(z.sum()) - G_TARGET
    if G_PEN != 0.0:
        j = j + G_PEN * e * np.ones(G_N)
    return j

def vol_fun(z):
    return float(z.sum()) - G_TARGET

def vol_jac(z):
    return np.ones((1, G_N))

def lbfgs_polish(h, n, deadline, pairs, maxit):
    best = project_feasible(h, n)
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * int(n)
    for a, p in pairs:
        if time.time() >= deadline:
            break
        set_state(n, a, p)
        z0 = project_feasible(best, n)
        try:
            res = minimize(cont_obj, z0, jac=cont_jac, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': int(maxit), 'ftol': 1e-15})
            x = getattr(res, 'x', None)
            z = project_feasible(x if x is not None else z0, n)
        except Exception:
            z = z0
        c = compute_c5(z, n)
        if c < bc:
            best, bc = z, c
    return best, bc

def slsqp_polish(h, n, deadline, alphas, maxit):
    best = project_feasible(h, n)
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * int(n)
    cons = [{'type': 'eq', 'fun': vol_fun, 'jac': vol_jac}]
    for a in alphas:
        if time.time() >= deadline:
            break
        set_state(n, a, 0.0)
        z0 = project_feasible(best, n)
        try:
            res = minimize(smooth_obj, z0, jac=smooth_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': int(maxit), 'ftol': 1e-14})
            x = getattr(res, 'x', None)
            z = project_feasible(x if x is not None else z0, n)
        except Exception:
            z = z0
        c = compute_c5(z, n)
        if c < bc:
            best, bc = z, c
    return best, bc

def binary_hill(hf, n, deadline, seed):
    if n % 2 != 0:
        h = project_feasible(hf, n)
        return h, compute_c5(h, n)
    rng = np.random.RandomState(seed)
    tg = n // 2
    h = np.zeros(n, dtype=np.int16)
    h[np.argsort(-hf)[:tg]] = 1
    b = 1 - h
    c = np.correlate(h, b, mode='full')
    cur = int(np.max(c))
    best = h.copy()
    bc = cur
    ones = set(np.nonzero(h)[0].tolist())
    zeros = set(np.nonzero(b)[0].tolist())
    stagnant = 0
    while time.time() < deadline:
        c = np.correlate(h, b, mode='full')
        cur = int(np.max(c))
        if cur < bc:
            bc, best = cur, h.copy()
            stagnant = 0
        act = np.nonzero(c >= cur - 1)[0]
        if act.size > 100:
            act = act[np.argsort(-c[act])[:100]]
        cand = set()
        for tt in act:
            L = int(abs(tt - n + 1))
            if L == 0:
                continue
            if tt > n - 1:
                idx = np.nonzero((h[:n - L] == 1) & (b[L:] == 1))[0]
                for i in idx[:15]:
                    cand.add(int(i) * n + int(i + L))
            else:
                idx = np.nonzero((h[L:] == 1) & (b[:n - L] == 1))[0]
                for i in idx[:15]:
                    cand.add(int(i + L) * n + int(i))
        if ones and zeros:
            ol = np.fromiter(ones, dtype=int)
            zl = np.fromiter(zeros, dtype=int)
            for _ in range(120):
                cand.add(int(ol[rng.randint(ol.size)]) * n + int(zl[rng.randint(zl.size)]))
        cl = list(cand)
        if len(cl) > 1100:
            cl = [cl[i] for i in rng.choice(len(cl), 1100, replace=False)]
        else:
            rng.shuffle(cl)
        found = False
        for code in cl:
            p = code // n
            q = code - p * n
            if h[p] != 1 or h[q] != 0:
                continue
            h[p] = 0
            h[q] = 1
            b[p] = 1
            b[q] = 0
            nm = int(np.max(np.correlate(h, b, mode='full')))
            h[p] = 1
            h[q] = 0
            b[p] = 0
            b[q] = 1
            if nm < cur:
                h[p] = 0
                h[q] = 1
                b[p] = 1
                b[q] = 0
                cur = nm
                ones.remove(p)
                ones.add(q)
                zeros.remove(q)
                zeros.add(p)
                found = True
                break
        if not found:
            stagnant += 1
            k = 1 + int(rng.randint(0, 2))
            if stagnant > 8:
                k += 1
            if stagnant > 22:
                k += 2
            if k > 5:
                k = 5
            for _ in range(k):
                if not ones or not zeros:
                    break
                p = int(rng.choice(list(ones)))
                q = int(rng.choice(list(zeros)))
                h[p] = 0
                h[q] = 1
                b[p] = 1
                b[q] = 0
                ones.remove(p)
                ones.add(q)
                zeros.remove(q)
                zeros.add(p)
            c = np.correlate(h, b, mode='full')
            cur = int(np.max(c))
            if cur > bc + 15:
                h = best.copy()
                b = 1 - h
                ones = set(np.nonzero(h)[0].tolist())
                zeros = set(np.nonzero(b)[0].tolist())
                cur = bc
            stagnant = 0
    if cur < bc:
        best, bc = h.copy(), cur
    return best.astype(float), float(bc) * 2.0 / n

def pairwise_local(h0, n, deadline, seed):
    cur = project_feasible(h0, n)
    y = 1.0 - cur
    best = cur.copy()
    bc = compute_c5(cur, n)
    cf = bc
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    sweep = 0
    while time.time() < deadline:
        if sweep % 8 == 0:
            cur = project_feasible(cur, n)
            y = 1.0 - cur
            cf = compute_c5(cur, n)
            if cf < bc:
                best, bc = cur.copy(), cf
        c = np.correlate(cur, y, mode='full') * dx
        mx = float(np.max(c))
        if mx < bc:
            best, bc, cf = cur.copy(), mx, mx
        cand = set()
        act = np.nonzero(c >= mx - dx)[0]
        if act.size > 60:
            act = act[np.argsort(-c[act])[:60]]
        for tt in act:
            l = int(tt - n + 1)
            if l == 0:
                continue
            if l > 0:
                m1 = n - l
                d = cur[:m1] - cur[l:]
                o = np.argsort(-d)[:12]
                for i in o:
                    if d[i] > 0.01:
                        cand.add((int(i), int(i + l)))
            else:
                L = -l
                m1 = n - L
                d = cur[L:] - cur[:m1]
                o = np.argsort(-d)[:12]
                for i in o:
                    if d[i] > 0.01:
                        cand.add((int(i + L), int(i)))
        hi = np.argsort(-cur)[:22]
        lo = np.argsort(cur)[:22]
        for i in hi:
            for j in lo:
                if i != j:
                    cand.add((int(i), int(j)))
        for _ in range(50):
            cand.add((int(rng.randint(n)), int(rng.randint(n))))
        cl = list(cand)
        if len(cl) > 850:
            cl = [cl[i] for i in rng.choice(len(cl), 850, replace=False)]
        else:
            rng.shuffle(cl)
        imp = False
        for a, b in cl:
            if time.time() >= deadline:
                break
            if a == b or cur[a] <= 1e-9 or cur[b] >= 1.0 - 1e-9:
                continue
            if cur[b] > cur[a]:
                a, b = b, a
            step = min(cur[a], 1.0 - cur[b])
            if step <= 1e-9:
                continue
            oa, ob = cur[a], cur[b]
            bv, bt = cf, 0.0
            for al in (1.0, 0.5, 0.25, 0.1):
                t = al * step
                if t <= 1e-11:
                    continue
                cur[a] = oa - t
                cur[b] = ob + t
                y[a] = 1.0 - cur[a]
                y[b] = 1.0 - cur[b]
                v = float(np.max(np.correlate(cur, y, mode='full')) * dx)
                if v < bv:
                    bv, bt = v, t
                cur[a] = oa
                cur[b] = ob
                y[a] = 1.0 - oa
                y[b] = 1.0 - ob
            if bt > 0.0:
                cur[a] = oa - bt
                cur[b] = ob + bt
                y[a] = 1.0 - cur[a]
                y[b] = 1.0 - cur[b]
                cf = bv
                if cf < bc:
                    best, bc = cur.copy(), cf
                imp = True
                break
        sweep += 1
        if not imp and sweep % 12 == 0:
            noise = rng.randn(n)
            noise -= float(noise.mean())
            h2 = project_feasible(cur + 0.03 * noise, n)
            f2 = compute_c5(h2, n)
            if f2 < bc + 0.003:
                cur, y, cf = h2, 1.0 - h2, f2
            elif cf > bc + 0.003:
                cur, y, cf = best.copy(), 1.0 - best, bc
    best = project_feasible(best, n)
    bc = compute_c5(best, n)
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    U = max(5.0, float(budget_s) - 15.0)
    dl = t0 + U
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = int(len(h0))
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best = h0.copy()
    bc = compute_c5(best, n)

    end = t0 + 0.10 * U
    if n % 2 == 0 and end > t0 + 20:
        h, c = binary_hill(best, n, end, seed)
        if c < bc:
            best, bc = h, c

    end = t0 + 0.22 * U
    if end > t0 + 20:
        h, c = pairwise_local(best, n, end, seed)
        if c < bc:
            best, bc = h, c

    maxit = int(max(40, min(420, 0.32 * U)))
    pairs = [(50.0, 1e3), (200.0, 1e4), (1000.0, 1e4),
             (4000.0, 1e5), (16000.0, 1e5), (64000.0, 1e6),
             (250000.0, 1e6), (1000000.0, 1e6)]
    end = t0 + 0.56 * U
    if end > t0 + 40:
        h, c = lbfgs_polish(best, n, end, pairs, maxit)
        if c < bc:
            best, bc = h, c

    end = t0 + 0.70 * U
    if end > t0 + 30:
        h, c = pairwise_local(best, n, end, seed + 1)
        if c < bc:
            best, bc = h, c

    end = t0 + 0.80 * U
    if end > t0 + 30:
        h, c = lbfgs_polish(best, n, end,
                            [(50000.0, 1e6), (200000.0, 1e6),
                             (1000000.0, 1e6)], max(60, maxit // 2))
        if c < bc:
            best, bc = h, c

    end = t0 + 0.89 * U
    if end > t0 + 40:
        h, c = slsqp_polish(best, n, end,
                            [10000.0, 50000.0, 200000.0, 500000.0],
                            max(50, maxit // 3))
        if c < bc:
            best, bc = h, c

    end = t0 + 0.94 * U
    if n % 2 == 0 and end > t0 + 20:
        h, c = binary_hill(best, n, end, seed + 2)
        if c < bc:
            best, bc = h, c
            if time.time() < dl - 25:
                h, c = lbfgs_polish(best, n, dl - 25,
                                    [(50000.0, 1e6), (500000.0, 1e6)],
                                    max(50, maxit // 4))
                if c < bc:
                    best, bc = h, c

    if time.time() < dl - 4:
        h, c = pairwise_local(best, n, dl - 3, seed + 3)
        if c < bc:
            best, bc = h, c

    best = project_feasible(best, n)
    bc = compute_c5(best, n)
    return best, bc, n
