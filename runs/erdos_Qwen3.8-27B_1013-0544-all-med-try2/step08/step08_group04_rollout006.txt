def grad_shift(h,n,k):
    dx=2.0/n; g=np.zeros(n)
    if k==0: return (1.0-2.0*h)*dx
    if k>0:
       m=n-k; g[:m]+=dx*(1.0-h[k:]); g[k:]+=-dx*h[:m]

       kk=-k; m=n-kk; g[kk:]+=dx*(1.0-h[:m]); g[:m]+=-dx*h[kk:]
    return g

import numpy as np
import time

SC = (5.0 ** 0.5 - 1.0) / 2.0

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def compute_c5_vec(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0.0:
            room = 1.0 - h
            m = room > 0.0
            s = room[m].sum()
            if s <= 1e-15:
                break
            h[m] += d * (room[m] / s)
        else:
            m = h > 0.0
            s = h[m].sum()
            if s <= 1e-15:
                break
            h[m] += d * (h[m] / s)
    return np.clip(h, 0.0, 1.0)

def initial_binary(h, n):
    y = np.zeros(n, dtype=float)
    y[np.argsort(-h)[:n // 2]] = 1.0
    return project_feasible(y, n)

def grad_shift(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k == 0:
        return (1.0 - 2.0 * h) * dx
    if k > 0:
        m = n - k
        g[:m] += dx * (1.0 - h[k:])
        g[k:] += -dx * h[:m]
    else:
        kk = -k
        m = n - kk
        g[kk:] += dx * (1.0 - h[:m])
        g[:m] += -dx * h[kk:]
    return g

def pair_eval(h, n, i, j, d):
    a = h[i]
    b = h[j]
    h[i] = a + d
    h[j] = b - d
    c = compute_c5(h, n)
    h[i] = a
    h[j] = b
    return c

def pair_golden(h, n, i, j, a, b, maxe, dl):
    if b <= a:
        return None
    x1 = b - SC * (b - a)
    x2 = a + SC * (b - a)
    f1 = pair_eval(h, n, i, j, x1)
    f2 = pair_eval(h, n, i, j, x2)
    bestd = x1
    bestv = f1
    if f2 < bestv:
        bestv = f2
        bestd = x2
    for _ in range(maxe):
        if time.time() > dl:
            break
        if b - a < 1e-12:
            break
        if f1 <= f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = b - SC * (b - a)
            f1 = pair_eval(h, n, i, j, x1)
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + SC * (b - a)
            f2 = pair_eval(h, n, i, j, x2)
        if f1 < bestv:
            bestv = f1
            bestd = x1
        if f2 < bestv:
            bestv = f2
            bestd = x2
    return bestv, bestd

def pair_line_search(h, n, i, j, best, dl, radius=0.05, npts=21, glob=False):
    hi = 1.0 - h[i]
    lo = -h[j]
    eps = 1e-13
    if hi <= lo + eps:
        return best
    if glob:
        a = lo + eps
        b = hi - eps
    else:
        a = max(lo, -radius)
        b = min(hi, radius)
    if b - a < 1e-13:
        return best
    ds = np.linspace(a, b, int(npts))
    bv = best
    bd = 0.0
    bi = -1
    for ii, d in enumerate(ds):
        if time.time() > dl:
            break
        if abs(d) < 1e-13:
            continue
        h[i] += d
        h[j] -= d
        c = compute_c5(h, n)
        h[i] -= d
        h[j] += d
        if c < bv:
            bv = c
            bd = d
            bi = ii
    if bd != 0.0 and bi > 0 and bi < len(ds) - 1:
        ga = ds[bi - 1]
        gb = ds[bi + 1]
        if gb - ga > 1e-12:
            r = pair_golden(h, n, i, j, ga, gb, 10, dl)
            if r is not None and r[0] < bv:
                bv = r[0]
                bd = r[1]

        h[i] += bd
        h[j] -= bd
        if h[i] < 0.0:
            h[i] = 0.0
        elif h[i] > 1.0:
            h[i] = 1.0
        if h[j] < 0.0:
            h[j] = 0.0
        elif h[j] > 1.0:
            h[j] = 1.0
        return bv
    return best

def dense_step(h, n, best, dl, step=0.05):
    cvec = compute_c5_vec(h, n)
    off = n - 1
    idx = np.argsort(cvec)[-4:]
    for ii in idx:
        if time.time() > dl:
            break
        g = grad_shift(h, n, off - int(ii))
        d = -g
        d -= d.mean()
        mx = float(np.max(np.abs(d)))
        if mx < 1e-13:
            continue
        d *= step / mx
        pos = d > 1e-14
        neg = d < -1e-14
        lo = -1.0
        hi = 1.0
        if pos.any():
            hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if neg.any():
            lo = max(lo, float(np.min(h[neg] / (-d[neg]))))
        hi -= 1e-12
        lo += 1e-12
        if hi <= lo:
            continue
        ds = np.linspace(lo, hi, 17)
        bv = best
        bd = 0.0

            if abs(a) < 1e-13:
                continue

            c = compute_c5(h, n)
            h -= a * d
            if c < bv:
                bv = c
                bd = a
        if bd != 0.0 and bv < best - 1e-14:
            h += bd * d
            best = bv
    return best

def candidate_pairs(h, n, maxc=45):
    cvec = compute_c5_vec(h, n)
    off = n - 1
    idx = np.argsort(cvec)[-5:]
    ti, tj = np.triu_indices(n, 1)
    scored = []
    seen = set()
    k = min(10, len(ti) - 1)
    for ii in idx:
        g = grad_shift(h, n, off - int(ii))
        s = np.abs(g[ti] - g[tj])
        top = np.argpartition(s, k)[-k:]
        top = top[np.argsort(s[top])[::-1]]
        for t in top:
            i = int(ti[t])
            j = int(tj[t])
            if (i, j) not in seen:
                seen.add((i, j))
                scored.append((float(s[t]), i, j))
    scored.sort(reverse=True)
    return scored[:maxc]

def polish_pair(h, n, best, dl):
    h = project_feasible(h.copy(), n)
    best = compute_c5(h, n)
    radii = [0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]

    h = project_feasible(h, n)
    best = compute_c5(h, n)
    return h, best
