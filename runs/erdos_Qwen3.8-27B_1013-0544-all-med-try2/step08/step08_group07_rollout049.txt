def slsqp_optimize(h, n, alpha, maxiter=2000, ftol=1e-15):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - target
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h.copy()

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj_grad(h, n, alpha, penalty=0.0):
    """Compute soft-max objective and its gradient."""
    dx = 2.0 / n
    target = n / 2.0
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    weights = np.exp(shifted - sm)
    wsum = weights.sum()
    weights = weights / wsum
    obj_val = m + (sm + np.log(wsum)) / alpha
    
    # Gradient of soft-max w.r.t. h
    # d(g_k)/d(h_i) = dx * [(1-h(i+k))*I(0<=i+k<n) - h(i-k)*I(0<=i-k<n)]
    # where k ranges over lags -(n-1) to (n-1)
    # In np.correlate(h, 1-h, 'full'), index i corresponds to lag (i - (n-1))
    grad = np.zeros(n)
    m_range = np.arange(-(n-1), n)
    for idx in range(len(m_range)):
        k = m_range[idx]
        w = weights[idx] * dx
        if abs(w) < 1e-18:
            continue
        # First part: (1-h(i+k)) for valid i+k
        # i+k in [0,n) => i in [-k, n-1-k] => i in [max(0,-k), min(n-1, n-1-k)]
        # For k >= 0: i in [0, n-1-k], h[i+k] = h[k:n]
        # For k < 0: i in [-k, n-1], h[i+k] = h[0:n+k] (since i+k = i+(-|k|))
        #   Wait: if k < 0, i+k >= 0 means i >= -k = |k|
        #   h[i+k] for i = |k|, ..., n-1 is h[0], h[1], ..., h[n-1-|k|] = h[0:n-|k|]
        
        # Second part: -h(i-k) for valid i-k
        # i-k in [0,n) => i in [k, n-1+k] => i in [max(0,k), min(n-1, n-1+k)]
        # For k >= 0: i in [k, n-1], h[i-k] = h[0:n-k]
        # For k < 0: i in [0, n-1+k] = [0, n-1-|k|], h[i-k] = h[|k|:n]
        
        if k >= 0:
            nk = n - k  # number of valid i for first part
            if nk > 0:
                grad[:nk] += w * (1.0 - h[k:n])
            nk2 = n - k  # number of valid i for second part
            if nk2 > 0:
                grad[k:] += -w * h[0:nk2]
        else:
            pk = -k  # |k|
            nkp = n - pk
            if nkp > 0:
                grad[pk:] += w * (1.0 - h[0:nkp])
            if nkp > 0:
                grad[:nkp] += -w * h[pk:n]
    
    # Add penalty gradient
    if penalty > 0:
        se = h.sum() - target
        obj_val += penalty * se * se
        grad += 2.0 * penalty * se
    
    return obj_val, grad

def coord_descent(h, n, max_sweeps=100, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(40):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def slsqp_step(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - target
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h.copy()
