def grad_from_lag_weights(h, w_dx):
    n = len(h)
    R = n - 1
    grad = np.zeros(n)
    om = 1.0 - h
    M = 2 * R + 1
    for l in range(M):
        w = w_dx[l]
        if w == 0.0:
            continue
        d = R - l
        if d >= 0:
            end = n - d
            grad[:end] += w * om[d:d+end]
            grad[d:d+end] -= w * h[:end]
        else:
            mlag = -d
            length = n - mlag
            grad[mlag:] += w * om[:length]
            grad[:length] -= w * h[mlag:]
    return grad

def exact_c(h):
    n = len(h)
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def clip_sum(h, tau):
    return float(np.sum(np.clip(h - tau, 0.0, 1.0)))


def sig_sum(z, beta):
    return float(np.sum(expit(z + beta)))


def project_feasible(h):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    target = CFG.get('target', float(n) / 2.0)
    s0 = clip_sum(h, 0.0)
    if s0 >= target:
        lo = 0.0
        hi = 1.0
        while clip_sum(h, hi) > target:
            hi *= 2.0
            if hi > 1e6:

def grad_from_lag_weights(h, w_dx):
