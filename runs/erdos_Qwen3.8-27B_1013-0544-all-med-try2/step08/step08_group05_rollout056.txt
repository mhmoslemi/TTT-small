def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n, rng):
    target = n // 2
    h = np.where(h >= 0.5, 1.0, 0.0)
    n_ones = int(h.sum())
    if n_ones > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:n_ones - target]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:target - n_ones]] = 1
    return h.astype(float)

                best_h = h.copy()

def slsqp_polish(h, n, deadline, alpha=100000, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)

        return project_feasible(h, n)

                best_c = c
                best_h = hp.copy()
                h_bin = to_binary(best_h, n, rng)

    # Phase 4: Final binary search + polish (15%)
    if time.time() < dl - 10:
        hb, cb = focused_hill_climb(best_h, n, dl - 15, seed=seed + 999)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        if time.time() < dl - 3:
            h_bin = to_binary(best_h, n, rng)
            hp = slsqp_polish(h_bin, n, dl - 1, alpha=2000000, maxiter=500)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
