from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

    idx = np.argsort(c)[-top_k:]
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    dx = 2.0 / n

        if w[i] > 1e-10:
            grad += w[i] * dx * grad_single(h, n, k)
    return grad

    return float(np.sum(c**p) ** (1.0/p) + 50.0*(h.sum() - n/2)**2)

    s = np.sum(c**p)

    s_val = h.sum() - n/2
    grad += 100.0 * s_val

    grad = np.zeros(n)
    for i in range(top_k):

                best_h[j] -= d

            no_improve = 0
        else:
            no_improve += 1
            step *= 0.8
    
    return best_h, best_c

    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)

            h += (diff / tot) * room

            if tot < 1e-14:
