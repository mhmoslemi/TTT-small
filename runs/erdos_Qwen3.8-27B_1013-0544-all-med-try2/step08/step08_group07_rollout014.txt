import time
import numpy as np
from scipy.optimize import minimize

SUM_PEN = 1e-4


def project_feasible(h, n):
    z = np.atleast_1d(np.asarray(h, dtype=float))
    if z.size != n:
        z = np.resize(z, n)
    z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, 0.0, 1.0)
    t = 0.5 * n
    diff = t - z.sum()
    if abs(diff) < 1e-7:
        mask = (z > 1e-10) & (z < 1.0 - 1e-10)
        if mask.any():
            z[mask] += diff / float(mask.sum())
            z = np.clip(z, 0.0, 1.0)
            if abs(z.sum() - t) < 1e-12:
                return z
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > t:
            lo = mid
        else:
            hi = mid
    z = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    diff = t - z.sum()
    if abs(diff) > 1e-12:
        mask = (z > 1e-12) & (z < 1.0 - 1e-12)
        if mask.any():
            z[mask] += diff / float(mask.sum())
            z = np.clip(z, 0.0, 1.0)
    return z


def c5(h, n):
    h = np.asarray(h, dtype=float)
    if not np.all(np.isfinite(h)):
        return float("inf")
    c = np.correlate(h, 1.0 - h, mode="full")
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return float("inf")
    return mx * (2.0 / n)


def sum_con(x):
    x = np.asarray(x, dtype=float)
    return float(x.sum() - 0.5 * len(x))


def sum_jac(x):
    return np.ones(len(x), dtype=float)


def soft_value(x, alpha, dx, n):
    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x)):
        return 1e100
    c = np.correlate(x, 1.0 - x, mode="full")
    g = c * dx
    m = float(g.max())
    if not np.isfinite(m):
        return 1e100
    e = alpha * (g - m)
    e = e - e.max()
    se = float(np.exp(e).sum())
    val = m + np.log(se) / alpha
    s = float(x.sum() - 0.5 * n)
    val += SUM_PEN * s * s
    return float(val)


def weighted_grad_direct(h, n, idx, w):
    center = n - 1
    grad = np.zeros(n, dtype=float)
    for jj, wt in zip(idx, w):
        if wt <= 1e-15:
            continue
        d = int(jj) - center
        if d > 0:
            grad[d:] += wt * (1.0 - h[:-d])
            grad[:-d] -= wt * h[d:]
        elif d < 0:
            p = -d
            grad[:-p] += wt * (1.0 - h[p:])
            grad[p:] -= wt * h[:-p]
        else:
            grad += wt * (1.0 - 2.0 * h)
    return grad


def soft_grad(x, alpha, dx, n):

    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)
    c = np.correlate(x, 1.0 - x, mode="full")

    m = float(g.max())
    e = alpha * (g - m)
    e = e - e.max()

    else:

        if su < 1.0 - 1e-8:
            wu = wu / su
    grad = dx * weighted_grad_direct(x, n, idx, wu)
    s = float(x.sum() - 0.5 * n)
    grad += (2.0 * SUM_PEN * s) * np.ones(n, dtype=float)
    return grad


def pen_value(x, alpha, mu, dx, n):
    x = np.asarray(x, dtype=float)
    s = float(x.sum() - 0.5 * n)
    return soft_value(x, alpha, dx, n) + 0.5 * mu * s * s


def pen_grad(x, alpha, mu, dx, n):
    x = np.asarray(x, dtype=float)
    s = float(x.sum() - 0.5 * n)
    return soft_grad(x, alpha, dx, n) + mu * s * np.ones(n, dtype=float)


def lbfgs_smooth(h, n, alpha, mu, maxiter):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            pen_value,
            h0,
            method="L-BFGS-B",
            jac=pen_grad,
            bounds=bounds,
            args=(alpha, mu, 2.0 / n, n),
            options={"maxiter": int(maxiter), "ftol": 1e-10, "disp": False},
        )
        x = np.clip(np.asarray(res.x, dtype=float), 0.0, 1.0)
        if not np.all(np.isfinite(x)):
            return h0
    except Exception:
        return h0
    return project_feasible(x, n)


def slsqp_smooth(h, n, alpha, maxiter, ftol):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con, "jac": sum_jac}]
    try:
        res = minimize(
            soft_value,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            jac=soft_grad,
            args=(alpha, 2.0 / n, n),
            options={"maxiter": int(maxiter), "ftol": float(ftol), "disp": False},
        )
        x = np.clip(np.asarray(res.x, dtype=float), 0.0, 1.0)
        if not np.all(np.isfinite(x)):
            return h0
    except Exception:
        return h0
    return project_feasible(x, n)


def active_weights(c, dx, beta, top=300):
    g = c * dx
    m = float(g.max())
    e = beta * (g - m)
    e = e - e.max()
    w = np.exp(e)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.ones_like(w) / len(w)
    else:
        w = w / s
    tol = max(1e-10, float(w.max()) * 1e-4)
    idx = np.nonzero(w > tol)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(w))])
    elif idx.size > top:
        idx = np.argsort(w)[::-1][:top]
    wu = w[idx]
    su = float(wu.sum())
    if su < 1.0 - 1e-8:
        wu = wu / su
    return idx, wu


def active_descent(h, n, deadline, max_eval=1000):
    h = project_feasible(h, n)
    cur = c5(h, n)
    dx = 2.0 / n
    evals = 0
    for beta in (20.0, 100.0, 500.0, 2000.0, 10000.0):
        while time.time() < deadline - 1.0 and evals < int(max_eval):
            cc = np.correlate(h, 1.0 - h, mode="full")
            idx, wu = active_weights(cc, dx, beta, 300)
            grad = dx * weighted_grad_direct(h, n, idx, wu)
            d = -grad
            d -= float(np.mean(d))
            norm = float(np.linalg.norm(d))
            if norm < 1e-14:
                break
            d *= 1.0 / norm
            best_h = h
            best_c = cur
            for step in (1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.3, 1.0):
                if time.time() > deadline - 1.0 or evals >= int(max_eval):
                    break
                cand = project_feasible(h + step * d, n)
                cv = c5(cand, n)
                evals += 1
                if cv < best_c - 1e-13:
                    best_c = cv
                    best_h = cand
            if best_c < cur - 1e-13:
                h = best_h
                cur = best_c
            else:
                break
    return h, cur


def random_polish(h, n, deadline, rng, max_eval=1000):
    h = project_feasible(h, n)
    cur = c5(h, n)
    evals = 0
    fails = 0
    while time.time() < deadline - 1.0 and evals < int(max_eval) and fails < 300:
        d = rng.randn(n)
        d -= d.mean()
        norm = float(np.linalg.norm(d))
        if norm < 1e-14:
            fails += 1
            continue
        d *= 1.0 / norm
        best_h = h
        best_c = cur
        for step in (1e-4, 5e-4, 2e-3, 1e-2, 5e-2, 0.2, 0.8):
            if time.time() > deadline - 1.0 or evals >= int(max_eval):
                break
            cand = project_feasible(h + step * d, n)
            cv = c5(cand, n)
            evals += 1
            if cv < best_c - 1e-13:
                best_c = cv
                best_h = cand
                break
        if best_c < cur - 1e-13:
            h = best_h
            cur = best_c
            fails = 0
        else:
            fails += 1
    return h, cur


def polish_candidate(h, n, deadline, quick=False):
    h = project_feasible(h, n)
    cur = c5(h, n)
    if time.time() > deadline - 25.0:
        return h, cur
    if quick:
        alphas = (50.0, 500.0, 3000.0, 30000.0)
        maxits = (60, 70, 60, 50)
    else:
        alphas = (20.0, 100.0, 500.0, 2000.0, 10000.0, 50000.0, 200000.0, 1000000.0)
        maxits = (80, 120, 120, 100, 100, 80, 70, 60)
    fails = 0
    for i, alpha in enumerate(alphas):
        now = time.time()
        left = deadline - now
        if left < 70.0:
            break
        mi = maxits[i]
        if left < 160.0:
            mi = min(mi, 40)
        elif left < 300.0:
            mi = min(mi, 80)
        if not quick and alpha <= 100.0:
            h2 = lbfgs_smooth(h, n, alpha, 1e4, min(mi, 80))
            c2 = c5(h2, n)
            if c2 < cur - 1e-13:
                h = h2
                cur = c2
                fails = 0
            else:
                fails += 1
        h2 = slsqp_smooth(h, n, alpha, mi, 1e-10)
        c2 = c5(h2, n)
        if c2 < cur - 1e-13:
            h = h2
            cur = c2
            fails = 0
        else:
            fails += 1
            if alpha > 10000.0 and fails >= 3:
                break
    if not quick and time.time() < deadline - 12.0:
        h, cur = active_descent(h, n, deadline - 4.0, max_eval=300)
    return h, cur


def make_fourier(n, rng, kmax=6):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    s = np.zeros(n, dtype=float)
    for k in range(1, kmax + 1):
        amp = rng.uniform(0.05, 0.30)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        s += amp * np.cos(np.pi * k * x + ph)
    sc = max(1e-12, float(np.max(np.abs(s))))
    h = 0.5 + 0.5 * (s / sc)
    return project_feasible(np.clip(h, 0.0, 1.0), n)


def to_binary(h, n):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(h)[::-1][: max(1, n // 2)]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return project_feasible(b, n)


def add_noise(h, n, amp, rng):
    h = np.asarray(h, dtype=float)
    return project_feasible(h + amp * rng.randn(n), n)


def resample(h, n_old, n_new):
    h = np.asarray(h, dtype=float)
    if n_old == n_new:
        return project_feasible(h, n_new)
    xo = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return project_feasible(np.interp(xn, xo, h), n_new)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(30.0, float(budget_s) - 20.0)
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    best_h = None
    best_n = 400
    try:
        p = np.atleast_1d(np.asarray(initial_h_values, dtype=float))
        if p.size > 0:
            best_n = int(p.size)
            best_h = project_feasible(p, best_n)
    except Exception:
        best_h = None

    if best_h is None:
        best_n = 400
        best_h = make_fourier(best_n, rng, 8)

    best_c = c5(best_h, best_n)

    dl1 = min(deadline - 12.0, t0 + max(30.0, 0.48 * total))
    best_h, best_c = polish_candidate(best_h, best_n, dl1, quick=False)

    starts = []
    if best_n > 0:
        if best_n < 600:
            nn = min(600, best_n + (100 if best_n < 500 else 50))
            if nn != best_n:
                starts.append(resample(best_h, best_n, nn))
                starts.append(nn)
        starts.append(to_binary(best_h, best_n))
        starts.append(best_n)
        starts.append(add_noise(best_h, best_n, 0.02, rng))
        starts.append(best_n)
        starts.append(add_noise(best_h, best_n, 0.08, rng))
        starts.append(best_n)

    rem_deadline = deadline - 45.0
    it = 0
    while it + 1 < len(starts):
        if time.time() > rem_deadline - 25.0:
            break
        h0 = starts[it]
        n0 = int(starts[it + 1])
        it += 2
        h0 = project_feasible(h0, n0)
        c0 = c5(h0, n0)
        if not np.isfinite(c0) or c0 > best_c + 0.05:
            continue
        nrem = (len(starts) - it) // 2
        if nrem <= 0:
            nrem = 1
        left = rem_deadline - time.time()
        alloc = min(260.0, max(40.0, left / float(nrem) * 0.85))
        if alloc < 45.0:
            continue
        dl = min(rem_deadline, time.time() + alloc)
        cand, cv = polish_candidate(h0, n0, dl, quick=True)
        if cv < best_c:
            best_h, best_c, best_n = cand, cv, n0

    if time.time() < rem_deadline - 60.0:
        hl = add_noise(best_h, best_n, 0.03, rng)
        cand, cv = polish_candidate(hl, best_n, min(rem_deadline, time.time() + 150.0), quick=True)
        if cv < best_c:
            best_h, best_c = cand, cv

    if time.time() < deadline - 12.0:
        variants = [
            best_h,
            1.0 - best_h,
            best_h[::-1],
            1.0 - best_h[::-1],
            0.5 * (best_h + 1.0 - best_h[::-1]),
        ]
        for vh in variants:
            cand = project_feasible(vh, best_n)
            cv = c5(cand, best_n)
            if cv < best_c:
                best_h, best_c = cand, cv

    if time.time() < deadline - 25.0:
        best_h, best_c = active_descent(best_h, best_n, deadline - 12.0, max_eval=1500)

    if time.time() < deadline - 10.0:
        best_h, best_c = random_polish(best_h, best_n, deadline - 8.0, rng, max_eval=1500)

    best_h = project_feasible(best_h, best_n)
    best_c = c5(best_h, best_n)
    if not np.isfinite(best_c):
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c = c5(best_h, best_n)

    return best_h, float(best_c), int(best_n)
