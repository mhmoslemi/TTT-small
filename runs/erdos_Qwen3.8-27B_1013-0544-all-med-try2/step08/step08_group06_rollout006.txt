    # Use bisection

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_box_sum(h, target):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = g.max()
    ex = np.exp(alpha * (g - mx))
    w = ex / ex.sum()
    val = mx + np.log(ex.sum()) / alpha
    # Gradient
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    jnp = np.arange(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = lags[k]
        i1 = jnp + m
        v1 = (i1 >= 0) & (i1 < n)
        i2 = jnp - m
        v2 = (i2 >= 0) & (i2 < n)
        t = np.zeros(n)
        t[v1] = 1.0 - h[i1[v1]]
        t[v2] -= h[i2[v2]]
        grad += wk * t
    grad *= dx
    return val, grad

    target = n / 2.0
    def fun(x):
        v, _ = smooth_obj_grad(x, n, alpha)
        return v + 500.0 * (x.sum() - target)**2

        _, g = smooth_obj_grad(x, n, alpha)
        return g + 1000.0 * (x.sum() - target)

                   bounds=[(0,1)]*n,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15})
    return project_box_sum(res.x, target)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_box_sum(h, target):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def precompute_indices(n):
    lags = np.arange(-(n - 1), n)
    jnp = np.arange(n)
    i1 = jnp[None, :] + lags[:, None]
    i2 = jnp[None, :] - lags[:, None]
    v1 = (i1 >= 0) & (i1 < n)
    v2 = (i2 >= 0) & (i2 < n)
    i1c = np.clip(i1, 0, n - 1)
    i2c = np.clip(i2, 0, n - 1)
    return lags, i1c, v1, i2c, v2

def smooth_obj_grad(h, n, alpha, pre):
    dx = 2.0 / n
    lags, i1c, v1, i2c, v2 = pre
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = g.max()
    ex = np.exp(alpha * (g - mx))
    esum = ex.sum()
    w = ex / esum
    val = mx + np.log(esum) / alpha
    t1 = w @ v1
    hi1 = np.where(v1, h[i1c], 0.0)
    t2 = w @ hi1
    hi2 = np.where(v2, h[i2c], 0.0)
    t3 = w @ hi2
    grad = dx * (t1 - t2 - t3)
    return val, grad

def lgbfgs_round(h, n, alpha, maxiter, pre, target):
    def fun(x):
        v, _ = smooth_obj_grad(x, n, alpha, pre)
        return v + 500.0 * (x.sum() - target) ** 2
    def gfun(x):
        _, gr = smooth_obj_grad(x, n, alpha, pre)
        return gr + 1000.0 * (x.sum() - target)
    try:
        res = minimize(fun, h, jac=gfun, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
        return res.x
    except Exception:
        return h

def greedy_binary(h, n, deadline, max_sweeps=300):
    dx = 2.0 / n
    order = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[order[:n // 2]] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    lags = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            best_j, best_nc = -1, cur_c
            im = i + lags
            vm = (im >= 0) & (im < n)
            im2 = i - lags
            vm2 = (im2 >= 0) & (im2 < n)
            for j in ones:
                jm = j + lags
                vj = (jm >= 0) & (jm < n)
                jm2 = j - lags
                vj2 = (jm2 >= 0) & (jm2 < n)
                delta = np.zeros(2 * n - 1)
                delta[vm] -= h[im[vm]]
                delta[vm2] -= h[im2[vm2]]
                delta[vj] += h[jm[vj]]
                delta[vj2] += h[jm2[vj2]]
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
                    best_delta = delta.copy()
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr += best_delta
                cur_c = best_nc
                ones.remove(best_j)
                zeros.remove(i)
                ones.append(i)
                zeros.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
