def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = compute_c5_all(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = x.sum() - n / 2.0
    return val + 1e-4 * s * s

def smooth_max_grad(x, n, alpha):
    c = compute_c5_all(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - x[k:])
            g[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - x[:nv])
            g[:nv] += wm * (-x[kk:])

                      options={'maxiter': max_iter, 'ftol': 1e-18,
                               'gtol': 1e-16, 'maxfun': max_iter * 40})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except:
        return h.copy(), compute_c5(h, n)

def nestorov_steps(h, n, deadline, alpha, base_lr=0.01, n_steps=500):
    """Nesterov-accelerated proximal gradient."""
    best_c = compute_c5(h, n)
    h_acc = h.copy()
    h_prev = h.copy()
    
    for t in range(n_steps):
        if time.time() > deadline:
            break
        # Nesterov momentum
        beta = (1 + np.sqrt(1 + 4 * (t+1)**2)) / (2 * (t+1) + 2)
        h_mom = h_acc + beta * (h_acc - h_prev)
        
        # Gradient step
        g = smooth_max_grad(h_mom, n, alpha)
        lr = base_lr / (1 + t * 0.01)
        h_new = h_mom - lr * g
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
        
        h_prev = h_acc
        h_acc = h_new
    
    return h_acc, best_c

                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                            break
                        h[i] += step
                        h[j] -= step
                    if h[j] >= step and h[i] <= 1.0 - step:
                        h[j] -= step
                        h[i] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                            break
                        h[j] += step
                        h[i] -= step
    return h, best_c

def multi_alpha_lbfgs(h, n, deadline):
    best_c = compute_c5(h, n)
    alphas = [2000, 10000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 2:
            break
        rem = deadline - time.time()
        if rem < 10:
            break
        iters = max(30, min(300, int(rem * 0.6)))
        h_new, c_new = lbfgs_run(h, n, time.time() + rem * 0.8, alpha=a, max_iter=iters)
        if c_new < best_c:
            h = h_new
            best_c = c_new
    return h, best_c

def perturbed_restart(h_best, n, rng, best_c, deadline, alloc):
    h = h_best + rng.randn(n) * 0.03
    h = project_feasible(h, n)
    h_new, c_new = multi_alpha_lbfgs(h, n, min(time.time() + alloc * 0.6, deadline))
    if c_new < best_c:
        h_new, c_new = deterministic_local_search(h_new, n, min(time.time() + alloc * 0.3, deadline), radius=20)
    return h_new, c_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 500
        h = np.zeros(n)
        half = n // 2
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        pos = np.floor(np.arange(half) * phi).astype(np.int64) % n
        h[pos] = 1.0

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: Multi-alpha L-BFGS (main optimization)
    t1 = min(t0 + 250, deadline - 600)
    h, c = multi_alpha_lbfgs(h, n, t1)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 2: Subgradient method focused on worst shifts
    if time.time() < deadline - 350:
        h, c = subgradient_method(best_h, n, min(time.time() + 120, deadline - 300), top_k=8)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: More L-BFGS with very high alpha
    if time.time() < deadline - 300:
        h, c = lbfgs_run(best_h, n, min(time.time() + 100, deadline - 250),
                        alpha=2000000, max_iter=400)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: Subgradient with more shifts
    if time.time() < deadline - 250:
        h, c = subgradient_method(best_h, n, min(time.time() + 80, deadline - 200), top_k=20)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 5: Deterministic local search (main polish)
    t5 = min(time.time() + 200, deadline - 100)
    h, c = deterministic_local_search(best_h, n, t5, radius=40)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 6: Stochastic local search with very fine steps
    if time.time() < deadline - 80:
        h, c = stochastic_local_search(best_h, n, min(time.time() + 120, deadline - 50),
                                        radius=30, n_iters=300000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 7: Final L-BFGS polish
    if time.time() < deadline - 50:
        h, c = lbfgs_run(best_h, n, min(time.time() + 40, deadline - 30),
                        alpha=5000000, max_iter=300)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 8: Perturbed restarts to escape local minima
    for _ in range(3):
        if time.time() > deadline - 60:
            break
        alloc = min(50, (deadline - time.time() - 25) * 0.5)
        if alloc < 20:
            break
        h_r, c_r = perturbed_restart(best_h, n, rng, best_c, deadline, alloc)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()

    # Phase 9: Final ultra-fine local search
    if time.time() < deadline - 5:
        t9 = deadline - 3
        h, c = deterministic_local_search(best_h, n, t9, radius=20)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
