def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    """Compute full correlation array for h and 1-h"""
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    val = m + np.log(np.sum(np.exp(sh - smax))) / alpha
    return val * (2.0 / n)

def smooth_max_grad(x, n, alpha):

    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)

Actually, let me try a different continuous refinement approach: **projected subgradient method** on the actual max function.

The subgradient of f(h) = max_k c_k(h) at a point h is the subgradient of c_k* where k* is the argmax. If there are multiple k* with the same max value, the subgradient is the convex hull.

For a single k*, the gradient of c_k*(h) = sum_i h[i+s](1-h[i]) (for shift s = k*-(n-1)) is:
- For i in valid range: d/dc_k* h[i+s] = (1-h[i]), d/dc_k* h[i] = -h[i+s]

So the subgradient is straightforward. We can do:
1. Find k* = argmax
2. Compute gradient of c_k*
3. Take a projected gradient step in the negative direction
4. Project to feasible set
