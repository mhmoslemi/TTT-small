import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    corr = np.correlate(h, 1 - h, mode='full')
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def smooth_grad(h, n, alpha):
    corr = np.correlate(h, 1 - h, mode='full')
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        idx = np.arange(n)
        ip = idx + lag
        im = idx - lag
        mp = (ip >= 0) & (ip < n)
        mm = (im >= 0) & (im < n)
        if mp.any():
            grad[mp] += wk * (1.0 - h[ip[mp]])
        if mm.any():
            grad[mm] -= wk * h[im[mm]]
    return grad

        if deadline is not None and time.time() > deadline:

        best_step = 0.0

            mask2 = (k2 >= 0) & (k2 < n)

            val0 = np.max(corr - B * h[i])
