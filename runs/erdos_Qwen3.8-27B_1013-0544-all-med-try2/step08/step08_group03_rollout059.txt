from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    c = compute_c5_vec(h, n)
    return float(np.max(c))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_max_obj_grad(h, n, alpha):
    """Compute smooth max objective and gradient."""
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    dx = 2.0 / n
    # Gradient: weighted sum of individual gradients
    top_k = min(100, len(w))
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n)
    for i in top_idx:
        if w[i] < 1e-12:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(h, n, k)
    return val, grad

def compute_c5(h, n):

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha + 100.0*(x.sum()-n/2.0)**2)

def smooth_max_grad(x, n, alpha, top_k=100):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n)
    for i in top_idx:
        if w[i] < 1e-12:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    s = x.sum() - n / 2.0
    grad += 200.0 * s
    return grad

def nesterov_gd(h, n, deadline, alpha=80.0, step=0.04, top_k=80):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    y = h.copy()
    t = 1.0
    stall = 0
    while time.time() < deadline:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - y)
        grad = smooth_max_grad(y, n, alpha, top_k)
        x_new = project_feasible(y - step * grad, n)
        c_new = compute_c5(x_new, n)
        x = x_new
        t = t_new
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = x_new.copy()
            stall = 0
        else:
            stall += 1
        if stall % 50 == 0:
            alpha *= 1.4
            step = min(step * 0.92, 0.01)
        if stall > 400:
            break
    return best_h, best_c

def precompute_G(h, n):
    """Precompute G[i,k] = dc[k]/dh[i] for all i, k. Shape (n, 2n-1)."""
    dx = 2.0 / n
    G = np.zeros((n, 2 * n - 1))
    for i in range(n):
        for k in range(-(n-1), n):
            d = 0.0
            ki = k + n - 1
            if k >= 0:
                if i <= n - 1 - k:
                    d += (1.0 - h[i + k])
                if i >= k:
                    d -= h[i - k]
            else:
                kk = -k
                if 0 <= i - kk <= n - 1 - kk:
                    d += (1.0 - h[i - kk])
                if i <= n - 1 - kk:
                    d -= h[i + kk]
            G[i, ki] = dx * d
    return G

def fast_local_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c = compute_c5_vec(h, n)
    G = precompute_G(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    radius = 40
    rounds = 0
    while time.time() < deadline - 1 and rounds < 200:
        improved = False
        for di, d in enumerate(deltas):
            if time.time() > deadline - 1:
                break
            for i in range(n):
                if time.time() > deadline - 1:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i]+=d, h[j]-=d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        delta_c = d * (G[i] - G[j])
                        new_max = (c + delta_c).max()
                        if new_max < best_c - 1e-14:
                            h[i] += d
                            h[j] -= d
                            c += delta_c
                            G = precompute_G(h, n)
                            best_c = float(new_max)
                            best_h = h.copy()
                            improved = True
                            break
                    # Try h[i]-=d, h[j]+=d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        delta_c = -d * (G[i] - G[j])
                        new_max = (c + delta_c).max()
                        if new_max < best_c - 1e-14:
                            h[i] -= d
                            h[j] += d
                            c += delta_c
                            G = precompute_G(h, n)
                            best_c = float(new_max)
                            best_h = h.copy()
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        rounds += 1
        if not improved:
            radius = max(20, radius // 2)
            if radius < 21:
                break
    return best_h, best_c

def lbfgs_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha + 50.0*(x.sum()-n/2.0)**2)

def lbfgs_grad(x, n, alpha, top_k=100):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(len(x))
    for i in top_idx:
        if w[i] < 1e-12:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    grad += 100.0 * (x.sum() - n / 2.0)
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.zeros(n)
        idx = rng.choice(n, size=n // 2, replace=False)
        h_init[idx] = 1.0
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    best_n = n
    # Phase 1: Nesterov GD
    t1 = min(t0 + 200, deadline - 350)
    best_h, best_c = nesterov_gd(best_h, best_n, t1, alpha=80.0, step=0.04)
    # Phase 2: L-BFGS-B refinement
    if time.time() < deadline - 200:
        t2 = min(time.time() + 150, deadline - 150)
        bounds = [(0.0, 1.0)] * best_n
        for alpha in [500, 2000, 8000]:
            if time.time() > t2:
                break
            try:
                res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad,
                              args=(best_n, alpha, 100), method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': 200, 'ftol': 1e-16})
                h_new = project_feasible(res.x, best_n)
                c_new = compute_c5(h_new, best_n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    # Phase 3: Try different resolutions
    for n_try in [400, 600, 700]:
        if time.time() > deadline - 150:
            break
        if n_try == best_n:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(80, deadline - time.time() - 120)
        if alloc < 30:
            continue
        bounds = [(0.0, 1.0)] * n_try
        try:
            res = minimize(lbfgs_obj, h_interp, jac=lbfgs_grad,
                          args=(n_try, 3000, 100), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 150, 'ftol': 1e-15})
            h_t = project_feasible(res.x, n_try)
            c_t = compute_c5(h_t, n_try)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try
        except Exception:
            pass
    # Phase 4: Nesterov GD on best resolution
    if time.time() < deadline - 120:
        t4 = min(time.time() + 80, deadline - 80)
        best_h, best_c = nesterov_gd(best_h, best_n, t4, alpha=200.0, step=0.02, top_k=60)
    # Phase 5: Fast local search
    if time.time() < deadline - 2:
        best_h, best_c = fast_local_search(best_h, best_n, deadline - 1)
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
