def project_exact(h, n):
    h = np.array(h, dtype=float, copy=True).flatten()
    if n != h.size:
        h = np.resize(h, n)
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * float(n)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15: break
            coef = diff / s
            if coef >= 1.0:
                h += room
            else:
                h += coef * room
        else:
            s = h.sum()
            if s <= 1e-15: break
            coef = -diff / s
            if coef >= 1.0:
                h *= 0.0
            else:
                h *= (1.0 - coef)
    diff = target - h.sum()
    for _ in range(100):
        if diff == 0.0: break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            step = min(diff, 1.0 - h[idx])
        else:
            idx = int(np.argmax(h))
            step = min(-diff, h[idx])
        if step == 0.0: break
        h[idx] += step
        diff = target - h.sum()
    return h

    h=np.array(h,float,copy=True).reshape(-1)

import numpy as np
import time
from scipy.optimize import minimize

def project_exact(h, n):
    h = np.array(h, dtype=float, copy=True).reshape(-1)
    if h.size != n:
        h = np.resize(h, n)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            if d <= s:
                h += d * room / s
            else:
                h += room
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            if -d <= s:
                h += d * h / s
            else:
                h -= h
    d = target - h.sum()
    for _ in range(100):
        if d == 0.0:
            break
        if d > 0.0:
            i = int(np.argmax(1.0 - h))

def project_exact(h, n):
    h = np.array(h, dtype=float, copy=True).reshape(-1)
    if h.size != n:
        h = np.resize(h, n)
    h = np.clip(h, 0.0, 1.0)

    for _ in range(200):

        if abs(d) < 1e-15:

        if d > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            if d <= s:

            else:
                h += room
        else:
            s = h.sum()
            if s <= 1e-15:
                break

    h = np.asarray(h, float).reshape(-1)

    xd = np.linspace(0.0, 2.0, n_dst)
    return project_exact(np.interp(xd, xs, h), n_dst)

    h = np.asarray(h, float).reshape(-1)
    if h.size != n:
        h = np.resize(h, n)
    klen = max(3, n // 16)
    if klen % 2 == 0:
        klen += 1
    r = klen // 2
    k = np.linspace(-r, r, klen)
    sig = max(0.5, r * 0.5)
    kern = np.exp(-0.5 * (k / sig) ** 2)
    kern /= kern.sum()
    return project_exact(np.convolve(h, kern, mode='same'), n)


def smooth_random(n, seed):
    rng = np.random.RandomState(int(seed))
    x = np.linspace(0.0, 2.0, n)
    f = np.zeros(n, float)
    for _ in range(4):
        freq = int(rng.randint(1, 6))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.normal(0.0, 1.0))
        f += amp * np.cos(np.pi * freq * x + ph)
        f += amp * np.sin(np.pi * freq * x + 0.37 * ph)
    return project_exact(0.5 + 0.5 * np.tanh(0.7 * f), n)


def periodic_start(n, p, seed):
    if p < 1:
        p = 2
    rng = np.random.RandomState(int(seed))
    bits = np.zeros(p, float)
    ones = p // 2
    if ones > 0:
        bits[rng.choice(p, ones, replace=False)] = 1.0
    base = np.tile(bits, n // p + 1)[:n]
    score = base + 0.02 * rng.random(n)
    order = np.argsort(score)[::-1]
    target = 0.5 * n
    k = int(target)
    hb = np.zeros(n, float)
    if k > 0:
        hb[order[:k]] = 1.0
    rem = target - k
    if rem > 1e-12 and k < n:
        hb[order[k]] = rem
    return project_exact(hb, n)


def grad_from_weights(h, n, weights):
    L = 2 * n - 1
    W = np.asarray(weights, float)
    if W.shape[0] != L:
        W = np.resize(W, L)
    A = np.convolve(1.0 - h, W[::-1], mode='full')
    B = np.convolve(h, W, mode='full')
    return A[n - 1:2 * n - 1] - B[n - 1:2 * n - 1]


def soft_eval(x, n, gamma):
    h = np.array(x, dtype=float, copy=True).reshape(-1)
    if h.size != n:
        h = np.resize(h, n)
    if not np.all(np.isfinite(h)):
        h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    g = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(g))
    if gamma <= 0.0:
        act = g >= m - 1e-8
        w = np.zeros_like(g)
        cnt = max(1, int(act.sum()))
        w[act] = 1.0 / cnt
        return m, grad_from_weights(h, n, w)
    sh = gamma * (g - m)
    sm = float(np.max(sh))
    sh = sh - sm
    e = np.exp(sh)
    se = float(np.sum(e))
    if not np.isfinite(se) or se <= 0.0:
        act = g >= m - 1e-8
        w = np.zeros_like(g)
        cnt = max(1, int(act.sum()))
        w[act] = 1.0 / cnt
        return m, grad_from_weights(h, n, w)
    val = m + (sm + float(np.log(se))) / gamma
    w = e / se
    return val, grad_from_weights(h, n, w)


def lin_eq_fun(x, n):
    return float(np.sum(x) - 0.5 * n)


def lin_eq_jac(x, n):
    return np.ones(x.size, dtype=float)


def slsqp_fun(x, n, gamma, dx):
    val, _ = soft_eval(x, n, gamma)
    return float(val * dx)


def slsqp_jac(x, n, gamma, dx):
    _, grad = soft_eval(x, n, gamma)
    return grad * dx


def slsqp_smooth(h, n, gamma, maxiter, deadline):
    h = project_exact(np.array(h, dtype=float, copy=True), n)
    dx = 2.0 / max(1, n)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': lin_eq_fun, 'jac': lin_eq_jac, 'args': (n,)}]
    try:
        res = minimize(
            slsqp_fun, h, jac=slsqp_jac, args=(n, gamma, dx),
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': int(max(1, maxiter)), 'ftol': 1e-14, 'disp': False}
        )
    except Exception:
        return h
    if res is None:
        return h
    x = np.asarray(res.x, float).reshape(-1)
    if x.size != n or not np.all(np.isfinite(x)):
        return h
    return project_exact(x, n)


def coord_line_search(h, n, max_sweeps=150, deadline=None):
    h = project_exact(np.array(h, dtype=float, copy=True), n)
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    L = 2 * n - 1
    p0 = n - 1
    act_tol = 1e-8
    for _ in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        clr = np.minimum(h, 1.0 - h)
        r = int(np.argmax(clr))
        if clr[r] < 1e-7:
            break
        m = float(c.max())
        improved = False
        for i in range(n):
            if i == r:
                continue
            if deadline is not None and time.time() > deadline:
                break
            old_i = float(h[i])
            old_r = float(h[r])
            lo = max(0.0, old_r + old_i - 1.0)
            hi = min(1.0, old_r + old_i)
            if hi - lo < 1e-13:
                continue
            bi = np.zeros(L, float)
            bi[n - 1 - i:2 * n - 1 - i] += 1.0 - h
            bi[i:i + n] -= h[::-1]
            br = np.zeros(L, float)
            br[n - 1 - r:2 * n - 1 - r] += 1.0 - h
            br[r:r + n] -= h[::-1]
            d = bi - br
            active = c >= m - act_tol
            if active.any():
                da = d[active]
                if float(da.min()) <= 1e-11 and float(da.max()) >= -1e-11:
                    continue
            blo, bhi = lo, hi
            mid_best = old_i
            line_best = m
            for __ in range(20):
                mid = 0.5 * (blo + bhi)
                v = c + d * (mid - old_i)
                mv = float(np.max(v))
                if mv < line_best:
                    line_best = mv
                    mid_best = mid
                av = v >= mv - 1e-12
                mn = float(d[av].min())
                mx = float(d[av].max())
                if mx > 1e-12:
                    bhi = mid
                elif mn < -1e-12:
                    blo = mid
                else:
                    mid_best = mid
                    break
            cands = [old_i, lo, hi, mid_best, blo, bhi]
            for dd in (1e-7, 1e-5, 5e-4, 0.002, 0.01):
                xa = old_i + dd
                if lo <= xa <= hi:
                    cands.append(xa)
                xb = old_i - dd
                if lo <= xb <= hi:
                    cands.append(xb)
            old_i0 = old_i * (1.0 - old_i)
            old_r0 = old_r * (1.0 - old_r)
            oldc0 = float(c[p0])
            p1 = p0 + (r - i)
            p2 = p0 - (r - i)
            best_val = m
            best_x = old_i
            best_rnew = old_r
            for x in cands:
                x = float(x)
                if x < lo - 1e-15 or x > hi + 1e-15:
                    continue
                if x < lo:
                    x = lo
                elif x > hi:
                    x = hi
                if abs(x - old_i) < 1e-15:
                    continue
                dt = x - old_i
                rn = old_r - dt
                v = c + d * dt
                qq = dt * dt
                v[p1] += qq
                v[p2] += qq
                v[p0] = oldc0 - old_i0 - old_r0 + x * (1.0 - x) + rn * (1.0 - rn)
                mv = float(np.max(v))
                if mv < best_val - 1e-14:
                    best_val = mv
                    best_x = x
                    best_rnew = rn
            if abs(best_x - old_i) > 1e-15:
                dt = best_x - old_i
                rnew = best_rnew
                qq = dt * dt
                c += d * dt
                c[p1] += qq
                c[p2] += qq
                c[p0] = oldc0 - old_i0 - old_r0 + best_x * (1.0 - best_x) + rnew * (1.0 - rnew)
                h[i] = best_x
                h[r] = rnew
                m = best_val
                improved = True
        if not improved:
            break
        h = project_exact(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, float(c.max() * (2.0 / max(1, n)))


def improve_candidate(h, n, alloc, deadline, seed=0):
    t = time.time()
    h = project_exact(h, n)
    best = compute_c5(h, n)
    if alloc <= 0.0 or time.time() >= deadline:
        return h, best
    gammas = (50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 8000000, 30000000)
    slq_end = min(deadline, t + 0.55 * alloc)
    if alloc < 20:
        gg = gammas[:4]
    elif alloc < 60:
        gg = gammas[:6]
    else:
        gg = gammas
    num = max(1, len(gg))
    gamma_time = max(0.0, (slq_end - time.time()) / num)
    for gamma in gg:
        if time.time() >= slq_end - 0.5:
            break
        maxiter = int(max(10, min(120, gamma_time * (600.0 / max(1, n)) + 12)))
        t_end = min(slq_end, time.time() + max(1.0, gamma_time * 1.2))
        h2 = slsqp_smooth(h, n, gamma, maxiter, t_end)
        c2 = compute_c5(h2, n)
        if c2 < best:
            h = h2
            best = c2
    coord_end = min(deadline, t + 0.90 * alloc)
    h, c3 = coord_line_search(h, n, max_sweeps=120, deadline=coord_end)
    h = project_exact(h, n)
    c3 = compute_c5(h, n)
    if c3 < best:
        best = c3
    final_end = min(deadline, t + alloc)
    if time.time() < final_end - 2.0:
        remf = final_end - time.time()
        maxiter = int(max(10, min(100, remf * (600.0 / max(1, n)) + 10)))
        h2 = slsqp_smooth(h, n, 50000000, maxiter, min(final_end, time.time() + max(1.0, remf * 0.4)))
        c4 = compute_c5(h2, n)
        if c4 < best:
            h = h2
            best = c4
        h, cc = coord_line_search(h, n, max_sweeps=40, deadline=min(final_end, time.time() + max(1.0, remf * 0.5)))
        h = project_exact(h, n)
        cc = compute_c5(h, n)
        if cc < best:
            best = cc
    return h, best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(30.0, float(budget_s) - 30.0)
    hard = t0 + total

    try:
        par = np.array(initial_h_values, dtype=float).reshape(-1)
        n_base = int(par.size)
    except Exception:
        par = None
        n_base = 400

    if par is None or n_base < 20:
        n_base = 400
        par = smooth_random(n_base, seed)
    if n_base > 900:
        par = np.interp(np.linspace(0.0, 2.0, 800), np.linspace(0.0, 2.0, n_base), par)
        n_base = 800

    h0 = project_exact(par, n_base)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n_base)
    best_n = n_base

    tA = t0 + 0.42 * total
    starts = [
        h0.copy(),
        make_binary_start(h0, n_base, seed + 1),
        smooth_parent(h0, n_base),
        smooth_random(n_base, seed + 2),
        periodic_start(n_base, 5, seed + 11),
        periodic_start(n_base, 7, seed + 13),
    ]
    for idx, hs in enumerate(starts):
        rem = tA - time.time()
        if rem < 8.0:
            break
        alloc = min(rem / max(1, len(starts) - idx), 0.5 * rem)
        if alloc < 5.0:
            break
        h2, c2 = improve_candidate(hs, n_base, alloc, hard, seed + 20 + idx)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
            best_n = n_base

    nF = min(900, max(n_base + 100, 500))
    tB = t0 + 0.72 * total
    if nF > n_base and time.time() < tB:
        hF = interp_to_n(best_h, best_n, nF)
        h2, c2 = improve_candidate(hF, nF, min(tB - time.time(), total * 0.25), hard, seed + 50)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
            best_n = nF

    tC = t0 + 0.90 * total
    nC = max(50, n_base // 2)
    cands = [interp_to_n(best_h, best_n, nC)]
    for pp in (3, 4, 5, 6, 7, 8, 10, 12, 16, 20, 25, 32):
        if pp < nC:
            cands.append(periodic_start(nC, pp, seed + 100 + pp))
    cands.append(smooth_random(nC, seed + 200))

    for idx, hc in enumerate(cands):
        rem = tC - time.time()
        if rem < 6.0:
            break
        alloc = min(8.0, rem / max(1, len(cands) - idx))
        if alloc < 3.0:
            alloc = rem * 0.8 if rem > 4.0 else 0.0
        if alloc < 2.0:
            break
        hc2, cc2 = improve_candidate(hc, nC, alloc, hard, seed + 300 + idx)
        if cc2 < best_c + 0.03 and nC < best_n:
            hn = interp_to_n(hc2, nC, best_n)
            aref = min(18.0, rem * 0.5)
            if aref > 5.0 and time.time() < tC:
                hn2, cf = improve_candidate(hn, best_n, aref, hard, seed + 400 + idx)
                if cf < best_c:
                    best_h = hn2.copy()
                    best_c = cf
                    best_n = best_n

    if time.time() < hard - 8.0:
        h2, c2 = improve_candidate(best_h, best_n, min(hard - time.time() - 2.0, total * 0.12), hard, seed + 500)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    rem = hard - time.time()
    if rem > 25.0:
        rng = np.random.RandomState(seed + 777)
        hn = project_exact(best_h + 0.01 * rng.normal(size=best_n), best_n)
        h2, c2 = improve_candidate(hn, best_n, min(rem - 3.0, total * 0.08), hard, seed + 600)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    rem = hard - time.time()
    if rem > 18.0:
        rng = np.random.RandomState(seed + 888)
        hn = project_exact(best_h + 0.003 * rng.normal(size=best_n), best_n)
        h2, c2 = improve_candidate(hn, best_n, min(rem - 2.0, total * 0.05), hard, seed + 700)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    best_h = project_exact(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
