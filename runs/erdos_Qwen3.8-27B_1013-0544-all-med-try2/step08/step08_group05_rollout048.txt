def compute_delta(h, i, j, n):
    """Compute delta correlation vector for swapping h[i]:0->1, h[j]:1->0"""
    delta = np.zeros(2*n-1)
    # h_new = h with h[i]=1, h[j]=0 (they already are in h for a binary function)
    # Actually for binary h, h[i]=0 and h[j]=1 before swap.
    
    # Part 1: adding h[i]=1 contributes (1-h_new[i+l]) to lag l
    # i+l ranges from i-(n-1) to i+(n-1), but must be in [0,n-1]
    # l ranges from -i to n-1-i
    l_start = max(0, 0 - i)  # l such that i+l >= 0, i.e., l >= -i
    l_end = min(2*n-2, n-1-i + n-1)  # l such that i+l <= n-1, i.e., l <= n-1-i
    # In the correlation array, index = l + (n-1)
    idx_start = max(0, -i + n-1)
    idx_end = min(2*n-2, (n-1-i) + n-1)
    lags = np.arange(idx_start - (n-1), idx_end - (n-1) + 1)
    positions = i + lags  # these are in [0, n-1]
    # (1 - h_new[positions]): h_new differs from h at i and j
    vals = 1 - h[positions]
    vals[positions == i] = 0  # h_new[i] = 1, so 1-h_new[i] = 0
    vals[positions == j] = 1  # h_new[j] = 0, so 1-h_new[j] = 1
    delta[idx_start:idx_end+1] += vals
    
    # Part 2: removing h[j]=1 subtracts (1-h_new[j+l]) from lag l
    idx_start2 = max(0, -j + n-1)
    idx_end2 = min(2*n-2, (n-1-j) + n-1)
    lags2 = np.arange(idx_start2 - (n-1), idx_end2 - (n-1) + 1)
    positions2 = j + lags2
    vals2 = 1 - h[positions2]
    vals2[positions2 == i] = 0  # 1-h_new[i] = 0
    vals2[positions2 == j] = 1  # 1-h_new[j] = 1
    delta[idx_start2:idx_end2+1] -= vals2
    
    # Part 3: (1-h) at position i changes from 1 to 0
    # This affects lag l where m+l = i, i.e., m = i-l
    # m must be in [0,n-1], so l = i-m, l ranges from i-(n-1) to i
    idx_start3 = max(0, i-(n-1) + n-1)  # = max(0, i)
    idx_end3 = min(2*n-2, i + n-1)
    lags3 = np.arange(idx_start3 - (n-1), idx_end3 - (n-1) + 1)
    positions3 = i - lags3  # = m
    vals3 = h[positions3]
    vals3[positions3 == i] = 0  # h_new[i] = 1... wait

    # Since m = i-l and m is in [0,n-1], m could equal j (if i-l=j, i.e., l=i-j)
    vals3[positions3 == j] = 1  # h_old[j] = 1 = h_new[j]? No, h_new[j]=0!
    # Wait, we're computing the change in C, which involves h_old values in this term.
    # The formula is: -h_old[i-l] for the (1-h[i]) change
    # h_old[i-l] = h[i-l] (since i-l ≠ i in general, and i-l = j is possible)
    # If i-l = j: h_old[j] = 1
    # If i-l ≠ j and i-l ≠ i: h_old[i-l] = h[i-l]
    # i-l = i means l=0, and h_old[i] = 0
    vals3[positions3 == i] = 0
    delta[idx_start3:idx_end3+1] -= vals3
    
    # Part 4: (1-h) at position j changes from 0 to 1
    # This affects lag l where m+l = j, i.e., m = j-l
