def fix_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / float(mask.sum())
        else:
            if diff > 0.0:
                i = int(np.argmax(1.0 - h))
                h[i] = min(1.0, h[i] + diff)
            else:
                i = int(np.argmax(h))
                h[i] = max(0.0, h[i] + diff)
        h = np.clip(h, 0.0, 1.0)
    return h


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    if x.size == 0:
        return x
    lo = -float(np.max(x))
    hi = 1.0 - float(np.min(x))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    return fix_sum(y, target)


def corr_full(a, b):
    n = len(a)
    if n <= 600 or sig_correlate is None:
        return np.correlate(a, b, mode="full")
    return sig_correlate(a, b, mode="full", method="fft")


def count_vec(h, n):
    return corr_full(h, 1.0 - h)


def max_count(h, n):
    return float(np.max(count_vec(h, n)))


def final_c5_direct(h, n):
    try:
        c = np.correlate(np.asarray(h, dtype=float), 1.0 - np.asarray(h, dtype=float), mode="full")
        return float(np.max(c) * (2.0 / float(n)))
    except Exception:
        return float(np.max(corr_full(h, 1.0 - h)) * (2.0 / float(n)))


def lag_gradient_counts(h, n, k):
    g = np.zeros(n, dtype=float)
    if k >= 0:
        nv = n - k
        if nv > 0:
            g[:nv] = 1.0 - h[k:]
            g[k:] -= h[:nv]
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            g[kk:] = 1.0 - h[:nv]
            g[:nv] -= h[kk:]
    return g


def select_active(C, M, max_lags, tol):
    max_lags = int(max_lags)
    if max_lags <= 0:
        return np.empty(0, dtype=int)
    if C.size <= max_lags:
        return np.arange(C.size, dtype=int)
    near = np.flatnonzero(C >= M - tol)
    top = np.argsort(C)[-max_lags:]
    idx = np.union1d(near, top)
    if idx.size > max_lags:
        order = np.argsort(C[idx])
        idx = idx[order[-max_lags:]]
    return idx.astype(int)


def lp_descent(h, C, idx, M, n):
    if linprog is None:
        return None
    m = len(idx)
    if m == 0 or n <= 1:
        return None
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    A = np.empty((m, n + 1), dtype=float)
    b = np.empty(m, dtype=float)
    for r in range(m):
        f = int(idx[r])
        k = f - (n - 1)
        A[r, :n] = lag_gradient_counts(h, n, k)
        A[r, n] = 1.0
        br = M - float(C[f])
        if br < 0.0:
            br = 0.0
        b[r] = br

    c = np.zeros(n + 1, dtype=float)
    c[n] = -1.0
    Aeq = np.zeros((1, n + 1), dtype=float)
    Aeq[0, :n] = 1.0
    beq = np.array([0.0], dtype=float)

    lower = -h
    upper = 1.0 - h
    bounds = list(zip(lower, upper))
    bounds.append((None, float(M) + 1.0))

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            A_eq=Aeq,
            b_eq=beq,
            bounds=bounds,
            method="highs"
        )
    except Exception:
        return None
    if res.status != 0:
        return None
    d = np.asarray(res.x[:n], dtype=float).copy()
    if not np.all(np.isfinite(d)):
        return None
    rho = float(res.x[n])
    if rho <= 1e-7:
        return None
    d = d - d.mean()
    amax = float(np.max(np.abs(d)))
    if amax <= 1e-12:
        return None
    return d / amax


def line_search_counts(h, d, n, current_M=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    d = np.asarray(d, dtype=float)
    target = 0.5 * float(n)
    if d.size != n or n <= 1:
        m = max_count(h, n) if current_M is None else current_M
        return h, m
    if not np.all(np.isfinite(d)):
        m = max_count(h, n) if current_M is None else current_M
        return h, m

    amax = float(np.max(np.abs(d)))
    if amax <= 1e-15:
        m = max_count(h, n) if current_M is None else current_M
        return h, m
    d = d / amax
    d = d - d.mean()
    amax = float(np.max(np.abs(d)))
    if amax <= 1e-15:
        m = max_count(h, n) if current_M is None else current_M
        return h, m
    d = d / amax

    pos = d > 1e-12
    neg = d < -1e-12
    alpha_max = 1.0
    if pos.any():
        v = (1.0 - h[pos]) / d[pos]
        if v.size:
            alpha_max = min(alpha_max, float(np.min(v)))
    if neg.any():
        v = (-h[neg]) / (-d[neg])
        if v.size:
            alpha_max = min(alpha_max, float(np.min(v)))
    if not math.isfinite(alpha_max) or alpha_max <= 1e-12:
        m = max_count(h, n) if current_M is None else current_M
        return h, m
    alpha_max = min(alpha_max, 1.0)

    Cbase = count_vec(h, n)
    Mbase = float(np.max(Cbase))
    if current_M is None:
        current_M = Mbase

    lin = corr_full(d, 1.0 - h) - corr_full(h, d)
    quad = -corr_full(d, d)

    fr = np.unique(np.concatenate((np.linspace(0.0, 1.0, 64), np.logspace(-4, 0.0, 24))))
    al = fr * alpha_max
    vals = Cbase[None, :] + al[:, None] * lin[None, :] + (al * al)[:, None] * quad[None, :]
    mx = np.max(vals, axis=1)
    i = int(np.argmin(mx))
    alpha = float(al[i])
    best_pred = float(mx[i])

    if alpha <= 1e-12 or best_pred >= current_M - 1e-8:
        return h.copy(), current_M

    h2 = h + alpha * d
    h2 = fix_sum(h2, target)
    m2 = max_count(h2, n)
    if m2 >= current_M:
        return h.copy(), current_M
    return h2, m2


def accept_if_improvement(h_new, m_new, h, best_h, best_m, cur_m, tol=1e-8):
    if m_new < cur_m - tol:
        h = h_new
        cur_m = m_new
        if m_new < best_m - tol:
            best_h = h_new.copy()
            best_m = m_new
        return h, best_h, best_m, cur_m, True
    return h, best_h, best_m, cur_m, False


def smooth_gradient_counts(h, C, n, alpha, top_m=80):
    M = float(C.max())
    m = min(int(top_m), int(C.size))
    idx = np.argsort(C)[-m:]
    vals = C[idx].astype(float)
    sh = alpha * (vals - M)
    sh -= float(np.max(sh))
    w = np.exp(sh)
    s = float(np.sum(w))
    g = np.zeros(n, dtype=float)
    if s <= 0.0:
        return g
    w = w / s
    for i in range(m):
        wi = float(w[i])
        if wi > 1e-10:
            k = int(idx[i]) - (n - 1)
            g += wi * lag_gradient_counts(h, n, k)
    return g


def random_direction(n, rng):
    d = rng.standard_normal(n)
    d = d - d.mean()
    amax = float(np.max(np.abs(d)))
    if amax > 0.0:
        d = d / amax
    return d


def random_pair_direction(n, rng):
    d = np.zeros(n, dtype=float)
    a = int(rng.randint(n))
    b = int(rng.randint(n))
    while b == a and n > 1:
        b = int(rng.randint(n))
    d[a] = 1.0
    d[b] = -1.0
    return d


def make_pair_perturb(h, n, rng, alpha_lo=0.25, alpha_hi=1.0):
    d = random_pair_direction(n, rng)
    pos = np.flatnonzero(d > 0.0)
    neg = np.flatnonzero(d < 0.0)
    m = max_count(h, n)
    if pos.size == 0 or neg.size == 0:
        return h.copy(), m
    i = int(pos[0])
    j = int(neg[0])
    room = min(1.0 - float(h[i]), float(h[j]))
    if room <= 1e-12:
        return h.copy(), m
    alpha = rng.uniform(alpha_lo, alpha_hi) * room
    h2 = h + alpha * d
    h2 = fix_sum(h2, 0.5 * float(n))
    return h2, max_count(h2, n)


def make_random_h(n, rng, kind=0):
    target = 0.5 * float(n)
    if n <= 1:
        return project_box_sum(np.array([0.5], dtype=float), target)
    if kind == 1:
        h = np.zeros(n, dtype=float)
        half = n // 2
        phi = (1.0 + math.sqrt(5.0)) / 2.0
        if half > 0:
            pos = np.floor(np.arange(half) * phi).astype(int) % n
            h[pos] = 1.0
        rem = target - float(h.sum())
        if abs(rem) > 1e-12:
            idx = int(rng.randint(n))
            h[idx] = min(1.0, max(0.0, float(h[idx]) + rem))
        return fix_sum(h, target)
    if kind == 2:
        h = (np.arange(n) % 2 == 0).astype(float)
        return fix_sum(h, target)
    h = np.zeros(n, dtype=float)
    half = n // 2
    if half > 0:
        idx = rng.choice(n, half, replace=False)
        h[idx] = 1.0
    rem = target - float(h.sum())
    if abs(rem) > 1e-12:
        idx = int(rng.randint(n))
        h[idx] = min(1.0, max(0.0, float(h[idx]) + rem))
    return fix_sum(h, target)


def resample_h(h, n_old, n_new):
    if n_new <= 1:
        return project_box_sum(np.array([0.5], dtype=float), 0.5 * float(n_new))
    h = np.asarray(h, dtype=float)
    if n_new == n_old:
        return project_box_sum(h, 0.5 * float(n_new))
    if n_new < n_old:
        idx = (np.arange(n_new) * float(n_old) / float(n_new)).astype(int)
        idx2 = ((np.arange(n_new) + 1) * float(n_old) / float(n_new)).astype(int)
        idx = np.clip(idx, 0, n_old - 1)
        idx2 = np.clip(idx2, 1, n_old)
        y = np.empty(n_new, dtype=float)
        for i in range(n_new):
            a = int(idx[i])
            b = int(idx2[i])
            if b <= a:
                y[i] = float(h[a])
            else:
                y[i] = float(np.mean(h[a:b]))
        return project_box_sum(y, 0.5 * float(n_new))
    idx = (np.arange(n_new) * float(n_old) / float(n_new)).astype(int)
    idx = np.clip(idx, 0, n_old - 1)
    y = h[idx]
    return project_box_sum(y, 0.5 * float(n_new))


def anti_shift_symmetry(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * float(n)
    m = n // 2
    h2 = np.zeros(n, dtype=float)
    if m > 0:
        left = 0.5 * (h[:m] + (1.0 - h[m:2 * m]))
        h2[:m] = left
        h2[m:2 * m] = 1.0 - left
    if n % 2 == 1:
        h2[-1] = 0.5
    return project_box_sum(h2, target)


def anti_reverse_symmetry(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    h2 = 0.5 * (h + (1.0 - h[::-1]))
    return project_box_sum(h2, 0.5 * float(n))

        rem = job_deadline - time.time()
        if rem < 25.0:
            break
        tw = 0.0
        for j in range(i, len(jobs)):
            tw += float(jobs[j][2])
        if tw <= 0.0:
            break
        alloc = rem * (float(jobs[i][2]) / tw)
        if alloc < 12.0:
            continue
        h0, n0, _ = jobs[i]
        t_end = min(time.time() + alloc, job_deadline)
        try:
            hr, mc, nr = run_job(h0, n0, t_end, seed=int(seed) + 97 * i + 13)
        except Exception:
            continue
        if hr is None or nr <= 0:
            continue
        hr = np.asarray(hr, dtype=float)
        dval = final_c5_direct(hr, nr)
        if dval < best_c - 1e-12:
            best_h = hr
            best_n = int(nr)
            best_c = float(dval)

    if best_h is not None and time.time() < deadline - 8.0:
        try:
            hr, mc, nr = run_job(best_h, best_n, deadline - 1.0, seed=int(seed) + 100003)
        except Exception:
            hr, nr = best_h, best_n
        if hr is not None and nr > 0:
            dval = final_c5_direct(hr, nr)
            if dval < best_c - 1e-12:
                best_h = hr
                best_n = int(nr)
                best_c = float(dval)

    best_n = int(best_n)
    best_h = project_box_sum(best_h, 0.5 * float(best_n))
    best_c = final_c5_direct(best_h, best_n)
    return best_h, float(best_c), best_n
