def sa_binary_search(h_init, n, deadline, seed=42):
    dx = 2.0 / n
    h = np.round(h_init)
    h = np.clip(h, 0, 1)
    # Ensure exact count
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target:
        # Randomly flip excess ones to zero
        excess = rng.choice(ones, len(ones)-target, replace=False)
        h[excess] = 0
    elif len(ones) < target:
        deficit = rng.choice(zeros, target-len(ones), replace=False)
        h[deficit] = 1
    
    corr = np.correlate(h, 1-h, mode='full')
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    T = 0.01  # initial temperature
    T_min = 1e-7
    cooling = 0.999
    
    while time.time() < deadline:
        if T < T_min:
            T = T_min
        # Pick random zero and one to swap
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        i = rng.choice(zeros)
        j = rng.choice(ones)
        
        # Compute delta
        delta = compute_swap_delta(h, i, j, n)  # O(n)
        new_corr = corr + delta
        new_c = new_corr.max() * dx
        
        # Accept or reject
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            h[i], h[j] = 1, 0
            corr = new_corr
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        
        T *= cooling
    
    return best_h, best_c

    """Convert to binary with exactly n//2 ones."""
    target = n // 2
    h = np.round(h)
    h = np.clip(h, 0, 1)

        # Pick random swap
        i = rng.choice(zeros)
        j = rng.choice(ones)

        h[j] = 0.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')

            h[j] = 1.0
