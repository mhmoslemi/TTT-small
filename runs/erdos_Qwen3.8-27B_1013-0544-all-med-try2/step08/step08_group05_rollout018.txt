    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        ones_arr = np.where(h == 1)[0]
        zeros_arr = np.where(h == 0)[0]
        nz = len(zeros_arr)
        no = len(ones_arr)
        
        if nz == 0 or no == 0:
            break
        
        # Precompute P for all zeros (net change in corr from setting h[i]=1)
        P = np.zeros((nz, 2*n-1))
        for ii in range(nz):
            i = zeros_arr[ii]
            im = i + m_range
            imn = i - m_range
            msk = (im >= 0) & (im < n)
            P[ii, msk] = 1.0 - h[im[msk]]
            msk2 = (imn >= 0) & (imn < n)
            P[ii, msk2] -= h[imn[msk2]]
        
        # Precompute Q for all ones (net change in corr from setting h[j]=0)
        Q = np.zeros((no, 2*n-1))
        for jj in range(no):
            j = ones_arr[jj]
            jm = j + m_range
            jmn = j - m_range
            msk = (jm >= 0) & (jm < n)
            Q[jj, msk] = -(1.0 - h[jm[msk]])
            msk2 = (jmn >= 0) & (jmn < n)
            Q[jj, msk2] += h[jmn[msk2]]
        
        # Find best swap
        best_val = cur_c
        best_ii = -1
        best_jj = -1
        
        for ii in range(nz):
            base = corr + P[ii]
            max_vals = (base[None, :] + Q).max(axis=1)
            top_k = min(8, no)
            top_idx = np.argsort(max_vals)[:top_k]
            
            for jj in top_idx:
                j = ones_arr[jj]
                i = zeros_arr[ii]
                candidate = base + Q[jj]
                candidate = candidate.copy()
                candidate[n-1] -= 2.0
                k1 = n - 1 + (j - i)
                k2 = n - 1 + (i - j)
                if 0 <= k1 < 2*n-1:
                    candidate[k1] += 1.0
                if 0 <= k2 < 2*n-1:
                    candidate[k2] += 1.0
                true_max = candidate.max()
                
                if true_max < best_val - 1e-10:
                    best_val = true_max
                    best_ii = ii
                    best_jj = jj
        
        # Accept/reject
        accepted = False
        if best_ii >= 0:
            if best_val < cur_c - 1e-10:
                accepted = True
            elif T > T_min:
                delta = cur_c - best_val
                if delta <= 0:
                    accepted = True
                else:
                    accepted = (rng.random() < np.exp(-delta / T))
        
        if accepted:
            i = zeros_arr[best_ii]
            j = ones_arr[best_jj]
            h[i], h[j] = 1.0, 0.0
            corr += P[best_ii] + Q[best_jj]
            corr[n-1] -= 2.0
            k1 = n - 1 + (j - i)
            k2 = n - 1 + (i - j)
            if 0 <= k1 < 2*n-1:
                corr[k1] += 1.0
            if 0 <= k2 < 2*n-1:
                corr[k2] += 1.0
            cur_c = best_val
            n_stalls = 0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            n_stalls += 1
        
        T = max(T * cooling, T_min)
        # Reheat if stuck
        if n_stalls > 500:
            T = T_init * 0.5
            n_stalls = 0
    
    return best_h, best_c * dx

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
