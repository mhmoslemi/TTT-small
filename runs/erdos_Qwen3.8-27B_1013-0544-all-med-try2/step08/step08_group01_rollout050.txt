        h400 = project_feasible(h400, 400)
        alloc = min(120, (deadline - time.time() - 150) * 0.5)
        if alloc > 30:
            h4, c4 = lbfgs_phase(h400, 400, time.time() + alloc * 0.7,
                                 [1000, 10000, 50000, 200000])
            if time.time() < deadline - 60:
                h4, c4 = guided_local_search(h4, 400, min(time.time() + 40, deadline - 30), candidate_k=10)
            if c4 < best_c:
                best_c = c4
                best_h = h4
                best_n = 400

    # Phase 5: Try n=300
    if time.time() < deadline - 180:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x300 = np.linspace(0, 2, 300, endpoint=False)
        h300 = np.interp(x300, x_old, best_h)
        h300 = project_feasible(h300, 300)
        alloc = min(80, (deadline - time.time() - 100) * 0.6)
        if alloc > 25:
            h3, c3 = lbfgs_phase(h300, 300, time.time() + alloc * 0.6,
                                 [1000, 10000, 50000, 200000])
            if c3 < best_c:
                best_c = c3
                best_h = h3
                best_n = 300

    # Phase 6: Restart at best_n with perturbation
    if time.time() < deadline - 100:
        h_pert = best_h + rng.randn(best_n) * 0.03
        h_pert = project_feasible(h_pert, best_n)
        alloc = min(50, (deadline - time.time() - 50) * 0.5)
        if alloc > 15:
            h_p, c_p = lbfgs_phase(h_pert, best_n, time.time() + alloc * 0.7,
                                   [5000, 50000, 200000])
            if c_p < best_c:
                best_c = c_p
                best_h = h_p

    # Phase 7: Fourier restart
    if time.time() < deadline - 70:
        h_f = fourier_start(best_n, rng)
        alloc = min(40, (deadline - time.time() - 30) * 0.5)
        if alloc > 12:
            h_fr, c_fr = lbfgs_phase(h_f, best_n, time.time() + alloc * 0.7,
                                     [2000, 20000, 100000])
            if c_fr < best_c:
                best_c = c_fr
                best_h = h_fr

    # Phase 8: Final L-BFGS polish on best
    if time.time() < deadline - 30:
        t8 = min(time.time() + 30, deadline - 10)
        best_h, best_c = lbfgs_phase(best_h, best_n, t8, [1000000, 5000000])

    # Phase 9: Final guided local search
    if time.time() < deadline - 5:
        best_h, best_c = guided_local_search(best_h, best_n, deadline - 2, candidate_k=8)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
