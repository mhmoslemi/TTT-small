            if tot < 1e-15:
                break
            h += (diff / tot) * h

    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

    m = c.max()

    mx = sh.max()

            nv = n - kk

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def lbfgs_opt(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [200, 1000, 5000, 25000, 100000, 500000, 2000000, 10000000, 50000000, 200000000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            return smooth_max_obj(x, n, alpha)
        def grad(x):
            return smooth_max_grad(x, n, alpha)
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-19,
                                   'gtol': 1e-18, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def slsqp_opt(h, n, deadline, alpha=1000000):
    def _obj(x):
        return smooth_max_obj(x, n, alpha)
    def _jac(x):
        return smooth_max_grad(x, n, alpha)
    def _eq(x):
        return float(x.sum() - n / 2.0)
    def _eq_j(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eq_j}]
    try:
        res = minimize(_obj, h, jac=_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 5000, 'ftol': 1e-19})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < compute_c5(h, n):
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def projected_grad_scan(h, n, deadline, alpha=500000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step_base = 0.02
    no_improve = 0
    while time.time() < deadline and no_improve < 30:
        g = smooth_max_grad(best_h, n, alpha)
        g = g - g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm * step_base
        improved = False
        for ls in range(25):
            trial = best_h - g
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_feasible(trial, n)
            c_trial = compute_c5(trial, n)
            if c_trial < best_c - 1e-16:
                best_h = trial
                best_c = c_trial
                improved = True
                break
            g *= 0.5
        if not improved:
            no_improve += 1
            alpha = max(alpha * 0.6, 100)
            step_base *= 0.5
        else:
            no_improve = 0
    return best_h, best_c

def fine_pair_search(h, n, deadline, top_k=60):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
    for d in steps:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            g = smooth_max_grad(best_h, n, 20000)
            dec_order = np.argsort(g)[:top_k]
            inc_order = np.argsort(-g)[:top_k]
            for i in dec_order:
                if time.time() > deadline:
                    return best_h, best_c
                if best_h[i] + d > 1.0:
                    continue
                for j in inc_order:
                    if i == j:
                        continue
                    if best_h[j] - d < 0.0:
                        continue
                    best_h[i] += d
                    best_h[j] -= d
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved_any = True
                    else:
                        best_h[i] -= d
                        best_h[j] += d
    return best_h, best_c
