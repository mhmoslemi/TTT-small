def _nextpow(n):
    N = 1
    while N < n:
        N *= 2
    return N

def _dfft(h, n, N=None):
    if N is None:
        N = _nextpow(2 * n - 1)
    xp = np.zeros(N)
    xp[:n] = h
    yp = np.zeros(N)
    yn = 1.0 - h[::-1]
    yp[:n] = yn
    c = np.fft.irfft(np.fft.rfft(xp) * np.fft.rfft(yp), N)
    m = n - 1
    o = np.empty(2 * n - 1)
    o[m:] = c[:m + 1][::-1]
    o[:m] = c[m + 1:2 * n - 1][::-1]
    return o

def _dcore(h, n, N=None):
    if FTT_OK and n > 130:
        return _dfft(h, n, N)
    return np.correlate(h, 1.0 - h, mode='full')

def c5_eval(h, n, N=None):
    d = _dcore(h, n, N)
    return float(d.max()) * 2.0 / n

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lsmax(h, n, alpha, N=None):
    d = _dcore(h, n, N)
    q = d * (2.0 / n)
    mm = float(q.max())
    e = np.exp(alpha * (q - mm))
    ws = float(e.sum())
    return q, mm, e / ws, np.log(ws)

def _lamp_g(h, n, alpha):
    m = n - 1
    sx = np.concatenate((np.zeros(1), np.cumsum(h)))
    p1 = np.zeros(n)
    p1[:m] = (m - np.arange(m, dtype=float)) - sx[1:n]
    return p1 - sx[:-1]

def _obj_p(x, n, alpha, mu, N):
    q, mm, e, lw = _lsmax(x, n, alpha, N)
    return float(mm + lw / alpha + mu * (x.sum() - n / 2.0) ** 2)

def _grad_p(x, n, alpha, mu, N):
    q, mm, e, lw = _lsmax(x, n, alpha, N)
    g = alpha * (2.0 / n) * e * _lamp_g(x, n, 0.0)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

import time
import numpy as np
from scipy.optimize import minimize


def _nextpow(m):
    N = 1
    while N < m:
        N *= 2
    return N


def _dfft(h, n, N=None):
    if N is None:
        N = _nextpow(2 * n - 1)
    xp = np.zeros(N)
    xp[:n] = h
    yp = np.zeros(N)
    yp[:n] = 1.0 - h[::-1]
    c = np.fft.irfft(np.fft.rfft(xp) * np.fft.rfft(yp), N)
    m = n - 1
    o = np.empty(2 * n - 1)
    o[m:] = c[:m + 1][::-1]
    o[:m] = c[m + 1:2 * n - 1][::-1]
    return o


def _fft_selftest():
    rng = np.random.RandomState(3)
    for n in (64, 200, 333):
        h = np.clip(rng.rand(n) * 1.4 - 0.2, 0.0, 1.0)
        a = np.correlate(h, 1.0 - h, mode='full')
        b = _dfft(h, n)
        if float(np.abs(a - b).max()) > 1e-9:
            return False
    return True


_FTF = _fft_selftest()


def _dcore(h, n, N=None):
    if _FTF and n > 130:
        return _dfft(h, n, N)
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of(h, n, N=None):
    return float(np.max(_dcore(h, n, N))) * (2.0 / n)


def _lai(o, X, i, n):
    o[n - 1 + i:2 * n - 1] += 1.0 - X[i:n]
    o[n:n + i] -= X[:i][::-1]
    X[i] = 1.0


def _lah(o, X, i, n):
    o[n - 1 + i:2 * n - 1] -= 1.0 - X[i:n]
    o[n:n + i] += X[:i][::-1]
    X[i] = 0.0


def _app(o, X, a, b, n):
    _lah(o, X, b, n)
    _lai(o, X, a, n)


def _rev(o, X, a, b, n):
    _lah(o, X, a, n)
    _lai(o, X, b, n)

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    tg = n / 2.0
    for _ in range(300):
        df = tg - h.sum()
        if abs(df) < 1e-12:
            break
        if df > 0:
            rm = 1.0 - h
            tt = rm.sum()
            if tt < 1e-12:
                break
            h += (df / tt) * rm
        else:
            tt = h.sum()
            if tt < 1e-12:
                break
            h += (df / tt) * h
    return np.clip(h, 0.0, 1.0)


def _lsmax(h, n, alpha, N=None):
    d = _dcore(h, n, N)
    q = d * (2.0 / n)
    mm = float(q.max())
    e = np.exp(alpha * (q - mm))
    ws = float(e.sum())
    return mm, e / ws


def _lamgg(h, n):
    m = n - 1
    sx = np.concatenate((np.zeros(1), np.cumsum(h)))
    p1 = np.zeros(n)
    p1[:m] = (m - np.arange(m, dtype=float)) - sx[1:n]
    return p1 - sx[:-1]


def _obj_p(x, n, alpha, mu, N):
    mm, e = _lsmax(x, n, alpha, N)
    return float(mm + np.log(e.sum()) / alpha + mu * (x.sum() - n / 2.0) ** 2)


def _grad_p(x, n, alpha, mu, N):
    mm, e = _lsmax(x, n, alpha, N)
    s = x.sum() - n / 2.0
    g = alpha * (2.0 / n) * e * _lamgg(x, n)
    g += 2.0 * mu * s
    return g

          (1e4, 1.5e3), (4e4, 5e3), (1.5e5, 1.6e4), (5e5, 5e4)]
    for al, mu in cf:
        left = deadline - time.time()
        if left < 10:

                          options={'maxiter': mit, 'maxfun': mit * 30,
                                   'ftol': 1e-20, 'gtol': 1e-16, 'maxls': 40})
            h2 = project_feasible(rs.x, n)
            c2 = c5_of(h2, n, NN)
            if c2 < best:
                h, best = h2, c2
        except Exception:
            pass
    return h, best


def _slq_eq(x, n):
    return float(x.sum() - n / 2.0)


def _slq_jac(x, n):
    return np.ones(n)


def _obj_slq(x, n, alpha, N):
    mm, e = _lsmax(x, n, alpha, N)
    return float(mm + np.log(e.sum()) / alpha)


def slsqp_polish(h, n, deadline, alpha=8e4):
    try:
        NN = _nextpow(2 * n - 1) if _FTF else None
        mit = int(np.clip(14.0 * (deadline - time.time()) * (500.0 / n), 150, 900))
        cons = [{'type': 'eq', 'fun': _slq_eq, 'jac': _slq_jac, 'args': (n,)}]
        h = project_feasible(h, n)
        rs = minimize(_obj_slq, h, args=(n, alpha, NN), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n, constraints=cons,
                      options={'maxiter': mit, 'ftol': 1e-16})
        h2 = project_feasible(rs.x, n)
        return h2, c5_of(h2, n)
    except Exception:
        return h, c5_of(h, n)


def grad_search(h, n, deadline):
    best = c5_of(h, n)
    NN = _nextpow(2 * n - 1) if _FTF else None
    for al, st in ((900.0, 2e-3), (5000.0, 8e-4), (3e4, 3e-4)):
        while time.time() < deadline - 2:
            _, e = _lsmax(h, n, al, NN)
            cv = e * _lamgg(h, n)
            od = np.argsort(cv)
            imp = False
            for ii in range(n):
                if time.time() > deadline:
                    return h, best
                i = od[ii]
                lo = max(0, ii - 40)
                hi = min(n, ii + 41)
                for jj in range(lo, hi):
                    j = od[jj]
                    if i == j or cv[j] - cv[i] < 1e-10:
                        continue
                    for sg in (1.0, -1.0):
                        vi = h[i] + sg * st
                        vj = h[j] - sg * st
                        if vi < 0.0 or vi > 1.0 or vj < 0.0 or vj > 1.0:
                            continue
                        oi = h[i]
                        oj = h[j]
                        h[i] = vi
                        h[j] = vj
                        c2 = c5_of(h, n, NN)
                        if c2 < best - 1e-15:
                            best, imp = c2, True
                        else:
                            h[i] = oi
                            h[j] = oj
            if not imp:
                break
    return h, best


def pair_polish(h, n, deadline, radius=None):
    if radius is None:
        radius = int(np.clip(n / 320.0, 4, 12))
    h = project_feasible(h, n)
    best = c5_of(h, n)
    NN = _nextpow(2 * n - 1) if _FTF else None
    for dd in (2e-3, 7e-4, 2e-4, 7e-5, 2e-5, 8e-6):
        imp = True
        while imp and time.time() < deadline:
            imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    for sg in (1.0, -1.0):
                        vi = h[i] + sg * dd
                        vj = h[j] - sg * dd
                        if vi < 0.0 or vi > 1.0 or vj < 0.0 or vj > 1.0:
                            continue
                        oi = h[i]
                        oj = h[j]
                        h[i] = vi
                        h[j] = vj
                        c2 = c5_of(h, n, NN)
                        if c2 < best - 1e-15:
                            best, imp = c2, True
                        else:
                            h[i] = oi
                            h[j] = oj
    return h, best


def anneal_bin(X, n, deadline, seed, T0, decay):
    rng = np.random.RandomState(seed)
    X = np.array(X, dtype=float).copy()
    o = _dcore(X, n)
    dm0 = float(o.max())
    dmB = dm0
    XB = X.copy()
    T = T0
    while time.time() < deadline:
        a = rng.randint(n)
        while X[a] > 0.5:
            a = rng.randint(n)
        b = -1
        if rng.rand() < 0.5:
            for _ in range(7):
                w = 1 + rng.randint(0, 46)
                cand = a + w
                if cand >= n:
                    cand = a - w
                if 0 <= cand < n and cand != a and X[cand] < 0.5:
                    b = cand
                    break
        if b < 0:
            b = rng.randint(n)
            while X[b] < 0.5:
                b = rng.randint(n)
        if b == a:
            continue
        _app(o, X, a, b, n)
        if not AIK:
            o = _dcore(X, n)
        dm1 = float(o.max())
        if dm1 <= dm0 or (T > 1e-9 and rng.rand() < np.exp(-(dm1 - dm0) / T)):
            dm0 = dm1
            if dm1 < dmB:
                dmB = dm1
                XB = X.copy()
        else:
            _rev(o, X, a, b, n)
            if not AIK:
                o = _dcore(X, n)
        T *= decay
    return XB, dmB


def _snap_bin(h, n, rng):
    X = (h >= 0.5).astype(float)
    ng = n // 2
    c1 = int(X.sum())
    if c1 > ng:
        ix = np.flatnonzero(X)
        X[ix[rng.permutation(ix.size)][:c1 - ng]] = 0.0
    elif c1 < ng:
        ix = np.flatnonzero(1.0 - X)
        X[ix[rng.permutation(ix.size)][:ng - c1]] = 1.0
    return X


def start_golden(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    pos = np.mod(np.floor(np.arange(half) * phi).astype(np.int64), n)
    h[pos] = 1.0
    return project_feasible(h, n)


def start_randbin(n, rng):
    X = (rng.rand(n) < 0.5).astype(float)
    return _snap_bin(X, n, rng)


def start_gauss(n, rng):
    h = np.full(n, 0.5)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    for f in rng.permutation(np.arange(1, 9))[:6]:
        h += rng.uniform(0.10, 0.30) * np.cos(np.pi * f * x + rng.uniform(0.0, 6.28318))
    return project_feasible(h, n)


def start_tilde(n, rng):
    P = int(rng.randint(11, 43))
    L = max(1, min(P - 1, int(round(n / 2.0 / P))))
    h = np.zeros(n)
    h[np.arange(n) % P < L] = 1.0
    return project_feasible(h, n)


def resample(h, n0, n1):
    if n0 == n1:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n1) + 0.5) * (2.0 / n1)
    return project_feasible(np.interp(x1, x0, h), n1)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + float(budget_s) - 3.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None
    if h0 is not None and h0.size > 0:
        best_h = project_feasible(h0, h0.size)
        best_n = h0.size
    else:
        best_n = 1000
        best_h = start_randbin(best_n, rng)
    best_c = c5_of(best_h, best_n)

    for nt, al in ((1000, 75), (2000, 115), (4000, 150)):
        if dl - time.time() < al + 640:
            break
        h1 = project_feasible(resample(best_h, best_n, nt) + rng.randn(nt) * 0.015, nt)
        h2, c2 = desc_con(h1, nt, min(time.time() + al, dl - 630))
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, nt

    bn = int(min(max(best_n, 800), 2400))
    if time.time() < dl - 310:
        alloc = min(240.0, dl - time.time() - 295)
        if alloc > 60:
            Xs = (_snap_bin(resample(best_h, best_n, bn), bn, rng),
                  start_randbin(bn, rng),
                  _snap_bin(start_tilde(bn, rng), bn, rng))
            for k, X0 in enumerate(Xs):
                fr = (0.5, 0.3, 0.2)[k]
                z = min(time.time() + alloc * fr, dl - 300)
                if z <= time.time():
                    break
                T0a = (2.0, 6.0, 4.0)[k]
                dc = (0.9998, 0.9997, 0.99975)[k]
                Xb, dmb = anneal_bin(X0, bn, z, seed + 101 + k, T0a, dc)
                cb = float(dmb) * 2.0 / bn
                if cb < best_c:
                    best_h, best_c, best_n = Xb.copy(), cb, bn
            if best_n == bn and time.time() < dl - 250:
                z = min(time.time() + 50.0, dl - 240)
                if z > time.time():
                    h2, c2 = desc_con(best_h, bn, z)
                    if c2 < best_c:
                        best_h, best_c = h2, c2

    if time.time() < dl - 170:
        dn = 1000 if best_n > 1500 else bn
        pool = (start_randbin(dn, rng), start_golden(dn), start_gauss(dn, rng))
        ks = 0
        for hs in pool:
            ks += 1
            if time.time() > dl - 165:
                break
            z = min(time.time() + 32.0, dl - 160)
            if z <= time.time():
                break
            h2, c2 = desc_con(project_feasible(hs, dn), dn, z)
            if c2 < best_c:
                best_h, best_c, best_n = h2, c2, dn
                if time.time() < dl - 150:
                    Xb, dmb = anneal_bin(_snap_bin(h2, dn, rng), dn,
                                         min(time.time() + 45.0, dl - 145),
                                         seed + 500 + ks, 3.0, 0.9997)
                    cb = float(dmb) * 2.0 / dn
                    if cb < best_c:
                        best_h, best_c, best_n = Xb.copy(), cb, dn

    if best_n <= 1400 and time.time() < dl - 70:
        h2, c2 = slsqp_polish(best_h, best_n, min(time.time() + 40.0, dl - 60), 8e4)
        if c2 < best_c:
            best_h, best_c = h2, c2
    if best_n <= 2300 and time.time() < dl - 80:
        best_h, best_c = grad_search(project_feasible(best_h, best_n), best_n,
                                     min(time.time() + 55.0, dl - 65))
    if time.time() < dl - 40:
        best_h, best_c = pair_polish(best_h, best_n, min(time.time() + 60.0, dl - 25))

    n2 = min(2 * best_n, 10000)
    if n2 > best_n and time.time() < dl - 12:
        h2 = resample(best_h, best_n, n2)
        c2 = c5_of(h2, n2)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, n2
            if time.time() < dl - 8:
                best_h, best_c = pair_polish(best_h, best_n, dl - 6, radius=5)

    best_h = project_feasible(best_h, best_n)
    best_c = c5_of(best_h, best_n)
    return best_h, float(best_c), int(best_n)
