    for _ in range(500):

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    return m + (smax + np.log(np.sum(np.exp(shifted - smax)))) / alpha

def softmax_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    w = np.exp(shifted - smax)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wm = w[m_idx]
        if wm < 1e-15:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def _slsqp_f(x, n, alpha):
    return softmax_obj(x, n, alpha)

def _slsqp_g(x, n, alpha):
    return softmax_grad(x, n, alpha)

def _eq_f(x, n):
    return x.sum() - n / 2.0

def _eq_g(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, deadline, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_f, 'jac': _eq_g, 'args': (n,)}]
    try:
        res = minimize(_slsqp_f, h, args=(n, alpha), jac=_slsqp_g,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def lbfgs_optimize(h, n, deadline, alphas, iters_per=150):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        def obj(x):
            return softmax_obj(x, n, alpha) + 5000.0 * (x.sum() - n / 2.0) ** 2
        def jac(x):
            return softmax_grad(x, n, alpha) + 10000.0 * (x.sum() - n / 2.0)
        try:
            res = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': iters_per, 'ftol': 1e-16, 'gtol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def binary_swap_search(h, n, deadline, rng):
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    while h_bin.sum() > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.randint(len(idx))]] = 0
    while h_bin.sum() < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.randint(len(idx))]] = 1
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    best_corr_max = np.max(np.correlate(h_bin, 1.0 - h_bin, mode='full'))
    no_improve = 0
    max_stall = 3000
    while time.time() < deadline and no_improve < max_stall:
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i] = 0
        h_bin[j] = 1
        c_new = np.max(np.correlate(h_bin, 1.0 - h_bin, mode='full'))
        if c_new < best_corr_max - 1e-12:
            best_corr_max = c_new
            best_c = c_new * (2.0 / n)
            best_h = h_bin.copy()
            no_improve = 0
        else:
            h_bin[i] = 1
            h_bin[j] = 0
            no_improve += 1
    return best_h, best_c

def directed_swap_search(h, n, deadline, rng):
    """Focus swaps on reducing the maximum correlation entries."""
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    while h_bin.sum() > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.randint(len(idx))]] = 0
    while h_bin.sum() < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.randint(len(idx))]] = 1
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    no_improve = 0
    max_stall = 5000
    while time.time() < deadline and no_improve < max_stall:
        corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        max_val = corr.max()
        # Find k indices at (or near) the maximum
        thresh = max_val * 0.99
        hot ks = np.where(corr >= thresh)[0]
        best_swap = None
        best_new_max = max_val
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        n_try = min(500, len(ones) * len(zeros))
        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0
            h_bin[j] = 1
            c_new = np.correlate(h_bin, 1.0 - h_bin, mode='full')
            nm = c_new.max()
            h_bin[i] = 1
            h_bin[j] = 0
            if nm < best_new_max - 1e-12:
                best_new_max = nm
                best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0
            h_bin[j] = 1
            best_c = best_new_max * (2.0 / n)
            best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    best_n = n

    # Phase 1: Quick L-BFGS-B polish (60s)
    t1 = min(t0 + 60, deadline - 500)
    best_h, best_c = lbfgs_optimize(best_h, best_n, t1,
                                     [50, 200, 1000, 5000, 20000])

    # Phase 2: SLSQP escalation (220s)
    for alpha, iters, dur in [(5000, 2000, 60), (50000, 3000, 80), (200000, 4000, 80)]:
        if time.time() > deadline - 300:
            break
        t_end = min(time.time() + dur, deadline - 290)
        h2, c2 = slsqp_optimize(best_h, best_n, t_end, alpha, iters)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Directed binary search (180s)
    if time.time() < deadline - 250:
        t3 = min(time.time() + 160, deadline - 240)
        h3, c3 = directed_swap_search(best_h, best_n, t3, rng)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: SLSQP after binary (180s)
    for alpha, iters, dur in [(100000, 5000, 90), (500000, 5000, 90)]:
        if time.time() > deadline - 200:
            break
        t4 = min(time.time() + dur, deadline - 190)
        h4, c4 = slsqp_optimize(best_h, best_n, t4, alpha, iters)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Try other n values (150s)
    for n_try in [400, 600, 750]:
        if time.time() > deadline - 100:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n_try)
        alloc = min(45, (deadline - time.time() - 80) * 0.45)
        if alloc < 15:
            continue
        h_t, c_t = slsqp_optimize(h_int, n_try, time.time() + alloc, 100000, 3000)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Phase 6: Final SLSQP very high alpha (60s)
    if time.time() < deadline - 15:
        t6 = min(time.time() + 55, deadline - 12)
        h6, c6 = slsqp_optimize(best_h, best_n, t6, 1000000, 8000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final binary polish
    if time.time() < deadline - 6:
        h7, c7 = binary_swap_search(best_h, best_n, deadline - 4, rng)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Final: L-BFGS-B micro-polish if time remains
    if time.time() < deadline - 3:
        h8, c8 = lbfgs_optimize(best_h, best_n, deadline - 2, [500000], 100)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
