            best_c = c.max()
            best_h = h_b.copy()
        else:
            break
    
    return best_h, best_c

- Δc from h[j]: 0→1: B[j] - A[j]
- Total Δc for swap (i,j): A[i] - B[i] + B[j] - A[j]

def binary_hill_climb_fast(h, n, deadline, max_rounds=200):
    h_b = h.copy()
    target = n // 2
    # Ensure exactly target ones
    while h_b.sum() > target:
        i = np.argmax(h_b)  # or random
        h_b[i] = 0
    while h_b.sum() < target:
        i = np.argmin(h_b)
        h_b[i] = 1
    
    c = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    best_c = float(c.max())
    best_h = h_b.copy()
    
    k = np.arange(2*n-1) - (n-1)  # lags
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        
        # Precompute A[p, m] = h_b[p+k[m]] if valid, 0 otherwise
        # B[p, m] = (1-h_b[p-k[m]]) if valid, 0 otherwise
        A = np.zeros((n, 2*n-1))
        B = np.zeros((n, 2*n-1))
        
        for p in range(n):
            pk = p + k  # p + lag for each m
            idx1 = (pk >= 0) & (pk < n)
            A[p, idx1] = h_b[pk[idx1]]
            
            pk2 = p - k  # p - lag for each m
            idx2 = (pk2 >= 0) & (pk2 < n)
            B[p, idx2] = 1 - h_b[pk2[idx2]]
        
        best_new_c = best_c
        best_i, best_j = -1, -1
        
        for i in ones:
            base = c + A[i] - B[i]
            # For each j in zeros: new max = max(base + B[j] - A[j])
            tmp = base[None, :] + B[zeros] - A[zeros]  # (n_zeros, 2n-1)
            new_maxes = tmp.max(axis=1)
            j_best = new_maxes.argmin()
            if new_maxes[j_best] < best_new_c - 1e-10:
                best_new_c = float(new_maxes[j_best])
                best_i, best_j = i, zeros[j_best]
        
        if best_i >= 0:
            h_b[best_i] = 0
            h_b[best_j] = 1
            c = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
            best_c = float(c.max())
            best_h = h_b.copy()
        else:
            break
    
    return best_h, best_c
