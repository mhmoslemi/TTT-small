import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h *= (target / tot) if tot > 0 else 1.0
            h = np.minimum(h, 1.0)
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return float(m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:

            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += -wm * x[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += -wm * x[kk:]
    return grad

def targeted_pushdown(h, n, deadline, top_k=5, steps=2000):
    """Directly push down the top-k worst shifts"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for step in range(steps):
        if time.time() > deadline:
            break
        c = np.correlate(best_h, 1 - best_h, mode='full')
        # Get top-k indices
        top_idx = np.argsort(c)[-top_k:]
        # Compute combined gradient of top-k shifts
        grad = np.zeros(n)
        for idx in top_idx:
            grad += grad_single_shift(best_h, n, idx)

def binary_hill_climb(h, n, deadline):

    h_bin = (h > 0.5).astype(float)
    target = n // 2
    # Fix to exactly target ones
    ones = np.where(h_bin == 1)[0]
    if len(ones) > target:
        # Remove worst ones
        for i in ones[len(ones)-target:]:
            h_bin[i] = 0
    elif len(ones) < target:
        zeros = np.where(h_bin == 0)[0]
        for i in zeros[:target-len(ones)]:
            h_bin[i] = 1
    
    best_c = compute_c5(h_bin, n)
    
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        best_move = None
        best_mc = best_c
        
        # Try all flips (for n=500, that's 250*250 = 62500, each O(n) = 500)
        # Total: ~31M ops per sweep, should take ~1s
        for i in ones:

                    best_move = (i, j)
        
        if best_move is not None:
            h_bin[best_move[0]] = 0
            h_bin[best_move[1]] = 1
            best_c = best_mc
            improved = True
    
    return h_bin, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = (1.0 - h)[mask]
            frac = min(1.0, diff / room.sum())
            h[mask] += frac * room
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            frac = min(1.0, -diff / h[mask].sum())
            h[mask] *= (1.0 - frac)
    return np.clip(h, 0.0, 1.0)

def smax_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    a = alpha * (c - m)
    am = a.max()
    return float(m + (am + np.log(np.sum(np.exp(a - am)))) / alpha)

def smax_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    a = alpha * (c - m)
    am = a.max()
    w = np.exp(a - am)
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        k = idx - (n - 1)
        wm = w[idx]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += -wm * x[:nv]
        else:
            t = -k
            nv = n - t
            grad[t:] += wm * (1.0 - x[:nv])
            grad[:nv] += -wm * x[t:]
    return grad

def obj_fn(x, n, alpha, mu):
    return smax_val(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def grad_fn(x, n, alpha, mu):
    g = smax_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def optimize_smooth(h, n, deadline, alphas, mu=1.0):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        rem = deadline - time.time()
        if rem < 5:
            break
        iters = min(500, max(20, int(rem * 4)))
        try:
            r = minimize(obj_fn, h, jac=grad_fn, args=(n, alpha, mu),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-15})
            h_new = project_feasible(r.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pushdown(h, n, deadline, k=5, lr0=0.05):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr0
    stalled = 0
