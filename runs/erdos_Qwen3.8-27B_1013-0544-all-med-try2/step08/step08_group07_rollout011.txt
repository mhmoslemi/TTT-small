import numpy as np

from scipy.optimize import minimize

def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def compute_c_int(h):
    n = len(h)
    L = 2 * n - 1
    a = h.astype(np.int32)
    b = 1 - a
    c = np.zeros(L, dtype=np.int16)
    c[n - 1] = int((a * b).sum())
    for k in range(1, n):
        c[n - 1 + k] = int((a[:n - k] * b[k:]).sum())
        c[n - 1 - k] = int((a[k:] * b[:n - k]).sum())
    return c

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            s = (1.0 - h).sum()
            if s < 1e-14:
                break
            h += d / s * (1.0 - h)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += d / s * h
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:

                add = min(1.0 - h[idx], d)
            else:
                add = max(-h[idx], d)
            if add != 0:
                h[idx] += add
                d -= add
            if abs(d) < 1e-12:

def to_binary(arr, n):
    if arr is None:
        arr = np.full(n, 0.5)
    arr = np.asarray(arr, dtype=float)
    if arr.size != n:

    idx = np.argsort(-arr)
    h = np.zeros(n, dtype=np.int8)
    h[idx[:n // 2]] = 1
    return h

def random_binary(n, rng):
    h = np.zeros(n, dtype=np.int8)

def structured_runs(n, rng):

    q = float(rng.randint(0, n))
    v = (np.arange(n).astype(float) * p + q) % float(n)
    out.append(to_binary(v + 0.05 * rng.randn(n), n))
    return out

def block_perturb(h, rng):
    n = len(h)
    h = h.astype(np.int8).copy()
    length = rng.randint(max(3, n // 10), n + 1)
    l = rng.randint(0, n - length)
    r = l + length
    seg = h[l:r]
    if rng.rand() < 0.5:
        h[l:r] = seg[::-1]
    else:
        d = rng.randint(1, seg.size)
        h[l:r] = np.concatenate((seg[d:], seg[:d]))
    return h

def apply_exchange(h, c, i, j):
    n = len(h)
    L = 2 * n - 1
    omh = (1 - h).astype(np.int16)
    hrev = h[::-1].astype(np.int16)
    c = c.astype(np.int16, copy=True)
    c[n - 1 - i:n - 1 - i + n] += omh
    c[n - 1 - j:n - 1 - j + n] -= omh
    c[i:i + n] -= hrev
    c[j:j + n] += hrev
    c[n - 1] -= 2
    c[n - 1 + i - j] += 1
    c[n - 1 + j - i] += 1
    h = h.astype(np.int8, copy=True)
    h[i] = 1
    h[j] = 0
    return h, c

def full_best_move(h, c, ones, zeros):
    n = len(h)
    L = 2 * n - 1
    m0 = len(zeros)
    m1 = len(ones)
    if m0 == 0 or m1 == 0:
        return None
    omh = (1 - h).astype(np.int16)
    hrev = h[::-1].astype(np.int16)
    c16 = c.astype(np.int16)
    base1 = np.zeros((m1, L), dtype=np.int16)
    for r, j in enumerate(ones):
        j = int(j)
        base1[r, n - 1 - j:n - 1 - j + n] -= omh
        base1[r, j:j + n] += hrev
    base0 = np.zeros((m0, L), dtype=np.int16)
    for r, i in enumerate(zeros):
        i = int(i)
        base0[r, n - 1 - i:n - 1 - i + n] += omh
        base0[r, i:i + n] -= hrev
        base0[r, n - 1] -= 2
    cur_m = int(c16.max())
    cur_w = int(np.count_nonzero(c16 == cur_m))
    best_m = cur_m
    best_w = cur_w
    best_i = -1
    best_j = -1
    rows = np.arange(m1, dtype=np.int32)
    ones_arr = np.asarray(ones, dtype=np.int32)
    for r0, i in enumerate(zeros):
        i = int(i)
        mat = np.empty((m1, L), dtype=np.int16)
        mat[:] = c16
        mat += base0[r0]
        mat += base1
        p = n - 1 + (i - ones_arr)
        q = n - 1 + (ones_arr - i)
        mat[rows, p] += 1
        mat[rows, q] += 1
        mx = mat.max(axis=1)
        mn = int(mx.min())
        if mn < best_m:
            idxs = np.flatnonzero(mx == mn)
            if idxs.size:
                r1 = int(idxs[0])
                best_m = mn
                best_i = i
                best_j = int(ones_arr[r1])
                best_w = int(np.count_nonzero(mat[r1] == best_m))
        elif mn == best_m and best_m == cur_m:
            idxs = np.flatnonzero(mx == best_m)
            if idxs.size:
                wts = np.count_nonzero(mat[idxs] == best_m, axis=1)
                pos = int(np.argmin(wts))
                if int(wts[pos]) < best_w:
                    r1 = int(idxs[pos])
                    best_w = int(wts[pos])
                    best_i = i
                    best_j = int(ones_arr[r1])
    if best_i < 0:
        return None
    if best_m < cur_m or (best_m == cur_m and best_w < cur_w):
        return best_i, best_j, best_m, best_w
    return None

def anneal_binary(h_init, c_init, timeout, rng):
    t0 = time.time()
    t_end = t0 + max(0.25, float(timeout))
    h = h_init.astype(np.int8).copy()
    c = c_init.astype(np.int16).copy()
    ones = list(np.flatnonzero(h).astype(int))
    zeros = list(np.flatnonzero(h == 0).astype(int))
    m = int(c.max())
    w = int(np.count_nonzero(c == m))
    best_score = m * 1000000 + w * 1000
    best_h = h.copy()
    steps = max(50, int(max(0.0, timeout) * 3000))
    T = 2.0
    for st in range(steps):
        if time.time() > t_end:
            break
        T = max(0.03, 2.0 * (0.95 ** (st / 200.0)))
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        nh, nc = apply_exchange(h, c, i, j)
        nm = int(nc.max())
        nw = int(np.count_nonzero(nc == nm))
        dm = nm - m
        if dm < 0 or (dm == 0 and nw < w) or (dm > 0 and rng.rand() < np.exp(-T * float(dm))):
            h, c = nh, nc
            m, w = nm, nw
            ones.remove(j)
            zeros.remove(i)
            ones.append(i)
            zeros.append(j)
            sc = m * 1000000 + w * 1000
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    return best_h

def optimize_binary(h_init, n, timeout, rng):
    t0 = time.time()
    t_end = t0 + max(1.0, float(timeout))
    h = np.asarray(h_init, dtype=np.int8).copy()
    if int(h.astype(np.int32).sum()) != n // 2:
        h = to_binary(h.astype(float), n)
    c = compute_c_int(h)
    ones = list(np.flatnonzero(h).astype(int))
    zeros = list(np.flatnonzero(h == 0).astype(int))
    m = int(c.max())
    w = int(np.count_nonzero(c == m))
    best_m = m
    best_w = w
    best_score = m * 1000000 + w * 1000
    best_h = h.copy()
    restarts = 0
    while time.time() < t_end - 0.5:
        min_time = 1.8 if n > 360 else 0.6
        if time.time() > t_end - min_time:
            break
        move = full_best_move(h, c, ones, zeros)
        if move is not None:
            i, j, nm, nw = move
            i = int(i)
            j = int(j)
            h, c = apply_exchange(h, c, i, j)
            ones.remove(j)
            zeros.remove(i)
            ones.append(i)
            zeros.append(j)
            m, w = int(nm), int(nw)
            sc = m * 1000000 + w * 1000
            if sc < best_score:
                best_score = sc
                best_m = m
                best_w = w
                best_h = h.copy()
            restarts = 0
        else:
            restarts += 1
            if restarts > 8 or time.time() > t_end - 1.0:
                break
            if restarts % 2 == 0:
                at = min(4.0, max(0.5, t_end - time.time() - 1.0))
                h2 = anneal_binary(h, c, at, rng)
            else:
                h2 = block_perturb(h, rng)
            if int(h2.astype(np.int32).sum()) != n // 2:
                h2 = to_binary(h2.astype(float), n)
            c2 = compute_c_int(h2)
            h, c = h2, c2
            ones = list(np.flatnonzero(h).astype(int))
            zeros = list(np.flatnonzero(h == 0).astype(int))
            m = int(c.max())
            w = int(np.count_nonzero(c == m))
            sc = m * 1000000 + w * 1000
            if sc < best_score:
                best_score = sc
                best_m = m
                best_w = w
                best_h = h.copy()
    return best_h, float(best_m) * 2.0 / float(n)

def soft_objective(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode="full") * dx
    gm = float(np.max(g))
    sh = alpha * (g - gm)
    sm = float(np.max(sh))
    return float(gm + (sm + np.log(np.exp(sh - sm).sum())) / alpha)

def sum_constraint(x, n):
    return float(np.sum(x) - n / 2.0)

def slsqp_refine(h, n, alpha, maxiter):
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": sum_constraint, "args": (n,)}]
        res = minimize(soft_objective, h, method="SLSQP",
                       bounds=bounds, constraints=cons, args=(n, alpha),
                       options={"maxiter": maxiter, "ftol": 1e-12, "disp": 0})
        if res is not None and hasattr(res, "x"):
            x = np.asarray(res.x, dtype=float)
            if x.size == n and np.all(np.isfinite(x)):
                return project_feasible(x, n)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s)) - 10.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
    except Exception:
        h0 = np.full(400, 0.5)
    n0 = int(h0.size)
    if n0 <= 0:
        h0 = np.full(400, 0.5)
        n0 = 400
    if n0 % 2:
        n0 += 1
        h0 = np.interp(np.linspace(0, 2, n0), np.linspace(0, 2, int(h0.size)), h0)
    best_h = h0.copy()
    best_n = n0
    best_c = compute_c5(best_h, best_n)
    for amp in (0.0, 0.02, 0.05, 0.1):
        hb = to_binary(h0 + amp * rng.randn(n0), n0)
        cb = int(compute_c_int(hb).max()) * 2.0 / float(n0)
        if cb < best_c:
            best_c = cb
            best_h = hb.astype(float)
    ns = set()
    for f in (0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0, 1.1):
        x = int(n0 * f)
        for y in (x - 2, x, x + 2):
            if 120 <= y <= 450 and y % 2 == 0:
                ns.add(y)
    for y in (120, 160, 200, 240, 280, 320, 360, 400, 440):
        if 120 <= y <= 450:
            ns.add(y)
    ns = sorted(ns)
    if not ns:
        ns = [n0 if n0 % 2 == 0 else n0 + 1]
    for n_try in ns:
        rem = deadline - time.time()
        if rem < 40.0:
            break
        if best_c <= 0.3790:
            break
        chunk = min(150.0, max(12.0, rem * 0.30))
        starts = starts_for_n(n_try, best_h, best_n, rng)
        if chunk < 35.0:
            starts = starts[:2]
        per = chunk / max(1, len(starts))
        for hs in starts:
            rem = deadline - time.time()
            if rem < 25.0:
                break
            opt_t = min(per, rem - 20.0)
            if opt_t < 1.0:
                break
            hopt, cval = optimize_binary(hs, n_try, opt_t, rng)
            if cval < best_c:
                best_c = cval
                best_h = hopt.astype(float)
                best_n = n_try
    if time.time() < deadline - 25.0:
        if best_n > 450:
            hn = 400
            hpol = np.interp(np.linspace(0, 2, hn),
                             np.linspace(0, 2, best_n),
                             np.asarray(best_h, dtype=float))
        else:
            hn = best_n
            hpol = np.asarray(best_h, dtype=float).copy()
        if hpol.size != hn:
            hpol = np.interp(np.linspace(0, 2, hn),
                             np.linspace(0, 2, int(hpol.size)),
                             hpol)
        hpol = project_feasible(hpol, hn)
        c_now = compute_c5(hpol, hn)
        if c_now < best_c:
            best_c = c_now
            best_h = hpol
        for alpha in (100, 1000, 10000, 100000, 1000000):
            rem = deadline - time.time()
            if rem < 8.0:
                break
            maxit = int(min(80, max(20, rem * 15)))
            h2 = slsqp_refine(hpol, hn, alpha, maxit)
            c2 = compute_c5(h2, hn)
            if c2 < c_now + 1e-12:
                hpol = h2
                c_now = c2
            if c2 < best_c:
                best_c = c2
                best_h = hpol
    bh = np.asarray(best_h, dtype=float)
    if bh.size != best_n:
        bh = np.interp(np.linspace(0, 2, best_n),
                       np.linspace(0, 2, int(bh.size)),
                       bh)
    if np.allclose(bh, np.round(bh), atol=1e-7):
        bf = np.round(bh).astype(float)
        if int(bf.sum()) != best_n // 2:
            bf = to_binary(bf, best_n).astype(float)
        best_h = bf
    else:
        best_h = project_feasible(bh, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
