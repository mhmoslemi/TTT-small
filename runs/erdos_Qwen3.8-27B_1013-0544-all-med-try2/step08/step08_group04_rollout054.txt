import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_max_val(c, alpha):
    s = alpha * (c - c.max())
    mx = s.max()
    return float(c.max() + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)

def _smooth_max_weights(c, alpha):
    s = alpha * (c - c.max())
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    return w

def smooth_max_obj(x, n, alpha, top_k):
    c = compute_c5_vec(x, n)
    if top_k and top_k < len(c):
        c_use = np.sort(c)[-top_k:]
    else:
        c_use = c
    return _smooth_max_val(c_use, alpha)

def smooth_max_grad(x, n, alpha, top_k, mu):
    c = compute_c5_vec(x, n)
    if top_k and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_use = c[idx]
    else:
        idx = np.arange(len(c))
        c_use = c
    w = _smooth_max_weights(c_use, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for ii, m in enumerate(idx):
        wi = w[ii]
        if wi < 1e-15:
            continue
        k = int(m) - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def pen_objective(x, n, alpha, top_k, mu):
    val = smooth_max_obj(x, n, alpha, top_k)
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def pen_jacobian(x, n, alpha, top_k, mu):
    return smooth_max_grad(x, n, alpha, top_k, mu)

def eq_fun(x, n):
    return float(x.sum() - n / 2.0)

def eq_jac(x, n):
    return np.ones(n)

def run_lbfgs(h, n, deadline, configs):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha, top_k, mu, maxiter in configs:
        if time.time() > deadline:
            break
        try:
            res = minimize(pen_objective, best_h, jac=pen_jacobian,
                          args=(n, alpha, top_k, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': maxiter * 12})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_slsqp(h, n, deadline, alpha, top_k, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, jac=lambda x, n2, a2, tk2: smooth_max_grad(x, n2, a2, tk2, 0.0),
                      args=(n, alpha, top_k), method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run_slsqp2(h, n, deadline, alpha, top_k, maxiter):

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    def slsqp_o(x):

            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = max(15, min(45, deadline - time.time() - 40))
            small_cfgs = [
                (200.0, 15, 50.0, 150),
                (5000.0, 5, 200.0, 200),
                (100000.0, 2, 1000.0, 200),
            ]
            h_t, c_t = run_lbfgs(h_interp, n_try, time.time() + alloc * 0.7, small_cfgs)
            if time.time() < deadline - 30:
                h_t, c_t = run_slsqp2(h_t, n_try, time.time() + min(15, deadline - 25),
                                      50000, 3, 800)
            if time.time() < deadline - 15:
                h_t, c_t = local_search_v2(h_t, n_try, min(time.time() + 10, deadline - 10), seed=seed+1)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n_init = n_try

    if time.time() < deadline - 8:
        nf = len(best_h)
        final_configs = [
            (500000.0, 3, 1000.0, 200),
            (5000000.0, 2, 5000.0, 200),
        ]
        best_h, best_c = run_lbfgs(best_h, nf, deadline - 5, final_configs)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
