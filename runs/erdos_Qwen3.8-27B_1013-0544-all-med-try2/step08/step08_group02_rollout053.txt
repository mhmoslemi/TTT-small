```python
                improved = True
            else:

        if not improved:
```

```python
def compute_full_corr(h, n):
    return np.correlate(h, 1-h, mode='full')

def swap_delta(h, i, j, n):
    """Compute the change in full correlation when h[i]:1->0, h[j]:0->1."""
    # corr has length 2n-1, indexed 0..2n-2
    # corr[m] corresponds to k = m - (n-1)
    # corr[k+(n-1)] = sum_{valid m} h[m] * (1-h[m+k])
    
    delta = np.zeros(2*n-1)
    # For each k from -(n-1) to (n-1):
    # k = m - (n-1) where m is the index in the full array
    # Term 1: -(1-h[i+k]) if 0 <= i+k < n
    # Term 2: +(1-h[j+k]) if 0 <= j+k < n
    # Term 3: +h[i-k] if 0 <= i-k < n (but wait, we need m+k=i, so m=i-k, and m must be valid: 0<=m<n)
    #   Also need the original term to be valid: 0 <= m < n and 0 <= m+k < n
    #   m = i-k, so 0 <= i-k < n and 0 <= i < n (always true since i is valid)
    #   So condition: 0 <= i-k < n
    # Term 4: -h[j-k] if 0 <= j-k < n
    
    # Let's vectorize over k
    k_vals = np.arange(-(n-1), n)  # length 2n-1
    # For term 1: valid when 0 <= i+k < n
    mask1 = (i + k_vals >= 0) & (i + k_vals < n)
    delta[mask1] -= (1 - h[i + k_vals[mask1]])
    # For term 2: valid when 0 <= j+k < n
    mask2 = (j + k_vals >= 0) & (j + k_vals < n)
    delta[mask2] += (1 - h[j + k_vals[mask2]])
    # For term 3: valid when 0 <= i-k < n (m = i-k must be in [0,n))
    #   Also the original term h[m]*(1-h[m+k]) must be in the sum, which requires 0<=m<n and 0<=m+k<n
    #   m = i-k, m+k = i, so 0<=i<n (always true) and 0<=i-k<n
    mask3 = (i - k_vals >= 0) & (i - k_vals < n)
    delta[mask3] += h[i - k_vals[mask3]]
    # For term 4: valid when 0 <= j-k < n
    mask4 = (j - k_vals >= 0) & (j - k_vals < n)
    delta[mask4] -= h[j - k_vals[mask4]]
    
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            if not mask.any():
                break
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return float(m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def pen_obj(x, n, alpha, mu):
    val = smooth_max_val(x, n, alpha)
    s = x.sum() - n / 2.0
    return val + mu * s * s

def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_polish(h, n, deadline, max_iters=500):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        if deadline - time.time() < 8:
            break
        mu = min(1000.0, alpha * 0.1)
        try:
            res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def binary_local_search(h, n, deadline, verbose=False):
    """Fast binary local search with O(n) per swap evaluation."""
    dx = 2.0 / n
    h = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h.sum())
    if cnt > target:
        idx = np.where(h == 1)[0]
        for i in idx[:cnt - target]:
            h[i] = 0
    elif cnt < target:
        idx = np.where(h == 0)[0]
        rng_fix = np.random.RandomState(0)
        for i in rng_fix.choice(idx, size=target - cnt, replace=False):
            h[i] = 1
    
    # Compute initial full correlation
    full = np.correlate(h, 1 - h, mode='full')  # length 2n-1
    best_c = full.max() * dx
    best_h = h.copy()
    
    k_vals = np.arange(-(n-1), n).astype(int)  # lag values, length 2n-1
    
    improved = True
    total_swaps = 0
    rng = np.random.RandomState(42)
    
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        n_try = min(n_ones * n_zeros, 100000)
        
        for _ in range(n_try):
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            
            # Compute delta vector
            delta = np.zeros(2 * n - 1)
            mask1 = (i + k_vals >= 0) & (i + k_vals < n)
            delta[mask1] -= (1.0 - h[i + k_vals[mask1]])
            mask2 = (j + k_vals >= 0) & (j + k_vals < n)
            delta[mask2] += (1.0 - h[j + k_vals[mask2]])
            mask3 = (i - k_vals >= 0) & (i - k_vals < n)
            delta[mask3] += h[i - k_vals[mask3]]
            mask4 = (j - k_vals >= 0) & (j - k_vals < n)
            delta[mask4] -= h[j - k_vals[mask4]]
            
            new_full_max = (full + delta).max()
            new_c = new_full_max * dx
            
            if new_c < best_c - 1e-14:
                # Accept
                h[i], h[j] = 0.0, 1.0
                full += delta
                best_c = new_c
                best_h = h.copy()
                improved = True
                total_swaps += 1
                # Refresh ones/zeros occasionally
                if total_swaps % 1000 == 0:
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    n_ones = len(ones)
                    n_zeros = len(zeros)
            else:
                total_swaps += 1
    
    return best_h, best_c
```
