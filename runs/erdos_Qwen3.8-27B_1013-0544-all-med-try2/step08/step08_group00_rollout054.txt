                            h[jj] += d

                    break

                return best_h, best_c
        h = best_h

def compute_c5_profile(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_profile(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = compute_c5_profile(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def smooth_grad(x, n, alpha):
    c = compute_c5_profile(x, n)
    mx = c.max()
    w = np.exp(alpha * (c - mx))
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:

        k = m - (n - 1)
        wm = w[m] * dx

            grad += wm * (1.0 - 2.0 * x)

            grad[k:] += wm * (-x[:n-k])
        else:
            kk = -k
            grad[:n-kk] += wm * (1.0 - x[kk:])
            grad[kk:] += wm * (-x[:n-kk])
    return grad
