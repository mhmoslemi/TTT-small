    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    g = 2.0 * mu * (x.sum() - n / 2.0)
    grad += g
    return grad

            if c_new < best_c:

def top_lag_optimize(h, n, deadline, top_k=3, step_init=0.01):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = step_init
    stalls = 0

        top_idx = np.argsort(c)[-top_k:]

        grad = np.zeros(n)
        for m in top_idx:
            k = m - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - best_h[k:]) * dx
                grad[k:] += (-best_h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - best_h[:nv]) * dx
                grad[:nv] += (-best_h[kk:]) * dx

        c_new = compute_c5(h_new, n)

            best_h = h_new

        else:

        top_idx = np.argsort(c)[-top_k:]
        dx = 2.0 / n
        grad = np.zeros(n)
        for m in top_idx:
            k = m - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - best_h[k:]) * dx
                grad[k:] += (-best_h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - best_h[:nv]) * dx
                grad[:nv] += (-best_h[kk:]) * dx
        h_new = project_feasible(best_h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c, best_h = c_new, h_new
            stalls = 0
            step = min(step * 1.03, 0.025)
        else:
            stalls += 1
            if stalls > 40:
                step *= 0.65
                stalls = 0
                if step < 1e-8:
                    break
    return best_h, best_c

def pair_search(h, n, deadline, radius=100):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in [0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]:
        imp = True
        while imp and time.time() < deadline:
            imp = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo, hi = max(0, i - radius), min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            imp = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[j] + d <= 1.0 and best_h[i] - d >= 0.0:
                        best_h[j] += d
                        best_h[i] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            imp = True
                        else:
                            best_h[j] -= d
                            best_h[i] += d
    return best_h, best_c

def try_different_n(h_best, n_best, c_best, deadline, seed):
    best_h, best_c, best_n = h_best, c_best, n_best
    for n_new in [500, 400, 350]:
        if time.time() > deadline - 40:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hi = project_feasible(hi, n_new)
        alloc = min(80, max(20, (deadline - time.time() - 30) * 0.5))
        ho, co = lbfgs_opt(hi, n_new, time.time() + alloc, [10000, 100000, 1000000], 400)
        if co < best_c:
            best_h, best_c, best_n = ho, co, n_new
    return best_h, best_c, best_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(500, 0.5)
        n = 500
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS-B main (use 55% of budget)
    t1 = min(t0 + 0.55 * budget_s, deadline - 300)
    best_h, best_c = lbfgs_opt(best_h, n, t1,
        [500, 5000, 50000, 500000, 5000000, 50000000], 800)

    # Phase 2: Top-lag subgradient
    if time.time() < deadline - 200:
        t2 = min(time.time() + 60, deadline - 200)
        h2, c2 = top_lag_grad_step(best_h, n, t2, top_k=3)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: L-BFGS-B with very high alpha
    if time.time() < deadline - 150:
        t3 = min(time.time() + 100, deadline - 150)
        h3, c3 = lbfgs_opt(best_h, n, t3,
            [50000000, 500000000, 5000000000], 500)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Top-lag with more lags
    if time.time() < deadline - 100:
        t4 = min(time.time() + 50, deadline - 100)
        h4, c4 = top_lag_grad_step(best_h, n, t4, top_k=5)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Try different n
    if time.time() < deadline - 80:
        best_h, best_c, n = try_different_n(best_h, n, best_c, deadline - 60, seed)

    # Phase 6: Final L-BFGS-B polish on best n
    if time.time() < deadline - 30:
        t6 = min(time.time() + 40, deadline - 30)
        h6, c6 = lbfgs_opt(best_h, n, t6, [100000000, 1000000000], 300)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Phase 7: Pair search (final polish)
    if time.time() < deadline - 3:
        best_h, best_c = pair_search(best_h, n, deadline - 1, radius=80)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
