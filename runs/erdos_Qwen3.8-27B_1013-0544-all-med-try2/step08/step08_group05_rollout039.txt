import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return float(m + (sm + np.log(np.exp(sh - sm).sum())) / alpha)
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def fast_local_search(h_init, n, deadline, max_sweeps=10000, n_restarts=1, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n-1), n)
    s_len = 2 * n - 1
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:

            for _ in range(n_pert):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    h[a], h[b] = 0.0, 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur_c = corr.max() * dx
            if cur_c > best_c + 0.01:
                h = best_h.copy()
                corr = np.correlate(h, 1.0 - h, mode='full')
                cur_c = best_c
                continue
        
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if len(zeros) == 0 or len(ones) == 0:
                break
            
            best_c_sweep = cur_c
            best_swap = None
            
            for i in zeros:
                if time.time() > deadline:
                    break
                i_plus_s = i + m_range
                mask_ip = (i_plus_s >= 0) & (i_plus_s < n)
                common = np.zeros(s_len)
                common[mask_ip] = (1.0 - h)[i_plus_s[mask_ip]]
                i_minus_s = i - m_range
                mask_im = (i_minus_s >= 0) & (i_minus_s < n)
                common[mask_im] -= h[i_minus_s[mask_im]]
                
                base = corr + common - 1.0
                base[n-1] -= 2.0

                s_idx = m_range[None, :]
                jp = j_idx + s_idx
                jm = j_idx - s_idx
                M_plus = np.where((jp >= 0) & (jp < n), h[np.clip(jp, 0, n-1)], 0.0)
                M_minus = np.where((jm >= 0) & (jm < n), h[np.clip(jm, 0, n-1)], 0.0)
                sum_hs = M_plus + M_minus
                
                new_corr_all = sum_hs + base[None, :]
                
                # Vectorized cross terms
                j_arr = ones
                k1 = n - 1 + (j_arr - i)
                valid_k1 = (k1 >= 0) & (k1 < s_len)
                if np.any(valid_k1):
                    idxs = np.where(valid_k1)[0]
                    new_corr_all[idxs, k1[valid_k1]] += 1.0
                k2 = n - 1 + (i - j_arr)
                valid_k2 = (k2 >= 0) & (k2 < s_len)
                if np.any(valid_k2):
                    idxs = np.where(valid_k2)[0]
                    new_corr_all[idxs, k2[valid_k2]] += 1.0
                
                max_vals = new_corr_all.max(axis=1)
                bi = max_vals.argmin()
                if max_vals[bi] * dx < best_c_sweep - 1e-15:
                    j = ones[bi]
                    best_c_sweep = max_vals[bi] * dx
                    best_swap = (i, j, new_corr_all[bi].copy())
            
            if best_swap is not None:
                i, j, new_corr = best_swap
                h[i], h[j] = 1.0, 0.0
                corr = new_corr
                cur_c = best_c_sweep
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                break
    
    return best_h, best_c
