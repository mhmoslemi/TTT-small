    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_shift(h, n, k):
    """Gradient of correlation at shift k with respect to h."""
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] += (1.0 - h[k:])
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] += (1.0 - h[:nv])
        grad[:nv] -= h[kk:]
    return grad

def smooth_max_grad(h, n, top_k, alpha):
    c = compute_c5_vec(h, n)
    idx = np.argsort(c)[-top_k:]
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(top_k):

        grad += w[i] * dx * grad_shift(h, n, k)
    obj = m + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha
    return grad, obj

A better approach: Just use a simple bisection or ternary search on t. Since f(t) is convex, ternary search will find the minimum in O(log(1/ε)) evaluations, each O(n). So per pair, it's O(n log(1/ε)). With O(n²) pairs, total is O(n³ log(1/ε)). For n=500 and ε=1e-6, that's about 500³ * 20 = 2.5 * 10^9, which is borderline.

Actually, we don't need to try all pairs. We can be smart about which pairs to try:
- Only try pairs (i,j) where the gradient suggests improvement
- Or use a priority queue

Let me take a step back and think about what's practical:

For n=500:
- compute_c5 via FFT: ~100μs (optimistic)
- Pair polish with 500*80 = 40000 pairs per delta, 10 deltas: 400000 * 100μs = 40 seconds per "improved" pass
- Maybe 5-10 passes: 200-400 seconds

That's a lot. Let me reduce:
- radius = 20 instead of 40: 500*40 = 20000 pairs per delta
- 5 deltas: 100000 * 100μs = 10 seconds per pass
- 10 passes: 100 seconds

That's more reasonable.

OK let me just write a clean, practical implementation. I'll focus on:
1. Good initial points (multiple starts)
2. L-BFGS-B for coarse optimization
3. Subgradient for medium-level optimization
4. Fast pair polish for fine optimization

The key difference from the previous code will be:
- More aggressive pair polish with adaptive deltas
- Better gradient computation (using only the top-k active shifts)
- Multiple starting points at different n values
- A "binary snap" step: after optimization, try snapping values close to 0 or 1 to exactly 0 or 1, then re-optimize

import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_shift(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] += (1.0 - h[k:])
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] += (1.0 - h[:nv])
        grad[:nv] -= h[kk:]
    return grad

def smooth_max_grad(h, n, top_k, alpha):
    c = compute_c5_vec(h, n)
    idx = np.argsort(c)[-top_k:]
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(len(idx)):
        k = int(idx[i]) - (n - 1)
        grad += w[i] * dx * grad_shift(h, n, k)
    obj = m + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha
    return grad, obj

def subgrad_optimize(h, n, deadline, top_k=50, alpha=300.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.03
    prev_h = h.copy()
    prev_g = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        g, _ = smooth_max_grad(h, n, top_k, alpha)
        if stall > 0:
            s = h - prev_h
            y = g - prev_g
            sy = s @ y
            ss = s @ s
            yy = y @ y
            if yy > 1e-20:
                step = ss / sy if sy > 1e-10 else ss / yy
                step = float(np.clip(step, 1e-4, 0.08))
        prev_h = h.copy()
        prev_g = g.copy()
        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            stall += 1
            h = project_feasible(h - step * 0.3 * g, n)
            c2 = compute_c5(h, n)
            if c2 < best_c:
                best_c = c2
                best_h = h.copy()
                stall = 0
            if stall > 100:
                break
    return best_h, best_c

def pair_polish(h, n, deadline, radius=25):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] >= d and h[j] <= 1.0 - d:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                    if h[j] >= d and h[i] <= 1.0 - d:
                        h[j] -= d
                        h[i] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[j] += d
                            h[i] -= d
            h = best_h.copy()
    return best_h, best_c

def golden_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = (np.arange(half) * phi).astype(int) % n
    h[positions] = 1.0
    return project_feasible(h, n)

def sine_start(n, f, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 * (1.0 + np.sin(f * x + phase))
    return project_feasible(h, n)

def random_binary_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def lbfgs_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha + 50.0 * (x.sum() - n / 2.0) ** 2)

def lbfgs_grad(x, n, alpha):
    g, _ = smooth_max_grad(x, n, 60, alpha)
    g += 100.0 * (x.sum() - n / 2.0)
    return g

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = golden_start(500, rng)
        best_c = compute_c5(best_h, 500)
    best_n = len(best_h)

    # Phase 1: L-BFGS on initial
    if time.time() < deadline - 400:
        bounds = [(0.0, 1.0)] * best_n
        for alpha in [300, 1000, 5000]:
            if time.time() > deadline - 350:
                break
            try:
                res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad,
                              args=(best_n, alpha), method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': 400, 'ftol': 1e-16})
                h_new = project_feasible(res.x, best_n)
                c_new = compute_c5(h_new, best_n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 2: Try alternative n values and starts
    for n_try in [400, 600, 300, 700]:
        if time.time() > deadline - 200:
            break
        if n_try == best_n:
            continue
        starts = [golden_start(n_try, rng), random_binary_start(n_try, rng)]
        for f in [1.5, 2.5]:
            starts.append(sine_start(n_try, f, rng))
        for h_s in starts:
            if time.time() > deadline - 200:
                break
            bounds = [(0.0, 1.0)] * n_try
            try:
                res = minimize(lbfgs_obj, h_s, jac=lbfgs_grad,
                              args=(n_try, 2000), method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': 300, 'ftol': 1e-15})
                h_opt = project_feasible(res.x, n_try)
                c_opt = compute_c5(h_opt, n_try)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
                    best_n = n_try
            except Exception:
                pass

    # Phase 3: Subgradient refinement
    if time.time() < deadline - 120:
        t_sub = min(time.time() + 80, deadline - 80)
        best_h, best_c = subgrad_optimize(best_h, best_n, t_sub, top_k=60, alpha=500.0)

    # Phase 4: Binary snap + re-optimize
    if time.time() < deadline - 150:
        h_snap = best_h.copy()
        n = best_n
        h_snap[h_snap > 0.7] = 1.0
        h_snap[h_snap < 0.3] = 0.0
        h_snap = project_feasible(h_snap, n)
        c_snap = compute_c5(h_snap, n)
        if c_snap < best_c:
            best_c = c_snap
            best_h = h_snap
        # Quick L-BFGS from snap
        if time.time() < deadline - 100:
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad,
                              args=(n, 5000), method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': 500, 'ftol': 1e-16})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 5: Subgradient again
    if time.time() < deadline - 80:
        t_sub2 = min(time.time() + 50, deadline - 40)
        best_h, best_c = subgrad_optimize(best_h, best_n, t_sub2, top_k=80, alpha=2000.0)

    # Phase 6: Pair polish
    if time.time() < deadline - 2:
        best_h, best_c = pair_polish(best_h, best_n, deadline - 1, radius=20)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
