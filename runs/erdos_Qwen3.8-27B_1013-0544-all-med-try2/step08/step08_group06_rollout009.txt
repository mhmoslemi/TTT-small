import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def obj_grad(x, n, alpha):
    q = 1.0 - x
    c = np.correlate(x, q, mode='full')
    m = c.max()
    ks = int(np.argmax(c))
    sh = alpha * (c - m)
    ms = sh.max()
    es = np.exp(sh - ms)
    p = es / es.sum()
    w = p.copy()
    w[ks] += 1.0
    val = m + np.log(es.sum()) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wv = w[k]
        if wv < 1e-16:
            continue
        off = k - (n - 1)
        js = max(0, -off)
        je = min(n, n - off)
        if je > js:
            g[js:je] += wv * q[js + off:je + off]
        js2 = max(0, off)
        je2 = min(n, n + off)
        if je2 > js2:
            g[js2:je2] -= wv * x[js2 - off:je2 - off]
    cv = x.sum() - n / 2.0
    g += 200.0 * cv
    return val, g

def lfb(h, n, alpha, maxiter, timeout_s):
    ts = time.time()
    def obj(x):
        if time.time() - ts > timeout_s:
            return np.inf
        v, _ = obj_grad(x, n, alpha)
        return v
    def jac(x):
        if time.time() - ts > timeout_s:
            return np.ones(n)
        _, g = obj_grad(x, n, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': maxiter, 'maxfun': maxiter * 5,
                              'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 100})
        return project_feasible(r.x, n)
    except Exception:
        return h

def greed_binary(h_init, n, deadline, max_sweeps=500, seed=42):
    rng = np.random.RandomState(seed)
    mr = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ol = [int(x) for x in np.where(h == 1)[0]]
    zl = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        rng.shuffle(zl)
        rng.shuffle(ol)
        for i in zl:
            if time.time() > deadline:
                break
            for j in ol:
                d = np.zeros(2 * n - 1)
                im = i + mr
                mk = (im >= 0) & (im < n)
                d[mk] += 1.0 - h[im[mk]]
                imn = i - mr
                mk = (imn >= 0) & (imn < n)
                d[mk] -= h[imn[mk]]
                jm = j + mr
                mk = (jm >= 0) & (jm < n)
                d[mk] -= 1.0 - h[jm[mk]]
                jmn = j - mr
                mk = (jmn >= 0) & (jmn < n)
                d[mk] += h[jmn[mk]]
                d[n - 1] -= 2.0
                d[n - 1 + j - i] += 1.0
                d[n - 1 + i - j] += 1.0
                nc = (corr + d).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur_c = nc
                    ol.remove(j)
                    zl.remove(i)
                    ol.append(i)
                    zl.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

    za = np.where(h == 0)[0].copy()

    for _ in range(n_iters):

            break
        ai = rng.randint(nh)
        bi = rng.randint(nh)
        i = za[ai]
        j = oa[bi]
        d = np.zeros(2 * n - 1)
        im = i + mr
        mk = (im >= 0) & (im < n)
        d[mk] += 1.0 - h[im[mk]]
        imn = i - mr
        mk = (imn >= 0) & (imn < n)
        d[mk] -= h[imn[mk]]
        jm = j + mr
        mk = (jm >= 0) & (jm < n)
        d[mk] -= 1.0 - h[jm[mk]]
        jmn = j - mr
        mk = (jmn >= 0) & (jmn < n)
        d[mk] += h[jmn[mk]]
        d[n - 1] -= 2.0
        d[n - 1 + j - i] += 1.0
        d[n - 1 + i - j] += 1.0
        nc = (corr + d).max() * dx
        if nc <= cur_c or (T > 1e-14 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += d
            cur_c = nc
            za[ai] = j
            oa[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c
