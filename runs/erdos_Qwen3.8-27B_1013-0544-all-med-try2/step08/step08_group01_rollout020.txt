```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    """Vectorized gradient of smooth max of correlation."""
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    
    # Build gradient using the structure:
    # g[j] = dx * sum_k w[k] * [(1-h[j+k]) - h[j-k]] (with bounds)
    # where k ranges from -(n-1) to (n-1)
    
    # To vectorize: for each k, the contribution to g is w[k]*dx * [(1-h[j+k]) - h[j-k]]
    # We can do this with a loop over k (2n-1 iterations) but each is O(n) vectorized
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wm = w[k + n - 1] * dx
        if wm < 1e-14:
            continue
        if k >= 0:
            # (1-h[j+k]) for j in 0..n-1-k
            grad[:n-k] += wm * (1.0 - h[k:])
            # -h[j-k] for j in k..n-1
            grad[k:] += wm * (-h[:n-k])
        else:
            kk = -k
            # (1-h[j+k]) = (1-h[j-kk]) for j in kk..n-1
            grad[kk:] += wm * (1.0 - h[:n-kk])
            # -h[j-k] = -h[j+kk] for j in 0..n-1-kk
            grad[:n-kk] += wm * (-h[kk:])
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wm = w[k + n - 1] * dx
        if wm < 1e-14:
            continue
        if k >= 0:
            grad[:n - k] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:n - k])
        else:
            kk = -k
            grad[kk:] += wm * (1.0 - h[:n - kk])
            grad[:n - kk] += wm * (-h[kk:])
    return grad

def smooth_max_objective(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_gradient(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_optimize(h, n, deadline, max_iters=300):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(200, 0.5), (1000, 2.0), (5000, 10.0),
               (20000, 40.0), (100000, 200.0),
               (500000, 800.0), (2000000, 3000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        tr = deadline - time.time() - 3
        if tr < 5:
            break
        try:
            res = minimize(smooth_max_objective, h, jac=smooth_max_gradient,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'ftol': 1e-18,
                                    'gtol': 1e-16, 'maxfun': max_iters * 25,
                                    'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def to_binary(h, n):
    target = n // 2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def binary_hill_climb(h_bin, n, deadline, rng, max_stalls=30000):

    best_c = c.max() * (2.0 / n)
    best_h = h_bin.copy()
    pad = 2 * n - 1
    hpad = np.concatenate([np.zeros(pad), h_bin, np.zeros(pad)])
    stall = 0

        i0 = ones[rng.randint(len(ones))]
        j0 = zeros[rng.randint(len(zeros))]

def slsqp_run(h, n, deadline, alpha=200000, maxiter=2000):
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    def eq(x):
        return float(x.sum() - n / 2.0)
    def jac_eq(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq, 'jac': jac_eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def nesterov_pg(h, n, deadline, alpha=5000, momentum=0.85, step=0.05):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    while time.time() < deadline - 2:
        g = grad_smooth_max(h, n, alpha)
        r = h.sum() - n / 2.0
        g += r
        v = momentum * v - step * g
        h_new = project_feasible(h + v, n)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-13:
            best_c = c
            best_h = h_new.copy()
        h = h_new
        if c > best_c + 0.02:
            step *= 0.6
        else:
            step = min(step * 1.05, 0.5)
    return best_h, best_c

            best_c, best_h = c2, h2

    # Phase 3: Binary conversion and hill-climb
    h_bin = to_binary(best_h, n)
    if time.time() < deadline - 150:
        t3 = min(time.time() + 250, deadline - 100)
        h_bin, c3 = binary_hill_climb(h_bin, n, t3, rng, max_stalls=40000)
        if c3 < best_c:
            best_c, best_h = c3, h_bin.copy()

    # Phase 4: L-BFGS starting from binary solution
    if time.time() < deadline - 100:
        h4 = h_bin.astype(float).copy()
        h4 = project_feasible(h4, n)
        t4 = min(time.time() + 120, deadline - 60)
        h4, c4 = lbfgs_optimize(h4, n, t4, max_iters=200)
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()

    # Phase 5: SLSQP polish
    if time.time() < deadline - 40:
        t5 = min(time.time() + 50, deadline - 25)
        h5, c5 = slsqp_run(best_h, n, t5, alpha=500000, maxiter=3000)
        if c5 < best_c:
            best_c, best_h = c5, h5.copy()

    # Phase 6: Another binary hill-climb from best continuous
    h_bin2 = to_binary(best_h, n)
    if time.time() < deadline - 15:
        t6 = min(time.time() + 60, deadline - 8)
        h_bin2, c6 = binary_hill_climb(h_bin2, n, t6, rng, max_stalls=20000)
        if c6 < best_c:
            best_c, best_h = c6, h_bin2.copy()

    # Phase 7: Final binary hill-climb on the best overall
    h_final_bin = to_binary(best_h, n)
    if time.time() < deadline - 3:
        t7 = min(time.time() + 20, deadline - 2)
        h_final_bin, c7 = binary_hill_climb(h_final_bin, n, t7, rng, max_stalls=10000)
        if c7 < best_c:
            best_c, best_h = c7, h_final_bin.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
