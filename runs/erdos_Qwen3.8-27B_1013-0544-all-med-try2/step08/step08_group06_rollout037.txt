def project_box_sum(h, target):
    """Project h onto {0 <= h <= 1, sum(h) = target}"""
    for _ in range(100):
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to increase some elements
            mask = h < 1
            if not mask.any():

            h[mask] += diff / mask.sum()
        else:
            mask = h > 0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
    return np.clip(h, 0, 1)

For binary h, the objective is:
C(h) = max_k dx * sum_m h[m]*(1-h[m+k-n+1])

When I flip h[i] from 0 to 1 and h[j] from 1 to 0, the change in c[k] is:
Δc[k] = (1-h[i+k-n+1])*δ1 - h[i-k+n-1]*δ1' - (1-h[j+k-n+1])*δ2 + h[j-k+n-1]*δ2'

where δ1 = [0≤i+k-n+1<n], etc.

For binary h, (1-h[i+k-n+1]) is either 0 or 1, and h[i-k+n-1] is either 0 or 1.

So Δc[k] ∈ {-2, -1, 0, 1, 2} for each k.

The new maximum is max_k (c[k] + Δc[k]).

Key insight: For a good swap, we want to decrease the lags that are near the maximum. The lags at the maximum are the "critical" ones.

**Strategy: Focus on critical lags**

Let L_max = {k : c[k] = C} (the set of lags achieving the maximum).
A swap (i,j) is "good" if it decreases c[k] for k in L_max without increasing any c[k'] above C.

For each critical lag k*, the change is:
Δc[k*] = (1-h[i+k*-n+1])*δ1 - h[i-k*+n-1]*δ1' - (1-h[j+k*-n+1])*δ2 + h[j-k*+n-1]*δ2'

For binary h, this is an integer in {-2,-1,0,1,2}.

To decrease c[k*], we want Δc[k*] < 0, which means:
(1-h[i+k*-n+1])*δ1 - h[i-k*+n-1]*δ1' < (1-h[j+k*-n+1])*δ2 - h[j-k*+n-1]*δ2'

The LHS is the "contribution of i" and the RHS is the "contribution of j" for lag k*.

The "contribution of position p for lag k*" is:
g(p, k*) = (1-h[p+k*-n+1])*[0≤p+k*-n+1<n] - h[p-k*+n-1]*[0≤p-k*+n+1<n]

For binary h, g(p, k*) ∈ {-1, 0, 1, 2}.

So I want to find i (zero) and j (one) such that g(i, k*) < g(j, k*) for all critical k*, and the overall max doesn't increase.

This gives a heuristic: for each zero position i, compute the vector g(i, k) for all k. For each one position j, compute g(j, k). Find pairs where g(i, k) < g(j, k) for the critical k's.

This is still O(n²) in the worst case, but with better pruning.

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def lse_gradient(h, n, alpha):
    """Compute gradient of log-sum-exp smoothed objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Softmax weights
    max_c = corr.max()
    exp_c = np.exp(alpha * (corr - max_c))
    w = exp_c / exp_c.sum()  # shape (2n-1,)
    
    # Compute gradient
    # grad[i] = dx * sum_k w[k] * [(1-h[i+k-n+1])*δ1 - h[i-k+n-1])*δ2]
    # Let shift_k = k - (n-1)
    shifts = np.arange(-(n-1), n)  # shape (2n-1,)
    
    grad = np.zeros(n)
    for i in range(n):
        # Direct: (1-h[i+shift]) for valid i+shift
        idx_d = i + shifts
        mask_d = (idx_d >= 0) & (idx_d < n)
        # Indirect: -h[i-shift] for valid i-shift
        idx_i = i - shifts
        mask_i = (idx_i >= 0) & (idx_i < n)
        
        dc = np.zeros(2*n-1)
        dc[mask_d] = 1 - h[idx_d[mask_d]]
        dc[mask_i] -= h[idx_i[mask_i]]
        grad[i] = np.dot(w, dc) * dx
    
    return grad, corr

def lse_objective(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    logsum = max_c + np.log(np.exp(alpha * (corr - max_c)).sum()) / alpha
    return logsum

def projected_gradient(h, n, alphas, max_iter, deadline):
    """Projected gradient with log-sum-exp smoothing."""
    h = h.copy()
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        max_iter_phase = max_iter // len(alphas)
        lr = 0.1 / alpha  # initial step size
        prev_grad = None
        
        for it in range(max_iter_phase):
            if time.time() > deadline:
                break
            
            grad, corr = lse_gradient(h, n, alpha)
            
            # Barzilai-Borwein step size
            if prev_grad is not None:
                s = h - prev_h  # need to store prev_h
                y = grad - prev_grad
                ss = np.dot(s, s)
                sy = np.dot(s, y)
                yy = np.dot(y, y)
                if yy > 1e-20:
                    lr = ss / sy if sy > 0 else 0.01 / alpha
            
            # Projected gradient step
            h_new = project_box_sum(h - lr * grad, target)
            
            # Line search (backtracking)
            f_new = lse_objective(h_new, n, alpha)
            f_old = lse_objective(h, n, alpha)
            while f_new > f_old + 1e-4 * np.dot(grad, h_new - h) and lr > 1e-10:
                lr *= 0.5
                h_new = project_box_sum(h - lr * grad, target)
                f_new = lse_objective(h_new, n, alpha)
            
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            
            prev_grad = grad.copy() if lr > 0 else None
            h = h_new
        
        # Verify constraint
        h = project_box_sum(h, target)
    
    return best_h, best_c

    mask_d = (idx_d >= 0) & (idx_d < n)

    mask_i = (idx_i >= 0) & (idx_i < n)

    try:

                       args=(n, alpha, precomp), bounds=bounds,
                       constraints=cons,

        return project_box_sum(h0, n/2.0)

    h_b[idx[:n//2]] = 1.0
    return h_b

    m_range = np.arange(-(n-1), n)
    h = to_binary(h_init, n).copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c

        if time.time() > deadline:
            break
        improved = False
        # Try all zero positions, find best one to swap with
        for i in sorted(zeros):
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in sorted(ones):
                # Compute delta efficiently
                im = i + m_range
                msk = (im >= 0) & (im < n)
                jm = j + m_range
                msk2 = (jm >= 0) & (jm < n)
                imn = i - m_range
                msk3 = (imn >= 0) & (imn < n)
                jmn = j - m_range
                msk4 = (jmn >= 0) & (jmn < n)
                
                delta = np.zeros(2*n-1)
                delta[msk] += 1.0 - h[im[msk]]
                delta[msk3] -= h[imn[msk3]]
                delta[msk2] -= 1.0 - h[jm[msk2]]
                delta[msk4] += h[jmn[msk4]]
                
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:

                h[best_j] = 0.0
                corr = np.correlate(h, 1-h, mode='full').astype(float)
                cur_c = best_nc
                zeros.discard(i)
                zeros.add(best_j)
                ones.discard(best_j)
                ones.add(i)
                improved = True
                if cur_c < best_c:

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def project_box_sum(h, target):

    for _ in range(80):

        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:

def precompute(n):
    shifts = np.arange(-(n-1), n)
    I, K = np.meshgrid(np.arange(n), shifts, indexing='ij')
    idx_d = I + K
    mask_d = (idx_d >= 0) & (idx_d < n)
    idx_i = I - K
    mask_i = (idx_i >= 0) & (idx_i < n)
    return idx_d, mask_d, idx_i, mask_i

def _obj(h, n, alpha, precomp):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    mc = corr.max()
    return mc + np.log(np.exp(alpha * (corr - mc)).sum()) / alpha

def _grad(h, n, alpha, precomp):
    dx = 2.0 / n
    idx_d, mask_d, idx_i, mask_i = precomp
    corr = np.correlate(h, 1 - h, mode='full') * dx
    mc = corr.max()
    ec = np.exp(alpha * (corr - mc))
    w = ec / ec.sum()
    t1 = np.where(mask_d, 1.0 - h[idx_d], 0.0)
    t2 = np.where(mask_i, h[idx_i], 0.0)
    return dx * ((t1 - t2) @ w)

    idx = np.argsort(h)[::-1]
