import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_c5(h, n, alpha):
    """Analytic gradient of (1/alpha)*log(sum exp(alpha*g_m)) where g_m = correlation * dx."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx
    max_g = g.max()
    sh = alpha * (g - max_g)
    sh -= sh.max()
    ev = np.exp(sh)
    se = ev.sum()
    w = ev / se
    value = max_g + np.log(se) / alpha
    
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wm = w[m_idx]
        if wm < 1e-14:
            continue
        L = m_idx - (n - 1)
        if L >= 0:
            grad[L:] += wm * (1.0 - h[:n-L])
            grad[:n-L] -= wm * h[L:]
        else:
            L2 = -L
            grad[:n-L2] += wm * (1.0 - h[L2:])
            grad[L2:] -= wm * h[:n-L2]
    grad *= dx
    return value, grad

def slsqp_smooth(h, n, alpha, maxiter=2000, ftol=1e-18):
    dx = 2.0 / n
    def obj(x):
        v, _ = compute_grad_c5(x, n, alpha)
        return v
    def grad(x):
        _, g = compute_grad_c5(x, n, alpha)
        return g
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, n, max_sweeps=200, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False

            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            # Bisection on target value t
            hi_v = np.max(corr + B * (1.0 - h[i])) if h[i] < 0.9999 else np.max(corr - B * h[i])
            lo_v = np.max(corr - B * h[i]) if h[i] > 0.0001 else np.max(corr + B * (1.0 - h[i]))
            # Actually, let me just use the current value as a starting point

            # We want to find x in [0,1] minimizing max(corr + B*(x-h[i]))
            # Binary search on t: find smallest t such that exists x in [0,1] with corr + B*(x-h[i]) ≤ t
            lo_t = -np.inf

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(h, n, alpha):
    """Smoothed C5 objective and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx
    max_g = g.max()
    sh = alpha * (g - max_g)
    sh -= sh.max()
    ev = np.exp(sh)
    se = ev.sum()
    w = ev / se
    value = max_g + np.log(se) / alpha
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wm = w[m_idx]
        if wm < 1e-14:
            continue
        L = m_idx - (n - 1)
        if L >= 0:
            grad[L:] += wm * (1.0 - h[:n-L])
            grad[:n-L] -= wm * h[L:]
        else:
            L2 = -L
            grad[:n-L2] += wm * (1.0 - h[L2:])
            grad[L2:] -= wm * h[:n-L2]
    grad *= dx
    return value, grad
