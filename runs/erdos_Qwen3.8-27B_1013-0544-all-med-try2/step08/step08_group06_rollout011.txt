    high = float(np.max(y))+30

def project_feasible(h,n):

    h=np.clip(h,0.0,1.0)
    target=0.5*n
    for _ in range(100):
        diff=target-h.sum()
        if abs(diff)<1e-12: break
        if diff>0.0:
            room=1.0-h
            s=room.sum()
            if s<1e-15: break
            h += diff*room/s
        else:

            if s<1e-15: break

def smooth_max(c,beta):
    m=float(c.max())
    if beta <= 1e-8:
        return m, np.full(c.shape, 1.0/len(c))
    raw=beta*(c-m)
    mx=float(raw.max())
    e=np.exp(raw-mx)
    s=e.sum()
    if s <= 0.0 or not np.isfinite(s):
        w=np.zeros_like(c); w[int(np.argmax(c))]=1.0
        return m,w
    w=e/s
    return m + (mx + float(np.log(s)))/beta, w

import numpy as np
from scipy.optimize import minimize
import time

OPT_DEADLINE = 1e18
MASK_CACHE = {}


def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff * room / s
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 0.0:
        idx = int(np.argmax(1.0 - h))
        h[idx] = min(1.0, h[idx] + diff)
    elif diff < 0.0:
        idx = int(np.argmax(h))
        h[idx] = max(0.0, h[idx] + diff)
    return h


def smooth_max(c, beta):
    m = float(c.max())
    if beta <= 1e-8:
        return m, np.full(c.shape, 1.0 / len(c))
    raw = beta * (c - m)
    mx = float(raw.max())
    e = np.exp(raw - mx)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    w = e / s
    return m + (mx + np.log(s)) / beta, w


def gradient_corr(h, weights, dx, n):
    g = np.zeros(n, dtype=float)
    w0 = float(weights[n - 1])
    if w0 != 0.0:
        g += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        # np.correlate(h, 1-h, 'full') has positive shift at n-1-d,
        # and negative shift at n-1+d.
        wp = float(weights[n - 1 - d])
        wm = float(weights[n - 1 + d])
        w = wp + wm
        if w == 0.0:
            continue
        g[:-d] += wp - w * h[d:]
        g[d:] += wm - w * h[:-d]
    return g * dx


def direct_value(x, n, beta, pen):
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    v, _ = smooth_max(c, beta)
    diff = x.sum() - 0.5 * n
    return float(v + 0.5 * pen * diff * diff)


def direct_grad(x, n, beta, pen):
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    _, w = smooth_max(c, beta)
    g = gradient_corr(x, w, dx, n)
    g += pen * (x.sum() - 0.5 * n)
    return g


def slsqp_val(x, n, beta):
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    v, _ = smooth_max(c, beta)
    return float(v)


def slsqp_grad(x, n, beta):
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    _, w = smooth_max(c, beta)
    return gradient_corr(x, w, dx, n)


def sum_eq(x, n):
    return float(x.sum() - 0.5 * n)


def sum_eq_jac(x, n):
    return np.ones_like(x)


def time_callback(*args):
    if time.time() > OPT_DEADLINE:
        raise RuntimeError("time")


def direct_polish(h0, n, stop_at, betas, pen, maxiter):
    global OPT_DEADLINE
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    mi = int(max(50, min(maxiter, 140000.0 / max(1, n))))
    for beta in betas:
        if time.time() > stop_at - 2.0:
            break
        OPT_DEADLINE = max(time.time() + 1.0, stop_at - 0.3)
        try:
            res = minimize(
                direct_value,
                h,
                method="L-BFGS-B",
                bounds=bounds,
                args=(n, beta, pen),
                jac=direct_grad,
                callback=time_callback,
                options={"maxiter": mi, "ftol": 1e-12, "gtol": 1e-9,
                         "maxcor": 10, "maxls": 10},
            )
            x = res.x
            if np.all(np.isfinite(x)):
                h = project_feasible(x, n)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        except Exception:
            pass
    return best_h, best_c


def slsqp_polish(h0, n, beta, maxiter, stop_at):
    global OPT_DEADLINE
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    OPT_DEADLINE = max(time.time() + 1.0, stop_at - 0.3)
    try:
        res = minimize(
            slsqp_val,
            h,
            method="SLSQP",
            bounds=bounds,
            args=(n, beta),
            constraints=[{"type": "eq", "fun": sum_eq, "jac": sum_eq_jac, "args": (n)}],
            jac=slsqp_grad,
            callback=time_callback,
            options={"maxiter": maxiter, "ftol": 1e-13},
        )
        x = res.x
        if np.all(np.isfinite(x)):
            h = project_feasible(x, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    except Exception:
        pass
    return best_h, best_c


def get_masks(n):
    if n not in MASK_CACHE:
        masks = [0] * n
        for d in range(1, n):
            masks[d] = (1 << (n - d)) - 1
        MASK_CACHE[n] = masks
    return MASK_CACHE[n]


def bits_max_limited(ones, n, limit, order=None, masks=None):
    if masks is None:
        masks = get_masks(n)
    mx = -1
    seq = order if order is not None else range(1, n)
    for d in seq:
        d = int(d)
        msk = masks[d]
        a = (ones & ~(ones >> d) & msk).bit_count()
        b = ((ones >> d) & ~ones & msk).bit_count()
        m = a if a > b else b
        if m >= limit:
            return m, False
        if m > mx:
            mx = m
    return mx, True


def bit_arrays(ones, n):
    oa = []
    za = []
    for i in range(n):
        if (ones >> i) & 1:
            oa.append(i)
        else:
            za.append(i)
    return np.array(oa, dtype=int), np.array(za, dtype=int)


def bits_from_h(h, n):
    idx = np.argsort(h)[::-1]
    ones = 0
    for k in range(n // 2):
        ones |= 1 << int(idx[k])
    return ones


def random_ones(n, seed):
    rng = np.random.RandomState(seed)
    if n <= 2:
        return 0
    idx = rng.choice(n, n // 2, replace=False)
    ones = 0
    for i in idx:
        ones |= 1 << int(i)
    return ones


def h_from_bits(ones, n):
    h = np.zeros(n, dtype=float)
    x = ones
    while x:
        lsb = x & -x
        h[lsb.bit_length() - 1] = 1.0
        x ^= lsb
    return h


def bits_local(ones, n, deadline, seed, max_steps=2000000):
    rng = np.random.RandomState(seed)
    ones_arr, zeros_arr = bit_arrays(ones, n)
    current_max, _ = bits_max_limited(ones, n, 10**9, None, get_masks(n))
    best_ones = ones
    best_max = current_max
    sill = 0
    plateau = 0
    in_plateau = False

    for _ in range(max_steps):
        if time.time() > deadline:
            break
        if len(ones_arr) == 0 or len(zeros_arr) == 0:
            break

        if in_plateau:
            plateau -= 1
            if plateau <= 0:
                in_plateau = False
                sill = 0
                if current_max == best_max:
                    ones = best_ones
                    current_max = best_max
                    ones_arr, zeros_arr = bit_arrays(ones, n)
        elif sill > 5000:
            in_plateau = True
            plateau = 1500
            sill = 0

        old = current_max
        limit = current_max + 1 if in_plateau else current_max
        ai = int(rng.randint(len(zeros_arr)))
        bi = int(rng.randint(len(ones_arr)))
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        new = ones ^ (1 << i) ^ (1 << j)
        mx, ok = bits_max_limited(new, n, limit, None, get_masks(n))

        if ok:
            accept = False
            if mx < old:
                accept = True
            elif in_plateau:
                accept = rng.random() < 0.20

            if accept:
                ones = new
                current_max = mx
                zeros_arr[ai] = j
                ones_arr[bi] = i
                if current_max < best_max:
                    best_max = current_max
                    best_ones = ones
                if mx < old:
                    in_plateau = False
                    plateau = 0
                    sill = 0
                else:
                    sill = 0
            else:
                sill += 1
        else:
            sill += 1

        if sill > 50000 and not in_plateau:
            ones = best_ones
            current_max = best_max
            ones_arr, zeros_arr = bit_arrays(ones, n)
            sill = 0

    return best_ones, best_max * 2.0 / float(n)


def segment_start(n, seed, nseg):
    rng = np.random.RandomState(seed)
    nseg = max(2, min(int(nseg), n - 1))
    if nseg % 2 == 1:
        nseg += 1
    if nseg > n - 1:
        nseg -= 1
    nseg = max(2, nseg)
    cuts = np.sort(rng.choice(np.arange(1, n), nseg - 1, replace=False))
    cuts_list = [0] + cuts.astype(int).tolist() + [n]
    h = np.zeros(n, dtype=float)
    for idx in range(0, nseg, 2):
        h[cuts_list[idx]:cuts_list[idx + 1]] = 1.0
    return project_feasible(h, n)


def sin_start(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n)
    f = float(rng.randint(1, 5))
    phi = rng.uniform(0.0, 2.0 * np.pi)
    amp = 0.35 + 0.15 * rng.random()
    h = 0.5 + amp * np.sin(np.pi * f * x + phi)
    h += 0.02 * rng.standard_normal(n)
    return project_feasible(h, n)
