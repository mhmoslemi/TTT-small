import numpy as np
import time
from scipy.optimize import minimize

FRACS = np.array([1.0/256.0, 1.0/128.0, 1.0/64.0, 1.0/48.0, 1.0/32.0, 1.0/24.0,
                  1.0/16.0, 1.0/12.0, 1.0/8.0, 1.0/6.0, 1.0/4.0, 1.0/2.0,
                  3.0/4.0, 1.0], dtype=float)
TRUSTS = [0.08, 0.04, 0.02, 0.01, 0.005, 0.0025, 0.00125, 0.000625,
          0.0003125, 0.000156, 7.8e-5, 3.9e-5, 1.95e-5, 9.8e-6, 4.9e-6,
          2.45e-6, 1.2e-6]

def compute_r_s(h):
    return np.correlate(h, 1.0 - h, mode='full')[::-1].copy()

def c_from_r(r_s, n):
    return float(r_s.max()) * (2.0 / float(n))

def compute_c5(h, n):
    return c_from_r(compute_r_s(h), n)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).ravel().copy()
    if h.size != n:
        h = h[:n]
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * float(n)
    for _ in range(2000):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s < target:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += ((target - s) / tot) * room
        else:
            if s < 1e-13:
                break
            h += ((target - s) / s) * h
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(n + 5):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            idx = int(np.argmin(h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

def top_weights(r_s, top=160, margin=0.5):
    rmax = float(r_s.max())
    idx = np.flatnonzero(r_s >= rmax - margin)
    if idx.size == 0:
        idx = np.array([int(np.argmax(r_s))], dtype=int)
    if idx.size > top:
        idx = idx[np.argsort(-r_s[idx])][:top]
    w = np.zeros(len(r_s), dtype=float)
    vals = r_s[idx]
    w[idx] = np.maximum(vals - rmax + margin, 1e-8)
    w /= w.sum()
    return w

def grad_weights(h, n, w):
    g = np.zeros(n, dtype=float)
    nz = np.nonzero(np.abs(w) > 1e-12)[0]
    nn = n - 1
    for t in nz:
        wt = float(w[t])
        s = t - nn
        if s >= 0:
            g[:n - s] += wt * (1.0 - h[s:])
            g[s:] -= wt * h[:n - s]
        else:
            ss = -s
            g[ss:] += wt * (1.0 - h[:n - ss])
            g[:ss] -= wt * h[ss:]
    return g

def smooth_obj(x, n, alpha, mu):
    r = compute_r_s(x)
    m = float(r.max())
    z = alpha * (r - m)
    zmax = float(z.max())
    val = m + (zmax + np.log(np.sum(np.exp(z - zmax)))) / alpha
    if mu != 0.0:
        s = x.sum() - 0.5 * float(n)
        val += 0.5 * mu * s * s
    return float(val)

def smooth_grad(x, n, alpha, mu):
    r = compute_r_s(x)
    m = float(r.max())
    z = alpha * (r - m)
    zmax = float(z.max())
    w = np.exp(z - zmax)
    w /= w.sum()
    g = grad_weights(x, n, w)
    if mu != 0.0:
        g += mu * (x.sum() - 0.5 * float(n))
    return g

def lbfgs_polish(h, n, deadline, maxiter=70):
    h = project_feasible(h, n)
    c = c_from_r(compute_r_s(h), n)
    bounds = [(0.0, 1.0)] * n
    alphas_mus = ((20.0, 1e-4), (80.0, 1e-3), (400.0, 0.02), (2000.0, 0.5))
    for alpha, mu in alphas_mus:
        if time.time() > deadline - 1.0:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad, args=(n, alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-13,
                                    'gtol': 1e-7, 'maxfun': maxiter * 20})
            h_new = project_feasible(res.x, n)
            c_new = c_from_r(compute_r_s(h_new), n)
            if c_new < c:
                h = h_new
                c = c_new
        except Exception:
            pass
    return h, c

def max_after(r_s, deltas, L, t0, t1, t2):
    vals = r_s[None, :] + deltas[:, None] * L[None, :]
    m = vals.max(axis=1)
    d2 = deltas * deltas
    m = np.maximum(m, vals[:, t0] - 2.0 * d2)
    m = np.maximum(m, vals[:, t1] + d2)
    m = np.maximum(m, vals[:, t2] + d2)
    return m

def eval_side(r_s, L, dmax, sign, t0, t1, t2, old_max):
    if dmax <= 0.0:
        return 0.0, float('inf')
    deltas = sign * dmax * FRACS
    vals = max_after(r_s, deltas, L, t0, t1, t2)
    bi = int(np.argmin(vals))
    bd = float(deltas[bi])
    bv = float(vals[bi])
    if bv < old_max - 1e-7:
        mag = abs(bd)
        gap = max(0.003, dmax / 16.0)
        if gap > dmax:
            gap = dmax
        lo = max(0.0, mag - gap)
        hi = min(dmax, mag + gap)
        if hi - lo > 1e-13:
            rd = sign * np.linspace(lo, hi, 5)
            vals2 = max_after(r_s, rd, L, t0, t1, t2)
            bi2 = int(np.argmin(vals2))
            if vals2[bi2] < bv:
                bd = float(rd[bi2])
                bv = float(vals2[bi2])
    return bd, bv

def try_pair(h, r_s, i, j, trust, prefer_pos, both_sides):
    n = len(h)
    max_pos = min(trust, 1.0 - h[i], h[j])
    max_neg = min(trust, h[i], 1.0 - h[j])
    if max_pos <= 0.0 and max_neg <= 0.0:
        return False
    old_max = float(r_s.max())
    v = 1.0 - h
    hrev = h[::-1]
    L = np.zeros(2 * n - 1, dtype=float)
    L[n - 1 - i:2 * n - 1 - i] += v
    L[i:i + n] -= hrev
    L[n - 1 - j:2 * n - 1 - j] -= v
    L[j:j + n] += hrev
    t0 = n - 1
    d = j - i
    t1 = n - 1 + d
    t2 = n - 1 - d
    bd = 0.0
    bv = old_max
    if both_sides:
        if max_pos > 0.0:
            cand_bd, cand_bv = eval_side(r_s, L, max_pos, 1.0, t0, t1, t2, old_max)
            if cand_bv < bv:
                bd, bv = cand_bd, cand_bv
        if max_neg > 0.0:
            cand_bd, cand_bv = eval_side(r_s, L, max_neg, -1.0, t0, t1, t2, old_max)
            if cand_bv < bv:
                bd, bv = cand_bd, cand_bv
    else:
        if prefer_pos:
            if max_pos > 0.0:
                bd, bv = eval_side(r_s, L, max_pos, 1.0, t0, t1, t2, old_max)
            elif max_neg > 0.0:
                bd, bv = eval_side(r_s, L, max_neg, -1.0, t0, t1, t2, old_max)
        else:
            if max_neg > 0.0:
                bd, bv = eval_side(r_s, L, max_neg, -1.0, t0, t1, t2, old_max)
            elif max_pos > 0.0:
                bd, bv = eval_side(r_s, L, max_pos, 1.0, t0, t1, t2, old_max)
    if bd != 0.0 and bv < old_max - 1e-10:
        r_s += bd * L
        q = bd * bd
        r_s[t0] -= 2.0 * q
        r_s[t1] += q
        r_s[t2] += q
        h[i] += bd
        h[j] -= bd
        idx = np.argmin(r_s)
        if r_s[idx] < 0.0 and r_s[idx] > -1e-9:
            r_s[idx] = 0.0
        return True
    return False

def pair_scan(h, n, r_s, trust, K, random_pairs, deadline, rng):

    K = min(int(K), n)
    improved = False
    for _ in range(180):

            if time.time() > deadline:
                return improved
            i = int(low[rng.randint(K)])
            j = int(high[rng.randint(K)])
            if i == j:
                continue
            if try_pair(h, r_s, i, j, trust, g[i] <= g[j], False):
                accepts += 1
                improved = True
        if accepts == 0:
            for _ in range(350):
                if time.time() > deadline:
                    return improved
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                if try_pair(h, r_s, i, j, trust, g[i] <= g[j], False):

def pair_polish(h, n, trusts, deadline, rng):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_r = compute_r_s(h)
    best_c = c_from_r(best_r, n)

                break
    return best_h, best_r, best_c

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return project_feasible(h_old.copy(), n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)

def make_candidate(h_base, n_base, n_new, scale, rng):
    h = interpolate_h(h_base, n_new)
    if scale > 0.0:
        h = project_feasible(h + rng.normal(0.0, scale, size=n_new), n_new)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(int(seed) % 2147483647)
    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel().copy()
        n0 = int(h0.size)
    except Exception:
        n0 = 650
        h0 = np.full(n0, 0.5)
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_n = n0
    best_c = compute_c5(best_h, best_n)

    if budget_s > 350.0:
        main_end = min(t0 + 0.45 * budget_s, deadline - max(220.0, 0.22 * budget_s))
        if main_end > t0 + 30.0:
            h, r, c = pair_polish(best_h, best_n, TRUSTS, main_end, rng)
            if c < best_c:
                best_h = h
                best_n = len(h)
                best_c = c

    if budget_s > 700.0 and best_c > 0.38075 and time.time() < deadline - 240.0:
        cand_specs = []
        cand_specs.append((best_n, 0.025))
        n_coarse = max(350, best_n - 80)
        if n_coarse != best_n:
            cand_specs.append((n_coarse, 0.025))
        n_fine = min(950, best_n + 80)
        if n_fine != best_n:
            cand_specs.append((n_fine, 0.055))
        for n_new, scale in cand_specs:
            remain = deadline - time.time()
            if remain < 150.0:
                break
            alloc = min(120.0, max(40.0, remain * 0.38))
            if alloc < 45.0:
                break
            h_c = make_candidate(best_h, best_n, n_new, scale, rng)
            d_lb = min(time.time() + min(28.0, alloc * 0.30), deadline - alloc - 10.0)
            h_c, _ = lbfgs_polish(h_c, n_new, d_lb, maxiter=55)
            d_pp = min(time.time() + alloc, deadline - 90.0)
            h_c, r_c, c_c = pair_polish(h_c, n_new, TRUSTS, d_pp, rng)
            if c_c < best_c:
                best_h = h_c
                best_n = n_new
                best_c = c_c

    if time.time() < deadline - 4.0:
        h, r, c = pair_polish(best_h, best_n, TRUSTS, deadline - 1.0, rng)
        if c < best_c:
            best_h = h
            best_n = len(h)
            best_c = c

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
