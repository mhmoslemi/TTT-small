            if s < 1e-14:

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()

        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

                    msk = (jmn >= 0) & (jmn < n)
                    if msk.any():
                        delta[msk] += h[jmn[msk]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - i)
                    if 0 <= k1 < 2 * n - 1:
                        delta[k1] += 1.0
                    k2 = n - 1 + (i - j)
                    if 0 <= k2 < 2 * n - 1:
                        delta[k2] += 1.0
                    new_corr = corr + delta
                    nc = new_corr.max() * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr = new_corr
                        cur_c = nc
                        ones.discard(j)
                        ones.add(i)
                        zeros.discard(i)
                        zeros.add(j)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: SLSQP from parent
    p1 = t0 + 0.10 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [500, 5000, 50000, 500000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 1000, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Binary local search (deterministic swaps)
    p2 = t0 + 0.30 * (budget_s - 5)
    hb, cb = fast_binary_search(best_h, n, p2, max_sweeps=500, n_restarts=5, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: Simulated annealing for escape
    p3 = t0 + 0.55 * (budget_s - 5)
    hb, cb = sa_binary_search(best_h, n, p3, n_restarts=3, seed=seed + 10)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 4: SLSQP polish from binary
    p4 = t0 + 0.65 * (budget_s - 5)
    if time.time() < p4:
        hp = project_feasible(best_h + 0.03 * rng.randn(n), n)
        for alpha in [100000, 1000000]:
            if time.time() > p4:
                break
            hp = slsqp_smooth(hp, n, alpha, 1500, p4)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()

    # Phase 5: Extended deterministic + SA on best
    p5 = t0 + 0.85 * (budget_s - 5)
    hb, cb = fast_binary_search(best_h, n, p5, max_sweeps=2000, n_restarts=5, seed=seed + 20)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    if time.time() < p5 + 30:
        hb, cb = sa_binary_search(best_h, n, min(p5 + 30, dl - 20), n_restarts=2, seed=seed + 30)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 6: Final SLSQP + binary polish
    if time.time() < dl - 10:
        h = slsqp_smooth(best_h.copy() + 0.02 * rng.randn(n), n, 5000000, 2000, dl - 8)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if time.time() < dl - 3:
            hb, cb = fast_binary_search(best_h, n, dl, max_sweeps=500, n_restarts=3, seed=seed + 40)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
