def compute_c5(h, n):

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_S(h, n):
    m_len = 2 * n - 1
    S = np.zeros((n, m_len))
    for m in range(m_len):
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            S[0:nv, m] = (1.0 - h[k:n])
            S[k:n, m] += (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            S[kk:n, m] = (1.0 - h[0:nv])
            S[0:nv, m] += (-h[kk:n])
    return S

def smooth_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        w = np.exp(s - mx)
    else:
        w = np.exp(s)
    w /= w.sum()
    return w

    c = compute_c5_vec(h, n)
    w = smooth_max_w(c, alpha)
    S = compute_S(h, n)
    dx = 2.0 / n
    return (S * (w * dx))[:, :].sum(axis=1)

    return float(smooth_max_val(c, alpha) + mu * (x.sum() - n / 2.0) ** 2)

    g += 2.0 * mu * (x.sum() - n / 2.0)

    return float(x.sum() - n / 2.0)

def slq_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [100, 500, 2000, 10000, 50000, 200000]
    mus = [1.0, 5.0, 20.0, 50.0, 100.0, 200.0]
    for idx, (alpha, mu) in enumerate(zip(alphas, mus)):
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_iters * 15, 'maxls': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=50000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slq_eq, 'jac': slq_jac, 'args': (n,)}]
    try:
        res = minimize(slq_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def fast_pair_search(h, n, deadline):
    best_c = compute_c5(h, n)
    c = compute_c5_vec(h, n).copy()
    dx = 2.0 / n
    m_len = 2 * n - 1
    S = compute_S(h, n)
    grade = S * dx
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                if h[i] + d > 1.0 + 1e-12:
                    continue
                gi = grade[i]
                hi_val = h[i]
                for j in range(n):
                    if i == j:
                        continue
                    if h[j] - d < -1e-12:
                        continue
                    diff = gi - grade[j]
                    c_new_max = np.max(c + d * diff)
                    if c_new_max < best_c - 1e-15:
                        h[i] += d
                        h[j] -= d
                        c += d * diff
                        best_c = c_new_max
                        improved = True
                        # Recompute S (O(n^2), fast for n=300)
                        S = compute_S(h, n)
                        grade = S * dx
                        gi = grade[i]
                if time.time() > deadline:
                    break
            if not improved:
                break
    # Also try reverse moves
    for d in deltas[:5]:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                if h[i] - d < -1e-12:
                    continue
                gi = grade[i]
                for j in range(n):
                    if i == j:
                        continue
                    if h[j] + d > 1.0 + 1e-12:
                        continue
                    diff = gi - grade[j]
                    c_new_max = np.max(c - d * diff)
                    if c_new_max < best_c - 1e-15:
                        h[i] -= d
                        h[j] += d
                        c -= d * diff
                        best_c = c_new_max
                        improved = True
                        S = compute_S(h, n)
                        grade = S * dx
                        gi = grade[i]
                if time.time() > deadline:
                    break
            if not improved:
                break
    return h, best_c

def block_search(h, n, deadline):
    best_c = compute_c5(h, n)
    c = compute_c5_vec(h, n).copy()
    dx = 2.0 / n
    S = compute_S(h, n)
    grade = S * dx
    block_sizes = [2, 3, 4, 5]
    deltas = [0.005, 0.002, 0.001, 0.0005]
    for bs in block_sizes:
        for d in deltas:
            while time.time() < deadline:
                improved = False
                for i in range(n - bs + 1):
                    if time.time() > deadline:
                        break
                    if h[i:i+bs].max() + d > 1.0 + 1e-12:
                        continue
                    for j in range(n - bs + 1):
                        if j >= i and j < i + bs:
                            continue
                        if h[j:j+bs].min() - d < -1e-12:
                            continue
                        diff = grade[i:i+bs].sum(axis=0) - grade[j:j+bs].sum(axis=0)
                        c_new_max = np.max(c + d * diff)
                        if c_new_max < best_c - 1e-14:
                            h[i:i+bs] += d
                            h[j:j+bs] -= d
                            c += d * diff
                            best_c = c_new_max
                            improved = True
                            S = compute_S(h, n)
                            grade = S * dx
                if not improved:
                    break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300
    
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: L-BFGS-B (30% budget)
    t1 = t0 + 0.30 * (budget_s - 5)
    best_h, best_c = lbfgs_phase(best_h, n, t1, max_iters=200)
    
    # Phase 2: Fast pair search (35% budget)
    t2 = t0 + 0.65 * (budget_s - 5)
    best_h, best_c = fast_pair_search(best_h, n, t2)
    
    # Phase 3: L-BFGS-B polish (10% budget)
    if time.time() < t0 + 0.78 * (budget_s - 5):
        t3 = t0 + 0.78 * (budget_s - 5)
        h3, c3 = lbfgs_phase(best_h, n, t3, max_iters=150)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: More pair search (10% budget)
    t4 = t0 + 0.88 * (budget_s - 5)
    best_h, best_c = fast_pair_search(best_h, n, t4)
    
    # Phase 5: Block search (5% budget)
    if time.time() < t0 + 0.93 * (budget_s - 5):
        t5 = t0 + 0.93 * (budget_s - 5)
        best_h, best_c = block_search(best_h, n, t5)
    
    # Phase 6: Final SLSQP + pair search
    if time.time() < deadline - 30:
        h6, c6 = slsqp_phase(best_h, n, min(time.time() + 25, deadline - 10), alpha=200000, maxiter=1000)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    if time.time() < deadline - 5:
        best_h, best_c = fast_pair_search(best_h, n, deadline - 2)
    
    # Final: try perturbing and re-optimizing
    if time.time() < deadline - 30:
        h_pert = best_h + rng.normal(0, 0.02, n)
        h_pert = project_feasible(h_pert, n)
        h_pert, c_pert = lbfgs_phase(h_pert, n, min(time.time() + 15, deadline - 15), max_iters=100)
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert
        if time.time() < deadline - 3:
            best_h, best_c = fast_pair_search(best_h, n, deadline - 1)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
