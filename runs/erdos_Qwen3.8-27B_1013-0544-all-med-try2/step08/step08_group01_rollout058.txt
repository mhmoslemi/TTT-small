def fast_pair_search(h, n, deadline, radius=50):
    """Fast local search using O(n) update per pair."""
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx  # current correlation values
    best_c = c.max()
    
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            # For each pair (i, j) with |i-j| <= radius
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d > 1.0 or h[j] - d < 0.0:
                        continue
                    # Compute Δc for all k
                    # Δc[k] * dx = -d * (h[i+k-(n-1)] + h[i-k+(n-1)] - h[j+k-(n-1)] - h[j-k+(n-1)])
                    # (with valid index checks)
                    delta = compute_delta_c(h, i, j, d, n)
                    new_max = (c + delta).max()
                    if new_max < best_c - 1e-15:
                        h[i] += d
                        h[j] -= d
                        c += delta
                        best_c = new_max
                        improved = True
                    # Try h[i] -= d, h[j] += d
                    if h[i] - d < 0.0 or h[j] + d > 1.0:
                        continue
                    delta = compute_delta_c(h, j, i, d, n)
                    new_max = (c + delta).max()
                    if new_max < best_c - 1e-15:
                        h[i] -= d
                        h[j] += d
                        c += delta
                        best_c = new_max
                        improved = True
            if not improved:
                break
    return h, best_c

def _delta_corr(h, i, j, d, n):

    k = np.arange(2 * n - 1)
    
    # ∂corr[m]/∂h[i] for all m:
    # (1-h[i+m-m_idx]) * [0≤i+m-m_idx<n] - h[i-m+m_idx] * [0≤i-m+m_idx<n]
    a = i + k - m_idx  # i+m-(n-1)
    va = (a >= 0) & (a < n)
    b = i - k + m_idx  # i-m+(n-1)
    vb = (b >= 0) & (b < n)
    
    gi = np.zeros(2 * n - 1)
    gi[va] += (1.0 - h[a[va]])
    gi[vb] -= h[b[vb]]

    c_idx = j + k - m_idx

    gj[vc] += (1.0 - h[c_idx[vc]])
    gj[vd] -= h[d_idx[vd]]

```python
def fast_local_search(h, n, deadline, radius=50, recompute_every=100):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c = corr * dx
    best_c = c.max()
    count = 0
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        delta = _delta_corr(h, i, j, d, n)
                        nm = (c + delta).max()
                        if nm < best_c - 1e-12:
                            h[i] += d; h[j] -= d
                            c += delta
                            best_c = nm
                            improved = True
                            count += 1
                            if count % recompute_every == 0:
                                c = np.correlate(h, 1-h, mode='full') * dx
                                best_c = c.max()
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        delta = _delta_corr(h, j, i, d, n)
                        nm = (c + delta).max()
                        if nm < best_c - 1e-12:
                            h[i] -= d; h[j] += d
                            c += delta
                            best_c = nm
                            improved = True
                            count += 1
                            if count % recompute_every == 0:
                                c = np.correlate(h, 1-h, mode='full') * dx
                                best_c = c.max()
            if not improved:
                break
    # Final recompute
    c = np.correlate(h, 1-h, mode='full') * dx
    best_c = c.max()
    return h, best_c
```

For a binary sequence of length n with n/2 ones, the correlation at shift k (in the full mode) is the number of positions i where h[i]=1 and h[i+k-(n-1)]=0 (with valid indices).

To minimize the maximum, we want the "inversion count" (number of 1→0 transitions at each relative position) to be as uniform as possible.

This is related to the concept of a "balanced incomplete block design" or "difference set". A difference set in Z_n is a subset D of size k such that every non-zero element of Z_n can be written as d1-d2 in exactly λ ways. This gives a perfectly uniform autocorrelation.

However, we're not working in Z_n (cyclic) but in the linear (full) correlation. So the boundary effects matter.

For the linear case, a related concept is a "Sidon set" or "B2 sequence" where all differences are distinct (or have bounded multiplicity).

Actually, for our problem, we want to minimize the maximum number of (1,0) pairs at any linear shift. This is equivalent to: for each shift k, minimize the number of i such that h[i]=1 and h[i+k]=0 (for the valid range of i).

If h is the indicator of a set S of size n/2, then for shift k, the count is |S ∩ (S-k)^c ∩ [valid range]| = |S ∩ [valid]| - |S ∩ (S+k) ∩ [valid]

For the center shift (k=0): count = 0 (since S ∩ S^c = ∅)
For shift k: count ≈ |S| - |S ∩ (S+k)| = n/2 - |S ∩ (S+k)|

To minimize the maximum, we want |S ∩ (S+k)| to be as large as possible for all k, i.e., the autocorrelation of S should be as uniform and high as possible.
