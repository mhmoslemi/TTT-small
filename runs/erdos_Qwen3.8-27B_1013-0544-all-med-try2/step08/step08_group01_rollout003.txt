def binary_local_search(h, n, deadline):
    # Convert to binary
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        # Remove excess ones (remove the ones that are most "isolated")
        idx = np.where(h_bin == 1)[0]
        # Sort by degree (number of neighboring ones)
        degrees = np.array([sum(h_bin[max(0,i-10):i+11]) for i in idx])
        order = np.argsort(degrees)
        for i in order[:n_ones - target]:
            h_bin[idx[i]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        # Add ones at positions that are most "surrounded" by zeros
        for i in idx[:target - n_ones]:
            h_bin[i] = 1
    
    dx = 2.0 / n
    best_c = float(np.max(np.correlate(h_bin, 1-h_bin, mode='full')) * dx)
    
    # Greedy flipping
    rng = np.random.RandomState(42)
    for _ in range(10000):
        if time.time() > deadline:
            break
        # Find a random 1 and 0 to flip
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        # Try flip: h[i] 1→0, h[j] 0→1
        h_bin[i] = 0
        h_bin[j] = 1
        c = float(np.max(np.correlate(h_bin, 1-h_bin, mode='full')) * dx)
        if c < best_c:
            best_c = c
        else:
            h_bin[i] = 1
            h_bin[j] = 0
    
    return h_bin, best_c

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c_max = c.max()
    cs = alpha * (c - c_max)
    mx = cs.max()
    w = np.exp(cs - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

        if wm < 1e-14:

            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])

        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    cs = alpha * (c - c_max)
    mx = cs.max()
    w = np.exp(cs - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        wm = wm * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def adam_opt(h, n, deadline, alpha, lr=0.003):
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    best_c = compute_c5(h, n)
    h = h.copy()
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    while time.time() < deadline:
        t += 1
        g = compute_grad(h, n, alpha)
        m_v = beta1 * m_v + (1.0 - beta1) * g
        v_v = beta2 * v_v + (1.0 - beta2) * g * g
        m_hat = m_v / (1.0 - beta1 ** t)
        v_hat = v_v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        if t % 200 == 0:
            lr *= 0.95
    return h, best_c

def _slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def _slsqp_eq(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_eq_jac(x):
    return np.ones(len(x))

def slsqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'maxfun': 500000})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def binary_search(h, n, deadline, seed=42):
    dx = 2.0 / n
    h_bin = (h > 0.5).astype(float).copy()
    target = n // 2
    n_ones = int(h_bin.sum())
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        rng = np.random.RandomState(seed)
        order = ones[rng.permutation(len(ones))]
        for i in order[:n_ones - target]:
            h_bin[i] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        rng = np.random.RandomState(seed)
        order = zeros[rng.permutation(len(zeros))]
        for i in order[:target - n_ones]:
            h_bin[i] = 1
    best_c = float(np.max(np.correlate(h_bin, 1 - h_bin, mode='full')) * dx)
    rng = np.random.RandomState(seed + 1)
    T = 0.01
    for step in range(20000):
        if time.time() > deadline:
            break
        T *= 0.9995
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h_bin[i] = 0
        h_bin[j] = 1
        c_new = float(np.max(np.correlate(h_bin, 1 - h_bin, mode='full')) * dx)
        if c_new < best_c or rng.random() < np.exp(-(c_new - best_c) / max(T, 1e-6)):
            best_c = min(best_c, c_new)
        else:
            h_bin[i] = 1
            h_bin[j] = 0
    return h_bin, best_c

def fine_tune(h, n, deadline, step=0.001):
    best_c = compute_c5(h, n)
    h = h.copy()
    for it in range(50000):
        if time.time() > deadline:
            break
        improved = False
        n_try = min(n, 200)
        for _ in range(n_try):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                h[i] += step
                h[j] -= step
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                else:
                    h[i] -= step
                    h[j] += step
            if h[i] - step >= 0.0 and h[j] + step <= 1.0:
                h[i] -= step
                h[j] += step
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                else:
                    h[i] += step
                    h[j] -= step
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    deadline = t0 + budget_s - 10

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200

    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_n = n

    # Phase 1: Adam optimization with increasing alpha
    alpha_configs = [(100, 0.01), (500, 0.005), (2000, 0.003), (10000, 0.002), (50000, 0.001)]
    for alpha, lr in alpha_configs:
        tl = min(time.time() + 80, deadline - 300)
        if time.time() > tl:
            break
        h_new, c_new = adam_opt(h, n, tl, alpha, lr)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new

    # Phase 2: SLSQP polish
    if time.time() < deadline - 250:
        tl = min(time.time() + 120, deadline - 200)
        h2, c2 = slsqp_polish(h, n, tl, alpha=1000000, maxiter=5000)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
        h = h2

    # Phase 3: Binary local search
    if time.time() < deadline - 200:
        tl = min(time.time() + 100, deadline - 150)
        h_bin, c_bin = binary_search(h, n, tl, seed=seed)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()

    # Phase 4: Interpolate to larger n and optimize
    for n_try in [800, 1200]:
        if time.time() > deadline - 150:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(80, (deadline - time.time() - 80) * 0.5)
        if alloc < 30:
            break
        h_t, c_t = adam_opt(h_interp, n_try, time.time() + alloc, 5000, 0.002)
        if time.time() < deadline - 60:
            h_t, c_t = slsqp_polish(h_t, n_try, min(time.time() + 40, deadline - 50), alpha=500000, maxiter=2000)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t.copy()
            best_n = n_try

    # Phase 5: Fine-tune on best solution
    if time.time() < deadline - 30:
        tl = min(time.time() + 50, deadline - 15)
        h_f, c_f = fine_tune(best_h, best_n, tl, step=0.0005)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()

    # Phase 6: Final SLSQP
    if time.time() < deadline - 8:
        tl = min(time.time() + 30, deadline - 3)
        h_f, c_f = slsqp_polish(best_h, best_n, tl, alpha=2000000, maxiter=3000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
