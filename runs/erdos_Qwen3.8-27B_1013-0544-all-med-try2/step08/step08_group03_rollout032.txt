def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    dx = 2.0 / n
    if k >= 0:
        grad[:nv] += (1.0 - h[k:]) * dx
        grad[k:] -= h[:nv] * dx
    else:
        kk = -k
        grad[kk:] += (1.0 - h[:nv]) * dx
        grad[:nv] -= h[kk:] * dx
    return grad

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 200.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, top_k=100):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    n_pts = len(x)
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n_pts)
    for i in top_idx:
        if w[i] < 1e-14:
            continue
        k = int(i) - (n - 1)
        grad += w[i] * grad_lag(x, n, k)
    s = x.sum() - n / 2.0
    grad += 400.0 * s
    return grad

def targeted_polish(h, n, deadline, max_iter=5000):

    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 50000.0

        if time.time() > deadline:
            break

        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()

        top_k = 50
        top_idx = np.argsort(w)[-top_k:]
        g = np.zeros(n)
        for i in top_idx:
            if w[i] < 1e-14:
                continue
            k = int(i) - (n - 1)
            g += w[i] * grad_lag(h, n, k)
        # Remove mean (project onto tangent space of constraint)
        g = g - g.mean()
        
        # Find best pair move: minimize g[i] - g[j] subject to feasibility
        # We want to decrease h[argmin g] and increase h[argmax g]
        # But we should try multiple candidates
        
        # Sort by gradient
        order = np.argsort(g)
        
        # Try top few moves
        tried = 0
        for a in range(min(10, n)):
            for b in range(n - 1 - min(10, n), n):
                i = order[a]  # low gradient (should decrease)
                j = order[b]  # high gradient (should increase)

                        tried += 1
                    else:

                        h[j] -= d
