import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return float(val)

def _grad(x, n, alpha):
    return grad_smooth_max(x, n, alpha)

def lbfgs_optimize(h, n, deadline, alphas=None, max_iters=300):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 50000]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        remaining = deadline - time.time() - 3
        if remaining < 5:
            break
        try:
            res = minimize(_obj, h, jac=_grad, args=(n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iters * 5,
                                   'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    
    def slsqp_obj(x, n, a):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = a * (c - m)
        mx = sh.max()
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / a
    
    def slsqp_constraint(x, n):
        return x.sum() - n / 2.0

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n

            grad[k:n] += wm * (-h[0:nv])

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])

    return grad

            h_new = project_feasible(res.x, n)

    # Phase 1: L-BFGS
    t1 = min(t0 + 150, deadline - 500)
    best_h, best_c = lbfgs_opt(best_h, n_init, t1, [50, 200, 1000, 5000, 50000], 200)

    # Phase 2: SLSQP
    if time.time() < deadline - 400:
        t2 = min(time.time() + 60, deadline - 400)
        for a in [200000, 1000000, 5000000]:
            if time.time() > t2:
                break
            h2, c2 = slsqp_opt(best_h, best_n, min(time.time() + 25, t2), alpha=a, maxiter=1000)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Gradient-guided search
    if time.time() < deadline - 300:
        t3 = min(time.time() + 150, deadline - 300)
        h3, c3 = grad_guided_search(best_h, best_n, t3)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Pair search with decreasing steps
    if time.time() < deadline - 150:
        t4 = min(time.time() + 250, deadline - 150)
        best_h, best_c = pair_search(best_h, best_n, t4,
                                      [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                      radius=40)

    # Phase 5: Try n=400
    if time.time() < deadline - 80:
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, 400)
        h400 = np.interp(x_new, x_old, best_h)
        h400 = project_feasible(h400, 400)
        alloc = min(80, deadline - time.time() - 40)
        if alloc > 20:
            h400, c400 = lbfgs_opt(h400, 400, time.time() + alloc * 0.5, [200, 2000, 20000], 150)
            if time.time() < deadline - 20:
                h400, c400 = pair_search(h400, 400, time.time() + min(20, deadline - time.time() - 10),
                                          [0.01, 0.005, 0.001, 0.0005], radius=30)
            if c400 < best_c:
                best_c = c400
                best_h = h400
                best_n = 400

    # Phase 6: Try n=250
    if time.time() < deadline - 40:
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, 250)
        h250 = np.interp(x_new, x_old, best_h)
        h250 = project_feasible(h250, 250)
        alloc = min(35, deadline - time.time() - 20)
        if alloc > 15:
            h250, c250 = lbfgs_opt(h250, 250, time.time() + alloc * 0.6, [200, 2000], 150)
            if time.time() < deadline - 10:
                h250, c250 = pair_search(h250, 250, deadline - 5,
                                          [0.01, 0.005, 0.001, 0.0005], radius=30)
            if c250 < best_c:
                best_c = c250
                best_h = h250
                best_n = 250

    # Phase 7: Final polish
    if time.time() < deadline - 2:
        best_h, best_c = pair_search(best_h, best_n, deadline - 1,
                                      [0.001, 0.0005, 0.0002, 0.0001], radius=30)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
