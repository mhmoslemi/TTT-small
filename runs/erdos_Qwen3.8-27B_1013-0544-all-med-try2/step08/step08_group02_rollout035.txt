def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)

        wm = w[m_idx]

            grad[:nv] += wm * (1.0 - x[k:])

            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])

Actually, let me try yet another approach. What about using **CVXPY** to solve the QP relaxation?

The problem is:
minimize T
s.t. for all k: sum_{valid i} h[i]*(1-h[i+k]) ≤ T
     sum h[i] = n/2
     0 ≤ h[i] ≤ 1

The constraints are quadratic (bilinear). CVXPY can handle QPs but the bilinear terms h[i]*h[i+k] make it non-convex.

However, we can use a different formulation. Note that:
h[i]*(1-h[i+k]) = h[i] - h[i]*h[i+k]

So the constraint becomes:
sum_{valid i} h[i] - sum_{valid i} h[i]*h[i+k] ≤ T

The term sum h[i] over the valid range is a linear function of h. The term sum h[i]*h[i+k] is a quadratic form.

For a fixed k, let's write it as:
(1^T A_k h) - h^T B_k h ≤ T

where A_k is a selection matrix and B_k is a matrix with B_k[i,i+k] = 1/2 and B_k[i+k,i] = 1/2 (to make it symmetric).

This is a non-convex QP (since B_k is not negative semidefinite in general).

So CVXPY won't directly help here. Let me stick with the iterative approach.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            h[h <= 0.0] = 0.0
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:k+nv])
            grad[k:k+nv] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:kk+nv] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:kk+nv])
    return grad

def pen_obj(x, n, alpha, mu):
    val = smooth_max_val(x, n, alpha)
    s = x.sum() - n / 2.0
    return float(val + mu * s * s)

def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def to_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt != target:
        if cnt > target:
            idx = np.where(h_bin == 1)[0]
            rem = rng.choice(idx, size=cnt - target, replace=False)
            h_bin[rem] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            add = rng.choice(idx, size=target - cnt, replace=False)
            h_bin[add] = 1
    return h_bin

def binary_ls(h_bin, n, deadline, rng, max_iters=200000):
    """Binary local search."""
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    cur = h_bin.copy()
    stall = 0
    temp = 0.05
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Try batch of random swaps
        n_try = min(len(ones) * len(zeros), 2000)
        best_move = None
        best_mc = compute_c5(cur, n)
        for _ in range(n_try):
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            cur[a] = 0; cur[b] = 1
            mc = compute_c5(cur, n)
            cur[a] = 1; cur[b] = 0
            if mc < best_mc - 1e-15:
                best_mc = mc
                best_move = (a, b)
        if best_move is not None:
            a, b = best_move
            cur[a] = 0; cur[b] = 1
            if best_mc < best_c - 1e-15:
                best_c = best_mc
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
            if stall % 500 == 0:
                temp *= 0.95
            # Accept a random (possibly worse) move to escape
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            cur[a] = 0; cur[b] = 1
            mc = compute_c5(cur, n)
            if mc - best_c > temp:
                cur[a] = 1; cur[b] = 0
    return best_h, best_c

def lbfgs_polish(h, n, deadline, max_iters=500):
    """L-BFGS-B continuous optimization."""
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in [(1000, 1.0), (10000, 10.0), (100000, 100.0), (1000000, 1000.0)]:
        if time.time() > deadline:
            break
        try:
            res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-20,
                                   'gtol': 1e-18, 'maxfun': max_iters * 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

**Key improvement: targeted swaps near bottleneck shifts**

If the current max is at shift k*, then we want to reduce c[k*]. A swap (a:1→0, b:0→1) reduces c[k*] if:
- h[a+k*]=0 and h[a-k*]=1 (removing a 1 that contributes to c[k*])
- h[b+k*]=1 and h[b-k*]=0 (adding a 0 that doesn't contribute to c[k*])

More precisely, from our earlier formula:
Δc[k*] = h[a+k*] - h[a-k*] - h[b+k*] + h[b-k*]

We want this to be negative, i.e., h[a+k*] + h[b-k*] < h[a-k*] + h[b+k*].

For binary h, this means:
- Best case: h[a+k*]=0, h[b-k*]=0, h[a-k*]=1, h[b+k*]=1 → Δ = -2
- Good: h[a+k*]=0, h[b-k*]=0, and (h[a-k*]=1 or h[b+k*]=1) → Δ = -1
- Bad: h[a+k*]=1, h[b-k*]=1, and (h[a-k*]=0 or h[b+k*]=0) → Δ = +1
- Worst: h[a+k*]=1, h[b-k*]=1, h[a-k*]=0, h[b+k*]=0 → Δ = +2

So for targeting, we should:
1. Find the bottleneck shifts k*
2. For each "1" position a, check if h[a+k*]=0 (removing it helps)
3. For each "0" position b, check if h[b+k*]=1 (adding it doesn't hurt)
4. Try swaps where both conditions hold

        if best_move is not None:
            a, b = best_move
            cur[a] = 0; cur[b] = 1
            if best_mc < best_c - 1e-15:
                best_c = best_mc
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
            if stall % 300 == 0:
                temp *= 0.93
                n_batch = max(200, n_batch - 50)
            if stall > 2000:
                n_kick = rng.randint(3, 8)
                for _ in range(n_kick):
                    o = np.where(cur == 1)[0]
                    z = np.where(cur == 0)[0]
                    if len(o) == 0 or len(z) == 0:
                        break
                    a = o[rng.randint(len(o))]
                    b = z[rng.randint(len(z))]
                    cur[a] = 0; cur[b] = 1
                stall = 0
    return best_h, best_c

def lbfgs_polish(h, n, deadline, max_iters=500):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 0.5), (2000, 2.0), (10000, 10.0),
               (50000, 50.0), (200000, 200.0), (1000000, 1000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-20,
                                   'gtol': 1e-18, 'maxfun': max_iters * 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.zeros(250)
        h_init[rng.choice(250, 125, replace=False)] = 1.0
        n_init = 250

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Binary local search on initial (90s)
    h_bin = to_binary(best_h, best_n, rng)
    t1 = min(t0 + 90, deadline - 500)
    h_b, c_b = binary_ls(h_bin, best_n, t1, rng)
    if c_b < best_c:
        best_c = c_b
        best_h = project_feasible(h_b, best_n)

    # Phase 2: Continuous polish (100s)
    t2 = min(time.time() + 100, deadline - 400)
    h_p, c_p = lbfgs_polish(best_h, best_n, t2, max_iters=800)
    if c_p < best_c:
        best_c = c_p
        best_h = h_p

    # Phase 3: Another binary round from polished solution (60s)
    if time.time() < deadline - 350:
        h_bin2 = to_binary(best_h, best_n, rng)
        t3 = min(time.time() + 60, deadline - 300)
        h_b2, c_b2 = binary_ls(h_bin2, best_n, t3, rng)
        if c_b2 < best_c:
            best_c = c_b2
            best_h = project_feasible(h_b2, best_n)

    # Phase 4: Multi-resolution exploration
    for n_small in [128, 192, 256]:
        if time.time() > deadline - 250:
            break
        h_small = upsample(best_h, best_n, n_small)
        h_sb = to_binary(h_small, n_small, rng)
        alloc = min(35, max(15, (deadline - time.time() - 200) // 3))
        t_sa = time.time() + alloc * 0.55
        h_s2, c_s2 = binary_ls(h_sb, n_small, t_sa, rng)
        h_back = upsample(h_s2, n_small, best_n)
        t_pol = time.time() + min(alloc * 0.35, 25)
        h_pol, c_pol = lbfgs_polish(h_back, best_n, t_pol, max_iters=400)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol

    # Phase 5: Try different n values
    for n_try in [best_n - 50, best_n, best_n + 50, best_n + 150]:
        if time.time() > deadline - 100:
            break
        if n_try < 50:
            continue
        h_t = upsample(best_h, best_n, n_try)
        alloc = min(30, max(12, (deadline - time.time() - 80) // 4))
        if alloc < 10:
            continue
        t_pol = time.time() + alloc
        h_nt, c_nt = lbfgs_polish(h_t, n_try, t_pol, max_iters=400)
        if c_nt < best_c:
            best_c = c_nt
            best_h = h_nt
            best_n = n_try

    # Phase 6: Final intensive polish
    if time.time() < deadline - 15:
        t6 = min(time.time() + 30, deadline - 5)
        h_f, c_f = lbfgs_polish(best_h, best_n, t6, max_iters=2000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Phase 7: Final binary refinement
    if time.time() < deadline - 5:
        h_fb = to_binary(best_h, best_n, rng)
        t7 = min(time.time() + 12, deadline - 3)
        h_f2, c_f2 = binary_ls(h_fb, best_n, t7, rng, max_iters=100000)
        if c_f2 < best_c:
            best_c = c_f2
            best_h = project_feasible(h_f2, best_n)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
