        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def default_start(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def compute_r(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(compute_r(h, n)) * 2.0 / n)

def top_indices(vals, k):
    k = int(min(k, vals.size))
    if k <= 0:
        return np.empty(0, dtype=int)
    if k >= vals.size:
        idx = np.argsort(-vals)
    else:
        idx = np.argpartition(-vals, k - 1)[:k]
    return idx[np.argsort(vals[idx])[::-1]]

def active_gradient(h, b, n, ord_):
    g = np.zeros(n)
    m = len(ord_)
    if m == 0:
        return g
    w = 1.0 / m
    m0 = n - 1
    for mm in ord_:
        s = int(mm) - m0
        if s == 0:
            g += w * (1.0 - 2.0 * h)
        elif s > 0:
            a = n - s
            if a > 0:
                g[s:] += w * b[:a]
                g[:a] -= w * h[s:]
        else:
            t = -s
            a = n - t
            if a > 0:
                g[:a] += w * b[t:]
                g[t:] -= w * h[:a]
    return g

def swap_newM(r, h, b, n, p, q, d):
    d.fill(0.0)
    br = b[::-1]
    d[p:p + n] -= br
    d[n - 1 - p:2 * n - 1 - p] += h
    d[q:q + n] += br
    d[n - 1 - q:2 * n - 1 - q] -= h
    d[n - 1 + p - q] += 1.0
    d[n - 1 + q - p] += 1.0
    d[n - 1] = 0.0
    r += d
    return float(np.max(r))

def random_pair(n, h, rng):
    while True:
        p = int(rng.randint(n))
        q = int(rng.randint(n))
        if p != q and h[p] > 0.5 and h[q] < 0.5:
            return p, q

def to_binary(h, n):
    out = np.zeros(n)
    out[np.argsort(-h)[:n // 2]] = 1.0
    return out

def binary_local(h, r, n, rng, deadline):
    b = 1.0 - h
    d = np.zeros(2 * n - 1)
    bh = h.copy()
    br = r.copy()
    bm = float(np.max(r))
    M = bm
    T = 0.3
    no = 0
    H = min(12, max(2, n // 8))
    while time.time() < deadline:
        if M < bm:
            bm = M
            bh = h.copy()
            br = r.copy()
        ord_ = top_indices(r, 80)
        g = active_gradient(h, b, n, ord_)
        best = M + 1.0
        bp = bq = -1
        os_ = g.copy()
        os_[h < 0.5] = 1e9
        zs = g.copy()
        zs[h > 0.5] = -1e9
        hi = top_indices(os_, H)
        lo = top_indices(-zs, H)
        for i in hi:
            if h[i] <= 0.5:
                continue
            for j in lo:
                if h[j] >= 0.5:
                    continue
                nm = swap_newM(r, h, b, n, i, j, d)
                if nm < best:
                    best = nm
                    bp, bq = i, j
                r -= d
        slist = []
        for mm in ord_[:16]:
            s = abs(int(mm) - (n - 1))
            if 0 < s < n and s not in slist:
                slist.append(s)
                if len(slist) >= 6:
                    break
        for s in slist:
            for p in range(n - s):
                if time.time() > deadline:
                    break
                a = h[p]
                c = h[p + s]
                if a == c:
                    continue
                p1 = p if a > 0.5 else p + s
                p0 = p + s if a > 0.5 else p
                nm = swap_newM(r, h, b, n, p1, p0, d)
                if nm < best:
                    best = nm
                    bp, bq = p1, p0
                r -= d
        for _ in range(50):
            if time.time() > deadline:
                break
            p, q = random_pair(n, h, rng)
            nm = swap_newM(r, h, b, n, p, q, d)
            if nm < best:
                best = nm
                bp, bq = p, q
            r -= d
        if bp >= 0 and best < M - 1e-12:
            swap_newM(r, h, b, n, bp, bq, d)
            h[bp] = 0.0
            h[bq] = 1.0
            b[bp] = 1.0
            b[bq] = 0.0
            r[n - 1] = 0.0
            M = best
            no = 0
            T = max(0.01, 0.94 * T)
            if M < bm:
                bm = M
                bh = h.copy()
                br = r.copy()
        else:
            no += 1
            if bp >= 0 and best - M < T and rng.random() < 0.15:
                swap_newM(r, h, b, n, bp, bq, d)
                h[bp] = 0.0
                h[bq] = 1.0
                b[bp] = 1.0
                b[bq] = 0.0
                r[n - 1] = 0.0
                M = best
                no = 0
                if M < bm:
                    bm = M
                    bh = h.copy()
                    br = r.copy()
            if no > 12:
                for _ in range(4):
                    if time.time() > deadline:
                        break
                    p, q = random_pair(n, h, rng)
                    nm = swap_newM(r, h, b, n, p, q, d)
                    if nm - M < 1.0:
                        h[p] = 0.0
                        h[q] = 1.0
                        b[p] = 1.0
                        b[q] = 0.0
                        r[n - 1] = 0.0
                        M = nm
                        if M < bm:
                            bm = M
                            bh = h.copy()
                            br = r.copy()
                    else:
                        r -= d
                no = 0
                T = min(1.0, 2.0 * T)
            else:
                T = max(0.01, 0.98 * T)
    if M < bm:
        bh = h.copy()
        br = r.copy()
    return bh, br

def binary_polish(h, r, n, deadline, max_s):
    b = 1.0 - h
    d = np.zeros(2 * n - 1)
    bh = h.copy()
    br = r.copy()
    bm = float(np.max(r))
    M = bm
    max_s = min(max_s, n - 1)
    passes = 0
    while time.time() < deadline and passes < 30:
        passes += 1
        improved = False
        for s in range(1, max_s + 1):
            if time.time() > deadline:
                break
            for p in range(n - s):
                a = h[p]
                c = h[p + s]
                if a == c:
                    continue
                p1 = p if a > 0.5 else p + s
                p0 = p + s if a > 0.5 else p
                nm = swap_newM(r, h, b, n, p1, p0, d)
                if nm < M - 1e-12:
                    h[p1] = 0.0
                    h[p0] = 1.0
                    b[p1] = 1.0
                    b[p0] = 0.0
                    r[n - 1] = 0.0
                    M = nm
                    improved = True
                    if M < bm:
                        bm = M
                        bh = h.copy()
                        br = r.copy()
                else:
                    r -= d
            if time.time() > deadline:
                break
        if not improved:
            break
    if M < bm:
        bh = h.copy()
        br = r.copy()
    return bh, br

def fw_refine(h0, n, deadline):
    h = project_bisect(h0.copy(), n)
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for K in (50, 120, 240):
        stall = 0
        while time.time() < deadline - 2:
            r = compute_r(h, n)
            c = float(np.max(r)) * dx
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
            b = 1.0 - h
            ord_ = top_indices(r, K)
            g = active_gradient(h, b, n, ord_)
            z = np.zeros(n)
            z[np.argpartition(g, n // 2)[:n // 2]] = 1.0
            d = z - h
            line_imp = False
            if float(np.max(np.abs(d))) > 1e-10:
                bt = 0.0
                btc = best_c
                for t in (1.0, 0.5, 0.25, 0.75, 0.125, 0.875, 0.375, 0.625):
                    ht = h + t * d
                    cc = float(np.max(compute_r(ht, n)) * dx)
                    if cc < btc - 1e-14:
                        btc = cc
                        bt = t
                if bt > 0.0 and btc < best_c - 1e-12:
                    h = project_bisect(h + bt * d, n)
                    best_c = btc
                    best_h = h.copy()
                    stall = 0
                    line_imp = True
            if line_imp:
                continue
            stall += 1
            imp = False
            for st in (0.2, 0.1, 0.05, 0.02, 0.01, 0.005):
                hp = project_bisect(h - st * g, n)
                cc = float(np.max(compute_r(hp, n)) * dx)
                if cc < best_c - 1e-12:
                    h = hp
                    best_c = cc
                    best_h = h.copy()
                    imp = True
                    break
            if imp:
                stall = 0
            if stall > 6:
                break
    best_h = project_bisect(best_h, n)
    return best_h, compute_c5(best_h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + max(20.0, float(budget_s) - 5.0)
    rng = np.random.RandomState(seed)
    try:
        par = np.asarray(initial_h_values, dtype=float).ravel()
    except Exception:
        par = default_start(500, rng)
    n = len(par)
    if n % 2:
        n -= 1
        par = par[:n]
    if n < 100:
        par = default_start(500, rng)
        n = 500
    h = project_bisect(par, n)
    bh = h.copy()
    bc = compute_c5(h, n)
    bn = n

    if time.time() < dl - 380:
        hb = to_binary(h, n)
        rb = compute_r(hb, n)
        rb[n - 1] = 0.0
        hb, rb = binary_local(hb, rb, n, rng, min(dl - 300, time.time() + 300))
        c = float(np.max(rb)) * 2.0 / n
        if c < bc:
            bh, bc, bn = hb.copy(), c, n
        if time.time() < dl - 240:
            hb = to_binary(hb, n)
            rb = compute_r(hb, n)
            rb[n - 1] = 0.0
            hb, rb = binary_polish(hb, rb, n, min(dl - 180, time.time() + 120), 150)
            c = float(np.max(rb)) * 2.0 / n
            if c < bc:
                bh, bc, bn = hb.copy(), c, n

    if time.time() < dl - 140:
        hf, cf = fw_refine(bh, bn, min(dl - 80, time.time() + 180))
        if cf < bc:
            bh, bc, bn = hf, cf, bn

    if time.time() < dl - 80:
        hb = to_binary(bh, bn)
        rb = compute_r(hb, bn)
        rb[bn - 1] = 0.0
        hb, rb = binary_local(hb, rb, bn, rng, min(dl - 50, time.time() + 55))
        hb, rb = binary_polish(hb, rb, bn, min(dl - 30, time.time() + 45), 120)
        c = float(np.max(rb)) * 2.0 / bn
        if c < bc:
            bh, bc, bn = hb.copy(), c, bn

    if time.time() < dl - 30:
        hf, cf = fw_refine(bh, bn, min(dl - 15, time.time() + 45))
        if cf < bc:
            bh, bc, bn = hf, cf, bn

    if time.time() < dl - 8:
        hb = to_binary(bh, bn)
        rb = compute_r(hb, bn)
        rb[bn - 1] = 0.0
        hb, rb = binary_polish(hb, rb, bn, min(dl - 4, time.time() + 15), 120)
        c = float(np.max(rb)) * 2.0 / bn
        if c < bc:
            bh, bc, bn = hb.copy(), c, bn

    bh = project_bisect(bh, bn)
    bc = compute_c5(bh, bn)
    return bh, bc, int(bn)
