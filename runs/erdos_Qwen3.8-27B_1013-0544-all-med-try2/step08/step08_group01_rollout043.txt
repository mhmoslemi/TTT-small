def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

    grad = np.zeros(n)

def nesterov_smooth(x, n, deadline, alpha, mu, L=None, max_iters=5000):

    x_prev = x.copy()
    t = 1.0
    
    if L is None:
        # Estimate L by running a few gradient steps
        g1 = smooth_max_grad(x, n, alpha, mu)
        g2 = smooth_max_grad(x + 0.01 * np.ones(n), n, alpha, mu)
        L = np.linalg.norm(g2 - g1) / 0.01 / np.linalg.norm(np.ones(n)) + 1e-6
        L = max(L, 1.0)
    
    step = 1.0 / L
    best_x = x.copy()
    best_c = compute_c5(x, n)

        x_new = np.clip(x_new, 0.0, 1.0)

    G has shape (top_m, n) where G[m, i] = ∂c[top_idx[m]]/∂h[i]
    Returns (i, j, d) or None.

    m = len(top_idx)
    c_top = c[top_idx]  # (m,)
    best_d = 0
    best_i = -1
    best_j = -1
    
    for i in range(n):
        # diff[mm] = G[mm, i] - G[mm, j] for all j
        diff = G[:, i] - G  # (m, n) via broadcasting
        mask = diff > 0  # (m, n)
        # d_max[mm, j] = (current_max - c_top[mm]) / diff[mm, j] if mask[mm, j] else inf
        with np.errstate(divide='ignore', invalid='ignore'):

        d_max_vec = np.min(candidate, axis=0)  # (n,)
        # Exclude j = i
        d_max_vec[i] = 0
        j = np.argmax(d_max_vec)
        if d_max_vec[j] > best_d:
            best_d = float(d_max_vec[j])
            best_i = i
            best_j = j

    return best_i, best_j, min(best_d * 0.9, 0.5)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[k:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[k:])
        else:
            kk = -k
            nv = n - kk
            grad[:nv] += wm * (1.0 - x[kk:])
            grad[kk:] += wm * (-x[:nv])
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def lbfgs_pass(x, n, deadline, alpha, mu, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, x, jac=smooth_max_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-16, 'maxfun': maxiter * 50, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return x.copy(), compute_c5(x, n)

def compute_shift_grad_matrix(h, n, top_m):
    """Compute G matrix (top_m x n) of partial derivatives for top shifts."""
    c = compute_c5_vec(h, n)
    top_idx = np.argsort(c)[-top_m:]
    dx = 2.0 / n
    G = np.zeros((top_m, n))
    for ii, mm in enumerate(top_idx):
        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            G[ii, k:] += (1.0 - h[:nv])
            G[ii, :nv] += (-h[k:])
        else:
            kk = -k
            nv = n - kk
            G[ii, :nv] += (1.0 - h[kk:])
            G[ii, kk:] += (-h[:nv])
    G *= dx
    return c, top_idx, G

def vectorized_best_move(G, c, top_idx, current_max, n):
    """Find best (i,j,d) pair move using vectorized computation."""
    m = len(top_idx)
    c_top = c[top_idx]
    best_d = 0.0
    best_i = -1
    best_j = -1
    for i in range(n):
        diff = G[:, i:i+1] - G  # (m, n)
        mask = diff > 1e-15
        with np.errstate(divide='ignore', invalid='ignore'):
            candidate = np.where(mask, (current_max - c_top)[:, None] / np.where(mask, diff, 1.0), np.inf)
        d_max_vec = np.min(candidate, axis=0)
        d_max_vec[i] = 0.0
        j = int(np.argmax(d_max_vec))
        dv = float(d_max_vec[j])
        if dv > best_d:
            best_d = dv
            best_i = i
            best_j = j
    if best_d <= 1e-12:
        return None
    return best_i, best_j, best_d

def accelerated_pair_search(h, n, deadline, top_m=20, max_iters=50000):
    """Fast pair search using precomputed shift gradients."""
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c, top_idx, G = compute_shift_grad_matrix(h, n, top_m)
        current_max = c.max()
        if abs(current_max - best_c) > 1e-12:
            best_c = current_max
        
        move = vectorized_best_move(G, c, top_idx, current_max, n)
        if move is None:
            # Try reverse: decrease i, increase j
            G_neg = -G
            move2 = vectorized_best_move(G_neg, c, top_idx, current_max, n)
            if move2 is None:
                break
            i, j, d_max = move2
            d = min(d_max * 0.7, h[i] * 0.8, (1.0 - h[j]) * 0.8)
            if d < 1e-10:
                break
            h[i] -= d
            h[j] += d
        else:
            i, j, d_max = move
            d = min(d_max * 0.7, (1.0 - h[i]) * 0.8, h[j] * 0.8)
            if d < 1e-10:
                break
            h[i] += d
            h[j] -= d
        
        # Verify
        new_c = compute_c5(h, n)
        if new_c >= best_c - 1e-15:
            # Revert
            if move is not None:
                h[i] -= d
                h[j] += d
            else:
                h[i] += d
                h[j] -= d
            # Reduce d and retry with smaller step
            d = d * 0.3
            if d < 1e-10:
                break
            if move is not None:
                h[i] = np.clip(h[i] + d, 0, 1)
                h[j] = np.clip(h[j] - d, 0, 1)
            else:
                h[i] = np.clip(h[i] - d, 0, 1)
                h[j] = np.clip(h[j] + d, 0, 1)
            new_c = compute_c5(h, n)
            if new_c >= best_c - 1e-15:
                if move is not None:
                    h[i] -= d
                    h[j] += d
                else:
                    h[i] += d
                    h[j] -= d
                break
        best_c = min(best_c, new_c)
        h = project_feasible(h, n)
        best_c = min(best_c, compute_c5(h, n))
    
    return h, best_c

def fine_polish(h, n, deadline, radius=80):
    """Final coordinate-descend style polish with small steps."""
    best_c = compute_c5(h, n)
    deltas = [0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def nesterov_pass(x, n, deadline, alpha, mu, L, max_iters=3000):
    x = project_feasible(x.copy(), n)
    x_prev = x.copy()
    t = 1.0
    step = 1.0 / L
    best_x = x.copy()
    best_c = compute_c5(x, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = x + beta * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        g = smooth_max_grad(y, n, alpha, mu)
        x_new = np.clip(y - step * g, 0.0, 1.0)
        x_new = project_feasible(x_new, n)
        x_prev = x
        x = x_new
        t = t_new
        if it % 200 == 0:
            c = compute_c5(x, n)
            if c < best_c:
                best_c = c
                best_x = x.copy()
    c = compute_c5(x, n)
    if c < best_c:
        best_c = c
        best_x = x
    return best_x, best_c
