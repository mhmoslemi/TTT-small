```python
    base[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    base[msk] -= h[imn[msk]]
    base[n-1] -= 2.0
    
    # For each j in ones, compute the additional delta
    best_j = -1
    best_val = np.inf
    for j in ones:
        delta = base.copy()
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n-1 + (j-i)] += 1.0
        delta[n-1 + (i-j)] += 1.0
        new_max = (corr + delta).max()
        if new_max < best_val:
            best_val = new_max
            best_j = j
    return best_j, best_val
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_delta(i, j, h, m_range, n):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def fast_anneal_binary(h_init, n, deadline, seed=42, n_batch=50):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max()
    best_c = cur_c
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    rng = np.random.RandomState(seed)
    T = 8.0
    cool_rate = 0.999997
    it = 0
    while time.time() < deadline:
        T *= cool_rate
        # Batch of moves
        for _ in range(n_batch):
            if time.time() > deadline:
                return best_h, best_c * dx
            ai = rng.randint(len(zeros))
            bi = rng.randint(len(ones))
            i = zeros[ai]
            j = ones[bi]
            delta = compute_delta(i, j, h, m_range, n)
            new_corr = corr + delta
            nc = new_corr.max()
            if nc <= cur_c or (T > 1e-8 and rng.random() < np.exp((cur_c - nc) / max(T, 1e-10))):
                h[i], h[j] = 1.0, 0.0
                corr = new_corr
                cur_c = nc
                zeros[ai] = j
                ones[bi] = i
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        it += 1
    return best_h, best_c * dx

def gradient_worst_lag(h, n, n_steps=5000, step_scale=0.01):
    """Projected gradient descent on the worst-lag direction."""
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = h.copy()
    for step in range(n_steps):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        m = k_max - (n - 1)
        # Gradient of corr at lag m w.r.t. h
        # corr[m] = sum_i h[i+m]*(1-h[i]) for valid i
        # d/dh[j]: appears as h[j] in first factor (when j=i+m, i.e., i=j-m)
        #           and as (1-h[j]) in second factor (when j=i, i.e., term h[j+m])
        grad = np.zeros(n)
        # First factor: h[j] appears when j = i+m, so i = j-m, need 0<=j-m<n and 0<=j<n
        # contribution: (1-h[j-m])
        im = m_range  # for the "first index" contribution
        # For position p in h, it appears as h[p] in the product h[p]*(1-h[p-m])
        # when p-m is valid: contributes (1-h[p-m])
        valid_first = (np.arange(n) - m)
        msk_f = (valid_first >= 0) & (valid_first < n)
        grad[msk_f] += (1.0 - h[valid_first[msk_f]]) if False else 0.0
        # Let me redo this properly

        # When p = i+k: i = p-k, term is h[p]*(1-h[p-k]), derivative = (1-h[p-k])
        # When p = i: term is h[p+k]*(1-h[p]), derivative = -h[p+k]

        # First part: p-k valid
        pk = np.arange(n) - m
        msk1 = (pk >= 0) & (pk < n)
        grad[msk1] += (1.0 - h[pk[msk1]])
        # Second part: p+m valid
        pm = np.arange(n) + m
        msk2 = (pm >= 0) & (pm < n)
        grad[msk2] -= h[pm[msk2]]
        # Projected gradient step
        lr = step_scale / (1.0 + step * 0.001)
        h = h - lr * grad
        h = project_feasible(h, n)
    return h

def slsqp_refine(h, n, alpha, maxiter=1000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h
```
