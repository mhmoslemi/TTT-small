import numpy as np, time, math
from scipy.optimize import minimize, linprog

def cv(h, n): return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))
def cc(h, n): return float(np.max(cv(h, n)))

def feasible(h, n):
    h = np.clip(np.asarray(h, float), 0.0, 1.0)
    t = n * 0.5
    for _ in range(5):
        d = t - h.sum()
        if abs(d) < 1e-14: break
        if d > 0:
            r = 1.0 - h; s = r.sum()
            if s <= 1e-15: break
            h += d * r / s
        else:
            s = h.sum()
            if s <= 1e-15: break
            h *= 1.0 + d / s
    return np.clip(h, 0.0, 1.0)

def exact_sum(h, n):
    for _ in range(3):
        h = feasible(h, n)
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-15: break
        i = int(np.argmax(np.minimum(h, 1.0 - h)))
        h[i] = np.clip(h[i] + d, 0.0, 1.0)
    return h

def tops(c, k):
    k = int(k); L = len(c)
    if k <= 0: return np.empty(0, int)
    if k >= L: return np.arange(L)
    return np.argpartition(c, -k)[-k:]

def smax(v, a):
    m = float(v.max()); s = a * (v - m); mx = float(s.max())
    return float(m + (mx + math.log(float(np.sum(np.exp(s - mx))))) / a)

def sgl(v, a):
    m = float(v.max()); s = a * (v - m); mx = float(s.max())
    w = np.exp(s - mx)
    return w / w.sum()

def ograd(h, n, idx):
    g = np.zeros(n); s = 2.0 / float(n); b = n - 1; k = int(idx) - b
    if k >= 0:
        nv = n - k; g[:nv] += s * (1.0 - h[k:]); g[k:] -= s * h[:nv]
    else:
        kk = -k; nv = n - kk; g[kk:] += s * (1.0 - h[:nv]); g[:nv] -= s * h[kk:]
    return g

def agradient(h, n, active, w):
    g = np.zeros(n); s = 2.0 / float(n); b = n - 1
    for idx, wm in zip(active, w):
        if wm < 1e-16: continue
        k = int(idx) - b
        if k >= 0:
            nv = n - k; g[:nv] += wm * (1.0 - h[k:]); g[k:] -= wm * h[:nv]
        else:
            kk = -k; nv = n - kk; g[kk:] += wm * (1.0 - h[:nv]); g[:nv] -= wm * h[kk:]
    return g * s

def robj(x, n, act, a, mu):
    v = cv(x, n)[act]
    val = smax(v, a)
    if mu:
        s = x.sum() - n * 0.5
        val += mu * s * s
    return float(val)

def rgrad(x, n, act, a, mu):
    c = cv(x, n)
    g = agradient(x, n, act, sgl(c[act], a))
    if mu: g += 2.0 * mu * (x.sum() - n * 0.5)
    return g

def req(x, n): return float(x.sum() - n * 0.5)
def reqj(x, n): return np.ones(n)

def lbfgs(h, n, dl, K=40):
    h = feasible(h, n); best = cc(h, n)
    bounds = [(0.0, 1.0)] * n
    al = (100.0, 600.0, 3000.0, 15000.0)
    mus = (1.0, 0.5, 0.1, 0.03)
    mi = 50 if n > 700 else 70
    for _ in range(3):
        if time.time() > dl - 3: break
        act = tops(cv(h, n), K)
        ch = False
        for i in range(4):
            if time.time() > dl - 3: break
            try:
                r = minimize(robj, h.copy(), jac=rgrad, args=(n, act, al[i], mus[i]),
                             method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': mi, 'ftol': 1e-10, 'gtol': 1e-8,
                                      'maxfun': mi * 20, 'maxls': 50})
                h2 = feasible(np.asarray(r.x), n); c2 = cc(h2, n)
                if c2 < best:
                    best = c2; h = h2; ch = True; break
            except Exception:
                pass
        if not ch: break

    return h, best

def pairs(h, n, dl, K=24):
    h = feasible(h, n); best = cc(h, n)
    fr = np.array([0.1, 0.25, 0.4, 0.55, 0.7, 0.85, 0.95, 1.0])
    while time.time() < dl - 2:
        c = cv(h, n)
        act = tops(c, K)
        g = agradient(h, n, act, sgl(c[act], 50000.0))
        up = np.argsort(g)[:K]; dn = np.argsort(g)[-K:]
        m = int(min(8, K)); imp = False
        for ai in range(m):
            for bi in range(m):
                if time.time() > dl - 2: break
                i = int(up[ai]); j = int(dn[bi])
                if i == j: continue
                hi = min(1.0 - h[i], h[j])
                if hi <= 1e-12: continue
                oi, oj = h[i], h[j]
                bs, bc = 0.0, best
                for f in fr:
                    st = f * hi
                    if st <= 1e-14: continue
                    h[i] = oi + st; h[j] = oj - st
                    c2 = float(np.max(cv(h, n)))
                    if c2 < bc: bc = c2; bs = st
                    h[i] = oi; h[j] = oj
                if bs > 0.0:
                    h[i] = float(np.clip(oi + bs, 0.0, 1.0))
                    h[j] = float(np.clip(oj - bs, 0.0, 1.0))
                    best = bc; imp = True
        if not imp: break
    return h, best

def lppolish(h, n, dl, K=40, D=0.02):
    h = feasible(h, n); best = cc(h, n)
    for _ in range(12):
        if time.time() > dl - 2: break
        act = tops(cv(h, n), K)
        if len(act) == 0: break
        A = np.zeros((len(act), n + 1)); A[:, n] = -1.0
        for r_i, idx in enumerate(act):
            A[r_i, :n] = ograd(h, n, idx)
        lo = -np.minimum(D, h)
        hi = np.minimum(D, 1.0 - h)
        Be = np.zeros((1, n + 1)); Be[0, :n] = 1.0
        cobj = np.zeros(n + 1); cobj[-1] = 1.0
        bds = list(zip(lo, hi)) + [(-1.0, 0.0)]
        try:
            try:
                res = linprog(cobj, A_ub=A, b_ub=np.zeros(len(act)), A_eq=Be,
                              b_eq=np.array([0.0]), bounds=bds, method='highs')
            except TypeError:
                res = linprog(cobj, A_ub=A, b_ub=np.zeros(len(act)), A_eq=Be,
                              b_eq=np.array([0.0]), bounds=bds)
            if not res.success or float(res.x[-1]) > -1e-7: break
            d = res.x[:n]; imp = False
            for s in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
                h2 = feasible(h + s * d, n); c2 = cc(h2, n)
                if c2 < best:
                    best = c2; h = h2; imp = True; break
            if not imp: break
        except Exception:
            break
    return h, best

def sfull(v, m): return np.concatenate((v, 1.0 - v))

def sobj(v, m, act, a):
    h = sfull(v, m)
    return smax(cv(h, 2 * m)[act], a)

def sgrad(v, m, act, a):
    h = sfull(v, m)
    v2 = cv(h, 2 * m)[act]
    g = agradient(h, 2 * m, act, sgl(v2, a))
    return g[:m] - g[m:]

def anti(h):
    n = len(h); m = n // 2
    if m <= 0: return h
    return sfull(0.5 * (h[:m] + 1.0 - h[m:]), m)

def sym_lbfgs(h, n, dl, K=32):

    return h, best

def interp(h, n2):
    n = len(h)
    if n == n2: return h.copy()
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    y = np.linspace(0.0, 2.0, n2, endpoint=False)
    return feasible(np.interp(y, x, h), n2)

def lfreq(n, rs):
    x = np.linspace(0.0, 2.0, n, endpoint=False)

        h += rs.uniform(0.08, 0.22) * np.sin(f * np.pi * x + rs.uniform(0.0, 2.0 * math.pi))
    return feasible(h, n)
