        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 100.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    n_pts = len(x)
    top_k = min(50, len(w))
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n_pts)
    for i in top_idx:
        if w[i] < 1e-14:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    s = x.sum() - n / 2.0
    grad += 200.0 * s
    return grad

def subgrad_exact(h, n, num_active=5):
    c = compute_c5_vec(h, n)
    m = float(c.max())
    tol = 1e-7
    active = np.where(c >= m - tol)[0]
    if len(active) > num_active:
        active = active[np.argsort(c[active])[-num_active:]]
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        grad += dx * grad_single(h, n, k)
    grad /= len(active)
    return grad, m, c, active

def lbfgs_optimize(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [200, 1000, 5000, 20000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-18})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def momentum_subgrad(h, n, deadline, num_active=5, lr=0.01):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_mom = h.copy()
    v = np.zeros(n)
    beta = 0.85
    lr_cur = lr
    stall = 0
    
    while time.time() < deadline:
        g, m, c, active = subgrad_exact(h_mom, n, num_active)
        v = beta * v - lr_cur * g
        h_new = project_feasible(h_mom + v, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h_mom = h_new.copy()
            stall = 0
        else:
            h_try = project_feasible(h_mom - lr_cur * 0.2 * g, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-14:
                best_c = c_try
                best_h = h_try.copy()
                h_mom = h_try.copy()
                stall = 0
            else:
                stall += 1
                lr_cur *= 0.8
                if stall > 30:
                    break
    
    return best_h, best_c

def pair_cd(h, n, deadline, num_active=8, max_sweeps=200):
    """Pair coordinate descent: exact minimization over pairs for active lags."""
    h_work = h.copy()
    best_c = compute_c5(h_work, n)
    dx = 2.0 / n
    B = n - 1
    
    c = compute_c5_vec(h_work, n)
    m = float(c.max())
    tol = 1e-6
    active = np.where(c >= m - tol)[0]
    if len(active) > num_active:
        active = active[np.argsort(c[active])[-num_active:]]
    
    # Precompute gradients for active lags
    grad_mats = np.zeros((len(active), n))
    for i_ax, idx in enumerate(active):
        k = idx - B
        grad_mats[i_ax] = grad_single(h_work, n, k) * dx
    c_active = c[active].copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        any_improve = False
        
        # Vectorized pair processing
        for i in range(n):
            if time.time() > deadline:
                return h_work, best_c
            # Slopes for moving from j to i (delta > 0: h[i]+delta, h[j]-delta)
            # f(delta) = max_a (c_active[a] + (grad[a,i] - grad[a,j]) * delta)
            # For fixed i, slopes[:, i] is fixed, we vary j
            slopes_i = grad_mats[:, i]  # shape (K,)
            
            for j in range(n):
                if i == j:
                    continue
                # Feasible delta range
                d_lo = max(h_work[j] - 1.0, -h_work[i])
                d_hi = min(1.0 - h_work[i], h_work[j])
                if d_hi - d_lo < 1e-12:
                    continue
                
                slopes_j = grad_mats[:, j]
                diffs = slopes_i - slopes_j  # shape (K,)
                
                # Find min of max(c_active + diffs * delta) over [d_lo, d_hi]
                # This is a 1D convex PL function; min at breakpoint or boundary
                f_at_lo = np.max(c_active + diffs * d_lo)
                f_at_hi = np.max(c_active + diffs * d_hi)
                f_at_0 = np.max(c_active)
                
                best_val = min(f_at_lo, f_at_hi, f_at_0)
                best_d = 0.0
                if f_at_lo < f_at_hi and f_at_lo < f_at_0:
                    best_d = d_lo
                elif f_at_hi < f_at_lo and f_at_hi < f_at_0:
                    best_d = d_hi
                
                # Check breakpoints
                K = len(diffs)
                for a in range(K):
                    for b in range(a + 1, K):
                        if abs(diffs[a] - diffs[b]) > 1e-15:
                            pt = (c_active[b] - c_active[a]) / (diffs[a] - diffs[b])
                            if d_lo <= pt <= d_hi:
                                f_pt = np.max(c_active + diffs * pt)
                                if f_pt < best_val - 1e-13:
                                    best_val = f_pt
                                    best_d = pt
                
                if abs(best_d) > 1e-10 and best_val < best_c - 1e-12:
                    h_work[i] = np.clip(h_work[i] + best_d, 0.0, 1.0)
                    h_work[j] = np.clip(h_work[j] - best_d, 0.0, 1.0)
                    best_c = best_val
                    any_improve = True
                    # Update c_active
                    c_active += diffs * best_d
                    # Update gradients for i and j
                    for idx_ax, idx in enumerate(active):
                        k = idx - B
                        grad_mats[idx_ax, i] = grad_single(h_work, n, k)[i] * dx
                        grad_mats[idx_ax, j] = grad_single(h_work, n, k)[j] * dx
        
        # Periodically recompute active set
        if sweep % 20 == 19 or not any_improve:
            c = compute_c5_vec(h_work, n)
            m = float(c.max())
            best_c = min(best_c, m)
            if not any_improve:
                break
            tol = 1e-6
            active = np.where(c >= m - tol)[0]
            if len(active) > num_active:
                active = active[np.argsort(c[active])[-num_active:]]
            for i_ax, idx in enumerate(active):
                k = idx - B
                grad_mats[i_ax] = grad_single(h_work, n, k) * dx
            c_active = c[active].copy()
    
    return h_work, best_c

Hmm wait, the pair_cd function has O(n²) per sweep with an inner O(K²) loop. For n=500, K=8: 500*500*64 = 16,000,000 operations per sweep. In Python, that's probably 5-15 seconds per sweep. With 200 sweeps, that's 1000-3000 seconds. Too slow!

I need to optimize this. Options:
1. Reduce the number of pairs tried per sweep (only try nearby pairs)
2. Vectorize the inner computation
3. Reduce n
4. Use a different approach

Let me try restricting to nearby pairs (radius = 50 or so). That gives 500 * 100 = 50,000 pairs per sweep. With K² = 64 inner ops, that's 3.2 million ops per sweep, about 1-3 seconds. With 100 sweeps, that's 100-300 seconds. Still a lot.

Actually, let me reconsider. The O(K²) per pair is the bottleneck. Can I do better?

The function f(δ) = max_k (c_k + d_k * δ) is the upper envelope of K lines. To find its minimum on an interval, I can:
1. Sort the lines by slope
2. The upper envelope can be computed in O(K log K)
3. Find the minimum of the envelope on the interval

But for K=5-8, the O(K²) approach is fine. The issue is the number of pairs.

Let me try a different strategy: instead of trying all pairs, use a gradient-guided approach:
1. Compute the gradient g (subgradient of the max)
2. For each i, the best j to pair with is the one that maximizes g[i] - g[j] (i.e., move mass from high gradient to low gradient)
3. Try the top few such pairs per coordinate

This reduces the number of pairs from O(n²) to O(n * top_m) where top_m is small (say 5-10).

Actually, let me think about this differently. The gradient tells us the direction of steepest descent. The best pair move is one where we move mass in the direction of the negative gradient. So for coordinate i, we want to move to coordinate j where g[j] is as small as possible (or g[i] is large and g[j] is small).

Let me implement a "greedy pair" approach:
1. Compute gradient g
2. For each i (in some order), find the best j to pair with
3. Do the exact 1D minimization for that pair
4. Update g
5. Repeat

This is O(n) pairs per sweep instead of O(n²).

Actually, even better: sort coordinates by gradient value. Pairs (i, j) where g[i] >> g[j] are the most promising. So we can try a limited set of such pairs.

Let me code this up:

This is much more efficient: O(n_try² * K²) per sweep where n_try=50, K=8. So 2500 * 64 = 160,000 ops per sweep. Very fast!

But wait, I'm only trying pairs between "high gradient" and "low gradient" indices. What if the optimal move is between two "medium" gradient indices? I should also try some random pairs or nearby pairs.

Let me combine: greedy pairs + random pairs + nearby pairs.

Actually, let me think about this more carefully. The gradient g[i] tells us how much the objective increases if we increase h[i]. So to decrease the objective, we want to:
- Decrease h[i] for large g[i]
- Increase h[j] for small g[j]

A pair move (h[i] - δ, h[j] + δ) decreases the objective by approximately (g[i] - g[j]) * δ. So the best pairs are (i, j) where g[i] is large and g[j] is small. This is exactly what the greedy approach does.

However, the exact objective change is not just (g[i]-g[j])*δ because the objective is the MAX of multiple functions. The linear approximation is only valid for small δ. For the exact computation (which we do in the pair descent), we account for this.

So the greedy approach should work well. Let me also add:
- A "recompute" step to update the active set
- Some random pairs for exploration

def golden_start(n, rng, salt=0):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(half) * phi + salt).astype(int) % n
    # Ensure uniqueness

    unique_pos = []
    for p in positions:

        if p not in seen:
            seen.add(p)
            unique_pos.append(p)
    h[unique_pos[:half]] = 1.0
    return project_feasible(h, n)

    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

One more thought: the `grad_single` function is called very frequently in the pair CD. Each call is O(n). But we only need grad_single[h[i]] and grad_single[h[j]] for a specific pair. Can we compute these more efficiently?

grad_single(h, n, k)[i] for k ≥ 0:
= (1 - h[i+k]) * dx if i < n-k
  + (-h[i-k]) * dx if i ≥ k

So for a specific (i, k):
- If i < n-k and i >= k: (1 - h[i+k] - h[i-k]) * dx
- If i < n-k and i < k: (1 - h[i+k]) * dx
- If i >= n-k and i >= k: (-h[i-k]) * dx
- (i >= n-k and i < k is impossible since n-k > k when k < n/2)

Actually this is just a quick formula. But in the pair CD, I'm calling grad_single(h_work, n, k) which computes the full gradient vector O(n), and then indexing into it. That's wasteful.

Let me write a helper that computes just grad_single[h[i]] for a given i and k:

```python
def grad_at(h, n, k, i):
    """Compute the i-th component of the gradient of c[k] w.r.t. h."""
    dx = 2.0 / n
    if k >= 0:
        val = 0.0
        if i < n - k:
            val += (1.0 - h[i + k])
        if i >= k:
            val -= h[i - k]
    else:
        kk = -k
        val = 0.0
        if i >= kk:
            val += (1.0 - h[i - kk])
        if i < n - kk:
            val -= h[i + kk]
    return val * dx
```

This is O(1) per call! Much better than O(n) for the full vector.

Let me use this in the pair CD. Then updating the gradient for a single (i, k) pair is O(1).
