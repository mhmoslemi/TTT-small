    """Compute new correlation after swapping h[i]=1→0 and h[j]=0→1.
    Returns the updated corr vector (a copy)."""
    n = len(h)
    new_corr = corr.copy()
    # Effect of i: 1→0
    for k in range(-(min(i, n-1)), min(n-1-i, n-1) + 1):
        idx = k + n - 1
        if 0 <= idx < 2*n-1:

    """Compute the delta in correlation vector when swapping h[i]:1→0 and h[j]:0→1."""
    delta = np.zeros(2*n-1)
    # Effect of i: 1→0
    # For shift k: delta_k = -(1-h[i+k])*I(0≤i+k<n) + h[i-k]*I(0≤i-k<n)
    k_min_i = max(-(n-1), -i)  # i+k ≥ 0 → k ≥ -i
    k_max_i = min(n-1, n-1-i)  # i+k < n → k < n-i
    ks = np.arange(k_min_i, k_max_i + 1)

    valid1 = (i + ks >= 0) & (i + ks < n)
    delta[idxs[valid1]] -= (1 - h[i + ks[valid1]])
    # h[i-k] for valid i-k
    valid2 = (i - ks >= 0) & (i - ks < n)
    delta[idxs[valid2]] += h[i - ks[valid2]]
    
    # Effect of j: 0→1

    k_min_j = max(-(n-1), -j)

    valid1_j = (j + ks_j >= 0) & (j + ks_j < n)
    delta[idxs_j[valid1_j]] += (1 - h[j + ks_j[valid1_j]])
    valid2_j = (j - ks_j >= 0) & (j - ks_j < n)
    delta[idxs_j[valid2_j]] -= h[j - ks_j[valid2_j]]

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]
        rem = rng.choice(idx, size=cnt - target, replace=False)
        h_bin[rem] = 0
    elif cnt < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)
        h_bin[add] = 1
    return h_bin

def binary_local_search(h_bin, n, deadline, rng, max_iters=5000000):
    """Fast binary local search using incremental correlation updates."""
    corr = np.correlate(h_bin, 1 - h_bin, mode='full')
    best_c = corr.max()
    best_h = h_bin.copy()
    ones_idx = np.where(h_bin == 1)[0]
    zeros_idx = np.where(h_bin == 0)[0]
    
    temp = 0.01
    iters = 0
    no_improve = 0
    
    while time.time() < deadline and iters < max_iters:
        if iters % 10000 == 0 and iters > 0:
            temp *= 0.9
            if temp < 0.001:
                break
        
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        if i == j:
            iters += 1
            continue
        
        # Compute delta in corr
        h = h_bin  # current state
        delta = np.zeros(2*n-1)
        # i: 1→0
        ks = np.arange(-(min(i, n-1)), min(n-1-i, n-1)+1)
        idxs = ks + n - 1
        valid1 = (i+ks >= 0) & (i+ks < n)
        delta[idxs[valid1]] -= (1 - h[i+ks[valid1]])
        valid2 = (i-ks >= 0) & (i-ks < n)
        delta[idxs[valid2]] += h[i-ks[valid2]]
        # j: 0→1
        ks_j = np.arange(-(min(j, n-1)), min(n-1-j, n-1)+1)
        idxs_j = ks_j + n - 1
        valid1_j = (j+ks_j >= 0) & (j+ks_j < n)
        delta[idxs_j[valid1_j]] += (1 - h[j+ks_j[valid1_j]])
        valid2_j = (j-ks_j >= 0) & (j-ks_j < n)
        delta[idxs_j[valid2_j]] -= h[j-ks_j[valid2_j]]
        
        new_corr = corr + delta
        new_max = new_corr.max()
        
        if new_max < best_c - 1e-15:

            h_bin[i] = 0
            h_bin[j] = 1
            corr = new_corr
            best_c = new_max
            best_h = h_bin.copy()
            no_improve = 0
        elif rng.random() < np.exp(-(new_max - best_c) / max(temp, 1e-10)):

            h_bin[i] = 0
            h_bin[j] = 1
            corr = new_corr
            # Don't update best
        else:
            # Reject
            pass
        
        iters += 1
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)

def cont_grad_descent(h, n, deadline, max_iters=2000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        for it in range(max_iters // len(alphas)):
            if time.time() > deadline - 3:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            m = corr.max()
            sh = alpha * (corr - m)
            smax = sh.max()
            w = np.exp(sh - smax)
            w /= w.sum()
            grad = np.zeros(n)
            for idx in range(2 * n - 1):
                if w[idx] < 1e-10:
                    continue
                k = idx - (n - 1)
                wm = w[idx]
                if k >= 0:
                    nv = n - k
                    grad[:nv] += wm * (1.0 - h[k:])
                    grad[k:] += wm * (-h[:nv])
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:] += wm * (1.0 - h[:nv])
                    grad[:nv] += wm * (-h[kk:])
            lr = 0.05 / (np.abs(grad).max() + 1e-10)
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def random_start(n, rng):

def structured_start(n, rng):
    h = np.zeros(n)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    vals = 0.5 + 0.3 * np.sin(2 * np.pi * x / n + phase) + 0.2 * np.sin(4 * np.pi * x / n + phase * 2)
    threshold = np.sort(vals)[n // 2]
    h[vals >= threshold] = 1.0
    return h

    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = random_start(500, rng)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Continuous refinement at n_init
    if time.time() < deadline - 500:
        t1 = min(time.time() + 80, deadline - 550)
        h_c, c_c = cont_grad_descent(best_h, best_n, t1)
        if c_c < best_c:
            best_c, best_h = c_c, h_c

    # Phase 2: Binary directed search at n_init
    if time.time() < deadline - 350:
        h_b = to_binary(best_h, best_n, rng)
        t2 = min(time.time() + 120, deadline - 300)
        h_b2, c_b2 = directed_binary_search(h_b, best_n, t2, rng)
        if c_b2 < best_c:
            best_c, best_h = c_b2, h_b2

    # Phase 3: Try n=1000
    n_try = 1000
    if time.time() < deadline - 400:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        t3 = min(time.time() + 60, deadline - 350)
        h_c3, c_c3 = cont_grad_descent(h_interp, n_try, t3)
        h_b3 = to_binary(h_c3, n_try, rng)
        t3b = min(time.time() + 100, deadline - 250)
        h_b3b, c_b3b = directed_binary_search(h_b3, n_try, t3b, rng)
        if c_b3b < best_c:
            best_c, best_h, best_n = c_b3b, h_b3b, n_try

    # Phase 4: Random restarts at best_n
    for ri in range(6):
        if time.time() > deadline - 150:
            break
        h_r = random_start(best_n, rng) if ri % 2 == 0 else structured_start(best_n, rng)
        t_r = min(40, (deadline - time.time() - 100) / (6 - ri))
        if t_r < 15:
            break
        h_rb, c_rb = directed_binary_search(h_r, best_n, time.time() + t_r, rng)
        if c_rb < best_c:
            best_c, best_h = c_rb, h_rb

    # Phase 5: Final random search polish
    if time.time() < deadline - 60:
        h_fin = to_binary(best_h, best_n, rng)
        t5 = min(time.time() + 50, deadline - 40)
        h_fin2, c_fin2 = random_binary_search(h_fin, best_n, t5, rng, max_iters=300000)
        if c_fin2 < best_c:
            best_c, best_h = c_fin2, h_fin2

    # Phase 6: Final directed search
    if time.time() < deadline - 20:
        h_fin3 = to_binary(best_h, best_n, rng)
        t6 = min(time.time() + 15, deadline - 5)
        h_fin4, c_fin4 = directed_binary_search(h_fin3, best_n, t6, rng)
        if c_fin4 < best_c:
            best_c, best_h = c_fin4, h_fin4

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
