def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-10 * n:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad
