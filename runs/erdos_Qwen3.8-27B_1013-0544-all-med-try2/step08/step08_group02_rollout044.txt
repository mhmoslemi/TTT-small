def pen_obj(x, n, alpha, mu):
    v = smooth_max_val(x, n, alpha)
    s = x.sum() - n / 2.0
    return v + mu * s * s

def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def focused_obj(x, n, lags, weights):
    c = np.correlate(x, 1 - x, mode='full')
    vals = c[lags]
    m = vals.max()
    w = weights / weights.sum()
    return float(np.sum(w * vals))

def focused_grad(x, n, lags, weights):
    grad = np.zeros(n)
    w = weights / weights.sum()
    for i, mi in enumerate(lags):
        if w[i] < 1e-12:
            continue
        k = mi - (n - 1)
        wm = w[i]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:k + nv])
            grad[k:k + nv] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def continued_lbfgs(h, n, deadline, alphas=None, max_iters=500, mu=1e8):
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 500000, 2000000]
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        mi = max(200, min(max_iters, int(100 + alpha / 20000)))
        try:
            res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': mi, 'ftol': 1e-19, 'gtol': 1e-15,
                                   'maxfun': mi * 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def focused_optimize(h, n, deadline, top_m=8):
    """Iteratively focus on the top lags and re-optimize"""
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    for it in range(10):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        idx = np.argsort(c)[-top_m:]
        vals = c[idx]
        weights = np.exp(50.0 * (vals - vals.max()))
        try:
            res = minimize(focused_obj, h, jac=focused_grad,
                          args=(n, idx, weights),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 1000, 'ftol': 1e-19, 'gtol': 1e-15})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            break
    return h, best_c

def slsqp_polish(h, n, deadline, alpha=2000000):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_val(x, n, alpha)

    def eq(x):
        return x.sum() - n / 2.0

    try:
        rem_iters = max(500, int((deadline - time.time()) * 200))
        res = minimize(obj, h, jac=lambda x: smooth_max_grad(x, n, alpha),
                      method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': eq,
                                   'jac': lambda x: np.ones(n)}],
                      options={'maxiter': min(rem_iters, 10000), 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def binary_from_h(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]
        rem = rng.choice(idx, size=cnt - target, replace=False)
        h_bin[rem] = 0
    elif cnt < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)
        h_bin[add] = 1
    return h_bin

def binary_local_search(h_bin, n, deadline, rng, max_rounds=500):
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    no_imp = 0
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        if no_imp > 100:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        best_delta = best_c
        best_i, best_j = -1, -1
        n_try = min(len(ones) * len(zeros), 8000)
        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i], h_bin[j] = 0, 1
            c_new = compute_c5(h_bin, n)
            h_bin[i], h_bin[j] = 1, 0
            if c_new < best_delta - 1e-15:
                best_delta = c_new
                best_i, best_j = i, j
        if best_i >= 0:
            h_bin[best_i] = 0
            h_bin[best_j] = 1
            best_c = best_delta
            best_h = h_bin.copy()
            no_imp = 0
        else:
            no_imp += 1
    return best_h, best_c

def perturb(h, n, rng, frac=0.08):
    h_new = h.copy()
    nP = max(3, int(n * frac))
    idx = rng.choice(n, size=nP, replace=False)
    h_new[idx] = rng.uniform(0, 1, size=nP)
    return project_feasible(h_new, n)

def interpolate_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.zeros(500)
        h_init[rng.choice(500, size=250, replace=False)] = 1.0
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Continuation L-BFGS-B (150s)
    t1 = min(t0 + 150, deadline - 500)
    best_h, best_c = continued_lbfgs(best_h, n, t1,
        alphas=[50, 200, 1000, 5000, 50000, 500000, 2000000])

    # Phase 2: Focused optimization (60s)
    if time.time() < deadline - 400:
        t2 = min(time.time() + 60, deadline - 380)
        hf, cf = focused_optimize(best_h, n, t2, top_m=10)
        if cf < best_c:
            best_c = cf
            best_h = hf

    # Phase 3: SLSQP polish (50s)
    if time.time() < deadline - 350:
        t3 = min(time.time() + 50, deadline - 330)
        hs, cs = slsqp_polish(best_h, n, t3, alpha=3000000)
        if cs < best_c:
            best_c = cs
            best_h = hs

    # Phase 4: Binary local search + polish (100s)
    if time.time() < deadline - 300:
        t4 = min(time.time() + 60, deadline - 280)
        hb = binary_from_h(best_h, n, rng)
        hb2, cb2 = binary_local_search(hb, n, t4, rng)
        if cb2 < best_c:
            best_c = cb2
            best_h = project_feasible(hb2, n)
        if time.time() < deadline - 250:
            t4b = min(time.time() + 40, deadline - 240)
            hp, cp = continued_lbfgs(hb2, n, t4b,
                alphas=[1000, 10000, 100000, 500000, 3000000])
            if cp < best_c:
                best_c = cp
                best_h = hp

    # Phase 5: Focused + SLSQP on best (50s)
    if time.time() < deadline - 200:
        t5 = min(time.time() + 30, deadline - 190)
        hf, cf = focused_optimize(best_h, n, t5, top_m=5)
        if cf < best_c:
            best_c = cf
            best_h = hf
        if time.time() < deadline - 180:
            t5b = min(time.time() + 20, deadline - 170)
            hs2, cs2 = slsqp_polish(best_h, n, t5b, alpha=5000000)
            if cs2 < best_c:
                best_c = cs2
                best_h = hs2

    # Phase 6: Perturbed restarts (200s)
    for trial in range(8):
        if time.time() > deadline - 100:
            break
        alloc = min(25, (deadline - time.time() - 80) / max(8 - trial, 1))
        if alloc < 12:
            break
        h_p = perturb(best_h, n, rng, frac=0.04 + 0.01 * trial)
        t_p = min(time.time() + alloc * 0.75, deadline - 80)
        hp, cp = continued_lbfgs(h_p, n, t_p,
            alphas=[500, 5000, 50000, 500000, 2000000])
        if time.time() < deadline - 70 and cp < best_c * 1.001:
            hp2, cp2 = focused_optimize(hp, n, min(time.time() + 10, deadline - 65), top_m=6)
            if cp2 < cp:
                hp, cp = hp2, cp2
        if cp < best_c:
            best_c = cp
            best_h = hp

    # Phase 7: Different n values (100s)
    for n_try in [450, 550, 600, 750, 1000]:
        if time.time() > deadline - 50:
            break
        h_int = interpolate_h(best_h, n, n_try)
        alloc = min(18, (deadline - time.time() - 40) * 0.7)
        if alloc < 8:
            continue
        ht, ct = continued_lbfgs(h_int, n_try, time.time() + alloc,
            alphas=[500, 5000, 50000, 500000])
        if ct < best_c:
            best_c = ct
            best_h = ht
            n = n_try

    # Phase 8: Final polish (remaining time)
    if time.time() < deadline - 5:
        h_f, c_f = focused_optimize(best_h, n, deadline - 3, top_m=5)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    if time.time() < deadline - 2:
        h_f2, c_f2 = slsqp_polish(best_h, n, deadline - 1, alpha=8000000)
        if c_f2 < best_c:
            best_c = c_f2
            best_h = h_f2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
