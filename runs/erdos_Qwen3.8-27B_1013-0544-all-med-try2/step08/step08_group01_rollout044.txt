    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: Focus optimization with moderate top_k
    t1 = min(t0 + 250, deadline - 500)
    h1, c1 = focus_optimize(h, n, t1, top_k=50)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Focus with larger top_k for broader improvement
    t2 = min(time.time() + 200, deadline - 350)
    h2, c2 = focus_optimize(best_h, n, t2, top_k=100)
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: Targeted coordinate descent
    t3 = min(time.time() + 100, deadline - 200)
    h3, c3 = targeted_cd(best_h, n, t3, top_k=30)
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Phase 4: Targeted pair polish
    t4 = min(time.time() + 120, deadline - 80)
    h4, c4 = pair_polish(best_h, n, t4, radius=60)
    if c4 < best_c:
        best_c = c4
        best_h = h4

    # Phase 5: Another focus round with smaller top_k (sharper)
    if time.time() < deadline - 60:
        t5 = min(time.time() + 80, deadline - 40)
        h5, c5 = focus_optimize(best_h, n, t5, top_k=20)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Final targeted CD + polish
    if time.time() < deadline - 20:
        t6 = min(time.time() + 50, deadline - 10)
        h6, c6 = targeted_cd(best_h, n, t6, top_k=20)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    if time.time() < deadline - 3:
        h7, c7 = pair_polish(best_h, n, deadline - 2, radius=40)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
