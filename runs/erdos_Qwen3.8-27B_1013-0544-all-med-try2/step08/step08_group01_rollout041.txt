    mx = sh.max()
    return (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    sh = alpha * c
    mx = sh.max()
    return (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def grad_c5(h, n, alpha):
    c = compute_c5_full(h, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _obj(x, n, alpha, mu):
    c = compute_c5_full(x, n)
    val = smooth_max_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu):
    g = grad_c5(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_opt(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        for mu in [1.0, 100.0]:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(_obj, best_h, jac=_grad,
                               args=(n, alpha, mu), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 300, 'ftol': 1e-18,
                                        'gtol': 1e-16, 'maxfun': 8000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def guided_local_search(h, n, deadline, radius=80, n_try=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        stall = 0
        while time.time() < deadline and stall < 50:
            stall = 0
            improved_this_round = False
            g = grad_c5(best_h, n, 50000)
            order_inc = np.argsort(g)
            order_dec = np.argsort(-g)
            for a in range(min(n_try, n)):
                if time.time() > deadline:
                    return best_h, best_c
                i = order_inc[a]

                    if best_h[i] + step <= 1.0 and best_h[j] - step >= 0.0:
                        best_h[i] += step

def golden_ratio_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(half) * phi).astype(int) % n
    h[positions] = 1.0
    return project_feasible(h, n)

def fourier_start(n, rng):

        h += amp * np.cos(2 * np.pi * freq * x + phase)
