import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def fft_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    v = 1.0 - 2.0 * h
    N = 1
    while N < 3 * n:
        N <<= 1
    wp = np.zeros(N)
    wp[:2*n-1] = w
    vp = np.zeros(N)
    vp[:n] = v
    g = np.fft.ifft(np.fft.fft(wp) * np.conj(np.fft.fft(vp))).real[:n]
    return g * (2.0 / n)

def sm_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def _eq_fun(x, n):
    return x.sum() - n / 2.0

def _eq_jac(x, n):
    return np.ones(n)

def _lbfgs_f(x, n, alpha):
    return sm_obj(x, n, alpha) + 1e4 * (x.sum() - n / 2.0) ** 2

def _lbfgs_g(x, n, alpha):
    return fft_grad(x, n, alpha) + 2e4 * (x.sum() - n / 2.0)

def _sm_g(x, n, alpha):
    return fft_grad(x, n, alpha)

def _slsqp_cons(n):
    return {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}

def opt_lbfgs(h, n, alpha, deadline, iters=3000):
    bounds = [(0.0, 1.0)] * n
    args = (n, alpha)
    try:
        r = minimize(_lbfgs_f, h, jac=_lbfgs_g, method='L-BFGS-B',
                    bounds=bounds, args=args,
                    options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-15,
                             'maxfun': iters * 15, 'maxls': 40})
        hn = project_feasible(r.x, n)
        return hn, compute_c5(hn, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def opt_slsqp(h, n, alpha, deadline, iters=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [_slsqp_cons(n)]
    args = (n, alpha)
    try:
        r = minimize(sm_obj, h, args=args, jac=_sm_g,
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': iters, 'ftol': 1e-15, 'maxls': 20})
        hn = project_feasible(r.x, n)
        return hn, compute_c5(hn, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def local_refine(h, n, deadline):
    best = compute_c5(h, n)
    for alpha in [50000, 500000]:
        for step in [0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            imp = True
            while imp and time.time() < deadline:
                imp = False
                g = fft_grad(h, n, alpha)
                idx_neg = np.argsort(g)[:30]
                idx_pos = np.argsort(g)[-30:]
                for i in idx_pos:
                    if h[i] < step:
                        continue
                    for j in idx_neg:
                        if i == j or h[j] > 1.0 - step:
                            continue
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best - 1e-15:
                            best = c
                            imp = True
                        else:
                            h[i] += step
                            h[j] -= step
                        if time.time() > deadline:
                            return h, best
    return h, best

def balance_active(h, n, deadline):
    best = compute_c5(h, n)
    for _ in range(50):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        k_max = int(np.argmax(c_vec))
        k = k_max - (n - 1)
        if k >= 0:
            nv = n - k
            contrib = h[:nv] * (1.0 - h[k:])
        else:
            kk = -k
            nv = n - kk
            contrib = h[kk:] * (1.0 - h[:nv])
        worst = int(np.argmax(contrib))
        step = 0.001
        if k >= 0:
            src = worst
            dst = worst + k
        else:
            dst = worst
            src = worst - k
        if src < 0 or src >= n or dst < 0 or dst >= n:
            continue
        if h[src] >= step and h[dst] <= 1.0 - step:
            sb = h.copy()
            h[src] -= step
            h[dst] += step
            if compute_c5(h, n) < best:
                best = compute_c5(h, n)
            else:
                h[:] = sb
        elif h[dst] >= step and h[src] <= 1.0 - step:
            sb = h.copy()
            h[dst] -= step
            h[src] += step
            if compute_c5(h, n) < best:
                best = compute_c5(h, n)
            else:
                h[:] = sb
    return h, best
