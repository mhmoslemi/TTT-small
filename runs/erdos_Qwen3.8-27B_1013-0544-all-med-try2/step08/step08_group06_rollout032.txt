    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def greed_binary(h_init, n, deadline, max_sweeps=300):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = compute_swap_delta(h, i, j, n, m_range)
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=3000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]
    T = 0.02
    cool = 0.99997
    scale = 5000.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = int(zeros_arr[ai])
        j = int(ones_arr[bi])
        delta = compute_swap_delta(h, i, j, n, m_range)
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(-scale * (nc - cur_c) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    scaled = corr * dx
    sh = alpha * (scaled - scaled.max())
    ev = np.exp(sh)
    esum = ev.sum()
    w = ev / esum
    val = scaled.max() + np.log(esum) / alpha
    # Gradient via loop
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        s = k - (n - 1)
        lo = max(0, -s)
        hi = min(n, n - s)
        if lo < hi:
            grad[lo:hi] += wk * omh[lo + s:hi + s]
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 - s:hi2 - s]
    grad *= dx
    return val, grad

def grad_optimize(h, n, a0, a1, nit, dl):
    h = h.copy()
    a = a0
    lr = np.log(max(a1, a0 * 1.01) / a0) / max(nit, 1)
    bh, bc = h.copy(), compute_c5(h, n)
    for it in range(nit):
        if time.time() > dl:
            break
        a *= np.exp(lr)
        v, g = smooth_max_grad(h, n, a)
        step = 1.0
        ok = False
        for _ in range(50):
            hn = project_feasible(h - step * g, n)
            vn, _ = smooth_max_grad(hn, n, a)
            if vn < v - 1e-15:
                h = hn
                c = compute_c5(h, n)
                if c < bc:
                    bc, bh = c, h.copy()
                ok = True
                break
            step *= 0.5
        if not ok and step < 1e-12:
            continue
    return bh, bc

def swap_delta(h, i, j, n, mr):
    d = np.zeros(2 * n - 1)
    im = i + mr
    m = (im >= 0) & (im < n)
    d[m] += 1.0 - h[im[m]]
    imn = i - mr
    m = (imn >= 0) & (imn < n)
    d[m] -= h[imn[m]]
    jm = j + mr
    m = (jm >= 0) & (jm < n)
    d[m] -= 1.0 - h[jm[m]]
    jmn = j - mr
    m = (jmn >= 0) & (jmn < n)
    d[m] += h[jmn[m]]
    d[n - 1] -= 2.0
    d[n - 1 + j - i] += 1.0
    d[n - 1 + i - j] += 1.0
    return d

def greed_bin(h0, n, dl, ms=300):
    mr = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h0, n).copy()
    cr = np.correlate(h, 1 - h, mode='full').astype(float)
    cc = cr.max() * dx
    bh, bc = h.copy(), cc
    ones = [int(x) for x in np.where(h == 1)[0]]
    zeros = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(ms):
        if time.time() > dl:
            break
        imp = False
        for i in zeros:
            if time.time() > dl:
                break
            for j in ones:
                d = swap_delta(h, i, j, n, mr)
                nc = (cr + d).max() * dx
                if nc < cc - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    cr += d
                    cc = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    imp = True
                    break
            if imp:
                break
        if not imp:
            break
        if cc < bc:
            bc, bh = cc, h.copy()
    return bh, bc

def anneal_bin(h0, n, dl, seed=42, ni=3000000):
    rng = np.random.RandomState(seed)
    mr = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h0, n).copy()
    cr = np.correlate(h, 1 - h, mode='full').astype(float)

    bh, bc = h.copy(), cc
    oa = np.where(h == 1)[0]
    za = np.where(h == 0)[0]
    T, cool, sc = 0.02, 0.99998, 5000.0
    for it in range(ni):
        if time.time() > dl:
            break
        ai = rng.randint(len(za))
        bi = rng.randint(len(oa))
        i, j = int(za[ai]), int(oa[bi])
        d = swap_delta(h, i, j, n, mr)
        nc = (cr + d).max() * dx
        if nc <= cc or (T > 1e-12 and rng.random() < np.exp(-sc * (nc - cc) / T)):
            h[i], h[j] = 1.0, 0.0
            cr += d
            cc = nc
            za[ai], oa[bi] = j, i
            if cc < bc:
                bc, bh = cc, h.copy()
        T *= cool
    return bh, bc

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_max_val(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    sh = alpha * (corr - corr.max())
    return float(corr.max() + np.log(np.exp(sh).sum()) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    scaled = corr * dx
    sh = alpha * (scaled - scaled.max())
    ev = np.exp(sh)
    esum = ev.sum()
    w = ev / esum
    val = scaled.max() + np.log(esum) / alpha
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        s = k - (n - 1)
        lo = max(0, -s)
        hi = min(n, n - s)
        if lo < hi:
            grad[lo:hi] += wk * omh[lo + s:hi + s]
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 - s:hi2 - s]
    grad *= dx
    return val, grad

def grad_optimize(h, n, a0, a1, nit, dl):
    h = h.copy()
    a = a0
    lr = np.log(max(a1, a0 * 1.01) / a0) / max(nit, 1)
    bh, bc = h.copy(), compute_c5(h, n)
    stall = 0
    for it in range(nit):
        if time.time() > dl:
            break
        a *= np.exp(lr)
        v, g = smooth_max_grad(h, n, a)
        step = 1.0
        ok = False
        for _ in range(40):

            vn = smooth_max_val(hn, n, a)
            if vn < v - 1e-14:
                h = hn
                c = compute_c5(h, n)
                if c < bc:
                    bc, bh = c, h.copy()
                    stall = 0
                else:
                    stall += 1
                ok = True
                break
            step *= 0.5
        if not ok:
            stall += 1
        if stall > 100:
            break
    return bh, bc

def swap_delta(h, i, j, n, mr):
    d = np.zeros(2 * n - 1)
    im = i + mr
    m = (im >= 0) & (im < n)
    d[m] += 1.0 - h[im[m]]
    imn = i - mr
    m = (imn >= 0) & (imn < n)
    d[m] -= h[imn[m]]
    jm = j + mr
    m = (jm >= 0) & (jm < n)
    d[m] -= 1.0 - h[jm[m]]
    jmn = j - mr
    m = (jmn >= 0) & (jmn < n)
    d[m] += h[jmn[m]]
    d[n - 1] -= 2.0
    d[n - 1 + j - i] += 1.0
    d[n - 1 + i - j] += 1.0
    return d

def greed_bin(h0, n, dl, ms=300):
    mr = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h0, n).copy()
    cr = np.correlate(h, 1 - h, mode='full').astype(float)
    cc = cr.max() * dx
    bh, bc = h.copy(), cc
    ones = [int(x) for x in np.where(h == 1)[0]]
    zeros = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(ms):
        if time.time() > dl:
            break
        imp = False
        for i in zeros:
            if time.time() > dl:
                break
            for j in ones:
                d = swap_delta(h, i, j, n, mr)
                nc = (cr + d).max() * dx
                if nc < cc - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    cr += d
                    cc = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    imp = True
                    break
            if imp:
                break
        if not imp:
            break
        if cc < bc:
            bc, bh = cc, h.copy()
    return bh, bc

def anneal_bin(h0, n, dl, seed=42, ni=3000000):
    rng = np.random.RandomState(seed)
    mr = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h0, n).copy()
    cr = np.correlate(h, 1 - h, mode='full').astype(float)
    cc = cr.max() * dx
    bh, bc = h.copy(), cc
    oa = np.where(h == 1)[0]
    za = np.where(h == 0)[0]
    T, cool, sc = 0.015, 0.99997, 5000.0
    for it in range(ni):
        if time.time() > dl:
            break
        ai = rng.randint(len(za))
        bi = rng.randint(len(oa))
        i, j = int(za[ai]), int(oa[bi])
        d = swap_delta(h, i, j, n, mr)
        nc = (cr + d).max() * dx
        if nc <= cc or (T > 1e-12 and rng.random() < np.exp(-sc * (nc - cc) / T)):
            h[i], h[j] = 1.0, 0.0
            cr += d
            cc = nc
            za[ai], oa[bi] = j, i
            if cc < bc:
                bc, bh = cc, h.copy()
        T *= cool
    return bh, bc
