def grad_c_weights(h,w,n):
    dx=2.0/float(n)
    P=w[n-1:]
    Q=w[:n][::-1]
    p0=float(P[0])
    cp=np.concatenate(([0.0], np.cumsum(P)))
    cq=np.concatenate(([0.0], np.cumsum(Q)))
    idx=np.arange(n)
    A=(cq[idx+1]-p0)+cp[n-idx]
    bpos=sp_correlate(h,P,mode='full',method='fft')[n-1:n-1+n]
    bneg=sp_fftconvolve(Q,h,mode='full')[:n]
    bneg=bneg-p0*h
    cpos=sp_fftconvolve(P,h,mode='full')[:n]
    cneg=sp_correlate(h,Q,mode='full',method='fft')[n-1:n-1+n]
    cneg=cneg-p0*h
    grad=dx*(A - (bpos+bneg) - (cpos+cneg))
    return np.asarray(grad, dtype=float)

import time
import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.signal import fftconvolve as sp_fftconvolve
from scipy.optimize import minimize


def enforce_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    for _ in range(4):
        diff = target - float(h.sum())
        if abs(diff) <= 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-13)[0]
        else:
            idx = np.where(h > 1e-13)[0]
        if idx.size == 0:
            break
        h[idx] += diff / float(idx.size)
        h = np.clip(h, 0.0, 1.0)
    if h.size > 1:
        res = target - float(h[:-1].sum())
        if -1e-12 <= res <= 1.0 + 1e-12:
            h[-1] = float(np.clip(res, 0.0, 1.0))
    return h


def project_feasible(h, n):
    target = 0.5 * float(n)
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) > 1e-10:
        lo = min(float(h.min()), 0.0) - 1.0
        hi = max(float(h.max()), 1.0) + 1.0
        for _ in range(70):
            mid = 0.5 * (lo + hi)
            sm = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
            if sm >= target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    return enforce_sum(h, target)


def compute_c_vec(h, n):
    dx = 2.0 / float(n)
    c = sp_correlate(h, 1.0 - h, mode="full", method="fft")
    return c * dx


def compute_c5(h, n):
    return float(np.max(compute_c_vec(h, n)))


def smooth_lse(c, alpha):
    m = float(c.max())
    if alpha <= 1.0:
        return m
    z = alpha * (c - m)
    return float(m + np.log(np.sum(np.exp(z))) / alpha)


def softmax_weights(c, alpha):
    m = float(c.max())
    z = alpha * (c - m)
    w = np.exp(z)
    s = float(w.sum())
    if s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    return w


def grad_single(h, n, k):
    nv = n - abs(k)
    g = np.zeros(n)
    if k >= 0:
        g[:nv] = 1.0 - h[k:]
        g[k:] -= h[:nv]
    else:
        kk = -k
        g[kk:] = 1.0 - h[:nv]
        g[:nv] -= h[kk:]
    return g


def direct_grad_from_weights(h, w, n):
    dx = 2.0 / float(n)
    g = np.zeros(n)
    order = np.argsort(w)[::-1]
    for t in order:
        wt = float(w[t])
        if wt < 1e-13:
            break
        k = int(t - (n - 1))
        g += wt * grad_single(h, n, k)
    return dx * g


def fft_grad_from_weights(h, w, n):
    dx = 2.0 / float(n)
    P = w[n - 1:]
    Q = w[:n][::-1]
    p0 = float(P[0])

    cp = np.concatenate(([0.0], np.cumsum(P)))
    cq = np.concatenate(([0.0], np.cumsum(Q)))
    idx = np.arange(n)
    A = (cq[idx + 1] - p0) + cp[n - idx]

    bpos = sp_correlate(h, P, mode="full", method="fft")[n - 1:n - 1 + n].real
    bneg = sp_fftconvolve(Q, h, mode="full")[:n].real - p0 * h
    cpos = sp_fftconvolve(P, h, mode="full")[:n].real
    cneg = sp_correlate(h, Q, mode="full", method="fft")[n - 1:n - 1 + n].real - p0 * h

    return dx * (A - (bpos + bneg) - (cpos + cneg))


def smooth_grad(h, n, alpha):
    c = compute_c_vec(h, n)
    w = softmax_weights(c, alpha)
    nz = int(np.count_nonzero(w > 1e-10))
    if nz < max(8, n // 4):
        return direct_grad_from_weights(h, w, n)
    return fft_grad_from_weights(h, w, n)


def lbfgs_obj(x, n, alpha, pen):
    c = compute_c_vec(x, n)
    val = smooth_lse(c, alpha)
    s = float(x.sum()) - 0.5 * float(n)
    return float(val + pen * s * s)


def lbfgs_grad(x, n, alpha, pen):
    c = compute_c_vec(x, n)
    w = softmax_weights(c, alpha)
    if int(np.count_nonzero(w > 1e-10)) < max(8, len(x) // 4):
        g = direct_grad_from_weights(x, w, n)
    else:
        g = fft_grad_from_weights(x, w, n)
    s = float(x.sum()) - 0.5 * float(n)
    return g + (2.0 * pen * s)


def slsqp_obj(x, n, alpha):
    return smooth_lse(compute_c_vec(x, n), alpha)


def slsqp_grad(x, n, alpha):
    return smooth_grad(x, n, alpha)


def slsqp_eq(x, n):
    return float(x.sum()) - 0.5 * float(n)


def slsqp_jac(x, n):
    return np.ones(len(x))


def choose_maxiter(n, remaining, cap):
    if remaining < 25.0:
        return 0
    if n < 300:
        cap = min(cap, 180)
    elif n < 650:
        cap = min(cap, 110)
    else:
        cap = min(cap, 70)
    limit = max(1, int(remaining / 3.0))
    return int(max(1, min(cap, limit)))


def interpolate_to(h, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h, n_new)
    xold = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    xnew = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    hnew = np.interp(xnew, xold, h, left=float(h[0]), right=float(h[-1]))
    return project_feasible(hnew, n_new)


def binary_start(n, rng):
    h = np.zeros(n)
    count = n // 2
    idx = rng.choice(n, size=count, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)


def sine_start(n, f, phase, amp):
    h = 0.5 + amp * np.sin(np.pi * float(f) * np.arange(n) + phase)
    return project_feasible(h, n)


def random_fourier_start(n, rng, freq, scale):
    h = np.zeros(n)
    x = (np.arange(n) / float(n)) * 2.0
    for f in range(1, freq + 1):
        amp = scale / np.sqrt(float(f))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.sin(np.pi * float(f) * x + ph)
    h = 0.5 + h
    return project_feasible(h, n)


def lbfgs_phase(h, n, alpha, pen, maxiter):
    if maxiter <= 0:
        return h, compute_c5(h, n)
    c0 = compute_c5(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(
            lbfgs_obj, h, jac=lbfgs_grad, args=(n, alpha, pen),
            method="L-BFGS-B", bounds=bounds,
            options={"maxiter": maxiter, "ftol": 1e-14, "gtol": 1e-9, "maxls": 25}
        )
        h2 = project_feasible(res.x, n)
        c2 = compute_c5(h2, n)
        if np.isfinite(c2) and c2 <= c0 + 1e-6:
            return h2, c2
    except Exception:
        pass
    return h, c0


def slsqp_phase(h, n, alpha, maxiter):
    if maxiter <= 0:
        return h, compute_c5(h, n)
    c0 = compute_c5(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": slsqp_eq, "jac": slsqp_jac, "args": (n,)}]
        res = minimize(
            slsqp_obj, h, jac=slsqp_grad, args=(n, alpha),
            method="SLSQP", bounds=bounds, constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-14}
        )
        h2 = project_feasible(res.x, n)
        c2 = compute_c5(h2, n)
        if np.isfinite(c2) and c2 < c0:
            return h2, c2
    except Exception:
        pass
    return h, c0


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 3.0
    rng = np.random.RandomState(seed)

    h_best = None
    c_best = np.inf
    n_best = 500

    try:
        par = np.asarray(initial_h_values, dtype=float)
        if par.size > 0:
            n0 = len(par)
            h0 = project_feasible(par, n0)
            c0 = compute_c5(h0, n0)
            h_best = h0.copy()
            c_best = float(c0)
            n_best = int(n0)
    except Exception:
        pass

    if h_best is None:
        n0 = 500
        h_best = sine_start(n0, 1, 0.0, 0.45)
        c_best = compute_c5(h_best, n0)
        n_best = n0

    h_cur = h_best.copy()
    c_cur = float(c_best)
    n_cur = n_best

    for alpha in (20.0, 100.0, 500.0, 2500.0):
        rem = deadline - time.time()
        maxit = choose_maxiter(n_cur, rem, 180)
        if maxit < 8:
            break
        h2, c2 = lbfgs_phase(h_cur, n_cur, alpha, 80.0, maxit)
        if c2 < c_best - 1e-12:
            c_best = float(c2)
            h_best = h2.copy()
            n_best = n_cur
        if c2 <= c_best + 1e-9:
            h_cur, c_cur = h2, float(c2)
        else:
            h_cur, c_cur = h_best.copy(), float(c_best)

    for tn in (512, 768, 960):
        rem = deadline - time.time()
        if rem < 200.0:
            break
        if tn <= n_best:
            continue
        h_up = interpolate_to(h_best, n_best, tn)
        c_up = compute_c5(h_up, tn)
        for alpha in (60.0, 400.0, 2500.0):
            rem = deadline - time.time()
            maxit = choose_maxiter(tn, rem, 90)
            if maxit < 6:
                break
            h_up, c_up = lbfgs_phase(h_up, tn, alpha, 120.0, maxit)
        if c_up < c_best - 1e-12:
            c_best = float(c_up)
            h_best = h_up.copy()
            n_best = int(tn)

    n_small = 256
    best_small = None
    best_small_c = np.inf

    cands = []
    cands.append(interpolate_to(h_best, n_best, n_small))
    hp = interpolate_to(h_best, n_best, n_small)
    cands.append(project_feasible(hp + 0.035 * rng.normal(size=n_small), n_small))
    cands.append(sine_start(n_small, 2, 0.8, 0.45))
    cands.append(sine_start(n_small, 3, 2.1, 0.42))
    cands.append(sine_start(n_small, 4, 3.5, 0.38))
    cands.append(random_fourier_start(n_small, rng, 4, 0.25))
    cands.append(random_fourier_start(n_small, rng, 3, 0.30))
    cands.append(binary_start(n_small, rng))

    for hs in cands:
        rem = deadline - time.time()
        if rem < 260.0:
            break
        hs, cs = lbfgs_phase(hs, n_small, 80.0, 50.0, choose_maxiter(n_small, rem, 150))
        rem = deadline - time.time()
        if rem < 230.0:
            break
        hs, cs = lbfgs_phase(hs, n_small, 700.0, 80.0, choose_maxiter(n_small, rem, 110))
        if cs < best_small_c:
            best_small_c = float(cs)
            best_small = hs.copy()

    if best_small is not None and (deadline - time.time()) > 240.0:
        target = n_best if n_best > 768 else 768
        if target > 960:
            target = 960
        ht = interpolate_to(best_small, n_small, target)
        ct = compute_c5(ht, target)
        for alpha in (120.0, 1200.0):
            rem = deadline - time.time()
            maxit = choose_maxiter(target, rem, 80)
            if maxit < 6:
                break
            ht, ct = lbfgs_phase(ht, target, alpha, 120.0, maxit)
        if ct < c_best - 1e-12:
            c_best = float(ct)
            h_best = ht.copy()
            n_best = int(target)

    for alpha in (600.0, 3000.0, 12000.0, 30000.0):
        rem = deadline - time.time()
        maxit = choose_maxiter(n_best, rem, 120)
        if maxit < 4:
            break
        h2, c2 = lbfgs_phase(h_best, n_best, alpha, 160.0, maxit)
        if c2 < c_best - 1e-12:
            c_best = float(c2)
            h_best = h2.copy()

    if n_best <= 600 and (deadline - time.time()) > 120.0:
        for alpha in (1500.0, 6000.0):
            rem = deadline - time.time()
            maxit = choose_maxiter(n_best, rem, 45)
            if maxit < 8:
                break
            h2, c2 = slsqp_phase(h_best, n_best, alpha, maxit)
            if c2 < c_best - 1e-12:
                c_best = float(c2)
                h_best = h2.copy()

    rem = deadline - time.time()
    maxit = choose_maxiter(n_best, rem, 35)
    if maxit > 0:
        h2, c2 = lbfgs_phase(h_best, n_best, 15000.0, 250.0, maxit)
        if c2 < c_best - 1e-12:
            c_best = float(c2)
            h_best = h2.copy()

    h_best = project_feasible(h_best, n_best)
    c_best = compute_c5(h_best, n_best)
    return np.asarray(h_best, dtype=float), float(c_best), int(n_best)
