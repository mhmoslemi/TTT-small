def compute_c5(h, n=None):
    h = np.asarray(h, dtype=float)
    if n is None:
        n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / float(n)))


def repair_sum(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = float(n) / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[idx])
            if add <= 0.0:
                break
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            sub = min(-diff, h[idx])
            if sub <= 0.0:
                break
            h[idx] -= sub
    return h


def proj_box_sum(y, n):

def smooth_obj(x, alpha):
    if not np.all(np.isfinite(x)):
        return 1.0e100
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / float(n))

    u = alpha * (c - smax)
    umax = float(np.max(u))
    w = np.exp(u - umax)
    w_sum = float(np.sum(w))
    if w_sum <= 0.0:
        return smax
    return float(smax + (umax + np.log(w_sum)) / alpha)


def smooth_grad(x, alpha):
    if not np.all(np.isfinite(x)):
        return np.zeros_like(x)
    n = len(x)
    dx = 2.0 / float(n)

    umax = float(np.max(u))

    w /= np.sum(w)

    term1 = np.convolve(w, 1.0 - x, mode="full")[center:center + n]
    term2 = np.convolve(w[::-1], x, mode="full")[center:center + n]
    return dx * (term1 - term2)

    d = x.sum() - len(x) / 2.0

    d = x.sum() - len(x) / 2.0

                elif h[i] > 1.0:
                    h[i] = 1.0
                if h[j] < 0.0:
                    h[j] = 0.0
                elif h[j] > 1.0:
                    h[j] = 1.0
                cur = float(np.max(c))
                curc = cur * dx
                accepts += 1
                no_improve = 0
                if curc < best_c:
                    best_c = curc
                    best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > stall_limit:
                break

    final_h = repair_sum(best_h, n)
    final_c = compute_c5(final_h, n)
    if final_c < best_c:
        best_c = final_c
    return final_h, best_c


def smooth_lbfgs_chain(x, n, alphas, lams, deadline):
    best = repair_sum(np.asarray(x, dtype=float), n)
    best_c = compute_c5(best, n)
    cur = best.copy()
    bounds = [(0.0, 1.0)] * n

    if n <= 300:
        base = 500
    elif n <= 450:
        base = 350
    else:
        base = 250

    for ai, alpha in enumerate(alphas):
        if deadline is not None and time.time() > deadline:
            break
        lam = float(lams[ai % len(lams)])
        maxit = base
        rem = (deadline - time.time()) if deadline is not None else 1000.0
        if rem < 60.0:
            maxit = min(maxit, 120)
        elif rem < 20.0:
            maxit = min(maxit, 50)
        try:
            res = minimize(
                pen_obj,
                cur,
                jac=pen_jac,
                method="L-BFGS-B",
                bounds=bounds,
                args=(alpha, lam),
                options={"maxiter": maxit, "maxfun": 3 * maxit, "maxls": 40,
                         "ftol": 1e-12, "gtol": 1e-8, "disp": False}
            )
            y = np.clip(res.x, 0.0, 1.0)
            y = proj_box_sum(y, n)
            c = compute_c5(y, n)
            if np.isfinite(c):
                if c < best_c:
                    best_c = c
                    best = y.copy()
                    cur = y.copy()
                elif c < best_c + 0.05:
                    cur = y.copy()
        except Exception:
            pass

    return best, best_c


def smooth_slsqp_chain(x, n, alphas, deadline):
    best = repair_sum(np.asarray(x, dtype=float), n)
    best_c = compute_c5(best, n)
    cur = best.copy()
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_constraint, "jac": sum_jac}

    if n <= 300:
        base = 350
    elif n <= 450:
        base = 220
    else:
        base = 150

    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        maxit = base
        rem = (deadline - time.time()) if deadline is not None else 1000.0
        if rem < 60.0:
            maxit = min(maxit, 90)
        elif rem < 20.0:
            maxit = min(maxit, 35)
        try:
            res = minimize(
                smooth_obj,
                cur,
                jac=smooth_grad,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                args=(alpha,),
                options={"maxiter": maxit, "ftol": 1e-11, "disp": False}
            )
            y = np.clip(res.x, 0.0, 1.0)
            y = proj_box_sum(y, n)
            c = compute_c5(y, n)
            if np.isfinite(c):
                if c < best_c:
                    best_c = c
                    best = y.copy()
                    cur = y.copy()
                elif c < best_c + 0.03:
                    cur = y.copy()
        except Exception:
            pass

    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    total = budget - 10.0
    if total < 30.0:
        total = max(10.0, budget - 2.0)
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = proj_box_sum(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    alphas = [20.0, 100.0, 500.0, 2000.0, 10000.0, 50000.0, 250000.0, 1200000.0]
    lams = [50.0, 150.0, 600.0, 2500.0]

    hp, cp = pair_descent(best_h, best_n, 70000, t0 + 0.20 * total, rng, 15000)
    if cp < best_c:
        best_h, best_c = hp, cp

    if time.time() < dl - 60.0:
        hp, cp = smooth_lbfgs_chain(best_h, best_n, alphas, lams, t0 + 0.54 * total)
        if cp < best_c:
            best_h, best_c = hp, cp
        hp, cp = pair_descent(best_h, best_n, 45000, t0 + 0.66 * total,
                              np.random.RandomState(seed + 101), 11000)
        if cp < best_c:
            best_h, best_c = hp, cp

    if time.time() < dl - 80.0:
        hp, cp = smooth_slsqp_chain(best_h, best_n, alphas[-4:], t0 + 0.76 * total)
        if cp < best_c:
            best_h, best_c = hp, cp
        hp, cp = pair_descent(best_h, best_n, 35000, t0 + 0.84 * total,
                              np.random.RandomState(seed + 102), 9000)
        if cp < best_c:
            best_h, best_c = hp, cp

    if time.time() < dl - 120.0:
        for sigma in (0.02, 0.06, 0.15):
            if time.time() > dl - 90.0:
                break
            hy = best_h + sigma * rng.normal(size=best_n)
            hy = proj_box_sum(hy, best_n)
            tsub = min(dl - 70.0, time.time() + 70.0)
            hy2, ch2 = pair_descent(hy, best_n, 30000, tsub, rng, 7000)
            if ch2 < best_c:
                best_h, best_c = hy2, ch2
            if time.time() < tsub - 10.0:
                hy3, ch3 = smooth_lbfgs_chain(hy2, best_n, alphas[2:], lams, tsub)
                if ch3 < best_c:
                    best_h, best_c = hy3, ch3
                if time.time() < min(dl - 60.0, time.time() + 40.0):
                    hy4, ch4 = pair_descent(hy3, best_n, 20000,
                                            min(dl - 60.0, time.time() + 25.0), rng, 5000)
                    if ch4 < best_c:
                        best_h, best_c = hy4, ch4

    if time.time() < dl - 100.0:
        offsets = (-30, -10, 10, 30, 100, 200)
        cands = []
        for off in offsets:
            nh = int(best_n + off)
            if 120 <= nh <= 900 and nh != best_n and nh not in cands:
                cands.append(nh)

        alphas_short = alphas[3:]
        for nh in cands:
            if time.time() > dl - 70.0:
                break
            xo = np.linspace(0.0, 2.0, best_n)
            xn = np.linspace(0.0, 2.0, nh)
            hh = np.interp(xn, xo, best_h)
            hh = proj_box_sum(hh, nh)
            tsub = min(dl - 55.0, time.time() + max(20.0, 0.04 * total))
            hh2, ch2 = pair_descent(hh, nh, 35000, tsub, rng, 8000)
            if ch2 < best_c:
                best_h, best_c, best_n = hh2, ch2, nh
            if time.time() < min(dl - 45.0, time.time() + 30.0):
                hh3, ch3 = smooth_lbfgs_chain(hh2, nh, alphas_short, lams, tsub)
                if ch3 < best_c:
                    best_h, best_c, best_n = hh3, ch3, nh
            if time.time() < min(dl - 40.0, time.time() + 15.0):
                hh4, ch4 = pair_descent(hh2, nh, 15000,
                                        min(dl - 40.0, time.time() + 12.0), rng, 4000)
                if ch4 < best_c:
                    best_h, best_c, best_n = hh4, ch4, nh

    if time.time() < dl - 10.0:
        hp, cp = pair_descent(best_h, best_n, 30000, dl - 5.0,
                              np.random.RandomState(seed + 207), 7000)
        if cp < best_c:
            best_h, best_c = hp, cp

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_box_sum(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
