def project_feasible(h):
    h = np.array(h, dtype=float, copy=True)

    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += diff * room / tot
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += diff * h / tot
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:

            idxs = np.argsort(1.0 - h)[::-1]
            for idx in idxs:
                room = 1.0 - h[idx]

        best_c = c
    return best_h, best_c, best_raw

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 4.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float, copy=True)
        n0 = int(h0.size)
    except Exception:
        h0 = default_start(500, rng)
        n0 = 500

    if n0 < 8:
        h0 = default_start(500, rng)
        n0 = 500

    h0 = project_feasible(h0)
    best_h, best_c, best_raw = local_search(h0, n0, deadline - 3.0, rng, max_pert=30)
    best_n = n0
    best_c5 = c5_from_raw(best_raw, best_n)

    if deadline - time.time() > 160.0:
        for n_try in (256, 384):
            if time.time() > deadline - 80.0:
                break
            if n_try == best_n:
                continue
            kept = False
            slice_s = min(130.0, deadline - time.time() - 40.0)
            if slice_s < 60.0:
                break
            dl_try = time.time() + slice_s
            try:
                h_try = resample_h(best_h, best_n, n_try)
                h_t, c_t, raw_t = local_search(
                    h_try, n_try, dl_try, rng, max_pert=8, gi_it=34, ri_it=32
                )
                c5_t = c5_from_raw(raw_t, n_try)
                if c5_t < best_c5 - 1e-12:
                    best_h = h_t
                    best_c = c_t
                    best_n = n_try
                    best_c5 = c5_t
                    kept = True
            except Exception:
                kept = False
            if kept:
                break

    best_h = project_feasible(best_h)
    c_fin = raw_correlate(best_h)
    best_c5 = c5_from_raw(float(np.max(c_fin)), best_n)
    return best_h, float(best_c5), int(best_n)
