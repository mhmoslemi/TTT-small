    """Find optimal h[j] given all other coordinates fixed."""
    dx = 2.0 / n
    # Compute base correlations (with h[j] set to 0)
    h[j] = 0.0
    c_base = np.correlate(h, 1-h, mode='full') * dx
    h[j] = 1.0
    c_one = np.correlate(h, 1-h, mode='full') * dx
    h[j] = 0.0
    
    # c[m] = c_base[m] + x * (c_one[m] - c_base[m]) for linear terms
    # But for k=0, there's also -x^2 term
    # c_base has h[j]=0, c_one has h[j]=1
    # For k=0: c[n-1] with h[j]=0: sum of h[i](1-h[i]) for i≠j
    #           with h[j]=1: sum + 1*(1-1) = sum (same! since 1*(1-1)=0)
    # Wait: h[j]*(1-h[j]) when h[j]=0 is 0, when h[j]=1 is 0.
    # So c_base[n-1] = sum_{i≠j} h[i](1-h[i])
    # c_one[n-1] = sum_{i≠j} h[i](1-h[i]) + 1*(1-1) = same
    # So the difference c_one[n-1] - c_base[n-1] = 0
    # But the true value at h[j]=x is c_base[n-1] + x - x^2
    # So we need to handle k=0 specially.

    # For m = n-1: true value is c_base[n-1] + x - x^2
    #              approximated as c_base[n-1] + x * diff[n-1] = c_base[n-1] (since diff[n-1]=0)
    #              So we need to add x - x^2 - x*diff[n-1] = x - x^2 (since diff[n-1]=0)

    # c_one[m] has h[j]=1

    obj = vals.max(axis=1)

    x_opt = xs[idx]

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def slsqp_constraint(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(n)

def coord_descent_sweep(h, n, mu=1.0, num_samples=100):
    """One sweep of coordinate descent with exact 1D optimization."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    S = h.sum()  # current sum
    for j in range(n):
        # Set h[j] to 0 and 1 to get base and difference
        old_val = h[j]
        h[j] = 0.0
        c0 = np.correlate(h, 1 - h, mode='full') * dx
        h[j] = 1.0
        c1 = np.correlate(h, 1 - h, mode='full') * dx
        h[j] = old_val
        
        diff = c1 - c0  # linear coefficients
        S_other = S - old_val  # sum of all other elements
        
        # Objective: max(c0 + x*diff, with c0[n-1] + x - x^2 for k=0) + mu*(S_other + x - n/2)^2
        # Evaluate on grid
        xs = np.linspace(0, 1, num_samples)
        vals = c0[None, :] + xs[:, None] * diff[None, :]
        vals[:, n-1] = c0[n-1] + xs - xs**2
        # Add penalty
        pen = mu * (S_other + xs - n / 2.0) ** 2
        vals_max = vals.max(axis=1) + pen
        idx = vals_max.argmin()
        x_opt = xs[idx]
        
        # Local refinement
        if 0 < idx < num_samples - 1:
            lo, hi = xs[idx-1], xs[idx+1]
            gr = 0.618025
            for _ in range(40):
                m1 = lo + (1-gr)*(hi-lo)
                m2 = lo + gr*(hi-lo)
                # Evaluate at m1
                v1_arr = c0 + m1 * diff
                v1_arr[n-1] = c0[n-1] + m1 - m1**2
                v1 = v1_arr.max() + mu * (S_other + m1 - n/2.0)**2
                v2_arr = c0 + m2 * diff
                v2_arr[n-1] = c0[n-1] + m2 - m2**2
                v2 = v2_arr.max() + mu * (S_other + m2 - n/2.0)**2
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            x_opt = (lo + hi) / 2
        
        h[j] = x_opt
        S = S_other + x_opt
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
    
    return h, best_c

def lbfgs_optimize(h, n, deadline, max_iters=500):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    configs = [
        (100, 10.0), (500, 50.0), (2000, 200.0),
        (10000, 1000.0), (50000, 5000.0), (200000, 20000.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(penalty_obj, h, jac=penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 20,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def slsqp_optimize(h, n, deadline, alpha=100000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_constraint, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

```python
def compute_coeff(j, h, n):
    """Compute coefficient of h[j] in each correlation c[m]."""
    ncorr = 2 * n - 1
    coeff = np.zeros(ncorr)
    # Positive shifts k = 1, ..., n-1 (m = n-1+k)
    ks = np.arange(1, n)
    # m indices for positive shifts
    ms = n - 1 + ks
    # (1-h[j+k]) where j+k < n
    valid1 = j + ks < n
    coeff[ms[valid1]] += (1.0 - h[j + ks[valid1]])
    # -h[j-k] where j >= k
    valid2 = j >= ks
    coeff[ms[valid2]] -= h[j - ks[valid2]]
    # Negative shifts: kk = 1, ..., n-1 (m = n-1-kk)
    msn = n - 1 - ks
    # (1-h[j-kk]) where j >= kk
    valid3 = j >= ks
    coeff[msn[valid3]] += (1.0 - h[j - ks[valid3]])
    # -h[j+kk] where j+kk < n
    valid4 = j + ks < n
    coeff[msn[valid4]] -= h[j + ks[valid4]]
    return coeff
```

This is O(n) with vectorized operations, which is fast.

Now let me put it all together. I'll aim for a clean, efficient implementation.

Actually, I just realized there might be an issue with my incremental update. When I update h[j] from old to new, the correlation c[m] changes by:
- For m ≠ n-1: delta * coeff[m] where coeff[m] is computed with the CURRENT h values (before updating h[j])
- For m = n-1: (new-old)*(1-new-old) = delta*(1-new-old)

But wait, the coefficient for m ≠ n-1 is:
coeff[m] = (1-h[j+k])*[j+k<n] - h[j-k]*[j>=k]  (for k>0)

This is computed with the current h (before updating h[j]). But h[j] doesn't appear in this coefficient (it's h[j+k] and h[j-k], not h[j] itself). So the coefficient is correct regardless of the value of h[j]. Good.

For m = n-1: the change is h[j]*(1-h[j]) going from old*(1-old) to new*(1-new).
= new - new² - old + old²
= (new-old) - (new+old)(new-old)
= delta * (1 - new - old)

So: c_new[n-1] = c[n-1] + delta*(1-new-old)

Let me verify: if old=0.3, new=0.7:
c_new[n-1] - c[n-1] = 0.7*0.3 - 0.3*0.7 = 0 (correct, since 0.7*0.3 = 0.3*0.7)
delta*(1-new-old) = 0.4*(1-0.7-0.3) = 0.4*0 = 0 ✓

If old=0.2, new=0.5:
c_new - c_old = 0.5*0.5 - 0.2*0.8 = 0.25 - 0.16 = 0.09
delta*(1-new-old) = 0.3*(1-0.5-0.2) = 0.3*0.3 = 0.09 ✓

Great, the incremental update is correct.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def slsqp_con(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(n)

def coord_sweep(h, n, mu=1.0, G=80, refine_iters=40):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    S = h.sum()
    nh = n / 2.0
    ncorr = 2 * n - 1
    ks = np.arange(1, n)
    xs_grid = np.linspace(0.0, 1.0, G)
    gr = 0.618025
    for j in range(n):
        old = h[j]
        So = S - old
        coeff = np.zeros(ncorr)
        ms = n - 1 + ks
        v1 = j + ks < n
        if v1.any():
            coeff[ms[v1]] += (1.0 - h[j + ks[v1]])
        v2 = j >= ks
        if v2.any():
            coeff[ms[v2]] -= h[j - ks[v2]]
        msn = n - 1 - ks
        if v2.any():
            coeff[msn[v2]] += (1.0 - h[j - ks[v2]])
        if v1.any():
            coeff[msn[v1]] -= h[j + ks[v1]]
        const = c - old * coeff
        c0n1 = c[n - 1] - old + old * old
        vals = const[None, :] + xs_grid[:, None] * coeff[None, :]
        vals[:, n - 1] = c0n1 + xs_grid - xs_grid ** 2
        pen = mu * (So + xs_grid - nh) ** 2
        obj = vals.max(axis=1) + pen
        idx = obj.argmin()
        x_opt = xs_grid[idx]
        if 0 < idx < G - 1:
            lo, hi = xs_grid[idx - 1], xs_grid[idx + 1]
            for _ in range(refine_iters):
                m1 = lo + (1.0 - gr) * (hi - lo)
                m2 = lo + gr * (hi - lo)
                v1 = np.max(const + m1 * coeff)
                vn1 = c0n1 + m1 - m1 * m1
                v1 = max(v1, vn1) + mu * (So + m1 - nh) ** 2
                v2 = np.max(const + m2 * coeff)
                vn1 = c0n1 + m2 - m2 * m2
                v2 = max(v2, vn1) + mu * (So + m2 - nh) ** 2
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            x_opt = (lo + hi) * 0.5
        delta = x_opt - old
        if abs(delta) > 1e-16:
            c += delta * coeff
            c[n - 1] = c0n1 + x_opt - x_opt * x_opt
            h[j] = x_opt
            S = So + x_opt
    return h, float(c.max())

def lbfgs_opt(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in [(100, 10.0), (500, 20.0), (2000, 50.0),
                      (10000, 200.0), (50000, 1000.0), (200000, 5000.0)]:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(penalty_obj, h, jac=penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 15,
                                   'maxls': 40})
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c - 1e-15:
                best_c = cn
                best_h = hn.copy()
                h = hn
        except Exception:
            pass
    return best_h, best_c

def slsqp_opt(h, n, deadline, alpha=200000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        hn = project_feasible(res.x, n)
        return hn, compute_c5(hn, n)
    except Exception:
        return h.copy(), compute_c5(h, n)
