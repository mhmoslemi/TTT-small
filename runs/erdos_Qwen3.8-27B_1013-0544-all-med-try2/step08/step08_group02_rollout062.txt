def compute_R(h, n):
    """Compute R[s] = sum h[i]*(1-h[i+s]) for shift s, indexed from -(n-1) to n-1."""
    R = np.zeros(2*n-1)
    for s in range(-(n-1), n):
        idx = s + (n-1)
        if s >= 0:
            R[idx] = np.sum(h[:n-s] * (1-h[s:]))
        else:
            R[idx] = np.sum(h[-s:] * (1-h[:n+s]))
    return R
```

def swap_delta(i, j, h, n):
    """Compute the change in R for swapping h[i]: 1->0, h[j]: 0->1"""
    # s ranges from -(n-1) to n-1, indexed as s + (n-1)
    s = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    # Term 1: -(1-h[i+s]) for valid i+s
    mask1 = (s >= -i) & (s <= n-1-i)
    if np.any(mask1):
        delta[mask1] -= (1 - h[i + s[mask1]])
    
    # Term 2: +h[i-s] for valid i-s
    mask2 = (s >= i-(n-1)) & (s <= i)
    if np.any(mask2):
        delta[mask2] += h[i - s[mask2]]
    
    # Term 3: +(1-h[j+s]) for valid j+s
    mask3 = (s >= -j) & (s <= n-1-j)
    if np.any(mask3):
        delta[mask3] += (1 - h[j + s[mask3]])
    
    # Term 4: -h[j-s] for valid j-s
    mask4 = (s >= j-(n-1)) & (s <= j)
    if np.any(mask4):
        delta[mask4] -= h[j - s[mask4]]
    
    return delta
```

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    a = alpha * (c - m)
    # a.max() = 0
    w = np.exp(a)
    val = m + np.log(w.sum()) / alpha
    s = x.sum() - n / 2.0
    return val + mu * s * s

def smooth_max_grad(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    a = alpha * (c - m)
    w = np.exp(a)
    w_sum = w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-15 * w_sum:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] / w_sum
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def lbfgs_optimize(h, n, deadline, alphas=None):
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    mu = 1000.0
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': 10000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c
