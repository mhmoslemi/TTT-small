def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return float(m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def pen_obj(x, n, alpha, mu):
    return smooth_max_val(x, n, alpha) + mu * (x.sum() - n/2.0)**2

def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - n/2.0
    g += 2.0 * mu * s

def compute_delta_fast(h, n, ip, iq):
    """Compute the change in the full correlation vector when h[ip]: 1->0, h[iq]: 0->1."""
    delta = np.zeros(2*n - 1)
    
    # For positive shift k (c index = n-1+k), k = 0..n-1:
    # Δc_k = -(1-h[ip+k])*[ip<=n-1-k] + h[ip-k]*[k<=ip] + (1-h[iq+k])*[iq<=n-1-k] - h[iq-k]*[k<=iq]
    
    # ip direct: valid for k <= n-1-ip, c index from n-1 to n-1+(n-1-ip)
    k_max_ip = n - 1 - ip
    for k in range(k_max_ip + 1):
        if ip + k < n:
            delta[n-1+k] += -(1 - h[ip+k])
    
    # ip indirect: valid for k <= ip, c index from n-1 to n-1+ip
    for k in range(min(ip, n-1) + 1):
        if k >= 1:  # k=0 means ip-k = ip, which is the same as direct
            delta[n-1+k] += h[ip-k]
    
    # iq direct: valid for k <= n-1-iq
    k_max_iq = n - 1 - iq
    for k in range(k_max_iq + 1):
        if iq + k < n:
            delta[n-1+k] += (1 - h[iq+k])
    
    # iq indirect: valid for k <= iq
    for k in range(min(iq, n-1) + 1):
        if k >= 1:
            delta[n-1+k] += -h[iq-k]
    
    # For negative shift -k (c index = n-1-k), k = 1..n-1:
    # Δc_{-k} = -(1-h[ip-k])*[k<=ip] + h[ip+k]*[ip<=n-1-k] + (1-h[iq-k])*[k<=iq] - h[iq+k]*[iq<=n-1-k]
    
    # ip direct (as h[i]): valid for k <= ip
    for k in range(1, min(ip, n-1) + 1):
        delta[n-1-k] += -(1 - h[ip-k])
    
    # ip indirect (as h[i-k]): valid for ip <= n-1-k, i.e., k <= n-1-ip
