import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def grad_active_set(h, n, alpha, active_indices):
    """Gradient of smooth max over only active_indices shifts."""
    c = compute_c5_vec(h, n)
    ca = c[active_indices]
    c_shift = alpha * (ca - ca.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx, m in enumerate(active_indices):
        if w[idx] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def smooth_max_val(h, n, alpha, active_indices=None):
    if active_indices is not None:
        c = compute_c5_vec(h, n)[active_indices]
    else:
        c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def lbfgs_optimize(h, n, deadline, alphas=None, max_iters=500):
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        tr = deadline - time.time() - 3
        if tr < 5:
            break
        # Define objective with penalty for constraint
        def obj(x):
            c = compute_c5_vec(x, n)
            m = c.max()
            sh = alpha * (c - m)
            mx = sh.max()
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
            s = x.sum() - target
            val += 1e4 * s * s
            return float(val)
        
        def gobj(x):
            g = grad_smooth_max(x, n, alpha)
            s = x.sum() - target
            g += 2e4 * s
            return g
        
        try:
            res = minimize(obj, h, jac=gobj, method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-18,
                                  'gtol': 1e-16, 'maxfun': max_iters * 20,
                                  'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def active_set_lbfgs(h, n, deadline, m_active=10, alphas=None, max_iters=300):

    if alphas is None:
        alphas = [5000, 50000, 500000, 5000000]

            s = x.sum() - target

            return g

def spsa_optimize(h, n, deadline, a=0.03, c=0.01, gamma=0.55, betas=0.55, seed=0):

        c_k = c / (k + 1)**betas

        d = target - w_plus.sum()
        if abs(d) > 1e-12:
            if d > 0:
                r = (1 - w_plus).sum()
                if r > 1e-12: w_plus += (d/r) * (1 - w_plus)
            else:
                t2 = w_plus.sum()
                if t2 > 1e-12: w_plus += (d/t2) * w_plus
            w_plus = np.clip(w_plus, 0, 1)
        
        w_minus = np.clip(w - c_k * delta, 0, 1)
        d = target - w_minus.sum()

            if d > 0:

                if t2 > 1e-12: w += (d/t2) * w
            w = np.clip(w, 0, 1)
        
        c_w = compute_c5(w, n)
        if c_w < best_c:
            best_c = c_w
            best_h = w.copy()
        k += 1
    return best_h, best_c

def pair_polish(h, n, deadline, radius=60):
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    best_c = compute_c5(h, n)
    for d in deltas:
        any_improved = True
        while any_improved and time.time() < deadline:
            any_improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from i to j
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try moving mass from j to i
                    if h[j] + d <= 1.0 and h[i] - d >= 0.0:
                        h[j] += d
                        h[i] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_improved = True
                        else:
                            h[j] -= d
                            h[i] += d
    return h, best_c

def gradient_pair_search(h, n, deadline, step=0.001, window=50):
    """Use gradient to guide pair moves."""
    best_c = compute_c5(h, n)
    alpha = 10000
    while time.time() < deadline:
        g = grad_smooth_max(h, n, alpha)
        order = np.argsort(g)
        improved = False
        for idx in range(n):
            if time.time() > deadline - 2:
                return h, best_c
            i = order[idx]
            lo = max(0, idx - window)
            hi = min(n, idx + window + 1)
            for j_idx in range(lo, hi):
                j = order[j_idx]
                if i == j:
                    continue
                if g[j] - g[i] < 1e-10:
                    continue
                # Move mass from high-gradient to low-gradient
                if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                    h[i] += step
                    h[j] -= step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= step
                        h[j] += step
                # Also try opposite direction
                if h[i] - step >= 0.0 and h[j] + step <= 1.0:
                    h[i] -= step
                    h[j] += step
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i] += step
                        h[j] -= step
        if not improved:
            if alpha < 100000:
                alpha *= 10
            else:
                break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, active_idx):
    c = compute_c5_vec(x, n)
    if active_idx is not None:
        c = c[active_idx]
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = x.sum() - n / 2.0
    val += 1e4 * s * s
    return float(val)

def smooth_max_grad(x, n, alpha, active_idx):
    c = compute_c5_vec(x, n)
    idx = active_idx if active_idx is not None else np.arange(len(c))
    ca = c[idx]
    c_shift = alpha * (ca - ca.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for ii, m in enumerate(idx):
        if w[ii] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[ii] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    s = x.sum() - n / 2.0
    grad += 2e4 * s
    return grad

def lbfgs_optimize(h, n, deadline, alphas, max_iters=400):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 4:
            break
        if deadline - time.time() < 8:
            break
        def obj(x):
            return smooth_max_obj(x, n, alpha, None)
        def gobj(x):
            return smooth_max_grad(x, n, alpha, None)
        try:
            res = minimize(obj, h, jac=gobj, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'ftol': 1e-18,
                                    'gtol': 1e-16, 'maxfun': max_iters * 25,
                                    'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def spsa_optimize(h, n, deadline, a=0.015, c=0.004, gamma=0.5, betas=0.5, seed=0):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    w = h.copy()
    k = 1
    target = n / 2.0
    while time.time() < deadline:
        a_k = a / (k + 10) ** gamma
        c_k = c / (k + 1) ** betas
        delta = rng.choice([-1, 1], size=n)
        wp = np.clip(w + c_k * delta, 0, 1)
        d = target - wp.sum()
        if abs(d) > 1e-12:
            if d > 0:
                r = (1 - wp).sum()
                if r > 1e-12:
                    wp += (d / r) * (1 - wp)
            else:
                t2 = wp.sum()
                if t2 > 1e-12:
                    wp += (d / t2) * wp
            wp = np.clip(wp, 0, 1)
        wm = np.clip(w - c_k * delta, 0, 1)
        d = target - wm.sum()
        if abs(d) > 1e-12:
            if d > 0:
                r = (1 - wm).sum()
                if r > 1e-12:

            else:
                t2 = wm.sum()
                if t2 > 1e-12:
                    wm += (d / t2) * wm
            wm = np.clip(wm, 0, 1)
        f_plus = compute_c5(wp, n)
        f_minus = compute_c5(wm, n)
        g = (f_plus - f_minus) / (2 * c_k) * delta
        w = np.clip(w - a_k * g, 0, 1)
        d = target - w.sum()
        if abs(d) > 1e-12:
            if d > 0:
                r = (1 - w).sum()

                    w += (d / r) * (1 - w)

                if t2 > 1e-12:
                    w += (d / t2) * w
            w = np.clip(w, 0, 1)
        c_w = compute_c5(w, n)
        if c_w < best_c:
            best_c = c_w
            best_h = w.copy()
        k += 1
    return best_h, best_c

def pair_polish(h, n, deadline, radius=60):
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    best_c = compute_c5(h, n)
    for d in deltas:
        any_imp = True
        while any_imp and time.time() < deadline:
            any_imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[j] + d <= 1.0 and h[i] - d >= 0.0:
                        h[j] += d
                        h[i] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_imp = True
                        else:
                            h[j] -= d
                            h[i] += d
    return h, best_c

def gradient_pair_search(h, n, deadline, step=0.0005, window=50):
