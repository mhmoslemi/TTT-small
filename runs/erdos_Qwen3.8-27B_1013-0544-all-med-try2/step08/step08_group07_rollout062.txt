```python
def sca_linearize(h, n):
    """Return (coeffs, consts) such that c_k ≈ consts[k] + coeffs[k] @ h for all k."""
    # c_k = sum_{i in I_k} h[i] * (1 - h[i+lag_k])
    # Linearized: c_k* + sum_{i in I_k} (1-h[i+lag_k])*(h[i]-h[i]_*) - sum_{i in I_k} h[i]_*(h[i+lag_k]-h[i+lag_k]_*)
    
    # For each k, the coefficient of h[j] is:
    # (1-h[j-lag_k]) if j-lag_k is a valid index for the "i" position
    # -h[j+lag_k] if j+lag_k is a valid index for the "i" position
    
    coeffs = np.zeros((2*n-1, n))
    consts = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        lag = k - (n-1)
        # Valid i: 0 <= i < n and 0 <= i+lag < n
        if lag >= 0:
            i_start, i_end = 0, n - lag
        else:
            i_start, i_end = -lag, n
        
        cc = 0.0
        for i in range(i_start, i_end):
            # c_k* contribution
            cc += h[i] * (1 - h[i+lag])
            # Linearized: (1-h[i+lag])*(h[i]-h[i]) - h[i]*(h[i+lag]-h[i+lag])
            # Coef of h[i]: (1-h[i+lag])
            # Coef of h[i+lag]: -h[i]
            coeffs[k, i] += (1 - h[i+lag])
            coeffs[k, i+lag] += -h[i]
        
        consts[k] = cc  # This is c_k* (the value at h*)
        # But the linearization is: c_k* + sum_j coeff[k,j]*(h[j]-h[j]_*)
        # = c_k* - sum_j coeff[k,j]*h[j]_* + sum_j coeff[k,j]*h[j]
        consts[k] = cc - np.dot(coeffs[k], h)
    
    return coeffs, consts
```

Then the LP is:
minimize t
subject to: consts[k] + coeffs[k] @ h ≤ t, for all k
            0 ≤ h ≤ 1
            sum h = n/2

This is a linear program with n+1 variables and 2n-1+n+1 = 3n constraints. For n=400, that's 401 variables and 1201 constraints. Very tractable!

But wait - the SCA approach might not converge to a good solution. It converges to a fixed point of the mapping h → argmin LP(h). This fixed point is a KKT point of the original problem.

For non-convex problems, KKT points can be local minima, maxima, or saddle points. In our case, since we're minimizing, a KKT point should be a local minimum (or at least not worse than the starting point, if we're careful).

Actually, for SCA to make progress, we need the linearization to be an underestimate of the true function. For a non-convex function (like a bilinear form), the linearization can be either an overestimate or underestimate depending on the direction.

Specifically, for f(x,y) = x*y, the linearization at (x*,y*) is:
f ≈ x*·y + x·y* - x*·y*

The true value is f = x·y. The difference is:
f - f_lin = x·y - x*·y - x·y* + x*·y* = (x-x*)(y-y*)

This can be positive or negative depending on the signs of (x-x*) and (y-y*). So the linearization is not a global underestimator.

    """SCA with cutting plane for speed."""
    import cvxpy as cp
    n2 = 2*n - 1

    corr = np.correlate(h, 1-h, mode='full')
    max_val = corr.max()
    active = np.where(corr > max_val - 0.1)[0]
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    base_constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n/2
    ]

        constraints = base_constraints.copy()

                i_idx = np.arange(0, n - lag)
                j_idx = i_idx + lag
            else:

            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(42):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        shifted = alpha * (g - g.max())
        sm = shifted.max()
        return g.max() + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def sca_step(h, n):
    n2 = 2 * n - 1
    coeffs = np.zeros((n2, n))
    consts = np.zeros(n2)
    for k in range(n2):
        lag = k - (n - 1)
        if lag >= 0:
            ii = np.arange(0, n - lag)
            jj = ii + lag
        else:
            ii = np.arange(-lag, n)
            jj = ii + lag
        if len(ii) == 0:
            consts[k] = 0.0
            continue
        cc = np.sum(h[ii] * (1 - h[jj]))
        coeffs[k, ii] += (1 - h[jj])
        coeffs[k, jj] += -h[ii]
        consts[k] = cc - np.dot(coeffs[k], h)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2,
                   consts + coeffs @ h_var <= t_var]
    solved = False
    for solver in [cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]:
        try:
            prob = cp.Problem(cp.Minimize(t_var), constraints)
            prob.solve(solver=solver)
            solved = True
            break
        except Exception:
            continue
    if not solved or h_var.value is None:
        return h, compute_c5(h, n)
    h_new = project_feasible(np.clip(h_var.value, 0, 1), n)
    c_new = compute_c5(h_new, n)
    c_curr = compute_c5(h, n)
    if c_new < c_curr:
        return h_new, c_new
    return h, c_curr

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 20
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Strong coordinate descent
    d1 = t0 + 0.22 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=800, deadline=d1, seed=seed)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 2: SCA iterations with interspersed coordinate descent
    d2 = t0 + 0.52 * total
    h = best_h.copy()
    for it in range(80):
        if time.time() > d2 - 5:
            break
        h, c = sca_step(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        hc2, _ = coord_descent(h, n, max_sweeps=15, deadline=dl, seed=seed + it)
        hc2 = project_feasible(hc2, n)
        cc2 = compute_c5(hc2, n)
        if cc2 < best_c:
            best_c, best_h = cc2, hc2.copy()
            h = hc2.copy()

    # Phase 3: SLSQP with gradual alpha
    d3 = t0 + 0.78 * total
    h = best_h.copy()
    alphas = [1e3, 5e3, 1e4, 5e4, 1e5, 5e5, 1e6, 5e6, 1e7, 5e7, 1e8, 5e8, 1e9]
    for alpha in alphas:
        if time.time() > d3 - 5:
            break
        h = slsqp_smooth(h, n, alpha, 3000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Final coordinate descent
    d4 = t0 + 0.88 * total
    hc3, _ = coord_descent(best_h, n, max_sweeps=400, deadline=d4, seed=seed + 500)
    hc3 = project_feasible(hc3, n)
    cc3 = compute_c5(hc3, n)
    if cc3 < best_c:
        best_c, best_h = cc3, hc3.copy()

    # Phase 5: Final SCA polish
    if time.time() < dl - 15:
        h = best_h.copy()
        for it in range(30):
            if time.time() > dl - 5:
                break
            h, c = sca_step(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
            hc4, _ = coord_descent(h, n, max_sweeps=10, deadline=dl - 3, seed=seed + 800 + it)
            hc4 = project_feasible(hc4, n)
            cc4 = compute_c5(hc4, n)
            if cc4 < best_c:
                best_c, best_h = cc4, hc4.copy()
                h = hc4.copy()

    # Phase 6: Try a slightly different resolution if time allows
    if time.time() < dl - 30 and n >= 350:
        nh = n // 2
        xo = np.linspace(0, 2, n + 1)
        xn = np.linspace(0, 2, nh + 1)
        hh = np.interp(xn[:-1], xo, np.append(best_h, best_h[-1]))
        hh = project_feasible(hh, nh)
        hh, _ = coord_descent(hh, nh, max_sweeps=100, deadline=dl - 20, seed=seed + 900)
        hh = project_feasible(hh, nh)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh
            for alpha in [1e6, 1e7, 1e8]:
                if time.time() > dl - 5:
                    break
                hh = slsqp_smooth(hh, nh, alpha, 2000, 1e-16)
                ch2 = compute_c5(hh, nh)
                if ch2 < best_c:
                    best_c, best_h, n = ch2, hh.copy(), nh

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
