    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        # For stability, subtract max before exp
        shifted = alpha * (g - m)
        sm = shifted.max()  # should be 0
        logsum = np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        weights = np.exp(shifted - shifted.max())
        weights /= weights.sum()
        # Gradient of log(sum(exp(alpha*(g-m)))) / alpha
        # = sum(weights * alpha * dg/dx) / alpha = sum(weights * dg/dx)
        # + gradient of m (subgradient of max)
        # For simplicity, use the smoothed gradient only
        grad = np.zeros(n)
        for k_idx in range(len(weights)):
            if weights[k_idx] < 1e-10:
                continue
            m_p = k_idx - (n - 1)
            # d/dx_j [h_j * (1-h_{j+m_p})] (with validity)
            # = (1 - h_{j+m_p}) for the h_j term
            # - h_{j-m_p} for the (1-h_{j+m_p}) term when j = i+m_p
            idx = np.arange(n)
            vr = (idx + m_p) >= 0
            vl = (idx + m_p) < n
            grad[vr & vl] += weights[k_idx] * (1 - x[idx[vr & vl] + m_p])
            vl2 = (idx - m_p) >= 0
            vr2 = (idx - m_p) < n
            grad[vl2 & vr2] -= weights[k_idx] * x[idx[vl2 & vr2] - m_p]
        grad *= dx
        return grad
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        return m + np.log(np.exp(shifted - shifted.max()).sum()) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        w = np.exp(shifted - shifted.max())
        w /= w.sum()
        grad = np.zeros(n)
        for k_idx in range(len(w)):
            wi = w[k_idx]
            if wi < 1e-10:
                continue
            m_p = k_idx - (n - 1)
            idx = np.arange(n)
            mask_r = (idx + m_p >= 0) & (idx + m_p < n)
            mask_l = (idx - m_p >= 0) & (idx - m_p < n)
            grad[mask_r] += wi * (1.0 - x[idx[mask_r] + m_p])
            grad[mask_l] -= wi * x[idx[mask_l] - m_p]
        grad *= dx
        return grad
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)

The third term: ∑_k w_k * h_{j-k} * 1{valid} is also a correlation.

            if s < 1e-14:

**Another idea: Use the fact that we're dealing with a max of smooth functions**

The "smooth max" (log-sum-exp) is a standard technique. The issue is that at very high alpha, the approximation is good but the optimization landscape becomes difficult (nearly non-differentiable). 

A better approach might be to use a **penalty method**:
min_h max_k f_k(h) ≈ min_h (max_k f_k(h) + μ * ∑_k (max(0, f_k(h) - max_k f_k(h) + ε))² )

But this is circular. 

Actually, a better penalty: 
min_h t + μ * ∑_k (max(0, f_k(h) - t))²

This is equivalent to the epigraph formulation but with a smooth penalty. For large μ, it approximates the original problem. And it's smooth everywhere!

The gradient is:
∂/∂h: ∑_k μ * 2 * max(0, f_k(h) - t) * ∇f_k(h)
∂/∂t: 1 - μ * ∑_k max(0, f_k(h) - t)

This is a nice smooth objective that we can minimize with standard methods (even BFGS or L-BFGS-B with the box constraint, handling the sum constraint with a penalty or parameterization).

def penalty_grad_x(x, n, t, mu):

    g = np.correlate(x, 1.0 - x, mode='full') * dx
    excess = np.maximum(0, g - t)
    # d/dx_j t + mu * sum_k excess_k^2
    # = mu * sum_k 2*excess_k * df_k/dx_j
    grad = np.zeros(n)

        if excess[k] < 1e-12:
            continue
        coeff = 2.0 * mu * excess[k]

        js_r = max(0, -mp)
        je_r = min(n, n - mp)

            grad[js_r:je_r] += coeff * (1.0 - x[js_r + mp:je_r + mp])
        js_l = max(0, mp)
        je_l = min(n, n + mp)
        if je_l > js_l:
            grad[js_l:je_l] -= coeff * x[js_l - mp:je_l - mp]
    grad *= dx
    return grad

    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    s = alpha * (g - m)

    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    s = alpha * (g - m)

    grad *= dx
