        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.zeros(n)
        h_init[rng.choice(n, n // 2, replace=False)] = 1.0

    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B with increasing alpha
    t1_end = min(t0 + 280, deadline - 320)
    for alpha in [100, 400, 1500, 5000, 15000, 50000]:
        if time.time() > t1_end:
            break
        try:
            res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-13})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass

    # Phase 2: PGD with all-lag gradient
    if time.time() < deadline - 250:
        t2_end = min(time.time() + 100, deadline - 200)
        for alpha in [800, 3000, 12000]:
            if time.time() > t2_end:
                break
            h, c_new = _pgd_loop(h, n, t2_end, alpha, rng)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()

    # Phase 3: High-alpha L-BFGS-B
    if time.time() < deadline - 120:
        t3_end = min(time.time() + 80, deadline - 100)
        for alpha in [50000, 100000, 200000]:
            if time.time() > t3_end:
                break
            try:
                res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(n, alpha),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-13})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
            except Exception:
                pass

    # Phase 4: Final PGD
    if time.time() < deadline - 80:
        t4_end = min(time.time() + 50, deadline - 60)
        h, c_new = _pgd_loop(h, n, t4_end, 50000, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()

    # Phase 5: Pair polish
    if time.time() < deadline - 2:
        best_h, best_c = _pair_polish(best_h, n, deadline - 1, radius=70)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
