```
import numpy as np
import time
try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

def corr_vals(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_count(h):
    return float(np.max(corr_vals(h)))

def c5_value(h, n):
    return c5_count(h) * (2.0 / n)

def project_simple(y, target):
    y = np.asarray(y, dtype=float)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        z = np.clip(y - mid, 0.0, 1.0)
        if float(z.sum()) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - float(z.sum())
    if abs(diff) > 1e-11:
        if diff > 0:
            room = 1.0 - z
            rs = float(room.sum())
            if rs > 1e-15:
                z = z + diff * room / rs
        else:
            ss = float(z.sum())
            if ss > 1e-15:
                z = z + diff * z / ss
    return np.clip(z, 0.0, 1.0)

def make_binary(h, n):
    h = np.asarray(h, dtype=float)
    if h.size != n:
        x_o = np.linspace(0.0, 1.0, h.size)
        x_n = np.linspace(0.0, 1.0, n)
        h = np.interp(x_n, x_o, h)
    m = n // 2
    idx = np.argsort(-h, kind='stable')[:m]
    x = np.zeros(n, dtype=np.int32)
    x[idx] = 1
    return x
```

```
def smooth_max(f, beta):
    f = np.asarray(f, dtype=float)
    m = float(np.max(f))
    s = float(np.sum(np.exp(f - m)))
    if s <= 0: return m
    return m + np.log(s) / beta

def lag_grad(h, n, s):
    g = np.zeros(n)
    if s >= 0:
        L = n - s
        if L > 0:
            g[:L] += 1.0 - h[s:]
            g[s:] -= h[:L]
    else:
        a = -s
        L = n - a
        if L > 0:
            g[a:] += 1.0 - h[:L]
            g[:L] -= h[a:]
    return g
```

```
def gradient_refine(h_in, n, deadline, betas=(10.0,30.0,100.0), step0=0.08, top_k=50):
    target = float(n) * 0.5
    h = project_simple(h_in, target)
    best_h = h.copy()
    best_c = c5_count(h)
    for beta in betas:
        if time.time() > deadline - 5: break
        f = corr_vals(h); S = smooth_max(f, beta)
        step = step0; noimp = 0
        while time.time() < deadline - 5 and noimp < 25:
            f = corr_vals(h); cur = float(np.max(f))
            m = cur
            ez = np.exp(beta*(f-m))
            p = ez / max(ez.sum(), 1e-300)
            if top_k >= len(p):
                active = np.arange(len(p))
            else:
                active = np.argsort(p)[-top_k:]
            pw = p[active]; ps = float(pw.sum())
            if ps <= 0: break
            pw = pw / ps
            G = np.zeros(n)
            mid = n - 1
            for idx, w in zip(active, pw):
                G += w * lag_grad(h, n, int(idx - mid))
            norm = float(np.linalg.norm(G))
            if norm < 1e-12:
                break
            Gn = G / norm
            oldS = S; oldC = cur

            for _ in range(10):

                    break
```

```
import numpy as np
```

```python
def corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def ccount(h):
    return float(np.max(corr(h)))

def c5(h, n):
    return ccount(h) * (2.0 / n)

def project_simple(y, target):
    y = np.asarray(y, float)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        z = np.clip(y - mid, 0.0, 1.0)
        if float(z.sum()) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(z.sum())
    if abs(diff) > 1e-11:
        if diff > 0:
            rs = float((1.0 - z).sum())
            z = np.clip(z + diff * (1.0 - z) / max(rs, 1e-15), 0.0, 1.0)
        else:
            ss = float(z.sum())
            z = np.clip(z + diff * z / max(ss, 1e-15), 0.0, 1.0)
        diff = target - float(z.sum())
        if abs(diff) > 1e-13:
            i = int(np.argmax((1.0 - z) if diff > 0 else z))
            z[i] += diff
    return np.clip(z, 0.0, 1.0)

def make_binary(h, n):
    h = np.asarray(h, float)
    if h.size != n:
        h = np.interp(np.linspace(0.0, 1.0, n), np.linspace(0.0, 1.0, h.size), h)
    x = np.zeros(n, np.int32)
    x[np.argsort(-h)[:n // 2]] = 1
    return x

def smooth_max(f, beta):
    m = float(f.max())
    s = float(np.exp(f - m).sum())
    return m + (np.log(s) / beta if s > 0 else 0.0)

def lag_grad(h, n, s):
    g = np.zeros(n)
    if s >= 0:
        L = n - s
        if L:
            g[:L] += 1.0 - h[s:]
            g[s:] -= h[:L]
    else:
        a = -s
        L = n - a
        if L:
            g[a:] += 1.0 - h[:L]
            g[:L] -= h[a:]
    return g

def gradient_refine(h_in, n, deadline, betas=(10.0, 30.0, 100.0), step0=0.08, top_k=50):
    target = 0.5 * n
    h = project_simple(h_in, target)
    best_h = h.copy()
    best_c = ccount(h)
    for beta in betas:
        if time.time() > deadline - 5:
            break
        f = corr(h)
        S = smooth_max(f, beta)
        step = step0
        noimp = 0
        while time.time() < deadline - 5 and noimp < 25:
            f = corr(h)
            cur = float(f.max())
            ez = np.exp(beta * (f - cur))
            p = ez / max(float(ez.sum()), 1e-300)
            active = np.arange(len(p)) if top_k >= len(p) else np.argsort(p)[-top_k:]
            pw = p[active]
            ps = float(pw.sum())
            if ps <= 0:
                break
            pw /= ps
            G = np.zeros(n)
            mid = n - 1
            for idx, w in zip(active, pw):
                G += w * lag_grad(h, n, int(idx - mid))
            norm = float(np.linalg.norm(G))
            if norm < 1e-12:
                break
            Gn = G / norm
            oldS = S
            oldC = cur
            ok = False
            for _ in range(10):
                h1 = project_simple(h - step * Gn, target)
                f1 = corr(h1)
                S1 = smooth_max(f1, beta)
                c1 = float(f1.max())
                if S1 < oldS - 1e-12 or (abs(S1 - oldS) < 1e-10 and c1 < oldC - 1e-9):
                    h = h1
                    S = S1
                    ok = True
                    step = min(0.5, step * 1.05)
                    noimp = 0
                    if c1 < best_c:
                        best_c = c1
                        best_h = h.copy()
                    break
                step *= 0.5
            if not ok:
                noimp += 1
    return best_h, best_c

def lp_refine(h_in, n, deadline):

def binary_AB(xf, zi, oj, n, Ls):
    A = np.zeros((zi.size, Ls.size), np.float32)
    B = np.zeros((oj.size, Ls.size), np.float32)
    im = zi[:, None] + Ls[None, :]
    v = (im >= 0) & (im < n)
    A[v] = 1.0 - xf[np.clip(im, 0, n - 1)[v]]
    im = zi[:, None] - Ls[None, :]
    v = (im >= 0) & (im < n)
    A[v] -= xf[np.clip(im, 0, n - 1)[v]]
    im = oj[:, None] + Ls[None, :]
    v = (im >= 0) & (im < n)
    B[v] = xf[np.clip(im, 0, n - 1)[v]] - 1.0
    im = oj[:, None] - Ls[None, :]
    v = (im >= 0) & (im < n)
    B[v] += xf[np.clip(im, 0, n - 1)[v]]
    return A, B

def find_best_swap(n, C, A, B, zi, oj):
    z = zi.size
    o = oj.size
    L = len(C)
    if z == 0 or o == 0:
        return np.inf, -1, -1
    D = oj[None, :] - zi[:, None]
    idxp = n - 1 + D
    idxm = n - 1 - D
    best_val = np.inf
    best_r = -1
    best_c = -1
    chunk = max(1, min(z, 2000000 // max(1, o * L)))
    col_rows = np.arange(o)[None, :]
    for a in range(0, z, chunk):
        b = min(a + chunk, z)
        abs_rows = np.arange(a, b)[:, None]
        S = A[a:b, None, :] + B[None, :, :]
        S += C[None, None, :]
        M = S.max(axis=2)
        del S
        idxpc = idxp[a:b]
        idxmc = idxm[a:b]
        base_d = C[idxpc] + A[abs_rows, idxpc] + B[col_rows, idxpc]
        base_m = C[idxmc] + A[abs_rows, idxmc] + B[col_rows, idxmc]
        np.maximum(M, base_d + 1.0, out=M)
        np.maximum(M, base_m + 1.0, out=M)
        flat = int(np.argmin(M))
        val = float(M.flat[flat])
        if val < best_val:
            best_val = val
            best_r = a + flat // o
            best_c = flat % o
    return best_val, best_r, best_c

def greedy_binary(x_in, n, deadline, allow_equal=0, max_moves=5000):
    x = np.asarray(x_in, np.int32).copy()
    m = n // 2
    if int(x.sum()) != m:
        x = make_binary(x.astype(float), n)
    xf = x.astype(np.float32)
    C = np.correlate(x, 1 - x, mode='full').astype(np.float32)
    cur = float(np.max(C))
    best_c = cur
    best_x = x.copy()
    zi = np.where(x == 0)[0].astype(np.intp)
    oj = np.where(x == 1)[0].astype(np.intp)
    Ls = np.arange(-(n - 1), n - 1, dtype=np.intp)
    equal_left = allow_equal
    moves = 0
    while moves < max_moves and time.time() < deadline - 2:
        A, B = binary_AB(xf, zi, oj, n, Ls)
        val, ri, cj = find_best_swap(n, C, A, B, zi, oj)
        if ri < 0:
            break
        strict = val < cur - 1e-6
        equal = val <= cur + 1e-6
        if strict or (equal and equal_left > 0):
            i = int(zi[ri])
            j = int(oj[cj])
            x[i] = 1
            x[j] = 0
            xf[i] = 1.0
            xf[j] = 0.0
            zi[ri] = j
            oj[cj] = i
            C = np.correlate(x, 1 - x, mode='full').astype(np.float32)
            cur = float(np.max(C))
            moves += 1
            if strict:
                equal_left = allow_equal if allow_equal > 0 else 0
            else:
                equal_left -= 1
            if cur < best_c - 1e-9:
                best_c = cur
                best_x = x.copy()
        else:
            break
    return best_x, best_c

def binary_optimize_from(h, n, deadline, seed=42, tries=3, allow_equal=300):
    rng = np.random.RandomState(seed)
    best_x = None
    best_c = np.inf
    for tr in range(tries):
        if time.time() > deadline - 20:
            break
        rem = deadline - time.time()
        if tr > 0 and rem < 60:
            break
        if tr > 1 and rem < 150:
            break
        if tr == 0:
            x = make_binary(h, n)
        elif tr == 1:
            if best_x is not None:
                x = best_x.copy()
                zi = np.where(x == 0)[0]
                oj = np.where(x == 1)[0]
                for _ in range(3):
                    a = int(rng.randint(zi.size))
                    b = int(rng.randint(oj.size))
                    i = int(zi[a])
                    j = int(oj[b])
                    x[i] = 1
                    x[j] = 0
                    zi[a] = j
                    oj[b] = i
            else:
                x = make_binary(h + 0.05 * rng.normal(size=n), n)
        else:
            if best_x is not None and rng.rand() < 0.5:
                x = best_x.copy()
                zi = np.where(x == 0)[0]
                oj = np.where(x == 1)[0]
                for _ in range(5):
                    a = int(rng.randint(zi.size))
                    b = int(rng.randint(oj.size))
                    i = int(zi[a])
                    j = int(oj[b])
                    x[i] = 1
                    x[j] = 0
                    zi[a] = j
                    oj[b] = i
            else:
                idx = rng.choice(n, n // 2, replace=False)
                x = np.zeros(n, np.int32)
                x[idx] = 1
        sub = time.time() + max(20.0, min(rem, rem / max(1, tries - tr) * 1.4))
        if sub > deadline - 5:
            sub = deadline - 5
        x, c = greedy_binary(x, n, sub, allow_equal=allow_equal if tr == 0 else 0)
        if c < best_c:
            best_c = c
            best_x = x.copy()
    if best_x is None:
        best_x = make_binary(h, n)
        best_c = ccount(best_x)
    return best_x, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    dl = t0 + max(5.0, total - 8.0)
    span = max(1.0, total - 12.0)
    try:
        h0 = np.asarray(initial_h_values, float).copy()
        n = int(n) if False else int(len(h0))
    except Exception:
        h0 = np.full(250, 0.5)
        n = 250
    if n < 20 or n % 2:
        nt = n + 1 if n % 2 else n
        if nt < 50:
            nt = 250
        if h0.size:
            h0 = np.interp(np.linspace(0.0, 1.0, nt), np.linspace(0.0, 1.0, h0.size), h0)
        else:
            h0 = np.full(nt, 0.5)
        n = nt
    target = 0.5 * n
    if float(np.std(h0)) < 1e-6:
        rr = np.random.RandomState(seed)
        h0 = 0.5 + 0.25 * rr.random(n)
    h0 = project_simple(h0, target)
    dx = 2.0 / n
    best_h = h0.copy()
    best_c = c5(best_h, n)

    p1 = t0 + 0.10 * span
    if p1 > time.time() + 3:
        if linprog is not None:
            h1, c1 = lp_refine(h0, n, p1)
        else:
            h1, c1 = gradient_refine(h0, n, p1)
        if c1 * dx < best_c:
            best_c = c1 * dx
            best_h = h1

    p2 = t0 + 0.48 * span
    if n <= 350 and p2 > time.time() + 30:
        xb, cb = binary_optimize_from(best_h, n, p2, seed=seed, tries=3, allow_equal=300)
        if cb * dx < best_c:
            best_c = cb * dx
            best_h = xb.astype(float)

    p3a = t0 + 0.65 * span
    if linprog is not None and p3a > time.time() + 30:
        h1, c1 = lp_refine(best_h, n, p3a)
        if c1 * dx < best_c:
            best_c = c1 * dx
            best_h = h1

    p3b = t0 + 0.75 * span
    if p3b > time.time() + 10:
        h1, c1 = gradient_refine(best_h, n, p3b)
        if c1 * dx < best_c:
            best_c = c1 * dx
            best_h = h1

    p4 = t0 + 0.88 * span
    if n <= 350 and p4 > time.time() + 30:
        xb, cb = binary_optimize_from(best_h, n, p4, seed=seed + 11, tries=2, allow_equal=150)
        if cb * dx < best_c:
            best_c = cb * dx
            best_h = xb.astype(float)

    if dl > time.time() + 15:
        if linprog is not None:
            h1, c1 = lp_refine(best_h, n, dl)
        else:
            h1, c1 = gradient_refine(best_h, n, dl)
        if c1 * dx < best_c:
            best_c = c1 * dx
            best_h = h1

    best_h = np.asarray(best_h, float)
    if abs(float(best_h.sum()) - target) > 1e-9 or np.any(best_h < -1e-12) or np.any(best_h > 1.0 + 1e-12):
        best_h = project_simple(best_h, target)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c = c5(best_h, n)
    return best_h, best_c, int(n)
```
