def grad_smooth(h,n,alpha):
    c=compute_cvec(h,n)
    cm=c.max()
    sh=alpha*(c-cm)
    # sh max maybe 0; but due floating maybe max 0. 
    mx=sh.max()
    if not np.isfinite(mx): return np.zeros(n)
    e=np.exp(sh-mx)
    s=e.sum()
    if s<=0 or not finite: w=np.zeros(len(c)); w[int(np.argmax(c))]=1.0
    else: w=e/s
    g=np.zeros(n)
    n1=n-1
    # active = np.where(w>1e-12)[0]
    active=np.nonzero(w>1e-12)[0]
    if len(active)==0:
       active=np.array([int(np.argmax(c))])
    for m in active:
       wm=float(w[m])
       if wm<=1e-14: continue
       s=m-n1
       if s>=0:
          nv=n-s
          g[:nv] += wm*(1.0-h[s:])
          g[s:] += -wm*h[:nv]
       else:
          kk=-s
          nv=n-kk
          g[kk:] += wm*(1.0-h[:nv])
          g[:nv] += -wm*h[kk:]
    return g

def proj(h, n):

    d = 0.5 * n - h.sum()
    if d > 1e-13:
        i = int(np.argmax(1.0 - h))
        if 0.0 <= h[i] + d <= 1.0:
            h[i] += d
        else:
            h = proj(h, n)
    elif d < -1e-13:
        i = int(np.argmax(h))
        if 0.0 <= h[i] + d <= 1.0:
            h[i] += d
        else:
            h = proj(h, n)
    return np.clip(h, 0.0, 1.0)


def grad(h, n, alpha):
    cv = cc(h, n)
    cm = cv.max()
    sh = alpha * (cv - cm)
    e = np.exp(sh)
    sw = float(e.sum())
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.zeros(len(cv))
        w[int(np.argmax(cv))] = 1.0
    else:
        w = e / sw
    g = np.zeros(n, dtype=float)
    n1 = n - 1
    for m in np.nonzero(w > 1e-14)[0]:
        wm = float(w[m])
        s = int(m) - n1
        if s >= 0:
            nv = n - s
            g[:nv] += wm * (1.0 - h[s:])
            g[s:] -= wm * h[:nv]
        else:
            kk = -s
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] -= wm * h[kk:]
    return g


def obj(x, n, alpha, mu):
    cv = cc(x, n)
    cm = cv.max()
    sh = alpha * (cv - cm)
    sw = float(np.exp(sh).sum())
    val = cm + np.log(max(sw, 1e-300)) / alpha
    val += mu * (x.sum() - 0.5 * n) ** 2
    return float(val)


def gobj(x, n, alpha, mu):
    g = grad(x, n, alpha)
    return g + 2.0 * mu * (x.sum() - 0.5 * n)


def lbfgs(h, n, dl, alphas, maxit=50):
    best_h = proj(h, n)
    best_c = c5(best_h, n)
    if time.time() >= dl:
        return best_h, best_c
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > dl - 4:
            break
        try:
            r = minimize(obj, best_h, jac=gobj, args=(n, a, 1.0), method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': maxit, 'ftol': 1e-16,
                                                 'gtol': 1e-13, 'maxfun': maxit * 25})
            h2 = proj(r.x, n)
            c2 = c5(h2, n)
            if c2 < best_c:
                best_h, best_c = h2, c2
        except Exception:
            pass
    return best_h, best_c


def search(h, d, n, ccur, step_cap):
    d = np.asarray(d, dtype=float).copy()
    d -= d.mean()
    dm = float(np.max(np.abs(d)))
    if dm < 1e-14:
        return None, None
    d /= dm
    pos = np.where(d > 0.0)[0]
    neg = np.where(d < 0.0)[0]
    lammax = float(step_cap)
    if pos.size:
        lammax = min(lammax, float(np.min((1.0 - h[pos]) / d[pos])) * 0.999)
    if neg.size:
        lammax = min(lammax, float(np.min((-h[neg]) / d[neg])) * 0.999)
    if not np.isfinite(lammax) or lammax <= 1e-15:
        return None, None
    for f in (1.0, 0.75, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02):
        lam = lammax * f
        if lam <= 1e-16:
            continue
        h2 = h + lam * d
        if h2.min() >= -1e-12 and h2.max() <= 1.0 + 1e-12:
            h2 = fix(h2, n)
            c2 = c5(h2, n)
            if c2 < ccur - 1e-15:
                return h2, c2
    return None, None


def active(h, n, best_c, dl):
    h = proj(h, n)
    best_c = c5(h, n)
    dx = 2.0 / n
    reds = (5e-5, 1e-4, 2e-4, 4e-4, 8e-4, 1.5e-3, 3e-3, 6e-3)
    for red in reds:
        if time.time() > dl:
            break
        rho = max(red / dx, 1e-8)
        prog = True
        while prog and time.time() < dl:
            cv = cc(h, n)
            ccur = float(cv.max() * dx)
            if ccur < best_c:
                best_c = ccur
            cmax = float(cv.max())
            idx = np.where(cv >= cmax - 0.5 * rho)[0]
            if idx.size == 0:
                idx = np.array([int(np.argmax(cv))], dtype=int)
            if idx.size > 12:
                idx = idx[np.argsort(cv[idx])[::-1]][:12]
            G = np.empty((int(idx.size), n), dtype=float)
            n1 = n - 1
            for a, m in enumerate(idx):
                s = int(m) - n1
                g = np.zeros(n, dtype=float)
                if s >= 0:
                    nv = n - s
                    g[:nv] = 1.0 - h[s:]
                    g[s:] -= h[:nv]
                else:
                    kk = -s
                    nv = n - kk
                    g[kk:] = 1.0 - h[:nv]
                    g[:nv] -= h[kk:]
                G[a] = g
            G -= G.mean(axis=1, keepdims=True)
            tgt = cv[idx] - (cmax - rho)
            tgt = np.minimum(tgt, 0.0)
            try:
                d = np.linalg.lstsq(G, tgt, rcond=None)[0]
            except Exception:
                d = None
            if d is None or not np.all(np.isfinite(d)) or float(np.max(np.abs(d))) < 1e-10:
                d = -grad(h, n, 20000.0)
            prog = False
            for cap in (0.25, 0.12, 0.06, 0.03, 0.015):
                h2, c2 = search(h, d, n, ccur, cap)
                if h2 is not None:
                    h = h2
                    best_c = min(best_c, c2)
                    prog = True
                    break
    return h, best_c


def pair(h, n, best_c, dl):
    h = proj(h, n)
    best_c = c5(h, n)
    K0 = 24
    for a in (2000.0, 20000.0, 200000.0, 2000000.0):
        for _ in range(300):
            if time.time() > dl:
                break
            cv = cc(h, n)
            ccur = float(cv.max() * (2.0 / n))
            if ccur < best_c:
                best_c = ccur
            g = grad(h, n, a)
            g -= g.mean()
            if float(g.max() - g.min()) < 1e-12:
                break
            K = min(K0, n)
            dest = np.argsort(g)[:K]
            src = np.argsort(g)[::-1][:K]
            D = g[dest, None] - g[None, src]
            D[np.arange(K), np.arange(K)] = np.inf
            found = False
            tried = 0
            for p in np.argsort(D.ravel()):
                if tried > 60:
                    break
                ii = int(p // K)
                jj = int(p % K)
                if D[ii, jj] >= 0.0:
                    break
                i = int(dest[ii])
                j = int(src[jj])
                base = min(float(h[j]), float(1.0 - h[i]))
                if base <= 1e-14:
                    continue
                tried += 1
                for fr in (0.5, 0.3, 0.15, 0.08, 0.04):
                    d = base * fr
                    if h[i] + d > 1.0 or h[j] - d < 0.0:
                        continue
                    hi, hj = h[i], h[j]
                    h[i] = hi + d
                    h[j] = hj - d
                    c2 = c5(h, n)
                    if c2 < best_c - 1e-15:
                        h = fix(h, n)
                        best_c = c2
                        found = True
                        break
                    h[i], h[j] = hi, hj
                if found:
                    break
            if not found:
                break
    return h, best_c


def binarize(h, n):
    h2 = np.zeros(n, dtype=float)
    half = n // 2
    if half > 0:
        idx = np.argsort(h)[::-1][:half]
        h2[idx] = 1.0
    if n % 2:
        i = int(np.argmax(h))
        if h2[i] == 0.0:
            h2[i] = 0.5
    return fix(h2, n)


def binary(h, n, best_c, dl):
    best_h = proj(h, n)
    best_c = c5(best_h, n)
    hb = binarize(h, n)
    bc = c5(hb, n)
    if bc < best_c:
        best_h, best_c = hb.copy(), bc
    K0 = 24
    for a in (5000.0, 50000.0, 500000.0):
        for _ in range(300):
            if time.time() > dl:
                break
            cv = cc(hb, n)
            ccur = float(cv.max() * (2.0 / n))
            if ccur < bc:
                bc = ccur
            if bc < best_c:
                best_h, best_c = hb.copy(), bc
            g = grad(hb, n, a)
            z = np.where(hb < 0.5)[0]
            o = np.where(hb > 0.5)[0]
            Kz = min(K0, int(z.size))
            Ko = min(K0, int(o.size))
            if Kz == 0 or Ko == 0:
                break
            dest = z[np.argsort(g[z])[:Kz]]
            src = o[np.argsort(g[o])[::-1][:Ko]]
            D = g[dest, None] - g[None, src]
            found = False
            tried = 0
            for p in np.argsort(D.ravel()):
                if tried > 50:
                    break
                ii = int(p // Ko)
                jj = int(p % Ko)
                if D[ii, jj] >= 0.0:
                    break
                i = int(dest[ii])
                j = int(src[jj])
                hb[i] = 1.0
                hb[j] = 0.0
                c2 = c5(hb, n)
                if c2 < bc - 1e-15:
                    bc = c2
                    found = True
                    if bc < best_c:
                        best_h, best_c = hb.copy(), bc
                else:
                    hb[i] = 0.0
                    hb[j] = 1.0
                if found:
                    break
            if not found:
                break
    return best_h, best_c


def st_golden(n):
    h = np.zeros(n, dtype=float)
    half = n // 2
    if half > 0:
        pos = (np.arange(half) * 1.618033988749895).astype(np.int64) % n
        h[pos] = 1.0
    if n % 2:
        h[-1] += 0.5
    return proj(h, n)


def st_rand(n, rng):
    h = np.zeros(n, dtype=float)
    half = n // 2
    if half > 0:
        idx = rng.choice(n, half, replace=False)
        h[idx] = 1.0
    if n % 2:
        i = int(rng.randint(n))
        h[i] += 0.5
    return proj(h, n)


def st_fourier(n, rng):
    h = np.full(n, 0.5, dtype=float)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    for f in range(1, 9):
        amp = rng.uniform(0.1, 0.4)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(np.pi * f * x + ph)
    return proj(h, n)


def seq(x, n):
    return float(x.sum() - 0.5 * n)


def seqj(x, n):
    return np.ones(x.size, dtype=float)


def slsqp(h, n, dl, alpha, maxit=30):
    best_h = proj(h, n)
    best_c = c5(best_h, n)
    if time.time() >= dl:
        return best_h, best_c
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': seq, 'jac': seqj, 'args': (n,)}]
    try:
        r = minimize(obj, best_h, jac=gobj, args=(n, alpha, 0.0), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        h2 = proj(r.x, n)
        c2 = c5(h2, n)
        if c2 < best_c:
            best_h, best_c = h2, c2
    except Exception:
        pass
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 6
    rng = np.random.RandomState(seed)
    n = 500

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None or len(h0) < 20:
        cands = [st_fourier(n, rng), st_golden(n), st_rand(n, rng)]
        best_h = cands[0]
        best_c = c5(best_h, n)
        for s in cands[1:]:
            ccand = c5(s, n)
            if ccand < best_c:
                best_h, best_c = s, ccand
    else:
        if len(h0) != n:
            xo = np.linspace(0.0, 2.0, len(h0), endpoint=False)
            xn = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(xn, xo, h0)
        best_h = proj(h0, n)
        best_c = c5(best_h, n)

    if time.time() < dl - 330:
        best_h, best_c = lbfgs(best_h, n, min(time.time() + 70, dl - 330),
                               (10000.0, 100000.0, 1000000.0), 55)
    if time.time() < dl - 270:
        best_h, best_c = active(best_h, n, best_c, min(time.time() + 240, dl - 270))
    if time.time() < dl - 210:
        best_h, best_c = pair(best_h, n, best_c, min(time.time() + 90, dl - 210))
    if time.time() < dl - 165:
        best_h, best_c = binary(best_h, n, best_c, min(time.time() + 80, dl - 165))
    if time.time() < dl - 145:
        best_h, best_c = active(best_h, n, best_c, min(time.time() + 70, dl - 145))

    if best_c > 0.38085 and time.time() < dl - 230:
        for st in (st_golden(n), st_fourier(n, rng)):
            if time.time() > dl - 120:
                break
            h2, c2 = lbfgs(st, n, min(time.time() + 35, dl - 100),
                           (50000.0, 500000.0), 40)
            h2, c2 = active(h2, n, c2, min(time.time() + 45, dl - 80))
            h2, c2 = pair(h2, n, c2, min(time.time() + 15, dl - 60))
            if c2 < best_c:
                best_h, best_c = h2, c2

    if time.time() < dl - 70:
        best_h, best_c = slsqp(best_h, n, dl, 2000000.0, 35)
    if time.time() < dl - 30:
        best_h, best_c = pair(best_h, n, best_c, min(time.time() + 20, dl - 30))
    if time.time() < dl - 10:
        best_h, best_c = active(best_h, n, best_c, min(time.time() + 10, dl - 10))

    best_h = proj(best_h, n)
    best_c = c5(best_h, n)
    return best_h, best_c, n
