import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def active_set_gradient(h, n, top_m=8):
    c = compute_c5_vec(h, n)
    m = c.max()
    active = np.where(c >= m - 1e-12)[0]
    if len(active) > top_m:
        active = np.argsort(c)[::-1][:top_m]
    elif len(active) == 0:
        active = np.array([np.argmax(c)])
    grad = np.zeros(n)
    dx = 2.0 / n
    for m_idx in active:
        k = m_idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[:nv] += (1.0 - h[k:]) / len(active)
            grad[k:] += (-h[:nv]) / len(active)
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += (1.0 - h[:nv]) / len(active)
            grad[:nv] += (-h[kk:]) / len(active)
    return grad * dx

def projected_grad_descent(h, n, deadline, top_m=5):

                h = h_new
                no_improve = 0
                improved = True
                break
        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
    return best_h, best_c

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_smooth(h, n, deadline, max_iters=600):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

            if c_new < best_c:

                h = h_new
        except Exception:

    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]

        any_improved = True
        while any_improved and time.time() < deadline:
            any_improved = False

                    return h, best_c

    best_c = compute_c5(h, n)
    d = 0.001
    iters = 0

        # Rank indices by gradient (most negative = should decrease, most positive = should increase)

                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
        iters += 1
        if not improved and iters > 20:
            break
    return h, best_c
