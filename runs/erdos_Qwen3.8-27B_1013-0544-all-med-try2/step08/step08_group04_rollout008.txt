import numpy as np
import time
import math
from scipy.optimize import minimize, linprog

def phase_deadline(s, off, res, dt):
    return min(max(s + 1.0, s + off), max(s + 1.0, dt - res))

def project_feasible(v, n):
    v = np.asarray(v, dtype=float).copy()
    t = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    d = t - float(v.sum())
    if abs(d) < 1e-12:
        if abs(d) > 1e-15:
            m = (v > 1e-14) & (v < 1.0 - 1e-14)
            if m.any():
                v[m] += d / float(m.sum())
            else:
                v += d / n
        return np.clip(v, 0.0, 1.0)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > t:
            lo = mid
        else:
            hi = mid
    o = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(20):
        d = t - float(o.sum())
        if abs(d) < 1e-14:
            break
        m = (o > 1e-14) & (o < 1.0 - 1e-14)
        if m.any():
            o[m] += d / float(m.sum())
        else:
            o += d / n
        o = np.clip(o, 0.0, 1.0)
    return o

def c5_full(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c, float(c.max())

def c5_val(h, n):
    return c5_full(h, n)[1]

def add_grad(g, h, n, k, dx):
    if k >= 0:
        nv = n - k
        g[:nv] += (1.0 - h[k:]) * dx
        g[k:] += -h[:nv] * dx
    else:
        kk = -k
        nv = n - kk
        g[kk:] += (1.0 - h[:nv]) * dx
        g[:nv] += -h[kk:] * dx

def grad_from_weights(h, n, w):
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for m, wm in enumerate(w):
        if wm > 1e-15:
            add_grad(g, h, n, m - off, dx * float(wm))
    return g

def active_grad_top(h, n, c, cnt=12):
    idx = np.argsort(c)[::-1][:max(1, min(cnt, len(c)))]
    g = np.zeros(n)
    f = (2.0 / n) / float(len(idx))
    off = n - 1
    for midx in idx:
        add_grad(g, h, n, midx - off, f)
    return g

def soft_value(c, a):
    m = float(c.max())
    z = a * (c - m)
    return m + float(np.log(np.sum(np.exp(z - z.max())))) / a

def soft_weights(c, a):
    m = float(c.max())
    z = a * (c - m)
    e = np.exp(z - z.max())
    s = float(e.sum())
    if s <= 0.0:
        return np.ones(len(c)) / len(c)
    return e / s

def smooth_obj(x, n, a, mu):
    c, _ = c5_full(x, n)
    return soft_value(c, a) + mu * (float(x.sum()) - n / 2.0) ** 2

def smooth_grad(x, n, a, mu):
    c, _ = c5_full(x, n)
    g = grad_from_weights(x, n, soft_weights(c, a))
    g += 2.0 * mu * (float(x.sum()) - n / 2.0)
    return g

def lbfgs_refine(h, n, dt, mx=30, nc=5):
    h = project_feasible(h, n)
    bh = h.copy()
    bc = c5_val(h, n)
    bd = [(0.0, 1.0)] * n
    cfg = [(100.0, 1e-4), (1000.0, 1e-3), (10000.0, 1e-2),
           (100000.0, 0.1), (1000000.0, 1.0)]
    for a, mu in cfg[:nc]:
        if time.time() > dt - 3:
            break
        it = min(mx, max(5, int((dt - time.time()) * 3)))
        r = None
        try:
            r = minimize(smooth_obj, h, jac=smooth_grad, args=(n, a, mu),
                         method='L-BFGS-B', bounds=bd,
                         options={'maxiter': it, 'ftol': 1e-15,
                                  'gtol': 1e-12, 'maxfun': min(300, it * 10)})
        except Exception:
            r = None
        if r is not None and r.x is not None:
            h2 = project_feasible(r.x, n)
            c2 = c5_val(h2, n)
            if c2 < bc - 1e-15:
                bc = c2
                bh = h2.copy()
                h = h2
    return bh, bc

def lp_refine(h, n, dt, mx=100):

    c, C = c5_full(h, n)
    M = 2 * n - 1
    off = n - 1
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    beq = np.array([n / 2.0])
    bd = [(0.0, 1.0)] * n + [(None, None)]
    rng = np.random.RandomState(12345)
    no = 0
    for _ in range(mx):
        if time.time() > dt - 2:
            break
        tk = min(M, 600)
        sel = np.argsort(c)[::-1][:tk]
        A = np.empty((tk, n + 1))
        b = np.empty(tk)
        dx = 2.0 / n
        gr = np.zeros(n)
        for r in range(tk):
            gr.fill(0.0)
            midx = sel[r]
            add_grad(gr, h, n, midx - off, dx)
            gh = float(gr.dot(h))
            A[r, :n] = gr
            A[r, n] = -1.0
            b[r] = c[midx] - gh
        co = np.zeros(n + 1)
        co[-1] = 1.0
        if no > 1:
            co[:n] = rng.normal(0.0, 1e-5, n)
        res = None
        try:
            res = linprog(co, A, b, Aeq, beq, bounds=bd, method='HiGHS')
        except Exception:
            break
        if res is None or res.x is None or not np.all(np.isfinite(res.x)):
            no += 1
            if no > 6:
                break
            continue
        u = res.x[:n]
        if float(np.linalg.norm(u - h)) < 1e-10:
            no += 1
            if no > 6:
                break
            continue
        acc = False
        a = 1.0
        for __ in range(30):
            x = project_feasible(h + a * (u - h), n)
            cc = c5_val(x, n)
            if cc < C - 1e-13:
                h = x
                c, C = c5_full(h, n)
                acc = True
                no = 0
                break
            a *= 0.5
        if not acc:
            no += 1
            if no > 8:
                break
    return h, float(c5_val(h, n))

def make_masks(n):
    M = 2 * n - 1
    off = n - 1
    out = []
    for i in range(n):
        lo = min(-i, i - (n - 1))
        hi = max(n - 1 - i, i)
        m = np.zeros(M, dtype=bool)
        m[lo + off:hi + off + 1] = True
        out.append(m)
    return out

def pair_value(ca, A, B, base, d):
    if ca.size == 0:
        return float(base)
    v = float(np.max(ca + A * d + B * d * d))
    return float(v if v > base else base)

def find_pair(h, n, c, C, i, j, mask, grid=17, ref=16):
    i = int(i)
    j = int(j)
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-11:
        return None
    aff = np.flatnonzero(mask)
    if aff.size == 0:
        return None
    base = float(c[~mask].max()) if np.any(~mask) else -1e300
    dx = 2.0 / n
    off = n - 1
    k = aff - off
    A = np.empty(aff.size)
    B = np.empty(aff.size)
    hi_i = h[i]
    hi_j = h[j]
    for t in range(aff.size):
        kk = int(k[t])
        a = 0.0
        bb = 0.0
        if kk == 0:
            a = dx * ((1.0 - 2.0 * hi_i) - (1.0 - 2.0 * hi_j))
            bb = -2.0 * dx
        else:
            ik = i + kk
            if 0 <= ik < n:
                a += dx * (1.0 - h[ik])
            i2 = i - kk
            if 0 <= i2 < n:
                a -= dx * h[i2]
            jk = j + kk
            if 0 <= jk < n:
                a -= dx * (1.0 - h[jk])
            j2 = j - kk
            if 0 <= j2 < n:
                a += dx * h[j2]
            if ik == j:
                bb += dx
            if jk == i:
                bb += dx
        A[t] = a
        B[t] = bb
    ca = c[aff]
    ds = np.linspace(lo, hi, grid)
    vals = ca[None, :] + A[None, :] * ds[:, None] + B[None, :] * ds[:, None] * ds[:, None]
    vc = vals.max(axis=1) if base < -1e299 else np.maximum(vals, base).max(axis=1)
    bi = int(np.argmin(vc))
    bc = float(vc[bi])
    bd = float(ds[bi])
    if bc >= C - 1e-13:
        return None
    span = (hi - lo) / max(1, grid - 1)
    if bi <= 0:
        aa, bb = lo, lo + 3 * span
    elif bi >= grid - 1:
        aa, bb = hi - 3 * span, hi
    else:
        aa, bb = ds[bi - 1], ds[bi + 1]
    if bb > aa:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        x1 = bb - gr * (bb - aa)
        x2 = aa + gr * (bb - aa)
        f1 = pair_value(ca, A, B, base, x1)
        f2 = pair_value(ca, A, B, base, x2)
        for _ in range(ref):
            if f1 <= f2:
                bb = x2
                x2 = x1
                f2 = f1
                x1 = bb - gr * (bb - aa)
                f1 = pair_value(ca, A, B, base, x1)
            else:
                aa = x1
                x1 = x2
                f1 = f2
                x2 = aa + gr * (bb - aa)
                f2 = pair_value(ca, A, B, base, x2)
        cd = x1 if f1 <= f2 else x2
        cc = min(f1, f2)
        if cc < bc:
            bc, bd = cc, cd
    if bc < C - 1e-13:
        return float(np.clip(bd, lo, hi))
    return None

def pair_polish(h, n, dt, ms, out=50, topB=None):
    h = project_feasible(h, n)
    if topB is None:
        topB = int(min(35, max(8, n // 8)))
    rng = np.random.RandomState(99)
    no = 0
    for r in range(out):
        if time.time() > dt:
            break
        c, C = c5_full(h, n)
        g = active_grad_top(h, n, c, 12)
        ordv = np.argsort(g)
        b = min(n, topB + (r % 3) * 5)
        low = ordv[:b]
        high = ordv[-b:]
        imp = False
        for i in high:
            if imp or time.time() > dt:
                break
            for j in low:
                if i == j:
                    continue
                if time.time() > dt:
                    return h, float(c5_val(h, n))
                d = find_pair(h, n, c, C, i, j, ms[i] | ms[j], grid=17, ref=14)
                if d is not None:
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
                    imp = True
                    break
        if imp:
            no = 0
            continue
        for _ in range(220):
            if time.time() > dt:
                return h, float(c5_val(h, n))
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            d = find_pair(h, n, c, C, i, j, ms[i] | ms[j], grid=13, ref=10)
            if d is not None:
                h[i] += d
                h[j] -= d
                h = project_feasible(h, n)
                imp = True
                break
        if imp:
            no = 0
            continue
        no += 1
        if no >= 3:
            break
    return h, float(c5_val(h, n))

def quick_refine(h, n, dt, mx=25):
    h = project_feasible(h, n)
    h, c = lp_refine(h, n, min(dt, time.time() + 45.0), mx)
    if time.time() < dt - 8:
        h2, c2 = lbfgs_refine(h, n, dt, 10, 2)
        if c2 < c:
            h, c = h2, c2
    return h, float(c5_val(h, n))

def initial_guess(n, seed):
    r = np.random.RandomState(seed)
    x = np.linspace(0, 1, n)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * x + r.uniform(0, 6.283))
    h += 0.2 * np.sin(4 * np.pi * x + r.uniform(0, 6.283))
    return project_feasible(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dt = t0 + max(10.0, float(budget_s)) - 5.0
    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel()
        if h0.size < 16:
            raise ValueError
        h = h0.copy()
        n = len(h)
    except Exception:
        n = 300
        h = initial_guess(n, seed)
    h = project_feasible(h, n)
    bh = h.copy()
    bc = float(c5_val(h, n))
    bn = n

    t = phase_deadline(t0, 260.0, 320.0, dt)
    if time.time() < t:
        h1, c1 = lp_refine(bh, bn, t, 160)
        if c1 < bc:
            bh, bc = h1, c1

    t = phase_deadline(time.time(), 120.0, 200.0, dt)
    if time.time() < t:
        h2, c2 = lbfgs_refine(bh, bn, t, 35, 5)
        if c2 < bc:
            bh, bc = h2, c2

    t = phase_deadline(time.time(), 160.0, 90.0, dt)
    if time.time() < t:
        h4, c4 = pair_polish(bh, bn, t, make_masks(bn), 60)
        if c4 < bc:
            bh, bc = h4, c4

    if time.time() < phase_deadline(time.time(), 180.0, 110.0, dt):
        base = bn
        alts = []
        for nt in (base // 2, base // 2 + 20, base - 32, base + 32, base + 64):
            if nt % 2:
                nt += 1
            if 120 <= nt <= 900 and nt != base and nt not in alts:
                alts.append(nt)
        for nt in alts[:3]:
            if time.time() > dt - 60:
                break
            x0 = np.linspace(0, 1, len(bh))
            x1 = np.linspace(0, 1, nt)
            ht = project_feasible(np.interp(x1, x0, bh), nt)
            al = min(150.0, dt - time.time() - 30.0)
            if al < 40:
                break
            ht, ct = lp_refine(ht, nt, time.time() + al * 0.8, 60)
            if ct < bc:
                if time.time() < dt - 30:
                    ht, ct = pair_polish(ht, nt, min(dt - 20, time.time() + 60.0), make_masks(nt), 20)
                if ct < bc:
                    bh, bc, bn = ht, ct, nt

    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    r = np.random.RandomState(seed + 7)
    for sc in scales:
        if time.time() > dt - 30:
            break
        for _ in range(2):
            if time.time() > dt - 20:
                break
            al = min(60.0, dt - time.time() - 20.0)
            if al < 25:
                break
            ht = project_feasible(bh + r.normal(0.0, sc, bn), bn)
            ht, ct = quick_refine(ht, bn, time.time() + al, 20)
            if ct < bc:
                bh, bc = ht, ct

    if time.time() < dt:
        h7, c7 = lp_refine(bh, bn, dt, 20)
        if c7 < bc:
            bh, bc = h7, c7
        if time.time() < dt - 1:
            h7, c7 = pair_polish(bh, bn, dt, make_masks(bn), 20)
            if c7 < bc:
                bh, bc = h7, c7

    bh = project_feasible(bh, bn)
    bc = float(c5_val(bh, bn))
    return bh, bc, int(bn)
