The problem is: min max_k f_k(h) where f_k(h) = ∫ h(x)(1-h(x+k)) dx

1. Start with a point h0
2. Compute the active set A = {k : f_k(h) ≈ max}
3. Solve: min sum_{k in A} f_k(h) s.t. constraints (this is a QP since f_k is quadratic in h)
4. Update the active set
5. Repeat

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = x.sum() - n / 2.0
    val += mu * s * s
    return float(val)

def smax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    cs = alpha * (c - c.max())
    mx = cs.max()
    w = np.exp(cs - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - x[k:])
            g[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - x[:nv])
            g[:nv] += wm * (-x[kk:])
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_opt(h, n, deadline, max_iters=600):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 0.1), (500, 0.5), (2000, 1.0), (10000, 5.0),
               (50000, 20.0), (200000, 100.0), (1000000, 500.0),
               (5000000, 2000.0), (20000000, 10000.0),
               (100000000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(smax_obj, h, jac=smax_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iters * 80,
                                   'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def block_search(h, n, deadline, block_sizes=(3, 5, 10, 20)):
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    for d in steps:
        while time.time() < deadline:
            any_imp = False
            for bs in block_sizes:
                if time.time() > deadline:
                    return h, best_c
                for start in range(0, n - bs + 1, max(1, bs // 2)):
                    if time.time() > deadline:
                        return h, best_c
                    block = h[start:start+bs].copy()
                    for start2 in range(0, n - bs + 1, max(1, bs // 2)):
                        if start2 == start:
                            continue
                        if abs(start2 - start) < bs:
                            continue
                        # Move: increase block1 by d (spread evenly), decrease block2 by d
                        new_b1 = block + d / bs
                        cur_b2 = h[start2:start2+bs]
                        new_b2 = cur_b2 - d / bs
                        if np.any(new_b1 > 1.0) or np.any(new_b2 < 0.0):
                            continue
                        h[start:start+bs] = new_b1
                        h[start2:start2+bs] = new_b2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_imp = True
                        else:
                            h[start:start+bs] = block
                            h[start2:start2+bs] = cur_b2
            if not any_imp:
                break
    return h, best_c

def pair_search(h, n, deadline, radius=150):
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    for d in steps:
        while time.time() < deadline:
            any_imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[j] + d <= 1.0 and h[i] - d >= 0.0:
                        h[j] += d
                        h[i] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_imp = True
                        else:
                            h[j] -= d
                            h[i] += d
            if not any_imp:
                break
    return h, best_c

def sa_search(h, n, deadline, rng, T0=0.003, T_min=1e-7, cool=0.9998):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    T = T0
    while time.time() < deadline:
        T = max(T * cool, T_min)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        step = rng.choice([0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5])
        sign = 1.0 if rng.rand() < 0.5 else -1.0
        h[i] += sign * step
        h[j] -= sign * step
        if h[i] < 0.0 or h[i] > 1.0 or h[j] < 0.0 or h[j] > 1.0:
            h[i] -= sign * step
            h[j] += sign * step
            continue
        c_new = compute_c5(h, n)
        delta = c_new - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i] -= sign * step
            h[j] += sign * step
    return best_h, best_c

def slsqp_polish(h, n, alpha=10000000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def o(x):
        return smax_obj(x, n, alpha, 0.0)
    def ef(x):
        return [x.sum() - target]
    def ej(x):
        return [np.ones(n)]
    cons = [{'type': 'eq', 'fun': ef, 'jac': ej}]
    try:
        res = minimize(o, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)
