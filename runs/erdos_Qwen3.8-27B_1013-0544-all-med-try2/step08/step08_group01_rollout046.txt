    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[:nv] += (1.0 - h[k:])

    """Fast O(n) descent targeting top correlations."""
    best_c = compute_c5(h, n)
    v = np.zeros(n)  # momentum
    stagnant = 0
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        top_idx = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for mi in top_idx:
            k = mi - (n - 1)
            g += grad_shift(h, n, k)
        # Normalize gradient (maintain sum)
        g -= g.mean()
        # Momentum

        h_new = project_feasible(h + v, n)

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)

    g = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_shift(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[:nv] += (1.0 - h[k:])
        g[k:] += -h[:nv]
    else:
        kk = -k
        nv = n - kk
        g[kk:] += (1.0 - h[:nv])
        g[:nv] += -h[kk:]
    return g

def top_shift_descent(h, n, deadline, top_m=5, lr=0.02, momentum=0.9):
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    stagnant = 0
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        top_idx = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for mi in top_idx:
            k = mi - (n - 1)
            g += grad_shift(h, n, k)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            h = h_new
            stagnant = 0
            lr = min(lr * 1.01, 0.3)
        else:
            stagnant += 1
            lr *= 0.85
            v *= 0.3
            if lr < 1e-8 or stagnant > 200:
                break
    return h, best_c

def soft_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def soft_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for mi in range(len(w)):
        if w[mi] < 1e-12:
            continue
        k = mi - (n - 1)
        g += w[mi] * grad_shift(x, n, k)
    return g * dx

def lbfgs_phase(h, n, deadline, alphas=(100, 1000, 10000)):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        def obj(x):
            return soft_max_obj(x, n, alpha)
        def jac(x):
            return soft_max_grad(x, n, alpha)
        try:
            res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def targeted_pairs(h, n, deadline, radius=60):
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            c = compute_c5_vec(h, n)
            k_star = np.argmax(c)
            k = k_star - (n - 1)
            if k >= 0:
                pairs = [(i, i + k) for i in range(n - k)]
            else:
                kk = -k
                pairs = [(i + kk, i) for i in range(n - kk)]
            for i, j in pairs:
                if time.time() > deadline:
                    return h, best_c
                hi, hj = h[i], h[j]
                if hi + d <= 1.0 and hj - d >= 0.0:
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved_any = True
                    else:
                        h[i] -= d
                        h[j] += d
                if hi - d >= 0.0 and hj + d <= 1.0:
                    h[i] -= d
                    h[j] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved_any = True
                    else:
                        h[i] += d
                        h[j] -= d
    return h, best_c

def multi_restart(h, n, rng, deadline, top_m=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_tries = 6
    per_try = max(1.0, (deadline - time.time()) / n_tries)
    for _ in range(n_tries):
        if time.time() > deadline - 5:
            break
        scale = rng.uniform(0.03, 0.12)
        h_try = project_feasible(h + rng.randn(n) * scale, n)
        t_end = min(time.time() + per_try * 0.75, deadline - 2)
        h_r, c_r = top_shift_descent(h_try, n, t_end, top_m=top_m, lr=0.01)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
    return best_h, best_c
