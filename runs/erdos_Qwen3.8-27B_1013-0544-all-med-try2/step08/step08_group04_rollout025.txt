    shmax = sh.max()
    val = cmax + (shmax + np.log(np.sum(np.exp(sh - shmax)))) / alpha

    best_h = h.copy()

        h_new = project_feasible(h - v, n)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single_shift(h, n, k):
    """Gradient of correlation at shift k (positive = right shift)."""
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[0:nv] += dx * (1.0 - h[k:n])
        g[k:n] += dx * (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        g[kk:n] += dx * (1.0 - h[0:nv])
        g[0:nv] += dx * (-h[kk:n])
    return g

def subgrad_max(h, n, tol=1e-6):
    """Subgradient of max of correlation values."""
    c = compute_c5_vec(h, n)
    cmax = c.max()
    active = np.where(c >= cmax - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        grad += grad_single_shift(h, n, k) / len(active)
    return grad

    cmax = c.max()
    sh = alpha * (c - cmax)
    shmax = sh.max()
    val = cmax + (shmax + np.log(np.sum(np.exp(sh - shmax)))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    shmax = sh.max()
