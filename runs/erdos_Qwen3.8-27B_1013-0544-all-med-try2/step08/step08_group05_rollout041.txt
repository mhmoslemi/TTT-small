        for i in zeros[:]:
            for j in ones[:]:
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                delta = np.zeros(2 * n - 1)
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + delta
                    cur_c = nc
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

def sa_binary(h_init, n, deadline, seed=42, T0=0.004, cooling=0.99995):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    idx = np.argsort(h_init)[::-1]
    target = n // 2
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    best_c = np.inf
    best_h = h.copy()
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_c = corr.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        T = T0
        stuck = 0
        phase_dl = min(deadline, time.time() + 70)
        while time.time() < phase_dl and stuck < 60000:
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h[i], h[j] = 1.0, 0.0
            new_c = np.correlate(h, 1.0 - h, mode='full').max() * dx
            dc = new_c - cur_c
            if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
                corr = np.correlate(h, 1.0 - h, mode='full')
                cur_c = corr.max() * dx
                ones.remove(j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(j)
                if cur_c < best_c - 1e-14:
                    best_c = cur_c
                    best_h = h.copy()
                    stuck = 0
                else:
                    stuck += 1
            else:
                h[i], h[j] = 0.0, 1.0
                stuck += 1
            T *= cooling
        hp, cp = steepest_polish(best_h.copy(), n, min(deadline, time.time() + 40), max_sweeps=2000)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
        h = best_h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    # Phase 1: SLSQP warm-up
    p1 = t0 + 0.10 * (budget_s - 15)
    h = best_h.copy()
    for alpha in [500, 5000, 50000, 500000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 1500, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: SA + steepest descent (main exploration)
    p2 = t0 + 0.55 * (budget_s - 15)
    hb, cb = sa_binary(best_h, n, p2, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 3: SLSQP polish from best binary
    p3 = t0 + 0.65 * (budget_s - 15)
    if time.time() < p3:
        hc = project_feasible(best_h + 0.02 * rng.randn(n), n)
        for alpha in [50000, 500000, 5000000]:
            if time.time() > p3:
                break
            hc = slsqp_smooth(hc, n, alpha, 3000, p3)
            c = compute_c5(hc, n)
            if c < best_c:
                best_c = c
                best_h = hc.copy()
    # Phase 4: Additional SA restarts with different seeds
    p4 = t0 + 0.87 * (budget_s - 15)
    for s in [seed + 1, seed + 7, seed + 13, seed + 23]:
        if time.time() > p4 - 30:
            break
        hb, cb = sa_binary(best_h, n, min(p4, time.time() + 70), seed=s)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Phase 5: Try alternate resolutions
    if time.time() < dl - 100:
        for nh in [200, 300]:
            if nh == n or time.time() > dl - 60:
                continue
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, nh)
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            hb, cb = sa_binary(hh, nh, dl - 60, seed=seed + 100)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
                n = nh
    # Phase 6: Final continuous polish
    if time.time() < dl - 10:
        hf = slsqp_smooth(best_h.copy() + 0.01 * rng.randn(n), n, 5000000, 5000, dl - 5)
        cf = compute_c5(hf, n)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
