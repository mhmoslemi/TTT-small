def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

    c = compute_c5_vec(h, n)
    idx = np.argsort(c)[-top_k:]

    grad = np.zeros(n)
    for m in idx:
        k = m - (n - 1)

        if k >= 0:

            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])

            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])

                    if i == j or best_h[i] < step or best_h[j] > 1.0 - step:
                        continue
                    best_h[i] -= step
                    best_h[j] += step
                    c = compute_c5(best_h, best_n)
                    if c < best_c - 1e-17:
                        best_c = c
                    else:
                        best_h[i] += step
                        best_h[j] -= step

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
