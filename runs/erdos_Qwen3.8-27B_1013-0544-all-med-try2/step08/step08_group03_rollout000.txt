    g=np.zeros(n+1); g[n]=1; return g

    h=x[:n]; tau=x[n]

    h=x[:n]

import time
import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize

_ORIENT = {}

def c_lag(h, n, k):
    dx = 2.0 / n
    if k >= 0:
        m = n - k
        if m <= 0:
            return 0.0
        return dx * float(np.dot(h[:m], 1.0 - h[k:]))
    kk = -k
    m = n - kk
    if m <= 0:
        return 0.0
    return dx * float(np.dot(h[kk:], 1.0 - h[:m]))

def grad_lag(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] = 1.0 - h[k:]
            g[k:] -= h[:m]
    else:
        kk = -k
        m = n - kk
        if m > 0:
            g[kk:] = 1.0 - h[:m]
            g[:m] -= h[kk:]
    return dx * g

def get_orientation(n):
    o = _ORIENT.get(n)
    if o is not None:
        return o
    if n <= 1:
        _ORIENT[n] = 1
        return 1
    ht = np.zeros(n)
    ht[0] = 1.0
    raw = sp_correlate(ht, 1.0 - ht, method='fft')
    o = 1 if abs(float(raw[n]) - 1.0) < 1e-5 else -1
    _ORIENT[n] = o
    return o

def compute_c5_vec(h, n):
    raw = sp_correlate(h, 1.0 - h, method='fft')
    if get_orientation(n) == -1:
        raw = raw[::-1]
    return raw * (2.0 / n)

def compute_c5(h, n):
    c = compute_c5_vec(h, n)
    return float(np.max(c)) if c.size else 0.0

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break

            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / float(mask.sum())
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] -= (-diff) / float(mask.sum())
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()

    sh = alpha * (c - m)

def lbfgs_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    val = entval(c, alpha)
    s = x.sum() - 0.5 * n
    val += 50.0 * s * s

def lbfgs_grad(x, n, alpha, top_k=160):

        wi = float(w[i])
        if wi > 1e-14:
            g += wi * grad_lag(x, n, int(i - (n - 1)))
    s = x.sum() - 0.5 * n
    g += 100.0 * s
    return g

def lbfgs_optimize(h, n, deadline, alpha=800, maxiter=80):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if time.time() >= deadline:
        return best_h, best_c
    rem = max(1.0, deadline - time.time())
    maxit = int(min(maxiter, max(20, int(rem / 1.2))))
    try:
        res = minimize(lbfgs_obj, h, jac=lbfgs_grad, args=(n, alpha, 160),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxit, 'ftol': 1e-12})
        hnew = project_feasible(np.clip(res.x, 0.0, 1.0), n)
        cnew = compute_c5(hnew, n)
        if cnew < best_c:
            return hnew, cnew
    except Exception:
        pass
    return best_h, best_c

def full_obj(x, n):
    return float(x[n])

def full_obj_jac(x, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def full_cons(x, n, active):
    h = x[:n]
    tau = float(x[n])
    out = np.empty(len(active))
    for i, k in enumerate(active):
        out[i] = tau - c_lag(h, n, k)
    return out

def full_jac(x, n, active):
    h = x[:n]
    J = np.zeros((n + 1, len(active)))
    for i, k in enumerate(active):
        J[:n, i] = -grad_lag(h, n, k)
        J[n, i] = 1.0
    return J

def full_eq(x, n):
    return float(x[:n].sum() - 0.5 * n)

def full_eq_jac(x, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def slsqp_full_polish(h, n, deadline, rng, maxrounds=4):
    cur = h.copy()
    cur_c = compute_c5(cur, n)
    best_h = cur.copy()
    best_c = cur_c
    stall = 0
    for r in range(maxrounds):
        if time.time() >= deadline:
            break
        c = compute_c5_vec(cur, n)
        cap = 50 if n <= 700 else 30
        active = choose_active_k(c, n, 25, 2e-4, rng, cap)
        x0 = np.empty(n + 1)
        x0[:n] = cur
        x0[n] = cur_c + 1e-7
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': full_cons, 'jac': full_jac, 'args': (n, active)},
            {'type': 'eq', 'fun': full_eq, 'jac': full_eq_jac, 'args': (n,)}
        ]
        rem = max(0.0, deadline - time.time())
        if rem < 45:
            maxit = 18
        elif n > 900:
            maxit = 32
        elif n > 600:
            maxit = 50
        else:
            maxit = 80
        improved = False
        try:
            res = minimize(full_obj, x0, jac=full_obj_jac, args=(n,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxit, 'ftol': 1e-13})
            hnew = project_feasible(np.clip(res.x[:n], 0.0, 1.0), n)
            cnew = compute_c5(hnew, n)
            if cnew < best_c - 1e-12:
                best_c = cnew
                best_h = hnew.copy()
                cur = hnew.copy()
                cur_c = cnew
                improved = True
            elif cnew < cur_c - 1e-9:
                cur = hnew.copy()
                cur_c = cnew
                improved = True
        except Exception:
            pass
        if improved:
            stall = 0
        else:
            stall += 1
        if stall >= 2 and rng is not None and time.time() + 25 < deadline:
            scale = 0.002 * min(stall, 3)
            hp = project_feasible(cur + rng.normal(0.0, scale, cur.size), n)
            cp = compute_c5(hp, n)
            if cp < cur_c + 0.002:
                cur = hp
                cur_c = cp
    return best_h, best_c

def central_from_f(f, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = f
    h[m:] = 1.0 - f[::-1]
    return h

def central_f_from_h(h, n):
    m = n // 2
    f = 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])
    return np.clip(f, 0.0, 1.0)

def central_obj(x, m, n):
    return float(x[m])

def central_obj_jac(x, m, n):
    g = np.zeros(m + 1)
    g[m] = 1.0
    return g

def central_cons(x, m, n, mirror, active):
    f = x[:m]
    tau = float(x[m])
    h = central_from_f(f, n)
    out = np.empty(len(active))
    for i, k in enumerate(active):
        out[i] = tau - c_lag(h, n, k)
    return out

def central_jac(x, m, n, mirror, active):
    f = x[:m]
    h = central_from_f(f, n)
    J = np.zeros((m + 1, len(active)))
    for i, k in enumerate(active):
        g = grad_lag(h, n, k)
        J[:m, i] = -g[:m] + g[mirror]
        J[m, i] = 1.0
    return J

def slsqp_central_polish(f, n, deadline, rng, maxrounds=4):
    m = n // 2
    mirror = n - 1 - np.arange(m)
    f = np.clip(f, 0.0, 1.0)
    cur = f.copy()
    h = central_from_f(cur, n)
    cur_c = compute_c5(h, n)
    best_f = cur.copy()
    best_c = cur_c
    stall = 0
    for r in range(maxrounds):
        if time.time() >= deadline:
            break
        h = central_from_f(cur, n)
        c = compute_c5_vec(h, n)
        cap = 60 if m <= 400 else 40
        active = choose_active_k(c, n, 25, 2e-4, rng, cap)
        x0 = np.empty(m + 1)
        x0[:m] = cur
        x0[m] = cur_c + 1e-7
        bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
        cons = [{'type': 'ineq', 'fun': central_cons, 'jac': central_jac,
                 'args': (m, n, mirror, active)}]
        rem = max(0.0, deadline - time.time())
        if rem < 45:
            maxit = 20
        elif m > 400:
            maxit = 45
        else:
            maxit = 80
        improved = False
        try:
            res = minimize(central_obj, x0, jac=central_obj_jac, args=(m, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxit, 'ftol': 1e-13})
            fnew = np.clip(res.x[:m], 0.0, 1.0)
            hnew = central_from_f(fnew, n)
            cnew = compute_c5(hnew, n)
            if cnew < best_c - 1e-12:
                best_c = cnew
                best_f = fnew.copy()
                cur = fnew.copy()
                cur_c = cnew
                improved = True
            elif cnew < cur_c - 1e-9:
                cur = fnew.copy()
                cur_c = cnew
                improved = True
        except Exception:
            pass
        if improved:
            stall = 0
        else:
            stall += 1
        if stall >= 2 and rng is not None and time.time() + 25 < deadline:
            scale = 0.002 * min(stall, 3)
            fp = np.clip(cur + rng.normal(0.0, scale, cur.size), 0.0, 1.0)
            hp = central_from_f(fp, n)
            cp = compute_c5(hp, n)
            if cp < cur_c + 0.002:
                cur = fp
                cur_c = cp
    return central_from_f(best_f, n), best_c

def active_pgd(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur = h.copy()
    stall = 0
    step = 0.04
    for _ in range(300):
        if time.time() >= deadline:
            break
        c = compute_c5_vec(cur, n)
        active = choose_active_k(c, n, 40, 1e-7, None, 50)
        vals = np.array([c[n - 1 + k] for k in active])
        sh = 40.0 * (vals - vals.max())
        w = np.exp(sh - sh.max())
        w /= w.sum()
        g = np.zeros(n)
        for k, wi in zip(active, w):
            g += wi * grad_lag(cur, n, k)
        g -= g.mean()
        if float(np.linalg.norm(g)) < 1e-13:
            break
        accepted = False
        st = step
        for _b in range(9):
            hnew = project_feasible(cur - st * g, n)
            cnew = compute_c5(hnew, n)
            if cnew <= best_c + 1e-13:
                if cnew < best_c - 1e-15:
                    best_c = cnew
                    best_h = hnew.copy()
                    stall = 0
                else:
                    stall += 1
                cur = hnew.copy()
                accepted = True
                break
            st *= 0.4
        if not accepted or stall > 50:
            break
    return best_h, best_c

def random_initial(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    f0 = rng.uniform(0.7, 2.5)
    ph = rng.uniform(0.0, 2.0 * np.pi)
    amp = rng.uniform(0.6, 1.0)
    h = np.clip(0.5 + 0.5 * amp * np.sin(2.0 * np.pi * f0 * x + ph), 0.0, 1.0)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s)) - 4.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 500
        h0 = random_initial(n0, rng)
    if n0 < 4:
        n0 = 500
        h0 = random_initial(n0, rng)

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_n = n0

    if time.time() < deadline - 320:
        t = min(time.time() + 40, deadline - 300)
        best_h, best_c = lbfgs_optimize(best_h, best_n, t, alpha=800, maxiter=50)

    if time.time() < deadline - 180:
        t = min(time.time() + 180, deadline - 130)
        best_h, best_c = slsqp_full_polish(best_h, best_n, t, rng, 5)

    if best_n % 2 == 0 and time.time() < deadline - 130:
        t = min(time.time() + 110, deadline - 90)
        hs, cs = slsqp_central_polish(central_f_from_h(best_h, best_n), best_n, t, rng, 5)
        if cs < best_c:
            best_h, best_c = hs, cs
            best_h = project_feasible(best_h, best_n)
            best_c = compute_c5(best_h, best_n)

    for ntry in (600, 700, 800, 1000):
        if time.time() > deadline - 150:
            break
        if ntry == best_n:
            continue
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, ntry, endpoint=False)
        htry = project_feasible(np.interp(x_new, x_old, best_h), ntry)
        if time.time() < deadline - 130:
            t = min(time.time() + 38, deadline - 110)
            htry, ctry = lbfgs_optimize(htry, ntry, t, alpha=1500, maxiter=45)
        else:
            ctry = compute_c5(htry, ntry)
        if ctry < best_c + 0.01 and time.time() < deadline - 90:
            t = min(time.time() + (55 if ntry <= 800 else 35), deadline - 60)
            htry, ctry = slsqp_full_polish(htry, ntry, t, rng, 2 if ntry <= 800 else 1)
        if ctry < best_c:
            best_h = project_feasible(htry, ntry)
            best_c = compute_c5(best_h, ntry)
            best_n = ntry

    if best_n % 2 == 0 and time.time() < deadline - 80:
        t = min(time.time() + 70, deadline - 45)
        hs, cs = slsqp_central_polish(central_f_from_h(best_h, best_n), best_n, t, rng, 3)
        if cs < best_c:
            best_h, best_c = hs, cs
            best_h = project_feasible(best_h, best_n)
            best_c = compute_c5(best_h, best_n)

    if time.time() < deadline - 12:
        t = min(time.time() + 35, deadline - 3)
        best_h, best_c = active_pgd(best_h, best_n, t)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
