                continue
            # Stage 2: verify with full correlation
            delta = np.zeros(ns)
            v = (i + all_shifts >= 0) & (i + all_shifts < n)
            delta[v] += 1.0 - h[i + all_shifts[v]]
            v2 = (i - all_shifts >= 0) & (i - all_shifts < n)
            delta[v2] -= h[i - all_shifts[v2]]
            v3 = (j + all_shifts >= 0) & (j + all_shifts < n)
            delta[v3] -= 1.0 - h[j + all_shifts[v3]]
            v4 = (j - all_shifts >= 0) & (j - all_shifts < n)
            delta[v4] += h[j - all_shifts[v4]]
            new_corr = corr + delta
            new_max = new_corr.max()
            if new_max < cur_max - 1e-13:
                h[i], h[j] = 1.0, 0.0
                corr = new_corr
                cur_max = new_max
                ones = np.setdiff1d(ones, j)
                zeros = np.setdiff1d(zeros, i)
                ones = np.append(ones, i)
                zeros = np.append(zeros, j)
                improved = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                break
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
    return best_h, best_max * dx

def anneal_restarts(h, n, deadline, seed=42, n_restarts=8):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = make_binary(h, n).astype(np.float64)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    ns = 2 * n - 1
    all_shifts = np.arange(ns) - (n - 1)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        # Do a few random swaps to perturb
        h = h.copy()
        corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        cur_max = corr.max()
        n_swaps = rng.randint(5, 15)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for _ in range(n_swaps):
            if len(ones) < 2 or len(zeros) < 2:
                break
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h[a], h[b] = 0.0, 1.0
            delta = np.zeros(ns)
            v = (b + all_shifts >= 0) & (b + all_shifts < n)
            delta[v] += 1.0 - h[b + all_shifts[v]]
            v2 = (b - all_shifts >= 0) & (b - all_shifts < n)
            delta[v2] -= h[b - all_shifts[v2]]
            v3 = (a + all_shifts >= 0) & (a + all_shifts < n)
            delta[v3] -= 1.0 - h[a + all_shifts[v3]]
            v4 = (a - all_shifts >= 0) & (a - all_shifts < n)
            delta[v4] += h[a - all_shifts[v4]]
            corr += delta
            cur_max = corr.max()
            ones = np.setdiff1d(ones, a)
            zeros = np.setdiff1d(zeros, b)
            ones = np.append(ones, b)
            zeros = np.append(zeros, a)
        # Local search from perturbed state
        none_improve = 0
        for sweep in range(2000):
            if time.time() > deadline:
                break
            if none_improve >= 3:
                break
            critical = np.where(corr >= cur_max - 2.0)[0]
            crit_shifts = all_shifts[critical]
            imp = False
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            n_try = min(len(zeros) * len(ones), 20000)
            for _ in range(n_try):
                if time.time() > deadline:
                    break
                ii = zeros[rng.randint(len(zeros))]
                jj = ones[rng.randint(len(ones))]
                dc = np.zeros(len(critical))
                v = (ii + crit_shifts >= 0) & (ii + crit_shifts < n)
                dc[v] += 1.0 - h[ii + crit_shifts[v]]
                v2 = (ii - crit_shifts >= 0) & (ii - crit_shifts < n)
                dc[v2] -= h[ii - crit_shifts[v2]]
                v3 = (jj + crit_shifts >= 0) & (jj + crit_shifts < n)
                dc[v3] -= 1.0 - h[jj + crit_shifts[v3]]
                v4 = (jj - crit_shifts >= 0) & (jj - crit_shifts < n)
                dc[v4] += h[jj - crit_shifts[v4]]
                if (corr[critical] + dc).max() >= cur_max - 1e-13:
                    continue
                delta = np.zeros(ns)
                v = (ii + all_shifts >= 0) & (ii + all_shifts < n)
                delta[v] += 1.0 - h[ii + all_shifts[v]]
                v2 = (ii - all_shifts >= 0) & (ii - all_shifts < n)
                delta[v2] -= h[ii - all_shifts[v2]]
                v3 = (jj + all_shifts >= 0) & (jj + all_shifts < n)
                delta[v3] -= 1.0 - h[jj + all_shifts[v3]]
                v4 = (jj - all_shifts >= 0) & (jj - all_shifts < n)
                delta[v4] += h[jj - all_shifts[v4]]
                nc = corr + delta
                nm = nc.max()
                if nm < cur_max - 1e-13:
                    h[ii], h[jj] = 1.0, 0.0
                    corr = nc
                    cur_max = nm
                    imp = True
                    break
            if not imp:
                none_improve += 1
        if cur_max * dx < best_max:
            best_max = cur_max * dx
            best_h = h.copy()
    return best_h, best_max

def slsqp_polish(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    # Phase 1: Binary local search from initial
    p1 = t0 + 0.25 * (budget_s - 10)
    hb, cb = binary_search_fast(best_h, n, p1, seed=seed, max_sweeps=15000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 2: Annealing restarts
    p2 = t0 + 0.55 * (budget_s - 10)
    hb, cb = anneal_restarts(best_h, n, p2, seed=seed + 100, n_restarts=15)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 3: Try different n values
    p3 = t0 + 0.75 * (budget_s - 10)
    for nh in [200, 300, 350, 400]:
        if time.time() > p3 - 20:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh = make_binary(hh, nh)
        hb2, cb2 = binary_search_fast(hh, nh, p3 - 10, seed=seed + nh, max_sweeps=5000)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
            n = nh
    # Phase 4: More annealing on best
    p4 = dl - 40
    hb, cb = anneal_restarts(best_h, n, p4, seed=seed + 500, n_restarts=10)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 5: SLSQP polish
    if time.time() < dl - 15:
        hp = project_feasible(best_h + 0.03 * rng.randn(n), n)
        for alpha in [10000, 100000, 1000000]:
            if time.time() > dl - 12:
                break
            hp = slsqp_polish(hp, n, alpha, 1000, dl - 12)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()
        # Final binary search on polished
        if time.time() < dl - 5:
            hb, cb = binary_search_fast(best_h, n, dl, seed=seed + 999, max_sweeps=5000)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
