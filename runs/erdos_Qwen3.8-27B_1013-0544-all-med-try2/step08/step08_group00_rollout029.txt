```python
import time
import numpy as np
from scipy.optimize import minimize


def proj_feasible(h, n):
    h = np.asarray(h, dtype=float)
    if h.size != n:
        if h.size < n:
            h = np.pad(h, (0, n - h.size), mode="constant")
        else:
            h = h[:n]
    else:
        h = h.copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(52):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-11:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            h[free] += diff / float(free.sum())
        else:
            h += diff / float(n)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def corr_vec(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def corr_val(h, n):
    return float(np.max(corr_vec(h, n)))


def smooth_max_value(c, alpha):
    m = float(c.max())
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    return m + float(np.log(e.sum())) / alpha


def softmax_weights(c, alpha):
    m = float(c.max())
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    return e / e.sum()


def weighted_grad(h, n, w):
    g = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-16:
        g += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[n - 1 - d]
        wm = w[n - 1 + d]
        c = wp + wm
        if c <= 1e-16:
            continue
        nd = n - d
        g[:nd] += wp - c * h[d:]
        g[d:] += wm - c * h[:nd]
    return g


def smooth_obj(x, n, alpha, mu):
    c = corr_vec(x, n)
    val = smooth_max_value(c, alpha)
    s = float(x.sum()) - 0.5 * n
    return float(val + 0.5 * mu * s * s)


def smooth_grad(x, n, alpha, mu):
    c = corr_vec(x, n)
    w = softmax_weights(c, alpha)
    g = weighted_grad(x, n, w)
    s = float(x.sum()) - 0.5 * n
    g += mu * s
    return g


def lbfgs_smooth(h, n, deadline, pairs):
    h = proj_feasible(h, n)
    best_c = corr_val(h, n)
    if time.time() >= deadline or not pairs:
        return h, best_c
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in pairs:
        if time.time() >= deadline - 2:
            break
        rem = deadline - time.time()
        if rem < 15:
            break
        maxiter = 250 if rem > 160 else (120 if rem > 70 else 60)
        try:
            res = minimize(
                smooth_obj, best_h, jac=smooth_grad,
                args=(n, alpha, mu), method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": maxiter, "ftol": 1e-15,
                         "gtol": 1e-12, "maxfun": maxiter * 10,
                         "maxls": 50}
            )
            if res.x.size == n and np.all(np.isfinite(res.x)):
                h_new = proj_feasible(res.x, n)
                c_new = corr_val(h_new, n)
                if c_new < best_c - 1e-15:
                    best_h = h_new
                    best_c = c_new
        except Exception:
            pass
    return best_h, best_c


def sigmoid_vec(s):
    out = np.empty_like(s)
    pos = s >= 0.0
    out[pos] = 1.0 / (1.0 + np.exp(-s[pos]))
    e = np.exp(s[~pos])
    out[~pos] = e / (1.0 + e)
    return out


def hinge_obj(x, n, tau, target, mu, guard_alpha, guard_scale):
    c = corr_vec(x, n)
    s = (c - target) / tau
    sp = np.empty_like(s)
    pos = s > 0.0
    sp[pos] = s[pos] + np.log1p(np.exp(-s[pos]))
    sp[~pos] = np.log1p(np.exp(s[~pos]))
    val = tau * float(sp.sum())
    if guard_scale > 0.0 and guard_alpha > 0.0:
        val += guard_scale * smooth_max_value(c, guard_alpha)
    ss = float(x.sum()) - 0.5 * n
    return float(val + 0.5 * mu * ss * ss)


def hinge_grad(x, n, tau, target, mu, guard_alpha, guard_scale):
    c = corr_vec(x, n)
    s = (c - target) / tau
    g = weighted_grad(x, n, sigmoid_vec(s))
    if guard_scale > 0.0 and guard_alpha > 0.0:
        g += guard_scale * weighted_grad(x, n, softmax_weights(c, guard_alpha))
    ss = float(x.sum()) - 0.5 * n
    g += mu * ss
    return g


def hinge_optimize(h, n, tau, target, deadline, mu, guard_alpha, guard_scale, maxiter):
    h = proj_feasible(h, n)
    best_c = corr_val(h, n)
    if time.time() >= deadline:
        return h, best_c
    rem = deadline - time.time()
    if rem < 15:
        return h, best_c
    maxiter = min(maxiter, 300 if rem > 130 else (150 if rem > 60 else 80))
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            hinge_obj, h, jac=hinge_grad,
            args=(n, tau, target, mu, guard_alpha, guard_scale),
            method="L-BFGS-B", bounds=bounds,
            options={"maxiter": maxiter, "ftol": 1e-15,
                     "gtol": 1e-12, "maxfun": maxiter * 10, "maxls": 50}
        )
        if res.x.size == n and np.all(np.isfinite(res.x)):
            h_new = proj_feasible(res.x, n)
            c_new = corr_val(h_new, n)
            if c_new < best_c - 1e-15:
                return h_new, c_new
    except Exception:
        pass
    return h, best_c


def single_grad(h, k, n):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        d = k
        nd = n - d
        g[:nd] += 1.0 - h[d:]
        g[d:] += -h[:nd]
    else:
        d = -k
        nd = n - d
        g[d:] += 1.0 - h[:nd]
        g[:nd] += -h[d:]
    return g


def slsqp_t_obj(x, n):
    return float(x[n])


def slsqp_t_jac(x, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def slsqp_ineq(x, n, indices):
    h = x[:n]
    c = corr_vec(h, n)
    return x[n] - c[indices]


def slsqp_ineq_jac(x, n, indices):
    h = x[:n]
    p = len(indices)
    rows = np.empty((p, n + 1))
    rows[:, n] = 1.0
    for a, m in enumerate(indices):
        k = n - 1 - m
        rows[a, :n] = -single_grad(h, k, n)
    return rows


def slsqp_sum(x, n):
    return float(x[:n].sum() - 0.5 * n)


def slsqp_sum_jac(x, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g


def active_polish(h, n, deadline, p_list):
    h = proj_feasible(h, n)
    best_c = corr_val(h, n)
    if time.time() >= deadline:
        return h, best_c
    for p in p_list:
        if time.time() >= deadline - 30:
            break
        p = int(min(p, 2 * n - 1))
        if p <= 0:
            continue
        c = corr_vec(h, n)
        if not np.all(np.isfinite(c)):
            break
        maxc = float(c.max())
        idx_top = np.argsort(c)[-p:]
        near = np.where(c >= maxc - 1e-5)[0]
        idx = np.unique(np.concatenate((idx_top, near)))
        cap = min(25, 3 * p)
        if idx.size > cap:
            idx = idx[np.argsort(c[idx])[-cap:]]
        idx = np.asarray(idx, dtype=int)
        rem = deadline - time.time()
        if p <= 4:
            maxit = 200
        elif p <= 8:
            maxit = 150
        else:
            maxit = 100
        if rem < 60:
            maxit = min(maxit, 60)
        elif rem < 120:
            maxit = min(maxit, 120)
        x0 = np.empty(n + 1)
        x0[:n] = h
        x0[n] = min(1.0 - 1e-12, maxc + 1e-7)
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        cons = [
            {"type": "ineq", "fun": slsqp_ineq, "jac": slsqp_ineq_jac, "args": (n, idx)},
            {"type": "eq", "fun": slsqp_sum, "jac": slsqp_sum_jac, "args": (n,)}
        ]
        try:
            res = minimize(
                slsqp_t_obj, x0, jac=slsqp_t_jac, args=(n,),
                method="SLSQP", bounds=bounds, constraints=cons,
                options={"maxiter": maxit, "ftol": 1e-16, "iprint": 0}
            )
            if res.x.size == n + 1 and np.all(np.isfinite(res.x)):
                h_new = proj_feasible(res.x[:n], n)
                c_new = corr_val(h_new, n)
                if c_new < best_c - 1e-15:
                    h = h_new
                    best_c = c_new
        except Exception:
            pass
    return h, best_c


def pair_polish(h, n, deadline, alpha, k, levels):
    h = proj_feasible(h, n)
    best_c = corr_val(h, n)
    steps = (0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001)
    levels = min(levels, len(steps))
    k = min(k, n)
    if time.time() >= deadline or levels <= 0 or k <= 0:
        return h, best_c
    for lev in range(levels):
        st = steps[lev]
        for _ in range(30):
            if time.time() >= deadline - 1:
                return h, best_c
            c = corr_vec(h, n)
            w = softmax_weights(c, alpha)
            g = weighted_grad(h, n, w)
            order = np.argsort(g)
            low = order[:k]
            high = order[-k:]
            improved = False
            for i in low:
                for j in high:
                    if i == j:
                        continue
                    d = st
                    if h[i] + d > 1.0:
                        d = 1.0 - h[i]
                    if h[j] - d < 0.0:
                        d = h[j]
                    if d <= 1e-11:
                        continue
                    oi, oj = h[i], h[j]
                    h[i] = oi + d
                    h[j] = oj - d
                    cv = corr_val(h, n)
                    if cv < best_c - 1e-15:
                        best_c = cv
                        improved = True
                    else:
                        h[i] = oi
                        h[j] = oj
                    if time.time() >= deadline - 1:
                        return h, best_c
            if not improved:
                break
    return h, best_c


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return proj_feasible(np.interp(x_new, x_old, h_old), n_new)


def fallback_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.5 * np.cos(np.pi * (x - 1.0))
    for q in range(1, 9):
        h += (0.02 / q) * np.sin(np.pi * q * x + rng.uniform(0.0, 2.0 * np.pi))
    return proj_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 3.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 < 60:
            raise ValueError("bad init")
    except Exception:
        n0 = 650
        h0 = fallback_start(n0, rng)

    best_h = proj_feasible(h0, n0)
    best_c = corr_val(best_h, n0)
    best_n = n0

    if time.time() < deadline - 60:
        parent_pairs = ((200, 0.1), (1000, 1.0), (5000, 10.0),
                        (20000, 100.0), (100000, 1000.0))
        lb_end = min(t0 + 180.0, deadline - 720.0)
        if lb_end > time.time() + 25.0:
            best_h, best_c = lbfgs_smooth(best_h, best_n, lb_end, parent_pairs)

    if time.time() < deadline - 320:
        act_end = min(time.time() + 340.0, deadline - 340.0)
        best_h, best_c = active_polish(best_h, best_n, act_end, (4, 8, 12))

    targets = (min(0.3808, best_c - 1e-6), min(0.38075, best_c - 1.5e-5))
    for tgt in targets:
        rem = deadline - time.time()
        if rem < 95.0:
            break
        limit = time.time() + min(80.0, rem - 95.0)
        if limit > time.time() + 20.0:
            h_new, c_new = hinge_optimize(
                best_h, best_n, 1e-4, tgt, limit,
                10000.0, 3000.0, 1e-6, 320
            )
            if c_new < best_c - 1e-15:
                best_h, best_c = h_new, c_new

    if time.time() < deadline - 130:
        h_new, c_new = active_polish(
            best_h, best_n, min(time.time() + 120.0, deadline - 90.0), (4, 8)
        )
        if c_new < best_c - 1e-15:
            best_h, best_c = h_new, c_new

    if time.time() < deadline - 190:
        for ratio in (0.8, 0.65):
            n2 = int(best_n * ratio)
            if n2 < 130 or n2 == best_n:
                continue
            rem = deadline - time.time()
            if rem < 170.0:
                break
            alloc = min(145.0, rem - 135.0)
            if alloc < 55.0:
                break
            h2 = interpolate_h(best_h, best_n, n2)
            c2 = corr_val(h2, n2)
            lb_end = time.time() + max(20.0, 0.45 * alloc)
            h2, c2 = lbfgs_smooth(
                h2, n2, min(lb_end, deadline - 90.0),
                ((1000, 1.0), (5000, 10.0), (20000, 100.0))
            )
            h2, c2 = active_polish(
                h2, n2, min(time.time() + alloc - 10.0, deadline - 70.0), (4, 8)
            )
            c2 = corr_val(proj_feasible(h2, n2), n2)
            if c2 < best_c - 1e-15:
                best_h, best_c, best_n = h2, c2, n2

    rem = deadline - time.time()
    if rem > 30.0:
        levels = 8 if rem > 120.0 else 5
        h_new, c_new = pair_polish(
            best_h, best_n, deadline - 2.0, 5000,
            min(25, best_n // 2), levels
        )
        if c_new < best_c - 1e-15:
            best_h, best_c = h_new, c_new

    if time.time() < deadline - 10.0:
        h_new, c_new = active_polish(best_h, best_n, deadline - 2.0, (4,))
        if c_new < best_c - 1e-15:
            best_h, best_c = h_new, c_new

    best_h = proj_feasible(best_h, best_n)
    best_c = corr_val(best_h, best_n)
    return best_h, float(best_c), int(best_n)
```
