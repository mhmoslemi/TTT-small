```python
def compute_gradient(h, n, w):
    """w is array of length 2n-1 (softmax weights for each lag)"""
    dx = 2.0 / n
    lags = np.arange(-(n-1), n)  # length 2n-1
    
    # pos_term[j] = sum_k w[k] * h[j + lags[k]] * I(0 <= j+lags[k] < n)
    # neg_term[j] = sum_k w[k] * h[j - lags[k]] * I(0 <= j-lags[k] < n)
    
    j_idx = np.arange(n)[:, None]  # (n, 1)
    lag_idx = lags[None, :]  # (1, 2n-1)
    
    pos_idx = j_idx + lag_idx  # (n, 2n-1)
    neg_idx = j_idx - lag_idx  # (n, 2n-1)
    
    pos_valid = (pos_idx >= 0) & (pos_idx < n)
    neg_valid = (neg_idx >= 0) & (neg_idx < n)
    
    # Safe index
    pos_safe = np.clip(pos_idx, 0, n-1)
    neg_safe = np.clip(neg_idx, 0, n-1)
    
    pos_term = (h[pos_safe] * pos_valid). @ w  # (n,)
    neg_term = (h[neg_safe] * neg_valid). @ w  # (n,)
    
    grad = dx * (1.0 - pos_term - neg_term)
    return grad
```

```python
    """Project y onto {x: 0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Find lambda such that sum(clip(y - lambda, 0, 1)) = target
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(y - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)
```

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_sum_box(y, n):
    target = n / 2.0
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def precompute(n):
    lags = np.arange(-(n - 1), n)
    j = np.arange(n)[:, None]
    L = lags[None, :]
    pi, ni = j + L, j - L
    pv, nv = (pi >= 0) & (pi < n), (ni >= 0) & (ni < n)
    return np.clip(pi, 0, n - 1), pv, np.clip(ni, 0, n - 1), nv

def smooth_and_grad(h, n, alpha, pre):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = g.max()
    s = alpha * (g - mx)
    sm = s.max()
    w = np.exp(s - sm)
    ws = w.sum()
    w /= ws
    val = mx + (sm + np.log(ws)) / alpha
    ps, pv, ns, nv = pre
    A = (pv * 1.0).dot(w)
    B = (h[ps] * pv).dot(w)
    C = (h[ns] * nv).dot(w)
    grad = dx * (A - B - C)
    return val, w, grad

def pg_loss(h, n, alpha, pre):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = g.max()
    s = alpha * (g - mx)
    sm = s.max()
    ws = np.exp(s - sm).sum()
    return mx + (sm + np.log(ws)) / alpha

def pg_descent(h, n, deadline, a0=50, a1=200000):
    pre = precompute(n)
    h = project_sum_box(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = a0
    step = 0.01
    fail = 0
    
    while time.time() < deadline:
        val, w, grad = smooth_and_grad(h, n, alpha, pre)
        h_new = project_sum_box(h - step * grad, n)
        val_new = pg_loss(h_new, n, alpha, pre)
        
        if val_new < val - 1e-16:
            h = h_new
            fail = 0
        else:
            step *= 0.7
            fail += 1
            if fail > 50:
                alpha *= 1.5
                step = 0.05
                fail = 0
                if alpha > a1:
                    break
        
        if fail % 30 == 0:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    c = compute_c5(h, n)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c
