def proj(h, n):
    h = np.asarray(h, dtype=float)
    t = 0.5 * n
    lo = -2.0 - max(0.0, -float(h.min()))
    hi = 2.0 + max(0.0, float(h.max()) - 1.0)
    for _ in range(50):
        m = 0.5 * (lo + hi)
        if float(np.clip(h - m, 0.0, 1.0).sum()) > t:
            lo = m
        else:
            hi = m
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def cvec(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def sv(c, a):
    M = float(np.max(c))
    s = a * (c - M)
    m = float(np.max(s))
    return float(M + (m + np.log(np.sum(np.exp(s - m)))) / a)


def sw(c, a):
    M = float(np.max(c))
    s = a * (c - M)
    m = float(np.max(s))
    e = np.exp(s - m)
    z = float(np.sum(e))
    return float(M + (m + np.log(z)) / a), e / z


def grad(h, n, w):
    dx = 2.0 / n
    g = np.zeros(n)
    m = n - 1
    v = w[m] * dx
    if v > 1e-14:
        g += v * (1.0 - 2.0 * h)
    for r in range(1, n):
        a = w[m - r] * dx
        if a > 1e-14:
            g[:n - r] += a * (1.0 - h[r:])
            g[r:] -= a * h[:n - r]
        b = w[m + r] * dx
        if b > 1e-14:
            g[r:] += b * (1.0 - h[:n - r])
            g[:n - r] -= b * h[r:]
    return g


def pnt(d, h, p, s, n):
    dx = 2.0 / n
    m = n - 1
    e = n - p - 1
    d[m] += s * dx * (1.0 - 2.0 * h[p])
    if e > 0:
        d[m - e:m] += s * dx * (1.0 - h[p + 1:p + e + 1])[::-1]
    if p > 0:
        d[m - p:m] += s * dx * (-h[:p])
        d[m + 1:m + p + 1] += s * dx * (1.0 - h[p - 1::-1])
    if e > 0:
        d[m + 1:m + e + 1] += s * dx * (-h[p + 1:p + e + 1])


def pdel(h, n, i, j):
    d = np.zeros(2 * n - 1)
    pnt(d, h, i, 1.0, n)
    pnt(d, h, j, -1.0, n)
    return d


def move(h, n, c, C, i, j):
    tmax = min(1.0 - float(h[i]), float(h[j]))
    if tmax <= 1e-10:
        return C
    dx = 2.0 / n
    d = pdel(h, n, i, j)
    r = abs(int(i) - int(j))
    m = n - 1
    q1 = m - r
    q2 = m + r
    ts = []
    for t in ABS:
        if t < tmax:
            ts.append(float(t))
    for f in FRC:
        t = f * tmax
        if t > 1e-12 and t <= tmax:
            ts.append(float(t))
    ts.sort()
    best = float(C)
    bt = 0.0
    last = -1.0
    for t in ts:
        if t - last <= 1e-10:
            continue
        last = t
        v = c + t * d
        if r:
            a = dx * t * t
            v[q1] += a
            v[q2] += a
        val = float(np.max(v))
        if val < best - 1e-13:
            best = val
            bt = t
    if bt > 0.0:
        h[i] += bt
        h[j] -= bt
        if h[i] > 1.0:
            h[i] = 1.0
        if h[j] < 0.0:
            h[j] = 0.0
        c += bt * d
        if r:
            a = dx * bt * bt
            c[q1] += a
            c[q2] += a
        return float(np.max(c))
    return C


def addc(lst, seen, i, j, n):
    i = int(i)
    j = int(j)
    if i == j or i < 0 or i >= n or j < 0 or j >= n:
        return
    e = i * n + j
    if e not in seen:
        seen.add(e)
        lst.append((i, j))

def pg(h, n, dl, a, it):
    h = proj(h, n)
    c = cvec(h, n)
    C = float(c.max())
    bh = h.copy()
    bc = C
    for _ in range(int(it)):
        if time.time() > dl:
            break
        S, w = sw(c, a)
        g = grad(h, n, w)
        g -= float(np.mean(g))
        nr = float(np.linalg.norm(g))
        if nr < 1e-13:
            break
        b = 0.04 / max(nr, 1e-12)
        ok = False
        for f in (1.0, 0.4, 0.15, 0.05, 0.015):
            y = proj(h - b * f * g, n)
            yv = cvec(y, n)
            Y = float(yv.max())
            SY = sv(yv, a)
            if Y < C - 1e-13 or (Y <= C + 1e-11 and SY < S - 1e-14):
                h = y
                c = yv
                C = Y
                ok = True
                break
        if not ok:
            b *= 0.5
            if b < 1e-10:
                break
        if C < bc:
            bc = C
            bh = h.copy()
    return bh, bc

def gold(n):
    h = np.zeros(n)
    m = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    if m > 0:
        h[np.floor(np.arange(m) * phi).astype(int) % n] = 1.0
    return proj(h, n)

def four(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0.0, 2.0, n, False)
    for f in range(1, 8):
        h += rng.uniform(0.05, 0.25) * np.cos(2.0 * np.pi * f * x / 2.0 + rng.uniform(0.0, 6.28))
    return proj(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = max(t0 + 1.0, t0 + float(budget_s) - 5.0)
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 500
        h0 = gold(500)
    if n0 < 2:
        n0 = 500
        h0 = gold(n0)
    if n0 > 1000:
        xo = np.linspace(0.0, 2.0, n0, False)
        xn = np.linspace(0.0, 2.0, 1000, False)
        h0 = np.interp(xn, xo, h0)
        n0 = 1000
    h0 = proj(h0, n0)
    C0 = float(cvec(h0, n0).max())
    bh = h0.copy()
    bc = C0
    bn = n0

    cc = 240 if bn <= 600 else 170 if bn <= 900 else 110
    pgp = 80.0 if bn <= 600 else 50.0 if bn <= 900 else 35.0
    t1 = min(dl - 180.0, t0 + min(520.0, (dl - t0) * 0.55))
    if t1 > time.time() + 20.0:
        h1, c1 = polish(h0, bn, t1, rng, cc, pgp)
        if c1 < bc:
            bh, bc = h1, c1

    for eps in (0.004, 0.012, 0.03):
        if time.time() > dl - 220.0:
            break
        st = proj(bh + rng.normal(0.0, eps, bn), bn)
        al = 50.0 if bn <= 600 else 35.0
        t2 = min(time.time() + al, dl - 160.0)
        if t2 <= time.time() + 15.0:
            break
        h2, c2 = polish(st, bn, t2, rng, 160, 20.0)
        if c2 < bc:
            bh, bc = h2, c2

    if time.time() < dl - 200.0:
        st = proj((bh > 0.5).astype(float), bn)
        t3 = min(time.time() + 50.0, dl - 160.0)
        if t3 > time.time() + 15.0:
            h3, c3 = polish(st, bn, t3, rng, 160, 20.0)
            if c3 < bc:
                bh, bc = h3, c3

    if time.time() < dl - 180.0:
        st = four(bn, rng)
        t3 = min(time.time() + 50.0, dl - 160.0)
        if t3 > time.time() + 15.0:
            h3, c3 = polish(st, bn, t3, rng, 160, 20.0)
            if c3 < bc:
                bh, bc = h3, c3

    nlist = []
    if bn < 600:
        nlist.append(600)
    if bn < 1000:
        nlist.append(1000)
    for nt in nlist:
        if time.time() > dl - 160.0:
            break
        xo = np.linspace(0.0, 2.0, bn, False)
        xn = np.linspace(0.0, 2.0, nt, False)
        ht = proj(np.interp(xn, xo, bh), nt)
        cc2 = 140 if nt <= 800 else 95
        pt2 = 25.0 if nt <= 800 else 20.0
        al = 160.0 if nt <= 800 else 240.0
        t4 = min(time.time() + al, dl - 40.0)
        if t4 <= time.time() + 20.0:
            break
        h4, c4 = polish(ht, nt, t4, rng, cc2, pt2)
        if c4 < bc:
            bh, bc, bn = h4, c4, nt

    bh = proj(bh, bn)
    bf = float(cvec(bh, bn).max())
    return bh, bf, int(bn)
