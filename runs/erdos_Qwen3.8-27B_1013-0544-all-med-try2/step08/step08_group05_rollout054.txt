    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta(h, n, i, j):
    """Delta in np.correlate(h,1-h,'full') when h[i]:0->1, h[j]:1->0"""
    m_range = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    if msk.any():
        delta[msk] += (1.0 - h[im[msk]])
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    if msk.any():
        delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    if msk.any():
        delta[msk] -= (1.0 - h[jm[msk]])
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    if msk.any():
        delta[msk] += h[jmn[msk]]
    delta[n-1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2*n-1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)

def local_search(h, n, deadline, max_sweeps=2000, rng=None):
    """First-improvement local search on binary h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try all zero positions
        for i in zeros.tolist():
            if time.time() > deadline:
                break
            best_j = -1
            best_val = cur_c
            for j in ones.tolist():
                d = swap_delta(h, n, i, j)
                new_corr = corr + d
                nc = new_corr.max() * dx
                if nc < best_val - 1e-15:
                    best_val = nc
                    best_j = j
                    best_new_corr = new_corr
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr = best_new_corr
                cur_c = best_val
                improved = True
                # Update ones and zeros
                ones = np.setdiff1d(ones, [best_j])
                ones = np.append(ones, i)
                zeros = np.setdiff1d(zeros, [i])
                zeros = np.append(zeros, best_j)
        if not improved:
            break
    return h, cur_c

def simulated_annealing(h, n, deadline, rng, t_start=0.1, t_end=0.001, n_iter_per_temp=50, n_temps=50):
    """Simulated annealing on binary h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    
    for t_idx in range(n_temps):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (t_idx / max(n_temps-1, 1))
        for _ in range(n_iter_per_temp):
            if time.time() > deadline:
                break
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = rng.choice(zeros_list)
            j = rng.choice(ones_list)
            d = swap_delta(h, n, i, j)
            new_corr = corr + d
            nc = new_corr.max() * dx
            delta_c = nc - cur_c
            if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-10)):
                h[i], h[j] = 1.0, 0.0
                corr = new_corr
                cur_c = nc
                ones_list.remove(j)
                ones_list.append(i)
                zeros_list.remove(i)
                zeros_list.append(j)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
    return best_h, best_c

def perturb(h, n, k, rng):
    """Randomly swap k pairs."""
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(k):
        i = rng.choice(zeros)
        j = rng.choice(ones)
        h[i], h[j] = 1.0, 0.0
        ones = np.setdiff1d(ones, [j])
        ones = np.append(ones, i)
        zeros = np.setdiff1d(zeros, [i])
        zeros = np.append(zeros, j)
    return h

    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
