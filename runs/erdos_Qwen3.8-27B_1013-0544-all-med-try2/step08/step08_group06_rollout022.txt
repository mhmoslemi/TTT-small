from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_optimize(h, n, alpha, maxiter=5000, penalty=1e6):
    """L-BFGS-B on log-sum-exp with penalty for equality constraint."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        m = corr.max()
        shifted = alpha * (corr - m)
        sm = shifted.max()
        lse = sm + np.log(np.exp(shifted - sm).sum())
        f = m + lse / alpha
        pen = penalty * (x.sum() - target) ** 2
        return f + pen
    
    def grad(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        m = corr.max()
        shifted = alpha * (corr - m)
        sm = shifted.max()
        w = np.exp(shifted - sm)
        w /= w.sum()
        # Gradient of log-sum-exp part
        g = np.zeros(n)
        # Only need top few weighted subgradients
        top_k = np.argsort(w)[::-1][:min(10, len(w))]
        for k in top_k:
            if w[k] < 1e-10:
                break
            mk = k - (n - 1)
            j = np.arange(n)
            jp = j + mk
            jm = j - mk
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)
            sg = np.zeros(n)
            sg[vp] += (1.0 - x[jp[vp]])
            sg[vm] -= x[jm[vm]]
            g += w[k] * sg * dx
        # Gradient of penalty
        g += 2 * penalty * (x.sum() - target)
        return g
    
    res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                   bounds=[(0, 1)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
    return project_feasible(res.x, n)

def subgrad_step(h, n, step_size):
    """One projected subgradient step on the true max."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    k_star = int(np.argmax(corr))
    mk = k_star - (n - 1)
    j = np.arange(n)
    jp = j + mk
    jm = j - mk
    vp = (jp >= 0) & (jp < n)
    vm = (jm >= 0) & (jm < n)
    sg = np.zeros(n)
    sg[vp] += (1.0 - h[jp[vp]])
    sg[vm] -= h[jm[vm]]
    # Project h - step_size * sg
    h_new = h - step_size * sg * dx
    return project_feasible(h_new, n)

def nesterov_subgrad(h, n, iters, base_step, deadline):

    dx = 2.0 / n
    target = n / 2.0

    h_prev = h.copy()
    best_h = h.copy()
    corr = np.correlate(h_curr, 1 - h_curr, mode='full')
    best_c = float(corr.max() * dx)

            break
        # Nesterov: evaluate at y = h_curr + (t-1)/(t+2) * (h_curr - h_prev)

        k_star = int(np.argmax(corr_y))
        mk = k_star - (n - 1)
        j = np.arange(n)
        jp = j + mk
        jm = j - mk
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        sg = np.zeros(n)
        sg[vp] += (1.0 - y[jp[vp]])
        sg[vm] -= y[jm[vm]]
        
        # Step

        h_new = y - step * sg * dx
        h_new = project_feasible(h_new, n)

            best_h = h_new.copy()

        h_curr = h_new
    
    return best_h, best_c

def binary_greedy(h_init, n, deadline, max_sweeps=200):
    """Greedy binary swap optimization."""
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx

        improved = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            for j in ones[:]:
                # Compute delta

                jn = np.clip(im, 0, n-1)
                msk = (im >= 0) & (im < n)
                imn = i - m_range
                mskn = (imn >= 0) & (imn < n)
                jm = j + m_range
                mskm = (jm >= 0) & (jm < n)
                jmn = j - m_range
                mskm2 = (jmn >= 0) & (jmn < n)
                
                delta = np.zeros(2*n - 1)
                delta[msk] += (1.0 - h[im[msk]])
                delta[mskn] -= h[imn[mskn]]
                delta[mskm] -= (1.0 - h[jm[mskm]])
                delta[mskm2] += h[jmn[mskm2]]
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)

                    delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_objective(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    lse = sm + np.log(np.exp(shifted - sm).sum())
    return m + lse / alpha + penalty * (x.sum() - n / 2.0) ** 2

def lse_gradient(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()
    g = np.zeros(n)
    top_idx = np.argsort(w)[::-1]
    count = 0
    for k in top_idx:
        if w[k] < 1e-12:
            break
        mk = k - (n - 1)
        j = np.arange(n)
        jp = j + mk
        jm = j - mk
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        sg = np.zeros(n)
        sg[vp] += (1.0 - x[jp[vp]])
        sg[vm] -= x[jm[vm]]
        g += w[k] * sg * dx
        count += 1
        if count >= 20:
            break
    g += 2.0 * penalty * (x.sum() - n / 2.0)
    return g

def lse_lbfgs(h, n, alpha, penalty, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        return lse_objective(x, n, alpha, penalty)
    def grad(x):
        return lse_gradient(x, n, alpha, penalty)
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-16,
                                'maxfun': maxiter * 3})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_refine(h, n, iters, base_lr, deadline):
    dx = 2.0 / n
    h_c = h.copy()
    h_p = h.copy()
    best_h = h.copy()
    corr = np.correlate(h_c, 1 - h_c, mode='full')
    best_c = float(corr.max() * dx)
    for t in range(iters):
        if time.time() > deadline:
            break
        if t > 0:
            beta = (t - 1) / (t + 2.0)
            y = project_feasible(h_c + beta * (h_c - h_p), n)
        else:
            y = h_c.copy()
        corr_y = np.correlate(y, 1 - y, mode='full')
        ks = int(np.argmax(corr_y))
        mk = ks - (n - 1)
        j = np.arange(n)
        jp = j + mk
        jm = j - mk
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        sg = np.zeros(n)
        sg[vp] += (1.0 - y[jp[vp]])
        sg[vm] -= y[jm[vm]]
        lr = base_lr / (1.0 + 0.005 * t)
        h_new = project_feasible(y - lr * sg * dx, n)
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        cn = float(corr_new.max() * dx)
        if cn < best_c:
            best_c = cn
            best_h = h_new.copy()
        h_p = h_c
        h_c = h_new
    return best_h, best_c

def binary_greedy(h_init, n, deadline, max_sweeps=150):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            for j in ones[:]:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                imn = i - m_range
                mskn = (imn >= 0) & (imn < n)
                jm = j + m_range
                mskm = (jm >= 0) & (jm < n)
                jmn = j - m_range
                mskm2 = (jmn >= 0) & (jmn < n)
                delta[msk] += (1.0 - h[im[msk]])
                delta[mskn] -= h[imn[mskn]]
                delta[mskm] -= (1.0 - h[jm[mskm]])
                delta[mskm2] += h[jmn[mskm2]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

    t0 = time.time()
    total = budget_s - 6
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
