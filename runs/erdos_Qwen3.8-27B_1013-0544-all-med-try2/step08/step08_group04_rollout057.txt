    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_fun, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < compute_c5(h, n):
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def coord_search(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for sgn in [1.0, -1.0]:
                    h_new = h.copy()
                    h_new[i] += sgn * d
                    h_new = project_feasible(h_new, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-16:
                        h = h_new
                        best_c = c_new
                        improved = True
                        break
    return h, best_c

def pair_search(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    if radius is None:
        radius = max(10, n // 8)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def interp_to_n(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B on initial resolution
    t1 = min(t0 + 120, deadline - 500)
    best_h, best_c = lbfgs_opt(best_h, n_init, t1,
                               alphas=(100, 500, 2000, 10000, 50000, 200000))

    # Phase 2: SLSQP refinement
    if time.time() < deadline - 400:
        for a in [50000, 200000, 1000000]:
            if time.time() > deadline - 400:
                break
            h2, c2 = slsqp_opt(best_h, n_init, min(time.time() + 40, deadline - 400),
                              alpha=a, maxiter=1000)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Coordinate descent
    if time.time() < deadline - 250:
        best_h, best_c = coord_search(best_h, n_init, deadline - 200)

    # Phase 4: Pair search
    if time.time() < deadline - 150:
        best_h, best_c = pair_search(best_h, n_init, deadline - 100,
                                     deltas=[0.005, 0.002, 0.001, 0.0005, 0.0002],
                                     radius=max(10, n_init // 6))

    # Phase 5: Multi-resolution optimization
    for n_try in [400, 500, 600, 800]:
        if time.time() > deadline - 80:
            break
        h_interp = interp_to_n(best_h, len(best_h), n_try)
        alloc = min(200, deadline - time.time() - 60)
        if alloc < 40:
            break
        # L-BFGS at new resolution
        t_lb = time.time() + alloc * 0.35
        h_t, c_t = lbfgs_opt(h_interp, n_try, t_lb,
                             alphas=(1000, 5000, 20000, 100000, 500000))
        # SLSQP
        if time.time() < deadline - 50 and alloc > 60:
            h_t, c_t = slsqp_opt(h_t, n_try, min(time.time() + alloc * 0.25, deadline - 50),
                                alpha=200000, maxiter=800)
        # Local search
        if time.time() < deadline - 40 and alloc > 80:
            h_t, c_t = coord_search(h_t, n_try, min(time.time() + alloc * 0.2, deadline - 40))
        if time.time() < deadline - 30 and alloc > 100:
            h_t, c_t = pair_search(h_t, n_try, min(time.time() + alloc * 0.15, deadline - 30),
                                   deltas=[0.002, 0.001, 0.0005, 0.0002],
                                   radius=max(10, n_try // 8))
        if c_t < best_c:
            best_c = c_t
            best_h = h_t

    # Phase 6: Final aggressive polish on best solution
    nf = len(best_h)
    if time.time() < deadline - 10:
        best_h, best_c = lbfgs_opt(best_h, nf, deadline - 8,
                                   alphas=(100000, 500000, 2000000))
    if time.time() < deadline - 4:
        best_h, best_c = pair_search(best_h, nf, deadline - 3,
                                     deltas=[0.001, 0.0005, 0.0002, 0.0001],
                                     radius=max(5, nf // 10))

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
