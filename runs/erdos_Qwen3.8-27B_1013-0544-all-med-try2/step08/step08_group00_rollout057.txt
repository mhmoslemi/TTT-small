```python
def exact_pair_line_search(h, n, i, j, c):
    """Find optimal δ to transfer from h[j] to h[i], minimizing max overlap."""
    dx = 2.0 / n
    # Compute a_k = dc_k/dδ for all k
    # c_k = sum_{valid m} h[m]*(1-h[m+k])*dx
    # d(c_k)/d(h[i]) = dx*(1-h[i+k]) - dx*h[i-k] (out of bounds = 0)
    # d(c_k)/d(h[j]) = dx*(1-h[j+k]) - dx*h[j-k]
    # a_k = d(c_k)/dδ = d(c_k)/d(h[i]) - d(c_k)/d(h[j])
    #      = dx*[(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]]
    #      = dx*[h[j+k] + h[j-k] - h[i+k] - h[i-k]]
    
    def h_ext(idx):
        return h[idx] if 0 <= idx < n else 0.0
    
    # Compute a_k for all shifts k from -(n-1) to n-1
    # a_k = dx * (h[j+k] + h[j-k] - h[i+k] - h[i-k]) with out-of-bounds = 0
    # Vectorized:
    def shift_sum(idx):
        # Returns h[idx+k] + h[idx-k] for k = -(n-1)..n-1, with OOB = 0
        # h[idx+k]: k ranges -(n-1) to n-1, so idx+k ranges idx-(n-1) to idx+n-1
        # Valid when 0 <= idx+k < n, i.e., -idx <= k < n-idx
        result = np.zeros(2*n-1)
        # k = m - (n-1) for m in range(2n-1)
        ks = np.arange(-(n-1), n)
        valid_plus = (idx + ks >= 0) & (idx + ks < n)
        valid_minus = (idx - ks >= 0) & (idx - ks < n)
        result[valid_plus] += h[idx + ks[valid_plus]]
        result[valid_minus] += h[idx - ks[valid_minus]]
        return result
    
    a = dx * (shift_sum(j) - shift_sum(i))
    
    # Now minimize max_k (c_k + a_k * δ) subject to bounds on δ
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if lo >= hi:
        return 0.0
    
    # Evaluate at boundaries and 0
    candidates = [lo, hi, 0.0]
    
    # Find top lines by current c_k
    top_idx = np.argsort(-c)[:30]  # top 30 by current value
    # Also include top by |a_k| (most impactful)
    top_a = np.argsort(-np.abs(a))[:20]
    combined = list(set(list(top_idx) + list(top_a)))
    
    # Find intersections of top lines
    for p in range(len(combined)):
        for q in range(p+1, len(combined)):
            k1, k2 = combined[p], combined[q]
            if abs(a[k1] - a[k2]) < 1e-15:
                continue
            delta = (c[k2] - c[k1]) / (a[k1] - a[k2])
            if lo <= delta <= hi:
                candidates.append(delta)
    
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= lo) & (candidates <= hi)]
    
    # Evaluate max at each candidate
    best_delta = 0.0
    best_val = compute_c5(h, n)  # current value
    for d in candidates:
        val = np.max(c + a * d)
        if val < best_val:
            best_val = val
            best_delta = d
    
    return best_delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        kk = k - (n - 1)
        wm = w[k] * dx
        if kk >= 0:
            nv = n - kk
            grad[:nv] += wm * (1.0 - x[kk:])
            grad[kk:] += wm * (-x[:nv])
        else:
            kk2 = -kk
            nv = n - kk2
            grad[kk2:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk2:])
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def lbfgs_run(h, n, deadline, max_iters=500, alphas=None, mus=None):
    best_c = compute_c5(h, n)
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    if mus is None:
        mus = [0.1, 0.5, 1.0, 5.0, 20.0, 100.0, 500.0, 2000.0]
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in zip(alphas, mus):
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iters * 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def exact_pair_delta(h, n, i, j, c):
    """Compute optimal transfer amount from j to i."""
    dx = 2.0 / n
    ks = np.arange(-(n-1), n)
    # h[j+k], h[j-k], h[i+k], h[i-k] with OOB=0
    jp = np.where((j+ks>=0) & (j+ks<n), j+ks, -1)
    jm = np.where((j-ks>=0) & (j-ks<n), j-ks, -1)
    ip = np.where((i+ks>=0) & (i+ks<n), i+ks, -1)
    im = np.where((i-ks>=0) & (i-ks<n), i-ks, -1)
    
    hjk = np.where(jp>=0, h[jp], 0.0)
    hjm = np.where(jm>=0, h[jm], 0.0)
    hik = np.where(ip>=0, h[ip], 0.0)
    him = np.where(im>=0, h[im], 0.0)
    
    a = dx * (hjk + hjm - hik - him)
    
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-15:
        return 0.0
    
    # Get top lines
    top_c = np.argsort(-c)[:40]
    top_a = np.argsort(-np.abs(a))[:20]
    combined = np.unique(np.concatenate([top_c, top_a]))
    
    candidates = [lo, hi, 0.0]
    for p in range(len(combined)):
        for q in range(p+1, len(combined)):
            k1, k2 = combined[p], combined[q]
            denom = a[k1] - a[k2]
            if abs(denom) < 1e-15:
                continue
            d = (c[k2] - c[k1]) / denom
            if lo <= d <= hi:
                candidates.append(d)
    
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= lo) & (candidates <= hi)]
    
    best_d = 0.0
    best_val = np.max(c)
    for d in candidates:
        val = np.max(c + a * d)
        if val < best_val - 1e-15:
            best_val = val
            best_d = d
    
    return best_d

        # Find top pairs: large gradient difference
```

```python
def _compute_a(h, n, i, j):
    """Compute a_k = dx * [h[j+k] + h[j-k] - h[i+k] - h[i-k]] for all k."""
    dx = 2.0 / n
    ks = np.arange(-(n-1), n)
    jp = j + ks
    jm = j - ks
    ip = i + ks
    im = i - ks
    a = np.zeros(2*n-1)
    vp = (jp >= 0) & (jp < n)
    vm = (jm >= 0) & (jm < n)
    vp2 = (ip >= 0) & (ip < n)
    vm2 = (im >= 0) & (im < n)
    a[vp] += h[jp[vp]]
    a[vm] += h[jm[vm]]
    a[vp2] -= h[ip[vp2]]
    a[vm2] -= h[im[vm2]]
    return a * dx

def _find_best_delta(c, a, h, i, j):
    """Find δ minimizing max(c + a*δ) subject to bounds."""
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-15:
        return 0.0
    
    # Top lines by c value
    top = np.argsort(-c)[:25]
    # Also add lines with large |a|
    top_a = np.argsort(-np.abs(a))[:10]
    combined = np.unique(np.concatenate([top, top_a]))
    
    candidates = np.array([lo, 0.0, hi])
    # Intersections
    n_c = len(combined)
    for p in range(n_c):
        for q in range(p+1, n_c):
            k1, k2 = combined[p], combined[q]
            denom = a[k1] - a[k2]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        kk = k - (n - 1)
        wm = w[k] * dx
        if kk >= 0:
            nv = n - kk
            grad[:nv] += wm * (1.0 - x[kk:])
            grad[kk:] += wm * (-x[:nv])
        else:
            kk2 = -kk
            nv = n - kk2
            grad[kk2:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk2:])
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def lbfgs_run(h, n, deadline, max_iters=400, alphas=None, mus=None):
    best_c = compute_c5(h, n)
    if alphas is None:
        alphas = [200, 1000, 5000, 25000, 100000, 500000, 2e6, 1e7, 5e7]
    if mus is None:
        mus = [0.1, 0.5, 2.0, 10.0, 50.0, 200.0, 1000.0, 5000.0, 2e4]
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in zip(alphas, mus):
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(_obj, h, jac=_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iters * 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def _compute_a(h, n, i, j):
    dx = 2.0 / n
    ks = np.arange(-(n - 1), n)
    jp, jm = j + ks, j - ks
    ip, im = i + ks, i - ks
    a = np.zeros(2 * n - 1)
    vp = (jp >= 0) & (jp < n)
    vm = (jm >= 0) & (jm < n)
    vp2 = (ip >= 0) & (ip < n)
    vm2 = (im >= 0) & (im < n)
    a[vp] += h[jp[vp]]
    a[vm] += h[jm[vm]]
    a[vp2] -= h[ip[vp2]]
    a[vm2] -= h[im[vm2]]
    return a * dx

def _find_best_delta(c, a, hi_i, lo_i, hi_j, lo_j):
    lo = max(lo_i, lo_j)
    hi = min(hi_i, hi_j)
    if hi - lo < 1e-14:
        return 0.0
    top_c = np.argsort(-c)[:20]
    top_a = np.argsort(-np.abs(a))[:8]
    combined = np.unique(np.concatenate([top_c, top_a]))
    n_c = len(combined)
    candidates = [lo, 0.0, hi]
    for p in range(n_c):
        for q in range(p + 1, n_c):
            k1, k2 = combined[p], combined[q]
            denom = a[k1] - a[k2]
            if abs(denom) < 1e-15:
                continue
            d = (c[k2] - c[k1]) / denom
            if lo < d < hi:
                candidates.append(d)
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= lo) & (candidates <= hi)]
    best_d, best_v = 0.0, c.max()
    for d in candidates:
        v = np.max(c + a * d)
        if v < best_v - 1e-15:
            best_v, best_d = v, d
    return best_d

def exact_pair_search(h, n, deadline, max_rounds=200, n_top=12):
    best_c = compute_c5(h, n)
    c = compute_c5_vec(h, n)
    pairs_since_recompute = 0
    for _r in range(max_rounds):
        if time.time() > deadline:
            break
        m = c.max()
        if m < 1e-15:
            break
        alpha = max(500.0, 50.0 / m)
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            kk = k - (n - 1)
            wm = w[k] * dx
            if kk >= 0:
                nv = n - kk
                grad[:nv] += wm * (1.0 - h[kk:])
                grad[kk:] += wm * (-h[:nv])
            else:
                kk2 = -kk
                nv = n - kk2
                grad[kk2:] += wm * (1.0 - h[:nv])
                grad[:nv] += wm * (-h[kk2:])
        top_down = np.argsort(grad)[-n_top:]
        top_up = np.argsort(grad)[:n_top]
        improved = False
        for ii in top_up:
            for jj in top_down:
                if ii == jj:
                    continue
                if h[ii] >= 0.999 or h[jj] <= 0.001:
                    continue
                a = _compute_a(h, n, ii, jj)
                d = _find_best_delta(c, a, 1.0 - h[ii], -h[ii], h[jj], h[jj] - 1.0)
                if abs(d) > 1e-12:
                    c += a * d
                    h[ii] += d
                    h[jj] -= d
                    np.clip(h, 0.0, 1.0, out=h)
                    pairs_since_recompute += 1
                    if c.max() < best_c - 1e-14:
                        best_c = c.max()
                        improved = True
                    if pairs_since_recompute >= 30:
                        c = compute_c5_vec(h, n)
                        pairs_since_recompute = 0
        if not improved:
            break
    return h, best_c

def slsqp_run(h, n, deadline, alpha=5e6, maxiter=3000):
    bounds = [(0.0, 1.0)] * n

    def _eq(x, nn):
        return float(x.sum() - nn / 2.0)

    def _eqj(x, nn):
        return np.ones(nn)

    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eqj, 'args': (n,)}]
    try:
        res = minimize(_obj, h, args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)
```
