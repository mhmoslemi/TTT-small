For the analytical gradient, let me simplify by using a direct (but vectorized) computation:

```python
def c5_grad(x, n, dx, alpha):
    """Compute gradient of log-sum-exp of correlations."""
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()  # length 2n-1
    
    # grad[j] = dx * sum_k p[k] * ((1-x[j+s_k])*I - x[j-s_k]*I)
    # where s_k = k - (n-1)
    
    # Vectorized computation:
    # For each j, we need:
    # sum_k p[k] * (1-x[j+k-(n-1)]) * I(0<=j+k-(n-1)<n)
    # - sum_k p[k] * x[j-k+(n-1)] * I(0<=j-k+(n-1)<n)
    
    # Use FFT for the convolutions
    N = 1
    while N < 4*n:
        N *= 2
    fx = np.fft.rfft(x, N)
    fp = np.fft.rfft(p, N)
    
    # Second term: sum_k p[k] * x[j-k+(n-1)]
    # = (p * x)[j+(n-1)] (convolution)
    px = np.fft.irfft(fp * fx, N)
    term2 = px[n-1:n-1+n]  # indices n-1 to 2n-2
    
    # First term: sum_k p[k] * x[j+k-(n-1)]
    # = (p_rev * x)[j+n-3] where p_rev = p[::-1]
    fp_rev = np.fft.rfft(p[::-1], N)
    pxx = np.fft.irfft(fp_rev * fx, N)
    # pxx[m] = sum_k p_rev[k] * x[m-k] = sum_k p[2n-2-k] * x[m-k]
    # Let k' = 2n-2-k: = sum_{k'} p[k'] * x[m-2n+2+k']
    # We want: sum_k p[k] * x[j-k+(n-1)] for the FIRST term (with +s_k = +k-(n-1))
    # sum_k p[k] * x[j+k-(n-1)] = sum_k p[k] * x[(j-(n-1))+k]
    # Let m = 2n-2+j: m-2n+2+k' = j+k' → k' as before
    # So: sum_k p[k] * x[j+k-(n-1)] = pxx[2n-2+j] ... 
    # Let me verify: pxx[2n-2+j] = sum_{k'} p[k'] * x[2n-2+j-2n+2+k'] = sum_{k'} p[k'] * x[j+k']
    # But we want sum_k p[k] * x[j+k-(n-1)], not x[j+k]
    # Hmm, off by (n-1).
    
    # Let me redo this. I want:
    # T1[j] = sum_k p[k] * x[j+k-(n-1)] * I(0<=j+k-(n-1)<n)
    # = sum_k p[k] * x[(j-(n-1))+k] * I(valid)
    # = (p ⋆ x)[j-(n-1)] (cross-correlation)
    
    # Cross-correlation (p ⋆ x)[m] = sum_k p[k] * x[m+k]
    # = (p_flipped * x)[m + len(p) - 1]
    # where * is convolution and p_flipped[k] = p[len(p)-1-k]
    
    # len(p) = 2n-1, so (p ⋆ x)[m] = (p_flipped * x)[m + 2n-2]
    
    p_flip = p[::-1]  # length 2n-1
    fpf = np.fft.rfft(p_flip, N)
    corr = np.fft.irfft(fpf * fx, N)
    # corr[i] = (p_flip * x)[i] = sum_k p_flip[k] * x[i-k]
    # = sum_k p[2n-2-k] * x[i-k]
    # Let k' = 2n-2-k: = sum_{k'} p[k'] * x[i-2n+2+k']
    # We want (p ⋆ x)[m] = sum_k p[k] * x[m+k]
    # Setting i-2n+2+k' = m+k' → i = m+2n-2
    # So (p ⋆ x)[m] = corr[m+2n-2]
    
    # T1[j] = (p ⋆ x)[j-(n-1)] = corr[j-(n-1)+2n-2] = corr[j+n-3]
    # For j in [0, n-1], j+n-3 ranges from n-3 to 2n-4
    # Need to check: is this within the valid range of corr?
    # corr has length N (>= 4n), and the linear convolution has length (2n-1)+n-1 = 3n-2
    # j+n-3 ranges from n-3 to 2n-4, which is within [0, 3n-3]. Good.
    
    term1_conv = corr[n-3:n-3+n]  # T1[j] for j=0,...,n-1
    
    # Now, the FIRST part of the gradient is:
    # sum_k p[k] * (1-x[j+s_k]) * I(valid) = S[j] - T1[j]
    # where S[j] = sum_k p[k] * I(0<=j+s_k<n) = sum of p[k] for valid k
    
    # S[j] = sum_{k: 0<=j+k-(n-1)<n} p[k] = sum_{k: n-1-j<=k<=n-1+j} p[k]
    # (clipped to [0, 2n-2])
    
    p_cum = np.concatenate([[0.0], np.cumsum(p)])
    lo = np.clip(n-1-np.arange(n), 0, 2*n-1)
    hi = np.clip(n-1+np.arange(n)+1, 0, 2*n)
    S = p_cum[hi] - p_cum[lo]
    
    grad = dx * (S - term1_conv - term2)
    return grad
```
