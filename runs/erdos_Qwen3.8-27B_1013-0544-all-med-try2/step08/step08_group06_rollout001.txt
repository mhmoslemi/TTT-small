```
def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).reshape(n)
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    mid = 0.5*(lo+hi)
    for _ in range(80):
        mid = 0.5*(lo+hi)
        s = float(np.sum(np.clip(h-mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h-mid, 0.0, 1.0)
    diff = target - float(p.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            r = 1.0 - p
            if r.sum() > 0: p += diff/r.sum()*r
        else:
            if p.sum() > 0: p += diff/p.sum()*p
        p = np.clip(p,0.0,1.0)
    return p
```

```
def project_feasible(h, n):
    h = np.asarray(h, dtype=float).reshape(n)
    target = n/2.0
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    mid = 0.5*(lo+hi)
    for _ in range(70):
        mid = 0.5*(lo+hi)
        s = float(np.sum(np.clip(h-mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h-mid, 0.0, 1.0)
    diff = target - float(p.sum())
    if abs(diff) > 1e-13:
        if diff > 0.0:
            r = 1.0 - p
            sr = float(r.sum())
            if sr > 0.0:
                p = p + diff/sr*r
        else:
            sp = float(p.sum())
            if sp > 0.0:
                p = p + diff/sp*p
        p = np.clip(p, 0.0, 1.0)
    return p
```

```python
import numpy as np
import time
from scipy.optimize import minimize

_G_ALPHA = 0.0
_G_W = None
_G_WR = None
_DEADLINE = 0.0


def set_params(alpha, w):
    global _G_ALPHA, _G_W, _G_WR
    _G_ALPHA = float(alpha)
    if w is None:
        _G_W = None
        _G_WR = None
    else:
        w = np.asarray(w, dtype=float)
        s = float(w.sum())
        if s <= 0.0:
            w = np.zeros_like(w)
            w[len(w) // 2] = 1.0
        else:
            w = w / s
        _G_W = w
        _G_WR = w[::-1].copy()


def set_deadline(t):
    global _DEADLINE
    _DEADLINE = float(t)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).reshape(n)
    target = n / 2.0
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    mid = 0.5 * (lo + hi)
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - mid, 0.0, 1.0)
    for _ in range(3):
        s = float(p.sum())
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            r = 1.0 - p
            sr = float(r.sum())
            if sr > 0.0:
                p = p + diff / sr * r
        else:
            sp = float(p.sum())
            if sp > 0.0:
                p = p + diff / sp * p
        p = np.clip(p, 0.0, 1.0)
    return p


def corr_T(x):
    x = np.asarray(x, dtype=float)
    n = len(x)
    return np.correlate(1.0 - x, x, mode='full') * (2.0 / n)


def c5_from_h(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))


def sum_constraint(x):
    return float(np.sum(x) - len(x) / 2.0)


def make_bounds(n):
    return [(0.0, 1.0)] * n


def make_weights(c, alpha):
    m = float(np.max(c))
    if alpha <= 0.0:
        w = (c >= m).astype(float)
        if w.sum() == 0.0:
            w[len(c) // 2] = 1.0
    else:
        shifted = float(alpha) * (c - m)
        e = np.exp(np.clip(shifted, -740.0, 0.0))
        w = e / e.sum()
    return w


def objective(x):
    if time.time() > _DEADLINE:
        raise RuntimeError('timeout')
    if _G_W is not None:
        return float(np.dot(_G_W, corr_T(x)))
    c = corr_T(x)
    a = _G_ALPHA
    m = float(np.max(c))
    if a <= 0.0:
        return m
    shifted = a * (c - m)
    e = np.exp(np.clip(shifted, -740.0, 0.0))
    s = float(e.sum())
    if s <= 0.0:
        return m
    return m + float(np.log(s)) / a


def jacobian(x):
    if time.time() > _DEADLINE:
        raise RuntimeError('timeout')
    x = np.asarray(x, dtype=float)
    n = len(x)
    y = 1.0 - x
    if _G_W is not None:
        w = _G_W
        wr = _G_WR
    else:
        w = make_weights(corr_T(x), _G_ALPHA)
        wr = w[::-1].copy()
    g1 = np.convolve(wr, y, mode='full')[n - 1:2 * n - 1]
    g2 = np.convolve(w, x, mode='full')[n - 1:2 * n - 1]
    return (2.0 / n) * (g1 - g2)


def slsqp_once(h, n, alpha, w, maxiter, bounds, cons):
    set_params(alpha, w)
    try:
        res = minimize(
            objective,
            h,
            jac=jacobian,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max(1, int(maxiter)), 'ftol': 1e-11}
        )
        out = res.x
        if out is None:
            out = h
    except Exception:
        out = h
    set_params(0.0, None)
    return project_feasible(out, n)


def active_polish(h, n, deadline, max_iters=100):
    set_deadline(deadline)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c5_from_h(h)
    tol = 1e-5
    for it in range(int(max_iters)):
        if time.time() > deadline:
            break
        c = corr_T(h)
        m = float(np.max(c))
        w = make_weights(c, 0.0)
        set_params(0.0, w)
        try:
            g = jacobian(h)
        except Exception:
            break
        g = g - float(np.mean(g))
        norm = float(np.linalg.norm(g))
        if norm < 1e-14:
            break
        base = 1.0 / (norm + 1e-14)
        improved = False
        for mult in (5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
            cand = project_feasible(h - mult * base * g, n)
            val = c5_from_h(cand)
            if val < best_c - 1e-13:
                h = cand.copy()
                best_c = val
                best_h = h.copy()
                improved = True
                break
        if not improved:
            tol = max(1e-10, tol * 0.5)
            if it > 15:
                break
    set_params(0.0, None)
    return best_h, float(best_c)


def improve(h, n, deadline, smooth_alphas, re_alphas, maxiter, r_maxiter, cycles, active_iters):
    set_deadline(deadline)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c5_from_h(h)
    bounds = make_bounds(n)
    cons = [{'type': 'eq', 'fun': sum_constraint}]

    for a in smooth_alphas:
        if time.time() > deadline - 5.0:
            break
        h_new = slsqp_once(h, n, a, None, maxiter, bounds, cons)
        c_new = c5_from_h(h_new)
        if c_new < best_c + 1e-13:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        elif c_new < best_c + 0.02:
            h = h_new
        else:
            h = project_feasible(0.5 * (h + h_new), n)

    for aw in re_alphas:
        for _ in range(int(cycles)):
            if time.time() > deadline - 5.0:
                break
            w = make_weights(corr_T(h), float(aw))
            h_new = slsqp_once(h, n, 0.0, w, r_maxiter, bounds, cons)
            c_new = c5_from_h(h_new)
            if c_new < best_c + 1e-13:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
            elif c_new < best_c + 0.02:
                h = h_new
            else:
                h = project_feasible(0.5 * (h + h_new), n)

    if active_iters > 0 and time.time() < deadline - 4.0:
        h2, c2 = active_polish(h, n, deadline - 3.0, active_iters)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    set_params(0.0, None)
    return best_h, float(best_c)


def make_start(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n)
    h = np.full(n, 0.5, dtype=float)
    for m in range(1, 11):
        amp = 0.12 / (1.0 + 0.25 * m)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h = h + amp * np.cos(m * np.pi * x + phase)
    h = project_feasible(h, n)
    r = np.zeros(n)
    if n >= 2:
        idx = rng.choice(n, n // 2, replace=False)
        r[idx] = 1.0
    return project_feasible(0.65 * h + 0.35 * r, n)


def to_binary(h, n):
    h = np.asarray(h, dtype=float)
    hb = np.zeros(n)
    k = n // 2
    if k > 0:
        if k < n:
            idx = np.argpartition(h, -k)[-k:]
        else:
            idx = np.arange(n)
        hb[idx] = 1.0
    return hb


def resample(h, n):
    m = len(h)
    xo = np.linspace(0.0, 2.0, m)
    xn = np.linspace(0.0, 2.0, n)
    return project_feasible(np.interp(xn, xo, h), n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = max(30.0, float(budget_s) - 20.0)
    deadline = t0 + B
    set_deadline(deadline)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 400
        h0 = make_start(n0, seed)

    best_h = project_feasible(h0, n0)
    best_n = n0
    best_c = c5_from_h(best_h)

    if B > 500.0:
        smooth_main = (50.0, 150.0, 500.0, 2000.0, 10000.0, 50000.0, 200000.0, 1000000.0)
        re_main = (500.0, 2000.0, 10000.0, 50000.0, 200000.0)
        mid_max = 105
        mid_rmax = 75
        mid_cyc = 2
        mid_act = 120
    else:
        smooth_main = (100.0, 1000.0, 10000.0, 100000.0)
        re_main = (1000.0, 10000.0, 100000.0)
        mid_max = 60
        mid_rmax = 40
        mid_cyc = 1
        mid_act = 50

    main_dl = t0 + (0.60 if B > 500.0 else 0.72) * B
    h, c = improve(best_h, best_n, main_dl, smooth_main, re_main, mid_max, mid_rmax, mid_cyc, mid_act)
    if c < best_c:
        best_h, best_c, best_n = h, c, best_n

    if best_n % 2 == 0 and time.time() + 15.0 < deadline:
        binary_dl = t0 + 0.76 * B
        if time.time() + 15.0 < binary_dl:
            hb = to_binary(best_h, best_n)
            hb, c = improve(hb, best_n, binary_dl, (500.0, 5000.0, 50000.0), (2000.0, 20000.0), 65, 45, 1, 30)
            if c < best_c:
                best_h, best_c = hb, c

    cand_ns = []
    for nn in (best_n + 100, best_n - 100, 500, 320):
        if 120 <= nn <= 700 and nn not in cand_ns:
            cand_ns.append(nn)
    other_dl = t0 + 0.90 * B
    for nn in cand_ns:
        if time.time() + 60.0 > other_dl:
            break
        hc = resample(best_h, nn)
        hc, c = improve(hc, nn, other_dl, (500.0, 5000.0, 50000.0), (2000.0, 20000.0), 55, 35, 1, 20)
        if c < best_c:
            best_h, best_c, best_n = hc, c, nn

    final_dl = deadline - 2.0
    if time.time() + 10.0 < final_dl:
        hf, c = active_polish(best_h, best_n, final_dl, 160)
        if c < best_c:
            best_h, best_c = hf, c

    if time.time() + 25.0 < final_dl:
        rng = np.random.RandomState(seed + 11)
        hn = project_feasible(best_h + 0.003 * (rng.random(best_n) - 0.5), best_n)
        hn, c = improve(hn, best_n, final_dl, (50000.0, 500000.0), (50000.0, 200000.0), 60, 40, 1, 25)
        if c < best_c:
            best_h, best_c = hn, c

    best_h = project_feasible(best_h, best_n)
    best_c = c5_from_h(best_h)
    return best_h, float(best_c), int(best_n)
```
