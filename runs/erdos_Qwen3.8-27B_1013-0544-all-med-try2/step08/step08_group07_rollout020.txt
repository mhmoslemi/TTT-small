def weighted_slsqp(h, n, weights, maxiter=3000, ftol=1e-16):
    dx = 2.0 / n
    ncorr = 2 * n - 1
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return float(np.dot(weights, c * dx))
    def obj_grad(x):
        # Analytic gradient
        # c[k] = sum_i x[i] * (1-x[i+k-(n-1)])
        # d c[k] / d x[j] = (1-x[j+k-(n-1)]) * valid1 - x[j-k+(n-1)] * valid2
        grad = np.zeros(n)
        for k in range(ncorr):
            if abs(weights[k]) < 1e-20:
                continue
            offset = k - (n - 1)
            # Term 1: (1-x[j+offset]) for j+offset in [0,n)
            # j in [max(0,-offset), min(n, n-offset))
            j_lo = max(0, -offset)
            j_hi = min(n, n - offset)
            if j_lo < j_hi:
                grad[j_lo:j_hi] += (1.0 - x[j_lo+offset:j_hi+offset]) * weights[k] * dx
            # Term 2: -x[j-offset] for j-offset in [0,n)
            # j in [max(0,offset), min(n, n+offset))
            j_lo2 = max(0, offset)
            j_hi2 = min(n, n + offset)
            if j_lo2 < j_hi2:
                grad[j_lo2:j_hi2] -= x[j_lo2-offset:j_hi2-offset] * weights[k] * dx
        return grad
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def softmax_slsqp(h, n, alpha, maxiter=3000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def compute_c5(h, n):

def project_feasible(h, n):

            s = h.sum()
            if s < 1e-15:
                break
