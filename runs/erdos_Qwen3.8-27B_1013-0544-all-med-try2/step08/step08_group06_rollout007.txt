import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(y, target):
    y = np.asarray(y, dtype=float)
    lo = float(np.min(y) - 1.0)
    hi = float(np.max(y) + 1.0)
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def to_binary(h, n):
    m = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n, dtype=np.int8)
    b[idx[:m]] = 1
    return b

def bin_state(hb, n):
    hf = hb.astype(float)
    O = np.correlate(1.0 - hf, hf, mode='full').astype(np.int32)
    L = 2 * n - 1
    off = n - 1
    D = np.zeros((n, L), dtype=np.int32)
    rows = np.arange(n, dtype=np.int32)
    for q in range(n):
        if hb[q] == 1:
            D[rows, off + rows - q] -= 1
        else:
            D[rows, off + q - rows] += 1
    D[:, off] = 0
    return O, D

def bin_update_D(D, rows, off, q, delta):
    D[rows, off + q - rows] -= delta
    D[rows, off + rows - q] -= delta
    D[q, off] = 0

def best_j_for_zero(i, O, D, ones, M, B, rowsB, off):
    np.add(O, D[i], out=M)
    np.subtract(M[None, :], D[ones, :], out=B)
    maxB = B.max(axis=1)
    l1 = off + i - ones
    l2 = off + ones - i
    val = np.maximum(B[rowsB, l1] + 1, B[rowsB, l2] + 1)
    cand = np.maximum(maxB, val)
    k = int(cand.argmin())
    return int(ones[k]), int(cand[k]), k

def swap_apply(hb, O, D, zeros, ones, rows, off, n, zi, oj, i, j):
    p1 = off + i - j
    p2 = off + j - i
    O += D[i]
    O -= D[j]
    O[p1] += 1
    O[p2] += 1
    bin_update_D(D, rows, off, i, 1)
    bin_update_D(D, rows, off, j, -1)
    hb[i] = 1
    hb[j] = 0
    zeros[zi] = j
    ones[oj] = i

def fft_size(n):
    need = max(2, 3 * n - 2)
    return 1 << (need - 1).bit_length()

def corr_from_circ(c, n):
    L = 2 * n - 1
    out = np.empty(L)
    if n > 1:
        out[:n - 1] = c[-(n - 1):]
    out[n - 1:] = c[:n]
    return out

def smooth_obj(x, n, alpha):
    x = np.asarray(x, dtype=float)
    dx = 2.0 / n
    N = fft_size(n)
    H = np.fft.rfft(x, N)
    V = np.fft.rfft(1.0 - x, N)
    c = np.fft.irfft(H * np.conj(V), N)
    g = corr_from_circ(c, n) * dx
    m = float(g.max())
    shift = alpha * (g - m)
    s = np.exp(shift)
    tot = float(s.sum())
    if tot <= 0.0:
        return m
    return m + math.log(tot) / alpha

def fgrad_smooth(x, n, alpha):
    x = np.asarray(x, dtype=float)
    dx = 2.0 / n
    L = 2 * n - 1
    off = n - 1
    N = fft_size(n)
    H = np.fft.rfft(x, N)
    V = np.fft.rfft(1.0 - x, N)
    c = np.fft.irfft(H * np.conj(V), N)
    g = corr_from_circ(c, n) * dx
    m = float(g.max())
    shift = alpha * (g - m)
    s = np.exp(shift)
    tot = float(s.sum())
    if tot <= 0.0:
        active = (g >= m).astype(float)
        w = active / max(1.0, float(active.sum()))
        f = m
    else:
        f = m + math.log(tot) / alpha
        w = s / tot
    active = (g >= m - 1e-12)
    if active.any():
        w = w + active / float(active.sum())

    Wf = np.fft.rfft(w, N)
    Wrf = np.fft.rfft(w[::-1], N)
    conv1 = np.fft.irfft(H * Wf, N)
    conv2 = np.fft.irfft(H * Wrf, N)
    H1 = np.asarray(conv1[off:off + n], dtype=float)
    H2 = np.asarray(conv2[off:off + n], dtype=float)

    wcum = np.empty(L + 1, dtype=float)
    wcum[0] = 0.0
    wcum[1:] = np.cumsum(w)
    p = np.arange(n, dtype=np.int64)
    min_idx = off - n + 1 + p
    max_idx = off + p
    W = wcum[max_idx + 1] - wcum[min_idx]
    grad = dx * (W - H1 - H2)
    return f, grad

            f_new = smooth_obj(x, n, alpha)

def slsqp_obj(x, n, alpha):
    return smooth_obj(x, n, alpha)

def slsqp_grad(x, n, alpha):
    _, g = fgrad_smooth(x, n, alpha)
    return g

def sum_con_fun(x):
    return float(x.sum() - len(x) / 2.0)

def sum_con_jac(x):
    return np.ones_like(x)

def slsqp_polish(h, n, deadline, alpha, maxiter=60):
    x0 = project_box_sum(np.asarray(h, dtype=float), n / 2.0)
    if time.time() > deadline - 25:
        return x0
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac},)
    try:
        res = minimize(
            slsqp_obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
            jac=slsqp_grad, args=(n, alpha),
            options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0}
        )
        x = np.asarray(res.x, dtype=float)
    except Exception:
        x = x0
    return project_box_sum(x, n / 2.0)

    h = project_box_sum(np.asarray(h, dtype=float), n / 2.0)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    hcur = h.copy()
    cur_c = best_c
    reserve = 45.0 if do_slsqp else 5.0
    for idx, a in enumerate(alphas):
        rem = deadline - reserve - time.time()
        if rem < 8.0:
            break
        t_alpha = time.time() + min(rem, 45.0 + 20.0 * idx)
        h1, c1 = gradient_alpha(hcur, n, float(a), t_alpha, seed + 31 * idx)
        if c1 < best_c:
            best_c = c1
            best_h = h1.copy()
        if c1 < cur_c:
            hcur = h1.copy()
            cur_c = c1
        else:
            hcur = best_h.copy()
            cur_c = best_c
    if do_slsqp and time.time() + 35 < deadline:
        x = slsqp_polish(best_h, n, deadline - 2, float(alphas[-1]),
                         maxiter=35 if n > 450 else 60)
        c = compute_c5(x, n)
        if c < best_c:
            best_c = c
            best_h = x.copy()
    return best_h, best_c

def interp_h(h, n_old, n_new):
    h = np.asarray(h, dtype=float)
    if n_new == n_old:
        return h.copy()
    xo = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(xn, xo, h, left=float(h[0]), right=float(h[-1]))
    return project_box_sum(y, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(60.0, float(budget_s) - 8.0)
    dl = t0 + total

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    if n0 % 2:
        if n0 > 2:
            h0 = h0[:n0 - 1]
        else:
            h0 = np.full(400, 0.5)
        n0 = len(h0)
    h0 = project_box_sum(h0, n0 / 2.0)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    t1 = t0 + 0.17 * total
    h1, c1 = continuous_refine(best_h, best_n, t1, seed,
                               [20, 60, 200, 600, 2000], False)
    if c1 < best_c:
        best_c, best_h, best_n = c1, h1, best_n

    t2 = t0 + 0.44 * total
    hb, cb = binary_refine(best_h, best_n, t2, seed, 350)
    if cb < best_c:
        best_c, best_h, best_n = cb, hb, best_n

    t3 = t0 + 0.71 * total
    h2, c2 = continuous_refine(best_h, best_n, t3, seed,
                               [100, 500, 2000, 8000, 20000], True)
    if c2 < best_c:
        best_c, best_h, best_n = c2, h2, best_n

    t4 = t0 + 0.88 * total
    for nh, sff in ((500, 101), (600, 202)):
        if time.time() > t4 - 45:
            break
        xh = interp_h(best_h, best_n, nh)
        c_xh = compute_c5(xh, nh)
        tb = min(t4 - 40, time.time() + 40)
        hbi, cbi = binary_refine(xh, nh, tb, seed + sff, 55)
        init = hbi if cbi < c_xh else xh
        h4, c4 = continuous_refine(init, nh, t4 - 18, seed + sff + 7,
                                   [200, 1000, 5000], nh <= 500)
        if c4 < best_c:
            best_c, best_h, best_n = c4, h4, nh
    hf, cf = continuous_refine(best_h, best_n, dl - 3, seed + 9999,
                               [10000, 30000, 80000], True)
    if cf < best_c:
        best_c, best_h, best_n = cf, hf, best_n

    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
