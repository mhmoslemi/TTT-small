    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:

- For a given k_star, fd[:, k_star] gives the change at k_star for each position
- For zeros: fd_zeros = fd[zeros_array, k_star]  (shape: n/2)
- For ones: fd_ones = fd[ones_array, k_star]  (shape: n/2)
- Total change at k_star for swap (i_idx, j_idx): fd_zeros[i_idx] + fd_ones[j_idx] + correction
- The correction depends on the specific i and j values (not just indices), so it's hard to fully vectorize
- But the correction is small (at most ±2), so we can ignore it for the initial ranking and add it back for verification

- scores = fd_zeros[:, None] + fd_ones[None, :]  (shape: n/2 × n/2)
- This gives the base change at k_star for each (i, j) pair
- Find the K smallest scores
- For each, add the correction and verify full max

Creating the scores matrix is O(n²) memory (15625 floats for n=250, trivial) and O(n²) time. Finding the K smallest is O(n² log K) or O(n²) with partial sort.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def compute_fd(h, n):
    ncorr = 2 * n - 1
    fd = np.zeros((n, ncorr))
    # Vectorized computation
    # For "first factor": fd[p, k] += sign * (1-h[p+s]) where s=k-(n-1), valid when 0<=p+s<n
    # For "second factor": fd[p, k] += sign2 * h[m] where m=p-k+n-1, valid when 0<=m<n
    for p in range(n):
        s_lo = max(0, -p)  # p+s >= 0
        s_hi = min(n-1, n-1-p)  # p+s <= n-1
        ks = np.arange(n-1+s_lo, n-1+s_hi+1)
        ps = p + ks - (n-1)
        if h[p] == 0:
            fd[p, ks] += (1.0 - h[ps])
        else:
            fd[p, ks] -= (1.0 - h[ps])
        
        ks2 = np.arange(p, p+n)
        ms = n-1+p-ks2
        if h[p] == 0:
            fd[p, ks2] -= h[ms]
        else:
            fd[p, ks2] += h[ms]
    return fd

def get_correction(i, j, n, ncorr):
    """Get the correction vector (length ncorr) for swap (i,j)."""
    corr_vec = np.zeros(ncorr)
    k0 = n - 1
    corr_vec[k0] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < ncorr:
        corr_vec[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < ncorr:
        corr_vec[k2] += 1.0
    return corr_vec

def binary_local_search(h_init, n, deadline, max_sweeps=200000, seed=42, top_k_count=5):
    dx = 2.0 / n
    ncorr = 2 * n - 1
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = corr.max()
    
    fd = compute_fd(h, n)
    
    best_c = cur_max
    best_h = h.copy()
    stale = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stale > 100:
            break
        
        ones_mask = (h == 1)
        zeros_mask = (h == 0)
        ones_arr = np.where(ones_mask)[0]
        zeros_arr = np.where(zeros_mask)[0]
        
        if len(ones_arr) == 0 or len(zeros_arr) == 0:
            break
        
        # Find top positions
        top_positions = np.argsort(corr)[::-1][:top_k_count]
        
        improved = False
        for k_star in top_positions:
            # Score all (i,j) pairs at position k_star
            fd_z = fd[zeros_arr, k_star]  # shape (n/2,)
            fd_o = fd[ones_arr, k_star]   # shape (n/2,)
            scores = fd_z[:, None] + fd_o[None, :]  # shape (n/2, n/2)
            
            # Find top K candidates (lowest scores = most reduction)
            flat_scores = scores.ravel()
            K = min(50, flat_scores.size)
            top_idx = np.argpartition(flat_scores, K)[:K]
            top_idx = top_idx[np.argsort(flat_scores[top_idx])]
            
            best_pair = None
            best_new_max = cur_max
            
            for tidx in top_idx:
                i_idx = tidx // len(ones_arr)
                j_idx = tidx % len(ones_arr)
                i_val = zeros_arr[i_idx]
                j_val = ones_arr[j_idx]
                
                # Compute full new correlation
                new_corr = corr + fd[i_val] + fd[j_val]
                # Add correction
                k0 = n - 1
                new_corr[k0] -= 2.0
                k1 = n - 1 + (j_val - i_val)
                if 0 <= k1 < ncorr:
                    new_corr[k1] += 1.0
                k2 = n - 1 + (i_val - j_val)
                if 0 <= k2 < ncorr:
                    new_corr[k2] += 1.0
                
                nm = new_corr.max()
                if nm < best_new_max:
                    best_new_max = nm
                    best_pair = (i_val, j_val, new_corr)
            
            if best_pair is not None and best_pair[2].max() < cur_max - 1e-10:
                i_val, j_val, new_corr = best_pair
                h[i_val] = 1.0
                h[j_val] = 0.0
                corr = new_corr
                cur_max = corr.max()
                fd = compute_fd(h, n)
                improved = True
                if cur_max < best_c:
                    best_c = cur_max
                    best_h = h.copy()
                break
        
        if not improved:
            stale += 1
        else:
            stale = 0
    
    return best_h, best_c * dx

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def sa_binary_search_fast(h_init, n, deadline, seed=42):
    dx = 2.0 / n
    ncorr = 2 * n - 1
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = corr.max()
    rng = np.random.RandomState(seed)
    
    best_c = cur_max
    best_h = h.copy()
    T = 0.05
    cooling = 0.99995
    step = 0
    
    while time.time() < deadline:
        step += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.choice(zeros)
        j = rng.choice(ones)
        
        # Compute delta using flip deltas (on-the-fly, O(n))
        # Flip i (0->1) and j (1->0)
        delta = np.zeros(ncorr)
        
        # First factor for i (0->1): add (1-h[i+s]) for valid s
        s_lo = max(0, -i)
        s_hi = min(n-1, n-1-i)
        ks = np.arange(n-1+s_lo, n-1+s_hi+1)
        delta[ks] += (1.0 - h[i + ks - (n-1)])
        
        # Second factor for i (1-h goes 1->0): subtract h[m] where m=i-s, k=n-1+i-m
        ks2 = np.arange(i, i+n)
        ms = n-1+i-ks2
        delta[ks2] -= h[ms]
        
        # First factor for j (1->0): subtract (1-h[j+s]) for valid s
        s_lo_j = max(0, -j)
        s_hi_j = min(n-1, n-1-j)
        ks_j = np.arange(n-1+s_lo_j, n-1+s_hi_j+1)
        delta[ks_j] -= (1.0 - h[j + ks_j - (n-1)])
        
        # Second factor for j (1-h goes 0->1): add h[m] where m=j-s
        ks2_j = np.arange(j, j+n)
        ms_j = n-1+j-ks2_j
        delta[ks2_j] += h[ms_j]
        
        # Corrections
        delta[n-1] -= 2.0
        k1 = n-1+(j-i)
        if 0 <= k1 < ncorr: delta[k1] += 1.0
        k2 = n-1+(i-j)
        if 0 <= k2 < ncorr: delta[k2] += 1.0
        
        new_corr = corr + delta
        new_max = new_corr.max()
        d = new_max - cur_max
        
        if d < 0 or rng.random() < np.exp(-d / T):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_max = new_max
            if cur_max < best_c:
                best_c = cur_max
                best_h = h.copy()
        # else: revert (h unchanged since we didn't modify it yet)
        
        T *= cooling
    
    return best_h, best_c * dx

Wait, for binary h, corr[0] = sum_i h[i]*(1-h[i]) = 0! Because h[i] is either 0 or 1, so h[i]*(1-h[i]) = 0.

So for binary h, corr[0] = 0. The max is at some other lag.

For a random binary sequence, E[corr[k]] = (n-|k|)/4 for k ≠ 0, and corr[0] = 0.
The max is expected to be around (n-1)/4 + some fluctuation ≈ 62 + O(√n).

But we want to MINIMIZE the max, so we want a sequence where the correlation is as flat as possible (excluding k=0).

This is essentially a "minimum peak sidelobe" problem, similar to radar sequence design.

For n=250 (125 ones, 125 zeros), the optimal max correlation is probably in the range of 40-60.
