    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Softmax weights
    max_c = c.max()
    exp_c = np.exp(alpha * (c - max_c))
    p = exp_c / exp_c.sum()

            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_box_sum(x, target):
    lo = x.min() - 1.0
    hi = x.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    max_c = c.max()
    shifted = alpha * (c - max_c)
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    val = max_c + np.log(sum_exp) / alpha
    p = exp_s / sum_exp
    # Gradient
    m_range = np.arange(2 * n - 1)
    d = m_range - (n - 1)
    J = np.arange(n)[:, None]
    idx1 = J + d[None, :]
    idx2 = J - d[None, :]
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    h_ext1 = np.where(mask1, x[idx1.clip(0, n-1)], 0.0)
    h_ext2 = np.where(mask2, x[idx2.clip(0, n-1)], 0.0)
    term1 = h_ext1 @ p
    term2 = h_ext2 @ p
    grad = dx * (1.0 - term1 - term2)
    return val, grad

def lfbgbsm(h, n, alpha, maxiter=5000, tol=1e-16):
    def fun(x):
        v, _ = softmax_obj_and_grad(x, n, alpha)
        return v
    def jac(x):
        _, g = softmax_obj_and_grad(x, n, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:

def coord_descent_fast(h, n, max_sweeps=200, deadline=None):

    h = h.astype(float).copy()

= min_h dx * ∑_i h[i] - dx * ∑_i,k p[k] h[i] h[i+k-n+1]

This is a quadratic optimization problem! Specifically:
= min_h dx * 1^T h - dx * h^T Q h

where Q is a matrix with Q[i,j] = ∑_k p[k] * δ_{j, i+k-n+1}

Since Q is symmetric (if p is symmetric around k=n-1) and the problem has bound constraints, we can solve it with standard QP methods.

But wait, the matrix Q might not be positive definite, so the problem might be non-convex. Let me check:

Q[i,j] = ∑_k p[k] * δ_{j, i+k-n+1} = p[i-j+n-1] (if i-j+n-1 is in [0, 2n-2])

So Q is a Toeplitz matrix with Q[i,j] = p[i-j+n-1].

The Hessian of the objective is -2 * dx * Q. For the problem to be convex, we need Q to be positive semi-definite, which means p needs to be a positive definite sequence (its Fourier transform must be non-negative).

In general, p is a probability distribution over k, and its Fourier transform is the characteristic function, which is always bounded but not necessarily non-negative. So the QP might be non-convex.

However, if p is concentrated near k = n-1 (i.e., the "zero lag"), then Q ≈ I * p[n-1], and the problem is approximately convex.

This suggests a strategy:
1. Start with p concentrated at the worst k (the one maximizing c[k])
2. Solve the QP
3. Update p based on the new worst k
5. Repeat (this is like a cutting-plane or column-generation approach)

This is essentially the **Dantzig-Wolfe decomposition** or **Benders decomposition** for the min-max problem.

Let me implement this:

```python
def dantzig_wolfe(h, n, max_iter=100, deadline=None):
    """Solve min_h max_k c[k]*dx using Dantzig-Wolfe style decomposition."""
    dx = 2.0 / n
    h = h.copy()
    
    # p is a distribution over k values
    n_k = 2 * n - 1
    p = np.zeros(n_k)
    p[n - 1] = 1.0  # Start with zero lag
    
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        
        # Find current worst k
        c = np.correlate(h, 1 - h, mode='full') * dx
        k_star = np.argmax(c)
        
        # Update p: put more weight on k_star
        p = 0.7 * p + 0.3 * np.zeros(n_k)
        p[k_star] += 0.3
        
        # Solve QP: min_h dx * sum(h) - dx * h^T Q(h, p)
        # subject to 0 <= h <= 1, sum(h) = n/2
        
        # Build Q matrix
        # Q[i,j] = p[i-j+n-1] (with p[k]=0 for k outside [0, 2n-2])
        # This is a Toeplitz matrix
        
        # Actually, let me just use the gradient directly
        # grad[h_i] = dx * sum_k p[k] * ((1 - h[i+k-n+1]) - h[i-k+n-1])
        
        # Use SLSQP on the weighted sum
        def obj(x):
            c = np.correlate(x, 1 - x, mode='full') * dx
            return float(np.dot(p, c))
        
        def cons(x):
            return x.sum() - n / 2.0
        
        try:
            res = minimize(obj, h, method='SLSQP',
                          bounds=[(0,1)]*n,
                          constraints=[{'type':'eq', 'fun': cons}],
                          options={'maxiter': 1000, 'ftol': 1e-12})
            h_new = project_feasible(np.clip(res.x, 0, 1), n)
            
            # Check if h_new is better
            c_new = np.correlate(h_new, 1 - h_new, mode='full') * dx
            c_old_max = c.max()
            c_new_max = c_new.max()
            
            if c_new_max < c_old_max:
                h = h_new
            else:
                # Reduce step
                h = 0.5 * h + 0.5 * h_new
                h = project_feasible(h, n)
                
        except Exception:
            break
    
    return h
```

def gradient_c5(h, n):
    """Compute gradient of C₅ = max_k c[k]*dx with respect to h."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx

    grad = np.zeros(n)

        idx2 = j - d
        term1 = (1 - h[idx1]) if 0 <= idx1 < n else 1.0
        term2 = h[idx2] if 0 <= idx2 < n else 0.0
        grad[j] = dx * (term1 - term2)
    return grad

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=500, deadline=None):

    for sweep in range(max_sweeps):

            k2 = i + m_range

            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
