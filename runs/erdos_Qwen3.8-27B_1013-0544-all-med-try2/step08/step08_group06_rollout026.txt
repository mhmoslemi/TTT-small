            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def obj_grad(h, n, p, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    S = np.sum(corr**p)
    if S < 1e-30:
        S = 1e-30
    f = S**(1.0/p) * dx
    factor = S**(1.0/p - 1.0) * dx
    W = corr**(p-1) * factor
    one_minus_h = 1.0 - h
    grad = np.empty(n)
    for i in range(n):
        a = W[i + n - 1 : i - 1 : -1]
        b = W[n - 1 - i : 2 * n - 1 - i]
        grad[i] = np.dot(a, one_minus_h) - np.dot(b, h)
    grad += 2.0 * penalty * (h.sum() - n / 2.0)
    return f, grad

def lbfgs_run(h_init, n, p, maxiter, deadline):
    dx = 2.0 / n
    penalty = 100.0
    def obj_wrap(x):
        f, _ = obj_grad(x, n, p, penalty)
        return f
    def grad_wrap(x):
        _, g = obj_grad(x, n, p, penalty)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_wrap, h_init, jac=grad_wrap, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10,
                                               'ftol': 1e-15, 'maxfun': maxiter * 3})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                if i != j:
                    delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B with increasing p
    p_schedule = [2, 5, 10, 20, 50, 100, 200, 500, 1000, 5000]
    alloc1 = 0.45 * total
    h_cur = best_h.copy()
    for p in p_schedule:
        if time.time() > t0 + alloc1:
            break
        h_cur = lbfgs_run(h_cur, n, p, 500, t0 + alloc1)
        c = compute_c5(h_cur, n)
        if c < best_c:
            best_c, best_h = c, h_cur.copy()

    # Phase 2: Binary local search on best continuous
    p2_end = t0 + 0.60 * total
    h_bin, c_bin = greedy_binary(best_h, n, p2_end, max_sweeps=300)
    if c_bin < best_c:
        best_c, best_h = c_bin, h_bin.copy()

    # Phase 3: Re-smooth from binary with high p
    p3_end = t0 + 0.75 * total
    h_cur = best_h.copy()
    for p in [100, 500, 2000, 10000]:
        if time.time() > p3_end:
            break
        h_cur = lbfgs_run(h_cur, n, p, 800, p3_end)
        c = compute_c5(h_cur, n)
        if c < best_c:
            best_c, best_h = c, h_cur.copy()

    # Phase 4: Multi-resolution
    p4_end = t0 + 0.90 * total
    for nh in [300, 500, 600]:
        if time.time() > p4_end - 20:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        h_new = np.interp(xn, xo, best_h)
        h_new = project_feasible(h_new, nh)
        for p in [50, 500, 5000]:
            if time.time() > p4_end - 15:
                break
            h_opt = lbfgs_run(h_new, nh, p, 400, p4_end - 15)
            c = compute_c5(h_opt, nh)
            if c < best_c:
                best_c, best_h, n = c, h_opt.copy(), nh
                h_new = h_opt.copy()

    # Phase 5: Noise perturbation + re-optimize
    p5_end = dl - 3
    for trial in range(4):
        if time.time() > p5_end - 5:
            break
        h_pert = best_h + 0.002 * rng.randn(n)
        h_pert = project_feasible(h_pert, n)
        for p in [1000, 10000]:
            if time.time() > p5_end - 3:
                break
            h_opt = lbfgs_run(h_pert, n, p, 500, p5_end - 3)
            c = compute_c5(h_opt, n)
            if c < best_c:
                best_c, best_h = c, h_opt.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
