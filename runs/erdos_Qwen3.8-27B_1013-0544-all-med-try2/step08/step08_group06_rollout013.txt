                grad[mp:] += w * (1.0 - x[:n - mp])
                grad[:n - mp] -= w * x[mp:]
    obj = max_g + np.log(exp_g.sum()) / alpha
    return obj, grad

def lbfqsb_optimize(h, n, alpha, maxiter=500):
    lam = 5000.0
    def pobj(x):
        o, _ = smooth_obj_grad(x, n, alpha)
        s = x.sum() - n / 2.0
        return o + lam * s * s
    def pgrad(x):
        _, g = smooth_obj_grad(x, n, alpha)
        s = x.sum() - n / 2.0
        return g + 2 * lam * s
    bounds = [(0.0, 1.0)] * n
    res = minimize(pobj, h, method='L-BFGS-B', jac=pgrad,
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-14})
    return project_feasible(res.x, n)

def greed_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_set = set(int(x) for x in np.where(h == 1)[0])
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_set:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_set.discard(j)
                    ones_set.add(i)
                    zeros_list.remove(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
