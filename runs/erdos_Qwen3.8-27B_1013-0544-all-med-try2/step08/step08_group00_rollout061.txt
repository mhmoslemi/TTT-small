import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def ls_value(x, n, alpha, topk=None):
    c = compute_c5_vec(x, n)
    if topk is not None and topk < len(c):
        # Only consider top-k correlations
        idx = np.argsort(c)[-topk:]
        vals = c[idx]
        m = vals.max()
        sh = alpha * (vals - m)
        smx = sh.max()
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)

def ls_grad_full(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15: continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def ls_grad_topk(x, n, alpha, topk):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w_all = np.exp(sh - smx)
    w_all /= w_all.sum()
    # Zero out weights for non-top-k
    idx_top = np.argsort(c)[-topk:]
    w = np.zeros_like(w_all)
    w[idx_top] = w_all[idx_top]
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15: continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def lbfgs_run(h, n, deadline, alphas=None, topk=None):
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        def obj(x):
            return ls_value(x, n, alpha, topk)
        def grad(x):
            if topk is not None:
                return ls_grad_topk(x, n, alpha, topk)
            return ls_grad_full(x, n, alpha)
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 50000,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_run(h, n, deadline, alpha=5000000, maxiter=5000, topk=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0,
             'jac': lambda x, n=n: np.ones(n)}]
    try:
        res = minimize(lambda x: ls_value(x, n, alpha, topk),
                      h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_pair_search(h, n, deadline, top_k=30, n_rounds=3):
    best_c = compute_c5(h, n)
    for alpha in [1000, 10000, 100000]:
        for rnd in range(n_rounds):
            if time.time() > deadline:
                break
            g = ls_grad_full(h, n, alpha)
            # Find best pair moves
            # Want to increase high-gradient indices and decrease low-gradient indices
            order = np.argsort(g)
            for i in order[:top_k]:
                if time.time() > deadline:
                    break
                for j in order[-top_k:]:
                    if i == j:
                        continue
                    for step in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
                        if h[i] + step <= 1.0 + 1e-12 and h[j] - step >= -1e-12:
                            h[i] += step
                            h[j] -= step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                            else:
                                h[i] -= step
                                h[j] += step
                            break
    return h, best_c

def local_refine(h, n, deadline, radius=80):
    best_c = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
    for d in steps:
        while time.time() < deadline:
            any_imp = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j: continue

    dx = 2.0 / n

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _ls_core(x, n, alpha, topk=None):
    c = compute_c5_vec(x, n)
    if topk is not None and topk < len(c):
        idx = np.argsort(c)[-topk:]
        vals = c[idx]
    else:
        vals = c
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

def _ls_grad(x, n, alpha, topk=None):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    if topk is not None and topk < len(w):
        idx = np.argsort(c)[-topk:]
        w2 = np.zeros_like(w)
        w2[idx] = w[idx]
        w = w2 / w2.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:

        else:

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_ls(x, n, alpha, topk):
    c = compute_c5_vec(x, n)
    if topk is not None and topk < len(c):
        idx = np.argsort(c)[-topk:]
        vals = c[idx]
    else:
        vals = c
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)

def grad_ls(x, n, alpha, topk):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    if topk is not None and topk < len(w):
        idx = np.argsort(c)[-topk:]
        w2 = np.zeros_like(w)
        w2[idx] = w[idx]
        s = w2.sum()
        if s > 1e-30:
            w = w2 / s
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

    if time.time() < deadline - 3:
        h, c = slsqp_opt(best_h, n, deadline - 1, 50000000, None, 2000)
        if c < best_c:
            best_h, best_c = h, c
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
