import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h_init, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        lse = sm + np.log(np.exp(shifted - sm).sum())
        return m + lse / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

def anneal_binary(h_init, n, deadline, seed=42, t0_temp=0.5, t1_temp=0.01):
    """Simulated annealing on binary configurations with precomputed contributions."""
    dx = 2.0 / n
    target = n // 2
    rng = np.random.RandomState(seed)
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    c = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
    cur_val = c.max()
    best_h = h.copy()
    best_val = cur_val
    total_steps = 0
    
    # Precompute contribution vectors for each position
    # contrib[i, m] = 1 - h[i + m - (n-1)] if valid, else 0
    # This changes when h changes, so we recompute periodically
    def get_contrib(pos, h_vec):
        m_lo = max(0, n - 1 - pos)
        m_hi = min(2 * n - 2, n - 1 + pos)
        vec = np.zeros(2 * n - 1)
        vec[m_lo:m_hi+1] = 1.0 - h_vec[n-1 + m_lo - pos : n-1 + m_hi - pos + 1]
        return vec, m_lo, m_hi
    
    steps_per_temp = 200
    n_temps = 30
    for temp_idx in range(n_temps):
        if time.time() > deadline:
            break
        T = t0_temp * (t1_temp / t0_temp) ** (temp_idx / max(n_temps - 1, 1))
        for step in range(steps_per_temp):
            if time.time() > deadline:
                break
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if len(zeros) < 2 or len(ones) < 2:
                break
            a = rng.choice(zeros)
            b = rng.choice(ones)
            if a == b:
                continue
            ca, alo, ahi = get_contrib(a, h)
            cb, blo, bhi = get_contrib(b, h)
            delta = ca - cb
            m_ab = b - a + n - 1
            m_ba = a - b + n - 1
            if 0 <= m_ab < 2 * n - 1:
                delta[m_ab] += 1.0
            if 0 <= m_ba < 2 * n - 1:
                delta[m_ba] -= 1.0
            new_c = c + delta
            new_val = new_c.max()
            dv = new_val - cur_val
            if dv < 0 or (T > 0.005 and rng.random() < np.exp(-dv / (T * cur_val))):
                h[a], h[b] = 1.0, 0.0
                c = new_c
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_h = h.copy()
            total_steps += 1
    return best_h, best_val * dx

def greedy_binary(h_init, n, deadline, seed=42, max_sweeps=500, n_restarts=3):
    """Greedy best-swap binary search with subset sampling for speed."""
    dx = 2.0 / n
    target = n // 2
    rng = np.random.RandomState(seed)
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    c = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
    cur_val = c.max()
    best_h = h.copy()
    best_val = cur_val
    n_lags = 2 * n - 1
    
    def get_contrib(pos, h_vec):
        m_lo = max(0, n - 1 - pos)
        m_hi = min(2 * n - 2, n - 1 + pos)
        vec = np.zeros(n_lags)
        vec[m_lo:m_hi+1] = 1.0 - h_vec[n-1 + m_lo - pos : n-1 + m_hi - pos + 1]
        return vec
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            for _ in range(rng.randint(3, 7)):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    h[rng.choice(oi)], h[rng.choice(zi)] = 0.0, 1.0
            c = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
            cur_val = c.max()
        
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if len(zeros) < 2 or len(ones) < 2:
                break
            improved = False
            # Sample a subset of zeros to check (for speed)
            sample_size = min(len(zeros), 20)
            z_sample = rng.choice(zeros, sample_size, replace=False)
            best_impr = 0
            best_a = best_b = -1
            best_delta = None
            for a in z_sample:
                ca = get_contrib(a, h)
                for b in ones:
                    cb = get_contrib(b, h)
                    delta = ca - cb
                    m_ab = b - a + n - 1
                    m_ba = a - b + n - 1
                    if 0 <= m_ab < n_lags:
                        delta[m_ab] += 1.0
                    if 0 <= m_ba < n_lags:
                        delta[m_ba] -= 1.0
                    new_max = (c + delta).max()
                    impr = cur_val - new_max
                    if impr > best_impr:
                        best_impr = impr
                        best_a, best_b = int(a), int(b)
                        best_delta = delta
            if best_a >= 0 and best_impr > 1e-10:
                h[best_a], h[best_b] = 1.0, 0.0
                c += best_delta
                cur_val = c.max()
                improved = True
                if cur_val < best_val:
                    best_val = cur_val
                    best_h = h.copy()
            if not improved:
                break
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
    return best_h, best_val * dx
