    alphas = [200, 1000, 5000, 15000, 40000]
    for alpha in alphas:
        if time.time() > deadline - 200:
            break
        max_iter = 400 if alpha < 5000 else 200
        try:
            res = minimize(smooth_val, best_h, jac=smooth_grad, args=(best_n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14})
            h_new = project_feasible(res.x, best_n)
            c_new = compute_c5(h_new, best_n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    # Phase 2: Damped FISTA
    if time.time() < deadline - 150:
        t1 = min(time.time() + 100, deadline - 80)
        best_h, best_c = damped_fista(best_h, best_n, t1, alpha=5000, top_k=50)

    # Phase 3: Focus on active set
    if time.time() < deadline - 60:
        best_h, best_c = focus_optimize(best_h, best_n, min(time.time() + 40, deadline - 30),
                                        alpha=20000, top_k=8)

    # Phase 4: Multi-start with perturbations
    n_restarts = 6
    remaining = deadline - time.time()
    per_restart = remaining * 0.5 / n_restarts
    if per_restart > 15:
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.03 + 0.02 * r
            pert = rng.randn(best_n) * scale
            h_pert = project_feasible(best_h + pert, best_n)
            c_pert = compute_c5(h_pert, best_n)
            t_end = min(time.time() + per_restart * 0.7, deadline - 35)
            try:
                res = minimize(smooth_val, h_pert, jac=smooth_grad, args=(best_n, 8000),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 250, 'ftol': 1e-17})
                h_new = project_feasible(res.x, best_n)
                c_new = compute_c5(h_new, best_n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
            if time.time() < t_end:
                h_f, c_f = damped_fista(h_new if c_new < best_c else h_pert, best_n,
                                        t_end, alpha=10000, top_k=40, init_step=0.015)
                if c_f < best_c:
                    best_c = c_f
                    best_h = h_f

    # Phase 5: Try smaller n (coarser grid, might find better local opt)
    for n_try in [350, 450]:
        if time.time() > deadline - 40:
            break
        if n_try == best_n:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        bounds_n = [(0.0, 1.0)] * n_try
        try:
            res = minimize(smooth_val, h_interp, jac=smooth_grad, args=(n_try, 8000),
                          method='L-BFGS-B', bounds=bounds_n,
                          options={'maxiter': 300, 'ftol': 1e-17})
            h_t = project_feasible(res.x, n_try)
            c_t = compute_c5(h_t, n_try)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try
        except Exception:
            pass

    # Phase 6: Final focus + FISTA polish
    if time.time() < deadline - 5:
        best_h, best_c = focus_optimize(best_h, best_n, min(time.time() + 15, deadline - 3),
                                        alpha=30000, top_k=6)
        if time.time() < deadline - 3:
            best_h, best_c = damped_fista(best_h, best_n, deadline - 2,
                                         alpha=20000, top_k=30, init_step=0.008)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
