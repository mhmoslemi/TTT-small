def compute_c5(h, n):

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    shifted = alpha * (c - c.max())
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:

        k = m - (n - 1)

            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

            return smooth_max_grad(x, _n, _a)

                         bounds=bounds,

                h = h_new
        except Exception:
            pass
    return h, best_c

            g = smooth_max_grad(h, n, 100000)

Actually, a better idea: **iterative relinearization**. 

The function I(k) = ∑_i h[i]·(1-h[i+k]) is bilinear in h. For a fixed set of "active" shifts K, we want to minimize max_{k∈K} I(k). 

We can use the following alternating approach:
1. Fix the active set K (top m shifts)
2. Linearize: for each k∈K, I(k) ≈ I(k)_current + ∇I(k)·(h - h_current)
3. Solve the resulting LP: minimize t subject to I(k)_current + ∇I(k)·(h - h_current) ≤ t for all k∈K, 0≤h≤1, ∑h = n/2

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    shifted = alpha * (c - c.max())
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    # Vectorized gradient
    k_vals = np.arange(-(n-1), n)
    # For each shift k, the gradient contribution is:
    # g += w[k] * dx * [(1-h) shifted - h shifted] (appropriate shifts)
    # Actually let me use the loop version for clarity
    grad = np.zeros(n)
    for m in range(len(w)):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def lbfgs_optimize(h, n, deadline, alphas, max_iters=300):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 8:
            break
        if deadline - time.time() < 5:
            break
        _n, _a = n, alpha
        def obj(x):
            return smooth_max_val(x, _n, _a)
        def grad(x):
            return smooth_max_grad(x, _n, _a)
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': max_iters * 50,
                                   'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def coord_descent(h, n, deadline, max_sweeps=200, n_cand=20):
    """Exact coordinate descent using the linearity of c[m] in each h[i]."""
    best_c = compute_c5(h, n)
    if n_cand < 4:
        n_cand = 4
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        # Recompute c from scratch every sweep to avoid drift
        c = compute_c5_vec(h, n).copy()
        current_max = c.max()
        
        any_improved = False
        k_vals = np.arange(-(n-1), n)
        
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            
            idx_plus = i + k_vals
            idx_minus = i - k_vals
            mask_plus = (idx_plus >= 0) & (idx_plus < n)
            mask_minus = (idx_minus >= 0) & (idx_minus < n)
            
            slopes = np.zeros(2*n-1)
            slopes[mask_plus] += (1.0 - h[idx_plus[mask_plus]])
            slopes[mask_minus] -= h[idx_minus[mask_minus]]
            
            # f(u) = max_m (slopes[m]*u + c[m]), u in [-h[i], 1-h[i]]
            lo_u = -h[i]
            hi_u = 1.0 - h[i]
            span = hi_u - lo_u
            if span < 1e-12:
                continue
                
            candidates = np.linspace(lo_u, hi_u, n_cand)
            # f(u) for all candidates: max over m of (slopes[m]*u + c[m])
            # Shape: (2n-1, n_cand)
            vals = np.max(slopes[:, None] * candidates[None, :] + c[None, :], axis=0)
            best_idx = np.argmin(vals)
            best_u = candidates[best_idx]
            best_val = vals[best_idx]
            
            if best_val < current_max - 1e-14 and abs(best_u) > 1e-12:
                h[i] += best_u
                c += slopes * best_u
                current_max = c.max()
                any_improved = True
        
        if not any_improved:
            break
        actual_c = compute_c5(h, n)
        if actual_c < best_c:
            best_c = actual_c
            c = compute_c5_vec(h, n)  # sync
    
    return h, best_c

def targeted_moves(h, n, deadline, top_m=10, steps=(0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002)):

            best_c = actual
    return h, best_c

def targeted_moves(h, n, deadline, top_m=8, steps=(0.01, 0.005, 0.002, 0.001, 0.0005)):
    best_c = compute_c5(h, n)
    k_vals = np.arange(-(n - 1), n)
    for _ in range(300):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n).copy()
        cm = c.max()
        top_idx = np.argsort(c)[::-1][:top_m]
        improved = False
        for mi in top_idx:
            if time.time() > deadline:
                return h, best_c
            k = mi - (n - 1)
            if k >= 0:
                nv = n - k
                ov = h[:nv] * (1.0 - h[k:])
                src = int(np.argmax(ov))
                dst = src + k
            else:
                kk = -k
                nv = n - kk
                ov = h[kk:] * (1.0 - h[:nv])
                src = int(np.argmax(ov)) + kk
                dst = src - kk
            for step in steps:
                if h[src] >= step and h[dst] <= 1.0 - step:
                    sp_s = np.zeros(2 * n - 1)
                    ip = src + k_vals
                    im = src - k_vals
                    mp2 = (ip >= 0) & (ip < n)
                    mm2 = (im >= 0) & (im < n)
                    sp_s[mp2] += (1.0 - h[ip[mp2]])
                    sp_s[mm2] -= h[im[mm2]]
                    sp_d = np.zeros(2 * n - 1)
                    ip = dst + k_vals
                    im = dst - k_vals
                    mp3 = (ip >= 0) & (ip < n)
                    mm3 = (im >= 0) & (im < n)
                    sp_d[mp3] += (1.0 - h[ip[mp3]])
                    sp_d[mm3] -= h[im[mm3]]
                    c_new = c + step * (sp_d - sp_s)
                    if c_new.max() < cm - 1e-14:
                        h[src] -= step
                        h[dst] += step
                        c = c_new
                        cm = c.max()
                        improved = True
                        break
        if improved:
            actual = compute_c5(h, n)
            if actual < best_c:
                best_c = actual
        else:
            break
    return h, best_c

def golden_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(half):
        h[int(np.floor((i + 1) * phi)) % n] = 1.0
    return project_feasible(h, n)

def random_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.arange(n)
    for freq in range(1, 10):
        amp = rng.uniform(0.05, 0.25)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init
    # Phase 1: L-BFGS with increasing alpha
    t1 = min(t0 + 180, deadline - 300)
    best_h, best_c = lbfgs_optimize(best_h, best_n, t1,
        [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000], 200)
    # Phase 2: Coordinate descent
    if time.time() < deadline - 200:
        t2 = min(time.time() + 100, deadline - 200)
        best_h, best_c = coord_descent(best_h, best_n, t2, 250, 28)
    # Phase 3: High-alpha L-BFGS
    if time.time() < deadline - 120:
        t3 = min(time.time() + 60, deadline - 120)
        best_h, best_c = lbfgs_optimize(best_h, best_n, t3,
            [500000, 2000000, 10000000, 50000000], 150)
    # Phase 4: Coordinate descent again
    if time.time() < deadline - 80:
        t4 = min(time.time() + 50, deadline - 80)
        best_h, best_c = coord_descent(best_h, best_n, t4, 200, 32)
    # Phase 5: Targeted moves
    if time.time() < deadline - 50:
        t5 = min(time.time() + 30, deadline - 50)
        best_h, best_c = targeted_moves(best_h, best_n, t5, 10)
    # Phase 6: Try higher n
    for n_try in [700, 1000, 1400]:
        if time.time() > deadline - 180:
            break
        alloc = deadline - time.time() - 60
        if alloc < 50:
            break
        xo = np.linspace(0, 2, best_n, endpoint=False)
        xn = np.linspace(0, 2, n_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, n_try)
        t6 = time.time() + alloc * 0.45
        h6, c6 = lbfgs_optimize(hi, n_try, t6, [500, 5000, 50000, 500000], 120)
        if time.time() < deadline - 30:
            t6b = min(time.time() + 25, deadline - 20)
            h6, c6 = coord_descent(h6, n_try, t6b, 80, 20)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = n_try
    # Phase 7: Diverse restarts
    if time.time() < deadline - 50:
        starts = [golden_start(best_n), random_start(best_n, rng), fourier_start(best_n, rng)]
        for s in starts:
            if time.time() > deadline - 15:
                break
            alloc = min(30, (deadline - time.time() - 10) * 0.5)
            if alloc < 8:
                break
            hs, cs = lbfgs_optimize(s, best_n, time.time() + alloc * 0.6,
                [500, 5000, 50000, 500000], 100)
            if cs < best_c:
                best_c = cs
                best_h = hs
    # Phase 8: Final polish
    if time.time() < deadline - 8:
        t8 = min(time.time() + 25, deadline - 4)
        best_h, best_c = coord_descent(best_h, best_n, t8, 100, 32)
    if time.time() < deadline - 1:
        best_h, best_c = targeted_moves(best_h, best_n, deadline - 0.5, 6,
            (0.001, 0.0005, 0.0002, 0.0001))
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
