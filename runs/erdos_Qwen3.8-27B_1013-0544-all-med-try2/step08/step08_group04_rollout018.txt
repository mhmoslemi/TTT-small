        best_h, best_c = pair_search(best_h, n, deadline - 2)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 200
        h0 = np.full(n0, 0.5)

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)

    # Phase 1: Optimize on original n
    t1 = min(t0 + 300, deadline - 400)
    best_h, best_c = optimize_block(best_h, n0, t1)

    # Phase 2: Try smaller n (sometimes coarser is easier to optimize)
    for n_try in [200, 150]:
        if time.time() > deadline - 200:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        hi = np.interp(x_new, x_old, best_h)
        hi = project_feasible(hi, n_try)
        t_alloc = min(120, deadline - time.time() - 100)
        if t_alloc < 30:
            break
        h_t, c_t = optimize_block(hi, n_try, time.time() + t_alloc)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t

    # Phase 3: Try larger n
    for n_try in [400, 500]:
        if time.time() > deadline - 150:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        hi = np.interp(x_new, x_old, best_h)
        hi = project_feasible(hi, n_try)
        t_alloc = min(120, deadline - time.time() - 80)
        if t_alloc < 30:
            break
        h_t, c_t = slsqp_opt(hi, n_try, time.time() + t_alloc * 0.7, 200000, maxiter=1500)
        if time.time() < deadline - 40:
            h_t2, c_t2 = coord_descent(h_t, n_try, min(time.time() + 40, deadline - 20))
            if c_t2 < c_t:
                c_t = c_t2; h_t = h_t2
        if c_t < best_c:
            best_c = c_t
            best_h = h_t

    # Phase 4: Perturb and re-optimize on best n
    nb = len(best_h)
    if time.time() < deadline - 80:
        for _ in range(3):
            if time.time() > deadline - 40:
                break
            h_pert = best_h + rng.randn(nb) * 0.02
            h_pert = project_feasible(h_pert, nb)
            t_alloc = min(50, deadline - time.time() - 15)
            if t_alloc < 15:
                break
            h_t, c_t = slsqp_opt(h_pert, nb, time.time() + t_alloc, 500000, maxiter=1000)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t

    # Phase 5: Final intensive local search
    if time.time() < deadline - 10:
        best_h, best_c = coord_descent(best_h, len(best_h), deadline - 10)
    if time.time() < deadline - 3:
        best_h, best_c = pair_search(best_h, len(best_h), deadline - 2,
                                     deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
