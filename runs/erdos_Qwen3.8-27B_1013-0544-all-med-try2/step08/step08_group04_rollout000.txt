import time
import numpy as np
from scipy.optimize import minimize

_LAG = {}

def _lag_info(n):
    if n not in _LAG:
        m = 2 * n - 1
        k = np.arange(m) - (n - 1)
        _LAG[n] = (k, n - np.abs(k))
    return _LAG[n]

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _c5(h):
    return float(np.max(_corr(h)))

def _proj(a):
    a = np.clip(np.asarray(a, dtype=float), 0.0, 1.0)
    n = len(a)
    t = 0.5 * n
    if t <= 0.0:
        return np.zeros(n)
    if t >= float(n):
        return np.ones(n)
    lo = -a.max() - 1e-14
    hi = 1.0 - a.min() + 1e-14
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(a + m, 0.0, 1.0)) < t:
            lo = m
        else:
            hi = m
    h = np.clip(a + 0.5 * (lo + hi), 0.0, 1.0)
    d = t - h.sum()
    if d > 0.0:
        for idx in np.argsort(1.0 - h)[::-1]:
            if d <= 1e-14:
                break
            add = min(1.0 - h[idx], d)
            h[idx] += add
            d -= add
    elif d < 0.0:
        for idx in np.argsort(h):
            if d >= -1e-14:
                break
            sub = min(h[idx], -d)
            h[idx] -= sub
            d += sub
    return h

def _smooth_val(x, alpha):
    c = _corr(x)
    cm = float(c.max())
    z = alpha * (c - cm)
    s = float(np.sum(np.exp(z)))
    if s <= 0.0:
        return cm
    return cm + np.log(s) / alpha

def _smooth_grad(x, alpha, mu=0.0):
    n = len(x)
    c = _corr(x)
    cm = float(c.max())
    z = alpha * (c - cm)
    w = np.exp(z)
    s = float(np.sum(w))
    if s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    g = np.zeros(n)
    if mu != 0.0:
        g += 2.0 * mu * (float(np.sum(x)) - 0.5 * n)
    kk, ll = _lag_info(n)
    dx = 2.0 / n
    for i in range(len(w)):
        wi = float(w[i])
        if wi <= 1e-14:
            continue
        k = int(kk[i])
        l = int(ll[i])
        wmi = wi * dx
        if k >= 0:
            g[k:k + l] += wmi * (1.0 - x[:l])
            g[:l] -= wmi * x[k:k + l]
        else:
            kkp = -k
            g[:l] += wmi * (1.0 - x[kkp:kkp + l])
            g[kkp:kkp + l] -= wmi * x[:l]
    return g

def _lb_obj(x, alpha, mu):
    return _smooth_val(x, alpha) + mu * (float(np.sum(x)) - 0.5 * len(x)) ** 2

def _lb_grad(x, alpha, mu):
    return _smooth_grad(x, alpha, mu)

def _sl_obj(x, alpha):
    return _smooth_val(x, alpha)

def _sl_grad(x, alpha):
    return _smooth_grad(x, alpha, 0.0)

def _eqf(x, t):
    return float(np.sum(x) - t)

def _eqj(x, t):
    return np.ones(len(x), dtype=float)

def _lbfgs(h, deadline):
    h = _proj(h)
    best_h = h.copy()
    best_c = _c5(h)
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    if n <= 450:

        except Exception:
            pass
    return best_h, best_c

def _slsqp(h, alpha, deadline, maxiter):
    h = _proj(h)
    n = len(h)
    t = 0.5 * n
    bounds = [(0.0, 1.0)] * n
    try:

                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eqf, 'jac': _eqj, 'args': (t,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if np.all(np.isfinite(res.x)):
            h2 = _proj(res.x)
            return h2, _c5(h2)
    except Exception:
        pass
    return h, _c5(h)
