                    g[b] = 0
                c = np.correlate(h, g, mode='full').astype(float)
                # Don't update best_h, just continue searching
                ones = np.where(h == 1)[0].astype(int)
                zeros = np.where(h == 0)[0].astype(int)
                stuck_count = 0
    
    return best_h, best_c * dx
```

This is O(n * n_try) per round (for the delta computation), which for n=500, n_try=3000 is 1.5 million operations per round. Each round should take about 1-2ms in numpy. So we can do many rounds per second.

Actually wait, the delta computation involves 4 array operations on arrays of length ~n, so it's O(n) per candidate. With n_try=3000, that's 3000*n = 1.5M per round. In numpy, this should be very fast.

But there's a problem: recomputing c = np.correlate(h, g, mode='full') after each accepted move is O(n²) = 250000 operations. That's fine, it happens at most once per round.

Actually, I can avoid the full recompute by accumulating the delta:
c = c + delta (for the accepted move)

But floating point errors will accumulate. Better to recompute every, say, 100 accepted moves.

Let me also think about the "targeted" approach: instead of random swaps, focus on swaps that are likely to reduce the current maximum.

If the current max is at index m*, then the shifts that are "close" to max are the ones we want to reduce. A swap (a, b) reduces c[m*] if:
- The delta at m* is negative

From the formula:
delta[m*] = -g[a+m*-nm1] (if m* in [nm1-a, 2n-2-a]) + h[a-m*+nm1] (if m* in [a, a+nm1]) + g[b+m*-nm1] (if m* in [nm1-b, 2n-2-b]) - h[b-m*+nm1] (if m* in [b, b+nm1])

For this to be negative, we want the negative terms to dominate.

The -g[a+m*-nm1] term is negative when g[a+m*-nm1] > 0, i.e., h[a+m*-nm1] < 1.
The -h[b-m*+nm1] term is negative when h[b-m*+nm1] > 0.

So if we pick a such that h[a+m*-nm1] = 0 (i.e., a+m* = nm1, or a = nm1-m*) and h[a]=1, then the first term is -1.
And if we pick b such that h[b-m*+nm1] = 1 (i.e., b-m* = nm1, or b = nm1+m*) and h[b]=0... wait, that doesn't work because we need h[b]=0 for it to be a valid swap target.

Hmm, let me think differently. The "targeted" approach is:
1. Find the index m* where c is maximum
2. Find all positions a where h[a]=1 and are "responsible" for c[m*] (i.e., a+m* is a valid index where h=0, or a-m* is a valid index where h=1... wait)

Actually, for the full correlation c[m*] = sum_i h[i]*g[i+m*-nm1]:
- The contributing positions are those i where h[i]=1 and g[i+m*-nm1]=1, i.e., h[i]=1 and h[i+m*-nm1]=0.

To reduce c[m*], we want to:
- Change one of these h[i] from 1 to 0 (by swapping with a zero elsewhere)
- Or change one of the h[i+m*-nm1] from 0 to 1 (by swapping with a one elsewhere)

The most direct swap: pick i where h[i]=1, h[i+m*-nm1]=0, and swap h[i] with h[i+m*-nm1]. This directly reduces c[m*] by 1.

But we also need to check that this doesn't increase other correlations too much.

Let me code up a "targeted best-improvement" approach:

```python
def targeted_binary_search(h, n, deadline, rng):
    g = 1 - h
    c = np.correlate(h, g, mode='full').astype(float)
    best_c = c.max()
    best_h = h.copy()
    nm1 = n - 1
    dx = 2.0 / n
    
    iterations = 0
    while time.time() < deadline:
        iterations += 1
        m_star = int(c.argmax())
        
        # Find positions contributing to c[m_star]
        # c[m_star] = sum_i h[i]*g[i + m_star - nm1]
        k = m_star - nm1  # the shift
        if k >= 0:
            # i ranges from 0 to n-1-k
            contrib_ones = np.where((h[:n-k] == 1) & (h[k:] == 0))[0]
        else:
            t = -k
            # c[m_star] = sum_i h[i+t]*g[i], i from 0 to n-1-t
            # = sum_i h[i+t]*(1-h[i])
            # Contributing: h[i+t]=1, h[i]=0
            contrib_ones = np.where((h[t:] == 1) & (h[:n-t] == 0))[0] + t
        
        if len(contrib_ones) == 0:
            break
        
        # For each contributor, try swapping it with a nearby zero
        # (or with h[i+k] directly)
        best_new_max = c.max()
        best_swap = None
        
        # Strategy: for the top contributors, try the "direct" swap
        # (swap h[i] with h[i+k] if k>0, or h[i] with h[i-k] if k<0)
        n_candidates = min(len(contrib_ones), 200)
        sample = rng.choice(contrib_ones, size=n_candidates, replace=False) if len(contrib_ones) > n_candidates else contrib_ones
        
        for a in sample:
            if k >= 0:
                b = a + k  # h[a]=1, h[b]=0
            else:
                b = a + k  # k<0, so b = a-k... 
                # Actually if k<0, then h[i+t]=1, h[i]=0, and a=i+t, so the zero is at a+t = a-k... 
                # Wait: k = -t, so b = a + k = a - t. And h[b] = h[a-t] = 0. Yes.
                b = a + k
            
            if b < 0 or b >= n or h[b] != 0:
                continue
            
            # Compute delta for swap (a: 1->0, b: 0->1)
            delta = np.zeros(2*n-1)
            
            m_lo = nm1 - a
            m_hi = 2*n-2-a
            delta[m_lo:m_hi+1] -= g[a + np.arange(m_lo, m_hi+1) - nm1]
            
            m_lo = a
            m_hi = a + nm1
            delta[m_lo:m_hi+1] += h[a - np.arange(m_lo, m_hi+1) + nm1]
            
            m_lo = nm1 - b
            m_hi = 2*n-2-b
            delta[m_lo:m_hi+1] += g[b + np.arange(m_lo, m_hi+1) - nm1]
            
            m_lo = b
            m_hi = b + nm1
            delta[m_lo:m_hi+1] -= h[b - np.arange(m_lo, m_hi+1) + nm1]
            
            new_max = (c + delta).max()
            if new_max < best_new_max - 1e-15:
                best_new_max = new_max
                best_swap = (a, b)
        
        if best_swap is not None:
            a, b = best_swap
            h[a] = 0
            h[b] = 1
            g[a] = 1
            g[b] = 0
            c = np.correlate(h, g, mode='full').astype(float)
            if c.max() < best_c - 1e-15:
                best_c = c.max()
                best_h = h.copy()
        else:
            # No improvement found, do a random perturbation
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            n_flip = rng.randint(2, 6)
            for _ in range(n_flip):
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h[a] = 0
                h[b] = 1
                g[a] = 1
                g[b] = 0
            c = np.correlate(h, g
