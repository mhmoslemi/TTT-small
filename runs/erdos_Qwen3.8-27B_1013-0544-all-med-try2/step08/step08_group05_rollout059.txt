def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def swap_delta(h, n, i, j):
    """Compute delta in correlation vector when flipping h[i]: 0->1, h[j]: 1->0"""
    delta = np.zeros(2 * n - 1)
    # For positive shift k (index n-1+k), k = 0..n-1
    for k in range(n):
        d = 0.0
        if i + k <= n - 1:
            d += (1.0 - h[i + k])
        if j + k <= n - 1:
            d -= (1.0 - h[j + k])
        if i >= k:
            d -= h[i - k]
        if j >= k:
            d += h[j - k]
        delta[n - 1 + k] = d
    # For negative shift k (index n-1-k), k = 1..n-1
    for k in range(1, n):
        d = 0.0
        if i >= k:

        if j >= k:
            d -= (1.0 - h[j - k])
        if i + k <= n - 1:
            d -= h[i + k]
        if j + k <= n - 1:
            d += h[j + k]
        delta[n - 1 - k] = d
    return delta

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            frac = min(1.0, diff / s)
            h = h + frac * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            frac = min(1.0, -diff / s)
            h = h - frac * h
    return np.clip(h, 0.0, 1.0)

    M1 = np.zeros((n, n))
    M2 = np.zeros((n, n))

    cur_c = corr.max() * dx

            # corr stays the same

        g = np.correlate(x, 1 - x, mode='full') * dx

        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0

                       constraints=cons,

            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)[:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def build_matrices(h, n):
    M1 = np.zeros((n, n))
    M2 = np.zeros((n, n))
    M3 = np.zeros((n, max(n - 1, 1)))
    M4 = np.zeros((n, max(n - 1, 1)))
    for j in range(n):
        M1[j, :n - j] = 1.0 - h[j:]
        M2[j, :j + 1] = h[j::-1]
        if n > 1:
            if j >= 1:
                M3[j, :j] = 1.0 - h[:j][::-1]
            if j < n - 1:
                M4[j, :n - 1 - j] = h[j + 1:]
    return M1, M2, M3, M4

def one_pass(h, corr, ones, zeros, M1, M2, M3, M4, n, rng, mode='best'):
    cur_max = corr.max()
    best_score = cur_max
    best_swap = None
    if len(ones) == 0 or len(zeros) == 0:
        return h, corr, False
    if mode == 'best':
        for i in zeros:
            A_pos = np.zeros(n)
            A_pos[:n - i] = 1.0 - h[i:]
            A_pos[:i + 1] -= h[i::-1]
            A_neg = np.zeros(n - 1)
            if i >= 1:
                A_neg[:i] = 1.0 - h[:i][::-1]
            if i < n - 1:
                A_neg[:n - 1 - i] -= h[i + 1:]
            bp = corr[n - 1:] + A_pos
            bn = corr[:n - 1][::-1] + A_neg
            sp = (bp[None, :] - M1[ones] + M2[ones]).max(axis=1)
            sn = (bn[None, :] - M3[ones] + M4[ones]).max(axis=1)
            scores = np.maximum(sp, sn)
            bi = scores.argmin()
            if scores[bi] < best_score:
                best_score = scores[bi]
                best_swap = (i, ones[bi])
    else:
        z_ord = zeros[rng.permutation(len(zeros))]
        for i in z_ord:
            A_pos = np.zeros(n)
            A_pos[:n - i] = 1.0 - h[i:]
            A_pos[:i + 1] -= h[i::-1]
            A_neg = np.zeros(n - 1)
            if i >= 1:
                A_neg[:i] = 1.0 - h[:i][::-1]
            if i < n - 1:
                A_neg[:n - 1 - i] -= h[i + 1:]
            bp = corr[n - 1:] + A_pos
            bn = corr[:n - 1][::-1] + A_neg
            sp = (bp[None, :] - M1[ones] + M2[ones]).max(axis=1)
            sn = (bn[None, :] - M3[ones] + M4[ones]).max(axis=1)
            scores = np.maximum(sp, sn)
            bi = scores.argmin()
            if scores[bi] < cur_max - 1e-12:
                best_swap = (i, ones[bi])
                best_score = scores[bi]
                break
    if best_swap is not None:
        i, j = best_swap
        h2 = h.copy()
        h2[i] = 1.0
        h2[j] = 0.0
        corr2 = np.correlate(h2, 1.0 - h2, mode='full').astype(float)
        return h2, corr2, True
    return h, corr, False

def binary_search(h_bin, n, deadline, rng, mode='best'):
    h = h_bin.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    M1, M2, M3, M4 = build_matrices(h, n)
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        h, corr, imp = one_pass(h, corr, ones, zeros, M1, M2, M3, M4, n, rng, mode)
        if not imp:
            break
        M1, M2, M3, M4 = build_matrices(h, n)
    return h, corr.max() * dx

def anneal_search(h_bin, n, deadline, rng, T0=2.0, T_min=0.05, cool=0.9995):
    h = h_bin.copy()
    dx = 2.0 / n
    corr_max = np.correlate(h, 1.0 - h, mode='full').max()
    best_h = h.copy()
    best_c = corr_max
    T = T0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(zeros))
        j = int(rng.choice(ones))
        h[i], h[j] = 1.0, 0.0
        nm = np.correlate(h, 1.0 - h, mode='full').max()
        if nm < corr_max or (T > T_min and rng.random() < np.exp(-(nm - corr_max) / T)):
            corr_max = nm
            if nm < best_c:
                best_c = nm
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
        T *= cool
    return best_h, best_c * dx

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n_best = n0
    # Phase 1: binary local search on initial n
    p1 = t0 + 0.18 * (budget_s - 10)
    hb = make_binary(best_h, n0)
    hb, cb = binary_search(hb, n0, p1, rng, 'best')
    if cb < best_c:
        best_c, best_h, n_best = cb, hb.copy(), n0
    # Phase 2: multi-n exploration
    p2 = t0 + 0.45 * (budget_s - 10)
    for nh in [200, 300, 400, 150]:
        if time.time() > p2 - 40:
            break
        if nh == n_best:
            continue
        xo = np.linspace(0, 2, n_best)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hb = make_binary(hh, nh)
        sub_dl = min(time.time() + 80, p2 - 30)
        hb, cb = binary_search(hb, nh, sub_dl, rng, 'best')
        if cb < best_c:
            best_c, best_h, n_best = cb, hb.copy(), nh
    # Phase 3: SLSQP polish
    p3 = t0 + 0.55 * (budget_s - 10)
    if time.time() < p3:
        hp = project_feasible(best_h + 0.03 * rng.randn(n_best), n_best)
        for alpha in [100000, 500000, 2000000]:
            if time.time() > p3 - 10:
                break
            hp = slsqp_smooth(hp, n_best, alpha, 2000, p3 - 5)
            c = compute_c5(hp, n_best)
            if c < best_c:
                best_c, best_h = c, hp.copy()
    # Phase 4: iterated local search with annealing kicks
    p4 = t0 + 0.82 * (budget_s - 10)
    n_iter = 0
    while time.time() < p4 and n_iter < 50:
        n_iter += 1
        hb = make_binary(best_h, n_best)
        nb = nb = np.correlate(hb, 1.0 - hb, mode='full').max() * (2.0 / n_best)
        kick = rng.randint(2, 8)
        for _ in range(kick):
            ones = np.where(hb == 1)[0]
            zeros = np.where(hb == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                a = int(rng.choice(ones))
                b = int(rng.choice(zeros))
                hb[a], hb[b] = 0.0, 1.0
        kick_dl = min(time.time() + 30, p4 - 10)
        hb2, cb2 = binary_search(hb, n_best, kick_dl, rng, 'best')
        if cb2 < best_c:
            best_c, best_h = cb2, hb2.copy()
        if time.time() < p4 - 20:
            hb3 = make_binary(best_h, n_best)
            hb3, cb3 = anneal_search(hb3, n_best, min(time.time() + 20, p4 - 15), rng, T0=1.5, T_min=0.02, cool=0.9998)
            if cb3 < best_c:
                best_c, best_h = cb3, hb3.copy()
                hb3, cb3b = binary_search(hb3, n_best, min(time.time() + 20, p4 - 10), rng, 'best')
                if cb3b < best_c:
                    best_c, best_h = cb3b, hb3.copy()
    # Phase 5: final extended binary search
    p5 = dl - 25
    if time.time() < p5:
        hb = make_binary(best_h, n_best)
        hb, cb = binary_search(hb, n_best, p5, rng, 'best')
        if cb < best_c:
            best_c, best_h = cb, hb.copy()
    # Phase 6: final SLSQP polish
    if time.time() < dl - 8:
        hp = project_feasible(best_h + 0.01 * rng.randn(n_best), n_best)
        hp = slsqp_smooth(hp, n_best, 10000000, 5000, dl - 5)
        c = compute_c5(hp, n_best)
        if c < best_c:
            best_c, best_h = c, hp.copy()
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    return best_h, best_c, n_best
