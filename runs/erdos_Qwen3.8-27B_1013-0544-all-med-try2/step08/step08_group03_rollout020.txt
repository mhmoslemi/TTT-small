    """Compute change in all c_k (as array of length 2n-1) when h[di] += d and h[dj] -= d"""
    delta = np.zeros(2*n-1)
    dx = 2.0 / n
    
    # For di gaining d:
    # As first element h[di] in c_k: valid when 0<=di+k<n → k in [-di, n-di-1]
    ks1 = np.arange(-di, n-di)
    delta[ks1 + n-1] += d * (1 - h[di + ks1])
    
    # As second element: h[di-k] in c_k: valid when 0<=di-k<n → k in [di-(n-1), di]
    ks2 = np.arange(di-(n-1), di+1)
    delta[ks2 + n-1] -= d * h[di - ks2]
    
    # For dj losing d (multiply by -d):
    ks1 = np.arange(-dj, n-dj)
    delta[ks1 + n-1] -= d * (1 - h[dj + ks1])
    
    ks2 = np.arange(dj-(n-1), dj+1)
    delta[ks2 + n-1] += d * h[dj - ks2]
    
    return delta * dx

import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_obj(x, n, alpha, top_k):
    c = compute_c5_vec(x, n)
    m = c.max()
    idx = np.argsort(c)[::-1][:top_k]
    ct = c[idx]
    mt = ct.max()
    sh = alpha * (ct - mt)
    mx = sh.max()
    val = mt + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 50.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, top_k):
    c = compute_c5_vec(x, n)
    m = c.max()
    idx = np.argsort(c)[::-1][:top_k]
    ct = c[idx]
    mt = ct.max()
    sh = alpha * (ct - mt)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    n_pts = len(x)
    grad = np.zeros(n_pts)
    for i in range(top_k):
        k = idx[i] - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    s = x.sum() - n / 2.0
    grad += 100.0 * s
    return grad

def fast_delta(h, n, di, dj, d):
    """Fast computation of delta in c when h[di]+=d, h[dj]-=d"""
    delta = np.zeros(2 * n - 1)
    # di gains d
    ks = np.arange(-di, n - di)
    delta[ks + n - 1] += d * (1.0 - h[di + ks])
    ks = np.arange(di - (n - 1), di + 1)
    delta[ks + n - 1] -= d * h[di - ks]
    # dj loses d
    ks = np.arange(-dj, n - dj)
    delta[ks + n - 1] -= d * (1.0 - h[dj + ks])
    ks = np.arange(dj - (n - 1), dj + 1)
    delta[ks + n - 1] += d * h[dj - ks]
    return delta * (2.0 / n)

def local_search_fast(h, n, deadline, radius=100):
    c = compute_c5_vec(h, n)
    best_c = float(c.max())
    h = h.copy()
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        delta = fast_delta(h, n, i, j, d)
                        new_max = (c + delta).max()
                        if new_max < best_c - 1e-14:
                            c = c + delta
                            h[i] += d
                            h[j] -= d
                            best_c = float(new_max)
                            improved = True
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        delta = fast_delta(h, n, j, i, d)
                        new_max = (c + delta).max()
                        if new_max < best_c - 1e-14:
                            c = c + delta
                            h[i] -= d
                            h[j] += d
                            best_c = float(new_max)
                            improved = True
    return h, best_c

def pgd_search(h, n, deadline, alpha=200, top_k=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    step = 0.02

        if time.time() > deadline:

                if c_new < best_c - 1e-15:
                    best_c = float(c_new)
                    best_h = h_new.copy()
                accepted = True
                break
            s *= 0.5
        if not accepted:
            break
        if it % 200 == 199:
            alpha = max(50, alpha * 0.85)
    return best_h, best_c

    h = np.zeros(n)

    positions = np.floor(np.arange(half) * phi).astype(int) % n

import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_obj(x, n, alpha, top_k):
    c = compute_c5_vec(x, n)
    idx = np.argsort(c)[::-1][:top_k]
    ct = c[idx]
    mt = ct.max()
    sh = alpha * (ct - mt)
    mx = sh.max()
    val = mt + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 50.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, top_k):
    c = compute_c5_vec(x, n)
    idx = np.argsort(c)[::-1][:top_k]
    ct = c[idx]
    mt = ct.max()
    sh = alpha * (ct - mt)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    n_pts = len(x)
    grad = np.zeros(n_pts)
    for i in range(top_k):
        if w[i] < 1e-14:
            continue
        k = idx[i] - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    s = x.sum() - n / 2.0
    grad += 100.0 * s
    return grad

def fast_delta(h, n, di, dj, d):
    delta = np.zeros(2 * n - 1)
    ks = np.arange(-di, n - di)
    delta[ks + n - 1] += d * (1.0 - h[di + ks])
    ks = np.arange(di - (n - 1), di + 1)
    delta[ks + n - 1] -= d * h[di - ks]
    ks = np.arange(-dj, n - dj)
    delta[ks + n - 1] -= d * (1.0 - h[dj + ks])
    ks = np.arange(dj - (n - 1), dj + 1)
    delta[ks + n - 1] += d * h[dj - ks]
    return delta * (2.0 / n)

def local_search_fast(h, n, deadline, radius=100):
    c = compute_c5_vec(h, n)
    best_c = float(c.max())
    h = h.copy()
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05]
    for d in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        delta = fast_delta(h, n, i, j, d)
                        nm = (c + delta).max()
                        if nm < best_c - 1e-14:
                            c = c + delta
                            h[i] += d
                            h[j] -= d
                            best_c = float(nm)
                            improved = True
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        delta = fast_delta(h, n, i, j, -d)
                        nm = (c + delta).max()
                        if nm < best_c - 1e-14:
                            c = c + delta
                            h[i] -= d
                            h[j] += d
                            best_c = float(nm)
                            improved = True
    return h, best_c

def pgd_search(h, n, deadline, alpha=300, top_k=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    step = 0.02
    for it in range(3000):
        if time.time() > deadline:
            break
        g = smooth_grad(h, n, alpha, top_k)
        s = step
        accepted = False
        for _ in range(15):
            h_new = project_feasible(h - s * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c + 1e-14:
                h = h_new
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
                accepted = True
                break
            s *= 0.5
        if not accepted:
            break
        if it % 300 == 299:
            alpha = max(30, alpha * 0.85)
    return best_h, best_c

def golden_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(half) * phi).astype(int) % n
    h[positions] = 1.0
    return project_feasible(h, n)

    for n_try in [400, 600, 800]:

                best_h = h_t
                best_n = n_try
                if time.time() < deadline - 60:

        except Exception:

    # Phase 5: Multi-start perturbation
