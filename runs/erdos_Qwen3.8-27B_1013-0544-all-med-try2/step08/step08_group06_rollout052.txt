def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - h; s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]

    hb[idx[:n//2]] = 1.0

def swap_delta(h, i, j, n):
    mr = np.arange(-(n-1), n)
    d = np.zeros(2*n-1)
    v = i + mr; m = (v>=0)&(v<n); d[m] += 1.0 - h[v[m]]
    v = i - mr; m = (v>=0)&(v<n); d[m] -= h[v[m]]
    v = j + mr; m = (v>=0)&(v<n); d[m] -= 1.0 - h[v[m]]
    v = j - mr; m = (v>=0)&(v<n); d[m] += h[v[m]]
    d[n-1] -= 2.0
    d[n-1+(j-i)] += 1.0
    d[n-1+(i-j)] += 1.0
