    h_c3 = diverse_start(40, rng, mode=2)
    c1 = compute_c5(h_c, 40)
    c2 = compute_c5(h_c2, 40)
    c3 = compute_c5(h_c3, 40)
    h_c = [h_c, h_c2, h_c3][int(np.argmin([c1, c2, c3]))]
    h_c, _ = lbfgs_optimize(h_c, 40, [100, 500], phase1_end, maxiter=300)
    h_c = interp_resample(h_c, 40, 80)
    h_c, _ = lbfgs_optimize(h_c, 80, [500, 2000], min(time.time() + 25, deadline - 800), maxiter=300)

    if time.time() < deadline - 700:
        h_c = interp_resample(h_c, 80, 160)
        h_c, _ = lbfgs_optimize(h_c, 160, [2000, 10000], min(time.time() + 25, deadline - 700), maxiter=300)

    if time.time() < deadline - 550:
        h_c = interp_resample(h_c, 160, 320)
        h_c, c_c = lbfgs_optimize(h_c, 320, [5000, 20000], min(time.time() + 30, deadline - 550), maxiter=300)
        if c_c < best_c:
            h_fine = interp_resample(h_c, 320, best_n)
            best_h = h_fine
            best_c = compute_c5(h_fine, best_n)

    # Phase 2: Fine L-BFGS-B on initial (250s)
    t2 = min(time.time() + 200, deadline - 300)
    h_a, c_a = lbfgs_optimize(best_h, best_n, [2000, 10000, 50000, 200000], t2, maxiter=400)
    if c_a < best_c:
        best_c, best_h = c_a, h_a

    # Phase 3: Multi-start exploration (150s)
    if time.time() < deadline - 150:
        t3 = min(time.time() + 130, deadline - 150)
        h_b, c_b = multi_start_lbfgs(best_h, best_n, rng, t3)
        if c_b < best_c:
            best_c, best_h = c_b, h_b

    # Phase 4: Coordinate descent polish (150s)
    if time.time() < deadline - 60:
        t4 = min(time.time() + 130, deadline - 60)
        best_h, best_c = coord_descent(best_h, best_n, t4, top_k=40)

    # Phase 5: Another L-BFGS-B pass with high alpha (50s)
    if time.time() < deadline - 20:
        t5 = min(time.time() + 40, deadline - 20)
        h_c, c_c = lbfgs_optimize(best_h, best_n, [100000, 500000, 2000000], t5, maxiter=300)
        if c_c < best_c:
            best_c, best_h = c_c, h_c

    # Phase 6: Final coordinate descent
    if time.time() < deadline - 5:
        best_h, best_c = coord_descent(best_h, best_n, deadline - 3, top_k=30)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
