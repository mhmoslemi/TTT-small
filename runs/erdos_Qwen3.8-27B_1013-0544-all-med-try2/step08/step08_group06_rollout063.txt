        else:

            h_vals = h[-s:]  # h[j-s] for j in [0, n+s-1]

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    d = target - h.sum()
    if d > 1e-12:
        r = 1.0 - h; h += d / r.sum() * r
    elif d < -1e-12:
        h += d / h.sum() * h
    return np.clip(h, 0.0, 1.0)

def sgrad(h, n, alpha):
    """Smoothed objective and gradient via log-sum-exp."""
    dx = 2.0 / n
    cf = np.correlate(h, 1 - h, mode='full').astype(float)
    ac = alpha * cf * dx
    m = ac.max()
    ev = np.exp(ac - m)
    sv = ev.sum()
    obj = (m + np.log(sv)) / alpha
    w = ev / sv
    g = np.zeros(n)
    act = np.where(w > 1e-10)[0]
    for k in act:
        s = k - (n - 1)
        wk = w[k]
        if s >= 0:
            g[s:] += wk * (1.0 - h[:n-s])
            g[:n-s] -= wk * h[s:]
        else:
            r = -s
            g[:n-r] += wk * (1.0 - h[r:])
            g[r:] -= wk * h[:n-r]
    return obj, g * dx

def fista(h, n, alpha, tlimit, max_it=20000):

        oy, gy = sgrad(y, n, alpha)
        for _ in range(40):

            ox, _ = sgrad(xn, n, alpha)
            if ox <= oy + gy @ (xn - y) + 0.5 * L * ((xn - y) @ (xn - y)) + 1e-10:
                break

        tn = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0

        x = xn; t = tn
        c = compute_c5(x, n)
        if c < bc: bc = c; bx = x.copy()
    return bx, bc

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    d = target - h.sum()
    if d > 1e-12:
        r = 1.0 - h
        if r.sum() > 1e-14:
            h += d / r.sum() * r
    elif d < -1e-12:
        if h.sum() > 1e-14:
            h += d / h.sum() * h
    return np.clip(h, 0.0, 1.0)

def sgrad(h, n, alpha):
    dx = 2.0 / n
    cf = np.correlate(h, 1 - h, mode='full').astype(float)
    ac = alpha * cf * dx
    m = ac.max()
    ev = np.exp(ac - m)
    sv = ev.sum()
    obj = (m + np.log(sv)) / alpha
    w = ev / sv
    g = np.zeros(n)
    act = np.where(w > 1e-10)[0]
    for k in act:
        s = k - (n - 1)
        wk = w[k]
        if s >= 0:
            g[s:] += wk * (1.0 - h[:n - s])
            g[:n - s] -= wk * h[s:]
        else:
            r = -s
            g[:n - r] += wk * (1.0 - h[r:])
            g[r:] -= wk * h[:n - r]
    return obj, g * dx

def fista(h, n, alpha, tlimit, max_it=30000):
    x = h.copy()
    y = h.copy()
    t = 1.0
    L = 50.0
    bc = compute_c5(x, n)
    bx = x.copy()
    for _ in range(max_it):
        if time.time() > tlimit:
            break
        oy, gy = sgrad(y, n, alpha)
        for _ in range(40):
            xn = project(y - gy / L, n)
            ox, _ = sgrad(xn, n, alpha)
            if ox <= oy + gy @ (xn - y) + 0.5 * L * ((xn - y) @ (xn - y)) + 1e-10:
                break
            L = min(L * 1.5, 1e6)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = project(xn + (t - 1.0) / tn * (xn - x), n)
        x = xn
        t = tn
        c = compute_c5(x, n)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def to_bin(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def binls(h, n, tlimit, rng, ni=3000000):
    dx = 2.0 / n
    h = to_bin(h, n).copy()
    cr = np.correlate(h, 1 - h, mode='full').astype(float)
    cc = cr.max() * dx
    bh = h.copy()
    bc = cc
    on = np.where(h == 1)[0]
    zo = np.where(h == 0)[0]
    mr = np.arange(-(n - 1), n)
    for _ in range(ni):
        if time.time() > tlimit:
            break
        i = zo[rng.randint(len(zo))]
        j = on[rng.randint(len(on))]
        d = np.zeros(2 * n - 1)
        im = i + mr
        ms = (im >= 0) & (im < n)
        d[ms] += 1.0 - h[im[ms]]
        im = i - mr
        ms = (im >= 0) & (im < n)
        d[ms] -= h[im[ms]]
        jm = j + mr
        ms = (jm >= 0) & (jm < n)
        d[ms] -= 1.0 - h[jm[ms]]
        jm = j - mr
        ms = (jm >= 0) & (jm < n)
        d[ms] += h[jm[ms]]
        d[n - 1] -= 2.0
        d[n - 1 + j - i] += 1.0
        d[n - 1 + i - j] += 1.0
        nc = (cr + d).max() * dx
        if nc < cc - 1e-15:
            h[i], h[j] = 1.0, 0.0
            cr += d
            cc = nc
            zi = np.where(zo == i)[0][0]
            oj = np.where(on == j)[0][0]
            zo[zi] = j
            on[oj] = i
            if cc < bc:
                bc = cc
                bh = h.copy()
    return bh, bc

def binls_greedy(h, n, tlimit, max_sweeps=200):
    dx = 2.0 / n
    h = to_bin(h, n).copy()
    cr = np.correlate(h, 1 - h, mode='full').astype(float)
    cc = cr.max() * dx
    bh = h.copy()
    bc = cc
    on = list(np.where(h == 1)[0])
    zo = list(np.where(h == 0)[0])
    mr = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > tlimit:
            break
        best_nc = cc
        best_pair = None
        for i in zo:
            for j in on:
                d = np.zeros(2 * n - 1)
                im = i + mr
                ms = (im >= 0) & (im < n)
                d[ms] += 1.0 - h[im[ms]]
                im = i - mr
                ms = (im >= 0) & (im < n)
                d[ms] -= h[im[ms]]
                jm = j + mr
                ms = (jm >= 0) & (jm < n)
                d[ms] -= 1.0 - h[jm[ms]]
                jm = j - mr
                ms = (jm >= 0) & (jm < n)
                d[ms] += h[jm[ms]]
                d[n - 1] -= 2.0
                d[n - 1 + j - i] += 1.0
                d[n - 1 + i - j] += 1.0
                nc = (cr + d).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_pair = (i, j, d)
        if best_pair is not None and best_nc < cc - 1e-15:
            i, j, d = best_pair
            h[i], h[j] = 1.0, 0.0
            cr += d
            cc = best_nc
            zo.remove(i)
            on.remove(j)
            zo.append(j)
            on.append(i)
            if cc < bc:
                bc = cc
                bh = h.copy()
        else:
            break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
