```python
        for j in range(n - 1):  # Leave last coordinate as slack
            # Compute g_m = d(corr_m)/d(h[j])
            g = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                lag = lags[m]
                im = j + lag
                if 0 <= im < n:
                    g[m] = (1.0 - h[im])
                im2 = j - lag
                if 0 <= im2 < n:
                    g[m] -= h[im2]
            
            # Also need g for h[n-1] (the slack)
            j_slack = n - 1
            g_s = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                lag = lags[m]
                im = j_slack + lag

                im2 = j_slack - lag
                if 0 <= im2 < n:
                    g_s[m] -= h[im2]

            # New corr = corr + g*delta + g_s*(-delta) = corr + (g - g_s)*delta
            g_eff = g - g_s
            
            # Minimize f(delta) = smooth_max(corr + g_eff * delta) over delta such that
            # 0 <= h[j] + delta <= 1 and 0 <= h[n-1] - delta <= 1
            # i.e., -h[j] <= delta <= 1-h[j] and h[n-1]-1 <= delta <= h[n-1]
```

Now, for the coordinate descent, I need g_eff = g_j - g_{n-1} (the effective gradient when I change h[j] and adjust h[n-1]).

g_j can be computed in O(n) using the slicing above.
g_{n-1} is the same formula with j = n-1:
- Forward: g[n-1-(n-1) : 2n-1-(n-1)] = g[0:n] += 1-h
- Backward: g[n-1 : n-1+n] = g[n-1:2n-1] -= h[::-1]

So g_{n-1} is precomputable once.

And g_j for arbitrary j:
- Forward: g[n-1-j : 2n-1-j] += 1-h
- Backward: g[j : j+n] -= h[::-1]

Both are O(n) operations (just adding shifted copies of precomputed vectors).

So the total cost per coordinate in the coordinate descent is O(n) for computing g, plus O(2n) * 50 for the ternary search evaluations. Total per coordinate: O(n). Per sweep: O(n²). Per full CD: O(n² * max_sweeps).

For n=400, max_sweeps=50: 50 * 160K = 8M. Very fast!

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_value(corr, alpha):
    m = corr.max()
    sh = alpha * (corr - m)
    ms = sh.max()
    return m + (ms + np.log(np.exp(sh - ms).sum())) / alpha

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m_val = corr.max()
    m_idx = int(np.argmax(corr))
    
    sh = alpha * (corr - m_val)
    ms = sh.max()
    exp_s = np.exp(sh - ms)
    S = exp_s.sum()
    w = exp_s / S
    value = m_val + (ms + np.log(S)) / alpha
    
    lags = np.arange(-(n-1), n)
    j_idx = np.arange(n)
    grad = np.zeros(n)
    
    # Max term
    lag_star = lags[m_idx]
    im = j_idx + lag_star
    mfk = (im >= 0) & (im < n)
    grad[mfk] += (1.0 - h[im[mfk]])
    im2 = j_idx - lag_star
    mbk = (im2 >= 0) & (im2 < n)
    grad[mbk] -= h[im2[mbk]]
    
    # Weighted sum
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        lag = lags[m]
        im = j_idx + lag
        mfk = (im >= 0) & (im < n)
        grad[mfk] += w[m] * (1.0 - h[im[mfk]])
        im2 = j_idx - lag
        mbk = (im2 >= 0) & (im2 < n)
        grad[mbk] -= w[m] * h[im2[mbk]]
    
    return value, grad

def optimize_lbfgsb(h, n, alpha, maxiter=1000, penalty=1e8):
    target = n / 2.0
    
    def fun(x):
        val, grad = smooth_max_grad(x, n, alpha)
        sv = x.sum() - target
        return val + 0.5 * penalty * sv * sv, grad + penalty * sv
    
    try:
        res = minimize(fun, h, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def coord_descent(h, n, alpha, max_sweeps=50):
    """Fast coordinate descent using O(n) per coordinate evaluation."""
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    
    # Precompute g for the last coordinate (n-1), used as slack
    g_slack = np.zeros(2 * n - 1)
    g_slack[0:n] += 1.0 - h       # forward for j=n-1: positions [0, n)
    g_slack[n-1:2*n-1] -= h[::-1]  # backward for j=n-1: positions [n-1, 2n-1)
    
    one_minus_h = 1.0 - h
    h_rev = h[::-1]
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n - 1):
            # Compute g_j
            g_j = np.zeros(2 * n - 1)
            sf = n - 1 - j
            g_j[sf:sf + n] += one_minus_h
            g_j[j:j + n] -= h_rev
            
            g_eff = g_j - g_slack
            
            # Bounds for delta
            lo = max(-h[j], h[n-1] - 1.0)
            hi = min(1.0 - h[j], h[n-1])
            if hi - lo < 1e-13:
                continue
            
            # Ternary search for optimal delta
            a, b = lo, hi
            for _ in range(60):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                c1 = corr + g_eff * m1
                c2 = corr + g_eff * m2
                f1 = smooth_max_value(c1, alpha)
                f2 = smooth_max_value(c2, alpha)
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            delta = (a + b) / 2.0
            
            if abs(delta) > 1e-13:
                h[j] += delta
                h[n-1] -= delta
                corr += g_eff * delta
                # Update precomputed arrays
                one_minus_h[j] = 1.0 - h[j]
                one_minus_h[n-1] = 1.0 - h[n-1]
                h_rev[n-1-j] = h[j]  # h_rev[k] = h[n-1-k], so h_rev[n-1-j] = h[j]
                h_rev[0] = h[n-1]
                improved = True
        
        if not improved:
            break
    
    h = project_feasible(h, n)
    return h
```

```python
    hi = float(h.max()) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_value(corr, alpha):
    m = float(corr.max())
    sh = alpha * (corr - m)
    ms = float(sh.max())
    return m + (ms + float(np.log(np.exp(sh - ms).sum()))) / alpha

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m_val = float(corr.max())
    m_idx = int(np.argmax(corr))
    sh = alpha * (corr - m_val)
    ms = float(sh.max())
    exp_s = np.exp(sh - ms)
    S = float(exp_s.sum())
    w = exp_s / S
    value = m_val + (ms + np.log(S)) / alpha
    lags = np.arange(-(n - 1), n)
    j_idx = np.arange(n)
    grad = np.zeros(n)
    lag_star = lags[m_idx]
    im = j_idx + lag_star
    mfk = (im >= 0) & (im < n)
    grad[mfk] += (1.0 - h[im[mfk]])
    im2 = j_idx - lag_star
    mbk = (im2 >= 0) & (im2 < n)
    grad[mbk] -= h[im2[mbk]]
    active = np.where(w > 1e-12)[0]
    for m in active:
        lag = lags[m]
        im = j_idx + lag
        mfk = (im >= 0) & (im < n)
        grad[mfk] += w[m] * (1.0 - h[im[mfk]])
        im2 = j_idx - lag
        mbk = (im2 >= 0) & (im2 < n)
        grad[mbk] -= w[m] * h[im2[mbk]]
    return value, grad * dx

def opt_lbfgsb(h, n, alpha, maxiter=500, penalty=1e6):
    target = n / 2.0
    def fun(x):
        val, grad = smooth_max_grad(x, n, alpha)
        sv = x.sum() - target
        return val + 0.5 * penalty * sv * sv, grad + penalty * sv
    try:
        res = minimize(fun, h, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def opt_slsqp(h, n, alpha, maxiter=100):
    target = n / 2.0
    ones_vec = np.ones(n)
    def fun(x):
        return smooth_max_grad(x, n, alpha)
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return ones_vec
    try:
        res = minimize(fun, h, jac=True, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def coord_descent(h, n, alpha, max_sweeps=30):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    omh = 1.0 - h
    hrev = h[::-1].copy()
    # g for slack coordinate (n-1)
    g_slack = np.zeros(2 * n - 1)
    g_slack[0:n] += omh
    g_slack[n-1:2*n-1] -= hrev
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n - 1):
            g_j = np.zeros(2 * n - 1)
            sf = n - 1 - j
            g_j[sf:sf + n] += omh
            g_j[j:j + n] -= hrev
            g_eff = g_j - g_slack
            lo = max(-h[j], h[n-1] - 1.0)
            hi = min(1.0 - h[j], h[n-1])
            if hi - lo < 1e-13:
                continue
            a, b = lo, hi
            for _ in range(50):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = smooth_max_value(corr + g_eff * m1, alpha)
                f2 = smooth_max_value(corr + g_eff * m2, alpha)
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            delta = 0.5 * (a + b)
            if abs(delta) > 1e-13:
                h[j] += delta
                h[n-1] -= delta
                corr += g_eff * delta
                omh[j] = 1.0 - h[j]
                omh[n-1] = 1.0 - h[n-1]
                hrev[n-1-j] = h[j]
                hrev[0] = h[n-1]
                improved = True
        if not improved:
            break
    return project_feasible(h, n)
```
