import time
import numpy as np
from scipy.optimize import minimize


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(300):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = float(room.sum())
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = float(h.sum())
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        room = np.minimum(h, 1.0 - h)
        idx = int(np.argmax(room))
        if room[idx] > 0.5 * abs(diff):
            h[idx] += diff
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def soft_parts(x, alpha, n):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    gm = float(np.max(g))
    if alpha < 1e-6:
        w = np.full(len(g), 1.0 / len(g))
        return gm, w, gm
    shifted = alpha * (g - gm)
    with np.errstate(under='ignore', over='ignore'):
        e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0:
        return gm, np.zeros(len(g)), gm
    w = e / se
    return gm + float(np.log(se)) / alpha, w, gm


def weights_lag_grad(w, x, dx, n):
    nm1 = n - 1
    grad = np.zeros(n, dtype=float)
    for p, wp in enumerate(w):
        if wp <= 1e-16:
            continue
        m = p - nm1
        if m >= 0:
            r = n - m
            grad[m:] += wp
            grad[m:] -= wp * x[:r]
            grad[:r] -= wp * x[m:]
        else:
            q = -m
            r = n - q
            grad[:r] += wp
            grad[:r] -= wp * x[q:]
            grad[q:] -= wp * x[:r]
    return grad * dx


def obj_h_scalar(x, alpha, n):
    F, _, _ = soft_parts(x, alpha, n)
    return F


def jac_h(x, alpha, n):
    _, w, _ = soft_parts(x, alpha, n)
    return weights_lag_grad(w, x, 2.0 / n, n)


def sum_constraint(x, n):
    return float(np.sum(x) - 0.5 * n)


def slsqp_smooth(h, n, alpha, maxiter):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * int(n)
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (int(n),)}]
    try:
        res = minimize(obj_h_scalar, h, args=(alpha, n), method='SLSQP',
                       bounds=bounds, constraints=cons, jac=jac_h,
                       options={'maxiter': max(1, int(maxiter)), 'ftol': 1e-12, 'disp': False})
        hx = np.asarray(res.x, dtype=float)
    except Exception:
        hx = h
    return project_feasible(np.clip(hx, 0.0, 1.0), int(n))


def resample_h(h_old, n_old, n_new):
    if int(n_new) == int(n_old):
        return project_feasible(h_old.copy(), int(n_new))
    xo = (np.arange(int(n_old)) + 0.5) * (2.0 / int(n_old))
    xn = (np.arange(int(n_new)) + 0.5) * (2.0 / int(n_new))
    h = np.interp(xn, xo, h_old)
    return project_feasible(h, int(n_new))


def random_feasible(n, rng):
    kind = int(rng.randint(3))
    n = int(n)
    if kind == 0:
        h = np.zeros(n)
        k = n // 2
        if k > 0:
            h[rng.choice(n, k, replace=False)] = 1.0
        if n % 2:
            h[rng.randint(n)] += 0.5
    elif kind == 1:
        x = np.linspace(0.0, 2.0, n)
        f = float(rng.uniform(0.5, 5.0))
        ph = float(rng.uniform(0.0, 6.283185307179586))
        h = 0.5 + 0.5 * np.sin(2.0 * 3.141592653589793 * f * x + ph)
    else:
        h = rng.rand(n)
    return project_feasible(h, n)


def variable_delta(idx, h, m_range):
    n = len(h)
    B = np.zeros(2 * n - 1, dtype=float)
    k1 = idx - m_range
    m1 = (k1 >= 0) & (k1 < n)
    B[m1] += 1.0 - h[k1[m1]]
    k2 = idx + m_range
    m2 = (k2 >= 0) & (k2 < n)
    B[m2] -= h[k2[m2]]
    return B


def binary_local(h0, n, deadline, max_sweeps=30, rng=None):
    if rng is None:
        rng = np.random.RandomState(12345)
    n = int(n)
    if n % 2 != 0:
        return h0, compute_c5(h0, n)
    k = n // 2
    h = np.zeros(n)
    ord_idx = np.argsort(h0)[::-1]
    h[ord_idx[:k]] = 1.0
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur = float(c.max())
    best_c = cur * (2.0 / n)
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    p0 = n - 1
    for _ in range(int(max_sweeps)):
        if time.time() > deadline:
            break
        zero_idx = np.where(h < 0.5)[0]
        one_idx = np.where(h > 0.5)[0]
        if len(zero_idx) == 0 or len(one_idx) == 0:
            break
        zero_idx = rng.permutation(zero_idx)
        one_idx = rng.permutation(one_idx)
        Bz = []
        for i in zero_idx:
            Bz.append(variable_delta(int(i), h, m_range))
        Bo = []
        for j in one_idx:
            Bo.append(variable_delta(int(j), h, m_range))
        improved = False
        for Bi, i in zip(Bz, zero_idx):
            if time.time() > deadline:
                break
            ii = int(i)
            for Bj, j in zip(Bo, one_idx):
                jj = int(j)
                cand = c + Bi - Bj
                cand[p0] -= 2.0
                cand[p0 + ii - jj] += 1.0
                cand[p0 + jj - ii] += 1.0
                nm = float(cand.max())
                if nm < cur - 1e-12:
                    h[ii] = 1.0
                    h[jj] = 0.0
                    c = cand
                    cur = nm
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        c = np.correlate(h, 1.0 - h, mode='full').astype(float)
        cur = float(c.max())
        cc = cur * (2.0 / n)
        if cc < best_c:
            best_c = cc
            best_h = h.copy()
    return best_h, best_c


def coord_descent(h, n, deadline, max_sweeps=30, bbis=22):
    h = project_feasible(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    m_range = np.arange(-(n - 1), n)
    L = 2 * n - 1
    for _ in range(int(max_sweeps)):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            B = np.zeros(L)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += 1.0 - h[k1[mask1]]
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            old = float(h[i])
            cur_max = float(corr.max())
            v0 = float(np.max(corr - B * old))
            v1 = float(np.max(corr + B * (1.0 - old)))
            loT = -1e-12
            hiT = min(v0, v1, cur_max) + 1e-12
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = float(corr[zero_B].max()) if zero_B.any() else -np.inf
            if hiT > loT:
                for __ in range(int(bbis)):
                    mid = 0.5 * (loT + hiT)
                    if zero_max > mid + 1e-14:
                        loT = mid
                        continue
                    lbx = 0.0
                    ubx = 1.0
                    if neg_B.any():
                        lbx = max(lbx, old + float(np.max((mid - corr[neg_B]) / B[neg_B])))
                    if pos_B.any():
                        ubx = min(ubx, old + float(np.min((mid - corr[pos_B]) / B[pos_B])))
                    if lbx <= ubx + 1e-12:
                        hiT = mid
                    else:
                        loT = mid
                lbx = 0.0
                ubx = 1.0
                if neg_B.any():
                    lbx = max(lbx, old + float(np.max((hiT - corr[neg_B]) / B[neg_B])))
                if pos_B.any():
                    ubx = min(ubx, old + float(np.min((hiT - corr[pos_B]) / B[pos_B])))
                if lbx <= ubx + 1e-12:
                    xopt = float(np.clip(0.5 * (lbx + ubx), 0.0, 1.0))
                else:
                    xopt = old
            else:
                xopt = old
            if abs(xopt - old) > 1e-12:
                corr += B * (xopt - old)
                h[i] = xopt
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    return h, corr
