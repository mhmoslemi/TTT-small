    w = np.ones(n_shifts) / n_shifts
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        # Build the Toeplitz matrix M where M[i,j] = w[j-i + (n-1)]
        # M[i,j] = w_{j-i} where index in w is (j-i) + (n-1)
        # For j >= i: index = j - i + (n-1), range [n-1, 2n-2]
        # For j < i: index = j - i + (n-1), range [0, n-2]
        
        # Actually, let's just use the gradient approach for the inner problem
        # Maximize h^T M h = sum_k w_k * r(k) where r(k) = sum_i h[i]h[i+k]
        
        # Use projected gradient ascent
        h_curr = best_h.copy()
        for pg_it in range(200):
            if time.time() > deadline - 5:
                break
            # Gradient of sum_k w_k * r(k) with respect to h
            # r(k) = sum_i h[i]*h[i+k] for k>=0
            # d(r(k))/d(h[m]) = sum_i [h[i+k] * delta(m,i) + h[i] * delta(m,i+k)]
            #                = h[m+k] + h[m-k] (for valid indices)
            # So the gradient is: grad[m] = sum_k w_k * (h[m+k] + h[m-k])
            #                    = sum_k 2*w_k*h[m±k] (by symmetry w_k = w_{-k})
            
            # Compute gradient
            g = np.zeros(n)
            for k in range(n):
                if k == 0:
                    wk = w[k + n - 1]
                    g += 2.0 * wk * h
                else:
                    wk_pos = w[k + n - 1]
                    wk_neg = w[(n-1) - k]
                    wsum = wk_pos + wk_neg
                    if wsum > 0:
                        # h[m+k] for m=0..n-1-k and h[m-k] for m=k..n-1
                        g[:n-k] += wsum * h[k:]
                        g[k:] += wsum * h[:n-k]
            
            # Add constraint force: push toward sum = n/2
            s = h_curr.sum() - n/2.0
            g -= 50.0 * s  # penalty gradient
            
            # Projected gradient step
            lr = 0.1 / (np.max(np.abs(g)) + 1e-10)
            h_curr = h_curr + lr * g
            h_curr = np.clip(h_curr, 0, 1)
            # Fix sum
            h_curr = project_feasible(h_curr, n)

        if c_new < best_c:
            best_c = c_new
            best_h = h_curr.copy()
        
        # Update weights
        c_vec = compute_c5_vec(h_curr, n)
        w = w * np.power(c_vec / (c_vec.mean() + 1e-10), beta)
        w /= w.sum()
    
    return best_h, best_c

  (valid if 0<=j-t<n and j-t != i, i.e., j-t != i means j != i+t)

    result = np.zeros(2*n-1)
    t_range = np.arange(-(n-1), n)
    
    # For each t, the affected m values are at most {i, j, i-t, j-t}
    # We need 0<=m<n and 0<=m+t<n
    for idx, t in enumerate(t_range):
        # Find all m in {i, j, i-t, j-t} that are valid (0<=m<n, 0<=m+t<n)
        ms = set()
        for m in [i, j, i-t, j-t]:
            if 0 <= m < n and 0 <= m+t < n:
                ms.add(m)
        
        delta = 0.0
        for m in ms:
            hp_m = 1.0 if m == i else (0.0 if m == j else h[m])
            hp_mt = 1.0 if (m+t) == i else (0.0 if (m+t) == j else h[m+t])
            delta += hp_m * (1.0 - hp_mt) - h[m] * (1.0 - h[m+t])
        result[idx] = delta
    
    return result

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    c = c * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0/n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    for mIdx in range(2 * n - 1):
        if w[mIdx] < 1e-15:
            continue
        k = mIdx - (n - 1)
        wm = w[mIdx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
