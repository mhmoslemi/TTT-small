def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            room = (1.0 - h); s = room.sum()
            if s < 1e-15:
                break
            h = h + (d / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + (d / s) * h
    return np.clip(h, 0.0, 1.0)

def proj_sum(x, n):
    target = 0.5 * n
    lo = float(x.min()); hi = float(x.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)

def c5_of(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + (d / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + (d / s) * h
    return np.clip(h, 0.0, 1.0)

def proj_sum(x, n):
    target = 0.5 * n
    lo = float(x.min()) - n
    hi = float(x.max()) + 1.0
    for _ in range(56):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(x - t, 0.0, 1.0)

def resample_to(h, n_old, n_new):
    if n_new == n_old:
        return project_feasible(h, n_old)
    x = np.linspace(0.0, 2.0, n_old)
    y = np.linspace(0.0, 2.0, n_new)
    return project_feasible(np.interp(y, x, h), n_new)

def lse_val_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    G = c * dx
    m = float(G.max())
    e = np.exp(alpha * (G - m))
    s = float(e.sum())
    w = e / s
    val = m + math.log(s) / alpha
    sl = slice(n - 1, 2 * n - 1)
    a = np.convolve(1.0 - h, w, mode="full")[sl]
    b = np.correlate(h, w, mode="full")[sl]
    return val, dx * (a - b)

def lse_polish(h, n, alphas, iters_per, deadline):
    h = project_feasible(h, n).copy()
    for alpha in alphas:
        if time.time() > deadline:
            break
        f, g = lse_val_grad(h, n, alpha)
        eta = 2.0 / (alpha * (2.0 / n))
        for _ in range(iters_per):
            if time.time() > deadline:
                break
            base_f = f; base_g = g
            ok = False
            for shr in range(14):
                h_try = proj_sum(h - eta * g, n)
                f_try, g_try = lse_val_grad(h_try, n, alpha)
                if f_try <= base_f + 1e-16:
                    ok = True
                    break
                eta *= 0.5
            if not ok:
                break
            dh = h_try - h
            dg = g_try - base_g
            dd = float(dh @ dh); gg = float(dh @ dg)
            h, f, g = h_try, f_try, g_try
            if gg > 1e-18:

def c5_of(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx


def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + (d / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + (d / s) * h
    return np.clip(h, 0.0, 1.0)


def proj_sum(x, n):
    target = 0.5 * n
    lo = float(x.min()) - n
    hi = float(x.max()) + 1.0
    for _ in range(56):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(x - t, 0.0, 1.0)


def resample_to(h, no, nn):
    if nn == no:
        return project_feasible(h, nn)
    x = np.linspace(0.0, 2.0, no)
    y = np.linspace(0.0, 2.0, nn)
    return project_feasible(np.interp(y, x, h), nn)


def snap_try(h, n, eps):
    h2 = h.copy()
    i0 = h2 < eps
    i1 = h2 > 1.0 - eps
    if not (i0.any() or i1.any()):

    h2[i0] = 0.0

def c5_of(hu, nn):

def project_feasible(hu, nn):
    target = 0.5 * nn
    hu = np.clip(hu, 0.0, 1.0).astype(float)
    for _ in range(80):
        dd = target - hu.sum()
        if abs(dd) < 1e-15:
            break
        if dd > 0:
            rw = 1.0 - hu
            ss = rw.sum()
            if ss < 1e-15:
                break
            hu = hu + (dd / ss) * rw
        else:
            ss = hu.sum()
            if ss < 1e-15:
                break
            hu = hu + (dd / ss) * hu
    return np.clip(hu, 0.0, 1.0)


def proj_sum(z, nn):
    target = 0.5 * nn
    lo = float(z.min()) - nn
    hi = float(z.max()) + 1.0
    for _ in range(56):
        mid = 0.5 * (lo + hi)
        ss = float(np.clip(z - mid, 0.0, 1.0).sum())
        if ss > target:
            lo = mid
        else:
            hi = mid
    tt = 0.5 * (lo + hi)
    return np.clip(z - tt, 0.0, 1.0)


def resample_to(hu, no, nn):
    if nn == no:
        return project_feasible(hu, nn)
    xo = np.linspace(0.0, 2.0, no)
    xn = np.linspace(0.0, 2.0, nn)
    return project_feasible(np.interp(xn, xo, hu), nn)


def snap_try(hu, nn, eps):
    h2 = hu.copy()
    i0 = h2 < eps
    i1 = h2 > 1.0 - eps
    if not (i0.any() or i1.any()):

    h2[i0] = 0.0

    dd = h2.sum() - 0.5 * nn

    k_arr = np.arange(pp) - (nn - 1)

def c5_of(hu, nn):
    a_dp = 2.0 / nn
    c = np.correlate(hu, 1.0 - hu, mode="full")
    return float(np.max(c)) * a_dp


def project_feasible(hu, nn):
    target = 0.5 * nn
    hu = np.clip(hu, 0.0, 1.0).astype(float)
    for _ in range(80):
        dd = target - hu.sum()
        if abs(dd) < 1e-15:
            break
        if dd > 0:
            rw = 1.0 - hu
            ss = rw.sum()
            if ss < 1e-15:
                break
            hu = hu + (dd / ss) * rw
        else:
            ss = hu.sum()
            if ss < 1e-15:
                break
            hu = hu + (dd / ss) * hu
    return np.clip(hu, 0.0, 1.0)


def proj_sum(z, nn):
    target = 0.5 * nn
    lo = float(z.min()) - nn
    hi = float(z.max()) + 1.0
    for _ in range(56):
        mid = 0.5 * (lo + hi)
        ss = float(np.clip(z - mid, 0.0, 1.0).sum())
        if ss > target:
            lo = mid
        else:
            hi = mid
    tt = 0.5 * (lo + hi)
    return np.clip(z - tt, 0.0, 1.0)


def resample_to(hu, no, nn):
    if nn == no:
        return project_feasible(hu, nn)
    xo = np.linspace(0.0, 2.0, no)
    xn = np.linspace(0.0, 2.0, nn)
    return project_feasible(np.interp(xn, xo, hu), nn)


def snap_try(hu, nn, eps):
    h2 = hu.copy()
    i0 = h2 < eps
    i1 = h2 > 1.0 - eps
    if not (i0.any() or i1.any()):
        return None
    h2[i0] = 0.0
    h2[i1] = 1.0
    dd = h2.sum() - 0.5 * nn
    m01 = (h2 > 0.0) & (h2 < 1.0)
    if not m01.any():
        if abs(dd) < 1e-9:
            return h2
        return None
    cap = float(np.minimum(h2[m01], 1.0 - h2[m01]).sum())
    if abs(dd) > cap - 1e-12:
        return None
    return project_feasible(h2, nn)


def lse_val_grad(hu, nn, a_w):
    a_dp = 2.0 / nn
    c = np.correlate(hu, 1.0 - hu, mode="full")
    gv = c * a_dp
    mm = float(np.max(gv))
    ee = np.exp(a_w * (gv - mm))
    ss = float(np.sum(ee))
    warr = ee / ss
    val = mm + math.log(ss) / a_w
    sl = slice(nn - 1, 2 * nn - 1)
    a_ac = np.convolve(1.0 - hu, warr, mode="full")[sl]
    b_av = np.correlate(hu, warr, mode="full")[sl]
    return val, a_dp * (a_ac - b_av)


def lwetes(hu, nn, alphas, iters, deadline):

    for a_gs in alphas:
        if time.time() > deadline:
            break
        fv, g = lse_val_grad(hu, nn, a_gs)
        if float(np.abs(g).max()) < 1e-19:
            continue
        eta = 1.0 / (a_gs * (2.0 / nn))
        for _ in range(iters):
            if time.time() > deadline:
                break
            base_fv = fv
            base_g = g
            base_eta = eta
            xok = False

                f_try, g_try = lse_val_grad(h_try, nn, a_gs)
                if f_try <= base_fv + 1e-16:
                    xok = True
                    break

            if not xok:

def cd(hu, nn, nsweep, deadline, rnd=None):

def dir_vec(mm, hu, nn, k_arr):
    pp = 2 * nn - 1
    v = np.zeros(pp)
    p = mm - k_arr
    m1 = (p >= 0) & (p < nn)

    q = mm + k_arr
    m2 = (q >= 0) & (q < nn)
    v[m2] -= hu[q[m2]]
    return v
