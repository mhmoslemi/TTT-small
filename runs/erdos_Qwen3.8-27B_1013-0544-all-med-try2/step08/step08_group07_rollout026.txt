    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    p = np.exp(shifted - sm)
    p = p / p.sum()
    
    # p has 2n-1 entries, indexed by k from -(n-1) to n-1
    # grad[i] = dx * sum_k p[k] * [(1 - h[i+k]) * 1_{0<=i+k<n} - h[i-k] * 1_{0<=i-k<n}]

    for idx in range(2*n-1):
        k = m_range[idx]
        w = p[idx]
        if w < 1e-12:
            continue
        # Term 1: (1 - h[i+k]) for valid i+k
        # i+k is valid when 0 <= i+k < n, i.e., -k <= i < n-k
        lo1 = max(0, -k)
        hi1 = min(n, n - k)

        hi2 = min(n, n + k)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2-k:hi2-k]
    grad *= dx

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        if sm > 500: return m + sm / alpha
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def cd(h, n, sweeps, deadline):

        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - mr; m1 = (k1 >= 0) & (k1 < n)
            B[m1] += 1.0 - h[k1[m1]]
            k2 = i + mr; m2 = (k2 >= 0) & (k2 < n)

        sm = sh.max()
