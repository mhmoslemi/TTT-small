import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lp_solve_fixed_g(h, g, n, maxiter=200):
    """Solve LP: min t s.t. c_s(h,g) <= t, 0<=h<=1, sum(h)=n/2"""
    dx = 2.0 / n
    # Variables: [h_0, ..., h_{n-1}, t]
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Inequality: for each s, sum_i h[i]*g[i+s]*dx <= t
    # => sum_i h[i]*g[i+s]*dx - t <= 0
    n_s = 2 * n - 1
    A = np.zeros((n_s, n + 1))
    b = np.zeros(n_s)
    
    m_range = np.arange(-(n-1), n)
    for idx_s, s in enumerate(m_range):
        # c_s = sum_i h[i] * g[i+s] * dx (valid indices)
        # For each i where 0<=i<n and 0<=i+s<n:
        i_min = max(0, -s)
        i_max = min(n, n - s)
        for i in range(i_min, i_max):
            A[idx_s, i] = g[i + s] * dx
        A[idx_s, -1] = -1.0  # -t
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return h

def alternating_lp(h_init, n, max_iters=100, deadline=None):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        g = 1.0 - h
        h_new = lp_solve_fixed_g(h, g, n)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        # Check convergence
        if np.max(np.abs(h_new - h)) < 1e-10:
            break
        h = h_new
    return best_h, best_c
