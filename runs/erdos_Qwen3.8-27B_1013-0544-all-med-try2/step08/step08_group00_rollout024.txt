import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    t = 0.5 * n
    for _ in range(80):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (d / s) * h
    h = np.clip(h, 0.0, 1.0)
    idxs = np.argsort(1.0 - h) if h.sum() < t else np.argsort(h)
    d = t - h.sum()
    for i in idxs:
        if abs(d) < 1e-15:
            break
        if d > 0:
            a = min(1.0 - h[i], d)
            h[i] += a
            d -= a
        else:
            a = min(h[i], -d)
            h[i] -= a
            d += a
    if abs(d) > 1e-15:
        cand = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(cand):
            h[cand[0]] = min(1.0, max(0.0, h[cand[0]] + d))
    return h

def default_start(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.25 * np.sin(2.0 * np.pi * x) + 0.10 * np.sin(4.0 * np.pi * x + 0.7)
    return project_feasible(h, n)

def compute_C(h):
    return np.correlate(h, 1.0 - h, mode="full").astype(float)

def c5(h):
    n = len(h)
    return float(compute_C(h).max() * (2.0 / n))

def smooth_grad_data(h, n, alpha):
    C = compute_C(h)
    a = alpha * C
    m = a.max()
    e = np.exp(a - m)
    p = e / e.sum()
    g = np.zeros(n)
    off = n - 1
    dx = 2.0 / n
    for idx, w in enumerate(p):
        if w <= 1e-16:
            continue
        wm = w * dx
        t = idx - off
        if t >= 0:
            nv = n - t
            g[t:t + nv] += wm * (1.0 - h[:nv])
            g[:nv] -= wm * h[t:t + nv]
        else:
            s = -t
            nv = n - s
            g[:nv] += wm * (1.0 - h[s:s + nv])
            g[s:s + nv] -= wm * h[:nv]
    return C, g

def lbfgs_obj(x, n, alpha, mu):
    if not np.all(np.isfinite(x)):
        return 1e100
    C = compute_C(x)
    a = alpha * C
    m = a.max()
    e = np.exp(a - m)
    s = x.sum() - 0.5 * n
    return float((2.0 / n) * (m + np.log(e.sum())) / alpha + mu * s * s)

def lbfgs_jac(x, n, alpha, mu):
    _, g = smooth_grad_data(x, n, alpha)
    s = x.sum() - 0.5 * n
    return g + 2.0 * mu * s

def lbfgs_refine(h, n, deadline, max_iters=160):
    best = c5(h)
    bounds = [(0.0, 1.0)] * n
    alphas = [200.0, 1000.0, 5000.0, 20000.0, 100000.0, 500000.0, 2000000.0]
    mus = [1.0e3, 5.0e3, 2.0e4, 1.0e5, 5.0e5, 2.0e6, 1.0e7]
    cyc = 0
    while time.time() < deadline - 2 and cyc < 3:
        improved = False
        for i, alpha in enumerate(alphas):
            if time.time() > deadline - 2:
                break
            mit = max(20, min(max_iters, int(12.0 * (deadline - time.time()))))
            try:
                res = minimize(lbfgs_obj, h, jac=lbfgs_jac, args=(n, alpha, mus[i]),
                               method="L-BFGS-B", bounds=bounds,
                               options={"maxiter": mit, "maxfun": 30 * mit,
                                        "ftol": 1e-18, "gtol": 1e-14})
                h2 = project_feasible(res.x, n)
                c2 = c5(h2)
                if c2 < best - 1e-12:
                    h = h2
                    best = c2
                    improved = True
            except Exception:
                pass
        if not improved and cyc > 0:
            break
        cyc += 1
    return h, best

def slsqp_obj(x, n, alpha):
    C = compute_C(x)
    a = alpha * C
    m = a.max()
    e = np.exp(a - m)
    return float((2.0 / n) * (m + np.log(e.sum())) / alpha)

def slsqp_jac(x, n, alpha):
    _, g = smooth_grad_data(x, n, alpha)
    return g

def _eq(x, n):
    return float(x.sum() - 0.5 * n)

def _eq_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, deadline):
    cur = c5(h)
    if deadline - time.time() < 18:
        return h, cur
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _eq, "jac": _eq_jac, "args": (n,)}]

                       options={"maxiter": mit, "ftol": 1e-18})

        c2 = c5(h2)
        return h2, c2
    except Exception:

def apply_pair(C, h, U, p, q, d, n):
    e1 = p + n
    e2 = (n - 1 - p) + n
    C[p:e1] += d * U
    C[n - 1 - p:e2] -= d * h
    d2 = d * d
    C[n - 1] -= 2.0 * d2
    C[n - 1 + p - q] += d2
    C[n - 1 + q - p] += d2

def reverse_pair(C, h, U, p, q, d, n):
    e1 = p + n
    e2 = (n - 1 - p) + n
    C[p:e1] -= d * U
    C[n - 1 - p:e2] += d * h
    d2 = d * d
    C[n - 1] += 2.0 * d2
    C[n - 1 + p - q] -= d2
    C[n - 1 + q - p] -= d2

def pair_polish(h, n, deadline, seed):
    C = compute_C(h)
    bm = C.max()
    U = np.empty(n)
    np.subtract(1.0, h[::-1], out=U)
    fr = [0.5, 0.2, 0.08, 0.03, 0.012, 0.004, 0.0015,
          0.0006, 0.00025, 0.0001, 4.0e-5, 1.5e-5, 5.0e-6]
    als = [2000.0, 20000.0, 200000.0, 1000000.0]
    K = 70
    ai = 0
    stale = 0
    acc = 0
    tend = deadline - 8
    while time.time() < tend:
        C, g = smooth_grad_data(h, n, als[ai])
        bm = C.max()
        np.subtract(1.0, h[::-1], out=U)
        order = np.argsort(g)
        down = order[:K]
        up = order[-K:]
        imp = False
        for f in fr:
            if time.time() > tend:
                break
            ppass = False
            for i in down:
                if time.time() > tend:
                    return h, bm
                for j in up:
                    if i == j:
                        continue
                    md = min(1.0 - h[i], h[j])
                    d = f * md
                    if d <= 1e-8:
                        continue
                    apply_pair(C, h, U, i, j, d, n)
                    nm = C.max()
                    if nm < bm - 1e-13:
                        h[i] += d
                        h[j] -= d
                        np.subtract(1.0, h[::-1], out=U)
                        bm = nm
                        imp = True

        if not imp:

            stale += 1
            if stale >= 2:

            stale = 0

    return h, C.max()
