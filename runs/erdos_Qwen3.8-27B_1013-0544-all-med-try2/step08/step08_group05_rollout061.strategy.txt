class OptimizationScheduler:
    """Manages optimization schedules with intelligent restarts"""
    def __init__(self, initial_budget=1100):
        self.budget = initial_budget
        self.current_budget = initial_budget
        
    def phase_schedule(self, time_remaining):
        if time_remaining > 0.5 * self.budget:
            return "global_peak", {"n": 100, "restarts": 10, "samples": 50}
        elif time_remaining > 0.3 * self.budget:
            return "local_opt_alpha", {"n": 150, "restarts": 5, "samples": 20}
        elif time_remaining > 0.1 * self.budget:
            return "local_opt_beta", {"n": 200, "restarts": 3, "samples": 15}
        else:
            return "final_click", {"n": 250, "restarts": 1, "samples": 10}
    
    def point_quality(self, h):
        return compute_erdos_objective(h, len(h))
    
    def decision_heuristic(self, current_score, neighbors_quality):
        if len(neighbors_quality) == 0:
            return "global_explore"
        if min(neighbors_quality) > current_score * 0.95:
            return "local_search"
        else:
            return "global_explore"
