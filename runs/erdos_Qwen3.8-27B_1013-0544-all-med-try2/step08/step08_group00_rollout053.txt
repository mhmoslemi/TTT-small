from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    """Gradient of f_k(h) = sum h[i+k]*(1-h[i])*dx for shift k."""
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            g[:nv] -= h[k:] * dx
            g[k:] += (1.0 - h[:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            g[kk:] -= h[:nv] * dx
            g[:nv] += (1.0 - h[kk:]) * dx
    return g

    """Find s = argmin <g,s> s.t. 0<=s<=1, sum(s)=n/2."""
    target = n / 2.0
    idx = np.argsort(g)
    k_full = int(np.floor(target))
    frac = target - k_full
    s = np.zeros(n)
    s[idx[:k_full]] = 1.0
    if frac > 1e-10:
        s[idx[k_full]] = frac
    return s

    """Frank-Wolfe method for min max_k f_k(h)."""
    best_c = compute_c5(h, n)
    it = 0
    while it < max_iters and time.time() < deadline:
        c = compute_c5_vec(h, n)
        k_star = int(np.argmax(c)) - (n - 1)  # convert to actual shift
        g = grad_active(h, n, k_star)
        s = fw_direction(g, n)

            h = h_new

Now let me think about the local search. The key insight is: for a pair (i,j), we want to find the optimal δ that minimizes max_k f_k. Since max_k f_k is the max of many linear functions of δ (for fixed i,j), the minimum over δ of the max is found at a "breakpoint" - where two f_k values cross.

def _smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-13:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - x[k:])
            g[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - x[:nv])
            g[:nv] += wm * (-x[kk:])
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def grad_guided_local(h, n, deadline, k_top=40):
    best_c = compute_c5(h, n)
    alphas = [10000, 50000, 200000, 1000000]
    for alpha in alphas:
        while time.time() < deadline:
            c = compute_c5_vec(h, n)
            top_m = np.argsort(c)[-5:]
            w = np.exp(c[top_m] - c[top_m].max())
            w /= w.sum()
            g = np.zeros(n)
            for wi, mi in zip(w, top_m):
                kv = int(mi) - (n - 1)
                g += wi * grad_active(h, n, kv)
            up = np.argsort(g)[-k_top:]
            down = np.argsort(g)[:k_top]
            improved = False
            for ii in down:
                if time.time() > deadline:
                    return h, best_c
                for jj in up:
                    if ii == jj:
                        continue
                    if h[ii] + 0.001 <= 1.0 and h[jj] - 0.001 >= 0.0:
                        for step in [0.001, 0.0005, 0.0002, 0.0001]:
                            if h[ii] + step <= 1.0 and h[jj] - step >= 0.0:

                                h[jj] -= step

                                    improved = True
                                else:
                                    h[ii] -= step
                                    h[jj] += step
                                if improved:
                                    break
            if not improved:
                break
    return h, best_c

def local_pair_search(h, n, deadline, radius=80):

    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]

    # Phase 3: SLSQP polish
    t3 = min(time.time() + 40, deadline - 250)
    if t3 > time.time() + 3:
        h3, c3 = slsqp_optimize(best_h, best_n, t3, alpha=5000000, maxiter=3000)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Gradient-guided local
    t4 = min(time.time() + 60, deadline - 150)
    if t4 > time.time() + 3:
        h4, c4 = grad_guided_local(best_h, best_n, t4, k_top=35)
        if c4 < best_c:
            best_c, best_h = c4, h4

    # Phase 5: Try n=500
    if time.time() < deadline - 120:
        n_new = 500
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_new)
        t5a = min(time.time() + 80, deadline - 80)
        if t5a > time.time() + 5:
            h5, c5 = lbfgs_optimize(h_interp, n_new, t5a, max_iters=300)
            if time.time() < deadline - 40:
                h5, c5 = slsqp_optimize(h5, n_new, time.time() + 20, alpha=10000000, maxiter=2000)
            if c5 < best_c:
                best_c, best_h, best_n = c5, h5, n_new

    # Phase 6: Try n=800
    if time.time() < deadline - 100:
        n_new = 800
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_new)
        t6a = min(time.time() + 60, deadline - 50)
        if t6a > time.time() + 5:
            h6, c6 = lbfgs_optimize(h_interp, n_new, t6a, max_iters=250)
            if c6 < best_c:
                best_c, best_h, best_n = c6, h6, n_new

    # Phase 7: Intensive local pair search on best
    if time.time() < deadline - 15:
        best_h, best_c = local_pair_search(best_h, best_n, deadline - 10, radius=60)

    # Phase 8: Final SLSQP
    if time.time() < deadline - 3:
        hf, cf = slsqp_optimize(best_h, best_n, deadline - 1, alpha=20000000, maxiter=1500)
        if cf < best_c:
            best_c, best_h = cf, hf

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
