def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):

        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:

            grad[0:nv] += wm * (1.0 - h[k:n])

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break

    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n

            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    sh = alpha * (c - m)

    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

    return h, best

def pair_search(h, n, deadline, deltas, radius):
    best = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best - 1e-14:
                            best = c
                            imp = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best - 1e-14:
                            best = c
                            imp = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not imp:
                break
    return h, best

def gradient_pair(h, n, deadline):
    best = compute_c5(h, n)
    alphas = [500, 2000, 10000]
    sizes = [0.005, 0.002, 0.001]
    for alpha, sz in zip(alphas, sizes):
        while time.time() < deadline:
            imp = False
            g = grad_smooth_max(h, n, alpha)
            order = np.argsort(g)
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return h, best
                ji = order[i]
                ki = order[n - 1 - i]
                mv = sz
                if h[ki] >= mv and h[ji] + mv <= 1.0:
                    h[ki] -= mv
                    h[ji] += mv
                    c = compute_c5(h, n)
                    if c < best - 1e-14:
                        best = c
                        imp = True
                    else:
                        h[ki] += mv
                        h[ji] -= mv
            if not imp:
                break
    return h, best

def anneal_pair(h, n, deadline, T0=0.001, T1=0.00001, deltas=None):
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005]
    best = compute_c5(h, n)
    best_h = h.copy()
    n_iter = 0
    while time.time() < deadline:
        T = T0 * (T1 / T0) ** (min(1.0, n_iter / max(1, deadline - time.time())))
        if T < 1e-7:
            T = 1e-7
        d = deltas[n_iter % len(deltas)]
        imp = False
        for _ in range(n // 5):
            if time.time() > deadline:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                h[i] += d
                h[j] -= d
                c = compute_c5(h, n)
                if c < best - 1e-14:
                    best = c
                    best_h = h.copy()
                    imp = True
                elif np.random.rand() < np.exp(-(c - best) / T):
                    imp = True
                else:
                    h[i] -= d
                    h[j] += d
        n_iter += 1
        if not imp and n_iter > 5:
            break
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(300, 0.5)
        n = 300

    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B
    t1 = min(t0 + 150, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, n, t1, maxit=150)

    # Phase 2: Gradient-guided pair search
    if time.time() < deadline - 350:
        t2 = min(time.time() + 50, deadline - 350)
        h2, c2 = gradient_pair(best_h, n, t2)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Extended pair search (main workhorse)
    if time.time() < deadline - 40:
        deltas_coarse = [0.02, 0.01, 0.005]
        t3a = min(time.time() + 120, deadline - 200)
        best_h, best_c = pair_search(best_h, n, t3a, deltas_coarse, radius=100)

    if time.time() < deadline - 30:
        deltas_fine = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        t3b = min(time.time() + 250, deadline - 30)
        best_h, best_c = pair_search(best_h, n, t3b, deltas_fine, radius=80)

    # Phase 4: Annealed search to escape local optima
    if time.time() < deadline - 40:
        t4 = min(time.time() + 60, deadline - 40)
        h4, c4 = anneal_pair(best_h, n, t4)
        if c4 < best_c:
            best_c = c4
            best_h = h4
        # Re-run pair search after annealing
        if time.time() < deadline - 20:
            best_h, best_c = pair_search(best_h, n, time.time() + min(40, deadline - time.time() - 20),
                                         [0.001, 0.0005, 0.0002, 0.0001], radius=60)

    # Phase 5: Final L-BFGS-B polish
    if time.time() < deadline - 5:
        best_h, best_c = lbfgs_phase(best_h, n, deadline - 3, maxit=100, a0=100000, a1=1000000)

    # Phase 6: Final pair search
    if time.time() < deadline - 2:
        best_h, best_c = pair_search(best_h, n, deadline - 1,
                                     [0.0005, 0.0002, 0.0001, 0.00005], radius=50)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
