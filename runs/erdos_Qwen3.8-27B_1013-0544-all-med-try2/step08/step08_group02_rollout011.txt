import time
import math
import numpy as np
from scipy.optimize import minimize


def c5(h):
    a = np.asarray(h, float)
    return float(np.max(np.correlate(a, 1.0 - a, mode="full")) * 2.0 / len(a))


def cval(h, n):
    u = 1.0 - h
    off = n - 1
    c = np.empty(2 * n - 1)
    c[off] = np.dot(h, u)
    for k in range(1, n):
        nv = n - k
        c[off + k] = np.dot(h[:nv], u[k:])
        c[off - k] = np.dot(h[k:], u[:nv])
    return c


def smc(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + math.log(float(np.sum(np.exp(sh - sm))))) / alpha


def smo(x, n, alpha):
    return (2.0 / n) * smc(cval(x, n), alpha)


def smg(x, n, alpha):
    h = x
    c = cval(h, n)
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    g = np.zeros(n)
    off = n - 1
    for idx, wm in enumerate(w):
        if wm < 1e-14:
            continue
        k = idx - off
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:n])
            g[k:n] += -wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            g[kk:n] += wm * (1.0 - h[:nv])
            g[:nv] += -wm * h[kk:n]
    return (2.0 / n) * g


def po(x, n, alpha, mu):
    s = x.sum() - n / 2.0
    return smo(x, n, alpha) + mu * s * s


def pg(x, n, alpha, mu):
    g = smg(x, n, alpha)
    s = x.sum() - n / 2.0
    return g + 2.0 * mu * s


def so(x, n, alpha):
    return smo(x, n, alpha)


def sg(x, n, alpha):
    return smg(x, n, alpha)


def eq(x, n):
    return x.sum() - n / 2.0


def ej(x, n):
    return np.ones(n)


def proj(h, n):
    t = n / 2.0
    h = np.clip(np.asarray(h, float), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > t:
            lo = mid
        else:
            hi = mid
    out = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(8):
        d = t - out.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            msk = (1.0 - out) > 1e-10
        else:
            msk = out > 1e-10
        if msk.sum():
            out[msk] += d / msk.sum()
            out = np.clip(out, 0.0, 1.0)
    d = t - out.sum()
    msk = (out > 1e-8) & (out < 1.0 - 1e-8)
    if msk.sum() and abs(d) > 1e-14:
        out[msk] += d / msk.sum()
    return np.clip(out, 0.0, 1.0)


def basis(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    cols = [np.ones(n), x, x * x]
    for m in range(1, 5):
        cols.append(np.cos(m * np.pi * x))
        cols.append(np.sin(m * np.pi * x))
    F = np.column_stack(cols)
    for j in range(1, F.shape[1]):
        c = F[:, j] - F[:, j].mean()
        nrm = np.linalg.norm(c)
        if nrm > 1e-12:
            F[:, j] = c / nrm
    return F


def bz(p, F):
    z = np.clip(p[0] * (F.dot(p[1:])), -12.0, 12.0)
    return 1.0 / (1.0 + np.exp(-z))


def bo(p, n, F):
    h = bz(p, F)
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    alpha = 4000.0
    sh = alpha * (c - m)
    sm = sh.max()
    v = (2.0 / n) * (m + (sm + math.log(float(np.sum(np.exp(sh - sm))))) / alpha)
    s = h.sum() - n / 2.0
    return v + 10.0 * s * s


def pbounds(d):
    return [(0.2, 8.0)] + [(-8.0, 8.0)] * d


def cold(h, n, dl):
    h = proj(h, n)
    bh = h.copy()
    bc = c5(h)
    bounds = [(0.0, 1.0)] * n
    base = 65 if n > 800 else 90 if n > 350 else 115
    stages = [
        (100.0, 1.0, 1.0),
        (1000.0, 1.0, 1.0),
        (5000.0, 10.0, 0.9),
        (20000.0, 50.0, 0.8),
        (100000.0, 200.0, 0.7),
        (500000.0, 1000.0, 0.6),
        (2000000.0, 5000.0, 0.5),
    ]
    for alpha, mu, f in stages:
        if time.time() > dl - 12:
            break
        try:
            r = minimize(po, h, jac=pg, args=(n, alpha, mu),
                         method="L-BFGS-B", bounds=bounds,
                         options={"maxiter": int(base * f), "ftol": 1e-15,
                                  "gtol": 1e-13, "maxfun": int(base * f) * 40})
            hn = proj(r.x, n)
            cn = c5(hn)
            if cn < bc:
                bh = hn.copy()
                bc = cn
            h = hn
        except Exception:
            pass
    return bh, bc


def sls_final(h, n, dl, alpha, mit):
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": eq, "jac": ej, "args": (n,)}]
    try:
        r = minimize(so, h, jac=sg, args=(n, alpha), method="SLSQP",
                     bounds=bounds, constraints=cons,
                     options={"maxiter": mit, "ftol": 1e-13})
        hn = proj(r.x, n)
        return hn, c5(hn)
    except Exception:
        return h, c5(h)


def tobin(h, n, r):
    b = (np.asarray(h, float) > 0.5).astype(np.int8)
    t = n // 2
    c = int(b.sum())
    if c > t:
        idx = np.flatnonzero(b)
        b[r.choice(idx, c - t, replace=False)] = 0
    elif c < t:
        idx = np.flatnonzero(1 - b)
        b[r.choice(idx, t - c, replace=False)] = 1
    return b


def cini(h, n):
    u = 1 - h
    off = n - 1
    c = np.zeros(2 * n - 1, np.int32)
    c[off] = int(np.dot(h, u))
    for k in range(1, n):
        nv = n - k
        c[off + k] = int(np.dot(h[:nv], u[k:]))
        c[off - k] = int(np.dot(h[k:], u[:nv]))
    return c


def pre(n):
    i = np.arange(n)[:, None]
    j = np.arange(n)[None, :]
    off = n - 1
    return (off + j - i).astype(np.intp), (off + i - j).astype(np.intp)


def flip(C, h, z, f, s, off, i, new):
    if new:
        C[f[i]] += z
        C[s[i]] -= h
    else:
        C[f[i]] -= z
        C[s[i]] += h
    C[off] = 0
    h[i] = new
    z[i] = 1 - new


def bw(C, h, z, f, s, off, i, j):
    oi = h[i]
    oj = h[j]
    if oi == oj:
        return None
    flip(C, h, z, f, s, off, i, 1 - oi)
    flip(C, h, z, f, s, off, j, 1 - oj)
    return oi, oj


def bu(C, h, z, f, s, off, i, j, sav):
    oi, oj = sav
    flip(C, h, z, f, s, off, j, oj)
    flip(C, h, z, f, s, off, i, oi)


def ms(C):
    return int(C.max()), int(np.dot(C, C))


def prop(h, z, C, off, m, n, r):
    if m > 0 and r.random() < 0.75:
        idxs = np.where(C >= max(0, m - 1))[0]
        if idxs.size:

                    if d > 0:
                        i = r.randint(n - d)
                        if h[i] and z[i + d]:
                            return i, i + d
                    else:
                        kk = -d
                        i = r.randint(kk, n)
                        if h[i] and z[i - kk]:
                            return i, i - kk
    for _ in range(30):
        i = r.randint(n)
        j = r.randint(n)
        if h[i] and not h[j]:
            return i, j
        if h[j] and not h[i]:
            return j, i
    for i in range(n):
        j = (i + 1) % n
        if h[i] and not h[j]:
            return i, j
    return 0, 1
