    amt = np.inf

def c5_vec(h, n):

def c5_max(h, n):
    return float(np.max(c5_vec(h, n)))

def build_L(h, n):

        L[:r, im] = 1.0 - h[k:]
        L[k:, im] = -h[:r]
    return L * dx

            break
    r = A @ w
    r -= r.mean()
    return -r, float(r @ r)

def line_search(h, p, n, base):
    p = np.asarray(p, dtype=float)
    p = p - p.mean()
    amp = float(np.max(np.abs(p)))
    if amp < 1e-13:
        return h, base
    amt = np.inf
    pos = p > 1e-12
    if np.any(pos):
        amt = min(amt, float(np.min((1.0 - h[pos]) / p[pos])))
    neg = p < -1e-12
    if np.any(neg):
        amt = min(amt, float(np.min(h[neg] / (-p[neg]))))
    if np.isfinite(amt) and amt > 1e-13:
        fr = (0.75, 0.5, 0.3, 0.15, 0.075, 0.037, 0.012, 0.004)
        alph = [amt * f for f in fr]
        proj = False
    else:
        fr = (0.25, 0.125, 0.062, 0.03, 0.015, 0.007, 0.003, 0.001)
        w = 1.0 / amp
        alph = [w * f for f in fr]
        proj = True
    bh, bb = h, base
    for s in alph:
        if s <= 1e-14:
            continue
        if proj:
            h2 = project_feasible(h + s * p, n)
        else:
            h2 = h + s * p
            sc = h2.sum() - 0.5 * n
            if abs(sc) > 1e-12:
                h2 = h2 - sc / n
            if h2.min() < -1e-11 or h2.max() > 1.0 + 1e-11:
                h2 = project_feasible(h2, n)
        c = c5_max(h2, n)
        if c < bb - 1e-14:
            bb, bh = c, h2
    return bh, bb

def pair_sweep(h, c, cur, L, deadline, passes=1, radius=None):
    n = len(h)
    i0 = n - 1
    dx = 2.0 / n
    imp = False
    for _ in range(int(passes)):
        any_imp = False
        for i in range(n - 1):
            if time.time() > deadline:
                return h, c, cur, imp
            hi = n if radius is None else min(n, i + int(radius) + 1)
            for j in range(i + 1, hi):
                if time.time() > deadline:
                    return h, c, cur, imp
                Dp = min(1.0 - h[i], h[j])
                Dn = min(h[i], 1.0 - h[j])
                if Dp + Dn <= 1e-12:
                    continue
                D = L[i] - L[j]
                dist = j - i
                ip = i0 + dist
                im = i0 - dist
                cand = (-Dn, -0.5 * Dn, -0.2 * Dn, -0.05 * Dn, 0.0,
                        0.05 * Dp, 0.2 * Dp, 0.5 * Dp, Dp)
                small = min(2.5e-3, 0.1 * max(Dp, Dn))
                if small > 1e-15:
                    for f in (1.0, 0.25, 0.05):
                        cand += (-small * f, small * f)
                z = -0.5 * D[ip]
                if -Dn <= z <= Dp:
                    cand = cand + (z,)
                z = -0.5 * D[im]
                if -Dn <= z <= Dp:
                    cand = cand + (z,)
                den = D[ip] - D[im]
                if abs(den) > 1e-14:
                    zi = (c[im] - c[ip]) / den
                    if -Dn <= zi <= Dp:
                        cand = cand + (zi,)
                da = np.unique(np.round(np.clip(np.asarray(cand, dtype=float), -Dn, Dp), 12))
                base = c + da[:, None] * D[None, :]
                vals = base.max(axis=1)
                vals = np.maximum(vals, c[i0] - 2.0 * dx * da * da)
                vals = np.maximum(vals, c[ip] + D[ip] * da + dx * da * da)
                vals = np.maximum(vals, c[im] + D[im] * da + dx * da * da)
                bi = int(np.argmin(vals))
                dd = float(da[bi])
                if abs(dd) < 1e-15:
                    continue
                newi = h[i] + dd
                newj = h[j] - dd
                if newi < -1e-11 or newi > 1.0 + 1e-11 or newj < -1e-11 or newj > 1.0 + 1e-11:
                    continue
                oldi, oldj = h[i], h[j]
                oldcur = cur
                h[i] = newi
                h[j] = newj
                c += dd * D
                c[i0] -= 2.0 * dx * dd * dd
                c[ip] += dx * dd * dd
                c[im] += dx * dd * dd
                nw = float(c.max())
                if nw < oldcur - 1e-13:
                    cur = nw
                    L = build_L(h, n)
                    any_imp = True
                    imp = True
                else:
                    h[i] = oldi
                    h[j] = oldj
                    c -= dd * D
                    c[i0] += 2.0 * dx * dd * dd
                    c[ip] -= dx * dd * dd
                    c[im] -= dx * dd * dd
        if not any_imp:
            break
    return h, c, cur, imp

def perturb_h(h, amp, rng):
    n = len(h)
    x = np.linspace(0.0, 2.0, n)
    s = np.zeros(n, dtype=float)
    for f in range(1, 6):
        s += (amp / (f + 0.5)) * float(rng.normal(0.4, 0.8)) * np.sin(np.pi * f * x + float(rng.uniform(0.0, 2.0 * np.pi)))
    s += 0.25 * amp * rng.normal(0.0, 1.0, n)
    return project_feasible(h + s, n)

def smooth_polish(h, n, deadline, rng):
    cur = project_feasible(h, n)
    best = cur.copy()
    cc = c5_vec(cur, n)
    curc = float(cc.max())
    bestc = curc
    L = build_L(cur, n)
    no = 0
    while time.time() < deadline:
        if no < 5:
            beta, q, fq = 50.0, 24, 16
        elif no < 15:
            beta, q, fq = 150.0, 48, 24
        elif no < 30:
            beta, q, fq = 400.0, 80, 40
        elif no < 60:
            beta, q, fq = 1000.0, 96, 48
        else:
            beta, q, fq = 2500.0, 80, 32
        improved = False
        if time.time() < deadline - 2:
            p, nr = fw_direction(cc, L, fq)
            if nr > 1e-16:
                h2, c2 = line_search(cur, p, n, curc)
                if c2 < curc - 1e-14:
                    cur, curc = h2, c2
                    cc = c5_vec(cur, n)
                    L = build_L(cur, n)
                    improved = True
        if not improved and time.time() < deadline - 2:
            w = smooth_weights(cc, beta, q)
            p = L.T @ w
            p = p - p.mean()
            h2, c2 = line_search(cur, p, n, curc)
            if c2 < curc - 1e-14:
                cur, curc = h2, c2
                cc = c5_vec(cur, n)
                L = build_L(cur, n)
                improved = True
        if improved:
            no = 0
            if curc < bestc - 1e-14:
                best, bestc = cur.copy(), curc
            continue
        no += 1
        if no in (5, 12, 25, 50, 80) and time.time() < deadline - 35:
            rad = 120 if n <= 300 else 80
            h2, c2, c3, imp2 = pair_sweep(cur, cc, curc, L, deadline - 15, passes=1, radius=rad)
            if imp2 and c3 < curc - 1e-14:
                cur, curc, cc, L = h2, c3, c2, build_L(h2, n)
                no = 0
                if curc < bestc - 1e-14:
                    best, bestc = cur.copy(), curc
                continue
        if no > 70:
            amp = 0.004 + 0.005 * min(2.0, (no - 70) / 20.0)
            hp = perturb_h(best, amp, rng)
            if c5_max(hp, n) <= bestc + 0.05:
                cur = hp
                curc = c5_max(hp, n)
                cc = c5_vec(cur, n)
                L = build_L(cur, n)
                no = 25
        if curc > bestc + 0.10:
            cur = best.copy()
            curc = bestc
            cc = c5_vec(cur, n)
            L = build_L(cur, n)
            no = 0
    return best, bestc

def optimize_n(h, n, deadline, rng):
    h = project_feasible(h, n)
    best, bc = smooth_polish(h, n, deadline, rng)
    if time.time() < deadline - 8:
        nh = len(best)
        hh = project_feasible(best, nh)
        cc = c5_vec(hh, nh)
        lc = float(cc.max())
        LL = build_L(hh, nh)
        hh, cc, lc, imp = pair_sweep(hh, cc, lc, LL, deadline, passes=2, radius=(None if nh <= 300 else 100))
        if lc < bc:
            best, bc = hh, lc
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 3.0
    rng = np.random.RandomState(seed)
    try:
        h = np.asarray(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 300
        h = np.full(n, 0.5, dtype=float)
    if n % 2:
        oldn = n
        n += 1
        h = np.interp(np.linspace(0.0, 2.0, n), np.linspace(0.0, 2.0, oldn), h)
    h = project_feasible(h, n)
    best = h.copy()
    bestc = c5_max(best, n)

    reserve = max(70.0, 0.25 * budget_s)
    main_dl = min(deadline - reserve, t0 + max(60.0, 0.55 * budget_s))
    if main_dl < t0 + 10.0:
        main_dl = min(deadline - 5.0, t0 + max(5.0, 0.8 * budget_s))
    b, c = optimize_n(best, n, main_dl, rng)
    if c < bestc:
        best, bestc, n = b, c, len(b)

    alt_dl = deadline - 70.0
    tried = {n}
    mults = (0.5, 0.67, 0.8, 1.25, 1.5, 0.4, 1.0)
    caps = (150, 100, 90, 70, 55, 45, 45)
    for idx, mm in enumerate(mults):
        nt = int(round(n * mm))
        if nt % 2:
            nt += 1
        if nt < 40 or nt > 650 or nt in tried:
            continue
        tried.add(nt)
        if time.time() > alt_dl - 30:
            break
        rem = alt_dl - time.time() - 20
        alloc = min(caps[idx] if idx < len(caps) else 40, rem)
        if alloc < 25:
            break
        xo = np.linspace(0.0, 2.0, len(best))
        xn = np.linspace(0.0, 2.0, nt)
        hi = np.interp(xn, xo, best)
        hi = project_feasible(hi, nt)
        b, c = optimize_n(hi, nt, time.time() + alloc, rng)
        if c < bestc:
            best, bestc, n = b, c, len(b)

    if time.time() < deadline - 3:
        nh = len(best)
        hh = project_feasible(best, nh)
        cc = c5_vec(hh, nh)
        lc = float(cc.max())
        LL = build_L(hh, nh)
        hh, cc, lc, imp = pair_sweep(hh, cc, lc, LL, deadline,
                                      passes=(2 if time.time() < deadline - 35 else 1),
                                      radius=(None if nh <= 300 else 80))
        if lc < bestc:
            best, bestc = hh, lc

    n = len(best)
    h = project_feasible(best, n)
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - h.sum()
    if abs(d) > 0:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if mask.any():
            h[mask] += d / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    val = c5_max(h, n)
    return h, float(val), n
