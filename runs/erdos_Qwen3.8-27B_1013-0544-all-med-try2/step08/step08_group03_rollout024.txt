Let me try yet another approach: **randomized rounding + local search**.

1. Start with the current best h
2. Create a "binary candidate" by thresholding: h_bin[i] = 1 if h[i] > 0.5 else 0
3. Adjust to ensure exactly n/2 ones
4. Do local search on the binary solution: try swapping pairs of bits
5. Compare the binary solution with the fractional one
6. Use the better one as the starting point for further optimization

The binary local search is:
- For each pair (i, j) where h_bin[i]=0 and h_bin[j]=1:
  - Try swapping: h_bin[i]=1, h_bin[j]=0
  - If C₅ improves, accept

This is O(n^2) per full pass, but with n=500 that's 250000 swaps to try, each requiring O(n log n) correlation. That's too slow.

Instead, we can use the same gradient-guided approach:
- Compute the gradient of C₅
- Find the best swap (i=0 → 1, j=1 → 0) based on the gradient
- Execute and check

OK let me just go with a pragmatic approach. I'll combine:
1. L-BFGS-B with high alpha
2. Momentum PGD with Nesterov-like acceleration
3. A more efficient pair polish that uses the gradient to guide moves
4. A "binary rounding" exploration step

Let me also add one more idea: **multi-start**. Try a few different random perturbations of the initial solution and see if any leads to a better local minimum.

import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def grad_lag(h, n, k):
    nv = n - abs(k)
    g = np.zeros(n)
    if k >= 0:
        g[:nv] = 1.0 - h[k:]
        g[k:] -= h[:nv]
    else:
        kk = -k
        g[kk:] = 1.0 - h[:nv]
        g[:nv] -= h[kk:]
    return g

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    sh = s.max()
    val = m + (sh + np.log(np.sum(np.exp(s - sh)))) / alpha
    val += 5.0 * (x.sum() - n / 2.0) ** 2
    return max(val, 0.0)

def smooth_grad(x, n, alpha, top_k=100):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    sh = s.max()
    w = np.exp(s - sh)
    w /= (w.sum() + 1e-300)
    dx = 2.0 / n
    top_idx = np.argsort(w)[-top_k:]
    g = np.zeros(n)
    for i in top_idx:
        if w[i] < 1e-12:
            continue
        k = i - (n - 1)
        g += w[i] * dx * grad_lag(x, n, k)
    g += 10.0 * (x.sum() - n / 2.0)
    return g

def momentum_pgd(h, n, deadline, alpha=500.0, top_k=100, mu=0.7):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    step = 0.01
    prev_h = h.copy()
    prev_g = None
    stall = 0
    while time.time() < deadline:
        g = smooth_grad(h, n, alpha, top_k)
        if prev_g is not None:
            s_vec = h - prev_h
            y_vec = g - prev_g
            sy = s_vec @ y_vec
            ss = s_vec @ s_vec
            if sy > 1e-12:
                step = ss / sy
            step = float(np.clip(step, 0.001, 0.04))
        prev_h = h.copy()
        prev_g = g.copy()
        v = mu * v - step * g
        h_new = project_feasible(h + v, n)
        accepted = False
        for _ in range(10):
            c_new = compute_c5(h_new, n)
            if c_new < best_c + 1e-12:
                accepted = True
                break
            v *= 0.4
            h_new = project_feasible(h + v, n)
        if not accepted:
            stall += 1
            if stall > 200:
                break
            continue
        stall = 0
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def pair_polish(h, n, deadline):
    best_c = compute_c5(h, n)
    deltas = [0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_c5_vec(h, n)
            m = c.max()
            s = 200.0 * (c - m)
            s -= s.max()
            w = np.exp(s)
            w /= (w.sum() + 1e-300)
            dx = 2.0 / n
            top_idx = np.argsort(w)[-40:]
            g = np.zeros(n)
            for i in top_idx:
                if w[i] < 1e-10:
                    continue
                k = i - (n - 1)
                g += w[i] * dx * grad_lag(h, n, k)
            vi = np.where((g > 0.0005) & (h > d * 0.8))[0]
            vj = np.where((g < -0.0005) & (h < 1.0 - d * 0.8))[0]
            if len(vi) == 0 or len(vj) == 0:
                break
            ic = vi[np.argsort(-g[vi])][:4]
            jc = vj[np.argsort(g[vj])][:4]
            for i in ic:
                for j in jc:
                    if i == j or time.time() > deadline:
                        continue
                    h[i] -= d
                    h[j] += d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 2
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 500
        h_init = np.zeros(n_init)
        half = n_init // 2
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        positions = np.floor(np.arange(half) * phi).astype(int) % n_init
        h_init[positions] = 1.0
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init
    # Phase 1: L-BFGS-B with increasing alpha
    t1 = min(t0 + 100, deadline - 350)
    for alpha in [300, 1500, 8000, 30000]:
        if time.time() > t1:
            break
        bounds = [(0.0, 1.0)] * best_n
        try:
            res = minimize(smooth_obj, best_h, jac=smooth_grad,
                          args=(best_n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-16})
            h_new = project_feasible(res.x, best_n)
            c_new = compute_c5(h_new, best_n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    # Phase 2: Momentum PGD
    if time.time() < deadline - 250:
        t2 = min(time.time() + 90, deadline - 160)
        best_h, best_c = momentum_pgd(best_h, best_n, t2, alpha=600.0, top_k=100)
    # Phase 3: Try n=400
    if time.time() < deadline - 180:
        n_try = 400
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        bounds = [(0.0, 1.0)] * n_try
        try:
            res = minimize(smooth_obj, h_int, jac=smooth_grad,
                          args=(n_try, 8000), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 200, 'ftol': 1e-15})
            h_t = project_feasible(res.x, n_try)
            c_t = compute_c5(h_t, n_try)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try
        except Exception:
            pass
    # Phase 4: Try n=600
    if time.time() < deadline - 120:
        n_try = 600
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        bounds = [(0.0, 1.0)] * n_try
        try:
            res = minimize(smooth_obj, h_int, jac=smooth_grad,
                          args=(n_try, 8000), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 150, 'ftol': 1e-15})
            h_t = project_feasible(res.x, n_try)
            c_t = compute_c5(h_t, n_try)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try
        except Exception:
            pass
    # Phase 5: Another PGD pass
    if time.time() < deadline - 60:
        t5 = min(time.time() + 30, deadline - 30)
        best_h, best_c = momentum_pgd(best_h, best_n, t5, alpha=1000.0, top_k=120, mu=0.5)
    # Phase 6: Pair polish
    if time.time() < deadline - 1:
        best_h, best_c = pair_polish(best_h, best_n, deadline - 1)
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
