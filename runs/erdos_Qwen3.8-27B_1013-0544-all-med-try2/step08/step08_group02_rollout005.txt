def top_grad(h, n, alpha, q):
    F = np.correlate(h, 1-h, mode='full')
    # select top q indices. Use argpartition.
    if len(F) <= q: idx = np.arange(len(F))
    else: idx = np.argpartition(F, -q)[-q:]
    M = F[idx].max()

import numpy as np
import time

DELTA_FRACS = (0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625)
ABS_STEPS = (0.01, 0.02, 0.05, 0.1)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = h < 1.0
            room = 1.0 - h[mask]
            tot = room.sum()
            if tot < 1e-14:
                break
            h[mask] += diff * room / tot
        else:
            mask = h > 0.0
            vals = h[mask]
            tot = vals.sum()
            if tot < 1e-14:
                break
            h[mask] += diff * vals / tot
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(idx):
            i = int(idx[0])
            adj = diff
            if adj > 1.0 - h[i]:
                adj = 1.0 - h[i]
            if adj < -h[i]:
                adj = -h[i]
            h[i] += adj
    return h


def raw_of(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))


def sync_state(h, n):
    F = np.correlate(h, 1.0 - h, mode="full")
    c0 = float(F[n - 1])
    pos = np.full(n, -1e300)
    neg = np.full(n, -1e300)
    if n > 1:
        P1 = float(np.dot(h[:n - 1], 1.0 - h[1:n]))
        N1 = float(np.dot(h[1:n], 1.0 - h[:n - 1]))
        if abs(F[n] - P1) <= abs(F[n - 2] - P1):
            pos[1:] = F[n:2 * n - 1]
            neg[1:] = F[:n - 1][::-1]
        else:
            pos[1:] = F[:n - 1][::-1]
            neg[1:] = F[n:2 * n - 1]
    raw = c0
    if n > 1:
        a = float(np.max(pos[1:]))
        b = float(np.max(neg[1:]))
        if a > raw:
            raw = a
        if b > raw:
            raw = b
    return pos, neg, c0, raw


def make_deltas(a, b):
    c = a + b
    low = max(-a, c - 1.0)
    high = min(1.0 - a, c)
    if high - low <= 1e-11:
        return ()
    w = high - low
    vals = [0.5 * (b - a)]
    for f in DELTA_FRACS:
        vals.append(f * w)
        vals.append(-f * w)
    for s in ABS_STEPS:
        vals.append(s)
        vals.append(-s)
    vals.append(low)
    vals.append(high)
    out = []
    for v in vals:
        if v < low:
            v = low
        elif v > high:
            v = high
        if abs(v) < 1e-10:
            continue
        ok = True
        for u in out:
            if abs(u - v) < 1e-8:
                ok = False
                break
        if ok:
            out.append(float(v))
    return out


def fill_diff(h, p, q, diff_pos, diff_neg, n):
    r = n - p - 1
    if r > 0:
        diff_pos[1:r + 1] += 1.0 - h[p + 1:n]
    if p > 0:
        diff_pos[1:p + 1] += -h[p - 1::-1]
    r = n - q - 1
    if r > 0:
        diff_pos[1:r + 1] -= 1.0 - h[q + 1:n]
    if q > 0:
        diff_pos[1:q + 1] += h[q - 1::-1]

    r = p
    if r > 0:
        diff_neg[1:r + 1] += 1.0 - h[p - 1::-1]
    r = n - p - 1
    if r > 0:
        diff_neg[1:r + 1] += -h[p + 1:n]
    r = q
    if r > 0:
        diff_neg[1:r + 1] -= 1.0 - h[q - 1::-1]
    r = n - q - 1
    if r > 0:
        diff_neg[1:r + 1] += h[q + 1:n]


def eval_delta(pos, neg, c0, diff_pos, diff_neg, d, lin0, delta):
    d2 = delta * delta
    val = c0 + lin0 * delta - 2.0 * d2

    arr = pos + delta * diff_pos
    m = float(np.max(arr))
    if d:
        vd = pos[d] + delta * diff_pos[d] + d2
        if vd > m:
            m = vd
    if m > val:
        val = m

    arr = neg + delta * diff_neg
    m2 = float(np.max(arr))
    if d:
        vd = neg[d] + delta * diff_neg[d] + d2
        if vd > m2:
            m2 = vd
    if m2 > val:
        val = m2
    return val


def try_pair_update(h, pos, neg, c0, raw, p, q):
    n = len(h)
    a = float(h[p])
    b = float(h[q])
    d = abs(p - q)
    if d == 0:
        return None
    deltas = make_deltas(a, b)
    if not deltas:
        return None
    diff_pos = np.zeros(n)
    diff_neg = np.zeros(n)
    fill_diff(h, p, q, diff_pos, diff_neg, n)
    lin0 = 2.0 * (b - a)
    best_d = 0.0
    best_val = raw
    for delta in deltas:
        v = eval_delta(pos, neg, c0, diff_pos, diff_neg, d, lin0, delta)
        if v < best_val - 1e-10:
            best_val = v
            best_d = delta
    if best_d == 0.0:
        return None
    d2 = best_d * best_d
    pos += best_d * diff_pos
    if d:
        pos[d] += d2
    neg += best_d * diff_neg
    if d:
        neg[d] += d2
    c0_new = c0 + lin0 * best_d - 2.0 * d2
    hp = a + best_d
    hq = b - best_d
    if hp < 0.0:
        hp = 0.0
    elif hp > 1.0:
        hp = 1.0
    if hq < 0.0:
        hq = 0.0
    elif hq > 1.0:
        hq = 1.0
    h[p] = hp
    h[q] = hq
    return c0_new, best_val


def choose_pairs(h, pos, neg, c0, raw, n, rng, max_pairs):
    pairs = []
    seen = set()
    if n < 2 or max_pairs <= 0:
        return pairs
    tol = max(0.5, 0.002 * raw)
    vals = np.empty(2 * n - 1)
    vals[0] = c0
    vals[1:n] = pos[1:]
    vals[n:2 * n - 1] = neg[1:]
    K = min(16, 2 * n - 2)
    if 2 * n - 1 <= K:
        tags = np.arange(2 * n - 1)
    else:
        tags = np.argpartition(vals, -K)[-K:]

    for t in tags:
        t = int(t)
        v = float(vals[t])
        if t == 0 or v < raw - tol:
            continue
        if t <= n - 1:
            is_pos = True
            r = t
        else:
            is_pos = False
            r = t - (n - 1)
        if r <= 0 or r >= n:
            continue
        m = n - r
        if m <= 0:
            continue
        if is_pos:
            contrib = h[:m] * (1.0 - h[r:r + m])
        else:
            contrib = h[r:r + m] * (1.0 - h[:m])
        topk = min(25, m)
        if topk <= 0:
            continue
        if m <= topk:
            topidx = np.arange(m)
        else:
            topidx = np.argpartition(contrib, -topk)[-topk:]
        take = min(4, len(topidx))
        for aidx in rng.choice(topidx, size=take, replace=False):
            i = int(aidx)
            p = i
            q = i + r
            if p > q:
                p, q = q, p
            key = p * n + q
            if key not in seen:
                seen.add(key)
                pairs.append((p, q))

    if len(pairs) < max_pairs:
        tries = 0
        while len(pairs) < max_pairs and tries < (max_pairs - len(pairs)) * 6 + 20:
            p = int(rng.randint(n))
            q = int(rng.randint(n))
            tries += 1
            if p == q:
                continue
            if p > q:
                p, q = q, p
            key = p * n + q
            if key not in seen:
                seen.add(key)
                pairs.append((p, q))
    return pairs


def pair_search(h, n, deadline, rng, max_pairs=500, max_stall=3, pass_time=8.0):
    pos, neg, c0, raw = sync_state(h, n)
    stall = 0
    while time.time() < deadline:
        pairs = choose_pairs(h, pos, neg, c0, raw, n, rng, max_pairs)
        if not pairs:
            break
        improved = False
        pstart = time.time()
        for p, q in pairs:
            if time.time() > deadline or time.time() - pstart > pass_time:
                break
            for attempt in range(2):
                res = try_pair_update(h, pos, neg, c0, raw, p, q)
                if res is None:
                    break
                c0, raw = res
                improved = True
        pos, neg, c0, raw = sync_state(h, n)
        if not improved:
            stall += 1
            if stall >= max_stall:
                break
        else:
            stall = 0
    return h, raw


def smooth_gradient(h, n, alpha, q):
    if n < 2:
        return np.zeros(n), 0.0
    pos, neg, c0, raw = sync_state(h, n)
    vals = np.empty(2 * n - 1)
    vals[0] = c0
    vals[1:n] = pos[1:]
    vals[n:2 * n - 1] = neg[1:]
    L = len(vals)
    if L <= q:
        idxs = np.arange(L)
    else:
        idxs = np.argpartition(vals, -q)[-q:]
    M = float(np.max(vals[idxs]))
    sh = alpha * (vals[idxs] - M)
    w = np.exp(sh - float(np.max(sh)))
    s = float(w.sum())
    if s <= 0.0:
        w = np.ones_like(w) / float(len(w))
    else:
        w = w / s
    g = np.zeros(n)
    for t, wi in zip(idxs, w):
        t = int(t)
        wi = float(wi)
        if wi < 1e-12:
            continue
        if t == 0:
            g += wi * (1.0 - 2.0 * h)
        elif t <= n - 1:
            r = t
            m = n - r
            if m > 0:
                g[:m] += wi * (1.0 - h[r:r + m])
                g[r:] += -wi * h[:m]
        else:
            r = t - (n - 1)
            m = n - r
            if m > 0:
                g[:m] += -wi * h[r:r + m]
                g[r:] += wi * (1.0 - h[:m])
    g -= g.mean()
    return g, raw


def gradient_polish(h, n, raw, deadline, alphas=(10.0, 40.0, 150.0), q=60, max_iter=600):
    for alpha in alphas:
        if time.time() > deadline:
            break
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            g, raw_cur = smooth_gradient(h, n, alpha, q)
            ginf = float(np.max(np.abs(g))) if len(g) else 0.0
            if ginf < 1e-12:
                break
            d = -g / ginf
            lam = 1.0
            posm = d > 1e-14
            if np.any(posm):
                lam = min(lam, float(np.min((1.0 - h[posm]) / d[posm])))
            negm = d < -1e-14
            if np.any(negm):
                lam = min(lam, float(np.min((-h[negm]) / (-d[negm]))))
            if lam <= 1e-12:
                break
            best_t = 0.0
            best_c = raw_cur
            best_h = None
            for t in (0.03, 0.015, 0.0075, 0.003, 0.001):
                if t > 0.95 * lam:
                    continue
                ht = h + t * d
                c = raw_of(ht, n)
                if c < best_c - 1e-11:
                    best_c = c
                    best_t = t
                    best_h = ht
            if best_t > 0.0:
                h[:] = best_h
                raw = best_c
            else:
                break
    return h, raw


def fourier_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for f in (1, 2, 3, 4, 5, 6):
        amp = 0.35 / (f + 1.0)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(np.pi * f * x + ph)
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 3.0
    rng = np.random.RandomState(int(seed))

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        h = fourier_start(500, rng)
        n = 500
    if n < 8:
        h = fourier_start(500, rng)
        n = 500
    h = project_feasible(h, n)

    pos, neg, c0, raw = sync_state(h, n)
    best_h = h.copy()
    best_raw = raw
    best_n = n
    best_c = best_raw * 2.0 / best_n

    remaining = deadline - time.time()
    if remaining <= 5.0:
        return best_h, best_c, best_n

    reserve = min(140.0, max(8.0, float(budget_s) * 0.15))
    main_deadline = max(t0 + 2.0, deadline - reserve)
    w_h = best_h.copy()
    w_raw = pair_search(w_h, best_n, main_deadline, rng, max_pairs=700, max_stall=3, pass_time=8.0)
    w_c = w_raw * 2.0 / best_n
    if w_c < best_c - 1e-12:
        best_h, best_raw, best_c = w_h, w_raw, w_c

    if time.time() < deadline - 30.0:
        g_h = best_h.copy()
        g_dead = min(deadline - 15.0, time.time() + min(90.0, (deadline - time.time()) * 0.6))
        g_raw = gradient_polish(g_h, best_n, best_raw, g_dead,
                                alphas=(10.0, 40.0, 150.0), q=60, max_iter=600)
        g_c = g_raw * 2.0 / best_n
        if g_c < best_c - 1e-12:
            best_h, best_raw, best_c = g_h, g_raw, g_c

    if time.time() < deadline - 8.0:
        p_h = best_h.copy()
        p_raw = pair_search(p_h, best_n, deadline - 3.0, rng, max_pairs=500, max_stall=2, pass_time=5.0)
        p_c = p_raw * 2.0 / best_n
        if p_c < best_c - 1e-12:
            best_h, best_raw, best_c = p_h, p_raw, p_c

    if time.time() < deadline - 50.0:
        for sigma in (0.02, 0.05, 0.08):
            if time.time() > deadline - 30.0:
                break
            s_h = project_feasible(best_h + sigma * rng.standard_normal(best_n), best_n)
            alloc = min(40.0, (deadline - time.time() - 15.0) * 0.5)
            if alloc < 8.0:
                break
            s_h, sr = pair_search(s_h, best_n, min(deadline - 10.0, time.time() + alloc),
                                  rng, max_pairs=300, max_stall=2, pass_time=4.0)
            sc = sr * 2.0 / best_n
            if sc < best_c - 1e-12:
                best_h, best_raw, best_c = s_h, sr, sc
        if time.time() < deadline - 30.0:
            f_h = fourier_start(best_n, rng)
            alloc = min(40.0, (deadline - time.time() - 15.0) * 0.5)
            if alloc > 10.0:
                f_h, fr = pair_search(f_h, best_n, min(deadline - 10.0, time.time() + alloc),
                                      rng, max_pairs=300, max_stall=2, pass_time=4.0)
                fc = fr * 2.0 / best_n
                if fc < best_c - 1e-12:
                    best_h, best_raw, best_c = f_h, fr, fc

    if time.time() < deadline - 80.0:
        base_n = best_n
        seen_n = set()
        for factor in (1.1, 1.25, 1.5, 2.0):
            if time.time() > deadline - 40.0:
                break
            nt = int(round(base_n * factor))
            if nt in seen_n:
                continue
            if nt <= best_n or nt < 50 or nt > 1000:
                continue
            seen_n.add(nt)
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, nt, endpoint=False)
            h_try = np.interp(x_new, x_old, best_h)
            h_try = project_feasible(h_try, nt)
            alloc = min(50.0, (deadline - time.time() - 25.0) * 0.5)
            if alloc < 10.0:
                break
            h_try, r_try = pair_search(h_try, nt, min(deadline - 15.0, time.time() + alloc),
                                       rng, max_pairs=400, max_stall=2, pass_time=5.0)
            c_try = r_try * 2.0 / nt
            if c_try < best_c - 1e-12:
                best_h, best_raw, best_n, best_c = h_try, r_try, nt, c_try

    best_h = project_feasible(best_h, best_n)
    final_raw = raw_of(best_h, best_n)
    final_c = final_raw * 2.0 / best_n
    return best_h, final_c, best_n
