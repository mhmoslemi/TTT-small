        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h

            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

def grad_smooth_max(h, n, alpha):
    c = compute_all_c(h, n)
    s = alpha * (c - c.max())
    mx = s.max()
    w = np.exp(s - mx)

        else:

                block = h[i:i+block_size].copy()
                # Generate a few random redistributions
                for _ in range(3):
                    new_block = block.copy()
                    # Move some mass from one position to another in the block
                    src = np.random.randint(block_size)
                    dst = np.random.randint(block_size)
                    if src == dst:
                        continue
                    amt = step * np.random.choice([-1, 1])

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k == 0:
            H[np.arange(n), np.arange(n)] += wm * (-2)
        else:
            kk = abs(k)
            nv = n - kk
            H[:nv, kk:kk+nv] += wm * (-1)
            H[kk:kk+nv, :nv] += wm * (-1)
    return H

def obj_pen(x, n, alpha, mu):
    c = compute_all_c(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(s))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def obj_pen_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def newton_method(h, n, deadline, alpha=50000, mu=1000, max_iters=200):
    """Newton method with line search for final optimization."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = obj_pen_grad(h, n, alpha, mu)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        
        H = hessian_smooth_max(h, n, alpha)
        H += 2.0 * mu * np.ones((n, n))
        
        # Regularize to ensure PD
        reg = 1e-6
        for _ in range(10):
            try:
                L = np.linalg.cholesky(H + reg * np.eye(n))
                d = np.linalg.solve(H + reg * np.eye(n), -g)
                break
            except np.linalg.LinAlgError:
                reg *= 10
        else:
            d = -g / g_norm
        
        # Line search
        improved = False
        step = 1.0
        for _ in range(30):
            h_new = project_feasible(h + step * d, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                improved = True
                break
            step *= 0.5
        if not improved:
            # Try gradient direction
            step = 0.01
            for _ in range(20):
                h_new = project_feasible(h - step * g / g_norm, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
                    break
                step *= 0.5
            else:
                break
    
    return best_h, best_c

def lbfgs_optimize(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)

    bounds = [(0.0, 1.0)] * n

def pair_search(h, n, deadline):
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - 40)
                hi = min(n, i + 41)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c
