from scipy.optimize import minimize

def compute_corr(h, n):
    """Compute correlation h * (1-h) using FFT for speed."""
    dx = 2.0 / n
    # np.correlate(h, 1-h, 'full') equivalent via cross-correlation
    # correlate(a, v)[k] = sum_n a[n+k] * conj(v[n])
    # For real: sum_n a[n+k] * v[n]
    # fftconvolve gives convolution: (a * v)[k] = sum_n a[n] * v[k-n]
    # cross-correlation = convolution with flipped v
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * dx

def compute_c5(h, n):
    return float(np.max(compute_corr(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_corr(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(val)

def smooth_max_grad(h, n, alpha):

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)

    dx = 2.0 / n

            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def penalty_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    # Use increasing alpha schedule
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    mus = [10.0, 50.0, 100.0, 500.0, 1000.0, 5000.0, 10000.0, 50000.0]
    for i in range(len(alphas)):

            break
        res = minimize(penalty_obj, h, jac=penalty_grad,

        if c_new < best_c - 1e-15:

    return h, best_c

        if diff > 0:

            if tot < 1e-14:

    if mx > 500:

from scipy.optimize import minimize
import time

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h, n)) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def smooth_max_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    return w

def smooth_max_grad(h, n, alpha):
    c = compute_corr(h, n)
    w = smooth_max_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

    c = compute_corr(x, n)
    val = smooth_max_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def pen_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    return smooth_max_val(c, alpha)

def sl_grad(x, n, alpha):
    return smooth_max_grad(x, n, alpha)

def sl_con(x, n):
    return float(x.sum() - n / 2.0)

def sl_jac(x, n):

    best_c = compute_c5(h, n)

                (20000, 1000), (100000, 5000), (500000, 10000),

    for alpha, mu in schedule:

        try:

                         method='L-BFGS-B', bounds=bounds,
