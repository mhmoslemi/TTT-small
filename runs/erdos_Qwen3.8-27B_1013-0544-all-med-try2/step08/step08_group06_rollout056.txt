def minimize_1d(B, D, dx, n):
    """Minimize max_l (B_l + t*D_l) over t in [0,1], where l=0 is special: B_0 + t - t²"""
    k0 = n - 1  # index of lag 0
    
    def f(t):
        vals = B + t * D
        vals[k0] = B[k0] + t - t*t
        return vals.max() * dx
    
    def subgrad(t):
        vals = B + t * D
        vals[k0] = B[k0] + t - t*t
        mx = vals.max()
        active = vals >= mx - 1e-12
        if active[k0]:
            # The quadratic term contributes (1-2t) to the subgradient
            pass
        # Subgradient is the convex hull of D_l for active l (and 1-2t for l=0)
        # For bisection, just use the D_l of the max (if unique)
        max_idx = np.argmax(vals)
        if max_idx == k0:
            return 1.0 - 2.0*t
        return D[max_idx]
    
    # Bisection on subgradient
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        g = subgrad(mid)
        if g > 0:
            hi = mid  # f is decreasing, move left
        elif g < 0:
            lo = mid  # f is increasing, move right
        else:
            break
    return (lo + hi) / 2

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, deadline, max_sweeps=5000, random_order=False, seed=42):
    """Coordinate descent for the Erdős overlap problem.
    Each 1D subproblem is convex (max of linear + one quadratic), solved by bisection."""
    dx = 2.0 / n
    k0 = n - 1
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_h = h.copy()

    # For coordinate i:
    #   lags where i-l is valid: l in [i-(n-1), i], clipped to [-(n-1), n-1]
    #   lags where i+l is valid: l in [-i, (n-1-i)], clipped to [-(n-1), n-1]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
            
        indices = np.random.permutation(n) if random_order else range(n)
        
        for i in indices:
            old = h[i]
            if old <= 1e-12 or old >= 1.0 - 1e-12:
                continue
            
            # Compute D vector
            # D[k] where k = lag + (n-1)
            # D[k] = (1-h[i-lag])*[0<=i-lag<n] + (-h[i+lag])*[0<=i+lag<n]
            # But for lag=0 (k=k0): D[k0] = 0 (handle separately as 1-2t)
            
            # Vectorized computation
            lags = np.arange(-(n-1), n)  # shape (2n-1,)
            
            # i - lags in [0, n)
            idx1 = i - lags
            valid1 = (idx1 >= 0) & (idx1 < n)

            idx2 = i + lags
            valid2 = (idx2 >= 0) & (idx2 < n)
            
            D = np.zeros(2*n-1)
            D[valid1] = 1.0 - h[idx1[valid1]]
            D[valid2] += -h[idx2[valid2]]
            D[k0] = 0.0  # Will handle lag=0 as quadratic
            
            # B = corr - old * D (for lags ≠ 0)
            # B[k0] = corr[k0] - old + old²

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def push_updown(h, n):
    """Ensure sum(h) = n/2 exactly"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            return h
        # Spread the correction to all non-boundary elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, deadline, max_sweeps=100000, random_order=False, seed=42):
    dx = 2.0 / n
    k0 = n - 1
    h = np.array(h, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_c = corr.max() * dx
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    lags = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        any_change = False
        for i in order:
            old = h[i]
            if old <= 1e-10 or old >= 1.0 - 1e-10:
                continue
            
            idx1 = i - lags
            valid1 = (idx1 >= 0) & (idx1 < n)
            idx2 = i + lags
            valid2 = (idx2 >= 0) & (idx2 < n)
            
            D = np.zeros(2*n-1)
            D[valid1] = 1.0 - h[idx1[valid1]]
            D[valid2] += -h[idx2[valid2]]
            D[k0] = 0.0
            
            B = corr - old * D
            B[k0] = corr[k0] - old + old*old
            
            lo, hi = 0.0, 1.0
            for _ in range(25):
                mid = (lo + hi) * 0.5
                vals = B + mid * D
                vals[k0] = B[k0] + mid - mid*mid
                mx = vals.max()
                active = vals >= mx - 1e-12
                na = active.sum()
                sg = D[active].sum() + (1.0 - 2.0*mid) * active[k0]
                sg /= na
                if sg > 1e-10:
                    hi = mid
                elif sg < -1e-10:
                    lo = mid
                else:
                    break
            
            t_new = np.clip((lo + hi) * 0.5, 0.0, 1.0)
            
            if abs(t_new - old) > 1e-10:
                # Verify improvement
                new_corr = B + t_new * D
                new_corr[k0] = B[k0] + t_new - t_new*t_new
                nc = new_corr.max() * dx
                if nc <= best_c + 1e-14:  # accept if not worse
                    h[i] = t_new
                    corr = new_corr
                    any_change = True
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
        
        if not any_change:
            break
    
    return best_h, best_c
