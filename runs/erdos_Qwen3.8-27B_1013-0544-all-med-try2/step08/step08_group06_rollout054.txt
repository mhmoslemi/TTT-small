def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

    h_b[idx[:n // 2]] = 1.0

                   lr=0.005, iters_per_phase=5000):

    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    alpha = alpha_start
    total_iters = 0
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    while time.time() < deadline:
        # Compute correlation and softmax
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = corr.max()
        shifted = alpha * (corr - cmax)
        exps = np.exp(shifted)
        p = exps / exps.sum()
        obj = cmax + np.log(exps.sum()) / alpha
        
        # Compute gradient
        Q = np.where(valid, p[p_idx], 0.0)
        grad = Q @ (1.0 - h) - Q.T @ h
        grad *= dx
        
        # ADAM update
        total_iters += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** total_iters)

        # Increase alpha periodically
        if total_iters % iters_per_phase == 0:
            alpha *= alpha_growth
            # Reset momentum when alpha changes
            m *= 0.5
            v *= 0.5
    
    return best_h, best_c
