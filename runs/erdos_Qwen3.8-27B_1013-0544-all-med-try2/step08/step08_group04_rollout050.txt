    gr = (np.sqrt(5) - 1) / 2

    d = a + gr * (b - a)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def obj_pen(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    v = smooth_max_val(c, alpha)
    v += mu * (x.sum() - n / 2.0) ** 2
    return float(v)

def grad_pen(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_sweep(h, n, deadline, max_iters=250):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [30, 80, 200, 500, 1500, 5000, 15000, 50000, 150000, 500000]
    mus = [5.0, 5.0, 10.0, 10.0, 50.0, 100.0, 300.0, 500.0, 2000.0, 5000.0]
    for alpha, mu in zip(alphas, mus):
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(obj_pen, h, jac=grad_pen,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_refine(h, n, deadline, alpha=300000, maxiter=800):
    bounds = [(0.0, 1.0)] * n
    def _obj(x):
        return float(smooth_max_val(compute_c5_vec(x, n), alpha))
    def _con(x):
        return float(x.sum() - n / 2.0)
    def _con_jac(x):
        return np.ones(n)
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': _con, 'jac': _con_jac}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def proj_grad(h, n, deadline, alpha=200.0, lr=0.005):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for _ in range(30000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        step = lr
        for _ in range(50):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                break
            step *= 0.5
        else:
            lr *= 0.6
            alpha *= 1.4
            if lr < 1e-13:
                break
    return best_h, best_c

def fast_pair_search(h, n, deadline, deltas, radius=100):

    best_c = compute_c5(h, n)
    for d in deltas:
        improved_any = False
        while time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved_any:
                break
    return h, best_c

def grad_guided_pairs(h, n, deadline, step=0.003):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    alphas = [100, 500, 2000, 10000, 50000]
    for alpha in alphas:
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                gi = g[i]
                for j in range(n):
                    if i == j:
                        continue
                    move = -step * (gi - g[j])
                    if abs(move) < 1e-12:
                        continue
                    hi_new = h[i] + move
                    hj_new = h[j] - move
                    if hi_new >= 0.0 and hi_new <= 1.0 and hj_new >= 0.0 and hj_new <= 1.0:
                        h[i] = hi_new
                        h[j] = hj_new
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            if not improved:
                break
    return h, best_c
