def swap_delta(a, b, h, n):
    """Compute delta[k] for all lags when h[a]: 1->0, h[b]: 0->1"""
    # c[k] = sum_i h[i] * (1-h[i-k]) where k ranges -(n-1) to n-1
    # Stored in array of size 2n-1, index j corresponds to k = j - n + 1

    # Effect of a: 1->0
    # a as first index: was contributing (1-h[a-k]) at lag k, now contributes 0
    #   valid when 0 <= a-k < n, i.e., a-n+1 <= k <= a
    #   in array index: a-n+1 + n-1 = a <= j <= a + n - 1
    #   delta[j] -= (1 - h[a - (j-n+1)]) = (1 - h[a-j+n-1])
    
    # a as second index: was contributing h[a+k]*(1-1)=0, now contributes h[a+k]*(1-0)=h[a+k]
    #   valid when 0 <= a+k < n, i.e., -a <= k < n-a
    #   in array index: n-1-a <= j <= 2n-2-a
    #   delta[j] += h[a + (j-n+1)] = h[a+j-n+1]
    
    # Effect of b: 0->1
    # b as first index: was contributing 0, now contributes (1-h[b-k])
    #   delta[j] += (1 - h[b-j+n-1]) for valid j
    # b as second index: was contributing h[b+k]*(1-0)=h[b+k], now contributes 0
    #   delta[j] -= h[b+j-n+1] for valid j
    
    # Vectorized computation:
    # For a as first index: j ranges from a to a+n-1 (clamped to [0, 2n-2])
    j_min_a1 = max(0, a)
    j_max_a1 = min(2*n-2, a+n-1)
    # h[a - (j-n+1)] = h[a - j + n - 1]
    idx_a1 = a - np.arange(j_min_a1, j_max_a1+1) + n - 1
    delta[j_min_a1:j_max_a1+1] -= (1 - h[idx_a1])
    
    # For a as second index: j ranges from n-1-a to 2n-2-a (clamped)
    j_min_a2 = max(0, n-1-a)
    j_max_a2 = min(2*n-2, 2*n-2-a)
    idx_a2 = a + np.arange(j_min_a2, j_max_a2+1) - n + 1
    delta[j_min_a2:j_max_a2+1] += h[idx_a2]
    
    # For b as first index:
    j_min_b1 = max(0, b)
    j_max_b1 = min(2*n-2, b+n-1)
    idx_b1 = b - np.arange(j_min_b1, j_max_b1+1) + n - 1
    delta[j_min_b1:j_max_b1+1] += (1 - h[idx_b1])
    
    # For b as second index:
    j_min_b2 = max(0, n-1-b)
    j_max_b2 = min(2*n-2, 2*n-2-b)
    idx_b2 = b + np.arange(j_min_b2, j_max_b2+1) - n + 1
    delta[j_min_b2:j_max_b2+1] -= h[idx_b2]
    
    return delta

        if diff > 0:

            s = room.sum()

            if s < 1e-14:

        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

- If i == b (i.e., a+k=b, k=b-a, j=n-1+b-a): h_new[b]=1, base uses h[b]=0, correction = +1

Case 3 (b as 1st index): i = b
- Old: h[b]*(1-h[b-k]) = 0
- New: h_new[b]*(1-h_new[b-k]) = 1*(1-h_new[b-k]) = 1-h_new[b-k]
- Base formula uses 1-h[b-k] (old value)
- If b-k != a: h_new[b-k] = h[b-k], no correction
- If b-k == a (i.e., k=b-a, j=n-1+b-a): h_new[a]=0, base uses 1-h[a]=0, new is 1-0=1, correction = +1

So total correction at j = n-1+(b-a): +1 + 1 = +2. ✓

So the final formula is:
delta[j] = [-(1-h[a-k]) + h[a+k]] + [(1-h[b-k]) - h[b+k]] + 2*(j == n-1+b-a)

where k = j-n+1, and all h[...] terms use valid indices (0 out of range).

This is a clean O(n) formula! Let me implement it:

```python
def compute_delta(a, b, h, n):
    delta = np.zeros(2*n-1)
    k_range = np.arange(2*n-1) - n + 1  # lags from -(n-1) to n-1
    
    # Term from a: -(1-h[a-k]) + h[a+k]
    # a-k valid: 0 <= a-k < n => a-n+1 <= k <= a
    valid1 = (a - k_range >= 0) & (a - k_range < n)
    delta[valid1] -= (1 - h[a - k_range[valid1]])
    
    valid2 = (a + k_range >= 0) & (a + k_range < n)
    delta[valid2] += h[a + k_range[valid2]]
    
    # Term from b: (1-h[b-k]) - h[b+k]
    valid3 = (b - k_range >= 0) & (b - k_range < n)
    delta[valid3] += (1 - h[b - k_range[valid3]])
    
    valid4 = (b + k_range >= 0) & (b + k_range < n)
    delta[valid4] -= h[b + k_range[valid4]]
    
    # Cross-term
    j_cross = n - 1 + (b - a)
    if 0 <= j_cross < 2*n-1:
        delta[j_cross] += 2
    
    return delta
```

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def make_binary(h, n, rng):
    target = n // 2
    h = (h >= 0.5).astype(float)
    n_ones = int(h.sum())
    if n_ones > target:
        pos = np.where(h == 1)[0]
        h[rng.choice(pos, n_ones - target, replace=False)] = 0
    elif n_ones < target:
        pos = np.where(h == 0)[0]
        h[rng.choice(pos, target - n_ones, replace=False)] = 1
    return h

def fast_binary_local_search(h, n, deadline, seed=42, n_candidates=50, top_k=15):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = make_binary(h, n, rng)
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    no_improve = 0
    k_range = np.arange(2*n-1) - n + 1
    
    while time.time() < deadline:
        # Find current max lags
        max_idx = np.argmax(corr)
        k_star = k_range[max_idx]
        
        # Score all ones and zeros at the max lag
        # score_a = -(1-h[a-k*]) + h[a+k*]
        a_ah = np.clip(ones_arr - k_star, 0, n-1)  # a-k_star (clamped)
        a_kh = np.clip(ones_arr + k_star, 0, n-1)  # a+k_star (clamped)
        a_valid_low = (ones_arr - k_star >= 0) & (ones_arr - k_star < n)
        a_valid_high = (ones_arr + k_star >= 0) & (ones_arr + k_star < n)
        score_a = np.zeros(len(ones_arr))
        score_a[a_valid_low] -= (1 - h[a_ah[a_valid_low]])
        score_a[a_valid_high] += h[a_kh[a_valid_high]]
        
        b_ah = np.clip(zeros_arr - k_star, 0, n-1)
        b_kh = np.clip(zeros_arr + k_star, 0, n-1)
        b_valid_low = (zeros_arr - k_star >= 0) & (zeros_arr - k_star < n)
        b_valid_high = (zeros_arr + k_star >= 0) & (zeros_arr + k_star < n)
        score_b = np.zeros(len(zeros_arr))
        score_b[b_valid_low] += (1 - h[b_ah[b_valid_low]])
        score_b[b_valid_high] -= h[b_kh[b_valid_high]]
        
        # Get top-k candidates
        top_a_idx = np.argsort(score_a)[:top_k]
        top_b_idx = np.argsort(score_b)[:top_k]
        
        # Also add some random candidates
        n_random = max(1, n_candidates - top_k * top_k)
        rand_a_idx = rng.randint(len(ones_arr), n_random)
        rand_b_idx = rng.randint(len(zeros_arr), n_random)
        
        candidates = []
        for ia in top_a_idx:
            for ib in top_b_idx:
                candidates.append((ones_arr[ia], zeros_arr[ib]))
        for i in range(n_random):
            candidates.append((ones_arr[rand_a_idx[i]], zeros_arr[rand_b_idx[i]]))
        
        best_imp = 0
        best_a, best_b = -1, -1
        
        for a, b in candidates:
            if a == b:
                continue
            # Compute delta
            delta = np.zeros(2*n-1)
            v1 = (a - k_range >= 0) & (a - k_range < n)
            delta[v1] -= (1 - h[a - k_range[v1]])
            v2 = (a + k_range >= 0) & (a + k_range < n)
            delta[v2] += h[a + k_range[v2]]
            v3 = (b - k_range >= 0) & (b - k_range < n)
            delta[v3] += (1 - h[b - k_range[v3]])
            v4 = (b + k_range >= 0) & (b + k_range < n)
            delta[v4] -= h[b + k_range[v4]]
            j_cross = n - 1 + (b - a)
            if 0 <= j_cross < 2*n-1:
                delta[j_cross] += 2
            
            new_max = (corr + delta).max()
            imp = cur_max - new_max
            if imp > best_imp:
                best_imp = imp
                best_a, best_b = a, b
        
        if best_a < 0:
            no_improve += 1
            if no_improve > 200:
                break
            if no_improve % 20 == 0:
                # Perturb: swap 2-3 random pairs
                n_pert = rng.randint(1, 4)
                for _ in range(n_pert):
                    a = ones_arr[rng.randint(len(ones_arr))]
                    b = zeros_arr[rng.randint(len(zeros_arr))]
                    h[a], h[b] = 0, 1
                ones_arr = np.where(h == 1)[0].copy()
                zeros_arr = np.where(h == 0)[0].copy()
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_max = corr.max()
        else:
            no_improve = 0
            h[best_a], h[best_b] = 0, 1
            ones_arr = np.where(h == 1)[0].copy()
            zeros_arr = np.where(h == 0)[0].copy()
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_max = corr.max()
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    
    return best_h, best_max * dx
