def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx  # g[m] = c[m] * dx
    g_max = g.max()
    shifted = alpha * (g - g_max)
    shifted_max = shifted.max()
    log_sum = shifted_max + np.log(np.exp(shifted - shifted_max).sum())
    f = g_max + log_sum / alpha
    p = np.exp(shifted - log_sum)  # probability weights
    
    # Compute gradient
    grad = np.zeros(n)
    m_range = np.arange(2 * n - 1)  # 0 to 2n-2
    for j in range(n):
        # A[j] = sum_m p[m] * (1-h[j+m-n+1]) * [0 <= j+m-n+1 < n]
        # B[j] = sum_m p[m] * h[j-m+n-1] * [0 <= j-m+n-1 < n]
        # l = m - (n-1), so m = n-1+l
        # A[j] = sum_l p[n-1+l] * (1-h[j+l]) * [0 <= j+l < n]
        # B[j] = sum_l p[n-1+l] * h[j-l] * [0 <= j-l < n]
        
        l_min = max(-(j), -(n-1))  # j+l >= 0 => l >= -j
        l_max = min(n-1-j, n-1)    # j+l < n => l < n-j => l <= n-1-j
        
        # For A: l from max(-j, -(n-1)) to min(n-1-j, n-1)
        # For B: j-l >= 0 => l <= j, j-l < n => l > j-n => l >= j-n+1
        #        l from max(j-n+1, -(n-1)) to min(j, n-1)
        
        # Actually since l ranges from -(n-1) to (n-1) always (because m ranges from 0 to 2n-2),
        # the additional constraints are just the [valid] indicators.
        
        a_val = 0.0
        b_val = 0.0
        for m in range(2 * n - 1):
            l = m - (n - 1)
            idx_a = j + l
            idx_b = j - l
            if 0 <= idx_a < n:
                a_val += p[m] * (1.0 - h[idx_a])
            if 0 <= idx_b < n:
                b_val += p[m] * h[idx_b]

    return f, grad, g_max

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    k2 = n - 1 + (i - j)
    delta[k1] += 1.0
    delta[k2] += 1.0
    return delta

def greed_binary(h_init, n, deadline, max_sweeps=500):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=5000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = 0.02
    cool = 0.99998
    scale = 2500.0
    no_improve = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i, j = zeros_arr[ai], ones_arr[bi]
        delta = swap_delta(h, n, i, j, m_range)
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai], ones_arr[bi] = j, i
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 50000 and T > 0.001:
            T *= 0.5
            no_improve = 0
        T *= cool
    return best_h, best_c

def smooth_max_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    log_sum = sm + np.log(np.exp(shifted - sm).sum())
    f = m + log_sum / alpha
    p = np.exp(shifted - log_sum)
    grad = np.zeros(n)
    for j in range(n):
        m_lo = max(0, n - 1 - j)
        m_hi = min(2 * n - 2, 2 * n - 2 - j)
        a_val = 0.0
        if m_lo <= m_hi:
            idx_a = j + np.arange(m_lo, m_hi + 1) - (n - 1)
            a_val = np.dot(p[m_lo:m_hi + 1], 1.0 - x[idx_a])
        m_lo_b = max(0, j - n + 1)
        m_hi_b = min(2 * n - 2, j + n - 1)
        b_val = 0.0
        if m_lo_b <= m_hi_b:
            idx_b = j - np.arange(m_lo_b, m_hi_b + 1) + (n - 1)
            b_val = np.dot(p[m_lo_b:m_hi_b + 1], x[idx_b])
        grad[j] = dx * (a_val - b_val)
    return f, grad

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    def obj(x):
        dx = 2.0 / n
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def proj_grad_smooth(h, n, alpha, lr, n_iters, deadline):
    h = h.copy()
    best = h.copy()
    best_c = compute_c5(h, n)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        f, grad = smooth_max_obj_and_grad(h, n, alpha)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        h = h_new
        if (it + 1) % 2000 == 0:
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best = c, h.copy()
    c = compute_c5(h, n)
    if c < best_c:
        best_c, best = c, h.copy()
    return best, best_c
