def project_feasible(z,n):
    target=n*0.5
    z=np.asarray(z,float).copy()
    # bisection on tau: x=clip(z-tau,0,1)
    lo=z.min()-1.0; hi=z.max()+1.0
    for _ in range(80):
       mid=(lo+hi)/2
       s=np.clip(z-mid,0,1).sum()
       if s>target: lo=mid
       else: hi=mid
    tau=(lo+hi)/2
    x=np.clip(z-tau,0,1)
    # correct tiny diff
    for _ in range(5):
       diff=target-x.sum()
       if abs(diff)<1e-18: break
       if diff>0:
          mask=x<1
          if not mask.any(): break
          i=int(np.argmax(np.where(mask,1-x,-1)))
          add=min(diff,1-x[i])
          x[i]+=add
       else:
          mask=x>0
          if not mask.any(): break
          i=int(np.argmin(np.where(mask,x,2)))
          sub=min(-diff,x[i])
          x[i]-=sub
    return x

def objective_smooth(h,n,alpha,K,mu=0.0):
    c=c_vec(h,n)
    ds=2.0/float(n)
    cm=float(c.max())
    sh=alpha*(c-cm)
    mx=float(sh.max())
    e=np.exp(sh-mx)
    se=float(e.sum())
    obj=cm + (mx + float(np.log(se)))/alpha
    w=e/se
    g=np.zeros(n)
    if K is None or K>=c.size:
       idx=np.arange(c.size)
    else:
       K=int(K)
       if K>0:
           if K<c.size: idx=np.argpartition(w,K-1)[-K:]
           else: idx=np.arange(c.size)
       else: idx=np.empty(0,dtype=int)
    base=n-1
    for ii in idx:
        wi=float(w[ii])
        if wi <= 1e-14: continue
        wm=wi*ds
        lag=int(ii)-base
        if lag==0:
            g += wm*(1.0-2.0*h)
        elif lag>0:
            k=lag; nv=n-k
            g[:nv] -= wm*h[k:k+nv]
            g[k:] += wm*(1.0-h[:nv])
        else:
            r=-lag; nv=n-r
            g[:nv] += wm*(1.0-h[r:r+nv])
            g[r:] -= wm*h[:nv]
    if mu != 0.0:
       s=float(h.sum())-0.5*n
       obj += 0.5*mu*s*s
       g += mu*s  # wait derivative of 0.5 mu s^2 is mu s, but before I used 2mu. Need consistency.
    return float(obj), g

def objective_smooth(h,n,alpha,K,mu):
    ds=2.0/float(n); c=np.correlate(h,1.0-h,mode='full')*ds
    cm=float(c.max()); sh=alpha*(c-cm); mx=float(sh.max())
    e=np.exp(sh-mx); se=float(e.sum()); w=e/se
    obj=cm+(mx+float(np.log(se)))/alpha
    g=np.zeros(n)
    L=c.size
    if K is None or K>=L: idx=np.arange(L)
    elif K<=0: return obj,g
    else: idx=np.argpartition(w,K-1)[-K:]
    base=n-1
    for q in idx:
        wi=float(w[q]); 
        if wi <= 1e-15: continue
        wm=wi*ds; lag=int(q)-base
        if lag==0: g+=wm*(1.0-2.0*h)
        elif lag>0:
            k=lag; nv=n-k; g[:nv]-=wm*h[k:k+nv]; g[k:]+=wm*(1.0-h[:nv])
        else:
            r=-lag; nv=n-r; g[:nv]+=wm*(1.0-h[r:r+nv]); g[r:]-=wm*h[:nv]
    if mu:
        s=float(h.sum())-0.5*n; obj+=0.5*mu*s*s; g+=mu*s
    return float(obj),g

def lp_squeeze(h,n,deadline,stepmax=0.05,topK=170):
    h=h.copy(); best=c_val(h,n)
    step=stepmax; fails=0; ds=2.0/n; base=n-1
    while time.time()<deadline:
        if step<5e-5: break
        c=np.correlate(h,1.0-h,mode='full')*ds
        if topK<c.size: act=np.argpartition(c,-topK)[-topK:]
        else: act=np.arange(c.size)
        A=np.zeros((act.size,n))
        for i,p in enumerate(act):
            lag=int(p)-base; row=A[i]
            if lag==0: row[:]=1.0-2.0*h
            elif lag>0:
                k=lag; nv=n-k; row[:nv]-=h[k:k+nv]; row[k:]+=1.0-h[:nv]
            else:
                r=-lag; nv=n-r; row[:nv]+=1.0-h[r:r+nv]; row[r:]-=h[:nv]
        A*=ds
        m=act.size; V=2*n+1
        co=np.zeros(V); co[n:2*n]=1e-6; co[2*n]=1.0
        AB=np.zeros((2*n+m,V)); ii=np.arange(n)
        AB[2*ii,ii]=1.0; AB[2*ii,n+ii]=-1.0
        AB[2*ii+1,ii]=-1.0; AB[2*ii+1,n+ii]=-1.0
        r0=2*n; AB[r0:r0+m,:n]=A; AB[:,2*n]=0.0; AB[r0:r0+m,2*n]=-1.0
        bb=np.zeros(2*n+m); bb[2*n:]=-c[act]
        AE=np.zeros((1,V)); AE[0,:n]=1.0
        lo=-np.minimum(h,step); up=np.minimum(1.0-h,step)
        bd=[(float(lo[i]),float(up[i])) for i in range(n)]+[(0.0,None)]*n+[(None,None)]
        delta=None
        try:

            if r.status==0: delta=r.x[:n]

        if delta is None:
            step*=0.7; fails+=1
            if fails>10: break
            continue
        h0=h; basec=best; bh=h0; bc=basec

            step*=0.7; fails+=1

       (15000.0, 100), (75000.0, 80), (350000.0, 64), (1500000.0, 48))
LBS = ((100.0, 250), (800.0, 150), (5000.0, 90), (50000.0, 60), (400000.0, 48))
SLA = (20000.0, 80000.0, 300000.0, 1000000.0)
FNA = (300000.0, 1000000.0, 3000000.0)

def project_feasible(z, n):
    t = 0.5 * n
    z = np.asarray(z, dtype=float)
    if z.size != n:
        z = np.resize(z, n)
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(70):
        m = 0.5 * (lo + hi)
        s = float(np.clip(z - m, 0.0, 1.0).sum())
        if s > t:
            lo = m
        else:
            hi = m
    x = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(5):
        d = t - float(x.sum())
        if abs(d) < 1e-17:
            break
        if d > 0:
            i = int(np.argmax(np.where(x < 1.0, 1.0 - x, -1.0)))
            a = min(d, 1.0 - x[i])
            x[i] += a
        else:
            i = int(np.argmin(np.where(x > 0.0, x, 2.0)))
            a = min(-d, x[i])
            x[i] -= a
    return x

def c_vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))

def c_val(h, n):
    return float(np.max(c_vec(h, n)))

def objective_smooth(h, n, alpha, K, mu):
    ds = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full') * ds
    cm = float(c.max())
    sh = alpha * (c - cm)
    mx = float(sh.max())
    e = np.exp(sh - mx)
    se = float(e.sum())
    w = e / se
    obj = cm + (mx + float(np.log(se))) / alpha
    g = np.zeros(n)
    L = c.size
    if K is None or K >= L:
        idx = np.arange(L)
    elif K <= 0:
        return float(obj), g
    else:
        idx = np.argpartition(w, K - 1)[-K:]
    base = n - 1
    for q in idx:
        wi = float(w[q])
        if wi <= 1e-15:
            continue
        wm = wi * ds
        lag = int(q) - base
        if lag == 0:
            g += wm * (1.0 - 2.0 * h)
        elif lag > 0:
            k = lag
            nv = n - k
            g[:nv] -= wm * h[k:k + nv]
            g[k:] += wm * (1.0 - h[:nv])
        else:
            r = -lag
            nv = n - r
            g[:nv] += wm * (1.0 - h[r:r + nv])
            g[r:] -= wm * h[:nv]
    if mu:
        s = float(h.sum()) - 0.5 * n
        obj += 0.5 * mu * s * s
        g += mu * s
    return float(obj), g

def obj_fun(x, n, alpha, K, mu):
    v, g = objective_smooth(x, n, alpha, K, mu)
    return v

def grad_fun(x, n, alpha, K, mu):
    v, g = objective_smooth(x, n, alpha, K, mu)
    return g

def lbfgs_stage(h, n, deadline, pairs, maxiter=70):
    h = h.copy()
    best = c_val(h, n)
    bd = [(0.0, 1.0)] * n
    for a, K in pairs:
        if time.time() >= deadline:
            break
        try:
            r = minimize(obj_fun, h, jac=grad_fun, args=(n, a, K, 0.1),
                         method='L-BFGS-B', bounds=bd,
                         options={'maxiter': maxiter, 'ftol': 1e-15,
                                  'gtol': 1e-12, 'maxfun': maxiter * 6,
                                  'maxls': 12})
            x = r.x
        except Exception:
            continue
        if x is None:
            continue
        h2 = project_feasible(x, n)
        c2 = c_val(h2, n)
        if c2 < best:
            h, best = h2, c2
    return h, best

def eq_sum(x, n):
    return float(x.sum() - 0.5 * n)

def eq_jac(x, n):
    return np.ones(n)

def slsqp_stage(h, n, deadline, alphas, maxiter=100, K=100):
    h = h.copy()
    best = c_val(h, n)
    bd = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': eq_sum, 'jac': eq_jac, 'args': (n,)}]
    for a in alphas:
        if time.time() >= deadline:
            break
        try:
            r = minimize(obj_fun, h, jac=grad_fun, args=(n, a, K, 0.0),
                         method='SLSQP', bounds=bd, constraints=con,
                         options={'maxiter': maxiter, 'ftol': 1e-13})
            x = r.x
        except Exception:
            continue
        if x is None:
            continue
        h2 = project_feasible(x, n)
        c2 = c_val(h2, n)
        if c2 < best:
            h, best = h2, c2
    return h, best

def lp_squeeze(h, n, deadline, stepmax=0.05, topK=170):
    h = h.copy()
    best = c_val(h, n)
    step = stepmax
    fails = 0
    ds = 2.0 / float(n)
    base = n - 1
    while time.time() < deadline:
        if step < 5e-5:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * ds
        if topK < c.size:
            act = np.argpartition(c, -topK)[-topK:]
        else:
            act = np.arange(c.size)
        A = np.zeros((act.size, n))
        for i, p in enumerate(act):
            lag = int(p) - base
            row = A[i]
            if lag == 0:
                row[:] = 1.0 - 2.0 * h
            elif lag > 0:
                k = lag
                nv = n - k
                row[:nv] -= h[k:k + nv]
                row[k:] += 1.0 - h[:nv]
            else:
                r = -lag
                nv = n - r
                row[:nv] += 1.0 - h[r:r + nv]
                row[r:] -= h[:nv]
        A *= ds
        m = act.size
        V = 2 * n + 1
        co = np.zeros(V)
        co[n:2 * n] = 1e-6
        co[2 * n] = 1.0
        AB = np.zeros((2 * n + m, V))
        ii = np.arange(n)
        AB[2 * ii, ii] = 1.0
        AB[2 * ii, n + ii] = -1.0
        AB[2 * ii + 1, ii] = -1.0
        AB[2 * ii + 1, n + ii] = -1.0
        r0 = 2 * n
        AB[r0:r0 + m, :n] = A
        AB[r0:r0 + m, 2 * n] = -1.0
        bb = np.zeros(2 * n + m)
        bb[2 * n:2 * n + m] = -c[act]
        AE = np.zeros((1, V))
        AE[0, :n] = 1.0
        lo = -np.minimum(h, step)
        up = np.minimum(1.0 - h, step)
        bd = [(float(lo[i]), float(up[i])) for i in range(n)]
        bd += [(0.0, None)] * n
        bd += [(None, None)]
        delta = None
        try:
            r = linprog(co, A_ub=AB, b_ub=bb, A_eq=AE, b_eq=[0.0],
                        bounds=bd, method='highs')
            if r.status == 0:
                delta = r.x[:n]
        except Exception:
            delta = None
        if delta is None:
            step *= 0.7
            fails += 1
            if fails > 12:
                break
            continue
        h0 = h
        c0 = best
        bh = h0
        bc = c0
        for s in (1.0, 0.6, 0.3, 0.12):
            cand = project_feasible(h0 + s * delta, n)
            cv = c_val(cand, n)
            if cv < bc:
                bh, bc = cand, cv
        if bc < c0 - 1e-15:
            h, best = bh, bc
            fails = 0
            if c0 - best > 1e-6:
                step = min(0.12, step * 1.15)
        else:
            step *= 0.7
            fails += 1
            if fails > 14:
                break
    return h, best

def pair_stage(h, n, deadline, alphas):
    h = h.copy()
    best = c_val(h, n)
    steps = (0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002)
    for a in alphas:
        top = 8
        while time.time() < deadline:
            _, g = objective_smooth(h, n, a, 64, 0.0)
            g -= g.mean()
            lo = np.argsort(g)[:top]
            hi = np.argsort(g)[-top:]
            moved = False
            for i in lo:
                for j in hi:
                    if i == j:
                        continue
                    for s in steps:
                        if h[i] + s <= 1.0 and h[j] - s >= 0.0:
                            h[i] += s
                            h[j] -= s
                            c2 = c_val(h, n)
                            if c2 < best:
                                best = c2
                                moved = True
                                break
                            h[i] -= s
                            h[j] += s
                    if moved:
                        break
                if moved:
                    break
            if not moved:
                top += 10
                if top > 48:
                    break
    return h, best

def interp_res(h, n0, n1):
    if n1 == n0:
        return h.copy()
    x0 = np.arange(n0) * (2.0 / float(n0))
    x1 = np.arange(n1) * (2.0 / float(n1))
    return project_feasible(np.interp(x1, x0, h), n1)

def random_start(n, rng):
    h = np.zeros(n)
    cnt = n // 2
    if cnt:
        h[rng.choice(n, cnt, replace=False)] = 1.0
    return project_feasible(h + 0.03 * rng.randn(n), n)

    t0 = time.time()
    deadline = t0 + max(20.0, float(budget_s)) - 8.0
    rng = np.random.RandomState(seed)
    try:

        if h0.ndim != 1 or h0.size == 0:

    except Exception:

    og = int(h0.size)
    n0 = og
    if n0 < 96:
        h0 = random_start(400, rng)
        n0 = 400
    elif n0 > 900:
        n0 = 900
        h0 = interp_res(h0, og, n0)
    h0 = project_feasible(h0, n0)
    best_h = h0
    best_n = n0
    best_c = c_val(best_h, best_n)

    if time.time() < deadline - 320:
        best_h, best_c = lbfgs_stage(best_h, best_n,
                                     min(time.time() + 100.0, deadline - 280.0), LBP, 70)
    if time.time() < deadline - 280:
        best_h, best_c = lp_squeeze(best_h, best_n,
                                    min(time.time() + 80.0, deadline - 220.0), 0.05, 180)
    if time.time() < deadline - 210:
        best_h, best_c = slsqp_stage(best_h, best_n,
                                     min(time.time() + 60.0, deadline - 170.0), SLA, 100, 100)

    for frac in (0.8, 0.6, 0.45):
        if time.time() > deadline - 150:
            break
        nn = int(round(best_n * frac))
        nn = max(96, nn if nn % 2 == 0 else nn + 1)
        if nn == best_n:
            continue
        alloc = min(80.0, (deadline - time.time()) * 0.30)
        if alloc < 35.0:
            break
        h2 = interp_res(best_h, best_n, nn)
        h2, c2 = lbfgs_stage(h2, nn, min(time.time() + alloc * 0.4, deadline - 90.0), LBS, 60)
        if time.time() < deadline - 80:
            h2, c2 = lp_squeeze(h2, nn, min(time.time() + alloc * 0.4, deadline - 70.0), 0.03, 110)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, nn
        if time.time() < deadline - 110:
            hr = random_start(nn, rng)
            hr, cr = lbfgs_stage(hr, nn, min(time.time() + 28.0, deadline - 90.0), LBS, 55)
            if cr < best_c and time.time() < deadline - 80:
                hr, cr = lp_squeeze(hr, nn, min(time.time() + 25.0, deadline - 70.0), 0.03, 100)
            if cr < best_c:
                best_h, best_c, best_n = hr, cr, nn

    if time.time() < deadline - 80:
        best_h, best_c = slsqp_stage(best_h, best_n,
                                     min(time.time() + 45.0, deadline - 55.0), FNA, 100, 80)
    if time.time() < deadline - 55:
        best_h, best_c = lp_squeeze(best_h, best_n,
                                    min(time.time() + 40.0, deadline - 25.0), 0.015, 140)
    if time.time() < deadline - 2.0:
        best_h, best_c = pair_stage(best_h, best_n, deadline - 1.0, FNA)

    best_h = project_feasible(best_h, best_n)
    best_c = c_val(best_h, best_n)
    return best_h, float(best_c), int(best_n)
