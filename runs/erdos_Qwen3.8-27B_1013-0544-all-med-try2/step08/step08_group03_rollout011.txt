```
def grad_weighted(h,n,weights,lag_indices):
```

```
def project(y,n,iters=35):
    y=np.asarray(y,dtype=float)
    target=0.5*n
    if len(y)!=n: y=np.resize(y,n) # not needed
    lo=np.min(y)-1.0; hi=np.max(y)+1.0
    tau=0.5*(lo+hi)
    x=np.clip(y-tau,0.0,1.0); s=x.sum()
    # quick if s close?
    for _ in range(iters):
        if s > target:
            lo=tau
        else:
            hi=tau
        tau=0.5*(lo+hi)
        x=np.clip(y-tau,0.0,1.0); s=x.sum()
        if abs(s-target)<1e-14: break
    # final adjust for exactness
    x=np.clip(x,0.0,1.0)
    diff=target-x.sum()
    if abs(diff)>1e-12:
        if diff>0:
            room=1.0-x
            idx=np.where(room>1e-15)[0]
            if idx.size:
                x[idx] += diff*room[idx]/room[idx].sum()
        else:
            idx=np.where(x>1e-15)[0]
            if idx.size:
                x[idx] += diff*x[idx]/x[idx].sum()
    x=np.clip(x,0.0,1.0)
    # one point fix residual
    diff=target-x.sum()
    if diff!=0:
       if diff>0:
          idx=np.where(x<1.0)[0]
          if idx.size:
             j=idx[np.argmax(1.0-x[idx])]
             x[j]=min(1.0,x[j]+diff)
       else:
          idx=np.where(x>0.0)[0]
          if idx.size:
             j=idx[np.argmax(x[idx])]
             x[j]=max(0.0,x[j]+diff)
    return x
```

```python
import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize


def cvec(h, n):
    hx = np.asarray(h, dtype=float)
    gx = 1.0 - hx
    if n < 96:
        y = np.correlate(hx, gx, mode="full")
    else:
        y = correlate(hx, gx, mode="full", method="fft")
    return y * (2.0 / n)


def max_c(h, n):
    return float(np.max(cvec(h, n)))


def project(y, n, iters=35):
    y = np.asarray(y, dtype=float)
    if len(y) != n:
        y = np.interp(np.linspace(0, 2, n, endpoint=False),
                      np.linspace(0, 2, len(y), endpoint=False), y)
    target = 0.5 * n
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    tau = 0.5 * (lo + hi)
    x = np.clip(y - tau, 0.0, 1.0)
    s = float(x.sum())
    for _ in range(iters):
        if s > target:
            lo = tau
        else:
            hi = tau
        tau = 0.5 * (lo + hi)
        x = np.clip(y - tau, 0.0, 1.0)
        s = float(x.sum())
        if abs(s - target) < 1e-14:
            break

    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - x
            idx = np.where(room > 1e-15)[0]
            if idx.size:
                x[idx] += diff * room[idx] / float(room[idx].sum())
        else:
            idx = np.where(x > 1e-15)[0]
            if idx.size:
                x[idx] += diff * x[idx] / float(x[idx].sum())
    x = np.clip(x, 0.0, 1.0)

    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size:
                j = int(idx[np.argmax(1.0 - x[idx])])
                x[j] = min(1.0, x[j] + diff)
        else:
            idx = np.where(x > 0.0)[0]
            if idx.size:
                j = int(idx[np.argmax(x[idx])])
                x[j] = max(0.0, x[j] + diff)
    return x


def grad_one(h, n, full_idx):
    g = np.zeros(n, dtype=float)
    k = int(full_idx) - (n - 1)
    if k >= 0:
        nv = n - k
        g[:nv] += 1.0 - h[k:]
        g[k:] -= h[:nv]
    else:
        a = -k
        nv = n - a
        g[a:] += 1.0 - h[:nv]
        g[:nv] -= h[a:]
    return g


def grad_top(h, n, weights, inds):
    g = np.zeros(n, dtype=float)
    for i in range(len(inds)):
        w = float(weights[i])
        if abs(w) < 1e-15:
            continue
        k = int(inds[i]) - (n - 1)
        if k >= 0:
            nv = n - k
            g[:nv] += w * (1.0 - h[k:])
            g[k:] -= w * h[:nv]
        else:
            a = -k
            nv = n - a
            g[a:] += w * (1.0 - h[:nv])
            g[:nv] -= w * h[a:]
    return g


def top_weights(c, alpha, topk):
    topk = max(1, min(int(topk), len(c)))
    if topk >= len(c):
        idx = np.argsort(c)[::-1]
    else:
        idx = np.argpartition(c, len(c) - topk)[-topk:]
        idx = idx[np.argsort(c[idx])[::-1]]
    vals = c[idx]
    sh = alpha * (vals - vals.max())
    w = np.exp(sh)
    s = float(w.sum())
    if s <= 0.0:
        w = np.full_like(vals, 1.0 / len(vals))
    else:
        w = w / s
    return idx, w


def active_direction(c, h, n, idx, top=32):
    if len(idx) < 4:
        return None
    idx = idx[:top]
    m = len(idx)
    G = np.empty((m, n), dtype=float)
    for r, ii in enumerate(idx):
        G[r] = grad_one(h, n, ii)

    cact = c[idx]
    cmax = float(cact.max())
    cmin = float(cact.min())
    spread = max(1e-9, cmax - cmin)
    target = cmax - max(1e-7, 0.025 * spread)
    b = target - cact
    b[b > 0.0] = 0.0
    if float(np.max(np.abs(b))) < 1e-14:
        b = -max(1e-6, 0.01 * spread) * np.ones_like(b)

    A = G @ G.T
    lam = 1e-5 * max(float(np.trace(A)), 1.0)
    A += lam * np.eye(m)
    rhs = G.T @ b
    try:
        u = np.linalg.solve(A, rhs)
    except Exception:
        u = np.linalg.lstsq(A, rhs, rcond=None)[0]

    d = G.T @ u
    d -= float(d.mean())
    std = float(np.linalg.norm(d)) / np.sqrt(n)
    if std < 1e-12:
        return None
    maxstd = 0.16
    if std > maxstd:
        d *= maxstd / std
    return d

def smooth_obj(x, n, alpha, pen):
    c = cvec(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = float(x.sum() - 0.5 * n)
    return float(val + pen * s * s)


def smooth_grad(x, n, alpha, pen, topk=160):
    c = cvec(x, n)
    m = float(c.max())
    idx, w = top_weights(c, alpha, topk)
    g = grad_top(x, n, w, idx)
    s = float(x.sum() - 0.5 * n)
    g += 2.0 * pen * s
    return g


def run_lbgfs(h, n, deadline, alpha, pen=800, maxiter=70):
    h = project(h, n)
    cur = max_c(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(
            smooth_obj, h, args=(n, alpha, pen), jac=smooth_grad,
            method="L-BFGS-B", bounds=bounds,
            options={"maxiter": maxiter, "ftol": 1e-15, "gtol": 1e-11,
                     "iprint": -1, "maxfun": max(200, 4 * maxiter)}
        )
        hn = project(res.x, n)
        cn = max_c(hn, n)
        if cn < cur:
            return hn, cn
    except Exception:
        pass
    return h, cur


def slsqp_obj(x, n, alpha):
    c = cvec(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)


def slsqp_eq(x, n):
    return float(x.sum() - 0.5 * n)


def slsqp_jac(x, n):
    return np.ones(len(x), dtype=float)


def run_slsqp(h, n, deadline, alpha):
    h = project(h, n)
    cur = max_c(h, n)
    if n > 56 or time.time() >= deadline - 10:
        return h, cur
    try:
        cons = [{"type": "eq", "fun": slsqp_eq, "jac": slsqp_jac, "args": (n,)}]
        bounds = [(0.0, 1.0)] * n
        res = minimize(
            slsqp_obj, h, args=(n, alpha), method="SLSQP",
            bounds=bounds, constraints=cons,
            options={"maxiter": 350, "ftol": 1e-16}
        )
        hn = project(res.x, n)
        cn = max_c(hn, n)
        if cn < cur:
            return hn, cn
    except Exception:
        pass
    return h, cur


def interp_array(h, n):
    h = np.asarray(h, dtype=float)
    m = len(h)
    if m == n:
        return h.copy()
    xo = np.linspace(0.0, 2.0, m, endpoint=False)
    xn = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(xn, xo, h)

                    oi, oj = h[i], h[j]
                    h[i] += d
                    h[j] -= d

                    if cn < best_c - 1e-15:

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(max(1, budget_s)) - 3.0
    rng = np.random.RandomState(seed)

    best_h = None
    best_n = 0
    best_c = 1e100

    try:
        ph = np.array(initial_h_values, dtype=float).copy()
        pn = len(ph)
    except Exception:
        ph = None
        pn = 512

    if ph is not None:
        ph = project(ph, pn)
        pc = max_c(ph, pn)
        best_h, best_n, best_c = ph, pn, pc
    else:
        ph = project(start_smooth_binary(512, rng), 512)
        pn = 512
        pc = max_c(ph, pn)
        best_h, best_n, best_c = ph, pn, pc

    if deadline - time.time() > 220:
        small_ns = [16, 20, 24, 32, 48]
        small_budget = min(120.0, max(0.0, deadline - time.time() - 360.0))
        if small_budget > 20:
            t_s = time.time()
            per = small_budget / len(small_ns)
            for nt in small_ns:
                if time.time() > t_s + small_budget - 5:
                    break
                alloc = min(per + 2.0, deadline - time.time() - 260.0)
                if alloc < 4:
                    continue
                try:
                    sh, sc = explore_n(nt, alloc, rng, ph)
                    if sh is not None and sc < best_c:
                        best_h, best_n, best_c = sh, nt, sc
                except Exception:
                    pass

    if ph is not None:
        rem = deadline - time.time()
        if rem > 180 and pn != best_n and best_c - max_c(ph, pn) < 0.02:
            end = min(time.time() + min(250.0, 0.35 * rem), deadline - 140)
            try:
                hh, cc = deep_opt(ph, pn, end, rng)
                if cc < best_c:
                    best_h, best_n, best_c = hh, pn, cc
            except Exception:
                pass

    rem = deadline - time.time()
    if rem > 40 and best_h is not None:
        end = time.time() + min(360.0, 0.58 * rem)
        if end > deadline - 18:
            end = deadline - 18
        try:
            bh, bc = deep_opt(best_h, best_n, end, rng)
            if bc < best_c:
                best_h, best_n, best_c = bh, best_n, bc
        except Exception:
            pass

    if deadline - time.time() > 4 and best_h is not None:
        try:
            fh = project(best_h, best_n)
            fc = max_c(fh, best_n)
            fh, fc = pair_micro(fh, best_n, deadline - 1, fc, rng)
            if fc < best_c:
                best_h, best_c = fh, fc
        except Exception:
            pass

    best_h = project(best_h, best_n)
    best_c = max_c(best_h, best_n)
    return best_h, float(best_c), int(best_n)
```
