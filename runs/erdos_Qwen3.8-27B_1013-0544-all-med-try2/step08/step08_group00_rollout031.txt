def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = 0.5 * float(n)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            room = 1.0 - h
            tot = room[mask].sum()
            if tot <= 1e-15:
                break
            h[mask] += diff * room[mask] / tot
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            tot = h[mask].sum()
            if tot <= 1e-15:
                break
            h[mask] += diff * h[mask] / tot
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)

def interpolate_to(h, new_n):
    old_n = len(h)
    if old_n == new_n: return h.copy()
    if old_n == 1: return np.full(new_n, h[0])
    x0 = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x1 = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x1, x0, h)

import numpy as np
import time

DEFAULT_ALPHAS = (
    1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005,
    0.002, 0.001, 0.0005, 0.0002, 0.0001,
    5e-05, 2e-05, 1e-05, 5e-06
)
DEFAULT_CAPS = (
    8, 10, 10, 12, 15, 20, 30, 35,
    45, 55, 65, 75, 85, 95, 105, 115, 120
)


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = 0.5 * float(n)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            room = 1.0 - h
            tot = room[mask].sum()
            if tot <= 1e-15:
                break
            h[mask] += diff * room[mask] / tot
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            tot = h[mask].sum()
            if tot <= 1e-15:
                break
            h[mask] += diff * h[mask] / tot
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / float(mask.sum())
            h = np.clip(h, 0.0, 1.0)
    return h


def make_c_unweighted(h):
    return np.correlate(h, 1.0 - h, mode="full")[::-1]


def weighted_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / n)


def consider_solution(h, best_h, best_w, best_n):
    n = len(h)
    h = project_feasible(h, n)
    w = weighted_c5(h)
    if w < best_w - 1e-15:
        return h, float(w), int(n)
    return best_h, best_w, best_n


def make_matrices(n):
    r = np.arange(n)
    P = r[:, None]
    R = r[None, :]
    diagA = (n - 1 + P - R).astype(np.int32)
    diagB = (n - 1 + R - P).astype(np.int32)
    d = np.abs(P - R).astype(np.int32)
    pos = (n - 1 + d).astype(np.int32)
    neg = (n - 1 - d).astype(np.int32)
    return r, diagA, diagB, pos, neg


def build_M(h, rows, diagA, diagB):
    n = len(h)
    M = np.zeros((n, 2 * n - 1), dtype=np.float64)
    om = 1.0 - h
    mh = -h
    for q in range(n):
        v1 = om[q]
        if v1 != 0.0:
            M[rows, diagA[q]] += v1
        v2 = mh[q]
        if v2 != 0.0:
            M[rows, diagB[q]] += v2
    return M


def pair_descent(h, alpha, t_end, rows, diagA, diagB, posmat, negmat):
    n = len(h)
    h = project_feasible(h, n).copy()
    c = make_c_unweighted(h)
    best = float(c.max())
    a = float(alpha)
    a2 = a * a
    tol = 1e-12
    M = build_M(h, rows, diagA, diagB)
    B = c - a * M
    updates = 0

    while time.time() < t_end:
        if not np.any(h >= a - tol) or not np.any(h + a <= 1.0 + tol):
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return h, best
            if h[i] + a > 1.0 + tol:
                continue
            valid = h >= a - tol
            valid[i] = False
            if not valid.any():
                continue

            D = B.copy()
            D += a * M[i]

            cand = D.max(axis=1)
            q = np.maximum(D[rows, posmat[i]] + a2, D[rows, negmat[i]] + a2)
            np.maximum(cand, q, out=cand)
            cand[~valid] = np.inf

            j = int(np.argmin(cand))
            val = float(cand[j])
            if val < best - 1e-10:
                if h[j] < a - tol or h[i] + a > 1.0 + tol:
                    continue

                mi = M[i]
                mj = M[j]
                delta_c = a * (mi - mj)
                p = posmat[i, j]
                ng = negmat[i, j]

                c += delta_c
                c[p] += a2
                c[ng] += a2

                B += delta_c
                B[rows, diagA[i]] += a2
                B[rows, diagB[i]] += a2
                B[rows, diagA[j]] -= a2
                B[rows, diagB[j]] -= a2

                h[i] += a
                h[j] -= a

                M[rows, diagA[i]] -= a
                M[rows, diagB[i]] -= a
                M[rows, diagA[j]] += a
                M[rows, diagB[j]] += a

                best = float(c.max())
                improved = True
                updates += 1

                if updates % 500 == 0:
                    c = make_c_unweighted(h)
                    best = float(c.max())
                    M = build_M(h, rows, diagA, diagB)
                    B = c - a * M

        if not improved:
            break

    return h, best


def polish_timed(h, mats, deadline, cap_scale=1.0, start_idx=0):
    n = len(h)
    rows, diagA, diagB, posmat, negmat = mats
    h = project_feasible(h, n).copy()
    best_unw = float(make_c_unweighted(h).max())
    best_h = h.copy()

    last_idx = len(DEFAULT_ALPHAS)
    for idx in range(max(0, int(start_idx)), last_idx):
        now = time.time()
        if now > deadline - 1.0:
            break
        cap = DEFAULT_CAPS[idx] * float(cap_scale)
        t_end = min(now + cap, deadline - 1.0)
        if t_end - now < 0.5:
            continue
        h_st, _ = pair_descent(
            best_h, DEFAULT_ALPHAS[idx], t_end, rows, diagA, diagB, posmat, negmat
        )
        h_st = project_feasible(h_st, n)
        unw = float(make_c_unweighted(h_st).max())
        if unw < best_unw - 1e-12:
            best_unw = unw
            best_h = h_st.copy()

    return best_h, best_unw


def interpolate_to(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    if old_n == 1:
        return np.full(new_n, h[0], dtype=float)
    x0 = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x1 = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x1, x0, h)


def binary_from_threshold(h, n, rng):
    b = (h >= 0.5).astype(float)
    target = n // 2
    cnt = int(b.sum())
    if cnt > target:
        ones = np.where(b == 1.0)[0]
        if ones.size:
            order = np.argsort(h[ones])
            for idx in order[:cnt - target]:
                b[idx] = 0.0
    elif cnt < target:
        zeros = np.where(b == 0.0)[0]
        if zeros.size:
            order = np.argsort(-h[zeros])
            for idx in order[:target - cnt]:
                b[idx] = 1.0

    cnt = int(b.sum())
    if cnt != target:
        if cnt > target:
            ones = np.where(b == 1.0)[0]
            if ones.size:
                b[ones[0]] = 0.0
        else:
            zeros = np.where(b == 0.0)[0]
            if zeros.size:
                b[zeros[0]] = 1.0
    return b


def random_binary(n, rng):
    b = np.zeros(n, dtype=float)
    k = n // 2
    if k:
        idx = rng.choice(n, size=k, replace=False)
        b[idx] = 1.0
    return b

        best_h, best_w, best_n = consider_solution(h_bin, best_h, best_w, best_n)
        t_end = min(deadline - 10.0, time.time() + 55.0)
        h_pol, _ = polish_timed(h_bin, mats_use, t_end, cap_scale=0.45, start_idx=0)
        best_h, best_w, best_n = consider_solution(h_pol, best_h, best_w, best_n)

    if time.time() < deadline - 40.0 and best_n % 2 == 0:
        mats_use = mats if best_n == n_init else make_matrices(best_n)
        for _ in range(2):
            if time.time() > deadline - 30.0:
                break
            h_r = random_binary(best_n, rng)
            best_h, best_w, best_n = consider_solution(h_r, best_h, best_w, best_n)
            t_end = min(deadline - 8.0, time.time() + 28.0)
            h_pol, _ = polish_timed(h_r, mats_use, t_end, cap_scale=0.25, start_idx=0)
            best_h, best_w, best_n = consider_solution(h_pol, best_h, best_w, best_n)

    for n_other in (600, 512):
        if time.time() > deadline - 55.0:
            break
        h2 = project_feasible(interpolate_to(best_h, n_other), n_other)
        best_h, best_w, best_n = consider_solution(h2, best_h, best_w, best_n)
        if time.time() > deadline - 45.0:
            continue
        mats2 = make_matrices(n_other)
        t_end = min(deadline - 10.0, time.time() + 55.0)
        h2, _ = polish_timed(h2, mats2, t_end, cap_scale=0.40, start_idx=0)
        best_h, best_w, best_n = consider_solution(h2, best_h, best_w, best_n)

    if time.time() < deadline - 3.0:
        mats_final = mats if best_n == n_init else make_matrices(best_n)
        h_pol, _ = polish_timed(best_h, mats_final, deadline - 1.0, cap_scale=0.20, start_idx=10)
        best_h, best_w, best_n = consider_solution(h_pol, best_h, best_w, best_n)

    best_h = project_feasible(best_h, best_n)
    best_w = weighted_c5(best_h)
    return best_h, float(best_w), int(best_n)
