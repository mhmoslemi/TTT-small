    c = np.correlate(h, 1.0 - h, mode='full')

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-13:
            break
        if diff > 0:

            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def slsqp_refine(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        if m < 1e-15:
            return 0.0
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

f(h) = max_k sum_i h[i](1-h[i+k]) * dx

This is the max of bilinear functions. It's not convex. But we can use a "smooth max" approximation:

where g_k(h) = sum_i h[i](1-h[i+k]) * dx.

This f_α is not convex either (since g_k is bilinear), but it's smooth. We can use gradient-based methods.

The gradient of g_k with respect to h[m] is:
∂g_k/∂h[m] = (1-h[m+k]) * dx + (if m-k is valid) -h[m-k] * dx
= dx * [(1-h[m+k]) - h[m-k]] (with appropriate boundary handling)

So ∂f_α/∂h[m] = sum_k p_k * ∂g_k/∂h[m] where p_k = exp(α*g_k) / sum_j exp(α*g_j).

This is a weighted average of the partial derivatives, weighted by the softmax distribution over shifts.

We can use this for gradient descent with projection onto the constraint set {0≤h≤1, sum(h)=n/2}.

        h = h_new

            A[ms:me] += ps * h[ms + s:me + s]
        ms = max(0, s)
        me = min(n, n + s)
        if ms < me:
            B[ms:me] += ps * h[ms - s:me - s]
    grad = dx * (1.0 - A - B)
    norm = np.max(np.abs(grad))
    if norm < 1e-18:
        return h
    h_new = h - (lr_scale / norm) * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h_new
            sr = room.sum()
            if sr > 0:
                h_new += (diff / sr) * room
        else:
            sr = h_new.sum()
            if sr > 0:
                h_new += (diff / sr) * h_new
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def binary_anneal(h_init, n, deadline, seed=42, n_restarts=5, sweeps=600):
    target = n // 2
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            n_flips = rng.randint(4, min(12, n // 20))
            for _ in range(n_flips):
                if not ones or not zeros:
                    break
                a = int(rng.choice(ones))
                b = int(rng.choice(zeros))
                h[a], h[b] = 0.0, 1.0
                ones.remove(a)
                ones.append(b)
                zeros.remove(b)
                zeros.append(a)
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur_c = corr.max() * dx

        T_start = 8e-4
        T_end = 1e-10
        for sweep in range(sweeps):
            if time.time() > deadline:
                break
            progress = sweep / max(1, sweeps - 1)
            T = T_start * (T_end / T_start) ** progress
            k_max = int(np.argmax(corr))
            s_max = k_max - (n - 1)
            contrib = [i for i in ones if 0 <= i + s_max < n and h[i + s_max] == 0]
            if contrib and rng.random() < 0.75:
                i = int(rng.choice(contrib))
            else:
                i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            if i == j:
                continue
            h[i], h[j] = 0.0, 1.0
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_c = float(new_corr.max()) * dx
            delta = new_c - cur_c
            accept = delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T))
            if accept:
                cur_c = new_c
                corr = new_corr
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                h[i], h[j] = 1.0, 0.0

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_c = float(corr.max()) * dx
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Binary annealing
    p1 = t0 + 0.22 * (budget_s - 5)
    hb, cb = binary_anneal(best_h, n, p1, seed=seed, n_restarts=5, sweeps=500)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: Smooth gradient descent
    p2 = t0 + 0.32 * (budget_s - 5)
    h_smooth = project_feasible(best_h.astype(float), n)
    alphas = [200, 500, 1000, 2000, 5000, 10000]
    for a in alphas:
        if time.time() > p2:
            break
        for _ in range(200):
            if time.time() > p2:
                break
            h_smooth = smooth_grad_step(h_smooth, n, a, lr_scale=0.5)
        c = compute_c5(h_smooth, n)
        if c < best_c:
            best_c = c
            best_h = h_smooth.copy()

    # Phase 3: SLSQP polish
    p3 = t0 + 0.42 * (budget_s - 5)
    if time.time() < p3:
        hp = slsqp_refine(best_h.copy(), n, 50000, 1500, p3)
        c = compute_c5(hp, n)
        if c < best_c:
            best_c = c
            best_h = hp.copy()

    # Phase 4: More binary annealing from best
    p4 = t0 + 0.68 * (budget_s - 5)
    hb, cb = binary_anneal(best_h, n, p4, seed=seed + 200, n_restarts=6, sweeps=700)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 5: Try other resolutions
    p5 = t0 + 0.82 * (budget_s - 5)
    for nh in [200, 300, 350]:
        if time.time() > p5 - 15:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, nh)
        idx2 = np.argsort(hi)[::-1]
        hbin = np.zeros(nh)
        hbin[idx2[:nh // 2]] = 1.0
        rem = max(20, p5 - time.time() - 5)
        hb2, cb2 = binary_anneal(hbin, nh, t0 + rem, seed=seed + nh, n_restarts=3, sweeps=300)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
            n = nh

    # Phase 6: Final gradient + SLSQP + binary
    if time.time() < dl - 20:
        hf = project_feasible(best_h.astype(float), n)
        for a in [5000, 20000]:
            if time.time() > dl - 15:
                break
            for _ in range(100):
                if time.time() > dl - 15:
                    break
                hf = smooth_grad_step(hf, n, a, lr_scale=0.3)
        c = compute_c5(hf, n)
        if c < best_c:
            best_c = c
            best_h = hf.copy()
        if time.time() < dl - 10:
            hb3, cb3 = binary_anneal(best_h, n, dl - 3, seed=seed + 500, n_restarts=2, sweeps=300)
            if cb3 < best_c:
                best_c = cb3
                best_h = hb3.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
