def ternary_pair(h, i, j, n, best_c, sign, max_iter=22):
    if sign > 0:
        hi = min(1.0 - h[i], h[j])
    else:
        hi = min(h[i], 1.0 - h[j])
    if hi < 1e-13:
        return h, best_c
    lo = 0.0
    for _ in range(max_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        h[i] += sign * m1
        h[j] -= sign * m1
        c1 = compute_c5(h, n)
        h[i] -= sign * m1
        h[j] += sign * m1
        h[i] += sign * m2
        h[j] -= sign * m2
        c2 = compute_c5(h, n)
        h[i] -= sign * m2
        h[j] += sign * m2
        if c1 < c2:
            hi = m2
        else:
            lo = m1
    delta = (lo + hi) / 2.0
    h[i] += sign * delta
    h[j] -= sign * delta
    c_new = compute_c5(h, n)
    if c_new < best_c:
        return h, c_new
    h[i] -= sign * delta
    h[j] += sign * delta
    return h, best_c

def quick_pair_check(h, i, j, n, best_c, delta=0.005):
    for sign in (1, -1):
        if sign > 0:
            if h[i] + delta > 1.0 or h[j] - delta < 0.0:
                continue
        else:
            if h[i] - delta > 1.0 or h[j] + delta < 0.0:
                continue
        h[i] += sign * delta
        h[j] -= sign * delta
        c = compute_c5(h, n)
        if c < best_c:
            h, c_new = ternary_pair(h, i, j, n, c, sign)
            return h, c_new, True
        h[i] -= sign * delta
        h[j] += sign * delta
    return h, best_c, False
