    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def lp_local_search(h_init, n, deadline, max_constraints=50, max_iter=500):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    active_lags = []
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_curr = compute_c5_vec(h, n)
        max_val = c_curr.max()
        if max_val < best_c - 1e-14:
            best_c = max_val
            best_h = h.copy()
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        A_ub_rows = []
        b_ub_vals = []
        for lag_idx in active_lags:
            k = lag_idx - (n - 1)
            g = grad_single(h, n, k) * (2.0 / n)
            row = np.zeros(n + 1)
            row[:n] = g
            row[-1] = -1.0
            A_ub_rows.append(row)
            b_ub_vals.append(-c_curr[lag_idx] + np.dot(g, h))
        if A_ub_rows:
            A_ub = np.array(A_ub_rows)
            b_ub = np.array(b_ub_vals)
        else:
            A_ub = np.zeros((1, n + 1))
            b_ub = np.array([best_c])
        bounds = [(0, 1)] * n + [(-np.inf, best_c + 0.01)]
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if not res.success:
            break
        h_new = res.x[:n]
        c_new = compute_c5_vec(h_new, n)
        max_idx = int(np.argmax(c_new))
        t_lp = res.x[-1]
        if c_new[max_idx] > t_lp + 1e-8:
            if max_idx not in active_lags:
                active_lags.append(max_idx)
                if len(active_lags) > max_constraints:
                    active_lags = active_lags[-max_constraints:]
        if np.max(np.abs(h_new - h)) < 1e-12:
            break
        h = project_feasible(h_new, n)
    return best_h, best_c

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 100.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    top_k = min(80, len(w))
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n)
    for i in top_idx:
        if w[i] < 1e-14:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    s = x.sum() - n / 2.0
    grad += 200.0 * s
    return grad

def momentum_pgd(h_init, n, deadline, top_k=30, base_step=0.04, momentum=0.85):
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    h = h_init.copy()
    velocity = np.zeros(n)
    step = base_step
    stall = 0
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        m = c.max()
        threshold = m - 0.003
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(c)[-top_k:]
        elif len(active) > top_k:
            active = np.argsort(c)[-top_k:]
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            grad += dx * grad_single(h, n, k)
        grad /= max(1, len(active))
        s = h.sum() - n / 2.0
        grad += 50.0 * s
        velocity = momentum * velocity - step * grad
        h_new = project_feasible(h + velocity, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            velocity *= 0.5
            step *= 0.85
            stall += 1
            if stall > 40:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h = np.zeros(500)
        h[rng.choice(500, 250, replace=False)] = 1.0
        h_init = project_feasible(h, 500)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    best_n = n
    # Phase 1: LP local search
    t1 = min(t0 + 150, deadline - 400)
    best_h, best_c = lp_local_search(best_h, best_n, t1, max_constraints=40, max_iter=1000)
    # Phase 2: Momentum PGD
    t2 = min(time.time() + 120, deadline - 350)
    best_h, best_c = momentum_pgd(best_h, best_n, t2, top_k=30)
    # Phase 3: L-BFGS
    t3 = min(time.time() + 120, deadline - 250)
    for alpha in [2000, 8000]:
        if time.time() > t3:
            break
        bounds = [(0.0, 1.0)] * best_n
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                         args=(best_n, alpha), method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': 200, 'ftol': 1e-16})
            h_new = project_feasible(res.x, best_n)
            c_new = compute_c5(h_new, best_n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    # Phase 4: LP again
    t4 = min(time.time() + 120, deadline - 200)
    best_h, best_c = lp_local_search(best_h, best_n, t4, max_constraints=50, max_iter=1000)
    # Phase 5: Multi-start with perturbations
    t5 = min(time.time() + 200, deadline - 100)
    strengths = [0.03, 0.06, 0.02, 0.05, 0.04]
    for s in strengths:
        if time.time() > t5:
            break
        h_try = project_feasible(best_h + rng.randn(best_n) * s, best_n)
        sub_deadline = min(time.time() + 35, t5)
        h_t, c_t = lp_local_search(h_try, best_n, sub_deadline, max_constraints=35, max_iter=400)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
    # Phase 6: Higher resolution
    for n_try in [600, 700]:
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        t_rem = min(50, deadline - time.time() - 40)
        if t_rem < 15:
            continue
        h_t, c_t = lp_local_search(h_interp, n_try, time.time() + t_rem,
                                   max_constraints=45, max_iter=600)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try
    # Phase 7: Final LP polish
    if time.time() < deadline - 3:
        best_h, best_c = lp_local_search(best_h, best_n, deadline - 1,
                                         max_constraints=50, max_iter=3000)
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
