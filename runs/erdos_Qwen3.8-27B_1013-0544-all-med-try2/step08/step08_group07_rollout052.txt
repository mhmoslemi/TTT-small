smooth_max(g; α) = m + (1/α) * log(Σ_k exp(α * (g_k - m)))

So ∂f_α/∂x_i = dx * Σ_k p_k * ∂g_k/∂x_i

where g_k = corr_k(x) = sum_j x[j+k]*(1-x[j]) (with appropriate indices).

∂g_k/∂x_i = (1-x[i-k]) - x[i+k] (for valid indices) = B_i[k] (the same B vector!)

```python
def smooth_gradient(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1-x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    p = np.exp(shifted)
    p /= p.sum()
    # grad[i] = dx * sum_k p[k] * B_i[k]
    # B_i[k] = (1-x[i-m_k]) - x[i+m_k] where m_k = k-(n-1)
    m_range = np.arange(-(n-1), n)
    # Vectorized:
    i_idx = np.arange(n)[:, None]
    im = i_idx - m_range[None, :]
    ip = i_idx + m_range[None, :]
    vm = (im >= 0) & (im < n)
    vp = (ip >= 0) & (ip < n)
    B = (1-x[np.clip(im, 0, n-1)]) * vm - x[np.clip(ip, 0, n-1)] * vp
    grad = dx * (B @ p)
    return grad, g, m
```

This is O(n²) per gradient evaluation, which is the same as computing the correlation. So it's not saving much over just computing the correlation directly.

But the advantage of projected gradient is that it's simple, stable, and can be run for many iterations. With a good step size (e.g., backtracking or Lipschitz-based), it can converge well.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greedy_binary(h_init, n, deadline, max_sweeps=300):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    best_h = h.copy()
    best_c = corr.max() * dx
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        for i in zeros:
            for j in ones:
                delta = np.zeros(2 * n - 1)
                im = i + m_range; msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range; msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range; msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range; msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    best_c = nc
                    best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def coord_descent(h, n, deadline, bs_iters=25, skip_thresh=5e-5):
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    for sweep in range(20000):
        if time.time() > deadline:
            break
        any_update = False
        for i in range(n):
            if h[i] < skip_thresh or h[i] > 1.0 - skip_thresh:
                continue
            B = np.zeros(2 * n - 1)
            k1 = i - m_range; m1 = (k1 >= 0) & (k1 < n)
            B[m1] += (1.0 - h[k1[m1]])
            k2 = i + m_range; m2 = (k2 >= 0) & (k2 < n)
            B[m2] -= h[k2[m2]]
            lo, hi = 0.0, 1.0
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zmax = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(bs_iters):
                mid = (lo + hi) * 0.5
                if zmax > mid + 1e-14:
                    lo = mid; continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, ((mid - corr[neg_B]) / B[neg_B]).max())
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, ((mid - corr[pos_B]) / B[pos_B]).min())
                if lb_x <= ub_x + 1e-12:
                    hi = mid
                else:
                    lo = mid
            x_opt = np.clip((lo + hi) * 0.5, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-11:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
                any_update = True
        if not any_update:
            break
    return h, corr

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def projected_smooth_hg(h, n, deadline, alpha, max_iters=2000):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    i_idx = np.arange(n)[:, None]
    im = i_idx - m_range[None, :]
    ip = i_idx + m_range[None, :]
    vm = (im >= 0) & (im < n)
    vp = (ip >= 0) & (ip < n)
    imc = np.clip(im, 0, n - 1)
    ipc = np.clip(ip, 0, n - 1)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        p = np.exp(shifted - sm)
        p /= p.sum()
        B = (1.0 - h[imc]) * vm - h[ipc] * vp
        grad = dx * (B @ p)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.7
            if step < 1e-6:
                break
    return best_h, best_c
```
