def get_fft_n(n):
    n = int(n)
    if n in _N_CACHE:
        return _N_CACHE[n]
    m = max(1, 2 * n - 1)
    N = 1
    while N < m:
        N <<= 1
    _N_CACHE[n] = N
    return N


def make_even(n):
    n = int(n)
    if n < 2:
        return 2
    return n if n % 2 == 0 else n - 1


def project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    n = v.size
    target = float(target)
    if n == 0:
        return v.copy()
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    if not np.isfinite(lo):
        lo = -1.0
    if not np.isfinite(hi):
        hi = 1.0

    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

    out = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - out.sum()
    if abs(diff) > 1e-14:
        if diff > 0.0:
            idx = np.where(out < 1.0 - 1e-15)[0]
            if idx.size:
                out[idx] += diff / float(idx.size)
            else:
                idx = np.where(out < 1.0)[0]
                if idx.size:
                    out[idx[0]] += min(diff, 1.0 - out[idx[0]])
        else:
            idx = np.where(out > 1e-15)[0]
            if idx.size:
                out[idx] += diff / float(idx.size)
            else:
                idx = np.where(out > 0.0)[0]
                if idx.size:
                    out[idx[0]] += max(diff, -out[idx[0]])
        out = np.clip(out, 0.0, 1.0)

    for _ in range(4):
        diff = target - out.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            idx = np.where(out < 1.0)[0]
            if idx.size == 0:
                break
            out[idx[0]] = min(1.0, out[idx[0]] + diff)
        else:
            idx = np.where(out > 0.0)[0]
            if idx.size == 0:
                break
            out[idx[0]] = max(0.0, out[idx[0]] + diff)
    return out


def compute_c_full(h, n):
    n = int(n)
    N = get_fft_n(n)
    h = np.asarray(h, dtype=float)
    b = 1.0 - h
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(np.conj(fa) * fb, n=N)
    c = np.empty(2 * n - 1, dtype=float)
    if n > 1:
        c[:n - 1] = r[N - (n - 1):N]
    c[n - 1:] = r[:n]
    return np.maximum(c, 0.0)


def compute_c_direct_asc(h, n):
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode='full')[::-1]
    return np.maximum(c, 0.0)


def compute_c5_direct(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / float(n)))


def build_gradient_from_weights(w, h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    L = 2 * n - 1
    active = np.where(w > 1e-15)[0]
    if active.size < 64:
        g = np.zeros(n, dtype=float)
        for m in active:
            wm = float(w[m])
            d = int(m) - (n - 1)
            if d == 0:
                g += wm * (1.0 - 2.0 * h)
            elif d > 0:
                nv = n - d
                g[d:] += wm * (1.0 - h[:nv])
                g[:nv] += -wm * h[d:]
            else:
                s = -d
                nv = n - s
                g[:nv] += wm * (1.0 - h[s:])
                g[s:] += -wm * h[:nv]
        return g

    P = np.zeros(n, dtype=float)
    M = np.zeros(n, dtype=float)
    if n > 1:
        P[1:n] = w[n:L]
        M[1:n] = w[0:n - 1]
    A = P + M
    rp = np.convolve(h, A[::-1], mode='full')[n - 1:2 * n - 1]
    rm = np.convolve(h, A, mode='full')[:n]
    g = w[n - 1] * (1.0 - 2.0 * h)
    if n > 1:
        g = g + np.cumsum(M)[::-1] + np.cumsum(P) - rp - rm
    return g


def _obj(x, n, alpha, mu):
    c = compute_c_full(x, int(n))
    m = float(c.max())
    z = float(alpha) * (c - m)
    zm = float(z.max())
    val = m + (zm + float(np.log(np.sum(np.exp(z - zm))))) / float(alpha)
    s = float(x.sum()) - 0.5 * float(n)
    if mu:
        val += float(mu) * s * s
    return float(val)


def _grad(x, n, alpha, mu):
    n = int(n)
    c = compute_c_full(x, n)
    m = float(c.max())
    z = float(alpha) * (c - m)
    zm = float(z.max())
    w = np.exp(z - zm)
    w = w / w.sum()
    g = build_gradient_from_weights(w, x, n)
    s = float(x.sum()) - 0.5 * float(n)
    if mu:
        g += 2.0 * float(mu) * s
    return g


def smooth_gradient(h, n, alpha):
    return _grad(h, int(n), float(alpha), 0.0)


def _eq_sum(x, target):
    return float(x.sum() - target)


def _eq_jac(x, target):
    return np.ones_like(x, dtype=float)


def precompute_pair_indices(n):
    n = int(n)
    if n in _PAIR_CACHE:
        return _PAIR_CACHE[n]
    rs = np.arange(n, dtype=np.int32).reshape(-1, 1)
    cs = np.arange(n, dtype=np.int32).reshape(1, -1)
    idxA = np.int32(n - 1) + rs - cs
    idxB = np.int32(n - 1) + cs - rs
    _PAIR_CACHE[n] = (idxA, idxB)
    return _PAIR_CACHE[n]


def local_pair_search(h, n, tlimit, alpha, k):

    Llen = 2 * n - 1
    c0 = n - 1
    drels = _DELTA_ARR
    no_imp = 0

                break

                low.append(int(idx))
        high = []
        for idx in order[::-1]:
            if len(high) >= k:
                break
            if h[idx] > 1e-7:
                high.append(int(idx))

        if not low or not high:
            break

            if time.time() > tlimit - 2.0:
                break

                L[idxB[i]] -= h

                D = D[D > 1e-12]
                if D.size == 0:
                    continue
                D2 = D * D
                off = abs(i - j)
                best_val = curmax
                best_d = 0.0

                for ii in range(D.size):
                    d = float(D[ii])
                    arr = C + d * L
                    q = float(D2[ii])
                    arr[c0] += -2.0 * q
                    if off:
                        arr[c0 + off] += q
                        arr[c0 - off] += q
                    val = float(arr.max())
                    if val < best_val:
                        best_val = val
                        best_d = d

                if best_d > 0.0 and best_val < curmax - 1e-11:
                    h[i] += best_d
                    h[j] -= best_d
                    h[i] = min(1.0, max(0.0, h[i]))
                    h[j] = min(1.0, max(0.0, h[j]))

                    C += best_d * L
                    q = best_d * best_d
                    C[c0] += -2.0 * q
                    if off:
                        C[c0 + off] += q
                        C[c0 - off] += q

                    curmax = best_val
                    improved += 1
                    c5 = curmax * (2.0 / float(n))
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()

        if improved == 0:
            no_imp += 1
            if no_imp >= 2:
                break
        else:
            no_imp = 0
            h = project_box_sum(h, target)
            C = compute_c_direct_asc(h, n)
            curmax = float(C.max())
            c5 = curmax * (2.0 / float(n))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    h = project_box_sum(h, target)
    c5 = compute_c5_direct(h, n)
    bh = project_box_sum(best_h, target)
    c5b = compute_c5_direct(bh, n)
    if c5b < c5:
        return bh, c5b
    return h, c5


def lbfgs_refine(h, n, tlimit, alphas, mus, maxiter):
    n = int(n)
    target = 0.5 * float(n)
    best_h = project_box_sum(h, target).copy()
    best_c = compute_c5_direct(best_h, n)
    bounds = [(0.0, 1.0)] * n

    for pos, alpha in enumerate(alphas):
        if time.time() > tlimit - 5.0:
            break
        mu = mus[pos % len(mus)]
        mt = int(max(20, maxiter))
        try:
            res = minimize(
                _obj, best_h, jac=_grad, args=(n, float(alpha), float(mu)),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': mt, 'ftol': 1e-15, 'gtol': 1e-11,
                         'maxfun': mt * 80, 'maxls': 30, 'maxcor': 20}
            )
            x = project_box_sum(res.x, target)
            c = compute_c5_direct(x, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = x
        except Exception:
            pass
    return best_h, best_c


def slsqp_refine(h, n, tlimit, alpha, maxiter):
    n = int(n)
    target = 0.5 * float(n)
    best_h = project_box_sum(h, target).copy()
    best_c = compute_c5_direct(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': _eq_sum, 'jac': _eq_jac, 'args': (target,)}

        c = compute_c5_direct(x, n)
        if c < best_c - 1e-15:

    except Exception:

    return best_h, best_c


def interp_h(h_old, n_old, n_new):
    n_old = int(n_old)
    n_new = int(n_new)
    if n_old == n_new:
        return h_old.copy()
    if n_old <= 0:
        return np.full(n_new, 0.5)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    y = np.interp(x_new, x_old, np.asarray(h_old, dtype=float))
    return project_box_sum(y, 0.5 * float(n_new))


def golden_start(n):
    n = int(make_even(max(8, n)))
    h = np.zeros(n, dtype=float)
    cnt = n // 2
    if cnt <= 0:
        return h
    step = int(round(0.6180339887498949 * n))
    if step <= 0 or step >= n:
        step = 1
    while gcd(step, n) != 1:
        step += 1
        if step >= n:
            step = 1
    pos = (np.arange(cnt, dtype=np.int64) * step) % n
    h[pos] = 1.0
    # If a collision somehow occurs, repair to exact half.
    have = int(np.count_nonzero(h))
    need = cnt - have
    if need > 0:
        idx = np.flatnonzero(h == 0.0)
        h[idx[:need]] = 1.0
    return project_box_sum(h, 0.5 * float(n))


def fourier_start(n, rng):
    n = int(make_even(max(8, n)))
    nr = len(np.fft.rfft(np.zeros(n)))
    spec = np.zeros(nr, dtype=complex)
    freq = np.arange(1, nr, dtype=float)
    if nr > 1:
        spec[1:] = (rng.randn(nr - 1) + 1j * rng.randn(nr - 1)) * (0.02 / np.sqrt(freq))
    h = 0.5 + np.fft.irfft(spec, n=n)
    return project_box_sum(h, 0.5 * float(n))

    return best_h, best_c, int(best_n)
