def compute_c5(h, n):
    dx = 2.0 / n

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0

        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def fast_anneal(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_c = float(corr.max())
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].astype(np.int32)
    zeros = np.where(h == 0)[0].astype(np.int32)
    half = n // 2
    T = 2.0
    cool = 0.999995
    scale = 50.0
    delta = np.zeros(2 * n - 1, dtype=np.float64)
    m_range = np.arange(-(n - 1), n)
    two_n_1 = 2 * n - 1
    while time.time() < deadline:
        ai = rng.randint(half)
        bi = rng.randint(half)
        i = int(zeros[ai])
        j = int(ones[bi])
        delta.fill(0.0)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < two_n_1:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < two_n_1:
            delta[k2] += 1.0
        new_max = float((corr + delta).max())
        if new_max <= cur_c or (T > 1e-8 and rng.random() < np.exp(scale * (cur_c - new_max) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = new_max
            zeros[ai] = j
            ones[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c * dx

def greedy_swap(h_init, n, deadline, max_sweeps=30):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    two_n_1 = 2 * n - 1
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_c = float(corr.max())
    ones = [int(x) for x in np.where(h == 1)[0]]
    zeros = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        best_gain = 0.0
        best_i, best_j = -1, -1
        best_delta = None
        for i in zeros:
            for j in ones:
                delta = np.zeros(two_n_1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < two_n_1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < two_n_1:
                    delta[k2] += 1.0
                nc = float((corr + delta).max())
                gain = cur_c - nc
                if gain > best_gain:
                    best_gain = gain
                    best_i, best_j = i, j
                    best_delta = delta
            if time.time() > deadline:
                break
        if best_i < 0:
            break
        h[best_i], h[best_j] = 1.0, 0.0
        corr += best_delta
        cur_c -= best_gain
        ones.remove(best_j)
        zeros.remove(best_i)
        ones.append(best_i)
        zeros.append(best_j)
    return h, cur_c * dx

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:

        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0
    # Phase 1: Greedy binary (n=400)
    p1 = min(t0 + 25, dl - 60)
    hb = to_binary(h0, n0)
    hg, cg = greedy_swap(hb, n0, p1, max_sweeps=5)
    if cg < best_c:
        best_c, best_h, best_n = cg, hg.copy(), n0
    # Phase 2: Main annealing (n=400) with restarts
    p2 = t0 + 0.55 * budget_s
    n_restarts = 4
    seg = (p2 - time.time()) / n_restarts
    for r in range(n_restarts):
        dl_r = time.time() + seg
        start_h = best_h if r > 0 else hb
        ha, ca = fast_anneal(start_h, n0, dl_r, seed=seed + r * 100)
        if ca < best_c:
            best_c, best_h, best_n = ca, ha.copy(), n0
    # Phase 3: Try other resolutions
    p3 = t0 + 0.75 * budget_s
    for nh in [200, 300, 500]:
        if time.time() > p3 - 8:
            break
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        seg3 = min(time.time() + 8, p3 - 5)
        hq, cq = greedy_swap(hh, nh, seg3 - 5, max_sweeps=3)
        if cq < compute_c5(hh, nh):
            hh = hq
        ha, ca = fast_anneal(hh, nh, seg3, seed=seed + 500 + nh)
        if ca < compute_c5(hh, nh):
            hh = ha
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, best_n = ch, hh.copy(), nh
    # Phase 4: SLSQP continuous refinement
    p4 = t0 + 0.90 * budget_s
    h_cont = best_h.copy()
    for alpha in [200, 2000, 20000, 200000]:
        if time.time() > p4 - 5:
            break
        h_cont = slsqp_optimize(h_cont, best_n, alpha, 400, 1e-14)
        c = compute_c5(h_cont, best_n)
        if c < best_c:
            best_c, best_h, best_n = c, h_cont.copy(), best_n
    # Phase 5: Final polish
    if time.time() < dl - 3:
        hp, cp = fast_anneal(best_h, best_n, dl - 2, seed=seed + 999)
        if cp < best_c:
            best_c, best_h, best_n = cp, hp.copy(), best_n
        if time.time() < dl - 1:
            h_cont = best_h.copy()
            for alpha in [5000, 50000]:
                if time.time() > dl - 0.5:
                    break
                h_cont = slsqp_optimize(h_cont, best_n, alpha, 200, 1e-14)
                c = compute_c5(h_cont, best_n)
                if c < best_c:
                    best_c, best_h, best_n = c, h_cont.copy(), best_n
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
