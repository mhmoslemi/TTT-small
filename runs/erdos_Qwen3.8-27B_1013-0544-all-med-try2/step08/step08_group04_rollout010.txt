def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    low = -float(np.max(h)) if n > 0 else 0.0
    high = 1.0 - float(np.min(h)) if n > 0 else 0.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        ss = float(np.clip(h + mid, 0.0, 1.0).sum())
        if ss < target:
            low = mid
        else:
            high = mid
    hp = np.clip(h + 0.5 * (low + high), 0.0, 1.0)
    diff = target - float(hp.sum())
    if abs(diff) > 1e-12:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            hp[mask] += diff / cnt
        else:
            if diff > 0:
                idx = int(np.argmin(hp))
            else:
                idx = int(np.argmax(hp))
            hp[idx] = float(np.clip(hp[idx] + diff, 0.0, 1.0))
    return np.clip(hp, 0.0, 1.0)

def smooth_grad_free(x,n,alpha,mu,target):
    dx=2.0/float(n)
    V=np.correlate(x,1.0-x,mode='full')*dx
    m=float(np.max(V))
    z=alpha*(V-m)
    e=np.exp(np.clip(z,-700.0,0.0))
    s=float(np.sum(e))
    if s <= 0.0:
       w=np.zeros_like(V); w[n-1]=1.0
    else:
       w=e/s
    grad = float(w[n-1]) * dx * (1.0 - 2.0*x)
    mid=n-1
    for d in range(1,n):
       wp=float(w[mid-d]); wn=float(w[mid+d])
       sw=wp+wn
       if sw <= 1e-15: continue
       nv=n-d
       wdx=dx*sw
       grad[:nv] += dx*wp - wdx*x[d:]
       grad[d:] += dx*wn - wdx*x[:nv]
    if mu != 0.0:
       grad = grad + 2.0*mu*(float(x.sum())-target)*np.ones(n)
    return grad
```

import numpy as np

def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    low = -float(np.max(h))
    high = 1.0 - float(np.min(h))
    for _ in range(60):
        mid = 0.5 * (low + high)
        ss = float(np.clip(h + mid, 0.0, 1.0).sum())
        if ss < target:
            low = mid
        else:
            high = mid
    hp = np.clip(h + 0.5 * (low + high), 0.0, 1.0)
    diff = target - float(hp.sum())
    if abs(diff) > 1e-12:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            hp[mask] += diff / cnt
        else:
            idx = int(np.argmin(hp)) if diff > 0 else int(np.argmax(hp))
            hp[idx] = float(np.clip(hp[idx] + diff, 0.0, 1.0))
    return np.clip(hp, 0.0, 1.0)


def c5_val(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def active_top_d(h, n, topd):
    dx = 2.0 / float(n)
    v = np.correlate(h, 1.0 - h, mode="full") * dx
    mid = n - 1
    items = []
    for d in range(1, n):
        items.append((max(float(v[mid - d]), float(v[mid + d])), d))
    if float(v[mid]) > 1e-15:
        items.append((float(v[mid]), 0))
    items.sort(reverse=True)
    return items[:topd]


def test_pair_move(h, n, i, j, amt, sgn, best, factors):
    bt = 0.0
    bv = best
    for f in factors:
        t = sgn * float(f) * float(amt)
        if abs(t) < 1e-15:
            continue
        x = h[i] + t
        y = h[j] - t
        if x < -1e-12 or x > 1.0 + 1e-12 or y < -1e-12 or y > 1.0 + 1e-12:
            continue
        h[i] = x
        h[j] = y
        c = c5_val(h, n)
        if c < bv - 1e-15:
            bv = c
            bt = t
        h[i] -= t
        h[j] += t
    if bt != 0.0:
        h[i] += bt
        h[j] -= bt
        h[i] = float(np.clip(h[i], 0.0, 1.0))
        h[j] = float(np.clip(h[j], 0.0, 1.0))
        c = c5_val(h, n)
        if c < best - 1e-15:
            return True, c
        h[i] -= bt
        h[j] += bt
    return False, best


def targeted_pair_search(h, n, deadline, rng=None, topd=12, tope=14):
    if rng is None:
        rng = np.random.RandomState(12345)
    factors = (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625, 0.001953125)
    h = project_feasible(np.array(h, dtype=float, copy=True), n)
    best_c = c5_val(h, n)
    best_h = h.copy()
    no_improve = 0
    while time.time() < deadline:
        top = active_top_d(h, n, topd)
        if not top:
            break
        improved = False
        for _, d in top:
            if time.time() >= deadline:
                break
            if d == 0:
                k = min(tope, n)
                hi = np.argpartition(h, n - k)[-k:] if k < n else np.arange(n)
                lo = np.argpartition(h, k)[:k] if k < n else np.arange(n)
                for a in hi:
                    for b in lo:
                        if a == b or h[a] < h[b] + 1e-8:
                            continue
                        tmax = float(min(h[b], 1.0 - h[a]))
                        if tmax > 1e-10:
                            ok, c = test_pair_move(h, n, int(a), int(b), tmax, 1.0, best_c, factors)
                            if ok:
                                best_c = c
                                best_h = h.copy()
                                improved = True
                                break
                    if improved:
                        break
                if improved:
                    break
                continue
            nv = n - d
            pvec = h[:nv] * (1.0 - h[d:])
            nvec = h[d:] * (1.0 - h[:nv])
            for vec, sgn in ((pvec, -1.0), (nvec, 1.0)):
                if float(np.max(vec)) > 1e-14:
                    k = min(tope, nv)
                    idx = np.argpartition(vec, nv - k)[-k:] if k < nv else np.arange(nv)
                    for pos in idx:
                        i = int(pos)
                        j = i + d
                        if sgn < 0.0:
                            tmax = float(min(h[i], 1.0 - h[j]))
                        else:
                            tmax = float(min(h[j], 1.0 - h[i]))
                        if tmax > 1e-10:
                            ok, c = test_pair_move(h, n, i, j, tmax, sgn, best_c, factors)
                            if ok:
                                best_c = c
                                best_h = h.copy()
                                improved = True
                                break
                if improved:
                    break
            if improved:
                break
        if improved:
            no_improve = 0
        else:
            rand_ok = False
            for _ in range(60):
                if time.time() >= deadline:
                    break
                val, d = top[int(rng.randint(len(top)))]
                if d == 0:
                    a = int(rng.randint(n))
                    b = int(rng.randint(n))
                    if a == b or h[a] < h[b] + 1e-8:
                        continue
                    tmax = float(min(h[b], 1.0 - h[a]))
                    if tmax > 1e-10:
                        ok, c = test_pair_move(h, n, a, b, tmax, 1.0, best_c, factors)
                        if ok:
                            best_c = c
                            best_h = h.copy()
                            rand_ok = True
                            break
                    continue
                nv = n - d
                i = int(rng.randint(nv))
                j = i + d
                if rng.rand() < 0.5:
                    tmax = float(min(h[i], 1.0 - h[j]))
                    if tmax > 1e-10:
                        ok, c = test_pair_move(h, n, i, j, tmax, -1.0, best_c, factors)
                        if ok:
                            best_c = c
                            best_h = h.copy()
                            rand_ok = True
                            break
                else:
                    tmax = float(min(h[j], 1.0 - h[i]))
                    if tmax > 1e-10:
                        ok, c = test_pair_move(h, n, i, j, tmax, 1.0, best_c, factors)
                        if ok:
                            best_c = c
                            best_h = h.copy()
                            rand_ok = True
                            break
            if rand_ok:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 2:
                    break
    out = project_feasible(best_h, n)
    co = c5_val(out, n)
    if co < best_c:
        return out, co
    return best_h, best_c


def smooth_value_free(x, n, alpha):
    dx = 2.0 / float(n)
    v = np.correlate(x, 1.0 - x, mode="full") * dx
    m = float(np.max(v))
    z = alpha * (v - m)
    e = np.exp(np.clip(z, -700.0, 0.0))
    return float(m + np.log(float(np.sum(e))) / float(alpha))


def smooth_grad_free(x, n, alpha, mu, target):
    dx = 2.0 / float(n)
    v = np.correlate(x, 1.0 - x, mode="full") * dx
    m = float(np.max(v))
    z = alpha * (v - m)
    e = np.exp(np.clip(z, -700.0, 0.0))
    s = float(np.sum(e))
    if s <= 0.0:
        w = np.zeros_like(v)
        w[n - 1] = 1.0
    else:
        w = e / s
    grad = float(w[n - 1]) * dx * (1.0 - 2.0 * x)
    mid = n - 1
    for d in range(1, n):
        wp = float(w[mid - d])
        wn = float(w[mid + d])
        sw = wp + wn
        if sw <= 1e-15:
            continue
        nv = n - d
        wdx = dx * sw
        grad[:nv] += dx * wp - wdx * x[d:]
        grad[d:] += dx * wn - wdx * x[:nv]
    if mu != 0.0:
        grad = grad + 2.0 * mu * (float(x.sum()) - target) * np.ones(n)
    return grad


def _obj_free(x, n, alpha, mu, target):
    val = smooth_value_free(x, n, alpha)
    if mu != 0.0:
        ds = float(x.sum()) - target
        val += mu * ds * ds
    return float(val)


def _grad_free(x, n, alpha, mu, target):
    return smooth_grad_free(x, n, alpha, mu, target)


def lbfgs_refine(h, n, deadline):
    h = project_feasible(np.array(h, dtype=float, copy=True), n)
    best_c = c5_val(h, n)
    target = 0.5 * n
    bounds = [(0.0, 1.0)] * n
    configs = ((30.0, 1e-5, 70), (80.0, 1e-4, 90), (300.0, 1e-3, 110),
               (1000.0, 1e-2, 130), (3000.0, 0.1, 160), (8000.0, 1.0, 200))
    for alpha, mu, it in configs:
        if time.time() > deadline:
            break
        try:
            res = minimize(_obj_free, h, jac=_grad_free,
                           args=(n, alpha, mu, target),
                           method="L-BFGS-B", bounds=bounds,
                           options={"maxiter": it, "ftol": 1e-15,
                                    "gtol": 1e-11,
                                    "maxfun": max(160, int(it * 3)),
                                    "maxls": 20})
        except Exception:
            break
        hp = project_feasible(res.x, n)
        cp = c5_val(hp, n)
        if cp < best_c - 1e-15:
            best_c = cp
            h = hp
    return h, best_c


def orbit_improve(h, n, best_c):
    h = project_feasible(np.array(h, dtype=float, copy=True), n)
    best_c = c5_val(h, n)
    rev = h[::-1]
    comp = 1.0 - h
    revcomp = 1.0 - rev
    cands = (h, rev, comp, revcomp, 0.5 * (h + rev), 0.5 * (h + revcomp))
    for c in cands:
        cp = project_feasible(c, n)
        cc = c5_val(cp, n)
        if cc < best_c - 1e-15:
            best_c = cc
            h = cp
    return h, best_c


def trig_start(n, freq, anti):
    m = n // 2
    x = np.linspace(0.0, 1.0, m)
    p = 0.5 + 0.5 * np.sin(np.pi * freq * x)
    h = np.empty(n, dtype=float)
    if anti:
        h[:m] = p
        h[m:] = 1.0 - p[::-1]
    else:
        h[:m] = p
        h[m:] = p[::-1]
    return project_feasible(h, n)


def grad_pair_search(h, n, deadline, rng, alpha=1000.0, top=12):
    h = project_feasible(np.array(h, dtype=float, copy=True), n)
    best_c = c5_val(h, n)
    best_h = h.copy()
    target = 0.5 * n
    factors = (1.0, 0.5, 0.25, 0.1, 0.05, 0.02)
    lr = 0.05
    while time.time() < deadline:
        g = smooth_grad_free(h, n, alpha, 0.0, target)
        if float(np.linalg.norm(g)) < 1e-14:
            break
        hi = np.argpartition(g, n - top)[-top:] if top < n else np.arange(n)
        lo = np.argpartition(g, top)[:top] if top < n else np.arange(n)
        improved = False
        for a in hi:
            for b in lo:
                if time.time() >= deadline:
                    break
                a = int(a)
                b = int(b)
                if a == b:
                    continue
                diff = float(g[a] - g[b])
                if diff <= 1e-12:
                    continue
                tmax = float(min(h[a], 1.0 - h[b]))
                if tmax <= 1e-12:
                    continue
                amt = lr * diff
                if amt > tmax:
                    amt = tmax
                if amt <= 1e-10:
                    continue
                ok, c = test_pair_move(h, n, a, b, amt, -1.0, best_c, factors)
                if ok:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-6:
                break
    out = project_feasible(best_h, n)
    co = c5_val(out, n)
    if co < best_c:
        return out, co
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float, copy=True)
    except Exception:
        h0 = np.full(300, 0.5)
    n = int(len(h0))
    if n < 20:
        h0 = np.full(200, 0.5)
        n = 200

    best_h = project_feasible(h0, n)
    best_c = c5_val(best_h, n)
    best_h, best_c = orbit_improve(best_h, n, best_c)

    left = deadline - time.time()
    if left > 50.0:
        end = min(t0 + min(320.0, 0.40 * left), deadline - 10.0)
        if end > time.time() + 15.0:
            best_h, best_c = targeted_pair_search(best_h, n, end, rng)
            best_h, best_c = orbit_improve(best_h, n, best_c)

    if n % 2 == 0 and time.time() < deadline - 420.0:
        extra_end = min(time.time() + 240.0, deadline - 300.0)
        for freq, anti in ((2, True), (4, False), (6, True), (10, True)):
            if time.time() > extra_end:
                break
            cand = trig_start(n, freq, anti)
            cc = c5_val(cand, n)
            if cc < best_c + 0.005:
                end2 = min(time.time() + 40.0, extra_end)
                c2, cc2 = targeted_pair_search(cand, n, end2, rng, topd=8, tope=10)
                c2, cc2 = orbit_improve(c2, n, cc2)
                if cc2 < best_c - 1e-15:
                    best_c = cc2
                    best_h = c2

    left = deadline - time.time()
    if left > 180.0:
        end = min(time.time() + min(250.0, 0.32 * left), deadline - 120.0)
        if end > time.time() + 20.0:
            best_h, best_c = lbfgs_refine(best_h, n, end)
            best_h, best_c = orbit_improve(best_h, n, best_c)

    left = deadline - time.time()
    if left > 90.0:
        end = min(time.time() + min(280.0, 0.42 * left), deadline - 30.0)
        if end > time.time() + 15.0:
            best_h, best_c = targeted_pair_search(best_h, n, end, rng, topd=16, tope=20)
            best_h, best_c = orbit_improve(best_h, n, best_c)

    left = deadline - time.time()
    if left > 70.0:
        end = min(time.time() + min(55.0, 0.25 * left), deadline - 10.0)
        if end > time.time() + 10.0:
            best_h, best_c = grad_pair_search(best_h, n, end, rng)
            best_h, best_c = orbit_improve(best_h, n, best_c)

    left = deadline - time.time()
    if left > 22.0:
        best_h, best_c = targeted_pair_search(best_h, n, deadline - 2.0, rng, topd=20, tope=26)
        best_h, best_c = orbit_improve(best_h, n, best_c)

    hp = project_feasible(best_h, n)
    cp = c5_val(hp, n)
    if cp < best_c:
        best_h = hp
        best_c = cp
    return best_h, float(best_c), int(n)
