import time

def c5v(h,n):
    return np.correlate(h,1.0-h,mode='full')*(2.0/n)

def c5c(h,n):
    return float(np.max(c5v(h,n)))

def proj(h,n):
    h=np.clip(h.copy(),0.0,1.0); t=0.5*n
    for _ in range(80):
        d=t-h.sum()
        if abs(d)<1e-14: break
        if d>0:
            m=h<1.0; s=(1.0-h)[m].sum()
            if s<1e-15: break
            h[m]+=d*(1.0-h[m])/s
        else:
            m=h>0.0; s=h[m].sum()
            if s<1e-15: break
            h[m]+=d*h[m]/s
    return np.clip(h,0.0,1.0)

def rsmp(h,n,n2):
    if n2==n: return h.copy()
    x=np.linspace(0.0,2.0,n,endpoint=False); y=np.linspace(0.0,2.0,n2,endpoint=False)
    return proj(np.interp(y,x,h),n2)

def gtop(h,lgs,n):
    g=np.zeros(n); h=np.asarray(h,float); o=n-1; k=len(lgs)
    for m in lgs:
        p=int(m)-o
        if p>=0:
            l=n-p; g[p:]+=1.0-h[:l]; g[:l]-=h[p:]
        else:
            q=-p; l=n-q; g[:l]+=1.0-h[q:]; g[q:]-=h[:l]
    return g/k

def gsmooth(h,n,al):
    c=c5v(h,n); z=al*c; z=z-z.max(); w=np.exp(z); w/=w.sum()
    dx=2.0/n; g=np.zeros(n); o=n-1
    for m in range(len(c)):
        wm=w[m]
        if wm<=1e-15: continue
        p=m-o
        if p>=0:
            l=n-p; g[p:]+=wm*dx*(1.0-h[:l]); g[:l]-=wm*dx*h[p:]
        else:
            q=-p; l=n-q; g[:l]+=wm*dx*(1.0-h[q:]); g[q:]-=wm*dx*h[:l]
    return g

def obj(x,n,al,mu):
    c=c5v(x,n); m=c.max(); z=al*(c-m)
    return float(m+np.log(np.sum(np.exp(z)))/al+mu*(x.sum()-0.5*n)**2)

def gr(x,n,al,mu):
    s=x.sum()-0.5*n
    return gsmooth(x,n,al)+2.0*mu*s

def lbfgs(h,n,dl):
    h=proj(h,n); bc=c5c(h,n)
    if time.time() > dl - 2: return h,bc
    bnd=[(0.0,1.0)]*n
    for al,mu in ((50,10),(200,50),(1000,100),(5000,500),(20000,1000),(100000,5000)):
        if time.time() > dl - 2: break
        try:
            r=minimize(obj,h,jac=gr,args=(n,al,mu),method='L-BFGS-B',bounds=bnd,
                       options={'maxiter':120,'ftol':1e-15,'gtol':1e-12,'maxfun':2400,'maxls':50})
            hn=proj(r.x,n); cn=c5c(hn,n)
            if cn<bc: h=hn; bc=cn
        except Exception: pass
    return h,bc

def cswap(h,C,i,j,d):
    n=len(h); o=n-1; om=1.0-h; dd=d*d
    C[i:o+i+1]+=d*om[::-1]
    C[o-i:o+n-i]-=d*h
    C[j:o+j+1]-=d*om[::-1]
    C[o-j:o+n-j]+=d*h
    C[o]-=2.0*dd
    C[o+i-j]+=dd
    C[o+j-i]+=dd
    h[i]+=d; h[j]-=d

def cpair(h,C,n,dl):
    h=h.copy(); C=C.copy(); bc=float(C.max()); ev=0
    kls=(1,2,4,8,16,32,64,128,256)
    ms=80; mc=50; fr=(0.25,0.125,0.0625,0.03125,0.015625)
    for K in kls:
        same=0
        while time.time()<dl:
            cm=float(C.max()); top=np.argsort(C)[-K:]; cj=float(C[top].sum())
            g=gtop(h,top,n)
            inc=np.argsort(g)[:ms]; dec=np.argsort(-g)[:ms]
            if len(inc)==0 or len(dec)==0: break
            if g[dec].max() - g[inc].min() <= 1e-13: break
            sc=g[dec,None]-g[inc]; fl=sc.ravel(); mn=min(mc,fl.size)
            if fl.max()<=1e-13: break
            idx=np.argpartition(-fl,mn-1)[:mn]; idx=idx[np.argsort(-fl[idx])]
            ok=False; I=len(inc)
            for q in idx:
                aidx=q//I; bidx=q%I; ii=inc[bidx]; jj=dec[aidx]
                if ii==jj: continue
                st=min(h[jj],1.0-h[ii])
                if st<1e-13: continue
                for f in fr:
                    d=f*st
                    if d<1e-14: continue
                    cswap(h,C,ii,jj,d); ev+=1
                    m=float(C.max())
                    if m < cm - 1e-13:
                        ok=True; break
                    js=float(C[top].sum())
                    if abs(m-cm)<=1e-13 and js < cj - 1e-10:
                        ok=True; break
                    cswap(h,C,jj,ii,d); ev+=1
                    if ev>80:
                        C=c5v(h,n); ev=0
            if not ok: break
            if ev>80:
                C=c5v(h,n); ev=0
            nm=float(C.max())
            if nm<bc: bc=nm
            if nm < cm - 1e-12: same=0
            else: same+=1
            if same>500: break
        C=c5v(h,n)
    return h,float(C.max())

def bini(h,n):
    a=np.zeros(n,np.int64)
    if n>1:
        idx=np.argpartition(h,n//2-1)
        a[idx[:n//2]]=1
    cnt=np.correlate(a,1-a,'full').astype(np.int64)
    return a,cnt

def sbit(a,C,i,j):
    n=len(a); o=n-1; om=1-a
    C[i:o+i+1]+=om[::-1]
    C[o-i:o+n-i]-=a
    C[j:o+j+1]-=om[::-1]
    C[o-j:o+n-j]+=a
    C[o]-=2
    C[o+i-j]+=1
    C[o+j-i]+=1
    a[i]=1; a[j]=0

def bpair(a,C,n,dl):
    a=a.copy(); C=C.copy(); bc=int(C.max())
    kls=(1,2,4,8,16,32,64,128,256)
    ms=100; mc=200
    for K in kls:
        same=0
        while time.time()<dl:
            cm=int(C.max()); top=np.argsort(C)[-K:]; cj=int(C[top].sum())
            g=gtop(a,top,n)
            z=np.flatnonzero(a==0); one=np.flatnonzero(a==1)
            if len(z)==0 or len(one)==0: break
            zg=g[z]; og=g[one]
            zi=z[np.argsort(zg)[:ms]]; oo=one[np.argsort(-og)[:ms]]
            if len(zi)==0 or len(oo)==0: break
            sc=og[oo][:,None]-zg[zi][None,:]; fl=sc.ravel(); mn=min(mc,fl.size)
            if fl.max()<=1e-12: break
            idx=np.argpartition(-fl,mn-1)[:mn]; idx=idx[np.argsort(-fl[idx])]
            ok=False; I=len(zi)
            for q in idx:
                aid=q//I; bid=q%I; iz=zi[bid]; io=oo[aid]
                if iz==io: continue
                sbit(a,C,iz,io)
                m=int(C.max())
                if m < cm:
                    ok=True; break
                if m==cm and int(C[top].sum()) < cj:
                    ok=True; break
                sbit(a,C,io,iz)
            if not ok: break
            nm=int(C.max())
            if nm<bc: bc=nm
            if nm < cm - 1: same=0
            else: same+=1
            if same>800: break
    return a,float(C.max()*(2.0/n))

def improve(h,n,dl,lb_t=60,pa_t=100,bi_t=40,cp_t=40):
    t=time.time()
    h=proj(h,n); C=c5v(h,n); c=float(C.max())
    if lb_t>5 and t<dl-5:
        t2=min(t+lb_t, dl-10)
        h2,c2=lbfgs(h,n,t2)
        if c2<c: h,c=h2,c2
    if pa_t>5 and time.time()<dl-5:
        t2=min(time.time()+pa_t, dl-5)
        h2,c2=cpair(h,c5v(h,n),n,t2)
        if c2<c: h,c=h2,c2
    if n%2==0 and bi_t>5 and time.time()<dl-5:
        t2=min(time.time()+bi_t, dl-10)
        a,cnt=bini(h,n)
        a,c2=bpair(a,cnt,n,t2)
        hb=a.astype(float); c0=c
        if c2<c0: h,c=hb,c2
        if cp_t>5 and time.time()<dl-5 and c2 < c0 + 0.01:
            t3=min(time.time()+cp_t, dl-5)
            h3,c3=cpair(hb,c5v(hb,n),n,t3)
            if c3<c: h,c=h3,c3
    return h,c

def c5c(h, n):
    return float(np.max(c5v(h, n)))

def proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)

        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            s = (1.0 - h)[m].sum()
            if s < 1e-15:
                break
            h[m] += d * (1.0 - h[m]) / s
        else:
            m = h > 0.0
            s = h[m].sum()
            if s < 1e-15:
                break
            h[m] += d * h[m] / s
    d = t - h.sum()
    if abs(d) > 1e-14:

            h[i] = min(1.0, h[i] + d)

def rsmp(h, n, n2):

        return h.copy()
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    y = np.linspace(0.0, 2.0, n2, endpoint=False)

def gtop(h, lgs, n):

            g[:l] -= h[p:]
        else:

            l = n - q
            g[:l] += 1.0 - h[q:]
            g[q:] -= h[:l]

def cswap(h, C, i, j, d):
    n = len(h)
    o = n - 1

    best_n = n0
    best_c = float(c5v(h0, n0).max())

    tc = min(t0 + 330, dl - 420)
    if tc > t0 + 10:
        h, c = improve(h0, n0, tc, 120, 120, 50, 30)
        if c < best_c:
            best_h, best_n, best_c = h, n, c

    if time.time() < dl - 150:
        t = min(time.time() + 120, dl - 120)
        hS = proj(0.5 * (best_h + best_h[::-1]), best_n)
        h, c = improve(hS, best_n, t, 20, 70, 15, 10)
        if c < best_c:
            best_h, best_n, best_c = h, best_n, c

    seen = set()
    for mm in (best_n // 2, 425, 350, best_n // 3):
        m = int(mm)
        if m % 2:
            m += 1
        if 120 < m < 1200 and m != best_n and m not in seen:
            seen.add(m)
            if time.time() > dl - 80:
                break
            t = min(time.time() + 100, dl - 60)
            hR = rsmp(best_h, best_n, m)
            h, c = improve(hR, m, t, 20, 50, 15, 10)
            if c < best_c:
                best_h, best_n, best_c = h, m, c

    if time.time() < dl - 10:
        t = min(time.time() + 300, dl - 5)
        h, c = improve(best_h, best_n, t, 30, 120, 80 if best_n % 2 == 0 else 0, 60)
        if c < best_c:
            best_h, best_n, best_c = h, best_n, c

    best_h = proj(best_h, best_n)
    best_c = float(c5v(best_h, best_n).max())
    return best_h, best_c, best_n
