```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    # Bisection on lambda: find lam such that sum(clip(h - lam, 0, 1)) = target
    def f(lam):
        return np.clip(h - lam, 0.0, 1.0).sum() - target
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def weighted_grad_descent(h, n, deadline, beta0=50, beta1=50000, n_iters=500000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        # Adaptive beta
        frac = min(it / n_iters, 1.0)
        beta = beta0 * (beta1 / beta0) ** frac
        
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        cur_c = corr.max()
        
        # Compute softmax weights
        shifted = beta * (corr - cur_c)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2*n-1)
            w[n-1] = 1.0
        else:
            w /= w_sum
        
        # Build W array: W[m] for m=0..n-1
        W = np.zeros(n)
        W[0] = w[n-1]
        W[1:] = w[n-2::-1] + w[n:]  # W[m] = w[n-1-m] + w[n-1+m]
        
        # Compute weighted gradient
        grad = W[0] * (1.0 - 2.0 * h)  # m=0 contribution
        for m in range(1, n):
            wm = W[m]
            if wm < 1e-15:
                continue
            grad[m:] += wm * (1.0 - h[:n-m])
            grad[:n-m] -= wm * h[m:]
        grad *= dx
        
        # Line search with several step sizes
        taken = False
        for step in [1.0, 0.5, 0.1, 0.05, 0.01, 0.005, 0.001, 0.0005]:
            h_new = project_feasible(h - step * grad, n)
            if not np.all(h_new >= 0) or not np.all(h_new <= 1):
                continue
            c_new = compute_c5(h_new, n)
            if c_new < cur_c * (1 - 1e-12):
                h = h_new
                cur_c = c_new
                taken = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        
        if not taken:
            # No improvement; try smaller steps or break
            # Try the subgradient direction (only active constraint)
            k_star = np.argmax(corr)
            m_abs = abs(k_star - (n - 1))
            if m_abs == 0:
                sg = (1.0 - 2.0 * h) * dx
            else:
                sg = np.zeros(n)
                sg[m_abs:] += (1.0 - h[:n-m_abs])
                sg[:n-m_abs] -= h[m_abs:]
                sg *= dx
            for step in [0.01, 0.001, 0.0001]:
                h_new = project_feasible(h - step * sg, n)
                c_new = compute_c5(h_new, n)
                if c_new < cur_c * (1 - 1e-12):
                    h = h_new
                    cur_c = c_new
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    break
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
```

**Approach: Direct 1D optimization per active constraint**

At each iteration:
1. Find the active lag k*
2. The gradient of c_{k*} is O(n) to compute
3. Take a step in the -gradient direction
4. Project

This is the subgradient method. It's simpler and each iteration is O(n) for the gradient + O(n^2) for finding the max (or O(n log n) with FFT).

The advantage: no Python loop for the gradient. The disadvantage: non-smooth, might oscillate.

To smooth it out, I can use a "momentum" approach (Nesterov-like):
- Maintain a velocity vector v
- At each step: v = momentum * v + lr * grad; h_new = h - v; project

Or I can use the "avaged" subgradient: maintain a running average of the gradients.

```python
def subgradient_descent(h, n, deadline, lr=0.01, momentum=0.9, n_iters=1000000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        # Find active lag
        corr = np.correlate(h, 1-h, mode='full') * dx
        k_star = np.argmax(corr)
        cur_c = corr[k_star]
        m = abs(k_star - (n - 1))
        
        # Compute subgradient
        if m == 0:
            g = (1.0 - 2.0 * h) * dx
        else:
            g = np.zeros(n)
            g[m:] += (1.0 - h[:n-m])
            g[:n-m] -= h[m:]
            g *= dx
        
        # Momentum update
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        
        # Check improvement
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Adaptive learning rate
        if it % 1000 == 0:
            lr *= (0.999 if c_new >= cur_c else 1.001)
            lr = min(lr, 0.1)
    
    return best_h, best_c
```

```python
def trust_constr_solve(h, n, maxiter=1000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        return g.max()
    
    # Use smooth approximation
    alpha = 10000
    def obj_smooth(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + np.log(np.exp(shifted - sm).sum()) / alpha
    
    def obj_grad(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        softmax = np.exp(shifted - sm)
        softmax /= softmax.sum()
        
        # Weighted gradient
        W = np.zeros(n)
        W[0] = softmax[n-1]
        W[1:] = softmax[n-2::-1] + softmax[n:]
        
        grad = W[0] * (1 - 2*x)
        for m_val in range(1, n):
            wm = W[m_val]
            if wm < 1e-15:
                continue
            grad[m_val:] += wm * (1 - x[:n-m_val])
            grad[:n-m_val] -= wm * x[m_val:]
        return grad * dx
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [LinearConstraint(np.ones(n), n/2, n/2)]
    
    res = minimize(obj_smooth, h, jac=obj_grad, method='trust-constr',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'gtol': 1e-12})
    return project_feasible(res.x, n)
```

```python
import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    def f(lam):
        return np.clip(h - lam, 0.0, 1.0).sum() - target
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def build_W(sw, n):
    W = np.zeros(n)
    W[0] = sw[n - 1]
    W[1:] = sw[n - 2::-1] + sw[n:]
    return W

def weighted_grad(h, n, W):
    dx = 2.0 / n
    S1 = np.zeros(n)
    if n > 1:
        S1[1:] = np.convolve(W[1:], 1.0 - h, mode='full')[:n - 1]
    S2 = np.zeros(n)
    if n > 1:
        S2[:n - 1] = np.correlate(h, W[1:][::-1], mode='full')[n - 1:]
    return (W[0] * (1.0 - 2.0 * h) + S1 - S2) * dx

def smooth_grad_descent(h, n, deadline, alpha0, alpha1, max_iter):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    mom = 0.85
    lr = 0.3
    stale = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        frac = min(it / max(max_iter, 1), 1.0)
        alpha = alpha0 * (alpha1 / alpha0) ** frac
        corr = np.correlate(h, 1 - h, mode='full') * dx
        mc = corr.max()
        shifted = alpha * (corr - mc)
        sm = np.exp(shifted)
        sw = sm / sm.sum()
        W = build_W(sw, n)
        g = weighted_grad(h, n, W)
        v = mom * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            stale = 0
        else:
            stale += 1
        h = h_new
        if stale > 2000:
            lr *= 0.5
            mom = min(mom + 0.01, 0.98)
            stale = 0
    return best_h, best_c

def slsqp_solve(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        s = alpha * (g - m)
        return m + (s.max() + np.log(np.exp(s - s.max()).sum())) / alpha
    cons_fun = lambda x: x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
```
