def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def soft_max_obj(x, n, alpha, lam):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.exp(sh - mx).sum())) / alpha
    if lam > 0:
        val += lam * (x.sum() - n / 2.0) ** 2
    return float(val)

def soft_max_grad(x, n, alpha, lam):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    top_k = min(150, len(w))
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(len(x))
    for i in top_idx:
        if w[i] < 1e-15:

        k = i - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    if lam > 0:
        grad += 2.0 * lam * (x.sum() - n / 2.0)
    return grad
