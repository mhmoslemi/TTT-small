import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

def fast_cd(h, n, max_sweeps=200, deadline=None, verbose=False):
    """Faster coordinate descent with early stopping."""
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    best_c = corr.max()
    no_improve_count = 0

            break
        sweep_best = best_c

def slsqp_smooth(h, n, alpha, maxiter=2000, ftol=1e-16):

    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def active_set_optimize(h, n, top_k=5, maxiter=2000, ftol=1e-16):
    """Optimize using only the top-k active constraints."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    # Find top-k indices
    top_idx = np.argsort(c)[::-1][:top_k]

        return x.sum() - n / 2.0

And dg_k/dh[i] = dx * [(1-h[i+k]) + h[i-k]] (with boundary handling)

So the gradient is a weighted sum of (1-h[i+k]) + h[i-k] terms, where the weights are the softmax probabilities. This can be computed in O(n * K) where K is the number of significant softmax components. For large alpha, K is small (only the top few k values have significant weight).

This is the key insight! For large alpha, the smooth max is close to the true max, and the gradient is dominated by a few k values. We can compute the gradient efficiently.

Let me implement this:

```python
def smooth_max_grad(h, n, alpha):
    """Compute gradient of smooth max of correlations."""
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx  # shape (2n-1,)
    m = g.max()
    logsumexp = m + np.log(np.exp(alpha * (g - m)).sum()) / alpha
    # softmax weights
    weights = np.exp(alpha * (g - m)) / np.exp(alpha * (g - m)).sum()
    
    # Gradient: sum_k weights[k] * dg/dh[i]
    # dg_k/dh[i] = dx * [(1-h[i+k]) + h[i-k]] where k = k_idx - (n-1)
    grad = np.zeros(n)
    m_range = np.arange(-(n-1), n)
    for k_idx in range(2*n-1):
        if weights[k_idx] < 1e-10:
            continue
        k = k_idx - (n - 1)
        # Contribution to grad[i] from this k:
        # dg_k/dh[i] = dx * [(1-h[i+k]) if i+k in [0,n) else 0 + h[i-k] if i-k in [0,n) else 0]
        # For all i: term1[i] = (1-h[i+k]) for i in [0, n-1-k] (when k>=0)
        #             = (1-h[i+k]) for i in [-k, n-1] (when k<0)
        #             = 0 otherwise
        # term2[i] = h[i-k] for i in [k, n-1] (when k>=0)
        #             = h[i-k] for i in [0, n-1+k] (when k<0)
        #             = 0 otherwise
        
        w = weights[k_idx] * dx
        if k >= 0:
            # (1-h[i+k]) valid for i in [0, n-1-k]
            grad[:n-k] += w * (1 - h[k:n])
            # h[i-k] valid for i in [k, n-1]
            grad[k:] += w * h[:n-k]
        else:
            kk = -k
            # (1-h[i+k]) = (1-h[i-kk]) valid for i in [kk, n-1]
            grad[kk:] += w * (1 - h[:n-kk])
            # h[i-k] = h[i+kk] valid for i in [0, n-1-kk]
            grad[:n-kk] += w * h[kk:n]
    
    return grad, logsumexp
```

Wait, this is O(n * K) where K is the number of non-negligible weights. For alpha=100000, K might be just 2-3. So the gradient is O(n) effectively!

This means I can use L-BFGS-B with this gradient, which is much faster than SLSQP for large n.

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    """Compute smooth max and its gradient."""
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    m = g.max()
    expo = alpha * (g - m)
    expo_max = expo.max()
    exps = np.exp(expo - expo_max)
    s = exps.sum()
    val = m + (expo_max + np.log(s)) / alpha
    w = exps / s  # softmax weights
    
    # Gradient
    grad = np.zeros(n)
    nonzero = w > 1e-12
    for k_idx in np.where(nonzero)[0]:
        k = k_idx - (n - 1)
        wv = w[k_idx] * dx
        if k >= 0:
            grad[:n - k] += wv * (1.0 - h[k:])
            grad[k:] += wv * h[:n - k]
        else:
            kk = -k
            grad[kk:] += wv * (1.0 - h[:n - kk])
            grad[:n - kk] += wv * h[kk:]
    return val, grad

def slsqp_analytic(h, n, alpha, maxiter=3000, ftol=1e-16):
    """SLSQP with analytic gradient of smooth max."""
    def obj(x):
        val, _ = smooth_max_and_grad(x, n, alpha)
        return val
    
    def obj_grad(x):
        _, grad = smooth_max_and_grad(x, n, alpha)
        return grad
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h

def fast_cd(h, n, max_sweeps=300, deadline=None):
    """Coordinate descent with early stopping."""
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    best_c = corr.max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        prev_best = best_c
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-10
            lo_v = min(val0, val1) - 1e-10
            
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_mask = ~(pos_B | neg_B)
            zero_max = corr[zero_mask].max() if zero_mask.any() else -np.inf
            
            for _ in range(30):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            
            if abs(x_opt - h[i]) > 1e-14:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
        
        if corr.max() < best_c - 1e-10:
            best_c = corr.max()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                break
    
    return h, corr

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    m = g.max()
    expo = alpha * (g - m)
    expo_max = expo.max()
    exps = np.exp(expo - expo_max)
    s = exps.sum()
    val = m + (expo_max + np.log(s)) / alpha
    w = exps / s
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-12:
            continue
        s_val = k_idx - (n - 1)
        wv = w[k_idx] * dx
        if s_val >= 0:
            # j+s in [0,n): j in [0, n-1-s]
            grad[:n - s_val] += wv * (1.0 - h[s_val:])
            # j-s in [0,n): j in [s, n-1]
            grad[s_val:] += wv * h[:n - s_val]
        else:
            sp = -s_val
            # j-s' in [0,n): j in [s', n-1]
            grad[sp:] += wv * (1.0 - h[:n - sp])
            # j+s' in [0,n): j in [0, n-1-s']
            grad[:n - sp] += wv * h[sp:]
    
    return val, grad

1. Compute gradient of smooth max
2. Take a step: h_new = h - step_size * grad
3. Project: h_proj = project(h_new)
4. Use backtracking or Armijo line search

1. y = h + momentum * (h - h_prev)
2. Compute gradient at y
3. h_new = y - step_size * grad(y)
4. Project: h_proj = project(h_new)
5. Update momentum

This should be much faster per iteration than SLSQP, though it might take more iterations.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    m = g.max()
    expo = alpha * (g - m)
    expo_max = expo.max()
    exps = np.exp(expo - expo_max)
    s = exps.sum()
    val = m + (expo_max + np.log(s)) / alpha
    w = exps / s
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-12:
            continue
        sv = k_idx - (n - 1)
        wv = w[k_idx] * dx
        if sv >= 0:
            grad[:n - sv] += wv * (1.0 - h[sv:])
            grad[sv:] += wv * h[:n - sv]
        else:
            sp = -sv
            grad[sp:] += wv * (1.0 - h[:n - sp])
            grad[:n - sp] += wv * h[sp:]
    return val, grad

def slsqp_h(h, n, alpha, maxiter=3000, ftol=1e-16):
    def obj(x):
        v, _ = smooth_max_grad(x, n, alpha)
        return v
    def objg(x):
        _, g = smooth_max_grad(x, n, alpha)
        return g
    def cf(x):
        return x.sum() - n / 2.0
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h

def fast_cd(h, n, max_sweeps=300, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    best_c = corr.max()
    no_improve = 0
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-10
            lo_v = min(val0, val1) - 1e-10
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_mask = ~(pos_B | neg_B)
            zero_max = corr[zero_mask].max() if zero_mask.any() else -np.inf
            for _ in range(30):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-14:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
        if corr.max() < best_c - 1e-10:
            best_c = corr.max()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                break
    return h, corr

def projected_grad(h, n, alpha, steps=500, deadline=None):
    """Nesterov-accelerated projected gradient."""
    h = h.astype(float).copy()
    h_prev = h.copy()
    momentum = 0.5
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(steps):
        if deadline and time.time() > deadline:
            break
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0, 1)
        _, grad = smooth_max_grad(y, n, alpha)
        step = 1.0 / (np.linalg.norm(grad) + 1e-8)
        step = min(step, 1.0)
        h_new = y - step * grad
        h_new = project_feasible(h_new, n)
        h_prev = h.copy()
        h = h_new
        momentum = (1 + t + 1) / (2 + t + 1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
