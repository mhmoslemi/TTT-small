    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = g - m
    # power mean approximation of max
    pp = (np.exp(p * shifted).sum()) / len(g)
    return m + np.log(pp) / p

def slsqp_power(h, n, p, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = g - m
        pp = np.exp(p * shifted).sum() / len(g)
        return m + np.log(pp) / p
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, n, max_sweeps=500, deadline=None, bisection_iters=50):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(bisection_iters):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def make_structured_pattern(n, mode, rng):
    h = np.zeros(n)
    if mode == 0:
        step = max(1, n // 200)
        for i in range(0, n, step * 2):
            h[i:i+step] = 1.0
    elif mode == 1:
        p = int(rng.randint(n // 4, 3 * n // 4))
        idx = (np.arange(n) * p) % n
        h[idx[:n // 2]] = 1.0
        # fix duplicates
        h = to_binary(h, n)
    elif mode == 2:
        period = int(rng.randint(max(4, n // 50), max(5, n // 20)))
        half = period // 2
        for i in range(n):
            h[i] = 1.0 if (i % period) < half else 0.0
    else:
        h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Binary greedy
    p1 = t0 + 0.05 * total
    hb, cb = greedy_binary(h0, n, p1, max_sweeps=200)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Coordinate descent
    p2 = t0 + 0.18 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=350, deadline=p2, bisection_iters=55)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 3: SLSQP soft-max escalation
    p3 = t0 + 0.45 * total
    alphas = [50, 100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000, 100000000, 500000000]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p3 - 5:
            break
        h = slsqp_softmax(h, n, alpha, 2500, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Coord descent from SLSQP result
    p4 = t0 + 0.56 * total
    hc2, _ = coord_descent(best_h, n, max_sweeps=250, deadline=p4, bisection_iters=55)
    hc2 = project_feasible(hc2, n)
    cc2 = compute_c5(hc2, n)
    if cc2 < best_c:
        best_c, best_h = cc2, hc2.copy()

    # Phase 5: Power-mean SLSQP (different landscape)
    p5 = t0 + 0.68 * total
    h = best_h.copy()
    for p in [5, 10, 20, 50, 100, 200, 500, 1000]:
        if time.time() > p5 - 4:
            break
        h = slsqp_power(h, n, p, 2000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 6: Final coord + SLSQP
    p6 = t0 + 0.78 * total
    hc3, _ = coord_descent(best_h, n, max_sweeps=200, deadline=p6, bisection_iters=55)
    hc3 = project_feasible(hc3, n)
    cc3 = compute_c5(hc3, n)
    if cc3 < best_c:
        best_c, best_h = cc3, hc3.copy()
    for alpha in [100000000, 500000000, 2000000000]:
        if time.time() > p6:
            break
        h = slsqp_softmax(best_h, n, alpha, 3000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 7: Multi-resolution exploration
    p7 = t0 + 0.89 * total
    for nh in [300, 500, 600, 800]:
        if time.time() > p7 - 8:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, _ = coord_descent(hh, nh, max_sweeps=50, deadline=p7 - 6, bisection_iters=40)
        hh = project_feasible(hh, nh)
        for alpha in [100000, 1000000, 10000000, 100000000]:
            if time.time() > p7 - 4:
                break
            hh = slsqp_softmax(hh, nh, alpha, 1500, 1e-16)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh

    # Phase 8: Diverse restarts from structured patterns + noise
    if time.time() < dl - 5:
        starts = []
        # noise perturbations
        for scale in [0.001, 0.003, 0.005, 0.01, 0.02]:
            starts.append(project_feasible(best_h + scale * rng.randn(n), n))
        # structured patterns
        for mode in range(4):
            starts.append(make_structured_pattern(n, mode, rng))
        # interpolate between best and a random pattern
        rand_pat = project_feasible(rng.uniform(0, 1, n), n)
        for w in [0.3, 0.5, 0.7]:
            starts.append(project_feasible(w * best_h + (1 - w) * rand_pat, n))

        for start_h in starts:
            if time.time() > dl - 2:
                break
            for alpha in [10000000, 100000000]:
                if time.time() > dl - 1:
                    break
                sh = slsqp_softmax(start_h, n, alpha, 1500, 1e-16)
                c = compute_c5(sh, n)
                if c < best_c:
                    best_c, best_h = c, sh.copy()
                    start_h = best_h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
