from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def active_gradient(h, n, K):
    """Compute subgradient from top K active lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    # Find top K lags
    idx = np.argsort(corr)[::-1][:K]
    grad = np.zeros(n)
    for k_idx in idx:
        k = int(k_idx) - (n - 1)
        # d(corr_k)/d(h[j]) = (1-h[j+k])[j+k valid] - h[j-k][j-k valid]
        if k >= 0:
            grad[:n-k] += (1.0 - h[k:])
            grad[k:] -= h[:n-k]
        else:
            grad[-k:] += (1.0 - h[:n+k])
            grad[:n+k] -= h[-k:]
    grad /= K
    grad *= dx
    return grad, corr.max() * dx

def focal_gradient_descent(h, n, deadline, K=20, n_steps=200000):
    """Focal active-set projected gradient descent with backtracking line search."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.1

            break
        grad, c_cur = active_gradient(h, n, K)
        # Backtracking line search
        alpha = lr
        for _ in range(20):
            h_new = project_feasible(h - alpha * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_cur - 1e-14:

            lr = max(alpha, 1e-6)
            if alpha < 1e-8:

                continue
        h = h_new
        lr = alpha
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def greedy_swap(h_init, n, deadline):
    m_range = np.arange(-(n-1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_c = float(corr.max()) * dx
    best_h = h.copy()
    best_c = cur_c
    ones = [int(x) for x in np.where(h == 1)[0]]
    zeros = [int(x) for x in np.where(h == 0)[0]]
    while time.time() < deadline:
        made = False
        for idx_i, i in enumerate(zeros[:]):
            if time.time() > deadline:
                return best_h, best_c
            bj = -1
            bnc = cur_c
            for j in ones:
                d = np.zeros(2*n - 1)
                im = i + m_range
                m = (im >= 0) & (im < n)
                d[m] += 1.0 - h[im[m]]
                imn = i - m_range
                m = (imn >= 0) & (imn < n)
                d[m] -= h[imn[m]]
                jm = j + m_range
                m = (jm >= 0) & (jm < n)
                d[m] -= 1.0 - h[jm[m]]
                jmn = j - m_range
                m = (jmn >= 0) & (jmn < n)
                d[m] += h[jmn[m]]
                d[n-1] -= 2.0
                d[n-1+j-i] += 1.0
                d[n-1+i-j] += 1.0
                nc = float((corr + d).max()) * dx
                if nc < bnc - 1e-14:
                    bnc = nc
                    bj = j
            if bj >= 0:
                h[i] = 1.0; h[bj] = 0.0
                zeros[idx_i] = bj
                ones.remove(bj)
                ones.append(i)
                corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
                cur_c = float(corr.max()) * dx
                made = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not made:
            break
    return best_h, best_c
