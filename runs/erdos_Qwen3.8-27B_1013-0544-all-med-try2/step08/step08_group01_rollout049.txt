        idx = active_indices

        idx = np.arange(len(c))

    mx = sh.max()

    w /= w.sum()

        k = m_idx - (n - 1)
        wm = w[i] * dx
        if k >= 0:

                    g += (1.0 - x[j - k])

                grad[j] += wm * g

            for j in range(n):

                if j <= n - 1 - kk:
                    g += (1.0 - x[j + kk])
                if j >= kk:
                    g -= x[j - kk]
                grad[j] += wm * g

For j in [0, n-1-k]: second term is -h[j+k]
Outside these ranges, the contribution is 0.

So: grad_k[j] = (1-h[j-k]) * 1[j >= k] - h[j+k] * 1[j <= n-1-k]

We can compute this vectorized:
- First part: for j in [k, n-1], value is (1-h[j-k]), which is (1-h[0:n-k]) placed at positions [k:n]
- Second part: for j in [0, n-1-k], value is -h[j+k], which is -h[k:n] placed at positions [0:n-k]

grad_k[k:] += (1 - h[:n-k])  # positions k to n-1
grad_k[:n-k] += -h[k:]      # positions 0 to n-k-1

grad_kk[j] = (1-h[j+kk]) * [j <= n-1-kk] - h[j-kk] * [j >= kk]

grad_kk = np.zeros(n)
grad_kk[:n-kk] += (1 - h[kk:])   # positions 0 to n-kk-1

def corr_gradient(x, n, weights):
    """Compute gradient of weighted sum of correlations.
    weights: array of length 2n-1, one weight per correlation value.
    Returns gradient of sum_m weights[m] * corr[m] with respect to x.
    """
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m] * dx
        if w < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - x[:n-k])
            grad[:n-k] += -w * x[k:]
        else:
            kk = -k
            grad[:n-kk] += w * (1.0 - x[kk:])
            grad[kk:] += -w * x[:n-kk]
    return grad

def subgradient_step(h, n, step_size, top_k=5):
    """Take a projected subgradient step targeting the top-k maximizing shifts."""
    c = compute_corr(h, n)
    # Get top-k shifts
    top_idx = np.argsort(c)[-top_k:]

    weights = np.zeros(2*n-1)

    # Projected gradient step

    if active is not None:

    else:

    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline, alpha_list=None, mu_list=None, active=None, maxiter=500):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alpha_list is None:
        alpha_list = [200, 1000, 5000, 20000, 50000, 200000, 500000, 2000000, 10000000]
    if mu_list is None:
        mu_list = [5.0, 20.0, 100.0, 500.0, 2000.0, 5000.0, 20000.0, 50000.0, 200000.0]
    for i, (alpha, mu) in enumerate(zip(alpha_list, mu_list)):
        if time.time() > deadline - 3:
            break
        tr = deadline - time.time() - 3
        if tr < 5:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu, active), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': maxiter * 40,
                                   'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def focused_optimize(h, n, deadline, top_k=20, alpha=500000, mu=10000):

def pair_polish(h, n, deadline, radius=80):
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    best_c = compute_c5(h, n)
    for d in deltas:
        any_improved = True
        while any_improved and time.time() < deadline:
            any_improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            any_improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            any_improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c
