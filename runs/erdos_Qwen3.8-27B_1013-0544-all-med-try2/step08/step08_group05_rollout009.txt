def coordinate_sweep(h, f, n, slack, coarse_g, refine, deadline, seed):
    start_c = float(np.max(f))
    cur_c = start_c
    order = np.arange(n)
    if 0 <= slack < n:
        order = np.delete(order, slack)
    rng = np.random.RandomState(seed)
    rng.shuffle(order)
    p0 = n - 1
    accepted = 0
    for u in order:
        if time.time() > deadline:
            break
        v = slack
        if u == v:
            continue
        hu = float(h[u]); hv = float(h[v])
        l = max(-hu, hv - 1.0)
        r = min(1.0 - hu, hv)
        if r - l < 1e-15:
            continue
        a = a_for_pair(h, n, u, v)
        p1 = p0 + v - u; p2 = p0 + u - v
        du, bv = search_delta(f, a, p1, p2, p0, l, r, coarse_g, refine)
        if abs(du) > 1e-15 and bv < cur_c - 1e-11:
            dd = du*du
            nf = f + a*du
            nf[p1] += dd
            nf[p2] += dd
            nf[p0] -= 2.0*dd
            nc = float(np.max(nf))
            if nc < cur_c - 1e-12:
                h[u] = hu + du
                h[v] = hv - du
                if h[u] < 0.0 and h[u] > -1e-12: h[u] = 0.0
                elif h[u] > 1.0 and h[u] < 1.0 + 1e-12: h[u] = 1.0
                if h[v] < 0.0 and h[v] > -1e-12: h[v] = 0.0
                elif h[v] > 1.0 and h[v] < 1.0 + 1e-12: h[v] = 1.0
                f = nf
                cur_c = nc
                accepted += 1
                if accepted % 12 == 0:
                    f = compute_f(h, n)
                    cur_c = float(np.max(f))
    f = compute_f(h, n)
    cur_c = float(np.max(f))
    return f, cur_c, cur_c < start_c - 1e-13

def pick_slack(h, rng, avoid=-1):
    n = h.size
    frac = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if frac.size:
       score = np.minimum(h[frac], 1.0 - h[frac])
       ord = np.argsort(-score)
       for pos in ord:
          idx = int(frac[pos])
          if idx != avoid:
              return idx
       return int(frac[ord[0]])
    cand = np.where(h < 0.5)[0]
    if cand.size == 0:
       cand = np.where(h >= 0.5)[0]
    if cand.size == 0:
       return 0
    # try not avoid
    choices = [int(x) for x in cand if int(x) != avoid]
    if not choices:
       choices = [int(cand[0])]
    return int(choices[int(rng.randint(len(choices)))])

import numpy as np
import time


def compute_f(h, n):
    omh = 1.0 - h
    k = 2 * n - 1
    f = np.zeros(k, dtype=float)
    f[n - 1] = float(np.dot(h, omh))
    for m in range(1, n):
        f[n - 1 + m] = float(np.dot(h[:n - m], omh[m:]))
        f[n - 1 - m] = float(np.dot(h[m:], omh[:n - m]))
    return f


def eval_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def project_sum(h, target):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    hq = np.clip(h - lam, 0.0, 1.0)
    diff = target - float(hq.sum())
    if diff > 1e-15:
        idx = int(np.argmax(1.0 - hq))
        hq[idx] = min(1.0, hq[idx] + diff)
    elif diff < -1e-15:
        idx = int(np.argmax(hq))
        hq[idx] = max(0.0, hq[idx] + diff)
    return hq


def a_for_pair(h, n, u, v):
    k = 2 * n - 1
    a = np.zeros(k, dtype=float)
    omh = 1.0 - h
    hr = h[::-1]
    a[n - 1 - u:2 * n - 1 - u] += omh
    a[u:u + n] -= hr
    a[n - 1 - v:2 * n - 1 - v] -= omh
    a[v:v + n] += hr
    return a


def max_quad_values(f, a, p1, p2, p0, d):
    t = f[:, None] + a[:, None] * d[None, :]
    dd = d * d
    t[p1, :] += dd
    t[p2, :] += dd
    t[p0, :] -= 2.0 * dd
    return t.max(axis=0)


def search_delta(f, a, p1, p2, p0, l, r, coarse_g, refine):
    if r - l <= 1e-14:
        return 0.0, float(np.max(f))
    g = max(8, int(coarse_g))
    d = np.linspace(l, r, g)
    if l < 0.0 < r and abs(d[0]) > 1e-13 and abs(d[-1]) > 1e-13:
        d = np.append(d, 0.0)
    vals = max_quad_values(f, a, p1, p2, p0, d)
    idx = int(np.argmin(vals))
    bd = float(d[idx])
    bv = float(vals[idx])
    step = (r - l) / float(g)
    for _ in range(int(refine)):
        lo = max(l, bd - step)
        hi = min(r, bd + step)
        if hi - lo <= 1e-15:
            break
        d2 = np.linspace(lo, hi, g)
        vals2 = max_quad_values(f, a, p1, p2, p0, d2)
        i2 = int(np.argmin(vals2))
        nv = float(vals2[i2])
        if nv >= bv - 1e-14:
            break
        bd = float(d2[i2])
        bv = nv
        step /= float(g)
    return bd, bv


def coordinate_sweep(h, f, n, slack, coarse_g, refine, deadline, seed):
    start_c = float(np.max(f))
    cur_c = start_c
    order = np.arange(n)
    if 0 <= slack < n:
        order = np.delete(order, slack)
    rng = np.random.RandomState(seed)
    rng.shuffle(order)
    p0 = n - 1
    accepted = 0
    for u in order:
        if time.time() > deadline:
            break
        v = slack
        if u == v:
            continue
        hu = float(h[u])
        hv = float(h[v])
        l = max(-hu, hv - 1.0)
        r = min(1.0 - hu, hv)
        if r - l < 1e-15:
            continue
        a = a_for_pair(h, n, u, v)
        p1 = p0 + v - u
        p2 = p0 + u - v
        du, bv = search_delta(f, a, p1, p2, p0, l, r, coarse_g, refine)
        if abs(du) > 1e-15 and bv < cur_c - 1e-11:
            dd = du * du
            nf = f + a * du
            nf[p1] += dd
            nf[p2] += dd
            nf[p0] -= 2.0 * dd
            nc = float(np.max(nf))
            if nc < cur_c - 1e-12:
                h[u] = hu + du
                h[v] = hv - du
                if -1e-12 < h[u] < 0.0:
                    h[u] = 0.0
                elif 1.0 < h[u] < 1.0 + 1e-12:
                    h[u] = 1.0
                if -1e-12 < h[v] < 0.0:

                    cur_c = float(np.max(f))
    f = compute_f(h, n)
    cur_c = float(np.max(f))
    return f, cur_c, cur_c < start_c - 1e-13


def pick_slack(h, rng, avoid=-1):
    n = h.size
    frac = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if frac.size:
        score = np.minimum(h[frac], 1.0 - h[frac])
        ordi = np.argsort(-score)
        for pos in ordi:
            idx = int(frac[pos])
            if idx != avoid:
                return idx
        return int(frac[ordi[0]])
    cand = np.where(h < 0.5)[0]
    if cand.size == 0:
        cand = np.where(h >= 0.5)[0]
    if cand.size == 0:
        return 0
    choices = []
    for x in cand:
        xi = int(x)
        if xi != avoid:
            choices.append(xi)
    if not choices:
        choices = [int(cand[0])]
    return int(choices[int(rng.randint(len(choices)))])


def zero_sum_perturb(h, n, sigma, rng):
    if n < 2:
        return h.copy()
    noise = rng.normal(0.0, sigma, n)
    noise -= noise.mean()
    m = float(np.max(np.abs(noise)))
    if m > 0.25:
        noise *= 0.25 / m
    return project_sum(h + noise, n / 2.0)


def pair_polish(h, f, n, deadline, seed, coarse_g, refine, max_no):
    start_c = float(np.max(f))
    cur_c = start_c
    rng = np.random.RandomState(seed)
    p0 = n - 1

def interp_to_n(parent, n_parent, n):
    if n_parent == n:
        return parent.copy()
    xo = np.linspace(0.0, 2.0, n_parent, endpoint=True)
    xn = np.linspace(0.0, 2.0, n, endpoint=True)
    return np.interp(xn, xo, parent)


def make_binary(h):
    n = h.size
    target = n // 2
    idx = np.argsort(-h)
    out = np.zeros(n, dtype=float)
    out[idx[:target]] = 1.0
    return out


def sine_initial(n, freq, amp, phase):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + amp * np.sin(np.pi * (freq * x + phase))
    return project_sum(np.clip(h, 0.0, 1.0), n / 2.0)


def quantize_exact(h, n):
    scale = 1 << 20
    target_num = (int(n) // 2) * scale
    q = np.rint(h * float(scale)).astype(np.int64)
    q = np.clip(q, 0, scale)
    diff = int(target_num - np.sum(q, dtype=np.int64))
    if diff > 0:
        order = np.argsort(-(scale - q))
        d = diff
        for idx in order:
            if d <= 0:
                break
            step = int(scale - q[idx])
            if step > d:
                step = d
            if step > 0:
                q[idx] += step
                d -= step
    elif diff < 0:
        d = -diff
        order = np.argsort(-q)
        for idx in order:
            if d <= 0:
                break
            step = int(q[idx])
            if step > d:
                step = d
            if step > 0:
                q[idx] -= step
                d -= step
    return q / float(scale)


def start_key(item):
