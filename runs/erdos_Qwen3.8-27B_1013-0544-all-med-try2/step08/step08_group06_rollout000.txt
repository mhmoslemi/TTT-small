```python
import numpy as np
import time
from scipy.optimize import minimize

OPT = {"n": 0, "mode": 0, "alpha": 1.0, "weights": None, "eq_arr": None, "deadline": 1e30}
CACHE = {"x": None, "f": 0.0, "g": None}


def cell_centers(n):
    return (np.arange(n) + 0.5) * (2.0 / float(n))


def eval_c5(h, n):
    hh = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    c = np.correlate(hh, 1.0 - hh, mode="full")
    return float(np.max(c)) * (2.0 / float(n))


def project_feasible(y, n):
    y = np.asarray(y, dtype=float)
    y = np.clip(y, 0.0, 1.0)
    target = 0.5 * float(n)
    if abs(float(y.sum()) - target) < 1e-13:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y + mid, 0.0, 1.0).sum())
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(4):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += diff / float(free.sum())
        else:
            h += diff / float(n)
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(h, n, nt):
    if nt == n:
        return np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    xo = cell_centers(n)
    xn = cell_centers(nt)
    return project_feasible(np.interp(xn, xo, np.asarray(h, dtype=float)), nt)


def weighted_grad(h, w):
    n = len(h)
    c = n - 1
    grad = np.zeros(n, dtype=float)
    w0 = w[c]
    if w0 != 0.0:
        grad += w0 * (1.0 - 2.0 * h)
    om = 1.0 - h
    nz = np.flatnonzero(w)
    if nz.size < 2 * (n - 1):
        for idx in nz:
            if idx == c:
                continue
            val = float(w[idx])
            if idx > c:
                s = idx - c
                m = n - s
                grad[:m] += val * om[s:]
                grad[s:] -= val * h[:m]
            else:
                s = c - idx
                m = n - s
                grad[s:] += val * om[:m]
                grad[:m] -= val * h[s:]
    else:
        for s in range(1, n):
            m = n - s
            a = w[n - 1 + s]
            if a != 0.0:
                ga = float(a)
                grad[:m] += ga * om[s:]
                grad[s:] -= ga * h[:m]
            b = w[n - 1 - s]
            if b != 0.0:
                gb = float(b)
                grad[s:] += gb * om[:m]
                grad[:m] -= gb * h[s:]
    return grad


def weighted_sum_fg(x, n, w):
    h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    g = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / float(n)
    f = dx * float(np.dot(w, g))
    return f, dx * weighted_grad(h, w)


def lse_fg(x, n, alpha):
    h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    g = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / float(n)
    mx = float(np.max(g))
    lam = alpha * dx
    if lam < 1e-8:
        ww = np.full(g.shape, 1.0 / g.shape[0])
        return dx * float(np.dot(ww, g)), dx * weighted_grad(h, ww)
    sh = lam * (g - mx)
    e = np.exp(sh)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        ww = np.zeros(g.shape)
        ww[int(np.argmax(g))] = 1.0
        extra = 0.0
    else:
        ww = e / s
        extra = float(np.log(s)) / alpha
    f = dx * mx + extra
    return f, dx * weighted_grad(h, ww)


def compute_fg(x):
    n = OPT["n"]
    if len(x) != n:
        return 1e6, np.zeros(n)
    if OPT["mode"] == 1:
        f, g = weighted_sum_fg(x, n, OPT["weights"])
    else:
        f, g = lse_fg(x, n, OPT["alpha"])
    if not np.isfinite(f):
        f = 1e6
    if not np.all(np.isfinite(g)):
        g = np.zeros(n)
    return f, g


def _obj_func(x):
    if time.time() > OPT["deadline"]:
        raise RuntimeError("time")
    f, g = compute_fg(x)
    CACHE["x"] = np.array(x, dtype=float, copy=True)
    CACHE["f"] = f
    CACHE["g"] = g
    return f


def _jac_func(x):
    if time.time() > OPT["deadline"]:
        raise RuntimeError("time")
    cx = CACHE["x"]
    if cx is None or len(cx) != len(x) or not np.array_equal(cx, x):
        _, g = compute_fg(x)
        CACHE["x"] = np.array(x, dtype=float, copy=True)
        CACHE["g"] = g
    return CACHE["g"]


def eq_fun(x):
    return float(x.sum() - 0.5 * OPT["n"]) / float(OPT["n"])


def eq_jac(x):
    return OPT["eq_arr"].copy()


def slsqp_step(h, n, mode, weights, alpha, maxiter, deadline, best_h, best_c):
    h = project_feasible(h, n)
    c0 = eval_c5(h, n)
    if c0 < best_c:
        best_c = c0
        best_h = h.copy()
    if time.time() > deadline - 8:
        return h, best_h, best_c

    OPT["n"] = int(n)
    OPT["mode"] = int(mode)
    OPT["alpha"] = float(alpha)
    OPT["weights"] = None if mode == 0 else np.asarray(weights, dtype=float)
    OPT["eq_arr"] = np.ones(n, dtype=float) / float(n)
    OPT["deadline"] = float(deadline)
    CACHE["x"] = None

    tleft = deadline - time.time()
    if tleft < 25:
        maxiter = min(maxiter, 10)
    elif tleft < 90:
        maxiter = min(maxiter, 40)
    elif tleft < 250:
        maxiter = min(maxiter, 90)

    cons = [{"type": "eq", "fun": eq_fun, "jac": eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            _obj_func,
            h,
            jac=_jac_func,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-12, "gtol": 1e-9, "disp": False},
        )
        h2 = project_feasible(res.x, n)
    except Exception:
        h2 = h

    c2 = eval_c5(h2, n)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()
    if c2 - best_c <= 1e-9:
        h_out = h2
    else:
        h_out = best_h
    return h_out, best_h, best_c


def slsqp_lse_sequence(h0, n, deadline, alphas):
    n = int(n)
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = eval_c5(h, n)
    scale = 400.0 / float(n)
    for base_a in alphas:
        if time.time() > deadline - 10:
            break
        alpha = float(base_a) * scale
        if alpha < 10.0:
            alpha = 10.0
        req = 140 if alpha < (1.0e4 * scale) else 70
        req = int(min(req, max(20, 55000 // max(1, n))))
        h, best_h, best_c = slsqp_step(h, n, 0, None, alpha, req, deadline, best_h, best_c)
    return best_h, best_c


def active_refine(h0, n, deadline):
    n = int(n)
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = eval_c5(h, n)
    tols = [2.0, 0.5, 0.1, 0.03, 0.01, 0.004, 0.0015]
    for tol in tols:
        if time.time() > deadline - 10:
            break
        g = np.correlate(h, 1.0 - h, mode="full")
        mx = float(np.max(g))
        active = np.where(g >= mx - tol)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(g))], dtype=int)
        if active.size > 80:
            order = np.argsort(g[active])[-80:]
            active = active[order]
        w = np.zeros(g.shape, dtype=float)
        vals = np.maximum(0.0, g[active] - (mx - tol))
        sv = float(vals.sum())
        if sv <= 1e-15:
            w[active] = 1.0 / float(active.size)
        else:
            w[active] = vals / sv
        maxiter = int(min(50, max(16, 25000 // max(1, n))))
        h, best_h, best_c = slsqp_step(h, n, 1, w, 0.0, maxiter, deadline, best_h, best_c)
    return best_h, best_c


def optimize_candidate(h0, n, deadline, base_alphas, do_active=False):
    hb, cb = slsqp_lse_sequence(h0, n, deadline, base_alphas)
    if do_active and time.time() < deadline - 18:
        hb, cb = active_refine(hb, n, deadline)
    return hb, cb


def make_fourier_start(n, rng, kind):
    n = int(n)
    x = cell_centers(n)
    if kind == 0:
        h = 0.5 + 0.45 * np.cos(np.pi * x) + 0.10 * np.cos(2.0 * np.pi * x)
    elif kind == 1:
        h = 0.5 + 0.40 * np.sin(np.pi * x) + 0.14 * np.sin(2.0 * np.pi * x)
    else:
        h = np.full(n, 0.5, dtype=float)
        for j in range(1, 12):
            amp = 0.32 / (j + 1.0)
            ph = float(rng.uniform(-np.pi, np.pi))
            h += amp * np.cos(j * np.pi * x + ph)
    return project_feasible(np.clip(h, 0.0, 1.0), n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        raw = float(budget_s)
    except Exception:
        raw = 1100.0
    if raw <= 0.0:
        raw = 60.0
    span = max(10.0, raw - 4.0)
    deadline = t0 + span
    rng = np.random.RandomState(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 < 50 or n0 > 950:
            nh = 400
            h0 = project_feasible(np.interp(cell_centers(nh), cell_centers(n0), h0), nh)
            n0 = nh
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    h = project_feasible(h0, n0)
    best_h = h.copy()
    best_c = eval_c5(h, n0)
    best_n = n0

    parent_dl = t0 + 0.45 * span
    parent_alphas = [1000, 10000, 100000, 1000000, 10000000]
    if time.time() < parent_dl - 8:
        h1, c1 = slsqp_lse_sequence(h, best_n, parent_dl, parent_alphas)
        if c1 < best_c:
            best_c = c1
            best_n = best_n
            best_h = h1
        if time.time() < parent_dl - 8:
            h1, c1 = active_refine(best_h, best_n, parent_dl)
            if c1 < best_c:
                best_c = c1
                best_h = h1

    alt_dl = t0 + 0.72 * span
    if time.time() < alt_dl - 8:
        if best_n < 500:
            n_alt = 500
        elif best_n < 600:
            n_alt = 600
        else:
            n_alt = 300
        if n_alt >= 600:
            alt_alphas = [10000, 100000, 1000000]
        elif n_alt >= 450:
            alt_alphas = [1000, 10000, 100000, 1000000]
        else:
            alt_alphas = [10000, 100000, 1000000]
        ha = project_feasible(interp_h(best_h, best_n, n_alt), n_alt)
        h1, c1 = optimize_candidate(ha, n_alt, alt_dl, alt_alphas, True)
        if c1 < best_c:
            best_c = c1
            best_n = n_alt
            best_h = h1

    alt2_dl = t0 + 0.86 * span
    if time.time() < alt2_dl - 8:
        n_alt2 = 300 if best_n != 300 else 250
        ha = project_feasible(interp_h(best_h, best_n, n_alt2), n_alt2)
        h1, c1 = optimize_candidate(ha, n_alt2, alt2_dl, [10000, 100000, 1000000], False)
        if c1 < best_c:
            best_c = c1
            best_n = n_alt2
            best_h = h1

    four_dl = t0 + 0.94 * span
    if time.time() < four_dl - 18:
        h4 = make_fourier_start(best_n, rng, 0)
        h1, c1 = optimize_candidate(h4, best_n, four_dl, [10000, 100000, 1000000], False)
        if c1 < best_c:
            best_c = c1
            best_h = h1
        if time.time() < deadline - 12:
            h4 = make_fourier_start(best_n, rng, 2)
            h1, c1 = optimize_candidate(h4, best_n, deadline, [100000, 1000000], False)
            if c1 < best_c:
                best_c = c1
                best_h = h1

    if time.time() < deadline - 6:
        h1, c1 = active_refine(best_h, best_n, deadline - 2)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    if time.time() < deadline - 3:
        std = 0.008
        hp = project_feasible(best_h + std * rng.normal(size=best_n), best_n)
        h1, c1 = slsqp_lse_sequence(hp, best_n, deadline - 1, [1000000, 10000000])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    best_h = project_feasible(best_h, best_n)
    best_c = eval_c5(best_h, best_n)
    return best_h, best_c, best_n
```
