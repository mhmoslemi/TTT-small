    best_h = h.copy()

        j = rng.randint(n)

        d = rng.uniform(0.001, 0.01)
        if rng.rand() < 0.5:
            di, dj = d, -d
        else:
            di, dj = -d, d

            # accept
            pass
        else:
            # reject
            h[i] -= di
            h[j] -= dj

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_weights(c, alpha):
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    w = smooth_max_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    s = x.sum() - n / 2.0
    return float(val + mu * s * s)

def lbfgs_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    s = x.sum() - n / 2.0
    g = g + 2.0 * mu * s
    return g

def slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha)

def slsqp_eq(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_eq_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, max_iters=600):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 0.5), (500, 2.0), (2000, 10.0),
               (10000, 50.0), (50000, 200.0),
               (200000, 1000.0), (1000000, 5000.0),
               (5000000, 20000.0), (20000000, 100000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(lbfgs_obj, h, jac=lbfgs_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-20,
                                   'gtol': 1e-18, 'maxfun': max_iters * 80})
            h_new = project_feasible(res.x, n)

            if c_new < best_c:

        except Exception:

    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=5000000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def constraint_focused_local(h, n, deadline, n_top=8):
    best_c = compute_c5(h, n)
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
    for d in steps:
        while time.time() < deadline:
            c_vec = compute_c5_vec(h, n)
            top_m = np.argsort(c_vec)[::-1][:n_top]
            any_imp = False
            for m in top_m:
                if time.time() > deadline:
                    return h, best_c
                k = m - (n - 1)
                if k >= 0:
                    nv = n - k
                    contrib = h[:nv] * (1.0 - h[k:])
                    top_pos = np.argsort(contrib)[::-1][:30]
                    for pos in top_pos:
                        i, j = pos, pos + k
                        if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                            h[i] -= d
                            h[j] += d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                any_imp = True
                            else:
                                h[i] += d
                                h[j] -= d
                else:
                    kk = -k
                    nv = n - kk
                    contrib = h[kk:] * (1.0 - h[:nv])
                    top_pos = np.argsort(contrib)[::-1][:30]
                    for pos in top_pos:
                        j, i = pos, pos + kk
                        if h[j] - d >= 0.0 and h[i] + d <= 1.0:
                            h[j] -= d
                            h[i] += d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                any_imp = True
                            else:
                                h[j] += d
                                h[i] -= d
            if not any_imp:
                break
    return h, best_c

def gradient_guided_local(h, n, deadline, radius=200):
    best_c = compute_c5(h, n)
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
    for d in steps:
        while time.time() < deadline:
            g = grad_smooth_max(h, n, 5000)
            ranked_inc = np.argsort(g)[:min(150, n)]
            ranked_dec = np.argsort(g)[::-1][:min(150, n)]
            any_imp = False
            for i in ranked_inc:
                if time.time() > deadline:
                    return h, best_c
                if h[i] + d > 1.0:
                    continue
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j or h[j] - d < 0.0:
                        continue
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        any_imp = True
                    else:
                        h[i] -= d
                        h[j] += d
                    if time.time() > deadline:
                        return h, best_c
            if not any_imp:
                break
    return h, best_c

def global_pair_search(h, n, deadline):
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
    for d in steps:
        while time.time() < deadline:
            any_imp = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[j] + d <= 1.0 and h[i] - d >= 0.0:
                        h[j] += d
                        h[i] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            any_imp = True
                        else:
                            h[j] -= d
                            h[i] += d
                    if time.time() > deadline:
                        return h, best_c
            if not any_imp:
                break
    return h, best_c

def golden_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(half) * phi).astype(int) % n
    h[positions] = 1.0
    return project_feasible(h, n)

def perturbed_start(h, n, rng, scale=0.03):
    h_p = h + rng.randn(n) * scale
    return project_feasible(h_p, n)

def multi_alpha_lbgfs(h, n, deadline):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 800, 3000, 12000, 50000, 200000, 1000000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        mu = 1.0
        try:
            res = minimize(lbfgs_obj, h, jac=lbfgs_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-20,
                                   'gtol': 1e-18, 'maxfun': 400 * 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = golden_start(650)
        n_init = 650

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS-B with increasing alpha (main optimization)
    t1 = min(t0 + 220, deadline - 650)
    best_h, best_c = lbfgs_optimize(best_h, best_n, t1, max_iters=800)

    # Phase 2: Multi-alpha refinement
    if time.time() < deadline - 500:
        t2 = min(time.time() + 100, deadline - 500)
        h2, c2 = multi_alpha_lbgfs(best_h, best_n, t2)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: SLSQP polish
    if time.time() < deadline - 350:
        t3 = min(time.time() + 70, deadline - 350)
        h3, c3 = slsqp_optimize(best_h, best_n, t3, alpha=10000000, maxiter=8000)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Constraint-focused local search
    if time.time() < deadline - 200:
        t4 = min(time.time() + 100, deadline - 200)
        best_h, best_c = constraint_focused_local(best_h, best_n, t4, n_top=10)

    # Phase 5: Gradient-guided local search
    if time.time() < deadline - 120:
        t5 = min(time.time() + 60, deadline - 120)
        h5, c5 = gradient_guided_local(best_h, best_n, t5, radius=200)
        if c5 < best_c:
            best_c, best_h = c5, h5

    # Phase 6: Try alternate grid sizes
    if time.time() < deadline - 100:
        for n_try in [550, 750]:
            if time.time() > deadline - 80:
                break
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(35, max(10, (deadline - time.time() - 60) * 0.4))
            h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc, max_iters=300)
            if c_t < best_c:
                best_c, best_h, best_n = c_t, h_t, n_try
                if time.time() < deadline - 50:
                    h_t, c_t = slsqp_optimize(best_h, best_n, deadline - 40,
                                             alpha=10000000, maxiter=5000)
                    if c_t < best_c:
                        best_c, best_h = c_t, h_t

    # Phase 7: Global pair fine search (most of remaining time)
    if time.time() < deadline - 10:
        best_h, best_c = global_pair_search(best_h, best_n, deadline - 5)

    # Phase 8: Final SLSQP squeeze
    if time.time() < deadline - 2:
        h_f, c_f = slsqp_optimize(best_h, best_n, deadline - 1,
                                   alpha=20000000, maxiter=3000)
        if c_f < best_c:
            best_c, best_h = c_f, h_f

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
