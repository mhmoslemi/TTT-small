def project_feasible(h, n):

            s = h.sum()
            if s < 1e-15: break

def lsq_obj_grad(x, alpha, n, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')[::-1]
    s = alpha * dx * corr
    m = float(s.max())
    ex = np.exp(s - m)
    es = float(ex.sum())
    obj = (m + math.log(es)) / alpha
    w = ex / es

    chr = np.convolve(x, wr, mode='full')

    sl = slice(n - 1, 2 * n - 1)
    grad = dx * (c1[sl] - chr[sl] - ch[sl])
    return obj, grad

_SL_ONEN = 0

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    for _ in range(20):
        d = target - h.sum()
        if abs(d) < 1e-14: break
        if d > 0:
            room = 1.0 - h; s = room.sum()
            if s < 1e-15: break
            h += (d / s) * room
        else:
            s = h.sum()
            if s < 1e-15: break
            h += (d / s) * h
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if d > 1e-12:
        idx = int(np.argmax(1.0 - h))
        h[idx] += min(d, 1.0 - h[idx])

        h[idx] -= min(-d, h[idx])

def lsq_obj_grad(x, alpha, n, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')[::-1]
    s = alpha * dx * corr
    m = float(s.max())
    ex = np.exp(s - m)
    es = float(ex.sum())
    obj = (m + math.log(es)) / alpha

    c1 = np.convolve(one, wr, mode='full')
    sl = slice(n - 1, 2 * n - 1)
    return obj, dx * (c1[sl] - chr[sl] - ch[sl])

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    for _ in range(20):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (d / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (d / s) * h
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if d > 1e-12:
        idx = int(np.argmax(1.0 - h))
        h[idx] += min(d, 1.0 - h[idx])
    elif d < -1e-12:
        idx = int(np.argmax(h))
        h[idx] -= min(-d, h[idx])
    return h

def thue_h(n):
    t = np.zeros(n, dtype=np.int8)
    for i in range(1, n):
        t[i] = int(t[i >> 1] ^ 1)
    return project_feasible(t.astype(float), n)

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def _get_ones(n):
    global _SL_ONE, _SL_ONEN
    if _SL_ONEN != n:
        _SL_ONE = np.ones(n)
        _SL_ONEN = n
    return _SL_ONE

def lsq_obj_grad(x, alpha, n, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')[::-1]
    s = alpha * dx * corr
    m = float(s.max())
    ex = np.exp(s - m)
    es = float(ex.sum())
    obj = (m + math.log(es)) / alpha
    w = ex / es
    wr = w[::-1]
    one = _get_ones(n)
    ch = np.convolve(x, w, mode='full')
    chr = np.convolve(x, wr, mode='full')
    c1 = np.convolve(one, wr, mode='full')
    sl = slice(n - 1, 2 * n - 1)
    return obj, dx * (c1[sl] - chr[sl] - ch[sl])

def _slq_obj(x):
    return lsq_obj_grad(x, _SL_ALPHA, _SL_N, _SL_DX)

def _slq_sum(x):
    return x.sum() - 0.5 * _SL_N

def _slq_sum_jac(x):
    return np.ones(_SL_N)

def slsqp_smooth(h, n, alpha, maxiter):
    global _SL_ALPHA, _SL_N, _SL_DX
    _SL_ALPHA = float(alpha)
    _SL_N = int(n)
    _SL_DX = 2.0 / n
    x0 = project_feasible(h, n)
    try:
        res = minimize(_slq_obj, x0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _slq_sum, 'jac': _slq_sum_jac}],
                       jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
        return project_feasible(res.x, n)
    except Exception:
        return x0

def coord_descent(h, n, max_sweeps, deadline, bis=24):
    h = project_feasible(np.asarray(h, dtype=float), n)
    M = 2 * n - 1
    c = n - 1
    m = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1.0 - h, mode='full')[::-1].astype(float)
    curmax = float(corr.max())
    eps = 1e-14
    tol = 1e-12

    for _sw in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False

        for i in range(n):
            old = float(h[i])
            q0 = float(corr[c])
            oldq = old * (1.0 - old)

            B = np.zeros(M)
            ks = i + m
            mask = (ks >= 0) & (ks < n)
            B[mask] = 1.0 - h[ks[mask]]

            km = i - m
            mask2 = (km >= 0) & (km < n)
            B[mask2] -= h[km[mask2]]
            B[c] = 0.0

            a = corr - old * B
            pos = B > eps
            neg = B < -eps
            usep = bool(pos.any())
            usen = bool(neg.any())

            if usep:
                invp = 1.0 / B[pos]
                cp = a[pos] * invp
            if usen:
                invn = 1.0 / B[neg]
                cn = a[neg] * invn

            zero = ~(pos | neg)
            zero[c] = False
            usez = bool(zero.any())
            azmax = float(a[zero].max()) if usez else -np.inf

            lo = 0.0
            if azmax > lo:
                lo = azmax
            hi = curmax
            if hi <= lo + 1e-15:
                continue

            for _b in range(bis):
                mid = 0.5 * (lo + hi)
                L = 0.0
                U = 1.0
                ok = True

                if usep:
                    u = float(np.min(mid * invp - cp))
                    if u < U:
                        U = u
                    if U < -tol:
                        ok = False

                if ok and usen:
                    l = float(np.max(mid * invn - cn))
                    if l > L:
                        L = l
                    if L > 1.0 + tol:
                        ok = False

                if ok and usez and azmax > mid + tol:
                    ok = False

                D = mid - q0 + oldq
                if ok:
                    if D < -tol or L > U + tol:
                        ok = False
                    elif D >= 0.25:
                        pass
                    elif D <= tol:
                        ok = (L <= tol and U >= -tol) or (L <= 1.0 + tol and U >= 1.0 - tol)
                    else:
                        sq = math.sqrt(max(0.0, 1.0 - 4.0 * D))
                        r = 0.5 * (1.0 - sq)
                        rr = 0.5 * (1.0 + sq)
                        ok = (L <= r + tol) or (U >= rr - tol)

                if ok:
                    hi = mid
                else:
                    lo = mid

                if hi - lo < 1e-13:
                    break

            if hi >= curmax - 1e-12:
                continue

            v = hi + 1e-12
            L = 0.0
            U = 1.0

            if usep:
                u = float(np.min(v * invp - cp))
                if u < U:
                    U = u
            if usen:
                l = float(np.max(v * invn - cn))
                if l > L:
                    L = l

            L = max(0.0, min(L, 1.0))
            U = max(0.0, min(U, 1.0))
            x = old

            if L <= U + 1e-10:
                D = v - q0 + oldq
                intervals = []

                if D >= 0.25:
                    l1 = max(L, 0.0)
                    u1 = min(U, 1.0)
                    if l1 <= u1 + tol:
                        intervals.append((l1, u1))
                elif D <= tol:
                    if L <= tol and U >= -tol:
                        intervals.append((0.0, 0.0))
                    if L <= 1.0 + tol and U >= 1.0 - tol:
                        intervals.append((1.0, 1.0))
                else:
                    sq = math.sqrt(max(0.0, 1.0 - 4.0 * D))
                    r = 0.5 * (1.0 - sq)
                    rr = 0.5 * (1.0 + sq)
                    l1 = max(L, 0.0)
                    u1 = min(r, U)
                    if l1 <= u1 + tol:
                        intervals.append((l1, u1))
                    l2 = max(rr, L)
                    u2 = min(1.0, U)
                    if l2 <= u2 + tol:
                        intervals.append((l2, u2))

                bestd = 1e100
                for l1, u1 in intervals:
                    l1 = max(0.0, min(l1, 1.0))
                    u1 = max(0.0, min(u1, 1.0))
                    if l1 > u1:
                        mp = 0.5 * (l1 + u1)
                        l1 = u1 = mp

                    if old < l1:
                        cand = l1
                    elif old > u1:
                        cand = u1
                    else:
                        cand = old

                    if abs(cand - old) < 1e-13:
                        cand = 0.5 * (l1 + u1)
                    if cand < l1:
                        cand = l1
                    elif cand > u1:
                        cand = u1

                    dv = abs(cand - old)
                    if dv < bestd:
                        bestd = dv
                        x = cand

            if abs(x - old) > 1e-14:
                newq = x * (1.0 - x)
                diffq = newq - oldq
                h[i] = x
                corr += B * (x - old)
                corr[c] += diffq
                nm = float(corr.max())
                if nm <= curmax + 1e-10:
                    curmax = nm
                    improved = True
                else:
                    h[i] = old
                    corr -= B * (x - old)
                    corr[c] -= diffq

        if not improved:
            break

    h = project_feasible(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full')[::-1].astype(float)
    return h, float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(60.0, float(budget_s) - 15.0)
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = thue_h(n)

    best = project_feasible(h0, n)
    best_c = compute_c5(best, n)
    best_n = n

    stage = min(t0 + 0.26 * total, dl - 25)
    h1, _ = coord_descent(best, best_n, 22, stage, 26)
    c = compute_c5(h1, best_n)
    if c < best_c:
        best, best_c = h1, c

    stage = min(t0 + 0.55 * total, dl - 35)
    for a, mi in ((20, 130), (100, 130), (500, 110), (2000, 90)):
        if time.time() > stage - 25:
            break
        h2 = slsqp_smooth(best, best_n, a, mi)
        c = compute_c5(h2, best_n)
        if c < best_c:
            best, best_c = h2, c

    stage = min(t0 + 0.66 * total, dl - 35)
    h3, _ = coord_descent(best, best_n, 10, stage, 24)
    c = compute_c5(h3, best_n)
    if c < best_c:
        best, best_c = h3, c

    stage = min(t0 + 0.81 * total, dl - 35)
    if time.time() < stage - 30:
        h4 = thue_h(best_n)
        h4 = slsqp_smooth(h4, best_n, 40, 100)
        h4, _ = coord_descent(h4, best_n, 8, stage, 22)
        c = compute_c5(h4, best_n)
        if c < best_c:
            best, best_c = h4, c

    stage = min(t0 + 0.89 * total, dl - 35)
    if time.time() < stage - 25:
        h5 = random_binary(best_n, rng)
        h5, _ = coord_descent(h5, best_n, 10, stage, 24)
        h5 = slsqp_smooth(h5, best_n, 150, 80)
        h5, _ = coord_descent(h5, best_n, 5, dl - 30, 22)
        c = compute_c5(h5, best_n)
        if c < best_c:
            best, best_c = h5, c

    stage = min(t0 + 0.965 * total, dl - 30)
    if time.time() < stage - 40 and best_n != 512:
        n5 = 512
        xo = np.linspace(0.0, 2.0, best_n)
        xn = np.linspace(0.0, 2.0, n5)
        h6 = project_feasible(np.interp(xn, xo, best), n5)
        h6, _ = coord_descent(h6, n5, 5, stage, 22)
        h6 = slsqp_smooth(h6, n5, 250, 90)
        h6, _ = coord_descent(h6, n5, 4, dl - 25, 20)
        c = compute_c5(h6, n5)
        if c < best_c:
            best, best_c, best_n = h6, c, n5

    stage = dl - 20
    if time.time() < stage - 20:
        for _ in range(3):
            if time.time() > stage - 10:
                break
            hv = best + 0.004 * rng.randn(best_n)
            hv = project_feasible(hv, best_n)
            hv = slsqp_smooth(hv, best_n, 1500, 70)
            hv, _ = coord_descent(hv, best_n, 4, min(stage, dl - 10), 20)
            c = compute_c5(hv, best_n)
            if c < best_c:
                best, best_c = hv, c

    best = project_feasible(best, best_n)
    return best, compute_c5(best, best_n), best_n
