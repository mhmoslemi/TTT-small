```python
    c = np.correlate(h_bin, 1-h_bin, mode='full')  # current correlation
    best_c = c.max()
    best_h = h_bin.copy()
    
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    temp = 0.1
    stall = 0
    
    while time.time() < deadline:
        # Try random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta
        a = i  # position being set to 0
        b = j  # position being set to 1
        # delta_c[m] = h[a+m-(n-1)] - h[b+m-(n-1)] (valid indices only)
        m_range = np.arange(2*n-1)
        valid_a = (a + m_range - (n-1) >= 0) & (a + m_range - (n-1) < n)
        valid_b = (b + m_range - (n-1) >= 0) & (b + m_range - (n-1) < n)
        
        delta = np.zeros(2*n-1)
        delta[valid_a] -= (1 - h_bin[a + m_range[valid_a] - (n-1)])
        delta[valid_b] += (1 - h_bin[b + m_range[valid_b] - (n-1)])

        new_max = new_c.max()
        
        if new_max < best_c or rng.random() < np.exp((best_c - new_max) / max(temp, 1e-10)):

            else:

        else:
            stall += 1
        
        if stall > 10000:

            stall = 0
    
    return best_h, best_c
```

The key insight: if I know the current correlation c, and I make a swap (a: 1→0, b: 0→1), the new correlation is:

- The term where i=a: h[a] * (1-h[a+m-(n-1)]) = 1 * (1-h[a+m-(n-1)])

The effect of adding h[b] (from 0 to 1):

```python
def compute_delta(h, a, b, n):
    """Compute delta to correlation when h[a] goes 1->0 and h[b] goes 0->1"""
    m_total = 2 * n - 1
    delta = np.zeros(m_total)
    
    # Effect of removing 1 at position a:
    # c[m] -= (1 - h[a + m - (n-1)]) for valid m
    # Valid: 0 <= a + m - (n-1) < n => (n-1-a) <= m <= (n-1-a) + (n-1) = 2(n-1)-a
    ma_start = max(0, n - 1 - a)
    ma_end = min(m_total - 1, 2 * (n - 1) - a)
    # indices into h: a + m - (n-1) for m in [ma_start, ma_end]
    # = a + (n-1-a) - (n-1) to a + 2(n-1)-a - (n-1) = 0 to n-1
    # So h indices go from 0 to (ma_end - ma_start)
    h_idx_start = a + ma_start - (n - 1)  # = 0
    h_idx_end = a + ma_end - (n - 1) + 1  # = ma_end - ma_start + 1
    delta[ma_start:ma_end+1] -= (1.0 - h[h_idx_start:h_idx_end])
    
    # Effect of adding 1 at position b:
    # c[m] += (1 - h[b + m - (n-1)]) for valid m
    mb_start = max(0, n - 1 - b)
    mb_end = min(m_total - 1, 2 * (n - 1) - b)
    h_idx_start = b + mb_start - (n - 1)  # = 0
    h_idx_end = b + mb_end - (n - 1) + 1
    delta[mb_start:mb_end+1] += (1.0 - h[h_idx_start:h_idx_end])
    
    return delta
```

```python
def swap_delta(h, a, b, n):
    """Compute delta to correlation when h[a] 1->0, h[b] 0->1"""
    m = 2 * n - 1
    delta = np.zeros(m)
    
    # h[a] goes 1->0:
    # Range 1: k in [a, a+n-1], delta -= (1 - h[a+n-1-k])
    #   h indices: a+n-1-k for k in [a, a+n-1] => h[n-1], h[n-2], ..., h[0]
    #   So delta[a:a+n] -= (1 - h[n-1::-1])  ... wait let me check
    #   k=a: h[a+n-1-a] = h[n-1]
    #   k=a+n-1: h[a+n-1-(a+n-1)] = h[0]
    #   So h indices go from n-1 down to 0 as k goes from a to a+n-1
    r1_start = a
    r1_end = a + n  # exclusive
    r1_start = max(0, r1_start)
    r1_end = min(m, r1_end)
    if r1_end > r1_start:
        h_indices = slice(n - 1 + r1_start - a, n - 1 + r1_end - a - 1, -1)
        # h[n-1 + (r1_start - a)] down to h[n-1 + (r1_end-1) - a] + 1 = ...
        # Let me just compute explicitly
        # For k in [r1_start, r1_end-1]: h_idx = a + n - 1 - k
        # h_idx ranges from a+n-1-r1_start to a+n-1-(r1_end-1)
        h_start = a + n - 1 - (r1_end - 1)  # smallest h index (when k is largest)
        h_end = a + n - 1 - r1_start + 1   # largest h index + 1 (exclusive)
        # delta[r1_start:r1_end] -= (1 - h[h_start:h_end][::-1])
        delta[r1_start:r1_end] -= (1.0 - h[h_start:h_end][::-1])
    
    # Range 2: k in [n-a-1, 2n-a-2], delta += h[a+k-n+1]
    r2_start = n - a - 1
    r2_end = 2 * n - a - 1  # exclusive (2n-a-2 + 1)
    r2_start = max(0, r2_start)
    r2_end = min(m, r2_end)
    if r2_end > r2_start:
        # h_idx = a + k - n + 1 for k in [r2_start, r2_end-1]
        # h_idx ranges from a+r2_start-n+1 to a+r2_end-1-n+1
        h_start = a + r2_start - n + 1
        h_end = a + (r2_end - 1) - n + 1 + 1  # = a + r2_end - n + 1
        delta[r2_start:r2_end] += h[h_start:h_end]
    
    # h[b] goes 0->1:
    # Range 3: k in [b, b+n-1], delta += (1 - h[b+n-1-k])
    r3_start = b
    r3_end = b + n
    r3_start = max(0, r3_start)
    r3_end = min(m, r3_end)
    if r3_end > r3_start:
        h_start = b + n - 1 - (r3_end - 1)
        h_end = b + n - 1 - r3_start + 1
        delta[r3_start:r3_end] += (1.0 - h[h_start:h_end][::-1])
    
    # Range 4: k in [n-b-1, 2n-b-2], delta -= h[b+k-n+1]
    r4_start = n - b - 1
    r4_end = 2 * n - b - 1
    r4_start = max(0, r4_start)
    r4_end = min(m, r4_end)
    if r4_end > r4_start:
        h_start = b + r4_start - n + 1
        h_end = b + (r4_end - 1) - n + 1 + 1
        delta[r4_start:r4_end] -= h[h_start:h_end]
    
    return delta
```
