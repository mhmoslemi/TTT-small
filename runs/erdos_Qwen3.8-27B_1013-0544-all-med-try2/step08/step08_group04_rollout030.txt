    h0 = project_feasible(h, n)
    c0 = compute_c5(h0, n)
    t0 = time.time()
    if deadline - t0 < 3:
        return h0, c0
    try:
        res = minimize(obj_pen, h0, jac=grad_pen, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxiter * 12,
                               'maxls': 40})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h0, c0

def momentum_gd(h, n, deadline, alpha=5000, lr_init=0.005):
    h = project_feasible(h, n)
    v = np.zeros(n)
    beta = 0.9
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    for _ in range(80000):
        if time.time() > deadline:
            break
        g = grad_smooth(h, n, alpha)
        v = beta * v + (1.0 - beta) * g
        h_new = project_feasible(h - lr * v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.85
            if lr < 1e-10:
                break
    return best_h, best_c

def focused_pair_search(h, n, deadline, top_k=5, m_cand=25):
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    best_c = compute_c5(h, n)
    nfull = 2 * n - 1
    for d in deltas:
        while time.time() < deadline:
            c = compute_c5_vec(h, n)
            top_indices = np.argsort(c)[-top_k:]
            improved = False
            for ki in top_indices:
                if time.time() > deadline:
                    return h, best_c
                k = ki - (n - 1)
                dx = 2.0 / n
                g_act = np.zeros(n)
                if k >= 0:
                    nv = n - k
                    g_act[:nv] = dx * (1.0 - h[k:])
                    g_act[k:] += dx * (-h[:nv])
                else:
                    kk = -k
                    nv = n - kk
                    g_act[kk:] = dx * (1.0 - h[:nv])
                    g_act[:nv] += dx * (-h[kk:])
                order = np.argsort(g_act)
                lows = order[:m_cand]
                highs = order[-m_cand:]
                for i in lows:
                    for j in highs:
                        if i == j:
                            continue
                        if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                            h[i] += d
                            h[j] -= d
                            c2 = compute_c5(h, n)
                            if c2 < best_c - 1e-14:
                                best_c = c2
                                improved = True
                            else:
                                h[i] -= d
                                h[j] += d
                        if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                            h[i] -= d
                            h[j] += d
                            c2 = compute_c5(h, n)
                            if c2 < best_c - 1e-14:
                                best_c = c2
                                improved = True
                            else:
                                h[i] += d
                                h[j] -= d
            if not improved:
                break
    return h, best_c

def local_radius_search(h, n, deadline, radius=40, deltas=None):
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c2 = compute_c5(h, n)
                        if c2 < best_c - 1e-14:
                            best_c = c2
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c2 = compute_c5(h, n)
                        if c2 < best_c - 1e-14:
                            best_c = c2
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def multi_start_refine(h_best, n, best_c, deadline, rng, n_starts=3):
    for s in range(n_starts):
        if time.time() > deadline - 60:
            break
        t_alloc = min(80, deadline - time.time() - 30)
        if t_alloc < 20:
            break
        # Structured perturbation
        scale = 0.02 * (s + 1)
        perturb = rng.randn(n) * scale
        h_pert = project_feasible(h_best + perturb, n)
        # L-BFGS-B phase
        t1 = time.time() + t_alloc * 0.5
        h_r, c_r = lbfgs_opt(h_pert, n, t1, alpha=10000, mu=50.0, maxiter=150)
        # Momentum GD
        if time.time() < deadline - 30:
            t2 = time.time() + min(30, deadline - time.time() - 20)
            h_r, c_r = momentum_gd(h_r, n, t2, alpha=50000, lr_init=0.002)
        # Focused pair search
        if time.time() < deadline - 10:
            h_r, c_r = focused_pair_search(h_r, n, min(time.time() + 20, deadline - 10),
                                          top_k=3, m_cand=20)
        if c_r < best_c:
            best_c = c_r
            h_best = h_r
    return h_best, best_c

def try_other_n(h_best, n_best, best_c, deadline, rng):
    for n_try in [200, 400, 500]:
        if time.time() > deadline - 40:
            break
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, h_best)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(50, deadline - time.time() - 20)
        if alloc < 15:
            break
        h_t, c_t = lbfgs_opt(h_interp, n_try, time.time() + alloc * 0.6,
                            alpha=5000, mu=10.0, maxiter=150)
        if time.time() < deadline - 10:
            h_t, c_t = focused_pair_search(h_t, n_try, min(time.time() + 20, deadline - 10),
                                          top_k=3, m_cand=20)
        if c_t < best_c:
            best_c = c_t
            h_best = h_t
            n_best = n_try
    return h_best, best_c, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n = 250

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick L-BFGS-B with increasing alpha
    alphas = [2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 400:
            break
        t_phase = min(time.time() + 40, deadline - 350)
        h1, c1 = lbfgs_opt(best_h, n, t_phase, alpha=a, mu=a * 1e-5, maxiter=150)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Momentum gradient descent
    if time.time() < deadline - 300:
        t2 = min(time.time() + 120, deadline - 250)
        for a in [10000, 100000]:
            if time.time() > t2:
                break
            h2, c2 = momentum_gd(best_h, n, min(time.time() + 60, t2),
                                 alpha=a, lr_init=0.003)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Focused pair search (main local refinement)
    if time.time() < deadline - 150:
        t3 = min(time.time() + 180, deadline - 100)
        best_h, best_c = focused_pair_search(best_h, n, t3, top_k=5, m_cand=30)

    # Phase 4: Multi-start refinement
    if time.time() < deadline - 80:
        t4 = min(time.time() + 200, deadline - 50)
        best_h, best_c = multi_start_refine(best_h, n, best_c, t4, rng, n_starts=4)

    # Phase 5: Try other n values
    if time.time() < deadline - 40:
        best_h, n, best_c = try_other_n(best_h, n, best_c, deadline - 10, rng)

    # Phase 6: Final polish - focused pair search with finer deltas
    if time.time() < deadline - 5:
        t6 = deadline - 2
        best_h, best_c = focused_pair_search(best_h, len(best_h), t6,
                                            top_k=3, m_cand=40)

    # Final: very fine local search
    if time.time() < deadline - 2:
        best_h, best_c = local_radius_search(best_h, len(best_h), deadline - 1,
                                            radius=30,
                                            deltas=[0.0001, 0.00005, 0.00002])

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
