```python
def greedy_step(h, corr, n):
    """Find the best single swap and apply it. Returns (h, corr, improved)."""
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    if len(zeros) == 0 or len(ones) == 0:
        return h, corr, False
    
    dx = 2.0 / n
    current_max = corr.max()
    m_range = np.arange(-(n-1), n)  # lags
    
    best_improvement = 0
    best_i, best_j = -1, -1
    best_new_corr = corr.copy()
    
    for i in zeros:
        # For this i (going 0->1), compute delta for all j in ones
        # delta[m] for lag m = m_range[m]
        # = (1-h[i+lag]) - (1-h[j+lag]) - h[i-lag] + h[j-lag]
        
        # Precompute terms that don't depend on j
        ip = i + m_range  # i + lag for each lag
        valid_ip = (ip >= 0) & (ip < n)
        term1 = np.where(valid_ip, 1.0 - h[np.clip(ip, 0, n-1)], 0.0)
        # But we need to be careful: where valid_ip is False, term1 = 0
        term1 = np.zeros(2*n-1)
        term1[valid_ip] = 1.0 - h[ip[valid_ip]]
        
        il = i - m_range  # i - lag for each lag
        valid_il = (il >= 0) & (il < n)
        term3 = np.zeros(2*n-1)
        term3[valid_il] = -h[il[valid_il]]
        
        base_ij = term1 + term3  # part that doesn't depend on j
        
        for j in ones:
            jp = j + m_range
            valid_jp = (jp >= 0) & (jp < n)
            term2 = np.zeros(2*n-1)
            term2[valid_jp] = -(1.0 - h[jp[valid_jp]])
            
            jl = j - m_range
            valid_jl = (jl >= 0) & (jl < n)
            term4 = np.zeros(2*n-1)
            term4[valid_jl] = h[jl[valid_jl]]
            
            delta = base_ij + term2 + term4
            new_corr = corr + delta
            new_max = new_corr.max()
            improvement = current_max - new_max
            if improvement > best_improvement + 1e-15:
                best_improvement = improvement
                best_i, best_j = i, j
                best_new_corr = new_corr
    
    if best_i >= 0:
        h[best_i] = 1.0
        h[best_j] = 0.0
        return h, best_new_corr, True
    return h, corr, False
```

```python
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n

    # Phase 2: Extended SA on best solution with lower temperature
    if time.time() < deadline - 80:
        h = best_h.copy()
        for _ in range(3):
            if time.time() > deadline - 60:
                break
            c_init = compute_c5(h, best_n)
            h_sa, corr_sa, c_sa, bc = sa_optimize(h, best_n, deadline - 50, rng, T0=0.15, T_end=0.002)
            if c_sa < best_c:
                best_c, best_h = c_sa, h_sa.copy()
            h = h_sa.copy()

    # Phase 3: Final greedy polish
    if time.time() < deadline - 15:
        h = best_h.copy()
        h, corr, c = greedy_optimize(h, best_n, deadline - 5, rng)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Block swap refinement
    if time.time() < deadline - 10:
        h = best_h.copy()
        corr = get_corr(h, best_n)
        bl_deadline = deadline - 5
        stagnant = 0
        while time.time() < bl_deadline:
            h, corr, improved = block_swap_step(h, corr, best_n, rng)
            c = corr.max() * (2.0 / best_n)
            if improved and c < best_c:
                best_c = c
                best_h = h.copy()
                stagnant = 0
            elif not improved:
                stagnant += 1
                if stagnant > 5:
                    break
            else:
                stagnant += 1
                if stagnant > 10:
                    break

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    # Ensure exact constraint
    s = best_h.sum()
    target = best_n / 2.0
    if abs(s - target) > 1e-10:
        idx = np.argsort(-best_h)
        best_h = np.zeros(best_n)
        best_h[idx[:int(target)]] = 1.0
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
```
