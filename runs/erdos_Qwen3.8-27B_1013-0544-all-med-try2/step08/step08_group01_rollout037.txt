    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2

    m = c.max()

    mx = sh.max()

            if tot < 1e-14:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-14:
            continue
        k = k_idx - (n - 1)
        wm = w[k_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def lbfgs_opt(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = (500, 5000, 50000, 500000, 2000000)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        if deadline - time.time() < 10:
            break
        mu = max(1.0, alpha * 0.01)
        try:
            res = minimize(_sm_obj, h, jac=_sm_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': 400, 'ftol': 1e-17,
                                  'gtol': 1e-15, 'maxfun': 12000,
                                  'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def _fourier_obj(params, n, m, alpha, beta):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(1, m + 1):
        h += params[2*(j-1)] * np.cos(j * np.pi * x) + \
             params[2*(j-1)+1] * np.sin(j * np.pi * x)
    c = compute_c5_vec(h, n)
    mx = c.max()
    sh = alpha * (c - mx)
    smx = sh.max()
    val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    viol = np.maximum(0, h - 1.0).sum() + np.maximum(0, -h).sum()
    val += beta * viol * viol
    return float(val)

def h_to_fourier(h, n, m):
    x = np.linspace(0, 2, n, endpoint=False)
    params = np.zeros(2 * m)
    for j in range(1, m + 1):
        params[2*(j-1)] = np.sum(h * np.cos(j * np.pi * x)) * (2.0/n)
        params[2*(j-1)+1] = np.sum(h * np.sin(j * np.pi * x)) * (2.0/n)
    return params

def sa_binary(h0, n, deadline, n_rounds=60, seed=42):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    one_list = list(np.where(h == 1)[0])
    zero_list = list(np.where(h == 0)[0])

            ii = zero_list[i]
            jj = one_list[j]
            h[ii] = 1.0
            h[jj] = 0.0
            new_c = compute_c5(h, n)
            delta = new_c - cur_c

                one_list[j] = ii
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[ii] = 0.0
                h[jj] = 1.0
    return best_h, best_c

def pair_polish(h, n, deadline, radius=30):
    best_c = compute_c5(h, n)
    deltas = (0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, np.tile(x_old, 2), np.tile(h_old, 2))
    return project_feasible(h_new, n_new)
