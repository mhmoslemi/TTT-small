This is what the previous solution does. The issue is that for large α, it becomes ill-conditioned.

Let me try a different approach: **Alternating optimization with bisection on C**

For a given C, I want to find h such that:
- 0 ≤ h ≤ 1
- sum(h) = n/2  
- overlap(k) ≤ C for all k

I'll solve this as a sequence of subproblems:
1. Fix h, compute B matrix
2. For each coordinate i, find the optimal h[i] given the constraint overlap(k) ≤ C for all k
   This is: find h[i] ∈ [0,1] such that corr[k] + B_i[k]*(h[i] - h[i]_old) ≤ C for all k
   Which gives: h[i] ∈ [max(0, max_k (C - corr[k])/B_i[k] for B_i[k] > 0), min(1, min_k (C - corr[k])/B_i[k] for B_i[k] < 0)]
   Plus we need the sum constraint.

This is essentially what the coordinate descent does, but with a hard constraint instead of minimizing the max.

The issue is the sum constraint. If I just clip each coordinate independently, the sum won't be right.

Let me try a different formulation:

**Projected subgradient method on C:**
1. Start with C = best_c
2. Compute subgradient of g(h) = max_k overlap(k) at current h
3. Take a step in the negative subgradient direction
4. Project onto the feasible set {0 ≤ h ≤ 1, sum(h) = n/2}
5. If g(h) < C, decrease C; else increase C
6. Repeat

The subgradient of g(h) = max_k overlap(k) at h is:
∇g(h) = (1/m_active) * ∑_{k ∈ active} ∇overlap(k)

where active = {k : overlap(k) = max_k overlap(k)}.

For the projection onto {0 ≤ h ≤ 1, sum(h) = n/2}, I can use the existing project_feasible function.

This is a simple and robust approach. The step size is the tricky part - I'll use a diminishing step size or backtracking.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def sum_eq(x, n):
    return x.sum() - n / 2.0

def make_bounds(n):
    return [(0.0, 1.0)] * n

def coord_descent(h, n, max_sweeps=200, deadline=None, bs_iters=55):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            m1 = (k1 >= 0) & (k1 < n)
            B[m1] += (1.0 - h[k1[m1]])
            k2 = i + m_range
            m2 = (k2 >= 0) & (k2 < n)
            B[m2] -= h[k2[m2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-10
            lo_v = min(val0, val1) - 1e-10
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(bs_iters):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-15:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-13:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
                improved = True
        if not improved:
            break
    return h

def slsqp_smooth(h, n, alpha, maxiter=2000, ftol=1e-18):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    try:
        res = minimize(obj, h, method='SLSQP', bounds=make_bounds(n),
                       constraints=[{'type': 'eq', 'fun': sum_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def penalty_optimize(h, n, C, maxiter=200, lr=None, deadline=None):
    """Minimize sum(max(0, g(k)-C)^2) via projected gradient"""
    dx = 2.0 / n
    h = h.astype(float).copy()
    m_range = np.arange(-(n - 1), n)
    if lr is None:
        lr = 0.05
    for it in range(maxiter):
        if deadline and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        viol = np.maximum(0.0, corr - C)
        if viol.max() < 1e-14:
            break
        B = np.zeros((n, 2 * n - 1))
        for i in range(n):
            k1 = i - m_range
            m1 = (k1 >= 0) & (k1 < n)
            B[i, m1] += (1.0 - h[k1[m1]])
            k2 = i + m_range
            m2 = (k2 >= 0) & (k2 < n)
            B[i, m2] -= h[k2[m2]]
        grad = 2.0 * dx * (B @ viol)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        c_cur = compute_c5(h, n)
        if c_new > c_cur + 1e-12:
            lr *= 0.5
            if lr < 1e-6:
                break
        h = h_new
    return h

def structured_init(n, mode=0):
    """Create structured initializations"""
    h = np.zeros(n)
    if mode == 0:
        # Golden ratio Beatty sequence
        alpha = (np.sqrt(5.0) - 1.0) / 2.0
        for i in range(n):
            if np.floor((i + 1) * alpha) != np.floor(i * alpha):
                h[i] = 1.0
    elif mode == 1:
        # Alternating blocks of size ~n/10
        bs = max(1, n // 10)
        for i in range(n):
            if (i // bs) % 2 == 0:
                h[i] = 1.0
    elif mode == 2:
        # Sine-modulated
        for i in range(n):
            h[i] = 0.5 + 0.5 * np.sin(2.0 * np.pi * i / n * 3.0)
    elif mode == 3:
        # Two-scale block
        bs1 = max(1, n // 4)
        bs2 = max(1, n // 20)
        for i in range(n):
            big_block = (i // bs1) % 2
            small_block = (i // bs2) % 3
            if big_block == 0 and small_block < 2:
                h[i] = 1.0
            elif big_block == 1 and small_block == 2:
                h[i] = 1.0
    h = project_feasible(h, n)
    return h

def optimize_at_n(h_init, n, deadline, alphas):
    """Full optimization pipeline at a given n"""
    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    # CD phase 1
    dl1 = min(deadline - 20, time.time() + 0.25 * (deadline - time.time()))
    h = coord_descent(h, n, max_sweeps=150, deadline=dl1, bs_iters=55)
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
    # SLSQP
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        h = slsqp_smooth(h, n, alpha)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
    # CD phase 2
    dl2 = min(deadline - 5, time.time() + 0.15 * (deadline - time.time()))
    h = coord_descent(h, n, max_sweeps=100, deadline=dl2, bs_iters=60)
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
    # Final SLSQP
    if time.time() < deadline - 5:
        for alpha in alphas[-2:]:
            h = slsqp_smooth(h, n, alpha, maxiter=3000)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
            h = coord_descent(h, n, max_sweeps=50, bs_iters=60)
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
    h = project_feasible(h, n)
    best_c = min(best_c, compute_c5(h, n))
    return h, best_c
