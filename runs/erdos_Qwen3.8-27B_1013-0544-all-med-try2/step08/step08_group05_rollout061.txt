```python
def make_binary(h, n):
    """Convert h to binary with exactly n//2 ones, preserving structure."""
    target = n // 2
    # Sort by value, threshold at the target-th largest
    thresh = np.sort(h)[-target]
    h_bin = (h >= thresh).astype(float)
    # Fix if we have too many/few ones
    n_ones = int(h_bin.sum())
    if n_ones > target:
        # Turn off the smallest among those that are 1
        one_vals = h[h_bin == 1]
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(one_vals)
        for k in order[n_ones - target:]:
            h_bin[one_idx[k]] = 0
    elif n_ones < target:
        zero_vals = h[h_bin == 0]
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(zero_vals)[::-1]
        for k in order[:target - n_ones]:
            h_bin[zero_idx[k]] = 1
    return h_bin

def compute_corr(h, n):
    """Compute full correlation vector."""
    return np.correlate(h, 1 - h, mode='full')

def swap_delta(h, i, j, n):
    """Compute the change in correlation when swapping h[i]:0->1, h[j]:1->0."""

    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)

    jm = j + m_range
```

```python
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}."""
    h = h.copy()
    # Use bisection on the Lagrange multiplier
    lo, hi = -10, 10

        s = x.sum()

            lo = mid
        else:
```

```python
def smooth_max_grad(h, n, alpha):
    """Gradient of dx * (1/alpha) * log(sum exp(alpha * corr[k]))"""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    # Softmax weights (numerically stable)
    a = alpha * (corr - corr.max())
    w = np.exp(a)
    w /= w.sum()
    
    # grad[i] = correlate(w, 1-h, 'full')[i] - conv(w, h)[n-1+i]
    # Use FFT for efficiency
    # correlate(w, 1-h, 'full')[i] for i=0..n-1
    # This is the first n elements of the full correlation
    
    # For n=250, direct computation is O(n * 2n) = O(n^2) = 125000, which is fine.
    # But let's use FFT for elegance.
    
    # Method 1: direct (O(n^2), fine for n=250)
    grad = np.zeros(n)
    for i in range(n):
        # First term: sum_k w[k] * (1-h)[i+k-(n-1)] for valid
        k_lo = max(0, n-1-i)
        k_hi = min(2*n-2, 2*n-2-i)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi+1)
            idx = i + ks - (n-1)
            grad[i] += np.dot(w[ks], (1-h)[idx])
        # Second term: -sum_k w[k] * h[i-k+(n-1)] for valid
        k_lo2 = max(0, i)
        k_hi2 = min(2*n-2, i+n-1)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2+1)
            idx2 = i - ks2 + (n-1)
            grad[i] -= np.dot(w[ks2], h[idx2])
    
    return grad * dx
```
