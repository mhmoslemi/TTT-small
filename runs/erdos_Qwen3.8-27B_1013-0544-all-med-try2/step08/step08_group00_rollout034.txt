import numpy as np
import time
import math
from scipy.optimize import minimize

GD = 0.0

def cb(x):
    return time.time() > GD

def project_feasible(h, n):
    t = 0.5 * n
    h = np.clip(np.array(h, dtype=float, copy=True), 0.0, 1.0)
    for _ in range(2000):
        s = float(h.sum()); d = t - s
        if abs(d) < 1e-14: break
        if d > 0:
            r = 1.0 - h; tot = float(r.sum())
            if tot < 1e-15: break
            h += (d / tot) * r
        else:
            if s < 1e-15: break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def c5vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def c5(h, n):
    return float(c5vec(h, n).max())

def sval(c, a):
    m = float(c.max()); sh = a * (c - m); mx = float(sh.max())
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / a

def swts(c, a):
    m = float(c.max()); sh = a * (c - m); mx = float(sh.max())
    e = np.exp(sh - mx); return e / e.sum()

def grad(h, n, a, c=None):
    if c is None: c = c5vec(h, n)
    w = swts(c, a); dx = 2.0 / n; ct = n - 1; g = np.zeros(n)
    w0 = w[ct] * dx
    if w0 > 1e-15: g += w0 * (1.0 - 2.0 * h)
    for k in range(1, n):
        nv = n - k
        wp = w[ct + k] * dx
        if wp > 1e-15:
            g[:nv] += wp * (1.0 - h[k:])
            g[k:] -= wp * h[:nv]
        wn = w[ct - k] * dx
        if wn > 1e-15:
            g[k:] += wn * (1.0 - h[:nv])
            g[:nv] -= wn * h[k:]
    return g

def hobj(x, n, a, mu):
    c = c5vec(x, n); v = sval(c, a); s = float(x.sum()) - 0.5 * n
    return float(v + 0.5 * mu * s * s)

def hgr(x, n, a, mu):
    g = grad(x, n, a); s = float(x.sum()) - 0.5 * n; g += mu * s
    return g

def hlbfgs(h, n, dl, a, mu, mi=220):
    h = np.array(h, dtype=float, copy=True); GD = dl
    mi = min(mi, max(60, int(160000 // n)))
    try:
        r = minimize(hobj, h, jac=hgr, args=(n, a, mu), method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n, options={'maxiter': mi, 'ftol': 1e-15, 'gtol': 1e-9, 'maxls': 40, 'iprint': 0, 'factr': 1e12}, callback=cb)
        x = r.x
    except Exception:
        x = h
    x = project_feasible(x, n); return x, c5(x, n)

def lbfgs_sweep(h, n, dl):
    best_h = h.copy(); best_c = c5(h, n)
    for a, mu, mi in [(80.0, 20.0, 280), (400.0, 50.0, 240), (2000.0, 100.0, 200), (10000.0, 200.0, 150)]:
        if time.time() > dl - 3: break
        al = min(55.0, dl - time.time())
        if al < 12: break
        x, c = hlbfgs(best_h, n, time.time() + al, a, mu, mi)
        if c < best_c:
            best_h, best_c = x, c
    return best_h, best_c

import numpy as np
import time
import math
from scipy.optimize import minimize

GD = 0.0

def cb(x):
    return time.time() > GD

def project_feasible(h, n):
    t = 0.5 * n
    h = np.clip(np.array(h, dtype=float, copy=True), 0.0, 1.0)
    for _ in range(3000):
        s = float(h.sum()); d = t - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h; tot = float(r.sum())
            if tot < 1e-15:
                break
            h += (d / tot) * r
        else:
            if s < 1e-15:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def c5vec(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def c5(h, n):
    return float(c5vec(h, n).max())

def sval(c, a):
    m = float(c.max()); sh = a * (c - m); mx = float(sh.max())
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / a

def swts(c, a):
    m = float(c.max()); sh = a * (c - m); mx = float(sh.max())
    e = np.exp(sh - mx)
    return e / e.sum()

def grad(h, n, a, c=None):
    if c is None:
        c = c5vec(h, n)
    w = swts(c, a); dx = 2.0 / n; ct = n - 1; g = np.zeros(n)
    w0 = w[ct] * dx
    if w0 > 1e-15:
        g += w0 * (1.0 - 2.0 * h)
    for k in range(1, n):
        nv = n - k
        wp = w[ct + k] * dx
        if wp > 1e-15:
            g[:nv] += wp * (1.0 - h[k:])
            g[k:] -= wp * h[:nv]
        wn = w[ct - k] * dx
        if wn > 1e-15:
            g[k:] += wn * (1.0 - h[:nv])
            g[:nv] -= wn * h[k:]
    return g

def hobj(x, n, a, mu):
    c = c5vec(x, n)
    v = sval(c, a)
    s = float(x.sum()) - 0.5 * n
    return float(v + 0.5 * mu * s * s)

def hgr(x, n, a, mu):
    g = grad(x, n, a)
    s = float(x.sum()) - 0.5 * n
    g += mu * s
    return g

def hlbfgs(h, n, dl, a, mu, mi=220):
    global GD
    h = np.array(h, dtype=float, copy=True)
    GD = dl
    mi = min(mi, max(60, int(160000 // max(1, n))))
    try:

                     callback=cb)

def apply_dt(c, h, p, dt, n):
    old = float(h[p]); new = old + float(dt)
    if new > 1.0:
        new = 1.0
    elif new < 0.0:
        new = 0.0
    d = new - old
    if d == 0.0:
        return
    ct = n - 1
    c[ct] += d - 2.0 * old * d - d * d
    r = n - p - 1
    if r > 0:

        c[n:n + r] -= d * seg

        c[n:n + l] += d * (1.0 - seg)

    if d <= 1e-15:
        return False, old

    apply_dt(c, h, i, -d, n)
    return False, old

def local(h, n, dl, rng, a=1500, steps=None, K=24):
    h = np.array(h, dtype=float, copy=True)
    n = len(h)
    if n < 2:
        return h, c5(h, n)
    if steps is None:
        steps = [1.0, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 1e-4, 5e-5]
    c = c5vec(h, n)
    best = float(c.max())
    for d in sorted(steps, reverse=True):
        if time.time() > dl:
            break
        no = 0
        while time.time() < dl:
            c = c5vec(h, n)
            best = float(c.max())
            g = grad(h, n, a, c=c)
            idxs = np.argsort(g)
            lim = min(n, 5 * K)
            low = []; high = []
            for idx in idxs[:lim]:
                if h[idx] < 1.0 - 1e-10:
                    low.append(int(idx))
            for idx in idxs[max(0, n - lim):]:
                if h[idx] > 1e-10:
                    high.append(int(idx))
            cands = []
            for i in low[:K]:
                for j in high[:K]:
                    if i != j:
                        cands.append((i, j))
            if n > 1:
                tl = np.argsort(c)[-4:]
                for mi in tl:
                    k = int(mi) - (n - 1)
                    if k == 0:
                        continue
                    if k > 0:
                        nv = n - k
                        ii = np.arange(nv); jj = ii + k
                        v = h[jj] * (1.0 - h[ii])
                        for ti in np.argsort(v)[-12:]:
                            i = int(ii[ti]); j = int(jj[ti])
                            if h[i] < 1.0 - 1e-10 and h[j] > 1e-10:
                                cands.append((i, j))
                    else:
                        kk = -k; nv = n - kk
                        ii = np.arange(nv); jj = ii + kk
                        v = h[ii] * (1.0 - h[jj])
                        for ti in np.argsort(v)[-12:]:
                            i = int(jj[ti]); j = int(ii[ti])
                            if h[i] < 1.0 - 1e-10 and h[j] > 1e-10:
                                cands.append((i, j))
            if low and high:
                lp = low[:min(50, len(low))]
                hp = high[:min(50, len(high))]
                for _ in range(120):
                    i = lp[int(rng.randint(len(lp)))]
                    j = hp[int(rng.randint(len(hp)))]
                    if i != j:
                        cands.append((i, j))
            for _ in range(50):
                i = int(rng.randint(n)); j = int(rng.randint(n))
                if i != j and h[i] < 1.0 - 1e-10 and h[j] > 1e-10:
                    cands.append((i, j))
            if cands:
                rng.shuffle(cands)
            acc = 0; tr = 0
            for i, j in cands:
                if time.time() >= dl:
                    break
                ok, nm = try_mv(c, h, i, j, d, n)
                tr += 1
                if ok:
                    best = nm; acc += 1
                if tr % 1500 == 0:
                    c = c5vec(h, n)
                    best = float(c.max())
            if acc == 0:
                no += 1
                if no >= 2:
                    break
    c = c5vec(h, n)
    return h, float(c.max())

def bround(h, n):
    hb = np.zeros(n)
    m = int(round(0.5 * n))
    hb[np.argsort(h)[::-1][:m]] = 1.0
    return project_feasible(hb, n)

def resize(h, nn):
    n = len(h)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), nn)

def golden(n):
    h = np.zeros(n)
    m = n // 2
    ph = (1.0 + np.sqrt(5.0)) / 2.0
    h[(np.floor(np.arange(m) * ph).astype(int)) % n] = 1.0
    return project_feasible(h, n)

def anneal(h, n, dl, rng, T=3e-4):
    h = np.array(h, dtype=float, copy=True)
    c = c5vec(h, n)
    cur = float(c.max()); best = cur; hb = h.copy()
    st = 0.08; it = 0; stb = 0
    lo_idx = np.arange(n); hi_idx = lo_idx
    while time.time() < dl:
        if it % 2000 == 0:
            g = grad(h, n, 800, c=c)
            od = np.argsort(g)
            m = max(1, n // 2)
            lo_idx = od[:m]
            hi_idx = od[-m:]
        i = int(lo_idx[int(rng.randint(len(lo_idx)))])
        j = int(hi_idx[int(rng.randint(len(hi_idx)))])
        d = st * (0.4 + 0.8 * rng.random())
        if i != j:
            d = min(d, 1.0 - h[i], h[j])
            if d > 1e-12:
                old = cur
                apply_dt(c, h, i, d, n)
                apply_dt(c, h, j, -d, n)
                nm = float(c.max())
                acc = nm <= old + 1e-13
                if not acc and T > 1e-10:
                    df = nm - old
                    if df < 20.0 * T and rng.uniform() < math.exp(-df / T):
                        acc = True
                if acc:
                    cur = nm
                    if nm < best - 1e-14:
                        best = nm; hb = h.copy(); stb = 0
                    else:
                        stb += 1
                else:
                    apply_dt(c, h, j, d, n)
                    apply_dt(c, h, i, -d, n)
        it += 1
        if it % 1000 == 0:
            c = c5vec(h, n)
            cur = float(c.max())
            if stb > 6000:
                T *= 0.6
                st *= 0.85
                stb = 0
            if cur > best + 0.01:
                h = hb.copy()
                c = c5vec(h, n)
                cur = best
    c = c5vec(hb, n)
    return hb, float(c.max())

def bases(bs):
    f = max(0.08, min(1.0, float(bs) / 1100.0))
    return [max(4.0, x * f) for x in (180, 140, 90, 360, 70, 70, 70)]

def run(seed=42, budget_s=1100, **kwargs):
    global GD
    t0 = time.time()
    dl = t0 + float(budget_s) - 1.5
    GD = dl
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 650
        h0 = golden(n0)

    h0 = project_feasible(h0, n0)
    bh = h0.copy(); bn = n0; bc = c5(bh, bn)
    du = bases(budget_s)

    e = min(dl - 20, time.time() + du[0])
    if time.time() < e:
        x, c = lbfgs_sweep(bh, bn, e)
        if c < bc:
            bh, bc = x, c

    e = min(dl - 20, time.time() + du[1])
    if time.time() < e:
        x, c = local(bh, bn, e, rng, a=1500)
        if c < bc:
            bh, bc = x, c

    e = min(dl - 25, time.time() + du[2])
    if time.time() < e:
        hb = bround(bh, bn)
        x, c = local(hb, bn, e, rng, a=1500,
                     steps=[1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002])
        if c < bc:
            bh, bc = x, c

    e_up = min(dl - 25, time.time() + du[3])
    if time.time() < e_up:
        nh = bn + max(50, int(0.35 * bn))
        if nh > 900:
            nh = 900
        if nh % 2:
            nh -= 1
        if nh > bn:
            hi = resize(bh, nh)
            hc = c5(hi, nh)
            e1 = min(e_up, time.time() + 0.50 * du[3])
            if time.time() < e1:
                hi, hc = lbfgs_sweep(hi, nh, e1)
            e2 = min(e_up, time.time() + 0.30 * du[3])
            if time.time() < e2:
                hi, hc = local(hi, nh, e2, rng, a=1500)
            e3 = min(e_up, time.time() + 0.18 * du[3])
            if time.time() < e3:
                hi, hc = local(hi, nh, e3, rng, a=5000,
                               steps=[0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 1e-4])
            if hc < bc:
                bh, bn, bc = hi, nh, hc

    e = min(dl - 30, time.time() + du[4])
    if time.time() < e:
        hp = project_feasible(bh + rng.randn(bn) * 0.03, bn)
        x, c = local(hp, bn, e, rng, a=1000,
                     steps=[0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
        if c < bc:
            bh, bc = x, c

    e = min(dl - 10, time.time() + du[5])
    if time.time() < e:
        x, c = anneal(bh, bn, e, rng)
        if c < bc:
            bh, bc = x, c

    if time.time() < dl - 2:
        x, c = local(bh, bn, dl - 1, rng, a=5000,
                     steps=[0.02, 0.005, 0.002, 0.001, 0.0005, 0.0002, 1e-4, 5e-5], K=30)
        if c < bc:
            bh, bc = x, c

    bh = project_feasible(bh, bn)
    bc = c5(bh, bn)
    return bh, bc, bn
