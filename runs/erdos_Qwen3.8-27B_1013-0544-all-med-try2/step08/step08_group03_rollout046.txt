import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def nesterov_pgd(h, n, deadline, top_k=15, alpha_start=200.0, alpha_end=15000.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    momentum = 0.8
    step = 0.025
    y = h.copy()
    h_cur = h.copy()
    it = 0
    stall = 0
    while time.time() < deadline:
        it += 1
        frac = min(float(it) / 2000.0, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        alpha = min(alpha, alpha_end)
        c = compute_c5_vec(y, n)
        idx = np.argsort(c)[-top_k:]
        vals = c[idx]
        m = vals.max()
        sh = alpha * (vals - m)
        w = np.exp(sh - sh.max())
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for i in range(top_k):
            if w[i] < 1e-10:
                continue
            k = idx[i] - (n - 1)
            grad += w[i] * dx * grad_single(y, n, k)
        h_new = project_feasible(y - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        y = h_new + momentum * (h_new - h_cur)
        np.clip(y, 0.0, 1.0, out=y)
        h_cur = h_new
        if c_new > best_c + 1e-5:
            step *= 0.5
        elif c_new < best_c - 1e-9:
            step = min(step * 1.03, 0.07)
        if stall > 300:
            break
    return best_h, best_c

def pair_polish(h, n, deadline, radius=50):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.002, 0.001, 0.0005, 0.00025, 0.000125, 6e-5, 3e-5, 1.5e-5, 7.5e-6]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def lbfgs_obj(x, n, alpha, top_k):
    c = compute_c5_vec(x, n)
    idx = np.argsort(c)[-top_k:]
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 500.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def lbfgs_grad(x, n, alpha, top_k):
    c = compute_c5_vec(x, n)
    idx = np.argsort(c)[-top_k:]
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(top_k):
        if w[i] < 1e-14:
            continue
        k = idx[i] - (n - 1)
        grad += w[i] * dx * grad_single(x, n, k)
    s = x.sum() - n / 2.0
    grad += 1000.0 * s
    return grad
