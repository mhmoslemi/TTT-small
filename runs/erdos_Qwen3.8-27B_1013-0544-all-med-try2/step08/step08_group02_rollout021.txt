            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx]
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def fast_binary_search(h, n, deadline, rng, max_flips=5000000):
    """Fast binary local search using O(n) per swap evaluation."""
    dx = 2.0 / n
    h = h.astype(np.float64).copy()
    c = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    best_c_int = int(c.max())
    best_c = best_c_int * dx
    best_h = h.copy()
    
    h_rev = h[::-1].copy()
    one_minus_h = 1 - h
    
    temp = 0.01
    flips = 0
    no_improve = 0
    
    while time.time() < deadline and flips < max_flips:
        if flips > 0 and flips % 100000 == 0:
            temp *= 0.9
            if no_improve > 500000:
                break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # For speed, precompute arrays periodically
        if flips % 10000 == 0:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            h_rev = h[::-1]
            one_minus_h = 1 - h
        
        i0 = ones[rng.randint(len(ones))]
        j0 = zeros[rng.randint(len(zeros))]
        
        # Compute delta for swap (i0: 1->0, j0: 0->1)
        delta = np.zeros(2 * n - 1)
        
        # i0: 1->0: delta[n-1-i0 : 2n-1-i0] -= (1-h), delta[i0 : n+i0] += h[::-1]
        delta[n-1-i0 : 2*n-1-i0] -= one_minus_h
        delta[i0 : n+i0] += h_rev
        
        # j0: 0->1: delta[n-1-j0 : 2n-1-j0] += (1-h), delta[j0 : n+j0] -= h[::-1]
        delta[n-1-j0 : 2*n-1-j0] += one_minus_h
        delta[j0 : n+j0] -= h_rev
        
        new_c = c + delta
        new_max = int(new_c.max())
        
        if new_max < best_c_int:
            # Accept
            h[i0] = 0
            h[j0] = 1
            c = new_c
            best_c_int = new_max
            best_c = best_c_int * dx
            best_h = h.copy()
            no_improve = 0
        elif rng.random() < np.exp(-(new_max - best_c_int) * dx / max(temp, 1e-10)):
            h[i0] = 0
            h[j0] = 1
            c = new_c
            no_improve += 1
        else:
            no_improve += 1
        
        flips += 1
    
    return best_h, best_c

        h_rev = h[::-1]

        best_move = best_c_int
        best_i, best_j = -1, -1
        best_delta = None
        
        n_try = min(len(ones) * len(zeros), 50000)
        for _ in range(n_try):
            i0 = ones[rng.randint(len(ones))]
            j0 = zeros[rng.randint(len(zeros))]

            delta[i0 : n+i0] += h_rev
            delta[n-1-j0 : 2*n-1-j0] += one_minus_h
            delta[j0 : n+j0] -= h_rev

            best_c_int = best_move
            best_c = best_c_int * dx
            best_h = h.copy()
        else:
            break
    
    return best_h, best_c

    x = np.linspace(0, 2, n, endpoint=False)
    for c, w in zip(centers, widths):
        p = 0.5 + 0.5 * np.exp(-((x - c) ** 2) / (2 * w ** 2))
        h += p
    h = (h > h.mean() * n / (n / 2)).astype(np.float64)
    cnt = int(h.sum())
    target = n // 2
    if cnt > target:
        h[rng.choice(np.where(h == 1)[0], cnt - target, replace=False)] = 0
    elif cnt < target:
        h[rng.choice(np.where(h == 0)[0], target - cnt, replace=False)] = 1
    return h

def fourier_binary(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 10):
        amp = rng.uniform(0.15, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(np.pi * freq * x + phase)
    h = (h > 0.5).astype(np.float64)
    cnt = int(h.sum())
    target = n // 2
    if cnt > target:
        h[rng.choice(np.where(h == 1)[0], cnt - target, replace=False)] = 0
    elif cnt < target:
        h[rng.choice(np.where(h == 0)[0], target - cnt, replace=False)] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = random_binary(500, rng)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    best_n = n

    # Phase 1: Continuous L-BFGS-B polish (90s)
    t1 = min(t0 + 90, deadline - 500)
    best_h, best_c = lbfgs_run(best_h, best_n, t1, max_iters=200)

    # Phase 2: Binary greedy from continuous (120s)
    if time.time() < deadline - 350:
        h_b = to_binary(best_h, best_n, rng)
        t2 = min(time.time() + 60, deadline - 300)
        h_g, c_g = greedy_binary(h_b, best_n, t2, rng)
        if c_g < best_c:
            best_c = c_g
            best_h = h_g
        # Polish binary in continuous
        t2b = min(time.time() + 50, deadline - 250)
        h_p, c_p = lbfgs_run(h_g, best_n, t2b, max_iters=150)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p

    # Phase 3: Fast SA from multiple starts (350s)
    starts = [to_binary(best_h, best_n, rng),
              random_binary(best_n, rng),
              gaussian_binary(best_n, rng),
              fourier_binary(best_n, rng)]
    per_alloc = 85
    for si, s in enumerate(starts):
        if time.time() > deadline - 200:
            break
        alloc = min(per_alloc, deadline - time.time() - 180)
        if alloc < 30:
            break
        h_s, c_s = sa_binary(s, best_n, time.time() + alloc * 0.6, rng)
        # Greedy polish
        t_gp = time.time() + min(alloc * 0.25, 25)
        h_gp, c_gp = greedy_binary(h_s, best_n, t_gp, rng)
        if c_gp < best_c:
            best_c = c_gp
            best_h = h_gp
        # Continuous polish
        t_cp = time.time() + min(alloc * 0.15, 20)
        h_cp, c_cp = lbfgs_run(h_gp, best_n, t_cp, max_iters=100)
        if c_cp < best_c:
            best_c = c_cp
            best_h = h_cp

    # Phase 4: Try other n values (150s)
    for n_try in [250, 375, 625, 750]:
        if time.time() > deadline - 60:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n_try)
        alloc = min(35, (deadline - time.time() - 50) * 0.45)
        if alloc < 15:
            continue
        h_t, c_t = lbfgs_run(h_int, n_try, time.time() + alloc * 0.5, max_iters=100)
        h_tb = to_binary(h_t, n_try, rng)
        t_bs = time.time() + alloc * 0.25
        h_bs, c_bs = sa_binary(h_tb, n_try, t_bs, rng)
        if c_bs < best_c:
            best_c = c_bs
            best_h = h_bs
            best_n = n_try
        if time.time() < deadline - 30:
            t_fp = time.time() + 15
            h_fp, c_fp = lbfgs_run(h_bs, best_n, t_fp, max_iters=80)
            if c_fp < best_c:
                best_c = c_fp
                best_h = h_fp

    # Phase 5: Final intensive SA on best (remaining - 20s)
    if time.time() < deadline - 20:
        h_final = to_binary(best_h, best_n, rng)
        t5 = deadline - 5
        h_f, c_f = sa_binary(h_final, best_n, t5, rng)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
        t5b = deadline - 2
        h_f2, c_f2 = greedy_binary(h_f, best_n, t5b, rng)
        if c_f2 < best_c:
            best_c = c_f2
            best_h = h_f2

    # Final continuous polish if time
    if time.time() < deadline - 2:
        h_fp, c_fp = lbfgs_run(best_h, best_n, deadline - 1, max_iters=80)
        if c_fp < best_c:
            best_c = c_fp
            best_h = h_fp

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
