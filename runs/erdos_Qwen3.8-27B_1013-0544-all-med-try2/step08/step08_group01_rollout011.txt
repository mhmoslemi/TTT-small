import numpy as np
import time
import math
from scipy.optimize import minimize, linprog

def corr_full(h):
    n = len(h)
    if n <= 600:
        return np.correlate(h, 1.0 - h, mode='full')
    N = 1 << (2 * n - 1).bit_length()
    y = np.fft.irfft(np.fft.rfft(h, N) * np.conj(np.fft.rfft(1.0 - h, N)), N)
    out = np.empty(2 * n - 1)
    out[:n] = y[n - 1::-1]
    out[n:] = y[N - 1:N - n:-1]
    return out

def c5_val(h):
    return float(corr_full(h).max() * (2.0 / len(h)))

def project(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    tgt = 0.5 * n
    for _ in range(70):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = (1.0 - h).sum()
            if r < 1e-14:
                break
            h = np.clip(h + (d / r) * (1.0 - h), 0.0, 1.0)
        else:
            r = h.sum()
            if r < 1e-14:
                break
            h = np.clip(h + (d / r) * h, 0.0, 1.0)
    for _ in range(12):
        d = tgt - h.sum()
        if abs(d) < 1e-12:
            break
        idxs = np.flatnonzero((h > 1e-10) & (h < 1.0 - 1e-10))
        if idxs.size == 0:
            break
        room = ((1.0 - h) if d > 0 else h)[idxs]
        s = room.sum()
        if s < 1e-14:
            break
        h[idxs] = np.clip(h[idxs] + d * room / s, 0.0, 1.0)
    return h

def soft_val(c, n, alpha):
    y = c * (2.0 / n)
    z = alpha * y
    zm = z.max()
    return float((zm + np.log(np.sum(np.exp(z - zm)))) / alpha)

def soft_w(c, n, alpha):
    y = c * (2.0 / n)
    z = alpha * y
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def grad_corr(w, h):
    n = len(h)
    g = np.zeros(n)
    for m in np.flatnonzero(w > 1e-14):
        wm = w[m]
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            if nv > 0:
                g[:nv] += wm * (1.0 - h[k:])
                g[k:] += -wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                g[kk:] += wm * (1.0 - h[:nv])
                g[:nv] += -wm * h[kk:]
    return g

def _obj_pen(x, n, alpha, mu):
    c = corr_full(x)
    val = soft_val(c, n, alpha)
    s = x.sum() - 0.5 * n
    return val + mu * s * s

def _jac_pen(x, n, alpha, mu):
    c = corr_full(x)
    w = soft_w(c, n, alpha)
    g = grad_corr(w, x) * (2.0 / n)
    s = x.sum() - 0.5 * n
    return g + 2.0 * mu * s

def lbfgs(h, n, deadline):
    h = project(h)
    best = c5_val(h)
    bounds = [(0.0, 1.0)] * n
    base = max(25, min(90, int(35000 // max(1, n))))
    for alpha, mu in ((100.0, 1.0), (500.0, 10.0), (2500.0, 60.0),
                      (12000.0, 300.0), (60000.0, 1500.0)):
        if deadline - time.time() < 10:
            break
        try:
            res = minimize(_obj_pen, h, jac=_jac_pen, args=(n, alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': base, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': base * 12,
                                    'maxls': 35})
            hn = project(res.x)
            c = c5_val(hn)
            if c < best:
                best = c
                h = hn
            base = min(base + 10, 140)
        except Exception:
            pass
    return h, best

def lag_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            g[:nv] += 1.0 - h[k:]
            g[k:] += -h[:nv]
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            g[kk:] += 1.0 - h[:nv]
            g[:nv] += -h[kk:]
    return g

def lp_direction(h, c, active, step_total, rho):
    n = len(h)
    m = len(active)
    if m == 0:
        return None
    G = np.empty((m, n))
    for r, ix in enumerate(active):
        G[r] = lag_grad(h, int(ix) - (n - 1))
    N = 1 + 2 * n
    cobj = np.zeros(N)
    cobj[0] = 1.0
    cobj[1 + n:1 + 2 * n] = rho
    A = np.zeros((m + 1, N))
    b = np.zeros(m + 1)
    A[:m, 0] = -1.0
    A[:m, 1:1 + n] = G
    b[:m] = -c[active]
    A[m, 1 + n:1 + 2 * n] = 1.0
    b[m] = step_total
    Aeq = np.zeros((1, N))
    Aeq[0, 1:1 + n] = 1.0
    bounds = [(-math.inf, math.inf)]
    for i in range(n):
        bounds.append((-h[i], 1.0 - h[i]))
    for _ in range(n):
        bounds.append((0.0, math.inf))
    try:
        res = linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=np.zeros(1),
                          bounds=bounds)
        except Exception:
            return None
    if not res.success:
        return None
    return res.x[1:1 + n], float(res.x[0])

def lp_improve(h, n, deadline):
    h = project(h)
    c = corr_full(h)
    raw = float(c.max())
    step = max(1.0, 0.015 * n)
    accepts = 0
    for _ in range(80):
        if time.time() > deadline - 5:
            break
        cmax = raw
        tol = max(0.1, 0.025 * cmax)
        cand = np.where(c >= cmax - tol)[0]
        if cand.size == 0:
            cand = np.array([int(np.argmax(c))])
        found = False
        for lim in (35, 20, 10, 5):
            if cand.size > lim:
                act = cand[np.argsort(c[cand])[-lim:]]
            else:
                act = cand
            r = lp_direction(h, c, act, step, 1e-7)
            if r is None:
                continue
            d, tpred = r
            if tpred < cmax - 1e-7:
                found = True
                break
        if not found:
            break
        if np.max(np.abs(d)) < 1e-9:
            break
        ok = False
        for al in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.01, 0.004):
            if time.time() > deadline - 3:
                break
            ht = project(h + al * d, n)
            nr = float(corr_full(ht).max())
            if nr < raw - 1e-9:
                h = ht
                raw = nr
                c = corr_full(h)
                accepts += 1
                ok = True
                break
        if not ok:
            step *= 0.5
            if step < 0.35 or accepts >= 40:
                break
    return h, raw * (2.0 / n)

def pair_basis(h, i, j):
    n = len(h)
    L = 2 * n - 1
    bidx = np.arange(n)
    a = h[i]
    b = h[j]
    A = np.zeros(L)
    Q = np.zeros(L)
    A[n - 1 + (bidx - i)] += 1.0 - h
    A[n - 1 + (bidx - j)] += -(1.0 - h)
    A[n - 1] += -a
    Q[n - 1] += -1.0
    A[n - 1 + (j - i)] += a
    Q[n - 1 + (j - i)] += 1.0
    A[n - 1] += 2.0 - 3.0 * b
    Q[n - 1] += -1.0
    A[n - 1 + (i - j)] += -b
    Q[n - 1 + (i - j)] += 1.0
    mask = np.ones(n, dtype=bool)
    mask[i] = False
    mask[j] = False
    qs = bidx[mask]
    A[n - 1 + (i - qs)] += -h[mask]
    A[n - 1 + (j - qs)] += h[mask]
    return A, Q

def pair_line(h, c, i, j, cur):
    a = h[i]
    b = h[j]
    if a >= 1.0 - 1e-13 or b <= 1e-13:
        return None
    thi = min(1.0 - a, b)
    if thi <= 1e-11:
        return None
    A, Q = pair_basis(h, i, j)
    fr = 10.0 ** (-np.linspace(0.0, 3.2, 30))
    ts = np.unique(fr * thi)
    ts = ts[(ts > 1e-12) & (ts <= thi + 1e-15)]
    if ts.size == 0:
        ts = np.array([thi])
    t2 = ts * ts
    vals = c[None, :] + ts[:, None] * A + t2[:, None] * Q
    mm = vals.max(axis=1)
    bi = int(np.argmin(mm))
    bt = float(ts[bi])
    bestv = float(mm[bi])
    lo = bt * 0.35 if bi > 0 else 0.0
    hi = bt * 1.6 if bi < ts.size - 1 else thi
    lo = max(0.0, lo)
    hi = min(thi, hi)
    if hi - lo > 1e-12 * max(1.0, thi):
        tr = np.linspace(lo, hi, 25)
        mr = (c[None, :] + tr[:, None] * A + tr[:, None] * tr[:, None] * Q).max(axis=1)
        bj = int(np.argmin(mr))
        if float(mr[bj]) < bestv:
            bestv = float(mr[bj])
            bt = float(tr[bj])
    if bestv < cur - 1e-8:
        H = h.copy()
        H[i] = float(np.clip(a + bt, 0.0, 1.0))
        H[j] = float(np.clip(b - bt, 0.0, 1.0))
        C = corr_full(H)
        return H, C, float(C.max())
    return None

def pair_polish(h, n, deadline):
    h = project(h)
    c = corr_full(h)
    raw = float(c.max())
    no = 0
    rng = np.random.RandomState(20240521)
    while time.time() < deadline - 2 and no < 5:
        c = corr_full(h)
        raw = float(c.max())
        cmax = raw
        tol = max(1e-8, 0.012 * cmax)
        cand = np.where(c >= cmax - tol)[0]
        if cand.size == 0:
            cand = np.array([int(np.argmax(c))])
        if cand.size > 220:
            cand = cand[np.argsort(c[cand])[-220:]]
        w = np.zeros(2 * n - 1)
        w[cand] = 1.0 / cand.size
        g = grad_corr(w, h)
        order = np.argsort(g)
        K = max(2, min(8, n // 40 + 1))
        found = False
        for ii in range(K):
            if found:
                break
            i = int(order[ii])
            for jj in range(K):
                j = int(order[-1 - jj])
                if i == j:
                    continue
                if h[i] < 1.0 - 1e-12 and h[j] > 1e-12:
                    r = pair_line(h, c, i, j, raw)
                    if r is not None:
                        h, c, raw = r
                        found = True
                        no = 0
                        break
                if h[j] < 1.0 - 1e-12 and h[i] > 1e-12:
                    r = pair_line(h, c, j, i, raw)
                    if r is not None:
                        h, c, raw = r
                        found = True
                        no = 0
                        break
        if not found:
            for _ in range(45):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                if h[i] < 1.0 - 1e-12 and h[j] > 1e-12:
                    r = pair_line(h, c, i, j, raw)
                    if r is not None:
                        h, c, raw = r
                        found = True
                        no = 0
                        break
                if h[j] < 1.0 - 1e-12 and h[i] > 1e-12:
                    r = pair_line(h, c, j, i, raw)
                    if r is not None:
                        h, c, raw = r
                        found = True
                        no = 0
                        break
        if not found:
            no += 1
    h = project(h)
    return h, c5_val(h)

def golden_start(n):
    h = np.zeros(n)
    m = n // 2
    phi = (1.0 + math.sqrt(5.0)) / 2.0
    pos = np.unique(np.floor(np.arange(m) * phi).astype(int) % n)
    h[pos] = 1.0
    if pos.size < m:
        rem = np.setdiff1d(np.arange(n), pos)
        h[rem[:m - pos.size]] = 1.0
    return project(h)

def sturm_start(n):
    x = np.mod(np.arange(n) * 0.6180339887498949, 1.0)
    h = (x < 0.5).astype(float)
    return project(h)

def keep_best(h, c, best_h, best_c, best_n):
    if c < best_c - 1e-15:
        return np.asarray(h, dtype=float).copy(), float(c), len(h)
    return best_h, best_c, best_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = max(60.0, float(budget_s))
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        n = len(h0)
    except Exception:
        h0 = golden_start(500)
        n = 500
    if n % 2:
        n -= 1
        h0 = h0[:n]
    if n < 80:
        h0 = golden_start(500)
        n = 500

    h = project(h0)
    best_h, best_c, best_n = h.copy(), c5_val(h), n
    if deadline - t0 <= 25:
        best_h = project(best_h)
        return best_h, c5_val(best_h), len(best_h)

    e = min(t0 + 0.31 * budget_s, deadline - 0.43 * budget_s)
    if e > time.time() + 25 and best_n <= 1100:
        h, c = lbfgs(best_h, best_n, e)
        best_h, best_c, best_n = keep_best(h, c, best_h, best_c, best_n)

    e = min(time.time() + 0.13 * budget_s, deadline - 0.30 * budget_s)
    if e > time.time() + 15 and best_n <= 1000:
        h, c = lp_improve(best_h, best_n, e)
        best_h, best_c, best_n = keep_best(h, c, best_h, best_c, best_n)

    e = min(time.time() + 0.10 * budget_s, deadline - 0.22 * budget_s)
    if e > time.time() + 12 and best_n <= 1100:
        h, c = pair_polish(best_h, best_n, e)
        best_h, best_c, best_n = keep_best(h, c, best_h, best_c, best_n)

    for n2 in (512, 768, 1024):
        if time.time() > deadline - 250:
            break
        if n2 == best_n:
            continue
        alloc = min(0.20 * budget_s, deadline - time.time() - 170)
        if alloc < 45:
            break
        xo = (np.arange(best_n) + 0.5) * (2.0 / best_n)
        xn = (np.arange(n2) + 0.5) * (2.0 / n2)
        hn = project(np.interp(xn, xo, best_h, left=best_h[0], right=best_h[-1]))
        e1 = min(time.time() + 0.50 * alloc, deadline - 130)
        hn, cn = lbfgs(hn, n2, e1)
        e2 = min(time.time() + 0.25 * alloc, deadline - 60)
        hn, cn = lp_improve(hn, n2, e2)
        hn, cn = pair_polish(hn, n2, deadline - 8)
        best_h, best_c, best_n = keep_best(hn, cn, best_h, best_c, best_n)

    if time.time() < deadline - 190:
        n_s = best_n if best_n <= 800 else 600
        starts = [sturm_start(n_s), golden_start(n_s)]
        hb = np.zeros(n_s)
        rb = np.zeros(n_s)
        rb.flat[:n_s // 4] = 1.0
        rb.flat[3 * n_s // 4:] = 1.0
        rb[:] = (hb + rb) / 2.0
        rb.flat[:n_s // 4] = 0.0
        starts.append(project(rb))
        starts.append(project(golden_start(n_s) + 0.15 * np.sin(np.linspace(0, 8.0, n_s))))
        for s in starts:
            if time.time() > deadline - 120:
                break
            alloc = min(95.0, deadline - time.time() - 80)
            if alloc < 35:
                break
            es = project(s)
            e1 = min(time.time() + 0.55 * alloc, deadline - 70)
            es, cs = lbfgs(es, n_s, e1)
            e2 = min(time.time() + 0.22 * alloc, deadline - 35)
            es, cs = lp_improve(es, n_s, e2)
            es, cs = pair_polish(es, n_s, deadline - 8)
            best_h, best_c, best_n = keep_best(es, cs, best_h, best_c, best_n)

    if time.time() < deadline - 35 and best_n <= 1000:
        h, c = lp_improve(best_h, best_n, deadline - 18)
        best_h, best_c, best_n = keep_best(h, c, best_h, best_c, best_n)

    if time.time() < deadline - 6:
        h, c = pair_polish(best_h, best_n, deadline - 2)
        best_h, best_c, best_n = keep_best(h, c, best_h, best_c, best_n)

    best_h = project(best_h)
    return best_h, c5_val(best_h), len(best_h)
