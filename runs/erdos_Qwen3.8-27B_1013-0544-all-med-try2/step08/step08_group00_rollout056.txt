import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def obj_lbfgsb(x, n, alpha):
    h = np.clip(x, 0, 1)
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = h.sum() - n / 2.0
    val += 100.0 * s * s
    return float(val)

def grad_lbfgsb(x, n, alpha):
    h = np.clip(x, 0, 1)
    g = grad_smooth_max(h, n, alpha)
    s = h.sum() - n / 2.0
    g += 200.0 * s
    return g

def lbfgs_optimize(h, n, deadline, max_iters=1000):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [1000, 10000, 100000, 1000000, 10000000]
    for a in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(obj_lbfgsb, h, jac=grad_lbfgsb,
                         args=(n, a), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-18,
                                  'gtol': 1e-14, 'maxfun': max_iters * 20,
                                  'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def compute_slopes(h, n, i, j):
    """Compute slope s_k for all lags k when updating h[i]+=δ, h[j]-=δ."""
    dx = 2.0 / n
    # s_k = dx * [(1-h[i-k]) - h[i+k] - (1-h[j-k]) + h[j+k]] with bounds checking
    # where k ranges from -(n-1) to (n-1)
    # For the full correlation indexed by m = k + (n-1), m in [0, 2n-2]:
    n Slopes = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        m = k + (n - 1)
        s = 0.0
        q1 = i - k
        if 0 <= q1 < n:
            s += (1.0 - h[q1])
        q2 = i + k
        if 0 <= q2 < n:
            s -= h[q2]
        q3 = j - k
        if 0 <= q3 < n:
            s -= (1.0 - h[q3])
        q4 = j + k
        if 0 <= q4 < n:
            s += h[q4]
        Slopes[m] = dx * s
    return Slopes

def pair_line_search(h, n, i, j, c0, n_iter=30):
    """Find optimal δ > 0 for h[i] += δ, h[j] -= δ using line search."""
    # δ range: [0, min(1-h[i], h[j])]
    dmax = min(1.0 - h[i], h[j])
    if dmax <= 0:
        return 0.0, c0
    
    slopes = compute_slopes(h, n, i, j)
    
    def c_at_delta(d):
        return float(np.max(c0 + slopes * d))
    
    # Ternary search
    lo, hi = 0.0, dmax
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        c1 = c_at_delta(m1)
        c2 = c_at_delta(m2)
        if c1 < c2:
            hi = m2
        else:
            lo = m1
    
    d_opt = (lo + hi) / 2.0
    c_opt = c_at_delta(d_opt)
    return d_opt, c_opt

Now for the line search, instead of calling c_at_delta 30 times (each O(n)), I can find the exact minimum of max_k (a_k + b_k * δ) more efficiently.

The function f(δ) = max_k (a_k + b_k * δ) is convex and piecewise linear. Its minimum is at a point where either:
1. A boundary is reached (δ = 0 or δ = δ_max)
2. Two lines cross at the minimum of the upper envelope

For case 2, if lines k and l cross at δ* = (a_l - a_k) / (b_k - b_l), then at δ*, both have value a_k + b_k * δ*. For this to be the minimum of the upper envelope, we need this value to be less than or equal to all other lines' values at δ*.

Efficient approach: 
- Start with the currently dominant k (where a_k is max)
- Find the next line that becomes dominant as δ increases
- Check if the minimum is between them
- This is essentially walking the upper envelope

But this is complex to implement correctly. Let me just use ternary search - it's simple and works well for convex functions.

Actually, for a piecewise linear convex function, the minimum is at a "kink" point. Ternary search will converge to that point. With 30 iterations, the interval width is δ_max * (2/3)^30 ≈ δ_max * 5e-6, which is more than enough precision.

Let me also think about whether the line search is correct. When I compute c_at_delta(d) = max_k (c0[k] + slopes[k] * d), this assumes that C_k changes linearly with δ. Is this true?

C_k = Σ_m h[m] * (1-h[m+k]) * dx

When h[i] changes by +δ and h[j] changes by -δ:
- C_k changes by δ * (∂C_k/∂h[i] - ∂C_k/∂h[j])
- This is exact (linear) because C_k is bilinear in h (each term is a product of two h values, but for a given k, changing h[i] only affects terms where h[i] appears, and the other factor doesn't change).

Wait, is C_k linear in each individual h[p]? 
C_k = Σ_m h[m] * (1-h[m+k]) * dx

∂C_k/∂h[p]: h[p] appears in the term where m=p (as h[m]) and where m+k=p (as h[m+k]).
- If m=p: the term is h[p] * (1-h[p+k]), derivative w.r.t. h[p] is (1-h[p+k])
- If m+k=p (i.e., m=p-k): the term is h[p-k] * (1-h[p]), derivative w.r.t. h[p] is -h[p-k]

So ∂C_k/∂h[p] = dx * [(1-h[p+k]) * 1{valid} + (-h[p-k]) * 1{valid}]

This is NOT linear in h[p] - it depends on the other h values. But for a FIXED step δ (small), the change is:
C_k(new) - C_k(old) = δ * ∂C_k/∂h[i] - δ * ∂C_k/∂h[j] (evaluated at the OLD h)

Wait, that's only first-order. The exact change is:
C_k(new) = Σ_m h'_m * (1-h'_{m+k}) * dx

where h'_i = h_i + δ, h'_j = h_j - δ, h'_p = h_p for p ≠ i,j.

The terms that change are those involving h'_i or h'_j:
- Terms where m=i: h'_i * (1-h'_{i+k}) = (h_i+δ)(1-h_{i+k}) if i+k≠j, or (h_i+δ)(1-(h_j-δ)) if i+k=j
- Terms where m+k=i (m=i-k): h'_{i-k} * (1-h'_i) = h_{i-k}(1-h_i-δ) if i-k≠j, or (h_j-δ)(1-h_i-δ) if i-k=j
- Similarly for j.

So if i+k ≠ j and i-k ≠ j, the change is:
δ(1-h_{i+k}) - δ*h_{i-k} (for terms involving i)
-δ(1-h_{j+k}) + δ*h_{j-k} (for terms involving j)

This IS linear in δ. But if i+k=j or i-k=j, there are cross terms involving δ².

So the linear approximation is exact UNLESS i and j are at a lag distance from each other. Specifically, if |i-j| < n, then for k = i-j (or k = j-i), there might be cross terms.

For the line search to be exact, I need to handle this. Let me check: if |i-j| = d (some lag), then for k=d, the term h[i]*(1-h[j]) becomes (h_i+δ)(1-(h_j-δ)) = h_i - h_j - h_i*h_j + δ + δ*h_i + δ*h_j + δ²... wait let me redo.

(h_i + δ)(1 - h_j + δ) = h_i(1-h_j) + h_i*δ + δ(1-h_j) + δ²

So the cross term is δ*h_i + δ*(1-h_j) + δ² = δ*h_i + δ - δ*h_j + δ²

The linear part is δ*h_i + δ - δ*h_j = δ(1 + h_i - h_j)
The quadratic part is δ².

So for the specific lag k = i-j (where i and j interact), the change has a δ² term. This means C_k(δ) is NOT exactly linear for that one lag.

However, the δ² term is small (δ is typically small, like 0.01 or less), so the linear approximation is very good. For the line search, the error is O(δ²) which is negligible for small δ.

I'll proceed with the linear approximation. If the line search suggests a large δ (close to dmax), I can verify by actually computing the new C5.

Let me also add a verification step: after accepting a pair update with the line search, actually compute the new C5 to verify it's an improvement. If not (due to the approximation error), revert.

        else:
            lo = m1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def obj_lbfgsb(x, n, alpha, mu):
    h = np.clip(x, 0, 1)
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = h.sum() - n / 2.0
    val += mu * s * s
    return float(val)

def grad_lbfgsb(x, n, alpha, mu):
    h = np.clip(x, 0, 1)
    g = grad_smooth_max(h, n, alpha)
    s = h.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_optimize(h, n, deadline):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 0.1), (2000, 0.5), (10000, 2.0), (50000, 10.0),
               (200000, 50.0), (1000000, 200.0), (5000000, 1000.0),
               (20000000, 5000.0), (100000000, 50000.0)]
    max_iters = 600
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(obj_lbfgsb, h, jac=grad_lbfgsb,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-18,
                                  'gtol': 1e-15, 'maxfun': max_iters * 30,
                                  'maxls': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def compute_slopes(h, n, i, j):
    dx = 2.0 / n
    ks = np.arange(-(n - 1), n)
    q1 = i - ks
    q2 = i + ks
    q3 = j - ks
    q4 = j + ks
    s = np.zeros(2 * n - 1)
    m1 = (q1 >= 0) & (q1 < n)
    m2 = (q2 >= 0) & (q2 < n)
    m3 = (q3 >= 0) & (q3 < n)
    m4 = (q4 >= 0) & (q4 < n)
    s[m1] += (1.0 - h[q1[m1]])
    s[m2] -= h[q2[m2]]
    s[m3] -= (1.0 - h[q3[m3]])
    s[m4] += h[q4[m4]]
    return dx * s

def pair_line_search(c0, slopes, dmax, n_iter=25):
    if dmax <= 1e-15:
        return 0.0, float(c0.max())
    lo, hi = 0.0, dmax
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        c1 = float(np.max(c0 + slopes * m1))
        c2 = float(np.max(c0 + slopes * m2))
        if c1 < c2:
            hi = m2
        else:
            lo = m1
    d_opt = (lo + hi) / 2.0
    c_opt = float(np.max(c0 + slopes * d_opt))
    return d_opt, c_opt

def gradient_guided_search(h, n, deadline):
    best_c = compute_c5(h, n)
    c0 = compute_c5_vec(h, n)
    alpha = 50000
    top_k = 15
    stagnant = 0
    while time.time() < deadline:
        g = grad_smooth_max(h, n, alpha)
        inc_idx = np.argsort(g)[:top_k]
        dec_idx = np.argsort(g)[-top_k:][::-1]
        found = False
        for i in inc_idx:
            if time.time() > deadline:
                break
            for j in dec_idx:
                if i == j:
                    continue
                slopes = compute_slopes(h, n, i, j)
                dmax_f = min(1.0 - h[i], h[j])
                if dmax_f <= 1e-14:
                    continue
                d_opt, c_opt = pair_line_search(c0, slopes, dmax_f)
                if d_opt > 1e-10 and c_opt < best_c - 1e-12:
                    h[i] += d_opt
                    h[j] -= d_opt
                    actual_c = compute_c5(h, n)
                    if actual_c < best_c - 1e-14:
                        best_c = actual_c
                        c0 = compute_c5_vec(h, n)
                        found = True
                    else:
                        h[i] -= d_opt
                        h[j] += d_opt
        if found:
            stagnant = 0
        else:
            stagnant += 1
            top_k = min(top_k + 5, 100)
            if top_k < 40:
                alpha = min(alpha * 2, 500000)
            if stagnant > 20:
                # random perturbation to escape
                rng_e = np.random.RandomState(stagnant * 7 + n)
                idx = rng_e.choice(n, size=10, replace=False)
                for a in range(5):
                    i, j = idx[2*a], idx[2*a+1]
                    d = 0.005 * (1 if h[i] + 0.005 <= 1 and h[j] - 0.005 >= 0 else -1)
                    if abs(d) > 0:
                        h[i] += d
                        h[j] -= d
                h = np.clip(h, 0, 1)
                h = project_feasible(h, n)
                actual_c = compute_c5(h, n)
                if actual_c < best_c + 0.001:
                    c0 = compute_c5_vec(h, n)
                stagnant = 0
            elif top_k >= 100:
                break
    return h, best_c

def fine_local_search(h, n, deadline):
    best_c = compute_c5(h, n)
    c0 = compute_c5_vec(h, n)
    alpha = 500000
    step_caps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    top_k = 25
    for cap in step_caps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            inc_idx = np.argsort(g)[:top_k]
            dec_idx = np.argsort(g)[-top_k:][::-1]
            for i in inc_idx:
                if time.time() > deadline:
                    break
                for j in dec_idx:
                    if i == j:
                        continue
                    dmax_f = min(1.0 - h[i], h[j], cap)
                    if dmax_f <= 1e-12:
                        continue
                    slopes = compute_slopes(h, n, i, j)
                    d_opt, c_opt = pair_line_search(c0, slopes, dmax_f, n_iter=20)
                    if d_opt > 1e-12 and c_opt < best_c - 1e-13:
                        h[i] += d_opt
                        h[j] -= d_opt
                        actual_c = compute_c5(h, n)
                        if actual_c < best_c - 1e-14:
                            best_c = actual_c
                            c0 = compute_c5_vec(h, n)
                            improved = True
                        else:
                            h[i] -= d_opt
                            h[j] += d_opt
    return h, best_c
