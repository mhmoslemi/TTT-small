import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_s_list(h, n):
    return np.correlate(h, 1.0 - h, mode='full').tolist()

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_from_probs(h, n):
    target = n // 2
    order = np.argsort(h)[::-1]
    b = np.zeros(n, dtype=np.uint8)
    idx = np.asarray(order[:target], dtype=int)
    b[idx] = 1

def init_binary_state(bits, n):
    off = n - 1
    C = np.correlate(bits.astype(float), 1.0 - bits.astype(float), mode='full').astype(int).tolist()
    ones = [int(i) for i in range(n) if bits[i] == 1]

    return C, ones, zeros, off

def swap_in_list(lst, old, new):
    for i, v in enumerate(lst):

            return
    if old in lst:

def smooth_max_obj(x,n,alpha,mu):
    c = np.correlate(x,1.0-x,mode='full')
    m=float(c.max())
    sh=alpha*(c-m)
    s=sh - sh.max()
    val=m + (s.max()+np.log(np.sum(np.exp(s))))/alpha
    if mu:
       val += mu*(x.sum()-0.5*n)**2
    return float(val)

def smooth_max_grad(x,n,alpha,mu):
    c = np.correlate(x,1.0-x,mode='full')
    m=c.max()
    sh=alpha*(c-m)
    s=sh - sh.max()
    w=np.exp(s); w/=w.sum()
    dx=2.0/n
    g=np.zeros(n)
    active=np.nonzero(w > 1e-10)[0]
    off=n-1
    for mi in active:
       wm = float(w[mi]) * dx
       d = off - int(mi)
       if d >= 0:
          nv = n - d
          g[:nv] += wm*(1.0 - x[d:])
          g[d:] += wm*(-x[:nv])
       else:
          kk = -d; nv = n - kk
          g[kk:] += wm*(1.0 - x[:nv])
          g[:nv] += wm*(-x[kk:])
    if mu:
       g += 2.0*mu*(x.sum()-0.5*n)
    return g

import numpy as np
import time
import math
from scipy.optimize import minimize

def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    tgt = 0.5 * n
    for _ in range(500):
        d = tgt - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0.0:
            r = (1.0 - h).sum()
            if r > 0.0:
                h += (d / r) * (1.0 - h)
        else:
            s = h.sum()
            if s > 0.0:
                h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def binary_from_probs(h, n):
    b = np.zeros(n, dtype=np.uint8)
    b[np.argsort(h)[::-1][:n // 2]] = 1
    return b

def init_binary_state(bits, n):
    off = n - 1
    C = np.correlate(bits.astype(float), 1.0 - bits.astype(float), mode="full").astype(int).tolist()
    ones = [i for i in range(n) if bits[i] == 1]
    zeros = [i for i in range(n) if bits[i] == 0]
    return C, ones, zeros, off

def swap_in_list(a, old, new):
    a[a.index(old)] = new

def apply_binary_move(C, ones, zeros, p, q, off, sgn):
    for s in ones:
        if s != p:
            C[off + s - p] += sgn
            C[off + s - q] -= sgn
    for t in zeros:
        if t != q:
            C[off + q - t] += sgn
            C[off + p - t] -= sgn
    C[off + q - p] += sgn
    C[off + p - q] -= sgn

def continuous_tail(S, m, tol=0.75):
    th = m - tol
    s = 0.0
    for v in S:
        if v > th:
            s += v - th
    return s

def apply_continuous_move(S, h, n, a, b, d, off, sgn):
    d *= sgn
    if d == 0.0:
        return
    ha = h[a]
    hb = h[b]
    dd = d * d
    S[off] += d * (1.0 - 2.0 * ha) - dd
    S[off] += -d * (1.0 - 2.0 * hb) - dd
    S[off + b - a] += -d * (1.0 - ha) - d * hb + dd
    S[off + a - b] += d * (1.0 - hb) + d * ha + dd
    for t in range(n):
        if t == a or t == b:
            continue
        ht = h[t]
        om = 1.0 - ht
        S[off + a - t] += d * om
        S[off + t - a] += -d * ht
        S[off + b - t] += -d * om
        S[off + t - b] += d * ht

    rng = np.random.RandomState(seed + 17)
    A = 12
    B = 10
    while time.time() < deadline:

        ct[off] = -1000000
        top = np.argsort(ct)[::-1][:A]
        found = False
        for m in top:
            d = off - int(m)
            if d == 0:
                continue
            kk = abs(d)
            nv = n - kk
            if d >= 0:
                sa = bits[:nv]
                ta = bits[kk:]
            else:
                sa = bits[kk:]
                ta = bits[:nv]
            idx = np.where((sa == 1) & (ta == 0))[0]

                tgt = cur[:nv]
            cap = np.minimum(src, 1.0 - tgt)
            term = src * (1.0 - tgt)
            idx = np.where(cap > 1e-7)[0]
            if idx.size == 0:
                continue
            if idx.size > B:
                part = idx[np.argpartition(term[idx], -B)[-B:]]
                part = part[np.argsort(term[part])[::-1]]
            else:
                part = idx[np.argsort(term[idx])[::-1]]
            for i in part:
                if d >= 0:
                    s = int(i)
                    t = int(i + d)
                else:
                    s = int(i + kk)
                    t = int(i)
                cmv = float(cap[i])
                if cmv <= 1e-9:
                    continue
                for f in fr:
                    dl = cmv * f
                    if dl <= 1e-10:
                        continue
                    apply_continuous_move(curS, cur, n, t, s, dl, off, 1.0)
                    nm = max(curS)
                    if nm < cm - 1e-12:
                        nt = continuous_tail(curS, nm)
                        accept = True
                    elif nm <= cm + 1e-12:
                        nt = continuous_tail(curS, nm)
                        accept = nt < ct - 1e-12
                    else:
                        nt = 0.0
                        accept = False
                    if accept:
                        cur[t] += dl
                        cur[s] -= dl
                        if cur[t] > 1.0:
                            cur[t] = 1.0
                        if cur[s] < 0.0:
                            cur[s] = 0.0
                        cm, ct = nm, nt
                        acc += 1
                        found = True
                        break
                    apply_continuous_move(curS, cur, n, t, s, dl, off, -1.0)
                if found:
                    break
            if found:
                break
        if not found:
            break
        if acc % 800 == 0:
            cur = project_feasible(cur, n)
            curS = np.correlate(cur, 1.0 - cur, mode="full").tolist()
            cm = max(curS)
            ct = continuous_tail(curS, cm)
    cur = project_feasible(cur, n)
    return cur, compute_c5(cur, n)

def smooth_obj(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    sh = alpha * (c - m)
    s = sh - sh.max()
    val = m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha
    if mu:
        val += mu * (x.sum() - 0.5 * n) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    sh = alpha * (c - m)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for mi in np.nonzero(w > 1e-10)[0]:
        wm = float(w[mi]) * dx
        d = off - int(mi)
        if d >= 0:
            nv = n - d
            g[:nv] += wm * (1.0 - x[d:])
            g[d:] += wm * (-x[:nv])
        else:
            kk = -d
            nv = n - kk
            g[kk:] += wm * (1.0 - x[:nv])
            g[:nv] += wm * (-x[kk:])
    if mu:
        g += 2.0 * mu * (x.sum() - 0.5 * n)
    return g

def lbfgs_phase(h0, n, deadline):
    h = project_feasible(h0, n)
    best = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = ((200, 1.0), (1000, 5.0), (5000, 25.0), (20000, 100.0), (80000, 400.0))
    for alpha, mu in configs:
        if time.time() > deadline - 3.0:
            break
        try:
            r = minimize(
                smooth_obj, h, jac=smooth_grad, args=(n, alpha, mu),
                method="L-BFGS-B", bounds=bounds,
                options={"maxiter": 120, "ftol": 1e-15, "gtol": 1e-12, "maxls": 30}
            )
            h2 = project_feasible(r.x, n)
            c2 = compute_c5(h2, n)
            if c2 < best:
                h, best = h2, c2
        except Exception:
            pass
    return h, best

def merge_records(h, c, bh, bc, sh, sc):
    if c < bc - 1e-12:
        bh, bc = h, c
    if c <= sc + 1e-4:
        sh, sc = h, c
    return bh, bc, sh, sc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 10.0
    if deadline <= t0 + 10.0:
        deadline = t0 + 10.0
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(240, 0.5)
    n = len(h0)
    if n < 20:
        n = 240
        h0 = np.full(n, 0.5)
    if n % 2:
        n -= 1
        h0 = h0[:n]
    h0 = project_feasible(h0, n)

    best_h = h0
    best_c = compute_c5(h0, n)
    search_h = h0.copy()
    search_c = best_c

    t = time.time()
    if t < deadline - 360.0:
        te = min(t + 220.0, deadline - 360.0)
        if te > t:
            h, c = continuous_targeted_search(search_h, n, te)
            best_h, best_c, search_h, search_c = merge_records(h, c, best_h, best_c, search_h, search_c)

    t = time.time()
    if t < deadline - 260.0:
        te = min(t + 120.0, deadline - 260.0)
        if te > t:
            start = search_h if search_c <= best_c + 0.02 else best_h
            b = binary_from_probs(start, n)
            h, c = binary_targeted_search(b, n, te, seed=seed + 1)
            best_h, best_c, search_h, search_c = merge_records(h, c, best_h, best_c, search_h, search_c)

    t = time.time()
    if t < deadline - 210.0:
        te = min(t + 110.0, deadline - 210.0)
        if te > t:
            start = search_h if search_c <= best_c + 0.02 else best_h
            h, c = continuous_targeted_search(start, n, te)
            best_h, best_c, search_h, search_c = merge_records(h, c, best_h, best_c, search_h, search_c)

    t = time.time()
    if t < deadline - 100.0:
        te = min(t + 150.0, deadline - 100.0)
        if te > t:
            h, c = lbfgs_phase(search_h, n, te)
            best_h, best_c, search_h, search_c = merge_records(h, c, best_h, best_c, search_h, search_c)

    mode = 0
    while time.time() < deadline - 8.0:
        rem = deadline - time.time()
        if rem < 25.0:
            break
        te = time.time() + min(45.0, rem - 6.0)
        start = search_h if search_c <= best_c + 0.01 else best_h
        if mode == 0:
            h, c = continuous_targeted_search(start, n, te)
        else:
            b = binary_from_probs(start, n)
            h, c = binary_targeted_search(b, n, te, seed=seed + 101 + mode)
        best_h, best_c, search_h, search_c = merge_records(h, c, best_h, best_c, search_h, search_c)
        mode = 1 - mode

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    search_h = project_feasible(search_h, n)
    sc = compute_c5(search_h, n)
    if sc < best_c:
        best_h, best_c = search_h, sc
    return best_h, best_c, n
