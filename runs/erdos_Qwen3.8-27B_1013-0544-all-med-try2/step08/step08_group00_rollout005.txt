import numpy as np
from scipy.optimize import minimize
import time

def c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_val(h, n):
    return float(np.max(c5_vec(h, n)))

def project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(x, n, alpha):
    c = c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dy = 2.0 / n
    grad = np.zeros(n)
    for idx in np.where(w > 1e-15)[0]:
        k = idx - (n - 1)
        wm = w[idx] * dy
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:k+nv])
            grad[k:k+nv] += -wm * x[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:kk+nv] += wm * (1.0 - x[:nv])
            grad[:nv] += -wm * x[kk:kk+nv]
    return grad

def lbfgs_run(h, n, deadline, alphas, max_iters=800):
    best_c = c5_val(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_max_obj, best_h, jac=lambda x: smooth_max_grad(x, n, a),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': max_iters, 'ftol': 1e-20, 'gtol': 1e-18,
                                    'maxfun': max_iters * 100, 'maxls': 200})
            h_new = project(res.x, n)
            c_new = c5_val(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def nesterov_smooth(h_init, n, deadline, alpha, lr_init=1e-4):
    h = h_init.copy().astype(float)
    y = h.copy().astype(float)
    t = 1.0
    lr = lr_init
    best_h = h.copy()
    best_c = c5_val(h, n)
    stall = 0
    while time.time() < deadline:
        gy = smooth_max_grad(y, n, alpha)
        h_new = project(y - lr * gy, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = h_new + momentum * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        c = c5_val(h, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 1.0 / alpha)
        else:
            stall += 1
            if stall > 50:
                lr *= 0.5
                stall = 0
                if lr < 1e-10:
                    break
        t = t_new
    return best_h, best_c

def anti_periodic_full(h_half, m):
    h = np.zeros(2 * m)
    h[:m] = h_half
    h[m:] = 1.0 - h_half
    return h

def anti_periodic_c5_val(h_half, m):
    h_full = anti_periodic_full(h_half, m)

def anti_periodic_obj(x, m, alpha):
    h_full = anti_periodic_full(x, m)

def anti_periodic_grad(x, m, alpha):
    h_full = anti_periodic_full(x, m)
    g_full = smooth_max_grad(h_full, 2 * m, alpha)
    return g_full[:m] - g_full[m:]

def optimize_antiperiodic(h_half, m, deadline, alphas, max_iters=600):

def active_set_refine(h, n, deadline, alpha=1.0e7, k_active=10):
    best_c = c5_val(h, n)
    for _ in range(200):
        if time.time() > deadline:

        c = c5_vec(h, n)
        c_max = c.max()
        # Find active lags
        active = np.where(c > c_max - 0.005)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        active = active[:k_active]
        # Average gradient over active lags only
        dy = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += dy * (1.0 - h[k:k+nv])
                grad[k:k+nv] += -dy * h[:nv]
            else:
                kk = -k
                nv = n - kk
                grad[kk:kk+nv] += dy * (1.0 - h[:nv])
                grad[:nv] += -dy * h[kk:kk+nv]
        grad /= len(active)
        # Line search
        best_step = 0
        best_c_trial = best_c
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_trial = project(h - step * grad, n)
            c_trial = c5_val(h_trial, n)

                best_c_trial = c_trial
                best_step = step
                h = h_trial

        if best_c_trial < best_c:

    return h, best_c

def fine_pair_search(h, n, deadline, k_top=30):
    best_c = c5_val(h, n)
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    for d in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            # Use gradient to pick most promising pairs
            c = c5_vec(h, n)
            active = np.where(c > c.max() - 0.01)[0]
            dy = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[:nv] += dy * (1.0 - h[k:k+nv])
                    grad[k:k+nv] += -dy * h[:nv]
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:kk+nv] += dy * (1.0 - h[:nv])
                    grad[:nv] += -dy * h[kk:kk+nv]
            # Decrease h where grad is most positive, increase where most negative
            up = np.argsort(grad)[-k_top:]
            down = np.argsort(grad)[:k_top]
            for i in down:
                if time.time() > deadline:
                    return h, best_c
                for j in up:
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c_new = c5_val(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
    return h, best_c
