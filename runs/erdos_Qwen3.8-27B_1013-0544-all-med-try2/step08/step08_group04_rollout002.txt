import numpy as np

def compute_c_vec(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c_vec(h, n)))

def smooth_value_weight(c, alpha):
    cmax = float(np.max(c))
    z = float(alpha) * (c - cmax)
    zmax = float(np.max(z))
    e = np.exp(z - zmax)
    s = float(np.sum(e))
    f = cmax + (zmax + float(np.log(s))) / float(alpha)
    w = e / s
    return f, w

def grad_from_w(h, n, w):
    dx = 2.0 / float(n)
    A = np.zeros(n)
    B = np.zeros(n)
    if n > 1:
        A[1:] = w[n:2*n-1]
        B[1:] = w[n-2::-1]
    W = A + B
    cump = np.cumsum(A)
    cumn = np.cumsum(B)
    const_left = cump
    const_right = cumn[n-1::-1]
    crossL = np.convolve(h, W, mode='full')[:n]
    crossR = np.convolve(h, W[::-1], mode='full')[n-1:2*n-1]
    w0 = float(w[n-1])
    return dx * (w0 * (1.0 - 2.0*h) + const_left + const_right - crossL - crossR)

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = 0.5 * float(n)
    diff = target - float(h.sum())
    if abs(diff) < 1e-7:
        if abs(diff) > 1e-15:
            m = (h > 1e-10) & (h < 1.0 - 1e-10)
            if m.any(): h[m] += diff / m.sum()
            else:
                i = int(np.argmin(h) if diff > 0 else np.argmax(h))
                h[i] += diff
        return np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(h + mid, 0.0, 1.0))) < target: lo = mid
        else: hi = mid
    h = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    for _ in range(8):
        diff = target - float(h.sum())
        if abs(diff) <= 1e-13: break
        m = (h > 0.0) & (h < 1.0)
        if m.any(): h[m] += diff / m.sum()
        else:
            i = int(np.argmax(h) if diff < 0 else np.argmin(h))
            h[i] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize, linprog

def compute_c_vec(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c_vec(h, n)))

def smooth_value_weight(c, alpha):
    cmax = float(np.max(c))
    z = float(alpha) * (c - cmax)
    zmax = float(np.max(z))
    e = np.exp(z - zmax)
    s = float(np.sum(e))
    f = cmax + (zmax + float(np.log(s))) / float(alpha)
    return f, e / s

def grad_from_w(h, n, w):
    dx = 2.0 / float(n)
    A = np.zeros(n)
    B = np.zeros(n)
    if n > 1:
        A[1:] = w[n:2 * n - 1]
        B[1:] = w[n - 2::-1]
    W = A + B
    const_left = np.cumsum(A)
    const_right = np.cumsum(B)[n - 1::-1]
    crossL = np.convolve(h, W, mode='full')[:n]
    crossR = np.convolve(h, W[::-1], mode='full')[n - 1:2 * n - 1]
    w0 = float(w[n - 1])
    return dx * (w0 * (1.0 - 2.0 * h) + const_left + const_right - crossL - crossR)

def smooth_obj(x, n, alpha):
    c = compute_c_vec(x, n)
    f, _ = smooth_value_weight(c, alpha)
    return float(f)

def smooth_jac(x, n, alpha):
    c = compute_c_vec(x, n)
    _, w = smooth_value_weight(c, alpha)
    return grad_from_w(x, n, w)

def aug_obj(x, n, alpha, mu, lam):

    f, _ = smooth_value_weight(c, alpha)
    diff = float(np.sum(x) - 0.5 * n)
    return float(f + lam * diff + 0.5 * mu * diff * diff)

def grad_lag_row(h, n, m):
    dx = 2.0 / float(n)
    k = m - (n - 1)

            if cs < c_best:
                h_best, c_best, n_best = hs.copy(), cs, ns
    now = time.time()
    if now < deadline - 30.0:
        h_f, c_f = lp_refine(h_best, n_best, min(deadline - 10.0, now + 70.0))
        if c_f < c_best:
            h_best, c_best = h_f, c_f
    now = time.time()
    if now < deadline - 5.0:
        h_f, c_f = top_pair_refine(h_best, n_best, deadline - 2.0)
        if c_f < c_best:
            h_best, c_best = h_f, c_f
    h_best = project_feasible(h_best, n_best)
    c_best = compute_c5(h_best, n_best)
    h_round = project_feasible(np.round(h_best, 6), n_best)
    c_round = compute_c5(h_round, n_best)
    if c_round < c_best:
        h_best, c_best = h_round, c_round
    return h_best, float(c_best), int(n_best)
