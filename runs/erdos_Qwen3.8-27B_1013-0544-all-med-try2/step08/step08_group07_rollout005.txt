    corr = np.correlate(x, 1.0 - x, mode='full')
    v = np.maximum(0.0, corr - C_dx)
    if not v.any():
        return np.zeros(n)
    # m_offset: corr[m+n-1] corresponds to shift m
    # corr[m] = sum_i x[i]*(1-x[i+m]) for valid i, i+m
    # d(corr[m])/dx[i] = (1-x[i+m])*[0<=i+m<n] - x[i-m]*[0<=i-m<n]
    
    grad = np.zeros(n)
    active = np.where(v > 0)[0]
    for m_idx in active:
        m = m_idx - (n - 1)  # actual shift
        vi = 2.0 * v[m_idx]
        # Contribution from (1-x[i+m]) term: valid for i where 0 <= i+m < n
        i_start = max(0, -m)
        i_end = min(n, n - m)
        if i_start < i_end:
            grad[i_start:i_end] += vi * (1.0 - x[i_start + m:i_end + m])
        # Contribution from -x[i-m] term: valid for i where 0 <= i-m < n
        i_start2 = max(0, m)
        i_end2 = min(n, n + m)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= vi * x[i_start2 - m:i_end2 - m]
    return grad

    target = n / 2.0

            break

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def penalty_func_and_grad(x, C_dx, n, M):

    corr = np.correlate(x, 1.0 - x, mode='full')
    v = np.maximum(0.0, corr - C_dx)
    f = np.sum(v ** 2)
    sum_viol = x.sum() - n / 2.0
    f += M * sum_viol ** 2
    
    grad = np.zeros(n)
    active = np.where(v > 1e-15)[0]
    for m_idx in active:
        m = m_idx - (n - 1)
        vi = 2.0 * v[m_idx]
        i1 = max(0, -m)
        i2 = min(n, n - m)
        if i1 < i2:
            grad[i1:i2] += vi * (1.0 - x[i1 + m:i2 + m])

        if i3 < i4:

    grad += 2.0 * M * sum_viol
    return f, grad

                    corr -= B * h[i]
                    h[i] = 0.0
