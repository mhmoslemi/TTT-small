def exact_line_search(h, n, i, j):
    """Find optimal δ for h[i]+=δ, h[j]-=δ"""
    dx = 2.0/n
    # Compute current C values and slopes
    c = np.correlate(h, 1-h, mode='full') * dx  # current values
    # Slope for h[i]: 
    # dC_m/dh[i] = dx * [(1-h[i+lag])*1[valid] - h[i-lag]*1[valid]]
    # where lag = m - (n-1)
    lag = np.arange(2*n-1) - (n-1)
    # For index i:
    # term1: (1-h[i+lag]) where 0 <= i+lag < n
    # term2: -h[i-lag] where 0 <= i-lag < n
    i_plus = i + lag
    i_minus = i - lag
    valid_plus = (i_plus >= 0) & (i_plus < n)
    valid_minus = (i_minus >= 0) & (i_minus < n)
    
    slope_i = np.zeros(2*n-1)
    slope_i[valid_plus] = dx * (1 - h[i_plus[valid_plus]])
    slope_i[valid_minus] += dx * (-h[i_minus[valid_minus]])
    
    # Similarly for j
    j_plus = j + lag
    j_minus = j - lag
    valid_jp = (j_plus >= 0) & (j_plus < n)
    valid_jm = (j_minus >= 0) & (j_minus < n)

    """Gradient of C at a specific lag index (0-indexed in full corr output)"""
    dx = 2.0 / n
    k = lag_idx - (n - 1)
    g = np.zeros(n)
    for i in range(n):
        ip = i + k
        im = i - k
        if 0 <= ip < n:
            g[i] += dx * (1 - h[ip])
        if 0 <= im < n:
            g[i] += dx * (-h[im])
    return g

    """Vectorized gradient of smooth max"""
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    lag = np.arange(2*n-1) - (n-1)
    g = np.zeros(n)
    for i in range(n):
        ip = i + lag
        im = i - lag
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        g[i] = (w[vp] * (1-h[ip[vp]]).sum() - w[vm] * h[im[vm]].sum()) * (2.0/n)

        vm = (im >= 0) & (im < n)
        g[i] = dx * (np.dot(w[vp], 1 - x[ip[vp]]) - np.dot(w[vm], x[im[vm]]))
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def quick_lbfgs(h, n, deadline):
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    for alpha, mu in [(100, 1.0), (1000, 10.0), (10000, 100.0)]:
        if time.time() > deadline:
            break
        try:
            res = minimize(_so, h, jac=_sg, args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 150, 'ftol': 1e-16, 'gtol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                h, best_c = h_new, c_new
        except Exception:
            pass
    return h, best_c

def weighted_gradient(h, n, c_vals, alpha):
    m = 2 * n - 1
    dx = 2.0 / n
    sh = alpha * (c_vals - c_vals.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    lag = np.arange(m) - (n - 1)
    g = np.zeros(n)
    for i in range(n):
        ip = i + lag
        im = i - lag
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        g[i] = dx * (np.dot(w[vp], 1 - h[ip[vp]]) - np.dot(w[vm], h[im[vm]]))
    return g

def exact_ls(h, n, i, j, c_vals):
    m = 2 * n - 1
    dx = 2.0 / n
    lag = np.arange(m) - (n - 1)
    i_p, i_m = i + lag, i - lag
    vp = (i_p >= 0) & (i_p < n)
    vm = (i_m >= 0) & (i_m < n)
    s_i = np.zeros(m)
    s_i[vp] = dx * (1 - h[i_p[vp]])
    if vm.any():
        s_i[vm] += -dx * h[i_m[vm]]
    j_p, j_m = j + lag, j - lag
    vjp = (j_p >= 0) & (j_p < n)
    vjm = (j_m >= 0) & (j_m < n)
    s_j = np.zeros(m)
    s_j[vjp] = dx * (1 - h[j_p[vjp]])
    if vjm.any():
        s_j[vjm] += -dx * h[j_m[vjm]]
    slopes = s_i - s_j
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_lo >= d_hi - 1e-14:
        return 0.0, c_vals.max()
    lo, hi = d_lo, d_hi
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        val = c_vals + mid * slopes
        mx = val.max()
        active = val >= mx - 1e-12
        if slopes[active].min() > 1e-13:
            hi = mid
        elif slopes[active].max() < -1e-13:
            lo = mid
        else:
            break
    best_d = 0.5 * (lo + hi)
    best_val = (c_vals + best_d * slopes).max()
    for dt in (0.0, d_lo, d_hi):
        if d_lo - 1e-14 <= dt <= d_hi + 1e-14:
            v = (c_vals + dt * slopes).max()
            if v < best_val:
                best_val, best_d = v, dt
    return best_d, best_val

def exact_pair_descent(h, n, deadline):
    best_c = compute_c5(h, n)
    alphas = [100, 500, 2000, 8000, 30000, 100000]
    for alpha in alphas:
        p = 6
        stuck = 0
        while time.time() < deadline:
            c_vals = compute_c5_vec(h, n)
            cur = c_vals.max()
            g = weighted_gradient(h, n, c_vals, alpha)
            inc = np.argsort(g)[:p]
            dec = np.argsort(g)[-p:][::-1]
            best_d, best_i, best_j, best_cv = 0.0, -1, -1, cur + 1e-14
            for i in inc:
                for j in dec:
                    if i == j:
                        continue
                    d, cv = exact_ls(h, n, i, j, c_vals)
                    if cv < best_cv:
                        best_cv, best_d, best_i, best_j = cv, d, i, j
            if best_i >= 0:
                h[best_i] += best_d
                h[best_j] -= best_d
                np.clip(h, 0.0, 1.0, out=h)
                if best_cv < best_c - 1e-14:
                    best_c = best_cv
                    stuck = 0
                else:
                    stuck += 1
            else:
                rng_local = np.random.RandomState(int(time.time() * 1000) % 10000)
                found = False
                for _ in range(30):
                    i = rng_local.randint(n)
                    j = rng_local.randint(n)
                    if i == j:
                        continue
                    d, cv = exact_ls(h, n, i, j, c_vals)
                    if cv < cur - 1e-14:
                        h[i] += d
                        h[j] -= d
                        np.clip(h, 0.0, 1.0, out=h)
                        best_c = min(best_c, cv)
                        found = True
                        break
                if not found:
                    stuck += 1
            if stuck > 300:
                break
    return h, best_c

def refine_small(h, n, deadline, scale=0.005):
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(123)
    for _ in range(200):
        if time.time() > deadline:
            break
        idx = rng.choice(n, size=2, replace=False)
        i, j = int(idx[0]), int(idx[1])
        c_vals = compute_c5_vec(h, n)
        d, cv = exact_ls(h, n, i, j, c_vals)
        h[i] += d
        h[j] -= d
        np.clip(h, 0.0, 1.0, out=h)
        if cv < best_c:
            best_c = cv
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(650, 0.5)
        n = 650
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    # Phase 1: Quick L-BFGS-B
    t1 = min(t0 + 25, deadline - 400)
    if t1 > t0 + 5:
        h, best_c = quick_lbfgs(h, n, t1)
        h = project_feasible(h, n)
        best_c = compute_c5(h, n)
    # Phase 2: Main exact pair descent
    t2 = deadline - 80
    h, best_c = exact_pair_descent(h, n, t2)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    # Phase 3: Random pair refinement
    if time.time() < deadline - 3:
        h, c3 = refine_small(h, n, deadline - 2)
        h = project_feasible(h, n)
        c3 = compute_c5(h, n)
        if c3 < best_c:
            best_c = c3
    # Phase 4: Perturbation restart
    if time.time() < deadline - 2:
        for scale in (0.02, 0.01, 0.005):
            if time.time() > deadline - 2:
                break
            h_p = h + rng.standard_normal(n) * scale
            h_p = project_feasible(h_p, n)
            # Quick exact pair search on perturbed
            c_vals = compute_c5_vec(h_p, n)
            g = weighted_gradient(h_p, n, c_vals, 5000)
            inc = np.argsort(g)[:5]
            dec = np.argsort(g)[-5:][::-1]
            for _ in range(100):
                if time.time() > deadline - 2:
                    break
                best_d, best_i, best_j, best_cv = 0.0, -1, -1, c_vals.max() + 1e-14
                for i in inc:
                    for j in dec:
                        if i == j:
                            continue
                        d, cv = exact_ls(h_p, n, i, j, c_vals)
                        if cv < best_cv:
                            best_cv, best_d, best_i, best_j = cv, d, i, j
                if best_i < 0:
                    break
                h_p[best_i] += best_d
                h_p[best_j] -= best_d
                np.clip(h_p, 0.0, 1.0, out=h_p)
                c_vals = compute_c5_vec(h_p, n)
                g = weighted_gradient(h_p, n, c_vals, 5000)
                inc = np.argsort(g)[:5]
                dec = np.argsort(g)[-5:][::-1]
            c_p = compute_c5(h_p, n)
            if c_p < best_c:
                h, best_c = h_p, c_p
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c, n
