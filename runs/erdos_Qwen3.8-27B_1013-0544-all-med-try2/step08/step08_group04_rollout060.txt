import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:n])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def sp_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sp_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_pass(h, n, alpha, mu, deadline, maxiter=800):
    bounds = [(0.0, 1.0)] * n
    t_alloc = deadline - time.time()
    if t_alloc < 5:
        return h, compute_c5(h, n)
    try:
        res = minimize(sp_obj, h, jac=sp_grad, args=(n, alpha, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-14, 'maxfun': maxiter * 8,
                                'maxls': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def grad_for_j(h, j, n, dx):
    g = np.zeros(2 * n - 1)
    ks = np.arange(n)
    ms = ks + (n - 1)
    va = ks < (n - j)
    pb = ks <= j
    g[ms] = dx * ((1.0 - np.where(va, h[j + np.minimum(ks, n-1-j)], 0.0))[va] +
                  (-np.where(pb, h[j - np.minimum(ks, j)], 0.0))[pb])
    # Recompute properly for safety
    g = np.zeros(2 * n - 1)
    for k in range(n):
        m = k + (n - 1)
        val = 0.0
        if j + k < n:
            val += (1.0 - h[j + k])
        if j >= k:
            val += (-h[j - k])
        g[m] = dx * val
    kps = np.ones(n - 1, dtype=int)
    for idx in range(n - 1):
        kp = n - 1 - idx
        m = idx
        val = 0.0
        if j >= kp:
            val += (1.0 - h[j - kp])
        if j + kp < n:
            val += (-h[j + kp])
        g[m] = dx * val
    return g

def coord_descend(h, n, deadline, max_rounds=50000):
    h = project_feasible(h, n)
    dx = 2.0 / n
    Q = np.correlate(h, 1 - h, mode='full') * dx
    best_c = Q.max()
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return h, best_c
            g = grad_for_j(h, j, n, dx)
            v = h[j]
            # Candidate values: 0, 1, and points based on gradient direction
            cands = [0.0, 1.0]
            direction = -g  # direction to decrease
            for step in [0.005, 0.02, 0.05, 0.1, 0.2, 0.5]:
                cv = v + step * direction[j] / (np.abs(direction[j]) + 1e-15)
                if 0.0 <= cv <= 1.0:
                    cands.append(cv)
                cv2 = v - step
                if 0.0 <= cv2 <= 1.0:
                    cands.append(cv2)
                cv3 = v + step
                if 0.0 <= cv3 <= 1.0:
                    cands.append(cv3)
            best_v = v
            best_c_new = Q.max()
            for cv in cands:
                new_Q = Q + (cv - v) * g
                c_new = new_Q.max()
                if c_new < best_c_new - 1e-15:
                    best_c_new = c_new
                    best_v = cv
            if best_v != v:
                Q = Q + (best_v - v) * g
                h[j] = best_v
                best_c = Q.max()
                improved = True
        if not improved:
            break
    return h, best_c

def pair_search_fast(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - 50)
                hi = min(n, i + 50)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    for sgn in [1, -1]:
                        new_vi = h[i] + sgn * d
                        new_vj = h[j] - sgn * d
                        if 0.0 <= new_vi <= 1.0 and 0.0 <= new_vj <= 1.0:
                            h[i] = new_vi
                            h[j] = new_vj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= sgn * d
                                h[j] += sgn * d
            if not improved:
                break
    return h, best_c

def perturb(h, n, rng, scale):
    h_new = h + rng.randn(n) * scale
    return project_feasible(h_new, n)
