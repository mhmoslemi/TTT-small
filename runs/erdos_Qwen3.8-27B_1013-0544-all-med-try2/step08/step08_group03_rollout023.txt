            if tot < 1e-14:

            h += (diff / tot) * h

def grad_single(h, n, k):
    nv = n - abs(k)

    if k >= 0:
        grad[:nv] += 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:

        grad[kk:] += 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad
