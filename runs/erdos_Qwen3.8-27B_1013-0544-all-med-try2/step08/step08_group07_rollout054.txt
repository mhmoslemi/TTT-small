def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

            s = room.sum()
            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
