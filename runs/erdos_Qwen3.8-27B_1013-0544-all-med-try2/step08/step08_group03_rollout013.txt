    nv = n - abs(k)

    if k >= 0:
        grad[:nv] += 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

    m = c.max()
    sh = alpha * (c - m)

        if not improved:

            if stalls > 5:
                break
            delta *= 0.6
            if delta < 2e-5:
                for _ in range(200):
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i == j:
                        continue
                    d = 0.008 * (1 if rng.random() > 0.5 else -1)
                    if 0 <= best_h[i] + d <= 1 and 0 <= best_h[j] - d <= 1:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c:
                            best_c = c
                            break
                else:
                    break
        else:
            stalls = 0
            delta = min(delta * 1.1, 0.005)
    return best_h, best_c

def simulated_annealing(h, n, deadline, top_k_lags=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    cur_c = compute_c5(cur_h, n)
    T = 0.02
    rng = np.random.RandomState(999)
    n_iter = 0
    while time.time() < deadline:
        n_iter += 1
        T *= 0.9997
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.005
        direction = 1 if rng.random() > 0.5 else -1
        if 0 <= cur_h[i] + direction * d <= 1 and 0 <= cur_h[j] - direction * d <= 1:
            cur_h[i] += direction * d
            cur_h[j] -= direction * d
            c_new = compute_c5(cur_h, n)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = cur_h.copy()
            elif rng.random() < np.exp((cur_c - c_new) / max(T, 1e-8)):
                cur_c = c_new
            else:
                cur_h[i] -= direction * d
                cur_h[j] += direction * d
        if n_iter % 200 == 0 and T < 0.001:
            break
    return best_h, best_c

def lbfgs_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 50.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def lbfgs_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    top_idx = np.argsort(w)[-50:]
    g = np.zeros(len(x))
    for i in top_idx:
        if w[i] < 1e-12:
            continue
        k = i - (n - 1)
        g += w[i] * dx * grad_single_lag(x, n, k)
    g += 100.0 * (x.sum() - n / 2.0)
    return g

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.zeros(500)
        idx = rng.choice(500, 250, replace=False)
        h[idx] = 1.0
        n = 500
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS-B
    t1 = min(t0 + 120, deadline - 300)
    for alpha in [800, 3000, 10000]:
        if time.time() > t1:
            break
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad, args=(n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-14})
            hnew = project_feasible(res.x, n)
            cn = compute_c5(hnew, n)
            if cn < best_c:
                best_c = cn
                best_h = hnew
        except Exception:
            pass

    # Phase 2: Nesterov PGD
    if time.time() < deadline - 200:
        t2 = min(time.time() + 150, deadline - 120)
        best_h, best_c = nesterov_pgd(best_h, n, t2, alpha=800.0, max_iter=6000)

    # Phase 3: Gradient-guided coordinate descent
    if time.time() < deadline - 100:
        t3 = min(time.time() + 400, deadline - 40)
        best_h, best_c = grad_guided_polish(best_h, n, t3, top_k_lags=30, delta=0.003)

    # Phase 4: Simulated annealing (short)
    if time.time() < deadline - 60:
        t4 = min(time.time() + 100, deadline - 20)
        h_sa, c_sa = simulated_annealing(best_h, n, t4, top_k_lags=20)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa

    # Phase 5: Final gradient-guided polish
    if time.time() < deadline - 5:
        best_h, best_c = grad_guided_polish(best_h, n, deadline - 1, top_k_lags=20, delta=0.001)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
