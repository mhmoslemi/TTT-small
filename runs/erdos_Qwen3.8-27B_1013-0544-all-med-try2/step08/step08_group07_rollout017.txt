def subgradient_method(h, n, steps=10000, deadline=None):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    h = h.copy()
    for step in range(steps):
        if deadline and time.time() > deadline:
            break
        # Compute gradient of g[k*] w.r.t. h
        corr = np.correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
        # g[k] = sum_i h[i] * (1 - h[i+k]) for appropriate i
        # dg[k]/dh[j] = (1 - h[j+k]) - h[j-k] (with appropriate bounds)
        # For lag k_star (0-indexed in 'full' mode, actual lag = k_star - (n-1))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        # For each j, contribution to g[lag]:
        # if j + lag in [0, n): grad[j] += (1 - h[j + lag])
        # if j - lag in [0, n): grad[j] -= h[j - lag]
        j_plus = np.arange(n) + lag
        mask_p = (j_plus >= 0) & (j_plus < n)
        grad[mask_p] += (1.0 - h[j_plus[mask_p]])
        j_minus = np.arange(n) - lag
        mask_m = (j_minus >= 0) & (j_minus < n)
        grad[mask_m] -= h[j_minus[mask_m]]
        
        # Step size (decreasing)
        lr = 0.1 / (1 + step * 0.01)
        h = h - lr * grad
        h = project_feasible(h, n)
    
    return h

                improved = True
        if not improved:
            break
    return h, corr

def fast_binary_search(h, n, deadline, max_sweeps=100):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n // 2]] = 1.0
    h = h_b.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_set = set(np.where(h == 0)[0].tolist())
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in sorted(zeros_set):
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in sorted(ones_set):
                if j == i:
                    continue
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                delta = np.zeros(2 * n - 1)
                im = i + m_range; msk = (im >= 0) & (im < n); delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range; msk = (imn >= 0) & (imn < n); delta[msk] -= h[imn[msk]]
                jm = best_j + m_range; msk = (jm >= 0) & (jm < n); delta[msk] -= (1.0 - h[jm[msk]])
                jmn = best_j - m_range; msk = (jmn >= 0) & (jmn < n); delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (best_j - i)] += 1.0
                delta[n - 1 + (i - best_j)] += 1.0
                corr += delta
                cur_c = best_nc
                zeros_set.discard(i); ones_set.add(i)
                ones_set.discard(best_j); zeros_set.add(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c, _ = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Binary local search
    p1 = t0 + 0.06 * total
    hb, cb = fast_binary_search(h0, n, p1, max_sweeps=80)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Coordinate descent
    p2 = t0 + 0.18 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=300, deadline=p2)
    hc = project_feasible(hc, n)
    cc, _ = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 3: SLSQP top-K ramp (main workhorse)
    p3 = t0 + 0.55 * total
    schedule = [
        (100, 60, 1500), (500, 40, 1500), (2000, 30, 2000),
        (10000, 20, 2000), (50000, 15, 2500), (200000, 10, 3000),
        (1000000, 8, 3000), (5000000, 5, 4000), (20000000, 4, 4000),
        (100000000, 3, 5000), (500000000, 2, 5000), (2000000000, 2, 5000),
    ]
    h = best_h.copy()
    for alpha, topk, iters in schedule:
        if time.time() > p3 - 8:
            break
        h = slsqp_topk(h, n, alpha, topk, iters, 1e-16)
        c, _ = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Coordinate descent refinement
    p4 = t0 + 0.65 * total
    hc2, _ = coord_descent(best_h, n, max_sweeps=400, deadline=p4)
    hc2 = project_feasible(hc2, n)
    cc2, _ = compute_c5(hc2, n)
    if cc2 < best_c:
        best_c, best_h = cc2, hc2.copy()

    # Phase 5: High-alpha SLSQP polish
    p5 = t0 + 0.80 * total
    h = best_h.copy()
    for alpha, topk, iters in [(100000000, 2, 5000), (500000000, 2, 6000),
                                (2000000000, 1, 6000), (10000000000, 1, 6000)]:
        if time.time() > p5 - 5:
            break
        h = slsqp_topk(h, n, alpha, topk, iters, 1e-18)
        c, _ = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 6: Diverse restarts
    p6 = t0 + 0.93 * total
    for trial in range(5):
        if time.time() > p6 - 3:
            break
        ns = 0.001 + 0.002 * trial
        h = best_h + ns * rng.randn(n)
        h = project_feasible(h, n)
        h, _ = coord_descent(h, n, max_sweeps=60, deadline=p6 - 6)
        h = project_feasible(h, n)
        for alpha, topk, iters in [(5000000, 3, 2000), (50000000, 2, 2000)]:
            if time.time() > p6 - 2:
                break
            h = slsqp_topk(h, n, alpha, topk, iters, 1e-16)
            c, _ = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 7: Try alternate resolution
    if time.time() < dl - 20:
        nh = 480
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, _ = coord_descent(hh, nh, max_sweeps=80, deadline=dl - 15)
        hh = project_feasible(hh, nh)
        for alpha, topk, iters in [(100000, 10, 1500), (5000000, 4, 2000), (100000000, 2, 2000)]:
            if time.time() > dl - 5:
                break
            hh = slsqp_topk(hh, nh, alpha, topk, iters, 1e-16)
            ch, _ = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh

    # Final: one more coord descent pass
    if time.time() < dl - 3:
        hf, _ = coord_descent(best_h, n, max_sweeps=100, deadline=dl - 1)
        hf = project_feasible(hf, n)
        cf, _ = compute_c5(hf, n)
        if cf < best_c:
            best_c, best_h = cf, hf.copy()

    best_h = project_feasible(best_h, n)
    best_c, _ = compute_c5(best_h, n)
    return best_h, best_c, n
