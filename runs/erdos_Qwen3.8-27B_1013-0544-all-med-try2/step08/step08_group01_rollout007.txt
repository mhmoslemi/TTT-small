def _smooth_obj_grad(h,n,alpha):
    N=1<<(2*n-1).bit_length(); dx=2.0/n
    fh=rfft(h,N); fa=rfft(1.0-h,N)

```python
import numpy as np
import time
from scipy.optimize import minimize, linprog
from numpy.fft import rfft, irfft


def _project_box_sum(y, target):
    y = np.asarray(y, dtype=float).copy()
    if y.size == 0:
        return y
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-15:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-15:
                h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def _compute_c5_direct(h, n):
    if n <= 0:
        return 1.0
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx


def _fft_c_all(h, n):
    N = 1 << (2 * n - 1).bit_length()
    dx = 2.0 / float(n)
    fh = rfft(h, N)
    fa = rfft(1.0 - h, N)
    cor = irfft(fh * np.conj(fa), N)
    cpos = cor[:n] * dx
    if n > 1:
        cneg = cor[N - 1:N - n:-1] * dx
        return np.concatenate((cneg, cpos))
    return cpos


def _fft_c_max(h, n):
    N = 1 << (2 * n - 1).bit_length()
    dx = 2.0 / float(n)
    fh = rfft(h, N)
    fa = rfft(1.0 - h, N)
    cor = irfft(fh * np.conj(fa), N)
    pm = float(np.max(cor[:n])) if n > 0 else 0.0
    nm = float(np.max(cor[N - 1:N - n:-1])) if n > 1 else -np.inf
    return max(pm, nm) * dx


def _smooth_obj_grad(h, n, alpha):
    N = 1 << (2 * n - 1).bit_length()
    dx = 2.0 / float(n)
    fh = rfft(h, N)
    fa = rfft(1.0 - h, N)
    cor = irfft(fh * np.conj(fa), N)
    cpos = cor[:n] * dx
    if n > 1:
        cneg = cor[N - 1:N - n:-1] * dx
        c = np.concatenate((cneg, cpos))
    else:
        c = cpos
    m = float(np.max(c))
    sh = alpha * (c - m)
    sm = float(np.max(sh))
    e = np.exp(sh - sm)
    S = float(np.sum(e))
    if not np.isfinite(S) or S <= 0.0:
        return m, np.zeros(n)
    val = m + (sm + np.log(S)) / alpha
    w = e / S

    wp = w[n - 1:]
    wn = np.zeros(n)
    if n > 1:
        wn[1:] = w[:n - 1]

    fwp = rfft(wp, N)
    fwn = rfft(wn, N)
    g = (irfft(fwp * fa, N)[:n]
         - irfft(fh * np.conj(fwp), N)[:n]
         + irfft(fa * np.conj(fwn), N)[:n]
         - irfft(fwn * fh, N)[:n])
    return float(val), np.asarray(g * dx, dtype=float)


def _smooth_slsqp_obj(x, n, alpha):
    return _smooth_obj_grad(x, n, alpha)[0]


def _smooth_slsqp_jac(x, n, alpha):
    return _smooth_obj_grad(x, n, alpha)[1]


def _pen_obj(x, n, alpha, mu):
    v, g = _smooth_obj_grad(x, n, alpha)
    s = float(x.sum()) - 0.5 * n
    return float(v + mu * s * s)


def _pen_grad(x, n, alpha, mu):
    v, g = _smooth_obj_grad(x, n, alpha)
    s = float(x.sum()) - 0.5 * n
    return g + (2.0 * mu * s)


def _eq_fun(x, n):
    return float(x.sum() - 0.5 * n)


def _eq_jac(x, n):
    return np.ones_like(x)


def _lbfgs_penalty(h, n, deadline, is_parent):
    target = 0.5 * n
    h = _project_box_sum(h, target)
    cbest = _fft_c_max(h, n)
    hbest = h.copy()
    bounds = [(0.0, 1.0)] * n
    if is_parent:
        alphas = (2000, 10000, 50000)
        mus = (1e-8, 1e-7, 1e-5)
        maxit = 90
    else:
        alphas = (1500, 8000)
        mus = (1e-7, 1e-5)
        maxit = 60
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 6:
            break
        mu = mus[min(i, len(mus) - 1)]
        try:
            res = minimize(_pen_obj, h, jac=_pen_grad, args=(n, alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-16,
                                    'gtol': 1e-14, 'maxls': 40})
            if res is not None and np.all(np.isfinite(res.x)):
                h2 = _project_box_sum(res.x, target)
                c2 = _fft_c_max(h2, n)
                if c2 < cbest - 1e-14:
                    cbest = c2
                    hbest = h2
                    h = h2
        except Exception:
            pass
    return _project_box_sum(hbest, target), cbest


def _lag_value_C(h, n, idx, dx):
    idx = int(idx)
    if idx < n - 1:
        k = idx + 1
        m = n - k
        return float(np.dot(h[:m], 1.0 - h[k:])) * dx
    L = idx - (n - 1)
    if L == 0:
        return float(np.dot(h, 1.0 - h)) * dx
    m = n - L
    return float(np.dot(h[L:], 1.0 - h[:m])) * dx


def _lag_grad(h, n, idx, dx):
    idx = int(idx)
    g = np.zeros(n)
    if idx < n - 1:
        k = idx + 1
        m = n - k
        g[:m] += (1.0 - h[k:]) * dx
        g[k:] -= h[:m] * dx
    else:
        L = idx - (n - 1)
        if L == 0:
            return (1.0 - 2.0 * h) * dx
        m = n - L
        g[:m] -= h[L:] * dx
        g[L:] += (1.0 - h[:m]) * dx
    return g


def _select_active(c_all, M):
    M = int(M)
    if M <= 0:
        return np.arange(0, dtype=int)
    if len(c_all) <= M:
        return np.arange(len(c_all), dtype=int)
    return np.argsort(c_all)[-M:]


def _lp_active_polish(h, n, deadline, M=24):

    h = _project_box_sum(h, target)
    best = _fft_c_max(h, n)
    trusts = (0.10, 0.04, 0.016, 0.006, 0.0025, 0.001, 0.0004, 0.00015)
    for trust in trusts:
        improved = True
        outer = 0
        while improved and outer < 50 and time.time() < deadline - 0.5:
            outer += 1
            c_all = _fft_c_all(h, n)
            cur = float(np.max(c_all))
            if cur < best:
                best = cur
            active = _select_active(c_all, min(M, len(c_all)))
            if active.size == 0:
                break
            dx = 2.0 / float(n)
            A = np.zeros((active.size, n + 1))
            b = np.empty(active.size)
            for r, idx in enumerate(active):
                gv = _lag_grad(h, n, idx, dx)
                cv = _lag_value_C(h, n, idx, dx)
                A[r, :n] = gv
                A[r, n] = -1.0
                b[r] = float(np.dot(gv, h)) - cv

            lo = np.maximum(0.0, h - trust)
            hi = np.minimum(1.0, h + trust)
            bad = lo > hi
            if bad.any():
                lo[bad] = h[bad]
                hi[bad] = h[bad]

            bounds = list(zip(lo, hi))
            bounds.append((0.0, 1.0))
            cobj = np.zeros(n + 1)
            cobj[-1] = 1.0
            Aeq = np.zeros((1, n + 1))
            Aeq[0, :n] = 1.0
            beq = np.array([target], dtype=float)

            res = None
            try:
                res = linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq,
                              bounds=bounds, method='highs')
            except Exception:
                try:
                    res = linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq,
                                  bounds=bounds)
                except Exception:
                    res = None

            ok = (res is not None and bool(getattr(res, 'success', False))
                  and np.all(np.isfinite(res.x)))
            if not ok:
                break

            u = _project_box_sum(res.x[:n], target)
            if float(np.max(np.abs(u - h))) < 1e-7:
                break

            thetas = (1e-6, 3e-6, 1e-5, 3e-5, 1e-4, 3e-4,
                      1e-3, 3e-3, 1e-2, 3e-2, 0.10, 0.25,
                      0.50, 0.75, 1.00)
            bth = 0.0
            bloc = best
            for th in thetas:
                hn = _project_box_sum((1.0 - th) * h + th * u, target)
                cn = _fft_c_max(hn, n)
                if np.isfinite(cn) and cn < bloc - 1e-15:
                    bloc = cn
                    bth = th

            if bth > 0.0 and bloc < best - 1e-13:
                h = _project_box_sum((1.0 - bth) * h + bth * u, target)
                best = bloc
                improved = True
            else:
                improved = False
    return _project_box_sum(h, target), best


def _t_obj(x):
    return float(x[-1])


def _t_jac(x):
    g = np.zeros(x.size)
    g[-1] = 1.0
    return g


def _eq_t_fun(x, n):
    return float(x[:n].sum() - 0.5 * n)


def _eq_t_jac(x, n):
    g = np.zeros(x.size)
    g[:n] = 1.0
    return g


def _active_ineq_fun(x, n, active):
    h = x[:n]
    t = float(x[-1])
    dx = 2.0 / float(n)
    out = np.empty(active.size, dtype=float)
    for r, idx in enumerate(active):
        out[r] = t - _lag_value_C(h, n, idx, dx)
    return out


def _active_ineq_jac(x, n, active):
    h = x[:n]
    dx = 2.0 / float(n)
    J = np.zeros((active.size, n + 1))
    for r, idx in enumerate(active):
        g = _lag_grad(h, n, idx, dx)
        J[r, :n] = -g
        J[r, n] = 1.0
    return J


def _slsqp_smooth(h, n, deadline, alphas, maxit):
    target = 0.5 * n
    h = _project_box_sum(h, target)
    best = _fft_c_max(h, n)
    hbest = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline - 8:
            break
        try:
            res = minimize(_smooth_slsqp_obj, h, jac=_smooth_slsqp_jac,
                           args=(n, alpha), method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': int(maxit), 'ftol': 1e-13})
            if res is not None and np.all(np.isfinite(res.x)):
                h2 = _project_box_sum(res.x, target)
                c2 = _fft_c_max(h2, n)
                if c2 < best - 1e-14:
                    best = c2
                    hbest = h2
                    h = h2
        except Exception:
            pass
    return _project_box_sum(hbest, target), best


def _slsqp_active_t(h, n, deadline, M=16, maxit=18, rounds=1):
    target = 0.5 * n
    h = _project_box_sum(h, target)
    best = _fft_c_max(h, n)
    hbest = h.copy()
    bounds = [(0.0, 1.0)] * n
    bounds.append((0.0, 1.0))
    for _ in range(int(rounds)):
        if time.time() > deadline - 15:
            break
        c_all = _fft_c_all(h, n)
        active = _select_active(c_all, min(M, len(c_all)))
        if active.size == 0:
            break
        x0 = np.concatenate((h, np.array([best + 1e-6])))
        cons = [
            {'type': 'eq', 'fun': _eq_t_fun, 'jac': _eq_t_jac, 'args': (n,)},
            {'type': 'ineq', 'fun': _active_ineq_fun,
             'jac': _active_ineq_jac, 'args': (n, active)},
        ]
        try:
            res = minimize(_t_obj, x0, jac=_t_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': int(maxit), 'ftol': 1e-12})
            if res is not None and np.all(np.isfinite(res.x)):
                h2 = _project_box_sum(res.x[:n], target)
                c2 = _fft_c_max(h2, n)
                if c2 < best - 1e-14:
                    best = c2
                    hbest = h2
                    h = h2
            else:
                break
        except Exception:
            break
    return _project_box_sum(hbest, target), best


def _pair_polish(h, n, deadline, top=60):
    target = 0.5 * n
    h = _project_box_sum(h, target)
    best = _fft_c_max(h, n)
    steps = (0.020, 0.010, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4)
    for step in steps:
        improved_out = True
        outer = 0
        while improved_out and outer < 25 and time.time() < deadline - 2:
            outer += 1
            g = _smooth_obj_grad(h, n, 8000)[1]
            order = np.argsort(g)
            low = order[:top]
            high = order[-top:]
            improved = False
            for i in high:
                if time.time() > deadline - 2:
                    return _project_box_sum(h, target), best
                hi_i = int(i)
                for j in low:
                    lo_j = int(j)
                    if hi_i == lo_j:
                        continue
                    if h[hi_i] > step and h[lo_j] < 1.0 - step:
                        oi = h[hi_i]
                        oj = h[lo_j]
                        h[hi_i] = oi - step
                        h[lo_j] = oj + step
                        c = _fft_c_max(h, n)
                        if np.isfinite(c) and c < best - 1e-15:
                            best = c
                            improved = True
                        else:
                            h[hi_i] = oi
                            h[lo_j] = oj
            improved_out = improved
    return _project_box_sum(h, target), best


def _resample(h, n0, n1):
    if n0 == n1:
        return _project_box_sum(h, 0.5 * n1)
    if n0 % n1 == 0:
        blk = n0 // n1
        try:
            y = np.mean(h.reshape(n1, blk), axis=1)
            return _project_box_sum(y, 0.5 * n1)
        except Exception:
            pass
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    y = np.interp(x1, x0, h)
    return _project_box_sum(y, 0.5 * n1)


def _sturmian_start(n, slope, phase=0.0):
    i = np.arange(n, dtype=float)
    h = (np.mod(i * slope + phase, 1.0) < 0.5).astype(float)
    return _project_box_sum(h, 0.5 * n)


def _adopt(h2, n, h, best_h, best_c, use_tol=1e-10):
    h2 = _project_box_sum(h2, 0.5 * n)
    c2 = _compute_c5_direct(h2, n)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()
    if c2 <= best_c + use_tol:
        h = h2
    return h, best_h, best_c


def _optimize_trial(h, n, deadline, is_parent):
    target = 0.5 * n
    h = _project_box_sum(h, target)
    best_h = h.copy()
    best_c = _compute_c5_direct(h, n)

    if time.time() < deadline - 25:
        lb_end = time.time() + min(110 if is_parent else 50,
                                   max(5, (deadline - time.time()) * 0.22))
        h2, _ = _lbfgs_penalty(h, n, lb_end, is_parent)
        h, best_h, best_c = _adopt(h2, n, h, best_h, best_c)

    if time.time() < deadline - 25:
        lp_end = time.time() + min(300 if is_parent else 130,
                                   max(5, (deadline - time.time()) * 0.52))
        M = 30 if n <= 700 else 22
        h2, _ = _lp_active_polish(h, n, lp_end, M)
        h, best_h, best_c = _adopt(h2, n, h, best_h, best_c)

    if time.time() < deadline - 30:
        ss_end = time.time() + min(65 if is_parent else 25,
                                   max(5, (deadline - time.time()) * 0.30))
        alphas = (2500, 12000) if is_parent else (5000,)
        maxit = 30 if is_parent else 18
        h2, _ = _slsqp_smooth(h, n, ss_end, alphas, maxit)
        h, best_h, best_c = _adopt(h2, n, h, best_h, best_c)

    if time.time() < deadline - 25:
        at_end = time.time() + min(50 if is_parent else 22,
                                   max(5, (deadline - time.time()) * 0.40))
        M = 16 if n <= 700 else 12
        maxit = 20 if is_parent else 12
        h2, _ = _slsqp_active_t(h, n, at_end, M, maxit, 1)
        h, best_h, best_c = _adopt(h2, n, h, best_h, best_c)

    if time.time() < deadline - 5:
        pp_end = time.time() + min(25 if is_parent else 12,
                                   max(2, deadline - time.time() - 2))
        top = 70 if n <= 700 else 40
        h2, _ = _pair_polish(h, n, pp_end, top)
        h, best_h, best_c = _adopt(h2, n, h, best_h, best_c)

    best_h = _project_box_sum(best_h, target)
    best_c = _compute_c5_direct(best_h, n)
    return best_h, best_c


def _finish_polish(h, n, deadline):
    target = 0.5 * n
    h = _project_box_sum(h, target)
    best_h = h.copy()
    best_c = _compute_c5_direct(h, n)

    if time.time() < deadline - 8:
        lp_end = time.time() + min(55, max(4, (deadline - time.time()) * 0.35))
        M = 28 if n <= 700 else 18
        h2, _ = _lp_active_polish(h, n, lp_end, M)
        h, best_h, best_c = _adopt(h2, n, h, best_h, best_c)

    if time.time() < deadline - 12:
        at_end = time.time() + min(35, max(4, (deadline - time.time()) * 0.35))
        M = 14 if n <= 700 else 10
        h2, _ = _slsqp_active_t(h, n, at_end, M, 14, 1)
        h, best_h, best_c = _adopt(h2, n, h, best_h, best_c)

    if time.time() < deadline - 3:
        pp_end = max(2.5, deadline - 1.5)
        h2, _ = _pair_polish(h, n, pp_end, 45 if n <= 700 else 30)
        h, best_h, best_c = _adopt(h2, n, h, best_h, best_c)

    best_h = _project_box_sum(best_h, target)
    best_c = _compute_c5_direct(best_h, n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    tau = max(1.0, float(budget_s) - 3.0)
    deadline = t0 + tau

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = int(len(h_parent))
    except Exception:
        n_parent = 500
        h_parent = _sturmian_start(n_parent, 0.5 * (np.sqrt(5.0) - 1.0))

    if n_parent < 2:
        n_parent = 500
        h_parent = _sturmian_start(n_parent, 0.5 * (np.sqrt(5.0) - 1.0))

    h_parent = _project_box_sum(h_parent, 0.5 * n_parent)
    best_h = h_parent.copy()
    best_n = n_parent
    best_c = _compute_c5_direct(best_h, best_n)

    reserve = max(60.0, 0.18 * float(budget_s))
    parent_dl = min(deadline - reserve, t0 + max(80.0, 0.55 * float(budget_s)))
    if parent_dl < t0 + 30.0:
        parent_dl = min(deadline - 15.0, t0 + max(20.0, 0.45 * float(budget_s)))

    if parent_dl > t0 + 20.0:
        h2, c2 = _optimize_trial(best_h, best_n, parent_dl, True)
        if c2 < best_c:
            best_h = h2
            best_n = best_n
            best_c = c2

    if time.time() < deadline - 120.0:
        n2 = 250
        if best_n < n2:
            n2 = max(128, int(round(best_n / 2.0)))
            if n2 % 2 == 1:
                n2 += 1
        h3 = _resample(best_h, best_n, n2)
        alloc = max(40.0, min(220.0, (deadline - time.time() - 60.0) * 0.45))
        if alloc > 35.0:
            h2, c2 = _optimize_trial(h3, n2, time.time() + alloc, False)
            if c2 < best_c:
                best_h = h2
                best_n = n2
                best_c = c2

    if time.time() < deadline - 90.0:
        n3 = 128
        h4 = _sturmian_start(n3, 0.5 * (np.sqrt(5.0) - 1.0), 0.23)
        alloc = max(25.0, min(110.0, (deadline - time.time() - 45.0) * 0.5))
        if alloc > 25.0:
            h2, c2 = _optimize_trial(h4, n3, time.time() + alloc, False)
            if c2 < best_c:
                best_h = h2
                best_n = n3
                best_c = c2

    if time.time() < deadline - 12.0:
        h2, c2 = _finish_polish(best_h, best_n, deadline - 2.0)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    best_h = _project_box_sum(best_h, 0.5 * best_n)
    best_c = _compute_c5_direct(best_h, best_n)
    return np.asarray(best_h, dtype=float), float(best_c), int(best_n)
```
