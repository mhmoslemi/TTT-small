def fix_sum(x,T):
    for _ in range(12):
        x = np.clip(x, 0.0, 1.0)
        d = T - x.sum()
        if abs(d) < 1e-13:
            return x
        if d > 0:
            m = x < 1.0
            t = (1.0 - x[m]).sum()
            if t <= 0: break
            x[m] += (d / t) * (1.0 - x[m])
        else:
            m = x > 0.0
            t = x[m].sum()
            if t <= 0: break
            x[m] += (d / t) * x[m]
    return np.clip(x,0,1)
```

def active_gradient(c, h, n, mid, alpha, top_lags, grad, coeff):
    grad.fill(0.0)
    L = len(c)
    if L <= top_lags:
        idxs = np.arange(L)
    else:
        idxs = np.argpartition(c, -top_lags)[-top_lags:]
    vals = c[idxs]
    w = alpha * (vals - vals.max())
    w = np.exp(w - w.max())
    s = w.sum()
    if s <= 0: return grad
    w /= s
    for t in range(len(idxs)):
        wi = w[t]
        if wi < 1e-12: continue
        make_lag_coeff(h, n, int(idxs[t]) - mid, coeff)
        grad += wi * coeff
    return grad
```

import numpy as np
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def fix_sum(x, T):
    for _ in range(14):
        x = np.clip(x, 0.0, 1.0)
        d = T - x.sum()
        if abs(d) < 1e-13:
            return x
        if d > 0.0:
            m = x < 1.0
            t = (1.0 - x[m]).sum()
            if t <= 0.0:
                break
            x[m] += (d / t) * (1.0 - x[m])
        else:
            m = x > 0.0
            t = x[m].sum()
            if t <= 0.0:
                break
            x[m] += (d / t) * x[m]
    return np.clip(x, 0.0, 1.0)


def project_feasible(z, n):
    x = np.asarray(z, dtype=float)
    if x.size != n:
        x = x.reshape(-1)[:n]
        if x.size < n:
            x = np.resize(x, n)
    return fix_sum(np.clip(x, 0.0, 1.0), 0.5 * n)


def make_lag_coeff(h, n, k, out):
    out.fill(0.0)
    if k == 0:
        out[:] = 1.0 - 2.0 * h
    elif k > 0:
        nv = n - k
        out[:nv] += 1.0 - h[k:]
        out[k:] += -h[:nv]
    else:
        t = -k
        nv = n - t
        out[t:] += 1.0 - h[:nv]
        out[:nv] += -h[t:]
    return out


def make_coord_coeff(h, n, p, mid, out):
    out.fill(0.0)
    out[mid] = 1.0 - 2.0 * h[p]
    r = n - 1 - p
    if r > 0:
        out[mid + 1:mid + 1 + r] += 1.0 - h[p + 1:n]
        out[mid - r:mid] += -h[n - 1:p:-1]
    if p > 0:
        out[mid + 1:mid + 1 + p] += -h[p - 1::-1]
        out[mid - p:mid] += 1.0 - h[:p]
    return out


def pair_eval(c, g, idxp, idxm, d, tmp):
    np.multiply(g, d, out=tmp)
    tmp += c
    if d != 0.0:
        d2 = d * d
        tmp[idxp] += d2
        tmp[idxm] += d2
    return float(tmp.max())


def pair_best_delta(c, h, i, j, n, mid, ai, aj, gbuf, tmp, cmax):
    lo = max(h[i] - 1.0, -h[j])
    hi = min(1.0 - h[j], h[i])
    if hi <= lo + 1e-12:
        return 0.0, cmax

    make_coord_coeff(h, n, i, mid, ai)
    make_coord_coeff(h, n, j, mid, aj)
    np.subtract(aj, ai, out=gbuf)

    idxp = mid + (j - i)
    idxm = mid + (i - j)

    a = lo
    b = hi
    c1 = a + (b - a) * 0.38196601125
    c2 = a + (b - a) * 0.61803398875
    f1 = pair_eval(c, gbuf, idxp, idxm, c1, tmp)
    f2 = pair_eval(c, gbuf, idxp, idxm, c2, tmp)

    for _ in range(26):
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = a + (b - a) * 0.38196601125
            f1 = pair_eval(c, gbuf, idxp, idxm, c1, tmp)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + (b - a) * 0.61803398875
            f2 = pair_eval(c, gbuf, idxp, idxm, c2, tmp)

    best_d = 0.0
    best_f = cmax
    for d in (a, b, 0.0, 0.5 * (a + b)):
        f = pair_eval(c, gbuf, idxp, idxm, d, tmp)
        if f < best_f:
            best_f = f
            best_d = d
    return float(best_d), best_f


def try_pair(c, h, a, b, n, mid, ai, aj, gbuf, tmp, cmax):
    d, bv = pair_best_delta(c, h, a, b, n, mid, ai, aj, gbuf, tmp, cmax)
    if d == 0.0 or bv >= cmax - 1e-11:
        return cmax, 0

    h[a] -= d
    h[b] += d
    if h[a] < 0.0:
        h[a] = 0.0
    elif h[a] > 1.0:
        h[a] = 1.0
    if h[b] < 0.0:
        h[b] = 0.0
    elif h[b] > 1.0:
        h[b] = 1.0

    np.multiply(gbuf, d, out=tmp)
    c += tmp
    d2 = d * d
    c[mid + (b - a)] += d2
    c[mid + (a - b)] += d2
    return bv, 1


def active_gradient(c, h, n, mid, alpha, top_lags, grad, coeff):
    grad.fill(0.0)
    L = len(c)
    if L <= top_lags:
        idxs = np.arange(L)
    else:
        idxs = np.argpartition(c, -top_lags)[-top_lags:]

    vals = c[idxs]
    w = alpha * (vals - vals.max())
    w = np.exp(w - w.max())
    s = w.sum()
    if s <= 0.0:
        return grad
    w /= s

    for r in range(len(idxs)):
        wi = w[r]
        if wi < 1e-12:
            continue
        make_lag_coeff(h, n, int(idxs[r]) - mid, coeff)
        grad += wi * coeff
    return grad


def pair_polish(h, n, deadline, rng, H=25, top_lags=50):
    h = project_feasible(h, n)
    dx = 2.0 / n
    mid = n - 1
    L = 2 * n - 1

    c = np.correlate(h, 1.0 - h, mode="full")
    best_c = float(c.max() * dx)
    best_h = h.copy()

    ai = np.empty(L)
    aj = np.empty(L)
    gbuf = np.empty(L)
    tmp = np.empty(L)
    grad = np.empty(n)
    coeff = np.empty(n)

    alphas = (5.0, 20.0, 100.0, 500.0)
    round_no = 0
    stuck = 0

    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode="full")
        cmax = float(c.max())
        if cmax * dx < best_c - 1e-12:
            best_c = cmax * dx
            best_h = h.copy()

        Huse = 45 if (round_no % 5 == 4) else H
        if n <= Huse:
            high = low = np.arange(n)
        else:
            active_gradient(c, h, n, mid, alphas[round_no % 4], top_lags, grad, coeff)
            order = np.argsort(grad)
            low = order[:Huse]
            high = order[::-1][:Huse]

        seen = set()
        moved = 0

        for ii in high:
            for jj in low:
                if ii == jj:
                    continue
                a = ii if ii < jj else jj
                b = jj if ii < jj else ii
                key = a * n + b
                if key in seen:
                    continue
                seen.add(key)

                new_cmax, ok = try_pair(c, h, a, b, n, mid, ai, aj, gbuf, tmp, cmax)
                if ok:
                    cmax = new_cmax
                    moved += 1
                    if cmax * dx < best_c - 1e-12:
                        best_c = cmax * dx
                        best_h = h.copy()
                    if time.time() > deadline - 4:
                        break
            if time.time() > deadline - 4:
                break

        if time.time() < deadline - 12:
            for _ in range(70):
                a = int(rng.randint(n))
                b = int(rng.randint(n))
                if a == b:
                    continue
                if a > b:
                    a, b = b, a
                key = a * n + b
                if key in seen:
                    continue
                seen.add(key)

                new_cmax, ok = try_pair(c, h, a, b, n, mid, ai, aj, gbuf, tmp, cmax)
                if ok:
                    cmax = new_cmax
                    moved += 1
                    if cmax * dx < best_c - 1e-12:
                        best_c = cmax * dx
                        best_h = h.copy()
                    if time.time() > deadline - 4:
                        break

        if moved == 0:
            stuck += 1
            if stuck >= 10:
                break
        else:
            stuck = 0
            if moved > 30:
                h = project_feasible(h, n)

        round_no += 1

    h = project_feasible(h, n)
    c2 = float(np.correlate(h, 1.0 - h, mode="full").max() * dx)
    bh = project_feasible(best_h, n)
    cb = float(np.correlate(bh, 1.0 - bh, mode="full").max() * dx)
    if c2 < cb:
        return h, c2
    return bh, cb
