import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    dx = 2.0 / n
    if k >= 0:
        grad[:nv] = dx * (1.0 - h[k:])
        grad[k:] -= dx * h[:nv]
    else:
        kk = -k
        grad[kk:] = dx * (1.0 - h[:nv])
        grad[:nv] -= dx * h[kk:]
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 100.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, top_k):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    idx = np.argsort(w)[-top_k:]
    w_top = np.zeros_like(w)
    w_top[idx] = w[idx]
    w_top /= w_top.sum()
    n_corr = len(c)
    grad = np.zeros(n)
    for i in idx:
        if w_top[i] < 1e-14:
            continue
        k = i - (n - 1)
        grad += w_top[i] * grad_lag(x, n, k)
    grad += 200.0 * (x.sum() - n / 2.0)
    return grad

def lbfgs_grad_wrap(x, n, alpha, top_k):
    return smooth_max_grad(x, n, alpha, top_k)

def binary_local_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h <= 0.5)[0]
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in ones:
            if time.time() > deadline:
                break

                best_h[i] = 0.0
                best_h[j] = 1.0
                c_new = compute_c5(best_h, n)

    return best_h, best_c
