import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_val(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = x.sum() - n / 2.0
    val += mu * s * s
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (-x[k:])
            grad[k:] += wm * (1.0 - x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[:nv] += wm * (1.0 - x[kk:])
            grad[kk:] += wm * (-x[:nv])
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = x.sum() - n / 2.0
    val += mu * s * s
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[k:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[k:])
        else:
            kk = -k
            nv = n - kk
            grad[:nv] += wm * (1.0 - x[kk:])
            grad[kk:] += wm * (-x[:nv])
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def lbfgs_optimize(h, n, deadline, max_iters=500):
    best_c = compute_c5_val(h, n)
    h = h.copy()
    bounds = [(0.0, 1.0)] * n
    configs = [(50, 0.1), (200, 1.0), (1000, 10.0),
               (5000, 100.0), (20000, 1000.0),
               (100000, 10000.0), (500000, 100000.0),
               (2000000, 500000.0), (10000000, 2000000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-20,
                                  'gtol': 1e-18, 'maxfun': max_iters * 60,
                                  'maxls': 200})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5_val(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=2000000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    def slsqp_obj(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    def slsqp_grad(x, n, alpha):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for m_idx in range(2 * n - 1):
            if w[m_idx] < 1e-14:
                continue
            k = m_idx - (n - 1)
            wm = w[m_idx] * dx
            if k >= 0:
                nv = n - k
                grad[k:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[k:])
            else:
                kk = -k
                nv = n - kk
                grad[:nv] += wm * (1.0 - x[kk:])
                grad[kk:] += wm * (-x[:nv])
        return grad
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n/2.0,
             'jac': lambda x, n=n: np.ones(n)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), jac=slsqp_grad,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5_val(h_new, n)
    except Exception:
        return h.copy(), compute_c5_val(h, n)

def grad_pair_search(h, n, deadline, top_k=40):
    best_c = compute_c5_val(h, n)
    h = h.copy()
    alphas = [500, 2000, 10000, 50000, 200000]
    for alpha in alphas:
        while time.time() < deadline:
            g = smooth_max_grad(h, n, alpha, 0.0)
            pos_idx = np.argsort(g)[::-1][:top_k]
            neg_idx = np.argsort(g)[:top_k]
            improved = False
            best_move = None
            best_new_c = best_c
            for i in pos_idx[:20]:
                if time.time() > deadline:
                    break
                for j in neg_idx[:20]:
                    if i == j:
                        continue
                    delta_max = min(1.0 - h[i], h[j])
                    if delta_max < 1e-10:
                        continue
                    for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
                        d = delta_max * f
                        h[i] += d
                        h[j] -= d
                        c = compute_c5_val(h, n)
                        if c < best_new_c:
                            best_new_c = c
                            best_move = (i, j, d)
                        else:
                            h[i] -= d
                            h[j] += d
                            break
                    else:
                        h[i] -= delta_max
                        h[j] += delta_max
                    h[i] -= d if best_move and best_move[0]==i and best_move[1]==j else 0
                    if best_move and best_move[0]==i and best_move[1]==j:
                        h[i] += best_move[2]
                        h[j] -= best_move[2]
            if best_move is not None:
                if best_new_c < best_c - 1e-15:
                    best_c = best_new_c
                    improved = True
                else:
                    i, j, d = best_move
                    h[i] -= d
                    h[j] += d
            if not improved:
                break
    return h, best_c

def local_fine_search(h, n, deadline, k_top=30):
    best_c = compute_c5_val(h, n)
    h = h.copy()
    alphas = [200000, 1000000, 5000000]
    for alpha in alphas:
        stagnation = 0
        while time.time() < deadline and stagnation < 5:
            g = smooth_max_grad(h, n, alpha, 0.0)
            pos_idx = np.argsort(g)[::-1][:k_top]
            neg_idx = np.argsort(g)[:k_top]
            found = False
            for i in pos_idx[:15]:
                if time.time() > deadline:
                    return h, best_c
                for j in neg_idx[:15]:
                    if i == j:
                        continue
                    delta_max = min(1.0 - h[i], h[j])
                    if delta_max < 1e-12:
                        continue
                    lo, hi = 0.0, delta_max
                    for _ in range(30):
                        mid = (lo + hi) / 2
                        h[i] += mid
                        h[j] -= mid
                        c = compute_c5_val(h, n)
                        h[i] -= mid
                        h[j] += mid
                        if c < best_c:
                            lo = mid
                        else:
                            hi = mid
                    d = lo
                    if d > 1e-12:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5_val(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            found = True
                            break
                if found:
                    break
            if not found:
                stagnation += 1
    return h, best_c

def fourier_start(n, seed):
    rng = np.random.RandomState(seed)
    h = np.full(n, 0.5)
    fft_h = np.fft.rfft(h)
    for k in range(1, len(fft_h)):
        fft_h[k] *= np.exp(1j * rng.uniform(0, 2 * np.pi))
    h = np.fft.irfft(fft_h, n=n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(650, 0.5)
        n_init = 650

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5_val(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS-B with fixed gradient
    t1 = min(t0 + 250, deadline - 500)
    best_h, best_c = lbfgs_optimize(best_h, best_n, t1, max_iters=600)

    # Phase 2: SLSQP polish
    if time.time() < deadline - 400:
        t2 = min(time.time() + 80, deadline - 400)
        h_s, c_s = slsqp_optimize(best_h, best_n, t2, alpha=2000000, maxiter=5000)
        if c_s < best_c:
            best_c, best_h = c_s, h_s

    # Phase 3: Gradient-guided pair search (coarse)
    if time.time() < deadline - 250:
        t3 = min(time.time() + 120, deadline - 250)
        h_p, c_p = grad_pair_search(best_h, best_n, t3, top_k=50)
        if c_p < best_c:
            best_c, best_h = c_p, h_p

    # Phase 4: Try n=500
    if time.time() < deadline - 200:
        n2 = 500
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n2)
        alloc = min(100, (deadline - time.time() - 100) * 0.5)
        if alloc > 30:
            h_t, c_t = lbfgs_optimize(h_new, n2, time.time() + alloc, max_iters=400)
            if c_t < best_c:
                best_c, best_h, best_n = c_t, h_t, n2

    # Phase 5: Fine local search (bisection line search)
    if time.time() < deadline - 30:
        t5 = min(time.time() + int((deadline - time.time()) * 0.5), deadline - 20)
        best_h, best_c = local_fine_search(best_h, best_n, t5, k_top=30)

    # Phase 6: Final SLSQP squeeze
    if time.time() < deadline - 5:
        h_f, c_f = slsqp_optimize(best_h, best_n, deadline - 2,
                                   alpha=5000000, maxiter=3000)
        if c_f < best_c:
            best_c, best_h = c_f, h_f

    # Phase 7: Ultra-fine local search
    if time.time() < deadline - 1:
        best_h, best_c = local_fine_search(best_h, best_n, deadline, k_top=20)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5_val(best_h, best_n)
    return best_h, best_c, best_n
