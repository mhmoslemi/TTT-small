def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    c = compute_c5_vec(h, n)
    return float(np.max(c))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_for_lag(h, n, k):
    """Gradient of f_k with respect to h"""
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] += 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    penalty = 100.0 * (x.sum() - n / 2.0) ** 2
    return float(val + penalty)

    prev_g = None

            y = g - prev_g

            sy = s @ y
            if sy > 1e-10:
                step = ss / sy
            else:
                yy = y @ y
                if yy > 1e-10:

import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    c = compute_c5_vec(h, n)
    return float(np.max(c))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_for_lag(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] += 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 80.0 * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, top_k=50):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n)
    for i in top_idx:
        if w[i] < 1e-14:
            continue
        k = i - (n - 1)
        grad += w[i] * dx * grad_for_lag(x, n, k)
    grad += 160.0 * (x.sum() - n / 2.0)
    return grad

def pgd_opt(h, n, deadline, top_k=30, alpha=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.02
    prev_h = None
    prev_g = None
    stall = 0
    while time.time() < deadline:
        g = smooth_grad(h, n, alpha, top_k)
        if prev_h is not None:
            s = h - prev_h
            y = g - prev_g
            ss = s @ s
            sy = s @ y
            if sy > 1e-10:
                step = min(ss / sy, 0.05)
            else:
                yy = y @ y
                if yy > 1e-10:
                    step = min(ss / yy, 0.05)
            step = max(step, 0.001)
        prev_h = h.copy()
        prev_g = g.copy()
        accepted = False
        for _ in range(15):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c + 1e-12:
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new.copy()
                    stall = 0
                else:
                    stall += 1
                h = h_new
                accepted = True
                break
            step *= 0.5
        if not accepted:
            step *= 0.5
            if step < 5e-6:
                break
        if stall > 80:
            break

def grad_guided_swaps(h, n, deadline, max_iter=3000):

    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(max_iter):

            break
        c = compute_c5_vec(best_h, n)
        m = c.max()

        grad = np.zeros(n)

        if not np.any(pos_mask) or not np.any(neg_mask):

def fine_polish(h, n, deadline, radius=35):

def structured_start(n, seed):

    for b in range(blocks):

        hi = min(n, lo + per_block)

        for i in range(lo, hi):
            if ((i - lo + shift) % (per_block)) < per_block // 2:
                h[i] = 1.0
    return project_feasible(h, n)
