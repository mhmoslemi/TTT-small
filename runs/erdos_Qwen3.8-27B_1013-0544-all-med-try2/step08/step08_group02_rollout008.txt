import numpy as np
import time
from scipy.optimize import linprog

PHI = 0.61803398874989488737
TARGET_C = 0.38075


def project_feasible(h, n):
    a = np.clip(np.asarray(h, float), 0.0, 1.0).copy()
    t = n * 0.5
    for _ in range(120):
        d = t - a.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            r = 1.0 - a
            m = r > 1e-15
            s = r[m].sum()
            if s <= 1e-15:
                break
            a[m] += (d / s) * r[m]
        else:
            m = a > 1e-15
            s = a[m].sum()
            if s <= 1e-15:
                break
            a[m] += (d / s) * a[m]
    return np.clip(a, 0.0, 1.0)


def make_counts(h, n):
    L = 2 * n - 1
    off = n - 1
    c = np.empty(L)
    om = 1.0 - h
    for k in range(n):
        l = n - k
        c[off + k] = np.dot(h[:l], om[k:])
    for k in range(1, n):
        l = n - k
        c[off - k] = np.dot(h[k:], om[:l])
    return c


def compute_O(h, n):
    L = 2 * n - 1
    off = n - 1
    O = np.zeros((n, L), dtype=float)
    om = 1.0 - h
    hrev = h[::-1]
    for p in range(n):
        row = O[p]
        row[off - p:off - p + n] += om
        row[off + p - n + 1:off + p + 1] -= hrev
    return O


def round_topm(h, n):
    h = project_feasible(h, n)
    m = n // 2
    out = np.zeros(n)
    if m > 0:
        idx = np.argpartition(h, -m)[-m:]
        out[idx] = 1.0
    return out


def golden_start(n, rng):
    m = n // 2
    x = np.arange(n, dtype=float)
    vals = (x * PHI) % 1.0
    order = np.argsort(vals)
    h = np.zeros(n)
    h[order[:m]] = 1.0
    return h


def random_start(n, rng):
    m = n // 2
    h = np.zeros(n)
    idx = rng.choice(n, size=m, replace=False)
    h[idx] = 1.0
    return h


def interval_start(n, rng, q):
    h = np.zeros(n)
    e = np.empty(q + 1)
    e[0] = 0.0
    e[-1] = 1.0
    if q > 1:
        e[1:-1] = np.sort(rng.uniform(0.0, 1.0, max(0, q - 1)))
    bit = int(rng.randint(0, 2))
    for b in range(q):
        i0 = int(e[b] * n)
        i1 = int(e[b + 1] * n)
        if i1 > i0 and ((b + bit) % 2 == 0):
            h[i0:i1] = 1.0
    return round_topm(h, n)


def interp_start(src, n):
    n0 = len(src)
    if n0 == 0:
        return golden_start(n, np.random.RandomState(0))
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x1, x0, src)
    return project_feasible(h, n)


def perturb_binary(h, n, rng, k):
    h = h.copy()
    for _ in range(k):
        on = np.where(h > 0.5)[0]
        zo = np.where(h <= 0.5)[0]
        if len(on) == 0 or len(zo) == 0:
            break
        h[on[rng.randint(len(on))]] = 0.0
        h[zo[rng.randint(len(zo))]] = 1.0
    return h


def binary_local_search(h, n, deadline, rng):
    cur = round_topm(h, n)
    cvec = make_counts(cur, n)
    cmax = float(cvec.max())
    best_h = cur.copy()
    best_c = cmax
    L = 2 * n - 1
    off = n - 1
    base = np.empty(L)
    tmp = np.empty(L)
    escapes = 0
    while time.time() < deadline:
        if best_c * (2.0 / n) <= TARGET_C:
            break
        O = compute_O(cur, n)
        improved = False
        best_move = cmax
        bi = -1
        bj = -1
        on = np.where(cur > 0.5)[0]
        zo = np.where(cur <= 0.5)[0]
        if len(on) == 0 or len(zo) == 0:
            break
        for ii in on:
            if time.time() > deadline:
                break
            np.subtract(cvec, O[ii], out=base)
            for jj in zo:
                np.add(base, O[jj], out=tmp)
                tmp[off] -= 2.0
                tmp[off + jj - ii] += 1.0
                tmp[off + ii - jj] += 1.0
                nm = float(tmp.max())
                if nm < best_move - 1e-8:
                    best_move = nm
                    bi = ii
                    bj = jj
        if bi >= 0:
            cur[bi] = 0.0
            cur[bj] = 1.0
            cvec = make_counts(cur, n)
            cmax = float(cvec.max())
            if cmax < best_c - 1e-12:
                best_c = cmax
                best_h = cur.copy()
            escapes = 0
            if best_c * (2.0 / n) <= TARGET_C:
                break
            continue
        if escapes < 3 and deadline - time.time() > 45:
            cur = perturb_binary(cur, n, rng, 2 + escapes)
            cvec = make_counts(cur, n)
            cmax = float(cvec.max())
            escapes += 1
            continue
        break
    return best_h, best_c


def lp_polish(h, n, deadline):
    h = project_feasible(h, n)
    c = make_counts(h, n)
    best_c = float(c.max())
    best_h = h.copy()
    L = 2 * n - 1
    cvec = np.zeros(n + 1)
    cvec[-1] = 1.0
    onem = np.ones((L, 1))
    Aeq = np.ones((1, n + 1))
    Aeq[0, -1] = 0.0
    iters = 30 if n <= 600 else 8
    for _ in range(iters):
        if time.time() > deadline - 3:
            break
        if best_c * (2.0 / n) <= TARGET_C:
            break
        O = compute_O(h, n)
        A = O.T
        A_ub = np.hstack((A, -onem))
        b_ub = -c
        improved = False
        for step in (0.25, 0.5):
            if time.time() > deadline - 2:
                break
            lo = np.maximum(-h, -step)
            hi = np.minimum(1.0 - h, step)
            bounds = []
            for i in range(n):
                bounds.append((float(lo[i]), float(hi[i])))
            bounds.append((0.0, None))
            res = None
            try:
                res = linprog(
                    cvec,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    A_eq=Aeq,
                    b_eq=0.0,
                    bounds=bounds,
                    method='highs',
                    options={'time_limit': max(1.0, deadline - 1.0 - time.time())}
                )
            except Exception:
                try:
                    res = linprog(cvec, A_ub=A_ub, b_ub=b_ub, A_eq=Aeq, b_eq=0.0, bounds=bounds)
                except Exception:
                    res = None
            if res is None or not getattr(res, 'success', False) or res.x is None:
                continue
            d = res.x[:n]
            if not np.all(np.isfinite(d)):
                continue
            if np.max(np.abs(d)) < 1e-12:
                continue
            for lam in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
                ht = project_feasible(h + lam * d, n)
                cn = make_counts(ht, n)
                mn = float(cn.max())
                if mn < best_c - 1e-9:
                    best_c = mn
                    best_h = ht.copy()
                    h = ht
                    c = cn
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c
