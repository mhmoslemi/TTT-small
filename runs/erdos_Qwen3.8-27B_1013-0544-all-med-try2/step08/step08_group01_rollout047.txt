import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val_and_weights(h, n, alpha, top_k=0):
    c = compute_c5_vec(h, n)
    m = c.max()
    w = np.ones(len(c))
    if top_k > 0:
        idx = np.argsort(c)[::-1]
        w[idx[:top_k]] = 20.0
        w[idx[top_k:2*top_k]] = 5.0
        w[idx[2*top_k:3*top_k]] = 2.0
    sh = alpha * w * (c - m)
    mx = sh.max()
    wexp = np.exp(sh - mx)
    wexp /= wexp.sum()
    return m, wexp

def smooth_max_obj(h, n, alpha, top_k=0):
    m, _ = smooth_max_val_and_weights(h, n, alpha, top_k)
    c = compute_c5_vec(h, n)
    sh = alpha * (c - m.max()) if top_k == 0 else None
    # Recompute properly
    c = compute_c5_vec(h, n)
    mv = c.max()
    w = np.ones(len(c))
    if top_k > 0:
        idx = np.argsort(c)[::-1]
        w[idx[:top_k]] = 20.0
        w[idx[top_k:2*top_k]] = 5.0
        w[idx[2*top_k:3*top_k]] = 2.0
    sh = alpha * w * (c - mv)
    mx2 = sh.max()
    return mv + (mx2 + np.log(np.sum(np.exp(sh - mx2)))) / alpha

def smooth_max_grad(h, n, alpha, top_k=0):
    c = compute_c5_vec(h, n)
    mv = c.max()
    w = np.ones(len(c))
    if top_k > 0:
        idx = np.argsort(c)[::-1]
        w[idx[:top_k]] = 20.0
        w[idx[top_k:2*top_k]] = 5.0
        w[idx[2*top_k:3*top_k]] = 2.0
    sh = alpha * w * (c - mv)
    mx2 = sh.max()
    wexp = np.exp(sh - mx2)
    wexp /= wexp.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if wexp[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        wm = wexp[mi] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def fourier_encode(h, n, K):
    x = np.arange(n)
    dc = h.mean()
    coeffs = np.zeros(2 * K)
    for k in range(1, K + 1):
        ck = np.cos(2.0 * np.pi * k * x / n)
        sk = np.sin(2.0 * np.pi * k * x / n)
        coeffs[2*(k-1)] = 2.0 / n * np.dot(h, ck)
        coeffs[2*(k-1)+1] = 2.0 / n * np.dot(h, sk)
    return dc, coeffs

def fourier_decode(dc, coeffs, n, K):
    x = np.arange(n)
    h = np.full(n, dc)
    for k in range(1, K + 1):
        ak = coeffs[2*(k-1)]
        bk = coeffs[2*(k-1)+1]
        h += ak * np.cos(2.0 * np.pi * k * x / n) + bk * np.sin(2.0 * np.pi * k * x / n)
    return h

def fourier_obj(params, n, K):
    h = fourier_decode(params[0], params[1:], n, K)
    h = project_feasible(h, n)
    return compute_c5(h, n)

def annealed_lbfgs(h, n, deadline, top_k=15):

                best_h = h_new
        except Exception:

    return best_h, best_c

def diverse_start(n, rng, method):
    x = np.arange(n)
    if method == 'fourier':
        h = np.full(n, 0.5)
        for k in range(1, 8):
            amp = rng.uniform(0.05, 0.3)
            phase = rng.uniform(0, 2 * np.pi)
            h += amp * np.cos(2 * np.pi * k * x / n + phase)
        return project_feasible(h, n)
    elif method == 'golden':
        h = np.zeros(n)
        half = n // 2
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        positions = np.floor(np.arange(half) * phi).astype(int) % n
        h[positions] = 1.0
        return project_feasible(h, n)
    elif method == 'blocks':
        h = np.zeros(n)
        nb = rng.randint(3, 7)
        bs = n // nb

        return project_feasible(h, n)
    else:
        h = np.zeros(n)
        half = n // 2
        idx = rng.choice(n, size=half, replace=False)
        h[idx] = 1.0
        return project_feasible(h, n)

def pair_polish(h, n, deadline, radius=40):

        while any_imp and time.time() < deadline:
            any_imp = False
            for i in range(0, n, 3):
                if time.time() > deadline - 1:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:

                        h[i] -= d

                            any_imp = True
