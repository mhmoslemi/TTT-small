def runs_start(n, rng):
    m = max(8, n // 25)
    if m % 2 == 0: m += 1
    w = rng.dirichlet(np.ones(m, dtype=float) * 2.0)
    lengths = np.floor(w * n).astype(int)
    lengths = np.maximum(1, lengths)
    total = int(lengths.sum())
    perm = rng.permutation(m)
    p = 0
    while total < n:
        idx = int(perm[p % m]); lengths[idx] += 1; total += 1; p += 1
    while total > n:
        idx = int(perm[p % m])
        if lengths[idx] > 1:
            lengths[idx] -= 1; total -= 1
        p += 1
    h=np.zeros(n)
    pos=0; state=1
    for l in lengths:
       if pos>=n: break
       vl=int(min(l,n-pos))
       if state: h[pos:pos+vl]=1.0
       pos += vl; state = 1-state
    target = int(n/2.0)
    cnt = int(h.sum())
    if cnt > target:
       ones = np.nonzero(h > 0.5)[0]
       flip = rng.choice(ones, size=cnt-target, replace=False)
       h[flip]=0.0
    elif cnt < target:
       zeros = np.nonzero(h < 0.5)[0]
       flip = rng.choice(zeros, size=target-cnt, replace=False)
       h[flip]=1.0
    return project_box_sum(h, n/2.0)

import numpy as np
import time
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize


def c5_vec(h):
    n = len(h)
    return sp_correlate(h, 1.0 - h, mode="full", method="fft") * (2.0 / n)


def c5_max(h):
    return float(np.max(c5_vec(h)))


def project_box_sum(z, target):
    z = np.asarray(z, dtype=float).copy()
    n = len(z)
    if n == 0:
        return z
    z[~np.isfinite(z)] = 0.5
    low = float(z.min()) - 1.0
    high = float(z.max())
    for _ in range(60):
        mid = 0.5 * (low + high)
        p = np.clip(z - mid, 0.0, 1.0)
        if p.sum() > target:
            low = mid
        else:
            high = mid
    p = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-13:
        order = np.arange(n)
        for t in range(10 * n):
            if abs(diff) <= 1e-13:
                break
            i = int(order[t % n])
            if diff > 0.0:
                step = min(diff, 1.0 - p[i])
            else:
                step = max(diff, -p[i])
            if step != 0.0:
                p[i] += step
                diff -= step
    diff = target - p.sum()
    if abs(diff) > 1e-13:
        idx = int(np.argmin(np.abs(p - 0.5)))
        if diff > 0.0 and p[idx] < 1.0:
            p[idx] += min(diff, 1.0 - p[idx])
        elif diff < 0.0 and p[idx] > 0.0:
            p[idx] += max(diff, -p[idx])
        p = np.clip(p, 0.0, 1.0)
    return p


def grad_single(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            g[:nv] += 1.0 - h[k:]
            g[k:] -= h[:nv]
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            g[kk:] = 1.0 - h[:nv]
            g[:nv] -= h[kk:]
    return g


def smoother_top(c, top_m, alpha):
    m = min(top_m, len(c))
    if m <= 0:
        return 0.0
    if m == len(c):
        vals = c
    else:
        idx = np.argpartition(c, len(c) - m)[-m:]
        vals = c[idx]
    base = float(vals.min())
    sh = alpha * (vals - base)
    lse = float(sh.max())
    return float(base + (lse + np.log(np.sum(np.exp(sh - lse)))) / alpha)


def top_weights(c, top_m, alpha):
    m = min(top_m, len(c))
    if m <= 0:
        return np.array([], dtype=int), np.array([], dtype=float)
    if m == len(c):
        idx = np.arange(len(c), dtype=int)
    else:
        idx = np.argpartition(c, len(c) - m)[-m:]
    vals = c[idx]
    base = float(vals.min())
    sh = alpha * (vals - base)
    sh = sh - sh.max()
    w = np.exp(sh)
    w /= np.sum(w)
    return idx, w


def lbfgs_obj_top(x, n, alpha, top_m, pen):
    c = c5_vec(x)
    val = smoother_top(c, top_m, alpha)
    s = x.sum() - n / 2.0
    val += 0.5 * pen * s * s
    return float(val)


def lbfgs_grad_top(x, n, alpha, top_m, pen):
    c = c5_vec(x)
    idx, w = top_weights(c, top_m, alpha)
    g = np.zeros(n)
    dx = 2.0 / n
    for t in range(len(idx)):
        ww = float(w[t])
        if ww > 1e-14:
            k = int(idx[t]) - (n - 1)
            g += ww * dx * grad_single(x, n, k)
    s = x.sum() - n / 2.0
    g += pen * s
    return g


def optimize_lbf_gs(h, n, deadline):
    best_c = c5_max(h)
    best_h = h.copy()
    rem = deadline - time.time()
    if rem > 700:
        configs = [(20, 20.0, 200), (16, 50.0, 180), (12, 150.0, 150), (10, 400.0, 120)]
    elif rem > 350:
        configs = [(20, 20.0, 120), (16, 50.0, 120), (12, 150.0, 100)]
    elif rem > 150:
        configs = [(16, 50.0, 80), (12, 150.0, 60)]
    else:
        configs = [(16, 50.0, 40)]
    for top_m, alpha, iters in configs:
        if time.time() > deadline:
            break
        x0 = project_box_sum(best_h, n / 2.0)
        bounds = [(0.0, 1.0)] * x0.size
        try:
            res = minimize(
                lbfgs_obj_top,
                x0,
                jac=lbfgs_grad_top,
                args=(n, alpha, top_m, 100.0),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": iters, "ftol": 1e-14, "gtol": 1e-10},
            )
            xp = project_box_sum(res.x, n / 2.0)
            c = c5_max(xp)
            if c < best_c - 1e-14:
                best_c = c
                best_h = xp
        except Exception:
            pass
    return best_h, best_c


def active_minimax_step(c, slopes, maxd):
    c = np.asarray(c, dtype=float)
    slopes = np.asarray(slopes, dtype=float)
    if maxd <= 1e-14:
        return 0.0, float(np.max(c))
    lo = 0.0
    hi = float(maxd)
    for _ in range(35):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(c + m1 * slopes))
        f2 = float(np.max(c + m2 * slopes))
        if f1 <= f2:
            hi = m2
        else:
            lo = m1
    d = 0.5 * (lo + hi)
    bestf = float(np.max(c + d * slopes))
    bestd = d
    f0 = float(np.max(c))
    if f0 < bestf - 1e-15:
        bestd = 0.0
        bestf = f0
    fh = float(np.max(c + hi * slopes))
    if fh < bestf - 1e-15:
        bestd = hi
        bestf = fh
    return bestd, bestf


def active_pass(h, n, best_c, top_r, probe, deadline, full_limit, max_accept):

    r = min(top_r, max(1, len(cvec)))

    C = cvec[idx]
    dx = 2.0 / n
    G = np.empty((r, n), dtype=float)
    for t in range(r):

        G[t] = grad_single(h, n, k) * dx
    if probe <= min_step:
        probe = min_step
    full_evals = 0

        if max_avail <= min_step:

        if pi <= min_step:
            continue
        scores = np.max(C[:, None] + pi * (G[:, i][:, None] - G), axis=0)

        allowed[i] = False
        cnt_allowed = int(np.count_nonzero(allowed))
        if cnt_allowed == 0:

            if full_evals >= full_limit or accepted >= max_accept:

            j = int(j)
            room = 1.0 - float(h[i])

            if md <= min_step:
                continue

            d, fmin = active_minimax_step(C, slopes, md)
            if d <= min_step or fmin >= best_c - 1e-10:
                continue
            orig_i = float(h[i])
            orig_j = float(h[j])
            best_local = best_c
            best_step = 0.0
            for frac in (1.0, 0.5, 0.25):
                if full_evals >= full_limit or time.time() > deadline:
                    break
                step = frac * d
                if step <= min_step:
                    continue
                if step > md:
                    step = md
                h[i] = orig_i + step
                h[j] = orig_j - step
                c = c5_max(h)
                full_evals += 1
                if c < best_local - 1e-14:
                    best_local = c
                    best_step = step
            h[i] = orig_i
            h[j] = orig_j
            if best_step > 0.0:
                h[i] = orig_i + best_step
                h[j] = orig_j - best_step
                best_c = best_local
                accepted += 1
    return h, best_c, accepted

def pair_sweep(h, n, deadline, radius, deltas):
    best_c = c5_max(h)
    for d in deltas:
        if time.time() >= deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() >= deadline:
                    break
                if h[i] >= 1.0 - d:
                    continue
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[j] >= d:
                        h[i] += d
                        h[j] -= d
                        c = c5_max(h)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                            if h[i] >= d and h[j] <= 1.0 - d:
                                h[i] -= d
                                h[j] += d
                                c = c5_max(h)
                                if c < best_c - 1e-14:
                                    best_c = c
                                    improved = True
                                else:
                                    h[i] += d
                                    h[j] -= d
    return h, best_c


def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_box_sum(h_new, n_new / 2.0)


def refine_candidate_n(h_best, n_best, n_new, time_end):
    if n_best == n_new:
        return None
    ht = interpolate_to_n(h_best, n_best, n_new)
    t_opt = min(time_end - 40.0, time.time() + 120.0)
    if time.time() < t_opt:
        ht, ct = optimize_lbf_gs(ht, n_new, t_opt)
    else:
        ct = c5_max(ht)
    if time.time() < time_end - 30.0:
        t_act = min(time_end - 5.0, time.time() + 60.0)
        ht, ct = active_pair_polish(ht, n_new, t_act)
    ht = project_box_sum(ht, n_new / 2.0)
    ct = c5_max(ht)
    if ct < c5_max(h_best) - 1e-12:
        return ht, ct, n_new
    return None


def runs_start(n, rng):
    m = max(8, n // 25)
    if m % 2 == 0:
        m += 1
    w = rng.dirichlet(np.ones(m, dtype=float) * 2.0)
    lengths = np.floor(w * n).astype(int)
    lengths = np.maximum(1, lengths)
    total = int(lengths.sum())
    perm = rng.permutation(m)
    p = 0
    while total < n:
        i = int(perm[p % m])
        lengths[i] += 1
        total += 1
        p += 1
    while total > n:
        i = int(perm[p % m])
        if lengths[i] > 1:
            lengths[i] -= 1
            total -= 1
        p += 1
    h = np.zeros(n)
    pos = 0
    state = 1
    for l in lengths:
        if pos >= n:
            break
        vl = int(min(lengths[pos] if pos < len(lengths) else l, n - pos))
        if state:
            h[pos:pos + vl] = 1.0
        pos += vl
        state = 1 - state
    target = int(n / 2.0)
    cnt = int(h.sum())
    if cnt > target:
        ones = np.nonzero(h > 0.5)[0]
        if ones.size:
            flip = rng.choice(ones, size=min(cnt - target, ones.size), replace=False)
            h[flip] = 0.0
    elif cnt < target:
        zeros = np.nonzero(h < 0.5)[0]
        if zeros.size:
            flip = rng.choice(zeros, size=min(target - cnt, zeros.size), replace=False)
            h[flip] = 1.0
    return project_box_sum(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 4.0
    rng = np.random.RandomState(int(seed))

    try:
        h_init = np.asarray(initial_h_values, dtype=float)
        if h_init.ndim != 1:
            h_init = h_init.ravel()
        if h_init.size == 0:
            raise ValueError
    except Exception:
        h_init = runs_start(500, rng)

    n = int(h_init.size)
    h = project_box_sum(h_init, n / 2.0)
    best_c = c5_max(h)

    hs = 0.5 * (h + h[::-1])
    hs = project_box_sum(hs, n / 2.0)
    hs_c = c5_max(hs)
    if hs_c < best_c:
        h = hs
        best_c = hs_c

    lbf_time = min(time.time() + 200.0, deadline - 450.0)
    if lbf_time < time.time() + 30.0:
        lbf_time = min(time.time() + 30.0, deadline - 80.0)
    if time.time() < lbf_time and deadline - time.time() > 60.0:
        h, best_c = optimize_lbf_gs(h, n, lbf_time)

    if best_c > 0.38083 and deadline - time.time() > 700.0:
        alt_n = 997 if n <= 800 else 700
        if alt_n != n:
            cand = refine_candidate_n(h, n, alt_n, time.time() + 170.0)
            if cand is not None:
                h, best_c, n = cand

    active_end = deadline - 70.0
    if active_end < time.time() + 30.0:
        active_end = deadline - 1.0
    if time.time() < active_end:
        h, best_c = active_pair_polish(h, n, active_end)

    if time.time() < deadline - 2.0:
        radius = 30 if n > 800 else 40
        h, best_c = pair_sweep(
            h,
            n,
            deadline - 1.0,
            radius,
            (1e-4, 5e-5, 2e-5),
        )

    hs = 0.5 * (h + h[::-1])
    hs = project_box_sum(hs, n / 2.0)
    hs_c = c5_max(hs)
    if hs_c < best_c:
        h = hs
        best_c = hs_c

    h = project_box_sum(h, n / 2.0)
    best_c = c5_max(h)
    return h, float(best_c), int(n)
