def true_subgradient(h, n, max_iters=10000, beta=0.9):
    """Momentum-based subgradient on the true max."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    c = compute_correlations(h, n)
    
    # Find the max lag
    max_idx = np.argmax(c)
    k = max_idx - (n - 1)
    
    v = np.zeros(n)
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for t in range(max_iters):
        # Compute gradient for the max lag
        g = np.zeros(n)
        if k >= 0:
            nv = n - k
            g[:nv] += (1.0 - h[k:]) * dx
            g[k:] += (-h[:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            g[kk:] += (1.0 - h[:nv]) * dx
            g[:nv] += (-h[kk:]) * dx
            
        # Update momentum
        v = beta * v + (1 - beta) * g
        
        # Try step sizes
        best_step_c = best_c
        best_h = h.copy()
        for alpha in step_sizes:
            h_new = h - alpha * v
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_step_c:
                best_step_c = c_new
                best_h = h_new
        
        if best_step_c < best_c - 1e-15:
            h = best_h
            best_c = best_step_c
            # Recompute max lag
            c = compute_correlations(h, n)
            max_idx = np.argmax(c)
            k = max_idx - (n - 1)
        else:
            # No improvement, try smaller steps or break
            step_sizes = [s * 0.5 for s in step_sizes]
            if step_sizes[0] < 1e-6:
                break
                
    return h, best_c

                    if i == j:

                    for move in [(i, j, d), (j, i, d)]:
                        a, b, s = move
                        if h[a] + s <= 1.0 and h[b] - s >= 0.0:
                            h[a] += s
                            h[b] -= s
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                any_imp = True
                            else:
                                h[a] -= s
                                h[b] += s
                if any_imp:
                    break
            if not any_imp:
                break
    return h, best_c

def multi_res_optimize(h_best, n_best, deadline, rng):
    best_h, best_c = h_best.copy(), compute_c5(h_best, n_best)
    for n_coarse in [300, 200]:
        if time.time() > deadline - 60:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_coarse, endpoint=False)
        h_c = np.interp(x_new, x_old, best_h)
        h_c = project_feasible(h_c, n_coarse)
        alloc = min(40, max(20, (deadline - time.time()) * 0.25))
        h_c, c_c = lbfgs_optimize(h_c, n_coarse, time.time() + alloc,
                                  alphas=[(500, 2.0), (5000, 20.0), (50000, 200.0)])
        if c_c * (n_best / n_coarse) * 0.95 < best_c:
            x_refine = np.linspace(0, 2, n_best, endpoint=False)
            h_up = np.interp(x_refine, x_new, h_c)
            h_up = project_feasible(h_up, n_best)
            c_up = compute_c5(h_up, n_best)
            if c_up < best_c:
                best_h, best_c = h_up, c_up
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Quick L-BFGS-B
    t1 = min(t0 + 80, deadline - 600)
    best_h, best_c = lbfgs_optimize(best_h, n_init, t1,
        alphas=[(500, 2.0), (5000, 20.0), (50000, 200.0), (500000, 2000.0)])
    
    # Phase 2: Momentum subgradient (main workhorse)
    if time.time() < deadline - 300:
        t2 = min(t0 + 550, deadline - 300)
        h_m, c_m = momentum_subgrad(best_h, n_init, t2, top_k=5)
        if c_m < best_c:
            best_c, best_h = c_m, h_m
    
    # Phase 3: High-alpha L-BFGS-B polish
    if time.time() < deadline - 200:
        t3 = min(time.time() + 120, deadline - 200)
        h_p, c_p = lbfgs_optimize(best_h, n_init, t3,
            alphas=[(200000, 200.0), (1000000, 1000.0), (5000000, 5000.0)])
        if c_p < best_c:
            best_c, best_h = c_p, h_p
    
    # Phase 4: Gradient-guided local moves
    if time.time() < deadline - 150:
        t4 = min(time.time() + 100, deadline - 150)
        h_l, c_l = grad_guided_local(best_h, n_init, t4, k_top=25)
        if c_l < best_c:
            best_c, best_h = c_l, h_l
    
    # Phase 5: Multi-resolution diversification
    if time.time() < deadline - 100:
        h_mr, c_mr = multi_res_optimize(best_h, n_init, deadline - 80, rng)
        if c_mr < best_c:
            best_c, best_h = c_mr, h_mr
    
    # Phase 6: Second momentum pass on best
    if time.time() < deadline - 40:
        t6 = min(time.time() + 60, deadline - 30)
        h_m2, c_m2 = momentum_subgrad(best_h, n_init, t6, top_k=3)
        if c_m2 < best_c:
            best_c, best_h = c_m2, h_m2
    
    # Phase 7: Final L-BFGS-B squeeze
    if time.time() < deadline - 10:
        h_f, c_f = lbfgs_optimize(best_h, n_init, deadline - 5,
            alphas=[(1000000, 1000.0), (5000000, 5000.0)])
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    
    best_h = project_feasible(best_h, n_init)
    best_c = compute_c5(best_h, n_init)
    return best_h, best_c, n_init
