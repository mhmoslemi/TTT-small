def coord_descent(h, n, max_sweeps=300, deadline=None, n_bisect=35):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            lo_v = min(val0, val1) - 1e-10
            hi_v = max(val0, val1) + 1e-10
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(n_bisect):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    t = (mid - corr[neg_B]) / B[neg_B]
                    lb_x = max(lb_x, h[i] + t.max())
                ub_x = 1.0
                if pos_B.any():
                    t = (mid - corr[pos_B]) / B[pos_B]
                    ub_x = min(ub_x, h[i] + t.min())
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                t = (hi_v - corr[neg_B]) / B[neg_B]
                lb_x = max(lb_x, h[i] + t.max())
            ub_x = 1.0
            if pos_B.any():
                t = (hi_v - corr[pos_B]) / B[pos_B]
                ub_x = min(ub_x, h[i] + t.min())
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-14:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
                improved = True
        if not improved:
            break
    return h, corr

                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
    return best_h, best_c
