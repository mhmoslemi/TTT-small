                            sign = 1 if d > 0 else -1
                            t = line_search_move(h, n, i, j, sign, best_c)
                            if t > 1e-8:
                                h[i] += sign * t
                                h[j] -= sign * t
                                best_c = compute_c5(h, n)
                        else:
                            h[i] -= d
                            h[j] += d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS with progressive alpha
    p1 = min(t0 + 180, deadline - 400)
    best_h, best_c = lbfgs_optimize(best_h, best_n, p1, max_iters=300)

    # Phase 2: SLSQP
    if time.time() < deadline - 250:
        for a in [100000, 500000, 2000000]:
            if time.time() > deadline - 200:
                break
            h2, c2 = slsqp_optimize(best_h, best_n, min(time.time() + 40, deadline - 180),
                                    alpha=a, maxiter=600)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Gradient-guided search (exploits which coordinates matter)
    if time.time() < deadline - 150:
        p3 = min(time.time() + 150, deadline - 130)
        best_h, best_c = gradient_guided_search(best_h, best_n, p3, step=0.005)

    # Phase 4: Pair search with line search
    if time.time() < deadline - 100:
        p4 = min(time.time() + 150, deadline - 80)
        best_h, best_c = pair_search(best_h, best_n, p4, radius=35)

    # Phase 5: Try other resolutions
    if time.time() < deadline - 100:
        for n_try in [150, 200, 250, 400, 500]:
            if time.time() > deadline - 50:
                break
            if n_try == best_n:
                continue
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(60, deadline - time.time() - 40)
            if alloc < 25:
                break
            h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc * 0.5, max_iters=200)
            if time.time() < deadline - 20:
                h_t, c_t = slsqp_optimize(h_t, n_try, min(time.time() + 25, deadline - 15),
                                          alpha=500000, maxiter=400)
            if time.time() < deadline - 10:
                h_t, c_t = pair_search(h_t, n_try, min(time.time() + 15, deadline - 5), radius=30)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try

    # Phase 6: Final polish
    if time.time() < deadline - 5:
        nf = len(best_h)
        best_h, best_c = lbfgs_optimize(best_h, nf, deadline - 5, max_iters=80)
    if time.time() < deadline - 2:
        best_h, best_c = pair_search(best_h, len(best_h), deadline - 1, radius=25)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
