    # We'll create a dict or sparse representation of changes
    # Actually for efficiency, let's just return the affected m range and compute the new max
    
    # Affected m range: union of all four ranges
    # i outer: [n-1-i, 2n-2-i]
    # i inner: [i, min(i+n-1, 2n-2)]
    # j outer: [n-1-j, 2n-2-j]
    # j inner: [j, min(j+n-1, 2n-2)]
    
    m_min = max(0, min(n-1-i, i, n-1-j, j))
    m_max = min(2*n-2, max(2*n-2-i, i+n-1, 2*n-2-j, j+n-1))
    
    # For each m in [m_min, m_max], compute the change
    # This is O(n) per swap
    
    delta = np.zeros(2*n-1)
    
    # i as outer: m in [n-1-i, 2n-2-i]
    lo = max(n-1-i, 0)
    hi = min(2*n-2-i, 2*n-2)
    partners = np.arange(lo, hi+1) + (i - n + 1)  # i + m - (n-1) for m in [lo, hi]
    # partners = i+m-(n-1) = m + i - n + 1
    # For m=lo: partner = n-1-i+i-n+1 = 0 ✓
    # For m=hi: partner = 2n-2-i+i-n+1 = n-1 ✓
    delta[lo:hi+1] += h[partners] - 1  # change = -(1-h[partner]) = h[partner]-1

def binary_local_search(h, n, deadline, rng, verbose=False):
    """Iterative improvement for binary h."""
    C = np.correlate(h, 1-h, mode='full')
    best_max = C.max()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    no_improve_count = 0
    rounds = 0
    
    while time.time() < deadline:
        rounds += 1
        best_swap_max = best_max
        best_i, best_j = -1, -1
        
        # Sample pairs
        n_try = min(len(ones) * len(zeros), 10000)
        for _ in range(n_try):
            i_idx = rng.randint(len(ones))
            j_idx = rng.randint(len(zeros))
            i = ones[i_idx]
            j = zeros[j_idx]
            
            # Try swap
            h[i] = 0; h[j] = 1
            new_C = np.correlate(h, 1-h, mode='full')
            new_max = new_C.max()
            h[i] = 1; h[j] = 0
            
            if new_max < best_swap_max - 1e-15:
                best_swap_max = new_max
                best_i, best_j = i, j
        
        if best_i >= 0:
            # Apply best swap
            h[best_i] = 0; h[best_j] = 1
            C = np.correlate(h, 1-h, mode='full')
            best_max = C.max()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 10:
                break
    
    return h, best_max
