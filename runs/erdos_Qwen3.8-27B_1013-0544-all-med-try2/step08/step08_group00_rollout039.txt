One more idea: **dual approach**. The problem can be written as:
min_h max_k c_k(h)

By minimax duality (Sion's theorem, if the problem were convex), we could write:
min_h max_k c_k(h) = max_{p ∈ Δ} min_h sum_k p_k * c_k(h)

where Δ is the probability simplex over k. Since the problem is non-convex, this equality doesn't hold in general, but it gives a lower bound:
max_{p ∈ Δ} min_h sum_k p_k * c_k(h) ≤ min_h max_k c_k(h)

The inner minimization is a QP (quadratic objective with box constraints and a linear equality constraint), which can be solved efficiently!

This suggests a "cutting plane" or "column generation" approach:
1. Start with a distribution p over lags
2. Solve the QP: min_h sum_k p_k * c_k(h) s.t. 0≤h≤1, sum(h)=n/2
3. Compute c_k(h) for the solution h
4. Update p to put more weight on the lags with highest c_k
5. Repeat

This is essentially the multiplicative weight update / boosting approach mentioned in the strategy!

The QP in step 2: min_h sum_k p_k * [sum_j h[j] - h[j]*h[j+k_k]] s.t. 0≤h≤1, sum(h)=n/2

This is a non-convex QP (because of the -h[j]*h[j+k] terms). But we can use scipy's minimize with SLSQP or trust-constr to solve it.

Actually, the objective sum_k p_k * c_k(h) = sum_k p_k * sum_j h[j](1-h[j+k_k])
= sum_j h[j] * sum_k p_k * (1-h[j+k_k])
= sum_j h[j] * (1 - sum_k p_k * h[j+k_k])
= sum_j h[j] - sum_j h[j] * (sum_k p_k * h[j+k_k])

This is still non-convex (bilinear terms). But it's a specific structure that might be tractable.

Alternatively, I can use the QP solver to solve a linearized version:
At the current h*, linearize c_k around h*:
c_k(h) ≈ c_k(h*) + sum_i G*[i,k] * (h[i] - h*[i])

Then: min_h sum_k p_k * [c_k(h*) + sum_i G*[i,k] * (h[i]-h*[i])]

This is a LINEAR program! (Linear objective, linear constraints, box constraints.)
min_h sum_i h[i] * (sum_k p_k * G*[i,k]) s.t. 0≤h≤1, sum(h)=n/2

This can be solved in O(n log n) by sorting (give h=1 to the n/2 coordinates with smallest coefficients, h=0 to the rest).

import numpy as np
from scipy.optimize import minimize
import time

# Global state for optimizer callbacks
_state = {}

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_grad_matrix(h, n):
    n2 = 2 * n - 1
    G = np.zeros((n, n2))
    for i in range(n):
        m_lo = n - 1 - i
        m_hi = n - 1 + i
        ks = np.arange(m_lo, m_hi + 1) - (n - 1)
        G[i, m_lo:m_hi + 1] += (1.0 - h[i + ks])
        m_lo2 = i
        m_hi2 = n - 1 + i
        ks2 = np.arange(m_lo2, m_hi2 + 1) - (n - 1)
        G[i, m_lo2:m_hi2 + 1] -= h[i - ks2]
    return G

def _obj(x):
    n, alpha = _state['n'], _state['alpha']
    c = compute_c5_vec(x, n)
    ac = alpha * c
    mx = ac.max()
    return float(mx / alpha + np.log(np.sum(np.exp(ac - mx))) / alpha)

def _grad(x):
    n, alpha = _state['n'], _state['alpha']
    c = compute_c5_vec(x, n)
    ac = alpha * c
    mx = ac.max()
    w = np.exp(ac - mx)
    w /= w.sum()
    G = build_grad_matrix(x, n)
    return G.T @ w

def _eq(x):
    return float(x.sum() - _state['n'] / 2.0)

def _eq_jac(x):
    return np.ones(_state['n'])

def lbfgs_optimize(h, n, deadline, max_iters=300):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        _state['n'] = n
        _state['alpha'] = alpha
        try:
            res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': max_iters,
                          'ftol': 1e-16, 'gtol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=2000000, maxiter=5000):
    _state['n'] = n
    _state['alpha'] = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eq_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def fast_local_search(h, n, deadline, steps=None):

    best_c = float(c.max())
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                 5e-5, 2e-5, 1e-5, 5e-6, 1e-6]

            G = build_grad_matrix(h, n)
            any_imp = False
            valid_i = h >= d

                v = c - d * G[i]

def mwt_lp_search(h, n, deadline):

    c = compute_c5_vec(h, n)
    best_c = float(c.max())
    n2 = 2 * n - 1
    p = np.ones(n2) / n2
    target = n / 2.0
    
    for iteration in range(2000):
        if time.time() > deadline:
            break
        # Update weights
        c_shift = c - c.max()
        p = p * np.exp(10.0 * c_shift)
        p /= p.sum()
        
        # Compute LP direction: w[i] = sum_k p[k] * G[i,k]
        G = build_grad_matrix(h, n)
        w = G @ p  # w[i] = sum_k G[i,k] * p[k] = dc/dh[i] weighted by p

        order = np.argsort(w)

        if remainder > 1e-10:
            h_lp[order[int(target)]] = remainder

            h = project_feasible(h, n)
            c = compute_c5_vec(h, n)
            best_c = float(c.max())
    return h, best_c
