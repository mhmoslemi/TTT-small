```python
def fast_binary_swap(h, n, deadline, max_sweeps=50000, seed=42, use_anneal=False):
    dx = 2.0 / n
    h = initial_binary(h, n)
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_max = corr.max()
    best_c = cur_max * dx
    best_h = h.copy()
    profiles = make_profiles(h, n)
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        best_new_max = cur_max + 1
        best_i, best_j = -1, -1
        
        for i in zeros_idx:
            base = corr - profiles[i]
            new_maxes = (base[None, :] + profiles[ones_idx]).max(axis=1)
            j_loc = np.argmin(new_maxes)
            if new_maxes[j_loc] < best_new_max:
                best_new_max = new_maxes[j_loc]
                best_i = i
                best_j = ones_idx[j_loc]
        
        if best_i < 0 or best_new_max >= cur_max - 1e-10:
            if use_anneal:
                # random kick
                i = rng.choice(zeros_idx)
                j = rng.choice(ones_idx)
                h[i] = 1.0; h[j] = 0.0
                corr = np.correlate(h, 1-h, mode='full').astype(float)
                # rebuild profiles
                profiles = make_profiles(h, n)
                cur_max = corr.max()
                continue
            else:
                break
        
        # Accept the swap (verified approximately; cross terms may make it slightly worse)
        h[best_i] = 1.0
        h[best_j] = 0.0
        corr = np.correlate(h, 1-h, mode='full').astype(float)
        cur_max = corr.max()
        # Update profiles incrementally
        update_profiles(profiles, best_i, 1.0, n)
        update_profiles(profiles, best_j, -1.0, n)
        
        if cur_max * dx < best_c:
            best_c = cur_max * dx
            best_h = h.copy()
    
    return best_h, best_c
```
