The "steepest feasible descent" is:
- At each step, find the direction d that most reduces max(c + t*d) for small t
- Subject to: ||d||₁ = 1 (transfer from one coord to another), d preserves sum (sum(d)=0)
- The best direction is: decrease the coordinate with highest "damage" and increase the one with lowest "damage"
- "Damage" of coordinate i = grad[i] (from smooth max gradient)

So the steepest feasible descent is:
- Find i* = argmin grad (most beneficial to increase)
- Find j* = argmax grad (most beneficial to decrease)
- Transfer a small amount from j* to i*

This is a simple O(n) operation per step, and we can do millions of steps!

                    b, c2, f2 = c2, c1, f1
                    c1 = b - gr * (b - a)
                    f1 = float(np.max(c + c1 * w))
                else:
                    a, c1, f1 = c1, c2, f2
                    c2 = a + gr * (b - a)
                    f2 = float(np.max(c + c2 * w))
            
            delta_opt = (a + b) / 2.0
            f_opt = float(np.max(c + delta_opt * w))
            
            if f_opt < best_c - 1e-13:
                h[i] = np.clip(h[i] + delta_opt, 0.0, 1.0)
                c += delta_opt * w
                best_c = f_opt
                improved = True
        
        if not improved:
            break
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    return best_h, best_c

def golden_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = (np.arange(half) * phi).astype(int) % n
    h[positions] = 1.0
    return project_feasible(h, n)

def fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 15):
        amp = rng.uniform(0.05, 0.3) / (freq ** 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = golden_start(500)
        n0 = 500
    
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_n = n0
    
    # Phase 1: L-BFGS on initial (40% budget)
    t1 = t0 + (deadline - t0) * 0.35
    h1, c1 = lbfgs_phase(best_h.copy(), best_n, t1)
    if c1 < best_c:
        best_c, best_h = c1, h1
    
    # Phase 2: Steepest feasible descent (25% budget)
    t2 = min(time.time() + (deadline - t0) * 0.25, deadline - (deadline - t0) * 0.3)
    h2, c2 = steepest_feasible(best_h.copy(), best_n, t2, alpha=200000)
    if c2 < best_c:
        best_c, best_h = c2, h2
    
    # Phase 3: Coordinate refinement (15% budget)
    t3 = min(time.time() + (deadline - t0) * 0.15, deadline - (deadline - t0) * 0.15)
    h3, c3 = coord_refine(best_h.copy(), best_n, t3, max_sweeps=80)
    if c3 < best_c:
        best_c, best_h = c3, h3
    
    # Phase 4: Try higher resolution
    if time.time() < deadline - 120:
        for n_try in [600, 1000]:
            if time.time() > deadline - 80:
                break
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h, right=best_h[-1], left=best_h[0])
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(80, (deadline - time.time() - 40) * 0.5)
            if alloc < 20:
                continue
            h_hi, c_hi = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.6)
            if time.time() < deadline - 30:
                h_hi, c_hi = steepest_feasible(h_hi, n_try, min(time.time() + 30, deadline - 20), alpha=50000)
            if c_hi < best_c:
                best_c, best_h, best_n = c_hi, h_hi, n_try
    
    # Phase 5: Diversified restarts
    if time.time() < deadline - 100:
        starts = [golden_start(best_n), fourier_start(best_n, rng)]
        for s in starts:
            if time.time() > deadline - 50:
                break
            alloc = min(40, (deadline - time.time() - 30) * 0.5)
            if alloc < 15:
                break
            h_r, c_r = lbfgs_phase(s, best_n, time.time() + alloc * 0.7)
            if c_r < best_c:
                best_c, best_h = c_r, h_r
    
    # Phase 6: Final fine polish
    if time.time() < deadline - 10:
        t6 = min(time.time() + 30, deadline - 5)
        h_f, c_f = steepest_feasible(best_h.copy(), best_n, t6, alpha=500000, max_steps=200000)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    
    if time.time() < deadline - 3:
        h_f2, c_f2 = coord_refine(best_h.copy(), best_n, deadline - 1, max_sweeps=30)
        if c_f2 < best_c:
            best_c, best_h = c_f2, h_f2
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
