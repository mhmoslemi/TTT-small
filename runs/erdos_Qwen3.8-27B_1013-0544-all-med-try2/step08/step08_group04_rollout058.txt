- Since corr[p] ≈ best_c (it's the max), we need R[j, p] - R[i, p] < ε/δ where ε is small
- So we want R[i, p] >> R[j, p]

2. For each j (source), find i (destination) that maximizes R[i, p] - R[j, p]
3. Try that move, check if it actually reduces the max
4. If yes, accept and recompute
5. If no, try next best

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def corr_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = corr_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m]
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def penalty_obj(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):

    g += 2.0 * mu * (x.sum() - n / 2.0)

                         args=(n, alpha, mu), method='L-BFGS-B',

def build_full_R(h, n):
    """R[i, m] = (1 - h[i-k_m]) - h[i+k_m] where k_m = m - (n-1)."""
    R = np.zeros((n, 2*n-1))
    idx = np.arange(n)
    for m in range(2*n-1):
        k = m - (n-1)
        a = idx - k  # i - k for all i
        b = idx + k  # i + k for all i
        va = np.zeros(n)
        vb = np.zeros(n)
        mask_a = (a >= 0) & (a < n)
        mask_b = (b >= 0) & (b < n)
        va[mask_a] = h[a[mask_a]]
        vb[mask_b] = h[b[mask_b]]
        R[:, m] = (1.0 - va) - vb
    return R

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def corr_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_R(h, n):
    R = np.zeros((n, 2 * n - 1))
    idx = np.arange(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        a = idx - k
        b = idx + k
        va = np.zeros(n)
        vb = np.zeros(n)
        ma = (a >= 0) & (a < n)
        mb = (b >= 0) & (b < n)
        va[ma] = h[a[ma]]
        vb[mb] = h[b[mb]]
        R[:, m] = (1.0 - va) - vb
    return R

def grad_smooth_max(h, n, alpha):
    c = corr_vec(h, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m]
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def p_obj(x, n, alpha, mu):
    c = corr_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def p_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_opt(h, n, deadline, max_iters=150):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(200, 5.0), (2000, 20.0), (20000, 200.0), (200000, 2000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(p_obj, h, jac=p_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iters * 5, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fast_pair_search(h, n, deadline, n_peaks=6, n_cand=12):
    best_c = compute_c5(h, n)
    corr = corr_vec(h, n)
    R = build_R(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    rng = np.random.RandomState(123)
    
    while time.time() < deadline:
        peak_indices = np.argsort(corr)[-n_peaks:]
        improved = False
        for d in deltas:
            if time.time() > deadline:
                return h, best_c
            valid_src = np.where(h >= d)[0]
            valid_dst = np.where(h + d <= 1.0 - 1e-12)[0]
            if len(valid_src) == 0 or len(valid_dst) == 0:
                continue
            for pi in peak_indices:
                if time.time() > deadline:
                    return h, best_c
                rs = R[valid_src, pi]
                rd = R[valid_dst, pi]
                i_order = np.argsort(rd)[::-1][:n_cand]
                j_order = np.argsort(rs)[:n_cand]
                for ii in i_order:
                    i = valid_dst[ii]
                    for jj in j_order:
                        j = valid_src[jj]
                        if i == j:
                            continue
                        if rd[ii] - rs[jj] <= 1e-12:
                            continue
                        new_corr = corr + d * (R[i, :] - R[j, :])
                        new_max = new_corr.max()
                        if new_max < best_c - 1e-14:
                            h[i] += d
                            h[j] -= d
                            best_c = new_max
                            corr = new_corr
                            R = build_R(h, n)
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            # Try random kicks to escape local optima
            if rng.random() < 0.3:
                n_kick = max(3, n // 20)
                kick_idx = rng.choice(n, size=n_kick, replace=False)
                kick_mag = rng.choice([0.01, 0.02, 0.05]) * rng.choice([-1, 1], size=n_kick)
                h_trial = h.copy()
                h_trial[kick_idx] += kick_mag
                h_trial = project_feasible(h_trial, n)
                c_trial = compute_c5(h_trial, n)
                if c_trial < best_c * 1.005:
                    h = h_trial
                    best_c = min(best_c, c_trial)
                    corr = corr_vec(h, n)
                    R = build_R(h, n)
            else:
                break
    return h, best_c

    # Phase 5: Try another n
    if time.time() < deadline - 60:
        for n_try in [250, 400]:
            if time.time() > deadline - 30:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)

    # Phase 6: Final refinement

        best_h, best_c = fast_pair_search(best_h, len(best_h), deadline, n_peaks=10, n_cand=25)
    
    best_h = project_feasible(best_h, len(best_h))
