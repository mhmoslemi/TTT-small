def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        softmax = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        return softmax
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        if time.time() > deadline:
            return h
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        if time.time() > deadline:
            return h
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def fast_binary_search(h_init, n, deadline, seed=42, n_restarts=200):
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    nx = 2 * n - 1
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    restarts = 0
    stale = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_ones = len(ones)
        # Precompute B matrix: B[p,k] = -(1-h[ones[p]+k]) + h[ones[p]-k]
        B = np.zeros((n_ones, nx))
        jm = ones[:, None] + m_range[None, :]
        jmn = ones[:, None] - m_range[None, :]
        msk_j = (jm >= 0) & (jm < n)
        msk_jn = (jmn >= 0) & (jmn < n)
        both = msk_j & msk_jn
        B[both] = -1.0 + h[jm[both]] + h[jmn[both]]
        only_j = msk_j & ~msk_jn
        B[only_j] = -1.0 + h[jm[only_j]]
        only_jn = ~msk_j & msk_jn
        B[only_jn] = h[jmn[only_jn]]
        improved = False
        # Shuffle zeros for diversity
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                return best_h, best_c
            A = np.zeros(nx)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            A[msk] = (1.0 - h[im[msk]])
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            A[msk] -= h[imn[msk]]
            candidate = corr[None, :] + A[None, :] + B
            candidate[:, n - 1] -= 2.0
            k1s = n - 1 + ones - i
            k2s = n - 1 + i - ones
            v1 = (k1s >= 0) & (k1s < nx)
            v2 = (k2s >= 0) & (k2s < nx)
            ar = np.arange(n_ones)
            candidate[ar[v1], k1s[v1]] += 2.0
            candidate[ar[v2], k2s[v2]] += 2.0
            new_maxes = candidate.max(axis=1)
            best_p = new_maxes.argmin()
            if new_maxes[best_p] * dx < cur_c - 1e-14:
                h[i] = 1.0
                h[ones[best_p]] = 0.0
                corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
                cur_c = corr.max() * dx
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
                stale = 0
                break
        if not improved:
            stale += 1
            if stale > 3 and restarts < n_restarts:
                restarts += 1
                n_rand = rng.randint(3, 10)
                for _ in range(n_rand):
                    oi = np.where(h == 1)[0]
                    zi = np.where(h == 0)[0]
                    if len(oi) > 0 and len(zi) > 0:
                        a = rng.choice(oi)
                        b = rng.choice(zi)
                        h[a], h[b] = h[b], h[a]
                corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
                cur_c = corr.max() * dx
            elif stale > 10:
                # More aggressive restart
                restarts += 1
                h[:] = 0
                new_idx = rng.choice(n, size=target, replace=False)
                h[new_idx] = 1.0
                corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
                cur_c = corr.max() * dx
                stale = 0
    return best_h, best_c

def multi_slsqp(h_bin, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    best = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    # Try several noise levels and alpha values
    noises = [0.02, 0.05, 0.1]
    alphas = [5000, 50000, 500000, 5000000]
    for noise in noises:
        if time.time() > deadline:
            break
        hp = project_feasible(h_bin + noise * rng.randn(n), n)
        for alpha in alphas:
            if time.time() > deadline:
                break
            hp = slsqp_smooth(hp, n, alpha, 2000, deadline)
            c = compute_c5(hp, n)
            if c < best:
                best = c
                best_h = hp.copy()
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Quick SLSQP warmup
    p1 = t0 + 0.08 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [2000, 20000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 1500, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: Main binary search
    p2 = t0 + 0.50 * (budget_s - 5)
    hb, cb = fast_binary_search(best_h, n, p2, seed=seed, n_restarts=200)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 3: SLSQP polish from binary
    p3 = t0 + 0.72 * (budget_s - 5)
    hp, ch = multi_slsqp(best_h, n, p3, seed=seed)
    if ch < best_c:
        best_c = ch
        best_h = hp.copy()
    # Phase 4: Second binary search with different seed
    p4 = t0 + 0.88 * (budget_s - 5)
    xb = np.linspace(0, 1, n)
    h_seed = project_feasible(best_h + 0.3 * rng.randn(n), n)
    hb2, cb2 = fast_binary_search(h_seed, n, p4, seed=seed + 100, n_restarts=100)
    if cb2 < best_c:
        best_c = cb2
        best_h = hb2.copy()
    # Phase 5: Final SLSQP
    p5 = dl - 3
    hp2, ch2 = multi_slsqp(best_h, n, p5, seed=seed + 200)
    if ch2 < best_c:
        best_c = ch2
        best_h = hp2.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
