            hb2, cb2 = greedy_polish(best_h, n, end, max_sweeps=20, seed=seed + 500 + r)
            if cb2 < best_c:
                best_c = cb2
                best_h = hb2.copy()

    # Phase 4: Long anneal on best (10%)
    p4 = t0 + 0.83 * budget_s
    if time.time() < p4:
        hb, cb = anneal_search(best_h, n, p4, seed=seed + 9999,
                                T0=0.002, cool_rate=0.99995,
                                max_iter=int(5e6), block_prob=0.05)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        hb2, cb2 = greedy_polish(best_h, n, p4, max_sweeps=50, seed=seed + 8888)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()

    # Phase 5: Final SLSQP + greedy (remainder)
    if time.time() < dl - 10:
        h = slsqp_smooth(best_h.copy() + 0.01 * rng.randn(n), n, 100000000, 2000, dl - 10)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if time.time() < dl - 3:
            hb, cb = greedy_polish(best_h, n, dl, max_sweeps=80, seed=seed + 7777)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
