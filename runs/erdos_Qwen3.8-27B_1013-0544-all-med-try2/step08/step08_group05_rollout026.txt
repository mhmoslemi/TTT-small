            s = h.sum()
            if s < 1e-14:
                break

    a = alpha * shifted
    a_max = a.max()
    return m + (a_max + np.log(np.exp(a - a_max).sum())) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        a = sh - sh.max()
        return m + (sh.max() + np.log(np.exp(a).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_search(h_init, n, deadline, max_sweeps=30000, n_restarts=12, seed=42):
    target = n // 2
    dx = 2.0 / n
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max()
    best_c = cur_c
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            h = best_h.copy()
            n_flip = rng.randint(1, max(2, n // 35))
            for _ in range(n_flip):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) == 0 or len(zi) == 0:
                    break
                a = rng.choice(oi)
                b = rng.choice(zi)
                h[a], h[b] = 0.0, 1.0
            s = int(h.sum())
            while s > target:
                h[rng.choice(np.where(h == 1)[0])] = 0.0
                s -= 1
            while s < target:
                h[rng.choice(np.where(h == 0)[0])] = 1.0
                s += 1
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max()
            if cur_c > best_c + 3:
                h = best_h.copy()
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max()

        ones = np.where(h == 1)[0].tolist()
        zeros = np.where(h == 0)[0].tolist()

        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            rng.shuffle(zeros)
            for i in zeros:
                if h[i] != 0:
                    continue
                if time.time() > deadline:
                    break
                rng.shuffle(ones)
                for j in ones:
                    if h[j] != 1:
                        continue
                    delta = np.zeros(2 * n - 1)
                    im = i + m_range
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    imn = i - m_range
                    msk = (imn >= 0) & (imn < n)
                    delta[msk] -= h[imn[msk]]
                    jm = j + m_range
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    jmn = j - m_range
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - i)
                    if 0 <= k1 < 2 * n - 1:
                        delta[k1] += 1.0
                    k2 = n - 1 + (i - j)
                    if 0 <= k2 < 2 * n - 1:
                        delta[k2] += 1.0
                    nc = (corr + delta).max()
                    if nc < cur_c - 1e-10:
                        h[i] = 1.0
                        h[j] = 0.0
                        corr += delta
                        cur_c = nc
                        improved = True
                        break
                if improved:
                    ones = np.where(h == 1)[0].tolist()
                    zeros = np.where(h == 0)[0].tolist()
                    break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:

    # Phase 4: Another binary search round

                best_h = hb.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
