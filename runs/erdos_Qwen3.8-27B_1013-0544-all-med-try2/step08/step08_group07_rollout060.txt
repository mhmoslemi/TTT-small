def compute_B(h, i, n):
    B = np.zeros(2*n-1)
    # (1-h[i-m]) term: for m where 0 <= i-m < n, i.e., i-(n-1) <= m <= i

    lo_m = max(-(n-1), i-(n-1))  # i-m >= 0 => m <= i

    m_vals2 = np.arange(lo_m2, hi_m2+1)
    B[m_vals2 + n - 1] -= h[i + m_vals2]

    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsumexp = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
        return logsumexp

        m = g.max()

        sm = shifted.max()

    hb, cb = binary_local_search(best_h, n, p1b, max_rounds=20)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: SLSQP with analytical gradient (45% time)
    p2 = t0 + 0.60 * total
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000, 50000000]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p2 - 10:
            break
        h = slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Coordinate descent polish (10% time)
    p3 = t0 + 0.70 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=300, deadline=p3)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 4: More SLSQP from polished result (15% time)
    p4 = t0 + 0.85 * total
    h = best_h.copy()
    for alpha in [5000000, 50000000, 500000000]:
        if time.time() > p4 - 5:
            break
        h = slsqp_smooth(h, n, alpha, maxiter=5000, ftol=1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: Multi-start perturbations (remaining time)
    for trial in range(6):
        if time.time() > dl - 5:
            break
        h = best_h + 0.005 * rng.randn(n)
        h = project_feasible(h, n)
        hc, _ = coord_descent(h, n, max_sweeps=100, deadline=dl - 20)
        hc = project_feasible(hc, n)
        for alpha in [1000000, 10000000]:
            if time.time() > dl - 5:
                break
            hc = slsqp_smooth(hc, n, alpha, maxiter=2000, ftol=1e-16)
            c = compute_c5(hc, n)
            if c < best_c:
                best_c, best_h = c, hc.copy()

    # Phase 6: Try smaller n for better resolution
    for nh in [300, 500]:
        if time.time() > dl - 10:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, _ = coord_descent(hh, nh, max_sweeps=80, deadline=dl - 10)
        hh = project_feasible(hh, nh)
        hh = slsqp_smooth(hh, nh, 1000000, maxiter=2000, ftol=1e-16)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
