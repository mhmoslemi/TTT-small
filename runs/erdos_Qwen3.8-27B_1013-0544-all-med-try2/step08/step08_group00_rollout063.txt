def binary_local_search(h, n, deadline):
    """Local search for binary h."""
    h = np.round(h).clip(0, 1)
    # Ensure sum is correct
    target = n // 2
    diff = target - int(h.sum())
    if diff > 0:
        # Need to flip diff zeros to ones
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(0)
        for _ in range(diff):
            idx = rng.choice(zeros)
            h[idx] = 1
            zeros = np.where(h == 0)[0]
    elif diff < 0:
        ones = np.where(h == 1)[0]
        rng = np.random.RandomState(0)
        for _ in range(-diff):
            idx = rng.choice(ones)
            h[idx] = 0
            ones = np.where(h == 1)[0]
    
    dx = 2.0 / n

    # Main loop
    while time.time() < deadline:
        # Compute all overlaps
        corr = np.correlate(h, 1-h, mode='full')
        # Find the worst lag
        worst = np.argmax(corr)
        worst_val = corr[worst]
        
        # Try to improve

        # Get the worst lag

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    corr = np.correlate(x, 1 - x, mode='full')
    for idx in range(2*n-1):
        if w[idx] < 1e-15:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def grad_penalty(x, n, alpha, mu):
    return smooth_max_grad(x, n, alpha) + mu * (x.sum() - n/2.0)

def lbfgs_run(h, n, deadline, alpha_schedule):

    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    for alpha, iters in alpha_schedule:
        if time.time() > deadline:
            break
        try:
            res = minimize(obj_penalty, h_cur, jac=grad_penalty,
                         args=(n, alpha, 1e6), method='L-BFGS-B',
                         bounds=bounds,

                                  'gtol': 1e-18, 'maxfun': iters*200,

            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h_cur = h_new
        except:
            pass
    return h_cur, best_c

def compute_swap_delta(h, i, j, n):
    """Compute change in corr vector when h[i] goes 1→0 and h[j] goes 0→1.
    i has h[i]=1, j has h[j]=0."""
    delta = np.zeros(2*n-1)
    
    # Effect of h[i]: 1 → 0
    # m >= 0: index n-1+m
    for m in range(0, n):
        idx = n - 1 + m
        d = 0.0
        if i + m < n:
            d -= (1.0 - h[i+m])
        if i >= m:
            d += h[i-m]
        delta[idx] += d
    # m < 0: m = -p, index n-1-p
    for p in range(1, n):
        idx = n - 1 - p
        d = 0.0
        if i >= p:
            d -= (1.0 - h[i-p])
        if i + p < n:
            d += h[i+p]
        delta[idx] += d
    
    # Effect of h[j]: 0 → 1
    for m in range(0, n):
        idx = n - 1 + m
        d = 0.0
        if j + m < n:
            d += (1.0 - h[j+m])
        if j >= m:
            d -= h[j-m]
        delta[idx] += d
    for p in range(1, n):
        idx = n - 1 - p
        d = 0.0
        if j >= p:
            d += (1.0 - h[j-p])
        if j + p < n:
            d -= h[j+p]
        delta[idx] += d
    
    return delta

def compute_swap_delta_vec(h, i, j, n):
    delta = np.zeros(2*n-1)
    # For m >= 0 (idx = n-1 to 2n-2):
    # m = idx - (n-1), ranges from 0 to n-1
    m_pos = np.arange(n)  # 0, 1, ..., n-1
    idx_pos = n - 1 + m_pos  # n-1, n, ..., 2n-2
    
    # h[i]: 1→0
    d_i = np.zeros(n)
    mask1 = (i + m_pos) < n
    d_i[mask1] -= (1.0 - h[i + m_pos[mask1]])
    mask2 = (i >= m_pos)
    d_i[mask2] += h[i - m_pos[mask2]]
    delta[idx_pos] += d_i
    
    # h[j]: 0→1
    d_j = np.zeros(n)
    mask1 = (j + m_pos) < n
    d_j[mask1] += (1.0 - h[j + m_pos[mask1]])
    mask2 = (j >= m_pos)
    d_j[mask2] -= h[j - m_pos[mask2]]
    delta[idx_pos] += d_j
    
    # For m < 0 (idx = 0 to n-2):
    # m = -(n-1-idx), p = n-1-idx, ranges from 1 to n-1
    p_range = np.arange(1, n)  # 1, ..., n-1
    idx_neg = n - 1 - p_range  # n-2, ..., 0
    
    # h[i]: 1→0
    d_i_neg = np.zeros(n-1)
    mask1 = (i >= p_range)
    d_i_neg[mask1] -= (1.0 - h[i - p_range[mask1]])
    mask2 = (i + p_range) < n
    d_i_neg[mask2] += h[i + p_range[mask2]]
    delta[idx_neg] += d_i_neg
    
    # h[j]: 0→1
    d_j_neg = np.zeros(n-1)
    mask1 = (j >= p_range)
    d_j_neg[mask1] += (1.0 - h[j - p_range[mask1]])
    mask2 = (j + p_range) < n
    d_j_neg[mask2] -= h[j + p_range[mask2]]
    delta[idx_neg] += d_j_neg
    
    return delta

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += 0.5 * mu * (x.sum() - n / 2.0) ** 2
