import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def obj_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = x.sum() - n / 2.0
    val += mu * s * s
    return float(val)

def grad_fn(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(50, 1.0), (200, 5.0), (1000, 20.0), (5000, 100.0),
               (20000, 500.0), (100000, 2000.0), (500000, 10000.0),
               (2000000, 50000.0), (10000000, 200000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(obj_fn, h, jac=grad_fn,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-18,
                                  'gtol': 1e-16, 'maxfun': max_iters * 100,
                                  'maxls': 200})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=5000000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:

                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

Actually, for a fixed k:

The quadratic part is h^T A h where A is a specific matrix (banded, with 1s on the k-th diagonal).

The Hessian of f(k) with respect to h is -2*dx*A (where A has 1s at positions (i, i+k) and (i+k, i) for valid i).

So we can do a Newton step on f(k*)! But we need to handle the max (multiple k might be active).

**Key algorithm: Active Newton method**
1. Find active set: k values where f(k) is within epsilon of the max
2. Form the combined objective: max of active f(k_i)
3. Since each f(k_i) is quadratic, the max of quadratics is piecewise quadratic
4. At the current point, the active quadratic dominates, so do a Newton step on it
5. Project the step to maintain feasibility
6. Update active set
7. Repeat

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def obj_lmbfgs(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = x.sum() - n / 2.0
    val += mu * s * s
    return float(val)

def grad_lmbfgs(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def exact_grad_single(h, n, k):
    """Exact gradient of f(k) for a single shift k"""
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[:nv] += (1.0 - h[k:])
        g[k:] += (-h[:nv])
    else:
        kk = -k
        nv = n - kk
        g[kk:] += (1.0 - h[:nv])
        g[:nv] += (-h[kk:])
    return g * dx

def lbfgs_optimize(h, n, deadline, max_iters=300):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 1.0), (500, 5.0), (2000, 20.0), (10000, 100.0),
               (50000, 500.0), (200000, 2000.0), (1000000, 10000.0),
               (5000000, 50000.0), (20000000, 200000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(obj_lmbfgs, h, jac=grad_lmbfgs,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-18,
                                  'gtol': 1e-16, 'maxfun': max_iters * 100,
                                  'maxls': 200})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=5000000, maxiter=4000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: smooth_max_value(x, n, alpha), h,
                      jac=lambda x: smooth_max_grad(x, n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def exact_gradient_polish(h, n, deadline):
    """Polish using exact per-k gradients with backtracking line search"""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(50000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        c_max = c.max()
        active = np.where(c >= c_max - 2e-4)[0]
        improved = False
        for idx in active:
            k = idx - (n - 1)
            g = exact_grad_single(h, n, k)
            g -= g.mean()
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-14:
                continue
            for lr_scale in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
                lr = lr_scale * 0.1 / g_norm
                h_t = h - lr * g
                h_t = project_feasible(h_t, n)
                c_t = compute_c5(h_t, n)
                if c_t < best_c - 1e-16:
                    best_c = c_t
                    best_h = h_t.copy()
                    h = h_t
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c

def bilinear_pair_polish(h, n, deadline):
    """Exploit bilinear structure: for pair (i,j), f(k) changes linearly in d"""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    c = compute_c5_vec(h, n)
    active = np.where(c >= c.max() - 5e-4)[0]
    active_ks = (active - (n - 1)).astype(int)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    for d in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c

                    continue

                        if c_new < best_c - 1e-16:

                        c_new = compute_c5(h, n)

def local_pair_search(h, n, deadline, radius=120):
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
    for d in steps:
        while time.time() < deadline:
            any_imp = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h if 'best_h' in dir() else h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            any_imp = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[j] + d <= 1.0 and h[i] - d >= 0.0:

                            h[j] -= d

            if not any_imp:
                break
    return h, best_c
