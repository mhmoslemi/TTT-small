def make_thue_morse(n):
    signs=np.zeros(n)
    for i in range(n):
       x=i; p=0
       while x:
          p ^= 1
          x >>=1
       signs[i] = 1.0 if p==0 else -1.0
     h=0.5+0.5*signs
     target=n//2
     cnt=int(np.sum(signs==1.0))
     if cnt>target:
        idx=np.where(signs==1.0)[0]
        signs[idx[-(cnt-target):]]=-1.0
     elif cnt<target:
        idx=np.where(signs==-1.0)[0]
        signs[idx[-(target-cnt):]]=1.0
     h=0.5+0.5*signs

import time
import numpy as np

def cand_score(x):
    return x[2]

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    for _ in range(3):
        h = np.clip(h, 0.0, 1.0)
        s = float(h.sum())
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            r = 1.0 - h
            rs = float(r.sum())
            if rs <= 1e-15:
                break
            h = h + (d / rs) * r
        elif s > 1e-15:
            h = h + (d / s) * h
    return np.clip(h, 0.0, 1.0)

def resample_to_n(h, n1, n2):
    if n1 == n2:
        return project_feasible(h, n2)
    x1 = np.linspace(0.0, 2.0, n1)
    x2 = np.linspace(0.0, 2.0, n2)
    return project_feasible(np.interp(x2, x1, h), n2)

def make_thue_morse(n):
    s = np.zeros(n)
    for i in range(n):
        x = i
        p = 0
        while x:
            p ^= 1
            x >>= 1
        s[i] = 1.0 if p == 0 else -1.0
    tgt = n // 2
    c1 = int(np.sum(s == 1.0))
    if c1 > tgt:
        idx = np.where(s == 1.0)[0]
        s[idx[-(c1 - tgt):]] = -1.0
    elif c1 < tgt:
        idx = np.where(s == -1.0)[0]
        s[idx[-(tgt - c1):]] = 1.0
    return project_feasible(0.5 + 0.5 * s, n)

def make_cosine(n, f, ph):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.5 * np.sin(np.pi * float(f) * x + float(ph))
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    k = n // 2
    if k > 0:
        h[rng.choice(n, k, replace=False)] = 1.0
    return project_feasible(h, n)

def make_random_continuous(n, rng):
    return project_feasible(rng.rand(n), n)

def corr_counts(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def objective_count(corr, n):
    return float(np.max(corr) * 2.0 / float(n))

def deriv_at(pos, h, off, idx):
    der = np.zeros(2 * len(idx) - 1)
    der[pos - idx + off] += 1.0 - h
    der[idx - pos + off] -= h
    return der

def gradient_active(h, n, active, weights, off):
    g = np.zeros(n)
    for ri in range(len(active)):
        ii = int(active[ri])
        w = float(weights[ri])
        if w <= 1e-16:
            continue
        k = ii - off
        if k >= 0:
            L = n - k
            if L <= 0:
                continue
            g[k:n] += w * (1.0 - h[:L])
            g[:L] -= w * h[k:n]
        else:
            t = -k
            L = n - t
            if L <= 0:
                continue
            g[:L] += w * (1.0 - h[t:n])
            g[t:n] -= w * h[:L]
    return g

def line_grid_vals(base, L, q0, q1, q2, max_t, grid_frac):
    G = len(grid_frac)
    M = len(base)
    ts = grid_frac * float(max_t)
    arr = np.empty((G, M), dtype=float)
    arr[:] = base[None, :]
    arr += ts[:, None] * L[None, :]
    col = (ts * ts)[:, 0]
    arr[:, q0] -= 2.0 * col
    arr[:, q1] += col
    arr[:, q2] += col
    return arr.max(axis=1), ts

def line_value_scalar(base, L, tmp, tmpL, q0, q1, q2, t):

    np.copyto(tmp, base)
    np.multiply(L, float(t), out=tmpL)
    tmp += tmpL
    tmp[q0] -= 2.0 * tt
    tmp[q1] += tt
    tmp[q2] += tt
    return float(tmp.max())

        else:
            a = c

    vb = line_value_scalar(base, L, tmp, tmpL, q0, q1, q2, b)
    if vb < v:

def line_search_one(corr, h, i, j, n, off, idx, cur, grid_frac):
    hi = float(min(1.0 - h[i], h[j]))
    if hi <= 1e-9:
        return None, None
    L = deriv_at(i, h, off, idx) - deriv_at(j, h, off, idx)

    q1 = off + i - j

    if best_v >= cur - 1e-11:

        cv = float(vals[ci])
        if cv > best_v + 1e-7:
