def c5_vec(h, n):
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * (2.0 / n)

def c5_max(h, n):
    return float(np.max(c5_vec(h, n)))

def proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    t = n / 2.0
    for _ in range(500):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h; s = r.sum()
            if s < 1e-13: break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-13: break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def smax(c, a):
    m = c.max()
    s = a * (c - m)
    mx = s.max()
    return m + (mx + np.log(np.sum(np.exp(s - mx)))) / a

def sgrad(h, n, a, tk=80):

    m = c.max()

    idx = np.argsort(w)[-tk:]
    wnd = np.zeros_like(w)
    wnd[idx] = w[idx]
    wnd /= wnd.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for i in idx:
        if wnd[i] < 1e-15:
            continue

            g[kk:kk+v] += wnd[i] * dx * (1.0 - h[:v])
