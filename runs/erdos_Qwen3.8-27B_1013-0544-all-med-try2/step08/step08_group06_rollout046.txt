```python
def compute_grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')  # O(k) for all k
    # k = m - (n-1) for m in [0, 2n-2]
    shifted = alpha * c * dx
    sm = shifted.max()
    p = np.exp(shifted - sm)
    p_sum = p.sum()
    if p_sum < 1e-300:
        return c.max(), np.zeros(n)
    p /= p_sum
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for m in range(len(p)):
        if p[m] < 1e-8:
            continue
        k = m - (n - 1)
        # shift(one_minus_h, k): element j gets one_minus_h[j-k] if 0<=j-k<n
        # shift(h, -k): element j gets h[j+k] if 0<=j+k<n
        if k >= 0:
            # one_minus_h[j-k] valid for j in [k, n-1]
            grad[k:] += p[m] * one_minus_h[:n-k]
            # h[j+k] valid for j in [0, n-1-k]
            grad[:n-k] -= p[m] * h[k:]
        else:
            # k < 0, let kk = -k > 0
            kk = -k
            # one_minus_h[j+kk] valid for j in [0, n-1-kk]
            grad[:n-kk] += p[m] * one_minus_h[kk:]
            # h[j-kk] valid for j in [kk, n-1]
            grad[kk:] -= p[m] * h[:n-kk]
    
    grad *= dx / alpha
    fval = (sm + np.log(p_sum)) / alpha
    return fval, grad
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_grad_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * c * dx
    sm = shifted.max()
    p = np.exp(shifted - sm)
    p_sum = p.sum()
    if p_sum < 1e-300:
        val = sm / alpha
        return val, np.zeros(n), c
    p /= p_sum
    grad = np.zeros(n)
    omh = 1.0 - h
    for m in range(len(p)):
        if p[m] < 1e-10:
            continue
        k = m - (n - 1)
        if k >= 0:
            grad[k:] += p[m] * omh[:n-k]
            grad[:n-k] -= p[m] * h[k:]
        else:
            kk = -k
            grad[:n-kk] += p[m] * omh[kk:]
            grad[kk:] -= p[m] * h[:n-kk]
    grad *= dx / alpha
    val = (sm + np.log(p_sum)) / alpha
    return val, grad, c

def projected_gradient(h, n, deadline, alphas=None):

    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in alphas:

        for _ in range(200):
            if time.time() > deadline:
                return best_h, best_c
            val, grad, c = compute_grad_smooth(h, n, alpha)
            # Line search
            for ls in [1.0, 0.5, 0.25, 0.1, 0.05]:
                h_new = project_feasible(h - ls * step * grad, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c or c_new < compute_c5(h, n):
                    h = h_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                    break
            else:

    return best_h, best_c

    step = 0.01

            best_h = h.copy()

        k_vals = top_idx - (n - 1)
        # Subgradient: uniform weight on active shifts
        grad = np.zeros(n)
        omh = 1.0 - h
        for k in k_vals:
            if k >= 0:
                grad[k:] += omh[:n-k]
                grad[:n-k] -= h[k:]

                kk = -k
                grad[:n-kk] += omh[kk:]
                grad[kk:] -= h[:n-kk]

            if c_new < cur_c:
                h = h_new
                break
        else:
            step *= 0.5
            if step < 1e-8:
```

```python
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-14 and rng.random() < np.exp(-8000.0 * (nc - cur_c) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T = max(T * 0.99995, 1e-14)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    dl = t0 + total

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary local search (8%)
    p1 = t0 + 0.08 * total
    hb, cb = greed_binary(h0, n, p1, max_sweeps=300)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Projected gradient smooth max (35%)
    p2 = t0 + 0.43 * total
    alphas = [30, 100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p2 - 3:
            break
        h, c = pg_optimize(h, n, min(p2 - 2, t0 + (time.time() - t0 + 0.03 * total) * (len(alphas) + alphas.index(alpha)) / len(alphas) * 0), [alpha])
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Hard max subgradient (12%)
    p3 = t0 + 0.57 * total
    h3, c3 = hard_max_opt(best_h, n, p3, K=5)
    if c3 < best_c:
        best_c, best_h = c3, h3.copy()

    # Phase 4: Binary anneal (15%)
    p4 = t0 + 0.73 * total
    ha, ca = anneal_binary(best_h, n, p4, seed=seed)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()

    # Phase 5: Re-smooth from best (12%)
    p5 = t0 + 0.86 * total
    h = best_h.copy()
    alphas2 = [10000, 100000, 1000000, 10000000, 100000000]
    for alpha in alphas2:
        if time.time() > p5 - 2:
            break
        h, c = pg_optimize(h, n, min(p5 - 1, time.time() + 80), [alpha])
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 6: Try n+100 resolution (6%)
    p6 = t0 + 0.93 * total
    nh = n + 100
    if nh <= 1000:
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, ch = pg_optimize(hh, nh, p6 - 2, [10000, 100000, 1000000, 10000000])
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh

    # Phase 7: Final hard max polish
    p7 = dl - 2
    if time.time() < p7:
        h7, c7 = hard_max_opt(best_h, n, p7, K=7)
        if c7 < best_c:
            best_c, best_h = c7, h7.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
