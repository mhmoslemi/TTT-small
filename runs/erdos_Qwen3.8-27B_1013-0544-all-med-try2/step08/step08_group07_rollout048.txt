def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-14
            s = room[mask].sum()
            if s < 1e-14:
                break
            h[mask] += diff / s * room[mask]
        else:
            mask = h > 1e-14
            s = h[mask].sum()
            if s < 1e-14:
                break

def coord_descent(h, n, max_sweeps=200, deadline=None):

    corr = np.correlate(h, 1 - h, mode='full').astype(float)

    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            # Binary search for optimal h[i] given current corr

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-10
            s = room[mask].sum()
            if s < 1e-14:
                break
            h[mask] += diff / s * room[mask]
        else:
            mask = h > 1e-10
            s = h[mask].sum()
            if s < 1e-14:
                break
            h[mask] += diff / s * h[mask]
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=200, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])

            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]

            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9

            pos_B = B > 1e-14
            neg_B = B < -1e-14

                    lo_v = mid

                lb_x = 0.0

                ub_x = 1.0

                if lb_x <= ub_x + 1e-12:

                else:
                    lo_v = mid

            if neg_B.any():

            ub_x = 1.0

    """Compute gradient of log-sum-exp approximation using FFT."""
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    w = exps / exps.sum()
    grad = dx * np.correlate(w, 1 - 2 * h, mode='valid')
    return grad

def proj_grad_lls(h, n, alpha, max_iter, deadline):
    """Projected gradient with backtracking on log-sum-exp."""
    h = project_feasible(h.astype(float).copy(), n)
    best_h = h.copy()

    dx = 2.0 / n

        if deadline and time.time() > deadline:
            break
        grad = lls_gradient(h, n, alpha)
        
        # Backtracking line search
        for _ in range(30):

            c_new = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
            c_old = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if c_new < c_old + 1e-16 or lr < 1e-10:

        lr = min(lr * 1.5, 1.0)  # increase step size after successful step
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        exps = np.exp(shifted - sm)
        w = exps / exps.sum()
        return np.correlate(w, 1.0 - 2.0 * x, mode='valid')
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if deadline and time.time() > deadline:
            return h
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def binary_local_search(h, n, deadline, max_rounds=50):

    dx = 2.0 / n
    h = h.copy()
    to_binary_idx = np.argsort(h)[::-1][:n//2]
    h_bin = np.zeros(n)
    h_bin[to_binary_idx] = 1.0

    best_h = h_bin.copy()
    best_c = cur_c
    
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    m_range = np.arange(-(n-1), n)

        if deadline and time.time() > deadline:
            break
        found = False
        for zi, i in enumerate(zeros):

                break

            k1 = i - m_range; m1 = (k1>=0)&(k1<n); Bi[m1] += (1-h_bin[k1[m1]])
            k2 = i + m_range; m2 = (k2>=0)&(k2<n); Bi[m2] -= h_bin[k2[m2]]
            for oj, j in enumerate(ones):
                Bj = np.zeros(2*n-1)
                k1 = j - m_range; m1 = (k1>=0)&(k1<n); Bj[m1] += (1-h_bin[k1[m1]])
                k2 = j + m_range; m2 = (k2>=0)&(k2<n); Bj[m2] -= h_bin[k2[m2]]
                # h[i]: 0->1, h[j]: 1->0
                # Change in corr: Bi*1 + Bj*(-1) = Bi - Bj, but also the cross term
                # Actually when we change h[i] from 0 to 1, corr changes by Bi
                # When we change h[j] from 1 to 0, corr changes by -Bj
                # But Bj was computed with the old h, so we need to be careful.
                # Let's just compute the new corr directly for the (i,j) swap.
                # corr_new = corr + Bi - Bj (approximately, since the changes interact)
                # Actually the interaction is: when we set h[i]=1, it affects B for j,
                # and when we set h[j]=0, it affects B for i.
                # The cross term is: -h[i_old]*Bj_at_j + ... this gets complicated.
                # Let me just compute it directly.
                # 
                # Actually, the exact change is:
                # delta_corr = Bi * (1-0) + Bj * (0-1) + cross_term
                # where cross_term accounts for the interaction between i and j.
                # The interaction: when h[i] changes from 0 to 1, it affects the
                # correlation at the lag corresponding to j-i and i-j.
                # Specifically, corr[n-1+(j-i)] includes h[i]*h[j] term.
                # When h[i] was 0, this term was 0. Now it's h[j]=1, so we add 1.
                # Similarly corr[n-1+(i-j)] includes h[j]*(1-h[i]) which was 1, now 0, so subtract 1.
                # Wait, I need to be more careful.
                # 
                # The exact formula: corr_new = corr + Bi + (-Bj) + correction
                # where correction accounts for the fact that Bi and Bj were computed
                # with the old h values.
                
                # Let me just do it the brute force way for correctness:
                nc = corr + Bi - Bj
                # Correction for the interaction between i and j:
                # When h[i] goes 0->1: the term h[i]*(1-h[j]) at relative position changes
                # In corr, the term involving h[i] and h[j] appears at:
                # corr index where i and j interact: i - (j - i) = ... 
                # Actually, corr[k] = sum_p h[p] * h[k-p] (for the h*h part)
                # The term h[i]*h[j] appears in corr[i+j] (from p=i, k-p=j => k=i+j)
                # Wait no: corr[k] = sum_p h[p]*(1-h[k-p])
                # = sum_p h[p] - sum_p h[p]*h[k-p]
                # The term h[i]*h[j] appears in the second sum when p=i and k-p=j, i.e., k=i+j
                # So in corr, the interaction term is at k=i+j: contributes -h[i]*h[j] to the second sum
                # When h[i] changes 0->1: the second sum at k=i+j gains -h[j] (since h[i] was 0, now 1)
                # When h[j] changes 1->0: the second sum at k=i+j loses -h[i]*h[j]... 
                
                # This is getting too complicated. Let me just compute the new corr directly.
                # For n=400, the swap of 2 elements and recomputing corr is O(n) per pair,
                # but we have O(n^2) pairs, so total O(n^3) = 64M operations. That's too slow.
                # 
                # Let me use the approximation: nc = corr + Bi - Bj (ignoring 2nd order terms)
                # The 2nd order correction is at most O(1) in the corr array, which corresponds
                # to O(dx) in the objective. For n=400, dx=0.005, so the error is ~0.01 in the
                # corr values, which is ~0.00005 in the objective. This might be too large.
                # 
                # Let me include the correction:
                # The interaction between changing h[i] and h[j] affects corr at positions
                # where both i and j contribute. Specifically:
                # In the sum sum_p h[p]*(1-h[k-p]):
                # - p=i contributes h[i]*(1-h[k-i]) to corr[k]
                # - p=j contributes h[j]*(1-h[k-j]) to corr[k]
                # - The product h[i]*h[k-i] and h[j]*h[k-j] are the bilinear terms.
                # When we change both h[i] and h[j]:
                # corr_new[k] = corr_old[k] + (1-h[k-i]) - (1-h[k-j]) + h[k-j] - h[k-i] + interaction
                # Wait, this is just Bi[k] - Bj[k] which is what we already have.
                # The "interaction" would be if the same term involves both h[i] and h[j].
                # That happens when k-i = j (i.e., k = i+j) or k-j = i (i.e., k = i+j).
                # In that case, the term h[i]*(1-h[j]) or h[j]*(1-h[i]) appears.
                # When h[i]: 0->1 and h[j]: 1->0:
                # At k=i+j: the term h[i]*(1-h[k-i]) = h[i]*(1-h[j]) goes from 0 to 0 (since 1-h[j] becomes 0)
                # Wait, h[i] goes 0->1, h[j] goes 1->0.
                # At k=i+j: h[i]*(1-h[k-i]) = h[i]*(1-h[j]). Old: 0*(1-1)=0. New: 1*(1-0)=1. Change: +1.
                # But also h[j]*(1-h[k-j]) = h[j]*(1-h[i]). Old: 1*(1-0)=1. New: 0*(1-1)=0. Change: -1.
                # So the net change at k=i+j from the linear terms is already captured by Bi[i+j] and Bj[i+j].
                # Bi[i+j] includes (1-h[k1]) where k1 = i+j-i = j, so Bi[i+j] += (1-h[j]) = 1-1 = 0.
                # Bj[i+j] includes (1-h[k1]) where k1 = i+j-j = i, so Bj[i+j] += (1-h[i]) = 1-0 = 1.
                # So the net from linear terms at k=i+j: Bi[i+j] - Bj[i+j] = 0 - 1 = -1.
                # But the actual change is +1 (from h[i]*(1-h[j])) + (-1) (from h[j]*(1-h[i])) = 0.
                # Discrepancy: the bilinear term h[i]*h[j] at k=i+j:
                # Old: h[i]*h[j] = 0*1 = 0. New: 1*0 = 0. No change!
                # So the linear approximation gives -1 but the true change is 0.
                # We need to add +1 to corr[i+j] (if i+j < 2n-1).
                
                # Similarly at k=|i-j| (the other interaction point):
                # Actually, let me think about this differently.
                # corr[k] = sum_p h[p] - h[p]*h[k-p]
                # = (number of p with h[p]=1 in range) - (number of p with h[p]=1 and h[k-p]=1 in range)
                # For binary h: corr[k] = (count of 1s in valid range) - (count of 11 pairs at distance k)
                
                # When we swap h[i] 0->1 and h[j] 1->0:
                # The "count of 1s" term: at each k, the valid range might include i or j.
                # If i is in the valid range for k: the count increases by 1.
                # If j is in the valid range for k: the count decreases by 1.
                # Net: +1 at k where i is valid but j is not, -1 where j is valid but i is not.
                # This is exactly Bi - Bj (the linear part).
                
                # The "11 pair" term: count of p where h[p]=1 and h[k-p]=1.
                # When h[i] goes 0->1: new pairs (i, k-i) for all k where k-i is a 1-position.
                # When h[j] goes 1->0: removed pairs (j, k-j) and (k-j, j) for all k.
                # Net new pairs at k: [h[k-i]=1] - [h[k-j]=1] - [h[i]=1 at old]... 
                # This is getting complicated. Let me just add the correction term.
                
                # The bilinear correction: when both h[i] and h[j] change, the term
                # h[i]*h[j] (or h[j]*h[i]) at k=i+j changes.
                # Old: h[i]*h[j] = 0. New: h[i]*h[j] = 0. No change.
                # But the linear terms Bi and Bj already account for the individual changes.
                # The issue is that Bi was computed with h[j]=1 and Bj with h[i]=0,
                # but in the new state, h[j]=0 and h[i]=1.
                # The correction is: at positions where Bi or Bj involves the other index.
                # Bi[k] involves h[k-i] (from the (1-h) part) and h[k+i]... 
                # This is exactly the 2nd-order term I described above.
                
                # For simplicity, let me just check: is i+j < 2n-1?
                if i + j < 2 * n - 1:
                    nc[i + j] += h_bin[j]  # correction: h[j] was 1, so we overcounted
                if abs(i - j) < 2 * n - 1:
                    # The other interaction point
                    pass  # I'll skip this for now and just use the approximation
                
                if nc.max() * dx < cur_c - 1e-14:
                    h_bin[i], h_bin[j] = 1.0, 0.0
                    corr = nc
                    cur_c = corr.max() * dx
                    zeros.remove(i)
                    ones.remove(j)
                    zeros.append(j)
                    ones.append(i)
                    found = True
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h_bin.copy()
                    break
            if found and deadline and time.time() > deadline:
                break
        if not found:
            break
    return best_h, best_c
