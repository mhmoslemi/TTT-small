def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()

            break

            if s < 1e-14:

def compute_delta(h, n):
    """Compute delta[i](m) = (1-h[i-m]) - h[i+m] for all i, m.
    Returns a 2D array of shape (n, 2n-1) where index corresponds to m = k-(n-1).
    """
    # delta[i, k] where k = n-1+m, so m = k-(n-1)
    # delta[i, k] = (1 - h[i - (k-(n-1))]) - h[i + (k-(n-1))]
    # Let m = k - (n-1), range -(n-1) to (n-1)
    # delta[i, k] = (1-h[i-m]) - h[i+m]
    # h[j] is 0 if j < 0 or j >= n
    
    delta = np.zeros((n, 2*n-1))
    for m_idx in range(2*n-1):
        m = m_idx - (n-1)
        # (1-h[i-m]) for i=0..n-1: index i-m ranges from -m to n-1-m
        # h[i+m] for i=0..n-1: index i+m ranges from m to n-1+m
        im = np.arange(n) - m  # i-m
        ip = np.arange(n) + m  # i+m
        # clip and set out-of-range to 0
        im_valid = (im >= 0) & (im < n)
        ip_valid = (ip >= 0) & (ip < n)

        h_ip = np.zeros(n)

    return delta

        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

                       constraints=cons,
