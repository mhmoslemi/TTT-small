            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_opt(h, n, deadline, max_iters=300):
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        mu = max(1.0, 0.01 * alpha)
        try:
            t_rem = min(80, deadline - time.time() - 1)
            res = minimize(sm_obj, h, jac=sm_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def smart_pair_search(h, n, deadline, n_crit=20):
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(123)
    alpha_seq = [200, 1000, 5000, 20000]
    for alpha in alpha_seq:
        while time.time() < deadline:
            g = sm_grad(h, n, alpha, 0.0)
            c = compute_c5_vec(h, n)
            thresh = np.partition(c, 2 * n - 1 - n_crit)[2 * n - 1 - n_crit]
            c_mask = c >= thresh
            improved = False
            n_try = 0
            while n_try < 20000 and time.time() < deadline:
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    n_try += 1
                    continue
                d_max = min(h[i], 1.0 - h[j])
                d_min = -min(h[j], 1.0 - h[i])
                if d_max - d_min < 1e-10:
                    n_try += 1
                    continue
                # Only try if gradient suggests improvement
                dg = g[i] - g[j]
                if dg < -1e-8:
                    trial_d = min(0.02 * dg / (abs(dg) + 1e-10), d_max)
                    steps = [trial_d * 0.5, trial_d, d_max * 0.5, -abs(d_max) * 0.3]
                else:
                    steps = [0.005 * np.sign(dg + 1e-10), -0.005]
                for d in steps:
                    if d < d_min or d > d_max or abs(d) < 1e-10:
                        continue
                    h[i] += d
                    h[j] -= d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                        break
                    h[i] -= d
                    h[j] += d
                else:
                    h[i] -= d if 'd' in dir() else 0
                    if 'd' in dir() and d_min < d < d_max and abs(d) > 1e-10:
                        h[j] += d
                n_try += 1
            if not improved:
                break
    return h, best_c

def iterated_local_search(h, n, deadline, scale=0.05):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(77)
    while time.time() < deadline:
        t_rem = deadline - time.time()
        if t_rem < 30:
            break
        # Perturb
        h_p = h.copy()
        for _ in range(n // 3):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-scale, scale)
            if 0 <= h_p[i] + d <= 1 and 0 <= h_p[j] - d <= 1:
                h_p[i] += d
                h_p[j] -= d
        h_p = project_feasible(h_p, n)
        # Local optimize
        t_opt = min(40, t_rem - 10)
        h_new, c_new = lbfgs_opt(h_p, n, time.time() + t_opt, max_iters=80)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            # Reduce scale on success
            scale *= 0.8
        else:
            scale *= 1.1
            scale = min(scale, 0.15)
    return best_h, best_c

def try_interp(h_src, n_src, n_dst, deadline):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    h_dst = project_feasible(h_dst, n_dst)
    t_opt = min(80, deadline - time.time() - 10)
    h_new, c_new = lbfgs_opt(h_dst, n_dst, time.time() + t_opt, max_iters=150)
    return h_new, c_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(300, 0.5)
    n0 = len(h0)
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)

    # Phase 1: L-BFGS polish (120s)
    t1 = min(t0 + 120, deadline - 500)
    best_h, best_c = lbfgs_opt(best_h, n0, t1, max_iters=200)

    # Phase 2: Smart pair search (250s)
    t2 = min(time.time() + 250, deadline - 350)
    best_h, best_c = smart_pair_search(best_h, n0, t2, n_crit=15)

    # Phase 3: L-BFGS again after pair search (80s)
    t3 = min(time.time() + 80, deadline - 250)
    best_h, best_c = lbfgs_opt(best_h, n0, t3, max_iters=150)

    # Phase 4: More pair search with tighter focus (200s)
    t4 = min(time.time() + 200, deadline - 200)
    best_h, best_c = smart_pair_search(best_h, n0, t4, n_crit=10)

    # Phase 5: Iterated local search (150s)
    t5 = min(time.time() + 150, deadline - 100)
    h_il, c_il = iterated_local_search(best_h, n0, t5, scale=0.03)
    if c_il < best_c:
        best_c = c_il
        best_h = h_il

    # Phase 6: Try different n values (150s)
    for n_try in [250, 400]:
        if time.time() > deadline - 40:
            break
        alloc = min(60, deadline - time.time() - 20)
        if alloc < 25:
            break
        h_t, c_t = try_interp(best_h, len(best_h), n_try, time.time() + alloc)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t

    # Phase 7: Final refinement (remaining time)
    nf = len(best_h)
    if time.time() < deadline - 30:
        t7 = min(time.time() + 60, deadline - 20)
        best_h, best_c = lbfgs_opt(best_h, nf, t7, max_iters=100)
    if time.time() < deadline - 10:
        t8 = min(time.time() + 40, deadline - 5)
        best_h, best_c = smart_pair_search(best_h, nf, t8, n_crit=10)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
