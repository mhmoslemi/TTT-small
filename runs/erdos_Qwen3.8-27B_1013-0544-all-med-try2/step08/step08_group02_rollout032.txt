    configs = [(500, 1.0), (2000, 10.0), (10000, 50.0),
               (50000, 200.0), (200000, 1000.0),
               (1000000, 5000.0), (5000000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        if deadline - time.time() < 5:
            break
        try:
            res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': max_iters * 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def to_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]
        rem = rng.choice(idx, size=cnt - target, replace=False)
        h_bin[rem] = 0
    elif cnt < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)
        h_bin[add] = 1
    return h_bin

def targeted_binary_opt(h, n, deadline, rng):
    h_bin = to_binary(h, n, rng)
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    stall_count = 0
    max_stall = 5000

    while time.time() < deadline - 2:
        c_vec = np.correlate(h_bin, 1 - h_bin, mode='full')
        c_max = c_vec.max()
        thresh = c_max * 0.92
        hot_ks = np.where(c_vec >= thresh)[0]
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        improved = False
        n_try = min(8000, max(500, int((deadline - time.time()) * 25)))
        for t in range(n_try):
            if time.time() > deadline - 2:
                break
            mode = rng.randint(4)
            if mode <= 1:
                # Targeted flip at a hot shift
                k_idx = hot_ks[rng.randint(len(hot_ks))]
                k = k_idx - (n - 1)
                if k >= 0:
                    if k >= n:
                        continue
                    valid = np.arange(0, n - k)
                    if len(valid) == 0:
                        continue
                    cand = np.where((h_bin[valid] == 1) & (h_bin[valid + k] == 0))[0]
                    if len(cand) == 0:
                        continue
                    i = valid[cand[rng.randint(len(cand))]]
                    j = i + k
                else:
                    kk = -k
                    if kk >= n:
                        continue
                    valid = np.arange(kk, n)
                    cand = np.where((h_bin[valid] == 0) & (h_bin[valid - kk] == 1))[0]
                    if len(cand) == 0:
                        continue
                    i = valid[cand[rng.randint(len(cand))]]
                    j = i - kk
                h_bin[i] = 0
                h_bin[j] = 1
                c_new = compute_c5(h_bin, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_bin.copy()
                    improved = True
                    c_vec = np.correlate(h_bin, 1 - h_bin, mode='full')
                    c_max = c_vec.max()
                    thresh = c_max * 0.92
                    hot_ks = np.where(c_vec >= thresh)[0]
                    ones = np.where(h_bin == 1)[0]
                    zeros = np.where(h_bin == 0)[0]
                    stall_count = 0
                else:
                    h_bin[i] = 1
                    h_bin[j] = 0
                    stall_count += 1
            else:
                # Random swap
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_bin[i] = 0
                h_bin[j] = 1
                c_new = compute_c5(h_bin, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_bin.copy()
                    improved = True
                    ones = np.where(h_bin == 1)[0]
                    zeros = np.where(h_bin == 0)[0]
                    stall_count = 0
                else:
                    h_bin[i] = 1
                    h_bin[j] = 0
                    stall_count += 1

        if not improved:
            stall_count += 1000
            if stall_count > max_stall * 3:
                break
        if stall_count > max_stall:
            # Perturb: flip 3-5 random pairs
            npairs = rng.randint(3, 6)
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            for _ in range(min(npairs, len(ones), len(zeros))):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_bin[i] = 0
                h_bin[j] = 1
                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
            stall_count = 0

    return best_h, best_c

def multi_bit_flip(h, n, deadline, rng):
    h_bin = to_binary(h, n, rng)
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()

    while time.time() < deadline - 2:
        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        n_try = min(5000, max(300, int((deadline - time.time()) * 12)))
        for _ in range(n_try):
            if time.time() > deadline - 2:
                break
            if len(ones) < 2 or len(zeros) < 2:
                break
            # 2-swap: move two 1s
            i1, i2 = ones[rng.randint(len(ones))], ones[rng.randint(len(ones))]
            j1, j2 = zeros[rng.randint(len(zeros))], zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                continue
            h_bin[i1] = 0; h_bin[i2] = 0; h_bin[j1] = 1; h_bin[j2] = 1
            c_new = compute_c5(h_bin, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_bin.copy()
                improved = True
                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
            else:
                h_bin[i1] = 1; h_bin[i2] = 1; h_bin[j1] = 0; h_bin[j2] = 0
        if not improved:
            break
    return best_h, best_c

def random_binary(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = random_binary(400, rng)
        n_init = 400

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS-B polish (120s)
    t1 = min(t0 + 120, deadline - 500)
    best_h, best_c = lbfgs_run(best_h, best_n, t1, max_iters=600)

    # Phase 2: Targeted binary optimization (250s)
    if time.time() < deadline - 350:
        t2 = min(time.time() + 130, deadline - 220)
        h_b, c_b = targeted_binary_opt(best_h, best_n, t2, rng)
        if c_b < best_c:
            best_c = c_b
            best_h = h_b
        # Polish with L-BFGS
        if time.time() < deadline - 180:
            t2b = min(time.time() + 50, deadline - 170)
            h_p, c_p = lbfgs_run(best_h, best_n, t2b, max_iters=400)
            if c_p < best_c:
                best_c = c_p
                best_h = h_p

    # Phase 3: Multi-bit flips (80s)
    if time.time() < deadline - 180:
        t3 = min(time.time() + 70, deadline - 120)
        h_m, c_m = multi_bit_flip(best_h, best_n, t3, rng)
        if c_m < best_c:
            best_c = c_m
            best_h = h_m

    # Phase 4: Try different n values
    for n_try in [250, 350, 450, 600, 800]:
        if time.time() > deadline - 80:
            break
        if abs(n_try - best_n) < 50:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n_try)
        alloc = min(35, max(15, (deadline - time.time() - 60) * 0.35))
        if alloc < 12:
            continue
        h_t, c_t = lbfgs_run(h_int, n_try, time.time() + alloc, max_iters=300)
        if time.time() < deadline - 40:
            t_bin = min(time.time() + 12, deadline - 28)
            h_bt, c_bt = targeted_binary_opt(h_t, n_try, t_bin, rng)
            if c_bt < c_t:
                h_t, c_t = h_bt, c_bt
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Phase 5: Iterated targeted binary with restarts
    if time.time() < deadline - 40:
        for restart in range(3):
            if time.time() > deadline - 20:
                break
            # Perturb best solution
            h_pert = best_h.copy()
            n_flip = rng.randint(10, 25)
            if n_flip % 2 == 0:
                ones = np.where(h_pert > 0.5)[0]
                zeros = np.where(h_pert <= 0.5)[0]
                for _ in range(n_flip // 2):
                    if len(ones) > 0 and len(zeros) > 0:
                        i = ones[rng.randint(len(ones))]
                        j = zeros[rng.randint(len(zeros))]
                        h_pert[i] = 1.0 - h_pert[i]
                        h_pert[j] = 1.0 - h_pert[j]
                        ones = np.where(h_pert > 0.5)[0]
                        zeros = np.where(h_pert <= 0.5)[0]
            h_pert = project_feasible(h_pert, best_n)
            t_sr = min(time.time() + 12, deadline - 15)
            h_sr, c_sr = targeted_binary_opt(h_pert, best_n, t_sr, rng)
            if c_sr < best_c:
                best_c = c_sr
                best_h = h_sr

    # Phase 6: Final L-BFGS polish
    if time.time() < deadline - 4:
        t6 = min(time.time() + 12, deadline - 3)
        h_f, c_f = lbfgs_run(best_h, best_n, t6, max_iters=200)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
