            grad[:nv] += wi * (1.0 - x[shift:])

        else:

            nv = n - s

def slsqp_lse(h, n, deadline, alpha=500000, maxiter=2000):

def gradient_local_search(h, n, deadline, top_k=10, n_steps=5):
    """Local search guided by gradient of top-k correlations."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for step_size in [0.01, 0.005, 0.002, 0.001, 0.0005]:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_c5_vec(h, n)
            top_idx = np.argsort(c)[-top_k:]

            ck = c[top_idx]
            m = ck.max()
            w = np.exp(ck - m)
            w /= w.sum()

            for ii in range(top_k):
                mi = top_idx[ii]
                shift = mi - (n - 1)
                wi = w[ii] * dx
                if shift >= 0:
                    nv = n - shift
                    grad[:nv] += wi * (1.0 - h[shift:])
                    grad[shift:] += wi * (-h[:nv])
                else:
                    s = -shift
                    nv = n - s
                    grad[s:] += wi * (1.0 - h[:nv])
                    grad[:nv] += wi * (-h[s:])
            # Find best pair swap
            # We want to decrease the objective, so move mass from high grad to low grad
            # (positive grad means increasing h[i] increases the objective)
            # So we want to decrease h where grad > 0 and increase h where grad < 0
            # Or: move from high-grad positions to low-grad positions
            order = np.argsort(grad)
            # Try moving from high-grad to low-grad positions
            n_try = min(50, n // 10)
            for a in range(n_try):
                for b in range(n_try, min(2 * n_try, n)):
                    i = order[n - 1 - a]  # high grad
                    j = order[b]            # low grad
                    if h[i] - step_size >= 0 and h[j] + step_size <= 1:
                        h[i] -= step_size
                        h[j] += step_size
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:

                            h[j] -= step_size
                    if h[j] - step_size >= 0 and h[i] + step_size <= 1:
                        h[j] -= step_size
                        h[i] += step_size
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[j] += step_size
                            h[i] -= step_size
    return h, best_c

    dx = 2.0 / n

def project_feasible(h, n):
    target = n / 2.0

        if diff > 0:

            tot = room.sum()

            h += (diff / tot) * room

            if tot < 1e-14:
