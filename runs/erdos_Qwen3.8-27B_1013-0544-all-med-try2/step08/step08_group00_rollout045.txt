def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy().clip(0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return h.clip(0.0, 1.0)

def smooth_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)

def smooth_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] -= wm * h[kk:]
    return grad

def lbfgs_round(h, n, t_end, alphas=None, maxiter=500):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for alpha in alphas:
        if time.time() > t_end:
            break
        try:
            res = minimize(smooth_obj, h.copy(), jac=smooth_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-22,
                                   'gtol': 1e-20, 'maxfun': maxiter * 20})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fine_grad_search(h, n, t_end):
    best_c = compute_c5(h, n)
    alphas_gs = [5000, 50000, 500000]
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    for alpha in alphas_gs:
        for d in steps:
            improved = True
            while improved and time.time() < t_end:
                improved = False
                g = smooth_grad(h, n, alpha)
                k = min(25, n // 4)
                up = np.argsort(g)[-k:]
                dn = np.argsort(g)[:k]
                for i in dn:
                    if h[i] < d:
                        continue
                    for j in up:
                        if i == j or h[j] > 1.0 - d:
                            continue
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
    return h, best_c

def local_exhaustive(h, n, t_end, radius=80):
    best_c = compute_c5(h, n)
    steps = [0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    for d in steps:
        improved = True
        while improved and time.time() < t_end:
            improved = False
            for i in range(n):
                if time.time() > t_end:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] >= 1.0 or h[j] <= 0.0:
                        continue
                    amt = min(d, 1.0 - h[i], h[j])
                    if amt < 1e-12:
                        continue
                    h[i] += amt
                    h[j] -= amt
                    c = compute_c5(h, n)
                    if c < best_c - 1e-17:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= amt
                        h[j] += amt
    return h, best_c

def try_grid(h_best, c_best, n_best, nt, t_end):
    if time.time() > t_end - 40:
        return h_best, c_best, n_best
    x_old = np.linspace(0, 2, n_best, endpoint=False)
    x_new = np.linspace(0, 2, nt, endpoint=False)
    ht = np.interp(x_new, x_old, h_best)
    ht = project_feasible(ht, nt)
    ht, ct = lbfgs_round(ht, nt, t_end - 10, alphas=[5000, 50000, 500000], maxiter=300)
    if time.time() < t_end - 5:
        ht, ct = fine_grad_search(ht, nt, t_end - 2)
    if ct < c_best:
        return ht, ct, nt
    return h_best, c_best, n_best

        best_h, best_c = lbfgs_round(best_h, best_n, t4,

    if time.time() < deadline - 10:

    if time.time() < deadline - 3:
        best_h, best_c = local_exhaustive(best_h, best_n, deadline - 1, radius=30)

    best_c = compute_c5(best_h, best_n)
