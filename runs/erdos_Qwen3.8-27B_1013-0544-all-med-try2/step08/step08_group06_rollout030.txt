def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def make_delta(h, i, j, n):
    """Correlation delta for swap i:0->1, j:1->0"""
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range

    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range

    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def make_delta(h, i, j, n):
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    delta[n - 1 + (j - i)] += 1.0
    delta[n - 1 + (i - j)] += 1.0
    return delta

def binary_greedy(h_init, n, deadline, max_rounds=2000):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            found = False
            for j in ones_list:
                delta = make_delta(h, i, j, n)
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    found = True
                    break
            if found:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def binary_sa(h_init, n, deadline, seed=42, T0=0.012, cool=0.99998):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = T0
    no_improve = 0
    patience = 500000
    while time.time() < deadline:
        it_end = time.time() + 2.0
        while time.time() < it_end:
            ai = rng.randint(len(zeros_arr))
            bi = rng.randint(len(ones_arr))
            i = zeros_arr[ai]
            j = ones_arr[bi]
            delta = make_delta(h, i, j, n)
            nc = (corr + delta).max() * dx
            if nc <= cur_c or (T > 1e-9 and rng.random() < np.exp((cur_c - nc) / T)):
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = nc
                zeros_arr[ai] = j
                ones_arr[bi] = i
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
            T *= cool
        if no_improve > patience:
            h = to_binary(best_h, n).copy()
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            ones_arr = np.where(h == 1)[0].copy()
            zeros_arr = np.where(h == 0)[0].copy()
            T = T0 * 1.5
            no_improve = 0
    return best_h, best_c

def slsqp_opt(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_cont(h, n, deadline, n_iter=50000, step_init=0.008):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    step = step_init
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_star = int(np.argmax(corr))
        k = k_star - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] -= h[k:]
        else:
            kk = -k
            grad[:n - kk] += (1.0 - h[kk:])
            grad[kk:] -= h[:n - kk]
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h = h_new
            best_c = c_new
            best_h = h_new.copy()
        else:
            step *= 0.99
            if step < 1e-7:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 12
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Greedy binary (quick)
    p1 = t0 + 0.04 * total
    hb, cb = binary_greedy(h0, n, p1, max_rounds=300)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Long SA with restarts (main workhorse)
    p2 = t0 + 0.52 * total
    ha, ca = binary_sa(best_h, n, p2, seed=seed, T0=0.012, cool=0.99998)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()

    # Phase 3: SLSQP with increasing alpha
    p3 = t0 + 0.78 * total
    h = best_h.copy()
    alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    for alpha in alphas:
        if time.time() > p3 - 4:
            break
        h = slsqp_opt(h, n, alpha, 1500)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Subgradient continuous polish
    p4 = t0 + 0.88 * total
    hs, cs = subgrad_cont(best_h, n, p4, n_iter=40000, step_init=0.005)
    if cs < best_c:
        best_c, best_h = cs, hs.copy()

    # Phase 5: Final high-alpha SLSQP from best
    p5 = dl - 8
    if time.time() < p5:
        h = slsqp_opt(best_h, n, 50000000, 2000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 6: Quick multi-resolution check
    if time.time() < dl - 3:
        for nh in [300, 256, 500]:
            if time.time() > dl - 3:
                break
            xo = np.linspace(0, 2, n + 1)
            xn = np.linspace(0, 2, nh + 1)
            hh = project_feasible(np.interp(xn[:-1], xo, best_h), nh)

            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
