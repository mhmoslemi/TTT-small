import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c5(h, n, alpha=100.0):
    """Compute gradient of smooth max C5."""
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    sh = sh - sh.max()
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # For each lag k, grad_k = d/dh [sum_i h[i](1-h[i+k])]
    # = for i in valid range: (1-h[i+k]) - h[i-k] (with boundary handling)
    # Vectorized: for lag k >= 0:
    #   grad[:n-k] += w * (1 - h[k:])
    #   grad[k:] += w * (-h[:n-k])
    # For lag k < 0 (let kk = -k):
    #   grad[kk:] += w * (1 - h[:n-kk])
    #   grad[:n-kk] += w * (-h[kk:])
    
    # Instead of looping over all n-1 lags, use the weighted sum
    # Actually, let's just loop over the significant ones
    top_k = min(100, 2*n-1)
    idx = np.argsort(w)[-top_k:]
    for i in idx:

        k = i - (n - 1)
        nv = n - abs(k)

            grad[k:] -= w[i] * h[:nv]
        else:
            grad[-k:] += w[i] * (1.0 - h[:nv])
            grad[:nv] -= w[i] * h[-k:]
    return dx * grad

    it = 0

            h = h  # don't update

def swap_descent(h, n, deadline, max_iters=5000):
    """Greedy swap-based coordinate descent."""
    best_c = compute_c5(h, n)
    h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = grad_c5(h, n, alpha=500.0)
        # Find the best swap: move from high grad to low grad
        # grad[i] > 0 means increasing h[i] increases C5
        # So we want to decrease h[i] where grad[i] > 0
        # and increase h[j] where grad[j] < 0
        i_max = np.argmax(g)  # most want to decrease
        i_min = np.argmin(g)  # most want to increase
        if g[i_max] <= -g[i_min] + 1e-12:
            break  # no improvement possible
        
        # Line search for the swap amount
        best_delta = 0.0
        best_val = best_c
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            delta = frac * min(h[i_max], 1.0 - h[i_min])
            if delta < 1e-6:

            h[i_max] -= delta
            h[i_min] += delta
            c = compute_c5(h, n)
            h[i_max] += delta
            h[i_min] -= delta
            if c < best_val:
                best_val = c
                best_delta = delta
        if best_delta > 0:
            h[i_max] -= best_delta
            h[i_min] += best_delta
            best_c = best_val
        else:

                    h[j] -= delta

                        h[j] += delta

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c5(h, n, alpha=200.0):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    sh = sh - sh.max()
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    top_k = min(100, 2 * n - 1)
    idx = np.argsort(w)[-top_k:]
    for i in idx:
        wi = w[i]
        if wi < 1e-10:
            continue
        k = i - (n - 1)
        nv = n - abs(k)
        if k >= 0:
            grad[:nv] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:nv]
        else:
            grad[-k:] += wi * (1.0 - h[:nv])
            grad[:nv] -= wi * h[-k:]
    return dx * grad

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    sh = sh - sh.max()
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    step = 0.05
    while time.time() < deadline:
        g = grad_c5(h, n, alpha)
        v = momentum * v - step * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.001, 0.1)
        else:
            step *= 0.7
            if step < 0.0005:
                break
    return best_h, best_c

    best_c = compute_c5(h, n)
    h = h.copy()
    stall = 0
    while time.time() < deadline and stall < 30:
        g = grad_c5(h, n, alpha)
        # Top sources (want to decrease) and targets (want to increase)
        sources = np.argsort(g)[-top_pairs:]  # highest gradient
        targets = np.argsort(g)[:top_pairs]   # lowest gradient
        improved = False
        for si in sources:
            if time.time() > deadline:
                break
            for ti in targets:
                if si == ti:
                    continue
                # Line search: move delta from si to ti
                max_d = min(h[si], 1.0 - h[ti])
                if max_d < 1e-8:
                    continue
                best_d = 0.0
                best_v = best_c
                for frac in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:

                    if d < 1e-8:

                    if c < best_v:

                    h[ti] -= d

                    h[ti] += best_d
                    best_c = best_v
                    improved = True
                    break
            if improved:
                break
        if not improved:
            stall += 1
        else:
            stall = 0
    return h, best_c

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c5(h, n, alpha=200.0):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    sh = sh - sh.max()
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    top_k = min(120, 2 * n - 1)
    idx = np.argsort(w)[-top_k:]
    for i in idx:
        wi = w[i]
        if wi < 1e-10:
            continue
        k = i - (n - 1)
        nv = n - abs(k)
        if k >= 0:
            grad[:nv] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:nv]
        else:
            grad[-k:] += wi * (1.0 - h[:nv])
            grad[:nv] -= wi * h[-k:]
    return dx * grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    sh = sh - sh.max()
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad(x, n, alpha):
    return grad_c5(x, n, alpha)

def pgd_momentum(h, n, deadline, alpha=300.0, momentum=0.8):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    step = 0.04
    while time.time() < deadline:
        g = grad_c5(h, n, alpha)
        v = momentum * v - step * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            step *= 0.7
            if step < 0.0003:
                break
    return best_h, best_c

def swap_polish(h, n, deadline, top_pairs=6, alpha=800.0):
    best_c = compute_c5(h, n)
    h = h.copy()
    stall = 0
    while time.time() < deadline and stall < 50:
        g = grad_c5(h, n, alpha)
        sources = np.argsort(g)[-top_pairs:]
        targets = np.argsort(g)[:top_pairs]
        improved = False
        for si in sources:
            if time.time() > deadline:
                return h, best_c
            for ti in targets:
                if si == ti:
                    continue
                max_d = min(h[si], 1.0 - h[ti])
                if max_d < 1e-8:
                    continue
                best_d = 0.0
                best_v = best_c
                for frac in [1.0, 0.6, 0.4, 0.25, 0.1, 0.04]:
                    d = frac * max_d
                    if d < 1e-8:
                        continue
                    h[si] -= d
                    h[ti] += d
                    c = compute_c5(h, n)
                    h[si] += d
                    h[ti] -= d
                    if c < best_v:
                        best_v = c
                        best_d = d
                if best_d > 0:
                    h[si] -= best_d
                    h[ti] += best_d
                    best_c = best_v
                    improved = True
                    break
            if improved:
                break
        if not improved:
            stall += 1
        else:
            stall = 0
    return h, best_c

def fine_line_search(h, n, i, j, best_c):
    max_d = min(h[i], 1.0 - h[j])
    if max_d < 1e-8:
        return 0.0, best_c
    best_d = 0.0
    best_v = best_c
    lo, hi = 0.0, max_d
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h[i] -= mid
        h[j] += mid
        c = compute_c5(h, n)
        h[i] += mid
        h[j] -= mid
        if c < best_v:
            best_v = c
            best_d = mid
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-7:
            break
    return best_d, best_v

def swap_polish_fine(h, n, deadline, top_pairs=10, alpha=2000.0):
    best_c = compute_c5(h, n)
    h = h.copy()
    stall = 0
    while time.time() < deadline and stall < 80:
        g = grad_c5(h, n, alpha)
        sources = np.argsort(g)[-top_pairs:]
        targets = np.argsort(g)[:top_pairs]
        improved = False
        for si in sources:
            if time.time() > deadline:
                break
            for ti in targets:
                if si == ti:
                    continue
                d, best_v = fine_line_search(h, n, si, ti, best_c)
                if d > 0:
                    h[si] -= d
                    h[ti] += d
                    best_c = best_v
                    improved = True
                    break
            if improved:
                break
        if not improved:
            stall += 1
        else:
            stall = 0
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 200
        h = np.zeros(n)
        h[::2] = 1.0

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    best_n = n

    # Phase 1: L-BFGS-B quick polish
    if time.time() < deadline - 200:
        bounds = [(0.0, 1.0)] * best_n
        t_alloc = min(80, deadline - time.time() - 150)
        if t_alloc > 15:
            try:
                res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                              args=(best_n, 1500), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 400, 'ftol': 1e-17, 'gtol': 1e-14,
                                      'maxfun': 150000, 'maxls': 50, 'disp': False,
                                      'max_time': t_alloc})
                h_new = project_feasible(res.x, best_n)
                c_new = compute_c5(h_new, best_n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 2: PGD with momentum
    if time.time() < deadline - 150:
        t2 = min(time.time() + 60, deadline - 120)
        best_h, best_c = pgd_momentum(best_h, best_n, t2, alpha=500.0)

    # Phase 3: Swap polish (main workhorse)
    if time.time() < deadline - 80:
        t3 = min(time.time() + 200, deadline - 60)
        best_h, best_c = swap_polish(best_h, best_n, t3, top_pairs=6, alpha=600.0)

    # Phase 4: Restart with perturbation
    if time.time() < deadline - 60:
        for restart in range(3):
            if time.time() > deadline - 40:
                break
            rng_r = np.random.RandomState(seed + restart * 7919)
            h_pert = project_feasible(best_h + rng_r.normal(0, 0.005, best_n), best_n)
            t_r = min(time.time() + 80, deadline - 30)
            h_r, c_r = swap_polish(h_pert, best_n, t_r, top_pairs=6, alpha=600.0)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r

    # Phase 5: Try different n
    for n_try in [300, 400, 600]:
        if time.time() > deadline - 50:
            break
        if n_try == best_n:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n_try)
        bounds = [(0.0, 1.0)] * n_try
        t_alloc = min(60, deadline - time.time() - 30)
        if t_alloc < 15:
            continue
        try:
            res = minimize(smooth_max_obj, h_new, jac=smooth_max_grad,
                          args=(n_try, 1000), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-16,
                                  'maxfun': 80000, 'maxls': 50, 'disp': False,
                                  'max_time': t_alloc})
            h_t = project_feasible(res.x, n_try)
            c_t = compute_c5(h_t, n_try)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = n_try
        except Exception:
            pass

    # Phase 6: Final fine polish with bisection line search
    if time.time() < deadline - 5:
        best_h, best_c = swap_polish_fine(best_h, best_n, deadline - 2, top_pairs=12, alpha=3000.0)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
