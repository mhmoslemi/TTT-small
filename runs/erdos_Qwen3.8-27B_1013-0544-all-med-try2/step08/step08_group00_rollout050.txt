import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lsfr_val(x, n, alpha, k_focus):
    c = compute_c5_vec(x, n)
    total = len(c)
    if k_focus and k_focus < total:
        idx = np.argpartition(c, -k_focus)[-k_focus:]
        cf = c[idx]
    else:
        cf = c
    m = cf.max()
    sh = alpha * (cf - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def lsfr_weights(x, n, alpha, k_focus):
    c = compute_c5_vec(x, n)
    total = len(c)
    if k_focus and k_focus < total:
        idx = np.argpartition(c, -k_focus)[-k_focus:]
        cf = c[idx]
    else:
        idx = np.arange(total)
        cf = c
    m = cf.max()
    sh = alpha * (cf - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    w_full = np.zeros(total)
    w_full[idx] = w
    return w_full

def lsfr_grad(x, n, alpha, k_focus):
    w_full = lsfr_weights(x, n, alpha, k_focus)
    dx = 2.0 / n
    total = len(w_full)
    grad = np.zeros(n)
    for mi in range(total):
        if w_full[mi] < 1e-15:
            continue
        k = mi - (n - 1)
        wm = w_full[mi] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def lbfgs_phase(h, n, deadline, alphas, k_focuses, mu, maxiters):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    for alpha, kf in zip(alphas, k_focuses):
        if time.time() > deadline:
            break
        def obj(x):
            return lsfr_val(x, n, alpha, kf) + mu * (x.sum() - target) ** 2
        def grad(x):
            g = lsfr_grad(x, n, alpha, kf)
            return g + 2.0 * mu * (x.sum() - target)
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiters,
                          'ftol': 1e-20, 'gtol': 1e-18, 'maxfun': maxiters * 80,
                          'maxls': 150})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha, kf):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    def obj_eq(x):
        return lsfr_val(x, n, alpha, kf)
    def jac_eq(x):
        return lsfr_grad(x, n, alpha, kf)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_eq, h, jac=jac_eq, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 5000, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h, best_c

def guided_local_search(h, n, deadline, k_top=40):
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    for d in steps:
        if time.time() > deadline:
            break
        for round in range(3):
            if time.time() > deadline:
                break
            c_vec = compute_c5_vec(h, n)
            k_max = np.argmax(c_vec)
            k = k_max - (n - 1)
            dx = 2.0 / n
            g = np.zeros(n)
            if k >= 0:
                nv = n - k
                g[:nv] += (1.0 - h[k:]) * dx
                g[k:] += -h[:nv] * dx
            else:
                kk = -k
                nv = n - kk
                g[kk:] += (1.0 - h[:nv]) * dx
                g[:nv] += -h[kk:] * dx
            up = np.argsort(g)[-k_top:]
            dn = np.argsort(g)[:k_top]
            improved = False
            for i in dn:
                if time.time() > deadline:
                    return h, best_c
                for j in up:
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:

                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[j] + d <= 1.0 and h[i] - d >= 0.0:
                        h[j] += d
                        h[i] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[j] -= d
                            h[i] += d
            if not improved:
                break
    return h, best_c

def random_restart(h, n, rng, scale):
    h_p = h + rng.randn(n) * scale
