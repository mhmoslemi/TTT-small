def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def find_1d_min(a, b):
    """Find t in [0,1] minimizing max_k (a[k]*t + b[k])."""
    # Subgradient bisection
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        k_star = np.argmax(vals)
        sg = a[k_star]
        if sg > 1e-15:
            hi = mid
        elif sg < -1e-15:
            lo = mid
        else:
            break
    t_mid = (lo + hi) * 0.5
    # Check all three candidates
    f_lo = np.max(b)
    f_mid = np.max(a * t_mid + b)
    f_hi = np.max(a + b)
    if f_lo <= f_mid and f_lo <= f_hi:
        return 0.0
    elif f_hi <= f_mid and f_hi <= f_lo:
        return 1.0
    else:
        return t_mid

def coord_descent(h, n, deadline):
    dx = 2.0 / n
    c = compute_c5_vec(h, n)
    best_c = c.max()
    ks = np.arange(-(n - 1), n)
    no_improve = 0
    
    while time.time() < deadline:
        improved = False
        for j in range(n):
            src_mask = (ks >= -j) & (ks <= n - 1 - j)
            tgt_mask = (ks >= j - (n - 1)) & (ks <= j)
            
            a = np.zeros(2 * n - 1)
            a[src_mask] = (1.0 - h[j + ks[src_mask]]) * dx
            a[tgt_mask] += (-h[j - ks[tgt_mask]]) * dx
            
            b = c - h[j] * a
            t_opt = find_1d_min(a, b)
            
            if abs(t_opt - h[j]) > 1e-14:

                c += delta * a

            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if cur_c < best_c:
            best_c = cur_c
    
    return h, best_c
