def project_feasible(h, n):

    for _ in range(200):

        if abs(diff) < 1e-13:

            room = 1.0 - h
            tot = room.sum()

        else:
            tot = h.sum()
            if tot < 1e-14: break

            h = np.clip(h * scale, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        room = 1.0 - h

    c_shift = c - cmax

    w = np.exp(t - mx)
    wsum = w.sum()

    """Gradient-directed local search with adaptive steps"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [base_step * f for f in [1.0, 0.5, 0.25, 0.125, 0.0625]]

                if time.time() > deadline:
                    return best_h, best_c

                        improved = True
                    else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h * (1.0 + diff / tot), 0.0, 1.0)
    return h

def correct_weighted_grad(h, n, w):
    """Correct gradient of sum_m w[m]*I_m(h)"""
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[k:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[k:])
        else:
            kk = -k
            nv = n - kk
            grad[:nv] += wm * (1.0 - h[kk:])
            grad[kk:] += wm * (-h[:nv])
    return grad

def soft_max_weights(c, alpha):
    cmax = c.max()
    t = alpha * (c - cmax)
    mx = t.max()
    w = np.exp(t - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        s = 1.0
    return w / s

def obj_smooth_max(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    cmax = c.max()
    t = alpha * (c - cmax)
    mx = t.max()
    e = np.exp(t - mx)
    s = e.sum()
    val = cmax + (mx + np.log(max(s, 1e-300))) / alpha
    if mu > 0:
        s2 = x.sum() - n / 2.0
        val += mu * s2 * s2
    return float(val)

def grad_smooth_max(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = soft_max_weights(c, alpha)
    g = correct_weighted_grad(x, n, w)
    if mu > 0:
        s = x.sum() - n / 2.0
        g = g + 2.0 * mu * s
    return g

def obj_active_mean(x, n, active_idx, mu):
    c = compute_c5_vec(x, n)
    val = float(np.mean(c[active_idx]))
    if mu > 0:
        s = x.sum() - n / 2.0
        val += mu * s * s
    return val

def grad_active_mean(x, n, active_idx, mu):
    c = compute_c5_vec(x, n)
    w = np.zeros(len(c))
    w[active_idx] = 1.0 / len(active_idx)
    g = correct_weighted_grad(x, n, w)
    if mu > 0:
        s = x.sum() - n / 2.0
        g = g + 2.0 * mu * s
    return g

def lbfgs_optimize(h, n, deadline, alphas, max_iters=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        mu = max(0.01, alpha * 1e-6)
        try:
            res = minimize(obj_smooth_max, best_h, jac=grad_smooth_max,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iters * 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_optimize(h, n, deadline, eps_start=0.005, eps_end=0.0001):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    eps = eps_start
    while eps > eps_end and time.time() < deadline - 5:
        c = compute_c5_vec(best_h, n)
        cmax = c.max()
        active = np.where(c >= cmax - eps)[0]
        if len(active) < 1:
            active = np.array([np.argmax(c)])
        mu = 0.1

                          args=(n, active, mu), method='L-BFGS-B',

            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
        eps *= 0.5
    return best_h, best_c

def local_search_grad(h, n, deadline, base_step=0.005):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [base_step * f for f in [1.0, 0.5, 0.25, 0.125, 0.0625]]
    for s in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_c5_vec(best_h, n)
            w = soft_max_weights(c, 50000)
            g = correct_weighted_grad(best_h, n, w)
            inc = np.argsort(g)[:40]
            dec = np.argsort(g)[-40:]
            for i_idx in inc:
                if time.time() > deadline:
                    return best_h, best_c
                i = int(i_idx)
                if best_h[i] + s > 1.0:
                    continue
                for j_idx in dec:
                    j = int(j_idx)
                    if i == j or best_h[j] - s < 0.0:
                        continue
                    old_i, old_j = best_h[i], best_h[j]
                    best_h[i] = old_i + s
                    best_h[j] = old_j - s
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        best_h[i] = old_i
                        best_h[j] = old_j
    return best_h, best_c

def golden_ratio_start(n):

    positions = np.floor(np.arange(half) * phi).astype(np.int64) % n
    h[positions] = 1.0
    return project_feasible(h, n)
