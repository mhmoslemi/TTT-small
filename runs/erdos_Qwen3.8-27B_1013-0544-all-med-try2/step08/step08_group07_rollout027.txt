import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_run(h_init, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h_init

def coord_descent(h, n, max_sweeps=500, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()

        if deadline is not None and time.time() > deadline:

            k1 = i - m_range

            B[mask1] += (1.0 - h[k1[mask1]])

            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]

            val1 = np.max(corr + B * (1.0 - h[i]))

            ub_x = 1.0
            if pos_B.any():

            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

For h of length n, 1-h also has length n.
np.correlate(h, 1-h, 'full') returns a vector of length 2n-1.
The element at index n-1+k (0-indexed) is:
sum_{i} h[i] * (1-h)[i+k] = sum_i h[i](1-h[i+k])

for k = -(n-1), ..., n-1.

So C_k = (n-1+k-th element) * dx for k = -(n-1), ..., n-1.

This matches the problem statement: C₅ = max_k ∫ h(x)(1-h(x+k)) dx.

Wait, but the integral is over x where both x and x+k are in [0,2]. The correlation naturally handles this because the sum only includes valid indices. Good.

But wait, the integral ∫₀² h(x)(1-h(x+k)) dx should be computed over the overlap of [0,2] and [-k, 2-k]. The discrete approximation with dx = 2/n gives:
C_k ≈ dx * sum_{i: i+k ∈ [0,n)} h[i] * (1-h[i+k])

This is exactly what np.correlate gives (with the appropriate indexing). Good.

Now, one important note: the maximum over k includes k=0, where C_0 = ∫ h(x)(1-h(x)) dx = ∫ h(x) dx - ∫ h(x)² dx = 1 - ∫ h(x)² dx.

Since h ∈ [0,1] and ∫h = 1, we have ∫h² ≤ ∫h = 1 (by Cauchy-Schwarz, with equality iff h is binary). So C_0 = 1 - ∫h² ≥ 0.
