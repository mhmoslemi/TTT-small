import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    nv = n - abs(k)
    grad = np.zeros(n)
    if k >= 0:
        grad[:nv] = 1.0 - h[k:]
        grad[k:] -= h[:nv]
    else:
        kk = -k
        grad[kk:] = 1.0 - h[:nv]
        grad[:nv] -= h[kk:]
    return grad

def smooth_max_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    return w

def compute_grad(h, n, alpha, top_k=50):
    c = compute_c5_vec(h, n)
    w = smooth_max_weights(c, alpha)
    if top_k is not None:
        top_idx = np.argsort(w)[-top_k:]
        mask = np.zeros(len(w), dtype=bool)
        mask[top_idx] = True
        w = np.where(mask, w, 0.0)
        w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]

            sh = alpha * (c - m)
            mx = sh.max()
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha + 50.0 * (x.sum() - nn/2.0)**2)
        def lbf_grad(x, nn, alpha):
            c = compute_c5_vec(x, nn)
            m = c.max()
            sh = alpha * (c - m)
            mx = sh.max()
            w = np.exp(sh - mx)
            w /= w.sum()
            top_k = 40
            top_idx = np.argsort(w)[-top_k:]
            mk = np.zeros(len(w), dtype=bool)
            mk[top_idx] = True
            w = np.where(mk, w, 0.0)
            w /= w.sum()
            dxx = 2.0 / nn
            grad = np.zeros(nn)
            for idx in np.where(w > 1e-12)[0]:
                kk = idx - (nn - 1)
                grad += w[idx] * dxx * grad_single(x, nn, kk)
            grad += 100.0 * (x.sum() - nn/2.0)
            return grad
        bounds = [(0.0, 1.0)] * best_n
        for alpha in [5000, 15000]:
            if time.time() > deadline - 300:
                break
            try:
                res = minimize(lbf_obj, best_h, jac=lbf_grad, args=(best_n, alpha),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-14})
                h_new = project_feasible(res.x, best_n)
                c_new = compute_c5(h_new, best_n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    # Phase 3: Guided local search (main workhorse)
    if time.time() < deadline - 30:
        t3 = deadline - 5
        best_h, best_c = guided_local_search(best_h, best_n, t3, k=14)
    # Phase 4: Final pair polish
    if time.time() < deadline - 1:
        best_h, best_c = pair_polish(best_h, best_n, deadline - 1, radius=40)
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
