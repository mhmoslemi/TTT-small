def make_sinusoid_start(n,rng):

def make_lowpass_start(n,rng):
    scores=rng.rand(n)
    y=(scores>0.5).astype(float)
    # balance
    s=y.sum(); target=n/2
    if s>target:
       idx=np.argwhere(y==1).ravel(); perm=rng.permutation(s); y[idx[perm[:s-target]]]=0
    elif s<target:
       idx=np.argwhere(y==0).ravel(); perm=rng.permutation(n-s); y[idx[perm[:target-s]]]=1
    y -= y.mean()
    f=np.fft.rfft(y)
    keep=max(2, int(n//6))
    f[keep:] = 0.0
    h=np.fft.irfft(f, n=n)
    h=0.5+h
    return project_feasible(h,n)

def resample_h(h, n_new):
    n_old=len(h)
    if n_old==n_new: return project_feasible(h,n_new)
    x_old=np.linspace(0.0,2.0,n_old,endpoint=False)
    x_new=np.linspace(0.0,2.0,n_new,endpoint=False)
    # periodic extension
    xd=np.concatenate((x_old-2.0, x_old, x_old+2.0))
    hd=np.concatenate((h,h,h))
    h_new=np.interp(x_new,xd,hd)

def try_orientation(h,c,a,b,n,idx,cur_c):
    if a == b: return None
    hi=h[a]; hj=h[b]
    maxd=min(1.0-hi, hj) - 1e-12
    if maxd <= 1e-10: return None
    dx=2.0/n
    ds=[]
    for f in LOCAL_FRACS:
       d=maxd*f
       if d > 1e-9: ds.append(d)
    for f in LOCAL_ABS:
       d=f if f < maxd else maxd
       if d > 1e-9:
          dup=False
          for x in ds:
             if abs(x-d) < 1e-9: dup=True; break
          if not dup: ds.append(d)

    if best_d is None: return None

import numpy as np
import time
from scipy.optimize import minimize

LBFGS_CONFIGS = ((10.0, 1e-5), (50.0, 1e-4), (200.0, 1e-3),
                 (1000.0, 1e-2), (5000.0, 0.1), (20000.0, 1.0))
LOCAL_FRACS = (0.02, 0.1, 0.4, 1.0)
LOCAL_ABS = (1e-4, 1e-3, 1e-2, 5e-2)

def c5_full(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    return np.correlate(h, 1.0 - h, mode='full')

def c5_value(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    return float(np.max(c5_full(h, n)) * (2.0 / n))

def exactify_sum(h, n):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    bad = ~np.isfinite(h)
    if bad.any():
        h[bad] = 0.5
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(10):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
        h = np.clip(h, 0.0, 1.0)
    for idx in range(n):
        val = target - (float(h.sum()) - h[idx])
        if -1e-12 <= val <= 1.0 + 1e-12:
            h[idx] = min(1.0, max(0.0, val))
            break
    return h

def project_feasible(x, n):
    x = np.asarray(x, dtype=float).reshape(-1)
    if x.size != n:
        x = np.resize(x, n)
    bad = ~np.isfinite(x)
    if bad.any():
        x[bad] = 0.5
    target = 0.5 * n
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(75):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(p.sum())
    for _ in range(12):
        if abs(diff) < 1e-13:
            break
        free = (p > 1e-10) & (p < 1.0 - 1e-10)
        nf = int(free.sum())
        if nf > 0:
            p[free] += diff / nf
        else:
            p += diff / n
        p = np.clip(p, 0.0, 1.0)
        diff = target - float(p.sum())
    return p

def grad_weighted(h, w, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    w = np.asarray(w, dtype=float).reshape(-1)
    g = np.zeros(n)
    dx = 2.0 / n
    n1 = n - 1
    nz = np.flatnonzero(w > 1e-15)
    for m in nz:
        wm = w[m] * dx
        k = int(m) - n1
        if k >= 0:
            nv = n - k
            g[0:nv] += wm * (1.0 - h[k:n])
            g[k:n] += -wm * h[0:nv]
        else:
            kk = -k
            nv = n - kk
            g[kk:n] += wm * (1.0 - h[0:nv])
            g[0:nv] += -wm * h[kk:n]
    return g

def smooth_obj_grad(x, n, alpha, mu=0.0):
    x = np.asarray(x, dtype=float).reshape(-1)
    if x.size != n:
        x = np.resize(x, n)
    csum = np.correlate(x, 1.0 - x, mode='full')
    c = csum * (2.0 / n)
    m = float(c.max())
    a = alpha * (c - m)
    amax = float(a.max())
    e = np.exp(a - amax)
    se = float(e.sum())
    f = m + (amax + np.log(se)) / alpha
    w = e / se
    g = grad_weighted(x, w, n)
    if mu != 0.0:
        r = float(x.sum()) - 0.5 * n
        f += mu * r * r
        g += 2.0 * mu * r
    return f, g

def lbfgs_phase(h, n, deadline, configs=None):
    if configs is None:
        configs = LBFGS_CONFIGS
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c5_value(best_h, n)
    bounds = [(0.0, 1.0)] * n
    rem = deadline - time.time()
    if rem < 5.0:
        return best_h, best_c
    per = rem / len(configs)
    for alpha, mu in configs:
        rem = deadline - time.time()
        if rem < 5.0:

        try:

                           method='L-BFGS-B', jac=True, bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-15,
                                    'gtol': 1e-12, 'maxfun': maxit * 6,
                                    'maxls': 40})
            if res is not None and res.x is not None:
                h2 = project_feasible(res.x, n)
                c2 = c5_value(h2, n)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2
        except Exception:
            pass
    return best_h, best_c

def get_weights(csum, n, alpha, topk):
    c = csum * (2.0 / n)
    w = np.zeros(c.size)
    if topk > 0:
        k = min(int(topk), int(c.size))
        if k >= int(c.size):
            w.fill(1.0 / c.size)
        else:
            inds = np.argpartition(c, int(c.size) - k)[-k:]
            w[inds] = 1.0 / k
    elif alpha > 0.0:
        m = float(c.max())
        a = alpha * (c - m)
        amax = float(a.max())
        e = np.exp(a - amax)
        w = e / float(e.sum())
    else:
        w.fill(1.0 / c.size)
    return w

def make_strategies():
    return ((0.0, 1), (20.0, 0), (0.0, 3), (200.0, 0), (0.0, 8),

def apply_move(h, c, a, b, d, n, idx):
    newc = c.copy()
    di = d
    dj = -d
    hi = h[a]
    hj = h[b]
    m0 = n - 1
    newc[m0] += di * (1.0 - 2.0 * hi) - di * di
    newc[m0] += dj * (1.0 - 2.0 * hj) - dj * dj
    if a != b:
        mi = n - 1 + (b - a)
        mj = n - 1 + (a - b)
        newc[mi] += di * (1.0 - hj) - hi * dj - di * dj
        newc[mj] += dj * (1.0 - hi) - hj * di - dj * di
        mask = (idx != a) & (idx != b)
        if np.any(mask):
            s = idx[mask]
            hs = h[mask]
            np.add.at(newc, n - 1 + (s - a), di * (1.0 - hs))
            np.add.at(newc, n - 1 + (a - s), -hs * di)
            np.add.at(newc, n - 1 + (s - b), dj * (1.0 - hs))
            np.add.at(newc, n - 1 + (b - s), -hs * dj)
    return newc

def try_orientation(h, c, a, b, n, idx, cur_c):
    if a == b:
        return None
    hi = h[a]
    hj = h[b]
    maxd = min(1.0 - hi, hj) - 1e-12
    if maxd <= 1e-10:
        return None
    dx = 2.0 / n
    ds = []
    for fr in LOCAL_FRACS:
        d = maxd * fr
        if d > 1e-9:
            ds.append(d)
    for ab in LOCAL_ABS:
        d = ab if ab < maxd else maxd
        if d > 1e-9:
            dup = False
            for old in ds:
                if abs(old - d) < 1e-9:
                    dup = True
                    break
            if not dup:
                ds.append(d)
    best_val = cur_c
    best_d = None
    best_newc = None
    for d in ds:
        if hi + d > 1.0 + 1e-10 or hj - d < -1e-10:
            continue
        newc = apply_move(h, c, a, b, d, n, idx)
        val = float(newc.max()) * dx
        if val < best_val - 1e-14:
            best_val = val
            best_d = d
            best_newc = newc
    if best_d is None:
        return None
    return best_d, best_newc, best_val

            if time.time() >= deadline:
                break
            res = try_orientation(h, c, a, b, n, idx, cur_c)
            if res is not None:
                d, newc, val = res
                h[a] += d
                h[b] -= d
                if h[a] < 0.0:
                    h[a] = 0.0
                elif h[a] > 1.0:
                    h[a] = 1.0
                if h[b] < 0.0:
                    h[b] = 0.0
                elif h[b] > 1.0:
                    h[b] = 1.0
                c = newc
                cur_c = val
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
                accepts += 1
                if accepts % 30 == 0:
                    c2 = c5_full(h, n)
                    cur2 = float(c2.max()) * (2.0 / n)
                    c = c2
                    cur_c = cur2
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
        if not improved:
            no_improve += 1
            si += 1
            if no_improve == len(strategies):
                amounts = (0.003, 0.008, 0.02, 0.04, 0.08)
                amt = amounts[noise_level % len(amounts)]
                h2 = project_feasible(h + amt * rng.randn(n), n)
                c2 = c5_full(h2, n)
                cur2 = float(c2.max()) * (2.0 / n)
                if cur2 <= best_c + 0.01:
                    h = h2
                    c = c2
                    cur_c = cur2
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                else:
                    h = best_h.copy()
                    c = c5_full(h, n)
                    cur_c = best_c
                noise_level += 1
                no_improve = 1
            if no_improve > 3 * len(strategies):
                break
        else:
            no_improve = 0
    c = c5_full(h, n)
    cur_c = float(c.max()) * (2.0 / n)
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()
    cb = c5_full(best_h, n)
    best_c = float(cb.max()) * (2.0 / n)
    return best_h, best_c

def resample_h(h, n_new):
    h = np.asarray(h, dtype=float).reshape(-1)
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(h, n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    xd = np.concatenate((x_old - 2.0, x_old, x_old + 2.0))
    hd = np.concatenate((h, h, h))
    h_new = np.interp(x_new, xd, hd)
    return project_feasible(h_new, n_new)

def make_sinusoid_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    s = np.zeros(n)
    for r in range(4):
        f = int(rng.randint(1, max(2, n // 8)))
        amp = rng.uniform(0.1, 0.45) / (0.6 * r + 1.0)
        s += amp * np.sin(2.0 * np.pi * f * x / 2.0 + rng.uniform(0.0, 2.0 * np.pi))
    s -= s.mean()
    return project_feasible(0.5 + s, n)

def make_lowpass_start(n, rng):
    scores = rng.rand(n)
    y = (scores > 0.5).astype(float)
    target = 0.5 * n
    s = int(y.sum())
    if s > target:
        idx = np.flatnonzero(y == 1.0)
        perm = rng.permutation(s)
        y[idx[perm[:s - int(target)]]] = 0.0
    elif s < target:
        idx = np.flatnonzero(y == 0.0)
        perm = rng.permutation(int(n) - s)
        y[idx[perm[:int(target) - s]]] = 1.0
    y -= y.mean()
    f = np.fft.rfft(y)
    keep = max(2, int(n // 6))
    if keep < len(f):
        f[keep:] = 0.0
    h = np.fft.irfft(f, n=n)
    return project_feasible(0.5 + h, n)

def quick_polish(h, n, deadline, rng):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c5_value(best_h, n)
    rem = deadline - time.time()
    if rem > 30.0:
        t1 = min(time.time() + min(25.0, rem / 2.0), deadline - 5.0)
        if t1 > time.time() + 5.0:
            h2, c2 = lbfgs_phase(h, n, t1, LBFGS_CONFIGS[:4])
            if c2 < best_c:
                best_h = h2
                best_c = c2
    if time.time() < deadline - 10.0:
        csum = c5_full(best_h, n)
        h2, c2 = local_search(best_h, csum, n, max(time.time() + 2.0, deadline - 2.0), rng)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    best_h = project_feasible(best_h, n)
    best_h = exactify_sum(best_h, n)
    best_c = c5_value(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0
    if budget < 8.0:
        budget = 8.0
    deadline = t0 + budget - 2.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        if h_init.size == 0:
            raise ValueError
    except Exception:
        h_init = None
    if h_init is None or len(h_init) < 10:
        h_init = make_lowpass_start(300, rng)
    n_main = len(h_init)
    if n_main < 300 and budget > 240:
        h_init = resample_h(h_init, 300)
        n_main = 300
    h = project_feasible(h_init, n_main)
    best_h = h.copy()
    best_n = n_main
    best_c = c5_value(best_h, best_n)

    rem = deadline - time.time()
    if rem > 130.0:
        t1 = min(time.time() + 80.0, deadline - 300.0)
        if t1 > time.time() + 10.0:
            h2, c2 = lbfgs_phase(best_h, best_n, t1)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = len(best_h)

    rem = deadline - time.time()
    if rem > 280.0:
        t2 = min(time.time() + 350.0, deadline - 130.0)
        if t2 > time.time() + 20.0:
            csum = c5_full(best_h, best_n)
            h2, c2 = local_search(best_h, csum, best_n, t2, rng)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = len(best_h)
    elif rem > 110.0:
        t2 = min(time.time() + 80.0, deadline - 30.0)
        if t2 > time.time() + 10.0:
            csum = c5_full(best_h, best_n)
            h2, c2 = local_search(best_h, csum, best_n, t2, rng)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = len(best_h)

    rem = deadline - time.time()
    if rem > 230.0:
        n_div = 150 if best_n >= 150 else max(80, best_n)
        starts = []
        starts.append(resample_h(best_h, n_div))
        starts.append(make_sinusoid_start(n_div, rng))
        starts.append(make_lowpass_start(n_div, rng))
        hd = resample_h(best_h, n_div)
        hd = project_feasible(hd + 0.05 * rng.randn(n_div), n_div)
        starts.append(hd)
        for idx_s in range(len(starts)):
            if time.time() > deadline - 45.0:
                break
            remaining = deadline - time.time() - 30.0
            if remaining < 35.0:
                break
            alloc = min(55.0, remaining / float(len(starts) - idx_s))
            tp = min(time.time() + max(20.0, alloc), deadline - 5.0)
            if tp <= time.time() + 10.0:
                break
            h2, c2 = quick_polish(starts[idx_s], n_div, tp, rng)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = len(best_h)

    if best_n < 300 and deadline - time.time() > 130.0:
        h_up = resample_h(best_h, 300)
        tp = min(time.time() + 90.0, deadline - 25.0)
        if tp > time.time() + 15.0:
            h2, c2 = quick_polish(h_up, 300, tp, rng)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = 300

    if deadline - time.time() > 25.0:
        n_final = len(best_h)
        csum = c5_full(best_h, n_final)
        h2, c2 = local_search(best_h, csum, n_final, max(time.time() + 5.0, deadline - 1.0), rng)
        if c2 < best_c:
            best_h = h2
            best_c = c2
            best_n = len(best_h)

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_h = exactify_sum(best_h, n_final)
    best_c = c5_value(best_h, n_final)
    return best_h, best_c, int(n_final)
