def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_passes=2000, deadline=None, random_order=False):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    C = np.correlate(h, 1 - h, mode='full').copy()
    d_off = np.arange(2 * n - 1) - (n - 1)
    m_len = 2 * n - 1
    order = np.arange(n)

    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        if random_order and p > 0:
            np.random.shuffle(order)
        improved = False
        for j in order:
            if deadline is not None and time.time() > deadline:
                h = project_feasible(h, n)
                best_c = min(best_c, compute_c5(h, n))
                return h, best_c
            t0 = h[j]
            jd = j + d_off
            jdv = (jd >= 0) & (jd < n)
            jdm = j - d_off
            jdmv = (jdm >= 0) & (jdm < n)
            a = np.zeros(m_len)
            a[jdv] = 1.0 - h[jd[jdv]]
            a[jdmv] -= h[jdm[jdmv]]
            B = C - a * t0
            lo, hi = 0.0, 1.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(B + a * m1) < np.max(B + a * m2):
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) * 0.5
            if abs(t_opt - t0) > 1e-14:
                h[j] = t_opt
                C += a * (t_opt - t0)
                improved = True
        if (p + 1) % 20 == 0:
            C = np.correlate(h, 1 - h, mode='full')
        if not improved:
            break
    h = project_feasible(h, n)
    best_c = min(best_c, compute_c5(h, n))
    return h, best_c

def subgradient_kick(h, n, step):
    C = np.correlate(h, 1 - h, mode='full')
    Cmax = C.max()
    active = C >= Cmax - 1e-8
    if not active.any():
        active[np.argmax(C)] = True
    w = np.zeros_like(C)
    w[active] = 1.0 / active.sum()
    d_off = np.arange(2 * n - 1) - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        jd = j + d_off
        jdv = (jd >= 0) & (jd < n)
        jdm = j - d_off
        jdmv = (jdm >= 0) & (jdm < n)
        g[j] = np.sum(w[jdv] * (1.0 - h[jd[jdv]])) - np.sum(w[jdmv] * h[jdm[jdmv]])
    h_new = h - step * g
    h_new = h_new - (h_new.sum() - n / 2.0) / n
    h_new = project_feasible(h_new, n)
    return h_new

def lbfgs_run(h, n, alpha, maxiter):
    dx = 2.0 / n

    def _sm_obj(x):
        C = np.correlate(x, 1 - x, mode='full')
        m = C.max()
        sh = alpha * (C - m)
        mx = sh.max()
        return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx)

    def _sm_grad(x):
        C = np.correlate(x, 1 - x, mode='full')
        m = C.max()
        sh = alpha * (C - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        d_off = np.arange(2 * n - 1) - (n - 1)
        g = np.zeros(n)
        for j in range(n):
            jd = j + d_off
            jdv = (jd >= 0) & (jd < n)
            jdm = j - d_off
            jdmv = (jdm >= 0) & (jdm < n)
            g[j] = np.sum(w[jdv] * (1.0 - x[jd[jdv]])) - np.sum(w[jdmv] * x[jdm[jdmv]])
        return g * dx

    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    rng = np.random.RandomState(seed)

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n_best = len(h_best)
    except Exception:
        h_best = np.full(650, 0.5)
        n_best = 650
        h_best[:325] = 1.0

    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)

    # Phase 1: L-BFGS-B quick polish
    if time.time() < deadline - 200:
        t1 = min(time.time() + 40, deadline - 200)
        for alpha in [5000, 50000]:
            if time.time() > t1:
                break
            h_t, c_t = lbfgs_run(h_best, n_best, alpha, 200)
            if c_t < best_c:
                best_c, h_best = c_t, h_t

    # Phase 2: Coordinate descent on parent
    if time.time() < deadline - 150:
        t2 = min(time.time() + 120, deadline - 150)
        h_best, best_c = coord_descent(h_best, n_best, max_passes=5000, deadline=t2)

    # Phase 3: Try nearby resolutions
    resolutions = [n_best + 50, n_best - 50, n_best + 100, n_best - 100, 550, 750]
    resolutions = sorted(set(r for r in resolutions if 200 <= r <= 1200 and r != n_best))
    for n_try in resolutions:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 40
        alloc = min(remaining * 0.35, 150)
        if alloc < 25:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_feasible(h_try, n_try)
        h_try, c_try = coord_descent(h_try, n_try, max_passes=3000, deadline=time.time() + alloc)
        if c_try < best_c:
            best_c, h_best, n_best = c_try, h_try, n_try

    # Phase 4: Iterated local search with perturbations
    remaining = deadline - time.time() - 20
    if remaining > 40:
        scales = [0.01, 0.02, 0.03, 0.05]
        for scale in scales:
            if time.time() > deadline - 30:
                break
            alloc = min(remaining * 0.25, 120)
            if alloc < 15:
                break
            h_pert = h_best + rng.randn(n_best) * scale
            h_pert = project_feasible(h_pert, n_best)
            h_p, c_p = coord_descent(h_pert, n_best, max_passes=2000, deadline=time.time() + alloc, random_order=True)
            if c_p < best_c:
                best_c, h_best = c_p, h_p
            remaining = deadline - time.time() - 20

    # Phase 5: Subgradient kicks
    remaining = deadline - time.time() - 10
    if remaining > 30:
        for step in [0.01, 0.005, 0.002, 0.001]:
            if time.time() > deadline - 20:
                break
            h_kick = subgradient_kick(h_best, n_best, step)
            alloc = min(remaining * 0.25, 80)
            if alloc < 10:
                break
            h_k, c_k = coord_descent(h_kick, n_best, max_passes=1500, deadline=time.time() + alloc)
            if c_k < best_c:
                best_c, h_best = c_k, h_k
            remaining = deadline - time.time() - 10

    # Phase 6: Final coordinate descent squeeze
    if time.time() < deadline - 1:
        h_best, best_c = coord_descent(h_best, n_best, max_passes=10000, deadline=deadline)

    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)
    return h_best, best_c, n_best
