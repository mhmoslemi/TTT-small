```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_max_and_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    exp_vals = np.exp(shifted - smax)
    weights = exp_vals / exp_vals.sum()
    val = m + (smax + np.log(exp_vals.sum())) / alpha
    
    grad = np.zeros(n)
    # Only process shifts with significant weight
    for k in range(len(c)):
        wk = weights[k]
        if wk < 1e-12:
            continue
        m_shift = k - (n - 1)
        if m_shift >= 0:
            nv = n - m_shift
            grad[:nv] += wk * (1.0 - h[m_shift:m_shift + nv])
            grad[m_shift:] += wk * (-h[:nv])
        else:
            mp = -m_shift
            nv = n - mp
            grad[mp:] += wk * (1.0 - h[:nv])
            grad[:nv] += wk * (-h[mp:mp + nv])
    
    return val, grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(x, n, alpha, mu):
    val, g = smooth_max_and_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    val += mu * s * s
    g = g + 2.0 * mu * s
    return val, g

def optimize_lbfgs(h, n, deadline, alphas, max_iters_per=200):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        if deadline - time.time() < 10:
            break
        mu = min(10000.0, alpha * 0.01)
        try:
            res = minimize(obj_and_grad, h, jac=True, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters_per, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': max_iters_per * 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c
```

```python
def smooth_max_and_grad(h, N, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    exp_vals = np.exp(shifted - smax)
    weights = exp_vals / exp_vals.sum()
    val = m + (smax + np.log(exp_vals.sum())) / alpha
    
    grad = np.zeros(N)
    for k in range(len(c)):
        wk = weights[k]
        if wk < 1e-12:
            continue
        s = k - (N - 1)
        if s >= 0:
            nv = N - s
            # (1-h[m+s]) for m in [0, nv-1]
            grad[:nv] += wk * (1.0 - h[s:s+nv])
            # -h[m-s] for m in [s, N-1]
            grad[s:] += wk * (-h[0:nv])
        else:
            p = -s
            nv = N - p
            # (1-h[m-p]) for m in [p, N-1]
            grad[p:] += wk * (1.0 - h[0:nv])
            # -h[m+p] for m in [0, nv-1]
            grad[:nv] += wk * (-h[p:p+nv])
    
    return val, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_max_and_grad(h, N, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    smax = shifted.max()
    ev = np.exp(shifted - smax)
    weights = ev / ev.sum()
    val = m + (smax + np.log(ev.sum())) / alpha
    grad = np.zeros(N)
    for k in range(len(c)):
        wk = weights[k]
        if wk < 1e-12:
            continue
        s = k - (N - 1)
        if s >= 0:
            nv = N - s
            grad[:nv] += wk * (1.0 - h[s:s+nv])
            grad[s:] += wk * (-h[0:nv])
        else:
            p = -s
            nv = N - p
            grad[p:] += wk * (1.0 - h[0:nv])
            grad[:nv] += wk * (-h[p:p+nv])
    return val, grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def obj_grad(x, n, alpha, mu):
    val, g = smooth_max_and_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    val += mu * s * s
    g = g + 2.0 * mu * s
    return val, g

def opt_lbfgs(h, n, deadline, alphas, iters=300):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        if deadline - time.time() < 8:
            break
        mu = min(5000.0, alpha * 0.005)
        try:
            res = minimize(obj_grad, h, jac=True, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': iters, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': iters * 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def binary_local_search(h, n, deadline, rng, max_rounds=20000):
    h_b = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_b.sum())
    if cnt > target:
        idx = np.where(h_b == 1)[0]
        h_b[rng.choice(idx, size=cnt - target, replace=False)] = 0
    elif cnt < target:
        idx = np.where(h_b == 0)[0]
        h_b[rng.choice(idx, size=target - cnt, replace=False)] = 1
    best_c = compute_c5(h_b, n)
    best_h = h_b.copy()
    no_improve = 0
    rounds = 0
    while time.time() < deadline and rounds < max_rounds:
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_b[i] = 0
        h_b[j] = 1
        c_new = compute_c5(h_b, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_b.copy()
            no_improve = 0
        else:
            h_b[i] = 1
            h_b[j] = 0
            no_improve += 1
        rounds += 1
        if no_improve > 5000:
            break
    return best_h, best_c

def fourier_init(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 15):
        h += rng.uniform(0.05, 0.25) * np.cos(2 * np.pi * freq * x + rng.uniform(0, 6.28))
    return project_feasible(h, n)
```
