def active_polish(h, n, deadline, max_iter=200, red_start=1e-5):
    h = quick_fix_sum(h, n).copy()
    c = corr_c(h,n); m=float(c.max())
    best_h=h.copy(); best_m=m; red=red_start
    for it in range(max_iter):
       if time.time() >= deadline: break
       # if m < best_m? keep
       tol=max(1e-8, 0.25*red)
       active = np.where(c >= m - tol)[0]
       if active.size ==0: active=np.array([int(c.argmax())])
       if active.size > 12:
           # keep sorted top 12 by c
           order=np.argsort(c[active])[::-1][:12]
           active=active[order]
       G=np.empty((active.size,n))
       for r,idx in enumerate(active):
           G[r]=grad_lag_one(h,n,int(idx)-(n-1))
       target=m - red
       e=c[active] - target
       e=np.maximum(e, 0.0)
       if e.sum() <= 0:
           # use max grad
           Gm = grad_lag_one(h,n,int(np.argmax(c)))
           delta=-Gm
       else:
           A=G.dot(G.T)
           a=A.shape[0]
           reg=1e-10 * max(1.0, A.trace()/a)
           A += np.eye(a)*reg
           try: sol=np.linalg.solve(A,e)
           except: sol=np.linalg.lstsq(A,e,rcond=None)[0]
           delta = -(G.T).dot(sol)
           # optional ellip
           gmax=grad_lag_one(h,n,int(np.argmax(c)))
           if np.dot(gmax,delta) >= -1e-15:
               delta=-gmax
       dnorm=float(np.max(np.abs(delta)))
       if dnorm < 1e-14: break
       accepted=False
       for target_step in (0.02,0.01,0.005,0.002,0.0005):
           h1=quick_fix_sum(h + delta * (target_step/dnorm), n)
           c1=corr_c(h1,n); m1=float(c1.max())

import numpy as np
import time
from scipy.optimize import minimize


def corr_c(h, n):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def compute_c5(h, n):
    return float(np.max(corr_c(h, n)))


def smooth_f(h, n, a, dx):
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(c.max())
    if a <= 0.0:
        return m
    z = np.clip(a * (c - m), -700.0, 0.0)
    s = float(np.exp(z).sum())
    if s <= 0.0:
        return m
    return m + float(np.log(s)) / a


def smooth_grad(h, n, a, dx):
    h = np.asarray(h, dtype=float)
    c = corr_c(h, n)
    m = float(c.max())
    z = np.clip(a * (c - m), -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    g = np.zeros(n, dtype=float)
    nc = len(c)
    if a < 80.0:
        idx = np.arange(nc)
    else:
        topk = 128 if a < 400.0 else 48
        if topk >= nc:
            idx = np.arange(nc)
        else:
            idx = np.argpartition(c, nc - topk)[-topk:]
    for ii in idx:
        w = float(e[ii] / s)
        if w < 1e-15:
            continue
        d = int(ii) - (n - 1)
        if d >= 0:
            L = n - d
            g[:L] += w * (1.0 - h[d:])
            g[d:] -= w * h[:L]
        else:
            e2 = -d
            L = n - e2
            g[e2:] += w * (1.0 - h[:L])
            g[:L] -= w * h[e2:]
    return g * dx


def grad_lag_one(h, n, p):
    d = int(p) - (n - 1)
    g = np.zeros(n, dtype=float)
    if d >= 0:
        L = n - d
        g[:L] += 1.0 - h[d:]
        g[d:] -= h[:L]
    else:
        e2 = -d
        L = n - e2
        g[e2:] += 1.0 - h[:L]
        g[:L] -= h[e2:]
    return g * (2.0 / float(n))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    target = 0.5 * n
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    if hi <= lo:
        hi = lo + 1.0
    for _ in range(48):
        mid = 0.5 * (lo + hi)
        v = np.clip(h - mid, 0.0, 1.0)
        s = float(v.sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    v = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(v.sum())
    if abs(diff) > 1e-12:
        if diff > 0.0:
            m = v < 1.0
            s = (1.0 - v[m]).sum()
            if s > 0.0:
                v[m] += diff * (1.0 - v[m]) / s
        else:
            m = v > 0.0
            s = v[m].sum()
            if s > 0.0:
                v[m] += diff * v[m] / s
    return np.clip(v, 0.0, 1.0)


def quick_fix_sum(h, n):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    target = 0.5 * n
    for _ in range(8):
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            return h
        if diff > 0.0:
            m = h < 1.0
            s = (1.0 - h[m]).sum()
            if s <= 1e-15:
                break
            h[m] += diff * (1.0 - h[m]) / s
        else:
            m = h > 0.0
            s = h[m].sum()
            if s <= 1e-15:
                break
            h[m] += diff * h[m] / s
    if abs(float(h.sum()) - target) > 1e-10:
        return project_feasible(h, n)
    return np.clip(h, 0.0, 1.0)


def to_binary(h, n):
    h = np.asarray(h, dtype=float)
    b = np.zeros(int(n), dtype=float)
    o = np.argsort(h)[::-1]
    b[o[: n // 2]] = 1.0
    if n % 2 and o.size > n // 2:
        b[o[n // 2]] = 0.5
    return b


def evaluate_fixed(h, n):
    h = quick_fix_sum(h, n)
    return h, compute_c5(h, n)


def resample_h(h, n, nt):
    xo = np.linspace(0.0, 2.0, int(n))
    xn = np.linspace(0.0, 2.0, int(nt))
    return np.interp(xn, xo, np.asarray(h, dtype=float))


def pen_obj(x, *args):
    n = len(x)
    a = float(args[0]) if len(args) > 0 else 1000.0
    dx = float(args[1]) if len(args) > 1 else 2.0 / n
    rho = float(args[2]) if len(args) > 2 else 0.0
    val = smooth_f(x, n, a, dx)
    if rho != 0.0:
        diff = float(x.sum()) - 0.5 * n
        val += rho * diff * diff
    return val


def pen_jac(x, *args):
    n = len(x)
    a = float(args[0]) if len(args) > 0 else 1000.0
    dx = float(args[1]) if len(args) > 1 else 2.0 / n
    rho = float(args[2]) if len(args) > 2 else 0.0
    g = smooth_grad(x, n, a, dx)
    if rho != 0.0:
        diff = float(x.sum()) - 0.5 * n
        g += 2.0 * rho * diff
    return g


def lbfgs_smooth(x, n, alpha, rho, maxiter=None):
    x = np.asarray(x, dtype=float).copy()
    n = int(n)
    dx = 2.0 / n
    if maxiter is None:
        if n <= 450:
            maxiter = 2500
        elif n <= 700:
            maxiter = 1200
        else:
            maxiter = 600
    maxiter = int(max(10, maxiter))
    try:
        opts = {
            "maxiter": maxiter,
            "maxfun": max(1000, 50 * maxiter),
            "ftol": 1e-12,
            "gtol": 1e-10,
            "maxls": 20,
            "disp": False,
        }
        res = minimize(
            pen_obj,
            x,
            method="L-BFGS-B",
            bounds=(0.0, 1.0),
            jac=pen_jac,
            args=(alpha, dx, rho),
            options=opts,
        )
        if res.x is not None:
            return np.asarray(res.x, dtype=float)
    except Exception:
        pass
    return x


def smo_obj(x, *args):
    n = len(x)
    a = float(args[0]) if len(args) > 0 else 1000.0
    dx = float(args[1]) if len(args) > 1 else 2.0 / n
    return smooth_f(x, n, a, dx)


def smo_jac(x, *args):
    n = len(x)
    a = float(args[0]) if len(args) > 0 else 1000.0
    dx = float(args[1]) if len(args) > 1 else 2.0 / n
    return smooth_grad(x, n, a, dx)


def sum_fun(x, *args):
    return float(x.sum() - 0.5 * len(x))


def sum_jac(x, *args):
    return np.ones(len(x), dtype=float)


def slsqp_smooth(x, n, alpha, maxiter=80):
    x = np.asarray(x, dtype=float).copy()
    n = int(n)
    dx = 2.0 / n
    maxiter = int(max(10, maxiter))
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": sum_fun, "jac": sum_jac}]
        opts = {"maxiter": maxiter, "ftol": 1e-12, "disp": False}
        res = minimize(
            smo_obj,
            x,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            jac=smo_jac,
            args=(alpha, dx),
            options=opts,
        )
        if res.x is not None:
            return np.asarray(res.x, dtype=float)
    except Exception:
        pass
    return x

            rhs[: e.size] = -e
            A = G.dot(G.T)
            r = 1e-8 * max(1.0, float(A.trace()) / float(A.shape[0]))
            A = A + np.eye(A.shape[0]) * r

                except Exception:
                    sol = np.zeros(A.shape[0])
            delta = G.T.dot(sol)
            delta -= delta.mean()
            if float(np.dot(g0, delta)) >= -1e-16:

                delta -= delta.mean()
        dn = float(np.max(np.abs(delta)))
        if dn < 1e-14:

        ok = False
