            for i in down:
                if time.time() > deadline:
                    return h, best_c
                for j in up:
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            improved_any = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if not improved and h[j] + step <= 1.0 and h[i] - step >= 0.0:
                        h[j] += step
                        h[i] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            improved_any = True
                        else:
                            h[j] -= step
                            h[i] += step
            if not improved and not improved_any:
                break
        if not improved_any:
            break
    return h, best_c

def directed_line_search(h, n, deadline, k_top=60):
    best_c = compute_c5(h, n)
    for _ in range(200):
        if time.time() > deadline:
            break
        g = topk_grad(h, n, 5, 50000, 0.0)
        up = np.argsort(g)[::-1][:k_top]
        down = np.argsort(g)[:k_top]
        improved = False
        for i in down:
            if time.time() > deadline:
                return h, best_c
            for j in up:
                if i == j:
                    continue
                for direction, ii, jj in [(1, i, j), (-1, j, i)]:
                    if direction == 1:
                        max_step = min(0.02, 1.0 - h[i], h[j])
                    else:
                        max_step = min(0.02, 1.0 - h[j], h[i])
                    if max_step < 1e-10:
                        continue
                    lo, hi_s = 0.0, max_step
                    for _ in range(12):
                        if time.time() > deadline:
                            return h, best_c
                        mid = (lo + hi_s) / 2.0
                        if mid < 1e-10:
                            break
                        h[ii] += mid
                        h[jj] -= mid
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            lo = mid
                            improved = True
                        else:
                            hi_s = mid
                        h[ii] -= mid
                        h[jj] += mid
                    if improved:
                        step = lo
                        h[ii] += step
                        h[jj] -= step
                        best_c = compute_c5(h, n)
        if not improved:
            break
    return h, best_c

def swap_search(h, n, deadline, radius=150):
    best_c = compute_c5(h, n)
    for step in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]:
        any_imp = False
        while time.time() < deadline - 2:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline - 2:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            any_imp = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if not improved and h[j] + step <= 1.0 and h[i] - step >= 0.0:
                        h[j] += step
                        h[i] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            any_imp = True
                        else:
                            h[j] -= step
                            h[i] += step
            if not improved and not any_imp:
                break
        if not any_imp:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(650, 0.5)
        n_init = 650

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS-B with top-k focus (major time)
    t1 = min(t0 + 300, deadline - 500)
    best_h, best_c = lbfgs_topk(best_h, best_n, t1)

    # Phase 2: SLSQP polish
    if time.time() < deadline - 400:
        t2 = min(time.time() + 80, deadline - 400)
        h, c = slsqp_polish(best_h, best_n, t2, alpha=10000000, maxiter=8000)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 3: Directed line search
    if time.time() < deadline - 300:
        t3 = min(time.time() + 60, deadline - 300)
        h, c = directed_line_search(best_h, best_n, t3, k_top=60)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 4: Fine gradient search
    if time.time() < deadline - 200:
        t4 = min(time.time() + 50, deadline - 200)
        h, c = fine_gradient_search(best_h, best_n, t4, k_top=60)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 5: Try n=700
    if time.time() < deadline - 180:
        n_new = 700
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_new)
        alloc = min(100, (deadline - time.time() - 130) * 0.5)
        if alloc > 30:
            h_t, c_t = lbfgs_topk(h_interp, n_new, time.time() + alloc)
            if time.time() < deadline - 100:
                t5 = min(time.time() + 40, deadline - 100)
                h_t, c_t = slsqp_polish(h_t, n_new, t5, alpha=10000000, maxiter=4000)
            if c_t < best_c:
                best_c, best_h, best_n = c_t, h_t, n_new

    # Phase 6: Local swap search on best
    if time.time() < deadline - 60:
        t6 = min(time.time() + 40, deadline - 55)
        best_h, best_c = swap_search(best_h, best_n, t6, radius=100)

    # Phase 7: Very fine gradient search
    if time.time() < deadline - 10:
        best_h, best_c = fine_gradient_search(best_h, best_n, deadline - 5, k_top=40)

    # Phase 8: Final SLSQP
    if time.time() < deadline - 2:
        h_f, c_f = slsqp_polish(best_h, best_n, deadline - 1, alpha=10000000, maxiter=2000)
        if c_f < best_c:
            best_c, best_h = c_f, h_f

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
