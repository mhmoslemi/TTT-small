    target = 0.5 * n

            if s < 1e-14: break

            s = h.sum()
            if s < 1e-14: break

def build_A(h,n):

import time, math
import numpy as np
from scipy.optimize import minimize

G = {'n': 2, 'dx': 1.0, 'alpha': 1.0, 'target': 1.0, 'ones': None, 'p': None}

def eval_c5(h, n):
    h = np.asarray(h, float)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_feasible(h, n):
    h = np.asarray(h, float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if idx.size:
            i = int(idx[0])
            nv = target - (h.sum() - h[i])
            if 0.0 <= nv <= 1.0:
                h[i] = nv
    return h

def binary_round(h, n):
    h = np.asarray(h, float)
    order = np.argsort(h)[::-1]
    hb = np.zeros(n, float)
    hb[order[:n // 2]] = 1.0
    return hb

def interp_h(h, n, nn):
    if nn == n:
        return np.asarray(h, float).copy()
    xo = np.linspace(0.0, 2.0, int(n))
    xn = np.linspace(0.0, 2.0, int(nn))
    return np.interp(xn, xo, np.asarray(h, float))

def get_cdx(h, n):
    h = np.asarray(h, float)
    return np.correlate(h, 1.0 - h, mode='full')[::-1].astype(float) * (2.0 / n)

def objective(x):
    x = np.asarray(x, float)
    n = G['n']
    dx = G['dx']
    c = np.correlate(x, 1.0 - x, mode='full')[::-1].astype(float) * dx
    p = G['p']
    if p is None:
        a = G['alpha']
        m = float(c.max())
        sh = a * (c - m)
        s = float(sh.max())
        return m + (s + math.log(float(np.exp(sh - s).sum()))) / a
    return float(np.dot(c, p))

def grad_fun(x):
    x = np.asarray(x, float)
    n = G['n']
    dx = G['dx']
    c = np.correlate(x, 1.0 - x, mode='full')[::-1].astype(float) * dx
    p = G['p']
    if p is None:
        a = G['alpha']
        m = float(c.max())
        sh = a * (c - m)
        s = float(sh.max())
        e = np.exp(sh - s)
        p = e / float(e.sum())
    g1 = 1.0 - x
    conv1 = np.convolve(p, g1, mode='full')
    conv2 = np.convolve(p[::-1], x, mode='full')
    st = n - 1
    return (conv1[st:st + n] - conv2[st:st + n]) * dx

def sum_fun(x):
    return float(np.asarray(x, float).sum() - G['target'])

def sum_jac(x):
    return G['ones'].copy()

def slsqp_solve(h, n, alpha=None, p_override=None, maxiter=80, ftol=1e-12):
    h = np.asarray(h, float).copy()
    G['n'] = n
    G['dx'] = 2.0 / n
    G['target'] = 0.5 * n
    G['ones'] = np.ones(n)
    if p_override is None:
        G['p'] = None
        G['alpha'] = float(alpha if alpha is not None else 1.0)
    else:
        G['p'] = np.asarray(p_override, float)
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(objective, h, method='SLSQP', jac=grad_fun,
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': int(maxiter), 'ftol': float(ftol), 'disp': 0})
        x = res.x
    except Exception:
        x = h
    return project_feasible(x, n)

def build_A(h, n):
    h = np.asarray(h, float)
    L = 2 * n - 1
    A = np.zeros((n, L), float)
    for s in range(-(n - 1), n):
        t = n - 1 + s
        if s >= 0:
            A[:n - s, t] += 1.0 - h[s:n]
            A[s:n, t] -= h[:n - s]
        else:
            u = -s
            A[:n - u, t] += 1.0 - h[:n - u]
            A[u:n, t] -= h[:n - u]
    return A

def apply_A_change(A, q, delta):
    n = A.shape[0]
    C = n - 1
    p = np.arange(n)
    A[p, C + q - p] -= delta
    A[p, C + p - q] -= delta

def binary_improve(h, n, deadline, seed=42, target_c=0.3808):
    h = np.asarray(h, float).copy()
    h = binary_round(h, n)
    target_count = int(math.floor(target_c * n / 2.0 + 1e-12))
    c = np.correlate(h, 1.0 - h, mode='full')[::-1].astype(float)
    C = n - 1
    cur = float(c.max())
    A = build_A(h, n)
    best_h = h.copy()
    best_count = int(math.ceil(cur - 1e-7))
    cur_equal = int(np.count_nonzero(c >= cur - 0.5))
    rng = np.random.RandomState(seed)
    streak = 0
    T = 8.0
    updates = 0
    while time.time() < deadline and cur > target_count + 1e-6:
        zeros = np.where(h <= 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        improved = False
        scan_lim = min(deadline, time.time() + 10.0)
        A1 = A[ones]
        K = ones.size
        block = 16
        for st in range(0, zeros.size, block):
            if time.time() > scan_lim:
                break
            zb = zeros[st:st + block]
            B = zb.size
            if B == 0 or K == 0:
                continue
            cand = c[None, None, :] + A[zb][:, None, :] - A1[None, :, :]
            N = B * K
            flat = np.arange(N)
            bidx = flat // K
            ridx = flat % K
            ia = zb[bidx]
            ja = ones[ridx]
            cand[flat, C] -= 2.0
            cand[flat, C + ja - ia] += 1.0
            cand[flat, C + ia - ja] += 1.0
            vals = cand.max(axis=2)
            minv = float(vals.min())
            if minv < cur - 1e-7:
                bi, oj = np.unravel_index(int(np.argmin(vals)), vals.shape)
                i = int(zb[bi])
                j = int(ones[oj])
                c = np.rint(c + A[i] - A[j] + 1e-9)
                c[C] -= 2.0
                p1 = C + j - i
                p2 = C + i - j
                c[p1] += 1.0
                c[p2] += 1.0
                cur = float(c.max())
                cur_equal = int(np.count_nonzero(c >= cur - 0.5))
                apply_A_change(A, i, 1.0)
                apply_A_change(A, j, -1.0)
                h[i] = 1.0
                h[j] = 0.0
                updates += 1
                if cur + 1e-7 < best_count:
                    best_count = int(math.ceil(cur - 1e-7))
                    best_h = h.copy()
                improved = True
                break
        if improved:
            streak = 0
            T = 8.0
        else:
            streak += 1
            tan = min(deadline, time.time() + max(4.0, 1.5 * streak))
            T = 8.0 if streak > 6 else max(T, 1.0)
            for _ in range(1000000):
                if time.time() > tan:
                    break
                ai = rng.randint(zeros.size)
                bi = rng.randint(ones.size)
                i = int(zeros[ai])
                j = int(ones[bi])
                if i == j:
                    continue
                val = c + A[i] - A[j]
                val[C] -= 2.0
                val[C + j - i] += 1.0
                val[C + i - j] += 1.0
                val = np.rint(val + 1e-9)
                nm = float(val.max())
                nd = nm - cur
                if nd < -1e-7:
                    acc = True
                elif abs(nd) < 1e-6:
                    ne = int(np.count_nonzero(val >= nm - 0.5))
                    acc = int(ne < cur_equal) or (cur_equal > 0 and int(rng.random() < 0.12))
                else:
                    acc = False
                    if nd <= 1.0 + 1e-6 and rng.random() < 0.02:
                        acc = True
                    elif T > 0.05:
                        acc = int(rng.random() < math.exp(-8.0 * nd / T))
                if acc:
                    c = val
                    cur = nm
                    cur_equal = int(np.count_nonzero(c >= cur - 0.5))
                    apply_A_change(A, i, 1.0)
                    apply_A_change(A, j, -1.0)
                    h[i] = 1.0
                    h[j] = 0.0
                    zeros[ai] = j
                    ones[bi] = i
                    updates += 1
                    if cur + 1e-7 < best_count:
                        best_count = int(math.ceil(cur - 1e-7))
                        best_h = h.copy()
                T = max(0.03, T * 0.999982)
        if updates > 5000:
            A = build_A(h, n)
            updates = 0
    cc = np.correlate(best_h, 1.0 - best_h, mode='full')[::-1]
    return best_h, float(cc.max()) * (2.0 / n)

def continuous_refine(h0, n, deadline, seed=42, starts=None):
    h0 = project_feasible(np.asarray(h0, float), n)
    if starts is None:
        starts = [h0]
    best_h = h0.copy()
    best_c = eval_c5(best_h, n)
    rem = deadline - time.time()
    if rem <= 5:
        return best_h, best_c
    if rem > 550:
        alphas = [20, 80, 350, 1500, 7000, 35000, 180000, 900000, 4000000, 20000000]
        maxit = 70
        do_top = True
    elif rem > 200:
        alphas = [30, 150, 800, 4000, 25000, 150000, 1000000, 7000000]
        maxit = 45
        do_top = True
    else:
        alphas = [80, 800, 8000, 80000, 800000]
        maxit = 25
        do_top = False
    rng = np.random.RandomState(seed)
    for si, hs in enumerate(starts):
        if time.time() > deadline - 3:
            break
        h = project_feasible(np.asarray(hs, float), n)
        lim = len(alphas) if si == 0 else max(3, len(alphas) // 2)
        for alpha in alphas[:lim]:
            if time.time() > deadline - 3:
                break
            h = slsqp_solve(h, n, alpha=alpha, maxiter=maxit, ftol=1e-12)
            ch = eval_c5(h, n)
            if ch < best_c - 1e-14:
                best_c = ch
                best_h = h.copy()
            if do_top and alpha >= 4000:
                cv = get_cdx(h, n)
                for q in (3, 7):
                    if time.time() > deadline - 5:
                        break
                    iq = np.argsort(cv)[-q:]
                    p = np.zeros(2 * n - 1, float)
                    p[iq] = 1.0 / float(q)
                    hq = slsqp_solve(h, n, p_override=p, maxiter=max(maxit // 2, 20), ftol=1e-12)
                    cq = eval_c5(hq, n)
                    if cq < best_c - 1e-14:
                        best_c = cq
                        best_h = hq.copy()
                        h = hq.copy()
                    elif cq < ch - 1e-14:
                        h = hq.copy()
                    cv = get_cdx(h, n)
        if time.time() < deadline - 20:
            hp = best_h + rng.normal(0.0, 0.005, n)
            hp = project_feasible(hp, n)
            for alpha in [5000, 50000, 500000, 5000000, 50000000]:
                if time.time() > deadline - 4:
                    break
                hp = slsqp_solve(hp, n, alpha=alpha, maxiter=max(maxit // 2, 20), ftol=1e-12)
                cp = eval_c5(hp, n)
                if cp < best_c - 1e-14:
                    best_c = cp
                    best_h = hp.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(60.0, float(budget_s)) - 10.0
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = int(len(h0))
    except Exception:
        h0 = None
        n0 = 0
    if n0 < 20 or (n0 % 2 == 1):
        if n0 % 2 == 1:
            n0 -= 1
        if n0 < 20:
            n0 = 400
        h0 = np.zeros(n0)
        h0[rng.permutation(n0)[:n0 // 2]] = 1.0
    else:
        h0 = h0[:n0]
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = eval_c5(best_h, n0)
    best_n = n0
    off1 = min(160.0, 0.20 * total)
    off2 = min(off1 + 420.0, 0.75 * total)
    off3 = min(off2 + 180.0, 0.90 * total)
    p1 = t0 + off1
    p2 = t0 + off2
    p3 = t0 + off3
    if time.time() + 15 < p1:
        h1, c1 = continuous_refine(h0, n0, p1, seed + 1)
        if c1 < best_c - 1e-14:
            best_h = h1.copy()
            best_c = c1
            best_n = n0
    if time.time() + 15 < p2:
        hb = binary_round(best_h, best_n)
        hb2, cb2 = binary_improve(hb, best_n, p2, seed + 2, 0.3808)
        if cb2 < best_c - 1e-14:
            best_h = hb2.copy()
            best_c = cb2
            best_n = best_n
    if best_c > 0.3808 - 1e-12 and time.time() + 15 < p3:
        other_n = []
        for nn in (300, 200, 480):
            if nn % 2 == 0 and nn != best_n and nn < 1000:
                other_n.append(nn)
        m = max(1, len(other_n))
        per = max(20.0, (p3 - time.time()) / m)
        for nn in other_n:
            end = min(p3, time.time() + per)
            if end < time.time() + 15:
                break
            hh = interp_h(best_h, best_n, nn)
            hh = project_feasible(hh, nn)
            hb = binary_round(hh, nn)
            hb2, cb2 = binary_improve(hb, nn, end, seed + 3 + nn, 0.3808)
            if cb2 < best_c - 1e-14:
                best_h = hb2.copy()
                best_c = cb2
                best_n = nn
            if best_c <= 0.3808 - 1e-12:
                break
    final_starts = [best_h]
    if n0 != best_n:
        final_starts.append(project_feasible(interp_h(h0, n0, best_n), best_n))
    else:
        final_starts.append(h0)
    if time.time() + 15 < dl - 5:
        hf, cf = continuous_refine(best_h, best_n, dl - 5, seed + 10, final_starts)
        if cf < best_c - 1e-14:
            best_h = hf.copy()
            best_c = cf
    best_h = project_feasible(best_h, best_n)
    best_c = eval_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
