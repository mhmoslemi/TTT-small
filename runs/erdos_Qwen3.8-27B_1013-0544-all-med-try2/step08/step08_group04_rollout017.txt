def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_corr(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(h, n, alpha):
    c = compute_corr(h, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # Build gradient efficiently
    for m in range(2 * n - 1):

            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax_obj(x, n, alpha, mu):
    c = compute_corr(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smax_grad(x, n, alpha, mu):
    c = compute_corr(x, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def subgrad_hess_grad(x, n):
    """Subgradient using uniform weights over near-max correlations."""
    c = compute_corr(x, n)
    mx = c.max()
    active = np.where(c > mx - 0.01)[0]
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad, mx, active

def subgrad_phase(h, n, deadline, eta_init=0.01):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_min = best_c
    x_avg = h.copy()
    eta = eta_init
    t = 1
    for it in range(20000):
        if time.time() > deadline:
            break
        g, mx, active = subgrad_hess_grad(h, n)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-20:
            break
        # Polyak step
        if c_min > 1e-10:
            eta = max(eta * 0.999, (mx - c_min + 0.01) / g_norm_sq)
        h_new = project_feasible(h - eta * g, n)
        # Nesterov averaging
        x_avg = x_avg + (h_new - x_avg) / (t + 1)
        t += 1
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        c_min = min(c_min, c_new)
        c_avg = compute_c5(project_feasible(x_avg, n), n)
        if c_avg < best_c:
            best_c = c_avg
            best_h = project_feasible(x_avg, n).copy()
    return best_h, best_c

def pair_search_optimal(h, n, deadline):
    """Pair search with exact 1D optimal move (linear approx)."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for _ in range(50):
        if time.time() > deadline:
            break
        improved = False
        c = compute_corr(h, n)
        mx = c.max()
        active = np.where(c > mx - 0.03)[0]
        # Find coordinates in active correlations
        coord_set = set()
        for m in active:
            k = m - (n - 1)
            if k >= 0:
                nv = n - k
                coord_set.update(range(0, nv))
                coord_set.update(range(k, n))
            else:
                kk = -k
                nv = n - kk
                coord_set.update(range(0, nv))
                coord_set.update(range(kk, n))
        coords = sorted(coord_set)
        if len(coords) > 150:
            coords = sorted(set(coords[::max(1, len(coords)//150)]))
        for i in coords:
            if time.time() > deadline:
                return h, best_c
            for j in coords:
                if i == j:
                    continue
                # Compute a_k = -dx*(h[i+k]+h[i-k]-h[j+k]-h[j-k])
                # Find optimal delta
                hi = h[i]
                hj = h[j]
                d_min = max(-hi, hj - 1.0)
                d_max = min(1.0 - hi, hj)
                if d_max <= d_min:
                    continue
                # Evaluate at boundaries and at crossing point
                candidates = [d_min, 0.0, d_max]
                # Find the two active correlations most affected
                best_cand = None
                best_val = best_c
                for delta in candidates:
                    if abs(delta) < 1e-12:
                        continue
                    h[i] = hi + delta
                    h[j] = hj - delta
                    c_new = compute_c5(h, n)
                    if c_new < best_val:
                        best_val = c_new
                        best_cand = delta
                # Also try the crossing: where two top correlations cross
                c_orig = compute_corr(h, n)
                top2 = np.argsort(-c_orig)[:4]
                for ii in range(len(top2)):
                    for jj in range(ii + 1, len(top2)):
                        m1, m2 = top2[ii], top2[jj]
                        k1 = m1 - (n - 1)
                        k2 = m2 - (n - 1)
                        # a_k = -dx*(h[i+k]+h[i-k]-h[j+k]-h[j-k])
                        def hk(idx, kk):
                            if 0 <= idx + kk < n:
                                return h[idx + kk]
                            return 0.0
                        def hk2(idx, kk):
                            if 0 <= idx - kk < n:
                                return h[idx - kk]
                            return 0.0
                        a1 = -dx * (hk(i, k1) + hk2(i, k1) - hk(j, k1) - hk2(j, k1))
                        a2 = -dx * (hk(i, k2) + hk2(i, k2) - hk(j, k2) - hk2(j, k2))
                        if abs(a1 - a2) > 1e-15:
                            delta_cross = (c_orig[m2] - c_orig[m1]) / (a1 - a2)
                            if d_min < delta_cross < d_max:
                                candidates.append(delta_cross)
                for delta in candidates:
                    if delta is None or not (d_min < delta < d_max):
                        continue
                    if abs(delta) < 1e-12:
                        continue
                    h[i] = hi + delta
                    h[j] = hj - delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                        break
                    h[i] = hi
                    h[j] = hj
                else:
                    h[i] = hi
                    h[j] = hj
                    continue
                break
            if time.time() > deadline:
                return h, best_c
        if not improved:
            break
    return h, best_c

def lbfgs_optimize(h, n, deadline):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100.0, 5.0), (500.0, 10.0), (2000.0, 50.0),
               (10000.0, 100.0), (50000.0, 200.0), (500000.0, 500.0),
               (2000000.0, 1000.0)]
    for alpha, mu in configs:
        if time.time() > deadline:
            break
        maxit = max(50, min(300, int((deadline - time.time()) * 2)))
        try:
            res = minimize(smax_obj, h, jac=smax_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': maxit * 15,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c
