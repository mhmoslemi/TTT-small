def coordinate_descent(h, n, max_passes=100, time_limit=None):
    """Cyclic coordinate descent exploiting linearity of c_k in each h[j]."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    
    for p in range(max_passes):
        if time_limit and time.time() > time_limit:
            break
        improved = False
        for j in range(n):
            old_val = h[j]
            # Compute c when h[j]=0 and h[j]=1
            h[j] = 0.0
            c0 = np.correlate(h, 1-h, mode='full') * dx
            h[j] = 1.0
            c1 = np.correlate(h, 1-h, mode='full') * dx
            h[j] = old_val
            
            slope = c1 - c0  # slope per k
            intercept = c0   # intercept per k
            
            # Find t in [0,1] minimizing max_k (slope[k]*t + intercept[k])
            # Evaluate at multiple points
            ts = np.linspace(0, 1, 100)
            vals = np.max(intercept + np.outer(ts, slope), axis=1)
            best_t = ts[np.argmin(vals)]
            
            if best_t != old_val:
                h[j] = best_t
                c_new = compute_c5(h, n)
                if c_new < best_c:
                    best_c = c_new
                    improved = True
                else:
                    h[j] = old_val
        if not improved:
            break
    
    return h, best_c

Let me use a different approach: find the minimum by solving a small LP or by bisection on the subgradient.

The subgradient approach:
- At point t, the function value is f(t) = max_k (a_k * t + b_k)
- The subgradient is conv{a_k : k ∈ active(t)} where active(t) = {k : a_k * t + b_k = f(t)}
- The minimum is where 0 ∈ subgradient, i.e., min(active slopes) ≤ 0 ≤ max(active slopes)

Bisection:
- At t=0: if max(active slopes) < 0, min is at 0. If min(active slopes) > 0, min is at 1.
- Otherwise, bisect between 0 and 1.

At each bisection step, I compute f(t) and the active set, check the subgradient condition.

This takes O(log(1/ε)) * O(n) = O(n log(1/ε)) per position.

def find_min_convex_pline(a, b, tol=1e-10, max_iter=100):
    """Find t in [0,1] minimizing max_i (a[i]*t + b[i]).
    a, b are arrays of same length. Returns (t*, f*)."""
    def f(t):
        return np.max(a * t + b)
    
    def subgrad_condition(t):
        val = f(t)
        active = a * t + b >= val - 1e-12
        min_slope = np.min(a[active])
        max_slope = np.max(a[active])
        return min_slope <= 0 <= max_slope or (max_slope < 0) or (min_slope > 0)
    
    # Check endpoints first
    f0 = f(0)
    f1 = f(1)
    if f0 <= f1:
        # Check if 0 is optimal
        active0 = b >= f0 - 1e-12
        if np.min(a[active0]) >= -1e-12:
            return 0, f0
    else:
        active1 = a + b >= f1 - 1e-12
        if np.max(a[active1]) <= 1e-12:
            return 1, f1
    
    # Bisection
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        val = f(mid)
        active = a * mid + b >= val - 1e-12
        min_s = np.min(a[active])
        max_s = np.max(a[active])
        if 0 <= min_s and max_s >= 0:
            # subgradient contains 0 (or all slopes >= 0) → min is at lo
            break
        if max_s <= 0 and min_s <= 0:
            # all slopes <= 0 → min is at hi
            break
        # The function is decreasing at mid (more negative slopes active)
        # or increasing (more positive slopes active)
        if min_s > 0:
            hi = mid  # function is increasing, go left
        elif max_s < 0:
            lo = mid  # function is decreasing, go right
        else:
            # 0 is in the subgradient at mid
            break
    t_star = (lo + hi) / 2
    return t_star, f(t_star)

def grad_c5(h, n, alpha):
    """Gradient of smooth max of correlation coefficients."""
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    shifted = alpha * (c - c.max())
    mx = shifted.max()
    w = np.exp(shifted - mx)

    dx = 2.0 / n

    # d(c[m])/d(h[j]) = dx * (1-h[j+m-(n-1)]) - dx * h[j-(m-(n-1))] (valid terms)

        wm = w[m] * dx

        # - h[j] * (1-h[j+k]) if j+k < n (j is the first factor)
        # - h[j-k] * (1-h[j]) if j-k >= 0 (j is the second factor)
        if k >= 0:

            grad[0:n-k] += wm * (1.0 - h[k:n])

            grad[k:n] += wm * (-h[0:n-k])

            grad[0:n-kk] += wm * (-h[kk:n])

def coordinate_descent_pass_v2(h, n, n_grid=100):
    dx = 2.0 / n
    improved = False
    best_c = np.max(np.correlate(h, 1-h, mode='full') * dx)
    
    for j in range(n):
        old = h[j]
        # Quick analytical slope
        k_vals = np.arange(2*n-1) - (n-1)
        valid1 = (j + k_vals >= 0) & (j + k_vals < n)
        valid2 = (j - k_vals >= 0) & (j - k_vals < n)
        slope = np.zeros(2*n-1)
        slope[valid1] += dx * (1.0 - h[j + k_vals[valid1]])
        slope[valid2] -= dx * h[j - k_vals[valid2]]
        
        # Get current c (with h at current values)
        c = np.correlate(h, 1-h, mode='full') * dx
        # c_k(t) = c_k + slope_k * (t - old)
        # So c_k(t) = (c_k - slope_k * old) + slope_k * t
        intercept = c - slope * old
        
        # Find best t
        ts = np.linspace(0, 1, n_grid)
        vals = np.max(intercept + np.outer(ts, slope), axis=1)
        idx = np.argmin(vals)
        best_t = ts[idx]
        
        if abs(best_t - old) > 1e-8:
            h[j] = best_t
            c_new = np.max(np.correlate(h, 1-h, mode='full') * dx)
            if c_new < best_c:
                best_c = c_new
                improved = True
            else:
                h[j] = old
    return h, improved

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h_1(x, k, n))
            grad[k:n] += wm * (-h_2(x, k, n))
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h_1(x, kk, n))
            grad[0:nv] += wm * (-h_2(x, kk, n))
    return grad

def h_1(x, k, n):
    return x[k:n]

def h_2(x, k, n):
    return x[0:n-k]

def lbfgs_obj(x, n, alpha, mu):
    v = smooth_max_value(x, n, alpha)
    v += mu * (x.sum() - n / 2.0) ** 2
    return float(v)

def lbfgs_grad(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_continuation(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        mu = 5.0
        try:
            res = minimize(lbfgs_obj, h, jac=lbfgs_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 800, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': 10000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def coordinate_descent(h, n, deadline, max_passes=200, n_grid=120):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    k_vals = np.arange(2 * n - 1) - (n - 1)
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            valid1 = (j + k_vals >= 0) & (j + k_vals < n)
            valid2 = (j - k_vals >= 0) & (j - k_vals < n)
            slope = np.zeros(2 * n - 1)
            if np.any(valid1):
                slope[valid1] += dx * (1.0 - h[j + k_vals[valid1]])
            if np.any(valid2):
                slope[valid2] -= dx * h[j - k_vals[valid2]]
            c = np.correlate(h, 1 - h, mode='full') * dx
            intercept = c - slope * old
            ts = np.linspace(0, 1, n_grid)
            vals = np.max(intercept + np.outer(ts, slope), axis=1)
            idx = np.argmin(vals)
            best_t = ts[idx]
            # Refine around best
            if 0 < idx < n_grid - 1:
                ts2 = np.linspace(ts[idx - 1], ts[idx + 1], 200)
                vals2 = np.max(intercept + np.outer(ts2, slope), axis=1)
                best_t = ts2[np.argmin(vals2)]
            if abs(best_t - old) > 1e-10:
                h[j] = best_t
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                else:
                    h[j] = old
        if not improved:
            break
    return h, best_c

def fine_pair_search(h, n, deadline, steps=None, radius=40):
    if steps is None:
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    best_c = compute_c5(h, n)
    for d in steps:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] - d >= 0 and h[j] + d <= 1:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                    if h[j] - d >= 0 and h[i] + d <= 1:
                        h[j] -= d
                        h[i] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[j] += d
                            h[i] -= d
            if not improved:
                break
    return h, best_c

def gradient_guided_search(h, n, deadline, step_size=0.003):
    best_c = compute_c5(h, n)
    alphas = [100, 500, 2000, 10000]
    for alpha in alphas:
        while time.time() < deadline:
            improved = False
            g = smooth_max_grad(h, n, alpha)
            order = np.argsort(g)
            for ii in range(len(order)):
                if time.time() > deadline:
                    return h, best_c
                i = order[ii]
                for jj in range(max(0, ii - 30), min(len(order), ii + 30)):
                    j = order[jj]
                    if i == j:
                        continue
                    move = step_size * (g[i] - g[j])
                    if move <= 0:
                        continue
                    if h[i] - move >= 0 and h[j] + move <= 1:
                        h[i] -= move
                        h[j] += move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += move
                            h[j] -= move
            if not improved:
                break
    return h, best_c
