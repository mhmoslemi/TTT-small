import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def top_m_smooth(h, n, alpha, m):
    c = compute_c5_vec(h, n)
    part = np.partition(c, len(c) - m)[-m:]
    ax = alpha * (part - part.max())
    return float(part.max() + np.log(np.sum(np.exp(ax))) / alpha)

def top_m_grad(h, n, alpha, m):
    c = compute_c5_vec(h, n)
    idx = np.argsort(c)[::-1][:m]
    vals = c[idx]
    ax = alpha * (vals - vals.max())
    w = np.exp(ax)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(m):
        if w[mi] < 1e-15:
            continue
        m_idx = idx[mi]
        k = m_idx - (n - 1)
        wm = w[mi] * alpha * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (-h[k:])
            grad[k:] += wm * (1.0 - h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (-h[:nv])
            grad[:nv] += wm * (1.0 - h[kk:])
    return grad

def lbfgs_phase(h, n, deadline, m_list=None, alpha_list=None):
    if m_list is None:
        m_list = [8, 15, 30, 60, 120, 200]
    if alpha_list is None:
        alpha_list = [5000, 50000, 500000]
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for m in m_list:
        if time.time() > deadline - 5:
            break
        for alpha in alpha_list:
            if time.time() > deadline - 3:
                break
            def obj(x, m=m, a=alpha):
                return top_m_smooth(x, n, a, m)
            def gr(x, m=m, a=alpha):
                return top_m_grad(x, n, a, m)
            try:
                res = minimize(obj, h, jac=gr, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 800, 'ftol': 1e-18,
                                       'gtol': 1e-16, 'maxfun': 10000,
                                       'maxls': 200})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def slsqp_squeeze(h, n, deadline, alpha=5000000, m=40):
    def obj(x):
        return top_m_smooth(x, n, alpha, m)
    def gr(x):
        return top_m_grad(x, n, alpha, m)
    def eq(x):
        return float(x.sum() - n / 2.0)
    def eq_j(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_j}]
    try:
        res = minimize(obj, h, jac=gr, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 5000, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_directed_pairs(h, n, deadline, k_top=40):
    best_c = compute_c5(h, n)
    alphas = [1000, 5000, 50000, 500000]
    for alpha in alphas:
        while time.time() < deadline - 2:
            g = top_m_grad(h, n, alpha, 50)
            up = np.argsort(g)[-k_top:]
            down = np.argsort(g)[:k_top]
            improved = False
            for i in down:
                if time.time() > deadline - 2:
                    return h, best_c
                for j in up:
                    if i == j:
                        continue
                    for step in [0.003, 0.001, 0.0005, 0.0002, 0.0001]:
                        if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                            h[i] += step
                            h[j] -= step
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= step
                                h[j] += step
                            if improved:
                                break
                if improved:
                    break
            if not improved:
                break
    return h, best_c
