```python
def project_feasible(h, n):

    delta[valid_i] += d * (1 - 2*h[idx_i[valid_i]])
    delta[valid_j] -= d * (1 - 2*h[idx_j[valid_j]])
    return delta
```

```python
            if tot < 1e-14:
```

```python
import numpy as np
from scipy.signal import correlate as sp_correlate
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return sp_correlate(h, 1.0 - h, mode='full', method='fft') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def precompute_vecs(h, n):
    m = np.arange(2*n-1)
    idx = np.arange(n)[:, None] + m[None, :] - n + 1
    valid = (idx >= 0) & (idx < n)
    vecs = np.zeros((n, 2*n-1))
    flat_pos = np.where(valid)
    vecs[flat_pos] = 1.0 - 2.0 * h[idx[valid]]
    return vecs

def fast_pair_polish(h, n, deadline, radius=None, deltas=None):
    if radius is None:
        radius = n
    if deltas is None:
        deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    c = compute_c5_vec(h, n)
    best_c = c.max()
    
    for d in deltas:
        max_passes = 200
        for _ in range(max_passes):
            if time.time() > deadline:
                break
            vecs = precompute_vecs(h, n)
            found = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                js = np.arange(lo, hi)
                js = js[js != i]
                if len(js) == 0:
                    continue
                
                # Direction 1: h[i] += d, h[j] -= d
                if h[i] + d <= 1.0:
                    mask = h[js] >= d
                    if mask.any():
                        js1 = js[mask]
                        diff = vecs[i][None, :] - vecs[js1]
                        new_maxs = np.max(c[None, :] + d * diff, axis=1)
                        good = new_maxs < best_c - 1e-15
                        if good.any():
                            bi = np.argmin(new_maxs[good])
                            j = int(js1[bi])
                            h[i] += d
                            h[j] -= d
                            c += d * (vecs[i] - vecs[j])
                            best_c = float(np.min(new_maxs[good]))
                            found = True
                            break
                
                # Direction 2: h[i] -= d, h[j] += d
                if h[i] - d >= 0.0:
                    mask = h[js] + d <= 1.0
                    if mask.any():
                        js2 = js[mask]
                        diff = vecs[i][None, :] - vecs[js2]
                        new_maxs = np.max(c[None, :] - d * diff, axis=1)
                        good = new_maxs < best_c - 1e-15
                        if good.any():
                            bi = np.argmin(new_maxs[good])
                            j = int(js2[bi])
                            h[i] -= d
                            h[j] += d
                            c += -d * (vecs[i] - vecs[j])
                            best_c = float(np.min(new_maxs[good]))
                            found = True
                            break
            if not found:
                break
    
    h, best_c = project_feasible(h, n), compute_c5(h, n)
    return h, best_c

def gradient_step(h, n, alpha=100.0, top_k=30, step_size=0.1):
    c = compute_c5_vec(h, n)
    idx = np.argsort(c)[-top_k:]
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    
    vecs = precompute_vecs(h, n)
    # Gradient of smoothed max = sum w_k * grad c_k = sum w_k * vecs[:, k]
    grad = vecs[:, idx].T @ w  # shape (n,)
    
    # Projected gradient step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    c_new = compute_c5(h_new, n)
    if c_new < compute_c5(h, n):
        return h_new, c_new
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):

    try:

    best_h = h.copy()

                best_h = h.copy()
    
    # Phase 2: Main fast pair polish (global radius)
    if time.time() < deadline - 100:
        t2 = min(t0 + budget_s * 0.6, deadline - 80)
        h, best_c = fast_pair_polish(best_h, n, t2, radius=n,
                                      deltas=[0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5])
        best_h = h.copy()
    
    # Phase 3: Try different n
    for n_try in [400, 600]:
        if time.time() > deadline - 120:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = deadline - time.time() - 60
        if alloc < 30:
            continue
        h_try, c_try = fast_pair_polish(h_interp, n_try, time.time() + alloc * 0.7,
                                          radius=n_try,
                                          deltas=[0.02, 0.01, 0.005, 0.001, 0.0005, 0.0001])
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
            n = n_try
    
    # Phase 4: Fine local polish
    if time.time() < deadline - 5:
        h, best_c = fast_pair_polish(best_h, n, deadline - 2, radius=30,
                                      deltas=[0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6])
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```

```python
def precompute_grad(h, n):
    """Precompute G[i,m] = ∂c[m]/∂h[i] for all i, m"""
    m = np.arange(2*n-1)
    # A[i,m] = (1-h[i+m-n+1]) if 0<=i+m-n+1<n else 0
    idx_a = np.arange(n)[:, None] + m[None, :] - n + 1  # i + m - n + 1
    valid_a = (idx_a >= 0) & (idx_a < n)
    A = np.zeros((n, 2*n-1))
    A[valid_a] = 1.0 - h[idx_a[valid_a]]
    
    # B[i,m] = h[i+n-1-m] if 0<=i+n-1-m<n else 0
    idx_b = np.arange(n)[:, None] + n - 1 - m[None, :]  # i + n - 1 - m
    valid_b = (idx_b >= 0) & (idx_b < n)
    B = np.zeros((n, 2*n-1))
    B[valid_b] = h[idx_b[valid_b]]
    
    return A - B  # G[i,m] = ∂c[m]/∂h[i]
```

Then for a swap (h[i]+=d, h[j]-=d):
Δc = d * (G[i] - G[j])

And new_max = max(c + d*(G[i] - G[j]))

Let me verify with the example:
n=4, h = [a, b, c, d] (using different letters to avoid confusion)
c[0] = h[3]*(1-h[0]) = d*(1-a)

G[0, 0] = A[0,0] - B[0,0]
A[0,0]: idx_a = 0+0-4+1 = -3, invalid → 0
B[0,0]: idx_b = 0+4-1-0 = 3, valid → h[3] = d
G[0,0] = 0 - d = -d ✓ (matches ∂c[0]/∂h[0] = -h[3] = -d)

G[3, 0]:
A[3,0]: idx_a = 3+0-4+1 = 0, valid → 1-h[0] = 1-a
B[3,0]: idx_b = 3+4-1-0 = 6, invalid → 0
G[3,0] = (1-a) - 0 = 1-a
∂c[0]/∂h[3] = (1-a) + 0 = 1-a ✓
