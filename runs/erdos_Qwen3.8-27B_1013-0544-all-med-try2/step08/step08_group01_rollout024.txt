    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)

    m = c.max()
    s = alpha * (c - m)
    mx = s.max()

For the local search, the key insight is: we can compute the change in C5 analytically for a pair swap without recomputing the full correlation.

When we swap δ from position j to position i (h[i] += δ, h[j] -= δ):
The change in I(k) for each shift k is:
ΔI(k) = δ * [(1-h[j+k])·[valid] - h[j-k]·[valid] - (1-h[i+k])·[valid] + h[i-k]·[valid]] * dx

Wait, that's the change for a single unit. For a swap of δ:
ΔI(k) = δ * [∂I(k)/∂h[i] - ∂I(k)/∂h[j]] * dx ... no wait.

Actually, I(i) → I(i) + δ * ∂I/∂h[i] (approximately, for small δ)
And I(i) also changes because h[j] changes: I → I + δ*∂I/∂h[i] - δ*∂I/∂h[j]

But this is only a first-order approximation. For the exact change:
I(k) = Σ_m h[m](1-h[m+k])
After swap: h'[i] = h[i]+δ, h'[j] = h[j]-δ
I'(k) = I(k) + δ(1-h[i+k])·[valid_i] - δ*h[i-k]·[valid_i'] - δ(1-h[j+k])·[valid_j] + δ*h[j-k]·[valid_j']

Where valid means the index is in range.

The key question is: does this change reduce the max? It reduces the max if:
- For the shift(s) at the maximum: ΔI(k*) < 0
- For all other shifts: I(k) + ΔI(k) ≤ C5 (the current max)

This gives us a way to quickly check if a swap is beneficial without recomputing the full correlation!

    """Compute the change in all I(k) when moving delta from j to i"""
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    dc = np.zeros(n_shifts)
    # Effect of h[i] increasing by delta
    for k_shift in range(-n+1, n):
        # h[i+k] term: (1-h[i+k]) if 0<=i+k<n
        if 0 <= i + k_shift < n:
            dc[k_shift + n - 1] += delta * (1.0 - h[i + k_shift])
        # h[i-k] term: -h[i-k] if 0<=i-k<n
        if 0 <= i - k_shift < n:
            dc[k_shift + n - 1] -= delta * h[i - k_shift]
    # Effect of h[j] decreasing by delta
    for k_shift in range(-n+1, n):
        if 0 <= j + k_shift < n:
            dc[k_shift + n - 1] -= delta * (1.0 - h[j + k_shift])
        if 0 <= j - k_shift < n:
            dc[k_shift + n - 1] += delta * h[j - k_shift]
    return dc * dx

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-15:
            continue
        k = mi - (n - 1)
        wmi = w[mi] * dx
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            grad[lo1:hi1] += wmi * (1.0 - x[lo1 + k:hi1 + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            grad[lo2:hi2] += wmi * (-x[lo2 - k:hi2 - k])
    return grad

def lbfgs_run(h, n, alpha, deadline, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                       args=(n, alpha), method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-16, 'maxfun': maxiter * 50,
                                'maxls': 60})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def gradient_pair_search(h, n, deadline, alpha=10000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    W = 40
    for ds in step_sizes:
        any_improved = True
        while any_improved and time.time() < deadline:
            any_improved = False
            g = smooth_max_grad(h, n, alpha)
            order = np.argsort(g)
            for idx_i in range(min(W, n)):
                if time.time() > deadline:
                    return best_h, best_c
                i = order[idx_i]
                if g[i] > -1e-10:
                    break
                for idx_j in range(n - 1, max(-1, n - 1 - W), -1):
                    j = order[idx_j]
                    if g[j] < 1e-10:
                        break
                    if i == j:
                        continue
                    if h[i] + ds <= 1.0 and h[j] - ds >= 0.0:
                        h[i] += ds
                        h[j] -= ds
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            any_improved = True
                        else:
                            h[i] -= ds
                            h[j] += ds
    return best_h, best_c

def shift_equalize(h, n, deadline):
    """Try to equalize the top shifts by targeted moves"""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    top_k = 8
    for round_num in range(50):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        top_idx = np.argsort(c)[-top_k:]
        c_max = c.max()
        improved = False
        for mi in top_idx:
            k = mi - (n - 1)
            if k >= 0:
                for i in range(0, max(1, n - k), max(1, (n - k) // 80)):
                    if time.time() > deadline:
                        return best_h, best_c
                    j = i + k
                    d = 0.001
                    if h[i] > 0.3 and h[j] < 0.7:
                        h[i] -= d
                        h[j] += d
                        c2 = compute_c5(h, n)
                        if c2 < best_c - 1e-15:
                            best_c = c2
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                    if h[i] < 0.7 and h[j] > 0.3:
                        h[i] += d
                        h[j] -= d
                        c2 = compute_c5(h, n)
                        if c2 < best_c - 1e-15:
                            best_c = c2
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            else:
                kk = -k
                for i in range(kk, n, max(1, (n - kk) // 80)):
                    if time.time() > deadline:
                        return best_h, best_c
                    j = i - kk
                    d = 0.001
                    if h[i] > 0.3 and h[j] < 0.7:
                        h[i] -= d
                        h[j] += d
                        c2 = compute_c5(h, n)
                        if c2 < best_c - 1e-15:
                            best_c = c2
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
        if not improved:
            break
    return best_h, best_c

def pair_polish(h, n, deadline, radius=25):
    step_sizes = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for d in step_sizes:
        any_improved = True
        while any_improved and time.time() < deadline:
            any_improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            any_improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            any_improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def momentum_descent(h, n, deadline, alpha=5000, lr0=0.005, mom=0.9):
    v = np.zeros(n)
    lr = lr0
    stall_count = 0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    while time.time() < deadline:
        g = smooth_max_grad(h, n, alpha)
        v = mom * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-16:
            best_c = c
            best_h = h_new.copy()
            h = h_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 10:
                lr *= 0.7
                v *= 0.3
                if lr < 1e-7:
                    break
            h = best_h.copy()
    return best_h, best_c

def make_fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 9):
        amp = rng.uniform(0.08, 0.25)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)

def make_binary_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

    deadline = t0 + budget_s - 5

    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: Progressive L-BFGS-B with increasing alpha
    alphas = [30, 100, 400, 1500, 6000, 25000, 100000, 400000, 1500000]
    for alpha in alphas:
        if time.time() > deadline - 150:
            break
        remaining = deadline - time.time() - 80
        alloc = min(100, remaining * 0.25)
        if alloc < 12:
            break
        h, c = lbfgs_run(h, n, alpha, time.time() + alloc,
                         maxiter=min(5000, int(alloc * 30)))
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Momentum descent for escaping shallow local minima
    if time.time() < deadline - 120:
        h_m, c_m = momentum_descent(best_h, n, time.time() + 80, alpha=8000)
        if c_m < best_c:
            best_c = c_m
            best_h = h_m.copy()

    # Phase 3: Gradient-directed pair search
    if time.time() < deadline - 80:
        h_g, c_g = gradient_pair_search(best_h, n, min(time.time() + 100, deadline - 30))
        if c_g < best_c:
            best_c = c_g
            best_h = h_g.copy()

    # Phase 4: Shift equalization
    if time.time() < deadline - 40:
        h_se, c_se = shift_equalize(best_h, n, min(time.time() + 50, deadline - 15))
        if c_se < best_c:
            best_c = c_se
            best_h = h_se.copy()

    # Phase 5: Perturbed restarts
    for attempt in range(3):
        if time.time() > deadline - 50:
            break
        scale = 0.04 * (attempt + 1) * 0.5
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, n)
        alloc = min(60, (deadline - time.time() - 20) * 0.5)
        if alloc < 15:
            break
        h_r, c_r = lbfgs_run(h_pert, n, 20000, time.time() + alloc, maxiter=2000)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
            # Quick polish after improvement
            if time.time() < deadline - 20:
                h_gp, c_gp = gradient_pair_search(best_h, n,
                                                  min(time.time() + 25, deadline - 5),
                                                  alpha=20000)
                if c_gp < best_c:
                    best_c = c_gp
                    best_h = h_gp.copy()

    # Phase 6: Try different starting structures
    starts = [make_fourier_start(n, rng), make_binary_start(n, rng)]
    for s in starts:
        if time.time() > deadline - 40:
            break
        alloc = min(40, (deadline - time.time() - 15) * 0.5)
        if alloc < 12:
            break
        h_s, c_s = lbfgs_run(s, n, 5000, time.time() + alloc, maxiter=1500)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s.copy()

    # Phase 7: Final pair polish
    if time.time() < deadline - 3:
        best_h, best_c = pair_polish(best_h, n, deadline - 1, radius=20)

    # Final verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
