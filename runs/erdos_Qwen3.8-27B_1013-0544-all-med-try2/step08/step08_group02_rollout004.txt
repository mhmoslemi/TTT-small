            x0 = np.linspace(0.0, 2.0, len(parent), endpoint=False)
            x1 = np.linspace(0.0, 2.0, n, endpoint=False)
            a = np.interp(x1, x0, parent)
    elif v == 2:
        p = int(rng.randint(max(4, n // 8), max(5, n // 4) + 1))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a = 0.5 + 0.5 * np.sin(2.0 * np.pi * x / p + ph)
    else:
        p = int(rng.randint(max(4, n // 12), max(5, n // 3) + 1))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a = 0.5 + 0.5 * np.sin(np.pi * x / float(n)) * np.sign(np.sin(2.0 * np.pi * x / p + ph))
    return to_bin(a, n, rng)

def ap(p, d, c, h):
    n = len(h)
    ce = n - 1
    r = n - 1 - p
    if r:
        hp = h[p + 1:]
        c[ce + 1:ce + r + 1] -= d * hp
        c[ce - r:ce][::-1] += d * (1.0 - hp)
    l = p
    if l:
        hl = h[p - 1::-1]
        c[ce + 1:ce + l + 1] += d * (1.0 - hl)
        c[ce - l:ce][::-1] -= d * hl

def sw(p, q, c, h):
    hp = h[p]
    hq = h[q]
    if hp == hq:
        return
    ap(p, hq - hp, c, h)
    ap(q, hp - hq, c, h)
    ce = len(h) - 1
    ell = abs(p - q)
    c[ce + ell] += 1.0
    c[ce - ell] += 1.0
    h[p], h[q] = hq, hp

def binsearch(h, n, dl, rng, tgt):
    h = to_bin(h, n, rng)
    c = np.correlate(h, 1.0 - h, mode="full")
    ce = n - 1
    cur = float(c.max())
    bs = cur
    bh = h.copy()
    t = 2.0
    ops = 0
    pa = 0
    while time.time() < dl:
        if tgt is not None and bs <= tgt + 1e-9:
            break
        if ops > 1000:
            c = np.correlate(h, 1.0 - h, mode="full")
            cur = float(c.max())
            ops = 0
            if cur < bs - 1e-12:
                bs = cur
                bh = h.copy()
            elif cur > bs + max(4.0, 0.02 * n):
                h = bh.copy()
                c = np.correlate(h, 1.0 - h, mode="full")
                cur = float(bs)
        pa += 1
        if pa % 10 == 0:
            t = max(0.05, 0.82 * t)
        ti = np.argpartition(c, -10)[-10:] if c.size > 10 else np.arange(c.size)
        ti = ti[np.abs(ti - ce) > 0]
        if ti.size:
            ti = ti[rng.permutation(ti.size)]
        for idx in ti:
            s = int(abs(idx - ce))
            if s == 0:
                continue
            k = int(idx - ce)
            if k > 0:
                m = (h[:n - s] < 0.5) & (h[s:] > 0.5)
            else:
                m = (h[:n - s] > 0.5) & (h[s:] < 0.5)
            ca = np.flatnonzero(m)
            if ca.size:
                if ca.size > 6:
                    ca = ca[rng.permutation(ca.size)][:6]
                for i in ca:
                    if time.time() > dl:
                        break
                    j = int(i + s)
                    old = cur
                    sw(i, j, c, h)
                    ops += 1
                    nm = float(c.max())
                    acc = nm < old - 1e-12
                    if not acc and nm <= old + 4.0 * t:
                        acc = rng.random() < np.exp((old - nm) / max(t, 1e-8))
                    if acc:
                        if nm < bs - 1e-12:
                            bs = nm
                            bh = h.copy()
                        cur = nm
                    else:
                        sw(i, j, c, h)
                        ops += 1
                        cur = old
                    if tgt is not None and bs <= tgt + 1e-9:
                        return bh, float(bs) * (2.0 / n)
        nr = 70 + int(70 * t)
        for _ in range(nr):
            if time.time() > dl:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if abs(h[i] - h[j]) < 0.5:
                continue
            old = cur
            sw(i, j, c, h)
            ops += 1
            nm = float(c.max())
            acc = nm < old - 1e-12
            if not acc and nm <= old + 4.0 * t:
                acc = rng.random() < np.exp((old - nm) / max(t, 1e-8))
            if acc:
                if nm < bs - 1e-12:
                    bs = nm
                    bh = h.copy()
                cur = nm
            else:
                sw(i, j, c, h)
                ops += 1
                cur = old
            if tgt is not None and bs <= tgt + 1e-9:
                return bh, float(bs) * (2.0 / n)
    c = np.correlate(h, 1.0 - h, mode="full")
    cur = float(c.max())
    if cur < bs - 1e-12:
        bs = cur
        bh = h.copy()
    return bh, float(bs) * (2.0 / n)

def lgrad(h, k, n):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        nv = n - k
        g[:nv] = -h[k:]
        g[k:] += 1.0 - h[:nv]
    else:
        s = -k
        nv = n - s
        g[:nv] = 1.0 - h[s:]
        g[s:] += -h[:nv]
    return g

def lppol(h, n, dl, A=8, steps=150):
    h = proj(h, n).copy()
    bh = h.copy()
    bc = f5(h, n)
    ce = n - 1
    cv = np.zeros(n + 1)
    cv[-1] = 1.0
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    bad = 0
    for _ in range(steps):
        if time.time() > dl:
            break
        cc = np.correlate(h, 1.0 - h, mode="full")
        m = float(cc.max())
        if m < bc - 1e-12:
            bc = m
            bh = h.copy()
        ac = np.argpartition(cc, -A)[-A:] if cc.size > A else np.arange(cc.size)
        if ac.size == 0:
            break
        rows = np.empty((ac.size, n + 1))
        bu = np.empty(ac.size)
        for r, idx in enumerate(ac):
            k = int(idx - ce)
            rows[r, :n] = lgrad(h, k, n)
            rows[r, n] = -1.0
            bu[r] = -cc[idx]
        bnds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(None, None)]
        try:
            res = linprog(cv, rows, bu, Aeq, 0.0, bounds=bnds)
        except Exception:
            break
        if res is None or getattr(res, "x", None) is None:
            break
        d = res.x[:n]
        if not np.all(np.isfinite(d)) or float(np.max(np.abs(d))) < 1e-8:
            break
        lams = np.linspace(0.0, 1.0, 17)
        bv = bc
        bl = 0.0
        for L in lams:
            hh = h + L * d
            np.clip(hh, 0.0, 1.0, out=hh)
            vv = f5(hh, n)
            if vv < bv - 1e-12:
                bv = vv
                bl = float(L)
        if bl > 0.0 and bv < bc - 1e-12:
            hn = proj(h + bl * d, n)
            cvv = f5(hn, n)
            if cvv < bc - 1e-12:
                bc = cvv
                bh = hn.copy()
                h = hn.copy()
                bad = 0
            else:
                bad += 1
        else:
            bad += 1
        if bad > 5:
            break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 8.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.asarray(initial_h_values, float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 500
        h0 = np.zeros(n0)
        h0[rng.choice(n0, n0 // 2, replace=False)] = 1.0
    if n0 < 4:
        n0 = 250
        h0 = np.zeros(n0)
        h0[rng.choice(n0, n0 // 2, replace=False)] = 1.0
    if n0 % 2:
        n0 -= 1
        h0 = h0[:n0]
    h0 = proj(h0, n0)
    best_h = h0.copy()
    best_c = f5(best_h, n0)
    best_n = n0
    if budget_s < 90:
        best_h = proj(best_h, best_n)
        return best_h, f5(best_h, best_n), best_n
    reserve = min(260, max(80, int(0.25 * budget_s)))
    hdl = dl - reserve
    if time.time() < hdl - 35:
        hh, cc = lppol(best_h, best_n, min(time.time() + 45.0, hdl - 35.0), 6, 90)
        if cc < best_c:
            best_h = hh
            best_c = cc
    want = [best_n, 250, 400, 300, 200, 350, 450]
    tn = []
    for x in want:
        x = int(x)
        if x >= 40 and x <= 800 and x % 2 == 0 and x not in tn:
            tn.append(x)
    if not tn:
        tn = [500]
    vcount = 0
    while time.time() < hdl - 16:
        nt = tn[0]
        variant = vcount % 4
        vcount += 1
        rem = hdl - time.time()
        alloc = min(55.0, max(18.0, 0.22 * rem))
        hs = make_start(nt, best_h, rng, variant)
        hb, su = binsearch(hs, nt, time.time() + alloc, rng, 0.3808 * nt / 2.0)
        c5 = su * (2.0 / nt)
        imp = False
        if c5 < best_c:
            best_h = hb.copy()
            best_c = c5
            best_n = nt
            imp = True
        if c5 < 0.386 and time.time() + 22.0 < hdl:
            hp, cp = lppol(hb, nt, min(time.time() + 24.0, hdl - 6.0), 6, 80)
            if cp < best_c:
                best_h = hp.copy()
                best_c = cp
                best_n = nt
                imp = True
        tn.pop(0)
        if imp:
            tn = [nt] + [x for x in tn if x != nt]
        else:
            tn.append(nt)
    best_h = proj(best_h, best_n)
    best_c = f5(best_h, best_n)
    if time.time() < dl - 48:
        hh, cc = lppol(best_h, best_n, min(time.time() + 75.0, dl - 48.0), 8, 220)
        if cc < best_c:
            best_h = hh.copy()
            best_c = cc
    if time.time() < dl - 26:
        hh, cc = lppol(best_h, best_n, min(time.time() + 40.0, dl - 26.0), 10, 180)
        if cc < best_c:
            best_h = hh.copy()
            best_c = cc
    if time.time() < dl - 3:
        hh, cc = lppol(best_h, best_n, dl - 2.0, 9, 260)
        if cc < best_c:
            best_h = hh.copy()
            best_c = cc
    best_h = proj(best_h, best_n)
    best_c = f5(best_h, best_n)
    return best_h, best_c, best_n
