**Subgradient descent on active constraints:**

At each step:
1. Compute C = c5_all(h, n)
2. Find k* = argmax C
3. Compute grad_k* = gradient of C_{k*}
4. Take step: h_new = h - η * grad_k*
5. Project to feasible set
6. If C₅ improved, accept; otherwise reduce η

The advantage is that we only compute ONE gradient per step (O(n) work) instead of all of them (O(n²)). But we might need many more steps.

Actually, for the subgradient method, we can be smarter. Instead of just the max, use a "top-k" weighted average:

g = (1/K) * sum_{k in top_K} w_k * grad_k

where w_k = exp(C_k / T) / sum_j exp(C_j / T) (softmax over top-K).

This is the same as what log-sum-exp does, but we can make it more "peaked" by using a larger T.

from scipy.optimize import minimize
import time

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1-h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def c5_grad_one(h, n, k):
    """Gradient of C_k (k can be negative)."""
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[:n-k] += (1.0 - h[k:])
            if k > 0:
                g[k:] -= h[:n-k]
    else:
        kk = -k
        if kk < n:
            g[kk:] += (1.0 - h[:n-kk])
            if kk > 0:
                g[:n-kk] -= h[kk:]
    return g * dx

def c5_grad_smax(h, n, alpha):
    c = c5_all(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    g = np.zeros(n)
    nw = len(w)

        k = idx - (n - 1)
        g += w[idx] * c5_grad_one(h, n, k)

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

    g = c5_grad_smax(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n/2.0)
    return g

def lbfgs_run(h, n, deadline, max_iters=300):

    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

               (100000, 10.0), (500000, 50.0), (2000000, 100.0),
               (10000000, 500.0), (50000000, 2000.0)]
    for alpha, mu in configs:

        try:

                         args=(n, alpha, mu), method='L-BFGS-B',

                                  'gtol': 1e-12, 'maxfun': max_iters * 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:

                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def subgrad_run(h, n, deadline, top_k=5, max_iter=3000):
    """Subgradient descent on top-k active constraints."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for step in steps:
        Improved = True
        while Improved and time.time() < deadline:
            Improved = False
            c = c5_all(h, n)
            m = c.max()
            # Get top-k active constraints
            top_idx = np.argsort(c)[-top_k:]
            g = np.zeros(n)
            for idx in top_idx:
                k = idx - (n - 1)
                g += c5_grad_one(h, n, k)

                    best_c = c_new
                    best_h = h_new.copy()
                    Improved = True
                    break
    return best_h, best_c

def binary_hill_climb(h, n, deadline, max_passes=10):
    """Project to binary and hill-climb."""
    # Round to binary, maintaining sum = n/2
    h_bin = (h >= 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    target = n // 2
    ones = int(h_bin.sum())
    if ones > target:
        # Need to flip some 1s to 0s: pick those with highest h contribution
        one_idx = np.where(h_bin == 1)[0]
        # Sort by original h value (lowest first - least "committed" to being 1)
        order = np.argsort(h[one_idx])
        for idx in order:
            if ones == target:
                break
            h_bin[one_idx[idx]] = 0
            ones -= 1
    elif ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])  # highest h first
        for idx in order:
            if ones == target:
                break
            h_bin[zero_idx[idx]] = 1
            ones += 1
    
    best_c = compute_c5(h_bin, n)
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        one_idx = np.where(h_bin == 1)[0]
        zero_idx = np.where(h_bin == 0)[0]
        # Try flipping: move a 1 to a 0 position
        for j in zero_idx:
            if time.time() > deadline:
                break
            for i in one_idx:
                h_bin[i] = 0
                h_bin[j] = 1
                c_new = compute_c5(h_bin, n)
                if c_new < best_c - 1e-14:

                    improved = True
                    one_idx = np.where(h_bin == 1)[0]
                else:
                    h_bin[i] = 1
                    h_bin[j] = 0
            if not improved:
                break
        if not improved:
            break
    
    # Blend binary solution with continuous
    if best_c < compute_c5(h, n):
        return h_bin, best_c
    return h, compute_c5(h, n)
