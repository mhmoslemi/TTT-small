```
def project_box_sum(v, n):
    v = np.asarray(v, dtype=float).copy()
    target = 0.5 * n
    v = np.clip(v, 0.0, 1.0)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        x = v - mid
        np.clip(x, 0.0, 1.0, out=x)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5*(lo+hi)
    x = v - lam
    np.clip(x,0.0,1.0,out=x)
    diff = target - x.sum()
    if abs(diff) > 0:
       if diff > 0:
           idx = np.where(x < 1.0)[0]
           for i in idx:
              add = min(diff, 1.0-x[i]); x[i]+=add; diff-=add
              if abs(diff)<1e-14: break
       else:
           idx = np.where(x > 0.0)[0]
           for i in idx:
              sub = min(-diff, x[i]); x[i]-=sub; diff+=sub
              if abs(diff)<1e-14: break
    return x
```

```
def eq_sum(x, n):
    return x.sum() - 0.5*n

def softmax_weights(c, alpha):
    m = float(np.max(c))
    z = alpha * (c - m)
    z = np.maximum(z, -700.0)
    e = np.exp(z)
    s = e.sum()
    if s <= 0.0 or not np.isfinite(s):
        e = np.ones_like(c)
        s = float(len(c))
    return e / s, m + float(np.log(s)) / alpha

def gradient_weights(h, w, n):
    off = n - 1
    comp = 1.0 - h
    g1 = np.convolve(w, comp, mode='full')
    g2 = np.convolve(w[::-1], h, mode='full')
    return (2.0 / n) * (g1[off:off+n] - g2[off:off+n])

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    return float(softmax_weights(c, alpha)[1])

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    w, _ = softmax_weights(c, alpha)
    return gradient_weights(x, w, n)

def slsqp_smooth(h, n, alpha, maxiter):
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': eq_sum, 'args': (n,)}]

        return project_box_sum(res.x, n)
    except Exception:
        return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_box_sum(v, n):
    v = np.asarray(v, dtype=float).copy()
    target = 0.5 * n
    v = np.clip(v, 0.0, 1.0)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        x = v - mid
        np.clip(x, 0.0, 1.0, out=x)
        s = float(x.sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = v - lam
    np.clip(x, 0.0, 1.0, out=x)
    diff = target - float(x.sum())
    if diff > 1e-14:
        idx = np.where(x < 1.0)[0]
        for i in idx:
            add = min(diff, 1.0 - x[i])
            x[i] += add
            diff -= add
            if abs(diff) < 1e-14:
                break
    elif diff < -1e-14:
        idx = np.where(x > 0.0)[0]
        for i in idx:
            sub = min(-diff, x[i])
            x[i] -= sub
            diff += sub
            if abs(diff) < 1e-14:
                break
    return x


def to_binary(h, n):
    hb = np.zeros(n, dtype=float)
    target = n // 2
    idx = np.argsort(h, kind="stable")[::-1]
    hb[idx[:target]] = 1.0
    return hb


def swap_pre_ok(a, b, h, D, top_pairs, n, M):
    limit = M - 1.0 + 1e-9
    ab = a - b
    ba = b - a
    for idx, s in top_pairs:
        val = 0.0
        y = a - s
        if 0 <= y < n:
            val += 1.0 - h[y]
        y = b - s
        if 0 <= y < n:
            val -= 1.0 - h[y]
        y = a + s
        if 0 <= y < n:
            val -= h[y]
        y = b + s
        if 0 <= y < n:
            val += h[y]
        if s == 0:
            val -= 2.0
        elif s == ab:
            val += 1.0
        elif s == ba:
            val += 1.0
        if D[idx] + val > limit:
            return False
    return True


def fill_swap_delta(delta, h, om, a, b, n, off):
    delta.fill(0.0)
    omr = om[::-1]
    delta[a:a + n] += omr
    delta[b:b + n] -= omr
    delta[off - a:off - a + n] -= h
    delta[off - b:off - b + n] += h
    delta[off] -= 2.0
    delta[off + a - b] += 1.0
    delta[off + b - a] += 1.0

            a = zeros[zi]
            b = ones[oi]

                if curM < bestM:
                    bestM = curM
                    best_h = h.copy()
            T *= 0.99995
            if T < 1e-4:
                T = 1e-4

    return best_h, compute_c5(best_h, n)


def softmax_from_c(c, alpha):
    m = float(np.max(c))
    z = alpha * (c - m)
    z = np.maximum(z, -700.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        e = np.ones_like(c)
        s = float(len(c))
    return e / s, m + float(np.log(s)) / alpha


def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    m = float(np.max(c))
    z = alpha * (c - m)
    z = np.maximum(z, -700.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        return m
    return m + float(np.log(s)) / alpha


def gradient_weights(h, w, n):
    off = n - 1
    comp = 1.0 - h
    g1 = np.convolve(w, comp, mode="full")
    g2 = np.convolve(w[::-1], h, mode="full")
    return (2.0 / n) * (g1[off:off + n] - g2[off:off + n])


def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    w, _ = softmax_from_c(c, alpha)
    return gradient_weights(x, w, n)


def eq_sum(x, n):
    return float(x.sum()) - 0.5 * n


def eq_sum_jac(x, n):
    return np.ones_like(x)


def slsqp_smooth(h, n, alpha, maxiter):
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': eq_sum, 'jac': eq_sum_jac, 'args': (n,)}]
        res = minimize(
            smooth_obj, h, method='SLSQP', jac=smooth_jac,
            args=(n, alpha), bounds=bounds, constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-12}
        )
        return project_box_sum(res.x, n)
    except Exception:
        return h


def weighted_obj(x, n, w, mu):
    D = np.correlate(x, 1.0 - x, mode="full")
    val = (2.0 / n) * float(np.dot(w, D))
    s = float(x.sum()) - 0.5 * n
    return val + 0.5 * mu * s * s


def weighted_jac(x, n, w, mu):
    g = gradient_weights(x, w, n)
    s = float(x.sum()) - 0.5 * n
    return g + mu * s


def weighted_lbfgs(h, n, tau, mu, maxiter):
    try:
        c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        w, _ = softmax_from_c(c, 1.0 / max(tau, 1e-6))
        bounds = [(0.0, 1.0)] * n
        res = minimize(
            weighted_obj, h, method='L-BFGS-B', jac=weighted_jac,
            args=(n, w, mu), bounds=bounds,
            options={'maxiter': int(maxiter), 'ftol': 1e-12, 'gtol': 1e-10}
        )
        return project_box_sum(res.x, n)
    except Exception:
        return h


def weighted_obj_exact(x, n, w):
    D = np.correlate(x, 1.0 - x, mode="full")
    return (2.0 / n) * float(np.dot(w, D))


def weighted_jac_exact(x, n, w):
    return gradient_weights(x, w, n)


def slsqp_weighted(h, n, w, maxiter):
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': eq_sum, 'jac': eq_sum_jac, 'args': (n,)}]
        res = minimize(
            weighted_obj_exact, h, method='SLSQP', jac=weighted_jac_exact,
            args=(n, w), bounds=bounds, constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-12}
        )
        return project_box_sum(res.x, n)
    except Exception:
        return h


def interp_to_n(h, n_old, n_new):
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    T = max(60.0, float(budget_s) - 20.0)
    dl = t0 + T
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
    except Exception:
        best_n = 400
        rr = np.random.RandomState(12345)
        best_h = np.zeros(best_n, dtype=float)
        best_h[rr.choice(best_n, best_n // 2, replace=False)] = 1.0

    best_h = project_box_sum(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    if best_n % 2 == 0 and time.time() < dl - 30.0:
        t_bin = t0 + 0.25 * T
        hb, cb = binary_search(best_h, best_n, t_bin, seed)
        if cb < best_c:
            best_c = cb
            best_h = hb

    t_wei = t0 + 0.60 * T
    taus = [0.05, 0.02, 0.01, 0.005, 0.002]
    for tau in taus:
        if time.time() > t_wei:
            break
        reps = 3 if tau <= 0.01 else 2
        for _ in range(reps):
            if time.time() > t_wei - 25.0:
                break
            maxit = 250 if tau > 0.005 else 170
            x = weighted_lbfgs(best_h, best_n, tau, 20.0, maxit)
            c = compute_c5(x, best_n)
            if c < best_c:
                best_c = c
                best_h = x

    t_slq = t0 + 0.80 * T
    alphas = [5000, 50000, 500000]
    for alpha in alphas:
        if time.time() > t_slq:
            break
        maxit = 180 if alpha < 100000 else 110
        x = slsqp_smooth(best_h, best_n, alpha, maxit)
        c = compute_c5(x, best_n)
        if c < best_c:
            best_c = c
            best_h = x
        if time.time() < t_slq - 25.0 and alpha == alphas[-1]:
            cv = np.correlate(best_h, 1.0 - best_h, mode="full") * (2.0 / best_n)
            w, _ = softmax_from_c(cv, 20000.0)
            x2 = slsqp_weighted(best_h, best_n, w, 120)
            c2 = compute_c5(x2, best_n)
            if c2 < best_c:
                best_c = c2
                best_h = x2

    t_new = t0 + 0.90 * T
    starts = []
    if best_n > 1:
        starts.append(project_box_sum(best_h[::-1].copy(), best_n))
        starts.append(project_box_sum(1.0 - best_h, best_n))
    starts.append(project_box_sum(best_h + 0.02 * rng.randn(best_n), best_n))

    for st in starts:
        if time.time() > t_new:
            break
        x = weighted_lbfgs(st, best_n, 0.01, 20.0, 140)
        c = compute_c5(x, best_n)
        if c < best_c:
            best_c = c
            best_h = x
            if time.time() < t_new - 10.0:
                x2 = slsqp_smooth(best_h, best_n, 50000, 110)
                c2 = compute_c5(x2, best_n)
                if c2 < best_c:
                    best_c = c2
                    best_h = x2

    for nh in (480, 512):
        if time.time() > dl - 60.0:
            break
        hi = project_box_sum(interp_to_n(best_h, best_n, nh), nh)
        x = weighted_lbfgs(hi, nh, 0.01, 20.0, 130)
        c = compute_c5(x, nh)
        if c < best_c:
            best_c = c
            best_h = x
            best_n = nh
            if time.time() < dl - 25.0 and nh <= 512:
                x2 = slsqp_smooth(best_h, best_n, 20000, 70)
                c2 = compute_c5(x2, best_n)
                if c2 < best_c:
                    best_c = c2
                    best_h = x2

    final_dl = dl - 2.0
    amps = [0.01, 0.005, 0.002]
    k = 0
    while time.time() < final_dl:
        amp = amps[k % 3]
        k += 1
        hp = project_box_sum(best_h + amp * rng.randn(best_n), best_n)
        if best_n <= 440:
            maxit = 80 if time.time() > final_dl - 30.0 else 50
            x = slsqp_smooth(hp, best_n, 100000, maxit)
        else:
            x = weighted_lbfgs(hp, best_n, 0.005, 30.0, 70)
        c = compute_c5(x, best_n)
        if c < best_c:
            best_c = c
            best_h = x

    best_h = project_box_sum(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
```
