def solve_minmax_lp(h0, n, t0, active_set, max_iters=50):
    """Solve LP: min t s.t. g_k(h) <= t for k in active_set, 0<=h<=1, sum(h)=n/2"""
    import cvxpy as cp
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    # For each active shift k, add constraint g_k(h) <= t
    # g_k(h) = sum_i h[i]*(1-h[i+k]) for k>=0, or sum_i h[i+k]*(1-h[i]) for k<0
    # Linearize at h0: g_k(h) ≈ g_k(h0) + grad_k(h0)·(h-h0)
    # = g_k(h0) + sum_i grad_k[i]*(h[i]-h0[i])
    # We need this <= t
    
    for k_idx in active_set:
        k = k_idx - (n - 1)
        # Compute g_k(h0) and grad_k(h0)
        # For k >= 0: g_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # grad[i] += (1-h[i+k]) for i in [0, n-1-k)
        # grad[i+k] += (-h[i]) for i in [0, n-1-k)
        g_val = 0.0
        grad = np.zeros(n)
        if k >= 0:
            nv = n - k
            for i in range(nv):
                g_val += h0[i] * (1 - h0[i+k])
                grad[i] += (1 - h0[i+k])
                grad[i+k] += (-h0[i])
        else:
            kk = -k
            nv = n - kk
            for i in range(nv):
                g_val += h0[i+kk] * (1 - h0[i])
                grad[i+kk] += (1 - h0[i])
                grad[i] += (-h0[i+kk])
        
        # g_k(h) ≈ g_val + grad·(h-h0) <= t
        # g_val - grad·h0 + grad·h <= t
        const = g_val - np.dot(grad, h0)
        constraints.append(cp.dot(grad, h_var) + const <= t_var)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS)  # or GLOP, CLARABEL
        if h_var.value is not None and t_var.value is not None:
            return np.array(h_var.value), float(t_var.value)
    except:
        pass
    return h0, t0

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            room[h >= 1.0] = 0.0
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += (diff / tot) * h[mask]
    return np.clip(h, 0.0, 1.0)

def corr_vec(h, n):

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    return m + (smax + np.log(np.sum(np.exp(sh - smax)))) / alpha

    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    return w

            nv = n - kk

                                     alphas=(5000000, 20000000), mu=50000.0)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r

    # Phase 6: Try different n (100s)
    for n_try in [400, 600, 750]:
        if time.time() > deadline - 50:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n_try)
        alloc = min(30, (deadline - time.time() - 40) * 0.6)
        if alloc < 10:
            continue
        h_t, c_t = do_lbfgs(h_int, n_try, time.time() + alloc,
                             alphas=(10000, 100000, 1000000), mu=10000.0)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Phase 7: Final high-alpha polish (remaining)
    if time.time() < deadline - 8:
        h_f, c_f = do_lbfgs(best_h, best_n, deadline - 3,
                             alphas=(1000000, 5000000, 20000000), mu=50000.0)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
