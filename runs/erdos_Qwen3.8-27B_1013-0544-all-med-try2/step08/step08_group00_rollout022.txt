def opt_pair_step(h, n, d_max=0.01):
    """Find the best single pair move and apply it."""
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    best_c = c.max()
    
    # For each pair (i, j), compute the optimal d
    # This is O(n^2 * n) = O(n^3) which is too slow for n=650
    # Instead, use gradient-based selection
    
    # Compute gradient of smooth max
    alpha = 10000
    c_shift = alpha * (c - best_c)
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w = w / w.sum()
    
    # Gradient: grad[j] = dx * sum_m w[m] * dc[m]/dh[j]
    # (same as before)
    grad = np.zeros(n)
    for m in range(2*n-1):
        wm = w[m] * dx
        if wm < 1e-14:
            continue
        k = m - (n-1)
        if k >= 0:
            nv = n - k
            grad[k:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[k:])
        else:
            kk = -k
            nv = n - kk
            grad[:nv] += wm * (1.0 - h[kk:])
            grad[kk:] += wm * (-h[:nv])
    
    # Top gradient (should increase) and bottom (should decrease)
    # We want to increase h where grad is most negative (reduces objective)
    # and decrease where grad is most positive
    # Wait: we're minimizing, so we want to move in -gradient direction
    # grad[i] > 0 means increasing h[i] increases objective -> decrease h[i]
    # grad[i] < 0 means increasing h[i] decreases objective -> increase h[i]
    
    # So: increase h where grad is most negative, decrease where most positive
    top_inc = np.argsort(grad)[:50]  # most negative gradient -> increase these
    top_dec = np.argsort(grad)[-50:]  # most positive gradient -> decrease these
    
    best_d = 0
    best_pair = None
    best_new_c = best_c
    
    for i in top_inc:
        for j in top_dec:
            if i == j:
                continue
            if grad[i] >= 0 or grad[j] <= 0:
                continue
            # Try to find optimal d: increase h[i] by d, decrease h[j] by d
            # d must satisfy: 0 <= d <= min(h[j], 1-h[i])
            d_max_ij = min(h[j], 1-h[i])
            if d_max_ij <= 0:
                continue
            # Binary search or grid for optimal d
            # For each d, new_c = max_k (c_k + d * delta_k)
            # where delta_k = dc_k/dh[i] - dc_k/dh[j]
            # We approximate with the gradient: delta ≈ grad[i] - grad[j] < 0
            # So all correlations decrease, and the best d is the largest possible
            # But that's only true if we're in the "smooth" regime
            # Let's just try a few values
            for d in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
                if d > d_max_ij:
                    continue
                h[i] += d
                h[j] -= d
                new_c = np.max(np.correlate(h, 1-h, mode='full') * dx)
                if new_c < best_new_c:
                    best_new_c = new_c
                    best_d = d
                    best_pair = (i, j)
                h[i] -= d
                h[j] += d
    
    if best_pair is not None:
        i, j = best_pair
        h[i] += best_d
        h[j] -= best_d
        return h, best_new_c
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_c5_vec(h, n)
    cm = c.max()
    sh = alpha * (c - cm)
    mx = sh.max()
    w = np.exp(sh - mx)
    w = w / w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-16:
            continue
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[k:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[k:])
        else:
            kk = -k
            nv = n - kk
            grad[:nv] += wm * (1.0 - h[kk:])
            grad[kk:] += wm * (-h[:nv])
    return grad

def lag_grad(h, n, lag):
    """Gradient of correlation at a specific lag."""
    k = lag - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[k:] += (1.0 - h[:nv])
        grad[:nv] += -h[k:]
    else:
        kk = -k
        nv = n - kk
        grad[:nv] += (1.0 - h[kk:])
        grad[kk:] += -h[:nv]
    return grad

def lbfgs_optimize(h, n, deadline, max_iters=800):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(200, 1.0), (1000, 5.0), (5000, 20.0),
               (20000, 100.0), (100000, 500.0),
               (500000, 2000.0), (2000000, 10000.0),
               (10000000, 50000.0), (50000000, 200000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        def obj(x):
            return smooth_max_obj(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
        def grad(x):
            return smooth_max_grad(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0)
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-20,
                                   'gtol': 1e-18, 'maxfun': max_iters * 30,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def smart_pair_search(h, n, deadline, k_sel=30, alpha=50000):
    """Gradient-directed pair moves with line search."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    iters = 0
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        cm = c.max()
        if cm <= best_c - 1e-15:
            best_c = cm
        grad = smooth_max_grad(h, n, alpha)
        inc_idx = np.argsort(grad)[:k_sel]
        dec_idx = np.argsort(grad)[-k_sel:]
        improved = False
        for i in inc_idx:
            if grad[i] >= -1e-12:
                continue
            for j in dec_idx:
                if grad[j] <= 1e-12:
                    continue
                if i == j:
                    continue
                d_max = min(h[j], 1.0 - h[i])
                if d_max <= 0:
                    continue
                trial_steps = [s for s in steps if s <= d_max]
                if d_max > 0 and d_max < 0.01:
                    trial_steps.append(d_max)
                for d in trial_steps:
                    h[i] += d
                    h[j] -= d
                    new_c = compute_c5(h, n)
                    if new_c < best_c - 1e-15:
                        best_c = new_c
                        improved = True
                        continue
                    h[i] -= d
                    h[j] += d
                # restore if we improved (already moved)
                # if not improved, already restored
        iters += 1
        if not improved:
            # Try smaller alpha for more focused gradient
            alpha = max(1000, alpha // 3)
            k_sel = max(15, k_sel // 2)
            if k_sel < 15:
                break
    return h, best_c

def lag_targeted_moves(h, n, deadline, n_top_lags=10):
    """Target the top correlations specifically."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    while time.time() < deadline:
        c = compute_c5_vec(h, n)
        cm = c.max()
        if cm <= best_c - 1e-15:
            best_c = cm
        c_sorted = np.sort(c)[::-1]
        thresh = c_sorted[min(n_top_lags - 1, len(c_sorted) - 1)]
        active_lags = np.where(c >= thresh * 0.999)[0]
        improved = False
        for m in active_lags:
            if time.time() > deadline:
                break
            lg = lag_grad(h, n, m)
            # We want to reduce c[m], so move in -lg direction
            # But must preserve sum(h), so pair moves
            neg_top = np.argsort(lg)[:20]  # increasing these decreases c[m]
            pos_top = np.argsort(lg)[-20:]  # decreasing these decreases c[m]
            for i in neg_top:
                if lg[i] >= 0:
                    continue
                for j in pos_top:
                    if lg[j] <= 0:
                        continue
                    if i == j:
                        continue
                    d_max = min(h[j], 1.0 - h[i])
                    if d_max <= 0:
                        continue
                    for d in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
                        if d > d_max:
                            continue
                        h[i] += d
                        h[j] -= d
                        new_c = compute_c5(h, n)
                        if new_c < best_c - 1e-15:
                            best_c = new_c
                            improved = True
                            break
                        h[i] -= d
                        h[j] += d
        if not improved:
            break
    return h, best_c

def anneal_polish(h, n, deadline, T_init=0.005, T_min=1e-6):
    """Simulated annealing on pair moves to escape local optima."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(123)
    cur_c = best_c
    T = T_init
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    iters = 0
    while time.time() < deadline and T > T_min:
        iters += 1
        if iters % 500 == 0:
            T *= 0.9
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = steps[rng.randint(len(steps))]
        if h[i] + d > 1.0 or h[j] - d < 0.0:
            continue
        h[i] += d
        h[j] -= d
        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
        h = project_feasible(h, n)
    return best_h, best_c

def multi_restart(h, n, deadline, rng):
    """Try perturbed starts."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    n_starts = 4
    per_start = max(15, (deadline - time.time()) / n_starts - 5)
    for s in range(n_starts):
        if time.time() > deadline:
            break
        scale = 0.02 * (s + 1)
        h0 = h.copy()
        h0 += rng.randn(n) * scale
        h0 = project_feasible(h0, n)
        c0 = compute_c5(h0, n)
        if c0 >= best_c:
            c0 = best_c * 1.01  # still try optimizing from here
        t_end = min(time.time() + per_start, deadline - 10)
        h_s, c_s = lbfgs_optimize(h0, n, t_end, max_iters=200)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s
    return best_h, best_c

def slsqq_polish(h, n, deadline, alpha=2000000):
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    def grad(x):
        return smooth_max_grad(x, n, alpha)
    def eq(x):
        return [x.sum() - n / 2.0]
    def eq_jac(x):
        return [np.ones(n)]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': eq, 'jac': eq_jac}],
                      options={'maxiter': 5000, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)
