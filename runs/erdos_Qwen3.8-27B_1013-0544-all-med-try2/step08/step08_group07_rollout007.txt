```python
def compute_grad(h, n, w):
    """Compute gradient of weighted sum of correlation values.
    w is the weight vector of length 2n-1."""
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        # First term: (1-h[j-k+n-1]) for valid k
        # j-k+n-1 in [0, n-1] means k in [j-n+1, j]
        k_lo = max(0, j - n + 1)
        k_hi = min(2*n-2, j)
        if k_lo <= k_hi:
            idx = j - np.arange(k_lo, k_hi+1) + n - 1
            g += np.dot(w[k_lo:k_hi+1], 1 - h[idx])
        # Second term: -h[j+k-n+1] for valid k
        # j+k-n+1 in [0, n-1] means k in [n-1-j, n-1+j]
        k_lo2 = max(0, n - 1 - j)
        k_hi2 = min(2*n-2, n - 1 + j)
        if k_lo2 <= k_hi2:
            idx2 = j + np.arange(k_lo2, k_hi2+1) - n + 1
            g -= np.dot(w[k_lo2:k_hi2+1], h[idx2])
        grad[j] = g
    return grad
```

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def softmax_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    ev = np.exp(sh - sm)
    return ev / ev.sum()

def compute_grad(h, n, w):
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        klo = max(0, j - n + 1)
        khi = min(2 * n - 2, j)
        if klo <= khi:
            ks = np.arange(klo, khi + 1)
            idx = j - ks + n - 1
            g += np.dot(w[klo:khi + 1], 1.0 - h[idx])
        klo2 = max(0, n - 1 - j)
        khi2 = min(2 * n - 2, n - 1 + j)
        if klo2 <= khi2:
            ks2 = np.arange(klo2, khi2 + 1)
            idx2 = j + ks2 - n + 1
            g -= np.dot(w[klo2:khi2 + 1], h[idx2])
        grad[j] = g
    return grad

def proj_gradient(h, n, alpha, max_iter, deadline):
    h = h.astype(float).copy()
    lr = 0.5 / (alpha + 1.0)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        w = softmax_w(c, alpha)
        grad = compute_grad(h, n, w)
        h = project_feasible(h - lr * grad, n)
        if it % 200 == 199:
            lr *= 0.95
    return h

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def gfunc(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        w = softmax_w(g, alpha)
        return compute_grad(x, n, w) * dx
    cons_fun = lambda x: x.sum() - n / 2.0
    cons_jac = lambda x: np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=gfunc, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, n, max_sweeps, deadline):
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    m_range = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(40):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr
