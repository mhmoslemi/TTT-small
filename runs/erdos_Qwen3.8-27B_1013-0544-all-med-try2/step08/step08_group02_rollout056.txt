import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            tot = h[mask].sum()
            if tot < 1e-15:
                break
            h[mask] += (diff / tot) * h[mask]

def smooth_max_and_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w_sum = w.sum()
    w /= w_sum
    val = m + np.log(w_sum) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue

        if s >= 0:
            nv = n - s

    bounds = [(0.0, 1.0)] * n
    if alpha_schedule is None:
        alpha_schedule = [(500, 5.0), (2000, 20.0), (10000, 100.0),
                          (50000, 500.0), (200000, 2000.0),
                          (1000000, 10000.0), (5000000, 50000.0)]
    for alpha, mu in alpha_schedule:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(pen_obj, h, jac=pen_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_run(h, n, deadline, alpha=2000000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def snap_optimize(h, n, deadline, threshold):
    h_snap = h.copy()
    h_snap[h_snap >= threshold] = 1.0
    h_snap[h_snap <= 1.0 - threshold] = 0.0
    h_snap = project_feasible(h_snap, n)
    c_snap = compute_c5(h_snap, n)
    c_orig = compute_c5(h, n)
    if c_snap > c_orig * 1.002:
        return h, c_orig
    t_sub = min(time.time() + 35, deadline - 20)
    h_opt, c_opt = lbfgs_polish(h_snap, n, t_sub,
        alpha_schedule=[(50000, 1000.0), (200000, 5000.0), (1000000, 20000.0)],
        max_iters=300)
    if c_opt < c_orig:
        return h_opt, c_opt
    if c_snap < c_orig:
        return h_snap, c_snap
    return h, c_orig

def binary_greedy_improve(h, n, deadline, rng, max_rounds=50):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        idx = np.where(h_bin == 1)[0]
        rem = rng.choice(idx, size=cnt - target, replace=False)
        h_bin[rem] = 0
    elif cnt < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, size=target - cnt, replace=False)
        h_bin[add] = 1
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        best_move_c = best_c
        best_i, best_j = -1, -1
        n_try = min(len(ones_idx) * len(zeros_idx), 20000)
        for _ in range(n_try):
            i = ones_idx[rng.randint(len(ones_idx))]
            j = zeros_idx[rng.randint(len(zeros_idx))]
            h_bin[i] = 0
            h_bin[j] = 1
            c_new = compute_c5(h_bin, n)
            h_bin[i] = 1
            h_bin[j] = 0
            if c_new < best_move_c - 1e-15:
                best_move_c = c_new
                best_i, best_j = i, j
        if best_i >= 0:
            h_bin[best_i] = 0
            h_bin[best_j] = 1
            best_c = best_move_c
            best_h = h_bin.copy()
        else:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    best_n = n

    # Phase 1: Main L-BFGS-B with aggressive alpha schedule
    if time.time() < deadline - 250:
        t1 = min(time.time() + 280, deadline - 220)
        h1, c1 = lbfgs_polish(best_h, best_n, t1,
            alpha_schedule=[(200, 5.0), (1000, 20.0), (5000, 100.0),
                           (20000, 500.0), (100000, 2000.0),
                           (500000, 10000.0), (2000000, 50000.0),
                           (5000000, 100000.0)],
            max_iters=500)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Snapping with increasing threshold
    if time.time() < deadline - 180:
        for threshold in [0.8, 0.9, 0.95, 0.99]:
            if time.time() > deadline - 130:
                break
            h_s, c_s = snap_optimize(best_h, best_n, deadline - 120, threshold)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s

    # Phase 3: Binary greedy + continuous polish
    if time.time() < deadline - 120:
        t_bg = min(time.time() + 30, deadline - 90)
        h_b, c_b = binary_greedy_improve(best_h, best_n, t_bg, rng, max_rounds=30)
        if c_b < best_c * 1.005:
            h_p = np.maximum(h_b, 0.0)
            h_p = project_feasible(h_p, best_n)
            t_bp = min(time.time() + 40, deadline - 70)
            h_bp, c_bp = lbfgs_polish(h_p, best_n, t_bp,
                alpha_schedule=[(10000, 200.0), (50000, 1000.0),
                               (200000, 5000.0), (1000000, 20000.0)],
                max_iters=400)
            if c_bp < best_c:
                best_c = c_bp
                best_h = h_bp

    # Phase 4: Random perturbation restarts
    for _ in range(4):
        if time.time() > deadline - 80:
            break
        noise_scale = 0.03
        h_pert = best_h + rng.normal(0, noise_scale, size=best_n)
        h_pert = project_feasible(h_pert, best_n)
        t_p = min(time.time() + 30, deadline - 60)
        h_r, c_r = lbfgs_polish(h_pert, best_n, t_p,
            alpha_schedule=[(5000, 100.0), (50000, 1000.0),
                           (500000, 10000.0)],
            max_iters=300)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r

    # Phase 5: Try different n values
    for n_try in [400, 625, 750]:
        if time.time() > deadline - 50:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n_try)
        alloc = min(40, (deadline - time.time() - 30) * 0.6)
        if alloc < 15:
            continue
        h_t, c_t = lbfgs_polish(h_int, n_try, time.time() + alloc,
            alpha_schedule=[(1000, 20.0), (10000, 200.0),
                           (100000, 2000.0), (500000, 10000.0)],
            max_iters=300)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 5:
        h_f, c_f = slsqp_run(best_h, best_n, deadline - 3,
                             alpha=5000000, maxiter=10000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    # Phase 7: Final high-alpha L-BFGS-B
    if time.time() < deadline - 2:
        h_f, c_f = lbfgs_polish(best_h, best_n, deadline - 1,
            alpha_schedule=[(5000000, 50000.0)],
            max_iters=100)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
