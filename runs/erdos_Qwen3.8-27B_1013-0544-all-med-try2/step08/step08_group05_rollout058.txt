import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones"""
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def compute_D(h, n):
    """Compute D matrix where D[p, s] = change in corr[s] if h[p] goes 0->1
    s is indexed 0 to 2n-2, corresponding to shift s-(n-1)
    """
    # D[p, s_idx] where s_idx = shift + (n-1), shift ranges from -(n-1) to (n-1)
    # D[p, s] = (1-h[p-s]) - h[p+s] for valid indices
    # In terms of s_idx: shift = s_idx - (n-1)
    # D[p, s_idx] = (1-h[p-(s_idx-(n-1))]) - h[p+(s_idx-(n-1))]
    #              = (1-h[p + (n-1) - s_idx]) - h[p + s_idx - (n-1)]
    
    p = np.arange(n)[:, None]  # (n, 1)
    s = np.arange(2*n-1)[None, :]  # (1, 2n-1)
    shift = s - (n-1)  # (1, 2n-1), values from -(n-1) to (n-1)
    
    # p - shift: (n, 2n-1)
    pt = p - shift  # p - (s - (n-1)) = p + (n-1) - s
    ps = p + shift  # p + (s - (n-1)) = p + s - (n-1)
    
    D = np.zeros((n, 2*n-1))
    valid_t = (pt >= 0) & (pt < n)
    valid_s = (ps >= 0) & (ps < n)
    D[valid_t] += (1 - h[pt[valid_t]])
    D[valid_s] -= h[ps[valid_s]]
    return D

def fast_local_search(h, n, max_sweeps, deadline, seed=42):
    """Fast binary local search using D matrix"""
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    D = compute_D(h, n)
    cur_max = corr.max()
    best_c = cur_max
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Process zeros in random order
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            if h[i] != 0:  # might have changed
                continue
            # Find best j among ones
            # new_corr = corr + D[i] - D[j]
            # We want to minimize max(new_corr)
            # Quick filter: D[i, s*] < D[j, s*] where s* = argmax(corr)
            s_star = np.argmax(corr)
            di = D[i]
            dj = D[ones]  # (n_ones, 2n-1)
            # Filter
            mask = di[s_star] < dj[:, s_star]
            if not mask.any():
                continue
            # For passing candidates, compute new max
            candidates = corr[None, :] + di[None, :] - dj[mask]  # (n_cand, 2n-1)
            new_maxes = candidates.max(axis=1)
            best_local = new_maxes.min()
            if best_local < cur_max - 1e-15:
                j = ones[mask][new_maxes.argmin()]
                # Accept swap
                h[i], h[j] = 1, 0
                # Update corr
                corr = corr + D[i] - D[j]
                cur_max = corr.max()
                # Update D (O(n) per swap)
                # h[i] went 0->1:
                for p in range(n):
                    si = p - i  # s_idx where p - shift = i, shift = p-i, s_idx = p-i+(n-1)
                    sj = i - p  # s_idx where p + shift = i, shift = i-p, s_idx = i-p+(n-1)
                    si_idx = si + (n-1)
                    sj_idx = sj + (n-1)
                    if 0 <= si_idx < 2*n-1:
                        D[p, si_idx] -= 1  # (1-h[i]) went from 1 to 0
                    if 0 <= sj_idx < 2*n-1:
                        D[p, sj_idx] -= 1  # -h[i] went from 0 to -1
                # h[j] went 1->0:
                for p in range(n):
                    si = p - j
                    sj = j - p
                    si_idx = si + (n-1)
                    sj_idx = sj + (n-1)
                    if 0 <= si_idx < 2*n-1:
                        D[p, si_idx] += 1  # (1-h[j]) went from 0 to 1
                    if 0 <= sj_idx < 2*n-1:
                        D[p, sj_idx] += 1  # -h[j] went from -1 to 0
                
                # Update ones and zeros
                ones = np.delete(ones, np.where(ones == j)[0][0])
                zeros = np.delete(zeros, np.where(zeros == i)[0][0])
                improved = True
                if cur_max < best_c:
                    best_c = cur_max
                    best_h = h.copy()
        if not improved:
            break
    
    return best_h, best_c * (2.0/n)

def parallel_sweep(h, corr, D, n, deadline):
    """Find a set of non-conflicting improving swaps and apply them"""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    cur_max = corr.max()
    s_star = np.argmax(corr)
    
    accepted_i = []
    accepted_j = []
    used_j = set()
    
    for i in zeros:
        if time.time() > deadline:
            break
        di = D[i]
        if di[s_star] >= D[ones, s_star].max():  # no possible improvement at s_star
            continue
        # Filter
        mask = di[s_star] < D[ones, s_star]
        if not mask.any():
            continue
        candidates = corr[None, :] + di[None, :] - D[ones[mask]]
        new_maxes = candidates.max(axis=1)
        best_local = new_maxes.min()
        if best_local < cur_max - 1e-12:
            j = ones[mask][new_maxes.argmin()]
            if j not in used_j:
                accepted_i.append(i)
                accepted_j.append(j)
                used_j.add(j)
    
    if not accepted_i:
        return 0
    
    # Apply all swaps
    for i, j in zip(accepted_i, accepted_j):
        h[i], h[j] = 1, 0
        corr = corr + D[i] - D[j]
    
    # Update D
    p = np.arange(n)
    for i, j in zip(accepted_i, accepted_j):
        # h[i] 0->1
        idx1 = p - i + (n-1)
        idx2 = i - p + (n-1)
        D[p, idx1] -= 1
        D[p, idx2] -= 1
        # h[j] 1->0
        idx3 = p - j + (n-1)
        idx4 = j - p + (n-1)
        D[p, idx3] += 1
        D[p, idx4] += 1
    
    return len(accepted_i)

        if len(ones) == 0 or len(zeros) == 0:
            break
        s_star = int(np.argmax(corr))
        zero_order = rng.permutation(zeros)
        improved = False

        for i in zero_order:
            if time.time() > deadline:
                break
            if h[i] != 0:
                continue
            di = D[i]
            dj_star = D[ones, s_star]
            mask = di[s_star] < dj_star
            if not mask.any():
                continue
            dj = D[ones[mask]]
            candidates = corr[None, :] + di[None, :] - dj
            new_maxes = candidates.max(axis=1)
            best_idx = int(new_maxes.argmin())
            new_max = new_maxes[best_idx]

            if anneal and new_max >= cur_max:
                delta = new_max - cur_max
                if rng.random() >= np.exp(-delta / max(T, 1e-10)):
                    continue

            j = int(ones[mask][best_idx])
            h[i], h[j] = 1.0, 0.0
            corr += di - D[j]
            cur_max = corr.max()
            update_D(D, i, 0, n)
            update_D(D, j, 1, n)
            ones = np.delete(ones, np.where(ones == j)[0][0])
            zeros = np.delete(zeros, np.where(zeros == i)[0][0])
            improved = True
            if cur_max < best_c:
                best_c = cur_max
                best_h = h.copy()

        if not improved:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        else:
            no_improve = 0

    return best_h, best_c * (2.0 / n)

def slsqp_smooth(h_init, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

def perturbed_binary(h_base, n, n_swaps, rng):
    h = h_base.copy()
    for _ in range(n_swaps):
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            break
        a = int(rng.choice(oi))
        b = int(rng.choice(zi))
        h[a], h[b] = 0.0, 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_c = np.inf
    best_h = None
    best_n = n0

    n_list = sorted(set([n0, 200, 250, 300]))
    time_per_n = max(60, (deadline - t0 - 120) / len(n_list))

    for n in n_list:
        if time.time() > deadline - 100:
            break
        phase_dl = min(time.time() + time_per_n, deadline - 100)

        if n == n0:
            h_base = h0.copy()
        else:
            xo = np.linspace(0, 2, n0)
            xn = np.linspace(0, 2, n)
            h_base = np.interp(xn, xo, h0)

        hb = make_binary(h_base, n)

        restarts = [(0, False), (5, False), (12, False), (5, True), (20, True)]
        for n_sw, use_anneal in restarts:
            if time.time() > phase_dl:
                break
            h_try = perturbed_binary(hb, n, n_sw, rng) if n_sw > 0 else hb.copy()
            h_res, c_res = local_search_binary(h_try, n, phase_dl,
                                               seed=seed + n + n_sw,
                                               anneal=use_anneal)
            if c_res < best_c:
                best_c = c_res
                best_h = h_res.copy()
                best_n = n

    # SLSQP polish
    if time.time() < deadline - 30 and best_h is not None:
        hp = best_h + 0.02 * rng.randn(best_n)
        hp = project_feasible(hp, best_n)
        for alpha in [50000, 500000, 5000000]:
            if time.time() > deadline - 10:
                break
            hp = slsqp_smooth(hp, best_n, alpha, 800, deadline - 5)
            c = compute_c5(hp, best_n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()

        # Binarize refined solution and do final local search
        if time.time() < deadline - 5:
            hb2 = make_binary(best_h, best_n)
            h_res, c_res = local_search_binary(hb2, best_n, deadline, seed=seed + 999)
            if c_res < best_c:
                best_c = c_res
                best_h = h_res.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
