def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

The problem is: min f(h) = max_k g_k(h) s.t. h ∈ P where P = {h : 0≤h≤1, sum(h)=n/2}

Frank-Wolfe iterates:
1. At current h, compute subgradient g = ∇g_{k*}(h) where k* is argmax
2. Solve linear oracle: s = argmin_{x ∈ P} g^T x
3. Update: h ← h + γ(g^T s) * (s - h) for appropriate step size γ

The linear oracle is easy: s_i = 0 if g_i > 0, s_i = 1 if g_i < 0, and we need exactly n/2 of them to be 1. So s is binary: set the n/2 indices with the smallest g_i to 1, rest to 0.

    for it in range(max_iters):
        corr = get_correlations(h, n)
        k_star = np.argmax(corr)
        # Subgradient: gradient of g_{k*}(h) = sum_i h_i(1-h_{i+k*})
        k = k_star - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            nv = n - k
            grad[:nv] += (1 - h[k:])  # d/dh_i of h_i(1-h_{i+k})
            grad[k:] += (-h[:nv])       # d/dh_{i+k} of h_i(1-h_{i+k})

            nv = n - kk
            grad[kk:] += (1 - h[:nv])
            grad[:nv] += (-h[kk:])
        
        # Linear oracle: find s in P that minimizes grad^T s
        # s_i = 1 for the n/2 indices with smallest grad_i

        s = np.zeros(n)

        # Gap

        if gap > -tol:
            break
        
        # Step size: backtracking or closed-form

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.maximum(1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.minimum(h + (diff / tot) * room, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.maximum(h + (diff / tot) * h, 0.0)
    return np.clip(h, 0.0, 1.0)

def grad_corr(h, n, s):
    """Gradient of correlation at shift s (s = k - (n-1) where k is numpy index)"""
    grad = np.zeros(n)
    if s >= 0:
        nv = n - s
        grad[:nv] += -h[s:]
        grad[s:] += 1.0 - h[:nv]
    else:
        kk = -s
        nv = n - kk
        grad[nv:] += -h[:nv]  # Wait, let me re-derive for s < 0
        # c_s = sum_{n=kk}^{n-1} h_n * (1-h_{n-kk})

        # So: grad[kk:n] += 1-h[0:n-kk]
        #     grad[0:n-kk] += -h[kk:n]

def frank_wolfe_run(h, n, deadline, max_iters=50000):
    h = h.copy()
    best_c = compute_c5(h, n)
    target_ones = n // 2
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        
        c5 = corr[k_star] * (2.0 / n)
        if c5 < best_c:
            best_c = c5
        
        grad = grad_corr(h, n, s)
        
        # Linear oracle: set target_ones indices with smallest grad to 1
        idx = np.argsort(grad)
        s_vec = np.zeros(n)
        s_vec[idx[:target_ones]] = 1.0
        
        gap = np.dot(grad, s_vec - h)
        if gap > -1e-13:
            break
        
        t = it + 1
        gamma = 2.0 / (t + 2)
        h = h + gamma * (s_vec - h)
        h = project_feasible(h, n)
    
    return h, best_c

def smooth_max_obj(x, n, alpha, top_m=None):
    """Smooth max of correlations, optionally only over top_m shifts"""
    c = np.correlate(x, 1 - x, mode='full')
    if top_m is not None and top_m < len(c):
        # Keep only top_m
        idx = np.argsort(c)[-top_m:]
        c = c[idx]
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    val = m + np.log(np.sum(np.exp(sh - smax))) / alpha
    return val

def smooth_max_grad(x, n, alpha, top_m=None):
    c = np.correlate(x, 1 - x, mode='full')
    if top_m is not None and top_m < len(c):
        top_idx = set(np.argsort(c)[-top_m:].tolist())
    else:
        top_idx = None
    m = c.max()
    sh = alpha * (c - m)
    smax = sh.max()
    w = np.exp(sh - smax)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx in range(len(c)):
        if w[idx] < 1e-12:
            continue
        if top_idx is not None and idx not in top_idx:
            continue
        s = idx - (n - 1)
        wm = w[idx]
        if s >= 0:
            nv = n - s
            grad[:nv] += wm * (-x[s:])
            grad[s:] += wm * (1.0 - x[:nv])
        else:
            kk = -s
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad
