import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    # log-sum-exp
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return float(val)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        return obj_smooth(x, n, alpha)
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(fun, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def optimal_pair_descent(h, n, deadline, radius=30, max_bisect=8):
    """For each nearby pair (i,j), find optimal transfer via bisection on convex objective."""
    best_c = compute_c5(h, n)
    n_total = 2 * n - 1  # number of lags
    
    for d_pass in range(4):
        improved_any = False
        for i in range(n):
            if time.time() > deadline:
                break
            lo_j = max(0, i - radius)
            hi_j = min(n, i + radius + 1)
            for j in range(lo_j, hi_j):
                if i == j:
                    continue
                if time.time() > deadline:
                    break
                # Feasible range for delta
                delta_lo = max(-h[i], h[j] - 1.0)
                delta_hi = min(1.0 - h[i], h[j])
                if delta_hi - delta_lo < 1e-10:
                    continue
                
                # Bisection to find optimal delta
                a, b = delta_lo, delta_hi
                # Check endpoints and midpoint
                best_delta = 0.0
                best_val = best_c
                
                for _ in range(max_bisect):
                    mid = (a + b) / 2
                    # Evaluate at mid
                    h[i] += mid; h[j] -= mid
                    c_mid = compute_c5(h, n)
                    h[i] -= mid; h[j] += mid
                    
                    if c_mid < best_val:
                        best_val = c_mid
                        best_delta = mid
                    
                    # Determine which side to search
                    # Evaluate at a/3 and 2a+b/3 points (ternary search)
                    t1 = a + (b - a) / 3
                    t2 = b - (b - a) / 3
                    
                    h[i] += t1; h[j] -= t1

                    if c1 < c2:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(len(x))

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            lo_j = max(0, i - radius)
            hi_j = min(n, i + radius + 1)
            for j in range(lo_j, hi_j):
                if i == j:
                    continue
                if time.time() > deadline:
                    return h, best_c
                if abs(h[i] - 0.0) < 1e-14 and abs(h[i] - 1.0) < 1e-14:
                    continue

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(len(x))

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def fast_pair_descent(h, n, deadline, max_sweeps=300, bisect_iter=10):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    best_c = float(corr.max() * dx)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            for j in range(i + 1, n):
                delta_lo = max(-h[i], h[j] - 1.0)
                delta_hi = min(1.0 - h[i], h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                d = np.zeros(n_corr)
                klo = max(0, n - 1 - i)
                khi = min(n_corr - 1, 2 * n - 2 - i)
                if klo <= khi:
                    d[klo:khi+1] += (1.0 - h[i + np.arange(klo, khi+1) - (n-1)])
                klo = max(0, i)
                khi = min(n_corr - 1, i + n - 1)
                if klo <= khi:
                    d[klo:khi+1] -= h[i + (n-1) - np.arange(klo, khi+1)]
                klo = max(0, n - 1 - j)
                khi = min(n_corr - 1, 2 * n - 2 - j)
                if klo <= khi:
                    d[klo:khi+1] -= (1.0 - h[j + np.arange(klo, khi+1) - (n-1)])
                klo = max(0, j)
                khi = min(n_corr - 1, j + n - 1)
                if klo <= khi:
                    d[klo:khi+1] += h[j + (n-1) - np.arange(klo, khi+1)]
                a_s, b_s = delta_lo, delta_hi
                for _ in range(bisect_iter):
                    m1 = a_s + (b_s - a_s) / 3.0
                    m2 = b_s - (b_s - a_s) / 3.0
                    if float(np.max(corr + m1 * d)) < float(np.max(corr + m2 * d)):
                        b_s = m2
                    else:
                        a_s = m1
                delta_opt = (a_s + b_s) / 2.0
                val_opt = float(np.max(corr + delta_opt * d))
                cur_max = float(np.max(corr))
                if val_opt < cur_max - 1e-12:
                    h[i] += delta_opt
                    h[j] -= delta_opt
                    corr += delta_opt * d
                    if val_opt * dx < best_c:
                        best_c = val_opt * dx
                    improved = True
        if not improved:
            break
    return h, best_c

def cosine_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for k in range(1, 6):
        amp = rng.uniform(0.05, 0.2) / k
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(np.pi * k * x + phase)
    return project_feasible(h, n)
