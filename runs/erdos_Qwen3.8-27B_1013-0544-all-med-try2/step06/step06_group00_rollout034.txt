```python
def c5_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    max_c = c.max()
    shifted = alpha * (c - max_c)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    obj = max_c + np.log(exp_c.sum()) / alpha
    
    A = np.convolve(w, 1-h, mode='full')[n-1 : 2*n-1]
    B = np.convolve(w[::-1], h, mode='full')[n-1 : 2*n-1]
    g = A - B
    return obj, g
```

```python
        if s > target:

        else:
```

```python
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_c = c.max()
    shifted = alpha * (c - max_c)
    shifted = np.clip(shifted, None, 700)
    exp_c = np.exp(shifted)
    lse = max_c + np.log(exp_c.sum()) / alpha
    penalty = mu * (h.sum() - n / 2.0) ** 2
    obj = lse + penalty
    
    w = exp_c / exp_c.sum()
    A = np.convolve(w, 1 - h, mode='full')[n-1:2*n-1]
    B = np.convolve(w[::-1], h, mode='full')[n-1:2*n-1]
    g = dx * (A - B) + 2 * mu * (h.sum() - n / 2.0)
    return obj, g
```
