            if s < 1e-14:

def smooth_max(c, alpha):
    """Compute log-sum-exp approximation of max(c) and its gradient w.r.t. c."""
    cm = c.max()
    shifted = alpha * (c - cm)
    logsum = cm + np.log(np.exp(shifted).sum()) / alpha
    # weights for gradient
    weights = np.exp(shifted) / np.exp(shifted).sum()
    return logsum, weights
