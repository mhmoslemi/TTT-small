    s = np.log(np.exp(alpha * (g - m)).sum())

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    s = np.log(np.exp(alpha * (g - m)).sum())
    return m + s / alpha

def lse_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    ez = np.exp(alpha * (g - m))
    w = ez / ez.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        wk = w[idx]
        if wk < 1e-13:
            continue
        if k >= 0:
            if k < n:
                js = np.arange(k, n)
                grad[js] += wk * (1.0 - x[js - k])
                js2 = np.arange(0, n - k)
                grad[js2] -= wk * x[js2 + k]
        else:
            l = -k
            if l < n:
                js = np.arange(l, n)
                grad[js] += wk * (1.0 - x[js - l])
                js2 = np.arange(0, n - l)
                grad[js2] -= wk * x[js2 + l]
    return grad

def slsqp_optimize(h0, n, alpha, maxiter, deadline):
    def obj(x):
        return lse_obj(x, n, alpha)
    def jac(x):
        return lse_grad(x, n, alpha)
    def cf(x):
        return x.sum() - n / 2.0
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def grad_descent_polish(h0, n, alpha, deadline, max_iter=5000):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 2.0 / (alpha * 1.0)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = lse_grad(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = lr
        improved = False
        for _ in range(40):
            ht = project_feasible(h - step * g, n)
            ct = compute_c5(ht, n)
            if ct < best_c:
                h = ht
                best_c = ct
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            if lr < 1e-15:
                break
            lr *= 0.5
    return best_h, best_c
