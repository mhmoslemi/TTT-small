import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def eq_fun(h, target_sum):
    return h.sum() - target_sum

def eq_jac(h, target_sum):
    return np.ones(len(h))

def subgrad_max(h, idx):
    n = len(h)
    l = idx - (n - 1)
    grad = np.zeros(n)
    if l == 0:
        grad[:] = 1.0 - 2.0 * h
    elif l > 0:
        grad[l:] += (1.0 - h[:n - l])
        grad[:n - l] -= h[l:]
    else:
        m = -l
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def slsqp_solve(h0, n, dx, target_sum, beta, maxiter=5000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (target_sum,)}]
    try:
        res = minimize(obj_smooth, h0, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return None, None

def subgradient_phase(h0, n, target_sum, n_iters, tl_fn):
    h = h0.copy()
    avg_h = np.zeros(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    for it in range(n_iters):
        if tl_fn() < 40:
            break
        k_idx = np.argmax(corr)
        top_k = np.argsort(corr)[-3:]
        k_idx = top_k[np.random.randint(len(top_k))]
        grad = subgrad_max(h, k_idx)
        lr = 0.005 / np.sqrt(1.0 + it * 0.002)
        h = h - lr * grad
        h = project_feasible(h, target_sum)
        avg_h += h / (it + 1)
        if it % 2000 == 1999:
            c_avg = compute_c5(avg_h)
            c_cur = compute_c5(h)
            h_keep = avg_h if c_avg < c_cur else h
            c_keep = min(c_avg, c_cur)
            if c_keep < compute_c5(h0):
                h0 = h_keep.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
    c1 = compute_c5(h)
    c2 = compute_c5(avg_h)
    if c2 < c1:
        h = avg_h
    return h, min(c1, c2)

def coord_descent(h0, n_passes, grid_size, tl_fn):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    grid = np.linspace(0.0, 1.0, grid_size)
    for p in range(n_passes):
        if tl_fn() < 20:
            break
        for j in range(n):
            best_val = h[j]
            best_obj = np.inf
            for v in grid:
                h[j] = v
                corr = np.correlate(h, 1.0 - h, mode='full')
                obj = np.max(corr)
                if obj < best_obj:
                    best_obj = obj
                    best_val = v
            h[j] = best_val
        h = project_feasible(h, target_sum)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP continuation
    for beta in [10, 50, 200, 1000, 5000, 20000, 100000]:
        if tl() < 80:
            break
        for trial in range(2):
            if tl() < 30:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += np.random.randn(n) * 0.008
                x0 = project_feasible(x0, target_sum)
            h_new, c5 = slsqp_solve(x0, n, dx, target_sum, beta, 4000)
            if h_new is not None and c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    # Phase 2: Subgradient with Polyak averaging
    if tl() > 120:
        h_sub, c5_sub = subgradient_phase(best_h, n, target_sum, 150000, tl)
        if c5_sub < best_c5:
            best_h = h_sub
            best_c5 = c5_sub

    # Phase 3: SLSQP after subgradient
    if tl() > 60:
        for beta in [10000, 50000, 200000]:
            if tl() < 20:
                break
            h_new, c5 = slsqp_solve(best_h, n, dx, target_sum, beta, 6000)
            if h_new is not None and c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    # Phase 4: Coordinate descent
    if tl() > 50:
        h_cd = coord_descent(best_h, 25, 35, tl)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    # Phase 5: SLSQP after coordinate descent
    if tl() > 30:
        for beta in [50000, 200000]:
            if tl() < 15:
                break
            h_new, c5 = slsqp_solve(best_h, n, dx, target_sum, beta, 8000)
            if h_new is not None and c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    # Phase 6: Multi-resolution search
    if tl() > 100:
        for n_try in [n + 2, n - 2, n + 4, n - 4, n + 8, n + 12]:
            if n_try < 30 or tl() < 50:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            improved = False
            for beta in [500, 5000, 50000]:
                if tl() < 20:
                    break
                h_new, c5 = slsqp_solve(h_t, n_try, dx_t, ts_t, beta, 3000)
                if h_new is not None and c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
                    n = n_try
                    dx = dx_t
                    target_sum = ts_t
                    improved = True
            if improved and tl() > 30:
                h_cd = coord_descent(best_h, 10, 25, tl)
                c5_cd = compute_c5(h_cd)
                if c5_cd < best_c5:
                    best_h = h_cd
                    best_c5 = c5_cd

    # Phase 7: Final polish
    if tl() > 15:
        h_new, c5 = slsqp_solve(best_h, n, dx, target_sum, 500000, 10000)
        if h_new is not None and c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    return best_h, best_c5, len(best_h)
