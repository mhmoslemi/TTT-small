def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()

            if s < 1e-14:

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_descent(h, n, deadline, eta_init=0.01, eta_final=0.0001, n_iters=1000000, seed=42):
    """Projected subgradient descent for minimizing max correlation."""
    dx = 2.0 / n
    target = n / 2.0
    rng = np.random.RandomState(seed)
    
    h = h.copy()
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    best_local_c = cur_c
    best_local_h = h.copy()
    no_improve = 0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        # Decay learning rate
        t = it / n_iters
        eta = eta_init * (eta_final / eta_init) ** t
        
        # Find argmax
        m_star = np.argmax(corr)
        
        # Compute subgradient
        # corr_full[m] = sum_i h[i] * (1-h[i+m-(n-1)])
        # d/dh[p] = (1-h[p+m-(n-1)]) - h[p-m+(n-1)] (with range checks)
        s = m_star - (n - 1)  # the actual shift
        grad = np.zeros(n)
        
        # First term: (1-h[p+s]) for p such that 0 <= p+s < n
        # p+s < n → p < n-s
        # p+s >= 0 → p >= -s
        p_lo = max(0, -s)
        p_hi = min(n, n - s)
        if p_hi > p_lo:
            grad[p_lo:p_hi] += 1.0 - h[p_lo + s : p_hi + s]
        
        # Second term: -h[p-s] for p such that 0 <= p-s < n
        # p-s < n → p < n+s
        # p-s >= 0 → p >= s
        p_lo2 = max(0, s)
        p_hi2 = min(n, n + s)
        if p_hi2 > p_lo2:
            grad[p_lo2:p_hi2] -= h[p_lo2 - s : p_hi2 - s]
        
        # Update
        h = h - eta * grad
        h = project_feasible(h, n)
        g = 1.0 - h
        
        # Compute new correlation
        corr = np.correlate(h, g, mode='full').astype(float)
        new_c = corr.max() * dx
        
        if new_c < best_local_c - 1e-10:
            best_local_c = new_c
            best_local_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if new_c < best_c:
            best_c = new_c
            best_h = h.copy()

        if no_improve > 5000:
