        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad, cmax

def active_set_grad_method(h, n, deadline, eps_init=0.02, lr_init=0.008):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    eps = eps_init
    stagnant = 0
    for it in range(200000):
        if time.time() > deadline:
            break
        g, _ = exact_subgradient(h, n, eps)
        g += 0.3 * (h.sum() - n / 2.0)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        improved = False
        step_lr = lr
        for _ in range(50):
            h_new = project_feasible(h - step_lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                stagnant = 0
                break
            step_lr *= 0.5
        if not improved:
            stagnant += 1
            if stagnant % 50 == 0:
                lr = max(0.001, lr * 0.7)
                eps = min(0.5, eps * 1.3)
            if lr < 1e-13:
                break
    return best_h, best_c

def smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full')
    dx = 2.0 / n
    c = c * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-13:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(5000, 5.0), (50000, 30.0), (200000, 100.0), (1000000, 500.0)]
    for alpha, mu in configs:
        if time.time() > deadline:
            break
        def obj_fn(x, a=alpha, mm=mu):
            v = smooth_max_obj(x, n, a)
            v += mm * (x.sum() - n / 2.0) ** 2
            return v
        def grad_fn(x, a=alpha, mm=mu):
            g = smooth_max_grad(x, n, a)
            g += 2.0 * mm * (x.sum() - n / 2.0) * np.ones(n)
            return g
        try:
            res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': max_iters * 12,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, deltas=None, radius=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius) if radius else 0
                hi = min(n, i + radius + 1) if radius else n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def multi_start_refine(best_h, best_c, n, deadline, rng, n_starts=5):
    for s in range(n_starts):
        if time.time() > deadline - 30:
            break
        h_pert = best_h + rng.randn(n) * 0.03 * (1.0 + s * 0.5)
        h_pert = project_feasible(h_pert, n)
        alloc = min(30, deadline - time.time() - 20)
        if alloc < 10:
            break
        h_p, c_p = active_set_grad_method(h_pert, n, time.time() + alloc * 0.5,
                                           eps_init=0.05, lr_init=0.005)
        if time.time() < deadline - 10:
            h_p, c_p = pair_search(h_p, n, time.time() + min(10, deadline - time.time() - 5),
                                    [0.001, 0.0005, 0.0002, 0.0001], radius=40)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p.copy()
    return best_h, best_c

def try_other_n(best_h, best_c, deadline, rng):
    n_orig = len(best_h)
    x_orig = np.linspace(0, 2, n_orig)
    for n_try in [250, 350, 400, 500]:
        if time.time() > deadline - 40:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(50, deadline - time.time() - 30)
        if alloc < 20:
            break
        h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.4, max_iters=150)
        if time.time() < deadline - 15:
            h_t, c_t = active_set_grad_method(h_t, n_try, time.time() + min(15, deadline - time.time() - 10),
                                               eps_init=0.02, lr_init=0.005)
        if time.time() < deadline - 8:
            h_t, c_t = pair_search(h_t, n_try, time.time() + min(8, deadline - time.time() - 5),
                                    [0.001, 0.0005, 0.0002, 0.0001], radius=35)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B
    t1 = min(t0 + 180, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, n_init, t1, max_iters=200)

    # Phase 2: Active set gradient (main workhorse)
    if time.time() < deadline - 250:
        t2 = min(time.time() + 120, deadline - 250)
        h2, c2 = active_set_grad_method(best_h, n_init, t2)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Another L-BFGS pass
    if time.time() < deadline - 200:
        t3 = min(time.time() + 80, deadline - 200)
        h3, c3 = lbfgs_phase(best_h, n_init, t3, max_iters=150)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Active set gradient again
    if time.time() < deadline - 150:
        t4 = min(time.time() + 80, deadline - 150)
        h4, c4 = active_set_grad_method(best_h, n_init, t4, eps_init=0.01, lr_init=0.004)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Pair search (coarse to fine)
    if time.time() < deadline - 80:
        t5 = min(time.time() + 100, deadline - 80)
        best_h, best_c = pair_search(best_h, n_init, t5, radius=50)

    # Phase 6: Multi-start perturbation
    if time.time() < deadline - 60:
        best_h, best_c = multi_start_refine(best_h, best_c, n_init, deadline - 20, rng, n_starts=5)

    # Phase 7: Try other n values
    if time.time() < deadline - 50:
        best_h, best_c = try_other_n(best_h, best_c, deadline - 15, rng)

    # Phase 8: Final refinement on the best
    nf = len(best_h)
    if time.time() < deadline - 10:
        best_h, best_c = active_set_grad_method(best_h, nf, deadline - 10,
                                                 eps_init=0.01, lr_init=0.003)
    if time.time() < deadline - 3:
        best_h, best_c = pair_search(best_h, nf, deadline - 1,
                                      [0.0005, 0.0002, 0.0001, 0.00005], radius=30)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
