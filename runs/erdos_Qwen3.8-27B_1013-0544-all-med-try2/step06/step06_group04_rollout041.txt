Let me also think about whether to use cvxpy for the outer approximation. The outer approximation solves:
min t

This is a linear program with 2n-1 + 1 constraints and n+1 variables. For n=300, that's about 600 constraints and 301 variables. GLOP or CBC should handle this quickly.

Actually, for the final polishing, let me try a different approach: **alternating minimization on pairs of lags**.

At the optimum, the top lags should be equal. If lags k1 and k2 are both at the maximum, we can try to find a direction that keeps both equal while decreasing them.

The condition for a direction d (with sum(d)=0) to keep c[k1] = c[k2] = t is:
grad_{k1}(h) * d = grad_{k2}(h) * d (first-order)

And to decrease t:
grad_{k1}(h) * d < 0 (which implies grad_{k2}(h) * d < 0 as well)

So we need: (grad_{k1} - grad_{k2}) * d = 0 and grad_{k1} * d < 0.

This is a system of linear equations/inequalities in d. We can solve it with LP.

Actually, a simpler approach: minimize (grad_{k1}(h) * d + grad_{k2}(h) * d) / 2 subject to (grad_{k1} - grad_{k2}) * d = 0, ||d||_inf <= epsilon, sum(d) = 0.

This is a small LP that gives us the steepest descent direction that equalizes the top lags.

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    """Correct gradient of c[lag] w.r.t. h (without dx factor)."""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:m] = 1.0 - h[m:]

        if n - m < n:

    else:
        grad[:m] = -h[m:]
        if n - m > m:

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w /= w.sum()
    return w

def slsqp_smooth(h, n, alpha, maxiter=5000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        return smooth_max_val(c, alpha)
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        w = smooth_max_grad(c / dx, alpha)  # weights on the unscaled c
        # Actually, let me be more careful
        # c_scaled = c_unscaled * dx
        # smooth_max(c_scaled, alpha) = smooth_max(dx * c_unscaled, alpha)
        # The gradient w.r.t. x is dx * sum_k w_k * grad(c_unscaled[k])
        # where w_k are the softmax weights of alpha * (c_scaled - max(c_scaled))
        # = alpha * dx * (c_unscaled - max(c_unscaled))
        # So the weights are the same whether we use scaled or unscaled
        g = np.zeros(n)

            g += w[k] * _lag_gradient(x, n, lag)
        return dx * g
    def sum_con(x):
        return np.sum(x) - n / 2.0
    def sum_con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})

    except Exception:
        return h.copy()

def outer_approx_lp(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.GLOP, cp.CBC, cp.CPLEX]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def gradient_descent_step(h, n, lr, top_m=10):
    """Take a projected gradient descent step on the top m lags."""
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    active_c = c[top_idx]
    w = smooth_max_grad(active_c / dx, 1000)
    
    g = np.zeros(n)
    for idx, k in enumerate(top_idx):
        if w[idx] < 1e-10:
            continue
        lag = k - (n - 1)
        g += w[idx] * _lag_gradient(h, n, lag)
    g *= dx
    
    # Project gradient: remove mean (to maintain sum constraint)
    g -= g.mean()
    # Clip to respect bounds (simple projection)
    step = h - lr * g
    return project_feasible(step, n)

Actually wait, I need to re-examine. The gradient in the previous code for lag > 0 is:
```

grad[m:] += (1.0 - h_curr[:n - m])
```

This gives:
- j in [0, m): grad[j] = -h[j+m]
- j in [m, n-m): grad[j] = -h[j+m] + 1-h[j-m]
- j in [n-m, n): grad[j] = 1-h[j-m]

The correct gradient is:
- j in [0, m): grad[j] = 1-h[j+m]
- j in [m, n-m): grad[j] = 1-h[j+m] - h[j-m]
- j in [n-m, n): grad[j] = -h[j-m]

So the errors are:
- j in [0, m): code gives -h[j+m], correct is 1-h[j+m]. Error = 1.
- j in [n-m, n): code gives 1-h[j-m], correct is -h[j-m]. Error = 1.

So the error is exactly 1 for 2m entries (m at each end). This is a significant constant error that would definitely affect optimization!

The sum of the gradient errors is: m*1 + m*1 = 2m (but with the dx factor, it's 2m*dx). Since we also have the constraint sum(h) = n/2, the projection onto this constraint would remove the mean of the gradient, which includes part of this error. But it's not a pure constant, so the projection wouldn't fully fix it.

This is a significant bug that could explain why the optimizer gets stuck. Let me make sure my code has the correct gradient.

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:m] = 1.0 - h[m:]
        if n - m > m:
            grad[m:n-m] = (1.0 - h[2*m:n]) - h[:n-2*m]
        grad[n-m:] = -h[:m]
    else:
        grad[:m] = -h[m:]
        if n - m > m:
            grad[m:n-m] = (1.0 - h[:n-2*m]) - h[2*m:n]
        grad[n-m:] = 1.0 - h[:m]
    return grad

def _softmax_weights(vals, alpha):
    shifted = alpha * (vals - vals.max())
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(vals)] = 1.0
    else:
        w /= s
    return w

def _smooth_max_val(vals, alpha):
    m = vals.max()
    shifted = alpha * (vals - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _weighted_lag_gradient(x, n, k_indices, weights):
    g = np.zeros(n)
    for idx, k in enumerate(k_indices):
        if weights[idx] < 1e-12:
            continue
        lag = k - (n - 1)
        g += weights[idx] * _lag_gradient(x, n, lag)
    return g

def slsqp_optimize(h, n, alpha, top_m=None, maxiter=3000):
    dx = 2.0 / n
    if top_m is None:
        k_indices = list(range(2 * n - 1))
    else:
        c = compute_all_lags(h, n)
        k_indices = list(np.argsort(c)[-top_m:])
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        if top_m is None:
            return _smooth_max_val(c, alpha)
        return _smooth_max_val(c[k_indices], alpha)
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        if top_m is None:
            w = _softmax_weights(c, alpha)
        else:
            w = _softmax_weights(c[k_indices], alpha)
        g = _weighted_lag_gradient(x, n, k_indices, w)
        return dx * g
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.GLOP, cp.CBC]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    if h_var.value is None:
        return h_curr.copy()
    return project_feasible(h_var.value, n)

        step = np.max(np.abs(h_new - h_curr))

        trust = max(0.002, trust * 0.88)
    return best_h, best_c

    return best_h, best_c
