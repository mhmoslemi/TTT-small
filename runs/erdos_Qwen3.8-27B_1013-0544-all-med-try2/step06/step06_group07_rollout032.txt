    n = len(h)

For coordinate descent with the sum constraint:
- Fix all h[j] for j ≠ i and j ≠ i+1
- Optimize (h[i], h[i+1]) subject to h[i] + h[i+1] = target_pair (where target_pair = target_sum - sum of all others)
- This is a 2D problem: minimize max_k c_k(h[i], h[i+1]) subject to h[i]+h[i+1] = s, 0 ≤ h[i] ≤ 1, 0 ≤ h[i+1] ≤ 1

Each c_k is linear in (h[i], h[i+1]) (since each c_k involves at most one term with h[i] and at most one with h[i+1]). So we're minimizing the max of linear functions in 2D with a linear constraint. This can be solved exactly!

The min of max of linear functions is a linear program in 2D, which can be solved by checking all pairs of active constraints. With 2n-1 linear functions, this is O((2n)^2) = O(n^2) per coordinate pair update. For n=70, that's about 19600 operations, which is fast.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def obj_grad_pen(h, dx, beta, target_sum, lam):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = smooth_max(corr, beta) * dx + lam * (h.sum() - target_sum)**2
    weights = np.exp(np.clip(beta * (corr - np.max(corr)), -700, 700))
    weights /= weights.sum()
    grad = _corr_grad(h, weights) * dx
    grad += 2.0 * lam * (h.sum() - target_sum)
    return val, grad

def _corr_grad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        off = k - (n - 1)
        # First term: (1 - h[j+off]) for valid j+off
        lo1 = max(0, -off)
        hi1 = min(n, n - off)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1+off:hi1+off])
        # Second term: -h[j-off] for valid j-off
        lo2 = max(0, off)
        hi2 = min(n, n + off)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2-off:hi2-off]
    return grad

def coord_descent_sweep(h, target_sum, n_grid=200):
    n = len(h)
    for it in range(2 * n):  # 2 full sweeps
        for i in range(n):
            j = (i + 1) % n
            s = target_sum - (h.sum() - h[i] - h[j])
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_lo >= t_hi:
                continue
            # Grid search
            best_t = h[i]
            best_obj = np.inf
            h_save_i = h[i]
            h_save_j = h[j]
            for t in np.linspace(t_lo, t_hi, n_grid):
                h[i] = t
                h[j] = s - t
                corr = np.correlate(h, 1.0 - h, mode='full')
                obj = np.max(corr)
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
            h[i] = best_t
            h[j] = s - best_t
    return h

    t0 = time.time()

    else:

    n_list = [n for n in n_list if n >= 30]

def obj_grad_pen(h, dx, beta, target_sum, lam):
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = smooth_max(corr, beta) * dx + lam * (h.sum() - target_sum) ** 2
    weights = np.exp(np.clip(beta * (corr - np.max(corr)), -700, 700))
    weights /= weights.sum()
    grad = _corr_grad(h, weights) * dx
    grad += 2.0 * lam * (h.sum() - target_sum)
    return val, grad

def coord_pair_optimize(h, i, j, target_sum, n_grid=300):
    n = len(h)
    s = target_sum - (h.sum() - h[i] - h[j])
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    if t_lo >= t_hi:
        return h
    ts = np.linspace(t_lo, t_hi, n_grid)
    h_save_i = h[i]
    h_save_j = h[j]
    best_t = h_save_i
    best_obj = np.inf
    for t in ts:
        h[i] = t
        h[j] = s - t
        corr = np.correlate(h, 1.0 - h, mode='full')
        obj = np.max(corr)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    h[i] = best_t
    h[j] = s - best_t
    return h

def coord_descent(h, target_sum, n_grid=200, max_sweeps=10, tol=1e-12):
    n = len(h)
    prev_obj = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            j = (i + 1) % n
            coord_pair_optimize(h, i, j, target_sum, n_grid)
        obj = compute_c5(h)
        if abs(prev_obj - obj) < tol:
            break
        if obj < prev_obj:
            improved = True
            prev_obj = obj
    return h, compute_c5(h)

def nesterov_subgrad(h, dx, target_sum, n_iter=20000, topk=5):
    n = len(h)
    v = np.zeros(n)
    momentum = 0.7
    prev_h = h.copy()
    best_local = h.copy()
    best_local_c5 = compute_c5(h)
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        topk_idx = np.argsort(corr)[-topk:]
        weights = np.zeros(len(corr))
        weights[topk_idx] = 1.0 / topk
        grad = _corr_grad(h, weights) * dx
        lr = 0.005 / (1.0 + 0.0002 * it)
        momentum = 0.7 + 0.25 * min(1.0, it / 5000.0)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target_sum)
        h = h_new
        if it % 2000 == 1999:
            c5 = compute_c5(h)
            if c5 < best_local_c5:
                best_local = h.copy()
                best_local_c5 = c5
    c5 = compute_c5(h)
    if c5 < best_local_c5:
        best_local = h.copy()
        best_local_c5 = c5
    return best_local, best_local_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS-B with penalty, continuation on multiple n
    n_list = [len(h_init)]
    for delta in [-10, -5, 5, 10, 20]:
        nn = len(h_init) + delta
        if nn >= 30:
            n_list.append(nn)
    n_list = sorted(set(n_list))

    for n in n_list:
        if tl() < 100:
            break
        dx = 2.0 / n
        target_sum = n / 2.0
        h = h_init.copy() if n == len(h_init) else resample_h(h_init, n)
        h = project_feasible(h, target_sum)
        for beta, lam in [(20, 50), (80, 500), (300, 5000), (1000, 50000)]:
            if tl() < 20:
                break
            try:
                res = minimize(obj_grad_pen, h, args=(dx, beta, target_sum, lam),
                               method='L-BFGS-B', jac=True,
                               bounds=[(0, 1)] * n,
                               options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-13})
                h = project_feasible(res.x, target_sum)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 2: Coordinate descent on best solution
    n = len(best_h)
    target_sum = n / 2.0
    if tl() > 120:
        h = best_h.copy()
        h, c5 = coord_descent(h, target_sum, n_grid=200, max_sweeps=20)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    # Phase 3: Nesterov subgradient
    if tl() > 100:
        h = best_h.copy()
        dx = 2.0 / len(h)
        h, c5 = nesterov_subgrad(h, dx, target_sum, n_iter=30000, topk=5)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    # Phase 4: More coordinate descent (finer grid)
    if tl() > 100:
        h = best_h.copy()
        h, c5 = coord_descent(h, target_sum, n_grid=400, max_sweeps=15)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    # Phase 5: L-BFGS-B high-precision polish
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    if tl() > 40:
        for beta, lam in [(3000, 50000), (10000, 500000)]:
            if tl() < 15:
                break
            try:
                res = minimize(obj_grad_pen, best_h, args=(dx, beta, target_sum, lam),
                               method='L-BFGS-B', jac=True,
                               bounds=[(0, 1)] * n,
                               options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
            except Exception:
                pass

    # Phase 6: Nesterov again from improved point
    if tl() > 60:
        h = best_h.copy()
        dx = 2.0 / len(h)
        h, c5 = nesterov_subgrad(h, dx, target_sum, n_iter=20000, topk=3)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    # Phase 7: Final fine coordinate descent
    if tl() > 20:
        h = best_h.copy()
        h, c5 = coord_descent(h, target_sum, n_grid=500, max_sweeps=10)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    # Phase 8: Try nearby n values from the best solution found
    if tl() > 120:
        n_best = len(best_h)
        for n_try in [n_best - 5, n_best + 5, n_best + 10, n_best - 10]:
            if n_try < 30 or tl() < 40:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            # Quick L-BFGS-B
            for beta, lam in [(200, 5000), (2000, 50000)]:
                if tl() < 15:
                    break
                try:
                    res = minimize(obj_grad_pen, h_t, args=(dx_t, beta, ts_t, lam),
                                   method='L-BFGS-B', jac=True,
                                   bounds=[(0, 1)] * n_try,
                                   options={'maxiter': 1500, 'ftol': 1e-16, 'gtol': 1e-14})
                    h_t = project_feasible(res.x, ts_t)
                except Exception:
                    pass
            # Coordinate descent
            if tl() > 15:
                h_t, c5_t = coord_descent(h_t, ts_t, n_grid=200, max_sweeps=8)
                if c5_t < best_c5:
                    best_h = h_t
                    best_c5 = c5_t

    # Final check
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
