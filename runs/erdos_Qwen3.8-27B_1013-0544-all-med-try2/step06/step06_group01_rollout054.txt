    return np.correlate(h, 1 - h, mode='full')

def coord_ascent_pass(h, n, deadline, values=(0.0, 0.25, 0.5, 0.75, 1.0)):
    """Coordinate descent: try each coordinate at several values."""
    h = h.copy()
    cur_c = compute_c5(h, n)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old = h[i]
            for v in values:
                if v == old:
                    continue
                h[i] = v
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < cur_c - 1e-15:
                    cur_c = c
                    improved = True
                    old = v
                    break
                h[i] = old
            else:
                h[i] = old  # restore if no improvement
    return h, cur_c

def binary_swap_search(h, n, deadline, max_sweeps=200, seed=42):
    """Efficient binary swap local search."""
    n = len(h)
    target = n // 2
    # Round to binary
    h = (h >= 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    n_ones = int(h.sum())
    if n_ones > target:
        # Remove excess ones (set to 0)
        ones = np.where(h == 1)[0]
        # Remove random ones
        rng = np.random.RandomState(seed)
        to_remove = rng.choice(ones, n_ones - target, replace=False)
        h[to_remove] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(seed)
        to_add = rng.choice(zeros, target - n_ones, replace=False)
        h[to_add] = 1
    
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1-h, mode='full').copy()
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        # Try all zero→one and one→zero swaps
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                # Compute delta for swapping h[i]:0→1, h[j]:1→0
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                
                delta = np.zeros(2*n-1)
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                msk = (imn >= 0) & (imn < n)  
                delta[msk] -= h[imn[msk]]
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    improved = True
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    break
            if improved:
                break
        if not improved:
            break
    
    return best_h, best_c

def subgrad_refine(h, n, deadline, top_k=10):
    """Subgradient refinement for continuous h."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.05
    patience = 0
    
    while time.time() < deadline and patience < 500:
        corr = corr_full(h, n)
        max_val = corr.max()
        m_idx = np.argmax(corr)
        
        # Find top lags
        sorted_idx = np.argsort(corr)[-top_k:]
        weights = np.exp(corr[sorted_idx] - corr[m_idx])
        weights = weights / weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for ki in sorted_idx:
            ki = int(ki)
            k = ki - (n - 1)
            w = weights[list(sorted_idx).index(ki)]
            # d/dh[j] corr[ki] = (1-h[j+k]) for valid j+k, minus h[j-k] for valid j-k
            if k >= 0:
                js = np.arange(n - k)
                grad[js] += w * (1.0 - h[js + k])
                js2 = np.arange(k, n)
                grad[js2] -= w * h[js2 - k]
            else:
                js = np.arange(-k, n)
                grad[js] += w * (1.0 - h[js + k])
                js2 = np.arange(n + k)
                grad[js2] -= w * h[js2 - k]
        
        # Step
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new.copy()
            patience = 0
            step = min(step * 1.01, 0.1)
        else:
            patience += 1
            step *= 0.8
            if step < 1e-5:
                break
    
    return best_h, best_c

**Key optimization for binary search**: Precompute the "flip delta" for each position.

For position i, define flip_delta[i] as the change in corr if h[i] changes by +1 (0→1) or -1 (1→0), ignoring the sum constraint.

For h[i] going 0→1:
flip_delta[i][ki] = (1-h[i+k]) - h[i-k] (with appropriate validity) where k = ki-(n-1)
(special case k=0: flip_delta[i][n-1] = -2*h[i] = 0 since h[i]=0)

For h[i] going 1→0:
flip_delta[i][ki] = -(1-h[i+k]) + h[i-k] (negated)

Then the total delta for swapping i(0→1) and j(1→0) is:
total_delta = flip_delta[i] + flip_delta_neg[j] + cross_correction(i,j)

where cross_correction only affects 2-3 positions (the diagonal and the i-j offset).

If I precompute flip_delta for all 250 positions, that's 250 * 499 = 124750 values. Then for each pair (i,j), I just add two precomputed vectors plus a tiny correction. The addition of two vectors of length 499 is ~5μs. So 15625 pairs * 5μs = 78ms per sweep. With 300 sweeps, that's 23 seconds. Much better!

```python
def precompute_flip_deltas(h, n):
    """Precompute the delta in corr for flipping each position."""
    # flip_delta[i] = change in corr if h[i] increases by 1
    # (regardless of whether it's currently 0 or 1)
    flip = np.zeros((n, 2*n-1))
    m_range = np.arange(-(n-1), n)
    for i in range(n):
        im = i + m_range
        imn = i - m_range
        delta = np.zeros(2*n-1)
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        flip[i] = delta
    return flip
```

Wait, but flip_delta depends on the current h, so I need to recompute it after each swap. That defeats the purpose.

Unless... I update it incrementally. When I swap h[i] and h[j]:
- flip_delta[i] changes (because h changed)
- flip_delta[j] changes
- All other flip_deltas change too (because they depend on h at other positions)

So incremental update is not straightforward.

**Alternative: Use the "effect on max" heuristic.**

Instead of computing the full delta, just look at the effect on the current maximum lag:
- corr[k*] is the maximum
- If I flip h[i] from 0 to 1, corr[k*] changes by: (1-h[i+k*]) - h[i-k*] (approximately)
- I want this to be negative (reduce the max)
- But I also need to check that no other lag becomes the new max

This gives a fast filtering step: only consider (i,j) pairs where the effect on k* is negative enough.

Let me implement a two-level approach:
1. Fast filter: for each zero i, compute the effect on the current max lag k*. Keep only those that reduce it.
2. Among the survivors, do the full check (compute full delta, check max).

This should be much faster in practice.

```python
def fast_binary_local(h, n, deadline, max_sweeps=500, seed=42):
    h = (h >= 0.5).astype(float)
    target = n // 2
    n_ones = int(h.sum())
    rng = np.random.RandomState(seed)
    if n_ones > target:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones, n_ones - target, replace=False)] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, target - n_ones, replace=False)] = 1
    
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1-h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    
    for sweep in range(max_sweeps):
        if time.time() > deadline: break
        improved = False
        k_star = np.argmax(corr)  # current max lag index
        k_actual = k_star - (n-1)  # actual lag value
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        # Fast filter: find zeros that reduce corr at k_star
        # Effect of h[i]:0→1 on corr[k_star]:
        # = (1-h[i+k_actual]) - h[i-k_actual] (with validity)
        i_val = zeros + k_actual
        effects_i = np.zeros(len(zeros))
        for idx, i in enumerate(zeros):
            val = 0.0
            if 0 <= i + k_actual < n:
                val += (1.0 - h[i + k_actual])
            if 0 <= i - k_actual < n:
                val -= h[i - k_actual]
            if k_actual == 0:
                val = -2.0 * h[i]  # = 0 since h[i]=0
            effects_i[idx] = val
        
        # Only consider zeros that reduce the max
        good_zeros = zeros[effects_i < -0.5 * cur_c / dx]
        if len(good_zeros) == 0:
            good_zeros = zeros  # fall back to all zeros
        
        # Try swaps with good zeros
        for i in good_zeros:
            if time.time() > deadline: break
            for j in ones:
                # Full delta computation
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                delta = np.zeros(2*n-1)
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n-1 + (j-i)
                if 0 <= k1 < 2*n-1: delta[k1] += 1.0
                k2 = n-1 + (i-j)
                if 0 <= k2 < 2*n-1: delta[k2] += 1.0
                
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + delta
                    cur_c = nc
                    improved = True
                    break
            if improved: break
        if not improved: break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c
```
