def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def precompute_c_matrix(h, n):
    """Compute c_i[m] for all i, where c_i[m] = (1-h[i+k]) - h[i-k] with k=m-(n-1)."""
    n_lags = 2 * n - 1
    C = np.zeros((n, n_lags))
    for i in range(n):
        lo1 = max(0, n - 1 - i)
        hi1 = min(n_lags - 1, n - 1 + i)
        if lo1 <= hi1:
            ks = np.arange(lo1, hi1 + 1) - (n - 1)
            idx = i + ks
            C[i, lo1:hi1+1] += (1.0 - h[idx])
        lo2 = max(0, i)
        hi2 = min(n_lags - 1, i + n - 1)
        if lo2 <= hi2:
            ks2 = np.arange(lo2, hi2 + 1) - (n - 1)
            idx2 = i - ks2
            C[i, lo2:hi2+1] -= h[idx2]
    return C

def find_delta(a, e, d_max):
    """Find delta in [0, d_max] minimizing max_k(a_k + delta * e_k)."""
    if d_max < 1e-15:
        return 0.0
    best_d = 0.0
    best_val = a.max()
    top = np.argsort(a)[-12:]
    cands = [0.0, d_max]
    for p in range(len(top)):
        for q in range(p + 1, len(top)):
            k1, k2 = top[p], top[q]
            de = e[k1] - e[k2]
            if abs(de) > 1e-14:
                dc = (a[k2] - a[k1]) / de
                if 0.0 <= dc <= d_max:
                    cands.append(dc)
    for dc in cands:
        val = np.max(a + dc * e)
        if val < best_val - 1e-15:
            best_val = val
            best_d = dc
    if 0 < best_d < d_max:
        a_new = a + best_d * e
        top2 = np.argsort(a_new)[-12:]
        for p in range(len(top2)):
            for q in range(p + 1, len(top2)):
                k1, k2 = top2[p], top2[q]
                de = e[k1] - e[k2]
                if abs(de) > 1e-14:
                    dc = (a[k2] - a[k1]) / de
                    if 0.0 <= dc <= d_max:
                        val = np.max(a + dc * e)
                        if val < best_val - 1e-15:
                            best_val = val
                            best_d = dc
    return best_d

def pair_descent(h, n, max_sweeps=80, deadline=None, rng=None):
    """Pair coordinate descent with incremental overlap tracking."""
    h = h.copy()
    n_lags = 2 * n - 1
    dx = 2.0 / n
    a = compute_all_lags(h, n)
    best_c = a.max()
    improved_sweeps = 0
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline - 5:
            break
        C = precompute_c_matrix(h, n)
        sweep_improved = False
        order = list(range(n))
        if rng is not None:
            rng.shuffle(order)
        for ii in range(n):
            i = order[ii]
            for j in range(i + 1, n):
                for sign in (1, -1):
                    if sign > 0:
                        d_max = min(1.0 - h[i], h[j])
                    else:
                        d_max = min(h[i], 1.0 - h[j])
                    if d_max < 1e-12:
                        continue
                    e = (C[i] - C[j]) * dx * sign
                    delta = find_delta(a, e, d_max)
                    if delta > 1e-10:
                        h[i] += sign * delta
                        h[j] -= sign * delta
                        a += delta * e
                        if a.max() < best_c - 1e-14:
                            best_c = a.max()
                            sweep_improved = True
        if sweep_improved:
            improved_sweeps = 0
        else:
            improved_sweeps += 1
            if improved_sweeps >= 2:
                break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def sa_local(h, n, deadline, rng, n_iters=50000):
    """Simulated annealing with pair moves."""
    h = h.copy()
    n_lags = 2 * n - 1
    dx = 2.0 / n
    a = compute_all_lags(h, n)
    best_c = a.max()
    best_h = h.copy()
    T = 0.005
    T_min = 1e-7
    cooling = 0.9997
    for it in range(n_iters):
        if time.time() > deadline - 3:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        sign = 1 if rng.rand() < 0.5 else -1
        if sign > 0:
            d_max = min(1.0 - h[i], h[j])
        else:
            d_max = min(h[i], 1.0 - h[j])
        if d_max < 1e-12:
            continue
        delta = rng.uniform(0.001, min(d_max, 0.05))
        C = precompute_c_matrix(h, n)
        e = (C[i] - C[j]) * dx * sign
        a_new = a + delta * e
        c_new = a_new.max()
        c_old = a.max()
        if c_new < c_old or rng.rand() < math.exp((c_old - c_new) / T):
            h[i] += sign * delta
            h[j] -= sign * delta
            a = a_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        T *= cooling
    return best_h, best_c

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, alpha=50000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
