import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def smooth_max_obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def slsqp(h, n, alpha, maxiter=5000, ftol=1e-16, top_idx=None):
    def obj(x):
        return smooth_max_obj(x, n, alpha, top_idx)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        cmax = c.max()
        if cmax < best_c:
            best_c = cmax
            h_best = h.copy()
        
        # Get active lags (top 5 or within 2% of max)
        thresh = cmax * 0.98
        active = np.where(c >= thresh)[0]
        if len(active) < 2:
            active = np.argsort(c)[-5:]
        
        # Average gradient of active lags
        g = np.zeros(n)
        for idx in active:
            lag = idx - (n - 1)
            g += _lag_grad(h, n, lag)
        g /= len(active)

            if c_try < cmax - 1e-12:

    return best_h, best_c

Let me focus on making the local search much more effective. The key insight is:

At the optimum, the top m lags should all be equal to C₅. The subgradient of max(c_k) is any convex combination of the gradients of the active lags. To decrease the maximum, we need to find a direction d such that:
- g_k · d < 0 for all active lags k
- d is in the feasible direction (maintains sum(h) = n/2 and 0 ≤ h ≤ 1)

This is a linear programming problem! We want to find d that minimizes the maximum rate of increase of the active lags:
min_t max_k (g_k · d)
s.t. sum(d) = 0
||d|| ≤ 1 (or some other normalization)
0 ≤ h + eps*d ≤ 1 (for small eps)

If the minimum t < 0, we have a descent direction.

This is a "Nemirovski-Wolfowitz" type approach. Let me implement it:

```python
def find_descent_direction(h, n, active_idx, deadline):
    """Find a descent direction using LP."""
    import cvxpy as cp
    d = cp.Variable(n)
    t = cp.Variable()
    cons = [cp.sum(d) == 0, cp.norm(d, 2) <= 1]
    for idx in active_idx:
        lag = idx - (n - 1)
        g = _lag_grad(h, n, lag)
        cons.append(cp.dot(g, d) <= t)
    # Also maintain 0 <= h + d <= 1 approximately
    cons.append(h + d >= -0.01)
    cons.append(h + d <= 1.01)
    prob = cp.Problem(cp.Minimize(t), cons)
    prob.solve(solver=cp.SCS)
    if prob.status == 'optimal':
        return d.value, t.value
    return None, 0

Let me try a different approach: **Frank-Wolfe (conditional gradient) method**.

The Frank-Wolfe method works as follows:
1. At current point h, compute the subgradient g of the objective
2. Solve: min_{x in feasible set} g · x
   The feasible set is {x : sum(x) = n/2, 0 ≤ x ≤ 1}
   The solution is: x_i = 1 if g_i < lambda, x_i = 0 if g_i > lambda, where lambda is chosen so sum(x) = n/2.
   (This is the solution of a simple water-filling problem.)
3. Take a step: h_new = (1-t)*h + t*x, where t is chosen by line search

The advantage of Frank-Wolfe is that the "linear minimization oracle" is trivial for our feasible set, and no projection is needed.

Let me implement this:

```python
def frank_wolfe(h, n, deadline, max_iters=10000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = compute_all_lags(h, n)
        cmax = c.max()
        if cmax < best_c:
            best_c = cmax
            best_h = h.copy()
        
        # Get active lags
        thresh = cmax - 0.02 * cmax
        active = np.where(c >= thresh)[0]
        if len(active) < 2:
            active = np.argsort(c)[-5:]
        
        # Subgradient: average of active lag gradients
        g = np.zeros(n)
        for idx in active:
            lag = idx - (n - 1)
            g += _lag_grad(h, n, lag)
        g /= len(active)
        
        # Frank-Wolfe: find x in feasible set minimizing g·x
        # x_i = 1 if g_i < lambda, 0 if g_i > lambda
        # Choose lambda so sum(x) = n/2
        sorted_g = np.sort(g)
        # We need exactly n/2 elements to be 1 (the smallest n/2 values of g)
        # So lambda is between sorted_g[n/2-1] and sorted_g[n/2]
        n_half = n // 2
        lambda_val = (sorted_g[n_half - 1] + sorted_g[n_half]) / 2
        x = (g < lambda_val).astype(float)
        # Fix sum
        diff = n_half - x.sum()
        if diff > 0:
            # Need more 1s: set the next smallest to 1
            idx_sorted = np.argsort(g)
            for i in idx_sorted:
                if x[i] == 0:
                    x[i] = 1
                    diff -= 1
                    if diff == 0:
                        break
        elif diff < 0:
            idx_sorted = np.argsort(g)[::-1]
            for i in idx_sorted:
                if x[i] == 1:
                    x[i] = 0
                    diff += 1
                    if diff == 0:
                        break
        
        # Line search
        best_t = 0
        best_val = cmax
        for t in [2.0/(it+2), 1.0/(it+1), 0.5, 0.25, 0.1]:
            h_try = (1-t)*h + t*x
            h_try = np.clip(h_try, 0, 1)
            # Project to maintain sum
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_val:
                best_val = c_try
                best_t = t
        
        if best_t == 0:
            break
        h = (1-best_t)*h + best_t*x
        h = project_feasible(h, n)
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def smooth_max_obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def slsqp(h, n, alpha, maxiter=5000, ftol=1e-16, top_idx=None):
    def obj(x):
        return smooth_max_obj(x, n, alpha, top_idx)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def frank_wolfe(h, n, deadline, max_iters=5000, top_m=8):
    """Frank-Wolfe / conditional gradient method."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        cmax = c.max()
        if cmax < best_c - 1e-14:
            best_c = cmax
            best_h = h.copy()
        
        # Active set
        thresh = cmax - 0.05 * cmax
        active = np.where(c >= thresh)[0]
        if len(active) > top_m:
            active = np.argsort(c)[-top_m:]
        elif len(active) < 2:
            active = np.argsort(c)[-5:]
        
        # Subgradient
        g = np.zeros(n)
        for idx in active:
            lag = idx - (n - 1)
            g += _lag_grad(h, n, lag)
        g /= len(active)
        
        # Linear oracle: x = argmin g·x s.t. sum(x)=n/2, 0<=x<=1
        # Solution: x_i = 1 for the n/2 smallest g_i, 0 otherwise
        n_half = n // 2
        idx_sorted = np.argsort(g)
        x = np.zeros(n)
        x[idx_sorted[:n_half]] = 1.0
        
        # Line search
        best_t = 0.0
        best_val = cmax
        gamma = 2.0 / (it + 2)
        for t in [gamma, gamma * 0.5, 0.5, 0.25, 0.1, 0.05]:
            h_try = (1 - t) * h + t * x
            h_try = np.clip(h_try, 0, 1)
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_val:
                best_val = c_try
                best_t = t
            if c_try < cmax * 0.999:
                break
        
        if best_t <= 0:
            break
        h = (1 - best_t) * h + best_t * x
        h = project_feasible(h, n)
    
    return best_h, best_c
