import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def obj_penalty(h, dx, beta, target_sum, gamma):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx + gamma * (h.sum() - target_sum) ** 2


def obj_smooth_only(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def sum_constraint_fun(h, target_sum):
    return h.sum() - target_sum


def sum_constraint_jac(h, n):
    return np.ones(n)


def time_remaining(t0, budget_s):
    return budget_s - (time.time() - t0)


def optimize_lbfgsb(h0, n, dx, target_sum, beta, gamma, maxiter):
    bounds = Bounds(np.zeros(n), np.ones(n))
    res = minimize(obj_penalty, h0, args=(dx, beta, target_sum, gamma),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    h_new = project_feasible(res.x, target_sum)
    return h_new


def optimize_slqp(h0, n, dx, target_sum, beta, maxiter):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint_fun, 'jac': sum_constraint_jac,
             'args': (target_sum, n)}]
    res = minimize(obj_smooth_only, h0, args=(dx, beta),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    h_new = project_feasible(res.x, target_sum)
    return h_new


def generate_structural_h(n, mode, rng):
    if mode == 0:
        h = np.full(n, 0.5)
    elif mode == 1:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
    elif mode == 2:
        h = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * np.arange(n) / n)
    elif mode == 3:
        h = (np.arange(n) % 3 == 0).astype(float)
    elif mode == 4:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 5 * np.arange(n) / n) + 0.1 * np.sin(2 * np.pi * 7 * np.arange(n) / n)
    elif mode == 5:
        h = np.clip(0.5 + rng.randn(n) * 0.3, 0, 1)
    else:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 4 * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    return project_feasible(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with increasing smoothness (fast local optimization)
    for beta in [30, 100, 500, 2000, 8000]:
        if time_remaining(t0, budget_s) < 15:
            break
        for gamma in [500, 5000]:
            if time_remaining(t0, budget_s) < 10:
                break
            try:
                h_new = optimize_lbfgsb(best_h, n, dx, target_sum, beta, gamma, 3000)
                c5 = compute_c5(h_new)

                    best_h = h_new
                    best_c5 = c5
            except Exception:
                pass

    # Phase 2: SLSQP with constraint (more accurate)
    for beta in [500, 2000, 10000, 40000]:

        try:
            h_new = optimize_slqp(best_h, n, dx, target_sum, beta, 2000)

                best_c5 = c5
        except Exception:
            pass

        n_t = n_try
        dx_t = 2.0 / n_t

                    c5_t = c5
                    improved = True
            except Exception:
                pass
        for beta in [2000, 10000]:
            if time_remaining(t0, budget_s) < 15:
                break
            try:
                h_new = optimize_slqp(h_t, n_t, dx_t, ts_t, beta, 1500)
                c5 = compute_c5(h_new)
                if c5 < c5_t:
                    h_t = h_new
                    c5_t = c5
                    improved = True
            except Exception:
                pass
        if c5_t < best_c5:
            best_h = h_t
            best_c5 = c5_t

    # Update best resolution
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    ts_best = n_best / 2.0

    # Phase 4: Random restarts with diverse structures
    for mode in range(6):
        if time_remaining(t0, budget_s) < 30:
            break
        h_r = generate_structural_h(n_best, mode, rng)
        h_r = project_feasible(h_r + rng.randn(n_best) * 0.02, ts_best)
        for beta in [200, 2000]:
            if time_remaining(t0, budget_s) < 15:
                break
            try:
                h_new = optimize_lbfgsb(h_r, n_best, dx_best, ts_best, beta, 2000, 2000)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
            except Exception:
                pass

    # Phase 5: Perturbation + re-optimize (escape local optima)
    for perturb_scale in [0.05, 0.03, 0.01]:
        for attempt in range(3):
            if time_remaining(t0, budget_s) < 20:
                break
            h_p = best_h + rng.randn(n_best) * perturb_scale
            h_p = project_feasible(h_p, ts_best)
            try:
                h_new = optimize_lbfgsb(h_p, n_best, dx_best, ts_best, 2000, 2000, 3000)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
            except Exception:
                pass

    # Phase 6: Final SLSQP polish with very high beta
    for beta in [20000, 80000, 200000]:
        if time_remaining(t0, budget_s) < 15:
            break
        try:
            h_new = optimize_slqp(best_h, n_best, dx_best, ts_best, beta, 5000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
        except Exception:
            pass

    # Phase 7: One more L-BFGS-B sweep at the end
    for beta in [5000, 20000]:
        if time_remaining(t0, budget_s) < 10:

        try:
            h_new = optimize_lbfgsb(best_h, n_best, dx_best, ts_best, beta, 5000, 5000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
