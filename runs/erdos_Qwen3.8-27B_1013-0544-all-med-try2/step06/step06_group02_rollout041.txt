            if tot < 1e-14:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for l in range(2 * n - 1):
        if w[l] < 1e-12:
            continue
        k = l - (n - 1)
        wm = w[l] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - h[k:])
                grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - h[:nv])
                grad[:nv] += wm * (-h[kk:])
    return grad

def penalty_obj(h, n, alpha, mu):
    v = smooth_max_obj(h, n, alpha)
    return v + mu * (h.sum() - n / 2.0) ** 2

def penalty_grad(h, n, alpha, mu):
    g = smooth_max_grad(h, n, alpha)
    g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, alphas=None, iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

    for a in alphas:

                                  'maxls': 30})

            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def gradient_line_search(h, n, deadline, alpha=1000000):

    best_h = h.copy()

                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; best_h = h.copy(); improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return best_h, best_c

def perturbation(h, n, rng, strength=0.03):
    h_new = h.copy()
    # Low-frequency sinusoidal perturbation
    x = np.arange(n) / n * 2.0
    for _ in range(rng.randint(1, 4)):
        freq = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(strength * 0.5, strength * 1.5)
        h_new += amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    # Small random noise
    h_new += rng.uniform(-strength * 0.3, strength * 0.3, n)
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 300
        h0 = np.full(n, 0.5)
    
    h = project_feasible(h0, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: L-BFGS-B from initial
    t1 = min(t0 + 300, deadline - 400)
    h, best_c = lbfgs_phase(h, n, t1, iters=300)
    best_h = h.copy()
    
    # Phase 2: Multi-start with perturbations
    for s in range(6):
        if time.time() > deadline - 400:
            break
        h_p = perturbation(best_h, n, rng, strength=0.02 + s * 0.01)
        alloc = min(60, deadline - time.time() - 350)
        if alloc < 15:
            break
        h_new, c_new = lbfgs_phase(h_p, n, time.time() + alloc, iters=150)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    
    # Phase 3: Gradient line search
    if time.time() < deadline - 200:
        t3 = min(time.time() + 100, deadline - 200)
        h_new, c_new = gradient_line_search(best_h, n, t3, alpha=5000000)
