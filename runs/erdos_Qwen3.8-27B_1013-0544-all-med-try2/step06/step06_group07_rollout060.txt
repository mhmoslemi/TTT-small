For each lag m (corresponding to L = m - (n-1)):
- The old value is corr[m] = sum_i h[i](1-h[i+L])
- The new value is sum_i h_new[i](1-h_new[i+L])
- The difference is the sum over affected i of the change in h_new[i](1-h_new[i+L])

The affected i values are: j, k (where h changes), and also i = j-L, k-L (where h_new[i+L] changes).

But we need to be careful not to double-count if, say, j-L = k.

Let me use a set to track which i values are affected:

```python
def compute_delta_f(h, j, k, L, n):
    """Return (a, b, c) such that Δf_L = a*δ² + b*δ + c*0 (constant is 0 since at δ=0, Δf=0)"""
    # Actually Δf_L(δ) = f_L_new(δ) - f_L_old = a*δ² + b*δ (no constant since f_L_new(0) = f_L_old)
    # Wait, f_L_new(0) should equal f_L_old, so the constant is 0.
    
    a_coeff = 0.0
    b_coeff = 0.0
    
    # Affected i values (where h_new differs from h_old, either directly or via i+L)
    affected = set()
    # Direct: i where h_new[i] ≠ h[i], i.e., i=j or i=k
    affected.add(j)
    affected.add(k)
    # Via i+L: i where h_new[i+L] ≠ h[i+L], i.e., i+L=j or i+L=k, so i=j-L or i=k-L
    if 0 <= j - L < n:
        affected.add(j - L)
    if 0 <= k - L < n:
        affected.add(k - L)
    
    # Remove out-of-range
    affected = {i for i in affected if 0 <= i < n}
    
    # For each affected i, compute the change in h_new[i] * (1 - h_new[i+L])
    for i in affected:
        if not (0 <= i + L < n):
            continue  # This term doesn't contribute to f_L
        
        h_i_old = h[i]
        h_ipL_old = h[i + L]
        
        # New values
        h_i_new = h_i_old
        h_ipL_new = h_ipL_old
        if i == j:
            h_i_new = h_i_old  # will add δ
        elif i == k:
            h_i_new = h_i_old  # will subtract δ
        
        # Change in h_i: δ if i==j, -δ if i==k, 0 otherwise
        delta_hi = (1.0 if i == j else (-1.0 if i == k else 0.0))
        # Change in h_{i+L}: δ if i+L==j, -δ if i+L==k, 0 otherwise
        delta_hipL = (1.0 if (i + L) == j else (-1.0 if (i + L) == k else 0.0))
        
        # Old term: h_i_old * (1 - h_ipL_old)
        # New term: (h_i_old + delta_hi*δ) * (1 - h_ipL_old - delta_hipL*δ)
        #          = h_i_old(1-h_ipL_old) + delta_hi*δ*(1-h_ipL_old) + h_i_old*(-delta_hipL*δ) + delta_hi*delta_hipL*δ²
        
        a_coeff += delta_hi * delta_hipL
        b_coeff += delta_hi * (1.0 - h_ipL_old) + (-h_i_old) * delta_hipL
    
    return a_coeff, b_coeff
```

Wait, I think there's an issue. If i is affected both directly (i=j) and via i+L (i+L=c where c is j or k), the term h_new[i](1-h_new[i+L]) has both the first and second factor changed. My formula above handles this correctly because I'm computing the full change.

1. For each pair (j, k), compute A[m], B[m] for all m (lags)

3. Find δ* = argmin_{δ∈[δ_min, δ_max]} max_L (A[L]*δ² + B[L]*δ + corr[L])
4. Update h[j] += δ*, h[k] -= δ*

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def pair_coefficients(h, j, k, n):
    """Compute A, B such that f_L(δ) = A[L]*δ² + B[L]*δ + corr[L] for all L."""
    n_lags = 2 * n - 1
    A = np.zeros(n_lags, dtype=float)
    B = np.zeros(n_lags, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for m in range(n_lags):
        L = m - (n - 1)
        a_c = 0.0
        b_c = 0.0
        # Affected i values
        affected = {j, k}
        if 0 <= j - L < n:
            affected.add(j - L)
        if 0 <= k - L < n:
            affected.add(k - L)
        for i in affected:
            if not (0 <= i < n and 0 <= i + L < n):
                continue
            delta_hi = 1.0 if i == j else (-1.0 if i == k else 0.0)
            delta_hipL = 1.0 if (i + L) == j else (-1.0 if (i + L) == k else 0.0)
            if delta_hi == 0.0 and delta_hipL == 0.0:
                continue
            a_c += delta_hi * delta_hipL
            h_i = h[i]
            h_ipL = h[i + L]
            b_c += delta_hi * (1.0 - h_ipL) - h_i * delta_hipL
        A[m] = a_c
        B[m] = b_c
    return A, B, corr

def pair_sweep(h, n, n_grid=200, n_refine_grid=200):
    """One full sweep: for each pair (j,k), find best δ and apply."""
    target_sum = n / 2.0
    improved = 0.0
    n_lags = 2 * n - 1
    for j in range(n):
        for k in range(j + 1, n):
            d_min = max(-h[j], h[k] - 1.0)
            d_max = min(1.0 - h[j], h[k])
            if d_min >= d_max - 1e-15:
                continue
            A, B, corr = pair_coefficients(h, j, k, n)
            deltas = np.linspace(d_min, d_max, n_grid)
            # f_grid[g, m] = A[m]*δ[g]² + B[m]*δ[g] + corr[m]
            d2 = deltas[:, None] ** 2
            d1 = deltas[:, None]
            f_grid = A[None, :] * d2 + B[None, :] * d1 + corr[None, :]
            obj_vals = np.max(f_grid, axis=1)
            best_idx = np.argmin(obj_vals)
            best_obj = obj_vals[best_idx]
            cur_obj = np.max(corr)
            if best_obj < cur_obj - 1e-14:
                # Refine around best_idx
                lo = max(d_min, deltas[best_idx] - (d_max - d_min) / n_grid)
                hi = min(d_max, deltas[best_idx] + (d_max - d_min) / n_grid)
                deltas_r = np.linspace(lo, hi, n_refine_grid)
                d2r = deltas_r[:, None] ** 2
                d1r = deltas_r[:, None]
                f_grid_r = A[None, :] * d2r + B[None, :] * d1r + corr[None, :]
                obj_r = np.max(f_grid_r, axis=1)
                best_idx_r = np.argmin(obj_r)
                best_delta = deltas_r[best_idx_r]
                h[j] += best_delta
                h[k] -= best_delta
                improved += (cur_obj - obj_r[best_idx_r])
    return improved

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP continuation
    for beta in [10, 50, 200, 1000, 5000]:
        if tl() < 80:
            break
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for trial in range(2):
            if tl() < 30:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += np.random.randn(n) * 0.015 / trial
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 2: Pair coordinate descent (main improvement)
    if tl() > 120:
        h_cd = best_h.copy()
        n_pairs = n * (n - 1) // 2
        est_sweep_time = 0.05 * n_pairs  # rough estimate
        max_sweeps = max(5, int((tl() - 100) / max(est_sweep_time, 1.0)))
        for sweep in range(min(max_sweeps, 200)):
            if tl() < 80:
                break
            imp = pair_sweep(h_cd, n, n_grid=300, n_refine_grid=300)
            c5_now = compute_c5(h_cd)
            if c5_now < best_c5:
                best_h = h_cd.copy()
                best_c5 = c5_now
            if imp < 1e-12:
                break

    # Phase 3: SLSQP polish from CD solution
    if tl() > 80:
        for beta in [5000, 20000, 80000]:
            if tl() < 30:
                break
            cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
            for trial in range(3):
                if tl() < 15:
                    break
                x0 = best_h.copy()
                if trial > 0:
                    x0 += np.random.randn(n) * 0.003 / trial
                    x0 = project_feasible(x0, target_sum)
                try:
                    res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 5000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 4: Another pair descent round
    if tl() > 60:
        h_cd2 = best_h.copy()
        for sweep in range(100):
            if tl() < 40:
                break
            imp = pair_sweep(h_cd2, n, n_grid=400, n_refine_grid=400)
            c5_now = compute_c5(h_cd2)
            if c5_now < best_c5:
                best_h = h_cd2.copy()
                best_c5 = c5_now
            if imp < 1e-12:
                break

    # Phase 5: Try other resolutions
    if tl() > 60:
        for n_try in [n, n + 10, n - 10, n + 20, n - 20]:
            if n_try < 40 or tl() < 35:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_h = h_t.copy()
                best_c5 = c5_t
                n = n_try
                dx = dx_t
                target_sum = ts_t
                bounds = Bounds(np.zeros(n_try), np.ones(n_try))
            # Quick SLSQP
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t, n_try)}]
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            for beta in [500, 5000]:
                if tl() < 15:
                    break
                try:
                    res = minimize(obj_smooth, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 2000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        n = n_try
                        dx = dx_t
                        target_sum = ts_t
                        bounds = bnds_t
                        h_t = h_new.copy()
                except Exception:
                    pass
            # Quick pair descent on this resolution
            if compute_c5(h_t) <= best_c5 + 1e-6 and tl() > 25:
                for sweep in range(30):
                    if tl() < 20:
                        break
                    imp = pair_sweep(h_t, n_try, n_grid=200, n_refine_grid=200)
                    if imp < 1e-12:
                        break
                c5_t2 = compute_c5(h_t)
                if c5_t2 < best_c5:
                    best_h = h_t.copy()
                    best_c5 = c5_t2
                    n = n_try
                    dx = 2.0 / n_try
                    target_sum = n_try / 2.0
                    bounds = Bounds(np.zeros(n_try), np.ones(n_try))

    # Phase 6: Final pair descent polish
    if tl() > 20:
        h_fin = best_h.copy()
        for sweep in range(50):
            if tl() < 10:
                break
            imp = pair_sweep(h_fin, len(h_fin), n_grid=500, n_refine_grid=500)
            c5_f = compute_c5(h_fin)
            if c5_f < best_c5:
                best_h = h_fin.copy()
                best_c5 = c5_f
            if imp < 1e-12:
                break

    return best_h, best_c5, len(best_h)
```
