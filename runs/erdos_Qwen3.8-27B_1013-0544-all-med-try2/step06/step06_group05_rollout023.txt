def compute_smoothed_gradient(h, n, alpha):
    dx = 2.0 / n
    # Compute correlations (only unique lags: 0, 1, ..., n-1)
    c_lags = np.zeros(n)
    c_lags[0] = np.sum(h * (1-h)) * dx
    for l in range(1, n):
        c_lags[l] = np.sum(h[l:] * (1-h[:-l])) * dx
    
    # The full correlation has 2n-1 values, symmetric
    # Full c at index k: for k = n-1, it's c_0
    # for k = n-1+l (l=1..n-1), it's c_l
    # for k = n-1-l (l=1..n-1), it's c_l
    # So weights: w[0] for lag 0, w[l] for each of the two positions at lag l
    
    # Softmax over the full 2n-1 values
    full_c = np.zeros(2*n-1)
    full_c[n-1] = c_lags[0]
    for l in range(1, n):
        full_c[n-1+l] = c_lags[l]
        full_c[n-1-l] = c_lags[l]
    
    shifted = alpha * (full_c - full_c.max())
    w = np.exp(shifted)
    w /= w.sum()
    
    # Gradient
    grad = np.zeros(n)
    # Lag 0 contribution
    grad += w[n-1] * (1 - 2*h)
    
    # Lag l contributions
    for l in range(1, n):
        wl = w[n-1+l] + w[n-1-l]  # = 2*w[n-1+l] by symmetry
        if wl < 1e-15:
            continue
        # (1-h[j-l]) for j >= l
        grad[l:] += wl * (1 - h[:n-l])
        # -h[j+l] for j < n-l

def pgd_optimize(h, n, deadline, alpha_start=100, alpha_end=500000, n_iters=5000):
    """Projected gradient descent with log-sum-exp smoothing."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        # Adaptive alpha
        progress = it / max(1, n_iters)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        full_c = np.correlate(h, 1 - h, mode='full') * dx
        
        # Softmax weights
        m = full_c.max()
        shifted = alpha * (full_c - m)
        smx = shifted.max()
        if smx > 500:
            w = np.exp(shifted - smx)
        else:
            w = np.exp(shifted)
        w /= w.sum()
        
        # Compute gradient
        # grad[j] = sum_k w[k] * dc_k/dh[j]
        # For lag l (where k = n-1+l for l>=0, k = n-1+l for l<0):
        # dc_l/dh[j] = (1-h[j-|l|]) * [j >= |l|] - h[j+|l|] * [j+|l| < n]
        # (for l != 0)
        # dc_0/dh[j] = 1 - 2*h[j]
        
        grad = np.zeros(n)
        
        # Lag 0
        grad += w[n-1] * (1 - 2*h)
        
        # Lags 1 to n-1
        for l in range(1, n):
            wl = w[n-1+l] + w[n-1-l]
            if wl < 1e-12:
                continue
            grad[l:] += wl * (1 - h[:n-l])
            grad[:n-l] -= wl * h[l:]

        step = 0.01
        f_curr = smooth_objective(h, n, alpha)
        for bs in range(30):
            h_new = project_feasible(h - step * grad, n)
            f_new = smooth_objective(h_new, n, alpha)
            if f_new < f_curr - 1e-4 * step * np.sum(grad**2):
                break
            step *= 0.5
        
        h = h_new
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def project_feasible(h, n):

        return np.sum(np.clip(h - tau, 0.0, 1.0)) - target

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    def f(tau):
        return np.sum(np.clip(h - tau, 0.0, 1.0)) - target
    lo, hi = -1.0, 1.0
    if f(lo) < 0:
        return np.clip(h, 0, 1)
    if f(hi) > 0:
        return np.clip(h, 0, 1)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sc(x, n):
    return float(np.sum(x) - n * 0.5)

def _scj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:

                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)  # momentum
    beta = 0.9
    lr = 0.05
    n_max = 200000

    dx = 2.0 / n
    
    for it in range(n_max):
        if time.time() > deadline:
            break
        p = min(1.0, it / n_max)
        alpha = alpha_lo * (alpha_hi / alpha_lo) ** p

        fc = np.correlate(h, 1 - h, mode='full') * dx
        m = fc.max()
        sh = alpha * (fc - m)
        smx = sh.max()
        w = np.exp(sh - smx) if smx > 500 else np.exp(sh)
        w /= w.sum()
        
        grad = np.zeros(n)
        grad += w[n-1] * (1.0 - 2.0 * h)
        for l in range(1, n):
            wl = w[n-1+l] + w[n-1-l]
            if wl < 1e-14:

            grad[l:] += wl * (1.0 - h[:n-l])
            grad[:n-l] -= wl * h[l:]
        
        # Momentum step
        v = beta * v - lr * grad
        h_new = project_feasible(h + v, n)
        
        # Check if we're making progress
        f_new = smooth_obj(h_new, n, alpha)
