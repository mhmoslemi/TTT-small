c[m] = sum_i h[i]*(1-h[i+m-(n-1)])

∂c[m]/∂h[p] = (1-h[p+m-(n-1)]) [from the i=p term in the first factor]

So the subgradient of f(h) = max_m c[m] at h is the gradient of c[m*] where m* = argmax_m c[m].

And the projected gradient descent step would be:
h_new = project(h - lr * grad, [0,1] ∩ {sum = n/2})

def subgradient_descent(h, n, lr=0.01, max_iter=100000, deadline=None):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        if it % 10000 == 0:
            lr = max(0.0001, lr * 0.999)
        
        c = np.correlate(h, 1-h, mode='full') * dx
        m_star = int(np.argmax(c))
        c5 = c[m_star]
        
        # Gradient of c[m_star] with respect to h
        # ∂c[m]/∂h[p] = (1-h[p+m-(n-1)]) - h[p+(n-1)-m]  (with boundary handling)
        grad = np.zeros(n)
        m = m_star
        for p in range(n):
            idx_a = p + m - (n-1)  # index for first factor
            idx_b = p + (n-1) - m  # index for second factor
            g = 0.0
            if 0 <= idx_a < n:
                g += (1 - h[idx_a])
            if 0 <= idx_b < n:
                g -= h[idx_b]
            grad[p] = g
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_correlation(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x):
    return float(np.sum(x) - len(x) / 2.0)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted - m * 0)))) / alpha

def _smooth_obj2(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con}]
    try:
        res = minimize(_smooth_obj2, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, max_passes=10000):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for d in deltas:
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 + 1e-14 and h[j] - d >= -1e-14:
                        h[i] = min(1.0, h[i] + d)
                        h[j] = max(0.0, h[j] - d)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-14 and h[j] + d <= 1.0 + 1e-14:
                        h[i] = max(0.0, h[i] - d)
                        h[j] = min(1.0, h[j] + d)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def subgradient_momentum(h, n, deadline, lr0=0.02, momentum=0.9):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    dx = 2.0 / n
    lr = lr0
    for it in range(500000):
        if time.time() > deadline:
            break
        if it % 5000 == 0:
            lr = max(0.0005, lr * 0.95)
        c = np.correlate(h, 1 - h, mode='full') * dx
        m_star = int(np.argmax(c))
        m = m_star
        grad = np.zeros(n)
        idx_a = np.arange(n) + m - (n - 1)
        valid_a = (idx_a >= 0) & (idx_a < n)
        grad[valid_a] += (1 - h[idx_a[valid_a]])
        idx_b = np.arange(n) + (n - 1) - m
        valid_b = (idx_b >= 0) & (idx_b < n)
        grad[valid_b] -= h[idx_b[valid_b]]
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c:
            best_c = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def random_01_init(n, rng, n_restarts=20):
    best_h = None
    best_c = float('inf')
    for _ in range(n_restarts):
        perm = rng.permutation(n)
        h = np.zeros(n)
        h[perm[:n // 2]] = 1.0
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # SLSQP with multiple alphas
    for a in [200, 1000, 5000, 20000, 100000]:
        if time.time() > deadline - 2:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()

    # Subgradient momentum
    if time.time() < deadline - 5:
        ht, ct = subgradient_momentum(best_h, n, deadline - 5, lr0=0.03, momentum=0.92)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()

    # Pair search
    if time.time() < deadline - 1:
        if n <= 150:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.05, 0.02, 0.01, 0.005, 0.002, 0.001], max_passes=20000)
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.02, 0.01, 0.005], max_passes=5000)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    best_h = None
    best_c = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = None
        n_parent = 250

    # Phase 1: Optimize from parent at its native resolution
    if h_parent is not None and time.time() < deadline - 60:
        alloc = min(120, (deadline - t0) * 0.15)
        h_opt, c_opt = optimize_at_n(h_parent, n_parent, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    # Phase 2: Try smaller resolutions (interpolate from best or parent)
    source = h_parent if h_parent is not None else np.full(250, 0.5)
    x_src = np.linspace(0, 2, len(source))

    for n_try in [100, 150, 200, 80, 50]:
        if time.time() > deadline - 40:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_src, source)
        h_interp = project_feasible(h_interp, n_try)
        remaining = deadline - time.time() - 20
        alloc = min(remaining * 0.3, 90)
        if alloc < 15:
            break
        h_opt, c_opt = optimize_at_n(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Random 0-1 restarts at small n
    if time.time() < deadline - 30:
        for n_try in [100, 80, 60]:
            if time.time() > deadline - 25:
                break
            h_rand, c_rand = random_01_init(n_try, rng, n_restarts=50)
            remaining = deadline - time.time() - 10
            alloc = min(remaining * 0.25, 60)
            if alloc < 10:
                break
            h_opt, c_opt = optimize_at_n(h_rand, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Final polish on best solution
    if best_h is not None and time.time() < deadline - 2:
        n = len(best_h)
        if n <= 120:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.005, 0.002, 0.001, 0.0005, 0.0002], max_passes=50000)
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.005, 0.002, 0.001], max_passes=10000)
        best_h = project_feasible(best_h, n)

    if best_h is None:
