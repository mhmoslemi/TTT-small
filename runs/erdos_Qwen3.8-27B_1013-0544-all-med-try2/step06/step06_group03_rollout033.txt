def compute_gradient_fast(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)

        w = np.exp(sh - sh.max())
    else:
        w = np.exp(sh)
    w = w / w.sum()
    
    k_vals = np.arange(2*n-1) - n + 1  # shape (2n-1,)
    
    # For each j, we need:
    # sum_m w[m] * (1-h[j+k_m]) * I(0≤j+k_m<n) - sum_m w[m] * h[j-k_m] * I(0≤j-k_m<n)
    
    # Vectorized:
    # J = j indices (0 to n-1), K = k values
    # We want: (J+K) in [0,n) for the first term, (J-K) in [0,n) for the second
    
    # Create a matrix: M[j, m] = I(0≤j+k_m<n)
    J = np.arange(n)[:, None]  # shape (n, 1)
    K = k_vals[None, :]  # shape (1, 2n-1)
    
    mask1 = (J + K >= 0) & (J + K < n)  # shape (n, 2n-1)
    mask2 = (J - K >= 0) & (J - K < n)  # shape (n, 2n-1)
    
    # First term: sum_m w[m] * (1-h[j+k_m]) * mask1[j,m]
    # h[j+k_m] where j+k_m is valid:
    idx1 = np.where(mask1, J + K, 0)  # shape (n, 2n-1)
    term1 = (1.0 - h[idx1]) * mask1  # shape (n, 2n-1)
    
    # Second term: sum_m w[m] * h[j-k_m] * mask2[j,m]
    idx2 = np.where(mask2, J - K, 0)  # shape (n, 2n-1)
    term2 = h[idx2] * mask2  # shape (n, 2n-1)
    
    g = (term1 @ w - term2 @ w) * dx  # shape (n,)
    return g, c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, n, alpha):
    """Vectorized gradient of soft-max objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    if sh.max() > 500:
        w = np.exp(sh - sh.max())
    else:
        w = np.exp(sh)
    w /= w.sum()
    
    # k_m = m - n + 1 for m = 0, ..., 2n-2
    k_vals = np.arange(2 * n - 1) - n + 1  # shape (2n-1,)
    J = np.arange(n)[:, None]  # (n, 1)
    K = k_vals[None, :]  # (1, 2n-1)
    
    # mask1: 0 <= J+K < n
    mask1 = (J + K >= 0) & (J + K < n)
    # mask2: 0 <= J-K < n
    mask2 = (J - K >= 0) & (J - K < n)
    
    idx1 = np.where(mask1, J + K, 0)
    idx2 = np.where(mask2, J - K, 0)
    
    # A[j] = sum_m w[m] * (1-h[j+k_m]) * mask1[j,m]
    # B[j] = sum_m w[m] * h[j-k_m] * mask2[j,m]
    A = ((1.0 - h[idx1]) * mask1).dot(w)  # (n,)
    B = (h[idx2] * mask2).dot(w)  # (n,)
    
    g = dx * (A - B)
    return g, c

def projected_grad_descent(h, n, alpha, deadline, max_iter=2000):
    """Projected gradient descent with backtracking."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    lr = 0.05
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g, c = compute_grad(h, n, alpha)
        
        # Backtracking line search
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr = 0.05
            # Try a few random perturbation steps
            break
    
    return h, best_c
