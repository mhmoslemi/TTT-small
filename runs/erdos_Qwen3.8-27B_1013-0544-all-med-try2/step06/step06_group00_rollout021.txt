    dx = 2.0 / n
    h = h.copy()
    corr = compute_corr(h, n)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    
    m_range = np.arange(-(n-1), n)
    
    for it in range(iterations):
        if time.time() > deadline:
            break
        # Find top K lags
        # Use a smoothed max
        top_idx = np.argsort(corr)[::-1][:K]
        # Weight by softmax
        top_vals = corr[top_idx]
        w = np.exp((top_vals - top_vals.max()) * 10)
        w /= w.sum()
        
        # Compute gradient as weighted sum
        grad = np.zeros(n)
        for w_k, k in zip(w, top_idx):
            m = k - (n-1)
            # d(full[k])/d(h[j]) = [0≤j-m<n]*(1-h[j-m]) - [0≤j+m<n]*h[j+m]
            # For j from max(0,-m) to min(n-1, n-1-m):
            #   first term: 1-h[j-m] (when 0≤j-m<n)
            # For j from max(0,-m) to min(n-1, n-1-m):
            #   second term: -h[j+m] (when 0≤j+m<n)
            
            # First term: j-m in [0,n-1] → j in [m, m+n-1], intersect [0,n-1]
            j_start1 = max(0, m)
            j_end1 = min(n-1, m + n - 1)
            if j_start1 <= j_end1:
                grad[j_start1:j_end1+1] += w_k * (1.0 - h[j_start1-m:j_end1-m+1])
            
            # Second term: j+m in [0,n-1] → j in [-m, n-1-m], intersect [0,n-1]
            j_start2 = max(0, -m)
            j_end2 = min(n-1, n-1-m)
            if j_start2 <= j_end2:
                grad[j_start2:j_end2+1] -= w_k * h[j_start2+m:j_end2+m+1]
        
        # Projected gradient step (minimize, so go opposite to gradient)
        h = h - lr * grad
        h = project_feasible(h, n)
        
        # Update correlation
        # Recompute (expensive but correct)
        corr = compute_corr(h, n)
        new_c = corr.max() * dx
        
        if new_c < best_c:
            best_c = new_c
            best_h = h.copy()
        
        # Adaptive learning rate
        if new_c > cur_c * 1.01:
            lr *= 0.9
        elif new_c < cur_c * 0.99:
            lr = min(lr * 1.05, 0.1)
        cur_c = new_c
    
    return best_h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def pgd_solve(h, n, deadline, lr=0.05, K=5, beta=30.0, momentum=0.9):
    h = h.copy().astype(float)
    v = np.zeros(n)  # momentum
    corr = compute_corr(h, n)
    cur_c = corr.max()
    best_h = h.copy()
    best_mc = cur_c
    lr_cur = lr
    it = 0
    while time.time() < deadline:
        top_idx = np.argpartition(corr, -K)[::-1][:K]
        top_vals = corr[top_idx]
        w = np.exp(beta * (top_vals - top_vals.max()))
        w /= w.sum()
        grad = np.zeros(n)
        for w_k, k in zip(w, top_idx):
            m = int(k) - (n - 1)
            j1s = max(0, m)
            j1e = min(n-1, m+n-1)
            if j1s <= j1e:
                grad[j1s:j1e+1] += w_k * (1.0 - h[j1s-m:j1e-m+1])
            j2s = max(0, -m)
            j2e = min(n-1, n-1-m)
            if j2s <= j2e:
                grad[j2s:j2e+1] -= w_k * h[j2s+m:j2e+m+1]
        v = momentum * v - lr_cur * grad
        h_new = project_feasible(h + v, n)
        corr = compute_corr(h_new, n)
        new_mc = corr.max()
        if new_mc < best_mc:
            best_mc = new_mc
            best_h = h_new.copy()
        if new_mc > cur_mc + 1e-6:
            lr_cur *= 0.5
            v *= 0.5
        else:
            lr_cur = min(lr_cur * 1.02, 0.3)
        h = h_new
        cur_mc = new_mc
        it += 1
    return best_h, best_mc

def binary_search(h_init, n, deadline):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = compute_corr(h, n)
    cur_mc = corr.max()
    best_h = h.copy()
    best_mc = cur_mc
    m_range = np.arange(-(n-1), n)
    while time.time() < deadline:
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        for i in zeros:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_mc
            for j in ones:
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta_i = np.zeros(2*n-1)
                delta_i[msk] = 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta_i[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta_j = np.zeros(2*n-1)
                delta_j[msk] = -(1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta_j[msk] += h[jmn[msk]]
                delta = delta_i + delta_j
                delta[n-1] -= 2.0
                k1 = n-1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n-1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                nc = (corr + delta).max()
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
                    best_delta = delta
            if best_j >= 0 and best_nc < cur_mc - 1e-14:
                h[i] = 1.0
                h[best_j] = 0.0
                corr += best_delta
                cur_mc = best_nc
                if cur_mc < best_mc:
                    best_mc = cur_mc
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_mc * dx

def anneal_single(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = compute_corr(h, n)
    cur_mc = corr.max()
    best_h = h.copy()
    best_mc = cur_mc
    m_range = np.arange(-(n-1), n)
    T = 0.01
    while time.time() < deadline:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        delta = np.zeros(2*n-1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n-1] -= 2.0
        k1 = n-1 + (j-i)
        if 0 <= k1 < 2*n-1: delta[k1] += 1.0
        k2 = n-1 + (i-j)
        if 0 <= k2 < 2*n-1: delta[k2] += 1.0
        nc = (corr + delta).max()
        if nc <= cur_mc or rng.random() < np.exp((cur_mc - nc) / T):
            h[i] = 1.0
            h[j] = 0.0
            corr += delta
            cur_mc = nc
            if cur_mc < best_mc:
                best_mc = cur_mc
                best_h = h.copy()
        T = max(T * 0.99999, 1e-8)
    return best_h, best_mc * dx
