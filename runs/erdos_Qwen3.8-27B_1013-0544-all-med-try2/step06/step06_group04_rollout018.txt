The issue is that minimizing t subject to all these linear constraints doesn't guarantee that the actual c_k(h) <= t. We need the cutting plane iteration to ensure this.

So the correct approach is:
1. Start with no cuts (or a few initial cuts)
2. Solve the LP
3. At the solution h_new, check all c_k(h_new)
4. Add cuts for any k where c_k(h_new) > t* (the LP optimal value)
5. Repeat

This is the standard cutting plane method for minimizing a convex function (each c_k is convex in h? Let me check).

This is a sum of terms h[i] - h[i]h[i+k], which is a sum of concave functions (the -h[i]h[i+k] part is concave). So c_k is NOT convex in h. It's actually non-convex!

This means the outer approximation (which assumes convexity for the cutting plane to be valid) is not directly applicable. The linearized constraint c_k(h_curr) + grad . (h - h_curr) >= c_k(h) is the tangent plane inequality, which holds for convex functions. Since c_k is NOT convex, the tangent plane can overestimate c_k, meaning the constraint might cut off feasible points.

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n: return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        for j in range(n):
            v = 0.0
            if j <= n - 1 - m: v += (1.0 - h[j + m])
            if j >= m: v -= h[j - m]
            grad[j] = v
    else:
        for j in range(n):
            v = 0.0
            if j >= m: v += (1.0 - h[j - m])
            if j <= n - 1 - m: v -= h[j + m]
            grad[j] = v
    return grad

def lp_outer(h, n, deadline, max_iters=150, top_k=5):

    h_curr = h.copy()
    best_h, best_c = h_curr.copy(), compute_c5(h_curr, n)
    h_var, t_var = cp.Variable(n), cp.Variable()
    base = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
    for it in range(max_iters):
        if time.time() > deadline - 2: break
        c_vals = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c_vals)[-top_k:]
        cons = list(base)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            g = _lag_grad(h_curr, n, lag)
            cons.append(c_vals[idx] + dx * (g @ (h_var - h_curr)) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try: prob.solve(solver=cp.GLOP)
        except Exception:
            try: prob.solve(solver=cp.CBC)
            except Exception: break
        if prob.status not in ('optimal', 'optimal_inaccurate'): break
        hn = h_var.value
        if hn is None: break
        hn = project_feasible(hn, n)
        cn = compute_c5(hn, n)
        if cn < best_c - 1e-12:
            best_c, best_h = cn, hn.copy()
        step = np.max(np.abs(hn - h_curr))
        h_curr = hn
        if step < 1e-10: break
    return best_h, best_c

def _smooth_max(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500: return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sum_con(x, n): return float(np.sum(x) - n / 2.0)
def _sum_con_jac(x, n): return np.ones(len(x))

def slsqp_polish(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_max, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception: return h.copy()
