import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _smooth_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _cons_eq(x, n):
    return float(x.sum() - n / 2.0)

def _cons_jac(x, n):
    return np.ones(n)

def _cons_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def lbfgs_phase(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 5.0), (500, 10.0), (2000, 50.0), (10000, 200.0),
               (50000, 1000.0), (200000, 5000.0), (1000000, 20000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_smooth_obj, h, jac=_smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_iters * 10,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac, 'args': (n,)}]
    try:

def simulated_annealing(h, n, deadline, T_init=0.005, T_final=0.0001, n_iter_factor=50):
    """Simulated annealing with pair swaps."""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_iter = n * n_iter_factor
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Temperature schedule (geometric)
        progress = it / n_iter
        T = T_init * (T_final / T_init) ** progress
        
        # Pick random pair
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # Pick random delta (positive or negative)
        delta = np.random.uniform(0.001, 0.05) * np.sign(np.random.uniform() - 0.5)
        
        # Check bounds
        if h[i] + delta < 0 or h[i] + delta > 1:
            continue
        if h[j] - delta < 0 or h[j] - delta > 1:
            continue
        
        # Make the move
        h[i] += delta
        h[j] -= delta
        h = project_feasible(h, n)  # Re-project to maintain sum constraint
        
        c_new = compute_c5(h, n)
        delta_c = c_new - best_c
        
        if delta_c < 0:
            # Always accept improving moves
            best_c = c_new
            best_h = h.copy()
        else:
            # Accept with probability exp(-delta_c / T)
            if T > 0 and np.random.uniform() < np.exp(-delta_c / T):
                pass  # Accept (keep the move)
            else:
                # Reject: undo
                h[i] -= delta
                h[j] += delta
                h = project_feasible(h, n)
    
    # After annealing, project to best
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13: continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    return float(x.sum() - n / 2.0)

    return np.ones(n)

def _obj2(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in [(100,5),(500,10),(2000,50),(10000,200),(50000,1000),(200000,5000),(1000000,20000)]:
        if time.time() > deadline - 2: break
        try:
            res = minimize(_obj, h, jac=_grad, args=(n,alpha,mu), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter':max_iters,'ftol':1e-16,'gtol':1e-13,'maxls':50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new; h = h_new
        except: pass
    return h, best_c

def slsqp(h, n, deadline, alpha=500000, maxiter=2000):

def binary_local_search(h, n, deadline, max_iters=50000):
    """Local search on the binary rounding of h."""
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        # Remove excess ones
        order = np.argsort(h_bin - 0.5)  # Most "0-like" first
        for i in order:
            if h_bin[i] == 1 and n_ones > target:
                h_bin[i] = 0; n_ones -= 1
    elif n_ones < target:
        order = np.argsort(0.5 - h_bin)  # Most "1-like" first  
        for i in order:
            if h_bin[i] == 0 and n_ones < target:
                h_bin[i] = 1; n_ones += 1
    
    best_c = compute_c5(h_bin, n)

        if time.time() > deadline: break
        improved = False
        # Try random pairs of swaps

            j = np.random.randint(n)
            if i == j or h_bin[i] == h_bin[j]: continue
            # Swap: h_bin[i] becomes 1, h_bin[j] becomes 0 (or vice versa)

            c = compute_c5(h_bin, n)

            else:
                h_bin[i], h_bin[j] = 1.0 - h_bin[i], 1.0 - h_bin[j]
        if not improved: break
    
    return h_bin, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline: return h, best_c
                lo = max(0, i - radius) if radius else 0
                hi = min(n, i + radius + 1) if radius else n
                for j in range(lo, hi):
                    if i == j: continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13: continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _eq(x, n):
    return float(x.sum() - n / 2.0)

def _eq_jac(x, n):
    return np.ones(n)

def _obj2(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def lbfgs(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in [(100,5),(500,10),(2000,50),(10000,200),(50000,1000),(200000,5000),(1000000,20000)]:
        if time.time() > deadline - 2: break
        try:
            res = minimize(_obj, h, jac=_grad, args=(n,alpha,mu), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter':max_iters,'ftol':1e-16,'gtol':1e-13,'maxls':50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new; h = h_new
        except: pass
    return h, best_c

def slsqp(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type':'eq','fun':_eq,'jac':_eq_jac,'args':(n,)}]
    try:
        res = minimize(_obj2, h, args=(n,alpha), method='SLSQP',
                      bounds=bounds, constraints=cons, options={'maxiter':maxiter,'ftol':1e-15})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except:
        return h.copy(), compute_c5(h, n)

def binary_local_search(h, n, deadline, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        order = np.argsort(h_bin[1-int(n_ones)])
        idx = np.where(h_bin == 1)[0]
        idx_sorted = idx[np.argsort(h_bin[idx])]
        for i in idx_sorted[:n_ones-target]:
            h_bin[i] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        idx_sorted = idx[np.argsort(-h_bin[idx])] if len(idx)>0 else idx
        for i in idx_sorted[:target-n_ones]:
            h_bin[i] = 1
    best_c = compute_c5(h_bin, n)
    if best_c >= 1.0: return h_bin, best_c
    while time.time() < deadline:
        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        rng.shuffle(ones); rng.shuffle(zeros)
        for i in ones:
            if time.time() > deadline: return h_bin, best_c
            for j in zeros:
                h_bin[i], h_bin[j] = 0.0, 1.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c; improved = True
                    break
                h_bin[i], h_bin[j] = 1.0, 0.0
            if improved: break
        if not improved: break
    return h_bin, best_c

def pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline: return h, best_c
                lo = max(0, i - radius); hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j: continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14: best_c = c; improved = True
                        else: h[i] -= d; h[j] += d
    return h, best_c
