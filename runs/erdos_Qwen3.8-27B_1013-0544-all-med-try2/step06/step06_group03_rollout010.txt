    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:

            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_sh = shifted.max()
    if max_sh > 500:
        return float(m + (max_sh + np.log(np.sum(np.exp(shifted - max_sh)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    optimize_candidate(h_init, n_init, alloc1, "init")

    # Phase 2: Try smaller n values (faster to optimize, might find better discrete opt)
    x_ref = np.linspace(0, 2, len(best_h) if best_h is not None else n_init)
    h_ref = best_h if best_h is not None else h_init
    for n_try in [100, 120, 150, 200]:
        if time.time() > deadline - 50:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_ref, h_ref)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(55, (deadline - time.time()) * 0.1)
        if alloc < 15:
            break
        optimize_candidate(h_interp, n_try, alloc, f"n={n_try}")

    # Phase 3: Structured starting points at n_init
    n = len(best_h)
    if time.time() < deadline - 50:
        tm = thue_morse_seq(n)
        tm = project_feasible(tm, n)
        alloc = min(35, (deadline - time.time()) * 0.06)
        if alloc > 10:
            optimize_candidate(tm, n, alloc, "thue-morse")

    if time.time() < deadline - 40:
        # Low-discrepancy binary: use van der Corput-like placement
        h_ld = np.zeros(n)
        positions = np.zeros(n // 2, dtype=int)
        for k in range(n // 2):
            pos = int(np.floor(2.0 * (k + 0.5) * (0.5 + 0.3 * np.sin(2 * np.pi * k / n))))
            pos = int(np.floor((k + 0.5) * 2.0)) % n
            positions[k] = pos
        h_ld[positions] = 1.0
        h_ld = project_feasible(h_ld, n)
        alloc = min(30, (deadline - time.time()) * 0.05)
        if alloc > 10:
            optimize_candidate(h_ld, n, alloc, "ld-binary")

    # Phase 4: Multi-restart with diverse perturbations
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(3, int(remaining / 25))
        for r in range(n_restarts):
            if time.time() > deadline - 15:
                break
            scale = 0.02 + 0.05 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            # Also try structured perturbation
            if r % 2 == 1:
                freq = rng.uniform(1, 5)
                phase = rng.uniform(0, 2 * np.pi)
                hp = best_h + scale * np.sin(freq * np.linspace(0, 2, n) * np.pi + phase)
            hp = project_feasible(hp, n)
            alloc = min(22, (deadline - time.time() - 10) * 0.35)
            if alloc < 8:
                break
            optimize_candidate(hp, n, alloc, f"restart-{r}")

    # Phase 5: Final intensive pair search
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        rem = deadline - time.time() - 3
        if rem > 3:
            best_h, best_c = random_pair_search(best_h, n, time.time() + rem * 0.6, seed=7777)
            n = len(best_h)
            if time.time() < deadline - 3 and n <= 150:
                best_h, best_c = greedy_pair_search(best_h, n, deadline - 2, radius=20)
            best_h = project_feasible(best_h, len(best_h))
            best_c = compute_c5(best_h, len(best_h))

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
