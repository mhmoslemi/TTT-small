    c = np.correlate(x, 1 - x, 'full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    return float(m + np.log(np.sum(np.exp(shifted - mx)) + 1.0) / alpha)

def slsqp_opt(h, n, alpha, maxiter=5000, active_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    if active_idx is not None:
        obj = smooth_obj_active
        args = (n, active_idx, alpha)
    else:
        obj = smooth_obj_full
        args = (n, alpha)
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient_component(h, n, k_idx):
    L = k_idx - (n - 1)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[:n - L] += (1.0 - h[L:])
        g[L:] += (-h[:n - L])
    else:
        m = -L
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] += -1.0
    return g

def projected_grad_descent(h, n, dl, alpha, max_iter=3000, step_init=0.005):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = step_init
    prev_c = best_c
    stuck = 0
    for it in range(max_iter):
        c = compute_all_lags(h, n)
        m = c.max()
        weights = np.exp(alpha * (c - m))
        weights /= weights.sum()
        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            if weights[k_idx] < 1e-10:
                continue
            grad += weights[k_idx] * lag_gradient_component(h, n, k_idx)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.05, 0.05)
            stuck = 0
        else:
            step *= 0.7
            stuck += 1
            if stuck > 50:
                break
            if step < 1e-9:
                break
    return best_h, best_c

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust * np.ones(n)]
    for k in range(2 * n - 1):
        L = k - (n - 1)
        grad = lag_gradient_component(h_curr, n, k)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx_loop(h, n, deadline, max_iters=80):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.003, trust * 0.9)
    return best_h, best_c

def multi_lag_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            top_lags = np.argsort(c)[-8:]
            for k_idx in top_lags:
                L = k_idx - (n - 1)
                if L == 0:
                    continue
                m = abs(L)
                if m >= n:
                    continue
                for i in range(min(50, n - m)):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[i] > h[j] + 1e-12:
                        delta = min(d, (h[i] - h[j]) * 0.4, h[i] * 0.5, (1.0 - h[j]) * 0.5)
                        if delta < 1e-10:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i] + 1e-12:
                        delta = min(d, (h[j] - h[i]) * 0.4, h[j] * 0.5, (1.0 - h[i]) * 0.5)
                        if delta < 1e-10:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
    return best_h, best_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: outer approximation
    if time.time() < deadline - 30:
        oo_end = time.time() + min(60, (deadline - time.time()) * 0.3)
        h_oo, c_oo = outer_approx_loop(best_h, n, oo_end)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo
    # Phase 2: SLSQP with full smooth max
    if time.time() < deadline - 20:
        for alpha in [5000, 50000, 200000]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_opt(best_h, n, alpha, maxiter=3000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    # Phase 3: projected gradient descent
    if time.time() < deadline - 15:
        pg_end = time.time() + min(40, (deadline - time.time()) * 0.35)
        h_pg, c_pg = projected_grad_descent(best_h, n, best_c, alpha=1000,
                                             max_iter=2000, step_init=0.003)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg
    # Phase 4: active set SLSQP
    if time.time() < deadline - 15:
        for _ in range(3):
            if time.time() > deadline - 10:
                break
            c = compute_all_lags(best_h, n)
            top_idx = np.argsort(c)[-6:]
            h_new = slsqp_opt(best_h, n, 50000, maxiter=3000, active_idx=top_idx)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
    # Phase 5: final SLSQP polish
    if time.time() < deadline - 10:
        h_new = slsqp_opt(best_h, n, 500000, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 6: local search
    if time.time() < deadline - 3:
        best_h, best_c = multi_lag_local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    # Optimize at the initial n
    if h_init is not None:
        alloc = min(300, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    # Try other n values
    all_n = sorted(set([n_init - 50, n_init, n_init + 50, n_init - 100, 200, 250]))
    all_n = [x for x in all_n if 80 <= x <= 500]
    for n_try in all_n:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 50
        if remaining < 60:
            break
        alloc = min(100, remaining * 0.3)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Multi-start restarts at best n
    if best_h is not None and time.time() < deadline - 60:
        n_best = len(best_h)
        n_restarts = max(1, int((deadline - time.time() - 40) / 50))
        n_restarts = min(n_restarts, 4)
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.02 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(55, (deadline - time.time() - 30) * 0.4)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_n(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Final polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_new = slsqp_opt(best_h, n, 1000000, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h, best_c = multi_lag_local_search(best_h, n, deadline - 2)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(200, 0.5), 200)
        best_c = compute_c5(best_h, 200)
    return best_h, best_c, len(best_h)
