    best_c = compute_c5(h, n)
    dx = 2.0 / n

        while time.time() < deadline:

    V[p, m] gives the change in corr[m] per unit increase in h[p]."""

    return base * (V @ weights) * dx

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_pnorm(h, n, p, maxiter=2000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_pnorm, h, args=(n, p), jac=grad_pnorm,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_influence(h, n):
    lags = np.arange(2 * n - 1) - (n - 1)
    p_idx = np.arange(n)[:, None]
    lag_idx = lags[None, :]
    pm = p_idx - lag_idx
    pp = p_idx + lag_idx
    V = np.where((pm >= 0) & (pm < n), 1.0 - h[np.clip(pm, 0, n - 1)], 0.0)
    V += np.where((pp >= 0) & (pp < n), -h[np.clip(pp, 0, n - 1)], 0.0)
    return V

def fast_pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    corr = compute_corr(h, n)
    V = compute_influence(h, n)

    for d in deltas:
        while time.time() < deadline:
            V = compute_influence(h, n)
            best_imp = 0.0
            best_move = None
            for i in range(n):
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        delta_c = d * (V[i] - V[j])
                        new_max = float(np.max(corr + delta_c))
                        imp = best_c - new_max
                        if imp > best_imp:
                            best_imp = imp
                            best_move = (i, j, d)
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        delta_c = -d * (V[i] - V[j])
                        new_max = float(np.max(corr + delta_c))
                        imp = best_c - new_max
                        if imp > best_imp:
                            best_imp = imp
                            best_move = (i, j, -d)
            if best_move is None or best_imp < 1e-15:
                break
            i, j, dd = best_move
            h[i] += dd
            h[j] -= dd
            corr = compute_corr(h, n)
            best_c = compute_c5(h, n)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=60):
    best_c = compute_c5(h, n)
    corr = compute_corr(h, n)

    for d in deltas:
        while time.time() < deadline:
            V = compute_influence(h, n)
            best_imp = 0.0
            best_move = None
            for i in range(n):
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        delta_c = d * (V[i] - V[j])
                        new_max = float(np.max(corr + delta_c))
                        imp = best_c - new_max
                        if imp > best_imp:
                            best_imp = imp
                            best_move = (i, j, d)
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        delta_c = -d * (V[i] - V[j])
                        new_max = float(np.max(corr + delta_c))
                        imp = best_c - new_max
                        if imp > best_imp:
                            best_imp = imp
                            best_move = (i, j, -d)
            if best_move is None or best_imp < 1e-15:
                break
            i, j, dd = best_move
            h[i] += dd
            h[j] -= dd
            corr = compute_corr(h, n)
            best_c = compute_c5(h, n)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    p_list = [10, 20, 50, 100, 200, 500]
    for p in p_list:
        if time.time() > deadline - 15:
            break
        ht = slsqp_pnorm(h, n, p, maxiter=2000, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            h = ht

    if n <= 130:
        if time.time() < deadline - 5:
            h, best_c = fast_pair_search(h, n, deadline - 5,
                                         [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    else:
        if time.time() < deadline - 5:
            h, best_c = near_pair_search(h, n, deadline - 5,
                                         [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                                         radius=80)

    if time.time() < deadline - 5:
        ht = slsqp_pnorm(h, n, 500, maxiter=2000, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht

    if n <= 130 and time.time() < deadline - 3:
        h, best_c = fast_pair_search(h, n, deadline - 1,
                                     [0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
    elif time.time() < deadline - 3:
        h, best_c = near_pair_search(h, n, deadline - 1,
                                     [0.0005, 0.0002, 0.0001, 0.00005],
                                     radius=50)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize at initial resolution
    alloc1 = min(250, (deadline - t0) * 0.30)
    h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt, c_opt

    # Phase 2: Try other resolutions by interpolation
    x_init = np.linspace(0, 2, n_init)
    for n_try in [100, 150, 200, 300]:
        if time.time() > deadline - 120:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        remaining = deadline - time.time() - 60
        if remaining < 40:
            break
        alloc = min(100, remaining * 0.25)
        if alloc < 30:
            break
        h_opt, c_opt = optimize_at_n(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multi-restart with structured perturbation
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(3, int(remaining / 35))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            if r == 0:
                # Shift perturbation
                hp = np.roll(best_h, 1)
            elif r == 1:
                # Scale perturbation
                hp = best_h + 0.05 * rng.randn(n)
            else:
                scale = 0.02 + 0.10 * (r / max(1, n_restarts - 1))
                hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 20) * 0.3)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Extended fine-grained search on best
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        if n <= 130:
            best_h, best_c = fast_pair_search(best_h, n, deadline - 5,
                                              [0.001, 0.0005, 0.0002, 0.0001,
                                               0.00005, 0.00002, 0.00001])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 5,
                                              [0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                              radius=100)

    # Phase 5: Final SLSQP polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        for p in [200, 500]:
            if time.time() > deadline - 3:
                break
            ht = slsqp_pnorm(best_h, n, p, maxiter=3000, ftol=1e-16)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht
        if time.time() < deadline - 2:
            if n <= 130:
                best_h, best_c = fast_pair_search(best_h, n, deadline - 1,
                                                  [0.0001, 0.00005, 0.00002, 0.00001])
            else:
                best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                                  [0.0001, 0.00005, 0.00002],
                                                  radius=60)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
