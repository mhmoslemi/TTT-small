def project_feasible(h, n):

        if abs(diff) < 1e-10:

        if diff > 0:

            s = h.sum()
            if s < 1e-10:

def fista(h, n, deadline, max_iter=10000, tol=1e-12):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_prev = h.copy()
    t = 1.0
    lr = 2.0 / (3.0 * n)  # initial learning rate, L is roughly 2/n * something
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        
        # Find all k's near the max for subgradient averaging
        threshold = m - 0.01  # k's within 0.01 of max
        k_near = np.where(g >= threshold)[0]
        
        # Average gradient over near-max k's
        grad = np.zeros(n)
        for k in k_near:
            grad += analytical_gradient(h, n, k)
        grad /= len(k_near)
        
        # FISTA step
        h_new = project_feasible(h - lr * grad, n)
        
        # Backtracking if needed
        c_new = compute_c5(h_new, n)
        if c_new > m + 1e-10:
            lr *= 0.5
            continue
        
        # Momentum
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        h_prev = h_new + (t/t_new - 1) * (h_new - h)
        h = project_feasible(h_prev, n)
        h_new = h.copy()
        t = t_new
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        if abs(best_c - c) < tol and it > 100:
            break
    
    return best_h, best_c

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_at_lag(h, n, lag):
    """Gradient of g at a specific lag."""
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:] += (1 - h[:n-lag])
        grad[:n-lag] -= h[lag:]
    else:
        ml = -lag
        grad[:n-ml] += (1 - h[ml:])
        grad[ml:] -= h[:n-ml]
    return grad

def smooth_max_gradient(h, n, alpha):
    """Gradient of (1/alpha)*log(sum exp(alpha*g[k]))"""
    g = compute_all_g(h, n)
    gm = g.max()
    w = np.exp(alpha * (g - gm))
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n), gm
    w /= wsum
    # Only iterate over significant weights
    sig = w > 1e-10
    if not sig.any():
        return np.zeros(n), gm
    ks = np.where(sig)[0]
    grad = np.zeros(n)
    for k in ks:
        lag = k - (n - 1)
        grad += w[k] * grad_at_lag(h, n, lag)
    return grad, gm

def fista_optimize(h, n, deadline, alpha, max_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    h_prev = h.copy()
    t = 1.0
    lr = 1.0  # will be adjusted by backtracking
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad, gm = smooth_max_gradient(h_curr, n, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        # Backtracking line search
        step = lr
        found = False
        for _ in range(30):
            h_try = project_feasible(h_curr - step * grad, n)
            g_try = compute_all_g(h_try, n)
            gm_try = g_try.max()
            # Descent condition for smooth max approx
            if gm_try <= gm - 0.1 * step * gn * gn + 1e-12:
                found = True
                break
            step *= 0.5
            if step < 1e-15:
                break
        
        if not found:
            break
        
        # FISTA momentum
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        h_new = project_feasible(h_try + (t / t_new - 1) * (h_try - h_curr), n)
        h_curr = h_new
        t = t_new
        
        c = compute_c5(h_curr, n)
        if c < best_c:
            best_c = c
            best_h = h_curr.copy()
    
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=100):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros[:]:
            for j in ones[:]:
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-12:
                    best_c = c
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                    improved = True
                    break
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

def difference_set_init(n):
    """Create initialization based on quadratic residues."""
    # Find a prime p close to n where p ≡ 3 (mod 4)
    p = n
    while True:
        if p % 4 == 3 and all(p % i != 0 for i in range(2, int(p**0.5)+1)):
            break
        p += 1 if p % 2 == 0 else 2
    
    # Quadratic residues mod p
    residues = set()
    for x in range(1, p):
        residues.add((x*x) % p)
    
    # Map to our n positions
    h = np.zeros(n)
    if n <= p:
        for r in residues:
            if r < n:
                h[r] = 1.0
    else:
        # Tile
        for r in residues:
            for k in range(n // p + 1):
                pos = r + k * p
                if pos < n:
                    h[pos] = 1.0
    
    return project_feasible(h, n)

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0

    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break

            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_at_lag(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:] += (1.0 - h[:m])
            grad[:m] -= h[lag:]
    else:
        ml = -lag
        m = n - ml
        if m > 0:
            grad[:m] += (1.0 - h[ml:])
            grad[ml:] -= h[:m]
    return grad
