```python
    h_new = np.clip(h_new, 0, 1)
    # Adjust to satisfy sum constraint
    target = n / 2.0
    diff = target - h_new.sum()
    if abs(diff) > 1e-15:
        # Proportional adjustment
        if diff > 0:
            room = 1.0 - h_new
            tot = room.sum()
            if tot > 1e-15:
                h_new += (diff / tot) * room
        else:
            tot = h_new.sum()
            if tot > 1e-15:
                h_new += (diff / tot) * h_new
    return np.clip(h_new, 0, 1)
```

```python
import cvxpy as cp

def linearized_qp(h0, n, mu=0.01, max_lags=None):
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    
    # Compute c_k at h0 and gradients
    c_vals = np.correlate(h0, 1 - h0, mode='full') * dx
    # For efficiency, only use top lags
    if max_lags is not None:
        idx = np.argsort(c_vals)[-max_lags:]
    else:
        idx = range(len(c_vals))
    
    constraints = [cp.sum(h) == n/2.0, h >= 0, h <= 1]
    
    for k_idx in idx:
        # lag = k_idx - (n-1)
        lag = k_idx - (n - 1)
        # c_k(h0) + grad^T (h - h0) <= t
        # grad[j] = (1 - h0[j+lag]) - h0[j-lag] (with out-of-bounds = 0)
        # Let me compute this properly
        grad = np.zeros(n)
        for j in range(n):
            g = 1.0
            if 0 <= j + lag < n:
                g -= h0[j + lag]
            if 0 <= j - lag < n:
                g -= h0[j - lag]
            grad[j] = g
        constraints.append(c_vals[k_idx] + grad @ (h - h0) <= t)
    
    obj = cp.Minimize(t + mu * cp.sum_squares(h - h0))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.CLARABEL)  # or ECOS, SCS
    if h.value is not None:
        return np.clip(h.value, 0, 1)
    return h0.copy()
```

```python
import numpy as np
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0, 1)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0, 1)

def compute_grad_matrix(h, n):
    """Returns matrix G of shape (2n-1, n) where G[k_idx, j] = d(c_k)/d(h_j)"""
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n-1), n):
        ki = k + n - 1
        if k == 0:
            G[ki] = 1.0 - 2.0 * h
        elif k > 0:
            G[ki, k:] = 1.0 - h[:n-k]
            G[ki, :n-k] -= h[k:]
        else:
            m = -k
            G[ki, :n-m] = 1.0 - h[m:]
            G[ki, m:] -= h[:n-m]
    return G

def solve_lin_qp(h0, n, mu):
    """Solve linearized QP at point h0 with proximal parameter mu."""
    try:
        c_vals = compute_all_c(h0, n)
        G = compute_grad_matrix(h0, n)
        n_lags = 2 * n - 1
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        delta = h_var - h0
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        # Linearized: c_k + G[k,:] @ (h - h0) <= t
        constraints.append(c_vals + G @ delta <= t_var)
        
        obj = cp.Minimize(t_var + mu * cp.sum_squares(delta))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if h_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            return np.clip(h_var.value, 0, 1)
    except Exception:
        pass
    return h0.copy()

def nesterov_opt(h, n, deadline, max_iter=5000):
```

```python
import numpy as np
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0, 1)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0, 1)

def compute_grad_matrix(h, n):
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        ki = k + n - 1
        if k == 0:
            G[ki] = 1.0 - 2.0 * h
        elif k > 0:
            G[ki, k:] = 1.0 - h[:n - k]
            G[ki, :n - k] -= h[k:]
        else:
            m = -k
            G[ki, :n - m] = 1.0 - h[m:]
            G[ki, m:] -= h[:n - m]
    return G

def solve_lin_qp(h0, n, mu, top_lags=None):
    try:
        c_vals = compute_all_c(h0, n)
        G = compute_grad_matrix(h0, n)
        n_lags = 2 * n - 1
        if top_lags is not None and top_lags < n_lags:
            idx = np.argsort(c_vals)[-top_lags:]
            c_use = c_vals[idx]
            G_use = G[idx]
        else:
            c_use = c_vals
            G_use = G
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        delta = h_var - h0
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        constraints.append(c_use + G_use @ delta <= t_var)
        obj = cp.Minimize(t_var + mu * cp.sum_squares(delta))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if h_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            return np.clip(h_var.value, 0, 1)
    except Exception:
        pass
    return h0.copy()

def smooth_opt(h, n, deadline, alpha=10000, max_iter=3000):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    t_mom = 1.0
    h_prev = h_curr.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vals = compute_all_c(h_curr, n)
        c_max = c_vals.max()
        shifted = alpha * (c_vals - c_max)
        shifted -= shifted.max()
        w = np.exp(shifted)
        w /= w.sum()
        G = compute_grad_matrix(h_curr, n)
        grad = G.T @ w
        t_mom = (1 + np.sqrt(1 + 4 * (it + 1) ** 2)) / 2
        if it == 0:
            y = h_curr.copy()
        else:

            y = h_curr + beta * (h_curr - h_prev)
        lr = 1.0 / (alpha * np.max(np.abs(grad)) + 1e-6)
        lr = min(lr, 0.05)
        h_new = project_feasible(y - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
        h_prev = h_curr
        h_curr = h_new
    return h_curr, best_c
```
