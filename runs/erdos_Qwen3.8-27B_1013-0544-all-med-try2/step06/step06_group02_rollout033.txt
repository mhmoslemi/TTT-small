def coord_minimize(h, i, n, dx):
    """Minimize the overlap objective with respect to h[i], keeping all else fixed."""
    # For each shift m, compute a_m and b_m such that c_m = a_m * x + b_m
    # where x = h[i]
    h_fixed = h.copy()
    x = h_fixed[i]
    
    # Overlap for shift k (where k ranges from -(n-1) to n-1):
    # c_k = sum_j h[j] * (1 - h[j+k]) * dx
    # For a given k, h[i] appears in c_k when:
    #   - j = i and j+k is valid: contributes h[i]*(1-h[i+k])*dx
    #   - j+k = i (i.e., j = i-k) and j is valid: contributes h[i-k]*(1-h[i])*dx
    
    a = np.zeros(2*n-1)
    b = np.zeros(2*n-1)
    
    for k in range(-(n-1), n):
        m = k + (n-1)  # index in the full correlation array
        # Term 1: h[i] appears as the first factor when i+k is in [0, n)
        if 0 <= i+k < n:
            a[m] += (1.0 - h_fixed[i+k]) * dx
        # Term 2: h[i] appears as (1-h[i]) when i-k is in [0, n)
        if 0 <= i-k < n:
            a[m] -= h_fixed[i-k] * dx
        b[m] += 0  # we'll compute b separately
    
    # Actually, let me compute b properly
    # c_m = sum_j h[j]*(1-h[j+k])*dx
    # = sum_{j ≠ i} h[j]*(1-h[j+k])*dx + terms involving h[i]
    # So b_m = sum_{j ≠ i} h[j]*(1-h[j+k])*dx
    
    # This is expensive to compute naively. Let me use a different approach.
    # c_m = (sum_j h[j]*(1-h[j+k])*dx) for the current h
    # The change when we modify h[i] to x:
    # dc_m/dx = a_m (computed above)
    # So c_m(x) = c_m(x_0) + a_m * (x - x_0)
    # Therefore b_m = c_m(x_0) - a_m * x_0
    
    c_current = np.correlate(h_fixed, 1 - h_fixed, mode='full') * dx
    b = c_current - a * x
    
    # Now minimize max_m (a[m]*t + b[m]) over t in [0,1]
    # This is a convex PL function. Find its minimum.
    
    # Approach: the minimum is where the subgradient contains 0
    # i.e., where there exist active constraints with slopes straddling 0
    # or at boundary t=0 or t=1
    
    # Simple approach: evaluate on a grid and refine

    # For efficiency, let's just use a fine grid + refinement

    t_best = t_grid[np.argmin(f_vals)]
    f_best = f_vals.min()
    
    # Refine with ternary search around t_best
    lo = max(0, t_best - 2/200)
    hi = min(1, t_best + 2/200)

        m2 = hi - (hi - lo) / 3

        if f1 < f2:

    return t_new

        if abs(diff) < 1e-14:

            tot = room.sum()

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])

            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n/2.0)**2
    return float(val)

def _grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for mm in range(2*n-1):
        if w[mm] < 1e-12:
            continue
        k = mm - (n-1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            g[0:nv] += wm * (1.0 - x[k:n])
            g[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:n] += wm * (1.0 - x[0:nv])
            g[0:nv] += wm * (-x[kk:n])
    g += 2.0 * mu * (x.sum() - n/2.0)
    return g

                      method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': max_it, 'ftol': 1e-16, 'gtol': 1e-14,
                               'maxfun': max_it*10, 'maxls': 40})
        h2 = project_feasible(res.x, n)
        c2 = compute_c5(h2, n)
        return h2, c2
    except Exception:
        return h, compute_c5(h, n)

    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    rng = np.random.RandomState(777)
