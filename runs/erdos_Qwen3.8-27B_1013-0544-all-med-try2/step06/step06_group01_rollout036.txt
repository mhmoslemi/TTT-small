            JL = j_arr[:, None] + m_range[None, :]
            JLm = j_arr[:, None] - m_range[None, :]
            msp = (JL >= 0) & (JL < n)
            msm = (JLm >= 0) & (JLm < n)
            A = np.where(msp, -(1.0 - h[np.clip(JL, 0, n - 1)]), 0.0)
            B = np.where(msm, h[np.clip(JLm, 0, n - 1)], 0.0)
            dj = A + B
            dj[:, n - 1] -= 1.0
            cross = np.zeros((no, K))
            k1 = n - 1 + (j_arr - i)
            k2 = n - 1 + (i - j_arr)
            ms1 = (k1 >= 0) & (k1 < K)
            ms2 = (k2 >= 0) & (k2 < K)
            idx_no = np.arange(no)
            cross[idx_no[ms1], k1[ms1]] += 1.0
            cross[idx_no[ms2], k2[ms2]] += 1.0
            base = corr[None, :] + di[None, :] + dj + cross
            max_vals = base.max(axis=1)
            bp = np.argmin(max_vals)
            if max_vals[bp] * dx < cur_c - 1e-15:
                j = j_arr[bp]
                h[i], h[j] = 1.0, 0.0
                corr = base[bp].copy()
                cur_c = max_vals[bp] * dx
                zeros_list[ii] = zeros_list[-1]
                zeros_list.pop()
                ones_list.remove(j)
                ones_list.append(i)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c

def iterated_ls(h_init, n, deadline, seed=42, max_restarts=80):
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)
    cur_h = best_h.copy()
    for r in range(max_restarts):
        if time.time() > deadline - 10:
            break
        h_r, c_r = local_search(cur_h, n, deadline - 5, max_sweeps=8000, seed=seed + r * 137)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
            cur_h = h_r.copy()
        h_p = best_h.copy()
        n_ones = int(np.sum(h_p == 1))
        n_zeros = n - n_ones
        if n_ones <= 2 or n_zeros <= 2:
            break
        n_flips = rng.randint(2, 5)
        ones = np.where(h_p == 1)[0].copy()
        zeros = np.where(h_p == 0)[0].copy()
        for _ in range(min(n_flips, len(ones), len(zeros))):
            ai = rng.randint(len(ones))
            bi = rng.randint(len(zeros))
            a, b = ones[ai], zeros[bi]
            h_p[a], h_p[b] = 0.0, 1.0
            ones = np.where(h_p == 1)[0]
            zeros = np.where(h_p == 0)[0]
        cur_h = h_p
    return best_h, best_c

def fourierbin(x, K, seed):
    rng = np.random.RandomState(seed)
    n = len(x)
    vals = np.full(n, 0.0)
    for k in range(1, K + 1):
        a = rng.uniform(-0.4, 0.4)
        b = rng.uniform(-0.4, 0.4)
        vals += a * np.cos(k * np.pi * x) + b * np.sin(k * np.pi * x)
    hb = np.zeros(n)
    hb[np.argsort(vals)[::-1][:n // 2]] = 1.0
    return hb

def slsqp_polish(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        if time.time() > deadline:
            return h
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            return project_feasible(res.x, n)
        return h
    except Exception:
        return h
