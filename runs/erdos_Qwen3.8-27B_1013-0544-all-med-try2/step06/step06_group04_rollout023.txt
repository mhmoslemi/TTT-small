import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _smx(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    sh_max = sh.max()
    if sh_max > 500:
        return float(m + (sh_max + np.log(np.sum(np.exp(sh - sh_max)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return _smx(c, alpha)

def _obj_act(x, n, idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return _smx(c[idx], alpha)

def _ceq(x, n):
    return x.sum() - n / 2.0

def _ceq_j(x, n):
    return np.ones(n)

def _lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            g[:n-m] -= h[m:]
            g[m:] += 1.0 - h[:n-m]
        else:
            g[:n-m] += 1.0 - h[m:]
            g[m:] -= h[:n-m]
    return g

def slsqp(h, n, alpha, maxiter=5000, ftol=1e-16):
    b = [(0.0, 1.0)] * n
    c = [{'type': 'eq', 'fun': _ceq, 'jac': _ceq_j, 'args': (n,)}]
    try:
        r = minimize(_obj, h, args=(n, alpha), method='SLSQP',
                     bounds=b, constraints=c,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

def slsqp_act(h, n, top_m, alpha, maxiter=3000):
    cc = compute_all_lags(h, n)
    idx = np.argsort(cc)[-top_m:]
    b = [(0.0, 1.0)] * n
    c = [{'type': 'eq', 'fun': _ceq, 'jac': _ceq_j, 'args': (n,)}]
    try:
        r = minimize(_obj_act, h, args=(n, idx, alpha), method='SLSQP',
                     bounds=b, constraints=c,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

def cut_step(h, n, trust):
    dx = 2.0 / n
    cc = compute_all_lags(h, n)
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [cp.sum(hv) == n / 2.0, hv >= 0, hv <= 1,
            cp.abs(hv - h) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        g = _lag_grad(h, n, lag)
        cons.append(cc[k] + dx * (g @ (hv - h)) <= tv)
    prob = cp.Problem(cp.Minimize(tv), cons)
    for solver in [cp.SCS, cp.ECOS, cp.CBC]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate'] or hv.value is None:
        return h.copy()
    return project_feasible(hv.value, n)

def outer_approx(h, n, deadline, max_it=30):
    hc = h.copy()
    bh = hc.copy()
    bc = compute_c5(hc, n)
    trust = 0.1
    for _ in range(max_it):
        if time.time() > deadline:
            break
        hn = cut_step(hc, n, trust)
        cn = compute_c5(hn, n)
        if cn < bc - 1e-12:
            bc = cn
            bh = hn.copy()
        step = np.max(np.abs(hn - hc))
        if step < 1e-9:
            break
        hc = hn
        trust = max(0.002, trust * 0.8)
    return bh, bc
