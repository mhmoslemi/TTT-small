def build_rudin_shapiro(n):

    # The Rudin-Shapiro sequence has period 2^k
    # For our purposes, we generate a low-autocorrelation binary sequence
    # Using the recursive definition
    length = 1
    r = np.array([1.0])
    s = np.array([1.0])
    while length < n:
        r_new = np.concatenate([r, -s])
        s_new = np.concatenate([r, s])
        r, s = r_new, s_new
        length *= 2
    # Convert to [0,1] range: 1 -> 1, -1 -> 0
    r = r[:n]
    h = (r + 1.0) / 2.0

            np.random.shuffle(zero_positions)

def binary_sa(h, n, t_start, t_end, n_iter, deadline):

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    return float(m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha)

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
