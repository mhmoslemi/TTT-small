import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # Distribute the residual
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0, 1)
    return h

def subgrad_single(h, k_worst, n):
    grad = np.zeros(n)
    center = n - 1
    lag = k_worst - center
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, left=0, right=0)

def coord_optimize(h, i, n):
    """Find optimal h[i] given all other coordinates fixed. Returns new value."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    
    # Compute coefficient a[k] for each k: f_k(h') = corr[k] + a[k]*(h'-h[i])
    a = np.ones(2 * n - 1)
    for k in range(2 * n - 1):
        s = k - center
        if 0 <= i + s < n:
            a[k] -= h[i + s]
        if 0 <= i - s < n:
            a[k] -= h[i - s]
    
    # Minimize max_k (a[k]*t + (corr[k] - a[k]*h[i])) over t in [0,1]
    q = corr - a * h[i]
    p = a
    
    # Evaluate at boundaries and find the minimum of max(p*t + q)
    # This is 1D convex PL minimization
    v0 = np.max(q)  # at t=0
    v1 = np.max(p + q)  # at t=1
    
    # Active indices at t=0 and t=1
    idx0 = np.argmax(q)
    idx1 = np.argmax(p + q)
    
    # Try intersection of the two "top" lines
    t_star = 0.0
    best_v = min(v0, v1)
    
    if p[idx0] != p[idx1] and abs(p[idx0] - p[idx1]) > 1e-15:
        t_int = (q[idx1] - q[idx0]) / (p[idx0] - p[idx1])
        t_int = float(np.clip(t_int, 0.0, 1.0))
        v_int = np.max(p * t_int + q)
        if v_int < best_v:
            best_v = v_int

def coord_descent(h, n, target_sum, max_sweeps=50, tol=1e-10):
    """Cyclic coordinate descent."""
    h = h.copy()
    for sweep in range(max_sweeps):
        max_change = 0.0
        for i in range(n):
            old = h[i]
            new = coord_optimize(h, i, n)
            h[i] = new
            max_change = max(max_change, abs(new - old))
        h = project_feasible(h, target_sum)
        if max_change < tol:
            break
    return h

def obj_smooth_pen(x, n, beta, target_sum):
    x_c = np.clip(x, 0, 1)
    corr = np.correlate(x_c, 1.0 - x_c, mode='full')
    dx = 2.0 / n
    val = smooth_max(corr, beta) * dx
    pen = 500.0 * (x_c.sum() - target_sum) ** 2
    return val + pen

def grad_smooth_pen(x, n, beta, target_sum):
    x_c = np.clip(x, 0, 1)
    corr = np.correlate(x_c, 1.0 - x_c, mode='full')
    m = np.max(corr)
    w = np.exp(np.clip(beta * (corr - m), -700, 700))
    w /= w.sum()
    g = np.zeros(n)
    center = n - 1
    dx = 2.0 / n
    for k in range(len(corr)):
        if w[k] < 1e-15:
            continue
        lag = k - center
        if lag >= 0:
            idx = np.arange(n - lag)
            g[idx] += (1.0 - x_c[idx + lag]) * w[k]
            g[idx + lag] -= x_c[idx] * w[k]
        else:
            la = -lag
            idx = np.arange(n - la)
            g[idx + la] += (1.0 - x_c[idx]) * w[k]
            g[idx] -= x_c[idx + la] * w[k]
    g *= dx
    g += 1000.0 * (x_c.sum() - target_sum)
    return g

def lbfgsb_optimize(h, n, target_sum, beta, max_iter):
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj_smooth_pen, h, jac=grad_smooth_pen,
                       args=(n, beta, target_sum), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_feasible(res.x, target_sum)
    except:
        return h

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0, 1)
    return h


def subgrad_single(h, k_worst, n):
    grad = np.zeros(n)
    center = n - 1
    lag = k_worst - center
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, left=0, right=0)


def coord_optimize(h, i, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    a = np.ones(2 * n - 1)
    for k in range(2 * n - 1):
        s = k - center
        if 0 <= i + s < n:
            a[k] -= h[i + s]
        if 0 <= i - s < n:
            a[k] -= h[i - s]
    q = corr - a * h[i]
    p = a
    lo, hi = 0.0, 1.0
    v0 = np.max(q)
    v1 = np.max(p + q)
    if p[np.argmax(q)] > 0:
        return 0.0
    if p[np.argmax(p + q)] < 0:
        return 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        gval = np.max(p * mid + q)
        active = np.abs(p * mid + q - gval) < 1e-12
        if p[active].max() < 0:
            lo = mid
        elif p[active].min() > 0:
            hi = mid
        else:
            break
    return float((lo + hi) / 2.0)


def coord_descent(h, n, target_sum, max_sweeps=30, tol=1e-12):
    h = h.copy()
    for sweep in range(max_sweeps):
        max_change = 0.0
        for i in range(n):
            old = h[i]
            new = coord_optimize(h, i, n)
            h[i] = new
            max_change = max(max_change, abs(new - old))
        h = project_feasible(h, target_sum)
        if max_change < tol:
            break
    return h


def shufflord_descent(h, n, target_sum, max_sweeps=30, rng=None, tol=1e-12):
    h = h.copy()
    if rng is None:
        rng = np.random.RandomState(42)
    for sweep in range(max_sweeps):
        max_change = 0.0
        order = rng.permutation(n)
        for i in order:
            old = h[i]
            new = coord_optimize(h, i, n)
            h[i] = new
            max_change = max(max_change, abs(new - old))
        h = project_feasible(h, target_sum)
        if max_change < tol:
            break
    return h


def obj_smooth_pen(x, n, beta, target_sum):
    x_c = np.clip(x, 0, 1)
    corr = np.correlate(x_c, 1.0 - x_c, mode='full')
    dx = 2.0 / n
    val = smooth_max(corr, beta) * dx
    pen = 500.0 * (x_c.sum() - target_sum) ** 2
    return val + pen


def grad_smooth_pen(x, n, beta, target_sum):
    x_c = np.clip(x, 0, 1)
    corr = np.correlate(x_c, 1.0 - x_c, mode='full')
    m = np.max(corr)
    w = np.exp(np.clip(beta * (corr - m), -700, 700))
    w /= w.sum()
    g = np.zeros(n)
    center = n - 1
    dx = 2.0 / n
    for k in range(len(corr)):
        if w[k] < 1e-15:
            continue
        lag = k - center
        if lag >= 0:
            idx = np.arange(n - lag)
            g[idx] += (1.0 - x_c[idx + lag]) * w[k]
            g[idx + lag] -= x_c[idx] * w[k]
        else:
            la = -lag
            idx = np.arange(n - la)
            g[idx + la] += (1.0 - x_c[idx]) * w[k]
            g[idx] -= x_c[idx + la] * w[k]
    g *= dx
    g += 1000.0 * (x_c.sum() - target_sum)
    return g


def lbfgsb_optimize(h, n, target_sum, beta, max_iter):
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj_smooth_pen, h, jac=grad_smooth_pen,
                       args=(n, beta, target_sum), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h
