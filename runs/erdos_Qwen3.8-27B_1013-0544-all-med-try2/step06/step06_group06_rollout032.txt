import time
import random

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:
        return h

def directed_binary_search(h, n, deadline, max_iter=100000):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    best_c = c.max()
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        k_star = int(np.argmax(c))  # This is the index in the full correlation
        # Convert to actual lag
        # In mode='full', index n-1+k corresponds to lag k
        # But we want the actual shift in terms of "how much x+k exceeds x"
        # The max could be at any position in [0, 2n-2]
        # lag = k_star - (n-1)
        lag = k_star - (n - 1)
        
        if lag == 0:
            # g(0) = sum h[i](1-h[i]) = 0 for binary, so this shouldn't be max
            break
        
        # Find best swap
        # For 1-positions a: score = h[a+lag] + h[a-lag] (with 0 for out of range)
        # For 0-positions b: score = h[b+lag] + h[b-lag]
        
        # We want to minimize g(lag): Δg(lag) = h[a+lag] + h[a-lag] - h[b+lag] - h[b-lag]
        # So we want small (h[a+lag]+h[a-lag]) and large (h[b+lag]+h[b-lag])
        
        a_scores = np.zeros(len(ones))
        for idx, a in enumerate(ones):
            s = 0
            if 0 <= a + lag < n:
                s += h[a + lag]
            if 0 <= a - lag < n:
                s += h[a - lag]
            a_scores[idx] = s
        
        b_scores = np.zeros(len(zeros))
        for idx, b in enumerate(zeros):
            s = 0
            if 0 <= b + lag < n:
                s += h[b + lag]
            if 0 <= b - lag < n:
                s += h[b - lag]
            b_scores[idx] = s
        
        # Best a: smallest score, Best b: largest score
        a_best = ones[np.argmin(a_scores)]
        b_best = zeros[np.argmax(b_scores)]
        
        # Check if this helps
        delta_lag = a_scores[np.argmin(a_scores)] - b_scores[np.argmax(b_scores)]
        if delta_lag >= 0:
            # Can't improve this lag, try a different approach
            # Do a random swap for diversity
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
        else:
            a, b = a_best, b_best
        
        # Compute full delta
        h[a] = 0
        h[b] = 1
        c_new = np.correlate(h, 1 - h, mode='full') * dx
        c_new_max = c_new.max()
        
        if c_new_max < best_c - 1e-12:
            best_c = c_new_max
            best_h = h.copy()
            c = c_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # Revert
            h[a] = 1
            h[b] = 0
    
    return best_h, best_c
