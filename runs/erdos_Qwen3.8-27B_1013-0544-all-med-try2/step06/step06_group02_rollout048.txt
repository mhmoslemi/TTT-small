def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(len(w)):
        if w[k_idx] < 1e-12:
            continue
        k = k_idx - (n - 1)
        wm = w[k_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return grad

def fix_binary(h, n, rng):
    h = (h > 0.5).astype(float).copy()
    n_ones = int(h.sum())
    half = n // 2
    if n_ones > half:
        excess = n_ones - half
        one_pos = np.where(h == 1)[0]
        to_rm = rng.choice(one_pos, excess, replace=False)
        h[to_rm] = 0
    elif n_ones < half:
        need = half - n_ones
        zero_pos = np.where(h == 0)[0]
        to_add = rng.choice(zero_pos, need, replace=False)
        h[to_add] = 1
    return h

def binary_sa(h, n, deadline, rng):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    temp = 0.03
    n_stuck = 0
    while time.time() < deadline:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h, n)
        delta = c - best_c
        if delta < -1e-15:
            best_c = c
            best_h = h.copy()
            n_stuck = 0
        elif rng.random() < np.exp(-max(delta, 0.0) / max(temp, 1e-10)):
            n_stuck += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            n_stuck += 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if n_stuck > 300:
            temp *= 0.95
            n_stuck = 0
        if temp < 1e-9:
            improved = False
            for _ in range(5000):
                if time.time() > deadline:
                    return best_h, best_c
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i] = 0.0

                c = compute_c5(h, n)

def binary_block_search(h, n, deadline, rng):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    while time.time() < deadline:
        improved = False
        for _ in range(2000):

                break
            i = rng.randint(0, n - 1)

                continue
            h = best_h.copy()
            h[i], h[j] = h[j], h[i]
            c = compute_c5(h, n)
            if c < best_c - 1e-15:

                best_h = h.copy()
                improved = True
        if not improved:

    return best_h, best_c

    bounds = [(0.0, 1.0)] * n

            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c:
                best_c = cn
                h = hn
        except:
            pass
    # Pair search
    rng = np.random.RandomState(123)
