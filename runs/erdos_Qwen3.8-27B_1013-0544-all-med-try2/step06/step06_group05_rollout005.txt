```python
def fast_eval(c, h, n, i, j, d):
    """Evaluate new C5 after h[i]+=d, h[j]-=d without recomputing full correlation."""
    # Compute delta for all lags
    # Δc[k] = d*((1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k])
    # where out-of-bounds are 0
    
    # This is O(n) but can be vectorized
    lags = np.arange(-n+1, n)
    delta = np.zeros(n*2-1)
    
    # (1-h[i+k]) for valid i+k
    valid1 = (i + lags >= 0) & (i + lags < n)
    delta[valid1] += (1 - h[i + lags[valid1]])
    
    # -h[i-k] for valid i-k
    valid2 = (i - lags >= 0) & (i - lags < n)
    delta[valid2] -= h[i - lags[valid2]]
    
    # -(1-h[j+k]) for valid j+k
    valid3 = (j + lags >= 0) & (j + lags < n)
    delta[valid3] -= (1 - h[j + lags[valid3]])
    
    # +h[j-k] for valid j-k
    valid4 = (j - lags >= 0) & (j - lags < n)
    delta[valid4] += h[j - lags[valid4]]
    
    delta *= d
    c_new = c + delta
    dx = 2.0 / n
    return float(np.max(c_new) * dx)
```

```python
def gradient_optimize(h, n, alpha, lr, n_iter, deadline, momentum=0.9):
    """Projected gradient descent with momentum on the smooth objective."""
    h = h.copy()
    v = np.zeros(n)  # momentum
    dx = 2.0 / n
    target_sum = n / 2.0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        c = np.correlate(h, 1-h, mode='full')
        # Compute gradient
        shifted = alpha * c * dx
        max_shifted = np.max(shifted)
        weights = np.exp(shifted - max_shifted)
        weights /= weights.sum()
        
        # Only keep significant weights
        threshold = np.max(weights) * 1e-3
        active = np.where(weights > threshold)[0]
        
        grad = np.zeros(n)
        for k_idx in active:
            k = k_idx - (n - 1)
            w = weights[k_idx] * dx
            lo = max(0, -k)
            hi = min(n, n - k)
            grad[lo:hi] += w * (1 - h[lo+k:hi+k])
            lo2 = max(0, k)
            hi2 = min(n, n + k)
            grad[lo2:hi2] -= w * h[lo2-k:hi2-k]
        
        # Gradient step with momentum
        v = momentum * v - lr * grad
        h_new = h + v
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        # Fix the sum constraint
        diff = target_sum - h_new.sum()
        if diff > 0:
            room = 1.0 - h_new
            tot = room.sum()
            if tot > 0:
                h_new += (diff / tot) * room
        else:
            tot = h_new.sum()
            if tot > 0:
                h_new += (diff / tot) * h_new
        h_new = np.clip(h_new, 0, 1)
        
        # Check if we improved
        c_new = np.correlate(h_new, 1-h_new, mode='full')
        if np.max(c_new) < np.max(c):
            h = h_new
        else:
            # Backtrack
            v *= 0.5
    
    return h
```
