def slsqp_smooth(h, n, alpha, maxiter=3000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        return smooth_max(c, alpha)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_active(h, n, active_lags, maxiter=5000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        return float(c[active_lags].max())
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def adam_optimize(h, n, deadline, max_steps=200000, init_lr=0.003, init_alpha=3000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    m_s = np.zeros(n)
    v_s = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = init_lr
    alpha = init_alpha
    stall = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        weights = smooth_max_weights(c, alpha)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] > 1e-10:
                lag = k - (n - 1)
                grad += weights[k] * lag_gradient(h, n, lag)
        m_s = b1 * m_s + (1 - b1) * grad
        v_s = b2 * v_s + (1 - b2) * grad * grad
        m_hat = m_s / (1 - b1 ** (step + 1))
        v_hat = v_s / (1 - b2 ** (step + 1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            alpha = min(alpha * 1.001, 200000)
        else:
            h = h_new
            stall += 1
            if stall > 100:
                lr *= 0.93
                alpha = max(500, alpha * 0.8)
                if lr < 1e-7:
                    lr = 0.003
                    stall = 0
    return best_h, best_c

def local_search_precise(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    n2 = 2 * n - 1
    for d in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if lag > 0:
                    if h[i] > 0 and h[j] < 1:
                        delta = min(d, h[i] * 0.3, (1.0 - h[j]) * 0.3)
                        if delta < 1e-10:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                            h = project_feasible(h, n)
                    if h[j] < 1 and h[i] > 0:
                        delta = min(d, (1.0 - h[i]) * 0.3, h[j] * 0.3)
                        if delta < 1e-10:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                            h = project_feasible(h, n)
                else:
                    if h[i] < 1 and h[j] > 0:
                        delta = min(d, (1.0 - h[i]) * 0.3, h[j] * 0.3)
                        if delta < 1e-10:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)

                            best_c = c_new
                            improved = True
                        else:

def active_set_iterate(h, n, deadline, max_rounds=30):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        cmax = c.max()
        tol = max(1e-6, cmax * 0.001)
        active = np.where(c >= cmax - tol)[0]
        if len(active) > 15:
            active = np.argsort(c)[-15:]
        for n_iter in [8000]:
            if time.time() > deadline:
                break
            h_new = slsqp_active(h, n, active, maxiter=n_iter, ftol=1e-16)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
            else:
                h = h_new
        c_check = compute_all_lags(h, n)
        if c_check.max() - compute_c5(h, n) < 1e-12:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 300
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: quick SLSQP warm-up
    if time.time() < deadline - 60:
        h_new = slsqp_smooth(h, n, alpha=5000, maxiter=2000, ftol=1e-13)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = best_h

    # Phase 2: Adam optimization
    if time.time() < deadline - 40:
        alloc = min(250, (deadline - time.time()) * 0.45)
        h_new, c_new = adam_optimize(h, n, time.time() + alloc)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = best_h

    # Phase 3: Active-set SLSQP iterations
    if time.time() < deadline - 30:
        alloc = min(200, (deadline - time.time()) * 0.45)
        h_new, c_new = active_set_iterate(h, n, time.time() + alloc)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = best_h

    # Phase 4: More Adam with refined settings
    if time.time() < deadline - 25:
        alloc = min(150, (deadline - time.time()) * 0.5)
        h_new, c_new = adam_optimize(h, n, time.time() + alloc, init_lr=0.001, init_alpha=10000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = best_h

    # Phase 5: Another active set round
    if time.time() < deadline - 15:
        alloc = min(100, (deadline - time.time()) * 0.5)
        h_new, c_new = active_set_iterate(h, n, time.time() + alloc, max_rounds=20)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = best_h

    # Phase 6: Local search
    if time.time() < deadline - 5:
        h_new, c_new = local_search_precise(h, n, deadline - 2)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = best_h

    # Phase 7: Final SLSQP polish with high alpha
    if time.time() < deadline - 2:
        h_new = slsqp_smooth(h, n, alpha=500000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
