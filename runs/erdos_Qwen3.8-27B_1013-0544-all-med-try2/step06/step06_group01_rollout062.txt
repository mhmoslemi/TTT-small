def compute_gradient(h, p):
    # p has length n (p[0], ..., p[n-1])
    # grad[i] = sum_k p[k] * (h[i-k] + h[i+k])
    n = len(h)
    grad = np.zeros(n)
    for k in range(n):
        if p[k] < 1e-15: continue
        # h[i-k] for i = k, ..., n-1
        grad[k:] += p[k] * h[:n-k]
        # h[i+k] for i = 0, ..., n-1-k
        grad[:n-k] += p[k] * h[k:]
    return -grad
```

def project_feasible(h, n):
    target = n / 2.0
    # Binary search for lambda
    lo, hi = -1, 2
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_autocorr(h, n):
    """Compute a[k] = sum_m h[m]*h[m+k] for k=0,...,n-1"""
    # Use FFT for efficiency
    size = 2 * n
    H = np.fft.rfft(h, size)
    ac = np.fft.irfft(H * np.conj(H), size)[:n]
    return ac

def project_feasible(h, n):
    """Euclidean projection onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def compute_grad_smooth_min(h, n, alpha):
    """Compute gradient of smooth min of autocorrelation.
    f(h) = -(1/alpha) * log(sum_k exp(-alpha * a_k))
    grad[i] = sum_k p_k * (h[i-k] + h[i+k])
    """
    a = compute_autocorr(h, n)
    # Softmax with negative sign
    log_w = -alpha * a
    log_w -= log_w.max()  # for numerical stability
    w = np.exp(log_w)
    p = w / w.sum()
    
    # Compute gradient: grad[i] = sum_k p[k] * (h[i-k] + h[i+k])
    grad = np.zeros(n)
    for k in range(n):
        if p[k] < 1e-18:
            continue
        pk = p[k]
        # h[i-k] for i=k,...,n-1
        grad[k:] += pk * h[:n-k]
        # h[i+k] for i=0,...,n-1-k
        grad[:n-k] += pk * h[k:]
    return grad

def gradient_optimize(h, n, alpha, max_iter, deadline, initial_lr=0.01):
    """Gradient ascent on smooth min of autocorrelation with projection."""
    h = h.copy()
    lr = initial_lr
    best_min_a = compute_autocorr(h, n).min()
    patience = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        grad = compute_grad_smooth_min(h, n, alpha)
        h_new = h + lr * grad
        h_new = project_feasible(h_new, n)
        
        min_a = compute_autocorr(h_new, n).min()
        if min_a > best_min_a:
            best_min_a = min_a
            patience = 0
            # Try increasing lr slightly
            lr = min(lr * 1.05, 0.1)
        else:
            patience += 1
            lr *= 0.9
            h_new = h  # don't accept
        
        h = h_new
        if patience > 100:
            # Try a perturbation
            h = project_feasible(h + 0.01 * np.random.randn(n), n)
            patience = 0
            lr = initial_lr
    
    return h

def binary_local_search(h, n, deadline, max_sweeps=1000, seed=42):
    """Binary local search: swap 1s and 0s to improve min autocorrelation."""
    rng = np.random.RandomState(seed)
    target = n / 2.0
    dx = 2.0 / n
    
    # Make binary
    h = np.clip(h, 0, 1)
    # Threshold to make it binary with exactly n/2 ones
    sorted_idx = np.argsort(h)[::-1]
    h[:] = 0
    h[sorted_idx[:n//2]] = 1.0
    
    a = compute_autocorr(h, n)
    cur_min_a = a.min()
    cur_c5 = (target - cur_min_a) * dx
    
    best_h = h.copy()
    best_c5 = cur_c5
    best_min_a = cur_min_a
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    sweep = 0
    while time.time() < deadline and sweep < max_sweeps:
        sweep += 1
        improved = False
        
        # Try all swaps
        best_delta_min = -np.inf
        best_i, best_j = -1, -1
        
        for i in ones:
            # Precompute the "removal" effect of position i
            # rem[k] = h[i-k] + h[i+k] (with h[m]=0 for m out of range)
            rem = np.zeros(n)
            # h[i-k] for k=0,...,i (i-k >= 0)
            if i >= 0:
                rem[:i+1] += h[i::-1]  # h[i], h[i-1], ..., h[0]
            # h[i+k] for k=0,...,n-1-i
            rem[:n-i] += h[i:]  # h[i], h[i+1], ..., h[n-1]
            
            for j in zeros:
                # add[k] = h[j-k] + h[j+k]
                add = np.zeros(n)
                if j >= 0:
                    add[:j+1] += h[j::-1]
                add[:n-j] += h[j:]
                
                # Delta
                delta = add - rem
                # Cross correction at k=|i-j|
                d = abs(i - j)
                if d < n:
                    delta[d] -= 1.0
                
                new_min_a = (a + delta).min()
                if new_min_a > best_delta_min:
                    best_delta_min = new_min_a
                    best_i, best_j = i, j
        
        if best_i >= 0 and best_delta_min > cur_min_a + 1e-12:
            h[best_i], h[best_j] = 0, 1
            a = compute_autocorr(h, n)
            cur_min_a = a.min()
            improved = True
            
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            new_c5 = (target - cur_min_a) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c5

        inf[k:, k] = h[:n-k]
        # h[m+k] for m=0,...,n-1-k: h[k], h[k+1], ..., h[n-1]
        inf[:n-k, k] += h[k:]
    return inf
```

delta[k] = inf[j, k] - inf[i, k] - (1 if k==|i-j| else 0)

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def compute_autocorr(h, n):

    H = np.fft.rfft(h, size)

    target = n / 2.0
    lo, hi = -2.0, 2.0

        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def compute_grad(h, n, alpha):
    a = compute_autocorr(h, n)
    log_w = -alpha * a
    log_w -= log_w.max()
    w = np.exp(log_w)
    p = w / w.sum()
    grad = np.zeros(n)

        if p[k] < 1e-18:
            continue
        pk = p[k]
        grad[k:] += pk * h[:n-k]
        grad[:n-k] += pk * h[k:]
    return grad

            break

        best_i_global = -1
