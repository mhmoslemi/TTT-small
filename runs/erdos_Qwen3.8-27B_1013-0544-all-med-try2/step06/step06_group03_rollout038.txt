    h = h.copy()
    for it in range(max_iter):
        dx = 2.0 / n
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        # Find top-k lags
        threshold = m - 1e-4
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Minimize sum of active lag values
        # Use a few gradient steps
        for _ in range(100):
            # Compute gradient of sum of active c_l
            grad = np.zeros(n)
            for l in active:
                # c_l = dx * sum_j h[j] * (1 - h[j+l-(n-1)])
                # Let offset = l - (n-1)
                offset = l - (n - 1)
                # For each j where both j and j+offset are valid:
                j_min = max(0, -offset)
                j_max = min(n, n - offset)
                for j in range(j_min, j_max):
                    k = j + offset
                    # ∂c_l/∂h[j] = dx * (1 - h[k])
                    # ∂c_l/∂h[k] = -dx * h[j]
                    grad[j] += dx * (1 - h[k])

            # Use simple projected gradient
            lr = 0.01
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)

            if np.max(np.abs(h_new - h)) < 1e-8:
                h = h_new
                break
            h = h_new
        
        # Update c
        c = np.correlate(h, 1 - h, mode='full') * dx
        if it > 5 and abs(c.max() - m) < 1e-10:
            break
    return h

def obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx  # length 2n-1
    m = c.max()
    sh = alpha * (c - m)
    sh = np.clip(sh, -500, 0)  # clip for numerical stability
    exp_sh = np.exp(sh)
    p = exp_sh / exp_sh.sum()  # probability weights
    f = m + np.log(exp_sh.sum()) / alpha
    
    # Compute gradient
    grad = np.zeros(n)
    # For each k in range(2n-1), offset = k - (n-1)
    # grad[i] += dx * p[k] * (1 - h[i+offset]) if 0 <= i+offset < n
    # grad[i] -= dx * p[k] * h[i-offset] if 0 <= i-offset < n
    
    # Vectorize: for each offset, the contribution is a shifted version of h
    offsets = np.arange(2*n-1) - (n-1)  # from -(n-1) to n-1
    for k in range(2*n-1):
        if p[k] < 1e-15:
            continue
        off = offsets[k]
        # Term 1: (1 - h[i+off]) for valid i
        i_min = max(0, -off)
        i_max = min(n, n - off)
        if i_max > i_min:
            idx = np.arange(i_min, i_max)
            grad[idx] += dx * p[k] * (1 - h[idx + off])
        # Term 2: -h[i-off] for valid i
        i_min2 = max(0, off)
        i_max2 = min(n, n + off)
        if i_max2 > i_min2:
            idx = np.arange(i_min2, i_max2)
            grad[idx] -= dx * p[k] * h[idx - off]
    
    return f, grad

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sh = np.clip(sh, -500, 0)
    exp_sh = np.exp(sh)
    s = exp_sh.sum()
    f = m + np.log(s) / alpha
    p = exp_sh / s
    
    grad = np.zeros(n)
    offsets = np.arange(2 * n - 1) - (n - 1)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-18:
            continue
        off = offsets[k]
        # grad[i] += dx * pk * (1 - h[i+off]) for valid i+off
        lo1 = max(0, -off)
        hi1 = min(n, n - off)
        if hi1 > lo1:
            idx = np.arange(lo1, hi1)
            grad[idx] += dx * pk * (1.0 - h[idx + off])
        # grad[i] -= dx * pk * h[i-off] for valid i-off
        lo2 = max(0, off)
        hi2 = min(n, n + off)
        if hi2 > lo2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= dx * pk * h[idx - off]
    
    return f, grad

    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    def func(x):
        return obj_and_grad(x, n, alpha)[0]

        return obj_and_grad(x, n, alpha)[1]
    
    def obj(x):
        f, g = obj_and_grad(x, n, alpha)
        return f, g

             'jac': lambda x: np.ones(n)}]

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})

    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for bs in block_sizes:
        d = 0.01
        while time.time() < deadline:
            improved = False
            for i in range(n - bs + 1):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + bs, n - bs + 1):
                    # Try moving block [i, i+bs) to [j, j+bs)
                    # i.e., increase h[i:i+bs] and decrease h[j:j+bs]
                    block_i = h[i:i+bs].copy()
                    block_j = h[j:j+bs].copy()
                    # Move d from block j to block i

                        # Wait, d here is the total amount to move, split among bs elements

    return fval, grad

def slsqp_opt(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def objfun(x):
        return _obj_and_grad(x, n, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(objfun, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    if radius is None:
        radius = n
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def optimize_local(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    alphas = [200, 1000, 5000, 20000, 100000, 300000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        h_new = slsqp_opt(h, n, a, maxiter=250)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-14:
            best_c = c
            h = h_new
    if time.time() < deadline - 5:
        if n <= 120:
            h, best_c = pair_search(h, n, deadline,
                [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        else:
            h, best_c = pair_search(h, n, deadline,
                [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001], radius=40)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(100, 0.5)
        n0 = 100

    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    x0 = np.linspace(0, 2, n0)

    # Phase 1: Optimize at original n with analytical gradient SLSQP
    alloc1 = min(280, (deadline - t0) * 0.3)
    h_opt, c_opt = optimize_local(h0, n0, t0 + alloc1)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    # Phase 2: Try other n values interpolated from best
    for n_try in [100, 150, 200, 80, 120]:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x0, h0)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(100, (deadline - time.time()) * 0.18)
        if alloc < 30:
            break
        h_opt, c_opt = optimize_local(h_interp, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multi-restart with increasing perturbation
    n = len(best_h)
    remaining = deadline - time.time()
    n_restarts = max(2, int(remaining * 0.35 / 45))
    for r in range(n_restarts):
        if time.time() > deadline - 35:
            break
        scale = 0.015 + 0.06 * (r / max(1, n_restarts - 1))
        hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        alloc = min(45, (deadline - time.time() - 25) * 0.35)
        if alloc < 15:
            break
        h_opt, c_opt = optimize_local(hp, n, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Extra restarts from random feasible points
    if time.time() < deadline - 60:
        for r in range(3):
            if time.time() > deadline - 30:
                break
            hp = np.clip(rng.beta(2, 2, n), 0, 1)
            hp = project_feasible(hp, n)
            alloc = min(30, (deadline - time.time() - 20) * 0.3)
            if alloc < 12:
                break
            h_opt, c_opt = optimize_local(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Final intensive refinement on best solution
    n = len(best_h)
    if time.time() < deadline - 15:
        ht = slsqp_opt(best_h, n, 500000, maxiter=400)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    if time.time() < deadline - 5:
        if n <= 120:
            best_h, best_c = pair_search(best_h, n, deadline - 2,
                [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 2,
                [0.001, 0.0005, 0.0002, 0.0001, 0.00005], radius=35)

    # Final SLSQP polish
    if time.time() < deadline - 3:
        ht = slsqp_opt(best_h, n, 1000000, maxiter=300)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
