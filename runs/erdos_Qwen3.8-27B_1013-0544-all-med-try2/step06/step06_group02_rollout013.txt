def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    max_shift = shifted.max()
    if max_shift > 500:

    return m + np.log(np.sum(np.exp(shifted))) / alpha

def simulated_annealing(h, n, deadline, T_init=0.05, T_end=0.001, n_iter=50000):
    """Simulated annealing with feasible moves."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    rng = np.random.RandomState(42)
    
    n_total = max(1, int((deadline - time.time()))
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        T = T_init * (T_end / T_init) ** (it / max(1, n_total))
        
        # Random feasible move: pick two random indices and swap part of their values
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        amount = rng.uniform(-0.02, 0.02)
        
        if h[i] + amount < 0 or h[i] + amount > 1 or h[j] - amount < 0 or h[j] - amount > 1:
            continue
        
        h[i] += amount
        h[j] -= amount
        
        c_new = compute_c5(h, n)
        delta = c_new - current_c
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_c = c_new
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i] -= amount
            h[j] += amount
    
    return best_h, best_c

    nf = len(best_h)
    if time.time() < deadline - 40:
        t5 = min(time.time() + 40, deadline - 40)
        best_h, best_c = full_pair_search(best_h, nf, t5,
                                         [0.002, 0.001, 0.0005, 0.0002],
                                         radius=80)

    # Phase 6: Final SLSQP + very fine search
    if time.time() < deadline - 10:
        nf = len(best_h)
        h6, c6 = slsqp_phase(best_h, nf, min(time.time() + 20, deadline - 10),
                            alpha=10000000, maxiter=1500)
        if c6 < best_c - 1e-16:
            best_c = c6
            best_h = h6

    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = full_pair_search(best_h, nf, deadline - 2,
                                         [0.0005, 0.0002, 0.0001, 0.00005],
                                         radius=60)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
