            grad[idx + la] += w * (1.0 - h[idx])

            if budget_s - (time.time() - t0) < 25:
                break
            order = np.random.permutation(n)
            improved = False
            for i in order:
                best_c5_local = compute_c5(h)
                best_local = h.copy()
                for d in deltas:
                    cand = h.copy()
                    cand[i] = np.clip(cand[i] + d, 0.0, 1.0)
                    cand = project_feasible(cand, target_sum)
                    c5 = compute_c5(cand)
                    if c5 < best_c5_local:
                        best_c5_local = c5
                        best_local = cand
                if best_local is not h:
                    h = best_local
                    improved = True
                if best_c5_local < best_c5:
                    best_c5 = best_c5_local
                    best_h = h.copy()
            if not improved and sweep > 5:
                break

    # Phase 4: Multi-resolution exploration
    if budget_s - (time.time() - t0) > 150:
        for n_try in [50, 60, 80, 100, 120, 65, 75]:
            if n_try < 20 or n_try == n or budget_s - (time.time() - t0) < 50:
                continue
            ts_t = n_try / 2.0
            dx_t = 2.0 / n_try
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t, n_try)}]
            for beta in [500, 3000]:
                if budget_s - (time.time() - t0) < 15:
                    break
                try:
                    res = minimize(obj_fn, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 3000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        n = n_try
                        dx = dx_t
                        target_sum = ts_t
                        bounds = bnds_t
                        cons = cons_t
                except Exception:
                    pass

    # Phase 5: Final high-beta SLSQP polish
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    bnds_b = Bounds(np.zeros(n_b), np.ones(n_b))
    cons_b = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_b, n_b)}]
    for beta in [5000, 20000, 50000]:
        if budget_s - (time.time() - t0) < 8:
            break
        try:
            res = minimize(obj_fn, best_h, args=(dx_b, beta), method='SLSQP',
                           bounds=bnds_b, constraints=cons_b,
                           options={'maxiter': 8000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, ts_b)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
        except Exception:
            pass

    # Final validation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
