import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def active_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=5000, ftol=1e-16, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    if top_idx is not None:
        obj = active_smooth_obj
        args = (n, top_idx, alpha)
    else:
        obj = smooth_max_obj
        args = (n, alpha)
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h_curr, n, lag):
    m = abs(lag)
    dx = 2.0 / n
    grad = np.zeros(n)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = (1.0 - 2.0 * h_curr) * dx
        return grad
    if lag > 0:
        grad[:n - m] -= h_curr[m:] * dx
        grad[m:] += (1.0 - h_curr[:n - m]) * dx
    else:
        grad[:n - m] += (1.0 - h_curr[m:]) * dx
        grad[m:] -= h_curr[:n - m] * dx
    return grad

def gradient_descent_active(h, n, top_idx, deadline, lr_init=0.005):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lr = lr_init
    stuck_count = 0
    for it in range(20000):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        active_c = c[top_idx]
        m = active_c.max()
        shifted = 2000 * (active_c - m)
        weights = np.exp(shifted)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wi, idx in zip(weights, top_idx):
            if wi < 1e-10:
                continue
            lag = idx - (n - 1)
            g = lag_gradient(h_curr, n, lag)
            grad += wi * g
        grad -= grad.mean()
        h_new = h_curr - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
            lr = min(lr * 1.03, 0.03)
            stuck_count = 0
        else:
            lr *= 0.85
            stuck_count += 1
            if lr < 1e-9 or stuck_count > 200:
                break
    return best_h, best_c

def lp_active_set(h_curr, n, top_idx, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
    for idx in top_idx:
        lag = idx - (n - 1)
        g = lag_gradient(h_curr, n, lag)
        cons.append(c[idx] + g @ (h_var - h_curr) <= t_var)
    cons.append(cp.norm(h_var - h_curr, np.inf) <= trust)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.ECOS)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = np.array(h_var.value).flatten()
    return project_feasible(h_new, n)

            best_h = h_new

        c = compute_all_lags(best_h, n)
        thresh = c.max() * 0.995
        top_idx = np.where(c >= thresh)[0]
        if len(top_idx) < 3:

        if len(top_idx) > 15:

        for alpha in [20000, 200000, 2000000]:
            if time.time() > deadline - 20:
                break
            h_new = slsqp_run(best_h, n, alpha, maxiter=3000, ftol=1e-16, top_idx=top_idx)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:

        gd_end = min(deadline - 10, time.time() + 50)

            else:
                trust *= 0.6
                if trust < 0.01:

    return project_feasible(best_h, n), best_c
