```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def obj_sum(x, n):
    return float(np.sum(x) - n / 2.0)

def obj_sum_jac(x, n):
    return np.ones(n)

def obj_full(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max(c, alpha)

def grad_lag(h, n, k):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
        return g
    m = abs(k)
    if m >= n:
        return g
    if k > 0:
        g[:m] = 1.0 - h[m:2*m]
        if n > 2*m:
            g[m:n-m] = (1.0 - h[2*m:n]) - h[:n-2*m]
        g[n-m:] = -h[:m]
    else:
        g[:m] = -h[m:2*m]
        if n > 2*m:
            g[m:n-m] = (1.0 - h[:n-2*m]) - h[2*m:n]
        g[n-m:] = 1.0 - h[:m]
    return g

def pgd_step(h, n, top_idx, base_step):
    """Projected gradient descent step on smooth max of top lags."""
    c = compute_all_lags(h, n)
    active = c[top_idx]
    # Softmax weights
    a_w = 2000.0
    w = np.exp(a_w * (active - active.max()))
    s = w.sum()
    if s < 1e-30:
        return h.copy(), compute_c5(h, n)
    w = w / s
    
    dx = 2.0 / n
    g = np.zeros(n)
    for idx, w_k in zip(top_idx, w):
        lag = idx - (n - 1)
        g += w_k * grad_lag(h, n, lag)
    g *= dx
    
    # Project onto sum constraint
    g = g - g.mean()
    
    # Project onto bounds
    at_lo = h < 1e-12
    at_hi = h > 1.0 - 1e-12
    g[at_lo & (g < 0)] = 0.0
    g[at_hi & (g > 0)] = 0.0
    
    # Line search
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = base_step
    for _ in range(30):
        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            step *= 1.5
        else:
            step *= 0.5
        if step < 1e-12:
            break
    return best_h, best_c

def pgd_loop(h, n, deadline, n_top=3):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    steps = [0.1, 0.05, 0.01, 0.005, 0.001, 0.0005, 0.0001, 0.00005]
    for base_step in steps:
        if time.time() > deadline:
            break
        for _ in range(200):
            if time.time() > deadline:
                break
            c = compute_all_lags(best_h, n)
            top_idx = np.argsort(c)[-n_top:]
            h_new, c_new = pgd_step(best_h, n, top_idx, base_step)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new.copy()
            else:

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

        g[:] = 1.0 - 2.0 * h
        return g
