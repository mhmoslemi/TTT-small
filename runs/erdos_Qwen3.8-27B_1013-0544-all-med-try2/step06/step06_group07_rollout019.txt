            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def con_fun(h, target_sum):
    return h.sum() - target_sum


def con_jac(h, target_sum):
    return np.ones(len(h))


def coord_descent_step(h, i, target_sum, lam):
    n = len(h)
    h_i = h[i]
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    n_shifts = 2 * n - 1
    slopes = np.zeros(n_shifts)
    for idx in range(n_shifts):
        s = idx - (n - 1)
        sl = 0.0
        if 0 <= i + s < n:
            sl += (1.0 - h[i + s])
        if 0 <= i - s < n:
            sl -= h[i - s]
        slopes[idx] = sl
    B = corr - h_i * slopes
    cur_sum = h.sum()
    t_grid = np.linspace(0.0, 1.0, 2001)
    lin = B[:, None] + t_grid[None, :] * slopes[:, None]
    mx = np.max(lin, axis=0)
    quad = lam * (cur_sum - target_sum + (t_grid - h_i)) ** 2
    total = mx + quad
    t_best = t_grid[np.argmin(total)]
    if 0.0 < t_best < 1.0:
        lo = max(0.0, t_best - 0.001)
        hi = min(1.0, t_best + 0.001)
        t_fine = np.linspace(lo, hi, 4001)
        lin_f = B[:, None] + t_fine[None, :] * slopes[:, None]
        mx_f = np.max(lin_f, axis=0)
        quad_f = lam * (cur_sum - target_sum + (t_fine - h_i)) ** 2
        total_f = mx_f + quad_f
        t_best = t_fine[np.argmin(total_f)]
    h[i] = t_best
    return h


def run_cd(h, target_sum, n_sweeps, lam, t_end):
    n = len(h)
    h = h.copy()
    order = np.random.permutation(n)
    for sw in range(n_sweeps):
        if time.time() > t_end:
            break
        for i in order:
            h = coord_descent_step(h, i, target_sum, lam)
        h = project_feasible(h, target_sum)
    return h


def slsqp_optimize(h, dx, target_sum, beta, maxiter=3000, ftol=1e-15):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
    try:
        res = minimize(obj_smooth, h, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n

    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    def try_update(h_new):
        nonlocal best_h, best_c5
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5

    # Phase 1: SLSQP coarse
    for beta in [20, 100, 400]:
        if rem() < 100:
            break
        h_new = slsqp_optimize(best_h, dx, target_sum, beta)
        try_update(h_new)

    # Phase 2: Coordinate descent (main workhorse)
    t_cd_end = t0 + budget_s * 0.55
    h_cd = best_h.copy()
    lam = 2000.0
    sweeps_done = 0
    while time.time() < t_cd_end:
        h_cd = run_cd(h_cd, target_sum, 50, lam, t_cd_end)
        sweeps_done += 50
        c5 = compute_c5(h_cd)
        if c5 < best_c5:
            best_h = h_cd.copy()
            best_c5 = c5

    # Phase 3: SLSQP refinement on best
    for beta in [1000, 5000]:
        if rem() < 40:
            break
        h_new = slsqp_optimize(best_h, dx, target_sum, beta, maxiter=4000, ftol=1e-16)
        try_update(h_new)

    # Phase 4: More coordinate descent
    t_cd2_end = min(t0 + budget_s * 0.75, time.time() + 120)
    h_cd2 = best_h.copy()
    while time.time() < t_cd2_end:
        h_cd2 = run_cd(h_cd2, target_sum, 50, 5000.0, t_cd2_end)
        c5 = compute_c5(h_cd2)
        if c5 < best_c5:
            best_h = h_cd2.copy()
            best_c5 = c5

    # Phase 5: Final SLSQP polish
    for beta in [10000, 50000]:
        if rem() < 20:
            break
        h_new = slsqp_optimize(best_h, dx, target_sum, beta, maxiter=6000, ftol=1e-16)
        try_update(h_new)

    # Phase 6: Try perturbed restarts
    if rem() > 60:
        for scale in [0.01, 0.02, 0.005]:
            if rem() < 40:
                break
            h_pert = best_h + np.random.randn(n) * scale
            h_pert = project_feasible(h_pert, target_sum)
            h_new = slsqp_optimize(h_pert, dx, target_sum, 2000, maxiter=2000)
            try_update(h_new)
            # Follow with short CD
            if rem() > 30:
                h_cd3 = best_h.copy()
                t_cd3_end = time.time() + 30
                while time.time() < t_cd3_end:
                    h_cd3 = run_cd(h_cd3, target_sum, 20, 3000.0, t_cd3_end)
                c5 = compute_c5(h_cd3)
                if c5 < best_c5:
                    best_h = h_cd3.copy()
                    best_c5 = c5

    # Phase 7: Try adjacent resolutions
    if rem() > 50:
        for n_try in [n + 5, n - 5, n + 10]:
            if n_try < 40 or rem() < 35:
                continue
            h_t = resample_h(best_h, n_try)
            ts_t = n_try / 2.0
            dx_t = 2.0 / n_try
            h_t = project_feasible(h_t, ts_t)
            h_new = slsqp_optimize(h_t, dx_t, ts_t, 1000, maxiter=2000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
                n = n_try
                target_sum = ts_t
                dx = dx_t
                if rem() > 30:
                    t_cd4_end = time.time() + 25
                    while time.time() < t_cd4_end:
                        best_h = run_cd(best_h, target_sum, 20, 3000.0, t_cd4_end)
                    try_update(best_h)

    # Phase 8: Final coordinate descent squeeze
    if rem() > 15:
        t_final_end = time.time() + max(5, rem() - 5)
        best_h = run_cd(best_h, target_sum, 100, 5000.0, t_final_end)
        try_update(best_h)

    return best_h, best_c5, len(best_h)
