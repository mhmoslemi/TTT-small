import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    ag = alpha * g
    ag_max = ag.max()
    exps = np.exp(ag - ag_max)
    w = exps / exps.sum()
    F = (ag_max + np.log(exps.sum())) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        d = k - (n - 1)
        if d >= 0:
            grad[d:] += wk * dx * (1.0 - h[:n - d])
            grad[:n - d] -= wk * dx * h[d:]
        else:
            dp = -d
            grad[:n - dp] += wk * dx * (1.0 - h[dp:])
            grad[dp:] -= wk * dx * h[:-dp]
    return F, grad

def lbfgsb_step(h0, n, alpha, maxiter, mu):

        return F + mu * (x.sum() - target) ** 2

        _, g = smooth_max_obj_and_grad(x, alpha)
        return g + 2 * mu * (x.sum() - target)

                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})

        return project_feasible(h0, n)

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    ag = alpha * g
    m = ag.max()
    e = np.exp(ag - m)
    w = e / e.sum()
    F = (m + np.log(e.sum())) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            grad[d:] += wk * dx * (1.0 - h[:n-d])
            grad[:n-d] -= wk * dx * h[d:]
        else:
            dp = -d
            grad[:n-dp] += wk * dx * (1.0 - h[dp:])
            grad[dp:] -= wk * dx * h[:-dp]
    return F, grad

def lbfgsb_opt(h0, n, alpha, maxiter, mu, deadline):

        F, _ = smooth_max_obj_grad(x, alpha)
        return F + mu * (x.sum() - target) ** 2

        _, g = smooth_max_obj_grad(x, alpha)
        return g + 2.0 * mu * (x.sum() - target)
    bounds = [(0.0, 1.0)] * n

    w = e / e.sum()
    F = (m + np.log(e.sum())) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            grad[d:] += wk * dx * (1.0 - h[:n - d])
            grad[:n - d] -= wk * dx * h[d:]
        else:
            dp = -d
            grad[:n - dp] += wk * dx * (1.0 - h[dp:])
            grad[dp:] -= wk * dx * h[:-dp]
    return F, grad

def lbfgsb_opt(h0, n, alpha, maxiter, mu, deadline):
    target = n / 2.0

    def obj(x):
        F, _ = smooth_max_obj_grad(x, alpha)
        return F + mu * (x.sum() - target) ** 2

    def jac(x):
        _, g = smooth_max_obj_grad(x, alpha)
        return g + 2.0 * mu * (x.sum() - target)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='L-BFGS-B', jac=jac, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

def subgrad_polish(h, n, deadline, max_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.08
    dx = 2.0 / n
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full') * dx
        m = g.max()
        k_max = int(np.argmax(g))
        d = k_max - (n - 1)
        sg = np.zeros(n)
        if d >= 0:
            sg[d:] += (1.0 - h[:n - d])
            sg[:n - d] -= h[d:]
        else:
            dp = -d
            sg[:n - dp] += (1.0 - h[dp:])
            sg[dp:] -= h[:-dp]
        sg *= dx
        ns = np.linalg.norm(sg)
        if ns < 1e-14:
            break
        ok = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            hn = project_feasible(h - lr * f * sg / ns, n)
            cn = compute_c5(hn, n)
            if cn < best_c - 1e-15:
                h, best_c, best_h = hn, cn, hn.copy()
                ok = True
                break
        if not ok:
            lr *= 0.93
            if lr < 1e-12:
                break
    return best_h, best_c

def pair_sweep(h, n, deadline, delta=0.01):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for _ in range(50):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full') * dx
        k_max = int(np.argmax(g))
        dm = k_max - (n - 1)
        improved = False
        idx = np.random.RandomState(int(time.time() * 1000) % 10000).permutation(n)
        for ii in range(min(200, len(idx))):
            i = idx[ii]
            for j in range(n):
                if j == i:
                    continue
                if time.time() > deadline:
                    return h, best_c
                if h[i] + delta <= 1.0 and h[j] - delta >= 0.0:
                    h[i] += delta
                    h[j] -= delta
                    c = compute_c5(h, n)
                    if c < best_c:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
                if h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
                    h[i] -= delta
                    h[j] += delta
                    c = compute_c5(h, n)
                    if c < best_c:
                        best_c = c
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 250
        h = np.full(n, 0.5)

    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS-B with smooth max, increasing alpha (50% budget)
    p1 = t0 + 0.50 * (budget_s - 10)
    alphas = [50, 200, 500, 2000, 5000, 20000, 50000, 200000, 500000]
    for alpha in alphas:
        if time.time() > p1:
            break
        ho = lbfgsb_opt(best_h.copy(), n, alpha, 400, 1000.0, p1)
        c = compute_c5(ho, n)
        if c < best_c:
            best_c, best_h = c, ho

    # Phase 2: Subgradient targeting exact max (15% budget)
    p2 = t0 + 0.65 * (budget_s - 10)
    ho, c = subgrad_polish(best_h.copy(), n, p2, 8000)
    if c < best_c:
        best_c, best_h = c, ho

    # Phase 3: Multi-resolution exploration (20% budget)
    p3 = t0 + 0.85 * (budget_s - 10)
    for nh in [150, 200, 300, 400, 500]:
        if time.time() > p3:
            break
        nc = len(best_h)
        xo = np.linspace(0, 2, nc)
        xn = np.linspace(0, 2, nh)
        hi = project_feasible(np.interp(xn, xo, best_h), nh)
        for alpha in [3000, 30000, 100000]:
            if time.time() > p3 - 5:
                break
            ho = lbfgsb_opt(hi, nh, alpha, 300, 1000.0, p3 - 5)
            c = compute_c5(ho, nh)
            if c < best_c:
                best_c, best_h = c, ho
                hi = ho
        if time.time() < p3 - 10:
            ho, c = subgrad_polish(best_h.copy(), len(best_h), p3, 2000)
            if c < best_c:
                best_c, best_h = c, ho

    # Phase 4: Pair search + final polish (15% budget)
    if time.time() < deadline - 40:
        ho, c = pair_sweep(best_h.copy(), len(best_h), deadline - 40, delta=0.01)
        if c < best_c:
            best_c, best_h = c, ho
        ho, c = pair_sweep(best_h.copy(), len(best_h), deadline - 20, delta=0.005)
        if c < best_c:
            best_c, best_h = c, ho
    if time.time() < deadline:
        nc = len(best_h)
        ho = lbfgsb_opt(best_h.copy(), nc, 1000000, 300, 5000.0, deadline)
        c = compute_c5(ho, nc)
        if c < best_c:
            best_c, best_h = c, ho

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
