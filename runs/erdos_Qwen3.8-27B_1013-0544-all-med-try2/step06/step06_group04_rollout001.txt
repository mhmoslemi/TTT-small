        lag = k_star - (n - 1)
        grad = lag_gradient(h, n, lag)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad_unit = grad / gnorm
        found = False
        for ss in [step, step * 0.5, step * 0.25, step * 0.1]:
            h_new = project_feasible(h - ss * grad_unit, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                step = min(ss * 1.3, 0.05)
                stall = 0
                found = True
                break
        if not found:
            step *= 0.6
            stall += 1
            if stall > 50 or step < 1e-7:
                break
    return best_h, best_c

def pair_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            stride = max(1, (n - m) // 300)
            for i in range(0, n - m, stride):
                if time.time() > deadline:
                    break
                j = i + m
                if lag > 0:
                    if h[i] > h[j]:
                        delta = min(d, (h[i] - h[j]) * 0.4)
                        if delta < 1e-9:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        if compute_c5(h, n) < best_c - 1e-14:
                            best_c = compute_c5(h, n)
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                else:
                    if h[j] > h[i]:
                        delta = min(d, (h[j] - h[i]) * 0.4)
                        if delta < 1e-9:
                            continue
                        h[j] -= delta
                        h[i] += delta
                        if compute_c5(h, n) < best_c - 1e-14:
                            best_c = compute_c5(h, n)
                            improved = True
                        else:
                            h[j] += delta
                            h[i] -= delta
    return h, best_c

def coordinate_ascent(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    n_terms = n
    for sweep in range(200):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        any_improve = False
        for idx in range(0, n, max(1, n // 200)):
            if time.time() > deadline:
                break
            i = idx
            j = (idx + m) % n
            if i == j:
                continue
            if lag > 0:
                if i >= n - m or j < m:
                    continue
            for delta in [0.01, 0.005, 0.002]:
                if lag > 0:
                    if h[i] > h[j]:
                        h[i] -= delta
                        h[j] += delta
                        if compute_c5(h, n) < best_c - 1e-14:
                            best_c = compute_c5(h, n)
                            any_improve = True
                            break
                        h[i] += delta
                        h[j] -= delta
                else:
                    if h[j] > h[i]:
                        h[j] -= delta
                        h[i] += delta
                        if compute_c5(h, n) < best_c - 1e-14:
                            best_c = compute_c5(h, n)
                            any_improve = True
                            break
                        h[j] += delta
                        h[i] -= delta
        if not any_improve:
            break
    return h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # SLSQP with increasing sharpness
    for alpha in [3000, 15000, 60000]:
        if time.time() > deadline - 15:
            break
        h_new = slsqp_opt(best_h, n, alpha, maxiter=2500, ftol=1e-14)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new
    # Gradient search
    if time.time() < deadline - 10:
        h_g, c_g = gradient_search(best_h, n, min(deadline - 5, time.time() + 40))
        if c_g < best_c:
            best_c = c_g
            best_h = h_g
    # Pair search
    if time.time() < deadline - 8:
        h_p, c_p = pair_search(best_h, n, min(deadline - 3, time.time() + 35))
        if c_p < best_c:
            best_c = c_p
            best_h = h_p
    # Final SLSQP
    if time.time() < deadline - 5:
        h_f = slsqp_opt(best_h, n, 200000, maxiter=3000, ftol=1e-16)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    # Final polish
    if time.time() < deadline - 2:
        h_fin, c_fin = pair_search(best_h, n, deadline - 1)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = min(350, (deadline - t0) * 0.4)
        best_h, best_c = optimize_at_n(h_init, n_init, t0 + alloc)
    # Try other resolutions
    for n_try in [250, 350, 200, 400]:
        if time.time() > deadline - 50:
            break
        alloc = min(100, (deadline - time.time() - 25) * 0.25)
        if alloc < 20:
            break
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, best_h)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Multi-start at best resolution
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        for r in range(8):
            if time.time() > deadline - 20:
                break
            if r % 3 == 0:
                hp = best_h + rng.normal(0, 0.015, n)
            elif r % 3 == 1:
                hp = best_h.copy()
                bs = rng.randint(0, max(1, n - 25))
                bl = rng.randint(8, 25)
                hp[bs:bs + bl] = 1.0 - hp[bs:bs + bl]
            else:
                hp = best_h + rng.normal(0, 0.04, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 20) * 0.35)
            if alloc < 12:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Final refinement
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        h_new = slsqp_opt(best_h, n, 500000, maxiter=4000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 3:
            best_h, best_c = gradient_search(best_h, n, deadline - 2)
        if time.time() < deadline - 1:
            best_h, best_c = pair_search(best_h, n, deadline - 0.5)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
