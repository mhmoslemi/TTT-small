    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):

        if abs(diff) < 1e-14:

            if tot < 1e-14:

def grad_smooth_max(h, n, alpha):
    """Compute gradient of smooth max of correlation values."""
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    # Gradient of c_k with respect to h_i
    # c_k = sum_j h[j]*(1-h[j+k]) * dx for valid j
    # d(c_k)/d(h_i) = (1-h[i+k])*dx - h[i-k]*dx (for valid indices)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-15:
            continue
        k = m - (n - 1)
        if k >= 0:

            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

And `np.correlate(h, 1-h, mode="full")` has length 2n-1. The shift k corresponds to index m = k + n - 1.

For k=0: m = n-1, this is the "zero lag" correlation = Σ h[i](1-h[i])
For k>0: m = n-1+k, this is Σ_{valid i} h[i](1-h[i+k])
For k<0: m = n-1+k, this is Σ_{valid i} h[i](1-h[i+k]) = Σ_{valid i} h[i+k](1-h[i])

The max is over all these. For a good solution, we want all of these to be small.

Notice that c_0 = Σ h[i](1-h[i]) = Σ h[i] - Σ h[i]² = n/2 - Σ h[i]².
To minimize c_0, we want to maximize Σ h[i]², which means making h as "spiky" as possible (values concentrated at 0 and 1).

But making h spiky increases c_k for k ≠ 0 (because adjacent values in h will be very different). So there's a tension.

This suggests the optimal h is a "balanced" function that's neither too smooth nor too spiky.
