        return np.max(c_top + delta * slope)

    for _ in range(50):

        m2 = a + gr * (b - a)
        if obj(m1) < obj(m2):
            b = m2
        else:
            a = m1
        if b - a < 1e-10:
            break

        if abs(diff) < 1e-14:
            break

            tot = room.sum()

        else:
            tot = h.sum()
            if tot < 1e-15:

**Alternative idea: Use the dual problem**

The problem is: min_h max_k ∫h(x)(1-h(x+k))dx subject to 0≤h≤1, ∫h=1.

**Alternative idea: Exploit the structure of the cross-correlation**

The cross-correlation of h with (1-h) can be written as:
∫h(x)(1-h(x+k))dx = ∫h(x)dx - ∫h(x)h(x+k)dx = 1 - (h ⋆ h)(k)

where (h ⋆ h)(k) = ∫h(x)h(x+k)dx is the autocorrelation.

So C₅ = max_k [1 - (h⋆h)(k)] = 1 - min_k (h⋆h)(k)

Wait, that's not right either. The cross-correlation in the discrete case gives values at different lags, and the integral of h(x)h(x+k) is the autocorrelation at lag k.

Actually, let me be more careful. In the continuous case:
∫₀² h(x)(1-h(x+k))dx - we need to be careful about the domain.

In the discretized case with numpy.correlate(h, 1-h, mode='full'):
c[m] = sum_n h[n+m-n1] * (1-h[n]) * dx

This is NOT simply 1 - autocorrelation because the sum is over a limited range (not all n from 0 to n-1).

For m = n-1 (k=0): c[n-1] = sum_n h[n]* (1-h[n]) * dx = (sum h - sum h²) * dx

For m ≠ n-1 (k ≠ 0): the sum is over a different range.

So it's not exactly 1 - autocorrelation. The boundary effects matter.

        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
            h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(250, 0.5)
        n = 250

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS-B with increasing alpha
    t1 = min(t0 + 200, deadline - 400)
    h, best_c = lbfgs_optimize(h, n, t1, max_iters=200)
    best_h = h.copy()

    # Phase 2: Exact 1D pair search (key improvement)
    if time.time() < deadline - 250:
        t2 = min(time.time() + 300, deadline - 250)
        h, best_c = pair_search_exact(h, n, t2, top_m=40, radius=25, n_random=5, max_passes=15)
        best_h = h.copy()

    # Phase 3: L-BFGS-B refinement
    if time.time() < deadline - 200:
        t3 = min(time.time() + 120, deadline - 200)
        h2, c2 = lbfgs_optimize(h, n, t3, max_iters=150)
        if c2 < best_c:
            best_c = c2
            best_h = h2
            h = h2

    # Phase 4: More exact pair search with tighter parameters
    if time.time() < deadline - 150:
        t4 = min(time.time() + 180, deadline - 150)
        h, best_c = pair_search_exact(h, n, t4, top_m=30, radius=15, n_random=3, max_passes=10)
        best_h = h.copy()

    # Phase 5: SLSQP polish
    if time.time() < deadline - 80:
        t5 = min(time.time() + 60, deadline - 80)
        h5, c5 = slsqp_optimize(h, n, t5, alpha=200000, maxiter=800)
        if c5 < best_c:
            best_c = c5
            best_h = h5
            h = h5

    # Phase 6: Final pair search
    if time.time() < deadline - 20:
        t6 = min(time.time() + 60, deadline - 5)
        h, best_c = pair_search_exact(h, n, t6, top_m=25, radius=10, n_random=2, max_passes=8)

    # Phase 7: Try perturbing and re-optimizing
    if time.time() < deadline - 10:
        h7, c7 = perturb_and_reopt(h, n, deadline - 2, rng, sigma=0.03)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
