import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def coord_descent(h, max_passes=200):
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target).copy()
    corr = get_corr(h)
    lags = np.arange(2 * n - 1) - (n - 1)
    best_local = np.max(corr)
    
    for p in range(max_passes):
        improved = False
        for i in range(n):
            idx1 = i + lags
            idx2 = i - lags
            s = np.zeros(2 * n - 1)
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            s[v1] = 1.0 - h[idx1[v1]]
            s[v2] -= h[idx2[v2]]
            base = corr - h[i] * s
            lo, hi = 0.0, 1.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(base + m1 * s) < np.max(base + m2 * s):
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) * 0.5
            if abs(t_opt - h[i]) > 1e-10:
                delta = t_opt - h[i]
                new_max = np.max(corr + delta * s)
                if new_max < best_local - 1e-14:
                    corr = corr + delta * s
                    h[i] = t_opt
                    best_local = new_max
                    improved = True
        if not improved:
            break
        h = project_feasible(h, target)
        corr = get_corr(h)
        best_local = np.max(corr)
    return h, float(np.max(get_corr(h)) * (2.0 / n))

def lbfgs_optimize(h, dx, target, beta, maxiter=3000):
    n = len(h)
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = get_corr(x)
        return smooth_max(corr, beta) * dx + 100.0 * (x.sum() - target) ** 2
    try:
        res = minimize(obj, h, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return h

def resample(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent from initial
    if tl() > 300:
        h_cd, c5_cd = coord_descent(best_h, max_passes=300)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 2: L-BFGS-B refinement
    n = len(best_h)
    dx = 2.0 / n
    target = n / 2.0
    for beta in [50, 200, 1000, 5000]:
        if tl() < 100:
            break
        h_new = lbfgs_optimize(best_h, dx, target, beta, maxiter=5000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5
        # Follow with coordinate descent
        if tl() > 150:
            h_cd, c5_cd = coord_descent(h_new, max_passes=200)
            if c5_cd < best_c5:
                best_h, best_c5 = h_cd, c5_cd

    # Phase 3: Random restarts with diverse initializations
    n_restarts = 30
    for r in range(n_restarts):
        if tl() < 120:
            break
        # Diverse random starts
        mode = r % 5
        if mode == 0:
            h_try = best_h + np.random.randn(n) * 0.15
        elif mode == 1:
            h_try = np.full(n, 0.5) + np.random.randn(n) * 0.3
        elif mode == 2:
            # Block-like
            h_try = np.zeros(n)
            half = n // 2
            h_try[:half] = 1.0
            h_try += np.random.randn(n) * 0.2
        elif mode == 3:
            # Periodic-like
            h_try = np.tile(0.5 + 0.3 * np.sin(2 * np.pi * np.arange(n) / n * 2), 1)
            h_try += np.random.randn(n) * 0.1
        else:
            # Smooth random
            h_try = np.cumsum(np.random.randn(n))
            h_try = (h_try - h_try.min()) / (h_try.max() - h_try.min() + 1e-10)
        
        h_try = np.clip(h_try, 0, 1)
        h_try = project_feasible(h_try, target)
        
        h_cd, c5_cd = coord_descent(h_try, max_passes=80)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd
            # Refine with L-BFGS-B
            if tl() > 100:
                for beta in [500, 2000]:
                    if tl() < 60:
                        break
                    h_new = lbfgs_optimize(best_h, dx, target, beta, maxiter=3000)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new, c5
                    h_cd2, c5_cd2 = coord_descent(h_new, max_passes=100)
                    if c5_cd2 < best_c5:
                        best_h, best_c5 = h_cd2, c5_cd2

    # Phase 4: Multi-resolution
    for n_try in [n + 10, n + 20, n - 10, n + 40]:
        if n_try < 30 or tl() < 80:
            continue
        dx_t = 2.0 / n_try
        target_t = n_try / 2.0
        h_t = resample(best_h, n_try)
        h_t = project_feasible(h_t, target_t)
        h_cd, c5_cd = coord_descent(h_t, max_passes=150)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd
        if tl() > 80:
            h_new = lbfgs_optimize(h_cd, dx_t, target_t, 1000, maxiter=3000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
            h_cd2, c5_cd2 = coord_descent(h_new, max_passes=150)
            if c5_cd2 < best_c5:
                best_h, best_c5 = h_cd2, c5_cd2

    # Phase 5: Final coordinate descent polish
    if tl() > 30:
        n_final = len(best_h)
        target_final = n_final / 2.0
        best_h, best_c5 = coord_descent(best_h, max_passes=500)

    return best_h, best_c5, len(best_h)
