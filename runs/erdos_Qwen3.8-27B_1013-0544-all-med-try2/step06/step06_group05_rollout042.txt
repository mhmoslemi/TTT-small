from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def fast_corr_delta(h, p, q, d, n):
    """Compute change in raw correlation (not scaled by dx) when h[p]+=d, h[q]-=d."""
    off = np.arange(-(n-1), n)
    ip = p + off
    ip2 = p - off
    iq = q + off
    iq2 = q - off
    
    delta = np.zeros(2*n - 1)
    m1 = (ip >= 0) & (ip < n)
    if np.any(m1):
        delta[m1] += d * (1.0 - h[ip[m1]])
    m2 = (ip2 >= 0) & (ip2 < n)
    if np.any(m2):

    m3 = (iq >= 0) & (iq < n)
    if np.any(m3):
        delta[m3] += -d * (1.0 - h[iq[m3]])
    m4 = (iq2 >= 0) & (iq2 < n)
    if np.any(m4):
        delta[m4] += d * h[iq2[m4]]
    return delta

def fast_pair_search(h, n, deadline, deltas, max_passes=50000):
    """Fast pair search with incremental correlation update."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    best_c_raw = c.max()
    best_c = best_c_raw * dx
    h = h.copy()
    
    for d in deltas:
        passes = 0
        for _pass in range(max_passes):
            if time.time() > deadline:

            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):

                            h[j] = hj2

            if not improved:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def corr_delta(h, p, q, d, n):
    """O(n) computation of correlation change when h[p]+=d, h[q]-=d (using old h)."""
    off = np.arange(-(n-1), n)
    # h[p] += d contributions
    ip = p + off
    ip2 = p - off
    # h[q] -= d contributions  
    iq = q + off
    iq2 = q - off
    delta = np.zeros(2*n - 1)
    m = (ip >= 0) & (ip < n)
    if m.any(): delta[m] += d * (1.0 - h[ip[m]])
    m = (ip2 >= 0) & (ip2 < n)
    if m.any(): delta[m] += -d * h[ip2[m]]
    m = (iq >= 0) & (iq < n)
    if m.any(): delta[m] += -d * (1.0 - h[iq[m]])
    m = (iq2 >= 0) & (iq2 < n)
    if m.any(): delta[m] += d * h[iq2[m]]
    return delta

def fast_pair_search(h, n, deadline, deltas):
    dx = 2.0 / n
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full')
    best = c.max()
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best * dx
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        delta = corr_delta(h, i, j, d, n)
                        nm = (c + delta).max()
                        if nm < best - 1e-13:
                            best = nm
                            c += delta
                            h[i], h[j] = hi, hj
                            improved = True
                    hi = h[i] - d
                    hj = h[j] + d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        delta = corr_delta(h, i, j, -d, n)
                        nm = (c + delta).max()
                        if nm < best - 1e-13:
                            best = nm
                            c += delta
                            h[i], h[j] = hi, hj
                            improved = True
            if not improved:
                break
    
    return h, best * dx

def subgrad_step(h, n, c, step=0.01):
    """Take a subgradient step on the max correlation."""
    k_max = c.argmax()
    off = k_max - (n - 1)
    # Subgradient: g[p] = 1 - h[p+off] - h[p-off] (with boundary)
    g = np.ones(n)
    ip = np.arange(n) + off
    valid = (ip >= 0) & (ip < n)
    g[valid] -= h[ip[valid]]
    ip2 = np.arange(n) - off
    valid2 = (ip2 >= 0) & (ip2 < n)
    g[valid2] -= h[ip2[valid2]]
    # Step: h -= step * g, then project
    h_new = h - step * g
    h_new = project_feasible(h_new, n)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sc(x, n):
    return float(np.sum(x) - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        r = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                     bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(r.x, n)
    except:
        return h.copy()

def corr_diff(h, p, q, d, n):
    """First-order approximation of correlation change for h[p]+=d, h[q]-=d."""
    off = np.arange(-(n-1), n)
    delta = np.zeros(2*n - 1)
    ip = p + off
    m = (ip >= 0) & (ip < n)
    if m.any(): delta[m] += d * (1.0 - h[ip[m]])
    ip2 = p - off
    m = (ip2 >= 0) & (ip2 < n)
    if m.any(): delta[m] += -d * h[ip2[m]]
    iq = q + off
    m = (iq >= 0) & (iq < n)
    if m.any(): delta[m] += -d * (1.0 - h[iq[m]])
    iq2 = q - off
    m = (iq2 >= 0) & (iq2 < n)
    if m.any(): delta[m] += d * h[iq2[m]]
    return delta

def pair_search(h, n, deadline, deltas, recompute_every=50):
    """Pair search with fast O(n) correlation updates."""
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full')
    best = c.max()
    dx = 2.0 / n
    moves = 0
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best * dx
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        delta = corr_diff(h, i, j, d, n)
                        nm = (c + delta).max()
                        if nm < best - 1e-10:
                            best = nm
                            c += delta
                            h[i], h[j] = hi, hj
                            improved = True
                            moves += 1
                            if moves % recompute_every == 0:
                                c = np.correlate(h, 1 - h, mode='full')
                                best = c.max()
                    hi = h[i] - d
                    hj = h[j] + d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        delta = corr_diff(h, i, j, -d, n)
                        nm = (c + delta).max()
                        if nm < best - 1e-10:
                            best = nm
                            c += delta
                            h[i], h[j] = hi, hj
                            improved = True
                            moves += 1
                            if moves % recompute_every == 0:
                                c = np.correlate(h, 1 - h, mode='full')
                                best = c.max()
            if not improved:
                break
    
    # Final exact computation
    c = np.correlate(h, 1 - h, mode='full')
    return h, c.max() * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sc(x, n):
    return float(np.sum(x) - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        r = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

def corr_diff(h, p, q, d, n):
    off = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    ip = p + off
    m = (ip >= 0) & (ip < n)
    if m.any():
        delta[m] += d * (1.0 - h[ip[m]])
    ip2 = p - off
    m = (ip2 >= 0) & (ip2 < n)
    if m.any():
        delta[m] += -d * h[ip2[m]]
    iq = q + off
    m = (iq >= 0) & (iq < n)
    if m.any():
        delta[m] += -d * (1.0 - h[iq[m]])
    iq2 = q - off
    m = (iq2 >= 0) & (iq2 < n)
    if m.any():
        delta[m] += d * h[iq2[m]]
    return delta

def pair_search(h, n, deadline, deltas, recompute_every=30):
