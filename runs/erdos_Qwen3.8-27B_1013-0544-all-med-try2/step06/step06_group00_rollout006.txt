    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary greedy (short)
    p1 = t0 + 12.0
    hb, cb = greedy_binary(h0, n, p1, max_sweeps=120)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: Projected gradient warm-up
    p2 = t0 + 60.0
    h = best_h.copy()
    for alpha in [200, 2000, 20000]:
        if time.time() > p2 - 5:
            break
        h, c = pg_phase(h, n, alpha, 500, 0.05, p2 - 5)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: SLSQP main optimization with careful alpha schedule
    p3 = t0 + total * 0.55
    h = best_h.copy()
    alphas_main = [500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000, 100000000]
    prev_c = compute_c5(h, n)
    for alpha in alphas_main:
        if time.time() > p3 - 15:
            break
        h = slsqp_optimize(h, n, alpha, 2000, 1e-14)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if abs(c - prev_c) < 1e-10:
            prev_c = c
        else:
            prev_c = c

    # Phase 4: Multiple restarts with decreasing noise
    p4 = t0 + total * 0.80
    n_restarts = 12
    for trial in range(n_restarts):
        if time.time() > p4 - 25:
            break
        scale = 0.03 * (0.7 ** trial)
        h = best_h + scale * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [50000, 500000, 5000000, 50000000]:
            if time.time() > p4 - 20:
                break
            h = slsqp_optimize(h, n, alpha, 1000, 1e-14)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 5: Try different resolutions
    p5 = dl - 30
    for nh in [300, 500]:
        if time.time() > p5 - 25:
            break
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = project_feasible(np.interp(xn, xo, best_h, period=2.0), nh)
        # Quick binary search at new resolution
        hhb, hhcb = greedy_binary(hh, nh, min(time.time() + 10, p5 - 20), max_sweeps=50)
        if hhcb < compute_c5(hh, nh):
            hh = hhb
        for alpha in [10000, 100000, 1000000, 10000000]:
            if time.time() > p5 - 20:
                break
            hh = slsqp_optimize(hh, nh, alpha, 800, 1e-14)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c = ch
                best_h = hh.copy()
                n = nh

    # Phase 6: Final fine-tuning
    p6 = dl - 2
    h = best_h.copy()
    for alpha in [10000000, 50000000, 200000000]:
        if time.time() > p6 - 3:
            break
        h = slsqp_optimize(h, n, alpha, 3000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # One more restart if time permits
    if time.time() < dl - 10:
        for trial in range(3):
            if time.time() > dl - 10:
                break
            h = best_h + 0.005 * rng.randn(n)
            h = project_feasible(h, n)
            h = slsqp_optimize(h, n, 100000000, 2000, 1e-16)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
