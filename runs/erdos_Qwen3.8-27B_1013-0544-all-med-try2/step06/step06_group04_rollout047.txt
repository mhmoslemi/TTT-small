import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    dx = 2.0 / n
    m = abs(lag)
    grad = np.zeros(n)
    if m >= n:
        return grad
    if lag >= 0:
        grad[:n - m] += (1.0 - h[lag:])
        grad[lag:] -= h[:n - lag]
    else:
        m = -lag
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    return dx * grad

def smooth_max_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)

    if mx > 500:
        logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
    else:
        logsum = np.log(np.sum(np.exp(shifted)))
    return float(m + logsum / alpha)

    return float(np.sum(h) - n / 2.0)
