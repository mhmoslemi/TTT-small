import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} approximately."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_logsumexp(h, n, alpha):
    """Compute gradient of (1/alpha)*log(sum(exp(alpha*v))) where v = corr*h_dx."""
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights
    v_shifted = alpha * (v - v.max())
    p = np.exp(v_shifted)
    p /= p.sum()
    # Gradient: sum_j p_j * dv_j/dh_i
    # dv_{n-1+k}/dh_i = dx * [(1-h[i-k]) - h[i+k]] (with bounds)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        if abs(p[n-1+k]) < 1e-15:
            continue
        pk = p[n-1+k]
        # For each i, dv_{n-1+k}/dh_i = dx * [(1-h[i-k]) - h[i+k]]
        # with boundary conditions (out of [0,n) is 0)
        # i-k must be in [0,n): i in [k, n+k)
        # i+k must be in [0,n): i in [-k, n-k)
        # Valid i: max(0,k) <= i < min(n, n+k) for first term
        #            max(0,-k) <= i < min(n, n-k) for second term
        i_start = max(0, k)
        i_end = min(n, n + k)
        if i_end > i_start:
            grad[i_start:i_end] += pk * dx * (1.0 - h[i_start - k:i_end - k])
        i_start2 = max(0, -k)
        i_end2 = min(n, n - k)
        if i_end2 > i_start2:
            grad[i_start2:i_end2] -= pk * dx * h[i_start2 + k:i_end2 + k]
    return grad / alpha  # chain rule: d/dh [(1/alpha)*log(sum(exp(alpha*v))] = sum p_j * dv_j/dh_i

Actually, I want to try one more thing: instead of log-sum-exp, use the "sum of top-k" which is a tighter upper bound on the max.

sum_topk(v, k) = sum of the k largest values of v.

This equals: min_λ {k*λ + sum_i max(v_i - λ, 0)}

The gradient with respect to v is: (1, 1, ..., 1, 0, ..., 0) where there are k ones (at the top-k positions).

This is actually easier to compute than log-sum-exp! And for k=1, it's exactly the max (but non-differentiable). For k=2 or k=3, it's a smooth approximation that's tighter than log-sum-exp.

But it's still non-differentiable at ties. In practice, ties are measure zero, so we can just pick the top-k positions and use the indicator.

Let me use k=3 (sum of top 3) as a practical smooth approximation.

Wait, sum of top-k is NOT differentiable. Its subgradient is the set of all convex combinations of the top-k indicator vectors. For optimization, we can just use one of them (e.g., the one at the current top-k).

So the "gradient" (subgradient) is:
∂f/∂v_i = 1{i ∈ top-k}

And then ∂f/∂h_i = sum_j ∂f/∂v_j * ∂v_j/∂h_i = sum_{j: v_j ∈ top-k} ∂v_j/∂h_i

This is easy to compute! Just sum the contributions from the top-k lags.

- At each iteration, find the top-k lags
- Compute the gradient as the sum of their individual gradients
- Take a projected gradient step

This is faster than log-sum-exp (no exp computation) and gives a tighter objective.

The downside is that it can "jump" between different sets of top-k lags, causing oscillation. To handle this, I can:

def topk_grad(h, n, k):
    """Compute subgradient of sum of top-k values of v=correlate(h,1-h)*dx"""
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode='full') * dx
    # Find top-k indices
    top_idx = np.argpartition(v, -k)[-k:]
    # Compute gradient
    grad = np.zeros(n)
    n1 = n - 1
    for idx in top_idx:
        kk = idx - n1
        i1 = max(0, kk)
        i2 = min(n, n + kk)

        i2b = min(n, n - kk)
        if i2b > i1b:
            grad[i1b:i2b] -= h[i1b+kk:i2b+kk]
