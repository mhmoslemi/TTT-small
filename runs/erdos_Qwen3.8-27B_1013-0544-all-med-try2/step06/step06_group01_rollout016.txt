def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_delta(h, i, j, n):
    """Compute change in correlation when swapping h[i]=0, h[j]=1 to h[i]=1, h[j]=0."""
    m_range = np.arange(-(n-1), n)
    delta = np.zeros(2*n - 1)
    
    im = i + m_range
    msk = (im >= 0) & (im < n)
    if msk.any():
        delta[msk] += 1.0 - h[im[msk]]
    
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    if msk.any():
        delta[msk] -= h[imn[msk]]
    
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    if msk.any():
        delta[msk] -= 1.0 - h[jm[msk]]
    
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    if msk.any():
        delta[msk] += h[jmn[msk]]
    
    # Cross terms
    delta[n-1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2*n-1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2*n-1:
        delta[k2] += 1.0
    
    return delta

    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1-h, mode='full').copy()
    cur_c = corr.max() * dx
    m_range = np.arange(-(n-1), n)
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Shuffle order for diversity
        rng.shuffle(zeros_list)
        rng.shuffle(ones_list)
        
        for i in zeros_list:
            if h[i] != 0:
                continue
            for j in ones_list:
                if h[j] != 1:
                    continue
                delta = compute_delta(h, i, j, n)
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    ones_list.remove(j)
                    ones_list.append(i)
                    zeros_list.remove(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return h, cur_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = np.max(sh)
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    m_range = np.arange(-(n - 1), n)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        rng.shuffle(zeros)
        rng.shuffle(ones)
        for i in zeros:
            if h[i] != 0:
                continue
            for j in ones:
                if h[j] != 1:
                    continue
                d = make_delta(h, i, j, n, m_range)
                nc = (corr + d).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur_c = nc
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, cur_c

def sim_anneal(h_bin, n, deadline, T0=0.008, cooling=0.9998, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h_bin.copy()
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    T = T0
    stuck = 0
    while time.time() < deadline:
        if not zeros or not ones:
            break
        ri = rng.randint(len(zeros))
        rj = rng.randint(len(ones))
        i = zeros[ri]
        j = ones[rj]
        d = make_delta(h, i, j, n, m_range)
        new_corr = corr + d
        new_c = new_corr.max() * dx
        dc = new_c - cur_c
        if dc <= 0 or rng.random() < np.exp(-dc / max(T, 1e-15)):
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = new_c
            zeros[ri] = j
            ones[rj] = i
            if cur_c < best_c - 1e-15:
                best_c = cur_c
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            stuck += 1
        T = max(T * cooling, 1e-6)
        if stuck > 8000:
            T = max(T, T0 * 0.5)
            stuck = 0
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        n = len(best_h)
    except Exception:
        n = 250
        best_h = np.full(n, 0.5)

    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP polish (8%)
    p1 = t0 + 0.08 * (budget_s - 3)
    h = best_h.copy()
    for alpha in [50000, 500000, 5000000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 800, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Greedy local search on binary (10%)
    p2 = t0 + 0.18 * (budget_s - 3)
    hb = binary_round(best_h, n)
    hg, cg = greedy_ls(hb, n, p2, max_sweeps=100, seed=seed)
    if cg < best_c:
        best_c = cg
        best_h = hg.copy()

    # Phase 3: Simulated annealing - main phase (55%)
    p3 = t0 + 0.73 * (budget_s - 3)
    n_list = [n]
    if n != 200:
        n_list.append(200)
    if n != 300:
        n_list.append(300)

    total_sa_time = p3 - time.time()
    time_per_n = total_sa_time / len(n_list)

    for ni in n_list:
        sa_deadline = min(time.time() + time_per_n, p3 - 2)
        if time.time() > sa_deadline - 10:
            continue
        # Build starting point for this n
        if ni == n:
            start = binary_round(best_h, n)
        else:
            xo = np.linspace(0, 2, n, endpoint=False)
            xn = np.linspace(0, 2, ni, endpoint=False)
            interp = np.interp(xn, xo, best_h)
            start = binary_round(interp, ni)
        # Quick greedy to get a good starting point
        start, _ = greedy_ls(start, ni, min(time.time() + 10, sa_deadline - 30),
                           max_sweeps=30, seed=seed + ni)
        # Multiple SA runs
        n_runs = 3
        sa_time_each = (sa_deadline - time.time() - 5) / n_runs
        for run_i in range(n_runs):
            if time.time() > sa_deadline - 5:
                break
            pert = start.copy()
            npert = rng.randint(2, 6)
            for _ in range(npert):
                oi = np.where(pert == 1)[0]
                zi = np.where(pert == 0)[0]
                if len(oi) and len(zi):
                    a, b = rng.choice(oi), rng.choice(zi)
                    pert[a], pert[b] = 0.0, 1.0
            T0_val = [0.004, 0.008, 0.016][run_i % 3]
            cool_val = [0.9999, 0.9997, 0.9995][run_i % 3]
            hd, cd = sim_anneal(pert, ni, time.time() + sa_time_each,
                               T0=T0_val, cooling=cool_val, seed=seed + ni * 10 + run_i)
            # Follow up greedy
            hd, cd = greedy_ls(hd, ni, min(time.time() + 5, sa_deadline - 3),
                              max_sweeps=50, seed=seed + ni * 10 + run_i + 100)
            if cd < best_c:
                best_c = cd
                best_h = hd.copy()
                n = ni

    # Phase 4: SLSQP polish (10%)
    p4 = t0 + 0.83 * (budget_s - 3)
    if time.time() < p4:
        h = best_h.copy()
        if np.all((h == 0) | (h == 1)):
            h = h + 0.08 * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [100000, 1000000, 10000000]:
            if time.time() > p4 - 2:
                break
            h = slsqp_smooth(h, n, alpha, 1500, p4 - 2)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 5: Final SA + greedy (10%)
    if time.time() < dl - 5:
        hfin = binary_round(best_h, n)
        for _ in range(rng.randint(3, 8)):
            oi = np.where(hfin == 1)[0]
            zi = np.where(hfin == 0)[0]
            if len(oi) and len(zi):
                a, b = rng.choice(oi), rng.choice(zi)
                hfin[a], hfin[b] = 0.0, 1.0
        hf, cf = sim_anneal(hfin, n, dl - 15, T0=0.012, cooling=0.9996, seed=seed + 99)
        hf, cf = greedy_ls(hf, n, dl - 3, max_sweeps=100, seed=seed + 100)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
        # Quick SLSQP from final
        if time.time() < dl - 1:
            hq = best_h.copy()
            if np.all((hq == 0) | (hq == 1)):
                hq = hq + 0.05 * rng.randn(n)
            hq = project_feasible(hq, n)
            hq = slsqp_smooth(hq, n, 5000000, 500, dl - 0.5)
            cq = compute_c5(hq, n)
            if cq < best_c:
                best_c = cq
                best_h = hq.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
