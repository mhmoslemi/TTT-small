```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta
```

Let me also think about the subgradient method more carefully. The subgradient of max_m C(m) at h is the subgradient of C(m*) where m* = argmax. The subgradient of C(m) w.r.t. h is:

The Polyak step size for subgradient methods is:

Actually, a better approach: use "mirror descent" with the entropy regularizer, or use the "Frank-Wolfe" (conditional gradient) method.

For Frank-Wolfe:
- At each iteration, compute the subgradient g = ∇C(m*)(h)
- Find the point s that minimizes <g, s> over the feasible set (0≤s≤1, sum(s)=n/2)
- Update: h ← (1-γ)h + γs

The linear minimization problem: min_s <g, s> subject to 0≤s≤1, sum(s)=n/2
This has a closed-form solution: s[i] = 1 if g[i] < threshold, 0 if g[i] > threshold, and the threshold is chosen so that sum(s) = n/2.

This is the "water-filling" solution! Sort g, set s[i]=1 for the smallest g[i] values and s[i]=0 for the largest, with one fractional value to make the sum exactly n/2.

This is very efficient! Let me implement Frank-Wolfe:

```python
def frank_wolfe(h, target_sum, max_iter=10000):
    n = len(h)
    h = h.copy()
    for it in range(max_iter):
        corr = np.correlate(h, 1-h, mode='full')
        m_star = np.argmax(corr)
        m = m_star - (n-1)
        
        # Compute subgradient
        g = np.zeros(n)
        if m == 0:
            g = 1.0 - 2.0 * h
        elif m > 0:
            for j in range(n - m):
                g[j] += (1.0 - h[j+m])
                g[j+m] -= h[j]
        else:
            p = -m
            for j in range(p, n):
                g[j] += (1.0 - h[j-p])
                g[j-p] -= h[j]
        
        # Find s that minimizes <g, s> over {0≤s≤1, sum(s)=n/2}
        # Sort g, set s=1 for smallest, s=0 for largest
        order = np.argsort(g)
        s = np.zeros(n)
        s[order[:n//2]] = 1.0
        # Adjust for exact sum
        s_sum = s.sum()
        if abs(s_sum - target_sum) > 1e-12:
            # Need fractional element
            # The (n//2)-th element should be fractional
            # Actually, if n is even, n//2 elements at 1 and n//2 at 0 gives sum = n/2 exactly
            # If n is odd, we need one fractional
            pass
        
        # Step size
        gamma = 2.0 / (it + 2.0)  # standard FW step size
        h = (1 - gamma) * h + gamma * s
        h = np.clip(h, 0, 1)
        h = project_feasible(h, target_sum)
    
    return h
```
