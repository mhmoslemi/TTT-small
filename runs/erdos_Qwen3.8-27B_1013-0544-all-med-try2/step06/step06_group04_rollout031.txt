def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, lag):
    """Gradient of unnormalized c[lag] w.r.t. h."""
    g = np.zeros(n)
    m = abs(lag)
    if m == 0:
        return 1.0 - 2.0 * h
    if m >= n:
        return g
    if lag > 0:
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    else:
        g[:m] -= h[m:]
        g[m:] += (1.0 - h[:n - m])
    return g

def smooth_max_obj(x, n, active_lags, alpha=50000.0):
    c = compute_all_lags(x, n)
    vals = c[active_lags]
    m = vals.max()
    s = alpha * (vals - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + float(np.log(np.sum(np.exp(s)))) / alpha)

def smooth_max_obj_jac(x, n, active_lags, alpha=50000.0):
    c = compute_all_lags(x, n)
    vals = c[active_lags]
    m = vals.max()
    s = alpha * (vals - m)
    # softmax gradient
    e = np.exp(s - s.max())

    dx = 2.0 / n

    for i, lag_idx in enumerate(active_lags):
        lag = lag_idx - (n - 1)
        g += probs[i] * dx * lag_grad(x, n, lag)
    return g

    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    trust = 0.1

        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        # Build LP
        nv = cp.Variable(n)
        t = cp.Variable()
        cons = [cp.sum(nv) == n * 0.5, nv >= 0, nv <= 1, cp.abs(nv - h_curr) <= trust]
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            g = lag_grad(h_curr, n, lag)
            cons.append(c[k] + dx * (g @ (nv - h_curr)) <= t)
        prob = cp.Problem(cp.Minimize(t), cons)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        nv_val = nv.value
        if nv_val is None:
            break
        h_new = project_feasible(nv_val, n)

        if step < 1e-9:

- Try to find a feasible h
- If we can, we've achieved our goal!

- At the current h, linearize the constraints
- Solve the LP: find h' such that the linearized constraints are satisfied
- If the LP is infeasible, t is too small
- If feasible, update h = h' and check the true constraints

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m == 0:
        return 1.0 - 2.0 * h
    if m >= n:
        return g
    if lag > 0:
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    else:
        g[:m] -= h[m:]
        g[m:] += (1.0 - h[:n - m])
    return g

def _smooth_obj(x, n, active, alpha):
    c = compute_all_lags(x, n)
    vals = c[active]
    m = vals.max()
    s = alpha * (vals - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + float(np.log(np.sum(np.exp(s)))) / alpha)

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones_like(x)

def outer_approx(h, n, deadline, max_iters=300):
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    trust = 0.12
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        nv = cp.Variable(n)
        t = cp.Variable()
        cons = [cp.sum(nv) == n * 0.5, nv >= 0, nv <= 1, cp.abs(nv - h_curr) <= trust]
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            g = lag_grad(h_curr, n, lag)
            cons.append(c[k] + dx * (g @ (nv - h_curr)) <= t)
        prob = cp.Problem(cp.Minimize(t), cons)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if nv.value is None:
            break
        h_new = project_feasible(nv.value, n)
        c_new = compute_c5(h_new, n)
        step = np.max(np.abs(h_new - h_curr))
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.003, trust * 0.88)
    return best_h, best_c

def proj_grad(h, n, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    step = 0.002
    for _ in range(8000):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        g = lag_grad(h, n, lag)
        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:

            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.03, 0.01)
        else:
            step *= 0.65

    return best_h, best_c

def slsqp_refine(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    c = compute_all_lags(h, n)
    top_m = min(12, 2 * n - 1)
    active = np.argsort(c)[-top_m:]
    try:
        res = minimize(_smooth_obj, h, args=(n, active, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

                            h[j] -= delta
                        else:

                            h[j] += delta

                        if c_new < best_c:
                            best_c = c_new
                            improved = True
                        else:
                            if h[i] > h[j]:
                                h[i] -= 2 * delta
                                h[j] += 2 * delta
                            else:
                                h[i] += 2 * delta
