2. For each position, try flipping and see if max overlap decreases
3. Or: swap pairs of bits

For the discrete case, the "gradient" check is even simpler: flipping bit i from 0 to 1 (and bit j from 1 to 0 to maintain the sum constraint) changes c[k] by:

- For each lag L where h[i+L]=0: c increases by 1
- For each lag L where h[i-L]=1: c decreases by 1 (because previously h[i]=0 so it didn't contribute, but now... wait, I need to be more careful)

Let me denote the contribution to c[k] from position i:
- If h[i]=1 and h[i+L_k]=0: contributes 1
- Otherwise: contributes 0

When h[i] flips from 0 to 1:
- For lag L_k: if h[i+L_k]=0, contribution goes from 0 to 1 (increase)
- Also, for lag -L_k: the term h[i-L_k]*(1-h[i]) was 0 (since h[i]=0), now becomes h[i-L_k]*0 = 0 (still 0)
  Wait, the term at lag -L_k involving position i is: h[i+(-L_k)]*(1-h[i]) = h[i-L_k]*(1-h[i]). When h[i] was 0, this was h[i-L_k]*1 = h[i-L_k]. When h[i]=1, this is h[i-L_k]*0 = 0. So c[-L_k] DECREASES by h[i-L_k].

So flipping h[i] from 0 to 1:
- For each lag L≥0: c[L] += 1 if h[i+L]=0, c[-L] -= h[i-L] (which is 1 if h[i-L]=1)

This is O(n) per flip (since there are 2n-1 lags). And we can precompute for each position i the "effect vector" (how c changes if we flip i). Total precomputation: O(n * 2n) = O(2n²).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    val += penalty * (np.sum(x) - n/2.0)**2
    return float(val)

def smooth_max_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    n_c = len(c)  # 2n-1
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    
    # Gradient of smooth max = sum_k w[k] * dc[k]/dx
    # dc[k]/dx[j] = dx * [(1-x[j+L]) * 1_{valid} - x[j-L] * 1_{valid}] where L = k-(n-1)
    
    grad = np.zeros(n)
    L = np.arange(n_c) - (n - 1)
    
    for k in range(n_c):
        wk = w[k]
        if wk < 1e-15:
            continue
        Lk = L[k]
        if Lk >= 0:
            # j+Lk valid: j in [0, n-1-Lk]
            if n - Lk > 0:
                grad[:n-Lk] += wk * (1 - x[Lk:n])  # 1-x[j+Lk] for j in [0,n-1-Lk]
            # j-Lk valid: j in [Lk, n-1]
            if n - Lk > 0:
                grad[Lk:] -= wk * x[:n-Lk]  # x[j-Lk] for j in [Lk, n-1]
        else:
            Mk = -Lk
            # j+Lk = j-Mk valid: j in [Mk, n-1]
            if n - Mk > 0:
                grad[Mk:] += wk * (1 - x[:n-Mk])
            # j-Lk = j+Mk valid: j in [0, n-1-Mk]
            if n - Mk > 0:
                grad[:n-Mk] -= wk * x[Mk:]
    
    grad *= dx
    
    # Add gradient of penalty term
    grad += 2 * penalty * (np.sum(x) - n/2.0) * np.ones(n)
    
    return grad

def lbfbs_opt(h, n, deadline, alpha=100000, penalty=500):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    if time.time() > deadline - 5:
        return best_h, best_c
    
    def obj(x):
        return smooth_max_obj(x, n, alpha, penalty)
    
    def grad(x):
        return smooth_max_grad(x, n, alpha, penalty)
    
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': 300, 'ftol': 1e-13, 'gtol': 1e-8,
                               'maxfun': 5000})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    except Exception:
        pass
    
    # Try with different alpha
    if time.time() < deadline - 10:
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-13, 'gtol': 1e-8,
                                   'maxfun': 3000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:

def discrete_local_search(h_bin, n, deadline, max_passes=500):
    """Local search on 0-1 valued array. h_bin has exactly n/2 ones."""
    n_c = 2 * n - 1
    dx = 2.0 / n
    
    def compute_c_discrete(h_bin):
        return np.correlate(h_bin.astype(float), 1 - h_bin.astype(float), mode='full') * dx
    
    h_bin = h_bin.copy()
    c = compute_c_discrete(h_bin)
    best_c = float(c.max())
    
    # Precompute effect vectors
    # effect[i] = change in c if we flip h_bin[i]
    # For 0->1: c[k] += 1 if h[i+L_k]=0 and valid, c[k] -= 1 if h[i-L_k]=1 and valid
    # For 1->0: opposite
    
    def compute_effects(h_bin, n):
        n_c = 2*n - 1
        effects = np.zeros((n, n_c))
        L = np.arange(n_c) - (n-1)
        for i in range(n):
            for k in range(n_c):
                Lk = L[k]
                eff = 0.0
                if h_bin[i] == 0:
                    # Flipping 0->1
                    if 0 <= i + Lk < n and h_bin[i+Lk] == 0:
                        eff += 1
                    if 0 <= i - Lk < n and h_bin[i-Lk] == 1:
                        eff -= 1
                else:
                    # Flipping 1->0
                    if 0 <= i + Lk < n and h_bin[i+Lk] == 0:
                        eff -= 1
                    if 0 <= i - Lk < n and h_bin[i-Lk] == 1:
                        eff += 1
                effects[i, k] = eff * dx
        return effects
    
    effects = compute_effects(h_bin, n)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        for i in zeros:
            for j in ones:
                # Swap: h[i] 0->1, h[j] 1->0
                # Net effect on c: effects[i] + effects[j] (since effects[j] for 1->0 is the negative of 0->1)
                # Wait, I defined effects[i] as the change when flipping i.
                # For i (0->1): effect is effects[i]
                # For j (1->0): effect is effects[j]
                # Net: effects[i] + effects[j]
                dc = effects[i] + effects[j]
                new_c = c + dc
                new_max = new_c.max()
                if new_max < best_c - 1e-15:
                    h_bin[i] = 1
                    h_bin[j] = 0
                    c = new_c
                    best_c = new_max
                    effects = compute_effects(h_bin, n)
                    improved = True
                    ones = np.where(h_bin == 1)[0]
                    zeros = np.where(h_bin == 0)[0]
        
        if not improved:
            break
    
    return h_bin, best_c

def directed_local_search(h, n, deadline, max_passes=1000, eps=0.001):
    dx = 2.0 / n
    n_c = 2*n - 1
    best_c = compute_c5(h, n)
    c = np.correlate(h, 1-h, mode='full') * dx
    
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for step in step_sizes:
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            # Recompute active set
            active = np.where(c > best_c - eps)[0]
            if len(active) == 0:
                active = np.array([np.argmax(c)])
            
            improved = False
            # For each pair (i,j), compute the directed score
            # score[i,j] = sum over active k of: (dc[k]/dh[i] - dc[k]/dh[j])
            # We want score < 0 (reduces the active overlaps)
            
            # Precompute gradient contributions for active shifts
            # G_active[k, i] = dc[k]/dh[i] for k in active
            G = np.zeros((len(active), n))
            L = np.arange(n_c) - (n-1)
            for idx, k in enumerate(active):
                Lk = L[k]
                for i in range(n):
                    g = 0.0
                    if 0 <= i + Lk < n:
                        g += (1 - h[i + Lk])
                    if 0 <= i - Lk < n:
                        g -= h[i - Lk]
                    G[idx, i] = g * dx
            
            # For each i, j: check if moving step from j to i reduces all active overlaps
            # Change in c[k] for active k: step * (G[k,i] - G[k,j])
            # We want: c[k] + step * G[k,i] - step * G[k,j] < best_c for all active k
            
            # This is: G[k,i] - G[k,j] < (best_c - c[k]) / step for all active k
            # Since c[k] >= best_c - eps for active k, (best_c - c[k])/step <= eps/step

            # More precisely: for all active k, c[k] + step*(G[k,i]-G[k,j]) < best_c

            # Then for pair (i,j): need max over active k of (gain[i,k] - step*G[k,j]) < best_c
            # i.e., max_k gain[i,k] - step*min_k G[k,j] < best_c... no that's not right either.

            # For each j, precompute min reduction = min_k (step * G[k,j])  (the best case for j)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    val += penalty * (np.sum(x) - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    n_c = len(c)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    grad = np.zeros(n)
    L = np.arange(n_c) - (n - 1)
    for k in range(n_c):
        wk = w[k]
        if wk < 1e-15:
            continue
        Lk = L[k]
        if Lk >= 0:
            if n - Lk > 0:
                grad[:n - Lk] += wk * (1 - x[Lk:n])
                grad[Lk:] -= wk * x[:n - Lk]
        else:
            Mk = -Lk
            if n - Mk > 0:
                grad[Mk:] += wk * (1 - x[:n - Mk])
                grad[:n - Mk] -= wk * x[Mk:]
    grad *= dx
    grad += 2 * penalty * (np.sum(x) - n / 2.0)
    return grad

def lbfbs_opt(h, n, deadline, alpha=100000, penalty=2000, maxiter=400):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    if time.time() > deadline - 3:
        return best_h, best_c
    bounds = [(0.0, 1.0)] * n
    for a in [alpha, alpha * 5]:
        if time.time() > deadline - 3:
            break
        p = penalty
        try:
            res = minimize(smooth_max_val, h, jac=lambda x: smooth_max_grad(x, n, a, p),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxfun': 8000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c, best_h = c_new, h_new
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def directed_search(h, n, deadline, eps=0.002):
    dx = 2.0 / n
    n_c = 2 * n - 1
    best_c = compute_c5(h, n)
    c = compute_c_vec(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    L = np.arange(n_c) - (n - 1)

                Lk = L[k]
                if Lk >= 0:
                    if n - Lk > 0:

                    for i in range(n):
                        g = 0.0
                        if 0 <= i + Lk < n:
                            g += (1 - h[i + Lk])
                        if 0 <= i - Lk < n:
                            g -= h[i - Lk]
                        G[idx, i] = g * dx
                else:
                    Mk = -Lk
                    for i in range(n):
                        g = 0.0
                        if 0 <= i - Mk < n:
                            g += (1 - h[i - Mk])
                        if 0 <= i + Mk < n:
                            g -= h[i + Mk]
                        G[idx, i] = g * dx
            improved = False
            for i in range(n):
                if h[i] + step > 1.0:
                    continue
                for j in range(n):
                    if i == j:
                        continue
                    if h[j] - step < 0.0:
                        continue
                    ok = True

                        h[j] -= step

                        best_c = float(c.max())
                        improved = True

            if not improved:
