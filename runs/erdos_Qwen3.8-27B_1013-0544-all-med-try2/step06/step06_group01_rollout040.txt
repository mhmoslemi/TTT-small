Actually, I realize there might be an even better approach: **coordinate descent on the C(d) values**.

The idea: identify the d values where C(d) is at or near the maximum. Then try swaps that specifically reduce C(d) for those d values.

For a given bottleneck d*, a swap (j→i) reduces C(d*) if:
- j+d* was in A (so C(d*) had a contribution from j being "0 side") → removing j from A removes this
- i-d* was not in A (so C(d*) didn't have a contribution from i-d* being "0 side") → adding i to A adds this... wait, that would increase it.

Let me re-examine. C(d) = |{k: h[k]=0, h[k+d]=1}|.

To DECREASE C(d*), we want to:
- Remove a contribution: find k where h[k]=0, h[k+d*]=1. Change h[k] to 1 (but that adds a 1 somewhere) or change h[k+d*] to 0.
- So: swap h[k]=0→1 and h[k+d*]=1→0. This removes the pair from C(d*).

But this swap might increase C(d) for other d values.

**Greedy approach**: 
1. Find d* = argmax C(d)
2. Find a pair (k, k+d*) where h[k]=0, h[k+d*]=1
3. Perform the swap (set h[k]=1, h[k+d*]=0)
4. Recompute C, check if max decreased
5. If yes, accept; if no, try another pair

This is essentially a specialized local search that focuses on the bottleneck.

Let me implement this:

```python
def bottleneck_local_search(h, n, max_iter, deadline):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    current_max = c.max()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Find bottleneck shifts
        bottleneck_d = np.where(c >= current_max - 1)[0]  # shifts near max
        found = False
        for d_idx in bottleneck_d:
            d = d_idx - (n-1)  # actual shift
            # Find pairs (k, k+d) where h[k]=0, h[k+d]=1
            valid_k = []
            for k in range(n):
                if 0 <= k+d < n and h[k] == 0 and h[k+d] == 1:
                    valid_k.append(k)
            for k in valid_k:
                # Try swap: h[k]=1, h[k+d]=0
                new_max = try_swap(h, c, k, k+d, n, current_max)
                if new_max < current_max:
                    h[k], h[k+d] = 1, 0
                    c = update_c(c, k, k+d, n, h)
                    current_max = new_max
                    found = True
                    break
            if found:
                break
        if not found:
            break
    return h, current_max * dx
```

```python
def delta_c_vec(h, j, i, n):
    """Vectorized delta C computation for swap h[j]:1->0, h[i]:0->1."""
    d = np.arange(-(n-1), n)  # all possible lags
    dc = np.zeros(len(d))
    
    # Term k=j: +I(0<=j+d<n) * h_eff[j+d] where h_eff[i]=1
    mask_j = (j + d >= 0) & (j + d < n)
    if mask_j.any():
        vals = h[j + d[mask_j]].copy()
        vals[j + d[mask_j] == i] = 1.0
        dc[mask_j] += vals
    
    # Term k=i: -I(0<=i+d<n) * h_eff2[i+d] where h_eff2[j]=0
    mask_i = (i + d >= 0) & (i + d < n)
    if mask_i.any():
        vals = h[i + d[mask_i]].copy()
        vals[i + d[mask_i] == j] = 0.0
        dc[mask_i] -= vals
    
    # Term k=j-d: -I(h[j-d]=0) if 0<=j-d<n and j-d != j and j-d != i
    kd = j - d
    mask_jd = (kd >= 0) & (kd < n) & (kd != j) & (kd != i)
    if mask_jd.any():
        dc[mask_jd] -= (1.0 - h[kd[mask_jd]])
    
    # Term k=i-d: +I(h[i-d]=0) if 0<=i-d<n and i-d != j and i-d != i
    kd2 = i - d
    mask_id = (kd2 >= 0) & (kd2 < n) & (kd2 != j) & (kd2 != i)
    if mask_id.any():
        dc[mask_id] += (1.0 - h[kd2[mask_id]])
    
    return dc
```

```python
def binary_sa(h, n, deadline, seed, T0=0.5, cooling=0.9999):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full').astype(float)
    cur_max = c.max()
    best_max = cur_max
    best_h = h.copy()
    T = T0

                    best_h = h.copy()
```

For the continuous case, the SLSQP approach from before should work, but it's slow. An alternative is to use a softmax parameterization:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def delta_c_vec(h, j, i, n):
    d = np.arange(-(n - 1), n)
    dc = np.zeros(len(d))
    jd = j + d
    mask = (jd >= 0) & (jd < n)
    if mask.any():
        vals = h[jd[mask]].copy()
        vals[jd[mask] == i] = 1.0
        dc[mask] += vals
    id_ = i + d
    mask = (id_ >= 0) & (id_ < n)
    if mask.any():
        vals = h[id_[mask]].copy()
        vals[id_[mask] == j] = 0.0
        dc[mask] -= vals
    kd = j - d
    mask = (kd >= 0) & (kd < n) & (kd != j) & (kd != i)
    if mask.any():
        dc[mask] -= (1.0 - h[kd[mask]])
    kd2 = i - d
    mask = (kd2 >= 0) & (kd2 < n) & (kd2 != j) & (kd2 != i)
    if mask.any():
        dc[mask] += (1.0 - h[kd2[mask]])
    return dc

def binary_sa(h, n, deadline, seed, T_max=5.0):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = c.max()
    best_max = cur_max
    best_h = h.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    T = T_max
    no_improve = 0
    while time.time() < deadline:
        T = max(0.01, T * 0.9998)
        j = rng.choice(sorted(ones))
        i = rng.choice(sorted(zeros))
        dc = delta_c_vec(h, int(j), int(i), n)
        new_max = (c + dc).max()
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / T):
            j, i = int(j), int(i)
            c = c + dc
            h[j], h[i] = 0, 1
            ones.discard(j)
            ones.add(i)
            zeros.discard(i)
            zeros.add(j)
            cur_max = new_max
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 5000:
                T = T_max  # reheat
                no_improve = 0
    return best_h, best_max * dx

def binary_grad_search(h, n, deadline, max_sweeps=1000):
    """Greedy: always accept the best improving swap."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = c.max()
    best_max = cur_max
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Try random subset of swaps to save time
        n_try = min(500, len(ones) * len(zeros))
        rng = np.random.RandomState(sweep * 7 + 13)
        best_dc_max = cur_max
        best_ji = None
        for _ in range(n_try):
            j = rng.choice(ones)
            i = rng.choice(zeros)
            dc = delta_c_vec(h, j, i, n)
            nm = (c + dc).max()
            if nm < best_dc_max:
                best_dc_max = nm
                best_ji = (j, i, dc)
        if best_ji is not None:
            j, i, dc = best_ji
            c = c + dc
            h[j], h[i] = 0, 1
            cur_max = best_dc_max
            improved = True
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        if not improved:
            break
    return best_h, best_max * dx

    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial solution

        zeros = np.where(h_bin == 0)[0]
        for idx in rng.choice(zeros, deficit, replace=False):
            h_bin[idx] = 1
    
    best_h = h_bin.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    
    # Phase 1: Try multiple n values with binary SA
    p1 = t0 + 0.15 * (budget_s - 5)
    for n_try in [n0, n0 - 10, n0 + 10, n0 - 30, n0 + 30]:
        if time.time() > p1 - 30:
            break
        if n_try < 50 or n_try > 500 or n_try % 2 != 0:
            n_try += 1
        # Create initial binary solution
        if n_try == n0:
            h_try = h_bin.copy()
        else:
            # Interpolate and binarize
            xo = np.linspace(0, 2, n0)
            xn = np.linspace(0, 2, n_try)
            h_cont = np.interp(xn, xo, h_bin)
            h_try = (h_cont >= 0.5).astype(float)
            target = n_try // 2
            if h_try.sum() > target:

    if cb < best_c:

    if cb < best_c:

        best_h = hb.copy()
    
    # Phase 4: Continuous polish with SLSQP
    if time.time() < dl - 20:
        dx = 2.0 / n
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            alpha = 10000

            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        def cons_fun(x):
            return x.sum() - n / 2.0
        h_init = project_feasible(best_h + 0.1 * rng.randn(n), n)
        try:
            res = minimize(obj, h_init, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': cons_fun}],
                          options={'maxiter': 3000, 'ftol': 1e-14})
            h_cont = project_feasible(res.x, n)
            c_cont = compute_c5(h_cont, n)
            if c_cont < best_c:
                best_c = c_cont
                best_h = h_cont.copy()
        except Exception:
            pass
    
    # Phase 5: Final binary polish
    if time.time() < dl - 3:
        h_final_bin = (best_h >= 0.5).astype(float)
        target = n // 2
        if h_final_bin.sum() != target:
            if h_final_bin.sum() > target:
                excess = int(h_final_bin.sum()) - target
                ones = np.where(h_final_bin == 1)[0]
                for idx in rng.choice(ones, excess, replace=False):
                    h_final_bin[idx] = 0
            else:
                deficit = target - int(h_final_bin.sum())
                zeros = np.where(h_final_bin == 0)[0]

                    h_final_bin[idx] = 1
        hb, cb = binary_grad_search(h_final_bin, n, dl)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
