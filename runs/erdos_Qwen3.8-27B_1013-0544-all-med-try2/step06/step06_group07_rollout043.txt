import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    e = np.exp(shifted)
    return m + np.log(np.sum(e)) / beta, e

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    _, e = smooth_max_val(corr, beta)
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += wk * (1.0 - h[idx + lag])
            grad[idx + lag] -= wk * h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += wk * (1.0 - h[idx])
            grad[idx] -= wk * h[idx]
    return grad * dx

def obj_grad(h, dx, beta, ts, pen):
    corr = np.correlate(h, 1.0 - h, mode='full')
    val, _ = smooth_max_val(corr, beta)
    val = val * dx + pen * (h.sum() - ts) ** 2
    g = grad_smooth(h, beta, dx) + 2.0 * pen * (h.sum() - ts)
    return val, g

def opt_single_var(h, j, beta):
    n = len(h)
    h_save = h[j]
    h[j] = 0.0
    corr_0 = np.correlate(h, 1.0 - h, mode='full').copy()
    h[j] = 1.0
    corr_1 = np.correlate(h, 1.0 - h, mode='full').copy()
    h[j] = h_save
    diff = corr_1 - corr_0

    def deriv(a):
        corr_a = corr_0 + a * diff
        m = np.max(corr_a)
        shifted = np.clip(beta * (corr_a - m), -700.0, 700.0)
        e = np.exp(shifted)
        w = e / e.sum()
        return np.dot(w, diff)

    d0 = deriv(0.0)
    d1 = deriv(1.0)
    if d0 >= 0:
        h[j] = 0.0
        return
    if d1 <= 0:
        h[j] = 1.0
        return
    a, b = 0.0, 1.0
    for _ in range(50):
        mid = 0.5 * (a + b)
        dm = deriv(mid)
        if dm > 0:
            b = mid
        else:
            a = mid
        if b - a < 1e-13:
            break
    h[j] = 0.5 * (a + b)

def block_cd_sweep(h, beta, ts, n_sweeps):
    n = len(h)
    for _ in range(n_sweeps):
        for j in range(n):
            opt_single_var(h, j, beta)
        h = project_feasible(h, ts)
    return h

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    best_h = project_feasible(h_init, ts)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with analytical gradient and continuation
    pen_val = 500.0
    betas_list = [10, 30, 100, 300, 1000, 3000, 10000, 30000]
    for beta in betas_list:
        if budget_s - (time.time() - t0) < 50:
            break
        for trial in range(5):
            if budget_s - (time.time() - t0) < 15:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += np.random.randn(n) * (0.002 / trial)
                x0 = project_feasible(x0, ts)
            try:
                res = minimize(obj_grad, x0, args=(dx, beta, ts, pen_val),
                             jac=True, method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': 30000, 'ftol': 1e-17, 'gtol': 1e-15,
                                      'maxls': 100})
                h_new = project_feasible(res.x, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 2: Block coordinate descent (very fast local refinement)
    if budget_s - (time.time() - t0) > 100:
        for beta_cd in [500, 2000, 10000, 50000]:
            if budget_s - (time.time() - t0) < 15:
                break
            h_cd = best_h.copy()
            h_cd = block_cd_sweep(h_cd, beta_cd, ts, 200)
            c5 = compute_c5(h_cd)
            if c5 < best_c5:
                best_h = h_cd.copy()
                best_c5 = c5

    # Phase 3: Another round of L-BFGS-B from improved point
    for beta in [5000, 20000, 100000]:
        if budget_s - (time.time() - t0) < 20:
            break
        try:
            res = minimize(obj_grad, best_h, args=(dx, beta, ts, 100.0),
                         jac=True, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 50000, 'ftol': 1e-17, 'gtol': 1e-16})
            h_new = project_feasible(res.x, ts)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
        except Exception:
            pass

    # Phase 4: More block CD at high beta
    if budget_s - (time.time() - t0) > 60:
        for beta_cd in [20000, 100000, 500000]:
            if budget_s - (time.time() - t0) < 10:
                break
            h_cd = best_h.copy()
            h_cd = block_cd_sweep(h_cd, beta_cd, ts, 500)
            c5 = compute_c5(h_cd)
            if c5 < best_c5:
                best_h = h_cd.copy()
                best_c5 = c5

    # Phase 5: Subgradient on non-smooth max
    if budget_s - (time.time() - t0) > 60:
        h = best_h.copy()
        for it in range(150000):
            if budget_s - (time.time() - t0) < 25:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_worst = int(np.argmax(corr))
            lag = k_worst - (n - 1)
            grad = np.zeros(n)
            if lag >= 0:
                idx = np.arange(n - lag)
                grad[idx] += (1.0 - h[idx + lag])
                grad[idx + lag] -= h[idx]
            else:
                la = -lag
                idx = np.arange(n - la)
                grad[idx + la] += (1.0 - h[idx])
                grad[idx] -= h[idx + la]
            step = 0.003 / (1.0 + it * 0.0003)
            h = h - step * grad
            h = project_feasible(h, ts)
            if it % 10000 == 9999:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 6: Multi-resolution search
    if budget_s - (time.time() - t0) > 60:
        for n_try in [60, 80, 90, 100, 75, 65, 110]:
            if budget_s - (time.time() - t0) < 20 or n_try == n:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            for beta_t in [500, 5000, 50000]:
                if budget_s - (time.time() - t0) < 10:
                    break
                try:
                    res = minimize(obj_grad, h_t, args=(dx_t, beta_t, ts_t, 500.0),
                                 jac=True, method='L-BFGS-B', bounds=bnds_t,
                                 options={'maxiter': 20000, 'ftol': 1e-16, 'gtol': 1e-14})
                    h_new = project_feasible(res.x, ts_t)
                    h_new = block_cd_sweep(h_new, 20000, ts_t, 100)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        n = n_try
                        dx = dx_t
                        ts = ts_t
                        bounds = bnds_t
                except Exception:
                    pass

    # Phase 7: Final polish with block CD + L-BFGS-B
    if budget_s - (time.time() - t0) > 15:
        best_h = block_cd_sweep(best_h, 200000, ts, 500)
        best_c5 = compute_c5(best_h)
        try:
            res = minimize(obj_grad, best_h, args=(dx, 200000, ts, 50.0),
                         jac=True, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 30000, 'ftol': 1e-17, 'gtol': 1e-16})
            h_new = project_feasible(res.x, ts)
            h_new = block_cd_sweep(h_new, 500000, ts, 200)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
