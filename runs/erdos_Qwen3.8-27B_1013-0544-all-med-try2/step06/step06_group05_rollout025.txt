    target = n / 2.0
    # Bisection on threshold
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2, 0.0, 1.0)

    if abs(diff) > 1e-10:
        # Distribute the residual
        room = 1.0 - projected
        active = room > 1e-10
        if diff > 0:
            projected[active] += diff / active.sum()
            projected = np.clip(projected, 0, 1)
        else:
            active2 = projected > 1e-10
            projected[active2] += diff / active2.sum()
            projected = np.clip(projected, 0, 1)
    return projected

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - projected.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = projected < 1.0 - 1e-12
            if mask.sum() > 0:
                projected[mask] += diff / mask.sum()
        else:
            mask = projected > 1e-12
            if mask.sum() > 0:
                projected[mask] += diff / mask.sum()
    return np.clip(projected, 0.0, 1.0)

def compute_grad_weighted(h, n, p):
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_corr):
        s = k - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        if i_lo > i_hi:
            continue
        pk = p[k]
        if pk < 1e-15:
            continue
        grad[i_lo:i_hi+1] += pk * (1.0 - h[i_lo+s:i_hi+s+1])
        grad[i_lo+s:i_hi+s+1] -= pk * h[i_lo:i_hi+1]
    return grad

def pg_solve(h, n, p, deadline, max_iter=2000):
    h = h.copy()
    best_h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_val = float(np.dot(p, c))
    p_sum = max(np.sum(p), 1e-10)
    lr = 0.8 / (p_sum * 2.0 / (2.0 * n - 1) + 1e-8)
    lr = min(lr, 2.0)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = compute_grad_weighted(h, n, p)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        val = float(np.dot(p, c_new))
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            lr = min(lr * 1.02, 5.0)
        else:
            lr *= 0.6
            if lr < 1e-7:
                break
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)
            c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            val = float(np.dot(p, c_new))
            if val < best_val:
                best_val = val
                best_h = h_new.copy()
        h = h_new
    return best_h

def minimax_optimize(h, n, deadline, max_outer=500):
    n_corr = 2 * n - 1
    p = np.ones(n_corr) / n_corr
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    tau = 0.05
    for outer in range(max_outer):
        if time.time() > deadline - 15:
            break
        c_pos = np.maximum(c, 1e-12)
        logits = c_pos / tau
        logits -= logits.max()
        p_target = np.exp(logits)
        p_target /= p_target.sum()
        alpha = 0.4 + 0.3 * (outer / max(outer + 1, max_outer))
        p = (1.0 - alpha) * p + alpha * p_target
        tau = max(0.005, tau * 0.97)
        inner_time = min(2.5, (deadline - time.time()) * 0.12)
        if inner_time < 0.3:
            break
        h = pg_solve(h, n, p, time.time() + inner_time, max_iter=1500)
        c = np.correlate(h, 1.0 - h, mode='full')
        c_val = float(np.max(c) * 2.0 / n)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
    return best_h, best_c

def pair_search_adaptive(h, n, deadline, max_passes=80):
    h = h.copy()
    best_c = compute_c5(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        npass = 0
        while improved and npass < max_passes and time.time() < deadline:
            improved = False
            npass += 1
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d > 1.0 + 1e-12 or h[j] - d < -1e-12:
                        continue
                    old_i, old_j = h[i], h[j]
                    h[i] = old_i + d
                    h[j] = old_j - d
                    c_new = np.correlate(h, 1.0 - h, mode='full')
                    c_val = float(np.max(c_new) * 2.0 / n)
                    if c_val < best_c - 1e-15:
                        best_c = c_val
                        c = c_new
                        improved = True
                    else:
                        h[i] = old_i
                        h[j] = old_j
    return h, best_c

def try_random_start(n, deadline, rng, best_c_ref):
    h = rng.binomial(1, 0.5, n).astype(float)
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    if c < best_c_ref + 0.01:
        alloc = min(60, (deadline - time.time()) * 0.3)
        if alloc > 20:
            h_opt, c_opt = minimax_optimize(h, n, time.time() + alloc, max_outer=150)
            if c_opt < best_c_ref:
                return h_opt, c_opt
    return None, best_c_ref

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    try:

    if c_opt < best_c:

        if c_opt < best_c:

            best_h = h_opt.copy()

    # Phase 3: Second minimax round

        remaining = deadline - time.time() - 40
        alloc3 = min(200, remaining * 0.55)
        h_opt, c_opt = minimax_optimize(best_h, n, time.time() + alloc3, max_outer=400)

        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

        for trial in range(3):
            if time.time() > deadline - 30:

    final_c = compute_c5(best_h, n)
