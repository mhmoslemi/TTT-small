import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    """Log-sum-exp based smooth max approximation."""
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
        w /= w.sum()
    else:
        w = np.exp(sh)
        w /= w.sum()
    dx = 2.0 / n

            grad[0:nv] += wm * (1.0 - h[lag:n])
            grad[lag:n] += wm * (-h[0:nv])

            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    val = smooth_max_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    return float(smooth_max_val(compute_c5_vec(x, n), alpha))

f_k = sum_m h[m] * (1 h[m+k]) * dx (for valid m, m+k)

The terms involving h[i]=t:
- If i+k < n: t * (1-h[i+k])
- If i-k >= 0: (1-t) * h[i-k]

The terms involving h[j]=s-t:
- If j+k < n: (s-t) * (1-h[j+k])
- If j-k >= 0: (1-(s-t)) * h[j-k] = (1-s+t) * h[j-k]

So f_k(t) = [t*(1-h[i+k]) if valid] + [(1-t)*h[i-k] if valid]

    - h[i-k] if i-k>=0 else 0
    - (1-h[j+k]) if j+k<n else 0
    + h[j-k] if j-k>=0 else 0

b_k = (rest) + s*(1-h[j+k]) if j+k<n else 0 + (1-s)*h[j-k] if j-k>=0 else 0

import numpy as np
from scipy.optimize import minimize
import time

def c5_of(h, n):
    return float(np.max(np.correlate(h, 1-h, mode='full')) * 2.0/n)

def c5v(h, n):
    return np.correlate(h, 1-h, mode='full') * 2.0/n

def proj(h, n):
    h = np.clip(h, 0., 1.)
    for _ in range(200):
        d = n/2. - h.sum()
        if abs(d) < 1e-14: break
        if d > 0:
            r = 1.-h; t = r.sum()
            if t < 1e-15: break
            h += d/t*r
        else:
            t = h.sum()
            if t < 1e-15: break
            h += d/t*h
    return np.clip(h, 0., 1.)

def smx(c, a):
    m = c.max(); s = a*(c-m); mx = s.max()
    if mx > 500: return m + (mx + np.log(np.sum(np.exp(s-mx))))/a
    return m + np.log(np.sum(np.exp(s)))/a

def gsm(h, n, a):
    c = c5v(h,n); m = c.max(); s = a*(c-m)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n); dx = 2./n
    for k in range(2*n-1):
        if w[k] < 1e-14: continue
        lag = k-(n-1); wm = w[k]*dx
        if lag >= 0:
            nv = n-lag; g[:nv] += wm*(1.-h[lag:]); g[lag:] -= wm*h[:nv]
        else:
            kk = -lag; nv = n-kk; g[kk:] += wm*(1.-h[:nv]); g[:nv] -= wm*h[kk:]
    return g

def _op(x, n, a, mu): return smx(c5v(x,n), a) + mu*(x.sum()-n/2)**2
def _gp(x, n, a, mu): return gsm(x,n,a) + 2*mu*(x.sum()-n/2)
def _os(x, n, a): return smx(c5v(x,n), a)
def _ce(x, n): return x.sum()-n/2
def _je(x, n): return np.ones(n)

def lbf(h, n, dl, al=None, mi=200):
    if al is None:
        al = [(50,5),(200,10),(1000,20),(5000,50),(20000,100),
              (100000,500),(500000,2000),(2000000,10000)]
    bd = [(0.,1.)]*n; bh, bc = h.copy(), c5_of(h,n)
    for a, mu in al:
        if time.time() > dl-2: break
        try:
            r = minimize(_op, bh, jac=_gp, args=(n,a,mu), method='L-BFGS-B',
                        bounds=bd, options={'maxiter':mi,'ftol':1e-18,
                        'gtol':1e-15,'maxfun':mi*20,'maxls':50})
            hn = proj(r.x, n); cn = c5_of(hn, n)
            if cn < bc: bc, bh = cn, hn
        except: pass
    return bh, bc

def slq(h, n, dl, a=500000, mi=1000):
    bd = [(0.,1.)]*n
    cs=[{'type':'eq','fun':_ce,'jac':_je,'args':(n,)}]
    try:
        r = minimize(_os, h, args=(n,a), method='SLSQP', bounds=bd,
                    constraints=cs, options={'maxiter':mi,'ftol':1e-16})
        hn = proj(r.x, n)
        return hn, c5_of(hn, n)
    except: return h.copy(), c5_of(h, n)

def sgd_go(h, n, dl, lr=0.005):
    bh, bc = h.copy(), c5_of(h,n)
    for _ in range(20000):
        if time.time() > dl: break
        c = c5v(h,n); top = np.argsort(c)[-3:]
        g = np.zeros(n); dx = 2./n
        for k in top:
            lag = k-(n-1)
            if lag >= 0:
                nv = n-lag; g[:nv] += dx*(1-h[lag:]); g[lag:] -= dx*h[:nv]
            else:
                kk=-lag; nv=n-kk; g[kk:] += dx*(1-h[:nv]); g[:nv] -= dx*h[kk:]
        g /= len(top); gn = np.linalg.norm(g)
        if gn < 1e-15: break
        g /= gn; step = lr; ok = False
        for _ in range(25):
            hn = proj(h-step*g, n); cn = c5_of(hn,n)
            if cn < bc-1e-15: h,bc,bh = hn,cn,hn.copy(); ok=True; break
            step *= 0.5
        if not ok:
            lr *= 0.7
            if lr < 1e-14: break
    return bh, bc

def exact_pair(h, n, dl):
    """Exact 1D golden-section for selected pairs."""
    bc = c5_of(h, n)
    for it in range(50):
        if time.time() > dl: break
        imp = False
        for i in range(n):
            if time.time() > dl: return h, bc
            js = [(i-1)%n, (i+1)%n, (i+n//2)%n, (i+7)%n, (i-7)%n]
            for j in js:
                if j == i: continue
                s = h[i]+h[j]
                lo = max(0., s-1.); hi = min(1., s)
                if hi-lo < 1e-12: continue
                gr = (np.sqrt(5)-1)/2
                a, b = lo, hi
                oi, oj = h[i], h[j]
                for _ in range(25):
                    m1 = b-gr*(b-a); m2 = a+gr*(b-a)
                    h[i]=m1; h[j]=s-m1; v1 = c5_of(h,n)
                    h[i]=m2; h[j]=s-m2; v2 = c5_of(h,n)
                    if v1 <= v2: b = m2
                    else: a = m1
                    if b-a < 1e-12: break
                h[i]=(a+b)/2; h[j]=s-h[i]
                c = c5_of(h,n)
                if c < bc-1e-14: bc=c; imp=True
                else: h[i]=oi; h[j]=oj
        if not imp: break
    return h, bc

def psearch(h, n, dl):
    bc = c5_of(h,n)
    for d in [0.01,0.005,0.001,0.0005,0.0001]:
        while time.time() < dl:
            imp = False
            g = gsm(h, n, 1000)
            for i in np.argsort(g):
                if time.time() > dl: return h, bc
                for j in range(max(0,i-50), min(n,i+51)):
                    if i==j: continue
                    di = d if g[i]<g[j] else -d
                    if 0<=h[i]+di<=1 and 0<=h[j]-di<=1:
                        h[i]+=di; h[j]-=di
                        c = c5_of(h,n)
                        if c < bc-1e-14: bc=c; imp=True
                        else: h[i]-=di; h[j]+=di
            if not imp: break
    return h, bc

def interp_h(h, nn):
    return np.interp(np.linspace(0,2,nn), np.linspace(0,2,len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time(); dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy(); n0 = len(h0)
    except:

    best_h = proj(h0, n0); best_c = c5_of(best_h, n0)
    
    # Phase 1: Multi-resolution
    for nt in [50, 100, 200]:
        if time.time() > dl-400: break
        ht = proj(interp_h(best_h, nt), nt)
        al = min(80, dl-time.time()-300)
        if al > 15:
            ht, ct = lbf(ht, nt, time.time()+al*0.7, mi=150)
            if time.time() < dl-250:
                ht, ct = slq(ht, nt, time.time()+min(20,dl-time.time()-200), mi=500)
        if nt < n0:
            hi = proj(interp_h(ht, n0), n0); ci = c5_of(hi, n0)
            if ci < best_c: best_c, best_h = ci, hi
        elif ct < best_c: best_c, best_h = ct, ht
    
    # Phase 2: L-BFGS-B aggressive
    nb = len(best_h)
    if time.time() < dl-250:
        best_h, best_c = lbf(best_h, nb, min(time.time()+250, dl-200),
            al=[(1000,20),(10000,50),(100000,500),(1000000,5000),
                (5000000,20000),(20000000,50000),(100000000,100000)], mi=300)
    
    # Phase 3: SLSQP
    if time.time() < dl-150:
        for a in [1000000, 5000000, 20000000]:
            if time.time() > dl-100: break
            h2, c2 = slq(best_h, nb, time.time()+min(40,dl-time.time()-80), a=a, mi=1500)
            if c2 < best_c: best_c, best_h = c2, h2
    
    # Phase 4: Subgradient
    if time.time() < dl-80:
        h3, c3 = sgd_go(best_h, nb, time.time()+min(50, dl-time.time()-50))
        if c3 < best_c: best_c, best_h = c3, h3
    
    # Phase 5: Exact pair optimization
    if time.time() < dl-40:
        best_h, best_c = exact_pair(best_h, nb, dl-10)
    
    # Phase 6: Fixed-step pair search
    if time.time() < dl-10:

            al = min(25, dl-time.time()-10)

                        al=[(10000,50),(100000,500),(1000000,5000)], mi=100)
            if cp < best_c: best_c, best_h = cp, hp
    
    best_h = proj(best_h, len(best_h))
    best_c = c5_of(best_h, len(best_h))
    return best_h, best_c, len(best_h)
