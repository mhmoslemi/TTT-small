```python
    """Compute C5 for a binary h (0-1 valued)."""
    dx = 2.0 / n
    c = np.correlate(h_binary.astype(float), 1 - h_binary.astype(float), mode='full')
    return np.max(c) * dx

def binary_local_search(h, n, deadline):
    """Local search on binary h."""
    h = h.copy()
    best_c = binary_c5(h, n)
    
    while time.time() < deadline:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            for j in zeros:
                # Try swapping: h[i]=0, h[j]=1
                h[i] = 0; h[j] = 1
                c = binary_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                else:
                    h[i] = 1; h[j] = 0
        
        if not improved:
            break
    
    return h, best_c
```

```python
def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n, rng):
    h = h.copy()
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    ones = int(h.sum())
    target = n // 2
    if ones > target:
        idx = np.where(h == 1)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:ones - target]]] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:target - ones]]] = 1
    return h

def thue_morse_binary(n, target_ones, rng):
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    ones = int(tm.sum())
    if ones > target_ones:
        idx = np.where(tm == 1)[0]
        perm = rng.permutation(len(idx))
        tm[idx[perm[:ones - target_ones]]] = 0
    elif ones < target_ones:
        idx = np.where(tm == 0)[0]
        perm = rng.permutation(len(idx))
        tm[idx[perm[:target_ones - ones]]] = 1
    return tm

def greedy_binary(n, target_ones, deadline):
    h = np.zeros(n)
    for step in range(target_ones):
        if time.time() > deadline:
            break
        best_pos, best_c = 0, float('inf')
        for pos in range(n):
            h[pos] = 1
            c = compute_c5(h, n)
            h[pos] = 0
            if c < best_c:
                best_c = c
                best_pos = pos
        h[best_pos] = 1
    return h

def binary_local_search(h, n, deadline, max_rounds=300):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            if time.time() > deadline:
                return h, best_c
            for j in zeros:
                h[i] = 0; h[j] = 1
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c; improved = True
                else:
                    h[i] = 1; h[j] = 0
        if not improved:
            break
    return h, best_c

def binary_sa(h, n, deadline, rng, t0=0.005, t1=0.00001):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    start_t = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_t
        total = max(deadline - start_t, 1)
        T = t0 * (t1 / t0) ** (elapsed / total)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0; h[j] = 1
        nc = compute_c5(h, n)
        delta = nc - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c = nc
            if nc < best_c:
                best_c = nc; best_h = h.copy()
        else:
            h[i] = 1; h[j] = 0
    return best_h, best_c

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    cs = alpha * (c - c.max())
    mx = cs.max()
    w = np.exp(cs - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_opt(h, n, deadline, max_iters=400):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(200, 5.0), (1000, 20.0), (5000, 100.0),
               (50000, 500.0), (500000, 2000.0), (2000000, 10000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters * 20, 'maxls': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def nesterov_opt(h, n, deadline, alpha=50000, lr=0.005, momentum=0.85):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    y = h.copy()
    for it in range(50000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(y, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        x_new = y - lr * g
        y = x_new + momentum * (x_new - y)
        h_new = project_feasible(y, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        if c_new > best_c + 0.01:
            lr *= 0.5
            y = best_h.copy()
        elif it % 200 == 0:
            alpha *= 1.2
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius) if radius else 0
                hi = min(n, i + radius + 1) if radius else n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c
```
