```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def grad_smooth_max(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ex = np.exp(np.clip(beta * (corr - np.max(corr)), -700.0, 700.0))
    w = ex / ex.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-9:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += wk * (1.0 - h[idx + lag])
            grad[idx + lag] -= wk * h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += wk * (1.0 - h[idx])
            grad[idx] -= wk * h[idx + la]
    return grad * dx


def subgrad_single(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def obj_lbfgs(x, dx, beta, target, lam):
    hc = np.clip(x, 0, 1)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    return smooth_max(corr, beta) * dx + lam * (hc.sum() - target) ** 2


def obj_lbfgs_grad(x, dx, beta, target, lam):
    hc = np.clip(x, 0, 1)
    g = grad_smooth_max(hc, dx, beta)
    g += 2.0 * lam * (hc.sum() - target) * np.ones(len(hc))
    return g


def optimize_resolution(h, n, tl):
    dx = 2.0 / n
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    h_curr = h.copy()
    c5_curr = compute_c5(h_curr)

    for beta in [20, 80, 300, 1200, 5000, 20000, 60000]:
        if tl() < 8:
            break
        lam = 50.0
        try:
            res = minimize(obj_lbfgs, h_curr,
                           args=(dx, beta, target, lam),
                           jac=lambda x: obj_lbfgs_grad(x, dx, beta, target, lam),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-13, 'maxls': 40})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_curr:
                h_curr = h_new
                c5_curr = c5_new
        except Exception:
            pass

    # SLSQP polish with gradient
    for beta in [5000, 30000]:
        if tl() < 6:
            break

        def f_slq(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return smooth_max(corr, beta) * dx

        def g_slq(x):
            return grad_smooth_max(x, dx, beta)

        def cfun(x):
            return x.sum() - target

        def cjac(x):
            return np.ones(n)

        try:
            res = minimize(f_slq, h_curr, jac=g_slq, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                           options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_curr:
                h_curr = h_new
                c5_curr = c5_new
        except Exception:
            pass

    return h_curr, c5_curr


def momentum_subgrad(h, n, tl, max_iter=80000):
    dx = 2.0 / n
    target = n / 2.0
    h_curr = h.copy()
    v = np.zeros(n)
    mu = 0.85
    best_local = compute_c5(h_curr)

    for it in range(max_iter):
        if tl() < 15:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        k_worst = np.argmax(corr)
        grad = subgrad_single(h_curr, k_worst, n)
        lr = 0.015 / (1.0 + it * 0.0004)
        v = mu * v - lr * grad
        h_new = project_feasible(h_curr + v, target)
        h_curr = h_new
        if it % 20000 == 19999:
            c5 = compute_c5(h_curr)
            if c5 < best_local:
                best_local = c5
            v = np.zeros(n)  # momentum restart

    c5 = compute_c5(h_curr)
    if c5 < best_local:
        best_local = c5
    # Return best of (final, best seen) - track both
    return h_curr, best_local


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n0 = len(h_init)
    best_h = project_feasible(h_init, n0 / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at original resolution
    h1, c5_1 = optimize_resolution(best_h, n0, tl)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Momentum subgradient
    h2, c5_2 = momentum_subgrad(best_h, len(best_h), tl, max_iter=60000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        # Recompute to get the h at best_c5
    # Re-evaluate h2
    c5_h2 = compute_c5(h2)
    if c5_h2 < best_c5:
        best_h, best_c5 = h2, c5_h2

    # Phase 3: Resolution sweep
    n_base = len(best_h)
    offsets = [-20, -15, -10, -5, 5, 10, 15, 20, 30, 50]
    for dn in offsets:
        if tl() < 20:
            break
        nt = n_base + dn
        if nt < 30 or nt > 300:
            continue
        h_t = resample_h(best_h, nt)
        h_t = project_feasible(h_t, nt / 2.0)
        h_t, c5_t = optimize_resolution(h_t, nt, tl)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 4: Another subgradient round at best resolution
    if tl() > 30:
        h3, _ = momentum_subgrad(best_h, len(best_h), tl, max_iter=40000)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Phase 5: Final SLSQP polish
    n = len(best_h)
    dx = 2.0 / n
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    for beta in [50000, 150000, 400000]:
        if tl() < 6:
            break

        def f_final(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return smooth_max(corr, beta) * dx

        def g_final(x):
            return grad_smooth_max(x, dx, beta)

        def cfun(x):
            return x.sum() - target

        def cjac(x):
            return np.ones(n)

        try:
            res = minimize(f_final, best_h, jac=g_final, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                           options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
```
