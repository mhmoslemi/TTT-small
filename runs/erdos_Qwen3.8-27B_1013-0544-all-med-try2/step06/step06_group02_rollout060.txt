                    h[i] -= delta; h[j] += delta
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:

                # Move from j to i

                    h[j] -= delta; h[i] += delta
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c; improved = True
                    else:
                        h[j] += delta; h[i] -= delta
        if not improved:
            delta *= 0.6
    return h, best_c

def nesterov_phase(h, n, deadline, alpha=5000, mu=100.0, lr=0.005):
    """Nesterov-accelerated gradient descent."""
    h = project_feasible(h, n)
    x = h.copy()
    y = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    t_val = 1.0
    for it in range(50000):
        if time.time() > deadline:
            break
        g = _grad(y, n, alpha, mu)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        x_new = project_feasible(y - lr * g / g_norm, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        momentum = (t_val - 1.0) / t_new
        y_new = x_new + momentum * (x_new - x)
        y_new = project_feasible(y_new, n)
        c_new = compute_c5(x_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = x_new.copy()
        x, y, t_val = x_new, y_new, t_new
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(30, 5.0), (100, 10.0), (500, 10.0),
               (2000, 50.0), (10000, 100.0), (50000, 500.0),
               (200000, 2000.0), (1000000, 5000.0), (5000000, 10000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_obj, h, jac=_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 10})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, start_delta=0.005, radius=50):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    delta = start_delta
    while delta >= 5e-5 and time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if i == j:
                    continue
                if h[i] >= delta and h[j] <= 1.0 - delta:
                    h[i] -= delta; h[j] += delta
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c; improved = True
                    else:
                        h[i] += delta; h[j] -= delta
                if h[j] >= delta and h[i] <= 1.0 - delta:
                    h[j] -= delta; h[i] += delta
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c; improved = True
                    else:
                        h[j] += delta; h[i] -= delta
        if not improved:
            delta *= 0.6
        else:
            delta *= 0.8
    return h, best_c

def coord_thresh_search(h, n, deadline):
    """Try setting each coordinate to 0 or 1, adjusting another to fix sum."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    target = n / 2.0
    for _ in range(5):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for val in [0.0, 1.0]:
                if abs(h[i] - val) < 1e-6:
                    continue
                diff = target - (h.sum() - h[i] + val)
                if diff > 0:
                    room = 1.0 - h
                    room[i] = 1.0 - val
                    tot = room.sum()
                    if tot < 1e-10:
                        continue
                    h_new = h.copy()
                    h_new[i] = val
                    h_new += (diff / tot) * room
                    h_new = np.clip(h_new, 0, 1)
                else:
                    if h.sum() - h[i] + val < 1e-10:
                        continue
                    h_new = h.copy()
                    h_new[i] = val
                    frac = (h.sum() - h[i] + val) / h.sum() if h.sum() > 1e-10 else 1
                    h_new = h_new * frac
                    h_new = np.clip(h_new, 0, 1)
                c = compute_c5(h_new, n)
                if c < best_c - 1e-14:
                    best_c = c
                    h = h_new
                    improved = True
        if not improved:
            break
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=1000):
    def _sobj(x, n2, a2):
        c = compute_c5_vec(x, n2)
        m = c.max()
        sh = a2 * (c - m)
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / a2)
    def _scon(x, n2):
        return float(x.sum() - n2 / 2.0)
    def _sac(x, n2):
        return np.ones(n2)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _scon, 'jac': _sac, 'args': (n,)}]
    try:
        res = minimize(_sobj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)
