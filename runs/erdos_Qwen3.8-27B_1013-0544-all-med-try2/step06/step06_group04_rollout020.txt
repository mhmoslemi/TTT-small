from scipy.optimize import minimize
import time
import cvxpy as cp

    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_all_lags(h, n)))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Gradient of c(lag) w.r.t. h, vectorized."""
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[m:] += (1.0 - h[:-m])
        grad[:-m] -= h[m:]
    else:
        grad[m:] += (1.0 - h[:-m])
        grad[:-m] -= h[m:]
    return grad

def active_subgradient(h, n, top_m=5, tol=0.001):
    """Average gradient of top active lags."""
    c = compute_all_lags(h, n)
    max_c = c.max()
    active = np.where(c >= max_c - tol)[0]
    # Include at least top_m
    if len(active) < top_m:
        top_idx = np.argsort(c)[-top_m:]
        active = np.unique(np.concatenate([active, top_idx]))
    total_grad = np.zeros(n)
    for idx in active:
        lag = idx - (n - 1)
        total_grad += lag_gradient(h, n, lag)
    total_grad /= len(active)
    return total_grad, max_c

def nesterov_subgrad(h, n, max_iter, deadline, top_m=5, tol=0.001):
    """Nesterov accelerated subgradient method."""
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    R = 1.0  # initial radius
    y = h_curr.copy()
    
    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        # Nesterov step
        alpha_t = 2.0 / (t + 3.0)
        y = project_feasible(h_curr + (y - h_curr) * (t - 1.0) / (t + 2.0), n)
        
        # Compute subgradient at y
        g, c_y = active_subgradient(y, n, top_m, tol)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g = g / g_norm
        
        # Step size
        eta = R / (np.sqrt(2.0 * t) * g_norm)  # actually R/(L*sqrt(2t))
        h_new = project_feasible(y - eta * g, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
    
    return best_h, best_c

def multi_lag_local_search(h, n, deadline, top_m=8, max_rounds=500):
    """Local search that considers multiple active lags simultaneously."""
    h_curr = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_curr, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        max_c = c.max()
        
        # Find active lags (within tolerance of max)
        tol = max(0.0005, 0.05 * (max_c - c.min()))
        active = np.where(c >= max_c - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        # Limit to top_m most active
        if len(active) > top_m:
            order = np.argsort(c[active])[::-1][:top_m]
            active = active[order]
        
        # Compute average gradient of active lags
        total_g = np.zeros(n)
        for idx in active:
            lag = idx - (n - 1)
            total_g += lag_gradient(h_curr, n, lag)
        total_g /= len(active)
        
        # Project gradient to sum-zero direction (to preserve sum constraint)
        total_g -= total_g.mean()
        
        # Also project to feasible direction (respect bounds)
        # For variables at 0, gradient should be >= 0 (can only increase)
        # For variables at 1, gradient should be <= 0 (can only decrease)
        mask_lo = h_curr < 1e-10
        mask_hi = h_curr > 1 - 1e-10
        # Zero out infeasible directions
        total_g[mask_lo & (total_g < 0)] = 0
        total_g[mask_hi & (total_g > 0)] = 0
        total_g -= total_g.mean()
        
        g_norm = np.linalg.norm(total_g)
        if g_norm < 1e-14:
            break
        
        # Line search
        best_step_c = best_c
        best_step_h = h_curr.copy()
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_try = project_feasible(h_curr - step * total_g / g_norm, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_step_c:
                best_step_c = c_try
                best_step_h = h_try
                break
        
        if best_step_c < best_c - 1e-14:
            best_c = best_step_c
            best_h = best_step_h.copy()
            h_curr = best_step_h
        elif best_step_c >= best_c:
            # Try smaller steps
            for step in [0.00005, 0.00002, 0.00001]:
                h_try = project_feasible(h_curr - step * total_g / g_norm, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-14:
                    best_c = c_try
                    best_h = h_try.copy()
                    h_curr = best_step_h
                    break
            else:
                break
    
    return best_h, best_c

            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_new = slsqp_smooth(best_h, n, 100000, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 3:
            best_h, best_c = lp_guided_search(best_h, n, deadline - 1, top_m=10, max_rounds=40)
            best_h = project_feasible(best_h, n)
            best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = compute_c5(best_h, 300)
    return best_h, best_c, len(best_h)
