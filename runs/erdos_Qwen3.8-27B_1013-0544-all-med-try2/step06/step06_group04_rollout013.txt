import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj_func(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    s = alpha * (c - m)
    s = s - s.max()
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _eq_con(x, n):
    return np.sum(x) - n / 2.0

def _eq_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_func, h, args=(n, alpha, top_idx),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def grad_descent(h, n, deadline, alpha_smooth=1000, top_k=5):
    """Projected gradient descent with adaptive step size."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.1
    n_inner = n
    
    while time.time() < deadline:
        c = compute_all_lags(h, n)
        # Get top_k active lags
        top_idx = np.argsort(c)[-top_k:]
        
        # Compute smoothed objective gradient
        # For each active lag k, gradient of c_k w.r.t. h[i] is:
        # For lag l > 0: dc_k/dh[i] = -dx*h[i+l] for i < n-l, and dx*(1-h[i-l]) for i >= l
        # For lag l = 0: dc_k/dh[i] = dx*(1-2*h[i])
        # For lag l < 0 (i.e., -(n-1-k)): similar
        
        grad = np.zeros(n)
        weights = np.zeros(len(top_idx))
        c_active = c[top_idx]
        m = c_active.max()
        exp_vals = np.exp(alpha_smooth * (c_active - m))
        weights = exp_vals / exp_vals.sum()
        
        for w, ki in zip(weights, top_idx):
            lag = ki - (n - 1)
            ml = abs(lag)
            if ml == 0:
                grad += w * (1.0 - 2.0 * h)
            elif lag > 0:
                grad[:n-ml] += w * (-h[ml:])
                grad[ml:] += w * (1.0 - h[:n-ml])
            else:
                grad[:n-ml] += w * (1.0 - h[ml:])
                grad[ml:] += w * (-h[:n-ml])
        
        # Project gradient to satisfy sum constraint (remove mean component)
        grad -= np.mean(grad)
        
        # Take step
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            step = min(step * 1.1, 0.5)
        else:
            h_new = h - (step/2) * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
            step *= 0.7
        
        h = best_h if c_new >= best_c else h_new
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj_func(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    s = alpha * (c - m)
    s = s - s.max()
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _eq_con(x, n):
    return np.sum(x) - n / 2.0

def _eq_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter, top_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_func, h, args=(n, alpha, top_idx),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, lag):
    m = abs(lag)
    grad = np.zeros(n)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += (1.0 - h[:n-m])
    else:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

def gradient_step(h, n, deadline, top_k=8, max_iters=50000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.02
    alpha_sm = 500
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_k:]
        c_act = c[top_idx]
        m = c_act.max()
        ev = np.exp(alpha_sm * (c_act - m))
        w = ev / ev.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            lag = ki - (n - 1)
            grad += wi * lag_gradient(h, n, lag)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm * step
        h_new = project_feasible(h - grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.05, 0.3)
        else:
            step *= 0.6
            if step < 1e-8:
                break
    return best_h, best_c

def pair_search(h, n, deadline, max_rounds=200000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(max_rounds):
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break
            if lag > 0:
                bad = np.where((h[:n-m] - h[m:]) > 0.05)[0]
            else:
                bad = np.where((h[m:] - h[:n-m]) > 0.05)[0]
            if len(bad) == 0:
                break
            idx = bad[np.argmax((h[bad] - h[bad + m]).abs() if lag > 0 else (h[bad+m] - h[bad]).abs())]
            i = int(idx)
            j = i + m
            if lag > 0:
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.4)
                    if delta < 1e-8:
                        continue
                    h[i] -= delta
                    h[j] += delta
                else:
                    continue
            else:
                if h[j] > h[i]:
                    delta = min(d, (h[j] - h[i]) * 0.4)
                    if delta < 1e-8:
                        continue
                    h[j] -= delta
                    h[i] += delta
                else:
                    continue
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h.copy()
            else:
                if lag > 0:
                    h[i] += delta
                    h[j] -= delta
                else:
                    h[j] += delta
                    h[i] -= delta
                h = project_feasible(h, n)
    return best_h, best_c

def active_set_loop(h, n, deadline, max_rounds=50):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [2000, 5000, 10000, 20000, 50000, 100000, 200000, 500000]
    ai = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-6:]
        alpha = alphas[min(ai, len(alphas)-1)]
        h_new = slsqp(h, n, alpha, maxiter=2000, top_idx=top_idx)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            if rnd % 5 == 0 and ai < len(alphas)-1:
                ai += 1
        else:
            if rnd % 3 == 0:
                alpha = min(alpha * 2, 2000000)
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    if time.time() < deadline - 40:
        h1 = slsqp(best_h, n, 5000, 3000)
        c1 = compute_c5(h1, n)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    if time.time() < deadline - 30:

        h2, c2 = active_set_loop(best_h, n, as_end)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    if time.time() < deadline - 20:

        if c3 < best_c:
            best_c = c3

            best_c = c5
            best_h = h5

    best_h = project_feasible(best_h, n)

    return best_h, best_c
