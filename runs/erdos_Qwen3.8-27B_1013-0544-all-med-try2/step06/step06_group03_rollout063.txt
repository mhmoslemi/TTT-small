import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = corr * dx
    m = c.max()
    shift = (c - m) * alpha
    max_sh = shift.max()
    exp_sh = np.exp(shift - max_sh)
    log_sum = max_sh + np.log(exp_sh.sum())
    obj_val = m + log_sum / alpha
    p = exp_sh / exp_sh.sum()
    A = 1.0 - x
    corr_p = np.correlate(A, p, mode='full')
    conv_p = np.convolve(x, p, mode='full')
    first_term = corr_p[n - 1:2 * n - 1]
    second_term = conv_p[n - 1:2 * n - 1]
    grad = dx * (first_term - second_term)
    return obj_val, grad

def lbfgs_opt(x, n, alpha, maxiter, deadline):
    def fun_and_grad(v):
        if time.time() > deadline:
            return 0.0, np.zeros(n)
        vc = np.clip(v, 0.0, 1.0)
        val, grad = smooth_max_and_grad(vc, n, alpha)
        s = vc.sum() - n / 2.0
        mu = 1e4
        val += mu * s * s
        grad += 2.0 * mu * s
        return val, grad
    try:
        res = minimize(fun_and_grad, x, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(x, n)

def compute_effects(h, n):
    eff = np.zeros((n, 2 * n - 1))
    for j in range(n):
        for m in range(2 * n - 1):
            idx1 = j + m - n + 1
            idx2 = j - m + n - 1
            if 0 <= idx1 < n:
                eff[j, m] += (1.0 - h[idx1])
            if 0 <= idx2 < n:
                eff[j, m] -= h[idx2]
    return eff

def fast_pair_search(h, n, deadline, max_delta=0.02):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    best_c = corr.max() * dx
    effects = compute_effects(h, n)
    updates_since_recompute = 0
    cur_max_delta = max_delta

    while time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            for j in range(i + 1, n):
                d = effects[i] - effects[j]
                d_lo = max(-h[i], h[j] - 1.0, -cur_max_delta)
                d_hi = min(1.0 - h[i], h[j], cur_max_delta)
                if d_lo >= d_hi - 1e-14:
                    continue
                # Try several delta values
                best_val = corr.max()
                best_delta = 0.0
                for frac in (0.15, 0.35, 0.5, 0.65, 0.85):
                    delta = d_lo + frac * (d_hi - d_lo)
                    if abs(delta) < 1e-14:
                        continue
                    val = (corr + delta * d).max()
                    if val < best_val - 1e-13:
                        best_val = val
                        best_delta = delta
                if best_delta != 0 and best_val < best_c / dx - 1e-13:
                    h[i] += best_delta
                    h[j] -= best_delta
                    corr = np.correlate(h, 1 - h, mode='full')
                    new_c = corr.max() * dx
                    if new_c < best_c:
                        best_c = new_c
                    else:
                        # Revert
                        h[i] -= best_delta
                        h[j] += best_delta
                        corr = np.correlate(h, 1 - h, mode='full')
                    improved = True
                    updates_since_recompute += 1
                    if updates_since_recompute >= 80:
                        effects = compute_effects(h, n)
                        updates_since_recompute = 0
        if not improved:
            effects = compute_effects(h, n)
            cur_max_delta *= 0.6
            if cur_max_delta < 1e-6:
                break
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    # L-BFGS-B with increasing alpha
    for alpha in (2000, 10000, 50000, 200000):
        if time.time() > deadline - 15:
            break
        h_new = lbfgs_opt(h, n, alpha, 200, deadline)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            h = h_new
    # Fast pair search
    if time.time() < deadline - 8:
        h, best_c = fast_pair_search(h, n, deadline - 3, max_delta=0.03)
    # Another L-BFGS polish
    if time.time() < deadline - 3:
        h_new = lbfgs_opt(h, n, 500000, 200, deadline)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            h = h_new
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()

    if c1 < best_c:

        if n_try == n_init:

        x_old = np.linspace(0, 2, len(best_h))

        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(120, (deadline - time.time()) * 0.2)

        h_t, c_t = optimize_at_n(h_interp, n_try, time.time() + alloc, rng)
        if c_t < best_c:

    # Phase 3: Multi-restart perturbation at best n
    n_best = len(best_h)
    remaining = deadline - time.time() - 30
    n_restarts = max(2, int(remaining / 50))
    for r in range(n_restarts):
        if time.time() > deadline - 25:
            break
        scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
        hp = best_h + rng.normal(0, scale, n_best)
        hp = project_feasible(hp, n_best)
        alloc = min(40, (deadline - time.time() - 15) * 0.3)

        h_r, c_r = optimize_at_n(hp, n_best, time.time() + alloc, rng)
        if c_r < best_c:

    # Phase 4: Final refinement
    if time.time() < deadline - 10:
        n_final_start = len(best_h)
        best_h, best_c = fast_pair_search(best_h, n_final_start, deadline - 5, max_delta=0.01)
    if time.time() < deadline - 3:
        n_f = len(best_h)

        c = compute_c5(h_new, n_f)
        if c < best_c:
            best_c, best_h = c, h_new
        if time.time() < deadline - 1:
            best_h, best_c = fast_pair_search(best_h, n_f, deadline - 1, max_delta=0.005)

    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out)
    best_c = compute_c5(best_h, n_out)
