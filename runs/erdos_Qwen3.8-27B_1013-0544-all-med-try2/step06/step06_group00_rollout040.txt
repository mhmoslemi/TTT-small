from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def c5_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    raw = np.correlate(x, 1.0 - x, mode='full')
    C = raw * dx
    m = C.max()
    shifted = alpha * (C - m)
    # Numerical stability
    shifted = shifted - shifted.max()
    exps = np.exp(shifted)
    logsum = np.log(exps.sum())
    W = exps / exps.sum()
    f = m + logsum / alpha
    # Gradient
    part1 = np.convolve(1.0 - x, W, mode='full')[n-1:2*n-1]
    part2 = np.correlate(x, W, mode='full')[n-1:2*n-1]
    grad = dx * (part1 - part2)
    return f, grad

def slsqp_analytic(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        return c5_obj_and_grad(x, n, alpha)[0]
    def grad(x):
        return c5_obj_and_grad(x, n, alpha)[1]
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def anneal_binary(h_init, n, deadline, seed=42, n_iters=5000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()

    scale = 5000.0

            break

        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]

        delta = np.zeros(2 * n - 1)
        im = i + m_range

        imn = i - m_range

        jm = j + m_range

            cur_c = nc

            ones_arr[bi] = i

                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
