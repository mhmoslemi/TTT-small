def single_step_delta(c_full, h, i, delta, n):
    """Update full correlation when h[i] changes by delta (assuming h[i]+delta is in [0,1])."""
    c_new = c_full.copy()
    for m in range(2*n - 1):

    delta = np.zeros(2*n - 1)

        if 0 <= i - lag < n:

        if 0 <= i + lag < n:

    # i as first: valid when 0 <= i - lag < n, i.e., i-n+1 <= lag <= i
    mask1_i = (lags >= (i - n + 1)) & (lags <= i) & (lags >= -(n-1)) & (lags <= n-1)

    idx1 = i - lags[mask1_i]

    delta1[mask1_i] = -(1.0 - h[idx1])
    
    # i as second: valid when 0 <= i + lag < n, i.e., -i <= lag <= n-1-i
    mask2_i = (lags >= (-i)) & (lags <= (n-1-i)) & (lags >= -(n-1)) & (lags <= n-1)

    mask1_j = (lags >= (j - n + 1)) & (lags <= j) & (lags >= -(n-1)) & (lags <= n-1)
