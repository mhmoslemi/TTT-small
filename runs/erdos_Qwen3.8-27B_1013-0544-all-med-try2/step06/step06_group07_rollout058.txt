Actually, numpy's correlate: np.correlate(a, v)[k] = sum_n a[n+k] * conj(v[n]) (for the "full" mode, with appropriate zero-padding).

This is getting complicated for a coordinate descent. Let me just use a simpler approach.

Actually, for the final stages of optimization (when we're very close to optimum), a simple approach that works well is:

**Projected gradient with backtracking line search on the true (non-smooth) objective.**

At each step:
1. Compute a subgradient of max_k f_k(h)
2. Take a step: h_new = h - lr * grad
3. Project onto feasible set
4. If f(h_new) > f(h), halve lr and retry
5. Accept if improved

This is like subgradient method with line search, which is more robust than fixed learning rate.

But actually, for the very fine improvements we need (0.00007), I think the best approach is:

1. Use SLSQP with very high beta (essentially solving the problem as if it's smooth)
2. Do many independent runs with small random perturbations
3. Keep the best

The key insight is that SLSQP with high beta is essentially doing a local optimization that treats the top few lags as the active constraints. With enough iterations and a good starting point, it should converge very close to the true optimum.

Let me also add a "Nelder-Mead" phase for the final few dimensions, and a "coordinate perturbation" phase where I try small changes to individual coordinates.

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-15:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def sum_con_fun(h, target_sum):
    return h.sum() - target_sum


def sum_con_jac(h, n):
    return np.ones(n)


def make_constraint(target_sum, n):
    return {'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac,
            'args': (target_sum, n)}


def multi_subgrad(h, overlaps, n, top_m=10):
    max_val = np.max(overlaps)
    if max_val <= 0:
        return np.zeros(n)
    threshold = max_val * (1.0 - 0.05)
    active = np.where(overlaps >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(overlaps)])
    if len(active) > top_m:
        active = np.argsort(overlaps)[::-1][:top_m]
    grad = np.zeros(n)
    for lag_idx in active:
        k = lag_idx - (n - 1)
        if k >= 0:
            idx = np.arange(n - k)
            grad[idx] += (1.0 - h[idx + k])
            grad[idx + k] -= h[idx]
        else:
            la = -k
            idx = np.arange(n - la)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad / len(active)


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-15:
            break

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:

        h = np.clip(h, 0.0, 1.0)

    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    max_val = np.max(overlaps)
    if max_val <= 0:
        return np.zeros(n)

    if len(active) == 0:
        active = np.array([np.argmax(overlaps)])
    if len(active) > top_m:
        active = np.argsort(overlaps)[::-1][:top_m]
