import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_objective(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = corr.max()
    exp_c = np.exp(alpha * (corr - m))
    lse_val = m + np.log(exp_c.sum()) / alpha
    sum_dev = x.sum() - n / 2.0
    return lse_val + penalty * sum_dev ** 2

def lse_gradient(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = corr.max()
    exp_c = np.exp(alpha * (corr - m))
    weights = exp_c / exp_c.sum()
    
    grad = np.ones(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-16:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n - s] -= w * x[s:]
            grad[s:] -= w * x[:n - s]
        else:
            sa = -s
            grad[:n - sa] -= w * x[sa:]
            grad[sa:] -= w * x[:n - sa]
    grad *= dx
    # Add penalty gradient
    sum_dev = x.sum() - n / 2.0
    grad += 2 * penalty * sum_dev
    return grad

def lbfgs_optimize(h, n, alpha, penalty=100.0, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    res = minimize(lse_objective, h, args=(n, alpha, penalty),
                   method='L-BFGS-B', bounds=bounds,
                   jac=lambda x: lse_gradient(x, n, alpha, penalty),
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
    return project_feasible(res.x, n)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = corr.max()
    e = np.exp(alpha * (corr - m))
    lse = m + np.log(e.sum()) / alpha
    sd = x.sum() - n / 2.0
    return lse + pen * sd * sd

def lse_grad(x, n, alpha, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = corr.max()
    e = np.exp(alpha * (corr - m))
    w = e / e.sum()
    
    g = np.ones(n)
    # Vectorized gradient computation
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s == 0:
            g -= 2.0 * wk * x
        elif s > 0:
            g[:n - s] -= wk * x[s:]
            g[s:] -= wk * x[:n - s]
        else:
            sa = -s
            g[:n - sa] -= wk * x[sa:]
            g[sa:] -= wk * x[:n - sa]
    g *= dx
    sd = x.sum() - n / 2.0
    g += 2.0 * pen * sd
    return g

def lbfgs_run(h, n, alpha, pen=1e4, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj, h, args=(n, alpha, pen), method='L-BFGS-B',
                       jac=lambda x: lse_grad(x, n, alpha, pen),
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

The KKT conditions give: x*[i] = clip(h[i] - λ, 0, 1) for some λ.

- When λ is very positive, clip(h[i] - λ, 0, 1) = 0 for all i, so sum = 0

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2, 0, 1)
    diff = target - result.sum()
    if abs(diff) > 1e-15:
        mask = (result > 1e-10) & (result < 1 - 1e-10)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0, 1)
    return result

def lse_obj(x, n, alpha, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = corr.max()
    e = np.exp(alpha * (corr - m))
    lse = m + np.log(e.sum()) / alpha
    sd = x.sum() - n / 2.0
    return lse + pen * sd * sd

def lse_grad(x, n, alpha, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = corr.max()
    e = np.exp(alpha * (corr - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s > 0:
            g[:n - s] += wk * (1 - x[s:]) + (-(x[:n - s]))
            g[s:] += wk * (1 - x[:n - s]) + (-(x[s:]))
        elif s < 0:
            sa = -s
            g[:n - sa] += wk * (1 - x[sa:]) + (-(x[:n - sa]))
            g[sa:] += wk * (1 - x[:n - sa]) + (-(x[sa:]))
        else:
            g += 2.0 * wk * (1 - x)
    g *= dx
    sd = x.sum() - n / 2.0
    g += 2.0 * pen * sd
    return g

def lbfgs_run(h, n, alpha, pen=1e4, maxiter=3000):

    try:
        res = minimize(lse_obj, h, args=(n, alpha, pen), method='L-BFGS-B',
                       jac=lse_grad, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
