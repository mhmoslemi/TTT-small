def compute_delta_for_swap(h, i, j, n):
    delta = np.zeros(2*n-1)
    hp_i = 1.0  # h'[i]
    hp_j = 0.0  # h'[j]
    
    for k in range(-(n-1), n):
        affected_ns = set()
        for n_val in [i, i-k, j, j-k]:
            if 0 <= n_val < n and 0 <= n_val+k < n:
                affected_ns.add(n_val)
        
        total = 0.0
        for n_val in affected_ns:
            # Get old values
            old_nk = h[n_val+k]
            old_n = h[n_val]
            # Get new values
            new_nk = hp_i if (n_val+k)==i else (hp_j if (n_val+k)==j else h[n_val+k])
            new_n = hp_i if n_val==i else (hp_j if n_val==j else h[n_val])
            total += new_nk*(1-new_n) - old_nk*(1-old_n)
        
        if total != 0:
            delta[n-1+k] = total
    
    return delta

def compute_delta_fast(h, i, j, n):

    delta = np.zeros(2*n-1)
    
    # For n=i (h[i] changes from 0 to 1):
    # Valid k: -i <= k <= n-1-i (so that 0<=i+k<n)
    k_lo = -i
    k_hi = n - 1 - i
    # delta at n-1+k for k in [k_lo, k_hi]:
    # old: h[i+k]*(1-0) = h[i+k]
    # new: h'[i+k]*(1-1) = 0
    # delta = -h[i+k]
    idx = n - 1 + np.arange(k_lo, k_hi+1)
    vals = np.arange(i+k_lo, i+k_hi+1)  # = np.arange(0, n-i)
    delta[idx] -= h[vals]
    
    # For n=i-k (h[n+k]=h[i] changes from 0 to 1):
    # n = i-k, valid when 0<=i-k<n, i.e., 0 < k <= i (k<i for i-k>0, k=i for i-k=0)
    # Wait: 0<=i-k<n means 0<=i-k and i-k<n, i.e., k<=i and k>i-n
    # And n+k=i must be in [0,n), which is 0<=i<n (always true)
    # So k ranges from max(0, i-n+1) to min(i, n-1) ... 
    # Actually k can be negative too. k > i-n means k >= i-n+1 (if i-n < 0, then k >= 0)
    # And k <= i (for i-k >= 0)
    # Also need n=i-k in [0, n-1]: 0 <= i-k <= n-1, so i-(n-1) <= k <= i
    k_lo2 = i - (n-1)  # = i-n+1
    k_hi2 = i
    k_lo2 = max(k_lo2, -(n-1))  # k >= -(n-1)
    k_hi2 = min(k_hi2, n-1)  # k <= n-1
    if k_lo2 <= k_hi2:
        kk = np.arange(k_lo2, k_hi2+1)
        nn = i - kk  # n values
        idx2 = n - 1 + kk
        # old: h[i]*(1-h[nn]) = 0
        # new: h'[i]*(1-h'[nn]) = 1*(1-h'[nn])
        # h'[nn] = 0 if nn==j, else h[nn]
        new_nn = h[nn].copy()
        new_nn[nn == j] = 0.0
        delta[idx2] += (1 - new_nn)
    
    # For n=j (h[j] changes from 1 to 0):
    # Valid k: -j <= k <= n-1-j
    k_lo3 = -j
    k_hi3 = n - 1 - j
    idx3 = n - 1 + np.arange(k_lo3, k_hi3+1)
    vals3 = np.arange(j+k_lo3, j+k_hi3+1)  # = np.arange(0, n-j)

    # new: h'[j+k]*(1-0) = h'[j+k]
    # h'[j+k] = 1 if j+k==i, else h[j+k]
    new_vals3 = h[vals3].copy()
    new_vals3[vals3 == i] = 1.0
    delta[idx3] += new_vals3
    
    # For n=j-k (h[n+k]=h[j] changes from 1 to 0):
