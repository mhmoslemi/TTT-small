                        if random.random() < dist_to_left / (2.0*n):
                            h[i] = max(0, 1 - 1e-5)
                        elif random.random() < dist_to_right / (2.0*n):
                            h[i] = min(1, 1e-5 + 1)
                    if strategy == ("move_peak_to_left", 25):
                        if i < n/3 and random.random() < 0.3:
                            h[i] += 0.2
                            h = np.clip(h, 0, 1)
                            break
        
        # Phase 4: Perform multiple assessments
        best_score = c
        best_confirmed = h
        
        if random.random() < 0.1:
            # Conduct a more detailed analysis with different measurements
            new_h = project_to_feasible_vectorized(h)
            _, best_lag = compute_overlap(new_h)
            
            # Pairwise genetic inversion
            best_lag_indices = set()
            try:
                for measure in range(3):
                    if random.random() < 0.5:
                        best_lag_indices.add(random.randint(0, len(h)-1))
            except:
                best_lag_indices = set([0, len(h)//2, len(h)-1])
            
            # Attempt to invert based on the best lags
            try:
                # This inverts the function around the best lags
                for idx in sorted(best_lag_indices, reverse=True):
                    h[idx] = 1.0 - h[idx]
                    if idx+1 < len(h):
                        h[idx+1] -= 0.1
                        h[idx-1] -= 0.1 if idx-1>=0 else 0
            except:
                # Simple variation
                h += 0.1 * (np.random.random(n) * 2 - 1)
                h = np.clip(h, 0, 1)
                
            new_score, _ = compute_overlap(h)
            if new_score < best_score * 0.9:
                best_score = new_score
                best_confirmed = copy.deepcopy(h)
        if best_score < c:
            c = best_score
            h = best_confirmed
    
    # Vary the solution
    for _ in range(runs):
        if time.time() > deadline:
            break
        
        h_copy = project_to_feasible_vectorized(initial_h)
        best_c = compute_overlap(h_copy)
        
        # Use replacement if initial is not good enough
        if best_c > 0.4:
            h = h_copy
            c = best_c
        
        # Track best solution
        best_h = h
        best_overlap = c
        
    # Final validation steps
    best_h = project_to_feasible_vectorized(best_h)
    overlap, lag = compute_overlap(best_h)
    
    # Return the best solution found
    return best_h, overlap, n_init
