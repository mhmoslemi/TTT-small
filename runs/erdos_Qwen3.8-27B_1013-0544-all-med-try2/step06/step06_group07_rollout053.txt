        h_cd2 = best_h.copy()
        n_cd2 = len(h_cd2)
        ts_cd2 = n_cd2 / 2.0
        order2 = np.zeros(n_cd2, dtype=int)

        for sweep in range(3000):
            if tl() < 30:
                break
            np.random.shuffle(order2)
            for i in order2:
                if tl() < 30:
                    break
                golden_coord_min(h_cd2, i, n_cd2, max_iter=15)
            h_cd2 = project_feasible(h_cd2, ts_cd2)
            if sweep % 100 == 99:
                c5_c2 = compute_c5(h_cd2)
                if c5_c2 < best_c5:
                    best_c5 = c5_c2
                    best_h = h_cd2.copy()

        c5_c2 = compute_c5(h_cd2)
        if c5_c2 < best_c5:
            best_c5 = c5_c2
            best_h = h_cd2.copy()

    # Phase 5: Final SLSQP polish
    if tl() > 15:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        dx_f = 2.0 / n_f
        for beta in [10000, 50000, 200000]:
            if tl() < 10:
                break
            h_new = slsqp_smooth(best_h, n_f, ts_f, dx_f, beta, maxiter=5000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 6: Quick coordinate descent polish
    if tl() > 10:
        n_p = len(best_h)
        ts_p = n_p / 2.0
        for sweep in range(500):
            if tl() < 5:
                break
            for i in range(n_p):
                golden_coord_min(best_h, i, n_p, max_iter=15)
            best_h = project_feasible(best_h, ts_p)
        c5_final = compute_c5(best_h)
        if c5_final < best_c5:
            best_c5 = c5_final

    return best_h, best_c5, len(best_h)
