                    improved = True
                    break
            if improved:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: SLSQP continuous optimization (15% of budget)
    p1 = t0 + 0.15 * (budget_s - 5)
    h1, c1 = slsqp_optimize(best_h, n, [2000, 10000, 100000, 1000000], p1)
    if c1 < best_c:
        best_c, best_h = c1, h1

    # Phase 2: Binary SA from continuous solution (50% of budget)
    p2 = t0 + 0.65 * (budget_s - 5)
    hb = to_binary(best_h, n)
    hb2, cb2 = sa_binary(hb, n, p2, seed=seed, t_init=0.15, cooling=0.99997, kick_freq=4000)
    if cb2 < best_c:
        best_c, best_h = cb2, hb2

    # Phase 3: Multi-resolution exploration (15% of budget)
    p3 = t0 + 0.80 * (budget_s - 5)
    for nh in [200, 240, 280, 300]:
        if time.time() > p3 - 20:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh = to_binary(hh, nh)
        hh2, ch2 = sa_binary(hh, nh, p3 - 15, seed=seed + nh, t_init=0.12, cooling=0.99998, kick_freq=3000)
        if ch2 < best_c:
            best_c, best_h = ch2, hh2
            n = nh

    # Phase 4: SLSQP polish from best binary (8% of budget)
    p4 = t0 + 0.88 * (budget_s - 5)
    if time.time() < p4:
        hp = project_feasible(best_h + 0.03 * rng.randn(n), n)
        hp2, cp2 = slsqp_optimize(hp, n, [50000, 500000, 5000000], p4)
        if cp2 < best_c:
            best_c, best_h = cp2, hp2

    # Phase 5: Final binary SA polish (remaining time)
    p5 = dl - 2
    hb = to_binary(best_h, n)
    hb3, cb3 = sa_binary(hb, n, p5, seed=seed + 1000, t_init=0.05, cooling=0.99999, kick_freq=6000)
    if cb3 < best_c:
        best_c, best_h = cb3, hb3

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
