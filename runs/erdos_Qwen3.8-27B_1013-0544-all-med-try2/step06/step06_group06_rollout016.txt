    top_k = np.argsort(c_all)[::-1][:num_active]

    # Variables: d[0..n-1], t (total n+1 variables)

    # For each active lag k: grad_k · d <= t - c_all[k]
    # sum(d) = 0
    # -h[i] <= d[i] <= 1-h[i]

        c_mat[idx, n] = -1  # -t

    A_eq = np.zeros((1, n + 1))

For binary h, c[k] = sum_m h[m] - sum_m h[m]*h[m+s] (where s = k-(n-1)).

Change in sum of products:
Old: h[i]*h[i+s] + h[j]*h[j+s] + h[i-s]*h[i] + h[j-s]*h[j] (terms involving i or j)
New: 0 + h[j]*h[j+s] ... wait this is getting complicated.

Let me think differently. The sum of products is: P[s] = sum_m h[m]*h[m+s]

After swap (i: 1→0, j: 0→1):
ΔP[s] = (new P[s]) - (old P[s])

Terms in P[s] that change:
- m=i: h[i]*h[i+s] → 0*h[i+s] = 0, change = -h[i+s]
- m=i-s: h[i-s]*h[i] → h[i-s]*0 = 0, change = -h[i-s]
- m=j: h[j]*h[j+s] → 1*h[j+s] = h[j+s], change = +h[j+s]
- m=j-s: h[j-s]*h[j] → h[j-s]*1 = h[j-s], change = +h[j-s]

(Where terms with out-of-bounds indices contribute 0.)

So: ΔP[s] = h[j+s] + h[j-s] - h[i+s] - h[i-s]
(where we use h[x]=0 if x is out of [0,n-1])

And since c[k] = sum(h) - P[s] and sum(h) doesn't change:
Δc[k] = -ΔP[s] = h[i+s] + h[i-s] - h[j+s] - h[j-s]

from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(s, h, n):
    grad = np.zeros(n)
    if s >= 0:
        rem = n - s
        if rem > 0:
            grad[:rem] += (1.0 - h[s:])
            grad[s:] -= h[:rem]
    else:
        sp = -s
        rem = n - sp
        if rem > 0:
            grad[sp:] += (1.0 - h[:rem])
            grad[:rem] -= h[sp:]
    return grad

def projected_gradient(h, n, deadline, max_iter=500000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.1
    n1 = n - 1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        # Top 3 active lags with softmax weighting
        top3 = np.argsort(c_raw)[::-1][:3]
        scale = 0.5 / (c_raw[top3[0]] + 1e-10)
        w = np.exp((c_raw[top3] - c_raw[top3[0]]) * scale)
        w /= w.sum()
        grad = w[0] * grad_lag(top3[0] - n1, h, n)
        grad += w[1] * grad_lag(top3[1] - n1, h, n)
        grad += w[2] * grad_lag(top3[2] - n1, h, n)
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.1, 1.0)
        else:
            step *= 0.8
            if step < 1e-10:
                break
    return best_h, best_c

    top_k = np.argsort(c_raw)[::-1][:num_active]
    n1 = n - 1

    b_ub = np.zeros(m + 1)
    for idx, k in enumerate(top_k):
        g = grad_lag(k - n1, h, n)
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
        b_ub[idx] = -c_raw[k]
    A_ub[m, :n] = 1.0
    b_ub[m] = 0.0
    A_ub_eq = None  # sum constraint is in A_ub as equality...

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(s, h, n):
    grad = np.zeros(n)
    if s >= 0:
        rem = n - s
        if rem > 0:
            grad[:rem] += 1.0 - h[s:]
            grad[s:] -= h[:rem]
    else:
        sp = -s
        rem = n - sp
        if rem > 0:
            grad[sp:] += 1.0 - h[:rem]
            grad[:rem] -= h[sp:]
    return grad

def proj_grad(h, n, deadline, max_iter=500000, top_k=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.1
    n1 = n - 1
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c_raw)[::-1][:top_k]
        # Softmax weights
        diff = c_raw[idx] - c_raw[idx[0]]
        w = np.exp(diff * 5.0)
        w /= w.sum()
        grad = np.zeros(n)
        for i, k in enumerate(idx):
            grad += w[i] * grad_lag(k - n1, h, n)
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.05, 2.0)
        else:
            step *= 0.75
            if step < 1e-12:
                break
    return best_h, best_c

def lp_direction(h, n, num_active=12):
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(c_raw)[::-1][:num_active]
    n1 = n - 1
    m = len(top)

    b_ub = np.zeros(m)

        g = grad_lag(k - n1, h, n)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -c_raw[k]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((0.0, None))
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n]
    return None

def lp_steps(h, n, deadline, max_iters=500):

    best_c = compute_c5(h, n)
    for _ in range(max_iters):

            if time.time() > deadline:

    return best_h, best_c
