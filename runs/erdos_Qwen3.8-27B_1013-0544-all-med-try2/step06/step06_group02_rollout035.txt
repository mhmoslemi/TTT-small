def smooth_max_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    val = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_refine(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in [(1000, 10.0), (5000, 50.0), (100000, 500.0)]:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iters * 5, 'maxls': 30})
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c - 1e-15:
                best_c = cn
                h = hn
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, deltas=None, radius=50):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, deadline, iters=300, grid_size=100, seed=None):
    dx = 2.0 / n
    offset = n - 1
    best_c = compute_c5(h, n)
    x_grid = np.linspace(0.0, 1.0, grid_size)
    rng = np.random.RandomState(seed)
    order = np.arange(n)
    for it in range(iters):
        if time.time() > deadline:
            break
        rng.shuffle(order)
        corr = np.correlate(h, 1 - h, mode='full')
        improved = False
        for i in order:
            if time.time() > deadline:
                return h, best_c
            hi = h[i]
            coef = np.zeros(2 * n - 1)
            k_lo = max(0, -i)
            k_hi = min(n - 1, n - 1 - i)
            if k_lo <= k_hi:
                k = np.arange(k_lo, k_hi + 1)
                coef[k + offset] += (1.0 - h[i + k])
            k2_lo = max(-(n - 1), i - (n - 1))
            k2_hi = min(n - 1, i)
            if k2_lo <= k2_hi:
                k2 = np.arange(k2_lo, k2_hi + 1)
                mask = k2 != 0
                if mask.any():
                    k2 = k2[mask]
                    coef[k2 + offset] += -h[i - k2]
            const = corr - hi * coef
            c0 = corr[n - 1] - hi + hi * hi
            aff = const[None, :] + x_grid[:, None] * coef[None, :]
            aff[:, n - 1] = c0 + x_grid - x_grid * x_grid
            f = aff.max(axis=1) * dx
            bi = f.argmin()
            bx, bv = x_grid[bi], f[bi]
            if 0 < bi < grid_size - 1:
                w = 2.0 / (grid_size - 1)
                lo = max(0.0, bx - 3 * w)
                hi2 = min(1.0, bx + 3 * w)
                xf = np.linspace(lo, hi2, 40)
                af = const[None, :] + xf[:, None] * coef[None, :]
                af[:, n - 1] = c0 + xf - xf * xf
                ff = af.max(axis=1) * dx
                fi = ff.argmin()
                if ff[fi] < bv:
                    bx, bv = xf[fi], ff[fi]
            if bv < best_c - 1e-13:

def smooth_max_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    val = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_refine(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in [(500, 5.0), (2000, 20.0), (10000, 100.0), (50000, 500.0)]:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iters * 5, 'maxls': 30})
            hn = project_feasible(res.x, n)
            cn = compute_c5(hn, n)
            if cn < best_c - 1e-15:
                best_c = cn
                h = hn
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, deltas=None, radius=50):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi2 = min(n, i + radius + 1)
                for j in range(lo, hi2):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def perturb_and_reopt(h, n, deadline, seed, n_perturb=None):

            best_h = h3

    # Phase 4: Very fine coordinate descent
    if time.time() < deadline - 150:
        t4 = min(time.time() + budget_s * 0.20, deadline - 150)
        best_h, best_c = coord_descent(best_h, n, t4, iters=200, grid_size=300, seed=seed+2)

    # Phase 5: Perturb and re-optimize (escape local optima)
    if time.time() < deadline - 100:
        for s in range(3):
            if time.time() > deadline - 60:
                break
            t5 = min(time.time() + 30, deadline - 60)
            hp, cp = perturb_and_reopt(best_h, n, t5, seed + 100 + s, n_perturb=max(3, n // 15))
            if cp < best_c:
                best_c = cp
                best_h = hp

    # Phase 6: Try other n values
    if time.time() < deadline - 80:
        for n_try in [200, 250, 400]:
            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(35, deadline - time.time() - 20)
            if alloc < 15:
                break
            ht, ct = coord_descent(h_interp, n_try, time.time() + alloc, iters=100, grid_size=100, seed=seed+n_try)
            if ct < best_c:
                best_c = ct
                best_h = ht
                n = n_try

    # Phase 7: Final coordinate descent + pair search
    if time.time() < deadline - 15:
        best_h, best_c = coord_descent(best_h, n, deadline - 10, iters=150, grid_size=300, seed=seed+7)

    if time.time() < deadline - 3:
        best_h, best_c = pair_search(best_h, n, deadline - 1,
                                     [0.001, 0.0005, 0.0002, 0.0001], radius=40)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
