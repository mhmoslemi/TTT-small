def find_optimal_t(c_current, A, t_old):
    # Minimize f(t) = max_k (c_current[k] + (t - t_old) * A[k]) over t in [0,1]
    # f is convex (max of linear functions)
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    for _ in range(60):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        f1 = np.max(c_current + (t1 - t_old) * A)
        f2 = np.max(c_current + (t2 - t_old) * A)
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2
