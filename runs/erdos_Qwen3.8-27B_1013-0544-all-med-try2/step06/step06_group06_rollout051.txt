    best_t = 0.0 if val_0 <= val_1 else 1.0

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coordinate_descent(h, n, max_sweeps=100, deadline=None):

    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)

                idx2 = j + (n-1) - m

                if 0 <= idx2 < n:

                a[m] = coeff

        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h.copy()
        elif not improved:
            break
    
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth_max(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

    best_h = h.copy()

    sum_sq = np.dot(h, h)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline is not None and time.time() > deadline:
                break
            # Compute a[m] for all m
            ns = 2 * n - 1
            a = np.zeros(ns)
            for m in range(ns):
                if m == n - 1:
                    continue
                idx1 = j + m - (n - 1)
                idx2 = j + (n - 1) - m
                coeff = 0.0
                if 0 <= idx1 < n:
                    coeff += 1.0 - h[idx1]
                if 0 <= idx2 < n:
                    coeff -= h[idx2]
                a[m] = coeff
            a[n-1] = 0  # unused
            
            # C_0' for the self-overlap term
            C0 = n / 2.0 - (sum_sq - h[j] * h[j])
            
            # b[m] = c[m] - a[m]*h[j]
            c_all = np.correlate(h, 1 - h, mode='full')
            b = c_all - a * h[j]
            b[n-1] = 0
            
            # 1D minimization: min_{t in [0,1]} max(C0 - t^2, max_m(a[m]*t + b[m]))
            # Coarse grid
            ts = np.linspace(0, 1, 50)
            quad = C0 - ts * ts
            lin = np.max(a[None, :] * ts[:, None] + b[None, :], axis=1)
            vals = np.maximum(quad, lin)
            
            best_idx = np.argmin(vals)
            # Golden section in neighborhood
            lo = max(0.0, ts[max(0, best_idx - 2)])
            hi = min(1.0, ts[min(49, best_idx + 2)])

                t_star = ts[best_idx]
            else:
                golden = (np.sqrt(5) - 1) / 2
                gr = golden  # golden ratio conjugate
                a_g = hi - gr * (hi - lo)
                b_g = lo + gr * (hi - lo)
                f_a = max(C0 - a_g**2, np.max(a * a_g + b))
                f_b = max(C0 - b_g**2, np.max(a * b_g + b))
                for _ in range(40):
                    if f_a < f_b:
                        hi = b_g
                        b_g = a_g
                        f_b = f_a
                        a_g = hi - gr * (hi - lo)
                        f_a = max(C0 - a_g**2, np.max(a * a_g + b))
                    else:
                        lo = a_g
                        a_g = b_g
                        f_a = f_b
                        b_g = lo + gr * (hi - lo)
                        f_b = max(C0 - b_g**2, np.max(a * b_g + b))
                t_star = (lo + hi) / 2.0
            
            if t_star != h[j]:
                sum_sq = sum_sq - h[j]**2 + t_star**2
                h[j] = t_star
                improved = True
        
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h.copy()
        elif not improved:
            break
    
    return best_h, best_c
