    return np.clip(h, 0.0, 1.0)

def exact_pair_optimize(h, c, n, radius, max_sweeps=50):
    """Optimize h using exact pair moves."""
    dx = 2.0 / n
    best_c = float(np.max(c))
    n_c = len(c)  # 2n-1
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            for di in range(-radius, radius+1):
                j = i + di
                if j < 0 or j >= n or j == i:
                    continue
                # Compute B[s] = change in c per unit delta transferred from j to i
                # c_new[s] = c[s] + delta * B[s]
                # B[s] = (1-h[i-s]) - h[i+s] - (1-h[j-s]) + h[j+s] (with validity)
                # where s ranges from -(n-1) to (n-1)
                
                # Compute B efficiently
                # For shift s (s = k - (n-1)), the change is:
                # B[s] = [0<=i-s<n]*(1-h[i-s]) - [0<=i+s<n]*h[i+s] - [0<=j-s<n]*(1-h[j-s]) + [0<=j+s<n]*h[j+s]
                
                # Let's compute this using numpy
                s = np.arange(-(n-1), n)  # all shifts
                # [0<=i-s<n] means s is in [i-n+1, i]
                # [0<=i+s<n] means s is in [-i, n-1-i]
                
                B = np.zeros(n_c)
                # Term 1: [0<=i-s<n]*(1-h[i-s])
                s1 = s[(s >= i-n+1) & (s <= i)]
                idx1 = (i - s1)  # = i-s, which is in [0, n-1]
                B[s1 + (n-1)] = 1.0 - h[idx1]
                
                # Term 2: -[0<=i+s<n]*h[i+s]
                s2 = s[(s >= -i) & (s <= n-1-i)]
                idx2 = (i + s2)  # = i+s, which is in [0, n-1]
                B[s2 + (n-1)] -= h[idx2]
                
                # Term 3: -[0<=j-s<n]*(1-h[j-s])
                s3 = s[(s >= j-n+1) & (s <= j)]
                idx3 = (j - s3)
                B[s3 + (n-1)] -= (1.0 - h[idx3])
                
                # Term 4: +[0<=j+s<n]*h[j+s]
                s4 = s[(s >= -j) & (s <= n-1-j)]
                idx4 = (j + s4)
                B[s4 + (n-1)] += h[idx4]
                
                # Now find optimal delta: minimize max(c + delta*B)
                # Subject to 0 <= h[i]+delta <= 1 and 0 <= h[j]-delta <= 1
                d_max = min(1.0 - h[i], h[j])
                d_min = max(h[i] - 1.0, -h[j])  # = -min(1-h[i], h[j])... 
                # Actually: h[i]+delta >= 0 → delta >= -h[i]
                # h[i]+delta <= 1 → delta <= 1-h[i]
                # h[j]-delta >= 0 → delta <= h[j]
                # h[j]-delta <= 1 → delta >= h[j]-1
                d_min = max(-h[i], h[j]-1.0)
                d_max = min(1.0 - h[i], h[j])
                
                if d_max <= d_min:
                    continue
                
                # Find optimal delta using bisection
                lo, hi = d_min, d_max
                delta = 0.0
                for _ in range(30):
                    mid = (lo + hi) / 2
                    new_c = c + mid * B
                    S_idx = np.where(new_c >= new_c.max() - 1e-10)[0]
                    B_S = B[S_idx]
                    if len(B_S) > 0:
                        if B_S.min() <= 0 <= B_S.max():
                            delta = mid
                            break
                        elif B_S.min() > 0:
                            hi = mid  # optimum is to the left
                        else:
                            lo = mid  # optimum is to the right
                    else:
                        delta = mid
                        break
                else:
                    delta = (lo + hi) / 2
                
                if abs(delta) < 1e-12:
                    continue
                
                # Evaluate
                new_c = c + delta * B
                new_max = float(np.max(new_c))
                if new_max < best_c - 1e-12:
                    # Apply the change
                    h[i] += delta
                    h[j] -= delta
                    h = np.clip(h, 0.0, 1.0)
                    # Recompute c (to be safe)
                    c = np.correlate(h, 1.0 - h, mode='full')
                    best_c = float(np.max(c))
                    improved = True
        
        if not improved:
            break
    
    return h, best_c * dx

            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_A(h, n):
    nc = 2 * n - 1
    A = np.zeros((n, nc))
    s = np.arange(-(n-1), n)
    for p in range(n):
        mask1 = (s >= p - n + 1) & (s <= p)
        A[p, mask1] = 1.0 - h[p - s[mask1]]
        mask2 = (s >= -p) & (s <= n - 1 - p)
        A[p, mask2] -= h[p + s[mask2]]
    return A

def update_A(A, i, j, delta, n, s):
    nc = 2 * n - 1
    # h[i] += delta
    mask = (s >= -i) & (s <= n - 1 - i)
    A[i + s[mask], s[mask] + (n-1)] -= delta
    mask = (s >= i - n + 1) & (s <= i)
    A[i - s[mask], s[mask] + (n-1)] -= delta
    # h[j] -= delta
    mask = (s >= -j) & (s <= n - 1 - j)
    A[j + s[mask], s[mask] + (n-1)] += delta
    mask = (s >= j - n + 1) & (s <= j)
    A[j - s[mask], s[mask] + (n-1)] += delta

def find_optimal_delta(c, B, d_min, d_max):
    """Find delta in [d_min, d_max] that minimizes max(c + delta*B)."""
    if d_max - d_min < 1e-15:
        return 0.0
    lo, hi = d_min, d_max
    for _ in range(20):
        mid = (lo + hi) * 0.5
        val = c + mid * B
        m = val.max()
        S = np.where(val >= m - 1e-12)[0]
        B_S = B[S]
        if B_S.size == 0:
            return mid
        if B_S.min() <= 0.0 <= B_S.max():
            return mid
        elif B_S.min() > 0.0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def pair_sweep(h, c, A, n, s, radius, max_sweeps=100, deadline=None):
    """Do sweeps of pair moves to improve h."""
    nc = 2 * n - 1
    best_c = float(c.max())
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        n_applied = 0
        for i in range(n):
            lo_j = max(0, i - radius)
            hi_j = min(n, i + radius + 1)
            for j in range(lo_j, hi_j):
                if j == i:
                    continue
                B = A[i] - A[j]
                d_max = min(1.0 - h[i], h[j])
                d_min = max(-h[i], h[j] - 1.0)
                if d_max - d_min < 1e-14:
                    continue
                delta = find_optimal_delta(c, B, d_min, d_max)
                if abs(delta) < 1e-13:
                    continue
                new_c = c + delta * B
                new_max = float(new_c.max())
                if new_max < best_c - 1e-13:
                    # Apply
                    h[i] += delta
                    h[j] -= delta
                    h[i] = max(0.0, min(1.0, h[i]))
                    h[j] = max(0.0, min(1.0, h[j]))
                    update_A(A, i, j, delta, n, s)
                    c = new_c
                    best_c = new_max
                    improved = True
                    n_applied += 1
        if not improved:
            break
    
    # Final recompute to be safe
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(c.max())
    return h, best_c, c

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(x.sum() - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
