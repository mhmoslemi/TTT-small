def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def find_optimal_1d(a, b):
    """Find v in [0,1] minimizing max_k (a[k] + b[k]*v) via bisection."""
    g0 = (a).max()
    g1 = (a + b).max()
    act0 = a >= g0 - 1e-14
    if b[act0].min() > 1e-10:
        return 0.0
    act1 = a + b

        mx_b = b[act].max()
        if mn_b > 1e-9:
            hi = mid
        elif mx_b < -1e-9:
            lo = mid
        else:

    return (lo + hi) * 0.5

def coord_descent(h, n, max_sweeps=30, deadline=None):
    h = h.copy()
    n_lags = 2 * n - 1
    f_vals = compute_all_lags(h, n)
    best_c = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline - 5:
            break
        any_change = False
        for i in range(n):
            s = np.zeros(n_lags)
            k_lo = -i
            k_hi = n - i - 1
            for ks in range(max(k_lo, -(n-1)), min(k_hi, n-1) + 1):
                s[ks + (n-1)] += (1.0 - h[i + ks])
            k_lo2 = i - n + 1
            k_hi2 = i

                s[ks + (n-1)] += (-h[i - ks])
            v = find_optimal_1d(f_vals, s)
            if abs(v - h[i]) > 1e-12:
                f_vals += (v - h[i]) * s
                h[i] = v
                any_change = True
        if not any_change:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

        c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        return m + np.log(np.sum(np.exp(sh))) / alpha
    def sumfun(x):
        return x.sum() - n / 2.0
    def sumjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sumfun, 'jac': sumjac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
