import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def max_count_binary(h_bin):
    corr = np.correlate(h_bin, 1 - h_bin, mode="full")
    return int(np.max(corr))

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_fn(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fn(h, target_sum):
    return h.sum() - target_sum

def con_jac(h, target_sum):
    return np.ones_like(h)

def binarize(h, n):
    h_bin = np.zeros(n, dtype=np.int8)
    idx = np.argsort(h)[::-1]
    h_bin[idx[:n // 2]] = 1
    return h_bin

def binary_sa(h_start, n, t0, time_budget, seed):
    rng = np.random.RandomState(seed)
    h_bin = binarize(h_start, n)
    cur_count = max_count_binary(h_bin)
    best_bin = h_bin.copy()
    best_count = cur_count
    T = 2.0
    total_iters = 8000000
    for it in range(total_iters):
        if time.time() - t0 > time_budget:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if h_bin[i] == h_bin[j]:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        new_count = max_count_binary(h_bin)
        delta = new_count - cur_count
        if delta <= 0 or (T > 0.05 and rng.random() < np.exp(-delta / T)):
            cur_count = new_count
            if new_count < best_count:
                best_count = new_count
                best_bin = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        T = max(0.05, 2.0 * (1.0 - it / float(total_iters)))
    return best_bin, best_count

def local_search_binary(h_bin, n, t0, time_budget):
    best_bin = h_bin.copy()
    best_count = max_count_binary(best_bin)
    improved = True
    while improved and time.time() - t0 < time_budget:
        improved = False
        ones = np.where(best_bin == 1)[0]
        zeros = np.where(best_bin == 0)[0]
        for i in ones:
            if time.time() - t0 > time_budget:
                break
            for j in zeros:
                best_bin[i] = 0
                best_bin[j] = 1
                nc = max_count_binary(best_bin)
                if nc < best_count:
                    best_count = nc
                    improved = True
                else:
                    best_bin[i] = 1
                    best_bin[j] = 0
    return best_bin, best_count

def multi_start_sa(h_ref, n, t0, time_budget, base_seed, n_starts):
    best_bin = None
    best_count = 999
    for s in range(n_starts):
        if time.time() - t0 > time_budget - 30:
            break
        t_per = (time_budget - (time.time() - t0)) / (n_starts - s)
        if s == 0:
            h_start = h_ref.copy()
        elif s == 1:
            h_start = np.random.RandomState(base_seed + 100).rand(n)
        else:
            h_start = h_ref + np.random.RandomState(base_seed + s).randn(n) * 0.1
        h_start = np.clip(h_start, 0, 1)
        h_bin, cnt = binary_sa(h_start, n, t0, time.time() - t0 + t_per, base_seed + s * 7)
        if cnt < best_count:
            best_count = cnt
            best_bin = h_bin
    return best_bin, best_count

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 70
        h_init = np.full(n_init, 0.5)

    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    best_n = n_init

    # Phase 1: Binary SA at initial resolution (if even)
    if n_init % 2 == 0:
        n = n_init
        dx = 2.0 / n
        tb = budget_s - 200
        best_bin, best_count = multi_start_sa(h_init, n, t0, tb, seed, 4)
        if best_bin is not None:
            c5_bin = float(best_count) * dx
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = best_bin.astype(float).copy()
                best_n = n

        # Local search polish
        if time.time() - t0 < budget_s - 120:
            best_bin2, best_count2 = local_search_binary(best_bin, n, t0, budget_s - 120)
            c5_ls = float(best_count2) * dx
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = best_bin2.astype(float).copy()
                best_n = n

    # Phase 2: Try nearby even resolutions
    for n_try in [n_init - 2, n_init + 2, n_init - 4, n_init + 4, n_init - 6, n_init + 6, n_init - 8, n_init + 8]:
        if n_try < 30 or n_try % 2 != 0 or time.time() - t0 > budget_s - 100:
            continue
        dx_t = 2.0 / n_try
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_r = np.interp(x_new, x_old, best_h)
        t_per = min(80, (budget_s - 100) - (time.time() - t0))
        if t_per < 20:
            break
        best_bin_t, best_count_t = binary_sa(h_r, n_try, t0, time.time() - t0 + t_per, seed + n_try)
        c5_t = float(best_count_t) * dx_t
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = best_bin_t.astype(float).copy()
            best_n = n_try

    # Phase 3: Continuous SLSQP refinement
    if time.time() - t0 < budget_s - 40:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        for beta in [300, 1000, 5000, 20000]:
            if time.time() - t0 > budget_s - 20:
                break
            cons = [{'type': 'eq', 'fun': con_fn, 'jac': con_jac, 'args': (target_sum,)}]
            try:
                res = minimize(obj_fn, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 4000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 4: Subgradient refinement
    if time.time() - t0 < budget_s - 10:
        n = best_n
        h = best_h.copy()
        target_sum = n / 2.0
        for it in range(200000):
            if time.time() - t0 > budget_s - 5:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_worst = int(np.argmax(corr))
            lag = k_worst - (n - 1)
            grad = np.zeros(n)
            if lag >= 0:
                idx = np.arange(n - lag)
                grad[idx] += (1.0 - h[idx + lag])
                grad[idx + lag] -= h[idx]
            else:
                la = -lag
                idx = np.arange(n - la)
                grad[idx + la] += (1.0 - h[idx])
                grad[idx] -= h[idx + la]
            lr = 0.003 / (1.0 + it * 0.0003)
            h = h - lr * grad
            h = project_feasible(h, target_sum)
            if it % 20000 == 19999:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5

    # Final check
    c5_final = compute_c5(best_h)
    if c5_final < best_c5:
        best_c5 = c5_final

    return best_h, best_c5, len(best_h)
