            i, j = candidates[idx]
            if h[i] == h[j]:
                continue
            if h[i] == 0 and h[j] == 1:
                h[i], h[j] = 1.0, 0.0
                new_corr = np.correlate(h, 1 - h, mode='full')
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-14:
                    corr = new_corr
                    cur_c = nc
                    improved = True
                    break
                else:
                    h[i], h[j] = 0.0, 1.0

        if not improved:
            for _ in range(8):
                i = rng.choice(zeros)
                j = rng.choice(ones)
                h[i], h[j] = 1.0, 0.0
                new_corr = np.correlate(h, 1 - h, mode='full')
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-14:
                    corr = new_corr
                    cur_c = nc
                    improved = True
                    break
                else:
                    h[i], h[j] = 0.0, 1.0

        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if not improved and cur_c >= best_c:
            if rng.rand() < 0.1:
                for _ in range(rng.randint(3, 10)):
                    i = rng.choice(np.where(h == 0)[0])
                    j = rng.choice(np.where(h == 1)[0])
                    h[i], h[j] = 1.0, 0.0
                corr = np.correlate(h, 1 - h, mode='full')
                cur_c = corr.max() * dx
    return best_h, best_c

def make_block_init(n, seed):
    rng = np.random.RandomState(seed)
    n_half = n // 2
    h = np.zeros(n)
    n_ones = n_half
    while n_ones > 0:
        block = min(rng.randint(1, max(2, n_ones // 3 + 1)), n_ones)
        start = rng.randint(0, n - block + 1)
        placed = 0
        for k in range(block):
            if h[start + k] == 0 and placed < n_ones:
                h[start + k] = 1.0
                placed += 1
        n_ones -= placed
    if h.sum() < n_half:
        remaining = n_half - int(h.sum())
        z = np.where(h == 0)[0]
        rng.shuffle(z)
        for k in range(min(remaining, len(z))):
            h[z[k]] = 1.0
    elif h.sum() > n_half:
        excess = int(h.sum()) - n_half
        o = np.where(h == 1)[0]
        rng.shuffle(o)
        for k in range(min(excess, len(o))):
            h[o[k]] = 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: SLSQP from initial (brief)
    p1 = t0 + 0.08 * (budget_s - 5)
    h = project_feasible(h0, n)
    for alpha in [1000, 10000, 100000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 500, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Binary local search (main workhorse)
    p2 = t0 + 0.55 * (budget_s - 5)
    hb = np.where(best_h > 0.5, 1.0, 0.0)
    if abs(hb.sum() - n / 2) > 1:
        idx = np.argsort(best_h)[::-1]
        hb = np.zeros(n)
        hb[idx[:n // 2]] = 1.0
    hb, cb = binary_local_search(hb, n, p2, max_sweeps=10000, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: SLSQP polish from binary
    p3 = t0 + 0.68 * (budget_s - 5)
    if time.time() < p3:
        hp = project_feasible(best_h + 0.03 * rng.randn(n), n)
        for alpha in [10000, 100000, 1000000, 10000000]:
            if time.time() > p3:
                break
            hp = slsqp_smooth(hp, n, alpha, 1000, p3)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()

    # Phase 4: More binary search on best
    p4 = t0 + 0.82 * (budget_s - 5)
    hb2 = np.where(best_h > 0.5, 1.0, 0.0)
    if abs(hb2.sum() - n / 2) > 1:
        idx = np.argsort(best_h)[::-1]
        hb2 = np.zeros(n)
        hb2[idx[:n // 2]] = 1.0
    hb2, cb2 = binary_local_search(hb2, n, p4, max_sweeps=15000, seed=seed + 100)
    if cb2 < best_c:
        best_c = cb2
        best_h = hb2.copy()

    # Phase 5: Try different resolutions
    p5 = dl - 30
    for nh in [200, 300, 350, 400]:
        if time.time() > p5 - 40:
            break
        xo = np.linspace(0, 2, n)
        hn = np.interp(np.linspace(0, 2, nh), xo, best_h)
        hn = project_feasible(hn, nh)
        hn_bin = np.where(hn > 0.5, 1.0, 0.0)
        if abs(hn_bin.sum() - nh // 2) > 2:
            idx = np.argsort(hn)[::-1]
            hn_bin = np.zeros(nh)
            hn_bin[idx[:nh // 2]] = 1.0
        limit = min(p5 - 20, time.time() + 60)
        hnb, cnb = binary_local_search(hn_bin, nh, limit, max_sweeps=3000, seed=seed + nh)
        if cnb < best_c:
            best_c = cnb
            best_h = hnb.copy()
            n = nh

    # Phase 6: Final polish
    if time.time() < dl - 5:
        h = slsqp_smooth(best_h + 0.01 * rng.randn(n), n, 10000000, 2000, dl - 3)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        hb3 = np.where(best_h > 0.5, 1.0, 0.0)
        if abs(hb3.sum() - n / 2) > 1:
            idx = np.argsort(best_h)[::-1]
            hb3 = np.zeros(n)
            hb3[idx[:n // 2]] = 1.0
        hb3, cb3 = binary_local_search(hb3, n, dl, max_sweeps=5000, seed=seed + 999)
        if cb3 < best_c:
            best_c = cb3
            best_h = hb3.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
