def binary_local_search(h, n, deadline):
    """Local search on 0-1 vectors with exactly n/2 ones."""
    # Round to 0-1
    h = (h >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > n//2:
        # Move some ones to zeros
        excess = len(ones) - n//2
        for idx in ones[:excess]:
            h[idx] = 0
            h[zeros[0]] = 1
            zeros = zeros[1:]
    elif len(ones) < n//2:
        deficit = n//2 - len(ones)
        for idx in zeros[:deficit]:
            h[idx] = 1
            h[ones[0]] = 0
            ones = ones[1:]
    
    c5 = compute_c5(h, n)
    best_c5 = c5
    best_h = h.copy()
    
    while time.time() < deadline:
        improved = False
        # Try all swaps
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0, 1
                new_c5 = compute_c5(h, n)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i], h[j] = 1, 0
            if improved:
                break
        if not improved:
            break
    
    return best_h, best_c5

- So 0-1 values minimize the "per-term" contribution, but the count of non-zero terms matters

Actually, for a 0-1 vector, c[k] is just a count (number of positions where h[i]=1 and h[i+k]=0). The maximum is at most n/2 (if all 1s are followed by 0s at lag k).

For a fractional vector, c[k] = Σ h[i](1-h[i+k]) ≤ Σ min(h[i], 1-h[i+k]) ≤ ... it's more complex.

I think the optimal solution might indeed be close to 0-1, with some fractional values at the "boundary" between the 1-region and 0-region.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    sorted_idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[sorted_idx[:n // 2]] = 1.0
    return h_b

def binary_local_search(h, n, deadline, rng, max_passes=50000):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        perm = rng.permutation(len(zeros))
        for i in ones:
            for jj in perm:
                j = zeros[jj]
                h[i] = 0
                h[j] = 1
                c5 = compute_c5(h, n)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
                    break
                else:
                    h[i] = 1
                    h[j] = 0
            if improved:
                break
        if not improved:
            break
    return h, best_c5

def binary_2swap(h, n, deadline, rng, max_iters=200000):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    for _ in range(max_iters):
        if time.time() > deadline:
            break

        i2 = ones[rng.randint(len(ones))]
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if i1 == i2 or j1 == j2:
            continue
        h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
        c5 = compute_c5(h, n)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
        else:
            h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
    return h, best_c5

def binary_sa(h, n, deadline, rng, T0=0.002, T_end=0.00003, max_iters=300000):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    cur = best_c5
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / max_iters)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0; h[j] = 1
        nc = compute_c5(h, n)
        d = nc - cur
        if d < 0 or rng.rand() < np.exp(-d / max(T, 1e-12)):
            cur = nc
            if nc < best_c5:
                best_c5 = nc
                best_h = h.copy()
        else:
            h[i] = 1; h[j] = 0
    return best_h, best_c5

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
