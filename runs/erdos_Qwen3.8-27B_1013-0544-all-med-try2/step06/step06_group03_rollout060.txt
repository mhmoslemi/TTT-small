    best_c = compute_c5(h_cur, n)

        b = 1.0 - h_cur
        corr = np.correlate(h_cur, b, mode='full') * dx

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, n_passes, deadline):
    """Fast coordinate descent for minimizing C5."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    n_corr = 2 * n - 1
    shifts = np.arange(n_corr) - (n - 1)
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            old_val = h[i]
            
            # Current correlation
            corr = np.correlate(h, 1.0 - h, mode='full')
            
            # Influence vector for coordinate i
            idx_a = i - shifts
            idx_b = i + shifts
            valid_a = (idx_a >= 0) & (idx_a < n)
            valid_b = (idx_b >= 0) & (idx_b < n)
            
            s = np.zeros(n_corr)
            if np.any(valid_a):
                s[valid_a] = 1.0 - h[idx_a[valid_a]]
            if np.any(valid_b):
                s[valid_b] -= h[idx_b[valid_b]]
            
            # Find best delta in [-old_val, 1-old_val]
            lo = -old_val
            hi = 1.0 - old_val
            if hi - lo < 1e-12:
                continue
            best_delta = 0.0
            best_obj = np.max(corr)
            
            # Check boundaries
            f_lo = np.max(corr + lo * s)
            f_hi = np.max(corr + hi * s)
            if f_lo < best_obj:
                best_obj = f_lo
                best_delta = lo
            if f_hi < best_obj:
                best_obj = f_hi
