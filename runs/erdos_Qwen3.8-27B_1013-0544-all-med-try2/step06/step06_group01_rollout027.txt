def flip_effect(h, n):
    """Precompute the effect of flipping each position.
    Returns:
        zero_effects[i] = change in correlation vector if h[i] goes 0->1
        one_effects[j] = change in correlation vector if h[j] goes 1->0
    """
    nz = 2 * n - 1
    zero_effects = np.zeros((n, nz))  # effect of 0->1 at position i
    one_effects = np.zeros((n, nz))   # effect of 1->0 at position j
    
    for i in range(n):
        # h[i]: 0 -> 1
        # As h[n+k] factor (n+k=i, n=i-k): valid when 0<=i-k<=n-1, i.e., k in [i-(n-1), i]
        #   In output: m = k+(n-1) in [i, i+n-1], value added = (1-h[n]) = (1-h[i-k]) = (1-h[m-(n-1)-... 
        #   n = i-k = i-(m-(n-1)) = i-m+n-1
        #   So value = 1-h[i-m+n-1] for m in [max(0,i), min(nz-1, i+n-1)]
        #   But we also need 0 <= n <= n-1, i.e., 0 <= i-m+n-1 <= n-1
        #   => i-m+n-1 >= 0 => m <= i+n-1
        #   => i-m+n-1 <= n-1 => m >= i
        lo = max(0, i)
        hi = min(nz - 1, i + n - 1)
        if lo <= hi:
            n_vals = i - np.arange(lo, hi + 1) + (n - 1)  # = i - m + n - 1
            zero_effects[i, lo:hi+1] += (1.0 - h[n_vals])
        
        # As (1-h[n]) factor (n=i): valid when 0<=i+k<=n-1, i.e., k in [-i, n-1-i]
        #   In output: m = k+(n-1) in [n-1-i, n-1+i]
        #   Value subtracted = h[i+k] = h[i+m-n+1]
        lo2 = max(0, n - 1 - i)
        hi2 = min(nz - 1, n - 1 + i)
        if lo2 <= hi2:
            val = np.arange(lo2, hi2 + 1) + i - (n - 1)  # = i + k = i + m - (n-1)
            zero_effects[i, lo2:hi2+1] -= h[val]
    
    for j in range(n):

        #   In output: m = k+(n-1) in [j, j+n-1], value removed = (1-h[n]) = (1-h[j-k])
        #   n = j-k = j-m+n-1
        lo = max(0, j)
        hi = min(nz - 1, j + n - 1)
        if lo <= hi:
            n_vals = j - np.arange(lo, hi + 1) + (n - 1)
            one_effects[j, lo:hi+1] -= (1.0 - h[n_vals])
        
        # As (1-h[n]) factor (n=j): valid when 0<=j+k<=n-1
        #   In output: m = k+(n-1) in [n-1-j, n-1+j]

def fast_local_search(h, n, deadline, max_sweeps=5000):
    """Fast greedy local search for binary h."""
    dx = 2.0 / n
    corr = corr_full(h)
    cur_c = corr.max() * dx
    nz = 2 * n - 1
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        mstar = np.argmax(corr)
        
        # For each zero i, compute effect of h[i]:0->1 on corr[mstar]
        # corr[mstar] = sum_n h[n+k]*(1-h[n]) where k=mstar-(n-1)
        # h[i] as h[n+k]: n=i-k=i-(mstar-(n-1))=i-mstar+n-1
        #   Valid if 0<=i-mstar+n-1<=n-1
        #   Contribution: (1-h[i-mstar+n-1])
        # h[i] in (1-h[n]): n=i, valid if 0<=i+k<=n-1 where k=mstar-(n-1)
        #   i+k = i+mstar-n+1
        #   Valid if 0<=i+mstar-n+1<=n-1
        #   Contribution: -h[i+mstar-n+1]
        
        best_gain = 0
        best_pair = None
        
        for i in zeros:
            gain_i = 0
            # As h[n+k] at mstar: n = i - mstar + n - 1
            n1 = i - mstar + n - 1
            if 0 <= n1 < n:
                gain_i += (1.0 - h[n1])
            # In (1-h[n]) at mstar: i+k = i + mstar - n + 1
            n2 = i + mstar - n + 1
            if 0 <= n2 < n:
                gain_i -= h[n2]
            
            for j in ones:
                # Effect of h[j]:1->0 on corr[mstar]
                # As h[n+k]: n = j - mstar + n - 1
                gain_j = 0
                n3 = j - mstar + n - 1
                if 0 <= n3 < n:
                    gain_j -= (1.0 - h[n3])
                # In (1-h[n]): j+k = j + mstar - n + 1
                n4 = j + mstar - n + 1
                if 0 <= n4 < n:
                    gain_j += h[n4]
                
                total_gain = gain_i + gain_j
                if total_gain > best_gain + 1e-15:
                    best_gain = total_gain
                    best_pair = (i, j)
        
        if best_pair is None:
            break
        
        i, j = best_pair
        h[i], h[j] = 1.0, 0.0
        corr = corr_full(h)  # Recompute (O(n²) but n=250 so ~62500 ops)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return float(np.max(c) * dx)

def corr_full(h):
    return np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')

def binarize(h):
    """Convert to binary by thresholding at 0.5, adjusting to get exactly n/2 ones."""
    n = len(h)
    target = n // 2
    h_sorted = np.sort(h)[::-1]
    h_bin = (h_sorted[:target] > 0)  # top n/2 become 1
    # Actually, just threshold
    h_bin = (h > 0.5).astype(float)
    # Adjust if needed
    while h_bin.sum() < target:
        idx = np.argmin(h * (1 - h_bin.astype(int)))
        h_bin[idx] = 1.0
    while h_bin.sum() > target:
        idx = np.argmax(h * h_bin.astype(int))
        h_bin[idx] = 0.0
    return h_bin

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:
        return h
