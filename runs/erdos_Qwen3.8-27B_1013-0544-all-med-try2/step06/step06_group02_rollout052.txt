import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlation(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(h, n, alpha):
    c = compute_correlation(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    # Use log-sum-exp for numerical stability
    mx = shifted.max()
    val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return val

def soft_max_grad(h, n, alpha):
    c = compute_correlation(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag in range(2 * n - 1):
        if w[lag] < 1e-14:
            continue
        k = lag - (n - 1)
        wm = w[lag] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def nesterov_grad_descent(h, n, deadline, alpha_init=100.0, lr_init=0.005, momentum=0.9):
    h = project_feasible(h, n)
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    y = h.copy()
    lr = lr_init
    alpha = alpha_init

            break
        
        g = soft_max_grad(y, n, alpha)
        v = momentum * v - lr * g
        y_new = y + v
        h_new = project_feasible(y_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:

            best_h = h_new.copy()
            h = h_new
            y = h_new

                lr = lr_init * 0.1
                alpha = alpha_init * 5

                    break
        y = h + momentum * (h - y)  # Nesterov extrapolation
    return best_h, best_c

            # Compute g_k as a function of h[j]
            # g_k = dx * sum_{i < n-k} h[i](1 - h[i+k])
            # Terms involving h[j]:
            # If j < n-k: h[j](1-h[j+k])*dx
            # If j >= k and j-k < n-k: h[j-k](1-h[j])*dx
            
            a = 0.0  # coefficient of h[j]^2
            b = 0.0  # coefficient of h[j]
            c = 0.0  # constant
            
            if j < n - k:
                # h[j] * (1 - h[j+k]) * dx = (1-h[j+k])*h[j]*dx - h[j]^2*dx
                b += (1 - h[j+k]) * dx
                a += -dx
            
            if j >= k and (j - k) < n - k:
                # h[j-k] * (1 - h[j]) * dx = h[j-k]*dx - h[j-k]*h[j]*dx
                b += -h[j-k] * dx
                # No h[j]^2 term here
            
            # g_k(h_j) = a*h_j^2 + b*h_j + c
            # Minimum at h_j* = -b/(2a) if a < 0, else at boundary
            if a < -1e-15:
                h_j_opt = -b / (2 * a)
                h[j] = np.clip(h_j_opt, 0.0, 1.0)
            else:
                # Linear: minimize at boundary
                if b > 0:
                    h[j] = 0.0
                elif b < 0:
                    h[j] = 1.0
                # else: any value works
        
        # Re-normalize to maintain sum(h) = n/2
        h = project_feasible(h, n)
    
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha):
    c = compute_corr(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 50:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def softmax_grad(x, n, alpha):
    c = compute_corr(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag in range(2 * n - 1):
        if w[lag] < 1e-14:
            continue
        k = lag - (n - 1)
        wm = w[lag] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def effect_matrix(h, n):
    """Compute effect_i[m] = d corr[m] / d h[i] for all i, m."""
    EM = 2 * n - 1
    eff = np.zeros((n, EM))
    center = n - 1
    for i in range(n):
        lo = center - i
        hi = center + i
        a = lo
        b = hi + 1
        for m in range(a, b):
            j1 = i + m - center
            j2 = i - m + center
            val = 0.0
            if 0 <= j1 < n:
                val += (1.0 - h[j1])
            if 0 <= j2 < n:
                val -= h[j2]
            eff[i, m] = val
    return eff

def lbfgs_run(h, n, deadline, max_iters=500):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [20, 100, 500, 2000, 10000, 50000, 200000, 1000000]
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        def obj(x):
            return softmax_obj(x, n, alpha)
        def jac(x):
            return softmax_grad(x, n, alpha)
        try:
            res = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def nesterov_run(h, n, deadline, alpha_init=50.0, lr_init=0.008, momentum=0.85):
    h = project_feasible(h, n)
    x = h.copy()
    x_prev = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_smooth = softmax_obj(h, n, alpha_init)
    lr = lr_init
    alpha = alpha_init

    for it in range(100000):
        if time.time() > deadline:
            break
        y = x + momentum * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        g = softmax_grad(y, n, alpha)
        x_new = project_feasible(y - lr * g, n)
        s_new = softmax_obj(x_new, n, alpha)
        if s_new <= best_smooth + 1e-12:
            x_prev = x
            x = x_new
            best_smooth = s_new
            c_new = compute_c5(x_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = x_new.copy()
        else:
            lr *= 0.5
            if lr < 1e-11:
                alpha *= 2.0
                lr = lr_init * 0.5
                x_prev = x.copy()
                if alpha > 1e7:
                    break
    return best_h, best_c

def pair_polish(h, n, deadline, max_rounds=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        for rnd in range(max_rounds):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    # Try moving from i to j
                    if h[i] - step >= 0.0 and h[j] + step <= 1.0:
                        h[i] -= step
                        h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += step
                            h[j] -= step
                    # Try moving from j to i
                    if h[j] - step >= 0.0 and h[i] + step <= 1.0:
                        h[j] -= step
                        h[i] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[j] += step
                            h[i] -= step
            if not improved:
                break
    return h, best_c

def gradient_pair_polish(h, n, deadline, step_size=0.003):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    for alpha in [50, 200, 1000, 5000, 20000]:
        while time.time() < deadline:
            improved = False
            g = softmax_grad(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    move = -step_size * (g[i] - g[j])
                    if abs(move) < 1e-10:
                        continue
                    hi_new = h[i] + move
                    hj_new = h[j] - move
                    if 0.0 <= hi_new <= 1.0 and 0.0 <= hj_new <= 1.0:
                        h[i] = hi_new
                        h[j] = hj_new
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            if not improved:
                break
        step_size *= 0.7
    return h, best_c
