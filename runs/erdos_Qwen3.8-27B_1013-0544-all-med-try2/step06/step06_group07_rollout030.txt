    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:

        grad[idx + lag] -= h[idx]

```python
def smooth_max_subgrad(h, beta):
    """Compute gradient of smooth_max(corr, beta) * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_c = np.exp(np.clip(beta * (corr - m), -700, 700))
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if 0 <= lag < n:
            idx = np.arange(n - lag)
            grad[idx] += wk * (1.0 - h[idx + lag])
            grad[idx + lag] -= wk * h[idx]
        elif -n < lag < 0:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += wk * (1.0 - h[idx])
            grad[idx] -= wk * h[idx + la]
    return grad * dx
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(20):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_subgrad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_c = np.exp(np.clip(beta * (corr - m), -700, 700))
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if 0 <= lag < n:
            idx = np.arange(n - lag)
            grad[idx] += wk * (1.0 - h[idx + lag])
            grad[idx + lag] -= wk * h[idx]
        elif -n < lag < 0:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += wk * (1.0 - h[idx])
            grad[idx] -= wk * h[idx + la]
    return grad * dx

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx

def con_fun(h, target_sum):
    return h.sum() - target_sum

def con_jac(h, target_sum):
    return np.ones(len(h))

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def adagrad_smooth(h_init, n, target_sum, max_time, beta_sg):
    h = h_init.copy()
    dx = 2.0 / n
    grad_sq = np.zeros(n)
    eps = 1e-8
    lr_base = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t_start = time.time()
    it = 0
    while time.time() - t_start < max_time:
        grad = smooth_subgrad(h, beta_sg)
        grad_sq += grad * grad
        lr = lr_base / np.sqrt(grad_sq + eps)
        h = h - lr * grad
        h = project_feasible(h, target_sum)
        it += 1
        if it % 3000 == 2999:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def slsqp_polish(h_init, n, target_sum, deadline):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    for beta in [500, 2000, 8000, 30000, 100000]:
        if time.time() > deadline - 5:
            break
        for trial in range(3):
            if time.time() > deadline - 3:
                break
            x0 = best_h.copy()
            if trial == 1:
                x0 += np.random.randn(n) * 0.005
                x0 = project_feasible(x0, target_sum)
            elif trial == 2:
                x0 += np.random.randn(n) * 0.001
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta),
                               method='SLSQP', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 30

    if 'initial_h_values' in globals() and initial_h_values is not None:

    n0 = len(h_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase A: Main optimization at n0
    # A1: AdaGrad with smooth subgradient
    h_curr = best_h.copy()
    t_start = time.time()
    ad_time = min(120, (deadline - t_start) * 0.3)
    h_ad, c5_ad = adagrad_smooth(h_curr, n0, n0 / 2.0, ad_time, beta_sg=100)
    if c5_ad < best_c5:
        best_h, best_c5 = h_ad, c5_ad

    # A2: SLSQP polish
    sl_time = min(200, (deadline - time.time()) * 0.4)
    deadline_sl = time.time() + sl_time
    h_sl, c5_sl = slsqp_polish(best_h, n0, n0 / 2.0, deadline_sl)
    if c5_sl < best_c5:
        best_h, best_c5 = h_sl, c5_sl

    # A3: More AdaGrad from improved point
    remaining = deadline - time.time()
    if remaining > 60:
        ad_time2 = min(150, remaining * 0.4)
        h_ad2, c5_ad2 = adagrad_smooth(best_h, n0, n0 / 2.0, ad_time2, beta_sg=200)
        if c5_ad2 < best_c5:
            best_h, best_c5 = h_ad2, c5_ad2
        # Final SLSQP
        if deadline - time.time() > 30:
            h_sl2, c5_sl2 = slsqp_polish(best_h, n0, n0 / 2.0, deadline)
            if c5_sl2 < best_c5:
                best_h, best_c5 = h_sl2, c5_sl2

    # Phase B: Try other resolutions
    for n_try in [n0 + 10, n0 - 10, n0 + 20, n0 + 40, n0 - 20, n0 + 60]:
        if n_try < 30 or time.time() > deadline - 40:
            continue
        h_t = resample_h(best_h, n_try)
        target_t = n_try / 2.0
        h_t = project_feasible(h_t, target_t)
        # Quick AdaGrad
        ad_t = min(40, (deadline - time.time()) * 0.3)
        h_at, c5_at = adagrad_smooth(h_t, n_try, target_t, ad_t, beta_sg=150)
        if c5_at < best_c5:
            best_h, best_c5 = h_at, c5_at
        # Quick SLSQP
        if time.time() > deadline - 15:
            break
        h_st, c5_st = slsqp_polish(h_at, n_try, target_t, deadline)
        if c5_st < best_c5:
            best_h, best_c5 = h_st, c5_st

    return best_h, best_c5, len(best_h)
