```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_binary(h, n):
    """Project to binary: top n/2 values become 1, rest become 0."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def greedy_binary_search(h, n, deadline, rng, max_rounds=200):
    """Greedy local search: at each step, find the best single swap."""
    best_c = compute_c5(h, n)
    target = n // 2
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        best_delta_c = 0
        best_i = -1
        best_j = -1
        
        # Try all swaps: move 1 from position j to position i
        for i in zeros:
            for j in ones:
                h[i], h[j] = 1.0, 0.0
                c = compute_c5(h, n)
                h[i], h[j] = 0.0, 1.0
                if c < best_c - 1e-12:
                    best_delta_c = c - best_c
                    best_i = i
                    best_j = j
        
        if best_i == -1:
            break  # No improvement found
        
        # Apply best swap
        h[best_i], h[best_j] = 1.0, 0.0
        best_c = best_c + best_delta_c
        ones = np.concatenate([ones[ones != best_j], [best_i]])
        zeros = zeros[zeros != best_i]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_raw(h, n):
    return float(np.max(np.correlate(h, 1 - h, mode='full')))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def greedy_search(h, n, deadline, max_rounds=200):
    """Best-improvement greedy swap search for binary solutions."""
    dx = 2.0 / n
    best_raw = compute_c5_raw(h, n)
    target = n // 2
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        best_r = best_raw
        best_i = -1
        best_j = -1
        
        for i in zeros:
            for j in ones:
                h[i], h[j] = 1.0, 0.0
                r = compute_c5_raw(h, n)
                h[i], h[j] = 0.0, 1.0
                if r < best_r:
                    best_r = r
                    best_i = i
                    best_j = j
        
        if best_i < 0:
            break
        h[best_i], h[best_j] = 1.0, 0.0
        best_raw = best_r
        ones.remove(best_j)
        ones.append(best_i)
        zeros.remove(best_i)
        zeros.append(best_j)
    
    return h, best_raw * dx

def kicked_search(h, n, deadline, rng, max_rounds=100, kick_size=3):
    """Greedy search with random kicks to escape local optima."""
    dx = 2.0 / n
    best_raw = compute_c5_raw(h, n)
    best_h = h.copy()
    target = n // 2
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        # Greedy improvement
        improved = True
        while improved and time.time() < deadline:
            improved = False
            best_r = best_raw
            best_i = -1
            best_j = -1
            for i in zeros:
                for j in ones:
                    h[i], h[j] = 1.0, 0.0
                    r = compute_c5_raw(h, n)
                    h[i], h[j] = 0.0, 1.0
                    if r < best_r:

                        best_j = j
            if best_i >= 0:

                best_raw = best_r
                ones.remove(best_j)
                ones.append(best_i)
                zeros.remove(best_i)
                zeros.append(best_j)
                improved = True
        
        if best_raw < compute_c5_raw(best_h, n):
            best_raw = best_raw

            ones.append(i)
```

```python
def project_feasible(h, n):
    target = n / 2.0

            break
        if diff > 0:

            s = room.sum()
            if s < 1e-14:
                break

            if s < 1e-14:
```

```python
def swap_update(c, h, i, j, n):
    """Update correlation vector after swapping h[i]:0->1, h[j]:1->0.
    c is the correlation vector (length 2n-1), indexed as c[k] = c[lag=k-(n-1)].
    Returns new c."""
    c_new = c.copy()
    offset = n - 1
    
    # From h[i]: 0->1
    # Case 1: d = i - n_pos for n_pos = 0..n-1
    #   c_new[offset + i - n_pos] += (1 - h[n_pos])
    n_pos = np.arange(n)
    d_vals = i - n_pos  # ranges from i-(n-1) to i
    valid = (d_vals >= -(n-1)) & (d_vals <= n-1)
    c_new[offset + d_vals[valid]] += (1 - h[n_pos[valid]])
    
    # Case 2: d from -i to n-1-i, c_new[offset+d] -= h[i+d]
    d_vals2 = np.arange(-i, n - i)
    c_new[offset + d_vals2] -= h[i + d_vals2]
    
    # From h[j]: 1->0
    # Case 1: d = j - n_pos, c_new[offset+d] -= (1-h[n_pos])
    d_vals3 = j - n_pos
    valid3 = (d_vals3 >= -(n-1)) & (d_vals3 <= n-1)
    c_new[offset + d_vals3[valid3]] -= (1 - h[n_pos[valid3]])
    
    # Case 2: d from -j to n-1-j, c_new[offset+d] += h[j+d]
    d_vals4 = np.arange(-j, n - j)
    c_new[offset + d_vals4] += h[j + d_vals4]
    
    return c_new
```

```python
    p2 = t0 + 0.68 * (budget_s - 5)
    if time.time() < p2 - 30 and n % 2 == 0:
        n2 = 300
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, n2)
        h2 = to_binary(np.interp(xn, xo, best_h), n2)
        h2, c = kicked_greedy(h2, n2, p2 - 10, rng, max_kicks=15, kick_strength=3)
        if c < best_c:
            best_c, best_h, n = c, h2.copy(), n2
    # Phase 3: Multi-swap random search on best binary (12% budget)
    p3 = t0 + 0.80 * (budget_s - 5)
    if n % 2 == 0 and time.time() < p3 - 10:
        hm = to_binary(best_h, n)
        hm, c = multi_swap_search(hm, n, p3 - 5, rng, max_iters=200000)
        if c < best_c:
            best_c, best_h = c, hm.copy()
    # Phase 4: Fractional refinement (15% budget)
    p4 = t0 + 0.95 * (budget_s - 5)
    if time.time() < p4 - 5:
        hf = project_feasible(best_h, n)
        hf, c = fractional_refine(hf, n, p4)
        if c < best_c:
            best_c, best_h = c, hf.copy()
    # Final: ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
