        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

- 4 fancy index operations: O(n)

To speed this up, I need a different approach. Options:
1. **Only check a subset of pairs**: Instead of all (i,j) pairs, only check a random subset or use a heuristic to select promising pairs.
2. **Precompute and cache**: For each position i, precompute which j would be good.
3. **Use FFT-based correlation**: Compute the full correlation via FFT, which is O(n log n) instead of O(n²). But we need to update it incrementally.

Actually, the key insight is: for the **greedy** approach (find the best swap), we don't need to check all pairs. We can:
- For each zero position i, the best j is the one that minimizes max(corr + delta_{i,j}).
- This is still O(n²) in the worst case, but we can use a **beam search** approach: only keep the top K candidates for each i.

Or better: use a **randomized** approach. Each sweep, try K random (i,j) pairs instead of all pairs. This is O(K) per sweep instead of O(n²).

def project_feasible(h, n, iterations=200):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(iterations):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
