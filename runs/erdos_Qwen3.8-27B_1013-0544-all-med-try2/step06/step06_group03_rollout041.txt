```python
def smax_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (np.max(c - m) + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + 1.0/alpha * np.log(np.sum(np.exp(sh))))

def smax_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx) / np.sum(np.exp(sh - mx))
    else:
        w = np.exp(sh) / np.sum(np.exp(sh))
    # Compute gradient using the derived formula
    # term1[j] = sum_m x[m] * w[m-j+n-1]
    # term2[j] = sum_m x[m] * w[j-m+n-1]
    # Use matrix approach
    m_idx = np.tile(np.arange(n), (n, 1))
    j_idx = np.repeat(np.arange(n)[:, None], n, axis=1)
    idx1 = m_idx - j_idx + n - 1
    idx2 = j_idx - m_idx + n - 1
    term1 = w[idx1] @ x
    term2 = w[idx2] @ x
    return dx * (1.0 - term1 - term2)

def eq_fun(x, n):
    return x.sum() - n / 2.0

def eq_jac(x, n):
    return np.ones(n)

def do_slsqp(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    try:
        res = minimize(smax_obj, h, args=(n, alpha), jac=lambda x: smax_grad(x, n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved_any = False
        while time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius) if radius else 0
                hi = min(n, i + radius + 1) if radius else n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i]+=d, h[j]-=d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c; improved_any = True
                        else:
                            h[i] -= d; h[j] += d
                    # Try h[i]-=d, h[j]+=d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c; improved_any = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved_any:
                break
    return h, best_c
```

     Let me simplify the first sum: sum_{k=j}^{j+n-1} w[k] = (cumulative sum of w)
     
     For the second sum, let m = j-k+n-1. When k=j, m=n-1. When k=j+n-1, m=0.
     So sum_{k=j}^{j+n-1} w[k]*h[j-k+n-1] = sum_{m=0}^{n-1} w[j-m+n-1]*h[m]
     
     For the third sum, let m = j+k-n+1. When k=n-1-j, m=0. When k=n-j-1, m=n-1.
     So sum_{k=n-1-j}^{n-j-1} w[k]*h[j+k-n+1] = sum_{m=0}^{n-1} w[m-j+n-1]*h[m]

     grad[j] = dx * [sum_{k=j}^{j+n-1} w[k] - sum_{m=0}^{n-1} w[j-m+n-1]*h[m] - sum_{m=0}^{n-1} w[m-j+n-1]*h[m]]

     For j=0: sum w[0..n-1]
     For j=n-1: sum w[n-1..2n-2]
     These are all valid ranges.

     For term2[j]: indices into w are [n-1-j, n-j, ..., n-1] (as m goes from 0 to n-1)

     idx2[j, m] = n - 1 - j + m
     term2 = w[idx2] @ h
     
     And the partial sum:
     w_partial[j] = sum_{k=j}^{j+n-1} w[k]

```python
def smax_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx) / np.sum(np.exp(sh - mx))
    else:
        w = np.exp(sh) / np.sum(np.exp(sh))
    
    # w has length 2n-1, indexed 0 to 2n-2
    # grad[j] = dx * [sum_{k=j}^{j+n-1} w[k] - sum_m h[m]*w[j+n-1-m] - sum_m h[m]*w[n-1-j+m]]
    
    # Partial sum of w
    w_cum = np.concatenate([[0], np.cumsum(w)])  # w_cum[i] = sum of w[0..i-1], length 2n
    w_partial = w_cum[n:] - w_cum[:n-1+1]  # w_partial[j] = w_cum[j+n] - w_cum[j] = sum w[j..j+n-1]
    # w_cum has length 2n (indices 0 to 2n-1)
    # w_cum[j+n] - w_cum[j] for j in [0, n-1]: indices j+n in [n, 2n-1], j in [0, n-1]
    # w_cum has length 2n, so max index is 2n-1. j+n max is (n-1)+n = 2n-1. OK.
    
    # term1[j] = sum_m x[m] * w[j+n-1-m]
    # idx1[j,m] = j+n-1-m
    m_arr = np.arange(n)
    idx1 = np.repeat(m_arr[None, :], n, axis=0)  # this gives idx1[j,m] = m... no
    # Let me be explicit:
    # idx1[j, m] = j + n - 1 - m
    J = np.repeat(np.arange(n)[:, None], n, axis=1)  # J[j,m] = j
    M = np.tile(np.arange(n)[None, :], n, axis=0)    # M[j,m] = m
    idx1 = J + n - 1 - M  # idx1[j,m] = j+n-1-m
    idx2 = n - 1 - J + M  # idx2[j,m] = n-1-j+m
    
    term1 = w[idx1] @ x
    term2 = w[idx2] @ x
    
    return dx * (w_partial - term1 - term2)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smax_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (np.max(c - m) + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _build_grad_cache(n):
    J = np.repeat(np.arange(n)[:, None], n, axis=1)
    M = np.tile(np.arange(n)[None, :], n, axis=0)
    idx1 = J + n - 1 - M
    idx2 = n - 1 - J + M
    w_cum = np.cumsum(np.ones(2 * n - 1))
    w_partial_idx = (np.arange(n) + n, np.arange(n))
    return idx1, idx2, w_partial_idx[0], w_partial_idx[1]

def _smax_grad(x, n, alpha, idx1, idx2):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

    w_partial = w_cum[n:2*n-1] - w_cum[:n-1]
    term1 = w[idx1] @ x
    term2 = w[idx2] @ x
    return dx * (w_partial - term1 - term2)

def _eq_fun(x, n):
    return x.sum() - n / 2.0

def _eq_jac(x, n):
    return np.ones(n)

def do_slsqp(h, n, alpha, maxiter=2000, idx1=None, idx2=None):
    if idx1 is None:
        idx1, idx2, _, _ = _build_grad_cache(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_smax_obj, h, args=(n, alpha),
                       jac=lambda x: _smax_grad(x, n, alpha, idx1, idx2),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved_any = False
        while time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius) if radius else 0
                hi = min(n, i + radius + 1) if radius else n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:

                        c = compute_c5(h, n)
                        if c < best_c:

    return h, best_c

                                             [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
            else:
                best_h, best_c = pair_search(best_h, n, deadline - 2,
                                             [0.002, 0.001, 0.0005, 0.0002], radius=60)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
