from scipy.optimize import minimize
import time

def compute_c5_full(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c, float(c.max())

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_delta_c(h, n, i, di, j, dj, crit_ks):
    k_arr = crit_ks - (n - 1)
    delta = np.zeros(len(crit_ks))
    idx_a = i - k_arr
    ma = (idx_a >= 0) & (idx_a < n)
    if ma.any():
        delta[ma] += di * (1.0 - h[idx_a[ma]])
    idx_b = i + k_arr
    mb = (idx_b >= 0) & (idx_b < n)
    if mb.any():
        delta[mb] -= di * h[idx_b[mb]]
    idx_c = j - k_arr
    mc = (idx_c >= 0) & (idx_c < n)
    if mc.any():
        delta[mc] += dj * (1.0 - h[idx_c[mc]])
    idx_d = j + k_arr
    md = (idx_d >= 0) & (idx_d < n)
    if md.any():
        delta[md] -= dj * h[idx_d[md]]
    return delta

    c, max_c = compute_c5_full(h, n)
    best_c = max_c
    best_h = h.copy()
    crit_ks = np.where(c >= max_c - gap)[0]
    crit_vals = c[crit_ks].copy()
    nc_mask = np.ones(len(c), dtype=bool)
    nc_mask[crit_ks] = False
    non_crit_max = c[nc_mask].max() if nc_mask.any() else 0.0
    accepted = 0
    
    for d in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > deadline:

                                nc_mask = np.ones(len(c), dtype=bool)

                    if not tried and h[i] - d >= 0.0 and h[j] + d <= 1.0:

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sc(x, n):
    return float(np.sum(x) - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    best_h = h.copy()
    best_c = compute_c5_full(h, n)[1]
    
    for alpha in [500, 2000, 10000, 50000]:

        c = compute_c5_full(ht, n)[1]

            best_h = ht.copy()
    
    if time.time() < deadline - 5:
        best_h, best_c = fast_local_search(best_h, n, deadline)
    
    return best_h, best_c
