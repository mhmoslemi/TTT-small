    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    g = compute_all_g(h, n)
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    lse = sm + np.log(np.exp(shifted - sm).sum())
    return m + lse / alpha

def smooth_max_grad(h, n, alpha):
    g = compute_all_g(h, n)
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wk * (1 - h[lag:])
            grad[lag:] -= wk * h[:n - lag]
        else:
            ml = -lag
            grad[ml:] += wk * (1 - h[:n - ml])
            grad[:n - ml] -= wk * h[ml:]
    grad *= dx
    return grad

def lbfgs_with_penalty(h, n, deadline, alpha, mu0, mu_max):
    """L-BFGS-B with penalty method for equality constraint."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    mu = mu0
    while time.time() < deadline and mu < mu_max:
        def obj(x):
            g = compute_all_g(x, n)
            m = g.max()
            shifted = alpha * (g - m)
            sm = shifted.max()
            lse = sm + np.log(np.exp(shifted - sm).sum())
            pen = mu * (x.sum() - n / 2.0) ** 2
            return m + lse / alpha + pen
        
        def grad(x):
            gg = compute_all_g(x, n)
            mm = gg.max()
            shifted = alpha * (gg - mm)
            sm = shifted.max()
            w = np.exp(shifted - sm)
            w /= w.sum()
            dx = 2.0 / n
            gr = np.zeros(n)
            for k2 in range(2 * n - 1):
                wk = w[k2]
                if wk < 1e-12:
                    continue
                lag = k2 - (n - 1)
                if lag >= 0:
                    gr[:n - lag] += wk * (1 - x[lag:])
                    gr[lag:] -= wk * x[:n - lag]
                else:
                    ml = -lag
                    gr[ml:] += wk * (1 - x[:n - ml])
                    gr[:n - ml] -= wk * x[ml:]
            gr *= dx
            gr += 2 * mu * (x.sum() - n / 2.0)
            return gr
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=[(0, 1)] * n,
                          options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12})
            h = project_feasible(res.x, n)
        except Exception:
            break
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Increase penalty
        constraint_viol = abs(h.sum() - n / 2.0)
        if constraint_viol < 1e-8:
            mu *= 5
        else:
            mu *= 2
            
    return best_h, best_c

```python
def project_box_simplex(h, n):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = n/2}."""
    target = n / 2.0
    # Use bisection to find threshold t such that sum(clamp(x_i - t, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Small correction
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        # Distribute remaining
        if diff > 0:
            room = 1.0 - proj
            s = room.sum()
            if s > 1e-15:
                proj += diff / s * room
        else:
            s = proj.sum()
            if s > 1e-15:
                proj += diff / s * proj
    return np.clip(proj, 0, 1)
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def project_box_hyperplane(h, n):
    target = n / 2.0
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    proj = np.clip(h - t, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - proj
            s = room.sum()
            if s > 1e-15:
                proj += diff / s * room
        else:
            s = proj.sum()
            if s > 1e-15:
                proj += diff / s * proj
    return np.clip(proj, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    g = compute_all_g(h, n)
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wk * (1.0 - h[lag:])
            grad[lag:] -= wk * h[:n - lag]
        else:
            ml = -lag
            grad[ml:] += wk * (1.0 - h[:n - ml])
            grad[:n - ml] -= wk * h[ml:]
    grad *= dx
    return grad

def smooth_max_val(h, n, alpha):
    g = compute_all_g(h, n)
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    lse = sm + np.log(np.exp(shifted - sm).sum())
    return m + lse / alpha

def lbfgs_penalty(h, n, deadline, alpha, mu_schedule):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for mu in mu_schedule:
        if time.time() > deadline:
            break
        def obj(x):
            gv = compute_all_g(x, n)
            mv = gv.max()
            sh = alpha * (gv - mv)
            smv = sh.max()
            lse = smv + np.log(np.exp(sh - smv).sum())
            return mv + lse / alpha + mu * (x.sum() - n / 2.0) ** 2
        def grad(x):
            gv = compute_all_g(x, n)
            mv = gv.max()
            sh = alpha * (gv - mv)
            smv = sh.max()
            ww = np.exp(sh - smv)
            ww /= ww.sum()
            dx = 2.0 / n
            gr = np.zeros(n)
            for k2 in range(2 * n - 1):
                wk = ww[k2]
                if wk < 1e-12:
                    continue
                lag = k2 - (n - 1)
                if lag >= 0:
                    gr[:n - lag] += wk * (1.0 - x[lag:])
                    gr[lag:] -= wk * x[:n - lag]
                else:
                    ml = -lag
                    gr[ml:] += wk * (1.0 - x[:n - ml])
                    gr[:n - ml] -= wk * x[ml:]
            gr *= dx
            gr += 2.0 * mu * (x.sum() - n / 2.0)
            return gr
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-10})
            h = project_feasible(res.x, n)
        except Exception:
            break
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def proj_grad_descent(h, n, deadline, alpha, max_iter=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 1.0 / (4.0 * alpha * 2.0 / n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = smooth_max_grad(h, n, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        f0 = smooth_max_val(h, n, alpha)
        s = step
        h_new = h
        for _ in range(40):
            h_new = project_box_hyperplane(h - s * grad, n)
            f_new = smooth_max_val(h_new, n, alpha)
            if f_new <= f0 - 1e-4 * s * gn * gn:
                break
            s *= 0.5
        h = h_new
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        step = s * 1.5
    return best_h, best_c

def gradient_guided_search(h, n, deadline, top_k=15, max_changes=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    changes = 0
    while changes < max_changes:
        if time.time() > deadline:
            break
        g = compute_all_g(best_h, n)
        m = g.max()
        top_idx = np.argsort(g)[-top_k:]
        lags = top_idx - (n - 1)
        g_top = g[top_idx]
        F = np.zeros((n, top_k))
        for k in range(top_k):
            lag = lags[k]
            if lag >= 0:
                F[:n - lag, k] += (1.0 - best_h[lag:])
                F[lag:, k] -= best_h[:n - lag]
            else:
                ml = -lag
                F[ml:, k] += (1.0 - best_h[:n - ml])
                F[:n - ml, k] -= best_h[ml:]
        best_gain = 1e-14
        best_pair = None
        best_d = 0.0
        i_cand = np.where(best_h > 0.15)[0]
        j_cand = np.where(best_h < 0.85)[0]
        if len(i_cand) > 120:
            rng_s = np.random.RandomState(changes)
            i_cand = i_cand[rng_s.choice(len(i_cand), 120, replace=False)]
        if len(j_cand) > 120:
            rng_s = np.random.RandomState(changes + 999)
            j_cand = j_cand[rng_s.choice(len(j_cand), 120, replace=False)]
        for i in i_cand:
            hi = best_h[i]
            Fi = F[i]
            for j in j_cand:
                if i == j:
                    continue
                hj = best_h[j]
                d_lo = max(-hi, hj - 1.0)
                d_hi = min(1.0 - hi, hj)
                if d_hi - d_lo < 1e-10:
                    continue
                D = Fi - F[j]
                if np.all(np.abs(D) < 1e-14):
                    continue
                cands = [d_lo, d_hi]
                for a in range(top_k):
                    for b in range(a + 1, top_k):
                        dd = D[a] - D[b]
                        if abs(dd) < 1e-14:
                            continue
                        dk = (g_top[b] - g_top[a]) / dd
                        if d_lo < dk < d_hi:
                            cands.append(dk)
                for d in cands:
                    nm = np.max(g_top + d * D)
                    if nm < m - best_gain:
                        best_gain = m - nm
                        best_pair = (i, j)
                        best_d = d
        if best_pair is not None:
            i, j = best_pair
            best_h[i] += best_d
            best_h[j] -= best_d
            best_h = np.clip(best_h, 0.0, 1.0)
            c = compute_c5(best_h, n)
            if c < best_c:
                best_c = c
                changes += 1
            else:
                best_h[i] -= best_d
                best_h[j] += best_d
                break
        else:
            break
    return best_h, best_c

def fista_smooth(h, n, deadline, alpha, max_iter=100000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    y = h.copy()
    tk = 1.0
    L = alpha * 4.0 / n
    step = 1.0 / L
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = smooth_max_grad(y, n, alpha)
        h_new = project_box_hyperplane(y - step * grad, n)
        tk_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
        y = h_new + ((tk - 1.0) / tk_next) * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        tk = tk_next
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
