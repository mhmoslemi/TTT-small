def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj_pb(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        sm = mx + np.log(np.sum(np.exp(sh - mx)))
    else:
        sm = np.log(np.sum(np.exp(sh)))
    return float(m + sm / alpha + mu * (x.sum() - n / 2.0) ** 2)

def _grad_pb(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _obj_sl(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        sm = mx + np.log(np.sum(np.exp(sh - mx)))
    else:
        sm = np.log(np.sum(np.exp(sh)))
    return float(m + sm / alpha)

def _con_sl(x, n):
    return x.sum() - n / 2.0

def _jac_sl(x, n):
    return np.ones(n)

def optimize_lbfgs(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(50, 5.0), (500, 50.0), (5000, 500.0), (50000, 5000.0), (500000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 30:
            break
        try:
            res = minimize(_obj_pb, h, jac=_grad_pb, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxfun': 2000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def optimize_slsqp(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_sl, 'jac': _jac_sl, 'args': (n,)}]
    for alpha in [50000, 200000, 1000000, 5000000]:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(_obj_sl, h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def local_search(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    eps_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for eps in eps_list:
        if time.time() > deadline:
            break
        max_rounds = 50
        for rnd in range(max_rounds):
            if time.time() > deadline:
                break
            improved = False
            g = grad_smooth_max(h, n, 500000)
            order = np.argsort(np.abs(g))[::-1]
            for i in order[:80]:
                if time.time() > deadline:
                    return h, best_c
                for d in [eps, -eps]:
                    if h[i] + d < -1e-10 or h[i] + d > 1 + 1e-10:
                        continue
                    if d > 0:
                        j_cands = np.argsort(g)[:15]
                    else:
                        j_cands = np.argsort(g)[-15:]
                    found = False
                    for j in j_cands:
                        if j == i:
                            continue
                        if h[j] - d < -1e-10 or h[j] - d > 1 + 1e-10:
                            continue

                        h[i] += d

                        h = project_feasible(h, n)
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                            found = True
                            break
                        h[i] = ov
                        h[j] = ow
                    if found:
                        break
                if improved:
                    break
            if not improved:
                break
    return h, best_c
