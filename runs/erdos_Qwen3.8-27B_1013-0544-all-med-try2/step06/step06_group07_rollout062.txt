import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

    return h.sum() - target_sum

    return np.ones(len(h))

    h = h.copy()
    for j in range(n):
        old_val = h[j]
        h[j] = 0.0
        corr0 = np.correlate(h, 1.0 - h, mode="full")
        h[j] = 1.0
        corr1 = np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h.copy()

def obj_smooth(h_flat, dx, beta):
    corr = np.correlate(h_flat, 1.0 - h_flat, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum):
    return h.sum() - target_sum

def con_jac(h, target_sum):
    return np.ones(len(h))

        h[j] = 1.0
        corr1 = np.correlate(h, 1.0 - h, mode="full")
        h[j] = old_val
        sum_others = h.sum() - old_val
        h[j] = coord_optimize_1d(corr0, corr1, n, dx, sum_others, target, lam)
    return h

    return np.interp(x_new, x_old, h_old)
