```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def gradient_at_k(h, n, k):
    """Compute gradient of g[k] with respect to h, where k is the index in the full correlation."""
    # k is index in the full correlation array (0 to 2n-2)
    # The shift is: shift = k - (n-1)
    dx = 2.0 / n
    shift = k - (n - 1)
    grad = np.zeros(n)
    
    if shift >= 0:
        # g = sum_i h[i+shift] * (1-h[i]) for valid i
        # Valid i: 0 <= i <= n-1-shift
        # First term: (1-h[j-shift]) for j in [shift, n-1]
        for j in range(shift, n):
            grad[j] += dx * (1 - h[j - shift])
        # Second term: -h[j+shift] for j in [0, n-1-shift]
        for j in range(0, n - shift):
            grad[j] -= dx * h[j + shift]
    else:
        # shift < 0, let s = -shift
        s = -shift
        # g = sum_i h[i-s] * (1-h[i]) for valid i, i.e., s <= i <= n-1
        # = sum_i h[i] * (1-h[i+s]) for valid i, i.e., 0 <= i <= n-1-s
        # First term: (1-h[j+s]) for j in [0, n-1-s]
        for j in range(0, n - s):
            grad[j] += dx * (1 - h[j + s])
        # Second term: -h[j-s] for j in [s, n-1]
        for j in range(s, n):
            grad[j] -= dx * h[j - s]
    
    return grad
```

```python
def projected_smooth_descent(h, n, deadline, alpha_list, max_iter_per_alpha=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        h = best_h.copy()
        lr = 1.0
        for it in range(max_iter_per_alpha):
            if time.time() > deadline:
                break
            g = compute_all_g(h, n)
            m = g.max()
            
            # Compute smooth gradient
            sh = alpha * (g - m)
            sm = sh.max()
            weights = np.exp(sh - sm)
            wsum = weights.sum()
            if wsum < 1e-10:
                break
            weights /= wsum
            
            # Only use significant components
            sig = np.where(weights > 1e-4)[0]
            grad = np.zeros(n)
            for k in sig:
                grad += weights[k] * gradient_at_k(h, n, k)
            
            # Project gradient to satisfy sum constraint
            grad -= grad.mean()  # projection onto sum-preserving subspace
            
            # Backtracking line search
            found = False
            for f in range(20):
                step = lr * (0.5 ** f)
                h_new = project_feasible(h - step * grad, n)
                c_new = compute_c5(h_new, n)
                if c_new < m - 1e-14:
                    h = h_new
                    found = True
                    break
            if not found:
                lr *= 0.5
                if lr < 1e-10:
                    break
            else:
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
    
    return best_h, best_c
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def gradient_smooth(h, n, alpha):
    """Compute the gradient of the smooth approximation of max_k g_k."""
    g = compute_all_g(h, n)
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    weights = np.exp(sh - sm)
    wsum = weights.sum()
    if wsum < 1e-15:
        return np.zeros(n)
    weights /= wsum

    grad = np.zeros(n)
    dx = 2.0 / n
    for k in sig:

        else:
            s = -shift
            grad[:n-s] += w * dx * (1 - h[s:])
            grad[s:] -= w * dx * h[:n-s]

    h = h.copy()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_single(h, n, k):
    dx = 2.0 / n
    shift = k - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        grad[shift:] += dx * (1.0 - h[:n - shift])
        grad[:n - shift] -= dx * h[shift:]
    else:
        s = -shift
        grad[:n - s] += dx * (1.0 - h[s:])
        grad[s:] -= dx * h[:n - s]
    return grad

def smooth_grad(h, n, alpha):
    g = compute_all_g(h, n)
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    ws = w.sum()
    if ws < 1e-15:
        return np.zeros(n), m
    w /= ws
    sig = np.where(w > 1e-5)[0]
    grad = np.zeros(n)
    for k in sig:
        grad += w[k] * grad_single(h, n, k)
    return grad, m

def optimize_smooth(h, n, deadline, a0, a1, max_steps, lr_init=0.3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    lr = lr_init
    for it in range(max_steps):
        if time.time() > deadline:
            break
        frac = min(1.0, 0.999 * it / max(1, max_steps))
        alpha = a0 * (a1 / a0) ** frac
        grad, m = smooth_grad(h, n, alpha)
        grad -= grad.mean()
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        found = False
        for f in range(25):
            step = lr * (0.5 ** f) / ng
            hn = project_feasible(h - step * ng * grad, n)
            cn = compute_c5(hn, n)
            if cn < m - 1e-14:
                h = hn
                found = True
                break
        if not found:
            lr *= 0.6
            if lr < 1e-13:
                break
        else:
            lr = min(lr * 1.08, 3.0)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def optimize_exact(h, n, deadline, max_steps, lr_init=0.2):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    lr = lr_init
    for it in range(max_steps):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        kmax = int(np.argmax(g))
        grad = grad_single(h, n, kmax)
        grad -= grad.mean()
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        found = False
        for f in range(25):
            step = lr * (0.5 ** f)
            hn = project_feasible(h - step * grad, n)
            cn = compute_c5(hn, n)
            if cn < m - 1e-14:
                h = hn
                found = True
                break
        if not found:
            lr *= 0.6
            if lr < 1e-13:
                break
        else:
            lr = min(lr * 1.05, 2.0)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def binary_search(h, n, deadline, max_sweeps=30):
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    best_c = compute_c5(hb, n)
    ones = sorted(set(np.where(hb == 1)[0].tolist()))
    zeros = sorted(set(np.where(hb == 0)[0].tolist()))
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        new_zeros = []
        for i in zeros:
            if time.time() > deadline:
                break
            hb[i] = 1.0
            g_new = compute_all_g(hb, n)
            best_j = None
            best_cj = best_c
            for j in ones:
                hb[j] = 0.0
                c = float(g_new.max())
                # Need to recompute since g_new was with j still 1
                c = compute_c5(hb, n)
                if c < best_cj:
                    best_cj = c
                    best_j = j
            if best_j is not None:
                hb[best_j] = 0.0
                ones.remove(best_j)
                ones.append(i)
                ones.sort()
                zeros.remove(i)
                zeros.append(best_j)
                zeros.sort()
                best_c = best_cj
                improved = True
                break
            else:
                hb[i] = 0.0
        if not improved:
            break
    return hb, best_c
```
