                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def optimize_n(h_start, n, deadline, alphas=None):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    if alphas is None:
        alphas = [300, 1000, 5000, 20000, 80000, 300000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = slsqp_opt(h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    # Gradient-guided local search
    if time.time() < deadline - 5:
        h, best_c = grad_pair_search(h, n, deadline - 5, alpha=100000)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize initial construction
    alloc1 = min(180, (deadline - t0) * 0.2)
    h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc1)
    best_h, best_c = h_opt, c_opt

    # Phase 2: Try different n values
    x_init = np.linspace(0, 2, n_init)
    for n_try in [100, 150, 200]:
        if time.time() > deadline - 100:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(100, (deadline - time.time()) * 0.15)
        if alloc < 40:
            break
        h_opt, c_opt = optimize_n(h_interp, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multi-restart with different perturbations
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(4, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            if r == 0:
                scale = 0.02
            elif r == 1:
                scale = 0.05
            elif r == 2:
                scale = 0.1
            else:
                scale = 0.15
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Try structured initializations
    if time.time() < deadline - 60:
        for n_struct in [100, 150]:
            if time.time() > deadline - 40:
                break
            # Cosine-based initialization
            x = np.linspace(0, 2, n_struct)
            h_cos = 0.5 + 0.4 * np.cos(np.pi * x)
            h_cos = np.clip(h_cos, 0, 1)
            h_cos = project_feasible(h_cos, n_struct)
            alloc = min(40, deadline - time.time() - 20)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_n(h_cos, n_struct, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            # Sine-based
            h_sin = 0.5 + 0.4 * np.sin(np.pi * x)
            h_sin = np.clip(h_sin, 0, 1)
            h_sin = project_feasible(h_sin, n_struct)
            if time.time() < deadline - 20:
                h_opt, c_opt = optimize_n(h_sin, n_struct, time.time() + 30)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Phase 5: Final intensive local search on best
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        if n <= 150:
            best_h, best_c = full_pair_search(best_h, n, deadline - 3,
                                              [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
        else:
            best_h, best_c = grad_pair_search(best_h, n, deadline - 5, alpha=200000)
            if time.time() < deadline - 3:
                best_h, best_c = full_pair_search(best_h, n, deadline - 2,
                                                  [0.005, 0.002, 0.001, 0.0005])

    # Phase 6: Final SLSQP polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        ht = slsqp_opt(best_h, n, 500000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if n <= 150 and time.time() < deadline - 2:
            best_h, best_c = full_pair_search(best_h, n, deadline - 1,
                                              [0.002, 0.001, 0.0005, 0.0002, 0.0001])

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
