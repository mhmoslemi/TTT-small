```python
def compute_gradient(h, n, w):
    """w is the softmax weights for each lag, length 2n-1"""
    dx = 2.0 / n
    offsets = np.arange(2*n-1) - (n-1)  # s_k values
    grad = np.zeros(n)
    for k, s in enumerate(offsets):
        if w[k] < 1e-15:
            continue
        # Contribution to grad from lag k
        # A part: (1-h[j-s]) for valid j-s
        if s >= 0:
            # j-s valid means j >= s
            j_idx = np.arange(s, n)
            grad[j_idx] += w[k] * (1 - h[j_idx - s])
        else:
            # j-s = j+|s| valid means j < n-|s|
            j_idx = np.arange(0, n + s)
            grad[j_idx] += w[k] * (1 - h[j_idx - s])
        # B part: -h[j+s] for valid j+s
        if s >= 0:
            # j+s valid means j < n-s
            j_idx = np.arange(0, n - s)
            grad[j_idx] -= w[k] * h[j_idx + s]
        else:
            # j+s = j-|s| valid means j >= |s|
            j_idx = np.arange(-s, n)
            grad[j_idx] -= w[k] * h[j_idx + s]
    return grad * dx
```

def project_feasible(h, n):

def lag_gradient(h, n, k):
    """Gradient of g_k with respect to h, where k is the index in the full correlation."""
    dx = 2.0 / n
    s = k - (n - 1)
    grad = np.zeros(n)

    else:
        a = -s
        grad[:n-a] += (1.0 - h[a:])
        grad[a:] -= h[:n-a]
    return grad * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = project_feasible(np.full(n0, 0.5), n0)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Smooth gradient on initial (35% budget)
    p1 = t0 + 0.35 * (budget_s - 10)
    h = best_h.copy()
    for seq in [[30, 100, 500], [200, 1000, 5000], [5000, 20000, 100000], [50000, 200000]]:
        if time.time() > p1:
            break
        h, c = smooth_optimize(h, n, p1, seq, lr0=0.4, max_iter_per_alpha=500)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: SLSQP polish (15% budget)
    p2 = t0 + 0.50 * (budget_s - 10)
    for alpha in [500, 2000, 10000, 50000]:
        if time.time() > p2:
            break
        h = slsqp_optimize(best_h.copy(), n, alpha, 800, p2)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 3: Multi-resolution (20% budget)
    p3 = t0 + 0.70 * (budget_s - 10)
    for nh in [150, 200, 300, 400, 500]:
        if time.time() > p3 - 20:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, c = smooth_optimize(hh, nh, p3 - 15,
                                [100, 500, 2000, 10000, 50000], lr0=0.5, max_iter_per_alpha=300)
        if c < best_c:
            best_c = c
            best_h = hh.copy()
            n = nh

    # Phase 4: Random binary restarts (15% budget)
    p4 = t0 + 0.85 * (budget_s - 10)
    for attempt in range(5):
        if time.time() > p4 - 10:
            break
        hb = random_binary_init(n, rng)
        hb, c = smooth_optimize(hb, n, p4 - 10,
                                [50, 200, 1000, 5000, 20000], lr0=0.5, max_iter_per_alpha=200)
        if c < best_c:
            best_c = c
            best_h = hb.copy()

    # Phase 5: Binary local search (5% budget)
    p5 = t0 + 0.92 * (budget_s - 10)
    if time.time() < p5 - 5:
        hbs, cbs = binary_local_search(best_h.copy(), n, p5, max_sweeps=200)
        if cbs < best_c:
            best_c = cbs
            best_h = hbs.copy()

    # Phase 6: Final smooth polish
    if time.time() < dl - 15:
        h, c = smooth_optimize(best_h.copy(), n, dl,
                                [20000, 100000, 500000], lr0=0.2, max_iter_per_alpha=500)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
