    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room

            s = h.sum()
            if s < 1e-14:

def corr_and_grad(h, n, alpha):
    """Compute log-sum-exp objective and its gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    shifted = alpha * (c - cmax)
    # For numerical stability
    smax = shifted.max()
    exps = np.exp(shifted - smax)
    logsum = smax + np.log(exps.sum())
    obj = cmax + logsum / alpha
    # Weights for gradient
    w = exps / exps.sum()
    # Compute gradient
    A = np.convolve(1.0 - h, w, mode='full')
    B = np.correlate(w, h, mode='full')
    j = np.arange(n)
    grad = (A[j + n - 1] - B[n - 1 - j]) / alpha
    # Add penalty for constraint
    pen = 100.0
    grad += 2.0 * pen * (h.sum() - n / 2.0) * (dx / dx)  # just the linear penalty gradient
    obj += pen * (h.sum() - n / 2.0) ** 2
    return obj, grad

def lbfgs_optimize(h, n, alpha, penalty, maxiter, deadline):
    """L-BFGS-B optimization with log-sum-exp objective."""
    dx = 2.0 / n
    npoints = n
    
    def fun_and_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        cmax = c.max()
        shifted = alpha * (c - cmax)
        smax = shifted.max()
        exps = np.exp(shifted - smax)
        sumexp = exps.sum()
        logsum = smax + np.log(sumexp)
        obj = cmax + logsum / alpha
        w = exps / sumexp
        A = np.convolve(1.0 - x, w, mode='full')
        B = np.correlate(w, x, mode='full')
        j_arr = np.arange(npoints)
        grad = (A[j_arr + npoints - 1] - B[npoints - 1 - j_arr]) / alpha
        s = x.sum() - npoints / 2.0
        obj += penalty * s * s
        grad = grad + 2.0 * penalty * s
        return obj, grad
    
    bounds = [(0.0, 1.0)] * npoints
    try:
        res = minimize(fun_and_grad, h, method='L-BFGS-B', jac=True,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, npoints)
    except Exception:
        return h

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    """Greedy swap local search for binary solutions."""
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    m_range = np.arange(-(n - 1), n)
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                # Compute delta for swapping i (0→1) and j (1→0)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta_im = np.zeros(2*n-1)
                delta_im[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta_im[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta_im[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta_im[msk] += h[jmn[msk]]
                delta_im[n-1] -= 2.0
                k1 = n - 1 + (j - i)
                delta_im[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta_im[k2] += 1.0
                nc = (corr + delta_im).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                j = best_j
                # Recompute delta for the chosen swap
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta = np.zeros(2*n-1)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c
