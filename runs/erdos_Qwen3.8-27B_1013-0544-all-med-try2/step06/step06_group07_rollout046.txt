from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_coords_A(h, j):
    """Compute A_m coefficients for coordinate j in the correlation."""
    n = len(h)
    A = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        coeff = 0.0
        if k >= 0:
            if j + k < n:
                coeff += (1.0 - h[j + k])
            if j >= k:
                coeff += -h[j - k]
        else:
            kp = -k
            if j >= kp:
                coeff += (1.0 - h[j - kp])
            if j + kp < n:
                coeff += -h[j + kp]
        A[m] = coeff
    return A

def coord_descent(h, target_sum, max_sweeps=200):
    """Coordinate descent for min max correlation."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = np.max(corr)
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            A = compute_coords_A(h, j)
            B = corr - A * h[j]
            # Minimize max_m (A_m * t + B_m) over t in [0,1]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(A * m1 + B)
                f2 = np.max(A * m2 + B)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            new_val = (lo + hi) / 2.0
            if abs(new_val - h[j]) > 1e-10:
                h[j] = new_val
                # Recompute affected correlations
                # Actually, let's just recompute all for simplicity (n is small)
                corr = np.correlate(h, 1.0 - h, mode="full")
                improved = True

        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            break
    
    return h, best_c5

            h[j] = (lo + hi) / 2.0

        s = h.sum()
        lam += 0.1 * (target_sum - s) / max(1, abs(target_sum))

        h = np.clip(h, 0, 1)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_A(h, j):
    n = len(h)
    A = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        c = 0.0
        if k >= 0:
            if j + k < n:
                c += (1.0 - h[j + k])
            if j >= k:
                c += -h[j - k]
        else:
            kp = -k
            if j >= kp:
                c += (1.0 - h[j - kp])
            if j + kp < n:
                c += -h[j + kp]
        A[m] = c
    return A

def coord_descent_min(h, target_sum, max_sweeps=300, tol=1e-10):
    n = len(h)
    h = h.copy()
    best = compute_c5(h)
    for sweep in range(max_sweeps):
        changed = False
        for j in range(n):
            A = compute_A(h, j)
            corr = np.correlate(h, 1.0 - h, mode="full")
            B = corr - A * h[j]
            lo, hi = 0.0, 1.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(A * m1 + B)
                f2 = np.max(A * m2 + B)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            new_val = (lo + hi) / 2.0
            if abs(new_val - h[j]) > 1e-12:
                h[j] = new_val
                changed = True
        h = project_feasible(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best:
            best = c5
        if not changed and sweep > 5:
            break
    return h, best

def coord_descent_maxmin(h, target_sum, max_sweeps=300):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        changed = False
        for j in range(n):
            A = compute_A(h, j)
            corr = np.correlate(h, 1.0 - h, mode="full")
            B = corr - A * h[j]
            lo, hi = 0.0, 1.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.min(1.0 - (A * m1 + B))
                f2 = np.min(1.0 - (A * m2 + B))
                if f1 > f2:
                    hi = m2
                else:
                    lo = m1
            new_val = (lo + hi) / 2.0
            if abs(new_val - h[j]) > 1e-12:
                h[j] = new_val
                changed = True
        h = project_feasible(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:

        if not changed and sweep > 5:
            break
    return h, best_c5

def slsqp_polish(h, target_sum, n, beta_list, maxiter=3000):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for beta in beta_list:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return smooth_max(corr, beta) * dx
        def cons_fun(x):
            return x.sum() - target_sum
        def cons_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
        except Exception:
            pass
    return best_h, best_c5

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

        dx_t = 2.0 / n_try

    best_h = project_feasible(best_h, len(best_h) / 2.0)

    return best_h, best_c5, len(best_h)
