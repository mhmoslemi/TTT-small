    k = int(np.argmax(g))
    shift = k - (n - 1)

        grad[:n-s] -= h[s:]

    for s in range(1, n):
        pk = p[n - 1 + s]
        if pk > 1e-10:
            grad[:n - s] += pk * (1.0 - h[s:])
            grad[s:] -= pk * h[:n - s]
        pk2 = p[n - 1 - s]
        if pk2 > 1e-10:
            grad[s:] += pk2 * (1.0 - h[:n - s])
            grad[:n - s] -= pk2 * h[s:]
    return grad

def pg_optimize(h, n, deadline, mode='exact', alpha=1000, max_iter=100000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.1
    stuck = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if mode == 'exact':
            grad = get_grad_exact(h, n)
        else:
            grad = get_grad_smooth(h, n, alpha)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        grad /= ng
        step = lr
        improved = False
        for _ in range(12):
            if time.time() > deadline:
                break
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < best_c - 1e-15:
                h = ht
                best_c = ct
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if improved:
            stuck = 0
            lr = min(lr * 1.02, 5.0)
        else:
            stuck += 1
            lr *= 0.75
            if lr < 1e-9:
                break
            if stuck > 80:
                lr = 0.1
                stuck = 0
    return best_h, best_c

def _slsqp_obj_c5(x, n, alpha):
    g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def _slsqp_grad_c5(x, n, alpha):
    g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    gm = g.max()
    p = np.exp(alpha * (g - gm))
    p /= p.sum()
    grad = np.zeros(n)
    grad += p[n - 1] * (1.0 - 2.0 * x)
    for s in range(1, n):
        pk = p[n - 1 + s]
        if pk > 1e-10:
            grad[:n - s] += pk * (1.0 - x[s:])
            grad[s:] -= pk * x[:n - s]
        pk2 = p[n - 1 - s]
        if pk2 > 1e-10:
            grad[s:] += pk2 * (1.0 - x[:n - s])
            grad[:n - s] -= pk2 * x[s:]
    return grad

_slqp_n_global = 250

def _slsqp_eq_n(x):
    return x.sum() - _slqp_n_global / 2.0

def slsqp_smooth(h, n, deadline, alphas=None):
    global _slqp_n_global
    if alphas is None:
        alphas = [2000, 10000, 50000, 200000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    _slqp_n_global = n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _slsqp_eq_n}]
        try:
            res = minimize(_slsqp_obj_c5, h,
                           jac=lambda x: _slsqp_grad_c5(x, n, alpha),
                           method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-14})
            h = project_feasible(res.x, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        except Exception:
            pass
    return best_h, best_c

def binary_init_from(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def multi_start(h, n, deadline, rng, n_starts=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for s in range(n_starts):
        if time.time() > deadline - 30:
            break
        pert = 0.05 * (s + 1)
        hp = project_feasible(h + pert * rng.randn(n), n)
        hh, c = pg_optimize(hp, n, min(deadline, time.time() + 40), mode='smooth', alpha=5000, max_iter=20000)
        if c < best_c:
            best_c = c
            best_h = hh.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Exact max PG (15% budget)
    p1 = t0 + 0.15 * (budget_s - 5)
    h, c = pg_optimize(h0.copy(), n, p1, mode='exact', max_iter=200000)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 2: Multi-start with perturbation (15% budget)
    p2 = t0 + 0.30 * (budget_s - 5)
    h, c = multi_start(best_h.copy(), n, p2, rng, n_starts=4)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 3: Smooth max PG with increasing alpha (30% budget)
    p3 = t0 + 0.60 * (budget_s - 5)
    for alpha in [200, 1000, 5000, 20000, 100000]:
        if time.time() > p3 - 10:
            break
        h, c = pg_optimize(best_h.copy(), n, min(p3, time.time() + 50),
                           mode='smooth', alpha=alpha, max_iter=50000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: SLSQP refinement (20% budget)
    p4 = t0 + 0.80 * (budget_s - 5)
    h, c = slsqp_smooth(best_h.copy(), n, p4, [5000, 20000, 100000])
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 5: Higher resolution (15% budget)
    p5 = dl
    if time.time() < p5 - 40:
        for nh in [350, 500]:
            if time.time() > p5 - 25:
                break
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, nh)
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            hh, c1 = pg_optimize(hh, nh, min(p5 - 5, time.time() + 35),
                                 mode='smooth', alpha=5000, max_iter=20000)
            c1 = compute_c5(hh, nh)
            if c1 < best_c:
                best_c = c1
                best_h = hh.copy()
                n = nh
                xo = np.linspace(0, 2, n)
    # Phase 6: Final exact max polish
    if time.time() < dl - 3:
        h, c = pg_optimize(best_h.copy(), n, dl, mode='exact', max_iter=100000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
