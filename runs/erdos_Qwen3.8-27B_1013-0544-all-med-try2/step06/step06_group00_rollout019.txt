import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha, pen_w):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    logsum = sm + np.log(np.exp(shifted - sm).sum())
    obj = m + logsum / alpha
    if pen_w > 0:
        s = x.sum() - n / 2.0
        obj += pen_w * s * s
    return obj

def grad_smooth(x, n, alpha, pen_w):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    weights = np.exp(shifted - sm)
    weights /= weights.sum()
    
    n_eff = 2 * n - 1
    grad = np.zeros(n)
    lags = np.arange(n_eff) - (n - 1)
    
    for k in range(n_eff):
        wk = weights[k]
        if wk < 1e-12:
            continue
        lag = lags[k]
        # (1 - h[i-lag]) * 1[0<=i-lag<n]
        if lag >= 0:
            grad[lag:] += wk * dx * (1.0 - x[:n-lag])
        else:
            grad[:n+lag] += wk * dx * (1.0 - x[-lag:])
        # -h[i+lag] * 1[0<=i+lag<n]
        if lag >= 0:
            grad[:n-lag] -= wk * dx * x[lag:]
        else:
            grad[-lag:] -= wk * dx * x[:n+lag]
    
    if pen_w > 0:
        s = x.sum() - n / 2.0
        grad += 2.0 * pen_w * s
    return grad

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greed_binary(h_init, n, deadline, max_sweeps=500):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0].astype(int))
    zeros_list = list(np.where(h == 0)[0].astype(int))
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 8
    deadline = t0 + total
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    
    n = n0
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Binary local search
    p1 = t0 + 0.04 * total
    hb, cb = greed_binary(best_h, n, p1, max_sweeps=150)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    
    # Phase 2: L-BFGS-B with increasing alpha
    p2 = t0 + 0.65 * total
    h = best_h.copy()
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000, 50000000]
    pen_w = 1.0
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > p2 - 5:
            break
        remaining = p2 - 5 - time.time()
        maxiter = min(300, max(20, int(remaining * 2)))
        try:
            res = minimize(obj_smooth, h, args=(n, alpha, pen_w),
                          jac=grad_smooth, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'maxfun': maxiter*3, 'ftol': 1e-15})
            h = project_feasible(res.x, n)
        except Exception:
            pass
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    
    # Phase 3: Noise restarts
    p3 = t0 + 0.85 * total
    for trial in range(8):
        if time.time() > p3 - 5:
            break
        noise_scale = 0.005 + 0.01 * trial / 8.0
        h = best_h + noise_scale * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [10000, 100000, 1000000, 10000000]:
            if time.time() > p3 - 5:
                break
            maxiter = min(200, max(10, int((p3 - 5 - time.time()) * 3)))
            try:
                res = minimize(obj_smooth, h, args=(n, alpha, pen_w),
                              jac=grad_smooth, method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': maxiter, 'maxfun': maxiter*3, 'ftol': 1e-15})
                h = project_feasible(res.x, n)
            except Exception:
                pass
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
    
    # Phase 4: Final fine-tuning
    if time.time() < deadline - 3:

        for alpha in [100000000, 500000000]:
            if time.time() > deadline - 3:

            except Exception:
                pass
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n

    if lag >= 0:
        # i+lag >= 0 (always true)
        # i+lag < n => i < n-lag
        # So: i ranges from 0 to n-1-lag, and h[i+lag] = h[lag..n-1]
        grad[:n-lag] -= wk * dx * x[lag:]
    else:
        # i+lag >= 0 => i >= -lag = |lag|
        # i+lag < n => i < n (always true)
        # So: i ranges from |lag| to n-1, and h[i+lag] = h[0..n-1-|lag|]
        ilag = -lag
        grad[ilag:] -= wk * dx * x[:n-ilag]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha, pen_w):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    logsum = sm + np.log(np.exp(shifted - sm).sum())
    obj = m + logsum / alpha
    if pen_w > 0:
        s = x.sum() - n / 2.0
        obj += pen_w * s * s
    return float(obj)

def grad_smooth(x, n, alpha, pen_w):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    weights = np.exp(shifted - sm)
    weights /= weights.sum()
    n_eff = 2 * n - 1
    lags = np.arange(n_eff) - (n - 1)
    grad = np.zeros(n)
    for k in range(n_eff):
        wk = weights[k]
        if wk < 1e-12:
            continue
        lag = lags[k]
        if lag >= 0:
            grad[lag:] += wk * dx * (1.0 - x[:n - lag])
            grad[:n - lag] -= wk * dx * x[lag:]
        else:
            ilag = -lag
            grad[:n - ilag] += wk * dx * (1.0 - x[ilag:])
            grad[ilag:] -= wk * dx * x[:n - ilag]
    if pen_w > 0:
        s = x.sum() - n / 2.0
        grad += 2.0 * pen_w * s
    return grad

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greed_binary(h_init, n, deadline, max_sweeps=500):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0].astype(int))
    zeros_list = list(np.where(h == 0)[0].astype(int))
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

    total = budget_s - 8
    deadline = t0 + total
    rng = np.random.RandomState(seed)
    try:

    bounds = [(0.0, 1.0)] * n
    # Phase 1: Binary local search

    # Phase 2: L-BFGS-B smooth optimization with increasing alpha
    p2 = t0 + 0.62 * total

    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000, 50000000]
