import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_swap_delta(h, i, j, n):
    """Compute the change in correlation when h[i]: 0->1, h[j]: 1->0"""
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def greed_binary(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    m_range = np.arange(-(n - 1), n)
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = binary_swap_delta(h, i, j, n)
                nc = (corr + delta).max() * dx
                if nc < best_nc:

                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                # Do the swap
                delta = binary_swap_delta(h, i, best_j, n)
                h[i] = 1.0
                h[best_j] = 0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

def proj_grad_smooth(h, n, deadline, alpha_start=100, alpha_end=100000, n_iters=50000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    T = 0.01  # temperature for accepting worse solutions
    lr = 0.01  # learning rate
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c_max = corr.max()
        
        # Adaptive alpha
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / n_iters)
        
        # Smoothed max gradient
        shifted = alpha * (corr - c_max)
        weights = np.exp(shifted - shifted.max())
        weights /= weights.sum()
        
        # Gradient of smoothed max
        # grad_j = sum_k weights[k] * d(corr[k])/dh[j]
        # d(corr[p])/dh[j] = dx * [(1-h[j+m]) - h[j-m]] where m = p - (n-1)

            best_c = c_new
            best_h = h_new.copy()
        
        T *= 0.9999
        lr *= 0.999

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            s = room.sum()
            if s < 1e-14:

            h += diff / s * room
        else:

            if s < 1e-14:

def swap_delta(h, i, j, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-14:
            break

            s = room.sum()
            if s < 1e-14:
                break

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

def swap_delta(h, i, j, n):
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range

    k1 = n - 1 + (j - i)

    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta
