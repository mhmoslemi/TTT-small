import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0, 1)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h * (1.0 + diff / tot), 0, 1)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)

        logsumexp = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        w = np.exp(shifted - mx)
    else:
        logsumexp = m + np.log(np.sum(np.exp(shifted))) / alpha
        w = np.exp(shifted)
    w = w / w.sum()

    grad_h = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        l = lags[idx]
        wt = w[idx]
        j1 = max(0, l)
        j2 = min(n, n + l)
        if j1 < j2:
            grad_h[j1:j2] += wt * (1.0 - h[j1 - l:j2 - l])
        j3 = max(0, -l)
        j4 = min(n, n - l)
        if j3 < j4:
            grad_h[j3:j4] -= wt * h[j3 + l:j4 + l]
    return logsumexp, grad_h

def sum_con(h, n):
    return float(np.sum(h) - n / 2.0)

def sum_con_jac(h, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(lambda x: smooth_obj_and_grad(x, n, alpha)[0],
                       h, args=(n, alpha),
                       jac=lambda x: smooth_obj_and_grad(x, n, alpha)[1],
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_refine(h, n, deadline, top_k=10):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    lags = np.arange(-(n - 1), n)
    for ss in [0.005, 0.003, 0.001, 0.0005, 0.0002]:
        stuck = 0
        for _ in range(20000):
            if time.time() > deadline:
                return best_h, best_c
            c = compute_all_c(h, n)
            c_max = c.max()
            if c_max < best_c:
                best_c = c_max
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
                if stuck > 2000:
                    break
            idx_active = np.argsort(c)[-top_k:]
            w = np.zeros(len(c))
            w[idx_active] = 1.0
            w /= w.sum()
            grad = np.zeros(n)
            nz = np.where(w > 1e-10)[0]
            for idx in nz:
                l = lags[idx]

                    grad[j3:j4] -= wt * h[j3 + l:j4 + l]
            h = h - ss * grad
            h = project_feasible(h, n)
    return best_h, best_c

                best_h = h_new.copy()
                h = h_new
            else:
