def smooth_max(x, beta):
    m = np.max(x)

When k < 0: let j = -k > 0: c[-j] = sum_{n=0}^{n-1-j} h[n] * (1-h[n+j]) = sum_{n=0}^{n-1-j} h[n] - sum_{n=0}^{n-1-j} h[n]*h[n+j]

So for k > 0: c[k] involves pairs (n, n+k) → sum h[n+k](1-h[n]) = sum h[n+k] - sum h[n]h[n+k]
For k < 0 (k = -j): c[-j] involves pairs (n, n+j) → sum h[n](1-h[n+j]) = sum h[n] - sum h[n]h[n+j]

For k < 0 (k = -j): t ≥ dx * (sum_{n=0}^{n-1-j} h[n] - sum_{n=0}^{n-1-j} z_n)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-15:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def grad_lag(h, n, k_idx):
    dx = 2.0 / n
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        npairs = n - lag
        if npairs <= 0:
            return grad
        idx = np.arange(npairs)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad * dx

def active_set_subgrad(h, target_sum, t0, wall, top_m, lr0, momentum, max_iters):
    n = len(h)
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    vel = np.zeros(n)
    for it in range(max_iters):
        if time.time() - t0 > wall:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        top = np.argsort(corr)[-top_m:]
        grad = np.zeros(n)
        for ki in top:
            grad += grad_lag(h_curr, n, ki)
        grad /= top_m
        lr = lr0 / (1.0 + it * 0.0002)
        vel = momentum * vel - lr * grad
        h_new = project_feasible(h_curr + vel, target_sum)
        h_curr = h_new
        if it % 150 == 149:
            c5 = compute_c5(h_curr)
            if c5 < best_c5:
                best_c5, best_h = c5, h_curr.copy()
    c5 = compute_c5(h_curr)
    if c5 < best_c5:
        best_c5, best_h = c5, h_curr.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation with increasing beta
    for beta in [15, 80, 400, 2000]:
        if tl() < 70:
            break
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for trial in range(3):
            if tl() < 20:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += np.random.randn(n) * 0.015 / trial
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 4000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 2: Active-set subgradient (top-5, heavy ball)
    if tl() > 180:
        h2, c52 = active_set_subgrad(best_h, target_sum, t0, 60, 5, 0.004, 0.88, 80000)
        if c52 < best_c5:
            best_h, best_c5 = h2, c52

    # Phase 3: High-beta SLSQP polish
    if tl() > 120:
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for beta in [8000, 40000, 150000]:
            if tl() < 25:
                break
            x0 = best_h.copy()
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 6000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 4: Structured multi-start
    if tl() > 100:
        xg = np.linspace(0, 2, n)
        perts = []
        for f in [2, 3, 4, 5, 8, 13]:
            perts.append(0.06 * np.sin(2 * np.pi * f * xg + np.random.uniform(0, 2 * np.pi)))
        for f in [2, 5, 9]:
            perts.append(0.06 * np.cos(2 * np.pi * f * xg + np.random.uniform(0, 2 * np.pi)))
        # Smooth random
        for _ in range(4):
            r = np.random.randn(n)
            w = np.fft.rfft(r)
            w[1:] *= 0.1
            sm = np.fft.irfft(w, n)
            sm -= sm.mean()
            sm /= (np.max(np.abs(sm)) + 1e-12)
            perts.append(0.04 * sm)

        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for p in perts:
            if tl() < 18:
                break
            x0 = project_feasible(best_h + p, target_sum)
            try:
                res = minimize(obj_smooth, x0, args=(dx, 8000), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 2500, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 5: Resolution sweep
    if tl() > 100:
        for n_try in [n - 10, n + 10, n - 5, n + 5, n + 20, n - 20, n + 30]:
            if n_try < 30 or n_try > 250 or tl() < 30:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = project_feasible(resample_h(best_h, n_try), ts_t)
            bt = Bounds(np.zeros(n_try), np.ones(n_try))
            ct = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t, n_try)}]
            for b in [3000, 15000]:
                if tl() < 15:
                    break
                try:
                    res = minimize(obj_smooth, h_t, args=(dx_t, b), method='SLSQP',
                                   bounds=bt, constraints=ct,
                                   options={'maxiter': 3000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new.copy(), c5
                        n, dx, target_sum = n_try, dx_t, ts_t
                        bounds = Bounds(np.zeros(n), np.ones(n))
                except Exception:
                    pass

    # Phase 6: Final active-set + SLSQP polish
    if tl() > 50:
        h6, c56 = active_set_subgrad(best_h, target_sum, t0, 20, 3, 0.001, 0.9, 50000)
        if c56 < best_c5:
            best_h, best_c5 = h6, c56

    if tl() > 15:
        n = len(best_h)
        dx = 2.0 / n
        target_sum = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        try:
            res = minimize(obj_smooth, best_h, args=(dx, 200000), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
