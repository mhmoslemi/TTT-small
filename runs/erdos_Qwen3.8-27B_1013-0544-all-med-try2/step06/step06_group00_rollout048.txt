        im = i + m_range; msk = (im >= 0) & (im < n); delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range; msk = (imn >= 0) & (imn < n); delta[msk] -= h[imn[msk]]
        jm = j + m_range; msk = (jm >= 0) & (jm < n); delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range; msk = (jmn >= 0) & (jmn < n); delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i); delta[k1] += 1.0
        k2 = n - 1 + (i - j); delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        accept = nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(scale * (cur_c - nc) / T))
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_list[ai] = j; ones_list[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 12
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick binary local search
    p1 = t0 + 0.04 * total
    hb, cb = binary_local_search(h0, n, p1, max_sweeps=120)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Projected gradient (fast, main workhorse)
    p2 = t0 + 0.50 * total
    alphas = [30, 80, 200, 500, 1000, 2000, 5000, 10000, 30000, 80000, 200000]
    h, c = pg_optimize(best_h, n, p2, alphas, max_iter=60000)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 3: Binary annealing for better discrete base
    p3 = t0 + 0.68 * total
    ha, ca = anneal_binary(best_h, n, p3, seed=seed)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()

    # Phase 4: Re-optimize from annealed + perturbations
    p4 = t0 + 0.90 * total
    alphas2 = [200, 1000, 5000, 20000, 80000, 200000]
    h, c = pg_optimize(best_h, n, p4, alphas2, max_iter=40000)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 5: Perturbation restarts for final polish
    p5 = dl - 2
    for trial in range(5):
        if time.time() > p5:
            break
        h = best_h + 0.003 * rng.randn(n)
        h = project_feasible(h, n)
        h, c = pg_optimize(h, n, p5, [50000, 200000, 500000], max_iter=20000)
        if c < best_c:
            best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
