def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Gradient of the lag-`lag` overlap w.r.t. h"""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        # f = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
        # d/dh[i] for i in [0, n-1-m]: (1 - h[i+m])
        # d/dh[i] for i in [m, n-1]: -h[i-m]
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    else:
        # For negative lag, by symmetry
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

def smooth_max_obj(h, n, alpha, top_k=10):

    # Use all lags but weight the top ones more
    idx = np.argsort(c)[-top_k:]
    active = c[idx]
    m = active.max()
    shifted = alpha * (active - m)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha, top_k=10):
    c = compute_all_lags(h, n)
    idx = np.argsort(c)[-top_k:]
    active = c[idx]
    m = active.max()
    weights = np.exp(alpha * (active - m))
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, k_idx in zip(weights, idx):
        lag = k_idx - (n - 1)
        grad += w * lag_gradient(h, n, lag)
    return grad

def subgradient_descent(h, n, deadline, max_iter=5000, lr0=0.01, decay=0.999):
    """Subgradient descent with backtracking"""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lr = lr0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        
        if lag == 0:
            grad = 1.0 - 2.0 * h_curr
        else:
            grad = lag_gradient(h_curr, n, lag)
        
        # Normalize gradient

        if gnorm < 1e-15:
            break
        grad_dir = -grad / gnorm
        
        # Step

        c_trial = compute_c5(h_trial, n)
        
        if c_trial < best_c:

            best_h = h_trial.copy()
            h_curr = h_trial
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.7

    return best_h, best_c

        return h.copy()

    cons = [cp.sum(h_var) == n/2.0, h_var >= 0, h_var <= 1,

Let me also add a key insight: since the problem is max of (convex) functions, and each individual lag function is NOT convex (it's quadratic with negative cross terms), the problem is genuinely non-convex. But the structure might allow us to find good solutions with the right approach.

One more idea: **alternating between reducing the max lag and maintaining feasibility**. At each step:
1. Find the worst lag k*
2. Take a step in the direction that reduces f_{k*} the most (while staying feasible)
3. This is basically Frank-Wolfe / conditional gradient adapted for this problem

The Frank-Wolfe direction would be: find h' in the feasible polytope {0≤h≤1, sum(h)=n/2} that minimizes the linearization of f_{k*} at the current point. This is:
min h'·g_subj_to {0≤h'≤1, sum(h')=n/2}
where g = gradient of f_{k*} at current h.

The optimal solution is: h'[j] = 1 if g[j] < λ, 0 if g[j] > λ, and the value that makes sum = n/2 if g[j] = λ, for some threshold λ.

This is a water-filling / threshold problem! Very efficient to solve.

Let me implement this:

```python
def frank_wolfe_step(h, n, shift):
    """One Frank-Wolfe step: find direction that minimizes linearization of f_shift"""
    g = lag_gradient(h, n, shift)
    # Solve: min g·h' s.t. 0≤h'≤1, sum(h')=n/2
    # Solution: h'[j] = 1 if g[j] < λ, 0 if g[j] > λ
    # Find λ such that sum = n/2
    sorted_g = np.sort(g)
    n_half = n / 2.0
    # If we set h'[j]=1 for all j where g[j] <= λ and 0 otherwise,
    # we need the count of "1"s to be n_half (if n_half is integer)
    # More precisely: count of g[j] < λ should be ≤ n_half ≤ count of g[j] ≤ λ
    # If n_half is integer: h'[j] = 1 for the n_half smallest g values
    # If n_half is not integer: fractional
    
    # For our case, n_half = n/2 which is integer (n is even)
    # So h'[j] = 1 for the n/2 indices with smallest g[j], 0 otherwise
    idx = np.argsort(g)
    h_lin = np.zeros(n)
    h_lin[idx[:n_half]] = 1.0
    return h_lin
```

Wait, but n/2 might not be integer if n is odd. Let me handle both cases.

Actually, the constraint is sum(h) = n/2. If n is even, n/2 is integer, and the FW vertex is binary. If n is odd, n/2 is half-integer, and we need one fractional component.

For the binary case: h'[j] = 1 for the n/2 indices with smallest g[j].

The FW step is then: h_new = h + α * (h_lin - h) for some step size α.

This is very efficient! And it naturally respects the box constraints.

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, shift):
    grad = np.zeros(n)
    m = abs(shift)
    if m >= n:
        return grad
    if m == 0:
        return 1.0 - 2.0 * h
    if shift > 0:
        grad[m:n] = 1.0
        grad[m:n] -= h[:n-m]
        grad[:n-m] -= h[m:]
    else:
        grad[:n-m] = 1.0
        grad[:n-m] -= h[m:]
        grad[m:n] -= h[:n-m]
    return grad

def fw_step(h, n, shift):
    """Frank-Wolfe linear minimization oracle"""
    g = lag_gradient(h, n, shift)
    n_half = n / 2.0
    idx = np.argsort(g)
    h_lin = np.zeros(n)
    k_full = int(n_half)
    frac = n_half - k_full
    h_lin[idx[:k_full]] = 1.0
    if frac > 1e-10:
        h_lin[idx[k_full]] = frac
    return h_lin

def fw_descent(h, n, deadline, max_iter=10000):
    """Frank-Wolfe descent for the max-lag problem"""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))
        shift = k_star - (n - 1)
        if shift == 0:
            # Special case: minimize sum h(1-h), which means make h more binary
            g = 1.0 - 2.0 * h_curr
        else:
            g = lag_gradient(h_curr, n, shift)
        
        h_lin = fw_step_from_grad(h_curr, n, g)
        # FW step size: 2/(t+2) or backtracking
        t = it + 1
        alpha = 2.0 / (t + 2.0)
        h_trial = project_feasible((1 - alpha) * h_curr + alpha * h_lin, n)
        c_trial = compute_c5(h_trial, n)
        
        if c_trial < best_c:
            best_c = c_trial
            best_h = h_trial.copy()
        h_curr = h_trial
        
        # Check convergence
        if it > 100 and (c[k_star] * (2.0/n) - best_c) < 1e-10:
            if it > 500:
                break
    
    return best_h, best_c

def fw_step_from_grad(h, n, g):
    n_half = n / 2.0
    idx = np.argsort(g)
    h_lin = np.zeros(n)
    k_full = int(n_half)
    frac = n_half - k_full
    h_lin[idx[:k_full]] = 1.0
    if frac > 1e-10:
        h_lin[idx[k_full]] = frac
    return h_lin

def slsqp_optimize(h, n, alpha, top_k, maxiter=3000, ftol=1e-14):
    def obj(x):
        c = compute_all_lags(x, n)
        idx = np.argsort(c)[-top_k:]
        active = c[idx]
        m = active.max()
        shifted = alpha * (active - m)
        if shifted.max() > 500:
            return float(m + (shifted.max() + np.log(np.sum(np.exp(shifted - shifted.max())))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def obj_grad(x):
        c = compute_all_lags(x, n)
        idx = np.argsort(c)[-top_k:]
        active = c[idx]
        m = active.max()
        weights = np.exp(alpha * (active - m))
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, ki in zip(weights, idx):
            shift = ki - (n - 1)
            grad += w * lag_gradient(x, n, shift)
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def active_set_lp(h_curr, n, n_active=30, trust=0.05):
    c = compute_all_lags(h_curr, n)
    top_idx = np.argsort(c)[-n_active:]
    dx = 2.0 / n
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n/2.0, h_var >= 0, h_var <= 1]
    # Trust region
    cons.append(cp.norm_1(h_var - h_curr) <= trust)
    
    for k_idx in top_idx:
        shift = k_idx - (n - 1)
        grad = lag_gradient(h_curr, n, shift)
        cons.append(c[k_idx] + grad @ (h_var - h_curr) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate'] and h_var.value is not None:
                return project_feasible(h_var.value, n)
        except:
            continue
    return h_curr.copy()

def targeted_reduction(h, n, deadline, max_iters=500):
    """Directly target reducing the worst lag using its gradient"""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))
        shift = k_star - (n - 1)
        
        if shift == 0:
            break  # Can't reduce lag 0 below 0 easily with sum constraint
        
        g = lag_gradient(h_curr, n, shift)
        
        # Try to move in -g direction while staying feasible
        # Use FW-like step
        h_lin = fw_step_from_grad(h_curr, n, g)
        
        for alpha in [0.5, 0.3, 0.1, 0.05, 0.02, 0.01]:
            h_trial = project_feasible((1-alpha)*h_curr + alpha*h_lin, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-15:
                best_c = c_trial
                best_h = h_trial.copy()
                h_curr = h_trial
                break
        else:
            # No improvement with any step size, try smaller
            h_curr = project_feasible(h_curr - 0.001 * g / (np.linalg.norm(g) + 1e-10), n)
            c_curr = compute_c5(h_curr, n)
            if c_curr < best_c:
                best_c = c_curr
                best_h = h_curr.copy()
            else:
                break
    
    return best_h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: FW descent (fast, many iterations)
    if time.time() < deadline - 20:
        fw_end = time.time() + min(40, (deadline - time.time()) * 0.35)
        h_fw, c_fw = fw_descent(best_h, n, fw_end, max_iter=20000)
        if c_fw < best_c:
            best_c = c_fw
            best_h = h_fw
    
    # Phase 2: Active set LP iterations
    if time.time() < deadline - 15:
        for _ in range(20):
            if time.time() > deadline - 15:
                break
            h_new = active_set_lp(best_h, n, n_active=30, trust=0.08)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new
            else:
                break
    
    # Phase 3: SLSQP with gradient
    if time.time() < deadline - 10:
        for alpha, tk in [(5000, 10), (20000, 20), (50000, 30)]:
            if time.time() > deadline - 10:
                break
            h_new = slsqp_optimize(best_h, n, alpha, tk, maxiter=2000, ftol=1e-14)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    
    # Phase 4: Targeted reduction of worst lag
    if time.time() < deadline - 5:
        h_new, c_new = targeted_reduction(best_h, n, min(deadline - 2, time.time() + 30))
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c
