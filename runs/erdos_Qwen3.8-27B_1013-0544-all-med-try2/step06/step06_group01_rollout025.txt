   import numpy as np
   from scipy.optimize import minimize
   import time

   def compute_c5(h, n):
       dx = 2.0 / n
       c = np.correlate(h, 1 - h, mode='full')
       return float(np.max(c) * dx)

   def project_feasible(h, n):
       target = n / 2.0
       h = np.clip(h, 0.0, 1.0)
       for _ in range(100):
           diff = target - h.sum()
           if abs(diff) < 1e-12:
               break
           if diff > 0:
               room = 1.0 - h
               s = room.sum()
               if s < 1e-14: break
               h += diff / s * room
           else:
               s = h.sum()
               if s < 1e-14: break
               h += diff / s * h
       return np.clip(h, 0.0, 1.0)

   def make_binary(h, n):
       target = n // 2
       idx = np.argsort(h)[::-1]
       b = np.zeros(n)
       b[idx[:target]] = 1.0
       return b

   def compute_delta(h, i, j, n):
       """Compute change in correlation when h[i]: 0→1 and h[j]: 1→0."""
       delta = np.zeros(2*n-1)
       # h[k-j] for k in [j, j+n-1]
       delta[j:j+n] += h
       # -h[k-i] for k in [i, i+n-1]
       delta[i:i+n] -= h
       # (1-h[k-i]) for k in [i, i+n-1]
       delta[i:i+n] += 1.0 - h
       # -(1-h[k-j]) for k in [j, j+n-1]
       delta[j:j+n] -= 1.0 - h
       # Cross terms
       delta[n-1] -= 2.0
       delta[n-1+j-i] += 1.0
       delta[n-1+i-j] += 1.0
       return delta

   def simulated_annealing(h, n, budget_s, seed=42):

           T = T_start * (T_end / T_start) ** progress

           sm = sh.max()
           return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
       def cons_fun(x):

                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
