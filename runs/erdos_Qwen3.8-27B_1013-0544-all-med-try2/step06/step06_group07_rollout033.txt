```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def subgrad(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

def fw_linear_min(g, target_sum):
    """Solve min <g, v> s.t. sum(v) = target_sum, 0 <= v <= 1"""
    n = len(g)
    # Sort indices by g value
    idx = np.argsort(g)
    v = np.zeros(n)
    # Set the smallest target_sum values to 1
    if target_sum == int(target_sum):
        k = int(target_sum)
        v[idx[:k]] = 1.0
    else:
        k = int(target_sum)
        v[idx[:k]] = 1.0
        v[idx[k]] = target_sum - k
    return v

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def fw_step(h, g, target_sum, gamma):
    """Regular FW step"""
    v = fw_linear_min(g, target_sum)
    return (1 - gamma) * h + gamma * v

def fw_away_step(h, g, target_sum, gamma):
    """FW with away steps"""
    v = fw_linear_min(g, target_sum)
    # Away step: try removing from the support
    # The active set of h is where h > 0
    active = np.where(h > 1e-10)[0]
    if len(active) == 0:
        return fw_step(h, g, target_sum, gamma)
    
    # FW step candidate
    h_fw = (1 - gamma) * h + gamma * v
    
    # Away step candidate: h_away = (1+gamma)*h - gamma * e_j where j is in active set
    # that maximizes <g, e_j> among active indices
    # Actually, the away step is: h_away = (1+gamma)*h - gamma * e_j
    # where j = argmax_{i in support(h)} g[i]
    j = active[np.argmax(g[active])]
    h_away = (1 + gamma) * h.copy()
    h_away[j] -= gamma
    
    # Check feasibility
    h_away = np.clip(h_away, 0, 1)
    h_away = project_feasible(h_away, target_sum)
    
    # Choose the better one based on <g, h>
    obj_fw = np.dot(g, h_fw)
    obj_away = np.dot(g, h_away)
    if obj_away < obj_fw:
        return h_away
    return h_fw
```

Let me think about the active-set approach more carefully. The idea is:
1. Compute corr = np.correlate(h, 1-h, mode='full')
2. Find all indices where corr is within some threshold of the max
3. Use smooth max only over those indices
4. Optimize with SLSQP
5. After optimization, check if any previously inactive index has become the new max
6. If yes, add it to the active set and re-optimize
7. Repeat until convergence

This is like a constraint generation / cutting plane method.

For the SLSQP objective, using only active indices makes the smooth max more accurate (less "leakage" from non-active indices) and potentially faster to optimize.

```python
def active_set_slsqp(h, dx, target_sum, bounds, max_iter=5000, threshold=None, beta_start=100):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    
    if threshold is None:
        threshold = 0.01 * dx  # 1% of dx below max
    
    active = np.where(corr >= max_corr - threshold)[0]
    
    for outer in range(20):
        # Optimize smooth max over active set
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        try:
            res = minimize(obj_smooth_active, h, args=(dx, beta_start, active),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-14})
            h = project_feasible(res.x, target_sum)
        except Exception:
            break
        
        # Check if active set needs updating
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        new_active = np.where(corr >= max_corr - threshold)[0]
        
        if np.array_equal(active, new_active):
            break
        active = new_active
    
    return h
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def subgrad(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

def fw_linear_min(g, target_sum):
    n = len(g)
    idx = np.argsort(g)
    v = np.zeros(n)
    k = int(target_sum)
    v[idx[:k]] = 1.0
    frac = target_sum - k
    if frac > 1e-12:
        v[idx[k]] = frac
    return v

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def obj_smooth_active(h, dx, beta, active):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr[active], beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def fw_with_away(h, n, target_sum, max_iters, time_limit_fn, use_greedy=False):
    best_local = compute_c5(h)
    stall = 0
    for it in range(max_iters):
        if time_limit_fn() <= 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_worst = np.argmax(corr)
        g = subgrad(h, k_worst, n)
        v = fw_linear_min(g, target_sum)
        gamma = 2.0 / (it + 2)
        h_new = (1 - gamma) * h + gamma * v
        active = np.where(h > 1e-10)[0]
        if len(active) > 0:
            j = active[np.argmax(g[active])]
            h_away = (1 + gamma) * h.copy()
            h_away[j] -= gamma
            h_away = project_feasible(h_away, target_sum)
            if np.dot(g, h_away) < np.dot(g, h_new):
                h_new = h_away
        h = h_new
        if it % 3000 == 2999:
            c = compute_c5(h)
            if c < best_local - 1e-8:
                best_local = c
                stall = 0
            else:
                stall += 1
            if stall > 4:
                return h, best_local
    c = compute_c5(h)
    return h, min(c, best_local)

def active_set_optimize(h, n, dx, target_sum, bounds, budget_fn, max_outer=8):
    best_c5 = compute_c5(h)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    threshold = 0.005 * dx
    active = np.where(corr >= max_corr - threshold)[0]
    for outer in range(max_outer):
        if budget_fn() <= 10:
            break
        for beta in [200, 2000]:
            if budget_fn() <= 5:
                break
            try:
                res = minimize(obj_smooth_active, h, args=(dx, beta, active),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 2000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    h = h_new
                    best_c5 = c5
            except Exception:
                pass
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        new_active = np.where(corr >= max_corr - threshold)[0]
        if len(new_active) == len(active) and np.all(new_active == active):
            break
        active = new_active
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: FW with away steps (primary global search)
    if tl() > 200:
        h = best_h.copy()
        for restart in range(5):
            if tl() < 60:
                break
            # Perturb on subsequent restarts
            if restart > 0:
                h = best_h.copy()
                h += np.random.randn(n) * (0.01 / restart)
                h = project_feasible(h, target_sum)
            h, c5 = fw_with_away(h, n, target_sum, 50000, tl)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    # Phase 2: Active-set SLSQP refinement
    if tl() > 80:
        best_h, best_c5 = active_set_optimize(
            best_h, n, dx, target_sum, bounds, tl, max_outer=10)

    # Phase 3: FW again after SLSQP
    if tl() > 60:
        h, c5 = fw_with_away(best_h.copy(), n, target_sum, 80000, tl)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 4: Multi-resolution search
    if tl() > 100:
        for n_try in [n + 10, n - 10, n + 20, n - 20, n + 30, n - 30, 50, 100]:
            if n_try < 30 or n_try > 200 or tl() < 30:
                continue
            ts_t = n_try / 2.0
            dx_t = 2.0 / n_try
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            # FW on new resolution
            h_f, c5_f = fw_with_away(h_t, n_try, ts_t, 30000, tl)
            if c5_f < best_c5:
                best_h = h_f.copy()
                best_c5 = c5_f
                n = n_try
                dx = dx_t
                target_sum = ts_t
                bounds = bnds_t
            # SLSQP polish on this resolution
            if tl() > 25:
                cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t, n_try)}]
                for beta in [500, 5000]:
                    if tl() < 10:
                        break
                    try:
                        res = minimize(obj_smooth, h_f, args=(dx_t, beta), method='SLSQP',
                                      bounds=bnds_t, constraints=cons_t,
                                      options={'maxiter': 1500, 'ftol': 1e-14})
                        h_new = project_feasible(res.x, ts_t)
                        c5 = compute_c5(h_new)
                        if c5 < best_c5:
                            best_h = h_new.copy()
                            best_c5 = c5
                            n = n_try
                            dx = dx_t
                            target_sum = ts_t
                            bounds = bnds_t
                    except Exception:
                        pass

    # Phase 5: Final active-set SLSQP with high beta
    if tl() > 20:
        best_h, best_c5 = active_set_optimize(
            best_h, n, dx, target_sum, bounds, tl, max_outer=5)

    # Phase 6: Final high-beta SLSQP polish
    if tl() > 10:
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for beta in [10000, 50000, 200000]:
            if tl() < 5:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 7: Very short final FW
    if tl() > 5:
        h, c5 = fw_with_away(best_h.copy(), n, target_sum, 30000, tl)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Final validation
    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
