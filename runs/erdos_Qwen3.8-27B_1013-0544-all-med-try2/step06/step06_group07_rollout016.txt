```python
from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum):
    return h.sum() - target_sum

def con_jac(h, target_sum):
    return np.ones(len(h))

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

        i = ones[np.random.randint(len(ones))]
```

```python
def frank_wolfe_step(h, grad, target_sum, n):
    """Find the vertex of the feasible polytope that minimizes grad·z."""
    # min grad·z s.t. sum(z) = target_sum, 0 <= z <= 1
    # Sort by grad values
    order = np.argsort(grad)
    z = np.zeros(n)
    # Set the smallest grad entries to 1, largest to 0
    # We need exactly target_sum ones (if target_sum is integer)
    # For target_sum = n/2 (which may not be integer for odd n)
    # For even n: set n/2 entries to 1
    # For odd n: set (n-1)/2 entries to 1 and one entry to 0.5
    if target_sum == int(target_sum):
        z[order[:int(target_sum)]] = 1.0
    else:
        k = int(target_sum)
        z[order[:k]] = 1.0
        z[order[k]] = target_sum - k
    return z
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def con_fun(h, target_sum):
    return h.sum() - target_sum


def con_jac(h, target_sum):
    return np.ones(len(h))


def subgrad_at(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def fix_binary_count(h, tgt):
    h = h.copy()
    cnt = int(h.sum())
    if cnt > tgt:
        idx = np.where(h > 0.5)[0]
        np.random.shuffle(idx)
        for i in idx[:cnt - tgt]:
            h[i] = 0.0
    elif cnt < tgt:
        idx = np.where(h < 0.5)[0]
        np.random.shuffle(idx)
        for i in idx[:tgt - cnt]:
            h[i] = 1.0
    return h


def binary_sa(h_bin, n, timeout):
    h = h_bin.copy()
    best = h.copy()
    corr_arr = np.correlate(h, 1.0 - h, 'full')
    best_corr = float(np.max(corr_arr))
    T = 0.5
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    it = 0
    while time.time() < timeout:
        it += 1
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        corr_arr = np.correlate(h, 1.0 - h, 'full')
        corr = float(np.max(corr_arr))
        if corr < best_corr:
            best_corr = corr
            best = h.copy()
            T = max(0.01, T * 0.9997)
        else:
            if np.random.random() < np.exp(-(corr - best_corr) / max(T, 1e-8)):
                T = max(0.01, T * 0.9997)
            else:
                h[i], h[j] = 1.0, 0.0
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
    return best, best_corr


def slsqp_solve(h0, dx, target_sum, beta, maxiter, ftol, timeout):
    n = len(h0)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
    try:
        res = minimize(obj_smooth, h0, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h0


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP continuation
    for beta in [10, 50, 200, 800, 3000]:
        if tl() < 50:
            break
        for trial in range(2):
            if tl() < 25:
                break
            x0 = best_h + (np.random.randn(n) * 0.025 if trial > 0 else 0)
            x0 = project_feasible(x0, target_sum)
            h_new = slsqp_solve(x0, dx, target_sum, beta, 6000, 1e-15, tl())
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    # Phase 2: Binary SA from best continuous
    if tl() > 70:
        h_bin = fix_binary_count((best_h >= 0.5).astype(float), n // 2)
        h_sa, sa_corr = binary_sa(h_bin, n, time.time() + min(35, tl() * 0.25))
        c5_sa = sa_corr * dx
        if c5_sa < best_c5:
            best_h, best_c5 = h_sa.copy(), c5_sa
        # Refine binary with SLSQP
        if tl() > 50:
            for beta in [500, 3000, 10000]:
                if tl() < 20:
                    break
                h_new = slsqp_solve(h_sa.copy(), dx, target_sum, beta, 6000, 1e-15, tl())
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5

    # Phase 3: Subgradient with Nesterov momentum
    if tl() > 50:
        h = best_h.copy()
        v = np.zeros(n)
        mu = 0.85
        for it in range(300000):
            if tl() < 15:
                break
            corr_arr = np.correlate(h, 1.0 - h, mode='full')
            k_worst = int(np.argmax(corr_arr))
            g = subgrad_at(h, k_worst, n)
            lr = 0.004 / (1.0 + it * 0.0004)
            v = mu * v - lr * g
            h = project_feasible(h + v, target_sum)
            if it % 15000 == 14999:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h, best_c5 = h.copy(), c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: Multi-resolution search
    if tl() > 100:
        cur_n = len(best_h)
        for delta in [4, -4, 8, -8, 12, 16, -12, 20]:
            n_try = cur_n + delta
            if n_try < 40 or n_try % 2 != 0 or tl() < 45:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = project_feasible(resample_h(best_h, n_try), ts_t)
            for beta in [300, 2000]:
                if tl() < 22:
                    break
                h_new = slsqp_solve(h_t, dx_t, ts_t, beta, 4000, 1e-14, tl())
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5, cur_n = h_new, c5, n_try

    # Phase 5: More binary SA with different thresholds
    if tl() > 60:
        cur_n = len(best_h)
        cur_dx = 2.0 / cur_n
        cur_ts = cur_n / 2.0
        for thresh in [0.3, 0.4, 0.6, 0.7]:
            if tl() < 30:
                break
            h_bin = fix_binary_count((best_h >= thresh).astype(float), cur_n // 2)
            h_sa, sa_corr = binary_sa(h_bin, cur_n, time.time() + min(15, tl() * 0.3))
            c5_sa = sa_corr * cur_dx
            if c5_sa < best_c5:
                best_h, best_c5 = h_sa.copy(), c5_sa

    # Phase 6: Final high-beta polish
    if tl() > 12:
        n = len(best_h)
        dx = 2.0 / n
        target_sum = n / 2.0
        for beta in [8000, 30000]:
            if tl() < 6:
                break
            h_new = slsqp_solve(best_h.copy(), dx, target_sum, beta, 10000, 1e-16, tl())
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    return best_h, best_c5, len(best_h)
```
