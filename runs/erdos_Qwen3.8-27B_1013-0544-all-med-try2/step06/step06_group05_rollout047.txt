    h = np.clip(h.copy(), 0.0, 1.0)

            if tot < 1e-14:

                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_single(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        remaining = deadline - time.time() - 25
        n_restarts = max(2, int(remaining / 40))
        scales = [0.003, 0.01, 0.02, 0.04, 0.06]
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(35, (deadline - time.time() - 20) * 0.4)
            if alloc < 12:
                break
            h_opt, c_opt = optimize_single(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final polish
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        best_h, best_c = gradient_polish(best_h, n, deadline - 8)
        best_h, best_c = pair_search(best_h, n, deadline - 1,
                                     [0.0002, 0.0001, 0.00005, 0.00002, 0.00001], max_passes=500)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
