def compute_gradient(h, n, alpha):

    c = np.correlate(h, 1 - h, mode='full')
    log_w = alpha * c * dx
    log_w -= log_w.max()
    w = np.exp(log_w)
    w /= w.sum()
    
    w_rev = w[::-1].copy()
    A = fftconvolve(w_rev, h, mode='full')[n-1 : 2*n-1]
    B = fftconvolve(w, h, mode='full')[n-1 : 2*n-1]

        if s > target:

        else:
            hi = mid

def fista(h0, n, alpha, max_iter, deadline, lr=None):
    h = h0.copy()
    prev_h = h0.copy()
    t = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    if lr is None:
        lr = 0.5 / alpha  # heuristic initial step size
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Nesterov momentum
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y = h + (t - 1) / t_new * (h - prev_h)
        
        # Gradient at y (projected to feasible set for gradient computation)
        y_clipped = np.clip(y, 0, 1)
        y_proj = project_sum_box(y_clipped, n)
        g = compute_gradient(y_proj, n, alpha)
        
        # Gradient step
        z = h - lr * g
        
        # Project
        h_new = project_sum_box(z, n)
        
        # Check improvement
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        
        prev_h = h
        h = h_new
        t = t_new
        
        # Adaptive step size: decrease if not improving
        # (simple approach: decay lr)
        lr *= 0.999
    
    return best_h, best_c
