def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)

        lags = [k_worst]
        wts = [1.0]

            grad[idx] += w * (1.0 - h[idx + lag])
            grad[idx + lag] -= w * h[idx]

            idx = np.arange(n - la)
            grad[idx + la] += w * (1.0 - h[idx])
            grad[idx] -= w * h[idx + la]

            if lag >= 0:
                idx = np.arange(n - lag)
                grad[idx] += w * (1.0 - h[idx + lag])
                grad[idx + lag] -= w * h[idx]
            else:
                la = -lag
                idx = np.arange(n - la)
                grad[idx + la] += w * (1.0 - h[idx])
                grad[idx] -= w * h[idx + la]
    return grad, corr


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def con_fun(h, target_sum, n):
    return h.sum() - target_sum


def con_jac(h, target_sum, n):
    return np.ones(n)


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def coord_descent(h, target_sum, n, max_rounds=50):
    """Try coordinate-wise perturbations to reduce max overlap."""
    best = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            for delta in [0.01, 0.005, -0.01, -0.005, 0.02, -0.02]:
                h2 = best.copy()
                h2[i] = np.clip(h2[i] + delta, 0.0, 1.0)
                h2 = project_feasible(h2, target_sum)
                c5 = compute_c5(h2)
                if c5 < best_c5 - 1e-12:
                    best = h2.copy()
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5


def slsqp_optimize(h0, dx, beta, bounds, cons, maxiter=5000, ftol=1e-15):
    try:
        res = minimize(obj_smooth, h0, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def remain():
        return budget_s - elapsed()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]

    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation
    for beta in [50, 200, 1000, 5000, 20000]:
        if remain() < 30:
            break
        x_opt = slsqp_optimize(best_h, dx, beta, bounds, cons)
        h_new = project_feasible(x_opt, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5

    # Phase 2: Top-k subgradient with adaptive parameters
    if remain() > 120:
        h = best_h.copy()
        for it in range(300000):
            if remain() < 60:
                break
            # Increase top_k as we get closer
            gap = best_c5 - compute_c5(h)
            top_k = 1 if gap > 0.002 else (2 if gap > 0.0005 else 4)
            soft_param = 5.0 + it * 0.005
            grad, corr = subgrad_topk(h, n, top_k, soft_param)
            lr = 0.004 / (1.0 + it * 0.002)**0.7
            h = h - lr * grad
            h = project_feasible(h, target_sum)
            if it % 3000 == 2999:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 3: SLSQP polish from subgradient result
    for beta in [5000, 20000, 100000]:
        if remain() < 25:
            break
        for trial in range(3):
            if remain() < 12:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += np.random.randn(n) * (0.002 / trial)
                x0 = project_feasible(x0, target_sum)
            x_opt = slsqp_optimize(x0, dx, beta, bounds, cons, maxiter=8000, ftol=1e-16)
            h_new = project_feasible(x_opt, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5

    # Phase 4: Coordinate descent refinement
    if remain() > 60:
        h_cd, c5_cd = coord_descent(best_h, target_sum, n, max_rounds=30)
        if c5_cd < best_c5:
            best_h = h_cd.copy()
            best_c5 = c5_cd
        # SLSQP from coordinate descent result
        if remain() > 20:
            x_opt = slsqp_optimize(best_h, dx, 50000, bounds, cons, maxiter=10000, ftol=1e-16)
            h_new = project_feasible(x_opt, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5

    # Phase 5: Try adjacent resolutions
    if remain() > 80:
        for n_try in [n + 5, n - 5, n + 10, n - 10, n + 15, n + 20, n - 20, n + 30]:
            if n_try < 40 or remain() < 35:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t, n_try)}]
            for beta in [2000, 20000]:
                if remain() < 15:
                    break
                x_opt = slsqp_optimize(h_t, dx_t, beta, bnds_t, cons_t, maxiter=4000, ftol=1e-15)
                h_new = project_feasible(x_opt, ts_t)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5

    # Phase 6: Final aggressive polish with tiny perturbations
    if remain() > 30:
        for attempt in range(20):
            if remain() < 10:
                break
            x0 = best_h.copy()
            x0 += np.random.randn(n) * 0.001
            x0 = project_feasible(x0, target_sum)
            x_opt = slsqp_optimize(x0, dx, 100000, bounds, cons, maxiter=6000, ftol=1e-16)
            h_new = project_feasible(x_opt, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5

    # Phase 7: One last shot with very high beta
    if remain() > 10:
        x_opt = slsqp_optimize(best_h, dx, 500000, bounds, cons, maxiter=15000, ftol=1e-17)
        h_new = project_feasible(x_opt, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5

    return best_h, best_c5, len(best_h)
