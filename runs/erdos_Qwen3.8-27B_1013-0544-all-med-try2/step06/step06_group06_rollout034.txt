import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def fast_analytical_grad(h, n, k_idx):
    """Analytical gradient of c[k_idx] from np.correlate(h, 1-h, 'full')."""
    # c[m] = sum_i h[i] * (1-h[i+m-n+1]) for valid indices
    # Let shift = k_idx - (n-1)
    # c[k_idx] = sum_i h[i] * (1 - h[i + shift])
    # d/dh[j] = (1-h[j+shift]) * [valid] - h[j-shift] * [valid]
    shift = k_idx - (n - 1)
    dx_scale = 2.0 / n
    grad = np.zeros(n)
    
    # (1-h[j+shift]) for valid j+shift
    if shift >= 0:
        grad[:n-shift] += (1 - h[shift:])
    else:
        grad[-shift:] += (1 - h[:n+shift])
    
    # -h[j-shift] for valid j-shift
    if shift >= 0:
        grad[shift:] -= h[:n-shift]
    else:
        grad[:n+shift] -= h[-shift:]
    
    return grad * dx_scale

def subgrad_fast(h, n, deadline, max_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5
    for it in range(max_iter):
        if time.time() > deadline:
            break
        ov = np.correlate(h, 1-h, 'full') * (2.0/n)
        m = ov.max()
        k_max = int(np.argmax(ov))
        grad = fast_analytical_grad(h, n, k_max)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                h = ht
                found = True
                break
        if found:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            lr *= 0.95
            if lr < 1e-12:
                break
    return best_h, best_c

def binary_ls_fast(h, n, deadline, max_sweeps=200, rng=None):

    if rng is None:

    target = n // 2
    idx = np.argsort(h)[::-1]

    h_bin[idx[:target]] = 1.0
    
    overlap = np.correlate(h_bin, 1 - h_bin, 'full')  # raw integer values

    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    ks = np.arange(-(n-1), n)  # shifts corresponding to correlate indices
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros_list = list(zeros)
        ones_list = list(ones)
        rng.shuffle(zeros_list)
        rng.shuffle(ones_list)
        
        for i in zeros_list:
            if time.time() > deadline:
                break

            valid_ik = (ik >= 0) & (ik < n)

            base_i = np.zeros(2*n-1)

def slsqp_smooth(h0, n, deadline, alpha, maxiter=500):
    """SLSQP with smooth max approximation."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1-x, 'full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        logsum = sm + np.log(np.exp(sh - sm).sum())
        return m + logsum / alpha
    
    def cons(x):
        return x.sum() - n/2.0
    
    try:
        res = minimize(obj, h0, method='SLSQP',
                      bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun':cons}],
                      options={'maxiter':maxiter, 'ftol':1e-15})
        return project_feasible(res.x, n)
    except:
        return h0

                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary local search (25%)
    p1 = t0 + 0.25 * (budget_s - 5)
    h, c = binary_ls(best_h, n, p1, max_sweeps=1000, rng=rng)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 2: Simulated annealing (25%)
    p2 = t0 + 0.50 * (budget_s - 5)
    h, c = anneal(best_h, n, p2, rng, max_iters=200000)
    if c < best_c:
        best_c, best_h = c, h.copy()
    h, c = binary_ls(best_h, n, p2 + 20, max_sweeps=300, rng=rng)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 3: Second anneal from perturbed start (15%)
    p3 = t0 + 0.65 * (budget_s - 5)
    hp = project_feasible(best_h + 0.3 * rng.randn(n), n)
    h, c = anneal(hp, n, p3, rng, max_iters=100000)
    if c < best_c:
        best_c, best_h = c, h.copy()
    h, c = binary_ls(best_h, n, p3 + 10, max_sweeps=200, rng=rng)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 4: Continuous subgradient (15%)
    p4 = t0 + 0.80 * (budget_s - 5)
    hc = best_h.copy()
    hc, c = subgrad_fast(hc, n, p4, max_iter=10000)
    if c < best_c:
        best_c, best_h = c, hc.copy()

    # Phase 5: SLSQP polish (10%)
    p5 = t0 + 0.90 * (budget_s - 5)
    for alpha in [500, 2000, 10000, 50000]:
        if time.time() > p5 - 5:
            break
        hn = slsqp_smooth(best_h, n, p5 - 2, alpha, maxiter=200)
        cn = compute_c5(hn, n)
        if cn < best_c:
            best_c, best_h = cn, hn.copy()

    # Phase 6: Final binary polish + subgradient
    if time.time() < dl - 10:
        h, c = binary_ls(best_h, n, dl - 10, max_sweeps=200, rng=rng)
        if c < best_c:
            best_c, best_h = c, h.copy()
    if time.time() < dl - 3:
        hc, c = subgrad_fast(best_h, n, dl, max_iter=3000)
        if c < best_c:
            best_c, best_h = c, hc.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
