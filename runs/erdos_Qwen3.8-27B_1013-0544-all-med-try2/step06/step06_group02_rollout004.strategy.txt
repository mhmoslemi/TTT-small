def compute_overlap(h, n):

    h2=np.roll(h,1)
    # Compute integrated correlation
    overlap = 0
    for i in range(2*n-1):
        inner = 0

def optimize_periodic(n, phases=[0.1,0.3,0.5,0.7], freq_range=[1,5], budget=1100):
    best_config = {'h': None, 'score': float('inf')}
    for phase in phases:

            h_init = np.sign(np.sin(np.arange(n)*2*np.pi*freq + phase))
            h_init = (1+h_init)/2
            h_init *= n/2/h_init.sum()

                    best_config['h']=res.x
