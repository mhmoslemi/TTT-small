```python
    # where s = k - n + 1
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # Vectorize: for each shift s, compute contribution to all j
    for k in range(2 * n - 1):
        s = k - n + 1
        wk = weights[k]
        if wk < 1e-14:
            continue
        if s >= 0:
            # j from 0 to n-1-s: contributes (1-h[j+s])
            j1 = np.arange(n - s)
            grad[j1] += wk * dx * (1.0 - h[j1 + s])
            # j from s to n-1: contributes -h[j-s]
            j2 = np.arange(s, n)
            grad[j2] -= wk * dx * h[j2 - s]
        else:
            t = -s
            # j from t to n-1: contributes (1-h[j+t])
            j1 = np.arange(t, n)
            grad[j1] += wk * dx * (1.0 - h[j1 + t])
            # j from 0 to n-1-t: contributes -h[j+t]
            j2 = np.arange(n - t)
            grad[j2] -= wk * dx * h[j2 + t]
    
    return grad, m
```

Let me look at this from a different angle. The problem is:
minimize max_s g_s subject to 0<=h<=1, sum(h)=n/2

At the optimum, by KKT conditions, we should have:
- If h[j] is in (0,1): the gradient of the active constraint is zero
- The active constraints are those shifts s where g_s = max

So at the optimum, a subset of shifts are "active" (achieving the max), and the gradient of their weighted combination is zero (for free variables).

This suggests an algorithm:
1. Identify the active set S (shifts achieving max)
2. Minimize the average of g_s for s in S (or just the max of those in S)
3. Check if any other shift now exceeds the max
4. If so, add it to S and repeat
5. When no change, we're at a local optimum

This is essentially what the smoothed max approach does, but with a sharper focus on the truly active shifts.

```python
import numpy as np
from scipy.optimize import minimize
import time

# Helper functions

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_g(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    """Gradient of (1/alpha)*log(sum(exp(alpha*g)))"""
    g = compute_g(h, n)
    m = g.max()
    ex = np.exp(alpha * (g - m))
    w_sum = ex.sum()
    weights = ex / w_sum
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - n + 1
        wk = weights[k]
        if wk < 1e-13:
            continue
        if s >= 0:
            j1 = np.arange(n - s)
            grad[j1] += wk * dx * (1.0 - h[j1 + s])
            j2 = np.arange(s, n)
            grad[j2] -= wk * dx * h[j2 - s]
        else:
            t = -s
            j1 = np.arange(t, n)
            grad[j1] += wk * dx * (1.0 - h[j1 + t])
            j2 = np.arange(n - t)
            grad[j2] -= wk * dx * h[j2 + t]
    return grad

def slsqp_step(h, n, alpha, maxiter, deadline):
    """One SLSQP call with softmax-smoothed max objective"""
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        return val
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pgd_refine(h, n, deadline, alpha_start=200, alpha_end=500000, max_iter=3000):
    """Projected gradient descent with momentum and increasing alpha"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    beta = 0.5
    lr = 0.05
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t = min(it / max_iter, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        
        grad = smooth_max_grad(h, n, alpha)
        v = beta * v + (1.0 - beta) * grad
        
        # Backtracking
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = lr * f
            h_new = project_feasible(h - step * v, n)
            c_new = compute_c5(h_new, n)
            c_cur = compute_c5(h, n)
            if c_new < c_cur - 1e-13:
                h = h_new
                lr = max(lr * 1.1, 0.5)
                found = True
                break
            else:
                h_new = h  # undo (wasn't actually applied)
        
        if not found:
            lr *= 0.9
            beta = max(beta * 0.99, 0.1)
            if lr < 1e-7:
                break
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
```

```python
def eval_swap(h, n, i, j, g):
    """Evaluate the change in g when h[i]:0->1, h[j]:1->0. O(n)"""
    # h is binary, g = compute_g(h, n) (already includes dx)
    dx = 2.0 / n
    s = np.arange(-(n-1), n)  # all shifts
    
    valid_ip = (i + s >= 0) & (i + s < n)
    valid_im = (i - s >= 0) & (i - s < n)
    valid_jp = (j + s >= 0) & (j + s < n)
    valid_jm = (j - s >= 0) & (j - s < n)
    
    # Base change
    delta = np.zeros(2*n-1)
    # (1-h[i+s]) for valid i+s
    idx_ip = i + s[valid_ip]
    delta[valid_ip] += (1.0 - h[idx_ip])
    # -h[i-s] for valid i-s
    idx_im = i - s[valid_im]
    delta[valid_im] -= h[idx_im]
    # -(1-h[j+s]) for valid j+s
    idx_jp = j + s[valid_jp]
    delta[valid_jp] -= (1.0 - h[idx_jp])
    # +h[j-s] for valid j-s
    idx_jm = j - s[valid_jm]
    delta[valid_jm] += h[idx_jm]
    
    # Correction at s = j-i
    s_corr = j - i
    if -(n-1) <= s_corr <= (n-1):
        delta[s_corr + n - 1] += 2.0
    
    return dx * delta
```
