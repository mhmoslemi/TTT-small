    return np.correlate(h, h, mode='full') * dx

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14: break
        if d > 0:
            r = 1.0 - h; s = r.sum()
            if s < 1e-14: break
            h += d / s * r
        else:
            s = h.sum()
            if s < 1e-14: break
            h += d / s * h
    return np.clip(h, 0.0, 1.0)

            g[s:] += h[:n-s]

import numpy as np

from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    return np.correlate(h, h, mode='full') * dx

def project(h, n):
    target = n / 2.0

    for _ in range(500):

            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14:
                break
            h = h + d / s * r
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + d / s * h
    return np.clip(h, 0.0, 1.0)

def grad_R_shift(h, s, dx):
    n = len(h)

    a = abs(s)
    if a >= n:
        return g
    if s >= 0:
        g[:n-a] += h[a:]
        g[a:] += h[:n-a]
    else:
        g[a:] += h[:n-a]

    return g * dx
