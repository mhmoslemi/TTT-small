    rng = np.random.RandomState(seed)

        # c(k) changes by delta * g(k) where:
        # g(k) = [(1-h[i+l]) * 1(0<=i+l<n) + h[j-l] * 1(0<=j-l<n)] - [(1-h[j+l]) * 1(0<=j+l<n) + h[i-l] * 1(0<=i-l<n)]

        # l ranges from -(n-1) to (n-1)
        l = np.arange(-(n-1), n)  # length 2n-1
        
        # term1: (1-h[i+l]) where 0 <= i+l < n
        idx1 = i + l
        m1 = (idx1 >= 0) & (idx1 < n)
        t1 = np.zeros(2*n-1)
        t1[m1] = 1.0 - h[idx1[m1]]
        
        # term2: h[j-l] where 0 <= j-l < n
        idx2 = j - l
        m2 = (idx2 >= 0) & (idx2 < n)
        t2 = np.zeros(2*n-1)
        t2[m2] = h[idx2[m2]]
        
        # term3: (1-h[j+l]) where 0 <= j+l < n
        idx3 = j + l
        m3 = (idx3 >= 0) & (idx3 < n)
        t3 = np.zeros(2*n-1)
        t3[m3] = 1.0 - h[idx3[m3]]
        
        # term4: h[i-l] where 0 <= i-l < n
        idx4 = i - l
        m4 = (idx4 >= 0) & (idx4 < n)
        t4 = np.zeros(2*n-1)
        t4[m4] = h[idx4[m4]]
        
        g = t1 + t2 - t3 - t4
        
        # Current correlation values
        c = np.correlate(h, 1 - h, mode='full')  # length 2n-1
        
        # We want to find delta that minimizes max_k (c[k] + delta * g[k])
        # subject to 0 <= h[i]+delta <= 1 and 0 <= h[j]-delta <= 1
        # i.e., -h[i] <= delta <= 1-h[i] and h[j]-1 <= delta <= h[j]
        dmin = max(-h[i], h[j] - 1.0)
        dmax = min(1.0 - h[i], h[j])
        
        if dmin >= dmax:
            continue
            
        # Find optimal delta: minimize max(c + delta*g) over delta in [dmin, dmax]
        # This is a convex function of delta. Use ternary search.
        # Or: the minimum is where the maximum of the lines c[k] + delta*g[k] is minimized.
        # At the optimum, at least two lines are tied at the maximum (unless at a boundary).
        
        # Simple approach: sample delta values
        # Better: use the fact that it's piecewise linear convex, so the minimum is at a "kink"
        # The kinks occur where c[k1] + delta*g[k1] = c[k2] + delta*g[k2]
        # i.e., delta = (c[k2] - c[k1]) / (g[k1] - g[k2])
        
        # For efficiency, just use a few trial values + boundaries
        candidates = [dmin, dmax, 0.0]
        # Add some quantiles
        candidates += list(np.quantile(np.concatenate([
            (c[n-1] - c[:n-1]) / np.where(g[:n-1] != g[n-1], g[:n-1] - g[n-1], 1e-10),
            (c[n+1:] - c[n:]) / np.where(g[n:] != g[n+1:], g[n:] - g[n+1:], 1e-10)
        ]), [0.1, 0.25, 0.5, 0.75, 0.9]))
        
        best_delta = 0.0
        best_val = np.inf
        for d in candidates:
            d = np.clip(d, dmin, dmax)
            val = np.max(c + d * g)
            if val < best_val:
                best_val = val
                best_delta = d
        
        # Verify improvement
        new_c_val = best_val * dx
        if new_c_val < cur_c - 1e-15:
            h[i] += best_delta
            h[j] -= best_delta
            cur_c = new_c_val
    
    return h, cur_c

def smooth_obj(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha + lam * (x.sum() - n/2.0)**2

        # +h[j-l] for valid: j-l in [0,n-1], i.e., l in [j-(n-1), j]
        lo = max(-(nm1), j - (n-1))

    rng = np.random.RandomState(seed)
    dx = 2.0 / n

    k_star = np.argmax(c)
    l_star = k_star - (n - 1)

    # sg[i] = (1-h[i+l_star]) * 1(0<=i+l_star<n) - h[i-l_star] * 1(0<=i-l_star<n)
    sg = np.zeros(n)
    valid1 = np.arange(-l_star, n - l_star)  # i such that i+l_star in [0,n-1]
    valid1 = valid1[(valid1 >= 0) & (valid1 < n)]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha + lam * (x.sum() - n / 2.0) ** 2

def lbfgsb_step(h, n, alpha, lam, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, method='L-BFGS-B', args=(n, alpha, lam),
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pairwise_cd(h, n, n_pairs, seed, structured=False):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full')
    cur_max = c.max() * dx
    nm1 = n - 1
    total_c = c.sum()

    for it in range(n_pairs):
        if structured and it < n:
            i = it
            j = (it + 1 + it * 7) % n
            if j == i:
                j = (j + 1) % n
        else:
            i, j = rng.choice(n, 2, replace=False)

        g = np.zeros(2 * n - 1)
        lo = max(-nm1, -i); hi = min(nm1, n - 1 - i)
        if lo <= hi:
            g[lo + nm1:hi + nm1 + 1] += 1.0 - h[i + np.arange(lo, hi + 1)]
        lo = max(-nm1, i - nm1); hi = min(nm1, i)
        if lo <= hi:
            g[lo + nm1:hi + nm1 + 1] -= h[i - np.arange(lo, hi + 1)]
        lo = max(-nm1, -j); hi = min(nm1, n - 1 - j)
        if lo <= hi:
            g[lo + nm1:hi + nm1 + 1] -= 1.0 - h[j + np.arange(lo, hi + 1)]
        lo = max(-nm1, j - nm1); hi = min(nm1, j)
        if lo <= hi:
            g[lo + nm1:hi + nm1 + 1] += h[j - np.arange(lo, hi + 1)]

        dmin = max(-h[i], h[j] - 1.0)
        dmax = min(1.0 - h[i], h[j])
        if dmin >= dmax - 1e-15:
            continue

        top_idx = np.argsort(c)[::-1][:12]
        cands = [dmin, dmax, 0.0]
        for a in range(len(top_idx)):
            for b in range(a + 1, len(top_idx)):
                ka, kb = top_idx[a], top_idx[b]
                dg = g[ka] - g[kb]
                if abs(dg) > 1e-12:
                    d = (c[kb] - c[ka]) / dg
                    if dmin < d < dmax:
                        cands.append(d)

        best_d = 0.0
        best_v = np.inf
        for d in cands:
            d = float(np.clip(d, dmin, dmax))
            v = np.max(c + d * g)
            if v < best_v:
                best_v = v
                best_d = d

        if best_v * dx < cur_max - 1e-14:
            h[i] += best_d
            h[j] -= best_d
            c += best_d * g
            cur_max = c.max() * dx
            total_c += best_d * g.sum()
            if total_c < -1e-6:
                c = np.correlate(h, 1 - h, mode='full')
                cur_max = c.max() * dx
                total_c = c.sum()

    return h, cur_max

def make_initial(n, seed, typ='random'):
    rng = np.random.RandomState(seed)
    if typ == 'blocks':
        h = np.zeros(n)
        pos = 0
        while pos < n:
            w = rng.randint(3, max(4, n // 15))
            h[pos:pos + w] = rng.uniform(0.3, 0.9)
            pos += w
    elif typ == 'sinusoidal':
        t = np.linspace(0, 2 * np.pi * rng.randint(3, 10), n)
        h = 0.5 + 0.3 * np.sin(t) + 0.05 * rng.randn(n)
    elif typ == 'binary_random':
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    else:
        h = rng.rand(n)
    return project_feasible(h, n)

    t0 = time.time()

    # Phase 1: L-BFGS-B polish
    p1 = t0 + 50
    h = best_h.copy()
    for alpha, lam in [(100, 500), (500, 2000), (2000, 10000), (10000, 50000)]:
        if time.time() > p1:

        h = lbfgsb_step(h, best_n, alpha, lam, 400)

        if c < best_c:

            best_h = h.copy()

    # Phase 3: Structured + random pairwise
    p3 = t0 + 650
    h = best_h.copy()
    s3 = seed + 100
    while time.time() < p3:
        h, c = pairwise_cd(h, best_n, 5000, s3, structured=True)
        s3 += 1
        h, c = pairwise_cd(h, best_n, 20000, s3)
        s3 += 1
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 4: Multi-resolution
    p4 = t0 + 800
    for nh in [best_n - 10, best_n + 10, best_n - 20, best_n + 20, best_n + 50]:
        if time.time() > p4 - 15:

        if nh < 100 or nh > 800:

        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nh)
        hi = project_feasible(np.interp(x_new, x_old, best_h), nh)
        hi = lbfgsb_step(hi, nh, 2000, 20000, 200)
        hi, ci = pairwise_cd(hi, nh, 20000, seed + 200)
        if ci < best_c:
            best_c = ci
            best_h = hi.copy()
            best_n = nh

    # Phase 5: Annealing + pairwise
    p5 = dl - 15
    h = best_h.copy()
    temp = 0.03
    for it in range(300):
        if time.time() > p5:
            break
        rng_a = np.random.RandomState(seed + 500 + it)
        h_n = project_feasible(h + temp * rng_a.randn(best_n), best_n)
        h_n, c_n = pairwise_cd(h_n, best_n, 25000, seed + 600 + it)
        if c_n < best_c:
            best_c = c_n
            best_h = h_n.copy()
            h = h_n.copy()
        elif it % 20 == 19:
            h = best_h.copy()
        temp = max(0.003, temp * 0.97)

    # Final polish
    if time.time() < dl - 3:
        h = lbfgsb_step(best_h, best_n, 50000, 500000, 500)
        h, c = pairwise_cd(h, best_n, 50000, seed + 999)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
