def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        # Don't overshoot boundaries
        if diff > 0:
            room = 1.0 - h[mask]
            max_step = room.min()
            if abs(step) > max_step:
                step = max_step
        else:
            room = h[mask]
            max_step = room.min()
            if abs(step) > max_step:
                step = -max_step
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta
