import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def find_optimal_t(a, b):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = np.max(a + c1 * b)
    f2 = np.max(a + c2 * b)
    for _ in range(45):
        if f1 < f2:
            hi = c2; c2 = c1; f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = np.max(a + c1 * b)
        else:
            lo = c1; c1 = c2; f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = np.max(a + c2 * b)
    return (lo + hi) / 2.0

def coord_descent(h, n, deadline, max_sweeps=300, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    c_vec = compute_c5_vec(h, n)
    best_c = c_vec.max()
    best_h = h.copy()
    k_arr = np.arange(2 * n - 1)
    offsets = k_arr - (n - 1)
    min_n = np.maximum(0, -offsets)
    max_n = np.minimum(n - 1, -offsets + (n - 1))
    stale = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        sweep_start_c = c_vec.max()
        for i in order:
            if time.time() > deadline:
                break
            old_val = h[i]
            m_first = i - offsets
            valid_f = (m_first >= min_n) & (m_first <= max_n)
            g = np.zeros(2 * n - 1)
            g[valid_f] = (1.0 - h[m_first[valid_f]])
            valid_s = (min_n <= i) & (i <= max_n)
            idx2 = i + offsets[valid_s]
            g[valid_s] -= h[idx2]
            a = c_vec - old_val * g
            t_opt = find_optimal_t(a, g)
            delta = t_opt - old_val
            if abs(delta) > 1e-14:
                h[i] = t_opt
                c_vec += delta * g
        cur_c = c_vec.max()
        if cur_c < best_c - 1e-13:
            best_c = cur_c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 20:
                h = project_feasible(h, n)
                c_vec = compute_c5_vec(h, n)
                best_c = c_vec.max()
                stale = 0
    return best_h, best_c

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 100:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    offsets = np.arange(2 * n - 1) - (n - 1)

        if w[kk] < 1e-12:

        shift = offsets[kk]
        wm = w[kk] * dx
        min_idx = max(0, -shift)
        max_idx = min(n, n - shift)

            if shift >= 0:

                grad[idx2] += wm * (-x[p])

                grad[idx2] += wm * (-x[p])

def vectorized_grad(x, n, alpha, mu):

    m = c.max()

    w /= w.sum()
    dx_val = 2.0 / n

        grad[p] += wm * (1.0 - x[q])

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
