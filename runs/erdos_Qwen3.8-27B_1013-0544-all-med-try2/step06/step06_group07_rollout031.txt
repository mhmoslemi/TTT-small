import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_multi(h, k_indices, n, dx):
    """Compute gradient using multiple lags."""
    grad = np.zeros(n)
    for k in k_indices:
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += (1.0 - h[idx + lag]) * dx
            grad[idx + lag] -= h[idx] * dx
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += (1.0 - h[idx]) * dx
            grad[idx] -= h[idx + la] * dx
    return grad

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def obj_smooth_grad(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    # gradient of smooth_max(corr, beta) * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    # gradient of corr[k] with respect to h[i]
    grad = np.zeros(len(h))
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        lag = k - (len(h) - 1)
        if lag >= 0:
            idx = np.arange(len(h) - lag)
            grad[idx] += (1.0 - h[idx + lag]) * dx * weights[k]
            grad[idx + lag] -= h[idx] * dx * weights[k]
        else:
            la = -lag
            idx = np.arange(len(h) - la)
            grad[idx + la] += (1.0 - h[idx]) * dx * weights[k]
            grad[idx] -= h[idx + la] * dx * weights[k]
    return smooth_max(corr, beta) * dx, grad

def resample_h(h_old, n_new):

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def coordinate_polish(h, target_sum, n):

    best_c5 = compute_c5(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def con_fun(h, target_sum):
    return h.sum() - target_sum

def con_jac(h, target_sum):
    return np.ones(len(h))

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def subgrad_multi(h, k_indices, n):
    grad = np.zeros(n)
    for k in k_indices:
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            if len(idx) == 0:
                continue
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            if len(idx) == 0:
                continue
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def compute_c5(h):

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):

        if abs(s - target_sum) < 1e-14:

        h = np.clip(h, 0.0, 1.0)
