        hi, hj = h[i], h[j]
        # h[i]=t, h[j]=hj-(t-hi), constraint: sum unchanged
        # corr[m] -> corr[m] + (t-hi)*di[m] + (hj-(t-hi)-hj)*dj[m] = corr[m] + (t-hi)*(di[m]-dj[m])
        dd = di - dj
        A = corr - hi * dd
        # also h[j] must be in [0,1]: hj - (t-hi) in [0,1] => t in [hj-1+hi, hj+hi]
        lo = max(0.0, hj + hi - 1.0)
        hi_b = min(1.0, hj + hi)
        if lo >= hi_b:
            continue
        def f2d(t):
            val = float(np.max(A + t * dd))
            hj_t = hj - (t - hi)
            if hj_t < -1e-10 or hj_t > 1.0 + 1e-10:
                return 1e10
            return val
        try:
            res = minimize_scalar(f2d, bounds=(lo, hi_b), method='bounded',
                                  options={'xatol': 1e-10})
            t_new = res.x
            h[i] = t_new
            h[j] = hj - (t_new - hi)
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target_sum)
        except Exception:
            pass
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def slsqp_polish(h, target_sum, beta=5000):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        return smooth_max(corr, beta) * dx
    def con_f(hv):
        return hv.sum() - target_sum
    def con_j(hv):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-15})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n_base = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Generate diverse starting points at base resolution
    starts = [h_init.copy()]
    for _ in range(8):
        idx = np.random.choice(n_base, n_base // 2, replace=False)
        hb = np.zeros(n_base)
        hb[idx] = 1.0
        starts.append(hb)
    for period in [3, 4, 5, 6, 7]:
        for ph in range(min(period, 3)):
            hp = np.array([1.0 if ((i + ph) % period) < (period + 1) // 2 else 0.0
                          for i in range(n_base)])
            starts.append(hp)
    for _ in range(4):
        x = np.linspace(0, 2, n_base)
        freq = np.random.choice([1, 2, 3, 4])
        ph = np.random.uniform(0, 2 * np.pi)
        hsn = 0.5 + 0.45 * np.sin(np.pi * freq * x + ph)
        starts.append(np.clip(hsn, 0, 1))
    for _ in range(3):
        hs = np.random.rand(n_base) * 0.8
        starts.append(hs)

    # Resolutions to try
    n_list = sorted(set([max(40, n_base - 10), max(40, n_base),
                         n_base + 10, n_base + 20, n_base + 30]))

    # Phase 1: Coordinate descent from diverse starts
    for sp in starts:
        if tl() < 40:
            break
        for n_try in n_list:
            if tl() < 15:
                break
            ts = n_try / 2.0
            ht = resample_h(sp, n_try)
            ht = project_feasible(ht, ts)
            n_passes = max(80, int(min(400, (tl() * 0.6) / (n_try * 0.001 / n_try))))
            ht = coord_descent(ht, ts, n_passes=n_passes, time_limit=tl() * 0.5, t0=t0)
            c5 = compute_c5(ht)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ht.copy()

    # Phase 2: Pair descent from best
    if tl() > 60:
        n_b = len(best_h)
        ts_b = n_b / 2.0
        best_h = pair_descent(best_h, ts_b, n_iter=5000, time_limit=tl() * 0.4, t0=t0)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5

    # Phase 3: SLSQP polish
    if tl() > 40:
        n_b = len(best_h)
        ts_b = n_b / 2.0
        for beta in [1000, 5000, 20000]:
            if tl() < 20:
                break
            hp = slsqp_polish(best_h, ts_b, beta=beta)
            c5 = compute_c5(hp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp

    # Phase 4: More coordinate descent + pair from polished
    if tl() > 30:
        n_b = len(best_h)
        ts_b = n_b / 2.0
        best_h = coord_descent(best_h, ts_b, n_passes=300, time_limit=tl() * 0.4, t0=t0)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5
        if tl() > 20:
            best_h = pair_descent(best_h, ts_b, n_iter=3000, time_limit=tl() * 0.4, t0=t0)
            c5 = compute_c5(best_h)
            if c5 < best_c5:
                best_c5 = c5

    # Phase 5: Try nearby resolutions from best
    if tl() > 30:
        for n_try in [len(best_h) - 5, len(best_h) + 5, len(best_h) + 10]:
            if n_try < 30 or tl() < 15:
                continue
            ts_t = n_try / 2.0
            ht = resample_h(best_h, n_try)
            ht = project_feasible(ht, ts_t)
            ht = coord_descent(ht, ts_t, n_passes=200, time_limit=tl() * 0.4, t0=t0)
            hp = slsqp_polish(ht, ts_t, beta=10000)
            for cand in [ht, hp]:
                c5 = compute_c5(cand)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = cand.copy()

    # Phase 6: Final coordinate descent
    if tl() > 10:
        n_b = len(best_h)
        ts_b = n_b / 2.0
        best_h = coord_descent(best_h, ts_b, n_passes=500, time_limit=tl() * 0.9, t0=t0)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5

    return best_h, best_c5, len(best_h)
