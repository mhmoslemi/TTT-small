**New idea: Use the fact that the optimal solution is nearly binary**

If h is nearly binary (values close to 0 or 1), then small perturbations around the binary values are what matter. We can parameterize h[i] = b[i] + ε[i] where b[i] ∈ {0,1} and |ε[i]| is small.

The constraint sum(h) = n/2 means sum(b) + sum(ε) = n/2, so sum(ε) = n/2 - sum(b).

If sum(b) = n/2 (i.e., exactly n/2 ones), then sum(ε) = 0.

The objective becomes max_k sum_i (b[i] + ε[i]) * (1 - b[i-lag] + ε[i-lag])
= max_k sum_i b[i](1-b[i-lag]) + b[i]ε[i-lag] + ε[i](1-b[i-lag]) + ε[i]ε[i-lag]
= max_k [base_k + linear_in_ε_k + quadratic_in_ε_k]

The base_k is fixed by the binary pattern. The linear and quadratic terms depend on ε.

For the quadratic terms (which are O(ε²)), if ε is small (say 0.01), they're negligible. So the problem reduces to minimizing max_k [base_k + linear_in_ε_k] subject to sum(ε) = 0 and |ε[i]| ≤ min(b[i], 1-b[i]).

This is a linear program (LP)! We can solve it with cvxpy or scipy.

def build_grad(b, n):

    for k in range(2*n-1):
        lag = k - (n+1) + 1  # k - (n-1)
        # Term 1: for i where 0 <= i - lag < n
        i_min1 = max(0, lag)
        i_max1 = min(n, n + lag)
        for i in range(i_min1, i_max1):
            gm[k, i] += 1 - b[i - lag]
        # Term 2: for i where 0 <= i + lag < n
        i_min2 = max(0, -lag)
        i_max2 = min(n, n - lag)
        for i in range(i_min2, i_max2):
            gm[k, i] -= b[i + lag]
    return gm

def build_grad(b, n):
    gm = np.zeros((2*n-1, n))
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    # For each k (row), lag = lags[k]
    # Term 1: gm[k, i] += (1 - b[i - lag]) for valid i
    #   i - lag in [0, n-1] means i in [lag, n-1+lag]
    # Term 2: gm[k, i] -= b[i + lag] for valid i
    #   i + lag in [0, n-1] means i in [-lag, n-1-lag]
    
    # Vectorized approach:
    # For each i, which k's have term 1 valid?
    # i - lag_k in [0, n-1] means lag_k in [i-(n-1), i]
    # lag_k = k - (n-1), so k - (n-1) in [i-(n-1), i] means k in [i, i+n-1]
    # But k is in [0, 2n-2], so k in [i, min(2n-2, i+n-1)]
    
    # This is still O(n²) but can be done with broadcasting...
    # Actually for n=250, a simple loop is fine.
    
    for i in range(n):
        for k in range(2*n-1):
            lag = k - (n-1)
            if 0 <= i - lag < n:
                gm[k, i] += 1 - b[i - lag]
            if 0 <= i + lag < n:
                gm[k, i] -= b[i + lag]
    return gm

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0, 1)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            room = 1 - h; tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0, 1)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = (c - m) / alpha
    return float(m + alpha * (s.max() + np.log(np.sum(np.exp(s - s.max())))))

def _sc(x, n): return float(x.sum() - n / 2)
def _scj(x, n): return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    try:
        r = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                     bounds=[(0, 1)] * n,
                     constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}],
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

def build_grad(b, n):
    gm = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        i1 = np.arange(max(0, lag), min(n, n + lag))
        if len(i1) > 0:
            gm[k, i1] += 1 - b[i1 - lag]
        i2 = np.arange(max(0, -lag), min(n, n - lag))
        if len(i2) > 0:
            gm[k, i2] -= b[i2 + lag]
    return gm

def lp_refine(h, n, max_iter=30, step_limit=0.15, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_iter):
        if deadline and time.time() > deadline: break
        b = (h > 0.5).astype(float)
        target = int(n // 2)
        while b.sum() > target:
            ones = np.where(b == 1)[0]
            i = ones[np.argmin(h[ones])]
            b[i] = 0
        while b.sum() < target:
            zeros = np.where(b == 0)[0]
            i = zeros[np.argmax(1 - h[zeros])]
            b[i] = 1
        c_base = np.correlate(b, 1 - b, mode='full')
        gm = build_grad(b, n)
        eps = cp.Variable(n)
        t = cp.Variable()
        cons = [cp.sum(eps) == 0,
                eps >= -b * step_limit,
                eps <= (1 - b) * step_limit,
                c_base + gm @ eps <= t]
        prob = cp.Problem(cp.Minimize(t), cons)
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver, verbose=False)
                solved = True
                break
            except Exception:
                continue
        if not solved or eps.value is None: break
        h_new = project_feasible(np.clip(b + eps.value, 0, 1), n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            h = h_new
            best_c = c_new
        else:
            break
    return h, best_c

def local_search(h, n, deadline, deltas, radius=0):
    best_c = compute_c5(h, n)
    for d in deltas:
        for _p in range(300):
            if time.time() > deadline: return h, best_c
            imp = False
            for i in range(n):
                if time.time() > deadline: return h, best_c
                jlo = 0 if radius == 0 else max(0, i - radius)
                jhi = n if radius == 0 else min(n, i + radius + 1)
                for j in range(jlo, jhi):
                    if i == j: continue
                    for sgn in (1, -1):
                        hi, hj = h[i] + sgn * d, h[j] - sgn * d
                        if 0 <= hi <= 1 and 0 <= hj <= 1:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c; imp = True
                            else:
                                h[i], h[j] = h[i] - sgn * d, h[j] + sgn * d
            if not imp: break
    return h, best_c

def optimize(h0, n, deadline, rng=None):
    h = project_feasible(h0.copy(), n)
    best = h.copy(); bestc = compute_c5(h, n)
    for a in [50, 200, 1000, 5000, 20000, 100000, 500000]:
        if time.time() > deadline - 25: break
        ht = slsqp(best, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < bestc - 1e-14:
            bestc, best = c, ht.copy()
    if time.time() < deadline - 10:
        ht, ct = lp_refine(best, n, max_iter=30, step_limit=0.15, deadline=deadline - 5)
        if ct < bestc:
            bestc, best = ct, ht.copy()
    if time.time() < deadline - 2:
        dl = [0.01, 0.005, 0.002, 0.001, 0.0005] if n <= 250 else [0.01, 0.005, 0.002]
        best, bestc = local_search(best, n, deadline - 1, dl, radius=0 if n <= 250 else 50)
    return best, bestc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    dl = t0 + budget_s - 3
    try:
        hi = np.array(initial_h_values, dtype=float).copy()
        ni = len(hi)
    except Exception:
        hi, ni = np.full(250, 0.5), 250
    bh, bc = None, float('inf')

    for nt in [200, 300]:
        if nt == ni or time.time() > dl - 45: continue
        alloc = min(75, (dl - time.time()) * 0.3)
        if alloc < 20: break
        hI = project_feasible(np.interp(np.linspace(0, 2, nt), xi, hi), nt)
        h, c = optimize(hI, nt, time.time() + alloc, rng)
        if c < bc: bc, bh = c, h.copy()
    if time.time() < dl - 18:
        n = len(bh)
        scales = [0.003, 0.008, 0.015, 0.03, 0.05, 0.08]
        for r in range(10):
            if time.time() > dl - 8: break
            sc = scales[r % len(scales)]
            hp = project_feasible(bh + rng.normal(0, sc, n), n)
            alloc = min(40, (dl - time.time() - 5) * 0.35)
            if alloc < 12: break
            h, c = optimize(hp, n, time.time() + alloc, rng)
            if c < bc: bc, bh = c, h.copy()
    if time.time() < dl - 2:
        n = len(bh)
        bh, bc = local_search(bh, n, dl - 1,
                              [0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                              radius=0 if n <= 250 else 70)
        bh = project_feasible(bh, n)
        bc = compute_c5(bh, n)
    if bh is None:
        bh = project_feasible(np.full(ni, 0.5), ni)
        bc = compute_c5(bh, ni)
    return bh, bc, len(bh)
