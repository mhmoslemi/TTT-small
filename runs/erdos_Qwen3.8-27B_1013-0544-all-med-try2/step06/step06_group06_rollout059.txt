def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

        else:
            s = h.sum()
            if s < 1e-14:

def slsqp_optimize(h, n, alpha, maxiter, ftol=1e-15):

        m = g.max()

        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

        # dg[k]/dx[i] for the smooth part

        return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary steepest descent from initial (30% budget)
    t1 = t0 + 0.30 * (budget_s - 3)
    hb, cb = binary_steep(best_h, n, t1)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: Try Weyl and random binaries, pick best (15% budget)
    t2 = t0 + 0.45 * (budget_s - 3)
    for attempt in range(3):
        if time.time() > t2 - 30:
            break
        if attempt == 0:
            hc = weyl_binary(n)
        else:
            hc = np.zeros(n)
            idx = rng.choice(n, n // 2, replace=False)
            hc[idx] = 1.0
        hc2, cc2 = binary_steep(hc, n, t2 - 10)
        if cc2 < best_c:
            best_c = cc2
            best_h = hc2.copy()

    # Phase 3: SLSQP continuous optimization (30% budget)
    t3 = t0 + 0.75 * (budget_s - 3)
    h = best_h.copy()
    alphas = [500, 2000, 10000, 50000, 200000, 500000, 2000000, 5000000]
    for alpha in alphas:
        if time.time() > t3 - 20:
            break
        h = slsqp(h, n, alpha, 1500)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            h = best_h.copy()

    # Phase 4: Pair coordinate descent fine-tuning (15% budget)
    t4 = t0 + 0.90 * (budget_s - 3)
    h, c = pair_cd(best_h, n, t4, strides=(1, 2, 5, 11, 23), max_sweeps=40)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 5: Final SLSQP polish (5% budget)
    if time.time() < dl - 15:
        h = slsqp(best_h, n, 10000000, 2000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 6: A few more pair_cd passes if time
    if time.time() < dl - 5:
        h, c = pair_cd(best_h, n, dl - 2, strides=(1, 3, 8), max_sweeps=20)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
