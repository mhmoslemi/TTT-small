def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def reweight_smooth(h, n, deadline, n_rounds=200):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        alpha = 30.0 + rnd * 0.5
        shifted = alpha * (c - c.max())
        max_s = shifted.max()
        v = np.exp(shifted - max_s)
        v /= v.sum()
        g = np.zeros(n)
        for i in range(n):
            l_lo = max(-(n - 1), -i)
            l_hi = min(n - 1, n - 1 - i)
            if l_lo <= l_hi:
                lags = l_lo + np.arange(l_hi - l_lo + 1)
                g[i] += np.dot(v[n - 1 + lags], 1.0 - h[i + lags])
            l_lo2 = max(-(n - 1), i - (n - 1))
            l_hi2 = min(n - 1, i)
            if l_lo2 <= l_hi2:
                lags2 = l_lo2 + np.arange(l_hi2 - l_lo2 + 1)
                g[i] -= np.dot(v[n - 1 + lags2], h[i - lags2])
        lr = 2.0 / (n * np.linalg.norm(g) + 1e-10)
        h = project_feasible(h - lr * g, n)
        cv = compute_c5(h, n)
        if cv < best_c:
            best_c = cv
            best_h = h.copy()
    return best_h, best_c

def iterated_local_search(h_init, n, deadline, seed=42, max_kicks=200):
    rng = np.random.RandomState(seed)
    h = to_binary(h_init, n).copy()
    best_h, best_c = h.copy(), compute_c5(h, n)
    kick_sizes = [2, 3, 3, 4, 5, 2, 2, 3]
    for kick in range(max_kicks):
        if time.time() > deadline:
            break
        ks = kick_sizes[kick % len(kick_sizes)]
        h_kicked = h.copy()
        ones_idx = np.where(h_kicked == 1)[0]
        zeros_idx = np.where(h_kicked == 0)[0]
        for _ in range(ks):
            a = rng.randint(len(ones_idx))
            b = rng.randint(len(zeros_idx))
            h_kicked[ones_idx[a]], h_kicked[zeros_idx[b]] = 0.0, 1.0
        h_new, c_new = greedy_binary(h_kicked, n, min(deadline, time.time() + 3.0), max_sweeps=50)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick binary greedy
    p1 = min(t0 + 0.06 * budget_s, dl)
    hb, cb = greedy_binary(h0, n, p1, max_sweeps=100)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Iterated local search
    p2 = min(t0 + 0.35 * budget_s, dl)
    hb2, cb2 = iterated_local_search(best_h, n, p2, seed=seed, max_kicks=150)
    if cb2 < best_c:
        best_c, best_h = cb2, hb2.copy()

    # Phase 3: Annealing from best
    p3 = min(t0 + 0.60 * budget_s, dl)
    ha, ca = anneal_binary(best_h, n, p3, seed=seed + 1, max_iters=2000000)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()

    # Phase 4: Reweighted smooth optimization
    p4 = min(t0 + 0.82 * budget_s, dl)
    hs, cs = reweight_smooth(best_h, n, p4, n_rounds=300)
    if cs < best_c:
        best_c, best_h = cs, hs.copy()

    # Phase 5: Binary search on smoothed, then re-smooth
    p5 = min(t0 + 0.90 * budget_s, dl)
    hb3, cb3 = greedy_binary(best_h, n, p5, max_sweeps=150)
    if cb3 < best_c:
        best_c, best_h = cb3, hb3.copy()

    # Phase 6: Final anneal + smooth
    p6 = dl - 2
    if time.time() < p6 - 30:
        ha2, ca2 = anneal_binary(best_h, n, p6 - 30, seed=seed + 100, max_iters=1500000)
        if ca2 < best_c:
            best_c, best_h = ca2, ha2.copy()
        hs2, cs2 = reweight_smooth(best_h, n, p6, n_rounds=150)
        if cs2 < best_c:
            best_c, best_h = cs2, hs2.copy()

    # Try a different resolution
    if time.time() < dl - 20:
        for nh in [500, 350]:
            if time.time() > dl - 15:
                break
            xo = np.linspace(0, 2, n, endpoint=False)
            xn = np.linspace(0, 2, nh, endpoint=False)
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            hq, cq = greedy_binary(hh, nh, dl - 12, max_sweeps=80)
            if cq < best_c:
                best_c, best_h, n = cq, hq.copy(), nh
                hs3, cs3 = reweight_smooth(best_h, n, dl - 5, n_rounds=100)
                if cs3 < best_c:
                    best_c, best_h = cs3, hs3.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
