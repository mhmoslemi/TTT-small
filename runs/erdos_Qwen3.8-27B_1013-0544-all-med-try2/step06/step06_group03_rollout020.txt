def subgradient_c5_fast(h, n, top_k=5):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    n_c = len(c)
    top_idx = np.argsort(c)[-top_k:]
    weights = c[top_idx] / c[top_idx].sum()
    
    g = np.zeros(n)
    for w, m in zip(weights, top_idx):
        lag = m - (n - 1)
        if lag >= 0:
            # j+lag valid for j in [0, n-lag)
            g[:n-lag] += w * dx * (1.0 - h[lag:n])
            # j-lag valid for j in [lag, n)
            g[lag:] -= w * dx * h[:n-lag]
        else:
            # lag < 0, let l = -lag
            l = -lag
            # j+lag = j-l valid for j in [l, n)
            g[l:] += w * dx * (1.0 - h[:n-l])
            # j-lag = j+l valid for j in [0, n-l)
            g[:n-l] -= w * dx * h[l:]
    return g

def project_sum(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):

            s = room.sum()
            if s < 1e-15: break
            h += (diff / s) * room

            s = h.sum()
            if s < 1e-15: break
            h += (diff / s) * h

def subgrad_lags(h, n, k_list):
    """Compute gradient of c[k] for lags in k_list. Returns array of shape (len(k_list), n)."""
    dx = 2.0 / n

            grads[idx, lag:] -= dx * h[:n-lag]
        else:
            l = -lag
            grads[idx, l:] += dx * (1.0 - h[:n-l])
            grads[idx, :n-l] -= dx * h[l:]
    return grads

        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
    return (x + 1.0) / 2.0, best_c

def informed_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for d in deltas:
        while time.time() < deadline:
            improved = False
            c = compute_corr(h, n)
            cmax = c.max()
            active = np.where(c > cmax - 0.008)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(c))])
            G = np.zeros((n, len(active)))
            for idx, m in enumerate(active):
                lag = m - (n - 1)
                col = np.zeros(n)
                if lag >= 0:
                    col[:n-lag] = dx * (1.0 - h[lag:n]) - dx * h[:n-lag]
                    col[lag:] -= dx * h[:n-lag]
                    col[:n-lag] += dx * (1.0 - h[lag:n])
                else:
                    l = -lag
                    col[l:] = dx * (1.0 - h[:n-l])
                    col[:n-l] -= dx * h[l:]
                G[:, idx] = col
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                gi = G[i]
                for j in range(i + 1, n):
                    if h[i] + d > 1.0 or h[j] - d < 0.0:
                        continue
                    if np.all(gi <= G[j] + 1e-12):
                        h[i] += d
                        h[j] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-13:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c

def near_search(h, n, deadline, radius=60):
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            order = np.random.permutation(n)
            for i in order:
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if j == i:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c
                            improved = True
                            break
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-13:
                            best_c = c
                            improved = True
                            break
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: SLSQP sweep on initial
    if time.time() < deadline - 80:
        for alpha in [1000, 10000, 100000]:
            if time.time() > deadline - 80:
                break
            h_new = slsqp_opt(best_h, n_init, alpha, maxiter=1500)
            c = compute_c5(h_new, n_init)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h_new

    # Phase 2: X-space subgradient
    if time.time() < deadline - 80:
        h_sg, c_sg = xspace_subgrad(best_h, n_init, steps=300, deadline=deadline - 80)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
        h2 = slsqp_opt(best_h, n_init, 50000, maxiter=1000)
        c2 = compute_c5(h2, n_init)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Try other resolutions
    x_orig = np.linspace(0, 2, n_init)
    for n_try in [100, 150, 200, 300]:
        if time.time() > deadline - 50:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_sum(h_interp, n_try)
        h_opt = slsqp_opt(h_interp, n_try, 50000, maxiter=1000)
        c = compute_c5(h_opt, n_try)
        if c < best_c:
            best_c = c
            best_h = h_opt
            n_init = n_try
            x_orig = np.linspace(0, 2, n_try)

    # Phase 4: Multi-restart
    n = len(best_h)
    if time.time() < deadline - 40:
        remaining = deadline - time.time() - 25
        n_restarts = max(3, int(remaining / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_sum(hp, n)
            h_opt = slsqp_opt(hp, n, 20000, maxiter=800)
            c = compute_c5(h_opt, n)
            if c < best_c:
                best_c = c
                best_h = h_opt

    # Phase 5: Informed local search
    n = len(best_h)
    if time.time() < deadline - 5:
        if n <= 200:
            best_h, best_c = informed_search(best_h, n, deadline - 3,
                                              [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
        else:
            best_h, best_c = near_search(best_h, n, deadline - 3, radius=80)

    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 2:
        n = len(best_h)
        h_fin = slsqp_opt(best_h, n, 200000, maxiter=1500)
        c = compute_c5(h_fin, n)
        if c < best_c:
            best_c = c
            best_h = h_fin
        if time.time() < deadline - 1:
            if n <= 200:
                best_h, best_c = informed_search(best_h, n, deadline,
                                                  [0.002, 0.001, 0.0005, 0.0002, 0.0001])
            else:
                best_h, best_c = near_search(best_h, n, deadline, radius=50)

    best_h = project_sum(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
