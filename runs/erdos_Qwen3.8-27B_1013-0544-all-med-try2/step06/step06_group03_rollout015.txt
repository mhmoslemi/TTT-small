        m = cc.max()
        sh = alpha * (cc - m)
        mx = sh.max()
        if mx > 400:
            s = np.log(np.sum(np.exp(sh - mx))) + mx
        else:
            s = np.log(np.sum(np.exp(sh)))
        return m + s / alpha

        return x.sum() - n / 2.0
    def sj(x):
        return np.ones(n)

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})

    freq = rng.uniform(1.0, 4.0)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.4 * np.sin(np.pi * freq * x + phase)
    return project_feasible(h, n)

def optimize_h(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    remain = deadline - time.time()
    # SLSQP
    if remain > 20:
        alloc_slq = min(remain * 0.4, 120)
        for alpha in [500, 5000, 50000]:
            if time.time() > deadline - 20:
                break
            alloc_a = min(alloc_slq / 3.0, (deadline - time.time()) / 3.0 - 5)
            if alloc_a < 5:
                break
            t_sub = time.time() + alloc_a
            ht = slsqp_opt(h, n, alpha, maxiter=600)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                h = ht
    # Greedy local search
    if time.time() < deadline - 10:
        t_sub = min(deadline - 5, time.time() + (deadline - time.time()) * 0.4)
        h, c = greedy_local_search(h, n, t_sub, rng)
        if c < best_c:
            best_c = c
    # SA local search
    if time.time() < deadline - 5:
        h, c = sa_local_search(h, n, deadline - 2, rng, T0=0.02, cooldown=0.985)
        if c < best_c:
            best_c = c
            h = h
    # Final SLSQP
    if time.time() < deadline - 3:
        ht = slsqp_opt(h, n, 100000, maxiter=500)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    # Final SA
    if time.time() < deadline - 1:
        h, c = sa_local_search(h, n, min(deadline, time.time() + 15), rng, T0=0.005, cooldown=0.995)
        if c < best_c:
            best_c = c
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    best_h = None
    best_c = float('inf')
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100
    x_init = np.linspace(0, 2, n_init, endpoint=False)
    # Phase 1: optimize initial
    alloc1 = min((deadline - t0) * 0.30, 300)
    h_opt, c_opt = optimize_h(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt, c_opt
    # Phase 2: try other n
    for n_try in [100, 200, 300, 400]:
        if time.time() > deadline * 0.7:
            break
        if n_try == n_init or n_try > 500:
            continue
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(150, (deadline - time.time()) * 0.18)
        if alloc < 40:
            break
        h_opt, c_opt = optimize_h(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Phase 3: multi-restart on best n
    n_best = len(best_h)
    remaining = deadline - time.time() - 30
    n_restarts = max(2, int(remaining / 80))
    strategies = []
    for r in range(n_restarts):
        if r % 4 == 0:
            strategies.append('perturb')
        elif r % 4 == 1:
            strategies.append('binary')
        elif r % 4 == 2:
            strategies.append('tm')
        else:
            strategies.append('smooth')
    for r, strat in enumerate(strategies):
        if time.time() > deadline - 40:
            break
        alloc = min(100, (deadline - time.time() - 30) / max(1, n_restarts - r))
        if alloc < 25:
            break
        if strat == 'perturb':
            scale = 0.02 + 0.02 * r
            hp = best_h + rng.normal(0, scale, n_best)
        elif strat == 'binary':
            hp = make_binary_start(n_best, rng)
        elif strat == 'tm':
            hp = make_thue_morse(n_best)
            hp = project_feasible(hp + rng.normal(0, 0.03, n_best), n_best)
        else:
            hp = make_smooth_start(n_best, rng)
        hp = project_feasible(hp, n_best)
        h_opt, c_opt = optimize_h(hp, n_best, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Phase 4: final intensive polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        ht = slsqp_opt(best_h, n, 500000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 2:
            best_h, best_c = sa_local_search(best_h, n, deadline - 1, rng, T0=0.003, cooldown=0.998, min_T=0.0001)
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
