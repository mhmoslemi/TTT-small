def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def analytic_grad(h, n, k_star):
    """O(n) gradient of g(k_star) w.r.t. h."""
    d = k_star - (n - 1)
    a = max(0, d)
    b = min(n - 1, n - 1 + d)
    grad = np.zeros(n)
    dx = 2.0 / n
    
    # Term 1: (1-h[m-d]) for m in [a, b]
    for m in range(a, b + 1):
        md = m - d
        if 0 <= md <= n - 1:
            grad[m] += (1.0 - h[md])
        else:
            grad[m] += 1.0  # h outside range is 0, so 1-h[md] = 1
    
    # Term 2: -h[m+d] for m in [a-d, b-d]
    lo = max(0, a - d)
    hi = min(n - 1, b - d)
    for m in range(lo, hi + 1):
        md = m + d
        if 0 <= md <= n - 1:
            grad[m] -= h[md]
    
    return grad * dx

    """Fast projected subgradient method with analytical gradient."""
    best_h = h.copy()
    g = compute_all_g(h, n)
    best_c = float(g.max())
    k_star = int(np.argmax(g))
    lr = 0.01

            elif time.time() > t0 + 1:  # Safety: don't loop forever
