import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-15:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_full(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def sum_con(h, target_sum):
    return h.sum() - target_sum


def sum_con_jac(h, target_sum):
    return np.ones(len(h))


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def block_obj(block, base, idx, dx, beta):
    h = base.copy()
    h[idx] = block
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def block_con(block, block_target):
    return block.sum() - block_target


def block_con_jac(block, block_target):
    return np.ones(len(block))


def subgrad_direction(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def do_bcd_step(h, n, target_sum, dx, beta, start, bs):
    idx = np.arange(start, start + bs)
    rest_sum = h.sum() - h[idx].sum()
    block_target = target_sum - rest_sum
    if block_target < 1e-10 or block_target > bs - 1e-10:
        return h
    base = h.copy()
    base[idx] = 0.0
    bounds_b = Bounds(np.zeros(bs), np.ones(bs))
    cons_b = [{'type': 'eq', 'fun': block_con, 'jac': block_con_jac,
               'args': (block_target,)}]
    try:
        res_b = minimize(block_obj, h[idx].copy(),
                         args=(base, idx, dx, beta),
                         method='SLSQP', bounds=bounds_b, constraints=cons_b,
                         options={'maxiter': 300, 'ftol': 1e-13})
        new_block = np.clip(res_b.x, 0.0, 1.0)
        new_block += (block_target - new_block.sum()) / bs
        new_block = np.clip(new_block, 0.0, 1.0)
        h[idx] = new_block
    except Exception:
        pass
    return h
