    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    max_steps = 8000
    for step in range(max_steps):
        if time.time() > deadline:
            break
        t += 1
        frac = min(t / max_steps, 1.0)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        lr_cur = lr_init * (1.0 + 0.5 * t) ** (-0.5)
        g = grad_smooth_max(h, n, alpha)
        m_arr = beta1 * m_arr + (1 - beta1) * g
        v_arr = beta2 * v_arr + (1 - beta2) * g ** 2
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        step_vec = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step_vec, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def lbfgs_phase(h, n, deadline, max_iters=300):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 5.0), (500, 10.0), (2000, 50.0),
               (10000, 200.0), (50000, 1000.0),
               (500000, 5000.0), (5000000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        def obj(x):
            return smooth_max_value(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
        def gr(x):
            g = grad_smooth_max(x, n, alpha)
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 20, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_value(x, n, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: float(x.sum() - n / 2.0),
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def pair_search(h, n, deadline, deltas, radius=None, grad_guided=False, alpha=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            if grad_guided:
                g = grad_smooth_max(h, n, alpha)
                order = np.argsort(-np.abs(g))
            else:
                order = np.arange(n)
            for i in order:
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def binary_push(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(200):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            if 0.01 < h[i] < 0.99:
                continue
            target = 0.0 if h[i] < 0.5 else 1.0
            delta = target - h[i]
            if abs(delta) < 1e-10:
                continue
            mask = (h > abs(delta)) & (h < 1.0 - abs(delta))
            mask[i] = False
            if not mask.any():
                mask = (h > 0) & (h < 1)
                mask[i] = False
            if not mask.any():
                continue
            h_new = h.copy()
            h_new[i] = target
            h_new[mask] -= delta / mask.sum()
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c - 1e-15:
                best_c = c
                h = h_new
                improved = True
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: L-BFGS-B baseline
    t1 = min(t0 + 120, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, n_init, t1, max_iters=250)
    # Phase 2: Adam with adaptive alpha
    if time.time() < deadline - 350:
        t2 = min(time.time() + 200, deadline - 350)
        h_a, c_a = adam_optimize(best_h, n_init, t2, alpha_init=50, alpha_final=200000, lr_init=0.015)
        if c_a < best_c:
            best_c, best_h = c_a, h_a
    # Phase 3: SLSQP fine-tuning
    if time.time() < deadline - 200:
        t3 = min(time.time() + 100, deadline - 200)
        for a in [50000, 200000, 1000000]:
            if time.time() > t3:
                break
            h3, c3 = slsqp_phase(best_h, n_init, min(time.time() + 40, t3), alpha=a, maxiter=1500)
            if c3 < best_c:
                best_c, best_h = c3, h3
    # Phase 4: Gradient-guided pair search
    if time.time() < deadline - 150:
        t4 = min(time.time() + 100, deadline - 150)
        best_h, best_c = pair_search(best_h, n_init, t4,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                     radius=80, grad_guided=True, alpha=5000)
    # Phase 5: Binary push
    if time.time() < deadline - 100:
        t5 = min(time.time() + 50, deadline - 100)
        h_b, c_b = binary_push(best_h, n_init, t5)
        if c_b < best_c:
            best_c, best_h = c_b, h_b
    # Phase 6: Fine pair search
    if time.time() < deadline - 60:
        t6 = min(time.time() + 60, deadline - 60)
        best_h, best_c = pair_search(best_h, n_init, t6,
                                     [0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                     radius=50)
    # Phase 7: Try other n
    if time.time() < deadline - 50:
        for n_try in [200, 250, 400]:
            if time.time() > deadline - 30:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            alloc = min(35, deadline - time.time() - 25)
            if alloc < 15:
                break
            h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.5, max_iters=200)
            if time.time() < deadline - 15:
                h_t, c_t = pair_search(h_t, n_try, min(time.time() + 15, deadline - 10),
                                       [0.005, 0.001, 0.0001], radius=40)
            if c_t < best_c:
                best_c, best_h = c_t, h_t
    # Phase 8: Final polish
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = pair_search(best_h, nf, deadline - 2,
                                     [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                     radius=30)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
