```python
    tol = 1e-6

    # c_t/dx = sum_{i=lo_t}^{hi_t} h[i]*(1-h[t-i])
    # dc_t/dh[j] = (1-h[t-j])*I(lo_t<=j<=hi_t) - h[t-j]*I(lo_t<=t-j<=hi_t)
    # where lo_t = max(0, t-n+1), hi_t = min(n-1, t)
    
    grad = np.zeros(n)
    for t in active:
        lo = max(0, t - n + 1)
        hi = min(n - 1, t)
        # First term: (1-h[t-j]) for j in [lo, hi]
        # t-j ranges from t-hi to t-lo
        j_range = np.arange(lo, hi + 1)
        s_range = t - j_range  # = t-lo to t-hi
        valid_s = (s_range >= 0) & (s_range < n)
        grad[j_range[valid_s]] += dx * (1 - h[s_range[valid_s]])
        
        # Second term: -h[t-j] for j such that t-j in [lo, hi]
        # t-j in [lo, hi] means j in [t-hi, t-lo]
        j_lo2 = max(0, t - hi)
        j_hi2 = min(n - 1, t - lo)
        if j_lo2 <= j_hi2:

            grad[j_range2[valid_s2]] -= dx * h[s_range2[valid_s2]]

    grad /= len(active)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        obj = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        obj = m + np.log(np.sum(np.exp(sh))) / alpha
    w = np.exp(sh - mx)
    w /= w.sum()
    g = 1.0 - 2.0 * h
    grad_vec = dx * np.correlate(w, g, mode='full')[:n]
    return float(obj), grad_vec

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(len(x))

def slsqp_with_grad(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        return lse_obj_and_grad(x, n, alpha)[0]
    def obj_g(x):
        return lse_obj_and_grad(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, jac=lambda x: obj_g(x)[1],
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lbfgs_aug(h, n, alpha, mu0=1.0, max_outer=30, max_inner=500):
    mu = mu0
    lam = 0.0
    best = h.copy()
    best_c = compute_c5(h, n)
    for outer in range(max_outer):
        def aug_obj(x):
            base, g = lse_obj_and_grad(x, n, alpha)
            s = x.sum() - n / 2.0
            return base + 0.5 * mu * s * s + lam * s
        def aug_grad(x):
            base, g = lse_obj_and_grad(x, n, alpha)
            s = x.sum() - n / 2.0
            return g + (mu * s + lam) * np.ones(n)
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(aug_obj, h, jac=aug_grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': max_inner, 'ftol': 1e-16})
            h = project_feasible(res.x, n)
        except Exception:
            break
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best = h.copy()
        s = h.sum() - n / 2.0
        if abs(s) < 1e-10:
            break
        lam = lam + mu * s
        mu = mu * 2.0
        if mu > 1e8:
            break
    return best, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c

    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    t_end = deadline - (deadline - time.time()) * beta_end

            h = ht

            h = ht
    # Pair search
    if time.time() < deadline - 5:

            h, best_c = pair_search(h, n, deadline - 3,

                                         radius=50)
    # Final SLSQP polish
    if time.time() < deadline - 2:
        ht = slsqp_with_grad(h, n, 1000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
```
