import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        # Log-sum-exp smooth max
        a2 = alpha * (g - m)
        sm = a2.max()
        return m + (sm + np.log(np.exp(a2 - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_grad(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_grad}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

        a = alpha * (g - m)
        sm = a.max()
        return m + (sm + np.log(np.exp(a - sm).sum())) / alpha

    bounds = [(0.0, 1.0)] * n

                oi, zi = np.where(h==1)[0], np.where(h==0)[0]
                if len(oi)>0 and len(zi)>0:
                    a, b = rng.choice(oi), rng.choice(zi)
                    h[a], h[b] = 0.0, 1.0
            corr = np.correlate(h, 1-h, mode='full').astype(float)
            cur_c = corr.max() * dx
        
        ones_list = np.where(h==1)[0].tolist()
        zeros_list = np.where(h==0)[0].tolist()
        
        for _ in range(max_sweeps):
            if time.time() > deadline: break
            best_delta_c = 0
            best_i, best_j = -1, -1
            best_new_corr = None
            
            for i in zeros_list:
                for j in ones_list:
                    im, imn = i + m_range, i - m_range
                    jm, jmn = j + m_range, j - m_range
                    delta = np.zeros(2*n-1)
                    m1 = (im>=0)&(im<n); delta[m1] += 1.0 - h[im[m1]]
                    m2 = (imn>=0)&(imn<n); delta[m2] -= h[imn[m2]]
                    m3 = (jm>=0)&(jm<n); delta[m3] -= 1.0 - h[jm[m3]]
                    m4 = (jmn>=0)&(jmn<n); delta[m4] += h[jmn[m4]]
                    delta[n-1] -= 2.0
                    delta[n-1+(j-i)] += 1.0
                    delta[n-1+(i-j)] += 1.0
                    nc = (corr + delta).max() * dx
                    if nc < cur_c + best_delta_c - 1e-15:
                        best_delta_c = nc - cur_c
                        best_i, best_j = i, j
                        best_new_corr = corr + delta
                
            if best_i >= 0:
                h[best_i], h[best_j] = 1.0, 0.0
                corr = best_new_corr
                cur_c = cur_c + best_delta_c
                ones_list.remove(best_j)
                ones_list.append(best_i)
                zeros_list.remove(best_i)
                zeros_list.append(best_j)
            else:
                break
        
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
    
    return best_h, best_c

The key innovation in Phase 3: Instead of random perturbations, I'll use **structured perturbations** that target the active lags. Specifically:
- Find the lags where the correlation is closest to the maximum
- Perturb h in directions that reduce those specific correlations
- This is like a "directed" simulated annealing

To compute the direction to reduce correlation at lag k:
∂g_k/∂h[j] = (1-h[j+k]) - h[j-k] (for valid indices)

So the steepest descent direction to reduce g_k is -∂g_k/∂h, projected onto the feasible set.

Let me implement this:

```python
def directed_perturb(h, n, top_k=5):
    """Perturb h in a direction that reduces the top-k correlations."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    # Find top k lags
    top_lags = np.argsort(corr)[-top_k:]
    # Compute gradient direction (negative = reduces correlations)
    grad = np.zeros(n)
    for k in top_lags:
        lag = k - (n-1)  # convert to actual lag
        for i in range(n):
            # d/dh[i] of h[i](1-h[i+lag]) for valid i
            if 0 <= i+lag < n:
                grad[i] += (1 - h[i+lag])
            if 0 <= i-lag < n:
                grad[i] -= h[i-lag]
    # Normalize and apply
    grad_norm = np.linalg.norm(grad)
    if grad_norm > 1e-10:
        grad = -grad / grad_norm
    return h + 0.02 * grad
```

Hmm, but this requires computing partial derivatives for each lag, which is O(n) per lag. For 5 lags and n=250, that's 1250 operations. Very fast.

But wait, I need to be more careful. The correlation at "full" index k corresponds to lag k-(n-1):
corr_full[k] = Σ_i h[i+k-(n-1)] * (1-h[i]) for valid i

So if the max is at full index k_max, the actual lag is k_max - (n-1).

Let me simplify: I'll just compute the gradient numerically or use the analytical form.

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def slsqp(h, n, alpha, maxiter=2000, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        a = alpha * (g - m)
        sm = a.max()
        return m + (sm + np.log(np.exp(a - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return project_feasible(res.x, n)
    except:
        return h
