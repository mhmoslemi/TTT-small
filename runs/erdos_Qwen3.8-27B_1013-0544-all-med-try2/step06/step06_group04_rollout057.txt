import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    """Compute smooth approximation of max(c) using log-sum-exp."""
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def binary_push(h, n, deadline):
    """Push values toward 0 or 1 to reduce lag-0 overlap."""
    h = h.copy()

    for _ in range(100):
        if time.time() > deadline:
            break
        # Find the "most middle" value
        idx = np.argmin(np.minimum(h, 1 - h))
        # Try pushing toward 0

        target = 0.0
        step = 0.05
        h_new[idx] += step * np.sign(target - h[idx])
        # Find compensating index

            comp_idx = np.argmax(1 - h)

def slsqp_refine(h, n, alpha, maxiter=3000):
    """SLSQP optimization with smooth max objective."""
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        return smooth_max(c, alpha)
    
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            weights = np.exp(shifted - mx)
            weights /= weights.sum()
        else:
            weights = np.exp(shifted)
            weights /= weights.sum()
        
        grad = np.zeros(n)
        center = n - 1
        for idx in range(2 * n - 1):
            w = weights[idx]
            if w < 1e-12:
                continue
            lag = idx - center
            if lag >= 0 and lag < n:
                grad[:n-lag] += w * (1.0 - x[lag:n])
                grad[lag:] += w * (-x[:n-lag])
            elif lag < 0:
                m2 = -lag
                if m2 < n:
                    grad[m2:] += w * (1.0 - x[:n-m2])
                    grad[:n-m2] += w * (-x[m2:n])
        grad *= dx
        return grad
    
    def constr(x):
        return np.sum(x) - n / 2.0
    
    def constr_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': constr, 'jac': constr_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    c = np.correlate(h, 1 - h, mode='full')

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def compute_smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()

        if w < 1e-12:
