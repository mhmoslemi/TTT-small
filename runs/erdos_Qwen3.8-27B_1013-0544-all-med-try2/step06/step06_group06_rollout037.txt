    beta = 0.9

        threshold = g_max - 0.01  # within 0.01 of max

        grad = np.zeros(n)
        for ik in active_ks:
            lag = ik - (n - 1)  # convert to lag
            # d g[lag] / d h[p] = (1-h[p-lag])*I[0<=p-lag<n] - h[p+lag]*I[0<=p+lag<n]

        # Evaluate

            h = h_new

            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.9
            if lr < 1e-8:
                break

    dx = 2.0 / n
    h = h.copy()
