        return x[-1]

        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def sum_cons(x):
        return x[:n].sum() - n / 2.0
    def sum_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    # Inequality constraints: c_k - t <= 0 => t - c_k >= 0
    for k in active_ks:

                h_v = x[:n]
                t_v = x[-1]

                i_start = max(0, -shift)
                i_end = min(n, n - shift)
                if i_end <= i_start:

                t_v = x[-1]

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 1.0

def _objective(x):
    n = _g_n
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    a = _g_alpha
    shifted = a * (c - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / a

def _constraint(x):
    return x.sum() - _g_target

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def _slsqp(h, n, alpha, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = n / 2.0
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _constraint}]
    try:
        res = minimize(_objective, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _project(res.x, n)
    except Exception:
        return h.copy()

def _project_sum_box(h, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_smooth(h, n, alpha):
    dx = 2.0 / n
    nm1 = n - 1
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    v = 1.0 - h
    grad = np.zeros(n)
    cw = np.correlate(c, w, mode='full')
    cv = np.correlate(c, v, mode='full')
    for j in range(n):
        k_lo = max(0, nm1 - j)
        k_hi = min(2 * n - 2, nm1 - j + n - 1)
        if k_hi >= k_lo:
            hidx = j + np.arange(k_lo, k_hi + 1) - nm1
            grad[j] = np.dot(w[k_lo:k_hi + 1], v[hidx])
        k_lo2 = max(0, j)
        k_hi2 = min(2 * n - 2, j + n - 1)
        if k_hi2 >= k_lo2:
            hidx2 = j + nm1 - np.arange(k_lo2, k_hi2 + 1)
            grad[j] -= np.dot(w[k_lo2:k_hi2 + 1], h[hidx2])
    return grad * dx

def _pgd(h, n, dl, lr_base=0.01, n_iters=500000):
    cur_c = _c5(h, n)
    alpha = 50000.0
    lr = lr_base
    stall = 0
    for it in range(n_iters):
        if time.time() > dl:
            break
        g = _grad_smooth(h, n, alpha)
        h_new = _project_sum_box(h - lr * g, n)
        c_new = _c5(h_new, n)
        if c_new < cur_c - 1e-15:
            h = h_new
            cur_c = c_new
            stall = 0
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.7
            stall += 1
            if stall > 50:
                lr = lr_base * 0.5
                stall = 0
        if it % 10000 == 0 and alpha < 5e6:
            alpha *= 2.0
    return h, cur_c
