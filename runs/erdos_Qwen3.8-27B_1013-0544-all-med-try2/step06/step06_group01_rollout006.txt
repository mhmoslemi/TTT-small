def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    
    # Phase 1: Binary local search (fast, finds good binary local opt)
    p1 = t0 + 0.25 * (budget_s - 5)
    hb, cb = binary_local_search(best_h, n, p1, max_swaps=200, n_restarts=8, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Snap to binary
    h_bin = (best_h > 0.5).astype(float)
    # Fix sum constraint
    target = n // 2
    if h_bin.sum() != target:
        if h_bin.sum() > target:
            order = np.argsort(h_bin)[::-1]
            h_bin[order[:int(h_bin.sum() - target)]] = 0
        else:
            order = np.argsort(h_bin)
            h_bin[order[:int(target - h_bin.sum())]] = 1
    best_h = h_bin.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 2: SLSQP polish with increasing alpha
    p2 = t0 + 0.55 * (budget_s - 5)
    h_cont = best_h.copy()
    for alpha in [2000, 10000, 50000, 200000, 1000000, 5000000]:
        if time.time() > p2:
            break
        h_cont = slsqp_smooth(h_cont, n, alpha, 1500, p2)
        c = compute_c5(h_cont, n)
        if c < best_c:
            best_c = c
            best_h = h_cont.copy()
    
    # Phase 3: Binary round + local search
    p3 = t0 + 0.70 * (budget_s - 5)
    if time.time() < p3:
        h_b3 = (best_h > 0.5).astype(float)
        if h_b3.sum() != target:
            if h_b3.sum() > target:
                order = np.argsort(h_b3)[::-1]
                h_b3[order[:int(h_b3.sum() - target)]] = 0
            else:
                order = np.argsort(h_b3)
                h_b3[order[:int(target - h_b3.sum())]] = 1
        hb3, cb3 = binary_local_search(h_b3, n, p3, max_swaps=300, n_restarts=10, seed=seed+100)
        if cb3 < best_c:
            best_c = cb3
            best_h = hb3.copy()
    
    # Phase 4: SLSQP from binary solution
    p4 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p4:
        h_s = project_feasible(best_h + 0.03 * rng.randn(n), n)
        for alpha in [50000, 200000, 1000000, 5000000]:
            if time.time() > p4:
                break
            h_s = slsqp_smooth(h_s, n, alpha, 2000, p4)
            c = compute_c5(h_s, n)
            if c < best_c:
                best_c = c
                best_h = h_s.copy()
    
    # Phase 5: Extended binary search
    p5 = dl - 15
    if time.time() < p5:
        h_b5 = (best_h > 0.5).astype(float)
        if h_b5.sum() != target:
            if h_b5.sum() > target:
                order = np.argsort(h_b5)[::-1]
                h_b5[order[:int(h_b5.sum() - target)]] = 0
            else:
                order = np.argsort(h_b5)
                h_b5[order[:int(target - h_b5.sum())]] = 1
        hb5, cb5 = binary_local_search(h_b5, n, p5, max_swaps=500, n_restarts=15, seed=seed+200)
        if cb5 < best_c:
            best_c = cb5
            best_h = hb5.copy()
    
    # Phase 6: Final SLSQP
    if time.time() < dl - 5:
        h_f = slsqp_smooth(project_feasible(best_h + 0.01 * rng.randn(n), n), n, 5000000, 3000, dl - 3)
        c = compute_c5(h_f, n)
        if c < best_c:
            best_c = c
            best_h = h_f.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
