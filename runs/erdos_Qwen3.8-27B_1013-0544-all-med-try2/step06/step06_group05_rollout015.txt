Let w[p] = c[p] + δ*M[i,p] (fixed for given i)
Then we want: min_j max_p (w[p] - δ*M[j,p])

This is a min-max problem. The optimal j is the one where the maximum residual is smallest.

In numpy, for all j simultaneously:
c_new = w[np.newaxis, :] - δ * M  # shape (n, 2n-1)
maxes = c_new.max(axis=1)  # shape (n,)
best_j = argmin(maxes)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_M(h, n):
    """Compute the gradient matrix M where M[i,p] = dc_p/dh[i]."""
    dx = 2.0 / n
    M = np.zeros((n, 2 * n - 1))
    for l in range(-(n - 1), n):
        p = l + n - 1
        # First term: h[i+l] contributes (1-h[i]) when valid
        # For position m in M (i.e., dh[m]), it appears as h[i+l]=h[m] when i=m-l
        # Valid when 0 <= m-l < n
        lo1 = max(0, l)
        hi1 = min(n, n + l)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            M[idx, p] += (1 - h[idx - l]) * dx
        # Second term: (1-h[i]) contributes -h[i+l] when i=m, valid when 0 <= m+l < n
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            M[idx, p] += (-h[idx + l]) * dx
    return M

def fast_pair_search(h, n, deadline, deltas, max_passes=300):
    """Fast pair search using precomputed gradient matrix."""
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    best_c = float(c.max())
    M = compute_M(h, n)
    
    for delta in deltas:
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if h[i] + delta > 1.0 + 1e-12:
                    continue
                # h[j] - delta >= 0 => h[j] >= delta
                valid = h >= delta - 1e-12
                valid[i] = False  # can't swap with self
                if not valid.any():
                    continue
                # c_new[j] = max_p (c[p] + delta * (M[i,p] - M[j,p]))
                w = c + delta * M[i]
                # For all valid j: max_p(w[p] - delta*M[j,p])
                candidate = w[np.newaxis, :] - delta * M[valid]
                maxes = candidate.max(axis=1)
                best_local = np.argmin(maxes)
                valid_idx = np.where(valid)[0]
                best_j = valid_idx[best_local]
                best_val = maxes[best_local]
                if best_val < best_c - 1e-14:
                    h[i] += delta
                    h[best_j] -= delta
                    h[i] = np.clip(h[i], 0, 1)
                    h[best_j] = np.clip(h[best_j], 0, 1)
                    best_c = best_val
                    improved = True
                    # Update c and M
                    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
                    M = compute_M(h, n)
            if not improved:
                break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_M(h, n):
    dx = 2.0 / n
    M = np.zeros((n, 2 * n - 1))
    for l in range(-(n - 1), n):
        p = l + n - 1
        lo1 = max(0, l)
        hi1 = min(n, n + l)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            M[idx, p] += (1 - h[idx - l]) * dx
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            M[idx, p] += (-h[idx + l]) * dx
    return M

def update_c_and_M(c, M, h, i, di, j, dj, n):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    c = c + di * M[i] + dj * M[j]
    h[i] += di
    h[j] += dj
    p1 = np.arange(max(0, n - 1 - i), min(n_lags, 2 * n - 1 - i))
    if len(p1) > 0:
        M[i + (p1 - n + 1), p1] -= di * dx
    p2 = np.arange(max(0, i), min(n_lags, i + n))
    if len(p2) > 0:
        M[i - (p2 - n + 1), p2] -= di * dx
    p1 = np.arange(max(0, n - 1 - j), min(n_lags, 2 * n - 1 - j))
    if len(p1) > 0:
        M[j + (p1 - n + 1), p1] -= dj * dx
    p2 = np.arange(max(0, j), min(n_lags, j + n))
    if len(p2) > 0:
        M[j - (p2 - n + 1), p2] -= dj * dx
    return c, M, h

def fast_pair_search(h, n, deadline, deltas, max_passes=200):
    dx = 2.0 / n
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full') * dx
    best_c = float(c.max())
    M = compute_M(h, n)
    for delta in deltas:
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                if h[i] + delta > 1.0 + 1e-12:
                    continue
                valid = (h >= delta - 1e-12)
                valid[i] = False
                if not valid.any():
                    continue
                w = c + delta * M[i]
                candidate = w[np.newaxis, :] - delta * M[valid]
                maxes = candidate.max(axis=1)
                best_j = np.where(valid)[0][np.argmin(maxes)]
                best_val = float(maxes.min())
                if best_val < best_c - 1e-14:
                    c, M, h = update_c_and_M(c, M, h, i, delta, best_j, -delta, n)
                    h[i] = np.clip(h[i], 0.0, 1.0)
                    h[best_j] = np.clip(h[best_j], 0.0, 1.0)
                    best_c = best_val
                    improved = True
            if not improved:
                break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def subgradient_step(h, n, deadline, n_steps=5000, step_scale=0.01):
    h = h.copy()
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(n_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(c)
        k = k_max - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            for ii in range(n - k):
                grad[ii + k] += (1 - h[ii]) * dx
                grad[ii] += (-h[ii + k]) * dx
        else:
            jj = -k
            for ii in range(jj, n):
                grad[ii - jj] += (1 - h[ii]) * dx
                grad[ii] += (-h[ii - jj]) * dx
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        step = step_scale / (1.0 + 0.1 * it)
        h_new = h - step * grad / g_norm
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c
