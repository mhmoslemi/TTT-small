   - Problem class: Nonlinear non-convex optimization with constraints
   - Constraints: 
     - ∑h · dx = 1 
     - 0 ≤ h[i] ≤ 1 for all i
   - Genetic representation: Floating-point values with projection constraint handler
   - Objective smoothing: Use exp-normalized penalty for overlap peaks
   - Key operations: Inversion mutation (flip regions) + box crossover + adaptive real crossover

   def individual_perturbation(ind):
     # Clone
     new_ind = ind.copy()
     # Adaptive inversion
     if random.random() < mutation_inv_switch:
         # Inversion mutation: Flip region from high to low or vice versa
         val_low = np.percentile(ind, 5)
         val_high = np.percentile(ind, 95)
         mask = (ind < val_low) | (ind > val_high)
         if np.random.rand() < 0.5:
             # Flip high to low
             new_ind[mask] = np.interp(new_ind[mask], [val_high, val_low], [val_low, val_high])
         else:
             # Flip low to high
             new_ind[mask] = np.interp(new_ind[mask], [val_low, val_high], [val_high, val_low])
     else:
         # Keep original random mutation
     return project_feasible(new_ind, n)
