import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_c = c.max()
    shifted = alpha * (c - max_c)
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    log_sum = max_c + (max_sh + np.log(exp_sh.sum())) / alpha
    obj = log_sum
    weights = exp_sh / exp_sh.sum()
    
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for idx in range(n_corr):
        w = weights[idx] * dx
        if w < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[:n-k] += w * (1.0 - h[k:])
            if k < n:
                grad[k:] -= w * h[:n-k]
        else:
            p = -k
            if p < n:
                grad[p:] += w * (1.0 - h[:n-p])
            if n - p > 0:
                grad[:n-p] -= w * h[p:]
    
    return obj, grad

def obj_only(h, n, alpha):
    return smooth_obj_and_grad(h, n, alpha)[0]

def grad_only(h, n, alpha):
    return smooth_obj_and_grad(h, n, alpha)[1]

def slsqp_analytical(h, n, alpha, maxiter=2000):
    def fun(x):
        return smooth_obj_and_grad(x, n, alpha)[0]
    def jac(x):
        return smooth_obj_and_grad(x, n, alpha)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search_sa(h, n, deadline, deltas, radius=None, sa_temp=0.001, sa_cool=0.95):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    temp = sa_temp
    n_points = n

        radius = n - 1
    
    for d in deltas:
        while time.time() < deadline and temp > 1e-8:
            improved = False
            for i in range(n_points):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n_points, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    for direction in [1, -1]:
                        hi_i = h[i] + direction * d
                        lo_j = h[j] - direction * d
                        if 0.0 - 1e-9 <= hi_i <= 1.0 + 1e-9 and 0.0 - 1e-9 <= lo_j <= 1.0 + 1e-9:
                            h[i] = np.clip(hi_i, 0, 1)
                            h[j] = np.clip(lo_j, 0, 1)
                            c = compute_c5(h, n)
                            delta = c - current_c
                            if delta < 0 or (temp > 1e-8 and np.random.random() < np.exp(-delta / temp)):

                                    best_h = h.copy()
                                improved = True
                            else:

                                h[j] = lo_j + direction * d
            temp *= sa_cool
            if not improved and temp < 0.01:
                break
    
    # Ensure we return the best found
    # Re-project
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

When we change h[i] by δ and h[j] by -δ, how does the correlation change?

c[k] = Σ_n h[n](1-h[n+k])

Δc[k] = (change in terms involving h[i]) + (change in terms involving h[j])

For h[i] → h[i] + δ:
- Terms where n = i: h[i](1-h[i+k]) changes by δ(1-h[i+k]) if i+k is valid
- Terms where n+k = i, i.e., n = i-k: h[i-k](1-h[i]) changes by -h[i-k]*δ if i-k is valid

For h[j] → h[j] - δ:
- Terms where n = j: h[j](1-h[j+k]) changes by -δ(1-h[j+k]) if j+k is valid
- Terms where n+k = j, i.e., n = j-k: h[j-k](1-h[j]) changes by h[j-k]*δ if j-k is valid

So:
Δc[k] = δ * [(1-h[i+k])*I(0≤i+k<n) - h[i-k]*I(0≤i-k<n)

def compute_delta_c(h, i, j, delta, n):
    """Compute change in correlation when h[i]+=delta, h[j]-=delta."""
    dx = 2.0 / n
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For h[i] change:
    # (1-h[i+lag]) for valid i+lag, -h[i-lag] for valid i-lag
    idx_plus = i + lags  # indices i+lag
    mask_plus = (idx_plus >= 0) & (idx_plus < n)
    
    idx_minus = i - lags  # indices i-lag
    mask_minus = (idx_minus >= 0) & (idx_minus < n)
    
    contrib_i = np.zeros(2*n-1)
    contrib_i[mask_plus] = (1.0 - h[idx_plus[mask_plus]])
    contrib_i[mask_minus] -= h[idx_minus[mask_minus]]
    
    # For h[j] change (opposite sign):
    idx_plus_j = j + lags
    mask_plus_j = (idx_plus_j >= 0) & (idx_plus_j < n)
    
    idx_minus_j = j - lags
    mask_minus_j = (idx_minus_j >= 0) & (idx_minus_j < n)
    
    contrib_j = np.zeros(2*n-1)
    contrib_j[mask_plus_j] = (1.0 - h[idx_plus_j[mask_plus_j]])
    contrib_j[mask_minus_j] -= h[idx_minus_j[mask_minus_j]]
    
    return delta * (contrib_i - contrib_j) * dx

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    mc = c.max()
    s = alpha * (c - mc)
    ms = s.max()
    es = np.exp(s - ms)
    return mc + (ms + np.log(es.sum())) / alpha

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    mc = c.max()
    s = alpha * (c - mc)
    ms = s.max()
    es = np.exp(s - ms)
    w = es / es.sum()
    
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for idx in range(2*n-1):
        wi = w[idx] * dx
        if wi < 1e-14:
            continue
        k = lags[idx]
        if k >= 0:
            if n-k > 0:
                grad[:n-k] += wi * (1.0 - h[k:])
            if k < n:
                grad[k:] -= wi * h[:n-k]
        else:
            p = -k
            if p < n:
                grad[p:] += wi * (1.0 - h[:n-p])
            if n-p > 0:
                grad[:n-p] -= wi * h[p:]
    return grad

    return smooth_obj(x, n, alpha)

def _grad_wrapper(x, n, alpha):
    return smooth_grad(x, n, alpha)

    return x.sum() - n / 2.0

def _cons_jac(x, n):
    return np.ones(len(x))

def slsqp_opt(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_wrapper, h, args=(n, alpha), jac=_grad_wrapper,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
