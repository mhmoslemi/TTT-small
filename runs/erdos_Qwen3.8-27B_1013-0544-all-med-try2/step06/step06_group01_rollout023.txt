Precompute for each position p the vector v_p[q] = (1-h[p+q]) - h[p-q] (with boundary handling). Then for a swap (i,j), delta[q] = v_i[q] - v_j[q] + cross_terms.

If I precompute all v_p as an (n, 2n-1) matrix, then for all pairs (i,j), I can compute the deltas as:
deltas[i,j,:] = V[i,:] - V[j,:] + cross_terms(i,j)

The cross terms only affect 3 positions (q=0, q=j-i, q=i-j), so they can be added after.

Then for each pair, I need max(corr + deltas[i,j,:]). This is a (n_zeros, n_ones, 2n-1) tensor, which for n=250 is 125*125*499 ≈ 7.8 million entries. That's manageable in NumPy.

new_max[i,j] = max_q (corr[q] + V[i,q] - V[j,q])

This is: new_max[i,j] = max_q (corr[q] + V[i,q]) - ... no, max of a sum is not the sum of maxes.

But I can compute it as: for each q, the value is corr[q] + V[i,q] - V[j,q]. The max over q is what I want.

One approach: compute M[i,q] = corr[q] + V[i,q] for all i,q (shape: n_zeros × (2n-1)), and N[j,q] = V[j,q] for all j,q (shape: n_ones × (2n-1)). Then new_max[i,j] = max_q (M[i,q] - N[j,q]).

This is still not trivially vectorizable in a memory-efficient way for large n. But for n=250, it should be fine.

            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:

            h = h + diff / s * h

def get_delta(h, i, j, n):
    """Compute the delta vector for swapping h[i]:0->1 and h[j]:1->0."""
    nq = 2 * n - 1
    delta = np.zeros(nq)
    for q_idx in range(nq):
        q = q_idx - (n - 1)
        ip = i + q
        iq = i - q
        jp = j + q
        jq = j - q
        val = 0.0
        if 0 <= ip < n:
            val += 1.0 - h[ip]
        if 0 <= iq < n:
            val -= h[iq]
        if 0 <= jp < n:
            val -= 1.0 - h[jp]
        if 0 <= jq < n:
            val += h[jq]
        delta[q_idx] = val
    # Cross terms
    delta[n-1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < nq:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < nq:
        delta[k2] += 1.0
    return delta

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return None, None, None, None

def binary_sa(h, n, deadline, max_sweeps=5000, seed=42, 
              T_start=0.005, T_end=0.0001, patience=500):
    """Binary local search with simulated annealing."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_local = cur_c
    no_improve = 0

        if time.time() > deadline:
            break
        
        result = vectorized_sweep(h, corr, n, dx)
        if result[0] is None:

            h[bi], h[bj] = 1.0, 0.0
            corr = bnc
            cur_c = bc

                best_h = h.copy()
                best_local = cur_c
                no_improve = 0
            else:
                no_improve += 1
        else:

            T = T_start * (T_end / T_start) ** (sweep / max(1, max_sweeps))
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]

            j = rng.choice(ones)

            acceptance = np.exp(-(nc - cur_c) / max(T, 1e-10))

                cur_c = corr.max() * dx
                no_improve = 0
                patience = min(patience * 2, 5000)

def subgradient_step(h, n, step_size):
    """One subgradient step for the continuous problem."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    k_star = np.argmax(corr)
    q = k_star - (n - 1)
    
    # Gradient of c[q] with respect to h
    grad = np.zeros(n)
    p = np.arange(n)
    pp = p + q
    pm = p - q
    m1 = (pp >= 0) & (pp < n)
    grad[m1] += 1.0 - h[pp[m1]]
    m2 = (pm >= 0) & (pm < n)
    grad[m2] -= h[pm[m2]]
    # For q=0, correction
    if q == 0:
        grad -= 2.0 * h  # subgradient correction (approximate)
    
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

def slsqp_optimize(h, n, alpha, maxiter, deadline):

        m = g.max()

        sm = sh.max()

                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def get_delta(h, i, j, n):
    nq = 2 * n - 1
    delta = np.zeros(nq)
    for q_idx in range(nq):
        q = q_idx - (n - 1)
        val = 0.0
        ip = i + q
        if 0 <= ip < n:
            val += 1.0 - h[ip]
        iq = i - q
        if 0 <= iq < n:
            val -= h[iq]
        jp = j + q
        if 0 <= jp < n:
            val -= 1.0 - h[jp]
        jq = j - q
        if 0 <= jq < n:
            val += h[jq]
        delta[q_idx] = val
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < nq:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < nq:
        delta[k2] += 1.0
    return delta

def vectorized_sweep(h, corr, n, dx, top_k=80):
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return None, None, None, None
    nq = 2 * n - 1
    V = np.zeros((n, nq))
    p = np.arange(n)
    for q_idx in range(nq):
        q = q_idx - (n - 1)
        pp = p + q
        pm = p - q
        v1 = np.zeros(n)
        m1 = (pp >= 0) & (pp < n)
        v1[m1] = 1.0 - h[pp[m1]]
        v2 = np.zeros(n)
        m2 = (pm >= 0) & (pm < n)
        v2[m2] = h[pm[m2]]
        V[:, q_idx] = v1 - v2
    A = corr[None, :] + V[zeros, :]
    B = V[ones, :]
    C = np.max(A[:, None, :] - B[None, :, :], axis=2)
    K = min(top_k, C.size)
    flat = C.ravel()
    top_idx = np.argsort(flat)[:K]
    best_val = np.inf
    best_i = -1
    best_j = -1
    best_nc = None
    for idx in top_idx:
        ii = int(idx // len(ones))
        jj = int(idx % len(ones))
        i = zeros[ii]
        j = ones[jj]
        delta = get_delta(h, i, j, n)
        new_corr = corr + delta
        nc = new_corr.max()
        if nc < best_val:
            best_val = nc
            best_i = i
            best_j = j
            best_nc = new_corr
    return best_i, best_j, best_nc, best_val * dx

def binary_sa(h, n, deadline, max_sweeps=5000, seed=42,
              T_start=0.004, T_end=0.0001, patience=300, top_k=80):

    idx = np.argsort(h)[::-1]
