   import time

   def compute_c5(h, n):
       dx = 2.0 / n
       c = np.correlate(h, 1 - h, mode='full')
       return float(np.max(c) * dx)

   def project_feasible(h, n):
       target = n / 2.0
       h = np.clip(h, 0.0, 1.0)
       for _ in range(200):
           diff = target - h.sum()
           if abs(diff) < 1e-13:
               break
           if diff > 0:
               room = 1.0 - h
               s = room.sum()
               if s < 1e-14:
                   break
               h += diff / s * room
           else:
               s = h.sum()
               if s < 1e-14:
                   break
               h += diff / s * h
       return np.clip(h, 0.0, 1.0)

   def to_binary(h, n):
       idx = np.argsort(h)[::-1]
       h_b = np.zeros(n)
       h_b[idx[:n // 2]] = 1.0
       return h_b

   def compute_gradient_power(h, n, p):
       """Compute gradient of power-mean approximation."""
       dx = 2.0 / n
       c = np.correlate(h, 1 - h, mode='full') * dx
       M = 2 * n - 1
       c_safe = np.maximum(c, 1e-14)
       if p <= 1.0:
           # Subgradient: just use the argmax
           m_idx = np.argmax(c)
           w = np.zeros(M)
           w[m_idx] = 1.0
       else:
           w = c_safe ** (p - 1) / (p * np.mean(c_safe ** p))
       grad = np.zeros(n)
       for m in range(M):
           wm = w[m]
           if wm < 1e-10:
               continue
           d = m - (n - 1)
           j_lo = max(0, d)
           j_hi = min(n, n + d)
           if j_lo < j_hi:
               grad[j_lo:j_hi] += wm * (1.0 - h[j_lo - d : j_hi - d])
           j_lo2 = max(0, -d)
           j_hi2 = min(n, n - d)
           if j_lo2 < j_hi2:
               grad[j_lo2:j_hi2] -= wm * h[j_lo2 + d : j_hi2 + d]
       return grad * dx, c

   def proj_grad_method(h_init, n, deadline, p_seq=None):
       """Projected gradient with adaptive power mean."""
       h = h_init.copy()
       best_c = compute_c5(h, n)
       best_h = h.copy()
       
       if p_seq is None:
           p_seq = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
       
       # Step size schedule
       base_alpha = 0.1
       
       for p in p_seq:
           if time.time() > deadline:
               break
           n_steps = 5000
           alpha = base_alpha / np.sqrt(p)
           for step in range(n_steps):
               if time.time() > deadline:
                   break
               grad, c = compute_gradient_power(h, n, p)
               grad_norm = np.linalg.norm(grad)
               if grad_norm < 1e-12:
                   break
               # Backtracking line search
               a = alpha
               for _ in range(20):
                   h_new = h - a * grad
                   h_new = project_feasible(h_new, n)
                   c_new = compute_c5(h_new, n)
                   if c_new <= best_c + 1e-12 or a < 1e-10:
                       break
                   a *= 0.5
               if c_new < best_c:
                   best_c = c_new
                   best_h = h_new.copy()
                   h = h_new
               else:
                   break
       return best_h, best_c
