- If achieved at multiple k: any convex combination of {∇f_k : k active}

f_k = sum_m h[m+k]*(1-h[m])*dx

∂f_k/∂h[j]:
- h[j] appears as h[m+k] when m = j-k (if valid): contributes (1-h[j-k])*dx
- h[j] appears in (1-h[m]) when m = j (if j+k valid): contributes -h[j+k]*dx

So ∂f_k/∂h[j] = dx * [(1-h[j-k])*I(0≤j-k<n) - h[j+k]*I(0≤j+k<n)]

This is O(1) per (j,k) pair, so the full gradient is O(n²) or O(n) per active lag.

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, lr, momentum_v, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_c = c.max()
    # Find active lags (within tolerance)
    tol = 0.005
    active = np.where(c > max_c - tol)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    for idx in active:
        lag = int(idx) - (n - 1)
        for j in range(n):
            gj = 0.0
            if 0 <= j - lag < n:
                gj += (1.0 - h[j - lag])
            if 0 <= j + lag < n:
                gj -= h[j + lag]
            grad[j] += gj * dx / len(active)
    
    # Momentum update
    momentum_v = beta * momentum_v - lr * grad
    h_new = h + momentum_v
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    
    return h_new, momentum_v, c

    grad = np.zeros(n)

    else:

- For the active lag k, changing h[i] and h[i+k] simultaneously can be beneficial
- This is a "paired" move that specifically targets the active lag

Here's the key insight for the pair search: if lag k is active, then f_k = sum_i h[i]*(1-h[i+k])*dx. To decrease f_k, we want to DECREASE h[i]*(1-h[i+k]) for the terms with large values. This means:
- If h[i] is large and h[i+k] is small: decrease h[i] or increase h[i+k]
- If h[i] is small and h[i+k] is large: increase h[i] or decrease h[i+k]

So the optimal pair move for lag k is: find (i, j) where j = i+k (or j-k = i), and transfer from the "large" to the "small" in a way that reduces the product h[i]*(1-h[i+k]).

This is much more targeted than random pair search!

    best_c = compute_c5(h, n)
    return h, best_c

def general_pair_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in [0.01, 0.005, 0.002, 0.001]:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    h = project_feasible(h, n)
                    return h, compute_c5(h, n)
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        cc = compute_c5(h, n)
                        if cc < best_c - 1e-14:
                            best_c = cc
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        cc = compute_c5(h, n)
                        if cc < best_c - 1e-14:
                            best_c = cc
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx2 = sh.max()
    if mx2 > 500:
        return float(m + (mx2 + np.log(np.sum(np.exp(sh - mx2)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _ones_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _ones_j, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n = 100

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase A: Subgradient momentum (fast global)
    dl = min(t0 + 60, deadline - 200)
    h_a, c_a = subgradient_momentum(h_init, n, 4000, dl)
    if c_a < best_c:
        best_c, best_h = c_a, h_a

    # Phase B: SLSQP polish
    if time.time() < deadline - 200:
        h_b = slsqp_opt(best_h, n, 200000, 2000)
        c_b = compute_c5(h_b, n)
        if c_b < best_c:
            best_c, best_h = c_b, h_b

    # Phase C: Coordinate descent (coarse)
    dl = min(t0 + 250, deadline - 150)
    h_c, c_c = coord_descent(best_h, n, 200, dl, fine=False)
    if c_c < best_c:
        best_c, best_h = c_c, h_c

    # Phase D: Lag-aware pair search
    dl = min(t0 + 500, deadline - 120)
    h_d, c_d = lag_pair_search(best_h, n, dl)
    if c_d < best_c:
        best_c, best_h = c_d, h_d

    # Phase E: Fine coordinate descent
    dl = min(t0 + 750, deadline - 80)
    h_e, c_e = coord_descent(best_h, n, 300, dl, fine=True)
    if c_e < best_c:
        best_c, best_h = c_e, h_e

    # Phase F: SLSQP with very high alpha
    if time.time() < deadline - 60:
        h_f = slsqp_opt(best_h, n, 800000, 3000)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c, best_h = c_f, h_f

    # Phase G: Random restarts
    n_restarts = 4
    for r in range(n_restarts):
        if time.time() > deadline - 80:
            break
        scale = 0.02 + 0.04 * r
        hp = project_feasible(best_h + rng.normal(0, scale, n), n)
        alloc = min(60, (deadline - time.time() - 30) * 0.4)
        if alloc < 15:
            break
        hr, cr = coord_descent(hp, n, 80, time.time() + alloc, fine=True)
        hr, cr = lag_pair_search(hr, n, time.time() + 20)
        if cr < best_c:
            best_c, best_h = cr, hr

    # Phase H: Final fine polish
    if time.time() < deadline - 20:
        h_h, c_h = coord_descent(best_h, n, 200, deadline - 10, fine=True)
        if c_h < best_c:
            best_c, best_h = c_h, h_h
        if time.time() < deadline - 10:
            h_h2, c_h2 = lag_pair_search(best_h, n, deadline - 5)
            if c_h2 < best_c:
                best_c, best_h = c_h2, h_h2

    # Phase I: Try different n
    for n_try in [200, 500]:
        if time.time() > deadline - 30:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        h_t, c_t = coord_descent(h_interp, n_try, 80, deadline - 10, fine=True)
        if c_t < best_c:
            best_c, best_h = c_t, h_t

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
