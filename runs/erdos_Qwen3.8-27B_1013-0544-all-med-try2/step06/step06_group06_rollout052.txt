def compute_gradient_softmax(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    # softmax weights
    c_shifted = c - c.max()
    p = np.exp(c_shifted / alpha)
    p /= p.sum()
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if p[k] < 1e-15:
            continue
        lag = k - (n-1)
        L = max(0, -lag)
        R = min(n-1, n-1-lag)
        for j in range(L, R+1):
            grad[j] += p[k] * (1.0 - h[j+lag])
        for j in range(L+lag, R+lag):  # j-lag in [L, R] means j in [L+lag, R+lag]
            if 0 <= j-lag < n:
                grad[j] -= p[k] * h[j-lag]
    return dx * grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h0, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        sh = (c - m) / alpha
        sm = sh.max()
        return m + alpha * (sm + np.log(np.exp(sh - sm).sum()))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h0

def proj_grad_descent(h, n, deadline, alpha=1000, max_iter=5000):
    """Projected gradient descent with analytical gradient on soft-max objective"""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 2.0 / (np.max(np.abs(h)) + 1e-10)  # Initialize learning rate
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1-h, mode='full') * dx
        c_shift = c - c.max()
        p = np.exp(c_shift / alpha)
        p /= p.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(2*n-1):
            if p[k] < 1e-14:
                continue
            lag = k - (n-1)
            L = max(0, -lag)
            R = min(n-1, n-1-lag)
            # First term: for j in [L, R], grad[j] += p[k] * (1 - h[j+lag])
            grad[L:R+1] += p[k] * (1.0 - h[L+lag:R+1+lag])
            # Second term: for j-lag in [L, R], i.e., j in [L+lag, R+lag]
            j_lo = max(0, L + lag)
            j_hi = min(n-1, R + lag)
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] -= p[k] * h[j_lo-lag:j_hi+1-lag]
        grad *= dx
        
        # Backtracking line search
        ng = np.linalg.norm(grad)
        if ng < 1e-15:
            break
        f_curr = compute_c5(h, n)
        step = lr
        improved = False
        for _ in range(20):
            h_new = project_feasible(h - step * grad, n)
            f_new = compute_c5(h_new, n)
            if f_new < f_curr - 1e-15:
                h = h_new
                improved = True
                if f_new < best_c:
                    best_c = f_new
                    best_h = h.copy()
                break
            step *= 0.5
        if not improved:
            lr *= 0.8
            if lr < 1e-15:
                break
    
    return best_h, best_c

        h_new = project_feasible(h - (lr/ng)*grad, n)

def binary_local_search(h, n, deadline, max_sweeps=300):
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    ones_set = set(ones)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            for j in ones:
                h_bin[i] = 1.0; h_bin[j] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c:
                    best_c = c
                    ones.remove(j); ones.append(i)
                    ones_set.discard(j); ones_set.add(i)
                    zeros.remove(i); zeros.append(j)
                    improved = True
                else:
                    h_bin[i] = 0.0; h_bin[j] = 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c
