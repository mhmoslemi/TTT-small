        else:
            step = max(step * 0.97, 0.003)
            if it % 100 == 0:
                h = project(h + 0.03 * rng.randn(n), n)
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + np.log(np.exp(sh - sm).sum()) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return project(res.x, n)
    except Exception:
        return h

def binary_worst_lag_search(h, n, deadline, seed=42):
    h = to_binary(h, n)
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    stuck = 0
    for it in range(2000000):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        sr = np.zeros(len(ones))
        for ii, a in enumerate(ones):
            sc = 0.0
            if 0 <= a + s < n:
                sc -= (1 - h[a + s])
            if 0 <= a - s < n:
                sc += h[a - s]
            sr[ii] = sc
        sa = np.zeros(len(zeros))
        for ii, b in enumerate(zeros):
            sc = 0.0
            if 0 <= b + s < n:
                sc += (1 - h[b + s])
            if 0 <= b - s < n:
                sc -= h[b - s]
            sa[ii] = sc
        ir = np.argsort(sr)
        ia = np.argsort(sa)
        best_tot = 99.0
        bi, bj = -1, -1
        for ii in ir[:40]:
            for jj in ia[:40]:
                tot = sr[ii] + sa[jj]
                if tot < best_tot:
                    best_tot = tot
                    bi, bj = ii, jj
        if best_tot < -0.5 and bi >= 0:
            a, b = ones[bi], zeros[bj]
            h[a], h[b] = 0.0, 1.0
            corr = np.correlate(h, 1 - h, mode='full')
            new_max = corr.max()
            if new_max <= cur_max:
                cur_max = new_max
                stuck = 0
            else:
                if rng.random() < 0.2 * np.exp(-(new_max - cur_max) * 5):
                    cur_max = new_max
                    stuck += 1
                else:
                    h[a], h[b] = 1.0, 0.0
                    stuck += 1
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h[a], h[b] = 0.0, 1.0
            corr = np.correlate(h, 1 - h, mode='full')
            new_max = corr.max()
            if new_max < cur_max + 3:
                cur_max = new_max
            else:
                h[a], h[b] = 1.0, 0.0
            stuck += 1
        if stuck > 500:
            for _ in range(8):
                o2 = np.where(h == 1)[0]
                z2 = np.where(h == 0)[0]
                a = o2[rng.randint(len(o2))]
                b = z2[rng.randint(len(z2))]
                h[a], h[b] = 0.0, 1.0
            corr = np.correlate(h, 1 - h, mode='full')
            cur_max = corr.max()
            stuck = 0
    return best_h, best_max * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: continuous descent
    p1 = t0 + 0.15 * (budget_s - 5)
    hc, cc = continuous_descent(best_h, n, p1, top_k=3, seed=seed)
    if cc < best_c:
        best_c, best_h = cc, hc
    # Phase 2: SLSQP
    p2 = t0 + 0.28 * (budget_s - 5)
    if time.time() < p2:
        for alpha in [500, 5000, 50000, 500000]:
            if time.time() > p2:
                break
            hs = slsqp_smooth(best_h, n, alpha, 800, p2)
            c = compute_c5(hs, n)
            if c < best_c:
                best_c, best_h = c, hs
    # Phase 3: binary worst-lag search (main phase)
    p3 = t0 + 0.65 * (budget_s - 5)
    hb, cb = binary_worst_lag_search(best_h, n, p3, seed=seed)
    if cb < best_c:
        best_c, best_h = cb, hb
    # Phase 4: continuous descent from binary
    p4 = t0 + 0.78 * (budget_s - 5)
    if time.time() < p4:
        hc2, cc2 = continuous_descent(best_h, n, p4, top_k=2, seed=seed + 50)
        if cc2 < best_c:
            best_c, best_h = cc2, hc2
    # Phase 5: final binary polish
    p5 = dl - 3
    if time.time() < p5:
        hb2, cb2 = binary_worst_lag_search(best_h, n, p5, seed=seed + 200)
        if cb2 < best_c:
            best_c, best_h = cb2, hb2
    best_h = project(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
