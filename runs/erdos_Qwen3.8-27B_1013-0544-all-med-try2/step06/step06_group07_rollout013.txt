import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def sm_corr_value(h):
    return smooth_max(np.correlate(h, 1.0 - h, mode='full'), 1.0)


def sm_corr_penalty(h, dx, beta, pen):
    n = len(h)
    tc = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    sm = smooth_max(corr, beta)
    f = dx * sm + pen * (h.sum() - tc) ** 2
    return f


def sm_corr_grad(h, dx, beta, pen):
    n = len(h)
    tc = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    sm = smooth_max(corr, beta)
    f = dx * sm + pen * (h.sum() - tc) ** 2
    shifted = np.clip(beta * (corr - np.max(corr)), -500.0, 500.0)
    w = np.exp(shifted)
    w = w / w.sum()
    grad = 2.0 * pen * (h.sum() - tc) * np.ones(n)
    threshold = 1e-10
    active = np.where(w > threshold)[0]
    for k in active:
        lag = k - (n - 1)
        wk = dx * w[k]
        if lag > 0:
            i1 = np.arange(lag, n)
            grad[i1] += wk * (1.0 - h[i1 - lag])
            i2 = np.arange(n - lag)
            grad[i2] -= wk * h[i2 + lag]
        elif lag == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        else:
            la = -lag
            i1 = np.arange(n - la)
            grad[i1] += wk * (1.0 - h[i1 + la])
            i2 = np.arange(la, n)
            grad[i2] -= wk * h[i2 - la]
    return f, grad


def sum_constraint(h, tc):
    return h.sum() - tc


def sum_constraint_jac(h, tc):
    return np.ones(len(h))


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    tc = n / 2.0
    h_init = project_feasible(h_init, tc)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with continuation
    if tl() > 400:
        for beta in [5, 20, 80, 300, 1000, 4000, 15000]:
            if tl() < 150:
                break
            pen = 50.0
            h_cur = best_h.copy()
            bnds = Bounds(np.zeros(n), np.ones(n))
            for attempt in range(2):
                if tl() < 50:
                    break
                h_start = h_cur.copy()
                if attempt > 0:
                    h_start += np.random.randn(n) * 0.01
                    h_start = project_feasible(h_start, tc)
                try:
                    res = minimize(sm_corr_grad, h_start, args=(dx, beta, pen),
                                   method='L-BFGS-B', jac=True, bounds=bnds,
                                   options={'maxiter': 3000, 'ftol': 1e-14,
                                            'gtol': 1e-12, 'maxfun': 15000})
                    h_cur = project_feasible(res.x, tc)
                    c5 = compute_c5(h_cur)
                    if c5 < best_c5:
                        best_h = h_cur.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 2: SLSQP with high beta for exact constraint handling
    if tl() > 200:
        cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac,
                 'args': (tc,)}]
        bnds = Bounds(np.zeros(n), np.ones(n))
        for beta in [2000, 10000, 50000]:
            if tl() < 80:
                break
            for trial in range(2):
                if tl() < 40:
                    break
                x0 = best_h.copy()
                if trial > 0:
                    x0 += np.random.randn(n) * 0.005
                    x0 = project_feasible(x0, tc)
                def obj_slsqp(x, _dx=dx, _beta=beta):
                    return smooth_max(np.correlate(x, 1.0 - x, mode='full'), _beta) * _dx
                # Can't use closure, use args instead
                try:
                    res = minimize(lambda x: smooth_max(np.correlate(x, 1.0 - x, mode='full'), beta) * dx,
                                   x0, method='SLSQP', bounds=bnds,
                                   constraints=cons,
                                   options={'maxiter': 5000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, tc)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 3: Subgradient with momentum on exact objective
    if tl() > 150:
        h = best_h.copy()
        momentum = np.zeros(n)
        beta_m = 0.9
        lr = 0.003
        for it in range(50000):
            if tl() < 40:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_worst = np.argmax(corr)
            lag = k_worst - (n - 1)
            grad = np.zeros(n)
            if lag > 0:
                i1 = np.arange(lag, n)
                grad[i1] += (1.0 - h[i1 - lag])
                i2 = np.arange(n - lag)
                grad[i2] -= h[i2 + lag]
            elif lag == 0:
                grad[:] = 1.0 - 2.0 * h
            else:
                la = -lag
                i1 = np.arange(n - la)
                grad[i1] += (1.0 - h[i1 + la])
                i2 = np.arange(la, n)
                grad[i2] -= h[i2 - la]
            momentum = beta_m * momentum + (1.0 - beta_m) * grad
            lr_t = lr / (1.0 + 0.001 * it)
            h = h - lr_t * momentum
            h = project_feasible(h, tc)
            if it % 500 == 499:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 4: Resolution search
    if tl() > 100:
        n_list = sorted(set([n + 10, n + 20, n - 10, n + 5, n - 5,
                             50, 60, 80, 90, 100, 120, 150]),
                        key=lambda x: abs(x - n))
        for n_try in n_list:
            if n_try < 30 or n_try > 300 or tl() < 40:
                continue
            dx_t = 2.0 / n_try
            tc_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, tc_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            for beta in [300, 2000]:
                if tl() < 20:
                    break
                try:
                    res = minimize(sm_corr_grad, h_t, args=(dx_t, beta, 50.0),
                                   method='L-BFGS-B', jac=True, bounds=bnds_t,
                                   options={'maxiter': 2000, 'ftol': 1e-13,
                                            'gtol': 1e-11, 'maxfun': 10000})
                    h_t = project_feasible(res.x, tc_t)
                    c5 = compute_c5(h_t)
                    if c5 < best_c5:
                        best_h = h_t.copy()
                        best_c5 = c5
                        n = n_try
                        dx = dx_t
                        tc = tc_t
                except Exception:
                    pass

    # Phase 5: Final L-BFGS-B polish with very high beta
    if tl() > 40:
        n = len(best_h)
        dx = 2.0 / n
        tc = n / 2.0
        bnds = Bounds(np.zeros(n), np.ones(n))
        for beta in [30000, 100000]:
            if tl() < 20:
                break
            try:
                res = minimize(sm_corr_grad, best_h, args=(dx, beta, 200.0),
                               method='L-BFGS-B', jac=True, bounds=bnds,
                               options={'maxiter': 5000, 'ftol': 1e-15,
                                        'gtol': 1e-14, 'maxfun': 25000})
                h_new = project_feasible(res.x, tc)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 6: Final SLSQP polish
    if tl() > 15:
        n = len(best_h)
        dx = 2.0 / n
        tc = n / 2.0
        bnds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac,
                 'args': (tc,)}]
        beta_final = 50000
        try:
            def _obj_fn(x):
                return smooth_max(np.correlate(x, 1.0 - x, mode='full'), beta_final) * dx
            res = minimize(_obj_fn, best_h, method='SLSQP', bounds=bnds,
                           constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, tc)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
