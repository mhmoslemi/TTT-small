    return float(np.max(compute_corr(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff / s * room
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def weighted_grad(h, n, weights):
    """Gradient of sum_k weights[k]*c(k) w.r.t. h.
    weights indexed by full correlation (len 2n-1), index (n-1+k) = lag k."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        w = weights[n - 1 + k]
        if abs(w) < 1e-18:
            continue
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:]) * dx
            grad[k:] -= w * h[:n-k] * dx
        else:
            kk = -k
            grad[kk:] += w * (1.0 - h[:n-kk]) * dx
            grad[:n-kk] -= w * h[kk:] * dx
    return grad

def smooth_obj_and_grad(h, n, alpha, penalty):
    """Log-sum-exp smoothed objective + penalty for sum constraint."""
    corr = compute_corr(h, n)
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    obj = m + np.log(exps.sum()) / alpha
    # Penalty
    s = h.sum() - n / 2.0
    obj += penalty * s * s
    # Gradient
    soft_w = exps / exps.sum()
    grad = weighted_grad(h, n, soft_w)
    grad += 2.0 * penalty * s * np.ones(n)
    return obj, grad

def lbfgs_smooth(h, n, alpha, penalty, deadline, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: smooth_obj_and_grad(x, n, alpha, penalty)[0],
            h, jac=lambda x: smooth_obj_and_grad(x, n, alpha, penalty)[1],
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h

    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0

    zeros_list = [int(x) for x in np.where(h == 0)[0]]

            break

                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0

                delta[k2] += 1.0

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_corr(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            r = 1.0 - h
            s = r.sum()
            if s > 1e-14:
                h += diff / s * r
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def wgrad(h, n, w):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(-(n - 1), n):
        wi = w[n - 1 + k]
        if wi == 0:
            continue
        if k >= 0:
            g[:n - k] += wi * (1.0 - h[k:]) * dx
            g[k:] -= wi * h[:n - k] * dx
        else:
            kk = -k
            g[kk:] += wi * (1.0 - h[:n - kk]) * dx
            g[:n - kk] -= wi * h[kk:] * dx
    return g

def obj_grad(h, n, alpha, pen):
    corr = compute_corr(h, n)
    m = corr.max()
    sh = alpha * (corr - m)
    sm = sh.max()
    ev = np.exp(sh - sm)
    f = m + np.log(ev.sum()) / alpha
    s = h.sum() - n / 2.0
    f += pen * s * s
    sw = ev / ev.sum()
    g = wgrad(h, n, sw) + 2.0 * pen * s
    return f, g
