    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

def multi_lag_gradient(h, n, top_k, beta_w):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    top_vals = corr[top_idx]
    w = np.exp(beta_w * (top_vals - top_vals.max()))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        lag = ki - (n - 1)
        grad += wi * lag_gradient(h, n, lag)
    return grad

def obj_slsqp(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones(len(h))

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def momentum_grad(h, n, target_sum, t_deadline):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_curr = h.copy()
    h_prev = h.copy()
    lr = 0.005
    beta_w = 30.0
    top_k = 8
    momentum = 0.85
    stuck = 0
    it = 0
    while time.time() < t_deadline:
        it += 1
        mom = momentum * np.exp(-it / 500.0) * 0.5 + 0.7 * (1 - np.exp(-it / 500.0))
        y = h_curr + mom * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        grad = multi_lag_gradient(y, n, top_k, beta_w)
        h_new = project_feasible(y - lr * grad, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.03, 0.03)
            stuck = 0
        else:
            stuck += 1
            lr = max(lr * 0.8, 0.0001)
        beta_w = min(beta_w * 1.002, 150.0)
        if stuck > 300:
            h_new = project_feasible(h_new + np.random.randn(n) * 0.008, target_sum)
            stuck = 0
            lr = 0.005
        h_prev = h_curr
        h_curr = h_new
    return best_h, best_c5

def coord_descent(h, n, target_sum, t_deadline):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_curr = h.copy()
    grid = np.linspace(0.0, 1.0, 13)
    while time.time() < t_deadline:
        order = np.random.permutation(n)
        improved_any = False
        for j in order:
            if time.time() > t_deadline:
                break
            best_v = h_curr[j]
            best_val = best_c5
            for v in grid:
                if abs(v - h_curr[j]) < 1e-6:
                    continue
                h_temp = h_curr.copy()
                h_temp[j] = v
                h_temp = project_feasible(h_temp, target_sum)
                c5_v = compute_c5(h_temp)
                if c5_v < best_val:
                    best_val = c5_v
                    best_v = v
            h_curr[j] = best_v
            h_curr = project_feasible(h_curr, target_sum)
            c5_curr = compute_c5(h_curr)
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()
                improved_any = True
        if not improved_any:
            break
    return best_h, best_c5

def slsqp_polish(h, n, dx, target_sum, beta, maxiter, t_deadline):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
    try:
        res = minimize(obj_slsqp, h, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Momentum gradient (50% budget)
    if remaining() > 150:
        deadline = t0 + remaining() * 0.45
        best_h, best_c5 = momentum_grad(best_h, n, target_sum, deadline)

    # Phase 2: Coordinate descent (20% budget)
    if remaining() > 80:
        deadline = t0 + remaining() * 0.25
        h_cd, c5_cd = coord_descent(best_h.copy(), n, target_sum, deadline)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    # Phase 3: SLSQP polish (15% budget)
    if remaining() > 60:
        for beta in [500, 3000, 10000]:
            if remaining() < 20:
                break
            deadline = t0 + remaining() - 10
            h_tmp, c5_tmp = slsqp_polish(best_h, n, dx, target_sum, beta, 5000, deadline)
            if c5_tmp < best_c5:
                best_h = h_tmp
                best_c5 = c5_tmp

    # Phase 4: Resolution sweep (10% budget)
    if remaining() > 100:
        for n_try in [n + 10, n - 10, n + 20, n - 20, n + 30, n - 30]:
            if n_try < 25 or remaining() < 40:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            deadline = t0 + remaining() - 15
            h_t, c5_t = momentum_grad(h_t, n_try, ts_t, deadline)
            if remaining() < 15:
                break
            h_t, c5_t2 = slsqp_polish(h_t, n_try, dx_t, ts_t, 3000, 3000, t0 + remaining() - 5)
            c5_t = min(c5_t, c5_t2)
            if c5_t < best_c5:
                best_h = h_t
                best_c5 = c5_t
                n = n_try
                dx = dx_t
                target_sum = ts_t

    # Phase 5: Final polish
    if remaining() > 10:
        h_final, c5_final = slsqp_polish(best_h, n, dx, target_sum, 30000, 10000, t0 + remaining() - 1)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    return best_h, best_c5, len(best_h)
