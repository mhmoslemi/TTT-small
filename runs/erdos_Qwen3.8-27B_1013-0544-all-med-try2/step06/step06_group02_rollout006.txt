    dx = 2.0 / n

    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0

    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.where((h > 1e-8) & (h < 1 - 1e-8))[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def topk_grad(h, n, k, alpha):
    c = compute_c5_vec(h, n)
    kk = min(k, len(c))
    top_idx = np.argsort(c)[-kk:]
    top_vals = c[top_idx]
    av = alpha * (top_vals - top_vals.max())
    w_top = np.exp(av)
    w_top /= w_top.sum()
    w = np.zeros(len(c))
    w[top_idx] = w_top
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k_shift = m - (n - 1)
        wm = w[m] * dx
        if k_shift >= 0:
            nv = n - k_shift
            grad[0:nv] += wm * (1.0 - h[k_shift:n])
            grad[k_shift:n] += wm * (-h[0:nv])
        else:
            kk2 = -k_shift
            nv = n - kk2
            grad[kk2:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk2:n])
    return grad, c

    kk = min(k, len(c))
    top_idx = np.argsort(c)[-kk:]
    top_vals = c[top_idx]
    m = top_vals.max()
    av = alpha * (top_vals - m)
    val = m + np.log(np.sum(np.exp(av))) / alpha
    return float(val)

    g, _ = topk_grad(x, n, k, alpha)
    return g
