```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_g(h, n):
    """Compute all correlation values (scaled by dx)."""
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def lbsmooth(h, n, alpha, penalty=100.0):
    """Smooth max objective with constraint penalty."""
    g = compute_g(h, n)
    m = g.max()
    shifted = alpha * (g - m)
    # Numerical stability
    shifted = np.clip(shifted, -100, 100)
    logsumexp = m + np.log(np.exp(shifted).sum()) / alpha
    # Constraint penalty
    cv = h.sum() - n / 2.0
    return logsumexp + penalty * cv * cv

def lbsmooth_grad(h, n, alpha, penalty=100.0):
    """Gradient of smooth max objective."""
    dx = 2.0 / n
    g_raw = np.correlate(h, 1 - h, mode='full')  # length 2n-1
    g = g_raw * dx
    
    m = g.max()
    shifted = alpha * (g - m)
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Gradient of g[k] with respect to h[j]
    # g[k] = dx * sum_i h[i] * (1 - h[i + lag_k]) where lag_k = k - n + 1
    # dg[k]/dh[j] = dx * [(1 - h[j + lag_k]) * I[j+lag_k valid] - h[j-lag_k] * I[j-lag_k valid]]
    
    grad = np.zeros(n)
    for k in range(len(g_raw)):
        if weights[k] < 1e-10:
            continue
        lag = k - n + 1
        w = weights[k] * dx
        if lag >= 0:
            # j + lag in [0, n-1] means j in [0, n-1-lag]
            # j - lag in [0, n-1] means j in [lag, n-1]
            # Term 1: (1 - h[j+lag]) for j in [0, n-1-lag]
            # Term 2: -h[j-lag] for j in [lag, n-1]
            j1 = np.arange(0, n - lag)
            grad[j1] += w * (1 - h[j1 + lag])
            j2 = np.arange(lag, n)
            grad[j2] += w * (-h[j2 - lag])
        else:
            # lag < 0, let l = -lag
            l = -lag
            # j + lag = j - l in [0, n-1] means j in [l, n-1]
            # j - lag = j + l in [0, n-1] means j in [0, n-1-l]
            j1 = np.arange(l, n)
            grad[j1] += w * (1 - h[j1 + lag])  # h[j-l]
            j2 = np.arange(0, n - l)
            grad[j2] += w * (-h[j2 - lag])  # h[j+l]
    
    # Add constraint gradient
    cv = h.sum() - n / 2.0
    grad += 2 * penalty * cv
    
    return grad
```

**Alternative: min-max game approach**
The problem is min_h max_k g_k(h). This is a zero-sum game where:

The inner problem (min_h ∑_k p_k * g_k(h)) is a convex optimization problem (quadratic in h, with linear constraints). We can solve it efficiently.

The outer problem (max_p) is over a simplex, which can be solved with multiplicative weights or other methods.

Is M PSD? M = ∑_k p_k * (S_lag + S_lag^T)/2. Each (S_lag + S_lag^T)/2 is the adjacency matrix of a "shift graph" which is... hmm, not necessarily PSD.

Actually, (S_lag)_{i,j} = 1 if j=i+lag. So (S_lag + S_lag^T)_{i,j} = 1 if j=i+lag or j=i-lag. This is the adjacency matrix of a circulant-like graph (but not truly circulant due to boundary effects). For a truly circulant matrix with a single nonzero in each row, the eigenvalues are on the unit circle, so the symmetrized version has eigenvalues in [-1, 1]. So M is not necessarily PSD.

```python
def project_sum(h, n):
    """Project h to satisfy sum(h) = n/2 while keeping 0<=h<=1."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def objective(h, n, alpha, penalty):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    shifted = np.clip(alpha * (g - m), -700, 700)
    lse = m + np.log(np.exp(shifted).sum()) / alpha
    cv = h.sum() - n / 2.0
    return lse + penalty * cv * cv

def grad_func(h, n, alpha, penalty):
    dx = 2.0 / n
    g_raw = np.correlate(h, 1 - h, mode='full')
    g = g_raw * dx
    m = g.max()

    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(g)] = 1.0
    else:
        w /= wsum
    grad_out = np.zeros(n)
    for k in range(len(g_raw)):
        if w[k] < 1e-10:
            continue
        lag = k - n + 1
        wk = w[k] * dx
        if lag >= 0:
            if n - lag > 0:

            if l > 0:
                grad_out[:n-l] -= wk * h[l:]
    cv = h.sum() - n / 2.0
    grad_out += 2.0 * penalty * cv
    return grad_out

def run_lbfgsb(h, n, alpha, penalty, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h, jac=grad_func, method='L-BFGS-B',
                      bounds=bounds, args=(n, alpha, penalty),
                      options={'maxiter': maxiter, 'ftol': 1e-15,
                               'gtol': 1e-12, 'maxls': 50})
        h_new = np.clip(res.x, 0, 1)
    except Exception:
        h_new = h.copy()
    return project_sum(h_new, n)

    g_max = g.max()
    active = np.where(g >= g_max - epsilon)[0]
    if len(active) == 0:
```
