import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(x, alpha):
    m = x.max()
    s = alpha * (x - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s)))) / alpha

def obj_all(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    return smooth_max(c, alpha)

def obj_active(x, n, idx, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    return smooth_max(c[idx], alpha)

def s_con(x, n):
    return float(x.sum() - n / 2.0)

def s_con_jac(x, n):
    return np.ones(n)

def do_slsqp(h0, n, obj_fn, args, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': s_con, 'jac': s_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h0, args=args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += 1.0 - h[:n - m]
    else:
        grad[:n - m] += 1.0 - h[m:]
        grad[m:] -= h[:n - m]
    return grad

def cvx_cutting_point(h0, n, top_idx, trust):
    dx = 2.0 / n
    c = compute_all_lags(h0, n)
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [cp.sum(hv) == n / 2.0, hv >= 0, hv <= 1, cp.abs(hv - h0) <= trust]
    for idx in top_idx:
        lag = idx - (n - 1)
        g = lag_gradient(h0, n, lag)
        cons.append(c[idx] + dx * (g @ (hv - h0)) <= tv)
    prob = cp.Problem(cp.Minimize(tv), cons)
    for solver in ['GLOP', 'CBC', 'CLARABEL']:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    if hv.value is None:
        return h0.copy()
    return project_feasible(hv.value, n)
