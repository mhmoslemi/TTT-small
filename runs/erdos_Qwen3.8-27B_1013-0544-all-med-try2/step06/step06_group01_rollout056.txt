from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h *= 1.0 + diff / s
    return np.clip(h, 0.0, 1.0)

def compute_dir(h, n, i):
    """Compute direction vector d for coordinate i."""
    s = np.arange(-(n-1), n)
    idx1 = i + s
    idx2 = i - s
    d = np.zeros(2*n - 1)
    m1 = (idx1 >= 0) & (idx1 < n)
    d[m1] += 1.0 - h[idx1[m1]]
    m2 = (idx2 >= 0) & (idx2 < n)
    d[m2] -= h[idx2[m2]]
    return d

def soft_coord_desc(h, n, deadline, n_grid=32):
    """Coordinate descent with exact 1D optimization per coordinate."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_max = corr.max()
    best_h = h.copy()
    best_c = cur_max * dx
    n_iter = 0
    while time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            v = h[i]
            d = compute_dir(h, n, i)
            # Find δ ∈ [-v, 1-v] minimizing max(corr + δ*d)
            lo, hi = -v, 1.0 - v
            if hi - lo < 1e-12:
                continue
            deltas = np.linspace(lo, hi, n_grid)
            best_val = np.inf
            best_d = 0.0
            for dt in deltas:
                val = np.max(corr + dt * d)
                if val < best_val:
                    best_val = val
                    best_d = dt
            if abs(best_d) > 1e-10:
                h[i] = v + best_d
                corr += best_d * d
                cur_max = corr.max()
                improved = True
                n_iter += 1
        # Project
        h = project_feasible(h, n)
        corr = np.correlate(h, 1 - h, mode='full')
        cur_max = corr.max()
        c = cur_max * dx
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

        m2 = (i - s_arr >= 0) & (i - s_arr < n)

                best_c = c
                best_h = h.copy()

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break

            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):

    def obj(x):

        m = g.max()

        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum() + 1e-30)) / alpha

                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
