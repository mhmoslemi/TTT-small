import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

                        h[i] += d

                        c += d * diff
                        current_max = new_max
                        update_G(G, h, n, i, d, j, -d)
                        improved = True

                            h[j] += d
                            c += d * diff2
                            current_max = new_max2
                            update_G(G, h, n, i, -d, j, d)
                            improved = True
                            stagnant = 0
            if not improved:
                break
        best_c = current_max * dx
    
    return h, best_c

def compute_G(h, n):
    G = np.zeros((n, 2*n-1))
    for s_idx in range(2*n-1):
        s = s_idx - (n-1)
        L = max(0, -s)
        R = min(n, n-s)
        if L >= R:
            continue
        # First term: (1-h[i-s]) for i in [L+s, R+s)
        i_lo = max(0, L + s)
        i_hi = min(n, R + s)
        if i_lo < i_hi:
            G[i_lo:i_hi, s_idx] += (1.0 - h[L:R])
        # Second term: -h[i+s] for i in [L, R)
        G[L:R, s_idx] -= h[L+s:R+s]
    return G

def update_G(G, h, n, i, delta_i, j, delta_j):
    """Update G after h[i] += delta_i and h[j] += delta_j"""
    for idx in (i, j):
        delta = delta_i if idx == i else delta_j
        if delta == 0:
            continue
        for s_idx in range(2*n-1):
            s = s_idx - (n-1)
            L = max(0, -s)
            R = min(n, n-s)
            # First term: G[k,s] has (1-h[k-s]) if L <= k-s < R
            # h[idx] appears in h[k-s] when k-s = idx, i.e., k = idx+s
            k1 = idx + s
            if 0 <= k1 < n and L <= idx < R:
                G[k1, s_idx] -= delta
            # Second term: G[k,s] has -h[k+s] if L <= k < R
            # h[idx] appears in h[k+s] when k+s = idx, i.e., k = idx-s
            k2 = idx - s
            if 0 <= k2 < n and L <= k2 < R:
                G[k2, s_idx] -= delta

G[k, s_idx] = A(k,s) - B(k,s)
where A(k,s) = (1-h[k-s]) if L_s <= k-s < R_s (and 0 <= k-s < n)
and B(k,s) = h[k+s] if L_s <= k < R_s (and 0 <= k+s < n)

h[idx] appears in A(k,s) when k-s = idx, i.e., k = idx+s.
Condition: L_s <= idx < R_s (since k-s = idx must satisfy L_s <= k-s < R_s)
         and 0 <= k < n (i.e., 0 <= idx+s < n)
         and 0 <= k-s < n (i.e., 0 <= idx < n, always true)

When h[idx] increases by delta, A(k,s) = 1-h[idx] decreases by delta.
So G[idx+s, s_idx] -= delta.

h[idx] appears in B(k,s) when k+s = idx, i.e., k = idx-s.
Condition: L_s <= k < R_s (i.e., L_s <= idx-s < R_s)
         and 0 <= k < n (i.e., 0 <= idx-s < n)
         and 0 <= k+s < n (i.e., 0 <= idx < n, always true)

When h[idx] increases by delta, B(k,s) = h[idx] increases by delta.
So G[idx-s, s_idx] -= delta (since G = A - B, and B increased).

So the update is:
- G[idx+s, s_idx] -= delta (if 0 <= idx+s < n and L_s <= idx < R_s)
- G[idx-s, s_idx] -= delta (if 0 <= idx-s < n and L_s <= idx-s < R_s)
