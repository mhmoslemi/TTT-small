def gradient_of_c_shift(h, n, k):
    """Compute gradient of c[k] (the k-th correlation value) w.r.t. h.
    k is the index in the full correlation output (0 to 2n-2).
    """
    grad = np.zeros(n)
    lag = k - (n - 1)  # can be negative, zero, or positive
    
    if lag == 0:
        # c = sum h[i](1-h[i]), grad = 1 - 2h[i]
        grad = 1.0 - 2.0 * h
    elif lag > 0:
        # c = sum_{j=lag}^{n-1} h[j]*(1-h[j-lag])
        # d/dh[i] = (1-h[i-lag]) if i >= lag, minus h[i+lag] if i+lag < n
        for i in range(lag, n):
            grad[i] += (1.0 - h[i - lag])
        for i in range(0, n - lag):
            grad[i] -= h[i + lag]
    else:
        # lag < 0, let m = -lag
        # c = sum_{j=0}^{n-1-m} h[j]*(1-h[j+m])
        # d/dh[i] = (1-h[i+m]) if i+m < n, minus h[i-m] if i >= m
        m = -lag
        for i in range(0, n - m):
            grad[i] += (1.0 - h[i + m])
        for i in range(m, n):
            grad[i] -= h[i - m]
    
    return grad * (2.0 / n)  # multiply by dx

    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break

        m = c.max()

            best_c = m

        active = np.where(c >= m - eta)[0]

        for k in active:
            grad += gradient_of_c_shift(h, n, k)
        grad /= len(active)

        # We want to move mass from high-grad positions to low-grad positions
        order = np.argsort(grad)
        # Most promising sources: high gradient (end of sorted array)
        # Most promising sinks: low gradient (beginning of sorted array)

                    amount = min(d, 1.0 - h[i], h[j])
                    orig_i, orig_j = h[i], h[j]
                    h[i], h[j] = orig_i + amount, orig_j - amount
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                        break
                    else:
                        h[i], h[j] = orig_i, orig_j
                if h[i] > 1e-12 and h[j] < 1.0 - 1e-12:
                    amount = min(d, h[i], 1.0 - h[j])
                    orig_i, orig_j = h[i], h[j]
                    h[i], h[j] = orig_i - amount, orig_j + amount
