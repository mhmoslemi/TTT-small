import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13: break
            h *= (target / tot)
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def _sum_con(x, n):
    return np.sum(x) - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha=50000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def fast_local_search(h, n, deadline, max_rounds=50000):
    dx = 2.0 / n
    h = h.astype(float).copy()
    c = np.correlate(h, 1 - h, mode='full') * dx
    best_c = c.max()
    best_h = h.copy()
    
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for d_idx, d_base in enumerate(deltas):
        if time.time() > deadline - 1:
            break
        round_in_phase = 0
        while round_in_phase < max_rounds and time.time() < deadline - 1:
            round_in_phase += 1
            # Find worst lag
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            m = abs(lag)
            if m == 0:
                # Special handling for lag 0
                # L_0 = sum(h*(1-h))*dx = (sum(h) - sum(h^2))*dx = (n/2 - sum(h^2))*dx
                # To decrease L_0, increase sum(h^2), push values toward 0 or 1
                improved = False
                sorted_idx = np.argsort(h)
                for _ in range(10):
                    lo = sorted_idx[0]
                    hi = sorted_idx[-1]
                    if h[hi] - h[lo] < 0.01:
                        break
                    dd = min(d_base * 0.5, (h[hi] - h[lo]) * 0.25)
                    if dd < 1e-10:
                        break
                    h[hi] += dd
                    h[lo] -= dd
                    h = np.clip(h, 0, 1)
                    c = np.correlate(h, 1 - h, mode='full') * dx
                    if c.max() < best_c:
                        best_c = c.max()
                        best_h = h.copy()
                        improved = True
                if not improved:
                    break
                continue
            
            # For the worst lag, try to decrease it
            # L_m = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m]) * dx
            # Gradient wrt h[j]:
            #   dx * [(1-h[j+m]) if j <= n-1-m else 0] - [h[j-m] if m <= j <= n-m else 0] * dx
            
            # Best move: for each pair (i, i+m) in this lag, try moving mass
            # to reduce h[i]*(1-h[i+m])
            best_move_val = 0  # improvement (positive = good)
            best_move = None
            
            # Scan pairs in the active lag
            step = max(1, m // 50)  # subsample for speed if m is large
            for i in range(0, n - m, step):
                a, b = i, i + m  # b = a + m when lag = m > 0
                # For lag > 0: term is h[a]*(1-h[b])
                # To decrease: increase h[b] relative to h[a]
                # i.e., h[a] -= dd, h[b] += dd
                if h[a] > 0.01:
                    dd = min(d_base, h[a] * 0.5, (1 - h[b]) * 0.5)
                    if dd > 1e-10:
                        # Change in this lag's term:
                        # (h[a]-dd)*(1-h[b]-dd... no wait
                        # h[a] decreases by dd, h[b] increases by dd
                        # term: h[a]*(1-h[b]) -> (h[a]-dd)*(1-(h[b]+dd)) = (h[a]-dd)*(1-h[b]-dd)
                        # = h[a](1-h[b]) - dd(1-h[b]) - dd*h[a] + dd^2
                        # change = -dd*(1-h[b]+h[a]) + dd^2 (negative = good)
                        change = -dd * (1 - h[b] + h[a]) + dd * dd
                        if change * dx < -best_move_val:
                            best_move_val = -change * dx
                            best_move = (a, b, dd)  # h[a] -= dd, h[b] += dd
                elif h[b] < 0.99:
                    # Try moving mass the other way (less likely to help for this lag)
                    pass
            
            # Also try moving mass from outside the lag to reduce its conflict
            # e.g., reduce h at a position that conflicts with many other lags
            if best_move is None and d_base < 0.001:
                # Try global moves: reduce a high h[i] that contributes to many lags
                top_idx = np.argsort(h)[-5:]
                for i in top_idx:
                    # Find a low-h position far from i
                    for j in [0, n//4, n//2, 3*n//4, n-1]:
                        if j == i or abs(j-i) < m:
                            continue
                        if h[i] > 0.5 and h[j] < 0.5:
                            dd = min(d_base * 0.3, (h[i]-0.5)*0.5)
                            if dd > 1e-10:
                                best_move = (i, j, dd)
                                break
                    if best_move:
                        break
            
            if best_move is None:
                break
                
            a, b, dd = best_move
            h[a] -= dd
            h[b] += dd
            h = np.clip(h, 0, 1)
            # Recompute c (expensive but necessary for correctness)
            c = np.correlate(h, 1 - h, mode='full') * dx
            new_max = c.max()
            if new_max < best_c:
                best_c = new_max
                best_h = h.copy()
            else:
                # Undo
                h[a] += dd
                h[b] -= dd
                h = np.clip(h, 0, 1)
                c = np.correlate(h, 1 - h, mode='full') * dx
    
    return best_h, best_c

Actually, I realize I should also try a **CVXPY-based approach** that's different from the outer approximation. Specifically, I can use the fact that for a fixed set of "active" lags, the problem is:

minimize t
subject to:
  ∑_i h[i](1-h[i+k_j]) ≤ t/dx for each active lag k_j
  ∑ h = n/2
  0 ≤ h ≤ 1

The constraint ∑_i h[i](1-h[i+k]) ≤ t/dx is equivalent to:
∑_i h[i] - ∑_i h[i]h[i+k] ≤ t/dx

The term ∑_i h[i]h[i+k] can be written as h^T P_k^T h where P_k is the appropriate permutation/shift matrix. This is a quadratic constraint.

In CVXPY, we can express this as:
cp.quad_form(h, P_k) >= sum_part - t/dx

Wait, but quad_form in CVXPY requires the matrix to be PSD for the constraint to be convex. The shift matrix is not PSD.

However! There's a trick. We can write:
h[i]*h[i+k] ≤ (h[i]² + h[i+k]²)/2 (by AM-GM)

So ∑_i h[i]*h[i+k] ≤ ∑_i (h[i]² + h[i+k]²)/2

This gives us a valid upper bound, but not tight.

Alternatively, we can use the fact that for 0 ≤ h ≤ 1:
h[i]*(1-h[i+k]) ≤ max(h[i], 1-h[i+k]) ≤ 1

This is too loose.

Let me try another approach: **linearization at the current point**. For a fixed h₀, we can linearize:
h[i]h[i+k] ≈ h₀[i]h₀[i+k] + h₀[i+k](h[i]-h₀[i]) + h₀[i](h[i+k]-h₀[i+k])

This gives a linear constraint, which is what the outer approximation does. But we can do it more efficiently by only considering the top few active lags.

        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    if time.time() < deadline - 30:
        sa_dl = min(deadline - 20, time.time() + 120)
        h_new, c_new = sa_local_search(best_h, n, sa_dl, rng, max_rounds=500000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    if time.time() < deadline - 15:
        h_new, c_new = greedy_local_search(best_h, n, min(deadline - 10, time.time() + 60))
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    if time.time() < deadline - 10:
        h_new = slsqp(best_h, n, alpha=200000, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new

        h_new, c_new = greedy_local_search(best_h, n, deadline - 1)

            best_c = c_new
            best_h = h_new

    best_c = compute_c5(best_h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    
    best_h = None
    best_c = float('inf')
    
    # Phase 1: Optimize at initial n
    alloc1 = min(350, (deadline - t0) * 0.4)
    h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt, c_opt
    
    # Phase 2: Try other n values
    n_candidates = [250, 200, 350, 400]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 80:
            break
        alloc = min(120, remaining * 0.35)
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Random restarts at best n
    n_best = len(best_h)
    if time.time() < deadline - 100:
        remaining = deadline - time.time() - 50
        n_restarts = max(1, int(remaining / 90))
        for r in range(n_restarts):
            if time.time() > deadline - 60:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(80, (deadline - time.time() - 40) * 0.4)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_at_n(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 4: Final polishing
    n_final = len(best_h)
    if time.time() < deadline - 30:
        h_new = slsqp(best_h, n_final, alpha=500000, maxiter=4000)
        c_new = compute_c5(h_new, n_final)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    if time.time() < deadline - 15:
        h_new, c_new = sa_local_search(best_h, n_final, min(deadline - 5, time.time() + 60), rng, max_rounds=200000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    if time.time() < deadline - 5:
        h_new, c_new = greedy_local_search(best_h, n_final, deadline - 2)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
