def lbfgs_objective(h_flat, n, alpha, mu):
    h = h_flat.reshape(-1)
    # clip for safety (L-BFGS-B should respect bounds, but just in case)
    h = np.clip(h, 0, 1)
    c = np.correlate(h, 1-h, mode='full') * (2.0/n)
    # smooth max
    shifted = alpha * (c - c.max())
    exps = np.exp(shifted)
    p = exps / exps.sum()
    F = (c.max() + np.log(exps.sum()) / alpha) * 1.0  # = (1/alpha)*log(sum exp(alpha*c))
    # Actually: (1/alpha)*log(sum exp(alpha*c)) = c.max() + (1/alpha)*log(sum exp(alpha*(c-c.max())))
    F = c.max() + np.log(exps.sum()) / alpha
    # penalty
    penalty = mu * (h.sum() - n/2.0)**2
    grad = compute_grad(h, n, alpha, p) + 2*mu*(h.sum()-n/2.0)*np.ones(n)
    return F + penalty, grad

    val = c_max + np.log(exps.sum()) / alpha
    
    # Gradient: d/dh[j] of smooth_max
    grad = np.zeros(n)
    omh = 1.0 - h
    for j in range(n):
        # Part 1: +dx * sum_l p[l+n-1] * (1-h[j-l]) for 0 <= j-l < n
        lo1 = max(j - n + 1, -(n-1))
        hi1 = min(j, n-1)
        if lo1 <= hi1:
            ls = np.arange(lo1, hi1+1)
            ms = j - ls
            grad[j] += dx * np.dot(p[ls + n - 1], omh[ms])
        # Part 2: -dx * sum_l p[l+n-1] * h[j+l] for 0 <= j+l < n
        lo2 = max(-j, -(n-1))
        hi2 = min(n-1-j, n-1)
        if lo2 <= hi2:
            ls2 = np.arange(lo2, hi2+1)
            ms2 = j + ls2
            grad[j] -= dx * np.dot(p[ls2 + n - 1], h[ms2])
    return val, grad

def lbfgs_solve(h, n, alpha, mu, maxiter=1000):
    """L-BFGS-B minimization with penalty for constraint."""
    dx = 2.0 / n
    
    def objective(x):
        x = np.clip(x, 0.0, 1.0)
        val, grad = grad_smooth_max(x, n, alpha)
        penalty = mu * (x.sum() - n/2.0)**2
        grad_pen = 2.0 * mu * (x.sum() - n/2.0)
        return val + penalty, grad + grad_pen
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_ls(h_init, n, deadline):
    """First-improvement binary local search."""
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    ones = list(np.where(h == 1)[0].astype(int))
    zeros = list(np.where(h == 0)[0].astype(int))
    
    for _ in range(50000):
        if time.time() > deadline:
            break
        improved = False
        for zi in range(len(zeros)):
            if time.time() > deadline:
                break
            i = zeros[zi]
            for oi in range(len(ones)):
                j = ones[oi]
                delta = np.zeros(2*n-1)
                im = i + m_range
                m = (im >= 0) & (im < n)
                delta[m] += 1.0 - h[im[m]]
                imn = i - m_range
                m = (imn >= 0) & (imn < n)
                delta[m] -= h[imn[m]]
                jm = j + m_range
                m = (jm >= 0) & (jm < n)
                delta[m] -= 1.0 - h[jm[m]]
                jmn = j - m_range
                m = (jmn >= 0) & (jmn < n)
                delta[m] += h[jmn[m]]
                delta[n-1] -= 2.0
                delta[n-1+j-i] += 1.0
                delta[n-1+i-j] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-14:
                    h[i] = 1.0; h[j] = 0.0
                    corr += delta
                    cur_c = nc
                    ones.pop(oi); ones.append(i)
                    zeros.pop(zi); zeros.append(j)
                    improved = True
                    if cur_c < best_c:
                        best_c, best_h = cur_c, h.copy()
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def anneal_bin(h_init, n, deadline, seed=42):
    """Simulated annealing for binary."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = 0.008
    cool = 0.99995
    scale = 500.0
    
    while time.time() < deadline:
        i = int(zeros[rng.randint(len(zeros))])
        j = int(ones[rng.randint(len(ones))])
        delta = np.zeros(2*n-1)
        im = i + m_range
        m = (im >= 0) & (im < n)
        delta[m] += 1.0 - h[im[m]]
        imn = i - m_range
        m = (imn >= 0) & (imn < n)
        delta[m] -= h[imn[m]]
        jm = j + m_range
        m = (jm >= 0) & (jm < n)
        delta[m] -= 1.0 - h[jm[m]]
        jmn = j - m_range
        m = (jmn >= 0) & (jmn < n)
        delta[m] += h[jmn[m]]
        delta[n-1] -= 2.0
        delta[n-1+j-i] += 1.0
        delta[n-1+i-j] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(scale*(cur_c-nc)/T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        T *= cool
    return best_h, best_c
