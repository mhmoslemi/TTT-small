    d[n-1] -= 2.0
    k1 = n-1 + (j-i)
    d[k1] += 1.0
    k2 = n-1 + (i-j)
    d[k2] += 1.0
    return d

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greed_binary(h_init, n, deadline, max_sweeps=300):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = [int(x) for x in np.where(h == 1)[0]]
    zeros = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        i = int(rng_greed.randint(len(zeros)))  # Random order to avoid bias
        # Actually let's just iterate in order for determinism
        for ii in range(len(zeros)):
            i = zeros[ii]
            if time.time() > deadline:
                break
            for j in ones:
                d = compute_swap_delta(h, i, j, n)
                nc = (corr + d).max() * dx
                if nc < cur_c - 1e-14:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur_c = nc
                    ones.remove(j)
                    zeros[ii] = j
                    zeros.append(i)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = 0.01
    # Cool from T to T*cool^iters
    # We want T to reach ~0.001 after ~2M iterations
    # 0.001 = 0.01 * cool^2000000 => cool = (0.1)^(1/2000000) ≈ 0.9999988
    cool = 0.9999988
    scale = 8000.0
    while time.time() < deadline:
        ai = rng.randint(len(zeros))
        bi = rng.randint(len(ones))
        i = zeros[ai]
        j = ones[bi]
        d = compute_swap_delta(h, i, j, n)
        nc = (corr + d).max() * dx
        if nc <= cur_c:
            accept = True
        elif T > 1e-12:
            accept = rng.random() < np.exp(-scale * (nc - cur_c) / T)
        else:
            accept = False
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr += d
            cur_c = nc
            zeros[ai] = j
            ones[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_swap_delta(h, i, j, n):
    mr = np.arange(-(n - 1), n)
    d = np.zeros(2 * n - 1)
    im = i + mr
    m = (im >= 0) & (im < n)
    d[m] += 1.0 - h[im[m]]
    imn = i - mr
    m = (imn >= 0) & (imn < n)
    d[m] -= h[imn[m]]
    jm = j + mr
    m = (jm >= 0) & (jm < n)
    d[m] -= 1.0 - h[jm[m]]
    jmn = j - mr
    m = (jmn >= 0) & (jmn < n)
    d[m] += h[jmn[m]]
    d[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    d[k1] += 1.0
    k2 = n - 1 + (i - j)
    d[k2] += 1.0
    return d

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greed_binary(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = [int(x) for x in np.where(h == 1)[0]]
    zeros = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for ii in range(len(zeros)):
            i = zeros[ii]
            if time.time() > deadline:
                break
            for jj in range(len(ones)):
                j = ones[jj]
                d = compute_swap_delta(h, i, j, n)
                nc = (corr + d).max() * dx
                if nc < cur_c - 1e-14:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur_c = nc
                    ones[jj] = i
                    zeros[ii] = j
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, T_start=0.005, T_end=0.0001):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    est_iters = 3000000
    cool = (T_end / T_start) ** (1.0 / est_iters)
    T = T_start
    while time.time() < deadline:
        ai = rng.randint(len(zeros))
        bi = rng.randint(len(ones))
        i = zeros[ai]
        j = ones[bi]
        d = compute_swap_delta(h, i, j, n)
        nc = (corr + d).max() * dx
        if nc <= cur_c:
            accept = True
        elif T > T_end:
            accept = rng.random() < np.exp(-(nc - cur_c) / T)
        else:
            accept = False
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr += d
            cur_c = nc
            zeros[ai] = j
            ones[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def perturb_binary(h, n, k, rng):
    h = h.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    for _ in range(k):
        if not ones or not zeros:
            break
        i = rng.choice(list(zeros))
        j = rng.choice(list(ones))
        h[i], h[j] = 1.0, 0.0
        zeros.discard(i)
        zeros.add(j)
        ones.discard(j)
        ones.add(i)
    return h
