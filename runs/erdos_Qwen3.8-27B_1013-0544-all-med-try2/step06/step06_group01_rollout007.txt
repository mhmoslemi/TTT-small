def smooth_max(c, alpha):
    """Smooth approximation of max(c)."""
    shifted = c - c.max()
    m = c.max()
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha

def fourier_to_grid(coeffs, n, K):
    """Convert Fourier coefficients to grid values.
    coeffs: array of length 2*K = [a1, b1, a2, b2, ..., aK, bK]
    Returns h on [0, 2] with n points (endpoint=False)."""
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(K):
        freq = (k + 1) * np.pi
        h += coeffs[2*k] * np.cos(freq * x)
        h += coeffs[2*k+1] * np.sin(freq * x)
    return h

def grid_to_fourier(h, n, K):
    """Fit Fourier coefficients to h via least squares."""
    x = np.linspace(0, 2.0, n, endpoint=False)
    A = np.zeros((n, 2*K))
    for k in range(K):
        freq = (k + 1) * np.pi
        A[:, 2*k] = np.cos(freq * x)
        A[:, 2*k+1] = np.sin(freq * x)
    # Least squares: minimize ||A @ coeffs - (h - 0.5)||^2
    rhs = h - 0.5
    coeffs, _, _, _ = np.linalg.lstsq(A, rhs, rcond=None)
    return coeffs

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / max(s, 1e-14) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / max(s, 1e-14) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = c - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha

def fourier_eval(coeffs, n, K):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(K):
        f = (k + 1) * np.pi
        h += coeffs[2*k] * np.cos(f * x)
        h += coeffs[2*k+1] * np.sin(f * x)
    return h

def fit_fourier(h, n, K):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    A = np.zeros((n, 2*K))
    for k in range(K):
        f = (k + 1) * np.pi
        A[:, 2*k] = np.cos(f * x)
        A[:, 2*k+1] = np.sin(f * x)
    coeffs, _, _, _ = np.linalg.lstsq(A, h - 0.5, rcond=None)
    return coeffs

def optimize_fourier(coeffs, n, K, alpha, maxiter):
    dx = 2.0 / n
    def obj(c):
        h = fourier_eval(c, n, K)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        sm = smooth_max(corr, alpha)
        # Penalty for bounds violation
        pen = 50.0 * (np.sum(np.maximum(0, -h)**2) + np.sum(np.maximum(0, h - 1)**2))
        # Integral constraint
        iv = h.sum() * dx - 1.0
        pen += 50.0 * iv * iv
        return sm + pen
    try:
        res = minimize(obj, coeffs, method='L-BFGS-B',
                      bounds=[(-0.5, 0.5)] * (2*K),
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except:
        return coeffs

def slsqp_grid(h, n, alpha, maxiter):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx

    def eqc(x):
        return x.sum() - target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eqc}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})

    except:
        return h
