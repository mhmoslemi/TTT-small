def solve_1d_convex(f_base, a, lo, hi):
    """Solve min_{u in [lo,hi]} max_k (f_base[k] + u * a[k])"""
    # Binary search for the optimal u
    # At the optimum, the max of the increasing group equals the max of the decreasing group
    for _ in range(60):  # enough for double precision
        mid = (lo + hi) / 2
        vals = f_base + mid * a
        m = np.max(vals)
        # Subgradient: if the max is achieved by a positive-slope function, we should decrease u
        # If by a negative-slope function, increase u
        # More precisely: the subdifferential of max_k(f_k + u*a_k) at u contains
        # conv{a_k : f_k + u*a_k = max}
        active = np.where(np.abs(vals - m) < 1e-12)[0]

        min_a = a[active].min()
        max_a = a[active].max()
        if min_a > 0:

        elif max_a < 0:
            lo = mid  # all active slopes negative, increase u
        else:
            # 0 is in the subdifferential, we're at optimum

    u_opt = (lo + hi) / 2

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, deadline, rng, max_sweeps=100000):
    h = h.copy()
    total = 2*n - 1
    km = np.arange(total) - (n-1)
    c_raw = np.correlate(h, 1-h, mode='full')
    best_c = c_raw.max()
    best_h = h.copy()
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:

            mask_fwd = (fwd_idx >= 0) & (fwd_idx < n)
            bwd_idx = j - km
            mask_bwd = (bwd_idx >= 0) & (bwd_idx < n)
            
            a = np.ones(total)
            if mask_fwd.any():
                a[mask_fwd] -= h[fwd_idx[mask_fwd]]
            if mask_bwd.any():
                a[mask_bwd] -= h[bwd_idx[mask_bwd]]

            hi = 1.0 - h[j]
            for _ in range(50):
                mid = (lo + hi) * 0.5
                vals = c_raw + mid * a
                m = vals.max()
                tol = 1e-11 * max(1.0, abs(m))

                t = 0.0
            elif t > 1.0:
                t = 1.0
            if abs(t - h[j]) > 1e-12:
                h[j] = t
                c_raw += u * a

            stall = 0

    c_best = compute_c5(h, n)
    attempts = 0
    while time.time() < deadline and attempts < 50:
        attempts += 1
        h_try = h_best.copy()

        idx = rng.choice(n, size=n_pert, replace=False)

        if c_r < c_best:
            c_best = c_r
            h_best = h_r
    return h_best, c_best

            for _ in range(50):
                mid = (lo + hi) * 0.5
                vals = c_raw + mid * a
                m = vals.max()
                tol = 1e-11 * max(1.0, abs(m))
                active = np.where(np.abs(vals - m) <= tol)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(vals)])
                if a[active].min() > 1e-13:
                    hi = mid
                elif a[active].max() < -1e-13:
                    lo = mid
                else:
                    break
            u = (lo + hi) * 0.5
            t = h[j] + u
            if t < 0.0:
                t = 0.0
            elif t > 1.0:
                t = 1.0
            if abs(t - h[j]) > 1e-12:
                h[j] = t
                c_raw += u * a
                improved = True
        if sweep % 50 == 0:
            c_raw = np.correlate(h, 1 - h, mode='full')
        cur_best = c_raw.max()
        if cur_best < best_c - 1e-11:
            best_c = cur_best
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if not improved:
            break
        if stall > 500:
            break
    return best_h, best_c * (2.0 / n)

def pair_refine(h, n, deadline, rng, radius=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    total = 2 * n - 1
    km = np.arange(total) - (n - 1)
    c_raw = np.correlate(h, 1 - h, mode='full')
    improved_rounds = 0
    for _ in range(200):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                return h, best_c
            lo_i = max(0, i - radius)
            hi_i = min(n, i + radius + 1)
            for j in range(lo_i, hi_i):
                if i == j:
                    continue
                fwd_i = i + km
                mf_i = (fwd_i >= 0) & (fwd_i < n)
                bw_i = i - km
                mb_i = (bw_i >= 0) & (bw_i < n)
                a1 = np.ones(total)
                if mf_i.any():
                    a1[mf_i] -= h[fwd_i[mf_i]]
                if mb_i.any():
                    a1[mb_i] -= h[bw_i[mb_i]]
                fwd_j = j + km
                mf_j = (fwd_j >= 0) & (fwd_j < n)
                bw_j = j - km
                mb_j = (bw_j >= 0) & (bw_j < n)
                a2 = np.ones(total)
                if mf_j.any():
                    a2[mf_j] -= h[fwd_j[mf_j]]
                if mb_j.any():
                    a2[mb_j] -= h[bw_j[mb_j]]
                lo1, hi1 = -h[i], 1.0 - h[i]
                lo2, hi2 = -h[j], 1.0 - h[j]
                best_uv = None
                best_val = c_raw.max()
                for u in np.linspace(lo1, hi1, 7):
                    c_temp = c_raw + u * a1
                    lo_map = max(lo2, lo1 + u - u)
                    for v in np.linspace(lo2, hi2, 7):
                        vals = c_temp + v * a2
                        m = vals.max()
                        if m < best_val - 1e-13:
                            best_val = m
                            best_uv = (u, v)
                if best_uv is not None:
                    u, v = best_uv
                    h[i] = np.clip(h[i] + u, 0, 1)
                    h[j] = np.clip(h[j] + v, 0, 1)
                    c_raw = np.correlate(h, 1 - h, mode='full')
                    if c_raw.max() < best_c - 1e-12:
                        best_c = c_raw.max()
                        improved = True
                        improved_rounds += 1
        if not improved:
            break
    return h, best_c * (2.0 / n)

def perturb_refine(h, n, deadline, rng):
    h_best = h.copy()
    c_best = compute_c5(h, n)
    attempts = 0
    while time.time() < deadline and attempts < 80:
        attempts += 1
        h_try = h_best.copy()
        n_pert = rng.randint(3, 12)
        idx = rng.choice(n, size=n_pert, replace=False)
        for i in idx:
            h_try[i] = rng.uniform(0, 1)
        h_try = project_feasible(h_try, n)
        alloc = min(25, deadline - time.time())
        if alloc < 8:
            break
        h_r, c_r = coord_descent(h_try, n, time.time() + alloc, rng, max_sweeps=3000)
        if c_r < c_best:
            c_best = c_r
            h_best = h_r
    return h_best, c_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    h = project_feasible(h_init, n_init)
    best_h = h.copy()
    best_c = compute_c5(h, n_init)
    best_n = n_init
    t1 = min(t0 + 350, deadline - 450)
    h, c = coord_descent(h, n_init, t1, rng)
    if c < best_c:
        best_c = c
        best_h = h.copy()
        best_n = n_init
    if time.time() < deadline - 350:
        t2 = min(time.time() + 130, deadline - 350)
        h_p, c_p = perturb_refine(best_h, best_n, t2, rng)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p.copy()
    if time.time() < deadline - 250:
        t3 = min(time.time() + 100, deadline - 250)
        h_pr, c_pr = pair_refine(best_h, best_n, t3, rng)
        if c_pr < best_c:
            best_c = c_pr
            best_h = h_pr.copy()
    for n_try in [200, 250]:
        if time.time() > deadline - 180:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(100, deadline - time.time() - 80)
        if alloc < 30:
            break
        h_t, c_t = coord_descent(h_interp, n_try, time.time() + alloc, rng)
        if time.time() < deadline - 40:
            h_t2, c_t2 = pair_refine(h_t, n_try, min(time.time() + 30, deadline - 40), rng)
            if c_t2 < c_t:
                h_t, c_t = h_t2, c_t2
        if c_t < best_c:
            best_c = c_t
            best_h = h_t.copy()
            best_n = n_try
    if time.time() < deadline - 70:
        n_rb = 200
        h_bin = np.zeros(n_rb)
        ones = rng.choice(n_rb, size=n_rb // 2, replace=False)
        h_bin[ones] = 1.0
        alloc = min(60, deadline - time.time() - 30)
        if alloc > 30:
            h_r, c_r = coord_descent(h_bin, n_rb, time.time() + alloc, rng)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r.copy()
                best_n = n_rb
    if time.time() < deadline - 5:
        h_f, c_f = coord_descent(best_h, best_n, deadline, rng, max_sweeps=30000)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
