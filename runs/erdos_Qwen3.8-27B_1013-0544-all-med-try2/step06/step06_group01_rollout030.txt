import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            s = (1.0 - h).sum()
            if s < 1e-12:
                break
            h = h + diff / s * (1.0 - h)
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()

    zeros = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_nc = cur_c
        best_pair = None
        best_delta = None
        len_zeros = len(zeros)
        len_ones = len(ones)
        
        for a in range(len_zeros):
            i = zeros[a]
            im = i + np.arange(-(n-1), n)
            imn = i - np.arange(-(n-1), n)
            msk_im = (im >= 0) & (im < n)
            msk_imn = (imn >= 0) & (imn < n)
            di = np.zeros(2*n-1)
            di[msk_im] += 1.0 - h[im[msk_im]]
            di[msk_imn] -= h[imn[msk_imn]]
            
            for b in range(len_ones):
                j = ones[b]
                jm = j + np.arange(-(n-1), n)
                jmn = j - np.arange(-(n-1), n)
                msk_jm = (jm >= 0) & (jm < n)
                msk_jmn = (jmn >= 0) & (jmn < n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            s = (1.0 - h).sum()
            if s < 1e-12:
                break
            h = h + diff / s * (1.0 - h)
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
