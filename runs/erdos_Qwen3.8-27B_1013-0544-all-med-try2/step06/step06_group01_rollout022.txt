        # Corrections

        if 0 <= k1 < L: delta[k1] += 1
        k2 = n - 1 + (i - j)
        if 0 <= k2 < L: delta[k2] += 1

            h[j] = 0.0

            cur_c = new_c

                best_h = h.copy()

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h_init, n):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    return h

def greedy_binary(h, n, deadline, max_sweeps=10000):
    dx = 2.0 / n
    L = 2 * n - 1
    m_arr = np.arange(L)
    best_c = np.inf
    best_h = h.copy()
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        cur_c = corr.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        nz, no = len(zeros), len(ones)
        if nz == 0 or no == 0:
            break
        
        # Compute dz: delta for setting each zero to 1
        a = zeros[:, None] + m_arr[None, :] - (n - 1)
        va = (a >= 0) & (a < n)
        t1 = np.where(va, 1 - h[np.clip(a, 0, n-1)], 0.0)
        b = zeros[:, None] - m_arr[None, :] + (n - 1)
        vb = (b >= 0) & (b < n)
        t2 = np.where(vb, -h[np.clip(b, 0, n-1)], 0.0)
        dz = t1 + t2  # (nz, L)
        
        # Compute do: delta for setting each one to 0
        a2 = ones[:, None] + m_arr[None, :] - (n - 1)
        va2 = (a2 >= 0) & (a2 < n)
        t3 = np.where(va2, -(1 - h[np.clip(a2, 0, n-1)]), 0.0)
        b2 = ones[:, None] - m_arr[None, :] + (n - 1)
        vb2 = (b2 >= 0) & (b2 < n)
        t4 = np.where(vb2, h[np.clip(b2, 0, n-1)], 0.0)
        do = t3 + t4  # (no, L)
        
        # Find best swap
        best_score = cur_c - 1e-15
        best_ii, best_jj = -1, -1
        
        for ii in range(nz):
            base = corr + dz[ii]  # (L,)
            # scores[jj] = max(base + do[jj])
            scores = (base[None, :] + do).max(axis=1) * dx
            jj = scores.argmin()
            if scores[jj] < best_score:
                # Verify with corrections
                i_pos = zeros[ii]
                j_pos = ones[jj]
                true = base + do[jj].copy()
                true[n-1] -= 2
                k1 = n - 1 + (j_pos - i_pos)
                if 0 <= k1 < L: true[k1] += 1
                k2 = n - 1 + (i_pos - j_pos)
                if 0 <= k2 < L: true[k2] += 1
                tv = true.max() * dx
                if tv < best_score:
                    best_score = tv
                    best_ii, best_jj = ii, jj
        if best_ii < 0:
            break
        h[zeros[best_ii]] = 1.0
        h[ones[best_jj]] = 0.0
    
    return best_h, best_c
