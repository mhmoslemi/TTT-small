def subgradient_step(h, n, c, top_m=5, step_size=0.01):
    """One subgradient step for minimizing max_k c[k]."""
    dx = 2.0 / n
    # Find top-m lags
    top_idx = np.argsort(c)[-top_m:]
    top_idx = top_idx[::-1]  # descending
    # Weight by softmax of top values
    c_top = c[top_idx]
    w = np.exp(100 * (c_top - c_top.max()))
    w /= w.sum()
    
    # Compute weighted gradient
    # c[k] = sum_i h[i+k] * (1-h[i]) for valid i
    # In the full correlation, index offset: c_full[j] corresponds to lag j-(n-1)
    n_full = 2*n - 1
    grad = np.zeros(n)
    for m_idx, lag_full in enumerate(top_idx):
        k = lag_full - (n - 1)  # actual shift
        # ∂c[k]/∂h[j] = (1-h[j-k]) - h[j+k] (with bounds checking)
        for j in range(n):
            g = 0.0
            if 0 <= j - k < n:
                g += (1.0 - h[j - k])
            if 0 <= j + k < n:
                g -= h[j + k]
            grad[j] += w[m_idx] * g
    
    # Descent direction: -grad
    # Take step
    h_new = h - step_size * grad
    return h_new

A better approach: after SLSQP, do a "max-reduction" step:

2. Compute the exact gradient of c_{k*}
3. Take a small step in the -gradient direction
4. Repeat until the max doesn't decrease

```python
def exact_max_reduction(h, n, deadline, max_iter=10000):
    """Directly reduce the max using exact subgradient."""
    h = h.copy()
    best_c, _ = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.001
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_val, c_arr = compute_c5(h, n)
        if c_val < best_c - 1e-15:
            best_c = c_val
            best_h = h.copy()
        
        # Find all lags within 1% of max
        threshold = c_arr.max() * 0.99
        active = np.where(c_arr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_arr)])
        
        # Compute average gradient over active lags
        offset = n - 1
        grad = np.zeros(n)
        n_active = len(active)
        for k_idx in active:
            k = k_idx - offset
            if k >= 0:
                nk = n - k
                if nk > 0:
                    grad[k:] += (1.0 - h[:nk])
                    grad[:nk] -= h[k:]
            else:
                ak = -k
                nak = n - ak
                if nak > 0:
                    grad[:nak] += (1.0 - h[ak:])
                    grad[ak:] -= h[:nak]
        grad /= n_active
        
        # Take step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Check if it helped
        c_new, _ = compute_c5(h_new, n)
        if c_new < c_val:
            h = h_new
            lr = min(lr * 1.05, 0.01)  # slightly increase step
        else:
            lr *= 0.7  # decrease step
        
        if lr < 1e-8:
            break
    
    return best_h, best_c
```
