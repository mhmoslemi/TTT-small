def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):

        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:

    result = np.clip(h - t, 0.0, 1.0)

def slsqp_power(h_init, n, p, maxiter, deadline):
    dx = 2.0 / n
    
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        gm = g.max()
        if gm < 1e-15:
            return 0.0
        r = g / gm
        return gm * np.sum(r ** p) ** (1.0 / p)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons,
