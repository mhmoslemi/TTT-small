def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return project_feasible(h, n), compute_c5(project_feasible(h, n), n)
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j: continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return project_feasible(h, n), compute_c5(project_feasible(h, n), n)

def lbfgs_quick(h, n, deadline, alpha=1000, mu=50.0):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return obj_pen(x, n, alpha, mu)
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for mi in range(len(w)):
            if w[mi] < 1e-13: continue
            k = mi - (n - 1)
            wm = w[mi] * dx
            if k >= 0:
                nv = n - k
                g[0:nv] += wm * (1.0 - x[k:n])
                g[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += wm * (1.0 - x[0:nv])
                g[0:nv] += wm * (-x[kk:n])
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-12})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(300, 0.5)
        n = 300

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    print(f"Initial: C5={best_c:.6f}, n={n}")

    # Phase 1: Quick L-BFGS to polish
    if time.time() < deadline - 600:
        t_alloc = min(80.0, deadline - time.time() - 600)
        h1, c1 = lbfgs_quick(h, n, time.time() + t_alloc, alpha=5000, mu=50.0)
        if c1 < best_c:
            best_c, best_h = c1, h1
        print(f"After L-BFGS: C5={best_c:.6f}")

    # Phase 2: Coordinate descent (main workhorse)
    if time.time() < deadline - 400:
        t_alloc = min(450.0, deadline - time.time() - 400)
        h2, c2 = coord_descent(best_h, n, time.time() + t_alloc, seed=seed)
        if c2 < best_c:
            best_c, best_h = c2, h2
        print(f"After coord descent: C5={best_c:.6f}")

    # Phase 3: Simulated annealing
    if time.time() < deadline - 200:
        t_alloc = min(150.0, deadline - time.time() - 200)
        h3, c3 = sim_annealing(best_h, n, time.time() + t_alloc, seed=seed + 1)
        if c3 < best_c:
            best_c, best_h = c3, h3
        print(f"After SA: C5={best_c:.6f}")

    # Phase 4: More coordinate descent after SA
    if time.time() < deadline - 100:
        t_alloc = min(120.0, deadline - time.time() - 100)
        h4, c4 = coord_descent(best_h, n, time.time() + t_alloc, seed=seed + 2)
        if c4 < best_c:
            best_c, best_h = c4, h4
        print(f"After 2nd coord descent: C5={best_c:.6f}")

    # Phase 5: Pair search polish
    if time.time() < deadline - 20:
        t_alloc = min(60.0, deadline - time.time() - 15)
        h5, c5 = pair_search(best_h, n, time.time() + t_alloc,
                             [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=50)
        if c5 < best_c:
            best_c, best_h = c5, h5
        print(f"After pair search: C5={best_c:.6f}")

    # Phase 6: Try different resolution
    if time.time() < deadline - 15:
        for n_try in [250, 350]:
            if time.time() > deadline - 10:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            t_rem = deadline - time.time() - 5
            if t_rem < 20:
                break
            h_t, c_t = coord_descent(h_interp, n_try, time.time() + t_rem * 0.7, seed=seed + 10)
            if c_t < best_c:
                best_c, best_h = c_t, h_t
                n = n_try
                print(f"Better at n={n_try}: C5={best_c:.6f}")

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    print(f"Final: C5={best_c:.6f}, n={len(best_h)}")
    return best_h, best_c, len(best_h)
