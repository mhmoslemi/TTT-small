def subgrad_descent(h, n, deadline, max_iter=10000, tol=1e-10):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 1.0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        # Find active set (k's within tol of max)
        active = np.where(g >= m - 1e-6)[0]
        # Average subgradient over active set
        grad = np.zeros(n)
        for k in active:
            grad += analytical_subgrad(h, n, int(k))
        grad /= len(active)
        
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        
        # Backtracking linesearch
        step = lr
        improved = False
        for tries in range(30):
            h_new = project_onto_box_sum(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < m - 1e-12:
                h = h_new
                improved = True
                break
            step *= 0.5
        if not improved:

            if lr < 1e-12:
                break
        else:
            lr = min(lr * 1.1, 5.0)
            c = compute_c5(h, n)
            if c < best_c:

    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

For a fixed k, g(k) = ∑_i h[i]*(1-h[i+j])*dx where j = n-1-k.
If we swap h[i]=0 to 1 and h[j']=1 to 0:
- Effect on g(k): 
  - i was 0, now 1: adds (1-h[i+j])*dx to g(k)
  - j' was 1, now 0: removes (1-h[j'+j])*dx from g(k)
  - Also, i might be the "second factor" for some term: h[m]*(1-h[m+j]) where m+j = i, i.e., m = i-j. If h[i-j] = 1, then the term h[i-j]*(1-h[i]) was (1-0)=1, now becomes (1-1)=0. So we lose h[i-j]*dx = 1*dx from g(k).
  - Similarly, j' might be the second factor: h[m]*(1-h[m+j]) where m+j = j', i.e., m = j'-j. If h[j'-j] = 1, the term was h[j'-j]*(1-h[j']) = 1*(1-1) = 0, now becomes 1*(1-0) = 1. So we gain dx.

So for a swap of (i: 0→1, j': 1→0), the change in g(k) is:
Δg(k) = dx * [(1-h[i+j]) - (1-h[j'+j]) - (h[i-j] if valid) + (h[j'-j] if valid)]

Where all indices must be valid (in [0, n-1]).

This is O(1) per swap for a fixed k! So we can evaluate all O(n²) swaps in O(n²) for a fixed k.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_onto_box_sum(h, n):
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return np.clip(h, 0, 1)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        val = np.clip(h - mid, 0, 1).sum()
        if val > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def analytical_subgrad(h, n, k):
    j = n - 1 - k
    dx = 2.0 / n
    grad = np.zeros(n)
    if j >= 0:
        v1 = np.arange(0, n - j)
        grad[v1] += (1.0 - h[v1 + j])
        v2 = np.arange(j, n)
        grad[v2] -= h[v2 - j]
    else:
        jj = -j
        v1 = np.arange(jj, n)
        grad[v1] += (1.0 - h[v1 - jj])
        v2 = np.arange(0, n - jj)
        grad[v2] -= h[v2 + jj]
    return grad * dx

def multi_subgrad(h, n, k_list):
    grad = np.zeros(n)
    for k in k_list:
        grad += analytical_subgrad(h, n, k)
    return grad / len(k_list)

def subgrad_descent(h, n, deadline, max_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        active = np.where(g >= m - 1e-8)[0]
        if len(active) == 0:
            active = np.array([np.argmax(g)])
        grad = multi_subgrad(h, n, active)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        step = lr
        improved = False
        for _ in range(40):
            h_new = project_onto_box_sum(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < m - 1e-13:
                h = h_new
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.7
            if lr < 1e-12:
                break
        else:
            lr = min(lr * 1.05, 3.0)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_onto_box_sum(res.x, n)
    except Exception:
        return h

def binary_local_search(h, n, deadline, max_sweeps=500):
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]

        if time.time() > deadline:
            break

        best_swap = None
        best_swap_c = best_c
        z_list = list(zeros)
        o_list = list(ones)
        for i in z_list:
            for j in o_list:

                h_bin[j] = 0.0

            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            improved = True
        if not improved:

    return h_bin, best_c

        need = target - len(ones)
        fill = rng.choice(zeros, need, replace=False)
        h[fill] = 1.0
    return h

        zeros = np.where(h == 0)[0]

            h[rem] = 0.0
        else:
            fill = rng.choice(zeros, target - curr, replace=False)
            h[fill] = 1.0
    return h
