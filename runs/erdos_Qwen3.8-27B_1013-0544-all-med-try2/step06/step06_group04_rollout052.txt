def compute_c5(h, n):

def compute_all_lags(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    m = abs(lag)
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            grad[:n-m] -= h[m:]
            grad[m:] += 1.0 - h[:n-m]

    idx = np.argsort(s)
    v = np.zeros(n)
    v[idx[:n//2]] = 1.0

def slsqp_optimize(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_lag_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    m = abs(lag)
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m < n:
        if lag > 0:
            grad[:n-m] -= h[m:]
            grad[m:] += 1.0 - h[:n-m]
        else:
            grad[:n-m] += 1.0 - h[m:]
            grad[m:] -= h[:n-m]
    return grad

def fw_oracle(s, n):
    idx = np.argsort(s)
    v = np.zeros(n)
    v[idx[:n // 2]] = 1.0
    return v

def frank_wolfe(h, n, max_iters=2000, deadline=None):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    avg_h = np.zeros(n)
    for t in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        grad = lag_gradient(h, n, lag)
        v = fw_oracle(grad, n)
        if np.max(np.abs(v - h)) < 1e-12:
            break
        gamma = 2.0 / (t + 2.0)
        accepted = False
        for _ in range(40):
            h_try = (1.0 - gamma) * h + gamma * v
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try <= best_c + 1e-13:

                avg_h = avg_h * (t / (t + 1)) + h / (t + 1)
                if c_try < best_c:
                    best_c = c_try
                    best_h = h.copy()
                accepted = True
                break
            gamma *= 0.5
            if gamma < 1e-8:
                break
        if not accepted:
            avg_h = avg_h * (t / (t + 1)) + h / (t + 1)
    c_avg = compute_c5(avg_h, n)
    if c_avg < best_c:
        best_h, best_c = avg_h, c_avg
    return best_h, best_c

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_solve(h, n, alpha, maxiter=4000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_slsqp(h, n, top_k, alpha, maxiter=4000):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_k:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_active_obj, h, args=(n, top_idx, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        for sweep in range(20):
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                break
            if lag > 0:
                for i in range(n - m):
                    if time.time() > deadline:
                        return h, best_c
                    j = i + m
                    diff = h[i] - h[j]
                    if abs(diff) < 1e-7:
                        continue
                    delta = min(d, abs(diff) * 0.4)
                    if h[i] > h[j]:
                        h[i] -= delta
                        h[j] += delta
                    else:
                        h[i] += delta
                        h[j] -= delta
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        if h[i] < h[j]:
                            h[i] += delta
                            h[j] -= delta
                        else:
                            h[i] -= delta
                            h[j] += delta
                        h = project_feasible(h, n)
            else:
                for i in range(m, n):
                    if time.time() > deadline:
                        return h, best_c
                    j = i + m
                    if j >= n:
                        break
                    diff = h[i] - h[j]
                    if abs(diff) < 1e-7:
                        continue
                    delta = min(d, abs(diff) * 0.4)
                    if h[i] > h[j]:
                        h[i] -= delta
                        h[j] += delta
                    else:
                        h[i] += delta
                        h[j] -= delta
                    h = project_feasible(h, n)
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                    else:
                        if h[i] < h[j]:
                            h[i] += delta
                            h[j] -= delta
                        else:
                            h[i] -= delta
                            h[j] += delta
                        h = project_feasible(h, n)
            if not improved:
                break
    return h, best_c

def optimize_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Frank-Wolfe
    if time.time() < deadline - 25:
        fw_dl = time.time() + min(80, (deadline - time.time()) * 0.30)
        h_fw, c_fw = frank_wolfe(best_h, n, max_iters=3000, deadline=fw_dl)
        if c_fw < best_c:
            best_c = c_fw
            best_h = h_fw
    # Phase 2: SLSQP sweep
    if time.time() < deadline - 20:
        for alpha in [2000, 10000, 50000, 200000, 1000000]:
            if time.time() > deadline - 20:
                break
            h_new = slsqp_solve(best_h, n, alpha, maxiter=4000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    # Phase 3: Active set SLSQP
    if time.time() < deadline - 15:
        for top_k in [5, 10, 15]:
            if time.time() > deadline - 15:
                break
            for alpha in [50000, 200000]:
                if time.time() > deadline - 15:
                    break
                h_new = active_slsqp(best_h, n, top_k, alpha, maxiter=3000)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
    # Phase 4: Final SLSQP
    if time.time() < deadline - 10:
        h_new = slsqp_solve(best_h, n, 5000000, maxiter=6000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 5: Local search
    if time.time() < deadline - 2:
        best_h, best_c = local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = min(380, (deadline - t0) * 0.38)
        h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt
    # Try other resolutions
    n_list = [200, 250, 300, 350, 400, 500]
    for n_try in n_list:
        if time.time() > deadline - 120:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 80:
            break
        alloc = min(120, remaining * 0.22)
        if best_h is not None:
            xb = np.linspace(0, 2, len(best_h))
            xn = np.linspace(0, 2, n_try)
            h_start = np.interp(xn, xb, best_h)
        elif h_init is not None:
            xi = np.linspace(0, 2, n_init)
            xn = np.linspace(0, 2, n_try)
            h_start = np.interp(xn, xi, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_n(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Perturbation restarts
    if best_h is not None and time.time() < deadline - 70:
        n = len(best_h)
        n_restarts = max(1, int((deadline - time.time() - 40) / 65))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.008 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(60, (deadline - time.time() - 35) * 0.35)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_solve(best_h, n, 10000000, maxiter=8000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h, best_c = local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
