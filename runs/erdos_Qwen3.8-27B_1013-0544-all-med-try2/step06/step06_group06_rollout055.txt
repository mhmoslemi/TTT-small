import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_correlation(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def hill_climb_binary(h, n, deadline, max_sweeps=10000):
    """Steepest descent: try all single swaps, take the best improving one."""
    target = n // 2
    hb = make_binary(h, n)
    c = float(np.max(compute_correlation(hb, n)) * 2.0 / n)
    
    ones = np.where(hb == 1)[0]
    zeros = np.where(hb == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        # Try all swaps, find best improving
        best_delta = 0
        best_a, best_b = -1, -1
        for a in ones:
            for b in zeros:
                hb[a] = 0.0
                hb[b] = 1.0
                new_c = float(np.max(compute_correlation(hb, n)) * 2.0 / n)
                delta = c - new_c
                if delta > best_delta:
                    best_delta = delta
                    best_a, best_b = a, b
                hb[a] = 1.0
                hb[b] = 0.0
        if best_a >= 0:
            # Apply best swap
