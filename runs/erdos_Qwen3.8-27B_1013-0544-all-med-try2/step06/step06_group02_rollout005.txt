from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, top_k=5, lr=0.01):
    """Subgradient method focusing on top-k lags"""
    c = compute_c5_vec(h, n)
    c_max = c.max()
    # Select top-k lags
    idx = np.argsort(c)[::-1][:top_k]
    # Weights: how close to max
    w = np.zeros(2*n-1)
    for i in idx:
        w[i] = 1.0  # equal weight for top-k
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in idx:
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    
    h_new = h - lr * grad
    return project_feasible(h_new, n)

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return float(val)

    return grad_smooth_max(x, n, alpha)

    sh = alpha * (c - m)

    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    t0 = time.time()
    
    for i in range(n_iter_max):
        if time.time() > deadline:

        # Choose random pair

        j_idx = np.random.randint(n)
        if i_idx == j_idx:
            continue
        # Choose random delta
        delta = np.random.choice([-1, 1]) * np.random.uniform(0.001, 0.05)

        if h[j_idx] - delta < 0 or h[j_idx] - delta > 1:

        h[i_idx] += delta
        h[j_idx] -= delta
        h = np.clip(h, 0, 1)

            h[j_idx] += delta

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for m in active:
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return float(val)

def _eq_constraint(x, n):
    return float(x.sum() - n / 2.0)

def _eq_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, max_alphas=8):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 50000, 500000, 2000000, 10000000]
    mus = [1.0, 5.0, 10.0, 50.0, 200.0, 2000.0, 5000.0, 10000.0]
    for i in range(min(max_alphas, len(alphas))):
        if time.time() > deadline - 2:
            break
        remaining = deadline - time.time() - 2
        if remaining < 3:
            break
        max_iter = min(250, int(remaining * 80))
        if max_iter < 10:
            break
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                          args=(n, alphas[i], mus[i]), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_iter * 8,
                                   'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def subgradient_solve(h, n, deadline, top_k=3):
    """Subgradient method focusing on top-k lags"""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.03
    for i in range(10000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        c_max = c.max()
        idx = np.argsort(c)[::-1][:top_k]
        dx = 2.0 / n
        grad = np.zeros(n)
        for m in idx:
            k = m - (n - 1)
            wm = dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
            h = h_new
        else:
            h = h_new  # still move, but don't update best
        lr = 0.03 / (1.0 + i * 0.005)
    return best_h, best_c

def simulated_annealing(h, n, deadline):

    temp = 0.005
    t_start = time.time()
    total_time = max(deadline - t_start, 1.0)
    
    for i in range(200000):
        if time.time() > deadline:
            break
        elapsed = time.time() - t_start
        frac = min(elapsed / total_time, 1.0)
        temp = 0.005 * (1.0 - frac) ** 2 + 0.00005
        
        i_idx = np.random.randint(n)
        j_idx = np.random.randint(n)
        if i_idx == j_idx:
            continue
        delta = np.random.choice([-1, 1]) * np.random.uniform(0.001, 0.03)
        if h[i_idx] + delta < 0 or h[i_idx] + delta > 1:

        if h[j_idx] - delta < 0 or h[j_idx] - delta > 1:

        h[i_idx] += delta
        h[j_idx] -= delta

            best_c = c_new
            best_h = h.copy()
        elif np.random.random() > np.exp(-min((c_new - best_c) / temp, 500)):
            pass
        else:
            h[i_idx] -= delta
            h[j_idx] += delta
    return best_h, best_c
