def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def slsqp_smooth(h, n, alpha, maxiter=2000):

        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()

            return float(m + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def constraint_fun(x):
        return float(x.sum() - n / 2.0)
    
    def constraint_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    if sh.max() > 500:
        return m + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def softmax_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    ex = np.exp(alpha * (c - m))
    p = ex / ex.sum()
    L = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    for j in range(n):
        v1 = (j - L >= 0) & (j - L < n)
        if v1.any():
            grad[j] += dx * np.dot(p[v1], 1 - h[j - L[v1]])
        v2 = (j + L >= 0) & (j + L < n)
        if v2.any():
            grad[j] -= dx * np.dot(p[v2], h[j + L[v2]])
    return grad

def fista_opt(h, n, alpha, lr, max_iter, deadline):
    h_curr = h.copy()
    y = h.copy()
    t_k = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = softmax_grad(y, n, alpha)
        y_new = project_feasible(y - lr * grad, n)
        t_new = (1 + np.sqrt(1 + 4*t_k*t_k)) / 2
        y = project_feasible(y_new + ((t_k-1)/t_new)*(y_new - h_curr), n)
        h_curr = y_new
        t_k = t_new
        if it % 10 == 0:
            c = compute_c5(h_curr, n)
            if c < best_c:
                best_c = c
                best_h = h_curr.copy()
    return best_h, best_c

def slsqp_opt(h, n, alpha, maxiter=2000):
    def obj(x):
        return soft_max_obj(x, n, alpha)
    def con_fun(x):
        return float(x.sum() - n/2.0)
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius:
                    lo, hi = max(0, i-radius), min(n, i+radius+1)
                else:
                    lo, hi = 0, n
                for j in range(lo, hi):
                    if i == j: continue
                    if h[i]+d <= 1.0 and h[j]-d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i]-d >= 0.0 and h[j]+d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def coordinate_descent(h, n, deadline, alpha=10000):
    """Gradient-guided coordinate descent."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    idx = np.arange(n)
    while time.time() < deadline:
        grad = softmax_grad(h, n, alpha)
        # Find the index with most negative gradient (most beneficial to decrease)
        # and most positive (beneficial to increase)
        i_dec = np.argmin(grad)  # want to decrease h[i_dec]
        i_inc = np.argmax(grad)  # want to increase h[i_inc]
        # Try moving from i_dec to i_inc
        for d in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            if h[i_dec] - d >= 0 and h[i_inc] + d <= 1:
                h[i_dec] -= d; h[i_inc] += d
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c; best_h = h.copy()
                else:
                    h[i_dec] += d; h[i_inc] -= d

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    if sh.max() > 500:
        return float(m + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def softmax_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    ex = np.exp(alpha * (c - m))
    p = ex / ex.sum()
    L = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        v1 = (j - L >= 0) & (j - L < n)
        if v1.any():
            grad[j] += dx * np.dot(p[v1], 1 - h[j - L[v1]])
        v2 = (j + L >= 0) & (j + L < n)
        if v2.any():
            grad[j] -= dx * np.dot(p[v2], h[j + L[v2]])
    return grad
