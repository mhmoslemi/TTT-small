        j = rng.choice(zeros)

        # delta in correlation
        # c_k = sum h[a] * (1-h[a+m]) where m = k-n+1

        # And: +h[j-m]*(-1) for the (1-h) terms involving h[j] (since 1-h[j] goes from 1 to 0)

        delta = new_c - cur_c

            corr = new_corr
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best_h, best_c

def compute_c5(h, n):

    return float(np.max(c) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-13:
        return h
    if s > target_sum:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c_max = c.max()
    sh = alpha * (c - c_max)
    sh_max = sh.max()
    log_sum = sh_max + np.log(np.exp(sh - sh_max).sum())
    sm = c_max + log_sum / alpha
    p = np.exp(sh - sh_max)
    p /= p.sum()
    # Gradient computation
    # grad[j] = sum_k p[k] * ((1-h[j+m_k]) - h[j-m_k]) where m_k = k-(n-1)
    h_pad = np.zeros(3 * n - 1)
    h_pad[n-1:2*n-1] = h
    offsets = np.arange(2*n-1) - (n-1)  # m values from -(n-1) to n-1
    # For each j, we need h[j+offsets] and h[j-offsets]
    # h[j+offsets] = h_pad[j + offsets + (n-1)]
    idx1 = np.arange(n)[:, None] + (offsets + n - 1)[None, :]  # (n, 2n-1)
    idx2 = np.arange(n)[:, None] - offsets[None, :] + (n-1)  # (n, 2n-1)
    term1 = 1.0 - h_pad[idx1]
    term2 = h_pad[idx2]
    grad = (term1 - term2) @ p
    return sm, grad

    best_h = h.copy()
