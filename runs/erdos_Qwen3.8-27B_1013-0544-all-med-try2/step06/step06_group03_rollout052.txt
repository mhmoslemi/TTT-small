```python
def fast_pair_eval(h, n, a, b, da, db):
    """Compute new max correlation after h[a]+=da, h[b]+=db without full correlate."""
    dx = 2.0 / n
    # Current correlations
    c = np.correlate(h, 1-h, mode='full') * dx
    old_max = c.max()
    
    # Compute delta for each lag
    # c[l + n - 1] corresponds to lag l
    # Δc[l] = da*(1-h[a+l])*valid(a+l) - da*h[a-l]*valid(a-l) 
    #          - db*(1-h[b+l])*valid(b+l) + db*h[b-l]*valid(b-l)
    # where da = +δ, db = -δ for a transfer
    
    for l in range(-(n-1), n):
        # Contribution from h[a] change
        if a+l >= 0 and a+l < n:
            c[l+n-1] += (h[a]+da - h[a]) * (1 - h[a+l])  # = da * (1-h[a+l])
        if a-l >= 0 and a-l < n:
            c[l+n-1] += -(h[a]+da - h[a]) * h[a-l]  # = -da * h[a-l]
        # Contribution from h[b] change  
        if b+l >= 0 and b+l < n:
            c[l+n-1] += (h[b]+db - h[b]) * (1 - h[b+l])  # = db * (1-h[b+l])
        if b-l >= 0 and b-l < n:
            c[l+n-1] += -(h[b]+db - h[b]) * h[b-l]  # = -db * h[b-l]
    
    return (c.max() - old_max) * dx
```

```python
def fast_pair_search(h, n, deadline, deltas):
    dx = 2.0 / n
    h = h.copy()
    best_c = compute_c5(h, n)
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            c_base = np.correlate(h, 1-h, mode='full')
            old_max = c_base.max()
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    if h[i] + d > 1.0 or h[j] - d < 0.0:
                        continue
                    # Compute new max
                    c_new = c_base.copy()  # O(n) copy
                    # Apply delta
                    da = d
                    db = -d
                    for l in range(-(n-1), n):
                        delta = 0.0
                        if 0 <= i+l < n:
                            delta += da * (1 - h[i+l])
                        if 0 <= i-l < n:
                            delta -= da * h[i-l]
                        if 0 <= j+l < n:

                        if 0 <= j-l < n:

                        h[i] += d
                        h[j] -= d
                        old_max = new_max
                        best_c = old_max * dx
                        improved = True
            if not improved:
                break
```

```python
def subgradient_step(h, n, step_size, top_k=5):
    """One step of projected subgradient on the max of correlations."""
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    m = c.max()
    # Find top-k active lags
    top_indices = np.argsort(c)[-top_k:]
    # Subgradient is average gradient of active lags
    # grad of c[l] wrt h[i]:
    # c[l] = sum_i h[i](1-h[i+l]) * dx
    # dc[l]/dh[i] = dx * [(1-h[i+l]) - h[i-l]] (with validity)
    grad = np.zeros(n)
    for idx in top_indices:
        l = idx - (n-1)  # actual lag
        for i in range(n):
            g = 0.0
            if 0 <= i+l < n:
                g += (1 - h[i+l])
            if 0 <= i-l < n:
                g -= h[i-l]
            grad[i] += dx * g
    grad /= len(top_indices)
    # Projected gradient step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new
```

```python
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d > 1.0 or h[j] - d < 0.0:
                        continue
                    delta_c = d * (effects[i] - effects[j])
                    new_max = (c_base + delta_c).max()
                    accept = new_max < old_max - 1e-12
                    if not accept and cooldown > 0:
                        diff = new_max - old_max
                        accept = np.random.random() < np.exp(-diff / max(cooldown, 1e-10))
                    if accept and new_max < old_max + 1e-3:
                        h[i] += d
                        h[j] -= d
                        c_base = np.correlate(h, 1 - h, mode='full')
                        old_max = c_base.max()
                        best_c = old_max * dx
                        if new_max < old_max - 1e-12:
                            improved = True
                            effects = compute_effect_matrix(h, n)
            if not improved:
                break
    return h, best_c

def subgradient_optimize(h, n, deadline, n_iter=200):
    dx = 2.0 / n
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.05
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        threshold = m - 0.005
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        effects = compute_effect_matrix(h, n)
        grad = effects[:, active].mean(axis=1) * dx
        h = project_feasible(h - step * grad, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        step *= 0.98
    return best_h, best_c

def structured_perturb(h, n, rng, method='sinusoidal'):
    hp = h.copy()
    if method == 'sinusoidal':
        freq = rng.randint(1, 5)
        amp = 0.05 + 0.1 * rng.random()
        phase = rng.random() * 2 * np.pi
        hp += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif method == 'bump':
        center = rng.randint(0, n)
        width = 5 + rng.randint(0, 15)
        amp = 0.1 + 0.15 * rng.random()
        x = np.arange(n)
        bump = amp * np.exp(-((x - center) ** 2) / (2 * width ** 2))
        side = rng.random()
        if side < 0.5:
            hp += bump
        else:
            hp -= bump
    else:
        hp += rng.normal(0, 0.05, n)
    return project_feasible(hp, n)

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    alphas = [500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        ht = slsqp(h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht

    for p in [20, 50, 100]:
        if time.time() > deadline - 10:
            break
        ht = slsqp_power(h, n, p, maxiter=1000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht

    if time.time() < deadline - 15:
        h, best_c = subgradient_optimize(h, n, time.time() + min(30, deadline - time.time() - 15))

    remaining = deadline - time.time() - 5
    if remaining > 5:
        dsets = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        h, best_c = fast_pair_search(h, n, time.time() + min(remaining, 60), dsets, cooldown=0.0)

    if time.time() < deadline - 5:
        ht = slsqp(h, n, 500000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
        h, best_c = fast_pair_search(h, n, deadline - 2, [0.001, 0.0005, 0.0002, 0.0001])

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Primary optimization at native resolution
    alloc1 = min(250, (deadline - t0) * 0.30)
    h_opt, c_opt = optimize_at_n(best_h, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    # Phase 2: Multi-restart with structured perturbations
    n = len(best_h)
    remaining = deadline - time.time() - 30
    n_restarts = max(4, int(remaining / 50))
    for r in range(n_restarts):
        if time.time() > deadline - 30:
            break
        method = ['sinusoidal', 'bump', 'gaussian'][r % 3]
        hp = structured_perturb(best_h, n, rng, method)
        alloc = min(50, (deadline - time.time() - 20) * 0.3)
        if alloc < 15:
            break
        h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Try different resolutions
    if time.time() < deadline - 80:
        for n_try in [100, 150, 200]:
            if n_try == n_init:
                continue
            if time.time() > deadline - 60:
                break
            x_new = np.linspace(0, 2, n_try)
            x_old = np.linspace(0, 2, n)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(60, (deadline - time.time()) * 0.15)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_n(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                n = len(best_h)

    # Phase 4: Final intensive pair search with simulated annealing
    if best_h is not None and time.time() < deadline - 10:
        nn = len(best_h)
        h_fin, c_fin = fast_pair_search(best_h, nn, deadline - 5,
                                         [0.01, 0.005, 0.002, 0.001, 0.0005,
                                          0.0002, 0.0001, 0.00005],
                                         cooldown=0.001)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin

    # Phase 5: Final SLSQP polish + pair search
    if best_h is not None and time.time() < deadline - 3:
        nn = len(best_h)
        ht = slsqp(best_h, nn, 1000000, maxiter=3000)
        c = compute_c5(ht, nn)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 2:
            best_h, best_c = fast_pair_search(best_h, nn, deadline - 1,
                                               [0.0005, 0.0002, 0.0001, 0.00005])

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
