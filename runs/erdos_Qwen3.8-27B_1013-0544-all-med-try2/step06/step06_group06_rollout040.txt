            grad[j] += (1 - h[j + m]) * dx
        for j in range(m, n):
            grad[j] -= h[j - m] * dx
    return grad

def subgrad_polish(h, n, deadline, max_iter=10000):
    """Subgradient descent on the exact max (not smoothed)."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 1.0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        k_star = int(np.argmax(g))
        grad = grad_gk(h, n, k_star)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        #_normalize and take step
        step_size = lr / ng
        h_new = project_feasible(h - step_size * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < m - 1e-13:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            lr = min(lr * 1.05, 10.0)
        else:
            lr *= 0.9
            if lr < 1e-8:
                break
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    """SLSQP with log-sum-exp objective."""
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        s = alpha * (g - m)
        sm = s.max()
        return m + (sm + np.log(np.exp(s - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h
