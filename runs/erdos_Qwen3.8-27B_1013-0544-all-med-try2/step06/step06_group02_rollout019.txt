        diff = target - h.sum()
        if abs(diff) < 1e-15:

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room

            if tot < 1e-15:

            h += (diff / tot) * h

def active_set_subgrad(h, n, eta=0.01, max_iters=5000):
    """Subgradient method on the active set with adaptive step size."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.01
    
    for it in range(max_iters):
        overlaps = compute_all_overlaps(h, n)
        c_max = overlaps.max()
        
        # Active set: shifts within eta of max
        active = overlaps >= c_max - eta
        
        # Compute average gradient of active set
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            if not active[m]:
                continue
            k = m - (n - 1)
            wm = 1.0 / active.sum() * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
        
        # Momentum
        momentum = beta * momentum - lr * grad
        
        # Projected step
        h_new = np.clip(h + momentum, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

            lr = min(lr * 1.1, 0.1)

            lr *= 0.7

                grad[0:nv] += wm * (-h[kk:n])
        order = np.argsort(grad)
        improved = False
        for rank_hi in range(n):
            if time.time() > deadline:
                break
            i = order[n - 1 - rank_hi]
            if h[i] < step:
                continue
            for rank_lo in range(min(30, n)):
                j = order[rank_lo]
                if i == j or h[j] + step > 1.0:
                    continue
                if grad[i] <= grad[j]:
                    continue
                h[i] -= step
                h[j] += step
                c_new = compute_c5(h, n)
                if c_new < c_max - 1e-15:
                    c_max = c_new
                    improved = True
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                    break
                else:
                    h[i] += step
                    h[j] -= step
        if not improved:
            break
    return h, best_c

def random_pair_search(h, n, deadline, step, seed=0):
    rng = np.random.RandomState(seed)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    while time.time() < deadline:
        improved = False
        for _ in range(20000):
            if time.time() > deadline:
                return best_h, best_c
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if h[i] >= step and h[j] + step <= 1.0:
                h[i] -= step
                h[j] += step
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] += step
                    h[j] -= step
            elif h[j] >= step and h[i] + step <= 1.0:
                h[j] -= step
                h[i] += step
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                else:
                    h[j] += step
                    h[i] -= step
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(300, 0.5)
        n = 300
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: L-BFGS with escalating alpha
    t1 = min(t0 + 200, deadline - 300)
    h, best_c = lbfgs_optimize(h, n, t1, alphas=[10000, 50000, 200000, 1000000, 5000000])
    best_h = h.copy()
    
    # Phase 2: Subgradient on active set
    if time.time() < deadline - 250:
        t2 = min(time.time() + 60, deadline - 250)
        h2, c2 = subgrad_active(h, n, t2, eta=0.005)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            h = h2
    
    # Phase 3: Gradient-guided pair search (medium steps)
    if time.time() < deadline - 200:
        t3 = min(time.time() + 80, deadline - 200)
        for step in [0.01, 0.005, 0.002, 0.001]:
            if time.time() > t3:
                break
            h, best_c = grad_pair_search(h, n, min(time.time() + 25, t3), step)
            best_h = h.copy()
    
    # Phase 4: Fine pair search (very small steps) - main time sink
    if time.time() < deadline - 150:
        t4 = deadline - 150
        for step in [0.0005, 0.0002, 0.0001, 0.00005, 0.00002]:
            if time.time() > t4 - 5:
                break
            t_alloc = min(60, t4 - time.time())
            if t_alloc < 10:
                break
            h, best_c = grad_pair_search(h, n, time.time() + t_alloc, step)
            best_h = h.copy()
    
    # Phase 5: Random pair search for extra fine-tuning
    if time.time() < deadline - 100:
        t5 = min(time.time() + 80, deadline - 100)
        h, best_c = random_pair_search(h, n, t5, 0.0001, seed=seed)
        if best_c < compute_c5(best_h, n):
            best_h = h.copy()
    
    # Phase 6: L-BFGS polish
    if time.time() < deadline - 50:
        t6 = min(time.time() + 40, deadline - 50)
        h6, c6 = lbfgs_optimize(best_h, n, t6, alphas=[500000, 2000000])
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Final fine pair search
    if time.time() < deadline - 10:
        t7 = deadline - 5
        for step in [0.0001, 0.00005]:
            if time.time() > t7 - 5:
                break
            best_h, best_c = grad_pair_search(best_h, n, min(time.time() + 30, t7), step)
    
    # Phase 8: Try n=500 if time allows
    if time.time() < deadline - 80:
        n2 = 500
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n2)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_feasible(h2, n2)
        t8 = min(time.time() + 50, deadline - 30)
        h2, c2 = lbfgs_optimize(h2, n2, t8, alphas=[50000, 200000, 1000000])
        if time.time() < deadline - 15:
            h2, c2 = grad_pair_search(h2, n2, min(time.time() + 20, deadline - 10), 0.001)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
