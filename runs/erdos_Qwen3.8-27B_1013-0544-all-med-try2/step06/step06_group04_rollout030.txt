                if time.time() > deadline - 20:
                    break
                h_new = slsqp_solve(h, n, alpha, top_idx, 4000)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    h = h_new

    if time.time() < deadline - 20:
        h_pg, c_pg = proximal_grad(h, n, min(deadline - 10, time.time() + 60))
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg
            h = h_pg

    if time.time() < deadline - 10:
        h_new = slsqp_solve(h, n, 20000000, None, 8000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 3:
        best_h, best_c = local_search(best_h, n, deadline - 1)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 300
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init.copy(), n_init)
    best_c = compute_c5(best_h, n_init)

    alloc1 = min(300, (deadline - t0) * 0.45)
    h_opt, c_opt = optimize_core(best_h, n_init, t0 + alloc1)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    for n_try in [200, 250, 350, 400]:
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 40
        if remaining < 60:
            break
        alloc = min(100, remaining * 0.3)
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_core(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        for r in range(4):
            if time.time() > deadline - 30:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_core(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_solve(best_h, n, 50000000, None, 5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h, best_c = local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
