```python
    c = np.correlate(h, 1-h, 'full') * dx
    m = c.max()

    sm = s.max()
    w = np.exp(s - sm) / np.exp(s - sm).sum()  # softmax weights, shape (2n-1,)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def analytical_gradient(h, n, alpha):
    """Compute gradient of log-sum-exp smoothed max of c(k)."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, 'full') * dx
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    w = np.exp(s - sm)
    w = w / w.sum()
    
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    
    for j in range(n):
        # Term 1: w[k] * (1-h[j-l_k]) for valid j-l_k
        valid1 = (j - lags >= 0) & (j - lags < n)
        # Term 2: w[k] * h[j+l_k] for valid j+l_k
        valid2 = (j + lags >= 0) & (j + lags < n)
        grad[j] = np.sum(w[valid1] * (1 - h[j - lags[valid1]])) - np.sum(w[valid2] * h[j + lags[valid2]])
    
    grad *= dx
    return grad, m, w

def projected_gradient_descent(h, n, alpha, deadline, max_iter=5000):
    """Projected gradient descent with backtracking line search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.01
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad, m, w = analytical_gradient(h, n, alpha)
        if np.linalg.norm(grad) < 1e-14:
            break
        
        # Backtracking line search
        c_old = m
        step = lr
        found = False
        for _ in range(30):
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_old - 1e-14:
                h = h_new
                found = True
                break
            step *= 0.5
        
        if not found:
            lr *= 0.9
            if lr < 1e-12:
                break
        else:
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
    
    return best_h, best_c

def binary_local_search_fast(h_bin, n, deadline, max_sweeps=100):

    dx = 2.0 / n
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    c = np.correlate(h_bin, 1 - h_bin, 'full')
    best_c = c.max() * dx
    
    lags = np.arange(2 * n - 1) - (n - 1)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_swap = None
        best_new_c = best_c
        
        for i in zeros:
            if time.time() > deadline:
                break
            # Precompute C_A for this i
            valid1_i = (i - lags >= 0) & (i - lags < n)
            valid2_i = (i + lags >= 0) & (i + lags < n)
            CA = c.copy()
            CA[valid1_i] += (1 - h_bin[i - lags[valid1_i]])
            CA[valid2_i] -= h_bin[i + lags[valid2_i]]
            
            for j in ones:
                if j == i:
                    continue
                valid1_j = (j - lags >= 0) & (j - lags < n)
                valid2_j = (j + lags >= 0) & (j + lags < n)
                new_c = CA.copy()
                new_c[valid1_j] -= (1 - h_bin[j - lags[valid1_j]])
                new_c[valid2_j] += h_bin[j + lags[valid2_j]]
                nc = new_c.max() * dx
                if nc < best_new_c - 1e-12:
                    best_new_c = nc
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        i, j = best_swap
        h_bin[i] = 1.0
        h_bin[j] = 0.0
        ones.remove(j)
        ones.append(i)
        zeros.remove(i)
        zeros.append(j)
        best_c = best_new_c
        c = np.correlate(h_bin, 1 - h_bin, 'full')
    
    return h_bin, best_c
```

```python
    return float(np.correlate(h, 1 - h, 'full').max() * dx)

def compute_all_c(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_logsumexp(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, 'full') * dx
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        v1 = (j - lags >= 0) & (j - lags < n)
        v2 = (j + lags >= 0) & (j + lags < n)
        grad[j] = np.sum(w[v1] * (1.0 - h[j - lags[v1]])) - np.sum(w[v2] * h[j + lags[v2]])
    return grad * dx, m

def pgd_solve(h, n, deadline, alphas=(500, 2000, 10000, 50000, 200000)):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h = h.copy()
        lr = 1.0 / (alpha * 2.0 / n + 1e-6)  # Lipschitz-based
        for it in range(2000):
            if time.time() > deadline:
                break
            grad, m = grad_logsumexp(h, n, alpha)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            # Backtracking
            step = min(lr, 0.1)
            improved = False
            for _ in range(40):
                h_new = project_feasible(h - step * grad, n)
                c_new = compute_c5(h_new, n)
                if c_new < m - 1e-13 or c_new < best_c:
                    h = h_new
                    improved = True
                    break
                step *= 0.5
            if improved:
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                break
    return best_h, best_c
```

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.correlate(h, 1 - h, 'full').max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_logsumexp_vec(h, n, alpha):
    """Vectorized gradient of alpha-smoothed max correlation."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, 'full') * dx
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    
    # A[j] = sum_i (1-h[i]) * w[j+n-1-i] = conv(1-h, w)[j+n-1]

    B = np.correlate(w, h, 'full')[:n][::-1]
    
    grad = (A - B) * dx
    return grad, m
