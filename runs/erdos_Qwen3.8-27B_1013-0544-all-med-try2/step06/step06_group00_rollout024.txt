```python
def fft_correlate(h):
    n = len(h)
    N = 2 * n
    H = np.fft.rfft(h, N)
    autocorr_full = np.fft.irfft(H * np.conj(H), N)
    # autocorr_full[d] for d=0..n-1 gives sum_{k} h[k]*h[k+d] (the linear autocorrelation)
    # Actually, for the zero-padded signal of length N=2n:
    # autocorr_full[d] = sum_{k=0}^{N-1-d} h_padded[k] * h_padded[k+d]
    # where h_padded = [h, 0, 0, ..., 0] (length N)
    # So autocorr_full[d] = sum_{k=0}^{n-1-d} h[k] * h[k+d] for d=0..n-1
    # This is exactly the autocorrelation we need!
    
    autocorr = autocorr_full[:n]  # autocorr[d] for d=0..n-1
    
    # Now compute S(d) for all lags
    cumsum = np.concatenate([[0], np.cumsum(h)])  # cumsum[i] = sum_{k=0}^{i-1} h[k]
    
    # Full correlation array of length 2n-1
    # Index k in output corresponds to lag d = k - (n-1)
    c = np.zeros(2*n-1)
    for d in range(0, n):
        # Lag d (positive): valid n from d to n-1
        S_pos = cumsum[n] - cumsum[d]
        c[n-1+d] = S_pos - autocorr[d]
        # Lag -d (negative): valid n from 0 to n-d-1
        if d > 0:
            S_neg = cumsum[n-d]
            c[n-1-d] = S_neg - autocorr[d]
    
    return c
```

    """
    Compute gradient of weighted correlation sum.
    h: array of length n
    w: weights of length 2n-1 (indexed 0 to 2n-2, corresponding to lags -(n-1) to n-1)
    Returns: gradient of length n

    grad = np.zeros(n)
    
    for d in range(n):
        wp = w[n-1+d]  # weight for positive lag d
        wn = w[n-1-d] if d > 0 else 0.0  # weight for negative lag d

            grad[d:] += (1 - h[:n-d]) * wp

            grad[:n-d] += (1 - h[d:]) * wn
            # -h[i-d] for i = d..n-1

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

    n = len(h)
    N = 2 * n
    H = np.fft.rfft(h, N)
    autocorr = np.fft.irfft(H * np.conj(H), N)[:n]

    c = np.zeros(2*n-1)

        c[n-1+d] = (cumsum[n] - cumsum[d]) - autocorr[d]
        if d > 0:
            c[n-1-d] = cumsum[n-d] - autocorr[d]
    return c

def corr_gradient(h, w):
    n = len(h)
    grad = np.zeros(n)
    for d in range(n):
        wp = w[n-1+d]
        if wp != 0:
            grad[d:] += (1 - h[:n-d]) * wp
            grad[:n-d] -= h[d:] * wp

            wn = w[n-1-d]
            if wn != 0:
                grad[:n-d] += (1 - h[d:]) * wn
                grad[d:] -= h[:n-d] * wn

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smoothed_max_obj(h, alpha):
    c = fft_correlate(h)
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    logsum = sm + np.log(np.exp(shifted - sm).sum())
    return m + logsum / alpha

def smoothed_max_grad(h, alpha):
    c = fft_correlate(h)
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()
    # Gradient of m is subgradient (use uniform over argmax for stability)
    mask = (c >= m - 1e-10)
    n_max = mask.sum()
    subgrad = np.zeros(len(c))
    subgrad[mask] = 1.0 / n_max
    # Gradient of logsum/alpha
    grad_logsum = corr_gradient(h, w) / alpha
    grad_sub = corr_gradient(h, subgrad)
    return grad_sub + grad_logsum

    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0

For continuous annealing moves:

- "Perturb" move: h[i] += eps * (1-h[i]), h[j] -= eps * h[j] (sum-preserving to first order)

The swap move is nice because it's exactly sum-preserving and can be evaluated with the same delta technique as binary swaps.

Let me define the delta for a swap of positions i and j (where a=h[i], b=h[j]):
The new h has h[i]=b, h[j]=a (swapped).

- Position j: h changes from b to a

import numpy as np
from scipy.optimize import minimize
import time

def fft_correlate(h):
    n = len(h)
    N = 2 * n
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N)[:n]
    cs = np.concatenate([[0.0], np.cumsum(h)])
    c = np.zeros(2 * n - 1)
    for d in range(n):
        c[n-1+d] = (cs[n] - cs[d]) - ac[d]
        if d > 0:
            c[n-1-d] = cs[n-d] - ac[d]
    return c

def corr_grad(h, w):
    n = len(h)
    grad = np.zeros(n)
    for d in range(n):
        wp = w[n-1+d]
        if wp != 0:
            if d == 0:
                grad[:] += (1.0 - 2.0*h) * wp
            else:
                grad[d:] += (1.0 - h[:n-d]) * wp
                grad[:n-d] -= h[d:] * wp
        if d > 0:
            wn = w[n-1-d]
            if wn != 0:
                grad[:n-d] += (1.0 - h[d:]) * wn
                grad[d:] -= h[:n-d] * wn
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def smooth_obj(h, alpha):
    c = fft_correlate(h)
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def smooth_grad(h, alpha):
    c = fft_correlate(h)
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    # Subgradient of max (uniform over argmax)
    mask = (c >= m - 1e-12)
    nmask = mask.sum()
    sw = np.zeros(len(c))
    sw[mask] = 1.0 / nmask
    return (corr_grad(h, sw) + corr_grad(h, w) / alpha)

def proj_grad_descent(h, n, alpha, maxiter, deadline):
    h = h.copy()
    step = 2.0 / (alpha * n)  # initial step size
    for it in range(maxiter):
        if time.time() > deadline:
            break
        g = smooth_grad(h, alpha)
        # Project gradient step
        h_new = project_feasible(h - step * g, n)
        f_new = smooth_obj(h_new, alpha)
        f_old = smooth_obj(h, alpha)
        if f_new < f_old:
            h = h_new
            step = min(step * 1.5, 1.0)
        else:
            step *= 0.5
            h = h  # don't update
    return h
