import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    nfull = 2 * n - 1

            h_pms[mask1] = h[p_minus_s[mask1]]

                    h_opt, c_opt = h_s, c_s
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()

    # Phase 3: Multi-start restarts on best resolution
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 60))
        scales = [0.005, 0.01, 0.02, 0.04]
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 25) * 0.5)
            if alloc < 15:
                break
            h_opt, c_opt = coord_descent(hp, n, max_sweeps=200,
                                         deadline=time.time() + alloc)
            if time.time() < deadline - 15:
                h_s = slsqp_refine(h_opt, n, deadline - 10)
                c_s = compute_c5(h_s, n)
                if c_s < c_opt:
                    h_opt, c_opt = h_s, c_s
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()

    # Phase 4: Final fine-tuning
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_opt, c_opt = coord_descent(best_h, n, max_sweeps=200, deadline=deadline - 3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        if time.time() < deadline - 2:
            h_f, c_f = fine_pair_search(best_h, n, deadline - 1)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
