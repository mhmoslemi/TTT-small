def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

    sh = alpha * (c - m)

The problem is: minimize max_k c(k) subject to 0≤h≤1 and ∫h=1.

This is a convex optimization problem! The function c(k) = ∫h(x)(1-h(x+k))dx is... wait, is it convex in h?

The term ∫h(x)h(x+k)dx is a quadratic form in h. Is it convex or concave?

In the discrete case: c(k) = sum_i h[i](1-h[i+k]) * dx = sum_i h[i]*dx - sum_i h[i]*h[i+k]*dx

The first term is linear. The second term is -sum_i h[i]*h[i+k]*dx, which is a quadratic form.

The matrix for the quadratic form: Q[i,j] = -1/2 if j=i+k or j=i-k (appropriately). This matrix is not positive semidefinite in general (it's like a circulant matrix with negative off-diagonal entries).

So the problem is NOT convex in general. That's why we need local search.

However, if we restrict to h being binary (0 or 1), the problem becomes combinatorial (and non-convex). The continuous relaxation allows fractional values, which helps.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _sm_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _cst_eq(x, n):
    return float(x.sum() - n / 2.0)

def _cst_jac(x, n):
    return np.ones(n)

def _sm_obj_nc(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def lbfgs_optimize(h, n, deadline, max_iters=200):
    alphas = [(100, 1.0), (500, 5.0), (2000, 20.0), (10000, 100.0),
              (50000, 500.0), (200000, 2000.0), (1000000, 10000.0)]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_sm_obj, h, jac=_sm_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_iters * 30, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cst_eq, 'jac': _cst_jac, 'args': (n,)}]
    try:
        res = minimize(_sm_obj_nc, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def projected_gradient(h, n, deadline, alpha_start=5000, alpha_end=2000000, iters=3000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.02
    for it in range(iters):
        if time.time() > deadline:
            break
        progress = it / max(iters, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        g = grad_smooth_max(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        step_lr = lr
        improved = False
        for _ in range(50):
            h_new = project_feasible(h - step_lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step_lr *= 0.5
        if not improved:
            lr *= 0.5
        if lr < 1e-14:
            break
    return best_h, best_c
