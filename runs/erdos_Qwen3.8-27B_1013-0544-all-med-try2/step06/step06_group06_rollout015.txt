    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    if s >= 0:
        grad[:n-s] += 1.0 - h[s:]
        grad[s:] -= h[:n-s]
    else:
        sp = -s
        grad[sp:] += 1.0 - h[:n-sp]
        grad[:n-sp] -= h[sp:]
    return grad

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def swap_contrib(h, n, pos):
    """Compute contribution vector when flipping h[pos] (0->1 or 1->0)."""
    c = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        v = 0.0
        if 0 <= pos + s < n:
            v += 1.0 - h[pos + s]
        if 0 <= pos - s < n:
            v -= h[pos - s]
        c[k] = v
    return c

def binary_ls_fast(h_bin, n, deadline, max_sweeps=500):
    best_c = compute_c5(h_bin, n)
    c_all = compute_all_g(h_bin, n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for a in zeros:
            if time.time() > deadline:
                break
            ca = swap_contrib(h_bin, n, a)
            c_a = c_all + ca
            # Vectorized: for each b in ones, compute max(c_a - contrib[b])
            cb_matrix = np.array([swap_contrib(h_bin, n, b) for b in ones])
            new_maxes = (c_a[None, :] - cb_matrix).max(axis=1)
            best_idx = int(new_maxes.argmin())
            if new_maxes[best_idx] < best_c - 1e-12:
                b = ones[best_idx]
                h_bin[a] = 1.0
                h_bin[b] = 0.0
                best_c = float(new_maxes[best_idx])
                c_all = c_a - cb_matrix[best_idx]
                improved = True
                break
        if not improved:
            break
    return h_bin, best_c

def pd_smooth(h, n, deadline, alpha_list, iters=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        h_cur = best_h.copy()
        lr = np.sqrt(n) / (2.0 * alpha)
        lr = max(lr, 1e-5)
        stuck = 0
        for it in range(iters):
            if time.time() > deadline or stuck > 80:
                break
            g = compute_all_g(h_cur, n)
            sg = alpha * g
            sgsh = sg - sg.max()
            p = np.exp(sgsh)
            p /= p.sum()
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                if p[k] < 5e-7:
                    continue
                s = k - (n - 1)
                if s >= 0:
                    grad[:n-s] += p[k] * (1.0 - h_cur[s:])
                    grad[s:] -= p[k] * h_cur[:n-s]
                else:
                    sp = -s
                    grad[sp:] += p[k] * (1.0 - h_cur[:n-sp])
                    grad[:n-sp] -= p[k] * h_cur[sp:]
            h_new = project_feasible(h_cur - lr * grad, n)
            c_new = compute_c5(h_new, n)
            c_cur = compute_c5(h_cur, n)
            if c_new < c_cur - 1e-15:
                h_cur = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_cur.copy()
                stuck = 0
                lr = min(lr * 1.05, 2.0 * np.sqrt(n) / alpha)
            else:
                lr *= 0.75
                stuck += 1
    return best_h, best_c

def greedy_binary(n, deadline):
    h = np.zeros(n)
    c_all = np.zeros(2 * n - 1)
    for step in range(n // 2):
        if time.time() > deadline:
            break
        best_pos = -1
        best_max = float('inf')
        for pos in range(n):
            if h[pos] == 1:
                continue
            ca = swap_contrib(h, n, pos)
            new_max = (c_all + ca).max()
            if new_max < best_max:
                best_max = new_max
                best_pos = pos
        if best_pos >= 0:
            c_all += swap_contrib(h, n, best_pos)
            h[best_pos] = 1.0
    return h, float(c_all.max() * 2.0 / n)

def round_to_binary(h, n, rng):
    target = int(n // 2)
    scores = h + 1e-4 * rng.randn(n)
    idx = np.argsort(scores)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

    xo = np.linspace(0, 2, n_old)
    xn = np.linspace(0, 2, n_new)
    return np.interp(xn, xo, h_old)
