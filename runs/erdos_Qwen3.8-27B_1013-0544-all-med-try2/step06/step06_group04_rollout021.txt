def compute_all_lags(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Compute gradient of lag correlation w.r.t. h"""
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return 1.0 - 2.0 * h
    if lag > 0:
        grad = np.zeros(n)
        # For i in [0, n-m): h[i] appears as h[i] with (1-h[i+m])

        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    else:
        grad = np.zeros(n)
        grad[:n-m] -= h[m:]
        grad[m:] += (1.0 - h[:n-m])
    return grad

def soft_max_grad(h, n, alpha):
    """Compute gradient of soft max of all lag correlations"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_c = c.max()
    # Soft max weights
    shifted = alpha * (c - max_c)
    max_shifted = shifted.max()
    weights = np.exp(shifted - max_shifted)
    weights /= weights.sum()
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] > 1e-10:
            lag = k - (n - 1)
            grad += weights[k] * lag_gradient(h, n, lag)
    return grad, float(max_c + np.log(weights.sum()) / alpha)
