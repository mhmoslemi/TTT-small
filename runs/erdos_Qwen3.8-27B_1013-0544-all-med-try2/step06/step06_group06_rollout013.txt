import numpy as np
from scipy.optimize import minimize
import time

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_all_g(h, n)))

def gradient_gk(h, n, k):
    """Analytic gradient of c[k] w.r.t. h components."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    # First term: (1-h[j+offset]) for j where both j and j+offset in [0,n)
    lo1 = max(0, -offset)
    hi1 = min(n, n - offset)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1+offset:hi1+offset])
    # Second term: -h[j-offset] for j where both j-offset and j in [0,n)
    lo2 = max(0, offset)
    hi2 = min(n, n + offset)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2-offset:hi2-offset]
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def multi_lag_gradient(h, n, K=5, temperature=0.01):
    """Compute weighted gradient from top K lags."""
    g = np.correlate(h, 1 - h, mode='full')
    # Softmax weights over top K
    top_k = np.argsort(g)[-K:][::-1]
    weights = np.zeros(2*n-1)
    for idx in top_k:
        weights[idx] = 1.0
    grad = np.zeros(n)
    for k in top_k:
        grad += weights[k] * gradient_gk(h, n, k)
    return grad / K

def projected_gradient(h, n, deadline, max_iter=50000, K=5, lr0=0.1):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        # Multi-lag gradient
        grad = multi_lag_gradient(h, n, K=K)
        ng = np.linalg.norm(grad)
        if ng < 1e-12:
            break
        grad_normed = grad / ng
        # Backtracking line search
        found = False
        for f in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            step = lr * f
            h_try = project_feasible(h - step * grad_normed, n)
            c_try = compute_c5(h_try, n)
            if c_try < m - 1e-14:
                h = h_try
                found = True
                break
        if not found:
            lr *= 0.9
            if lr < 1e-12:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            lr = min(lr * 1.05, 1.0)
    return best_h, best_c

    offset = k - (n - 1)

    lo1 = max(0, -offset)
    hi1 = min(n, n - offset)

        grad[lo1:hi1] += (1.0 - h[lo1+offset:hi1+offset])
    lo2 = max(0, offset)

                    ones_list.append(i)
                    zeros_list.remove(i)
                    zeros_list.append(j)
                    ones_list = sorted(ones_list)
                    zeros_list = sorted(zeros_list)
                    improved = True
                    break
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

def lp_polish_binary(h_bin, n):
    dx = 2.0 / n
    g_base = np.correlate(h_bin, 1.0 - h_bin, mode='full') * dx
    m0 = float(np.max(g_base))
    n_lags = 2 * n - 1
    A_ub = np.zeros((n_lags, n + 1))
    b_ub = np.zeros(n_lags)
    for k in range(n_lags):
        A_ub[k, :n] = dx * gradient_gk(h_bin, n, k)
        A_ub[k, -1] = -1.0
        b_ub[k] = -g_base[k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h_bin[i], 1.0 - h_bin[i]) for i in range(n)] + [(None, None)]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            h_new = project_feasible(h_bin + res.x[:n], n)
            c_new = compute_c5(h_new, n)
            if c_new < m0:
                return h_new, c_new
    except Exception:
        pass
    return h_bin, m0

def slsqp_opt(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Projected gradient with increasing K (40% budget)
    p1 = t0 + 0.40 * (budget_s - 5)
    for K in [1, 2, 3, 5, 8, 12, 20]:
        if time.time() > p1 - 30:
            break
        h, c = projected_gradient(best_h.copy(), n, p1 - 30, K=K, lr0=0.5)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: Binary swap search (15% budget)
    p2 = t0 + 0.55 * (budget_s - 5)
    if time.time() < p2 - 5:
        hb, cb = binary_swap_search(best_h.copy(), n, p2 - 5)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Phase 3: LP polish around binary (5% budget)
    p3 = t0 + 0.60 * (budget_s - 5)
    if time.time() < p3 - 2:
        # Make a binary version and LP polish it
        idx = np.argsort(best_h)[::-1]
        hb = np.zeros(n)
        hb[idx[:n // 2]] = 1.0
        hl, cl = lp_polish_binary(hb, n)
        if cl < best_c:
            best_c = cl
            best_h = hl.copy()
    # Phase 4: More projected gradient on best (20% budget)
    p4 = t0 + 0.80 * (budget_s - 5)
    for K in [3, 5, 10]:
        if time.time() > p4 - 20:
            break
        h, c = projected_gradient(best_h.copy(), n, p4 - 20, K=K, lr0=0.3)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 5: SLSQP polish (10% budget)
    p5 = t0 + 0.90 * (budget_s - 5)
    if time.time() < p5 - 5:
        for alpha in [100000, 500000, 2000000]:
            if time.time() > p5 - 10:
                break
            h = slsqp_opt(best_h.copy(), n, alpha, 1000, p5 - 10)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    # Phase 6: Try n=200 (5% budget)
    p6 = t0 + 0.95 * (budget_s - 5)
    if time.time() < p6 - 5:
        nh = 200
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, c = projected_gradient(hh, nh, p6 - 5, K=5, lr0=0.3)
        if c < best_c:
            best_c = c
            best_h = hh.copy()
            n = nh
    # Final
    if time.time() < dl - 2:
        h, c = projected_gradient(best_h.copy(), n, dl - 2, K=3, lr0=0.1)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
