```python
def subgradient_step(h, n, active_threshold=0.01):
    """Compute subgradient at the active set."""
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full')
    g_max = g.max()
    active = np.where(g > g_max - active_threshold)[0]
    
    grad = np.zeros(n)
    for k in active:
        t = k - (n - 1)
        # d/dh[i] g_k = (1-h[i-t]) - h[i+t] with 0 boundary
        # For the (1-h[i-t]) term: valid when 0 <= i-t < n, i.e., t <= i < n+t
        # For the h[i+t] term: valid when 0 <= i+t < n, i.e., -t <= i < n-t
        if t >= 0:
            # (1-h[i-t]) for i in [t, n-1]
            grad[t:] += (1.0 - h[:n-t])
            # -h[i+t] for i in [0, n-1-t]
            grad[:n-t] -= h[t:]
        else:
            s = -t
            # (1-h[i+s]) for i in [0, n-1-s]
            grad[:n-s] += (1.0 - h[s:])
            # -h[i-s] for i in [s, n-1]
            grad[s:] -= h[:n-s]
    
    grad /= len(active)
    return grad * dx, g_max
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def subgradient_at_active(h, n, threshold=0.01):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full')
    g_max = g.max()
    active = np.where(g > g_max - threshold)[0]
    grad = np.zeros(n)
    for k in active:
        t = k - (n - 1)
        if t >= 0:
            grad[t:] += (1.0 - h[:n - t])
            grad[:n - t] -= h[t:]
        else:
            s = -t
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
    grad /= max(len(active), 1)
    return grad * dx, g_max * dx

def projected_subgrad(h, n, deadline, max_iter=100000, thr_list=None):

    best_c = compute_c5(h, n)
    h = h.copy()
    lr = 0.05
    for it in range(max_iter):
        if time.time() > deadline:
            break
        thr = thr_list[min(it // 2000, len(thr_list) - 1)]

            best_c = c_new

            h = h_new
            lr = min(lr * 1.02, 0.2)
        else:
            lr *= 0.92
            if lr < 1e-8:
                break
    return best_h, best_c

def slsqp_polish(h, n, deadline, alpha=200000, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
```
