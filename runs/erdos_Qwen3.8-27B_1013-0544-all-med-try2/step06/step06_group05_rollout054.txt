So: i should have h[m*+n-1-i] small, j should have h[m*+n-1-j] large.

But we also need to check that other positions don't increase. This is the hard part.

**Alternative: Just use many small steps of subgradient descent.**

The subgradient of max_m c[m] at h is (for the active set A = {m: c[m] = max(c)}):
grad[i] = (1/|A|) * sum_{m in A} (1 - 2*h[m+n-1-i])  (for valid indices)

This is computed in O(|A| * n) time. For |A| small (usually 1-5), this is O(n).

Then: h_new = h - lr * grad, project to [0,1] and sum=n/2.

This is O(n) per step! With 100,000 steps, that's O(n * 100000) = 25M operations for n=250. Very fast in numpy.

def subgradient_descent(h, n, deadline, lr0=0.01, lr_decay=0.9999):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = np.max(np.correlate(h, 1-h, mode='full')) * dx
    lr = lr0
    
    for it in range(200000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1-h, mode='full')
        max_val = c.max()
        
        if max_val * dx < best_c:
            best_c = max_val * dx
            best_h = h.copy()
        
        # Find active set (positions within 1e-6 of max)
        active = np.where(c >= max_val - 1e-6)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for m in active:
            # grad[i] += (1 - 2*h[m+n-1-i]) for valid i
            # m+n-1-i in [0, n-1] means i in [m+n-1-(n-1), m+n-1] = [m, m+n-1] intersect [0, n-1] = [m, n-1]
            # Wait: m+n-1-i >= 0 means i <= m+n-1 (always true since i < n and m >= 0)
            # m+n-1-i <= n-1 means i >= m
            # So valid i for active m is: i in [m, n-1] (if m < n) or i in [0, n-1] (if m >= n)
            
            # Actually m ranges from 0 to 2n-2
            # m+n-1-i in [0, n-1]
            # => i in [m+n-1-(n-1), m+n-1-0] = [m, m+n-1]
            # intersect with [0, n-1]:
            # if m < n: i in [m, n-1]
            # if m >= n: i in [0, n-1] (since m+n-1 >= n-1 means the lower bound is m >= n, but i max is n-1 < n <= m, so... )
            # Wait: if m >= n, then m > n-1 >= i for all i, so i < m, meaning i < m which means m+n-1-i > n-1. So no valid i?
            
            # Let me redo: m+n-1-i >= 0 => i <= m+n-1 (always true for i < n, m >= 0)
            # m+n-1-i <= n-1 => i >= m
            # So we need i >= m AND i <= n-1
            # If m < n: i in [m, n-1]
            # If m >= n: no valid i (since i <= n-1 < n <= m)
            
            if m < n:
                idx = m + n - 1 - np.arange(m, n)  # = n-1-m, n-2-m, ..., 0
                grad[m:] += (1.0 - 2.0 * h[idx])
        
        grad /= len(active)
        
        # Update
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        
        # Project sum
        target = n / 2.0
        diff = target - h.sum()
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 0:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 0:
                h += (diff / tot) * h
        h = np.clip(h, 0, 1)
        
        lr *= lr_decay
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if abs(diff) > 50:
            # Large correction: scale
            if diff > 0:
                h = np.clip(h + diff / n, 0, 1)
            else:
                h = np.clip(h * (target_sum / max(h.sum(), 1e-10)), 0, 1)
        elif diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_descent(h, n, deadline, lr0=0.05, momentum=0.9, max_iter=50000):
    dx = 2.0 / n
    h = h.copy()
    vel = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr0
    lr_decay = (0.01 / lr0) ** (1.0 / max_iter)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        max_val = float(c.max())
        if max_val * dx < best_c:
            best_c = max_val * dx
            best_h = h.copy()
        
        # Active set: positions within 1% of max
        threshold = max_val * 0.99
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Compute subgradient
        grad = np.zeros(n)
        for m in active:
            if m < n:
                idx = np.arange(n - 1 - m, -1, -1)  # n-1-m, n-2-m, ..., 0
                grad[m:] += (1.0 - 2.0 * h[idx])
            # if m >= n: no valid i (as derived above)
        
        grad /= len(active)
        
        # Momentum update
        vel = momentum * vel - lr * grad
        h = h + vel
        h = np.clip(h, 0.0, 1.0)
        
        # Project sum
        target = n / 2.0
        diff = target - h.sum()
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-13:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-13:
                h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
        
        lr *= lr_decay
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_step(h, n, lr, momentum, vel):
    c = np.correlate(h, 1 - h, mode='full')
    max_val = float(c.max())
    threshold = max_val * 0.995
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    grad = np.zeros(n)
    for m in active:
        if m < n:
            idx = np.arange(n - 1 - m, -1, -1)
            grad[m:] += (1.0 - 2.0 * h[idx])
    grad /= len(active)
    gn = np.linalg.norm(grad)
    if gn > 1.0:
        grad /= gn
    vel = momentum * vel - lr * grad
    h_new = h + vel
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n / 2.0
    diff = target - h_new.sum()
    if diff > 0:
        room = 1.0 - h_new
        tot = room.sum()
        if tot > 1e-13:
            h_new += (diff / tot) * room
    else:
        tot = h_new.sum()
        if tot > 1e-13:
            h_new += (diff / tot) * h_new
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new, vel, max_val
