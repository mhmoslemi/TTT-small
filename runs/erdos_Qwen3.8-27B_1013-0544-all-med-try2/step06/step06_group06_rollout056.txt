# Global state for SLSQP callbacks
_g_n = 0
_g_alpha = 0.0
_g_dx = 0.0

def _slsqp_obj(x):
    g = np.correlate(x, 1 - x, mode='full') * _g_dx
    m = g.max()
    shifted = _g_alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / _g_alpha

def _slsqp_cons(x):
    return x.sum() - _g_n / 2.0

def _single_obj(x):
    # Optimize a single lag
    lag = _g_lag
    n = _g_n
    if lag >= 0:
        return np.sum(x[:n-lag] * (1 - x[lag:])) * _g_dx
    else:
        m = -lag
        return np.sum(x[m:] * (1 - x[:n-m])) * _g_dx

    dx = 2.0 / n

    return float(np.max(c) * dx)

            if s < 1e-14:
