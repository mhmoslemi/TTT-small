```python
def subgradient_step(h, n, c, alpha, lr):
    """One subgradient step for the smooth max objective."""
    # Compute smooth gradients: weight_k = exp(alpha * (c_k - max_c)) / sum(exp(...))
    m = c.max()
    weights = np.exp(alpha * (c - m))
    weights /= weights.sum()
    
    # Gradient of smooth max = weighted sum of individual gradients
    # G[i,k] = ∂c_k/∂h_i
    # grad[i] = sum_k weights[k] * G[i,k]
    
    # Compute G matrix (n x (2n-1))
    # For lag s (corresponding to index k = n-1+s in the full correlation):
    # G[i, n-1+s] = ∂c_s/∂h_i
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] < 1e-10:
            continue
        s = k - (n-1)
        # Compute gradient of c_s w.r.t. h
        if s == 0:
            g = 1.0 - 2.0 * h
        elif s > 0:
            g = np.zeros(n)
            valid1 = slice(0, n-s)  # h[i] term, i from 0 to n-1-s
            g[valid1] += (1.0 - h[s:n])  # 1-h[i+s] for i in 0..n-1-s
            g[s:] -= h[0:n-s]  # -h[i-s] for i in s..n-1
        else:
            s_abs = -s
            g = np.zeros(n)
            g[s_abs:] += (1.0 - h[0:n-s_abs])  # 1-h[i+s] where i+s >= 0, i >= s_abs
            g[0:n-s_abs] -= h[s_abs:n]  # -h[i+s] where i+s < n, i < n-s_abs
        grad += weights[k] * g
    
    # Projected gradient step
    h_new = h - lr * grad
    return project_feasible(h_new, n)
```

```python
def compute_gradient_matrix(h, n):
    """Compute G[i, k] = ∂c_k/∂h_i for all i, k."""
    G = np.zeros((n, 2*n-1))

            G[m:, k] += (1.0 - h[:n-m])
            G[:n-m, k] -= h[m:]
    return G
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0, 1)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0, 1)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
        return m + (shifted.max() + np.log(np.sum(np.exp(shifted)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_grad(h, n, c, alpha):
    """Gradient of smooth max of correlations."""
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    grad = np.zeros(n)
    grad += w[n-1] * (1.0 - 2.0*h)
    for s in range(1, n):
        ws = w[n-1+s]
        wm = w[n-1-s]
        if ws > 1e-12:
            grad[:n-s] += ws * (1.0 - h[s:])
            grad[s:] -= ws * h[:n-s]
        if wm > 1e-12:
            grad[s:] += wm * (1.0 - h[:n-s])
            grad[:n-s] -= wm * h[s:]
    return grad

def subgradient_method(h, n, deadline, alpha=5000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    t = 1
    lr_base = 0.1
    
    while time.time() < deadline:
        c = compute_all_lags(h_curr, n)
        grad = smooth_grad(h_curr, n, c, alpha)
        # Normalize gradient
        gnrm = np.linalg.norm(grad)
        if gnrm < 1e-15:
            break
        lr = lr_base / np.sqrt(t)
        h_new = h_curr - lr * grad / gnrm
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # Sometimes insert the current point to avoid drifting
        if t % 10 == 0:
            h_curr = 0.5 * h_new + 0.5 * h_curr
            h_curr = project_feasible(h_curr, n)
        else:
            h_curr = h_new
        t += 1
        if t > 5000:
            break
    
    return best_h, best_c
```

For each pair (i, j), compute the "descent direction" d_k = G[i,k] - G[j,k]. The best move is the one that most reduces the max.

A practical approach:
1. Compute all correlations c[k]
2. Identify the "active" set: lags where c[k] > C_max - ε (i.e., near the maximum)
3. For each position i, find the best partner j such that moving from j to i reduces all active lags
4. Make the move

To make this efficient, I can use the following heuristic:
- For the single worst lag s*, the best move is to move from the position with highest G[j,s*] to the position with lowest G[i,s*]
- Check if this also helps (or at least doesn't hurt) the other active lags
- If not, try the second best, etc.

But this is still O(n²) in the worst case. For n=300, that's 90K pairs, which is manageable if each evaluation is O(1) or O(T) where T is the number of active lags.

Let me think about this more carefully. For a given pair (i, j) and the active set A of lags:
Δc[k] = δ * (G[i,k] - G[j,k]) for k in A

We want all Δc[k] ≤ 0 (or at least Δc[s*] < 0 where s* is the worst).

If we just look at the worst lag: we need G[i,s*] < G[j,s*].

The "best" i is the one with minimum G[i,s*], and the "best" j is the one with maximum G[j,s*].

The question is: is the steepest descent pair always the best? Not necessarily. Sometimes a "shallow" move that helps multiple lags is better than a "steep" move that only helps the worst lag.

Let me modify: instead of just the worst lag, consider the top T lags and find a move that helps all of them.

```python
def multi_lag_coordinate_descent(h, n, deadline, top_T=5, max_rounds=2000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_T:]
        
        # For each active lag, compute the gradient
        # Find the move (i, j, delta) that most reduces the max active correlation
        
        # Strategy: for each position i, compute the "score" = weighted sum of G[i,k] for active k
        # Then find i with min score and j with max score
        
        scores = np.zeros(n)
        for k in top_idx:
            s = k - (n-1)
            if s == 0:
                G_k = 1.0 - 2.0 * h
            elif s > 0:
                G_k = np.zeros(n)
                G_k[:n-s] += (1.0 - h[s:])
                G_k[s:] -= h[:n-s]
            else:
                m = -s
                G_k = np.zeros(n)
                G_k[m:] += (1.0 - h[:n-m])
                G_k[:n-m] -= h[m:]
            # Weight by how close to max
            weight = c[k] / c[top_idx[0]]  # normalize
            scores += weight * G_k
        
        i_best = int(np.argmin(scores))
        j_best = int(np.argmax(scores))
        
        if i_best == j_best:
            break
        
        # Move from j (high score) to i (low score)
        if h[j_best] <= h[i_best]:
            i_best, j_best = j_best, i_best
        
        delta = min(0.01, (h[j_best] - h[i_best]) * 0.5)
        if delta < 1e-8:
            break
        
        h_new = h.copy()
        h_new[j_best] -= delta
        h_new[i_best] += delta
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller delta
            for frac in [0.5, 0.1, 0.01]:
                delta2 = delta * frac
                h_new = h.copy()
                h_new[j_best] -= delta2
                h_new[i_best] += delta2
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new
                    break
            else:
                # No improvement, stop
                h = best_h.copy()
                break
    
    return best_h, best_c
```
