def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_gradient(h, n, k_abs):
    """Compute gradient of g_k w.r.t. h for positive shift k_abs."""
    dx = 2.0 / n
    idx = np.arange(n)
    grad = np.zeros(n)
    if k_abs == 0:
        return dx * (1 - 2 * h)
    mask1 = idx >= k_abs  # h[i] appears as h[n+k]
    mask2 = idx <= (n - 1 - k_abs)  # h[i] appears as h[n]
    i1 = idx[mask1]
    i2 = idx[mask2]
    grad[mask1] += (1 - h[i1 - k_abs])
    grad[mask2] -= h[i2 + k_abs]
    return dx * grad

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_pair_search(h, n, deadline, L=5, delta0=0.005):
    """O(n) per step gradient-guided pair search."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    delta = delta0
    total_shift = n - 1
    
    while time.time() < deadline:
        c = np.correlate(h, 1-h, mode='full')
        c_vals = c * (2.0 / n)
        max_c = c_vals.max()
        
        # Top L active shifts
        top_idx = np.argsort(c_vals)[-L:]
        # Filter: only use shifts with value close to max
        active = [i for i in top_idx if c_vals[i] > max_c - 0.002]
        if not active:
            active = [int(top_idx[-1])]
        
        # Average gradient over active shifts
        grad = np.zeros(n)
        for idx in active:
            k = abs(int(idx) - total_shift)
            grad += get_gradient(h, n, k)
        grad /= len(active)
        
        # Best pair: max grad (decrease) and min grad (increase)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        
        if i == j:
            # Find second extreme
            sorted_g = np.argsort(grad)
            j = int(sorted_g[0]) if i != int(sorted_g[0]) else int(sorted_g[1])
            i = int(sorted_g[-1]) if i != int(sorted_g[-1]) else int(sorted_g[-2])
            if i == j:
                break
        
        # Ensure valid move
        move_delta = min(delta, h[i], 1.0 - h[j])
        if move_delta < 1e-8:
            delta *= 0.5
            if delta < 1e-6:
                break
            continue
        
        h[i] -= move_delta
        h[j] += move_delta
        c_new = compute_c5(h, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
            delta = min(delta * 1.02, 0.03)
        else:
            h[i] += move_delta
            h[j] -= move_delta
            delta *= 0.97
            if delta < 1e-6:
                # Reset delta and try a different approach
                delta = 0.002
    
    return best_h, best_c

def optimize_with_hybrid(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    alphas = [500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 30:

        ht = slsqp_optimize(best_h, n, a, maxiter=3000, ftol=1e-15)

    # Phase 3: Final SLSQP polish
    if time.time() < deadline - 5:

        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht.copy()
    
    # Phase 4: Final gradient refinement
    if time.time() < deadline - 2:

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_gradient(h, n, k_abs):
    if k_abs == 0:
        return (2.0 / n) * (1 - 2 * h)
    dx = 2.0 / n
    idx = np.arange(n)
    grad = np.zeros(n)
    mask1 = idx >= k_abs
    mask2 = idx <= (n - 1 - k_abs)
    i1 = idx[mask1]
    i2 = idx[mask2]
    grad[mask1] += (1 - h[i1 - k_abs])
    grad[mask2] -= h[i2 + k_abs]
    return dx * grad

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=4000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_pair_search(h, n, deadline, L=5, delta0=0.005):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    delta = delta0
    total_shift = n - 1
    stuck_count = 0
    
    while time.time() < deadline:
        c_raw = np.correlate(h, 1 - h, mode='full')
        c_vals = c_raw * (2.0 / n)
        max_c = c_vals.max()
        
        top_idx = np.argsort(c_vals)[-L:]
        active = [int(i) for i in top_idx if c_vals[i] > max_c - 0.003]
        if not active:
            active = [int(top_idx[-1])]
        
        grad = np.zeros(n)
        for idx in active:
            k = abs(idx - total_shift)
            grad += get_gradient(h, n, k)
        grad /= len(active)
        
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            sg = np.argsort(grad)
            j = int(sg[0])
            i = int(sg[-1])
        if i == j:
            break
        
        move_delta = min(delta, h[i], 1.0 - h[j])
        if move_delta < 1e-9:
            delta *= 0.5
            if delta < 1e-7:
                break
            continue
        
        h[i] -= move_delta
        h[j] += move_delta
        c_new = compute_c5(h, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
            stuck_count = 0
            delta = min(delta * 1.03, 0.025)
        else:
            h[i] += move_delta
            h[j] -= move_delta
            stuck_count += 1
            delta *= 0.95
            if stuck_count > 200:
                delta = max(delta0 * 0.1, 0.0005)
                stuck_count = 0
    
    return best_h, best_c

def optimize_with_hybrid(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    alphas = [300, 1000, 5000, 20000, 100000, 500000]
    for a in alphas:

        ht = slsqp_optimize(best_h, n, a, maxiter=3000, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()
    
    if time.time() < deadline - 15:

            best_h = ht.copy()
    
    if time.time() < deadline - 2:

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
