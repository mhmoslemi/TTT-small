        if diff > 0:

        else:

            if tot < 1e-14:

            h += (diff / tot) * h

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = (c - m) * alpha
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, deadline, rng, max_alphas=5):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    
    alphas = [200, 1000, 5000, 20000, 100000, 500000]
    alphas = alphas[:max_alphas]
    
    for a in alphas:
        if time.time() > deadline - 30:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, h, args=(n, a), method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                h = h_new
        except:
            pass
    
    return h, best_c

def fast_local_search(h, n, deadline, rng):
    """Fast stochastic local search."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    step_idx = 2  # Start with moderate step
    stuck = 0
    no_improve = 0
    
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        if iter_count % 50000 == 0 and no_improve > 200000:
            break
        
        s = steps[min(step_idx, len(steps)-1)]
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        direction = 1 if rng.rand() < 0.5 else -1
        d = direction * s
        hi = h[i] + d
        hj = h[j] - d
        if 0 <= hi <= 1 and 0 <= hj <= 1:
            h[i], h[j] = hi, hj
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
                stuck = 0
                no_improve = 0
                if step_idx < len(steps) - 1 and rng.rand() < 0.4:
                    step_idx += 1
            else:
                h[i], h[j] = hi - d, hj + d
                stuck += 1
                no_improve += 1
                if stuck > 5000:
                    stuck = 0
                    if step_idx > 0:
                        step_idx = max(0, step_idx - 1)
                    # Occasionally make a bigger move
                    if rng.rand() < 0.1:
                        k = rng.randint(n)
                        m = rng.randint(n)

                                h[k] += dk
                                h[m] = hm
                                c = compute_c5(h, n)
                                if c < best_c + 0.001:  # Allow small worsening
                                    if c < best_c:
                                        best_c = c
                                        best_h = h.copy()
                                        no_improve = 0
    
    return best_h, best_c

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def soft_max_obj(x, n, alpha):
    c = compute_corr(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def soft_max_grad(h, n, alpha):
    """Compute gradient of soft max objective."""
    c = compute_corr(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        weights = np.exp(shifted - mx)
    else:
        weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Compute gradient
    g = np.zeros(n)
    for p in range(2 * n - 1):
        w = weights[p]
        if w < 1e-10:
            continue
        k = p - (n - 1)  # lag
        ak = abs(k)
        if ak >= n:
            continue
        if k >= 0:
            # df_k/dh[j] = (1-h[j-k]) * 1_{k<=j<n} + (-h[j+k]) * 1_{0<=j<n-k}
            for j in range(ak, n):
                g[j] += w * (1 - h[j - ak])
            for j in range(0, n - ak):
                g[j] += w * (-h[j + ak])
        else:
            # k < 0, ak = -k
            # df_k/dh[j] = (1-h[j+ak]) * 1_{0<=j<n-ak} + (-h[j-ak]) * 1_{ak<=j<n}
            for j in range(0, n - ak):
                g[j] += w * (1 - h[j + ak])
            for j in range(ak, n):
                g[j] += w * (-h[j - ak])
    
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, deadline, rng, alphas=None):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(soft_max_obj, h, args=(n, a), method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 2000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def soft_max_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shift = alpha * (c - m)
    mx = shift.max()
    w = np.exp(shift - mx)
    w /= w.sum()
    g = np.zeros(n)
    center = n - 1
    for p in range(2 * n - 1):
        if w[p] < 1e-12:
            continue
        ak = abs(p - center)
        if ak >= n:
            continue
        g[ak:] += w[p] * (1.0 - h[:n - ak])
        g[:n - ak] += w[p] * (-h[ak:])
    return g

def gradient_descent(h, n, deadline, alpha_start=100, alpha_end=20000, lr0=0.003, momentum=0.85):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_iters = 20000
    lr = lr0
    v = np.zeros(n)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = it / n_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        g = soft_max_gradient(h, n, alpha)
        g = g - g.mean()
        v = momentum * v - lr * g
        h_try = h + v
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            h = h_try
            best_c = c_try
            best_h = h.copy()
            if c_try < compute_c5(best_h, n) - 1e-8:
                lr = min(lr0, lr * 1.02)
        else:
            lr *= 0.9
            v *= 0.5
            if lr < 1e-8:
                lr = lr0 * 0.05
    return best_h, best_c

def fast_local_search(h, n, deadline, rng):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    step_idx = 1
    stuck = 0
    total_iters = 0
    while time.time() < deadline:
        total_iters += 1
        if total_iters % 100000 == 0 and stuck > 50000:
            break
        s = steps[min(step_idx, len(steps) - 1)]
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = (1 if rng.rand() < 0.5 else -1) * s
        hi = h[i] + d
        hj = h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
                stuck = 0
                if step_idx < len(steps) - 1 and rng.rand() < 0.3:
                    step_idx += 1
            else:
                h[i], h[j] = hi - d, hj + d
                stuck += 1
                if stuck > 3000:
                    stuck = 0
                    if step_idx > 0:
                        step_idx -= 1
                    if rng.rand() < 0.15:
                        k = rng.randint(n)
                        m2 = rng.randint(n)

                                c2 = compute_c5(h, n)
                                if c2 < best_c:
                                    best_c = c2
                                    best_h = h.copy()
    return best_h, best_c
