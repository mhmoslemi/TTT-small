   ```python
   def compute_weighted_grad(h, p, dx):
       n = len(h)
       grad = np.zeros(n)
       # k = 0
       idx0 = n - 1
       grad += p[idx0] * (1 - 2*h)
       # k > 0
       for k in range(1, n):
           pk = p[n-1+k]
           if abs(pk) < 1e-12:
               continue
           grad[k:] += pk * (1 - h[:n-k])
           grad[:n-k] -= pk * h[k:]
       # k < 0
       for k in range(1, n):
           pk = p[n-1-k]
           if abs(pk) < 1e-12:
               continue
           grad[:n-k] += pk * (1 - h[k:])
           grad[k:] -= pk * h[:n-k]
       return dx * grad
   ```

   ```python
import numpy as np
from scipy.optimize import minimize, Bounds
   ```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_weighted_grad(h, p, dx):
    n = len(h)
    grad = np.zeros(n)
    idx0 = n - 1
    grad += p[idx0] * (1.0 - 2.0 * h)
    for k in range(1, n):
        pk = p[n - 1 + k]
        if pk > 1e-14:
            grad[k:] += pk * (1.0 - h[:n - k])
            grad[:n - k] -= pk * h[k:]
        pk2 = p[n - 1 - k]
        if pk2 > 1e-14:
            grad[:n - k] += pk2 * (1.0 - h[k:])
            grad[k:] -= pk2 * h[:n - k]
    return dx * grad


def mirror_descent(h_init, n, dx, target_sum, t0, budget, max_time):

    p = np.ones(n_lags) / n_lags
    h = h_init.copy()

    last_improve = 0
    it = 0
    while time.time() - t0 < budget - 30:

        grad = compute_weighted_grad(h, p, dx)
        lr = 0.08 / np.sqrt(it + 10.0)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        corr = np.correlate(h_new, 1.0 - h_new, mode='full')

        if n_try < 30 or tl() < 50:
            continue
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_t = resample_h(best_h, n_try)
        h_t = project_feasible(h_t, ts_t)
        h_md, c5_md = mirror_descent(h_t, n_try, dx_t, ts_t, t0, budget_s, min(60, tl() - 20))
        if c5_md < best_c5:
            best_h, best_c5 = h_md, c5_md
            best_n, best_dx, best_ts = n_try, dx_t, ts_t
        if tl() > 25:
            h_new, c5_new = slsqp_opt(best_h, best_n, best_dx, best_ts, 5000, 2000)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
                best_n, best_dx, best_ts = len(h_new), 2.0 / len(h_new), len(h_new) / 2.0

    # Phase 4: Diversification with structured perturbations
    if tl() > 80:
        for restart in range(6):
            if tl() < 20:
                break
            h_pert = best_h.copy()
            mode = restart % 3
            if mode == 0:
                bl = np.random.randint(5, min(15, n // 3))
                st = np.random.randint(0, n - bl)
                h_pert[st:st + bl] = 1.0 - h_pert[st:st + bl]
            elif mode == 1:
                h_pert += np.random.randn(n) * 0.03
            else:
                idx = np.random.choice(n, size=max(2, n // 5), replace=False)
                h_pert[idx] = np.random.uniform(0, 1, size=len(idx))
            h_pert = project_feasible(h_pert, best_ts)
            h_md, c5_md = mirror_descent(h_pert, best_n, best_dx, best_ts, t0, budget_s, min(30, tl() - 10))
            if c5_md < best_c5:
                best_h, best_c5 = h_md, c5_md

    # Phase 5: Coordinate descent polish
    if tl() > 50:
        h_cp, c5_cp = coord_polish(best_h, best_n, best_dx, best_ts, t0, budget_s)
        if c5_cp < best_c5:
            best_h, best_c5 = h_cp, c5_cp

    # Phase 6: Final SLSQP polish
    if tl() > 15:
        h_new, c5_new = slsqp_opt(best_h, best_n, best_dx, best_ts, 50000, 5000)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Verify and return
    best_h = project_feasible(best_h, best_ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
