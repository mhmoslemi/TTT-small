    gr = (np.sqrt(5) - 1) / 2
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)

    f2 = eval_max(a, b, c, x2)

        if f1 < f2:
            hi = x2; x2 = x1; f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = eval_max(a, b, c, x1)
        else:
            lo = x1; x1 = x2; f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = eval_max(a, b, c, x2)
    return (lo + hi) / 2

    stall = 0
    total_iters = 0
    while time.time() < deadline and delta > min_delta:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            lo = 0 if max_radius >= n else max(0, i - max_radius)
            hi = n if max_radius >= n else min(n, i + max_radius + 1)
            for j in range(lo, hi):
                if i == j:
                    continue
                # Try +delta
                if h[i] + delta <= 1.0 and h[j] - delta >= 0.0:
                    h[i] += delta; h[j] -= delta
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c; improved = True
                    else:
                        h[i] -= delta; h[j] += delta
                # Try -delta
                if h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
                    h[i] -= delta; h[j] += delta
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c; improved = True
                    else:
                        h[i] += delta; h[j] -= delta
                total_iters += 1
            if time.time() > deadline:
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 2:
                delta *= 0.5
                stall = 0
    return h, best_c

def simulated_annealing(h, n, deadline, rng, T_init=0.01, cooling=0.9995, min_T=1e-6):
    """SA with pair moves to escape local optima."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    cur_c = best_c
    T = T_init
    while time.time() < deadline and T > min_T:
        T *= cooling
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.1, 0.1)
        if h[i] + d > 1.0 or h[j] - d < 0.0:
            continue
        h[i] += d; h[j] -= d
        c = compute_c5(h, n)
        delta_f = c - cur_c
        if delta_f < 0 or rng.rand() < np.exp(-delta_f / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i] -= d; h[j] += d
    return best_h, best_c

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h
