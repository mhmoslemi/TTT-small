```python
def efficient_binary_local_search(h, n, deadline, max_sweeps=1000, seed=42):
    dx = 2.0 / n
    target = n // 2
    # Convert to binary
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_max = corr.max()
    cur_c = cur_max * dx
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Precompute base arrays
    # base_pos[p] = contribution to delta[m] when h[p] goes from 0 to 1 (treating others as fixed)
    # base_neg[p] = contribution to delta[m] when h[p] goes from 1 to 0 (treating others as fixed)
    
    m_range = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For each position p, base_pos[p][k] = (1-h[p+m_range[k]]) - h[p-m_range[k]] (with boundary)
    # This is expensive to precompute for all p, but we can compute on the fly
    
    # Instead, let's compute delta for a swap (i,j) as:
    # delta = base_pos[i] + base_neg[j] + cross(i,j)
    # where base_pos[i][k] = (1-h[i+m_k]) if valid, 0 otherwise  -  h[i-m_k] if valid, 0 otherwise
    # and base_neg[j][k] = -(1-h[j+m_k]) if valid, 0 otherwise  +  h[j-m_k] if valid, 0 otherwise
    
    best_c = cur_c
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            # Compute base_pos for i
            m_pos = i + m_range  # where second factor is: i + m
            m_neg = i - m_range  # where first factor is: i - m (i.e., h[i-m] appears)

            msk = (m_pos >= 0) & (m_pos < n)
            bp[msk] += 1.0 - h[m_pos[msk]]
            msk = (m_neg >= 0) & (m_neg < n)
            bp[msk] -= h[m_neg[msk]]

            # Vectorize: for all j, compute base_neg[j] + cross(i,j)
            # base_neg[j][k] = -(1-h[j+m_k]) + h[j-m_k]
            
            # This is still O(n_ones * n) per i, which is O(n^2) per sweep
            # Let's try a different approach: just iterate but break early
            
            for j in ones:
                jn = n - 1
                m_pos_j = j + m_range
                m_neg_j = j - m_range
                bn = np.zeros(2*n-1)
                msk = (m_pos_j >= 0) & (m_pos_j < n)
                bn[msk] -= 1.0 - h[m_pos_j[msk]]
                msk = (m_neg_j >= 0) & (m_neg_j < n)
                bn[msk] += h[m_neg_j[msk]]

                total_delta[n-1] -= 2.0
                d = j - i
                k1 = n - 1 + d
                if 0 <= k1 < 2*n-1:
                    total_delta[k1] += 1.0
                k2 = n - 1 - d
                if 0 <= k2 < 2*n-1:
                    total_delta[k2] += 1.0
                
                new_corr = corr + total_delta
                nm = new_corr.max()
                if nm < cur_max - 1e-15:

                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_max = nm
                    zeros = np.where(h == 0)[0]
                    ones = np.where(h == 1)[0]
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_max * dx < best_c:
            best_c = cur_max * dx
            best_h = h.copy()
    
    return best_h, best_c
```

```python
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h_init, n, deadline, max_iters=20000, seed=42):
    """Fast binary local search with precomputed base arrays."""
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = float(corr.max())
    cur_c = cur_max * dx
    best_c = cur_c
    best_h = h.copy()
    
    m_vals = np.arange(-(n - 1), n)  # shape (2n-1,)
    n_corr = 2 * n - 1
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    # Precompute base arrays for all positions
    # base_pos[p] = delta[m] when h[p] goes 0->1 (others fixed)
    # base_neg[p] = delta[m] when h[p] goes 1->0 (others fixed)
    # base_pos[p][k] = (1-h[p+m_k]) if valid, 0 else 0  -  h[p-m_k] if valid, 0 else 0
    # base_neg[p][k] = -(1-h[p+m_k]) if valid, 0 else 0  +  h[p-m_k] if valid, 0 else 0
    
    # These change as h changes, so we can't fully precompute.
    # But we CAN compute them on the fly efficiently.
    
    it = 0
    while it < max_iters:
        if time.time() > deadline:
            break
        it += 1
        improved = False
        
        for i in list(zeros):
            if time.time() > deadline:
                break
            
            # Compute base_pos for i
            m_pos = i + m_vals  # positions of second factor
            m_neg = i - m_vals  # positions of first factor (h[i - m] = h[m_neg])
            
            bp = np.zeros(n_corr)
            msk = (m_pos >= 0) & (m_pos < n)
            bp[msk] += 1.0 - h[m_pos[msk]]
            msk = (m_neg >= 0) & (m_neg < n)
            bp[msk] -= h[m_neg[msk]]
            
            # For each j in ones, try the swap
            for j in list(ones):
                # Compute base_neg for j
                m_pos_j = j + m_vals
                m_neg_j = j - m_vals
                
                bn = np.zeros(n_corr)
                msk = (m_pos_j >= 0) & (m_pos_j < n)
                bn[msk] -= 1.0 - h[m_pos_j[msk]]
                msk = (m_neg_j >= 0) & (m_neg_j < n)
                bn[msk] += h[m_neg_j[msk]]
                
                total = bp + bn
                total[n-1] -= 2.0
                d = j - i
                k1 = n - 1 + d
                if 0 <= k1 < n_corr:
                    total[k1] += 1.0
                k2 = n - 1 - d
                if 0 <= k2 < n_corr:
                    total[k2] += 1.0
                
                new_corr = corr + total
                nm = float(new_corr.max())
                if nm < cur_max - 1e-15:
                    # Accept
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = new_corr
                    cur_max = nm
                    cur_c = cur_max * dx
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                    improved = True
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_h, best_c
```

This is still O(n²) per improvement, but with decent numpy constants.

Actually, I realize I can speed this up significantly by vectorizing over j. For fixed i, I can compute base_neg for ALL j simultaneously:

For all j in ones, base_neg[j][k] = -(1-h[j+m_k]) + h[j-m_k]

This is essentially a "correlation" between the ones and the m_vals offsets. I can compute it as:

base_neg_all[j_idx, k] = -(1 - h[ones[j_idx] + m_vals[k]]) + h[ones[j_idx] - m_vals[k]]

With proper boundary handling. This is a 2D array of shape (n_ones, 2n-1) which for n=250 is (125, 499) ≈ 62K elements. Very manageable!

Let me write the vectorized version:

```python
s_arr = np.where(h == 0)[0]

            msk = (m_neg >= 0) & (m_neg < n)
            bp[msk] -= h[m_neg[msk]]
            
            # Compute base_neg for all j in ones (shape (n_ones, 2n-1))
            # m_pos_j[j_idx, k] = ones_arr[j_idx] + m_vals[k]
            # m_neg_j[j_idx, k] = ones_arr[j_idx] - m_vals[k]
            mp = ones_arr[:, None] + m_vals[None, :]  # (n_ones, 2n-1)
            mn = ones_arr[:, None] - m_vals[None, :]
            
            # base_neg[j_idx, k] = -(1-h[mp[j_idx,k]]) + h[mn[j_idx,k]] (with boundary)
            bn = np.zeros((len(ones_arr), n_corr))
            msk_p = (mp >= 0) & (mp < n)
            bn[msk_p] -= (1.0 - h[mp[msk_p]])
            msk_m = (mn >= 0) & (mn < n)
            bn[msk_m] += h[mn[msk_m]]
            
            # Add base_pos to all
            total = bn + bp[None, :]  # (n_ones, 2n-1)
            
            # Cross terms
            total[:, n-1] -= 2.0
            # d[j_idx] = ones_arr[j_idx] - i
            d = ones_arr - i
            k1 = n - 1 + d  # (n_ones,)
            valid_k1 = (k1 >= 0) & (k1 < n_corr)
            total[np.arange(len(ones_arr))[valid_k1], k1[valid_k1]] += 1.0
            k2 = n - 1 - d
            valid_k2 = (k2 >= 0) & (k2 < n_corr)
            total[np.arange(len(ones_arr))[valid_k2], k2[valid_k2]] += 1.0
            
            # Find the j that minimizes max(corr + total[j_idx])
            new_corrs = corr[None, :] + total  # (n_ones, 2n-1)
            new_maxs = new_corrs.max(axis=1)  # (n_ones,)
            
            best_j_idx = int(np.argmin(new_maxs))
            best_nm = float(new_maxs[best_j_idx])
            
            if best_nm < cur_max - 1e-15:
                j = ones_arr[best_j_idx]
                h[i] = 1.0
                h[j] = 0.0
                corr = corr[None, :] + total[best_j_idx]  # or just corr + total[best_j_idx]
                corr = corr + total[best_j_idx]
                cur_max = best_nm
                improved = True
                if cur_max * dx < best_c:
                    best_c = cur_max * dx
                    best_h = h.copy()
                break
        
        if not improved:
            break
    
    return best_h, best_c
```

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
