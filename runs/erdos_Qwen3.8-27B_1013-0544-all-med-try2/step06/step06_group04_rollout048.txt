            break

            tot = room.sum()

            if tot < 1e-15:

                       bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})

    """Compute gradient of smooth max of all lags."""
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    m = c.max()
    sh = alpha * (c - m)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    grad = np.zeros(n)

        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        m_abs = abs(lag)
        if m_abs >= n:
            continue
        if m_abs == 0:
            grad[:] += w[k] * dx * (1.0 - 2.0 * h)
        elif lag > 0:

            grad[m_abs:] -= w[k] * dx * h[:n - m_abs]
        else:
            grad[m_abs:] += w[k] * dx * (1.0 - h[:n - m_abs])
            grad[:n - m_abs] -= w[k] * dx * h[m_abs:]
    return grad

Let me provide the analytical gradient. The gradient of the smooth max:
f(h) = (1/alpha) * log(∑_k exp(alpha * c(k))) + max_k c(k)

∇_h f = (1/∑_k exp(alpha*c(k))) * ∑_k alpha * exp(alpha*c(k)) * ∇_h c(k)
        = ∑_k w_k * ∇_h c(k)

where w_k = exp(alpha*c(k)) / ∑_j exp(alpha*c(j)) = softmax(alpha * c)

And ∇_h c(k) for lag m = k - (n-1):
- If m = 0: dx * (1 - 2h)
- If m > 0: dx * [(1 - h[m:]) for i < n-m, -h[:n-m] for i ≥ m]
- If m < 0: dx * [(1 - h[:n-m]) for i ≥ |m|, -h[|m|:] for i < n-|m|]

Let me implement this:

```python
def _smooth_full_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    ex = np.exp(sh - sh.max())
    w = ex / ex.sum()
    val = float(m + (sh.max() + np.log(ex.sum())) / alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        m_abs = abs(lag)
        if m_abs >= n:
            continue
        if m_abs == 0:
            grad[:] += w[k] * dx * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[:n - m_abs] += w[k] * dx * (1.0 - x[m_abs:])
            grad[m_abs:] -= w[k] * dx * x[:n - m_abs]
        else:
            grad[m_abs:] += w[k] * dx * (1.0 - x[:n - m_abs])
            grad[:n - m_abs] -= w[k] * dx * x[m_abs:]
    return val, grad
```

With this, SLSQP will use the analytical gradient, which is much faster. Each SLSQP iteration now costs ~n for the gradient (since most w[k] are negligible), plus the correlate computation.

Actually, the loop over k in the gradient computation is O(n) in the worst case (when all lags have non-negligible weight). With alpha large, only the top few lags have significant weight, so it's effectively O(1) per iteration.
