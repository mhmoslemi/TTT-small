def swap_delta(h, i, j, n):
    """Compute change in correlation when h[i] 0->1 and h[j] 1->0."""
    m_range = np.arange(-(n-1), n)
    dx_unused = None  # delta is in correlation units, not yet multiplied by dx

    delta[k1] += 1.0

    corr = np.correlate(h, 1.0-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h==1)[0].copy()
    zeros = np.where(h==0)[0].copy()
    
    stagnation = 0
    for it in range(max_iters):
        if time.time() > deadline: break
        if stagnation > 2000: break
        
        ai = rng.randint(len(zeros))
