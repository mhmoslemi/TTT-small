```python
def coord_desc_1d(corr_at_0, diff, lam, dx, n):
    """Solve the 1D subproblem for one coordinate."""
    # f(x) = max(corr_at_0 + diff*x, with k=0 term adjusted) + lam*x
    # For k=0 (index n-1): S_0(x) = corr_at_0[n-1] + dx*x*(1-x)
    # For k≠0: S_k(x) = corr_at_0[k] + diff[k]*x
    
    # We'll use minimize_scalar on the bounded interval [0,1]
    def f1d(x):
        vals = corr_at_0 + diff * x
        vals[n-1] = corr_at_0[n-1] + dx * x * (1.0 - x)
        return vals.max() + lam * x
    
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10})
    return res.x

def coord_descent(h, n, dx, max_sweeps, t0, time_budget):
    """Full coordinate descent with Lagrange multiplier."""
    target_sum = n / 2.0
    lam = 0.0
    lr_lam = 0.05
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_budget - 120:
            break
        
        order = np.random.permutation(n)
        
        for i in order:
            x = h[i]
            h[i] = 0.0
            corr_at_0 = np.correlate(h, 1.0 - h, mode="full") * dx
            h[i] = 1.0
            corr_at_1 = np.correlate(h, 1.0 - h, mode="full") * dx
            h[i] = x
            
            diff = corr_at_1 - corr_at_0
            
            x_new = coord_desc_1d(corr_at_0, diff, lam, dx, n)
            h[i] = x_new
        
        # Update Lagrange multiplier
        s = h.sum()
        lam += lr_lam * (s - target_sum)
        lam = np.clip(lam, -5.0, 5.0)
        
        # Project to feasible
        h = project_feasible(h, target_sum)
        
        if sweep % 10 == 9:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5
```

One more idea: **Use the structure of the problem more directly.**

The objective is max_k S_k. At the optimum, suppose the maximum is achieved at lags k_1, k_2, ..., k_m. Then (by KKT conditions) the gradient of the objective is in the cone generated by the gradients of S_{k_1}, ..., S_{k_m}.

The gradient of S_k with respect to h[i] is:
∂S_k/∂h[i] = dx * [(1-h[i+k]) - h[i-k]] (for k≥1, when both terms are valid)
(or similar for k<0)

At the optimum (ignoring the sum constraint for a moment):
∑_m α_m * ∂S_{k_m}/∂h[i] = 0 for all i (with α_m > 0, ∑α_m = 1)

This gives a system of equations. If I know which lags are active (k_1, ..., k_m), I can solve for h.

This is the idea behind the "active set" method. The challenge is that the active set changes as h changes.

A practical algorithm:
1. Start with some h
2. Find the active lags (those where S_k = max)
3. Solve the KKT system for the active set
4. Update h
5. Re-check active set
6. Repeat

This is similar to what the previous SLSQP approach does, but more directly.

If h is symmetric: h[i] = h[n-1-i], then S_k = S_{-(n-1-k)}... actually not exactly. Let me think.

Actually, S_k = dx * ∑_{j=0}^{n-1-k} h[j]*(1-h[j+k]) for k≥0.
By the symmetry h[j] = h[n-1-j]:
S_k = dx * ∑_{j=0}^{n-1-k} h[n-1-j]*(1-h[n-1-(j+k)])
= dx * ∑_{j=0}^{n-1-k} h[n-1-j]*(1-h[n-1-j-k])

Let i = n-1-j, so j goes from 0 to n-1-k means i goes from n-1 to k:
= dx * ∑_{i=k}^{n-1} h[i]*(1-h[i-k])
= S_{-k} (the definition of S for negative lag)

So S_k = S_{-k} by symmetry! This means we only need to consider k ≥ 0, and the max over all k is the same as the max over k ≥ 0.

This doesn't reduce the dimensionality (we still need to optimize all n coordinates), but it means the correlation is symmetric, which is a nice check.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s

            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = mask.sum()

        h = np.clip(h, 0.0, 1.0)
