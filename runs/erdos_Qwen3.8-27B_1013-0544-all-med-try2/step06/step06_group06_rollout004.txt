def swap_delta(h, n, a, b):
    """Compute delta in g when h[a]:1->0, h[b]:0->1. Returns array of length 2n-1."""
    # Build new h for the affected positions
    h_new = h.copy()
    h_new[a] = 0
    h_new[b] = 1
    
    delta = np.zeros(2*n-1)
    for m in range(2*n-1):
        shift = m - (n-1)
        # Affected i values: a, b, a-shift, b-shift
        affected = set()
        for i in [a, b, a-shift, b-shift]:
            if 0 <= i < n:
                j = i + shift
                if 0 <= j < n:
                    affected.add(i)
        if not affected:
            continue
        old_sum = 0.0
        new_sum = 0.0
        for i in affected:
            j = i + shift
            old_sum += h[i] * (1 - h[j])
            new_sum += h_new[i] * (1 - h_new[j])
        delta[m] = (new_sum - old_sum) * (2.0 / n)
    
    return delta
