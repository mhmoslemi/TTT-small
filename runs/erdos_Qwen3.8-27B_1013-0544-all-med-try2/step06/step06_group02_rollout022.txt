        if abs(diff) < 1e-15:

    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        top_idx = np.argsort(c_full)[-5:]
        w[top_idx] = 1.0
        w_sum = 5.0
    w = w / w_sum
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        wm = w[n - 1 + k]
        if k > 0:
            wm += w[n - 1 - k]
        if wm < 1e-14:
            continue
        wm_dx = wm * dx
        nv = n - k
        grad[:nv] -= wm_dx * h[k:]
        grad[k:] += wm_dx * (1.0 - h[:nv])
    return grad

def _lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _lbfgs_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def _con_eq(x, n):
    return float(x.sum() - n / 2.0)

def _con_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, bounds):
    best_c = compute_c5(h, n)
    configs = [
        (20, 1.0, 300), (100, 1.0, 300), (500, 1.0, 300),
        (2000, 0.5, 300), (10000, 0.1, 300), (50000, 0.1, 200),
        (200000, 0.01, 200), (1000000, 0.01, 150), (5000000, 0.001, 150)
    ]
    for alpha, mu, max_iter in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(_lbfgs_obj, h, jac=_lbfgs_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, bounds):
    best_c = compute_c5(h, n)
    for alpha in [50000, 200000, 1000000, 10000000]:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': _con_eq,
                                       'jac': _con_jac, 'args': (n,)}],
                          options={'maxiter': 800, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline):
    best_c = compute_c5(h, n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def gradient_step(h, n, alpha, lr, deadline):
    best_c = compute_c5(h, n)
    for it in range(500):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        g += 2.0 * (h.sum() - n / 2.0)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        step = lr
        improved = False
        for _ in range(50):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new; best_c = c_new; improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.7
            if lr < 1e-14:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B coarse to fine
    t1 = min(t0 + 200, deadline - 300)
    h, best_c = lbfgs_optimize(h, n, t1, bounds)
    best_h = h.copy()

    # Phase 2: SLSQP
    if time.time() < deadline - 200:
        t2 = min(time.time() + 150, deadline - 200)
        h, c2 = slsqp_optimize(best_h, n, t2, bounds)
        if c2 < best_c:
            best_c = c2
            best_h = h.copy()

    # Phase 3: Gradient descent with backtracking
    if time.time() < deadline - 150:
        t3 = min(time.time() + 60, deadline - 150)
        h3, c3 = gradient_step(best_h, n, 50000, 0.01, t3)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()

    # Phase 4: More SLSQP refinement
    if time.time() < deadline - 100:
        t4 = min(time.time() + 60, deadline - 100)
        h4, c4 = slsqp_optimize(best_h, n, t4, bounds)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()

    # Phase 5: Pair search
    if time.time() < deadline - 30:
        t5 = min(time.time() + 120, deadline - 15)
        best_h, best_c = pair_search(best_h, n, t5)

    # Phase 6: Final L-BFGS + pair search
    if time.time() < deadline - 10:
        t6 = min(time.time() + 30, deadline - 8)
        h6, c6 = lbfgs_optimize(best_h, n, t6, bounds)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        if time.time() < deadline - 3:
            best_h, best_c = pair_search(best_h, n, deadline - 1)

    # Phase 7: Try perturbation restarts
    for _ in range(3):
        if time.time() > deadline - 20:
            break
        h_pert = best_h + 0.01 * rng.randn(n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c * 1.02:
            t_r = min(time.time() + 40, deadline - 10)
            h_r, c_r = slsqp_optimize(h_pert, n, t_r, bounds)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r.copy()

    # Phase 8: Try a different n if time allows
    if time.time() < deadline - 60:
        for n_try in [200, 250]:
            if time.time() > deadline - 30:
                break
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            bounds_t = [(0.0, 1.0)] * n_try
            t_t = min(time.time() + 40, deadline - 20)
            h_t, c_t = lbfgs_optimize(h_interp, n_try, t_t, bounds_t)
            if time.time() < deadline - 10:
                h_t, c_t = slsqp_optimize(h_t, n_try, min(time.time()+20, deadline-5), bounds_t)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n = n_try

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
