```python
def compute_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    
    grad = np.zeros(n)
    nhalf = n - 1
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        shift = k - nhalf
        # Term 1: (1-x[i-shift]) for i where 0 <= i-shift < n
        # i ranges from max(0, shift) to min(n-1, shift+n-1)
        lo = max(0, shift)
        hi = min(n, n + shift)  # exclusive
        if lo < hi:
            # i-shift ranges from lo-shift to hi-1-shift
            grad[lo:hi] += w[k] * dx * (1.0 - x[lo - shift:hi - shift])
        # Term 2: -x[i+shift] for i where 0 <= i+shift < n
        # i ranges from max(0, -shift) to min(n-1, n-1-shift)
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)  # exclusive
        if lo2 < hi2:
            # i+shift ranges from lo2+shift to hi2-1+shift
            grad[lo2:hi2] -= w[k] * dx * x[lo2 + shift:hi2 - 1 + shift]
    
    return grad
```
