```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, shift):
    grad = np.zeros(n)
    m = abs(shift)
    if m >= n:
        return grad
    if m == 0:
        return 1.0 - 2.0 * h
    if shift > 0:
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def subgradient_method(h, n, max_iter=300, deadline=None):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.01
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        c_max = c.max()
        if c_max < best_c * dx_:
            pass
        threshold = c_max - 0.001
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        g = np.zeros(n)
        for idx in active:
            shift = idx - (n - 1)
            g += lag_gradient(h, n, shift)
        g /= len(active)
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        h_new = project_feasible(h - eta * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        eta *= 0.995
    return best_h, best_c

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lp_active_set(h, n, top_m, trust, deadline):
    dx = 2.0 / n
    c_full = compute_all_lags(h, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
    top_idx = np.argsort(c_full)[-top_m:]
    for idx in range(2 * n - 1):
        shift = idx - (n - 1)
        grad = lag_gradient(h, n, shift)
        cons.append(c_full[idx] + dx * (grad @ (h_var - h)) <= t_var)
    if trust is not None:
        cons.append(cp.abs(h_var - h) <= trust)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.GLOP, cp.CBC, cp.CPLEX]:
        try:
            prob.solve(solver=solver, timestamp=False)
            break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h.copy()
    if h_var.value is None:
        return h.copy()
    return project_feasible(h_var.value, n)

def gradient_local_search(h, n, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    top_m = 5
    ni = 25
    nj = 25
    delta_base = 0.02
    for sweep in range(500):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_m:]
        grads = np.zeros((len(top_idx), n))
        for li, idx in enumerate(top_idx):
            shift = idx - (n - 1)
            grads[li] = lag_gradient(h, n, shift)
        active_sum = grads.sum(axis=0)
        ci = np.argsort(active_sum)[-ni:]
        cj = np.argsort(active_sum)[:nj]
        improved = False
        for i in ci:
            if time.time() > deadline:
                break
            for j in cj:
                if i == j:
                    continue
                ok = True
                for li in range(len(top_idx)):
                    if grads[li, i] >= grads[li, j]:
                        ok = False
                        break
                if not ok:
                    continue
                delta = min(delta_base, h[i], 1.0 - h[j])
                if delta < 1e-10:
                    continue
                h[i] -= delta
                h[j] += delta
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    delta_base = max(0.005, delta_base * 0.9)
                else:
                    h[i] += delta
                    h[j] -= delta
        if not improved:
            delta_base = max(0.002, delta_base * 0.7)
            if delta_base < 0.001:
                break
    return best_h, best_c

def pair_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.4, h[i], 1.0 - h[j])
                    if delta < 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    if compute_c5(h, n) < best_c - 1e-15:
                        best_c = compute_c5(h, n)
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i]:
                    delta = min(d, (h[j] - h[i]) * 0.4, h[j], 1.0 - h[i])
                    if delta < 1e-10:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    if compute_c5(h, n) < best_c - 1e-15:
                        best_c = compute_c5(h, n)
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c
```
