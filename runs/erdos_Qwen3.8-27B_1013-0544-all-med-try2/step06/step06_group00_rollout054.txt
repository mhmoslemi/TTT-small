import numpy as np
from scipy.optimize import minimize
import time

def fft_corr_full(a, b):
    n = len(a)
    N = 2 * n - 1
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    result = np.empty(N)
    result[n - 1] = c[0]
    result[n:] = c[1:n]
    result[:n - 1] = c[n:][::-1]
    return result

def compute_c5(h, n):
    dx = 2.0 / n
    c = fft_corr_full(h, 1.0 - h)
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    c = fft_corr_full(x, 1.0 - x)
    cmax = c.max()
    ec = np.exp(alpha * (c - cmax))
    se = ec.sum()
    f = (cmax + np.log(se) / alpha) * dx + penalty * (x.sum() - n / 2.0) ** 2
    w = ec / se
    grad = np.zeros(n)
    offs = np.arange(-(n - 1), n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        off = offs[k]
        lo1 = max(0, off)
        hi1 = min(n, n + off)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - x[lo1 - off:hi1 - off])
        lo2 = max(0, -off)
        hi2 = min(n, n - off)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * x[lo2 + off:hi2 + off]
    grad *= dx
    grad += 2.0 * penalty * (x.sum() - n / 2.0)
    return f, grad

def lbfgs_opt(h, n, alpha, penalty, maxiter=300):
    h = project_feasible(h.copy(), n)
    try:
        res = minimize(lse_obj_grad, h, args=(n, alpha, penalty), method='L-BFGS-B',
                       jac=True, bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_local_search(h_init, n, deadline, max_sweeps=300):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = fft_corr_full(h, 1 - h).astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        top3 = np.argsort(corr)[::-1][:3]
        for zi in range(len(zeros)):
            if time.time() > deadline:
                break
            i = zeros[zi]
            best_j = -1
            best_mc = cur_c
            for j in ones:
                if i == j:
                    continue
                mc = 0.0
                for tl in top3:
                    off = tl - (n - 1)
                    d = 0.0
                    if 0 <= i - off < n:
                        d += 1.0 - h[i - off]
                    if 0 <= i + off < n:
                        d -= h[i + off]
                    if 0 <= j - off < n:
                        d -= 1.0 - h[j - off]
                    if 0 <= j + off < n:
                        d += h[j + off]
                    if i - off == j:
                        d += 1.0
                    if j - off == i:
                        d -= 1.0
                    v = corr[tl] + d
                    if v > mc:
                        mc = v
                if mc < best_mc - 1e-14:
                    best_mc = mc
                    best_j = j
            if best_j >= 0:
                h[i] = 1.0
                h[best_j] = 0.0
                corr = fft_corr_full(h, 1 - h).astype(float)
                cur_c = corr.max() * dx
                zeros[zi] = best_j
                ones[ones.index(best_j)] = i
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, max_iters=2000000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = fft_corr_full(h, 1 - h).astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    T = 0.01
    cool = 0.99997
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros))
        bi = rng.randint(len(ones))
        i = zeros[ai]
        j = ones[bi]
        if i == j:
            continue
        h[i] = 1.0
        h[j] = 0.0
        corr = fft_corr_full(h, 1 - h).astype(float)
        nc = corr.max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(-(nc - cur_c) / T)):
            cur_c = nc
            zeros[ai] = j
            ones[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
            corr = fft_corr_full(h, 1 - h).astype(float)
        T *= cool
    return best_h, best_c
