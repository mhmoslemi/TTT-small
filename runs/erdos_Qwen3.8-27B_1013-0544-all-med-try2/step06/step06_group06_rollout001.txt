def compute_gradient_smoothed(h, n, alpha):
    dx = 2.0/n
    c = np.correlate(h, 1-h, 'full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    # w has length 2n-1, corresponding to shifts -(n-1) to (n-1)
    
    grad = np.zeros(n)
    for i in range(n):
        for k in range(2*n-1):
            s = k - (n-1)  # shift
            # Term 1: (1-h[i+s]) if i+s valid
            j = i + s
            if 0 <= j < n:
                grad[i] += w[k] * (1 - h[j])
            # Term 2: -h[i-s] if i-s valid
            j2 = i - s
            if 0 <= j2 < n:
                grad[i] -= w[k] * h[j2]
        grad[i] *= dx
    
    return grad
```

This is O(n * 2n) = O(n²) which for n=250 is about 125K operations. Very fast!

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones"""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def smooth_objective(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted -= shifted.max()
    return c_max + np.log(np.exp(shifted).sum()) / alpha

def smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    # For efficiency, use the fact that this is a convolution
    # w has length 2n-1, index k corresponds to shift s = k - (n-1)
    for i in range(n):
        for k in range(2*n-1):
            s = k - (n-1)
            j = i + s
            if 0 <= j < n:
                grad[i] += w[k] * (1.0 - h[j])
            j2 = i - s
            if 0 <= j2 < n:
                grad[i] -= w[k] * h[j2]
    grad *= dx
    return grad

def projected_gradient_descent(h, n, alpha, lr, max_iter, deadline):
    best_h = h.copy()
    best_val = compute_c5(h, n)
    h_curr = h.copy()
    h_prev = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = smooth_gradient(h_curr, n, alpha)
        h_new = h_curr - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        # Backtracking if needed
        if smooth_objective(h_new, n, alpha) > smooth_objective(h_curr, n, alpha) + lr/2 * np.dot(grad, h_new - h_curr):
            lr *= 0.5
            h_new = h_curr - lr * grad
            h_new = project_feasible(h_new, n)
        
        h_prev = h_curr
        h_curr = h_new
    
    return best_h, best_val

def binary_local_search(h, n, deadline, max_sweeps=200):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = compute_all_c(h, n)
        improved = False
        
        # Focus on the worst shift
        s_max_idx = np.argmax(c)
        shift = s_max_idx - (n - 1)
        
        # Find "bad" positions: h[i]=1 and h[i+shift]=0
        if shift >= 0:
            bad_ones = [i for i in range(n - shift) if h[i] == 1 and h[i+shift] == 0]
            good_zeros = [j for j in range(n - shift) if h[j] == 0 and h[j+shift] == 1]
        else:
            sh = -shift
            bad_ones = [i for i in range(sh, n) if h[i] == 1 and h[i-sh] == 0]
            good_zeros = [j for j in range(sh, n) if h[j] == 0 and h[j-sh] == 1]
        
        # Try targeted swaps
        for i in bad_ones:
            if improved:
                break
            for j in good_zeros:
                if i == j:
                    continue
                h[i], h[j] = h[j], h[i]
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i], h[j] = h[j], h[i]
        
        if not improved:
            # Try all pairs (more expensive but thorough)
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            for zi in zeros:
                if improved or time.time() >
