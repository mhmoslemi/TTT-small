import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_cube_sum(h_new, n):
    """Project onto {h: 0<=h<=1, sum(h)=n/2} using bisection on Lagrange multiplier."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h_tmp = np.clip(h_new - mid, 0.0, 1.0)
        s = h_tmp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h_new - (lo + hi) / 2, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greedy_binary(h_init, n, deadline, max_sweeps=200):

    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    m_range = np.arange(-(n - 1), n)
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break

    """Projected subgradient descent on max correlation."""
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    best_c = corr.max() * dx
    cur_c = best_c

    no_improve = 0
    patience = 2000
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        cur_c = corr.max() * dx
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:

            h = best_h + 0.01 * np.random.randn(n)
            h = project_cube_sum(h, n)
            no_improve = 0
            alpha = 0.05 / n

        top_idx = np.argsort(corr)[::-1][:top_k]

        grad = np.zeros(n)
        for k_idx in top_idx:

            lo1 = max(0, -m)
            hi1 = min(n, n - m)
            if hi1 > lo1:

            lo2 = max(0, m)
            hi2 = min(n, n + m)
            if hi2 > lo2:

        grad /= top_k
        
        # Line search
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Try a few step sizes
        accepted = False

            h_new = project_cube_sum(h - step * grad, n)
            c_new = np.correlate(h_new, 1 - h_new, mode='full').max() * dx
            if c_new < cur_c - 1e-15:
                h = h_new
                cur_c = c_new
                alpha = min(alpha * 1.1, 1.0)
                accepted = True
                break
        
        if not accepted:
            alpha *= 0.5
            if alpha < 1e-10:

    return best_h, best_c

    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]
    T = 0.005
    cool = 0.99998
    
    while time.time() < deadline:
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)

                best_h = h.copy()
        T *= cool
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_cube_sum(h_new, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h_tmp = np.clip(h_new - mid, 0.0, 1.0)
        s = h_tmp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h_new - (lo + hi) / 2, 0.0, 1.0)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def gradient_max_corr(h, n, top_k=5):
    corr = np.correlate(h, 1 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for k_idx in top_idx:
        m = k_idx - (n - 1)
        lo1 = max(0, -m)
        hi1 = min(n, n - m)
        if hi1 > lo1:
            grad[lo1:hi1] += (1.0 - h[lo1 + m:hi1 + m])
        lo2 = max(0, m)
        hi2 = min(n, n + m)
        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2 - m:hi2 - m]
    grad /= top_k
    return grad, corr

def smooth_gradient(h, n, alpha):
    corr = np.correlate(h, 1 - h, mode='full')
    shifted = alpha * (corr - corr.max())
    weights = np.exp(shifted)
    weights /= weights.sum()
    active_idx = np.where(weights > 1e-8)[0]
    grad = np.zeros(n)
    for k_idx in active_idx:
        w = weights[k_idx]
        m = k_idx - (n - 1)
        lo1 = max(0, -m)
        hi1 = min(n, n - m)
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 + m:hi1 + m])
        lo2 = max(0, m)
        hi2 = min(n, n + m)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2 - m:hi2 - m]
    return grad, corr

def subgradient_descent(h_init, n, deadline, top_k=5, use_smooth=False, alpha_smooth=500.0):
    dx = 2.0 / n
    h = h_init.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    best_c = corr.max() * dx
    best_h = h.copy()
    cur_c = best_c
    momentum = np.zeros(n)
    beta = 0.9
    alpha = 0.5 / n
    no_improve = 0
    patience = 3000

    for it in range(200000):
        if time.time() > deadline:
            break
        if use_smooth:
            grad, corr = smooth_gradient(h, n, alpha_smooth)
        else:
            grad, corr = gradient_max_corr(h, n, top_k)
        cur_c = corr.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > patience:
            h = best_h + 0.01 * np.random.randn(n)
            h = project_cube_sum(h, n)
            momentum *= 0.5
            alpha = 0.3 / n
            no_improve = 0
            continue
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        momentum = beta * momentum - alpha * grad
        accepted = False
        for trial in range(12):
            h_new = project_cube_sum(h + momentum, n)
            c_new = np.correlate(h_new, 1 - h_new, mode='full').max() * dx
            if c_new < cur_c - 1e-14:
                h = h_new
                cur_c = c_new
                alpha = min(alpha * 1.05, 1.0)
                accepted = True
                break
            momentum = momentum * 0.5 - alpha * 0.5 * grad
        if not accepted:
            alpha *= 0.5
            if alpha < 1e-12:
                break
    return best_h, best_c

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    m_range = np.arange(-(n - 1), n)
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    m_range = np.arange(-(n - 1), n)
    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]
    T = 0.004
    cool = 0.99999
    while time.time() < deadline:
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        delta[n - 1 + (j - i)] += 1.0
        delta[n - 1 + (i - j)] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(-(nc - cur_c) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
