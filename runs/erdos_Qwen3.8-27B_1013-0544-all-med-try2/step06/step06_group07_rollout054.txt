        # Valid if 0 <= i-lag < n
        if 0 <= i - lag < n:
            b_k = h[i - lag]  # h[i-lag] * (1-x)
        
        # The correlation contribution from position i:
        # a_k * x + b_k * (1-x) = (a_k - b_k) * x + b_k
        slope_k = a_k - b_k
        intercept_k = b_k  # contribution when x=0
        
        # But we also need the rest of the correlation (all j != i terms)
        # For now, let's compute the full correlation at x=0 and x=1
        # Actually, let me just compute slopes and intercepts directly
        slopes.append(slope_k)
        intercepts.append(intercept_k)
    
    # Now f(x) = max_k (slopes[k]*x + intercepts[k] + rest_k) * dx
    # where rest_k is the part of corr[k] not involving h[i]
    # Hmm, this is getting complicated. Let me just compute corr at x=0 and x=1
    # to get the full picture, then interpolate.
    
    # Actually, the key insight: corr[k](x) = corr[k](0) + x * (partial corr[k] / partial x)
    # The partial is:
    #   (1-h[i+lag])*1_{valid} - h[i-lag]*1_{valid}
    #   = a_k - b_k (as computed above)
    
    # So corr[k](x) = corr[k](0) + x * slope_k
    # And f(x) = max_k (corr[k](0) + x * slope_k) * dx
    
    # Since we just need argmax, we can work with the unscaled values
    # Let's compute corr at x=0 (set h[i]=0)
    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    corr1 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 0.0  # reset
    
    # corr[k](x) = corr0[k] + x * (corr1[k] - corr0[k])
    slopes = corr1 - corr0  # this is the slope for each k
    
    # f(x) = max_k (corr0[k] + x * slopes[k]) * dx
    # Minimize over x in [0,1]
    # This is a convex piecewise linear function
    
    # Find minimum: check x=0, x=1, and breakpoints
    # Breakpoints: corr0[a] + x*slopes[a] = corr0[b] + x*slopes[b]
    # => x = (corr0[b] - corr0[a]) / (slopes[a] - slopes[b])
    
    candidates = [0.0, 1.0]
    
    # For efficiency, only consider the "active" set (top few)
    # But let's be thorough: find all breakpoints in (0,1)
    # This is O(m^2) where m = 2n-1
    m = len(corr0)
    # To save time, only consider pairs where both lines could be near the max
    # Simple approach: just check a subset
    
    # Actually for n=70, m=139, m^2/2 ≈ 9700 which is fine
    for a in range(m):
        for b in range(a+1, m):
            ds = slopes[a] - slopes[b]
            if abs(ds) < 1e-15:
                continue
            x_bp = (corr0[b] - corr0[a]) / ds
            if 0.0 < x_bp < 1.0:
                candidates.append(x_bp)
    
    # Evaluate f at all candidates
    best_x = 0.0
    best_val = np.inf
    for x in candidates:
        vals = corr0 + x * slopes
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_x = x
    
    return best_x

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_fn(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def solve_1d_coord(h, i, n, topk=20):
    """Find optimal h[i] given all other coordinates fixed.
    Returns the optimal value in [0,1]."""
    orig = h[i]
    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    corr1 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = orig
    slopes = corr1 - corr0
    
    # Active set: top lines at x=0 and x=1
    top0 = np.argsort(corr0)[-topk:]
    top1 = np.argsort(corr1)[-topk:]
    active = np.unique(np.concatenate([top0, top1]))
    
    candidates = [0.0, 1.0]
    for a_idx in range(len(active)):
        for b_idx in range(a_idx + 1, len(active)):
            a, b = active[a_idx], active[b_idx]
            ds = slopes[a] - slopes[b]
            if abs(ds) < 1e-15:
                continue
            x_bp = (corr0[b] - corr0[a]) / ds
            if 1e-10 < x_bp < 1.0 - 1e-10:
                candidates.append(x_bp)
    
    best_x = 0.0
    best_val = np.inf
    for x in candidates:
        v = np.max(corr0 + x * slopes)
        if v < best_val - 1e-15:
            best_val = v
            best_x = x
    
    return best_x

def coord_descent(h, n, target_sum, max_sweeps=100):
    """Run coordinate descent with exact 1D subproblems."""
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            new_val = solve_1d_coord(h, i, n)
            if abs(new_val - h[i]) > 1e-10:
                h[i] = new_val
                improved = True
        # Project to satisfy sum constraint
        h = project_feasible(h, target_sum)
        if not improved:
            break
    return h

    np.random.seed(seed)
    t0 = time.time()
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Get initial point
    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        h_init = np.full(70, 0.5)
    
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Coordinate descent (exact 1D subproblems)
    for sweepbatch in range(20):
        if remaining() < 30:
            break
        h = best_h.copy()
        h = coord_descent(h, n, target_sum, max_sweeps=50)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_h = h.copy()
            best_c5 = c5
        else:
            # Try perturbing
            h = best_h + np.random.randn(n) * 0.02
            h = project_feasible(h, target_sum)
            h = coord_descent(h, n, target_sum, max_sweeps=30)
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-10:
                best_h = h.copy()
                best_c5 = c5
    
    # Phase 2: SLSQP polish
    if remaining() > 40:
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': lambda h, ts=target_sum: h.sum() - ts}]
        for beta in [200, 1000, 5000, 20000]:
            if remaining() < 20:
                break
            try:
                res = minimize(obj_fn, best_h, args=(dx, beta), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
            except:
                pass
    
    # Phase 3: More coordinate descent + perturb
    if remaining() > 60:
        for attempt in range(50):
            if remaining() < 20:
                break
            h = best_h.copy()
            # Add structured perturbation
            scale = 0.01 * (1.0 + np.random.rand())
            h += np.random.randn(n) * scale
            h = project_feasible(h, target_sum)
            h = coord_descent(h, n, target_sum, max_sweeps=30)
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_h = h.copy()
                best_c5 = c5
    
    # Phase 4: Try different resolutions
    if remaining() > 80:
        for n_try in [n - 5, n + 5, n - 10, n + 10, n + 20]:
            if n_try < 30 or remaining() < 40:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new_arr = np.linspace(0, 2, n_try, endpoint=False)
            h_t = np.interp(x_new_arr, x_old, best_h)
            h_t = project_feasible(h_t, ts_t)
            
            # Coordinate descent at new resolution
            h_t = coord_descent(h_t, n_try, ts_t, max_sweeps=50)
            c5_t = compute_c5(h_t)
            
            if c5_t < best_c5:
                best_h = h_t.copy()
                best_c5 = c5_t
                n = n_try
                dx = dx_t
                target_sum = ts_t
                continue
            
            # SLSQP at new resolution
            if remaining() > 20:
                bounds_t = Bounds(np.zeros(n_try), np.ones(n_try))
                cons_t = [{'type': 'eq', 'fun': lambda h, ts=ts_t: h.sum() - ts}]
                try:
                    res = minimize(obj_fn, h_t, args=(dx_t, 1000), method='SLSQP',
                                  bounds=bounds_t, constraints=cons_t,
                                  options={'maxiter': 3000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new
                        best_c5 = c5
                        n = n_try
                        dx = dx_t
                        target_sum = ts_t
                except:
                    pass
    
    # Phase 5: Final polish
    if remaining() > 15:
        h = best_h.copy()
        h = coord_descent(h, n, target_sum, max_sweeps=100)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    return best_h, best_c5, len(best_h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_fn(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def solve_1d(h, i, n, topk=25):
    orig = h[i]
    h[i] = 0.0
    c0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    c1 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = orig
    sl = c1 - c0
    t0 = np.argsort(c0)[-topk:]
    t1 = np.argsort(c1)[-topk:]
    act = np.unique(np.concatenate([t0, t1]))
    cands = [0.0, 1.0]
    for a in range(len(act)):
        for bi in range(a + 1, len(act)):
            sa = sl[act[a]] - sl[act[bi]]
            if abs(sa) < 1e-15:
                continue
            xbp = (c0[act[bi]] - c0[act[a]]) / sa
            if 1e-12 < xbp < 1.0 - 1e-12:
                cands.append(xbp)
    bx, bv = 0.0, np.inf
    for x in cands:
        v = np.max(c0 + x * sl)
        if v < bv:
            bv = v
            bx = x
    return bx

    for _ in range(sweeps):
        changed = False
        for i in range(n):
            nv = solve_1d(h, i, n)
            nv = h[i] + damp * (nv - h[i])
            if abs(nv - h[i]) > 1e-10:
                h[i] = nv
                changed = True
        h = project_feasible(h, ts)
        if not changed:
            break
    return h
