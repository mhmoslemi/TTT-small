import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    elif lag > 0:
        grad[:n - lag] = -dx * h[lag:]
        grad[lag:] = dx * (1.0 - h[:n - lag])
    else:
        m = -lag
        grad[:n - m] = dx * (1.0 - h[m:])
        grad[m:] = -dx * h[:n - m]
    return grad

def outer_approx_step(h_curr, n, trust, cuts=None):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    cutoff = c.max() - 0.02
    for k in range(2 * n - 1):
        if c[k] < cutoff and cuts is None:
            continue
        lag = k - (n - 1)
        grad = lag_gradient(h_curr, n, lag)
        cons.append(c[k] + grad @ (h_var - h_curr) <= t_var)
    if cuts is not None:
        for (h_ref, c_ref, grad_ref) in cuts:
            cons.append(c_ref + grad_ref @ (h_var - h_ref) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for s in [cp.PDLP, cp.GLOP, cp.ECOS, cp.CBC]:
        try:
            prob.solve(solver=s)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    break
            except Exception:
                continue
    if h_var.value is None:
        return h_curr.copy(), None
    h_new = np.clip(h_var.value, 0.0, 1.0)
    return project_feasible(h_new, n), c

def active_slsqp(h, n, top_idx, alpha, maxiter=3000, ftol=1e-15):
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        active = c[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        return val
    def obj_grad(x):
        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        active = c[top_idx]
        m = active.max()
        shifted = alpha * (active - m)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            lag = ki - (n - 1)
            grad += wi * lag_gradient(x, n, lag)
        return grad
    def sum_con(x):
        return np.sum(x) - n / 2.0
    def sum_con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_polish(h, n, deadline, top_m=8):
    h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [2000, 5000, 10000]
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for alpha in alphas:
        for ss in steps:
            if time.time() > deadline:
                break
            for _ in range(100):
                if time.time() > deadline:
                    break
                c = compute_all_lags(h, n)
                top_idx = np.argsort(c)[-top_m:]
                active = c[top_idx]
                m = active.max()
                shifted = alpha * (active - m)
                mx = shifted.max()
                w = np.exp(shifted - mx)
                w /= w.sum()
                grad = np.zeros(n)
                for wi, ki in zip(w, top_idx):
                    lag = ki - (n - 1)
                    grad += wi * lag_gradient(h, n, lag)
                grad -= np.mean(grad)
                h_new = np.clip(h - ss * grad, 0.0, 1.0)
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-13:
                    best_c = c_new
                    h = h_new
                else:
                    break
    return h, best_c

def coord_search(h, n, deadline, max_rounds=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.004, 0.002, 0.001]
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(max_rounds):
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            improved = False
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j]:
                    delta = min(d, (h[i] - h[j]) * 0.4, h[i] * 0.8, (1.0 - h[j]) * 0.8)
                    if delta < 1e-7:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i]:
                    delta = min(d, (h[j] - h[i]) * 0.4, h[j] * 0.8, (1.0 - h[i]) * 0.8)
                    if delta < 1e-7:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
            if not improved:
                break
    return h, best_c

def optimize_n(h_start, n, deadline):
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Phase 1: Outer approximation
    if time.time() < deadline - 40:
        oo_end = min(time.time() + 100, deadline - 40)
        h_oo = h.copy()
        trust = 0.2
        for it in range(50):
            if time.time() > oo_end:
                break
            h_new, _ = outer_approx_step(h_oo, n, trust)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new.copy()
            step_sz = np.max(np.abs(h_new - h_oo))
            if step_sz < 1e-9:
                break
            h_oo = h_new
            trust = max(0.003, trust * 0.85)
        h = best_h.copy()
    # Phase 2: Active set SLSQP
    if time.time() < deadline - 20:
        for alpha in [5000, 15000, 40000]:
            if time.time() > deadline - 20:
                break
            c_lags = compute_all_lags(h, n)
            top_idx = np.argsort(c_lags)[-10:]
            top_idx = top_idx[np.argsort(c_lags[top_idx])[::-1]]
            h_new = active_slsqp(h, n, top_idx, alpha, maxiter=2500, ftol=1e-14)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
    # Phase 3: Gradient polish
    if time.time() < deadline - 5:
        h_new, c_new = gradient_polish(h, n, deadline - 3, top_m=10)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        h = best_h.copy()
    # Phase 4: Coordinate search
    if time.time() < deadline - 1:
        h_new, c_new = coord_search(h, n, deadline - 0.5)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
