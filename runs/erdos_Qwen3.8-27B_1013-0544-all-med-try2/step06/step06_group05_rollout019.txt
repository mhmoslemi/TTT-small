        grad[:n-L] += (1 - h[L:])
    else:
        # j+L >= 0 means j >= -L, so j in [-L, n-1]
        # j+L in [0, n-1+L]
        grad[-L:] += (1 - h[:n+L])
    
    # Second term: -h[j-L] where 0 <= j-L < n
    if L >= 0:
        # j-L >= 0 means j >= L, so j in [L, n-1]
        # j-L in [0, n-1-L]
        grad[L:] -= h[:n-L]
    else:
        # j-L < n means j < n+L, so j in [0, n+L-1] (since L<0, n+L < n)
        # j-L = j+|L| in [|L|, n-1]
        grad[:n+L] -= h[-L:]
    
    return grad * dx
```

def single_lag_grad(h, L):
    n = len(h)
    grad = np.zeros(n)
    if L >= 0:
        if L < n:
            grad[:n-L] += (1 - h[L:])

    else:
        negL = -L
        if negL < n:
            grad[negL:] += (1 - h[:n-negL])
            grad[:n-negL] -= h[negL:]
    return grad

def topk_subgradient(h, k=5, sharpen=50):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    top_idx = np.argsort(c)[-k:]
    top_vals = c[top_idx]
    max_val = top_vals.max()
    weights = np.exp(sharpen * (top_vals - max_val))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        L = int(idx) - (n - 1)
        grad += w * single_lag_grad(h, L)
    
    return grad * dx, float(c.max() * dx)

Actually, let me think about this differently. The key insight is:

The C₅ is the maximum over lags of the correlation. If we can identify which lags are "active" (close to the max), we can focus our optimization on reducing those specific correlations.

For a given active lag L, the contribution to c_L is: sum_i h[i]*(1-h[i+L]). To reduce this, we want to either:
- Decrease h[i] where h[i] is high and h[i+L] is low (reduces the product h[i]*(1-h[i+L]))
- Or increase h[i+L] where h[i] is high (same effect)

This gives a natural "gradient" direction for each active lag.

    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:

            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, L):
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        if L < n:
            g[:n - L] += 1.0 - h[L:]
            g[L:] -= h[:n - L]
    else:
        m = -L
        if m < n:
            g[m:] += 1.0 - h[:n - m]
            g[:n - m] -= h[m:]
    return g

def obj_lse(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    return m + (sm + np.log(np.exp(s - sm).sum())) / alpha

def s1_con(x, n):
    return x.sum() - n / 2.0

def s1_con_j(x, n):
    return np.ones(n)

def slsqp_step(h, alpha, maxiter=2000):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': s1_con, 'jac': s1_con_j, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x)
    except Exception:
        return h.copy()

    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    n_it = 3000

        if time.time() > deadline:

        c = np.correlate(h, 1 - h, mode='full') * dx
        top = np.argsort(c)[-k:]
        tv = c[top]
        mw = np.exp(sharp * (tv - tv.max()))
        mw /= mw.sum()
        g = np.zeros(n)
        for idx, w in zip(top, mw):
            g += w * lag_grad(h, int(idx) - (n - 1))
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        h_new = project_feasible(h - lr * g / gn)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def gradient_guided_search(h, deadline, n_iters=50000):
    n = len(h)
    dx = 2.0 / n
    best_c = compute_c5(h)
    best_h = h.copy()
    
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in steps:
        if time.time() > deadline:
            break
        for it in range(n_iters):
            if time.time() > deadline:
                break
            # Compute subgradient of the max
            c = np.correlate(best_h, 1 - best_h, mode='full')
            k_star = int(c.argmax())
            L = k_star - (n - 1)
            g = lag_grad(best_h, L)
            
            # Find best pair: i with most negative g (increase h[i]), j with most positive g (decrease h[j])
            # Or i with most positive g (decrease h[i]), j with most negative g (increase h[j])
            
            # Option 1: increase h[i], decrease h[j]
            i = int(g.argmin())  # most negative -> increasing helps
            j = int(g.argmax())  # most positive -> decreasing helps
            
            if best_h[i] < 1 and best_h[j] > 0:
                h_trial = best_h.copy()
                h_trial[i] += d
                h_trial[j] -= d
                if 0 <= h_trial[i] <= 1 and 0 <= h_trial[j] <= 1:
                    c_new = compute_c5(h_trial)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h_trial
            
            # Option 2: decrease h[i], increase h[j]
            i2 = int(g.argmax())
            j2 = int(g.argmin())
            if best_h[i2] > 0 and best_h[j2] < 1:
                h_trial = best_h.copy()
                h_trial[i2] -= d
                h_trial[j2] += d
                if 0 <= h_trial[i2] <= 1 and 0 <= h_trial[j2] <= 1:
                    c_new = compute_c5(h_trial)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h_trial
            
            # Also try top-5 candidates for diversity
            neg_idx = np.argsort(g)[:5]
            pos_idx = np.argsort(g)[-5:][::-1]
            for i in neg_idx:
                for j in pos_idx:
                    if i == j:
                        continue
                    if best_h[i] < 1 and best_h[j] > 0:
                        h_trial = best_h.copy()
                        h_trial[i] += d
                        h_trial[j] -= d
                        if 0 <= h_trial[i] <= 1 and 0 <= h_trial[j] <= 1:
                            c_new = compute_c5(h_trial)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                best_h = h_trial
    
    return best_h, best_c
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, L):
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        if L < n:
            g[:n - L] += 1.0 - h[L:]
            g[L:] -= h[:n - L]
    else:
        m = -L
        if m < n:
            g[m:] += 1.0 - h[:n - m]
            g[:n - m] -= h[m:]
    return g

def obj_lse(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    e = np.exp(s - sm)
    return m + (sm + np.log(e.sum())) / alpha

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_con_j(x, n):
    return np.ones(n)

def slsqp_step(h, alpha, maxiter=2000):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_j, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x)
    except Exception:
        return h.copy()

def subgrad_opt(h, deadline, k=5, sharp=40, lr0=0.03, lr1=5e-6, n_it=5000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    for it in range(n_it):
        if time.time() > deadline:
            break
        lr = lr0 * (lr1 / lr0) ** (it / n_it)
        c = np.correlate(h, 1 - h, mode='full')
        top = np.argsort(c)[-k:]
        tv = c[top]
        mw = np.exp(sharp * (tv - tv.max()))
        mw /= mw.sum()
        g = np.zeros(n)
        for idx, w in zip(top, mw):
            g += w * lag_grad(h, int(idx) - (n - 1))
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = project_feasible(h - lr * g / gn)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def grad_guided_search(h, deadline, steps=None, top_k=4):
    n = len(h)
    best_c = compute_c5(h)
    best_h = h.copy()
    if steps is None:
        steps = [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
    for d in steps:
        if time.time() > deadline:
            break
        stall = 0
        while time.time() < deadline:
            c = np.correlate(best_h, 1 - best_h, mode='full')
            k_star = int(c.argmax())
            L = k_star - (n - 1)
            g = lag_grad(best_h, L)
            neg_idx = np.argsort(g)[:top_k]
            pos_idx = np.argsort(g)[-top_k:][::-1]
            found = False
            for i in neg_idx:
                if best_h[i] >= 1.0:
                    continue
                for j in pos_idx:
                    if i == j or best_h[j] <= 0.0:
                        continue
                    amt = min(d, 1.0 - best_h[i], best_h[j])
                    if amt < 1e-10:
                        continue
                    h_t = best_h
                    h_t[i] += amt
                    h_t[j] -= amt
                    c_new = compute_c5(h_t)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_h = h_t.copy()
                        h_t[i] -= amt
                        h_t[j] += amt
                        found = True
                    else:
                        h_t[i] -= amt
                        h_t[j] += amt
            if not found:
                break
    return best_h, best_c

def pair_search_near(h, deadline, radius=30, steps=None):
    n = len(h)
    best_c = compute_c5(h)
    best_h = h.copy()
    if steps is None:
        steps = [0.002, 0.001, 0.0005, 0.0002, 8e-5, 4e-5, 2e-5, 1e-5]
    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                jlo = max(0, i - radius)
                jhi = min(n, i + radius + 1)
                for j in range(jlo, jhi):
                    if i == j:
                        continue
                    if best_h[i] < 1.0 and best_h[j] > 0.0:
                        a = min(d, 1.0 - best_h[i], best_h[j])
                        best_h[i] += a
                        best_h[j] -= a
                        c_new = compute_c5(best_h)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            best_h[i] -= a
                            best_h[j] += a
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy())
    best_h = h.copy()
    best_c = compute_c5(h)

    for a in [300, 1000, 5000, 20000, 100000]:
        if time.time() > deadline - 30:
            break
        ht = slsqp_step(best_h, a, maxiter=1500)
        c = compute_c5(ht)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    if time.time() < deadline - 20:
        ht, c = subgrad_opt(best_h, deadline - 15, k=5, sharp=40, lr0=0.02, lr1=5e-6, n_it=4000)
        if c < best_c:
            best_c = c
            best_h = ht

    if time.time() < deadline - 10:
        ht, c = grad_guided_search(best_h, deadline - 5, top_k=4)
        if c < best_c:
            best_c = c
            best_h = ht

    if time.time() < deadline - 2:
        ht, c = pair_search_near(best_h, deadline - 0.5, radius=20,
                                  steps=[5e-5, 2e-5, 1e-5, 5e-6, 2e-6])
        if c < best_c:
            best_c = c
            best_h = ht

    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
