```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_gradient(h, alpha):
    """Compute gradient of LSE approximation of max(c)."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    scaled = alpha * c * dx
    scaled -= scaled.max()
    exps = np.exp(scaled)
    w = exps / exps.sum()
    
    # Use FFT for gradient computation
    # T1[j] = sum_i (1-h[i]) * w[j-i+n-1]
    # T2[j] = sum_i h[i] * w[i-j+n-1]
    
    # T1 = conv(1-h, w) at position n-1+j
    N = 1
    while N < 3 * n:
        N *= 2
    t1 = np.real(np.fft.ifft(np.fft.fft(1-h, N) * np.fft.fft(w, N)))[n-1:n-1+n]
    
    # T2[j] = sum_i h[i] * w[i-j+n-1]
    # = conv(h, w_flipped) at position j+n-1  where w_flipped = w[::-1]
    t2 = np.real(np.fft.ifft(np.fft.fft(h, N) * np.fft.fft(w[::-1], N)))[n-1:n-1+n]
    
    grad = dx * (t1 - t2)
    return grad

def pgd_optimize(h, n, alpha, lr, n_iters, deadline):
    """Projected gradient descent with LSE surrogate."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(n_iters):
        if time.time() > deadline:
            break
        grad = lse_gradient(h, alpha)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        h = h_new
        if it % 100 == 0:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    scaled = alpha * c * dx
    scaled -= scaled.max()
    exps = np.exp(scaled)
    w = exps / exps.sum()
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    nm = n - 1
    
    # T1[j] = sum_i (1-h[i]) * w[j-i+nm]
    # = conv(1-h, w)[j+nm]
    N = 1
    while N < 3 * n:
        N *= 2
    t1_conv = np.real(np.fft.ifft(np.fft.fft(1-h, N) * np.fft.fft(w, N)))
    t1 = t1_conv[nm : nm+n]
    
    # T2[j] = sum_i h[i] * w[i-j+nm]
    # Let's verify: this equals conv(h, w_rev)[j + something]
    # w_rev[k] = w[2*nm - k] (reversed w, length 2n-1)
    # conv(h, w_rev)[m] = sum_k h[k] * w_rev[m-k] = sum_k h[k] * w[2*nm-(m-k)]
    # I want: sum_k h[k] * w[k-j+nm]
    # So: 2*nm-(m-k) = k-j+nm => 2nm-m+k = k-j+nm => 2nm-m = -j+nm => m = j+nm
    # Check: conv(h, w_rev)[j+nm] = sum_k h[k] * w[2*nm-((j+nm)-k)] = sum_k h[k] * w[2*nm-j-nm+k] = sum_k h[k] * w[nm-j+k]
    # I need: w[k-j+nm] = w[nm+k-j]. Yes! ✓
    
    w_rev = w[::-1]
    t2_conv = np.real(np.fft.ifft(np.fft.fft(h, N) * np.fft.fft(w_rev, N)))
    t2 = t2_conv[nm : nm+n]  # positions nm+0 to nm+n-1, i.e., j+nm for j=0..n-1
    
    grad = dx * (t1 - t2)
    return grad
```
