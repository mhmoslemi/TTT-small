import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if m >= n:
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def subgradient_method(h, n, deadline, max_iters=50000):
    """Projected subgradient method for bottleneck problem."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lr = 0.001
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        if c.max() < best_c:
            best_c = c.max()
            best_h = h_curr.copy()
        # Use top few active constraints for better gradient
        top_k = np.argsort(c)[-3:]
        # Weight by how close to max (softmax-like)
        weights = np.exp(100 * (c[top_k] - c.max()))
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, ki in zip(weights, top_k):
            lag = ki - (n - 1)
            grad += w * _lag_gradient(h_curr, n, lag)
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        step = lr * grad / grad_norm  # normalized step
        h_new = h_curr - step
        h_new = project_feasible(h_new, n)
        # Check if we improved
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # Decay learning rate
        lr *= 0.9999
        if lr < 1e-6:
            lr = 1e-6
        h_curr = h_new
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if m >= n:
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def _smooth_max_obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def projected_subgradient(h, n, deadline, n_iter=None):
    """Projected subgradient with momentum for bottleneck problem."""
    h_curr = h.copy()
    vel = np.zeros(n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lr = 0.005
    momentum = 0.9
    count = 0
    stuck = 0
    while time.time() < deadline:
        c = compute_all_lags(h_curr, n)
        if c.max() < best_c - 1e-12:
            best_c = c.max()
            best_h = h_curr.copy()
            stuck = 0
        else:
            stuck += 1
        # Weighted gradient from top constraints
        top_k = np.argsort(c)[-5:]
        # Exponential weights based on gap to max
        gaps = c.max() - c[top_k]
        weights = np.exp(-gaps * 50)
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, ki in zip(weights, top_k):
            lag = ki - (n - 1)
            grad += w * _lag_gradient(h_curr, n, lag)
        # Momentum step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        vel = momentum * vel - lr * grad / grad_norm
        h_new = project_feasible(h_curr + vel, n)
        h_curr = h_new
        count += 1
        # Adaptive learning rate
        if stuck > 100:
            lr *= 0.9
            stuck = 0
        if lr < 1e-7:
            lr = 1e-7
        if n_iter and count >= n_iter:
            break
    return best_h, best_c
