    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, k):
    """Gradient of lag k overlap w.r.t. h"""
    grad = np.zeros(n)
    m = abs(k)
    if m >= n:
        return grad
    if m == 0:

        return grad
    if k > 0:

        # dL/dh[i] = (1-h[i+m]) for i < n-m

        grad[:n-m] += (1.0 - h[m:])

    c = compute_all_lags(h, n)
    # Find top lags
    top_idx = np.argsort(c)[-top_m:]

        grad += lag_gradient(h, n, k)

So we solve:
[A A^T, A 1; 1^T A, n] [μ; ν] = [1; 0]

This is an (m+1) x (m+1) system, very fast to solve!

def steepest_descent_dir(h, n, top_idx):
    """Find direction that decreases all active lags at equal rate, with sum=0"""
    m = len(top_idx)
    A = np.zeros((m, n))
    for i, idx in enumerate(top_idx):
        lag = idx - (n - 1)
        A[i] = lag_grad(h, n, lag)
    # Solve augmented system
    AA = A @ A.T  # m x m
    A1 = A @ np.ones(n)  # m x 1
    n_val = float(n)
    # Matrix: [AA, A1; A1^T, n]
    M = np.zeros((m+1, m+1))
    M[:m, :m] = AA
    M[:m, m] = A1
    M[m, :m] = A1
    M[m, m] = n_val
    b = np.zeros(m+1)
    b[:m] = 1.0
    # b[m] = 0
    try:
        sol = np.linalg.solve(M, b)
    except:
        sol = np.linalg.lstsq(M, b, rcond=None)[0]
    mu = sol[:m]
    nu = sol[m]
    d = -A.T @ mu - nu * np.ones(n)
    return d

def binary_snap(h, n, threshold=0.3):
    """Push values towards 0 or 1, keeping sum constraint"""
    h_new = np.where(h > threshold, 1.0, 0.0)
    # Adjust to meet sum constraint
    ones = (h_new == 1.0).sum()
    target = n // 2

        # Turn off the ones with smallest original h values
        one_indices = np.where(h_new == 1.0)[0]
        order = np.argsort(h[one_indices])[:ones - target]
        h_new[one_indices[order]] = 0.0

        zero_indices = np.where(h_new == 0.0)[0]
        order = np.argsort(-h[zero_indices])[:target - ones]
        h_new[zero_indices[order]] = 1.0
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[lag:] += (1.0 - h[:n-lag])
        grad[:n-lag] -= h[lag:]
    else:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

def steepest_dir(h, n, top_idx):
    m = len(top_idx)
    A = np.zeros((m, n))
    for i, idx in enumerate(top_idx):
        lag = int(idx) - (n - 1)
        A[i] = lag_grad(h, n, lag)
    AA = A @ A.T
    A1 = A.sum(axis=1)
    M = np.zeros((m+1, m+1))
    M[:m, :m] = AA
    M[:m, m] = A1
    M[m, :m] = A1
    M[m, m] = float(n)
    b = np.zeros(m+1)
    b[:m] = 1.0
    try:
        sol = np.linalg.solve(M, b)
    except:
        sol = np.linalg.lstsq(M, b, rcond=None)[0]
    mu = sol[:m]
    nu = sol[m]
    d = -A.T @ mu - nu * np.ones(n)
    return d

    if top_m is None:

        if pos.any():
            t_max = min(t_max, ((1-h[pos]) / d[pos]).min() * 0.99)
        t_max = max(t_max, 1e-8)
        improved = False
        for frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:

def coord_search(h, n, deadline, max_rounds=500):

def smooth_obj(x, n, alpha, top_idx):
    dx = 2.0 / n
    c_full = np.correlate(x, 1 - x, mode='full') * dx
    c = c_full[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
