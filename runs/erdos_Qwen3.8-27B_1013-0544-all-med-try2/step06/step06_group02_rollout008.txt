from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    """Convert to binary, maintaining sum = n/2."""
    target = n // 2
    # Sort by value, keep top 'target' as 1
    h_copy = h.copy()
    # Use thresholding with tie-breaking
    thresh = np.sort(h_copy)[::-1][target-1] if target > 0 else 0
    binary = (h_copy > thresh).astype(float)
    # Adjust for exact count
    n_ones = int(binary.sum())
    if n_ones > target:
        # Set some 1s to 0 (those closest to threshold)
        ones_idx = np.where(binary == 1)[0]
        sorted_ones = ones_idx[np.argsort(h_copy[ones_idx])]
        for i in range(n_ones - target):
            binary[sorted_ones[i]] = 0
    elif n_ones < target:
        zeros_idx = np.where(binary == 0)[0]
        sorted_zeros = zeros_idx[np.argsort(-h_copy[zeros_idx])]
        for i in range(target - n_ones):
            binary[sorted_zeros[i]] = 1
    return binary

def local_search_binary(h, n, deadline, max_rounds=100):
    """Local search on binary solutions: try all single swaps."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Try moving each 1 to each 0
        for i in list(ones):
            if time.time() > deadline:
                return h, best_c
            h[i] = 0
            for j in zeros:
                if j == i: continue
                h[j] = 1
                c = compute_c5(h, n)
                h[j] = 0
                if c < best_c - 1e-14:
                    h[i] = 0
                    h[j] = 1
                    ones.remove(i)
                    ones.add(j)
                    zeros.remove(j)
                    zeros.add(i)
                    best_c = c
                    improved = True
                    break
            if h[i] == 0 and i not in ones:
                # i was already moved
                continue
            elif i in ones and h[i] == 0:
                h[i] = 1  # restore
        if not improved:
            break
    return h, best_c

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

I need the O(n) delta approach. Let me implement it properly.

For a binary vector h, the correlation c[m] = sum_i h[i]*(1-h[i+m-(n-1)])

When we swap h[p]=1→0 and h[q]=0→1:
- c changes by: (effect of p going to 0) + (effect of q going to 1)

Effect of h[p] going from 1 to 0:
- c[m] = sum_i h[i]*(1-h[i+k]) where k=m-(n-1)
- Terms involving h[p]:
  - i=p: h[p]*(1-h[p+k]) = 1*(1-h[p+k]) for valid p+k. After: 0. Change: -(1-h[p+k]) = h[p+k]-1
  - i+k=p, i.e., i=p-k: h[p-k]*(1-h[p]) = h[p-k]*0 = 0. After: h[p-k]*1 = h[p-k]. Change: +h[p-k]

So delta_p[m] (for h[p] going 1→0):
- For m = p+k-(n-1) where 0≤p+k<n: delta = h[p+k]-1 (which is -1 if h[p+k]=0, 0 if h[p+k]=1)
- For m = p-k-(n-1) where 0≤p-k<n: delta = h[p-k] (which is 1 if h[p-k]=1, 0 if h[p-k]=0)

Effect of h[q] going from 0 to 1:
- i=q: 0*(1-h[q+k]) = 0. After: 1*(1-h[q+k]) = 1-h[q+k]. Change: +(1-h[q+k])
- i+k=q, i=q-k: h[q-k]*(1-0) = h[q-k]. After: h[q-k]*(1-1) = 0. Change: -h[q-k]

So delta_q[m] (for h[q] going 0→1):
- For m = q+k-(n-1) where 0≤q+k<n: delta = 1-h[q+k]
- For m = q-k-(n-1) where 0≤q-k<n: delta = -h[q-k]

Total delta for swap (p→0, q→1):
delta[m] = delta_p[m] + delta_q[m]

Each delta is non-zero only at O(n) positions (specifically, at most 2*(n-1) positions for each).

To compute max(c + delta), we need to check all 2n-1 positions. But most of c + delta is unchanged. Only the O(n) positions affected by p and q need to be checked, plus we need to verify that the new max isn't at an unaffected position (which would be the same as before, so max(c+delta) ≤ max(max_affected, max_unaffected) = max(max_affected, current_max)).

Wait, that's the key insight! The new max is:
max(c + delta) = max(max over affected positions of (c[m] + delta[m]), max over unaffected positions of c[m])

The unaffected positions have value c[m], so their max is just the current max (or the second max if the current max position is affected).

So we can compute:
- current_max = c.max()
- current_max_pos = argmax(c)
- For the O(n) affected positions, compute c[m] + delta[m]
- new_max = max(max_affected, max_unaffected)
  where max_unaffected = current_max if current_max_pos is not affected, else second_max

This makes each swap evaluation O(n) instead of O(n²)!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _pt_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _pt_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def _slsqp_con(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_j(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 10.0), (5000, 100.0), (50000, 1000.0), (500000, 10000.0), (2000000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:

        try:
            res = minimize(_pt_obj, h, jac=_pt_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

                                   'gtol': 1e-13, 'maxls': 50})

            if c_new < best_c - 1e-15:

        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

                        else:
                            h[i] -= d; h[j] += d

                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved
