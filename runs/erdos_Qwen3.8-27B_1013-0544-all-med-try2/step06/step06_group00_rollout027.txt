def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    dx = 2.0 / n
    N = len(h)
    
    def obj_and_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        M = corr.max()
        g = alpha * (corr - M)
        g_max = g.max()
        w = np.exp(g - g_max)
        w_sum = w.sum()
        obj_val = M + np.log(w_sum) / alpha
        
        # Gradient
        w = w / w_sum
        rows = np.arange(n)[:, None]
        cols = np.arange(n)[None, :]
        idx_P = n - 1 + cols - rows
        idx_Q = n - 1 + rows - cols
        W_P = w[idx_P]
        W_Q = w[idx_Q]
        P = W_P @ (1.0 - x)
        Q = W_Q @ x
        grad = dx * (P - Q)
        
        return obj_val, grad
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    def cons_grad(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_grad}]
    
    res = minimize(obj_and_grad, h, method='SLSQP', bounds=bounds,
