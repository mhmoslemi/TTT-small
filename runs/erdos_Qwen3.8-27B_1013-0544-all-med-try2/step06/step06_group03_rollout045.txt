def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def gradient_c5(h, n, alpha=50):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    c_max = c.max()
    shifted = np.clip(alpha * (c - c_max), -40, 0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for m in range(len(c)):
        if w[m] < 1e-5: continue
        lag = m - (n - 1)
        wm = w[m] * dx
        if lag == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wm * (1.0 - h[lag:])
            grad[lag:] -= wm * h[:n - lag]
        else:
            lag2 = -lag
            grad[lag2:] += wm * (1.0 - h[:-lag2])
            grad[:-lag2] -= wm * h[lag2:]
    return grad

def grad_descent(h, n, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 30.0
    lr = 0.1

        gn = np.linalg.norm(grad)

            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if improved:
            stall = 0
            lr = min(lr * 1.01, 0.3)
        else:
            stall += 1
            lr *= 0.6
            if lr < 1e-9: break
            if stall > 20: break
        alpha = min(alpha * 1.03, 300.0)
    return best_h, best_c

            improved = False

                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):

                        else:
