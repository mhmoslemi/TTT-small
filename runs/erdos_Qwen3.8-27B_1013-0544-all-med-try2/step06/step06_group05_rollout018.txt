                    if h[i_dec] < 0: h[i_dec] = 0
                    if h[i_dec] > 1: h[i_dec] = 1
                    if h[i_inc] < 0: h[i_inc] = 0
                    if h[i_inc] > 1: h[i_inc] = 1
                    stall += 1
            else:
                stall += 1
    return best_h, best_c

def stochastic_pair_search(h, n, deadline, rng, max_steps=200000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    count = 0
    while time.time() < deadline and count < max_steps:
        count += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(0.0001, 0.005) * (1 if rng.rand() < 0.5 else -1)
        hi = h[i] + delta
        hj = h[j] - delta
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            c = compute_c5(h, n)
            if c < best_c - 1e-16:
                best_c = c
                best_h = h.copy()
            else:
                h[i], h[j] = hi - delta, hj + delta
    return best_h, best_c

def optimize_single(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alphas = [2000, 20000, 200000, 1000000, 5000000, 20000000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=3000, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()
    if time.time() < deadline - 10:
        best_h, best_c = guided_pair_search(best_h, n, deadline - 10, rng)
    if time.time() < deadline - 3:
        ht = slsqp_optimize(best_h, n, 50000000, maxiter=2000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    if time.time() < deadline - 1:
        best_h, best_c = stochastic_pair_search(best_h, n, deadline - 1, rng, max_steps=100000)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    best_h = None
    best_c = float('inf')
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250
    if h_init is not None:
        alloc1 = min(400, (deadline - t0) * 0.4)
        h_opt, c_opt = optimize_single(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt
        x_init = np.linspace(0, 2, n_init)
        for n_try in [200, 300]:
            if n_try == n_init or time.time() > deadline - 80:
                continue
            alloc = min(70, (deadline - time.time() - 60) * 0.3)
            if alloc < 25:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_single(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
        if best_h is not None and time.time() < deadline - 50:
            n = len(best_h)
            n_restarts = min(5, max(2, int((deadline - time.time() - 30) / 50)))
            for r in range(n_restarts):
                if time.time() > deadline - 30:
                    break
                scale = 0.01 + 0.015 * r
                x = np.linspace(0, 2, n)
                freq = rng.randint(2, 12)
                phase = rng.uniform(0, 2 * np.pi)
                pert = best_h + scale * np.sin(2 * np.pi * freq * x / 2.0 + phase)
                pert = pert + rng.normal(0, scale * 0.4, n)
                pert = project_feasible(pert, n)
                alloc = min(50, (deadline - 30 - time.time()) * 0.3)
                if alloc < 15:
                    break
                h_opt, c_opt = optimize_single(pert, n, time.time() + alloc, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    if time.time() < deadline - 2:
        n = len(best_h)
        best_h, best_c = guided_pair_search(best_h, n, deadline - 1, rng)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c, len(best_h)
