    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx  # current correlation
    best_c = c.max()
    
    for delta in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += delta, h[j] -= delta
                    if h[i] + delta <= 1.0 and h[j] - delta >= 0.0:
                        # Compute delta_c
                        d = delta
                        hc = h.copy()  # We won't actually modify h yet
                        # For the delta computation, we use current h
                        # delta_c[m] for k = m - (n-1):
                        # = d * [(1-h[i+k])*[0<=i+k<n] - h[i-k]*[0<=i-k<n] 
                        #          - (1-h[j+k])*[0<=j+k<n] + h[j-k]*[0<=j-k<n]]

                        # h[i-k] valid when 0 <= i-k < n, i.e., i-n+1 <= k <= i
                        # For h[j] -= d:
                        # (1-h[j+k]) valid when 0 <= j+k < n
                        # h[j-k] valid when 0 <= j-k < n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return float(val)

def _con(x, n):
    return float(x.sum() - n / 2.0)

def _jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con, 'jac': _jac, 'args': (n,)}]
    try:
        t_remain = deadline - time.time()
        mi = min(maxiter, max(100, int(t_remain * 5)))
        res = minimize(_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': mi, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_opt(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 5.0), (2000, 20.0), (10000, 100.0),
               (50000, 500.0), (200000, 2000.0), (1000000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        def _pobj(x):
            return _obj(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
        def _pgrad(x):
            return grad_smooth_max(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0)
        mi = min(max_iters, max(50, int((deadline - time.time() - 3) * 2)))
        try:
            res = minimize(_pobj, h, jac=_pgrad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': mi, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': mi * 5, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fast_pair_search(h, n, deadline, deltas, radius=None):
    dx = 2.0 / n
    c = compute_c5_vec(h.copy(), n)
    best_c = c.max()
    kc = np.arange(-(n - 1), n)
    for delta in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    for d in (delta, -delta):
                        hi_new = h[i] + d
                        hj_new = h[j] - d
                        if hi_new < 0.0 or hi_new > 1.0 or hj_new < 0.0 or hj_new > 1.0:
                            continue
                        # Compute delta_c
                        delta_c = np.zeros(2 * n - 1)
                        mk1 = (kc + i >= 0) & (kc + i < n)
                        mk2 = (i - kc >= 0) & (i - kc < n)
                        delta_c[mk1] += d * (1.0 - h[i + kc[mk1]])
                        delta_c[mk2] -= d * h[i - kc[mk2]]
                        mk3 = (kc + j >= 0) & (kc + j < n)
                        mk4 = (j - kc >= 0) & (j - kc < n)
                        delta_c[mk3] -= d * (1.0 - h[j + kc[mk3]])
                        delta_c[mk4] += d * h[j - kc[mk4]]
                        new_max = (c + delta_c * dx).max()
                        if new_max < best_c - 1e-14:
                            h[i] += d
                            h[j] -= d
                            c += delta_c * dx
                            best_c = new_max
                            improved = True
            if not improved:
                break
    return h, best_c

def binary_projection_search(h, n, deadline):
    best_c = compute_c5(h, n)
    for threshold in [0.3, 0.5, 0.7]:
        if time.time() > deadline:
            break
        hb = (h >= threshold).astype(float)
        # Adjust to meet sum constraint
        n_ones = int(n / 2)
        if hb.sum() != n_ones:
            # Sort by h values and pick top n_ones
            idx = np.argsort(-h)
            hb = np.zeros(n)
            hb[idx[:n_ones]] = 1.0
        hb = project_feasible(hb, n)
        c_b = compute_c5(hb, n)
        if c_b < best_c:
            best_c = c_b
    return h, best_c
