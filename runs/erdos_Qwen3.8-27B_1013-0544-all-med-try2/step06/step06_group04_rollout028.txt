import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n, rng=None):
    target = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def binary_move_search(h_bin, n, deadline, max_iter=20000):
    h = h_bin.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    O = compute_all_lags(h, n)
    n_lags = 2 * n - 1
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(O))
        best_delta_c = best_c + 1e-12
        best_j = None
        best_jp = None
        best_new_O = None
        for j in ones:
            rem = np.zeros(n_lags)
            for k_idx in range(n_lags):
                lag = k_idx - (n - 1)
                if lag > 0:
                    d = 0.0
                    if j + lag < n:
                        d -= (1.0 - h[j + lag])
                    if 0 <= j - lag <= n - 1 - lag:
                        d += h[j - lag]
                    rem[k_idx] = d
                elif lag < 0:
                    m = -lag
                    d = 0.0
                    if 0 <= j - m <= n - 1 - m:
                        d -= (1.0 - h[j - m])
                    if j + m < n:
                        d += h[j + m]
                    rem[k_idx] = d
            add_all = np.zeros((len(zeros), n_lags))
            for zi, jp in enumerate(zeros):
                for k_idx in range(n_lags):
                    lag = k_idx - (n - 1)
                    d = 0.0
                    if lag > 0:
                        if jp + lag < n:
                            d += (1.0 - h[jp + lag])
                        if 0 <= jp - lag <= n - 1 - lag:
                            d -= h[jp - lag]
                    elif lag < 0:
                        m = -lag
                        if 0 <= jp - m <= n - 1 - m:
                            d += (1.0 - h[jp - m])
                        if jp + m < n:
                            d -= h[jp + m]
                    add_all[zi, k_idx] = d
            new_O = O[None, :] + rem[None, :] + add_all
            new_max = new_O.max(axis=1)
            bi = np.argmin(new_max)
            if new_max[bi] < best_delta_c * n:
                best_delta_c = new_max[bi]
                best_j = j
                best_jp = zeros[bi]
                best_new_O = new_O[bi]
        if best_j is None:
            stall += 1
            if stall > 200:
                if len(ones) > 2 and len(zeros) > 2:
                    j = ones[np.random.randint(len(ones))]
                    jp = zeros[np.random.randint(len(zeros))]
                    h[j] = 0
                    h[jp] = 1
                    O = compute_all_lags(h, n)
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    stall = 0
                break
            continue
        h[best_j] = 0
        h[best_jp] = 1
        O = compute_all_lags(h, n)
        c_new = float(O.max())
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
            stall = 0
        else:
            h[best_j] = 1
            h[best_jp] = 0
            O = compute_all_lags(h, n)
            stall += 1
            if stall > 300:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    def sum_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def active_set_slsqp(h, n, top_m, alpha, maxiter=3000):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    top_idx = np.argsort(c)[-top_m:]
    def obj(x):
        cc = np.correlate(x, 1 - x, mode='full') * dx
        a = cc[top_idx]
        m = a.max()
        shifted = alpha * (a - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    def sum_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Parameter(nonneg=True)
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = np.zeros(n)
        m = abs(lag)
        if m == 0:
            grad[:] = 1.0 - 2.0 * h_curr
        elif lag > 0:
            grad[:n - m] -= h_curr[m:]
            grad[m:] += (1.0 - h_curr[:n - m])
        else:
            grad[:n - m] += (1.0 - h_curr[m:])
            grad[m:] -= h_curr[:n - m]
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy(), best_c if 'best_c' in dir() else compute_c5(h_curr, n)
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy(), compute_c5(h_curr, n)
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy(), compute_c5(h_curr, n)
    return project_feasible(h_new, n), compute_c5(project_feasible(h_new, n), n)

def fine_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            for i in range(0, n - m, 2):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-6:
                    delta = min(d, (h[i] - h[j]) * 0.4)
                    if delta < 1e-9:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-6:
                    delta = min(d, (h[j] - h[i]) * 0.4)
                    if delta < 1e-9:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:

                    else:

def optimize_full(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Binary local search
    if time.time() < deadline - 60:
        h_bin = to_binary(h, n, rng)
        bin_deadline = min(deadline - 50, time.time() + 120)
        h_b, c_b = binary_move_search(h_bin, n, bin_deadline)
        if c_b < best_c:
            best_c = c_b
            best_h = h_b.copy()
        # Also try fractional from binary
        h_frac = h_b.copy()
        h_frac[h_b > 0.5] = np.clip(h_b[h_b > 0.5] + 0.1, 0, 1)
        h_frac[h_b <= 0.5] = np.clip(h_b[h_b <= 0.5] - 0.1, 0, 1)
        h_frac = project_feasible(h_frac, n)
        if time.time() < deadline - 40:
            h_new = slsqp_optimize(h_frac, n, 50000, maxiter=2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    # Phase 2: SLSQP on best
    if time.time() < deadline - 30:
        for alpha in [20000, 100000, 500000]:
            if time.time() > deadline - 25:
                break
            h_new = slsqp_optimize(best_h, n, alpha, maxiter=2500)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new

    # Phase 3: Active set
    if time.time() < deadline - 25:
        for tm in [3, 5]:
            if time.time() > deadline - 20:
                break
            h_new = active_set_slsqp(best_h, n, tm, 100000, maxiter=2000)

    # Phase 4: Fine local search
    if time.time() < deadline - 5:

    best_h = project_feasible(best_h, n)
