import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    s = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(s))) / beta

def smooth_max_weights(x, beta):
    m = np.max(x)
    s = np.clip(beta * (x - m), -700.0, 700.0)
    e = np.exp(s)
    return e / e.sum()

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(h, dx, beta, pen_factor):
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = smooth_max_val(corr, beta) * dx + pen_factor * (h.sum() - target)**2
    w = smooth_max_weights(corr, beta)
    gfull = 2 * n - 1
    grad = np.zeros(n)
    for k in range(gfull):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            a = np.zeros(n)
            a[:n - lag] = 1.0 - h[lag:]
            b = np.zeros(n)
            b[lag:] = -h[:n - lag]
            grad += wk * (a + b)
        else:
            la = -lag
            a = np.zeros(n)
            a[la:] = 1.0 - h[:n - la]
            b = np.zeros(n)
            b[:n - la] = -h[la:]
            grad += wk * (a + b)
    grad *= dx
    grad += 2.0 * pen_factor * (h.sum() - target)
    return f, grad

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, left=0.0, right=0.0)

def subgrad_polish(h, n, max_iters, t_rem_func, lr_base=0.003):
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_local = compute_c5(h)
    for it in range(max_iters):
        if t_rem_func() < 20:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_worst = int(np.argmax(corr))
        lag = k_worst - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            a = np.zeros(n)
            a[:n - lag] = 1.0 - h[lag:]
            b = np.zeros(n)
            b[lag:] = -h[:n - lag]
            grad = a + b
        else:
            la = -lag
            a = np.zeros(n)
            a[la:] = 1.0 - h[:n - la]
            b = np.zeros(n)
            b[:n - la] = -h[la:]
            grad = a + b
        lr = lr_base / (1.0 + it * 0.001)
        h -= lr * grad
        h = project_feasible(h, target)
        if (it + 1) % 500 == 0:
            c5 = compute_c5(h)
            if c5 < best_local:
                best_local = c5
    c5 = compute_c5(h)

    return h, best_local

def make_block(n, frac=0.5):
    h = np.zeros(n)
    h[:int(n * frac)] = 1.0
    return h

def make_sine(n, freq=1.0, offset=0.5):
    x = np.linspace(0, 2, n, endpoint=False)
    return offset + (1 - offset) * np.cos(np.pi * freq * x)

    rng = np.random.RandomState(seed)
