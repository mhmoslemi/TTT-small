import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _active_smooth_obj(x, n, top_idx, alpha):
    cc = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _compute_lag_gradient(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h_curr
        return grad
    if lag > 0:
        grad[:n - m] -= h_curr[m:]
        grad[m:] += (1.0 - h_curr[:n - m])
    else:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] -= h_curr[:n - m]
    return grad

def fast_outer_approx(h_curr, n, deadline, top_m=12, max_iters=200):
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.08
    active_set = None
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-top_m:]
        # Add lags close to current max
        thresh = c.max() - 0.002
        near_max = np.where(c >= thresh)[0]
        active_set = np.unique(np.concatenate([top_idx, near_max]))
        active_set = active_set[:min(len(active_set), 25)]

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
                cp.abs(h_var - h_curr) <= trust]
        dx = 2.0 / n
        for k in active_set:
            lag = k - (n - 1)
            grad = _compute_lag_gradient(h_curr, n, lag)
            cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            try:
                prob.solve(solver=cp.GLOP)
            except Exception:
                return best_h, best_c
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            return best_h, best_c
        h_new = h_var.value
        if h_new is None:
            return best_h, best_c
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        h_curr = h_new
        trust = max(0.003, trust * 0.88)
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_search(h, n, deadline):
    """Projected gradient with adaptive step on smooth max."""
    h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 20000
    step = 0.005
    dx = 2.0 / n
    stalled = 0
    for it in range(20000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        w = alpha * (c - m)
        wmx = w.max()
        w = w - wmx
        p = np.exp(w)
        psum = p.sum()
        if psum < 1e-30:
            break
        p = p / psum
        # Only use top contributing lags for speed
        n_active = min(20, len(p))
        top_p_idx = np.argsort(p)[-n_active:]
        p_top = p[top_p_idx]
        p_top = p_top / p_top.sum()

        grad = np.zeros(n)
        for idx, k_idx in enumerate(top_p_idx):
            if p_top[idx] < 1e-6:
                continue
            lag = k_idx - (n - 1)
            grad += p_top[idx] * _compute_lag_gradient(h, n, lag)

        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            h = h_new
            stalled = 0
            step = min(step * 1.02, 0.02)
            alpha = min(alpha * 1.05, 5000000)
        else:
            stalled += 1
            step *= 0.8
            if stalled > 200 or step < 1e-7:
                break
    return h, best_c
