    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Core optimization on parent
    t1 = t0 + min(180, budget_s * 0.2)
    best_h, best_c = refine(best_h, n_init, t1)

    # Phase 2: Gradient descent from refined solution
    if time.time() < deadline - 200:
        t2 = min(time.time() + 60, deadline - 200)
        h2, c2 = gradient_descent(best_h.copy(), n_init, t2, lr_init=0.003, alpha_init=1000.0)
        if c2 < best_c:
            best_c = c2
            best_h = project_feasible(h2, n_init)

    # Phase 3: Multi-start with diverse perturbations
    if time.time() < deadline - 120:
        t3_end = deadline - 80
        candidates = [best_h.copy()]
        x = np.linspace(0, 2, n_init)
        # Sine perturbations at various frequencies
        for freq in [1, 2, 3, 4, 5, 7]:
            for amp in [0.03, 0.08]:
                p = best_h + amp * np.sin(2 * np.pi * freq * (x / 2.0))
                candidates.append(project_feasible(p, n_init))
        # Random perturbations
        for amp in [0.02, 0.05, 0.1]:
            p = best_h + amp * (rng.rand(n_init) * 2 - 1)
            candidates.append(project_feasible(p, n_init))
        # Block flips
        for blk in [10, 20, 30]:
            start = rng.randint(0, max(1, n_init - blk))
            p = best_h.copy()
            p[start:start+blk] = 1.0 - p[start:start+blk]
            candidates.append(project_feasible(p, n_init))
        # Binary-like
        p = np.where(rng.rand(n_init) < 0.5, 0.0, 1.0)
        candidates.append(project_feasible(p, n_init))

        # Evaluate each candidate with quick refinement
        remaining = t3_end - time.time()
        per_cand = max(15, remaining / len(candidates))
        for cand in candidates[1:]:
            if time.time() > t3_end - 10:
                break
            alloc = min(per_cand, t3_end - time.time() - 5)
            if alloc < 12:
                break
            h_t, c_t = refine(cand.copy(), n_init, time.time() + alloc)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t

    # Phase 4: Try different n values
    if time.time() < deadline - 60:
        nf = len(best_h)
        x_old = np.linspace(0, 2, nf)
        for n_try in [200, 250, 400, 500]:
            if time.time() > deadline - 40:
                break
            if n_try == nf:
                continue
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = max(25, (deadline - 40 - time.time()) / 3)
            h_t, c_t = refine(h_interp, n_try, time.time() + alloc)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                nf = n_try
                x_old = np.linspace(0, 2, nf)

    # Phase 5: Final intensive refinement
    if time.time() < deadline - 5:
        nf = len(best_h)
        best_h, best_c = lbfgs_optimize(best_h, nf, deadline - 4, max_iters=150)
        if time.time() < deadline - 3:
            best_h, best_c = coord_descent(best_h, nf, deadline - 2)
        if time.time() < deadline - 1:
            best_h, best_c = pair_search(best_h, nf, deadline - 0.5,
                                         [0.001, 0.0005, 0.0002, 0.0001, 0.00005], radius=40)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
