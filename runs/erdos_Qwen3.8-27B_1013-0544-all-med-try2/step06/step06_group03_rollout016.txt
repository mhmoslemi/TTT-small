def project_feasible(h, n):

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def _cons_fun(x, n):
    return x.sum() - n / 2.0

def _cons_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def active_set_cd(h, n, max_iter=500, tol=0.01, top_k=5):
    """
    Coordinate descent on the active-set quadratic.
    """
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        c_all = compute_c5_all(h, n)
        max_c = c_all.max()
        
        # Find active lags
        active = c_all >= (max_c - tol)
        active_idx = np.where(active)[0]
        if len(active_idx) > top_k:
            thresh = np.sort(c_all[active_idx])[-top_k]
            active_idx = active_idx[c_all[active_idx] >= thresh]
        
        if len(active_idx) == 0:
            break
        
        # Convert to lags
        lags = active_idx - (n - 1)
        
        # Coordinate descent: for each j, compute s_j and update
        changed = False
        for j in range(n):
            s = 0.0
            for k in lags:
                if 0 <= j + k < n:
                    s += h[j + k]
                if 0 <= j - k < n:
                    s += h[j - k]
            # We want to maximize sum h[i]*h[i+k], so:
            # If s > 0, push h[j] up; if s < 0, push h[j] down
            # Use a soft update
            delta = 0.1 * s
            h_new_j = np.clip(h[j] + delta, 0.0, 1.0)
            if abs(h_new_j - h[j]) > 1e-10:
                h[j] = h_new_j
                changed = True
        
        # Project
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
        
        if not changed:
            break
    
    return best_h, best_c

                            improved = True
                        else:

        active = c_all >= (max_c - tol)
        active_idx = np.where(active)[0]
        if len(active_idx) > top_k:
            thresh = np.sort(c_all[active_idx])[-top_k]
            active_idx = active_idx[c_all[active_idx] >= thresh]
        if len(active_idx) == 0:
            break
        lags = active_idx - (n - 1)
        # Compute gradient of sum of active c_k (we want to minimize max, 
        # so gradient of max tells us direction to reduce it)
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active_idx:
            k = idx - (n - 1)
            if k >= 0:
                j1 = np.arange(0, n - k)
                grad[j1] += (1.0 - h[j1 + k]) * dx
                j2 = np.arange(k, n)
                grad[j2] -= h[j2 - k] * dx
            else:
                kp = -k
                j1 = np.arange(kp, n)
                grad[j1] += (1.0 - h[j1 - kp]) * dx
                j2 = np.arange(0, n - kp)
                grad[j2] -= h[j2 + kp] * dx
        grad /= len(active_idx)
        momentum = beta * momentum + (1 - beta) * grad
        h = h - lr * momentum
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-11:
            best_c = c
            best_h = h.copy()
            stall = 0
            if np.random.random() < 0.05:
                lr *= 1.3
        else:
            stall += 1
            if stall > 20:
                lr *= 0.8
                stall = 0
            if lr < 0.001:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except:
        n = 250
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: SLSQP with increasing alpha
    for alpha in [1000, 5000, 30000, 100000, 500000]:
        if time.time() > deadline - 120:
            break
        h_new = slsqp(best_h, n, alpha, maxiter=1200)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-11:
            best_c = c
            best_h = h_new

    # Phase 2: Active-set coordinate descent
    if time.time() < deadline - 100:
        h2, c2 = active_set_cd(best_h, n, max_iter=400, tol=0.03, top_k=5, step=0.12)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: SLSQP from CD result
    if time.time() < deadline - 80:
        h_new = slsqp(best_h, n, 200000, maxiter=1500)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new

    # Phase 4: Subgradient with momentum
    if time.time() < deadline - 80:
        h2, c2 = subgrad_momentum(best_h, n, deadline - 80, top_k=3, tol=0.008)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 5: Active-set CD with tighter params
    if time.time() < deadline - 60:
        h2, c2 = active_set_cd(best_h, n, max_iter=300, tol=0.01, top_k=3, step=0.08)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 6: SLSQP polish
    if time.time() < deadline - 40:
        h_new = slsqp(best_h, n, 1000000, maxiter=2000)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new

    # Phase 7: Multi-restart via perturbation + CD + SLSQP
    if time.time() < deadline - 60:
        n_restarts = 3
        remaining = deadline - time.time() - 40
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.02 + 0.04 * r
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            h2, c2 = active_set_cd(hp, n, max_iter=200, tol=0.02, top_k=4, step=0.1)
            if c2 < best_c:
                h_new = slsqp(h2, n, 300000, maxiter=1000)
                c3 = compute_c5(h_new, n)
                if c3 < best_c:
                    best_c = c3
                    best_h = h_new
                else:
                    best_c = c2
                    best_h = h2

    # Phase 8: Final local search
    if time.time() < deadline - 5:
        radius = 50 if n <= 300 else 30
        best_h, best_c = local_search(best_h, n, deadline - 3, radius=radius)

    # Phase 9: Final SLSQP
    if time.time() < deadline - 2:
        h_new = slsqp(best_h, n, 2000000, maxiter=1500)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
