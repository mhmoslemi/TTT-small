import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15: break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n//2]] = 1.0
    return hb

def lse_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def lse_grad(x, n, alpha):
    dx = 2.0 / n

    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    ex = np.exp(sh - sm)

    # grad[i] = sum_k w[k] * [(1-x[i+m_k]) - x[i-m_k]]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def lse_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(g.max())
    sh = alpha * (g - m)
    sm = float(sh.max())
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def lse_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(g.max())
    sh = alpha * (g - m)
    sm = float(sh.max())
    ex = np.exp(sh - sm)
    w = ex / ex.sum()
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            term1[:n - lag] += wk * x[lag:]
            term2[lag:] += wk * x[:n - lag]
        else:
            neg = -lag
            term1[neg:] += wk * x[:n - neg]
            term2[:n - neg] += wk * x[neg:]
    return 1.0 - term1 - term2

def slsqp_opt(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return lse_obj(x, n, alpha)
    def g(x):
        return lse_grad(x, n, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)
