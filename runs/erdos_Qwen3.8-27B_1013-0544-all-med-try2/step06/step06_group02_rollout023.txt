            if tot < 1e-14:

            h += (diff / tot) * h

def corr_gradient_matrix(h, n):
    """Compute G[m, i] = d corr[m] / d h[i]. Shape: (2n-1, n)."""
    m_total = 2 * n - 1
    G = np.zeros((m_total, n))
    for m in range(m_total):
        lag = m - (n - 1)
        if lag >= 0:
            k = lag
            if k == 0:
                G[m, :] = 1.0 - 2.0 * h
            else:
                G[m, k:] += (1.0 - h[:n-k])
                G[m, :n-k] -= h[k:]
        else:
            k = -lag
            G[m, :n-k] += (1.0 - h[k:])
            G[m, k:] -= h[:n-k]
    return G

    """Solve the linearized LP to get a new h."""
    c_vec = compute_c5_vec(h, n)
    G = corr_gradient_matrix(h, n)
    
    # LP: min t
    # s.t. G @ h_new - t <= G @ h - c_vec  (for each row)
    #      sum(h_new) = n/2
    #      0 <= h_new[i] <= 1
    #      t >= -inf (or t >= 0 since correlations are non-negative)
    
    # Variables: [h_new(0), ..., h_new(n-1), t]
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0  # minimize t

    # A_ub has shape (2n-1, n+1)
    A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
    b_ub = G @ h - c_vec

    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

                  bounds=bounds, method='highs')

        t_new = res.x[n]
        return h_new, t_new

**Approach D: Coordinate descent with exact optimization**
- For each coordinate i, find the optimal value of h[i] given all other h[j] fixed

- This is a convex piecewise-linear function of h[i], so we can find the minimum in O(n) time
- Cycle through all coordinates

  - i = j for some j in [0, n-1-lag]: i in [0, n-1-lag], contributes h[i+lag]*(1-h[i])

If lag < 0 (k = -lag): corr[m] = sum_{j=0}^{n-1-k} h[j]*(1-h[j+k])
- h[i] appears if:

  - i = j+k: i in [k, n-1], contributes h[i-k]*(1-h[i])

This is quadratic in h[i]! The coefficient of h[i]² is -1, so it's concave in h[i]. The maximum of a concave function on [0,1] is at the endpoints or at the vertex.

Actually, let me reconsider. For lag ≠ 0:

- Wait: (1-h[i])*h[j] = h[j] - h[i]*h[j]. This is linear in h[i]!

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def single_corr_grad(h, n, m):
    """Compute gradient of corr[m] w.r.t. h. Shape: (n,)."""
    lag = m - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        k = lag
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
    else:
        k = -lag
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    return g

def subgradient_step(h, n, tol=1e-8):
    """Compute subgradient of the max correlation objective."""
    c = compute_c5_vec(h, n)
    max_val = c.max()
    active = np.where(np.abs(c - max_val) < tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    grad = np.zeros(n)
    for m in active:
        grad += single_corr_grad(h, n, m)
    grad /= len(active)
    return grad, max_val

def subgradient_method(h, n, deadline, max_iters=5000):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.05
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad, _ = subgradient_step(h, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        # Backtracking line search
        step_lr = lr
        improved = False
        for _ in range(30):
            h_try = project_feasible(h - step_lr * grad, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-16:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                improved = True
                break
            step_lr *= 0.5
        if not improved:
            lr *= 0.7
        if lr < 1e-14:
            break
    return best_h, best_c

def lp_iteration(h, n):
    """One step of successive linearization."""
    c_vec = compute_c5_vec(h, n)
    m_total = 2 * n - 1
    G = np.zeros((m_total, n))
    for m in range(m_total):
        G[m] = single_corr_grad(h, n, m)
    
    # LP: min t s.t. G @ h_new - t <= G @ h - c_vec, sum(h_new) = n/2, 0<=h_new<=1
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((m_total, 1))])
    b_ub = G @ h - c_vec
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n]
    return h

def successive_lp(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h_new = lp_iteration(h, n)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Try a damped step
            for alpha in [0.5, 0.3, 0.1]:
                h_damped = project_feasible((1-alpha)*h + alpha*h_new, n)
                c_damped = compute_c5(h_damped, n)
                if c_damped < best_c - 1e-15:
                    best_c = c_damped
                    best_h = h_damped.copy()
                    h = h_damped
                    break
            else:
                break
    return best_h, best_c
