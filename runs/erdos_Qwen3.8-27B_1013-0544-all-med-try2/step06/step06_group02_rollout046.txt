    return np.clip(h, 0.0, 1.0)

def _sm_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sm_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w[idx]
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad * dx

def lbfgs_optimize(h, n, bounds, alphas, deadline, max_iter_per=300):
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        remaining = deadline - time.time() - 30
        if remaining < 5:
            break
        mi = min(max_iter_per, max(10, int(remaining // 0.08)))
        try:
            res = minimize(_sm_obj, h, args=(n, alpha), jac=_sm_grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': mi, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def grad_guided_pairs(h, n, d, deadline, radius=80):
    c = compute_c5(h, n)
    g = _sm_grad(h, n, 50000)
    improved_any = False
    for _ in range(200):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, c, improved_any
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if i == j:
                    continue
                if h[i] + d <= 1.0 and h[j] - d >= 0.0 and g[j] > g[i]:
                    h[i] += d; h[j] -= d
                    c_new = compute_c5(h, n)
                    if c_new < c - 1e-15:
                        c = c_new
                        improved = True
                        improved_any = True
                        g = _sm_grad(h, n, 50000)
                    else:
                        h[i] -= d; h[j] += d
                elif h[i] - d >= 0.0 and h[j] + d <= 1.0 and g[i] > g[j]:
                    h[i] -= d; h[j] += d
                    c_new = compute_c5(h, n)
                    if c_new < c - 1e-15:
                        c = c_new
                        improved = True
                        improved_any = True
                        g = _sm_grad(h, n, 50000)
                    else:
                        h[i] += d; h[j] -= d
        if not improved:
            break
    return h, c, improved_any

def brute_pairs(h, n, d, deadline, radius=40):
    c = compute_c5(h, n)
    improved_any = False
    for _ in range(100):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, c, improved_any
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if i == j:
                    continue
                if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                    h[i] += d; h[j] -= d
                    c_new = compute_c5(h, n)
                    if c_new < c - 1e-15:
                        c = c_new; improved = True; improved_any = True
                    else:
                        h[i] -= d; h[j] += d
                if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                    h[i] -= d; h[j] += d
                    c_new = compute_c5(h, n)
                    if c_new < c - 1e-15:
                        c = c_new; improved = True; improved_any = True
                    else:
                        h[i] += d; h[j] -= d
        if not improved:
            break
    return h, c, improved_any

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)

    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B with sweeping alpha
    alphas_p1 = [50, 200, 1000, 5000, 20000, 100000, 500000]
    h, best_c = lbfgs_optimize(h, n, bounds, alphas_p1, deadline - 300, max_iter_per=400)
    best_h = h.copy()

    # Phase 2: Multi-restart with perturbation
    for r in range(5):
        if time.time() > deadline - 250:
            break
        noise_scale = 0.005 + 0.005 * r
        h_pert = h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, n)
        alphas_p2 = [5000, 50000, 200000]
        h_p, c_p = lbfgs_optimize(h_pert, n, bounds, alphas_p2, deadline - 250, max_iter_per=200)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p
            h = h_p

    # Phase 3: Fine-tune with L-BFGS-B at very high alpha
    if time.time() < deadline - 150:
        h, best_c = lbfgs_optimize(h, n, bounds, [200000, 500000, 1000000], deadline - 150, max_iter_per=500)
        best_h = h.copy()

    # Phase 4: Gradient-guided pair search (medium steps)
    if time.time() < deadline - 100:
        h = best_h.copy()
        c = best_c
        for d in [0.005, 0.002, 0.001, 0.0005]:
            h, c, _ = grad_guided_pairs(h, n, d, deadline - 100, radius=80)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 5: Brute force pair search (fine steps)
    if time.time() < deadline - 20:
        h = best_h.copy()
        c = best_c
        for d in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
            h, c, _ = brute_pairs(h, n, d, deadline - 20, radius=40)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 6: Try alternative n values
    for n_alt in [250, 350, 400]:
        if time.time() > deadline - 60:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_alt)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_alt)
        alloc = min(50, deadline - time.time() - 30)
        if alloc < 15:
            continue
        bounds_alt = [(0.0, 1.0)] * n_alt
        h_alt, c_alt = lbfgs_optimize(h_interp, n_alt, bounds_alt,
                                       [10000, 100000], time.time() + alloc * 0.7, max_iter_per=200)
        if time.time() < deadline - 40:
            h_alt, c_alt = brute_pairs(h_alt, n_alt, 0.001, time.time() + alloc * 0.25, radius=30)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
