def gradient_guided_pair_search(h, n, c_vec, deadline):
    dx = 2.0 / n
    best_c_raw = float(c_vec.max())
    best_c = best_c_raw * dx
    k_all = np.arange(-(n-1), n)
    
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for d in deltas:
        while time.time() < deadline:
            improved = False
            # Compute subgradient
            max_c = c_vec.max()
            active = c_vec >= max_c - 1e-10
            n_active = active.sum()
            
            # Compute subgradient for each coordinate
            grad = np.zeros(n)
            for m in range(2*n-1):
                if not active[m]:
                    continue
                k = k_all[m]
                # Upper: for i in [max(0,k), min(n, n+k)): grad[i] += (1-h[i-k])/n_active
                lo_u = max(0, k)
                hi_u = min(n, n+k)
                if lo_u < hi_u:
                    idx = np.arange(lo_u, hi_u)
                    grad[idx] += (1.0 - h[idx - k]) / n_active
                # Lower: for i in [max(0,-k), min(n, n-k)): grad[i] += -h[i+k]/n_active
                lo_l = max(0, -k)
                hi_l = min(n, n-k)
                if lo_l < hi_l:
                    idx = np.arange(lo_l, hi_l)
                    grad[idx] -= h[idx + k] / n_active
            
            # Find best pair: large grad[i] (decrease h[i]) and large -grad[j] (increase h[j])
            # Score for pair (i,j): grad[i] - grad[j] (we want to move from high-grad to low-grad)
            
            # For efficiency, try top candidates
            sorted_idx = np.argsort(-grad)  # highest grad first
            candidates = sorted_idx[:50]
            
            for i in range(len(candidates)):
                for j in range(len(candidates)):
                    if i == j:
                        continue
                    ii = candidates[i]
                    jj = candidates[j]
                    if h[ii] < d or h[jj] > 1.0 - d:
                        continue
                    # Move d from ii to jj
                    h[ii] -= d; h[jj] += d
                    c_new_raw = float(np.max(np.correlate(h, 1-h, mode='full')))
                    if c_new_raw < best_c_raw - 1e-12:

                        c_vec = np.correlate(h, 1-h, mode='full')
                        improved = True
                        break
                    else:
                        h[ii] += d; h[jj] -= d
                if improved:
                    break

                break
