def analytic_grad_g(h, n, worst_idx):

    k = worst_idx - (n - 1)
    grad = np.zeros(n)
    # Term 1: (1-h[m-k]) for valid m-k
    if k >= 0:

        grad[k:] += (1.0 - h[:n-k])

For binary h (values 0 or 1, with exactly n/2 ones):
- A "flip" changes one 1 to 0 (or vice versa), but this violates the sum constraint.
- A "swap" exchanges a 1 and a 0: h[i] was 1, h[j] was 0 → h[i] becomes 0, h[j] becomes 1.

The effect of a swap (i: 1→0, j: 0→1) on g_k:
  Δg_k = dx * [change in sum of h[n+k]*(1-h[n]) due to the swap]

The swap affects g_k for all k where i or j is involved. Specifically:
- For the term where n = i: h[i+k]*(1-h[i]) changes from h[i+k]*0 to h[i+k]*1 → Δ = h[i+k]
- For the term where n+k = i (i.e., n = i-k): h[i]*(1-h[i-k]) changes from 1*(1-h[i-k]) to 0 → Δ = -(1-h[i-k])
- For the term where n = j: h[j+k]*(1-h[j]) changes from h[j+k]*1 to h[j+k]*0 → Δ = -h[j+k]
- For the term where n+k = j (i.e., n = j-k): h[j]*(1-h[j-k]) changes from 0 to 1*(1-h[j-k]) → Δ = (1-h[j-k])

So Δg_k = dx * [h[i+k] - (1-h[i-k]) - h[j+k] + (1-h[j-k])]

         = dx * [(h[i+k] - h[j+k]) + (h[i-k] - h[j-k])]

(with appropriate boundary handling: h[m] = 0 if m < 0 or m >= n)

This is O(1) per k per swap! So evaluating the effect of a swap on all k is O(n).

But wait, the delta formula I derived was for the correlation at a specific lag k. Let me re-derive for the binary case.

For binary h, g_k = dx * sum_{valid n} h[n+k]*(1-h[n]).

After swap (i:1→0, j:0→1):
- h[i] changes from 1 to 0
- h[j] changes from 0 to 1

The change in g_k:
- Term where n+k = i (i.e., n = i-k): was h[i]*(1-h[i-k]) = 1*(1-h[i-k]), now 0*(1-h[i-k]) = 0. Δ = -(1-h[i-k])
- Term where n = i: was h[i+k]*(1-h[i]) = h[i+k]*0 = 0, now h[i+k]*(1-0) = h[i+k]. Δ = h[i+k]
- Term where n+k = j (i.e., n = j-k): was h[j]*(1-h[j-k]) = 0, now 1*(1-h[j-k]) = 1-h[j-k]. Δ = 1-h[j-k]
- Term where n = j: was h[j+k]*(1-h[j]) = h[j+k]*1 = h[j+k], now h[j+k]*0 = 0. Δ = -h[j+k]

So Δg_k = dx * [-(1-h[i-k]) + h[i+k] + (1-h[j-k]) - h[j+k]]
         = dx * [(h[i+k] - h[j+k]) - (1-h[i-k]) + (1-h[j-k])]
         = dx * [(h[i+k] - h[j+k]) + (h[i-k] - h[j-k])]

(where h[m] = 0 for m outside [0,n))

This is correct! And it's O(1) per lag.

def binary_anneal_fast(h_cont, n, deadline, rng, max_iter=200000):
    target = n // 2
    h = np.zeros(n)
    top = np.argsort(h_cont)[::-1][:target]
    h[top] = 1.0
    g = compute_all_g(h, n)
    best_h = h.copy()
    best_c = g.max()
    cur_max = best_c
    cur_idx = int(np.argmax(g))
    T = max((best_c - 0.25) * 0.3, 0.001)
    ones = set(top.tolist())
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Random swap
        i = list(ones)[rng.randint(len(ones))]
        j = rng.randint(n)  # random position
        if h[j] == 1:
            continue
        # Compute delta
        dx = 2.0 / n
        delta = np.zeros(2 * n - 1)
        # For each lag k (indexed as kidx = k + n - 1):
        # Δg_{kidx} = dx * [(h[i+k] - h[j+k]) + (h[i-k] - h[j-k])]
        # k = kidx - (n-1)
        # We need h at positions i+k, j+k, i-k, j-k
        # Valid when position is in [0, n)
        
        # Vectorize: for kidx in range(2n-1), k = kidx - (n-1)
        ks = np.arange(2 * n - 1) - (n - 1)
        # h[i+k]: valid when 0 <= i+k < n, i.e., -i <= k < n-i
        # h[j+k]: valid when -j <= k < n-j
        # h[i-k]: valid when 0 <= i-k < n, i.e., i-n < k <= i
        # h[j-k]: valid when j-n < k <= j
        
        def safe_h(idx):
            # idx is an array, return h[idx] with 0 for out of bounds
            out = np.zeros_like(idx, dtype=float)
            mask = (idx >= 0) & (idx < n)
            out[mask] = h[idx[mask]]
            return out
        
        dh = safe_h(i + ks) - safe_h(j + ks) + safe_h(i - ks) - safe_h(j - ks)
        delta = dx * dh
        new_g = g + delta
        new_max = new_g.max()
        
        if new_max < cur_max or rng.random() < np.exp((cur_max - new_max) / T):
            # Accept
            h[i] = 0.0
            h[j] = 1.0
            g = new_g
            cur_max = new_max
            ones.remove(i)
            ones.add(j)
            if new_max < best_c:
                best_c = new_max
                best_h = h.copy()
        T *= 0.99995
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_of_g(h, n, idx):
    dx = 2.0 / n
    k = idx - (n - 1)
    grad = np.zeros(n)
    m1s = max(0, k)
    m1e = min(n, n + k)
    if m1e > m1s:
        grad[m1s:m1e] += (1.0 - h[m1s - k:m1e - k])
    m2s = max(0, -k)
    m2e = min(n, n - k)
    if m2e > m2s:
        grad[m2s:m2e] -= h[m2s + k:m2e + k]
    return grad * dx

def subgrad_descent(h, n, deadline, max_iter=30000):

def slsqp_solve(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        idx = int(np.argmax(g))
        return grad_of_g(x, n, idx)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greedy_binary(n, deadline):
    h = np.zeros(n)
    dx = 2.0 / n
    g = np.zeros(2 * n - 1)
    target = n // 2
    ks = np.arange(2 * n - 1) - (n - 1)
    for step in range(target):
        if time.time() > deadline:
            break
        best_pos = 0
        best_val = np.inf
        best_delta = None
        for pos in range(n):
            pk = pos - ks
            pk2 = pos + ks
            mask1 = (pk >= 0) & (pk < n)
            mask2 = (pk2 >= 0) & (pk2 < n)
            delta = np.zeros(2 * n - 1)
            delta[mask1] += 1.0 - h[pk[mask1]]
            delta[mask2] -= h[pk2[mask2]]
            delta *= dx
            mv = (g + delta).max()

                best_val = mv
                best_pos = pos
                best_delta = delta
        h[best_pos] = 1.0
        g += best_delta
    return h, float(g.max())
