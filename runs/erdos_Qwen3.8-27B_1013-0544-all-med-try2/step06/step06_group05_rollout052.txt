    dx = 2.0 / n
    # Compute all c_k values
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Softmax weights
    shifted = alpha * (c - c.max())
    p = np.exp(shifted)
    p /= p.sum()
    # Gradient: for each j, sum over k of p[k] * dc_k/dh_j
    # c_k = sum_i h[i]*(1-h[i+k])*dx for valid i
    # dc_k/dh[j] = (1-h[j+k])*dx - h[j-k]*dx (with bounds)
    
    # k ranges from -(n-1) to (n-1), indexed as 0 to 2n-2 in the correlate output
    # k = index - (n-1)
    grad = np.zeros(n)
    for idx in range(2*n - 1):
        k = idx - (n - 1)  # shift in grid units
        if p[idx] < 1e-15:
            continue
        # For shift k: c_k = sum_i h[i]*(1-h[i+k])*dx
        # valid i: 0 <= i < n and 0 <= i+k < n
        # So i ranges from max(0, -k) to min(n, n-k) - 1
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        if i_lo >= i_hi:

        # = (1-h[j+k])*dx if 0 <= j < n and 0 <= j+k < n (i.e., j in [max(0,-k), min(n, n-k)))
        # - h[j-k]*dx if 0 <= j < n and 0 <= j-k < n (i.e., j in [max(0,k), min(n, n+k)))

        # For j in [max(0,k), min(n, n+k)): contributes -h[j-k]*dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    return float(m + np.log(np.exp(shifted).sum()) / alpha)

def _smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    p = np.exp(shifted)
    p /= p.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if p[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wt = p[idx] * dx
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            grad[lo1:hi1] += wt * (1.0 - x[lo1 + k:hi1 + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wt * x[lo2 - k:hi2 - k]
    return grad

def _sum_con(x, n):
    return float(x.sum() - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search(h, n, deadline, step_sizes=None, offsets=None):
    if step_sizes is None:
        step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    if offsets is None:
        offsets = [1, -1, 2, -2, n // 4, -n // 4, n // 2, -n // 2]
    h = h.copy()
    best_c = compute_c5(h, n)
    for s in step_sizes:
        if time.time() > deadline:
            break
        for _pass in range(30):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for off in offsets:
                    j = (i + off) % n
                    if j == i:
                        continue
                    for sign in [1, -1]:
                        d = sign * s
                        hi = h[i] + d
                        hj = h[j] - d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                                break
                            else:
                                h[i], h[j] = hi - d, hj + d
                if improved:
                    break
            if not improved:
                break
    return h, best_c

def binary_push(h, n, deadline):
    """Push values toward 0 or 1 while maintaining feasibility and not increasing C5."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for _it in range(200):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):

            if 0.05 < h[i] < 0.95:
                target = 0.0 if h[i] < 0.5 else 1.0
                shift = target - h[i]
                # Distribute -shift over other coordinates proportionally
                others = np.ones(n)
                others[i] = 0
                # Only shift coordinates that have room
                if shift > 0:
                    room = 1.0 - h
                    room[i] = 0
                    tot = room.sum()
                    if tot < 1e-10:

                    delta = -shift * room / tot
                else:
                    tot = h.sum() - h[i]
                    if tot < 1e-10:
                        continue
                    delta = -shift * h / tot
                    delta[i] = 0
                h_new = h + delta
                h_new[i] = target
                h_new = np.clip(h_new, 0, 1)
                h_new = project_feasible(h_new, n)
                c = compute_c5(h_new, n)
                if c < best_c + 1e-12:
                    if c < best_c - 1e-15:
                        improved = True
                    h = h_new
                    best_c = min(best_c, c)
    return h, best_c
