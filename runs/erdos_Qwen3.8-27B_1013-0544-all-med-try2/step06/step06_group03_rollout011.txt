import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_all_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return float(val)

def obj_topk(x, n, k=5):
    """Penalize top-k correlations"""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-k:]
    return float(np.sum(c[idx]) / k)

def slsqp_smooth(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, radius=None):

                            h[j] += d

                        break
                    amt = d * 0.5
                    if h[i] - d >= 0 and h[j] + amt <= 1 and h[k] + amt <= 1:
                        h[i] -= d
                        h[j] += amt
                        h[k] += amt
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= amt
                            h[k] -= amt
                    elif h[j] - amt >= 0 and h[k] - amt >= 0 and h[i] + d <= 1:
                        h[j] -= amt
                        h[k] -= amt
                        h[i] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[j] += amt
                            h[k] += amt
                            h[i] -= d
        if not improved:
            break
    return h, best_c

def optimize_round(h, n, deadline, rng):
    """One round of SLSQP + local search"""
    best_c = compute_c5(h, n)
    alphas = [500, 2000, 10000, 50000, 200000, 500000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        alloc = min(40, (deadline - time.time()) * 0.2)
        if alloc < 5:
            break
        ht = slsqp_smooth(h, n, a, maxiter=min(1500, int(alloc * 40)))
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    # Local search
    if time.time() < deadline - 5:
        r = n // 3 if n > 100 else None
        h, best_c = pair_search(h, n, min(deadline - 2, time.time() + 30),
                                [0.01, 0.005, 0.002, 0.001, 0.0005], radius=r)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(80, 0.5)
        n_init = 80

    best_h = h_init.copy()
    best_c = compute_c5(h_init, n_init)
    best_n = n_init
    history = [h_init.copy()]

    # Phase 1: Optimize initial at its resolution
    alloc1 = min(180, (deadline - t0) * 0.18)
    h, c = optimize_round(h_init, n_init, t0 + alloc1, rng)
    if c < best_c:
        best_c, best_h, best_n = c, h, n_init
        history.append(h.copy())

    # Phase 2: Resolution sweep - try several resolutions
    resolutions = sorted(set([n_init, n_init - 50, n_init + 50, 150, 200]))
    resolutions = [r for r in resolutions if 50 <= r <= 500 and r != n_init]
    for n_try in sorted(resolutions, key=lambda x: abs(x - n_init)):
        if time.time() > deadline - 120:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        c0 = compute_c5(h_interp, n_try)

        alloc = min(80, (deadline - time.time()) * 0.1)
        if alloc < 20:
            break
        h, c = optimize_round(h_interp, n_try, time.time() + alloc, rng)
        if c < best_c:
            best_c, best_h, best_n = c, h, n_try
            history.append(h.copy())

    # Phase 3: Triple moves on best solution
    if time.time() < deadline - 60 and best_n <= 250:
        alloc = min(40, (deadline - time.time()) * 0.15)
        h, c = triple_search(best_h, best_n, time.time() + alloc, d=0.02)
        if c < best_c:
            best_c, best_h = c, h
            history.append(h.copy())

    # Phase 4: Multi-restart with varied perturbations
    if time.time() < deadline - 60:
        n_b = best_n
        remaining = deadline - time.time() - 30
        n_restarts = max(4, int(remaining / 35))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            # Vary perturbation: mix of Gaussian and uniform
            scale = 0.015 + 0.04 * (r / max(1, n_restarts - 1))
            if r % 3 == 0:
                hp = best_h + rng.normal(0, scale, n_b)
            elif r % 3 == 1:
                hp = best_h + rng.uniform(-scale, scale, n_b)
            else:
                # Sparse perturbation
                hp = best_h.copy()
                n_pert = max(5, n_b // 5)
                idx = rng.choice(n_b, n_pert, replace=False)
                hp[idx] += rng.normal(0, scale * 2, n_pert)
            hp = project_feasible(hp, n_b)

            alloc = min(35, (deadline - time.time() - 20) * 0.3)
            if alloc < 10:
                break
            h, c = optimize_round(hp, n_b, time.time() + alloc, rng)
            if c < best_c:
                best_c, best_h = c, h
                history.append(h.copy())

    # Phase 5: Blend with history (interpolation between good solutions)
    if len(history) >= 2 and time.time() < deadline - 40:
        for i in range(min(3, len(history) - 1)):
            if time.time() > deadline - 20:
                break
            h_blend = 0.5 * history[i] + 0.5 * best_h
            h_blend = project_feasible(h_blend, best_n)
            cb = compute_c5(h_blend, best_n)
            if cb < best_c:
                # Refine the blend
                alloc = min(20, deadline - time.time() - 10)
                if alloc > 5:
                    h, c = optimize_round(h_blend, best_n, time.time() + alloc, rng)
                    if c < best_c:
                        best_c, best_h = c, h

    # Phase 6: Final aggressive polish
    if time.time() < deadline - 8:
        n_b = best_n
        # High alpha SLSQP
        ht = slsqp_smooth(best_h, n_b, 1000000, maxiter=3000)
        c = compute_c5(ht, n_b)
        if c < best_c:
            best_c, best_h = c, ht
        # Very fine pair search
        if time.time() < deadline - 3:
            r = n_b // 2 if n_b > 100 else None
            best_h, best_c = pair_search(best_h, n_b, deadline - 1,
                                         [0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                         radius=r)

    # Phase 7: If n is large, try downsampled refinement
    if best_n > 150 and time.time() < deadline - 10:
        n_small = 120
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_small)
        h_s = np.interp(x_new, x_old, best_h)
        h_s = project_feasible(h_s, n_small)
        alloc = min(30, deadline - time.time() - 5)
        if alloc > 10:
            h, c = optimize_round(h_s, n_small, time.time() + alloc, rng)
            if c < best_c:
                best_c, best_h, best_n = c, h, n_small

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
