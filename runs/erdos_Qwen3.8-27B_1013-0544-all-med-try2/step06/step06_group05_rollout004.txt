import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    lags = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = lags.max()
    shifted = alpha * (lags - m)
    mx = shifted.max()
    if mx > 500:
        s = float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    else:
        s = float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    return s

def _sc(x, n):
    return float(np.sum(x) - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    """Target the worst lag for focused moves."""
    best_c = compute_c5(h, n)
    for _pass in range(300):
        if time.time() > deadline:
            break
        lags = compute_lags(h, n)
        idx = int(np.argmax(lags))
        k = idx - (n - 1)
        ak = abs(k)
        if ak == 0 or ak >= n:
            break
        improved = False
        sign = 1 if k > 0 else -1
        for i in range(n - ak):
            if time.time() > deadline:
                break
            j = i + ak
            if sign > 0:
                # h[i+ak]*(1-h[i]) is the term; reduce by lowering h[i+ak] or raising h[i]
                pair_lo, pair_hi = i, j
            else:
                pair_lo, pair_hi = j, i
            val = h[pair_hi] * (1 - h[pair_lo])
            if val > 0.02:
                d = min(0.005, 0.3 * h[pair_hi], 0.3 * (1 - h[pair_lo]))

                        improved = True

                        h[pair_lo] -= d
        if not improved:
            break
