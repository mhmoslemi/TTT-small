import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    shifted = alpha * (c - cmax)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-30:
        return cmax, np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        k = ki - (n - 1)
        wk = w[ki]
        if wk < 1e-14:
            continue
        i_lo = max(0, -k)
        i_hi = min(n - 1, n - 1 - k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += wk * (1.0 - h[i_lo+k:i_hi+k+1])
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, n - 1 + k)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] -= wk * h[i_lo2-k:i_hi2-k+1]
    grad *= dx
    return cmax, grad

def grad_desc(h, n, alpha, lr, n_iters, deadline):

    best_h = h.copy()

            lr = min(lr * 1.1, 0.5)
        elif c_try > cmax + 1e-12:
            lr *= 0.8
    return best_h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    shifted = alpha * (c - cmax)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-30:
        return cmax, np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        k = ki - (n - 1)
        wk = w[ki]
        if wk < 1e-12:
            continue
        i_lo = max(0, -k)
        i_hi = min(n - 1, n - 1 - k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += wk * (1.0 - h[i_lo + k:i_hi + k + 1])
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, n - 1 + k)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2 + 1] -= wk * h[i_lo2 - k:i_hi2 - k + 1]
    grad *= dx
    return cmax, grad

def grad_desc(h, n, alpha, lr, n_iters, deadline):
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

            break
        cmax, grad = lse_grad(h, n, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        step = lr
        for _ in range(25):
            h_try = project_feasible(h - step * grad, n)
            c_try = compute_c5(h_try, n)
            if c_try <= cmax + 1e-14:
                break
            step *= 0.5
        h = h_try
        if c_try < best_c:
            best_c = c_try
            best_h = h.copy()
        if c_try < cmax - 1e-9:
            lr = min(lr * 1.08, 0.4)
        elif c_try > cmax + 1e-11:
            lr = max(lr * 0.75, 1e-5)
    return best_h, best_c

def swap_delta(h, n, i, j):
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def greedy_binary(h_init, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for ii in range(len(zeros_list)):
            if time.time() > deadline:
                break
            i = zeros_list[ii]
            best_j, best_nc, best_delta = -1, cur_c, None
            for jj in range(len(ones_list)):
                j = ones_list[jj]
                d = swap_delta(h, n, i, j)
                nc = (corr + d).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc, best_j, best_delta = nc, j, d
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr += best_delta
                cur_c = best_nc
                ones_list.remove(best_j)
                ones_list.append(i)
                zeros_list[ii] = best_j
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx

    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = 0.08
    cool = 0.99998
    scale = 8000.0
    stall = 0
    for it in range(20000000):
        if time.time() > deadline:
            break
        if T < 1e-9:
            break
        if stall > 200000:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i, j = int(zeros_arr[ai]), int(ones_arr[bi])
        if i == j:
            continue
        d = swap_delta(h, n, i, j)
        nc = (corr + d).max() * dx
        if nc <= cur_c or (T > 1e-9 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += d
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            stall = 0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            stall += 1
        T *= cool
    return best_h, best_c
