            if s < 1e-14:

            s = h.sum()
            if s < 1e-14:

            h = h + diff / s * h

def subgrad_analytical(h, n, k_star):
    """Analytical subgradient of g_{k_star} at h, scaled by dx."""
    dx = 2.0 / n
    shift = k_star - (n - 1)
    grad = np.zeros(n)
    # Term 1: (1-h[j-shift]) for j where 0<=j-shift<=n-1
    js = max(0, shift)
    je = min(n-1, n-1+shift)
    if js <= je:
        idx = np.arange(js, je+1)
        grad[idx] += (1 - h[idx - shift])
    # Term 2: -h[j+shift] for j where 0<=j+shift<=n-1
    js2 = max(0, -shift)
    je2 = min(n-1, n-1-shift)
    if js2 <= je2:
        idx2 = np.arange(js2, je2+1)
        grad[idx2] -= h[idx2 + shift]
    return grad * dx

def projected_subgrad(h, n, deadline, max_iter=50000):

    best_h = h.copy()
    best_c = compute_c5(h, n)

        m = g.max()

        grad = subgrad_analytical(h, n, k_star)

                delta = (ci_shift + ci_rshift) - (cj_shift + cj_rshift)
                if delta < best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
        if best_i == -1:
            break
        # Apply best swap
        h_bin[best_i] = 1.0
        h_bin[best_j] = 0.0
        c = compute_c5(h_bin, n)
        if c < best_c:
            best_c = c
        else:
            # Revert
            h_bin[best_i] = 0.0
            h_bin[best_j] = 1.0
            break
    
    return h_bin, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgrad_analytical(h, n, k_star):
    dx = 2.0 / n
    shift = k_star - (n - 1)
    grad = np.zeros(n)
    js = max(0, shift)
    je = min(n - 1, n - 1 + shift)
    if js <= je:
        idx = np.arange(js, je + 1)
        grad[idx] += (1 - h[idx - shift])
    js2 = max(0, -shift)
    je2 = min(n - 1, n - 1 - shift)
    if js2 <= je2:
        idx2 = np.arange(js2, je2 + 1)
        grad[idx2] -= h[idx2 + shift]
    return grad * dx

def projected_subgrad(h, n, deadline, max_iter=200000):

            break

        m = g.max()
        k_star = int(np.argmax(g))
        grad = subgrad_analytical(h, n, k_star)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:

        gnorm = grad / ng
        step = lr

def slsqp_smoothed(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        if time.time() > deadline:
            return h
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def eff_binary_search(h, n, deadline, max_sweeps=300):
