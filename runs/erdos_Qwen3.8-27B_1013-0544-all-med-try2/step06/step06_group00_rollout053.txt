    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        corr = np.correlate(h, 1 - h, mode='full')
        vals = corr * dx
        
        # Find critical lags
        top_idx = np.argsort(vals)[::-1][:K]
        top_vals = vals[top_idx]
        
        # Log-sum-exp weighting
        m = top_vals.max()
        w = np.exp(alpha * (top_vals - m))
        w = w / w.sum()
        
        # Gradient: d/dh[i] of sum_k w_k * corr[k] * dx
        # dcorr[k]/dh[i] = (1-h[i+k_offset]) - h[i-k_offset] (with boundary handling)
        grad = np.zeros(n)
        for ki, k in enumerate(top_idx):
            d = k - (n - 1)  # lag offset
            # corr[k] = sum_i h[i] * (1-h[i+d]) for valid i
            # dcorr[k]/dh[i] = (1-h[i+d]) for i where i+d is valid
            #                  - h[i-d] for i where i-d is valid

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greedy_binary(h_init, n, deadline, max_sweeps=300):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    m_range = np.arange(-(n - 1), n)
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try best-improvement swap
        best_nc = cur_c
        best_i, best_j = -1, -1
        best_delta = None
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta = np.zeros(2 * n - 1)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk2 = (imn >= 0) & (imn < n)
                delta[msk2] -= h[imn[msk2]]
                jm = j + m_range
                msk3 = (jm >= 0) & (jm < n)
                delta[msk3] -= 1.0 - h[jm[msk3]]
                jmn = j - m_range
                msk4 = (jmn >= 0) & (jmn < n)
                delta[msk4] += h[jmn[msk4]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_i, best_j = i, j
                    best_delta = delta
                    break
        if best_i >= 0:
            h[best_i], h[best_j] = 1.0, 0.0
            corr += best_delta
            cur_c = best_nc
            ones_list.remove(best_j)
            zeros_list.remove(best_i)
            ones_list.append(best_i)
            zeros_list.append(best_j)
            improved = True
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_fast(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greedy_binary(h_init, n, deadline, max_sweeps=300):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta = np.zeros(2 * n - 1)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk2 = (imn >= 0) & (imn < n)
                delta[msk2] -= h[imn[msk2]]
                jm = j + m_range
                msk3 = (jm >= 0) & (jm < n)
                delta[msk3] -= 1.0 - h[jm[msk3]]
                jmn = j - m_range
                msk4 = (jmn >= 0) & (jmn < n)

                    zeros_list.append(j)

            if improved:
                break
        if not improved:
            break

            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def grad_descent(h_init, n, alpha, lr, deadline, K=50, max_iters=500000, mu=0.85):
    dx = 2.0 / n
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        vals = corr * dx
        K_eff = min(K, 2 * n - 1)
        top_idx = np.argpartition(vals, -K_eff)[-K_eff:]
        top_vals = vals[top_idx]
        m = top_vals.max()
        log_w = alpha * (top_vals - m)
        log_w -= log_w.max()
        w = np.exp(log_w)
        w /= w.sum()
        grad = np.zeros(n)
        offsets = (top_idx - (n - 1)).astype(int)
        for ki in range(K_eff):
            d = offsets[ki]
            wk = w[ki]
            if d > 0:
                grad[d:] += wk * (1.0 - h[:n - d])
                grad[:n - d] -= wk * h[d:]
            elif d < 0:
                d2 = -d
                grad[:n - d2] += wk * (1.0 - h[d2:])
                grad[d2:] -= wk * h[:n - d2]
            else:
                grad += wk * (1.0 - 2.0 * h)
        grad *= dx
        v = mu * v - lr * grad
        h = project_fast(h + v, n)
        if (it + 1) % 100 == 0:
            c = compute_c5(h, n)
            if c < best_c - 1e-10:
                best_c = c
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 50:
                v *= 0.5
                stall = 0
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def slsqp_polish(h, n, alpha, maxiter=500, ftol=1e-15):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = alpha * (c - m)
        sm = s.max()
        return m + (sm + np.log(np.exp(s - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_fast(res.x, n)
    except Exception:
        return h
