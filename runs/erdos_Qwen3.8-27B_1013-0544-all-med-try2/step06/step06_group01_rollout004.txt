def project_to_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s > target:
        excess = s - target
        idx = np.argsort(h)[::-1]  # highest first
        for i in idx:
            if excess <= 0:
                break
            reduce = min(h[i], excess)
            h[i] -= reduce
            excess -= reduce
    else:
