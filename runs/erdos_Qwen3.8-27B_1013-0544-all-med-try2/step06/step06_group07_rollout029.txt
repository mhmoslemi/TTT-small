**New idea: Alternating optimization on active constraints**

At any point, only a few lags are "active" (near the max). We can:
1. Identify the top m active lags
2. Solve the problem: minimize max_{k in active} corr[k]*dx subject to the constraints
   This is equivalent to: minimize t subject to corr[k]*dx ≤ t for k in active
   Which is a QP (since corr[k] is quadratic in h)
3. Check if any inactive constraint is violated
4. Add violated constraints to the active set
5. Repeat

This is essentially a cutting-plane / active-set method for the nonsmooth problem.

The QP in step 2 is:
min t
s.t. dx * Σ h[i]*(1-h[i+k_j]) ≤ t for j = 1, ..., m
     Σ h[i] = n/2
     0 ≤ h[i] ≤ 1

This is a non-convex QP (because of the h[i]*h[i+k] terms). But we can try to solve it with scipy's trust-constr method or SLSQP.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_at(h, k_worst, n):
    grad = np.zeros(n)
    lag = k_worst - (n - 1)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

def active_subgradient(h, n, tol_frac=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    tol = mx * tol_frac
    active = np.where(corr >= mx - tol)[0]
    grad = np.zeros(n)
    for k in active:
        grad += subgradient_at(h, k, n)
    grad /= len(active)
    return grad, mx * dx

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

    return h.sum() - target_sum

    return np.ones(n)

    pen = pen_weight * (hc.sum() - target_sum) ** 2
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    return smooth_max(corr, beta) * dx + pen
