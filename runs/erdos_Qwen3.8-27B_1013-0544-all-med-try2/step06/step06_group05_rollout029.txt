```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def c5_value(h, n):
    return float(np.max(compute_corr(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def single_corr_grad(h, n, lag):
    g = np.zeros(n)
    if lag >= 0:
        v = n - lag
        if v > 0:
            g[:v] += (1.0 - h[lag:])
            g[lag:] -= h[:v]
    else:
        m = -lag
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    return g

def lse_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    cmax = c.max()
    sh = alpha * (c - cmax)
    if sh.max() > 400:
        return float(cmax)
    return float(cmax + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha)

def lse_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    cmax = c.max()
    sh = alpha * (c - cmax)
    if sh.max() > 400:
        k = int(np.argmax(c))
        return dx * single_corr_grad(x, n, k - (n - 1))
    probs = np.exp(sh - sh.max())
    probs /= probs.sum()
    g = np.zeros(n)
    lags = np.arange(len(c)) - (n - 1)
    for idx in range(len(c)):
        if probs[idx] < 1e-12:
            continue
        g += probs[idx] * dx * single_corr_grad(x, n, lags[idx])
