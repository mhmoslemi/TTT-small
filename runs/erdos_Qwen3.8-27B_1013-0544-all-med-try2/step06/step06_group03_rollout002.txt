```python
def pair_optimal_delta(c, h, i, j, n):
    """Find optimal delta for pair (i,j): h[i]+=delta, h[j]-=delta"""
    dx = 2.0 / n
    # g_k = change in corr[k] per unit delta
    # corr[k] = sum_m h[m] * (1-h[m+k-(n-1)])
    # Δcorr[k] from h[i]+=δ: δ*(1-h[i+k-(n-1)]) - δ*h[i-(k-(n-1))]
    # Δcorr[k] from h[j]-=δ: -δ*(1-h[j+k-(n-1)]) + δ*h[j-(k-(n-1))]
    # g_k = [(1-h[i+off]) - h[i-off]] - [(1-h[j+off]) - h[j-off]]
    #      where off = k-(n-1)
    
    # Precompute g for all k
    g = np.zeros(2*n-1)
    for k in range(2*n-1):
        off = k - (n-1)
        gi = 0.0
        j1 = i + off
        if 0 <= j1 < n:
            gi += (1.0 - h[j1])
        j2 = i - off
        if 0 <= j2 < n:
            gi += h[j2]
        gj = 0.0
        j3 = j + off
        if 0 <= j3 < n:
            gj += (1.0 - h[j3])
        j4 = j - off
        if 0 <= j4 < n:
            gj += h[j4]
        g[k] = dx * (gi - gj)
    
    # Now minimize max_k (c[k] + delta * g[k]) over delta in [delta_min, delta_max]
    delta_min = max(-h[i], 0.0)  # h[i] + delta >= 0
    delta_max = min(1.0 - h[i], h[j])  # h[i]+delta <= 1 and h[j]-delta >= 0
    
    if delta_min >= delta_max:
        return 0.0
    
    # Ternary search on f(delta) = max_k (c[k] + delta*g[k])
    a, b = delta_min, delta_max
    for _ in range(30):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(c + m1 * g)
        f2 = np.max(c + m2 * g)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0
```

```python
        p_lo2 = max(0, off)
        p_hi2 = min(n, n + off)
        if p_lo2 < p_hi2:
            K[p_lo2:p_hi2, k] += h[p_lo2 - off : p_hi2 - off]
    return K
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smax_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    if sh.max() > 500:
        sh -= sh.max()
        return float(m + (sh.max() + np.log(np.sum(np.exp(sh)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return x.sum() - n / 2.0

def _sum_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    res = minimize(smax_obj, h, args=(n, alpha), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_feasible(res.x, n)

def compute_K(h, n):
    K = np.zeros((n, 2 * n - 1))
    for k in range(2 * n - 1):
        off = k - (n - 1)
        p_lo = max(0, -off)
        p_hi = min(n, n - off)
        if p_lo < p_hi:
            K[p_lo:p_hi, k] += (1.0 - h[p_lo + off:p_hi + off])
        p_lo2 = max(0, off)
        p_hi2 = min(n, n + off)
        if p_lo2 < p_hi2:
            K[p_lo2:p_hi2, k] += h[p_lo2 - off:p_hi2 - off]
    return K

def pair_search_opt(h, n, deadline, radius=30):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    outer_improved = True
    while outer_improved and time.time() < deadline:
        outer_improved = False
        K = compute_K(h, n)
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if i == j:
                    continue
                g = dx * (K[i] - K[j])
                c = np.correlate(h, 1 - h, mode='full') * dx
                dmin = -h[i]
                dmax = min(1.0 - h[i], h[j])
                if dmin >= dmax - 1e-12:
                    continue
                a, b = dmin, dmax
                for _ in range(20):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    if np.max(c + m1 * g) < np.max(c + m2 * g):
                        b = m2
                    else:
                        a = m1
                delta = (a + b) / 2.0
                new_c = float(np.max(c + delta * g))
                if new_c < best_c - 1e-14:
                    h[i] += delta
                    h[j] -= delta
                    np.clip(h, 0.0, 1.0, out=h)
                    best_c = new_c
                    outer_improved = True
                    K = compute_K(h, n)
    return h, best_c

def pair_search_fixed(h, n, deadline, deltas, radius=30):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    for d in deltas:
        outer_improved = True
        while outer_improved and time.time() < deadline:
            outer_improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; outer_improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; outer_improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def optimize_single(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    alphas = [1000, 5000, 20000, 100000]
    for a in alphas:
        if time.time() > deadline - 20:
            break
        ht = slsqp_opt(h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    if time.time() < deadline - 10:
        h, best_c = pair_search_opt(h, n, deadline - 10, radius=35)
    if time.time() < deadline - 3:
        h, best_c = pair_search_fixed(h, n, deadline - 3,
                                      [0.005, 0.002, 0.001, 0.0005], radius=40)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c
```
