import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def analytical_grad_k(h, n, k):
    s = k - (n - 1)
    grad = np.zeros(n)
    n_lo = max(0, -s)
    n_hi = min(n - 1, n - 1 - s)
    for t in range(n_lo, n_hi + 1):
        grad[t + s] += (1.0 - h[t])
        grad[t] -= h[t + s]
    return grad

def subgrad_analytical(h, n, deadline, max_iter=100000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.005
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()
        k_star = int(np.argmax(g))
        grad = analytical_grad_k(h, n, k_star)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        step = lr / ng
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < m:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            lr *= 0.92
            if lr < 1e-13:
                break
    return best_h, best_c

def coord_descent_fast(h, n, deadline, max_sweeps=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    n_coarse = 40
    n_fine = 150
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        # Recompute c periodically or at start
        if sweep % 10 == 0:
            c = np.correlate(h, 1 - h, mode='full') * dx
        current_max = c.max()
        use_fine = sweep >= max_sweeps - 30
        n_eval = n_fine if use_fine else n_coarse
        
        order = range(n) if sweep % 2 == 0 else range(n - 1, -1, -1)
        for i in order:
            if time.time() > deadline:
                break
            x_old = h[i]
            s_lo = max(-(n - 1), i - (n - 1))
            s_hi = min(n - 1, i)
            slopes = np.zeros(2 * n - 1)
            if s_lo <= s_hi:
                sv = np.arange(s_lo, s_hi + 1)
                kv = sv + (n - 1)
                slopes[kv] += (1.0 - h[i - sv])
            s_lo2 = max(-(n - 1), -i)
            s_hi2 = min(n - 1, n - 1 - i)
            if s_lo2 <= s_hi2:
                sv2 = np.arange(s_lo2, s_hi2 + 1)
                kv2 = sv2 + (n - 1)
                slopes[kv2] -= h[i + sv2]
            
            best_x = x_old
            best_val = current_max
            if n_eval <= 60:
                cands = np.concatenate(([0.0, 1.0, x_old], np.linspace(0, 1, n_eval)))
            else:
                cands = np.concatenate(([0.0, 1.0, x_old], np.linspace(0, 1, n_eval)))
            for x in cands:
                if abs(x - x_old) < 1e-9:
                    continue
                new_max = (c + (x - x_old) * slopes).max()
                if new_max < best_val:
                    best_val = new_max
                    best_x = x
            # Top-2 equalization candidate
            top2 = np.argsort(c)[-2:][::-1]
            if len(top2) == 2 and abs(slopes[top2[0]] - slopes[top2[1]]) > 1e-12:
                x_eq = x_old + (c[top2[1]] - c[top2[0]]) / (slopes[top2[0]] - slopes[top2[1]])
                if 0 < x_eq < 1 and abs(x_eq - x_old) > 1e-9:
                    new_max = (c + (x_eq - x_old) * slopes).max()
                    if new_max < best_val:
                        best_val = new_max
                        best_x = x_eq
            
            if best_x != x_old:
                delta = best_x - x_old
                h[i] = best_x
                c += delta * slopes
        
        h = project_feasible(h, n)
        c = np.correlate(h, 1 - h, mode='full') * dx
        c_val = c.max()
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=200):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in sorted(zeros):
            if time.time() > deadline:
                break
            for j in list(ones):
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.add(i)
                    ones.discard(j)
                    zeros.add(j)
                    zeros.discard(i)
                    improved = True
                    break
                else:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

def golden_start(n):
    phi = (1 + np.sqrt(5)) / 2
    h = np.zeros(n)
    h[(np.arange(n) * (1.0 / phi)) % 1.0 < 0.5] = 1.0
    return project_feasible(h, n)
