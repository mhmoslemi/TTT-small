import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project onto {sum(h)=n/2, 0<=h<=1} via bisection."""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    # If sum is already correct, return
    if abs(h.sum() - target) < 1e-12:
        return h
    # Bisection on lambda: find lambda s.t. sum(clip(h-lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_gradient(h, n, alpha):
    """Compute gradient of (1/alpha)*log(sum exp(alpha*c_k)) * dx"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    # Softmax
    c_scaled = alpha * c
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    p = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        wk = p[k_idx] * dx
        if wk < 1e-14:
            continue
        # Term 1: (1-h[j-k]) for j in [max(0,k), min(n, k+n))
        j_lo1 = max(0, k)
        j_hi1 = min(n, k + n)
        if j_hi1 > j_lo1:
            grad[j_lo1:j_hi1] += wk * (1.0 - h[j_lo1 - k:j_hi1 - k])
        # Term 2: -h[j+k] for j in [max(0,-k), min(n, n-k))
        j_lo2 = max(0, -k)
        j_hi2 = min(n, n - k)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= wk * h[j_lo2 + k:j_hi2 + k]
    return grad

def gradient_descent(h, n, alpha, steps, step_size, deadline):
    """Gradient descent with projection."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for step in range(steps):
        if time.time() > deadline:
            break
        g = smooth_gradient(h, n, alpha)
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm  # normalized direction
        # Line search (simple: fixed step with occasional adjustment)
        lr = step_size
        for trial in range(20):
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c or c_new < compute_c5(h, n):
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
            lr *= 0.5
            if lr < 1e-10:
                break
    return best_h, best_c

from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-13:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    sc = alpha * c
    sc = sc - sc.max()
    e = np.exp(sc)
    p = e / e.sum()
    obj = c.max() / alpha * 0 + (sc.max() + np.log(e.sum())) / alpha
    pwx = p * dx
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        k = ki - (n - 1)
        wk = pwx[ki]
        if wk < 1e-14:
            continue
        a1, b1 = max(0, k), min(n, k + n)
        if b1 > a1:
            grad[a1:b1] += wk * (1.0 - h[a1 - k:b1 - k])
        a2, b2 = max(0, -k), min(n, n - k)
        if b2 > a2:
            grad[a2:b2] -= wk * h[a2 + k:b2 + k]
    return obj, grad

def custom_gd(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000, 500000]
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        n_steps = 3000 if alpha < 5000 else 1500
        base_lr = 0.5 / (1.0 + alpha * 0.005)
        for it in range(n_steps):
            if time.time() > deadline - 2:
                break
            _, g = softmax_obj_grad(h, n, alpha)
            gn = np.linalg.norm(g)
            if gn < 1e-13:
                break
            g = g / gn
            cur_c = compute_c5(h, n)
            lr = base_lr
            best_trial_c = cur_c
            best_trial_h = h.copy()
            t_accepted = False
            for _t in range(12):
                cand = project_feasible(h - lr * g, n)
                cc = compute_c5(cand, n)
                if cc < best_trial_c:
                    best_trial_c = cc
                    best_trial_h = cand.copy()
                if cc < cur_c:
                    t_accepted = True
                    break
                lr *= 0.5
            if not t_accepted and best_trial_c < cur_c - 1e-14:
                t_accepted = True
            if t_accepted:
                h = best_trial_h
                if best_trial_c < best_c:
                    best_c = best_trial_c
                    best_h = h.copy()
    return best_h, best_c

def slsqp_pass(h, n, alpha, deadline, maxiter=3000):
    def obj(x):
        _,_ = 0,0
        c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        sc = alpha * c
        sc -= sc.max()
        e = np.exp(sc)
        return float(c.max() * 0.01 + (sc.max() + np.log(e.sum())) / alpha)
    def obj_grad(x):
        _, g = softmax_obj_grad(x, n, alpha)
        return g
    def sfun(x):
        return float(x.sum() - n / 2.0)
    def sjac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sfun, 'jac': sjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, max_passes=200):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _p in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi, hj = h[i] + d, h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def binary_search(h, n, deadline, max_iter=50000):
    """Local search on the binary pattern (which positions are 1)."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    idx = np.argpartition(h, n // 2 - 1)[:n // 2]
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or (i in idx and j in idx) or (i not in idx and j not in idx):
            continue
        if i in idx:
            idx = np.setdiff1d(idx, [i])
            idx = np.union1d(idx, [j])
        else:
            idx = np.setdiff1d(idx, [j])
            idx = np.union1d(idx, [i])
        h_new = np.zeros(n)
        h_new[idx] = 1.0
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

        c1 = compute_c5(h1, n_init)
        if c1 < best_c:

        # Phase 2: Custom GD with increasing alpha (60s)
        alloc2 = min(60, (deadline - t0) * 0.1)
        h2, c2 = custom_gd(best_h, n_init, t0 + alloc2)
        if c2 < best_c:
            best_c, best_h = c2, h2

        # Phase 3: SLSQP with high alpha (30s)
        if time.time() < deadline - 40:
            h3 = slsqp_pass(best_h, n_init, 100000, time.time() + 30, maxiter=5000)
            c3 = compute_c5(h3, n_init)
            if c3 < best_c:
                best_c, best_h = c3, h3

        # Phase 4: Try different n values
        for n_try in [100, 150, 200]:
            if time.time() > deadline - 80:
                break
            h_interp = interpolate_h(best_h, n_init, n_try)
            alloc = min(50, (deadline - time.time()) * 0.12)
            if alloc < 15:
                break
            h_opt, c_opt = custom_gd(h_interp, n_try, time.time() + alloc,
                                     alphas=[500, 2000, 10000, 50000])
            if time.time() < deadline - 20:
                h_opt = slsqp_pass(h_opt, n_try, 20000, time.time() + 20, maxiter=3000)
                c_opt = compute_c5(h_opt, n_try)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
                n_init = n_try

        # Phase 5: Random restarts
        if time.time() < deadline - 50:
            n_cur = len(best_h)
            remaining = deadline - time.time() - 40
            n_restarts = max(2, int(remaining / 40))
            scales = [0.003, 0.01, 0.02, 0.04, 0.06]
            for r in range(n_restarts):
                if time.time() > deadline - 35:
                    break
                scale = scales[r % len(scales)]
                hp = project_feasible(best_h + rng.normal(0, scale, n_cur), n_cur)
                alloc = min(35, (deadline - time.time() - 35) * 0.4)
                if alloc < 10:
                    break
                h_opt, c_opt = custom_gd(hp, n_cur, time.time() + alloc,
                                         alphas=[1000, 5000, 20000, 100000])
                if c_opt < best_c:
                    best_c, best_h = c_opt, h_opt

    # Phase 6: Final SLSQP + pair search polish
    if best_h is not None and time.time() < deadline - 8:
        n_final = len(best_h)
        if time.time() < deadline - 10:
            hf = slsqp_pass(best_h, n_final, 500000, time.time() + 15, maxiter=5000)
            cf = compute_c5(hf, n_final)
            if cf < best_c:
                best_c, best_h = cf, hf

        if time.time() < deadline - 3:
            dts = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
            maxp = 300 if n_final <= 150 else 150
            best_h, best_c = pair_search(best_h, n_final, deadline - 1, dts, max_passes=maxp)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
