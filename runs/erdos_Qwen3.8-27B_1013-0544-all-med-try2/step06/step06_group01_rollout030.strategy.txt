def project_feasible(h, n):
    s = h.sum()
    target = n/2.0
    diff = target - s
    if abs(diff) < 1e-5: return h
    room = np.clip(1-h, 0, 1) # Room for improvement if deficit
    if diff > 0: 
        # Can add 1 where h=0
        if np.sum(room) < 1e-5: 
            # No room, accept as is
            pass
        else:
            h += diff/s * room
            s = np.round(h.sum(),15) - n/2.0
            h = np.clip(h,0,1)
    else:  # diff < 0
        # Can remove some h[i] where h[i]>0
        if s < 1e-5:
            h = np.clip(h - diff, 0, 1)
    return h
