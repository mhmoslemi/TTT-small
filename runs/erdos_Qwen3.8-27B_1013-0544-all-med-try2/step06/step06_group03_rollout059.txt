Alternatively, we can use the fact that at the optimum, the active constraints (those k where f_k = C) determine the solution. This is the basis of the previous SLSQP approach with log-sum-exp.

            best_c = c_t; best_h = h_t; best_n = n_try

    # Phase 3: Further refine best solution
    remaining = deadline - time.time() - 20
    if remaining > 30:
        h_opt, c_opt = exact_pair_search(best_h, best_n, time.time() + remaining * 0.5)
        if c_opt < best_c:
            best_c = c_opt; best_h = h_opt

    # Phase 4: Ultra-fine delta search
    if time.time() < deadline - 5:
        h_opt, c_opt = full_delta_search(best_h, best_n, deadline - 2,
                                         [0.0002, 0.0001, 0.00005, 0.00002])
        if c_opt < best_c:
            best_c = c_opt; best_h = h_opt

    # Phase 5: Final subgradient polish
    if time.time() < deadline - 3:
        h_opt, c_opt = subgrad_refine(best_h, best_n, deadline - 2, max_steps=500)
        if c_opt < best_c:
            best_c = c_opt; best_h = h_opt

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
