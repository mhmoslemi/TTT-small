        while time.time() < tl:
            imp = False
            for i in range(n):
                if time.time() > tl:
                    return h, bc
                lo = max(0, i - rad)
                hi = min(n, i + rad + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                        h[i] += d
                        h[j] -= d
                        cc = compute_c5(h, n)
                        if cc < bc - 1e-15:
                            bc = cc
                            imp = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if 0 <= h[i] - d <= 1 and 0 <= h[j] + d <= 1:
                        h[i] -= d
                        h[j] += d
                        cc = compute_c5(h, n)
                        if cc < bc - 1e-15:
                            bc = cc
                            imp = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not imp:
                break
    return h, bc

def multi_alpha_lbfgs(h, n, tl):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [5000, 20000, 100000, 500000]
    for a in alphas:
        if time.time() > tl - 20:
            break
        t_alloc = min(60, tl - time.time() - 10)
        if t_alloc < 15:
            break
        hh, cc = lbfgs_opt(best_h, n, time.time() + t_alloc, alpha=a, mi=300)
        if cc < best_c:
            best_c = cc
            best_h = hh
    return best_h, best_c

def multi_n_try(h_src, n_src, c_src, tl):
    x_src = np.linspace(0, 2, n_src)
    best_h, best_c = h_src.copy(), c_src
    for n_t in [100, 150, 200, 250]:
        if time.time() > tl - 40:
            break
        ht = project_feasible(np.interp(np.linspace(0, 2, n_t), x_src, h_src), n_t)
        ct = compute_c5(ht, n_t)
        alloc = min(100, tl - time.time() - 30)
        if alloc < 25:
            break
        t1 = time.time() + alloc * 0.4
        ht, ct = lbfgs_opt(ht, n_t, t1, alpha=8000, mi=300)
        if time.time() < tl - 25:
            ht, ct = bin_push(ht, n_t, time.time() + 20, thr=0.12)
        if time.time() < tl - 22:
            ht, ct = pair_srch(ht, n_t, time.time() + 25,
                               [0.01, 0.005, 0.001, 0.0005, 0.0002], rad=35)
        if ct < best_c:
            best_h, best_c = ht, ct
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    try:
        hi = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        hi = np.full(200, 0.5)
    n0 = len(hi)
    h = project_feasible(hi, n0)
    bc = compute_c5(h, n0)
    bh = h.copy()
    bn = n0

    # Phase 1: L-BFGS multi-alpha (180s)
    t1 = min(t0 + 180, dl - 600)
    bh, bc = multi_alpha_lbfgs(bh, bn, t1)

    # Phase 2: Binary push with decreasing threshold (60s)
    if time.time() < dl - 500:
        for thr in [0.15, 0.10, 0.06, 0.03]:
            tt = min(time.time() + 15, dl - 500)
            if time.time() >= tt:
                break
            hh, cc = bin_push(bh, bn, tt, thr=thr)
            if cc < bc:
                bc, bh = cc, hh

    # Phase 3: Try other n values (250s)
    if time.time() < dl - 250:
        bh, bc = multi_n_try(bh, bn, bc, dl - 200)
        bn = len(bh)

    # Phase 4: Re-optimize at best n with high alpha (100s)
    if time.time() < dl - 180:
        t4 = min(time.time() + 100, dl - 180)
        hh, cc = multi_alpha_lbfgs(bh, bn, t4)
        if cc < bc:
            bc, bh = cc, hh

    # Phase 5: Binary push again (40s)
    if time.time() < dl - 140:
        for thr in [0.12, 0.06, 0.03]:
            tt = min(time.time() + 12, dl - 140)
            if time.time() >= tt:
                break
            hh, cc = bin_push(bh, bn, tt, thr=thr)
            if cc < bc:
                bc, bh = cc, hh

    # Phase 6: Heavy pair search (130s)
    if time.time() < dl - 50:
        t6 = min(time.time() + 130, dl - 50)
        bh, bc = pair_srch(bh, bn, t6,
                          [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005], rad=50)

    # Phase 7: Final binary push (20s)
    if time.time() < dl - 25:
        hh, cc = bin_push(bh, bn, min(time.time() + 20, dl - 25), thr=0.05)
        if cc < bc:
            bc, bh = cc, hh

    # Phase 8: Final fine pair search (remaining)
    if time.time() < dl - 3:
        bh, bc = pair_srch(bh, bn, dl - 2,
                          [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001], rad=40)

    bh = project_feasible(bh, len(bh))
    bc = compute_c5(bh, len(bh))
    return bh, bc, len(bh)
