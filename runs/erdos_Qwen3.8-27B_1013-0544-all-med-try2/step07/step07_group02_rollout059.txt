def project_feasible(h, n):

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(x, n, alpha):
    """Compute smooth max of corr(x, 1-x)*dx and its gradient."""
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w_sum = w.sum()
    w = w / w_sum
    obj_val = (sm + np.log(w_sum)) / alpha  # = log(sum exp(alpha*g))/alpha
    
    # Gradient: sum_k w[k] * dg_k/dx
    grad = np.zeros(n)
    for k in range(len(g)):
        wk = w[k]
        if wk < 1e-12:
            continue
        ks = k - (n - 1)
        if ks >= 0:
            if n - ks > 0:
                grad[:n-ks] += wk * dx * (1.0 - x[ks:])
                grad[ks:] -= wk * dx * x[:n-ks]
        else:
            ks2 = -ks
            if n - ks2 > 0:
                grad[ks2:] += wk * dx * (1.0 - x[:n-ks2])
                grad[:n-ks2] -= wk * dx * x[ks2:]
    return obj_val, grad

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def obj_grad(x):
        return smooth_max_obj_grad(x, n, alpha)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent_exact(h, n, max_sweeps, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            pos = B > 1e-14
            neg = B < -1e-14
            zero = ~(pos | neg)
            zero_max = corr[zero].max() if zero.any() else -np.inf
            # Bisection on optimal M
            val0 = np.max(np.where(pos, corr - B*h[i], np.where(neg, corr - B*h[i], corr)))
            val1 = np.max(np.where(pos, corr + B*(1-h[i]), np.where(neg, corr + B*(1-h[i]), corr)))
            M_lo = min(val0, val1) - 1e-10
            M_hi = max(val0, val1) + 1e-10
            for _ in range(45):
                M_mid = (M_lo + M_hi) / 2
                if zero_max > M_mid + 1e-14:
                    M_lo = M_mid
                    continue
                t_lo = 0.0
                if neg.any():
                    t_lo = max(t_lo, np.max((M_mid - corr[neg]) / B[neg])) + h[i]
                t_hi = 1.0
                if pos.any():
                    t_hi = min(t_hi, np.min((M_mid - corr[pos]) / B[pos])) + h[i]
                if t_lo <= t_hi + 1e-13:
                    M_hi = M_mid
                else:
                    M_lo = M_mid
            M_opt = M_hi
            t_lo = 0.0
            if neg.any():
                t_lo = max(0.0, np.max((M_opt - corr[neg]) / B[neg])) + h[i]
            t_hi = 1.0
            if pos.any():
                t_hi = min(1.0, np.min((M_opt - corr[pos]) / B[pos])) + h[i]
            t_opt = np.clip((t_lo + t_hi) / 2.0, 0.0, 1.0)
            if abs(t_opt - h[i]) > 1e-15:
                corr += B * (t_opt - h[i])
                h[i] = t_opt
    return h, corr

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w_sum = w.sum()
    w = w / w_sum
    obj_val = (sm + np.log(w_sum)) / alpha
    grad = np.zeros(n)
    for k in range(len(g)):
        wk = w[k]
        if wk < 1e-12:
            continue
        ks = k - (n - 1)
        if ks >= 0:
            if n - ks > 0:
                grad[:n-ks] += wk * dx * (1.0 - x[ks:])
                grad[ks:] -= wk * dx * x[:n-ks]
        else:
            ks2 = -ks
            if n - ks2 > 0:
                grad[ks2:] += wk * dx * (1.0 - x[:n-ks2])
                grad[:n-ks2] -= wk * dx * x[ks2:]
    return obj_val, grad

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def obj_grad(x):
        return smooth_max_grad(x, n, alpha)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, n, max_sweeps, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            m1 = (k1 >= 0) & (k1 < n)
            B[m1] += (1.0 - h[k1[m1]])
            k2 = i + m_range
            m2 = (k2 >= 0) & (k2 < n)
            B[m2] -= h[k2[m2]]
            pos = B > 1e-14
            neg = B < -1e-14
            zero = ~(pos | neg)
            zmax = corr[zero].max() if zero.any() else -np.inf
            v0 = np.max(np.where(pos, corr - B*h[i], np.where(neg, corr - B*h[i], corr)))
            v1 = np.max(np.where(pos, corr + B*(1-h[i]), np.where(neg, corr + B*(1-h[i]), corr)))
            M_lo = min(v0, v1) - 1e-10
            M_hi = max(v0, v1) + 1e-10
            for _ in range(50):
                Mm = (M_lo + M_hi) / 2.0
                if zmax > Mm + 1e-14:
                    M_lo = Mm
                    continue
                t_lo = 0.0
                if neg.any():
                    t_lo = max(t_lo, np.max((Mm - corr[neg]) / B[neg])) + h[i]
                t_hi = 1.0
                if pos.any():
                    t_hi = min(t_hi, np.min((Mm - corr[pos]) / B[pos])) + h[i]
                if t_lo <= t_hi + 1e-13:
                    M_hi = Mm
                else:
                    M_lo = Mm
            Mo = M_hi
            t_lo = 0.0
            if neg.any():
                t_lo = max(0.0, np.max((Mo - corr[neg]) / B[neg])) + h[i]
            t_hi = 1.0
            if pos.any():
                t_hi = min(1.0, np.min((Mo - corr[pos]) / B[pos])) + h[i]
            t_opt = np.clip((t_lo + t_hi) / 2.0, 0.0, 1.0)
            if abs(t_opt - h[i]) > 1e-15:
                corr += B * (t_opt - h[i])
                h[i] = t_opt
    return h, corr

    return best_h, best_c, n
