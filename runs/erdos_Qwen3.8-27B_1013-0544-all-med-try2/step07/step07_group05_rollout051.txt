            if dc < 0 or (t > 1e-15 and rng.random() < np.exp(-dc / t)):
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i] -= sgn * d
                h[j] += sgn * d
        
        t *= cool
        step = max(0.02, step * 0.998)
    
    return best_h, best_c
