import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                h[:] = 0.5
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_active(c, active_idx, alpha):
    ac = c[active_idx]
    m = ac.max()
    shifted = alpha * (ac - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_smooth(h0, n, alpha, maxiter=5000, ftol=1e-16, active_idx=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    
    def obj(x):
        c = compute_all_lags(x, n)
        if active_idx is not None:
            return smooth_max_active(c, active_idx, alpha)
        return smooth_max(c, alpha)
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def thue_morse_binary(n):
    """Generate Thue-Morse based binary pattern with sum = n/2"""
    # Thue-Morse: t(n) = popcount(n) mod 2
    tm = np.array([bin(i).count('1') % 2 for i in range(n)])
    h = tm.astype(float)
    # Adjust to get sum = n/2
    h = project_feasible(h, n)
    return h

            h_start = np.full(n_t, 0.5)
        h_start = project_feasible(h_start, n_t)
        h_opt, c_opt = full_pipeline(h_start, n_t, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if best_h is not None and time.time() < deadline - 70:
        n = len(best_h)
        n_restarts = max(1, int((deadline - time.time() - 50) / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 45:
                break
            scale = 0.015 * (1.0 + 0.5 * r)
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(50, deadline - time.time() - 35)
            if alloc < 30:
                break
            h_opt, c_opt = full_pipeline(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 6:
        n = len(best_h)
        h_f, c_f = full_pipeline(best_h, n, deadline - 2, rng)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
