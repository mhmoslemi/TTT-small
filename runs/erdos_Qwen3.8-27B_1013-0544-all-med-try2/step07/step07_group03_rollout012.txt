```python
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

        if time.time() > deadline:
            break

        c = np.correlate(h, 1-h, mode='full') * dx
        c_max = c.max()
        
        # Softmax weights
        alpha = alpha_start * (alpha_end/alpha_start)**(step/n_steps)
        shifted = alpha * (c - c_max)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[c.argmax()] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient
        grad = np.zeros(n)
        for k_idx in range(len(c)):
            if w[k_idx] < 1e-12:
                continue
            s = k_idx - (n-1)
            j_min = max(0, -s)
            j_max = min(n-1, n-1-s)

            grad[idx1] += w[k_idx] * (1 - h[idx1 + s])
            # Second term: -h[i-s] for i-s in [j_min, j_max], i.e., i in [j_min+s, j_max+s]
            i_min2 = j_min + s
            i_max2 = j_max + s

            i_max2 = min(n-1, i_max2)

                grad[idx2] += -w[k_idx] * h[idx2 - s]
        
        # Gradient step with adaptive learning rate

        # Scale step based on gradient magnitude
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
```

```python
            g[i_lo2:i_hi2+1] += -x[i_lo2-s:i_hi2-s+1]
            return g * dx
        w /= wsum
        g = np.zeros(n)
        for k_idx in range(len(c)):
            wk = w[k_idx]
            if wk < 1e-10:
                continue
            s = k_idx - (n - 1)
            i_lo = max(0, -s)
            i_hi = min(n - 1, n - 1 - s)
            if i_lo <= i_hi:
                g[i_lo:i_hi+1] += wk * (1.0 - x[i_lo+s:i_hi+s+1])
            i_lo2 = max(0, s)
            i_hi2 = min(n - 1, n - 1 + s)
            if i_lo2 <= i_hi2:
                g[i_lo2:i_hi2+1] += -wk * x[i_lo2-s:i_hi2-s+1]
        return g * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def weighted_gradient(x, n, alpha, c):
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -60, 0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        k = int(np.argmax(c))
        s = k - (n - 1)
        g = np.zeros(n)
        i_lo = max(0, -s); i_hi = min(n - 1, n - 1 - s)
        if i_lo <= i_hi:
            g[i_lo:i_hi+1] = (1.0 - x[i_lo+s:i_hi+s+1])
        i_lo2 = max(0, s); i_hi2 = min(n - 1, n - 1 + s)
        if i_lo2 <= i_hi2:
            g[i_lo2:i_hi2+1] += -x[i_lo2-s:i_hi2-s+1]
        return g
    w /= wsum
    g = np.zeros(n)
    dx = 2.0 / n
    for k_idx in range(len(c)):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        s = k_idx - (n - 1)
        i_lo = max(0, -s); i_hi = min(n - 1, n - 1 - s)
        if i_lo <= i_hi:
            g[i_lo:i_hi+1] += wk * (1.0 - x[i_lo+s:i_hi+s+1])
        i_lo2 = max(0, s); i_hi2 = min(n - 1, n - 1 + s)
        if i_lo2 <= i_hi2:
            g[i_lo2:i_hi2+1] += -wk * x[i_lo2-s:i_hi2-s+1]
    return g * dx

def slsqp_jac(h_init, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = np.clip(alpha * (c - m), -60, 0)
        return m + np.log(np.exp(shifted).sum()) / alpha
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        return weighted_gradient(x, n, alpha, c)
    bounds = [(0.0, 1.0)] * n
    ones_vec = np.ones(n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: ones_vec}]
    try:
        res = minimize(obj, h_init, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

def subgrad_opt(h_init, n, deadline, alpha_start=20, alpha_end=5e6, n_iters=8000, mom=0.85):
    dx = 2.0 / n
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max(n_iters, 1))
        c = np.correlate(h, 1 - h, mode='full') * dx
        g = weighted_gradient(h, n, alpha, c)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        eta = 0.03 / (1.0 + 0.005 * it)
        v = mom * v - eta * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new <= best_c + 0.002:
            h = h_new
        else:
            v *= 0.3
    return best_h, best_c

def greed_binary(h_init, n, deadline, max_sweeps=150):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for idx_i in range(len(zeros_list)):
            if time.time() > deadline:
                break
            i = zeros_list[idx_i]
            best_nc = cur_c
            best_j = -1
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                j = best_j
                delta = np.zeros(2 * n - 1)
                im = i + m_range; msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range; msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range; msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range; msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)

    # Phase 1: binary greedy
    p1 = t0 + 0.05 * total

    # Phase 2: SLSQP with analytic jac (main)
    p2 = t0 + 0.55 * total

    h = best_h.copy()
    for alpha in alpha_schedule:

        c = compute_c5(h, n)
        if c < best_c:

    # Phase 3: subgradient from best
    p3 = t0 + 0.72 * total
    hs, cs = subgrad_opt(best_h, n, p3, alpha_start=10, alpha_end=1e7, n_iters=6000)
    if cs < best_c:

    # Phase 4: binary polish
    p4 = t0 + 0.82 * total
    hb, cb = greed_binary(best_h, n, p4, max_sweeps=60)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 5: final SLSQP high-alpha polish
    p5 = dl - 3
    for alpha in [5000000, 20000000, 100000000]:
        if time.time() > p5 - 2:

    # Phase 6: try n=200 for a different perspective

        nx = 200
        xo = np.linspace(0, 2, n)

        hh = project_feasible(np.interp(xn, xo, best_h), nx)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
