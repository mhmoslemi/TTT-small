import numpy as np

from scipy.optimize import minimize

_SLQ_N = 0

def enforce_sum(h, target):
    for _ in range(10):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        tol = 1e-12
        if diff > 0.0:
            idx = np.where(h < 1.0 - tol)[0]
            if idx.size == 0:
                break
            room = float((1.0 - h[idx]).sum())
            if diff <= room:
                h[idx] += diff / idx.size
            else:
                h[idx] = 1.0
        else:
            idx = np.where(h > tol)[0]
            if idx.size == 0:
                break
            room = float(h[idx].sum())
            if -diff <= room:
                h[idx] += diff / idx.size
            else:
                h[idx] = 0.0
    return np.clip(h, 0.0, 1.0)


def project_polytope(v, target=None):
    v = np.asarray(v, dtype=float)
    n = len(v)
    if target is None:
        target = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    if hi <= lo:
        hi = lo + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return enforce_sum(y, target)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def interp_to(h, nh):
    n = len(h)
    xo = np.linspace(0.0, 2.0, n)
    xn = np.linspace(0.0, 2.0, nh)
    y = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_polytope(y, 0.5 * nh)


def to_binary(h):
    n = len(h)
    order = np.argsort(h)[- (n // 2):]
    b = np.zeros(n)
    b[order] = 1.0
    return b


def pair_params(n):
    if n <= 450:
        return 26, 100, 14
    if n <= 700:
        return 20, 80, 12
    return 15, 55, 10


def fill_pair_lin(lin, h, rev, i, j, n):
    lin.fill(0.0)
    lin[i:i + n] += rev
    lin[n - 1 - i:2 * n - 1 - i] -= h
    lin[j:j + n] -= rev
    lin[n - 1 - j:2 * n - 1 - j] += h


def pair_value(tmp, carr, lin, b1, b2, delta):
    np.multiply(lin, delta, out=tmp)
    tmp += carr
    d2 = delta * delta
    tmp[b1] += d2
    tmp[b2] += d2
    return float(tmp.max())


def minimize_pair_delta(carr, lin, b1, b2, dmax, base_raw, tmp, iters):
    if dmax <= 1e-12:
        return 0.0, base_raw
    a = 0.0
    b = float(dmax)
    phi = 0.6180339887498949
    c1 = b - phi * (b - a)
    c2 = a + phi * (b - a)
    f1 = pair_value(tmp, carr, lin, b1, b2, c1)
    f2 = pair_value(tmp, carr, lin, b1, b2, c2)
    for _ in range(max(1, int(iters))):
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - phi * (b - a)
            f1 = pair_value(tmp, carr, lin, b1, b2, c1)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + phi * (b - a)
            f2 = pair_value(tmp, carr, lin, b1, b2, c2)
    delta = 0.5 * (a + b)
    fbest = pair_value(tmp, carr, lin, b1, b2, delta)
    fend = pair_value(tmp, carr, lin, b1, b2, dmax)
    if fend < fbest:
        delta = dmax
        fbest = fend
    if base_raw < fbest:
        delta = 0.0
        fbest = base_raw
    return delta, fbest


def pair_polish(h, deadline, alpha=1000.0, R=24, test_limit=80, iters=12,
                rng=None, recomp=15):
    n = len(h)
    target = 0.5 * n
    h = project_polytope(h, target).astype(float).copy()
    dx = 2.0 / n
    m = 2 * n - 1
    if rng is None:
        rng = np.random.RandomState(0)

    c = np.correlate(h, 1.0 - h, mode="full").astype(float)
    cur = float(c.max()) * dx
    best_h = h.copy()
    best_c = cur
    rev = 1.0 - h
    ones = np.ones(n)
    lin = np.zeros(m)
    tmp = np.empty(m)
    stall = 0
    sweep = 0

    while time.time() < deadline:
        sweep += 1
        if sweep % recomp == 0:
            c = np.correlate(h, 1.0 - h, mode="full").astype(float)
            cur = float(c.max()) * dx
            if cur < best_c:
                best_c = cur
                best_h = h.copy()

        base_raw = float(c.max())
        sh = float(alpha) * dx * (c - base_raw)
        sh -= sh.max()
        w = np.exp(sh)
        ws = float(w.sum())
        if ws <= 0.0:
            w = np.full(m, 1.0 / m)
        else:
            w /= ws

        D = np.convolve(w, ones, mode="full")[n - 1:2 * n - 1]
        A = np.convolve(w, h, mode="full")[n - 1:2 * n - 1]
        S = np.convolve(h[::-1], w, mode="full")
        B = S[n - 1:2 * n - 1][::-1]
        grad = dx * (D - A - B)

        order = np.argsort(grad)
        incs = order[:R]
        decs = order[-R:][::-1]

        cand = []
        for i in incs:
            gi = grad[i]
            hi = 1.0 - h[i]
            if hi <= 1e-12:
                continue
            for j in decs:
                if i == j:
                    continue
                hj = h[j]
                if hj <= 1e-12:
                    continue
                dm = hi if hi < hj else hj
                if dm < 1e-10:
                    continue
                dv = gi - grad[j]
                if dv < -1e-12:
                    cand.append((dv, int(i), int(j), float(dm)))

        if not cand:
            for _ in range(2 * R):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                if h[i] >= 1.0 - 1e-12 or h[j] <= 1e-12:
                    continue
                if grad[i] < grad[j]:
                    dm = min(1.0 - h[i], h[j])
                    if dm > 1e-10:
                        cand.append((float(grad[i] - grad[j]), i, j, float(dm)))

        if not cand:
            stall += 1
            if stall >= 3:
                if stall >= 8:
                    h = best_h.copy()
                strength = 0.035 / (1.0 + 0.2 * stall)
                h = h + strength * (rng.rand(n) - 0.5)
                h = project_polytope(h, target).astype(float).copy()
                rev = 1.0 - h
                c = np.correlate(h, 1.0 - h, mode="full").astype(float)
                cur = float(c.max()) * dx
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            continue

        if len(cand) > 220:
            cand.sort()
            cand = cand[:220]
        else:
            cand.sort()

        improved = False
        tested = 0
        for _score, i, j, dm in cand:
            if tested >= test_limit or time.time() >= deadline:
                break
            tested += 1
            fill_pair_lin(lin, h, rev, i, j, n)
            b1 = i - j + n - 1
            b2 = j - i + n - 1
            delta, new_raw = minimize_pair_delta(c, lin, b1, b2, dm, base_raw, tmp, iters)
            if new_raw < base_raw - 1e-12:
                h[i] += delta
                h[j] -= delta
                rev[i] -= delta
                rev[j] += delta
                c += delta * lin
                dd = delta * delta
                c[b1] += dd
                c[b2] += dd
                new_raw2 = float(c.max())
                if new_raw2 >= base_raw:
                    h[i] -= delta
                    h[j] += delta
                    rev[i] += delta
                    rev[j] -= delta
                    c = np.correlate(h, 1.0 - h, mode="full").astype(float)
                else:
                    cur = new_raw2 * dx
                    if cur < best_c:
                        best_c = cur
                        best_h = h.copy()
                    base_raw = new_raw2
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1

        if sweep % 60 == 0:
            enforce_sum(h, target)
            rev = 1.0 - h

    best_h = project_polytope(best_h, 0.5 * len(best_h))
    bc = compute_c5(best_h)
    if bc < best_c:
        best_c = bc
    return best_h, best_c


def slsqp_objective(x):
    x = np.asarray(x, dtype=float)
    n = _SLQ_N
    dx = _SLQ_DX
    alpha = _SLQ_ALPHA
    c = np.correlate(x, 1.0 - x, mode="full")
    C = c * dx
    mx = float(C.max())
    if alpha <= 1e-8:
        return mx
    sh = alpha * (C - mx)
    sh -= sh.max()
    s = float(np.exp(sh).sum())
    if s <= 0.0:
        return mx
    return mx + math.log(s) / alpha


def slsqp_gradient(x):
    x = np.asarray(x, dtype=float)
    n = _SLQ_N
    dx = _SLQ_DX
    alpha = _SLQ_ALPHA
    c = np.correlate(x, 1.0 - x, mode="full")
    C = c * dx
    mx = float(C.max())
    if alpha <= 1e-8:
        w = np.full(len(C), 1.0 / len(C))
    else:
        sh = alpha * (C - mx)
        sh -= sh.max()
        w = np.exp(sh)
        ws = float(w.sum())
        if ws <= 0.0:
            w = np.full(len(C), 1.0 / len(C))
        else:
            w /= ws
    ones = np.ones(n)
    D = np.convolve(w, ones, mode="full")[n - 1:2 * n - 1]
    A = np.convolve(w, x, mode="full")[n - 1:2 * n - 1]
    S = np.convolve(x[::-1], w, mode="full")
    B = S[n - 1:2 * n - 1][::-1]
    return dx * (D - A - B)


def slsqp_constraint(x):
    return float(np.asarray(x).sum()) - 0.5 * _SLQ_N


def slsqp_smooth(h, alpha, maxiter=80):
    global _SLQ_N, _SLQ_ALPHA, _SLQ_DX
    h = project_polytope(h, 0.5 * len(h)).astype(float).copy()
    n = len(h)
    _SLQ_N = n
    _SLQ_ALPHA = float(alpha)
    _SLQ_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [
        {"type": "eq", "fun": slsqp_constraint},
    ]
    try:
        res = minimize(
            slsqp_objective,
            h,
            jac=slsqp_gradient,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max(10, int(maxiter)), "ftol": 1e-12},
        )
        x = np.asarray(res.x, dtype=float)
        if np.all(np.isfinite(x)):
            return project_polytope(x, 0.5 * n)
    except Exception:
        pass
    return project_polytope(h, 0.5 * n)


def thue_values(n):
    a = np.array([0.0])
    while len(a) < n:
        a = np.concatenate((a, 1.0 - a))
    return a[:n].copy()


def weyl_values(n):
    alpha = (math.sqrt(5.0) - 1.0) / 2.0
    x = (np.arange(n, dtype=float) * alpha) % 1.0
    return (x < 0.5).astype(float)


def symmetric_candidate(h):
    s = 0.5 * (h + h[::-1])
    return project_polytope(s, 0.5 * len(h))


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    span = float(max(1.0, float(budget_s)))
    hard_end = t0 + span - 12.0
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(400, 0.5)

    n = len(h)
    if n < 50:
        n = 400
        h = np.full(n, 0.5)
    if n % 2 == 1:
        h = interp_to(h, n + 1)
        n = n + 1
    h = project_polytope(h, 0.5 * n).astype(float).copy()

    best_h = h.copy()
    best_c = compute_c5(best_h)
    best_n = n

    # Quick local SLSQP polish from the supplied construction.
    p1 = t0 + 0.08 * span
    if time.time() < p1:
        hp = slsqp_smooth(best_h, 250.0, 100)
        cp = compute_c5(hp)
        if cp < best_c:
            best_h, best_c, best_n = hp, cp, len(hp)

    # Cheap resolution probes before the main local search.
    p2 = t0 + 0.45 * span
    qend = min(p2, t0 + 25.0)
    for add in (80, 160, 320):
        nh = best_n + add
        if nh > 880 or nh <= best_n:
            continue
        if nh % 2 == 1:
            nh += 1
        if time.time() >= qend:
            break
        hh = interp_to(best_h, nh)
        cc = compute_c5(hh)
        if cc < best_c - 1e-12:
            best_h, best_c, best_n = hh, cc, nh

    R, TL, IT = pair_params(best_n)
    hp, cp = pair_polish(best_h, p2, alpha=450.0, R=R, test_limit=TL, iters=IT, rng=rng)
    if cp < best_c:
        best_h, best_c, best_n = hp, cp, len(hp)

    p3 = t0 + 0.60 * span
    if time.time() < p3:
        hp = slsqp_smooth(best_h, 4000.0, 110)
        cp = compute_c5(hp)
        if cp < best_c:
            best_h, best_c, best_n = hp, cp, len(hp)

    R, TL, IT = pair_params(best_n)
    hp, cp = pair_polish(best_h, t0 + 0.78 * span, alpha=5000.0, R=R,
                         test_limit=min(140, TL + 30), iters=IT + 2, rng=rng)
    if cp < best_c:
        best_h, best_c, best_n = hp, cp, len(hp)

    # Binary restart at the current resolution.
    p5 = t0 + 0.86 * span
    if best_n % 2 == 0 and time.time() < p5:
        hb = to_binary(best_h)
        hb = project_polytope(hb, 0.5 * best_n)
        d = min(p5, time.time() + 90.0)
        if d > time.time() + 5:
            Rb, TLb, ITb = pair_params(best_n)
            hb, cb = pair_polish(hb, d, alpha=3000.0, R=Rb, test_limit=TLb,
                                 iters=ITb, rng=rng)
            if cb < best_c:
                best_h, best_c, best_n = hb, cb, best_n

    # Resolution expansion.
    res_end = min(hard_end, t0 + 0.97 * span)
    seen = set()
    for target in (512, 640, 768, 896):
        if target in seen or target <= best_n or target > 900:
            continue
        seen.add(target)
        if time.time() > res_end - 35:
            break
        hh = interp_to(best_h, target)
        cc0 = compute_c5(hh)
        budget_here = 110.0 if target <= 640 else 65.0
        d = min(res_end - 18, time.time() + budget_here)
        Rh, TLh, ITh = pair_params(target)
        if d > time.time() + 5:
            hh, cc = pair_polish(hh, d, alpha=3500.0, R=Rh, test_limit=TLh,
                                 iters=ITh, rng=rng)
        else:
            cc = cc0
        if cc < best_c - 1e-12:
            best_h, best_c, best_n = hh, cc, target
        elif cc0 < best_c - 1e-12:
            best_h, best_c, best_n = hh, cc0, target

    # Alternative structured starts if time remains.
    alt_end = min(hard_end, time.time() + 160.0)
    if time.time() < alt_end - 25 and best_n >= 100:
        alts = []
        alts.append(symmetric_candidate(best_h))
        alts.append(project_polytope(weyl_values(best_n), 0.5 * best_n))
        alts.append(project_polytope(thue_values(best_n), 0.5 * best_n))
        alts.append(project_polytope(rng.rand(best_n), 0.5 * best_n))
        Radt, TLadt, ITadt = pair_params(best_n)
        for hstart in alts:
            if time.time() > alt_end - 12:
                break
            d = min(alt_end - 8, time.time() + 55.0)
            if d <= time.time() + 3:
                continue
            hh, cc = pair_polish(hstart, d, alpha=2500.0, R=Radt, test_limit=TLadt,
                                 iters=ITadt, rng=rng)
            if cc < best_c:
                best_h, best_c, best_n = hh, cc, len(hh)

    best_h = project_polytope(best_h, 0.5 * best_n).astype(float)
    best_c = compute_c5(best_h)
    return best_h, best_c, best_n
