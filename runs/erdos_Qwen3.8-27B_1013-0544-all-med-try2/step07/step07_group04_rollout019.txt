1. Precompute total_effect (n × (2n-1))
2. Generate a batch of candidate swaps (e.g., all zero positions × top K one positions)
3. Evaluate all candidates in batch
4. Accept the best improving swap
5. Update corr and total_effect (only need to update the two affected rows)
6. Repeat

Updating total_effect after a swap:
- total_effect[i] changes because h[i] changed (affects the h[p-k+(n-1)] term for all k)
  and g[i] changed (affects the g[p+k-(n-1)] term for all k)
  
Actually, total_effect[p][k] = g[p+k-(n-1)] - h[p-k+(n-1)] (with bounds)

After swapping h[i]: 0→1 and h[j]: 1→0:
- For total_effect[p][k]:
  - g[p+k-(n-1)] changes if p+k-(n-1) = i (g[i] went 1→0) or p+k-(n-1) = j (g[j] went 0→1)
  - h[p-k+(n-1)] changes if p-k+(n-1) = i (h[i] went 0→1) or p-k+(n-1) = j (h[j] went 1→0)

So total_effect[p] only changes for specific p values:
- From g change at i: p+k-(n-1) = i → p = i-k+(n-1). For all k, this means p = i-(k-(n-1)) = i + (n-1-k). As k ranges over [0, 2n-2], (n-1-k) ranges over [-(n-1), n-1]. So p ranges over i-(n-1) to i+(n-1).
  The change is: g[i] went 1→0, so g[p+k-(n-1)] decreases by 1 when p+k-(n-1)=i.
  So total_effect[p][k] decreases by 1 when k = i-p+(n-1) (for valid k).

- From g change at j: total_effect[p][k] increases by 1 when k = j-p+(n-1) (for valid k).

- From h change at i: h[i] went 0→1, so h[p-k+(n-1)] increases by 1 when p-k+(n-1)=i, i.e., k = p-i+(n-1).
  total_effect[p][k] decreases by 1 at k = p-i+(n-1).

- From h change at j: h[j] went 1→0, so h[p-k+(n-1)] decreases by 1 when p-k+(n-1)=j, i.e., k = p-j+(n-1).
  total_effect[p][k] increases by 1 at k = p-j+(n-1).

So for each p, total_effect[p] changes at at most 4 positions. This means I can update total_effect efficiently after each swap!

Total update cost: O(n) (update 4 positions for each of n rows).

This makes the algorithm very fast:
- Precomputation: O(n²)
- Each iteration: O(n²) for batch evaluation (vectorized) + O(n) for update
- Can run for tens of thousands of iterations

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_correlation(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_total_effect(h, n):
    """Precompute total_effect[p][k] for all p, k."""
    n_corr = 2 * n - 1
    g = 1 - h
    total_effect = np.zeros((n, n_corr))
    for p in range(n):
        # g[p+k-(n-1)] for valid k
        k_min = max(0, n - 1 - p)
        k_max = min(n_corr - 1, 2 * n - 1 - p)
        if k_min <= k_max:
            ks = np.arange(k_min, k_max + 1)
            total_effect[p, ks] += g[p + ks - (n - 1)]
        # -h[p-k+(n-1)] for valid k
        k_min2 = max(0, p)
        k_max2 = min(n_corr - 1, p + n - 1)
        if k_min2 <= k_max2:
            ks2 = np.arange(k_min2, k_max2 + 1)
            total_effect[p, ks2] -= h[p - ks2 + (n - 1)]
    return total_effect

def update_total_effect(total_effect, h, n, i, j):
    """Update total_effect after swapping h[i]:0→1 and h[j]:1→0."""
    n_corr = 2 * n - 1
    # h[i] went 0→1, h[j] went 1→0
    # g[i] went 1→0, g[j] went 0→1
    for p in range(n):
        # g change at i: total_effect[p][k] changes by -1 at k = i-p+(n-1)
        k = i - p + (n - 1)
        if 0 <= k < n_corr:
            total_effect[p, k] -= 1.0
        # g change at j: total_effect[p][k] changes by +1 at k = j-p+(n-1)
        k = j - p + (n - 1)
        if 0 <= k < n_corr:
            total_effect[p, k] += 1.0
        # h change at i: total_effect[p][k] changes by -1 at k = p-i+(n-1)
        k = p - i + (n - 1)
        if 0 <= k < n_corr:
            total_effect[p, k] -= 1.0
        # h change at j: total_effect[p][k] changes by +1 at k = p-j+(n-1)
        k = p - j + (n - 1)
        if 0 <= k < n_corr:
            total_effect[p, k] += 1.0

def batch_evaluate(corr, total_effect, i_list, j_list, n):
    """Evaluate multiple swaps. Returns new max correlation values."""
    n_corr = 2 * n - 1
    deltas = total_effect[i_list] - total_effect[j_list]
    deltas[:, n - 1] -= 2.0
    k1 = n - 1 + (j_list - i_list)
    m1 = (k1 >= 0) & (k1 < n_corr)
    if m1.any():
        deltas[m1, k1[m1]] += 1.0
    k2 = n - 1 + (i_list - j_list)
    m2 = (k2 >= 0) & (k2 < n_corr)
    if m2.any():
        deltas[m2, k2[m2]] += 1.0
    return (corr[None, :] + deltas).max(axis=1)

def vectorized_binary_search(h, n, deadline, batch_size=2000, max_iters=50000, seed=42):
    """Fast vectorized binary local search."""
    n_corr = 2 * n - 1
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    corr = get_correlation(h, n).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    total_effect = compute_total_effect(h, n)
    rng = np.random.RandomState(seed)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        # Generate candidates: all zeros × subset of ones (or all if small enough)
        if len(zeros) * len(ones) <= batch_size * 4:
            i_list, j_list = np.meshgrid(zeros, ones, indexing='ij')
            i_list, j_list = i_list.ravel(), j_list.ravel()
        else:
            # Sample a batch
            n_pairs = batch_size
            i_list = rng.choice(zeros, n_pairs, replace=True)
            j_list = rng.choice(ones, n_pairs, replace=True)
            # Remove duplicates where i==j (shouldn't happen since i in zeros, j in ones)
        
        new_maxes = batch_evaluate(corr, total_effect, i_list, j_list, n)
        # Find improving swaps
        improving = np.where(new_maxes * dx < cur_c - 1e-12)[0]
        
        if len(improving) == 0:
            # Try with a lower threshold (allow tiny improvements)
            improving = np.where(new_maxes * dx < cur_c - 1e-15)[0]
            if len(improving) == 0:
                break
        
        # Pick the best improving swap
        best_idx = improving[np.argmin(new_maxes[improving])]
        i, j = i_list[best_idx], j_list[best_idx]
        
        # Apply the swap
        h[i], h[j] = 1.0, 0.0
        corr = batch_evaluate(corr, total_effect, np.array([i]), np.array([j]), n)[0]
        update_total_effect(total_effect, h, n, i, j)
        cur_c = corr.max() * dx
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
