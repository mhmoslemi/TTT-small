```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(h - t, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def grad_pnorm(h, n, p, k_active=10):
    """Compute F_p and its gradient, using only top-k active terms"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    f = corr[n-1:].astype(float)
    
    # Get top-k indices
    if k_active < n:
        top_idx = np.argpartition(f, -k_active)[-k_active:]
    else:
        top_idx = np.arange(n)
    
    f_top = f[top_idx]
    fp = f_top ** p
    S = fp.sum()
    if S < 1e-30:
        return dx * f.max(), np.zeros(n)
    
    w = S ** (1.0/p - 1.0) * f_top ** (p - 1)
    F = dx * S ** (1.0/p)
    
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        d = top_idx[idx]
        wd = w[idx]
        if wd < 1e-15:
            continue
        grad[:n-d] += wd * (1.0 - h[d:])
        grad[d:] -= wd * h[:n-d]
    
    return F, grad * dx

def nesterov_optimize(h, n, deadline, p=100, step=0.05, momentum=0.9, k_active=10, n_iter=100000):
    """Nesterov accelerated projected gradient"""
    h = project_feasible(h.astype(float).copy(), n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        # Compute F and gradient
        F, g = grad_pnorm(h, n, p, k_active)
        
        # Nesterov lookahead
        h_look = project_feasible(h + momentum * v, n)
        F_look, g_look = grad_pnorm(h_look, n, p, k_active)
        
        # Update velocity
        v = momentum * v - step * g_look
        
        # Position update
        h_new = project_feasible(h + v, n)
        
        # Backtracking: check if we improved
        c_new = compute_c5(h_new, n)
        if c_new > best_c * 1.01 and it > 100:
            # Too bad, reduce step
            step *= 0.95
            v = v * 0.5
        elif c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Adaptive step: occasionally increase
        if it % 1000 == 0 and it > 0:
            step = min(step * 1.1, 0.5)
    
    return best_h, best_c

def lbfgs_optimize(h, n, deadline, p=50, maxiter=5000, k_active=20):
    """L-BFGS-B with p-norm penalty for sum constraint"""
    dx = 2.0 / n
    penalty = 1000.0
    
    def objective(x):
        F, g = grad_pnorm(x, n, p, k_active)
        sum_pen = penalty * (x.sum() - n/2.0) ** 2
        sum_grad = 2 * penalty * (x.sum() - n/2.0)
        return F + sum_pen, g + sum_grad * np.ones(n)
    
    # Use L-BFGS-B with bounds
    bounds = [(0.0, 1.0)] * n
    x0 = project_feasible(h.astype(float).copy(), n)
    
    best_h = x0.copy()
    best_c = compute_c5(x0, n)
    
    t_end = deadline
    try:
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    except Exception:
        pass
    
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=200):
    """Greedy binary local search"""
    dx = 2.0 / n
    h_b = to_binary(h, n).copy()
    corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    ones_list = list(np.where(h_b == 1)[0])
    zeros_list = list(np.where(h_b == 0)[0])
    m_range = np.arange(-(n-1), n)
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_delta = 0
            best_j = -1
            for j in ones_list:
                # Compute change in max correlation
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta = np.zeros(2*n - 1)
                delta[msk] += 1.0 - h_b[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h_b[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h_b[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h_b[jmn[msk]]
                delta[n-1] -= 2.0
                delta[n-1 + (j-i)] += 1.0
                delta[n-1 + (i-j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    best_delta = nc
                    best_j = j
                    break
            if best_j >= 0:
                h_b[i], h_b[best_j] = 1.0, 0.0
                corr += delta
                cur_c = best_delta
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_b.copy()
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(h - t, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def grad_pnorm(h, n, p, k_active=8):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    f = corr[n - 1:].astype(float)
    if k_active < n:
        top_idx = np.argpartition(f, -k_active)[-k_active:]
    else:
        top_idx = np.arange(n)
    f_top = f[top_idx]
    S = (f_top ** p).sum()
    if S < 1e-30:
        return dx * f.max(), np.zeros(n)
    w = S ** (1.0 / p - 1.0) * f_top ** (p - 1)
    F = dx * S ** (1.0 / p)
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        d = top_idx[idx]
        wd = w[idx]
        if wd < 1e-12:
            continue
        grad[:n - d] += wd * (1.0 - h[d:])
        grad[d:] -= wd * h[:n - d]
    return F, grad * dx

def nesterov_opt(h, n, deadline, p=100, step=0.08, momentum=0.9, k_act=6, n_iter=500000):
    h = project_feasible(h.astype(float).copy(), n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        _, g = grad_pnorm(h, n, p, k_act)
        v = momentum * v - step * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-10:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                v = np.zeros(n)
                step = max(step * 0.8, 0.001)
                stall = 0
        if stall % 5000 == 0 and stall > 0:
            step = min(step * 1.05, 0.5)
        h = h_new
    return best_h, best_c

def lbfgs_opt(h, n, deadline, p=50, maxiter=3000, k_act=15):
    penalty = 5000.0
    def obj_grad(x):
        F, g = grad_pnorm(x, n, p, k_act)
        sm = x.sum() - n / 2.0
        sg = 2.0 * penalty * sm
        return F + penalty * sm * sm, g + sg
    x0 = project_feasible(h.astype(float).copy(), n)
    bounds = [(0.0, 1.0)] * n
    best_h = x0.copy()
    best_c = compute_c5(x0, n)
    try:
        res = minimize(obj_grad, x0, jac=True, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': 200000})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    except Exception:
        pass
    return best_h, best_c

def binary_ls(h, n, deadline, max_sweeps=300):
    dx = 2.0 / n
    h_b = to_binary(h, n).copy()
    corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    ones_list = list(np.where(h_b == 1)[0])
    zeros_list = list(np.where(h_b == 0)[0])
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h_b[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h_b[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h_b[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h_b[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h_b[i], h_b[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c, best_h = cur_c, h_b.copy()
    return best_h, best_c

def anneal_binary(h, n, deadline, seed=42, n_iter=2000000):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h_b = to_binary(h, n).copy()
    corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    zeros_arr = np.where(h_b == 0)[0]
    ones_arr = np.where(h_b == 1)[0]
    m_range = np.arange(-(n - 1), n)
    T = 0.01
    cool = 0.99997
    scale = 2000.0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h_b[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h_b[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h_b[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h_b[jmn[msk]]
        delta[n - 1] -= 2.0
        delta[n - 1 + (j - i)] += 1.0
        delta[n - 1 + (i - j)] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h_b[i], h_b[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c, best_h = cur_c, h_b.copy()
        T *= cool
    return best_h, best_c
```
