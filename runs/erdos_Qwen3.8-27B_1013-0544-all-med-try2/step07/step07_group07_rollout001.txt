import numpy as np
import time
from scipy.optimize import minimize_scalar


def _c5(h):
    n = len(h)
    if n == 0:
        return float("inf")
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def _project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n

    for _ in range(30):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 1e-15:
                break
            h += (d / tot) * room
        else:
            tot = h.sum()
            if tot <= 1e-15:
                break
            h += (d / tot) * h
        np.clip(h, 0.0, 1.0, out=h)

    d = target - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:
            if d > 0.0:
                add = min(d, 1.0 - h[idx])
            else:
                add = max(d, -h[idx])
            if add == 0.0:
                continue
            h[idx] += add
            d -= add
            if abs(d) < 1e-14:
                break

    return h


def _cos_h(n, a):
    x = np.arange(n, dtype=float)
    return 0.5 + float(a) * np.cos((2.0 * np.pi * (x + 0.5)) / n)


def _cos_obj_raw(a, n):
    return _c5(_cos_h(n, a))


def _tune_cos_amplitude(n):
    base = 1.0 / np.sqrt(6.0)
    grid = np.array(
        [0.20, 0.30, 0.35, 0.40,
         base - 0.02, base - 0.01, base,
         base + 0.01, base + 0.02,
         0.45, 0.48, 0.50]
    )

    best_a = float(grid[0])
    best_v = float("inf")
    for a in grid:
        v = _cos_obj_raw(float(a), n)
        if v < best_v:
            best_v = v
            best_a = float(a)

    lo = max(0.0, best_a - 0.02)
    hi = min(0.5, best_a + 0.02)
    if hi - lo > 1e-10:
        try:
            res = minimize_scalar(
                _cos_obj_raw,
                args=(n,),
                bounds=(lo, hi),
                method="bounded",
                options={"xatol": 1e-12},
            )
            if res.fun < best_v:
                best_a = float(np.clip(res.x, 0.0, 0.5))
                best_v = float(res.fun)
        except Exception:
            pass

    return best_a


def _candidate_from_cos(n):
    a = _tune_cos_amplitude(n)
    h = _cos_h(n, a)
    h = _project_feasible(h, n)
    return h, _c5(h)
