    for _ in range(100):

            s = room.sum()

            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

def topk_obj_grad(h, n, K, alpha=1.0):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Get top K indices
    topk_idx = np.argsort(c)[::-1][:K]
    f = c[topk_idx].sum()
    
    # Gradient: sum of gradients of top K correlations
    # dc[m]/dh[k] = (1-h[k-p_m])*I(0<=k-p_m<n) - h[k+p_m]*I(0<=k+p_m<n)
    # where p_m = m - (n-1)

    # S1[k] = sum_{m in topK} (1-h[k-p_m]) * I(0<=k-p_m<n)
    # S2[k] = sum_{m in topK} h[k+p_m] * I(0<=k+p_m<n)
    
    K_idx = np.arange(n)[:, None]  # (n, 1)
    P_top = p_top[None, :]  # (1, K)
    
    idx1 = K_idx - P_top  # (n, K) = k - p_m
    mask1 = (idx1 >= 0) & (idx1 < n)

    idx2_clip = np.clip(idx2, 0, n-1)
    S2 = (h[idx2_clip] * mask2).sum(axis=1)
    
    grad = dx * (S1 - S2)
    return f, grad

        if c2 < best_c:
            best_c, best_h = c2, h2
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
