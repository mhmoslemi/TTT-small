```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        h = np.clip(h, 0.0, 1.0)
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    max_c = c.max()
    shifted = alpha * (c - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    obj = (max_c + np.log(sum_exp) / alpha) * dx
    p = exp_shifted / sum_exp
    grad_max = np.ones(n)
    for idx in range(len(p)):
        k = idx - (n - 1)
        pk = p[idx]
        if pk < 1e-14:
            continue
        i_lo = max(0, -k)
        i_hi = min(n - 1, n - 1 - k)
        if i_lo <= i_hi:
            grad_max[i_lo:i_hi+1] -= pk * h[i_lo+k:i_hi+k+1]
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, k + n - 1)
        if i_lo2 <= i_hi2:
            grad_max[i_lo2:i_hi2+1] -= pk * h[i_lo2-k:i_hi2-k+1]
    return obj, grad_max * dx

def lbfgs_optimize(h, n, alpha, penalty, maxiter, dl):
    def obj(x):
        s, _ = smooth_max_and_grad(x, alpha)
        return s + penalty * (x.sum() - n / 2.0) ** 2
    def obj_grad(x):
        _, g = smooth_max_and_grad(x, alpha)
        return g + 2.0 * penalty * (x.sum() - n / 2.0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='L-BFGS-B', jac=obj_grad, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    return project_feasible(res.x, n)

def binary_local_search(h_init, n, dl, max_sweeps=10000, n_restarts=3, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    
    for restart in range(n_restarts):
        if time.time() > dl:
            break
        if restart > 0:
            for _ in range(rng.randint(2, 6)):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    h[int(rng.choice(oi))] = 0.0
                    h[int(rng.choice(zi))] = 1.0
            corr = np.correlate(h, 1 - h, mode='full')
            cur_c = corr.max() * dx
        
        for sweep in range(max_sweeps):
            if time.time() > dl:
                break
            # Precompute M and N
            pos = np.arange(n)[:, None]
            mr = m_range[None, :]
            im = pos + mr
            jm = pos - mr
            M = np.zeros((n, 2*n-1))
            N = np.zeros((n, 2*n-1))
            msk = (im >= 0) & (im < n)
            M[msk] = 1.0 - h[im[msk]]
            msk2 = (jm >= 0) & (jm < n)
            N[msk2] = h[jm[msk2]]
            
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if len(zeros) == 0 or len(ones) == 0:
                break
            
            best_swap = None
            best_nc = cur_c
            
            for i in zeros:
                im_i = i + m_range
                imn_i = i - m_range
                part_i = np.zeros(2*n-1)
                msk = (im_i >= 0) & (im_i < n)
                part_i[msk] += 1.0 - h[im_i[msk]]
                msk = (imn_i >= 0) & (imn_i < n)
                part_i[msk] -= h[imn_i[msk]]
                part_i[n-1] -= 2.0
                
                delta_all = part_i[None, :] - M[ones, :] + N[ones, :]
                for jj, j in enumerate(ones):
                    delta_all[jj, n-1+j-i] += 1.0
                    delta_all[jj, n-1+i-j] += 1.0
                
                new_max = (corr[None, :] + delta_all).max(axis=1) * dx
                bi = new_max.argmin()
                if new_max[bi] < best_nc:
                    best_nc = new_max[bi]
                    best_swap = (i, ones[bi], delta_all[bi])
            
            if best_swap is not None:
                i, j, delta = best_swap
                h[i] = 1.0
                h[j] = 0.0
                corr = corr + delta
                cur_c = best_nc
            else:
                break
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c
