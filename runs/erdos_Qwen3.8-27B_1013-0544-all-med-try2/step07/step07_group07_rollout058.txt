import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_polytope(v, n):
    """Project v onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on threshold
    lo, hi = v.min() - target, v.max() + target
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0, 1)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2, 0, 1)
    return h

def smooth_max(values, alpha):
    """Compute smooth maximum alpha * log(sum(exp(alpha * x)))"""
    m = values.max()
    s = np.exp(alpha * (values - m))
    return m + np.log(s.sum()) / alpha

def gradient_smooth_max(h, n, alpha, active_idx=None):
    """Gradient of smooth max of all (or active) lags."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if active_idx is not None:
        c_active = c[active_idx]
        weights = np.exp(alpha * (c_active - c_active.max()))
        weights /= weights.sum()
        c_ext = np.zeros(len(c))
        c_ext[active_idx] = weights
    else:
        c_ext = np.exp(alpha * (c - c.max()))
        c_ext /= c_ext.sum()
    
    grad = np.zeros(n)
    for k in range(len(c)):
        if c_ext[k] < 1e-15:
            continue
        lag = k - (n - 1)
        w = c_ext[k]
        if lag == 0:
            grad += w * (1 - 2 * h)
        elif lag > 0 and lag < n:
            grad[:n - lag] -= w * h[lag:]
            grad[lag:] += w * (1 - h[:n - lag])
        elif lag < 0 and abs(lag) < n:
            m = abs(lag)
            grad[:n - m] += w * (1 - h[m:])
            grad[m:] -= w * h[:n - m]
    return grad

def fista_optimize(h0, n, alpha, deadline, max_iter=10000):
    """FISTA optimization of smooth max with projection."""
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    
    # Lipschitz constant estimate
    dx = 2.0 / n
    L = 4.0 * n / dx  # rough upper bound
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = gradient_smooth_max(y, n, alpha)
        # Step size
        step = 1.0 / L
        h_new = project_polytope(y - step * grad, n)
        
        # Backtracking if needed
        while True:
            if time.time() > deadline:
                break
            c_new = compute_c5(h_new, n)
            if c_new < best_c + 0.01:
                break
            L *= 2
            step = 1.0 / L
            h_new = project_polytope(y - step * grad, n)
        
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y = h_new + ((t - 1) / t_new) * (h_new - h)
        h = h_new
        t = t_new
        
        c_curr = compute_c5(h, n)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h.copy()
    
    return best_h, best_c

        grads.append(lag_grad(h, n, lag))

        g2 = lag_grad(h, n, top_idx[1] - (n-1))

        g1g2 = np.dot(g1, g2)
        g2norm2 = np.dot(g2, g2)

            d = -g1
        else:
            lam = g1g2 / g2norm2
            d = -g1 + lam * g2

        # We need: h + td stays in [0,1]^n and sum(h+td) = n/2

        d -= sum_d / n  # Make sum(d) = 0
        
        # Find max step in this direction
        pos_mask = d > 0
        neg_mask = d < 0
        t_max = np.inf
        if pos_mask.any():

        if neg_mask.any():

        t_max = max(t_max, 1e-10)
        
        # Line search

import numpy as np
from scipy.optimize import minimize
import time
from scipy.interpolate import CubicSpline

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_polytope(v, n):
    target = n / 2.0
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def lag_grad(h, n, lag):
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if lag > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += 1.0 - h[:n-m]
    else:
        grad[:n-m] += 1.0 - h[m:]
        grad[m:] -= h[:n-m]
    return grad

def fista_opt(h0, n, top_k, alpha, deadline):

    y = h.copy()
    t = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)

    c = compute_all_lags(h, n)

def leveling_opt(h0, n, deadline, n_rounds=200):

        top_idx = np.argsort(c)[-4:]
        c_top = c[top_idx]

        if c_top[-1] - c_top[0] < 1e-9:
            break

        g_main = lag_grad(h, n, top_idx[-1] - (n-1))
        # Combine other top gradients as constraint
        g_other = np.zeros(n)
        for k in top_idx[:-1]:
            g_other += lag_grad(h, n, k - (n-1))
        g_other /= (len(top_idx) - 1)

        g1g2 = np.dot(g_main, g_other)
        g2n2 = np.dot(g_other, g_other)
        if g2n2 < 1e-15:
            d = -g_main
        else:
            lam = g1g2 / g2n2
            d = -g_main + lam * g_other

        sum_d = d.sum()
        d = d - sum_d / n

        dn = np.dot(d, d)
        if dn < 1e-20:
            break

        pos = d > 1e-12
        neg = d < -1e-12
        t_max = 1.0

                improved = True
                break

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    try:
        cs = CubicSpline(x_old, h, bc_type='not-a-knot')
        h_new = cs(x_new)
    except Exception:
        h_new = np.interp(x_new, x_old, h)
    return project_polytope(h_new, n_new)

def slsqp_polish(h, n, alpha, deadline):
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    def obj_grad(x):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        w = np.exp(alpha * (c - m))
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-10:
                lag = k - (n - 1)
                grad += w[k] * lag_grad(x, n, lag)
        return grad

    def sum_fun(x):
        return float(np.sum(x) - n / 2.0)

    def sum_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-15})
        return project_polytope(res.x, n)
    except Exception:
        return h.copy()
