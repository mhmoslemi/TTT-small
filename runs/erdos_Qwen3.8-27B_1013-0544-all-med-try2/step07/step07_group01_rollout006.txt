def project(h,n):
    target=0.5*float(n)
    h=np.asarray(h,dtype=float).copy()
    for _ in range(8):
        h=np.clip(h,0.0,1.0)
        s=float(h.sum())
        d=target-s
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            r=1.0-h
            rs=float(r.sum())
            if rs < 1e-15:
                break
            h += (d/rs)*r
        else:
            if s < 1e-15:
                break
            h += (d/s)*h
    h=np.clip(h,0.0,1.0)
    d=target-float(h.sum())
    if abs(d)>1e-12:
        idx=np.where((h>1e-10)&(h<1.0-1e-10))[0]
        if idx.size:

def smax_obj(x,n,alpha,mu):
    q=q_vec(x,n)
    m=float(q.max())
    z=alpha*(q-m); z=z-z.max()
    val=m + float(np.log(np.sum(np.exp(z))))/alpha

def smax_grad(x,n,alpha,mu):
    q=q_vec(x,n)
    m=float(q.max())
    z=alpha*(q-m); z=z-z.max()
    e=np.exp(z); e/=e.sum()
    g=grad_from_weights(x,e,n)

def lbfgs_phase(h,n,deadline,alpha=500.0,maxiter=80):

import numpy as np
import time
from scipy.optimize import minimize


def grid_x(n):
    n = int(n)
    return (np.arange(n) + 0.5) * (2.0 / float(n))


def nearest_even(x):
    x = int(x)
    if x % 2:
        x += 1
    if x < 8:
        x = 8
    return x


def q_vec(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def c5_of(h, n):
    return float(np.max(q_vec(h, n)))


def qc(h, n):
    q = q_vec(h, n)
    return q, float(np.max(q))


def project(h, n):
    target = 0.5 * float(n)
    h = np.asarray(h, dtype=float).copy()
    for _ in range(8):
        h = np.clip(h, 0.0, 1.0)
        s = float(np.sum(h))
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            r = 1.0 - h
            rs = float(np.sum(r))
            if rs < 1e-15:
                break
            h += (d / rs) * r
        else:
            if s < 1e-15:
                break
            h += (d / s) * h
    h = np.clip(h, 0.0, 1.0)
    d = target - float(np.sum(h))
    if abs(d) > 1e-12:
        idx = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if idx.size:
            h[idx[0]] += d
            h = np.clip(h, 0.0, 1.0)
            d = target - float(np.sum(h))
            if abs(d) > 1e-14 and idx.size > 1:
                h[idx[1]] += d
                h = np.clip(h, 0.0, 1.0)
    return h


def make_binary(h, n):
    h = np.asarray(h, dtype=float)
    b = np.zeros(int(n), dtype=float)
    k = int(n) // 2
    order = np.argsort(-h)
    b[order[:k]] = 1.0
    if int(n) % 2 and k < int(n):
        b[order[k]] = 0.5
    return b


def interp_to_n(h, n0, n1):
    n0 = int(n0)
    n1 = int(n1)
    if n0 == n1:
        return project(h, n1)
    if n0 <= 1:
        return np.full(n1, 0.5, dtype=float)
    return project(np.interp(grid_x(n1), grid_x(n0), np.asarray(h, dtype=float)), n1)


def soft_weights(q, alpha):
    m = float(np.max(q))
    z = alpha * (q - m)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0:
        w = np.zeros_like(q)
        w[int(np.argmax(q))] = 1.0
        return w
    return e / s


def active_weights(q, eps=1e-7):
    m = float(np.max(q))
    w = np.zeros_like(q, dtype=float)
    mask = q >= (m - max(eps, 1e-12))
    if not np.any(mask):
        w[int(np.argmax(q))] = 1.0
    else:
        w[mask] = 1.0 / float(np.sum(mask))
    return w


def grad_from_weights(h, w, n):
    dx = 2.0 / float(n)
    a = np.convolve(1.0 - h, w[::-1], mode="full")[n - 1:2 * n - 1]
    b = np.convolve(h, w, mode="full")[n - 1:2 * n - 1]
    return dx * (a - b)


def try_move(h, i, j, s, n, c):
    s = float(s)
    if s <= 0.0:
        return False, None, None
    cap = min(1.0 - float(h[i]), float(h[j]))
    if cap <= 1e-13:
        return False, None, None
    amax = cap * (1.0 - 1e-12)
    if s > amax:
        s = amax
    hi = float(h[i])
    hj = float(h[j])
    h[i] = hi + s
    h[j] = hj - s
    q, c2 = qc(h, n)
    if c2 < c - 1e-14:
        return True, q, c2
    h[i] = hi
    h[j] = hj
    return False, None, None


def smax_obj(x, n, alpha, mu):
    q = q_vec(x, n)
    m = float(np.max(q))
    z = alpha * (q - m)
    z = z - float(np.max(z))
    val = m + float(np.log(np.sum(np.exp(z)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - 0.5 * float(n)) ** 2
    return val


def smax_grad(x, n, alpha, mu):
    q = q_vec(x, n)
    m = float(np.max(q))
    z = alpha * (q - m)
    z = z - float(np.max(z))
    e = np.exp(z)
    e /= float(np.sum(e))
    g = grad_from_weights(x, e, n)
    g += 2.0 * mu * (float(np.sum(x)) - 0.5 * float(n))
    return g


def lbfgs_phase(h, n, deadline, alpha=500.0, maxiter=80):
    n = int(n)
    h = project(h, n)
    c0 = c5_of(h, n)
    if time.time() > deadline - 2.0 or n > 800:
        return h, c0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            smax_obj,
            h,
            jac=smax_grad,
            args=(n, float(alpha), 1e-3),
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": int(maxiter), "ftol": 1e-16, "gtol": 1e-14,
                     "maxfun": int(maxiter * 5), "maxls": 30},
        )
        h2 = project(np.asarray(res.x, dtype=float), n)
        c2 = c5_of(h2, n)
        if c2 < c0 - 1e-15:
            return h2, c2
    except Exception:
        pass
    return h, c0


def local_continuous(h, n, deadline, seed=0):
    n = int(n)
    h = project(h, n)
    q, c = qc(h, n)
    best_h = h.copy()
    best_c = c
    if n < 4:
        return best_h, best_c
    rng = np.random.RandomState(int(seed) % (2 ** 31 - 1))
    alphas = (20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0)
    steps = (0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.003)
    K = min(30, max(2, n // 10))
    it = 0
    stall = 0
    ai = 0
    while time.time() < deadline:
        it += 1
        if it % 6 == 0 or stall > 30:
            w = active_weights(q, 1e-7)
        else:
            w = soft_weights(q, alphas[ai % len(alphas)])
        g = grad_from_weights(h, w, n)
        g -= float(np.mean(g))
        order = np.argsort(g)
        lows = order[:K]
        highs = order[-K:]
        improved = False
        for _ in range(50):
            if time.time() > deadline:
                break
            i = int(lows[int(rng.randint(K))])
            j = int(highs[int(rng.randint(K))])
            if i == j:
                continue
            cap = min(1.0 - float(h[i]), float(h[j]))
            if cap <= 1e-13:
                continue
            for f in steps:
                s = f * cap
                if s <= 1e-12:
                    continue
                ok, q2, c2 = try_move(h, i, j, s, n, c)
                if ok:
                    q, c = q2, c2
                    improved = True
                    break
        for _ in range(50):
            if time.time() > deadline:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            if g[i] < g[j]:
                a, b = i, j
            elif g[i] > g[j]:
                a, b = j, i
            else:
                a, b = i, j
            if rng.rand() < 0.1:
                a, b = b, a
            cap = min(1.0 - float(h[a]), float(h[b]))
            if cap <= 1e-13:
                continue
            for f in steps[2:]:
                s = f * cap
                if s <= 1e-12:
                    continue
                ok, q2, c2 = try_move(h, a, b, s, n, c)
                if ok:
                    q, c = q2, c2
                    improved = True
                    break
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
        if improved:
            stall = 0
            if it % 50 == 0:
                ai = (ai + 1) % len(alphas)
        else:
            stall += 1
            ai = (ai + 1) % len(alphas)
            if stall == 40 or stall == 80:
                p = rng.uniform(-0.04, 0.04, n)
                p -= float(np.mean(p))
                h = project(h + p, n)
                q, c = qc(h, n)
            if stall > 120:
                h = best_h.copy()
                q, c = qc(h, n)
                stall = 0
                ai = 0
        if c > best_c + 0.02 and stall > 10:
            h = best_h.copy()
            q, c = qc(h, n)
            stall = 0
    return best_h, best_c


def binary_local(h, n, deadline, seed=0):
    n = int(n)
    h = project(h, n)
    if n % 2 != 0:
        return h, c5_of(h, n)
    b = make_binary(h, n)
    c = c5_of(b, n)
    best = b.copy()
    best_c = c
    if n < 4:
        return best, best_c
    rng = np.random.RandomState(int(seed) % (2 ** 31 - 1))
    perturb = 0
    while time.time() < deadline:
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        rng.shuffle(ones)
        rng.shuffle(zeros)
        improved = False
        cnt = 0
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                cnt += 1
                if (cnt & 255) == 0 and time.time() > deadline:
                    break
                if b[i] > 0.5 or b[j] < 0.5:
                    continue
                b[i] = 1.0
                b[j] = 0.0
                c2 = c5_of(b, n)
                if c2 < c - 1e-14:
                    c = c2
                    improved = True
                else:
                    b[i] = 0.0
                    b[j] = 1.0
            if time.time() > deadline:
                break
        if c < best_c - 1e-14:
            best_c = c
            best = b.copy()
        if not improved:
            if perturb >= 8:
                break
            ones2 = np.where(b > 0.5)[0]
            zeros2 = np.where(b < 0.5)[0]
            msw = max(4, n // 25)
            for _ in range(msw):
                if time.time() > deadline:

                ii = int(rng.choice(ones2))

                if b[ii] > 0.5 and b[jj] < 0.5:
                    b[ii] = 0.0
                    b[jj] = 1.0

def pattern_binary(n, seed):
    n = nearest_even(int(n))
    rng = np.random.RandomState(int(seed) % (2 ** 31 - 1))
    kind = int(rng.randint(3))
    x = grid_x(n)
    if kind == 0:
        idx = rng.choice(n, n // 2, replace=False)
        b = np.zeros(n, dtype=float)
        b[idx] = 1.0
        return b
    if kind == 1:
        p = int(rng.randint(2, 64))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        s = np.sin(2.0 * np.pi * float(p) * x + ph) + 0.25 * rng.normal(0.0, 1.0, n)
    else:
        w = rng.uniform(0.10, 0.45)
        s = np.exp(-((x - 1.0) / w) ** 2) + 0.15 * rng.normal(0.0, 1.0, n)
    order = np.argsort(-s)
    b = np.zeros(n, dtype=float)
    b[order[:n // 2]] = 1.0
    return b
