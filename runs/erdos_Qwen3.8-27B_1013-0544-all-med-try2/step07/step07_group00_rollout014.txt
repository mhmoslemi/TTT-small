def coord_descent_exact(h, n, deadline, n_passes=200):
    dx = 2.0 / n
    c_all = np.correlate(h, 1 - h, mode='full') * dx
    best_c = c_all.max()
    
    m = np.arange(2*n - 1)
    k = m - (n - 1)
    
    for sweep in range(n_passes):
        if time.time() > deadline:
            break
        any_change = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            ik = i + k
            ik_valid = (ik >= 0) & (ik < n)
            ikk = i - k
            ikk_valid = (ikk >= 0) & (ikk < n)
            
            d = np.where(ik_valid, 1 - h[ik], 0.0) - np.where(ikk_valid, h[ikk], 0.0)
            d *= dx
            
            a = c_all  # c_k values (these are the intercepts when t=0)
            # f(t) = max_m (a[m] + (t - h[i]) * d[m])
            # Let s = t - h[i], so f(s) = max_m (a[m] + s * d[m])
            # We want to minimize over s in [-h[i], 1-h[i]]
            
            lo_s = -h[i]
            hi_s = 1.0 - h[i]
            
            # Binary search for the minimum
            for _ in range(50):
                mid = (lo_s + hi_s) / 2.0
                vals = a + mid * d
                f_val = vals.max()
                active = np.isclose(vals, f_val, atol=1e-12)
                d_max_active = d[active].max()
                d_min_active = d[active].min()
                if d_min_active >= -1e-12:
                    # 0 is in subgradient or all active have d>=0, minimum is at or before mid
                    hi_s = mid
                elif d_max_active <= 1e-12:
                    lo_s = mid
                else:
                    # 0 is in the subgradient, we've found the minimum
                    lo_s = hi_s = mid
                    break
            
            t_new = lo_s + h[i]
            t_new = np.clip(t_new, 0.0, 1.0)
            
            if abs(t_new - h[i]) > 1e-10:
                h[i] = t_new
                c_all = np.correlate(h, 1 - h, mode='full') * dx
                new_c = c_all.max()
                if new_c < best_c:
                    best_c = new_c
                any_change = True
        if not any_change:
            break
    return h, best_c

def binary_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 10):
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)

def golden_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.unique(np.floor(np.arange(half) * phi).astype(int) % n)
    h[positions[:half]] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Coordinate descent on initial construction
    t1 = min(t0 + 250, deadline - 300)
    h1, c1 = coord_descent(best_h, best_n, t1, max_passes=8000)
    if c1 < best_c:
        best_c, best_h, best_n = c1, h1.copy(), n_init

    # Phase 2: Try higher resolutions
    for n_try in [600, 800]:
        if time.time() > deadline - 150:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(200, deadline - time.time() - 80)
        if alloc < 40:
            continue
        ht, ct = coord_descent(h_interp, n_try, time.time() + alloc, max_passes=6000)
        if ct < best_c:
            best_c, best_h, best_n = ct, ht.copy(), n_try

    # Phase 3: Diverse restarts with coordinate descent
    starts = [binary_start(best_n, rng), fourier_start(best_n, rng), golden_start(best_n)]
    for s in starts:
        if time.time() > deadline - 60:
            break
        alloc = min(80, deadline - time.time() - 30)
        if alloc < 25:
            break
        hs, cs = coord_descent(s, best_n, time.time() + alloc, max_passes=4000)
        if cs < best_c:
            best_c, best_h = cs, hs.copy()

    # Phase 4: Perturbation of best
    for pert in [0.03, 0.08]:
        if time.time() > deadline - 30:
            break
        h_p = best_h + rng.randn(best_n) * pert
        h_p = project_feasible(h_p, best_n)
        alloc = min(50, deadline - time.time() - 10)
        if alloc < 15:
            break
        hp, cp = coord_descent(h_p, best_n, time.time() + alloc, max_passes=3000)
        if cp < best_c:
            best_c, best_h = cp, hp.copy()

    # Final polish: one more round of coordinate descent on best
    if time.time() < deadline - 3:
        hf, cf = coord_descent(best_h, best_n, deadline - 2, max_passes=2000)
        if cf < best_c:
            best_c, best_h = cf, hf

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
