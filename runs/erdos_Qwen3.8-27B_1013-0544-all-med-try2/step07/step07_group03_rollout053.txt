        # Penalty for sum constraint
        s = x.sum() - n / 2.0
        obj += penalty * s * s
        t1 = np.where(valid1, 1.0 - x[idx1c], 0.0) @ w
        t2 = np.where(valid2, x[idx2c], 0.0) @ w
        grad = dx * (t1 - t2) + 2 * penalty * s * np.ones(n)
        return obj, grad
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_and_grad, h, method='L-BFGS-B', jac=True,
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12})
    return project_feasible(res.x, n)

def greed_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1: delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1: delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def make_grad_context(n):
    j = np.arange(n)[:, None]
    k = np.arange(2 * n - 1)[None, :]
    lag = k - (n - 1)
    idx1 = j - lag
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx1c = np.clip(idx1, 0, n - 1).astype(int)
    idx2 = j + lag
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx2c = np.clip(idx2, 0, n - 1).astype(int)
    return valid1, idx1c, valid2, idx2c

def lbfads_step(h, n, alpha, ctx, maxiter=3000, penalty=1e4):
    dx = 2.0 / n
    valid1, idx1c, valid2, idx2c = ctx
    def obj_and_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        max_c = c.max()
        shifted = alpha * (c - max_c)
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()
        obj = dx * (max_c + np.log(exp_c.sum()) / alpha)
        s = x.sum() - n / 2.0
        obj += penalty * s * s
        A = np.where(valid1, 1.0 - x[idx1c], 0.0)
        B = np.where(valid2, x[idx2c], 0.0)
        grad = dx * ((A @ w) - (B @ w)) + 2.0 * penalty * s
        return obj, grad
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()

    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break

                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0

                    delta[k1] += 1.0

                    delta[k2] += 1.0
