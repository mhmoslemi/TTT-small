```python
def ternary_search_coord(h, j, n, max_iters=40):
    """Find the best value of h[j] to minimize max correlation."""
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    coeff = np.zeros(2*n-1)
    # Compute coeff for changing h[j]
    coeff[n-1] = (1 - 2*h[j]) * dx
    for s in range(1, n):
        c_val = 0.0
        if s <= n - 1 - j:
            c_val += (1 - h[j+s]) * dx
        if s <= j:
            c_val -= h[j-s] * dx
        coeff[s + n - 1] = c_val
        
        c_val2 = 0.0
        if s <= j:
            c_val2 += (1 - h[j-s]) * dx
        if s <= n - 1 - j:
            c_val2 -= h[j+s] * dx
        coeff[n - 1 - s] = c_val2
    
    lo = -h[j]  # h[j] + lo = 0
    hi = 1 - h[j]  # h[j] + hi = 1
    
    def obj(t):
        return np.max(c + t * coeff)
    
    for _ in range(max_iters):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1
    
    best_t = (lo + hi) / 2
    return h[j] + best_t
```

Actually, let me think about this more carefully. For a convex piecewise-linear function, ternary search works but might not be the fastest. A better approach for piecewise-linear convex functions:

The function f(t) = max_k (c[k] + t*coeff[k]). The derivative (where it exists) is coeff[k*] where k* is the argmax. As t increases, the argmax changes at breakpoints.

We can find the minimum by:
1. Compute the current argmax set: S = {k : c[k] + t*coeff[k] = f(t)}
2. If all coeff[k] > 0 for k in S, decrease t
3. If all coeff[k] < 0 for k in S, increase t
4. Otherwise, we're at a minimum

This is essentially a form of subgradient descent, but for 1D piecewise-linear convex functions, we can do better by jumping directly to the next breakpoint.

The next breakpoint in the direction of t is:
t_next = min{(c[k]-c[k*]) / (coeff[k*]-coeff[k]) : k not in S, coeff[k] > coeff[k*]} when increasing t
or
t_next = max{(c[k]-c[k*]) / (coeff[k*]-coeff[k]) : k not in S, coeff[k] < coeff[k*]} when decreasing t

This is O(n) per step, and in practice converges in O(n) steps (since each step eliminates at least one function from consideration). Total: O(n²) per coordinate. For n=300, that's 90000 per coordinate, 27M per sweep. In numpy, this should be fast.

```python
import numpy as np
```

```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

                                'maxfun': maxiter * 30})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def get_coeff(j, h, n):
    dx = 2.0 / n
    coeff = np.zeros(2 * n - 1)
    coeff[n - 1] = (1.0 - 2.0 * h[j]) * dx
    if n <= 1:
        return coeff
    nf = n - 1 - j
    ns = min(j, n - 1)
    cv_pos = np.zeros(n - 1)
    if nf > 0:
        cv_pos[:nf] += (1.0 - h[j + 1:j + nf + 1]) * dx
    if ns > 0:
        cv_pos[:ns] -= h[j - 1:j - ns - 1:-1] * dx
    cv_neg = np.zeros(n - 1)
    if ns > 0:
        cv_neg[:ns] += (1.0 - h[j - 1:j - ns - 1:-1]) * dx
    if nf > 0:
        cv_neg[:nf] -= h[j + 1:j + nf + 1] * dx
    coeff[n:] = cv_pos
    coeff[0:n - 1] = cv_neg[::-1]
    return coeff

            c = np.correlate(h, 1 - h, mode='full') * dx
            cur = float(c.max())
            if cur < best_c:

        order = rng.permutation(n) if sweep % 3 == 2 else range(n)

                m2 = hi - (hi - lo) / 3.0
```
