def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

The idea:
- Start with a binary sequence
- At each step, pick a random (i,j) pair (i is a 0, j is a 1)
- Compute the change in C₅ (the max correlation)
- Accept if it improves, or with probability exp(-Δ/T) if it worsens

- New value at k*: C(k*) + Δ(k*)
- The new max is at most max(C(k*) + Δ(k*), second_max)
- If Δ(k*) < 0, the new max is max(C(k*) + Δ(k*), second_max)
- If Δ(k*) > 0, the new max is C(k*) + Δ(k*)

So we only need to compute Δ(k*) and know the second_max. This makes each evaluation O(n) (to compute Δ at k*) plus O(1) for the acceptance decision.

def compute_delta(h, i, j, n):
    """Compute Δ vector when flipping h[i]:0->1 and h[j]:1->0"""
    m_range = 2 * n - 1
    delta = np.zeros(m_range)
    # For each lag index in [0, m_range-1], lag = idx - (n-1)
    lags = np.arange(-(n-1), n)
    
    # Terms: (1-h[i+lag]) - h[i-lag] - (1-h[j+lag]) + h[j-lag]
    # = h[j+lag] - h[i+lag] - h[i-lag] + h[j-lag]
    
    i_plus = i + lags
    i_minus = i - lags
    j_plus = j + lags
    j_minus = j - lags
    
    valid_i_plus = (i_plus >= 0) & (i_plus < n)
    valid_i_minus = (i_minus >= 0) & (i_minus < n)
    valid_j_plus = (j_plus >= 0) & (j_plus < n)
    valid_j_minus = (j_minus >= 0) & (j_minus < n)
    
    delta[valid_j_plus] += h[j_plus[valid_j_plus]]
    delta[valid_i_plus] -= h[i_plus[valid_i_plus]]
    delta[valid_i_minus] -= h[i_minus[valid_i_minus]]
    delta[valid_j_minus] += h[j_minus[valid_j_minus]]
    
    return delta

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
