from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_descent(h, n, deadline, n_grid=50, max_sweeps=200):
    """Coordinate descent with grid search for each coordinate."""
    best_c = compute_c5(h, n)
    sweep = 0
    while time.time() < deadline and sweep < max_sweeps:
        improved = False
        improvement = 0.0
        for i in range(n):
            old_val = h[i]
            old_c = compute_c5(h, n)
            # Grid search: focus near current value in later sweeps
            if sweep < 5:
                grid = np.linspace(0, 1, n_grid)
            else:
                # Local search around current value
                radius = max(0.05, 0.5 * (1 - sweep / max_sweeps))
                lo = max(0, old_val - radius)
                hi = min(1, old_val + radius)
                grid = np.linspace(lo, hi, n_grid)
            
            best_local = old_val
            best_local_c = old_c
            for v in grid:
                h[i] = v
                c = compute_c5(h, n)
                if c < best_local_c - 1e-15:
                    best_local_c = c
                    best_local = v
            h[i] = best_local
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                improved = True
                improvement += best_c - c
                best_c = c
        sweep += 1
        if not improved:
            break
    # Project to ensure feasibility
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_descent(h, n, deadline, n_grid=80, max_sweeps=500):
    """Coordinate descent with grid search."""
    best_c = compute_c5(h, n)
    sweep = 0
    while time.time() < deadline and sweep < max_sweeps:
        improved = False
        for i in range(n):
            old_val = h[i]
            # Determine search range
            if sweep < 3:
                lo, hi = 0.0, 1.0
                ng = n_grid
            elif sweep < 10:
                radius = 0.3
                lo = max(0.0, old_val - radius)
                hi = min(1.0, old_val + radius)
                ng = n_grid
            else:
                radius = max(0.02, 0.3 * (10.0 / (sweep + 1)))
                lo = max(0.0, old_val - radius)
                hi = min(1.0, old_val + radius)
                ng = max(20, n_grid // 2)
            
            grid = np.linspace(lo, hi, ng)
            best_local = old_val
            best_local_c = compute_c5(h, n)
            
            for v in grid:
                if v == old_val:
                    continue
                h[i] = v
                c = compute_c5(h, n)
                if c < best_local_c - 1e-15:
                    best_local_c = c
                    best_local = v
            
            h[i] = best_local
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                improved = True
                best_c = c
        
        sweep += 1
        if not improved:
            break
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c
