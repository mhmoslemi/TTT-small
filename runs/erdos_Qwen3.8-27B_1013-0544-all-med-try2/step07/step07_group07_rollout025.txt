    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_active(h, n, active_idx, alpha, deadline, maxiter=3000):
    def obj(x):
        c = compute_all_lags(x)
        a = c[active_idx]
        m = a.max()
        s = alpha * (a - m)
        sm = s.max()
        if sm > 500:
            return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
        return m + np.log(np.sum(np.exp(s))) / alpha
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def targeted_search(h, n, deadline):
    """Fast local search focusing on the worst lag pairs"""
    h = h.copy()
    best_c = compute_c5(h)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001]
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(5000):
            if time.time() > deadline:
                break
            c = compute_all_lags(h)
            k_star = np.argmax(c)
            lag = int(k_star - (n - 1))
            if lag == 0:
                break
            m = abs(lag)
            # Find worst pair for this lag
            if lag > 0:
                # f contribution: h[i+m]*(1-h[i])
                # Want to decrease h[i+m] or increase h[i]
                vals = h[m:][:n-m] * (1 - h[:n-m])
                i_worst = np.argmax(vals)
                i, j = i_worst, i_worst + m
                # Move: decrease h[j], increase h[i]
                delta = min(d, h[j] * 0.4, 1.0 - h[i] * 0.999)
            else:
                # f contribution: h[i]*(1-h[i+m])
                vals = h[:n-m] * (1 - h[m:][:n-m])
                i_worst = np.argmax(vals)
                i, j = i_worst, i_worst + m
                # Move: decrease h[i], increase h[j]
                delta = min(d, h[i] * 0.4, 1.0 - h[j] * 0.999)
            if delta < 1e-9:
                break
            if lag > 0:
                h[j] -= delta
                h[i] += delta
            else:
                h[i] -= delta
                h[j] += delta
            h = np.clip(h, 0, 1)
            h = project_feasible(h, n)
            c_new = compute_c5(h)
            if c_new < best_c - 1e-15:
                best_c = c_new
            else:
                # Revert
                if lag > 0:
                    h[j] += delta
                    h[i] -= delta
                else:
                    h[i] += delta
                    h[j] -= delta
                h = project_feasible(h, n)
    return h, best_c
