from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

            if tot < 1e-14:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def _grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

            grad[kk:] += wm * (1.0 - x[:nv])

    val = _obj(x, n, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def _grad_pen(x, n, alpha, mu):
    g = _grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    return x.sum() - n / 2.0

    return np.ones(n)

        res = minimize(_obj_pen, h, jac=_grad_pen, args=(n, alpha, mu),

    except Exception:
        return h
