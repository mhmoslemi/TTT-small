For k < 0: x ∈ [-k, 2], integral = ∫_{-k}^{2} h(x)(1-h(x+k))dx

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:

    """Take a subgradient step on the max correlation."""
    c = compute_c5_vec(h, n)
    km = np.argmax(c)
    dx = 2.0 / n
    k = km - (n - 1)
    grad = np.zeros(n)

    else:
        kk = -k
        nv = n - kk
        grad[kk:] += (1.0 - h[:nv]) * dx
