import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= (target / tot) if tot > 0 else 0
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-13:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_optimize(h, n, deadline, alpha=10000, maxiter=500):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_obj(x, n, alpha) + 0.01*(x.sum()-n/2.0)**2
    def grad(x):
        return smooth_max_grad(x, n, alpha) + 0.02*(x.sum()-n/2.0)*np.ones(n)
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = (1.0 - h).sum()
            if r < 1e-14: break
            h += (d / r) * (1.0 - h)
        else:
            if h.sum() < 1e-14: break
            h *= target / h.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        v = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    else:
        v = m + np.log(np.sum(np.exp(s))) / alpha
    v += mu * (x.sum() - n / 2.0) ** 2
    return float(v)

def smooth_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    g = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-13: continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - x[k:])
            g[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - x[:nv])
            g[:nv] += wm * (-x[kk:])
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    bc = compute_c5(h, n)
    bh = h.copy()

    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,

Let me also think about whether there's a more principled approach. The problem is essentially:
- Find a binary sequence of length n with exactly n/2 ones
- That minimizes the maximum "cross-correlation" at any non-zero lag

This is related to the problem of finding "low autocorrelation" sequences, but with a cross-correlation (between the sequence and its complement).

For a binary sequence with density 1/2, the cross-correlation at lag k is:
C(k) = #{i: s[i]=1, s[i+k]=0} (with appropriate bounds for non-circular)

Note that C(k) + C(-k) = #{i: s[i]=1, s[i+k]=0} + #{i: s[i]=0, s[i+k]=1}
= #{i: s[i] ≠ s[i+k]} (for circular) or bounded version for non-circular.

So C(k) + C(-k) = D(k) where D(k) is the "Hamming distance" at lag k.

For a random sequence, D(k) ≈ n/2 (half the bits differ). So C(k) + C(-k) ≈ n/2, meaning C(k) ≈ n/4 on average.

The goal is to make max_k C(k) as small as possible. Since C(k) + C(-k) = D(k), and D(k) ≥ 0, we have C(k) ≤ D(k). But we want C(k) to be small for all k.

The minimum possible value of max_k C(k) is at least n/4 * (something related to the structure). For a truly optimal sequence, we'd want C(k) to be as uniform as possible across lags.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = (1.0 - h).sum()
            if r < 1e-14:
                break
            h += (d / r) * (1.0 - h)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h *= target / s
    return np.clip(h, 0.0, 1.0)

def _obj(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        v = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    else:
        v = m + np.log(np.sum(np.exp(s))) / alpha
    v += mu * (x.sum() - n * 0.5) ** 2
    return float(v)

def _grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    g = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-13:
            continue
        k = mi - n + 1
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - x[k:])
            g[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - x[:nv])
            g[:nv] += wm * (-x[kk:])
    g += 2.0 * mu * (x.sum() - n * 0.5)
    return g

def lbfgs_opt(h, n, deadline, alpha=10000, mu=0.05, maxiter=200):
    h = project_feasible(h, n)
    bc = compute_c5(h, n)
    bh = h.copy()
    try:
        res = minimize(_obj, h, jac=_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        h2 = project_feasible(res.x, n)
        c2 = compute_c5(h2, n)
        if c2 < bc:
            return h2, c2
    except Exception:
        pass
    return bh, bc

def to_binary(h, n):
    t = n // 2
    idx = np.argsort(-h, kind='stable')
    b = np.zeros(n)
    b[idx[:t]] = 1.0
    return b

def bin_local(h, n, deadline):
    bc = compute_c5(h, n)
    while time.time() < deadline:
        imp = False
        Z = np.where(h == 0)[0].tolist()
        O = np.where(h == 1)[0].tolist()
        for a in Z:
            if time.time() > deadline:
                return h, bc
            if h[a] != 0:
                continue
            for b in O:
                if h[b] != 1:
                    continue
                h[a] = 1.0
                h[b] = 0.0
                c = compute_c5(h, n)
                if c < bc - 1e-15:
                    bc = c
                    imp = True
                    break
                h[a] = 0.0
                h[b] = 1.0
            if imp:
                break
        if not imp:
            break
    return h, bc

def multi_start(h_best, n, deadline, rng, n_starts=15):
    bc = compute_c5(h_best, n)
    for _ in range(n_starts):
        if time.time() > deadline:
            break
        h = h_best.copy()
        n_flips = rng.randint(3, 15)
        for _ in range(n_flips):
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if len(zeros) == 0 or len(ones) == 0:
                break
            a = zeros[rng.randint(len(zeros))]
            b = ones[rng.randint(len(ones))]
            h[a] = 1.0
            h[b] = 0.0
        h, c = bin_local(h, n, min(time.time() + 80, deadline))
        if c < bc:
            bc = c
            h_best = h
    return h_best, bc
