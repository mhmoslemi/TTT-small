        g[:n - k] += -h[k:]
        grad += wk * g
    return grad, m

def sg_descent(h, n, deadline, max_iter=80000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.005
    improvement_count = 0
    last_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 500 == 0:
            c_now = compute_c5(h, n)
            if c_now < best_c - 1e-14:
                best_c = c_now
                best_h = h.copy()
            if it - last_improve > 15000:
                break
        grad, m = subgradient_at(h, n, alpha=300)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            last_improve = it
            step = min(step * 1.02, 0.03)
        elif c_new < m + 1e-6:
            h = h_new
        else:
            h = h_new
            step *= 0.97
            if step < 1e-7:
                step = 0.003
                h = best_h.copy()
    return best_h, best_c

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full')[:n] * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def local_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break
    return h, best_c

def optimize_h(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    if time.time() < deadline - 40:
        h, c = sg_descent(h, n, min(time.time() + 60, deadline - 40))
        if c < best_c:
            best_c = c
    for alpha in [3000, 15000, 60000, 200000]:
        if time.time() > deadline - 25:
            break
        ht = slsqp_opt(h, n, alpha, maxiter=1200)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    if n <= 130:
        if time.time() < deadline - 5:
            h, best_c = pair_search(h, n, deadline - 3,
                [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    else:
        if time.time() < deadline - 5:
            h, best_c = local_search(h, n, deadline - 3,
                [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005], radius=45)
    if time.time() < deadline - 4:
        ht = slsqp_opt(h, n, 500000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    if time.time() < deadline - 2:
        if n <= 130:
            h, best_c = pair_search(h, n, deadline - 1,
                [0.0005, 0.0002, 0.0001, 0.00005])
        else:
            h, best_c = local_search(h, n, deadline - 1,
                [0.0005, 0.0002, 0.0001], radius=35)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = None
    best_c = float('inf')
    # Phase 1: Optimize at initial resolution
    alloc1 = min(280, (deadline - t0) * 0.30)
    h_opt, c_opt = optimize_h(h_init, n_init, t0 + alloc1)
    best_h, best_c = h_opt, c_opt
    # Phase 2: Try coarser resolutions for speed
    for n_try in [100, 150]:
        if time.time() > deadline - 180:
            break
        x_src = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_src, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(90, (deadline - time.time()) * 0.12)
        if alloc < 25:
            break
        h_opt, c_opt = optimize_h(h_interp, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Phase 3: Finer resolution
    n_fine = min(500, n_init * 2)
    if n_fine > n_init and time.time() < deadline - 140:
        x_src = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_fine)
        h_interp = np.interp(x_new, x_src, best_h)
        h_interp = project_feasible(h_interp, n_fine)
        alloc = min(110, (deadline - time.time()) * 0.18)
        if alloc > 35:
            h_opt, c_opt = optimize_h(h_interp, n_fine, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Phase 4: Multi-restart at best resolution
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(3, int(remaining / 35))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            if r % 4 == 0:
                hp = best_h + rng.normal(0, 0.04, n)
            elif r % 4 == 1:
                t_arr = np.linspace(0, 4 * np.pi * (r + 1), n)
                hp = best_h + 0.02 * np.sin(t_arr)
            elif r % 4 == 2:
                idx = rng.choice(n, size=n // 5, replace=False)
                hp = best_h.copy()
                hp[idx] = np.clip(hp[idx] + rng.uniform(-0.08, 0.08, len(idx)), 0, 1)
            else:
                hp = best_h + rng.normal(0, 0.015, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 20) * 0.3)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_h(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Phase 5: Final intensive polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        if n <= 150:
            best_h, best_c = pair_search(best_h, n, deadline - 3,
                [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = local_search(best_h, n, deadline - 3,
                [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001], radius=55)
    # Phase 6: Final SLSQP + micro pair
    if best_h is not None and time.time() < deadline - 4:
        n = len(best_h)
        ht = slsqp_opt(best_h, n, 1000000, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 2:
            if n <= 150:
                best_h, best_c = pair_search(best_h, n, deadline - 1,
                    [0.0003, 0.0001, 0.00005])
            else:
                best_h, best_c = local_search(best_h, n, deadline - 1,
                    [0.0003, 0.0001], radius=30)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
