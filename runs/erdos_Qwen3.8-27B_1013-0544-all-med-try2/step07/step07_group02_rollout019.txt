import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_grad_ascent(h, n, alpha_list, eta0, deadline, max_steps_per_alpha=500):
    """Smooth gradient ascent on the log-sum-exp approximation of max."""
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        eta = eta0
        for step in range(max_steps_per_alpha):
            if time.time() > deadline:
                break
            
            c = np.correlate(h, 1 - h, mode='full')
            
            # Log-sum-exp weights
            shifted = alpha * c
            shifted -= shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            
            # Compute gradient
            grad = np.zeros(n)
            for m in range(2 * n - 1):
                km = m - (n - 1)
                wm = w[m]
                if wm < 1e-14:
                    continue
                if km >= 0:
                    nv = n - km
                    grad[:nv] += wm * (1.0 - h[km:])
                    grad[km:] += wm * (-h[:nv])
                else:
                    kk = -km
                    nv = n - kk
                    grad[kk:] += wm * (1.0 - h[:nv])
                    grad[:nv] += wm * (-h[kk:])
            
            # Update with adaptive step size
            h_new = h + eta * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, n)
            
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            
            # Adaptive step size
            if c_new > best_c + 1e-12:
                eta *= 0.8
            else:
                eta = min(eta * 1.05, 2.0)
            
            h = h_new
            # Decay step size
            eta *= 0.999
    
    return best_h, best_c

def coord_descent(h, n, max_sweeps=200, deadline=None):
    """Coordinate descent for the non-smooth max problem."""
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            
            # Binary search for optimal h[i]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-12
            lo_v = min(val0, val1) - 1e-12
            
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            
            for _ in range(40):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    
    return h, corr

def greedy_binary_short(h_init, n, deadline, max_sweeps=30):
    """Quick binary greedy improvement."""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                if i == j:
                    continue
                # Quick evaluation
                delta_c = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta_c[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta_c[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta_c[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta_c[msk] += h[jmn[msk]]
                delta_c[n - 1] -= 2.0
                delta_c[n - 1 + j - i] += 1.0
                delta_c[n - 1 + i - j] += 1.0
                nc = (corr + delta_c).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                # Apply the swap
                j = best_j
                delta_c = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta_c[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta_c[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta_c[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta_c[msk] += h[(jmn[msk])]
                delta_c[n - 1] -= 2.0
                delta_c[n - 1 + j - i] += 1.0
                delta_c[n - 1 + i - j] += 1.0
                h[i], h[j] = 1.0, 0.0
                corr += delta_c
                cur_c = best_nc
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

                jm = j + m_range

                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + j - i] += 1.0
                delta[n - 1 + i - j] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                j = best_j
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + j - i] += 1.0
                delta[n - 1 + i - j] += 1.0
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: Binary greedy
    p1 = t0 + 0.06 * total
    hb, cb = binary_greedy(h0, n, p1, max_sweeps=30)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 2: Smooth gradient descent with increasing alpha
    alphas = [5, 20, 80, 300, 1000, 4000, 15000, 50000, 150000, 400000]
    p2 = t0 + 0.65 * total
    h = best_h.copy()
    eta_start = 0.15
    for alpha in alphas:
        if time.time() > p2 - 3:
            break
        hh, ch = smooth_grad_descent(h, n, alpha, eta_start, 600, p2 - 2)
        if ch < best_c:
            best_c, best_h = ch, hh.copy()
        h = hh
    # Phase 3: Coordinate descent
    p3 = t0 + 0.80 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=100, deadline=p3)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()
    # Phase 4: More gradient descent at high alpha
    p4 = t0 + 0.92 * total
    h = best_h.copy()
    for alpha in [500000, 1500000]:
        if time.time() > p4 - 3:
            break
        hh, ch = smooth_grad_descent(h, n, alpha, 0.08, 400, p4 - 2)
        if ch < best_c:
            best_c, best_h = ch, hh.copy()
        h = hh
    # Phase 5: Final coord descent
    if time.time() < dl - 3:
        hc, _ = coord_descent(best_h, n, max_sweeps=60, deadline=dl - 1)
        hc = project_feasible(hc, n)
        cc = compute_c5(hc, n)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()
    # Phase 6: Try lower resolution
    for nh in [200, 300]:
        if time.time() > dl - 4:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        hh, ch = smooth_grad_descent(hh, nh, 5000, 0.15, 300, dl - 2)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
