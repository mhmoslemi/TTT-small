import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def _slsqp_constraint(x, n):
    return float(x.sum() - n / 2.0)

def _slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, max_iters=150):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (100, 5.0), (200, 10.0), (400, 10.0),
        (1000, 50.0), (2000, 100.0), (5000, 500.0),
        (10000, 2000.0), (30000, 5000.0)
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxfun': max_iters * 10,
                                   'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=1e6, maxiter=1000):

    cons = [{'type': 'eq', 'fun': _slsqp_constraint, 'jac': _slsqp_jac, 'args': (n,)}]

def pair_search(h, n, deadline, deltas, radius=20):

            h_trial = np.interp(x_target, x_base, base_h)
            noise = rng.normal(0, 0.05, target_n)
            h_trial = h_trial + noise
        h_trial = project_feasible(h_trial, target_n)
        
        t_phase_end = deadline - 20
        t_lbfgs = min(t0 + 80, t_phase_end - 100)
        h_t, c_t = lbfgs_optimize(h_trial, target_n, t_lbfgs, max_iters=100)
        
        if time.time() < t_phase_end - 50:
            h_t, c_t = slsqp_optimize(h_t, target_n, min(time.time() + 40, t_phase_end - 30), alpha=5e5, maxiter=500)
        
        if time.time() < t_phase_end - 10:
            h_t, c_t = pair_search(h_t, target_n, t_phase_end, [0.001, 0.0005], radius=8)
        
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Optimize initial grid (300) for ~400s
    t1_end = min(t0 + 400, deadline - 500)
    if time.time() < t1_end:
        best_h, best_c = lbfgs_optimize(best_h, n_init, min(time.time() + 120, t1_end - 250), max_iters=150)
    
    if time.time() < t1_end - 150:
        best_h, best_c = slsqp_optimize(best_h, n_init, min(time.time() + 60, t1_end - 100), alpha=5e5, maxiter=800)
    
    if time.time() < t1_end - 100:
        best_h, best_c = gradient_descent(best_h, n_init, min(time.time() + 40, t1_end - 60), lr_init=0.008, alpha_init=150.0)
    
    if time.time() < t1_end - 60:
        best_h, best_c = pair_search(best_h, n_init, min(time.time() + 80, t1_end), [0.005, 0.002, 0.001, 0.0005], radius=30)
    
    # Phase 2: Upscale to 600 and optimize
    n_up = 600
    if time.time() < deadline - 450:
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_up)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n_up)
        c_up = compute_c5(h_up, n_up)
        
        if c_up < best_c or True:  # Always try 600, might find better
            h_up, c_up = multistart_600(h_up, n_up, n_up, deadline, seed)
            if c_up < best_c:
                best_c = c_up
                best_h = h_up
                n_init = n_up
    
    # Final polish on best solution
    if time.time() < deadline - 10:
        best_h, best_c = lbfgs_optimize(best_h, n_init, deadline - 5, max_iters=50)
    
    best_h = project_feasible(best_h, n_init)
    best_c = compute_c5(best_h, n_init)
    return best_h, best_c, n_init
