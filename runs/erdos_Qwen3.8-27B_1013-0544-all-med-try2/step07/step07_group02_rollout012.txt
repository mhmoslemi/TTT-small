    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

        else:
            s = h.sum()
            if s < 1e-14:
                break

        # Term 1: (1-h[i-shift]) valid for max(shift,0) <= i <= min(n-1, n-1+shift)
        lo1 = max(0, shift)
        hi1 = min(n - 1, n - 1 + shift)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += p * dx * (1.0 - h[idx - shift])
        # Term 2: -h[i+shift] valid for max(0,-shift) <= i <= min(n-1, n-1-shift)
        lo2 = max(0, -shift)
        hi2 = min(n - 1, n - 1 - shift)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] -= p * dx * h[idx + shift]
    
    return grad

def lbfgs_optimize(h, n, alpha, maxiter=5000, deadline=None):
    """Optimize using L-BFGS-B with smoothed max objective."""
    dx = 2.0 / n
    t = n / 2.0
    
    def obj(v):
        if deadline and time.time() > deadline:
            return np.inf
        g = np.correlate(v, 1.0 - v, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        pen = 100.0 * (v.sum() - t) ** 2
        return m + logsum / alpha + pen
    
    def grad(v):
        if deadline and time.time() > deadline:
            return np.zeros(n)
        gv = compute_grad_c5_smoothed(v, n, alpha)
        gv += 200.0 * (v.sum() - t)
        return gv
    
    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-14,
                            'gtol': 1e-8, 'maxls': 50})
    if deadline and time.time() > deadline:
        return h
    return project_feasible(np.clip(res.x, 0, 1), n)
