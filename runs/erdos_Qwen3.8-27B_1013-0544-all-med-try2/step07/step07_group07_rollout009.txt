    elif lag > 0:

            g[m:] -= h[:n - m]
    else:
        m = -lag
        if m < n:
            g[m:] = 1.0 - h[:n - m]
            g[:n - m] -= h[m:]
    return g

        if wi < 1e-14:

        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_grad(h, n, lag):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < lag < n:
        g[:n - lag] = 1.0 - h[lag:]
        g[lag:] -= h[:n - lag]
    elif -n < lag < 0:
        m = -lag
        g[m:] = 1.0 - h[:n - m]
        g[:n - m] -= h[m:]
    return g

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    exp_v = np.exp(shifted)
    s = exp_v.sum()
    obj = m + np.log(s) / alpha
    w = exp_v / s
    active = np.where(w > 1e-10)[0]
    grad = np.zeros(n)
    for k in active:
        grad += w[k] * dx * lag_grad(h, n, k - (n - 1))
    return obj, grad

def projected_grad_descent(h, n, alpha, max_iter, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    base_step = 0.1
    stuck = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stuck > 50:
            h = project_feasible(h + np.random.randn(n) * 0.01, n)
            base_step = 0.1
            stuck = 0
            continue
        _, grad = smooth_obj_grad(h, n, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = base_step / gnorm
        improved = False
        for _ in range(25):
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                improved = True
                base_step = min(base_step * 1.3, 1.0)
                break
            step *= 0.5
        if not improved:
            stuck += 1
            base_step *= 0.6
            if base_step < 1e-9:
                break
    return best_h, best_c

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:

                break

            k_star = int(np.argmax(c))

            m = abs(lag)
            found = False
            for i in range(n - m):
                if time.time() > deadline:
                    return h, best_c
                j = i + m

                    best_c = c_new
                    found = True
                    break
                h[i] -= delta
                h[j] += delta
            if not found:
                break
    return h, best_c

def _s_obj(x, n, alpha):
    v, _ = smooth_obj_grad(x, n, alpha)
    return v

def _s_grad(x, n, alpha):
    _, g = smooth_obj_grad(x, n, alpha)
    return g

def _s_sum(x, n):
    return x.sum() - n / 2.0

def _s_sum_jac(x, n):
    return np.ones(len(x))

def slsqp_refine(h, n, alpha, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _s_sum, 'jac': _s_sum_jac, 'args': (n,)}]
    try:
        res = minimize(_s_obj, h, args=(n, alpha), jac=_s_grad,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 2000, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < compute_c5(h, n):
            return h_new
    except Exception:
        pass
    return h.copy()
