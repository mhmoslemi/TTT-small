```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Compute gradient of c_k w.r.t. h for a given lag."""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def softmax_smooth_obj(h, n, alpha):
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    # numerically stable log-sum-exp
    mx = shifted.max()
    return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

def softmax_smooth_grad(h, n, alpha):
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx) / np.sum(np.exp(shifted - mx))
    # grad = sum_k weights[k] * grad(c_k)
    grad = np.zeros(n)
    for k in range(len(c)):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        grad += weights[k] * lag_gradient(h, n, lag)
    return grad

def smooth_gradient_descent(h, n, deadline, alphas=None, max_iters=50000):
    if alphas is None:
        alphas = [200, 500, 1000, 2000, 5000]
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        
        # For this alpha, do gradient descent with momentum
        velocity = np.zeros(n)
        mu = 0.9  # momentum
        eta = 2.0 / (alpha * n * 0.02)  # initial step size (rough Lipschitz estimate)
        
        for it in range(max_iters):
            if time.time() > deadline:
                break
            
            grad = softmax_smooth_grad(h_curr, n, alpha)
            
            # Momentum update
            velocity = mu * velocity - eta * grad
            h_new = h_curr + velocity
            h_new = project_feasible(h_new, n)
            
            # Check improvement
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            
            # Simple convergence check
            if np.max(np.abs(h_new - h_curr)) < 1e-10:
                break
            
            h_curr = h_new
    
    return best_h, best_c
```

```python
    best_h = h_start.copy()
    best_c = compute_c5(h_start, n)
    h_curr = h_start.copy()
    
    # Store cuts: (c_k value, gradient, k)
    cuts = []
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = compute_all_lags(h_curr, n)
        c_max = c.max()
        if c_max < best_c:
            best_c = c_max
            best_h = h_curr.copy()
        
        # Add top cuts
        top_k = np.argsort(c)[-5:]
        new_cuts = []
        for k_idx in top_k:
            lag = k_idx - (n - 1)
            g = lag_gradient(h_curr, n, lag)
            new_cuts.append((c[k_idx], g, k_idx))
        
        # Keep only the most relevant cuts (avoid too many)
        cuts = cuts[-20:] + new_cuts
        
        # Solve LP: min t s.t. c_k(h) <= t (linearized), sum(h) = n/2, 0<=h<=1
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n/2.0, h_var >= 0, h_var <= 1, t_var >= 0]

            except:

        h_new = project_feasible(h_new, n)

        h_curr = h_new
    
    return best_h, best_c
```

```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def soft_gradient_descent(h, n, deadline, alpha=1000, max_iters=20000):
    """Smoothed gradient descent with adaptive step size."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    vel = np.zeros(n)
    mu = 0.85
    eta = 0.5  # initial step size
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = compute_all_lags(h_curr, n)
        c_max = c.max()
        if c_max < best_c:
            best_c = c_max
            best_h = h_curr.copy()
        
        # Softmax weights (top-k for efficiency)
        shifted = alpha * (c - c_max)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w /= w_sum
        
        # Only use top-weighted lags
        threshold = 1e-6
        active = np.where(w > threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for idx in active:
            lag = idx - (n - 1)
            grad += w[idx] * lag_gradient(h_curr, n, lag)
        
        # Momentum update
        vel = mu * vel - eta * grad
        h_new = h_curr + vel
        h_new = project_feasible(h_new, n)
        
        # Adaptive step size
        c_new = compute_c5(h_new, n)
        if c_new > c_max + 0.01:
            eta *= 0.5
            h_new = h_curr  # revert
        elif it % 50 == 0 and eta < 2.0:
            eta *= 1.1  # slowly increase
        
        h_curr = h_new
    
    return best_h, best_c
```

The problem is:

Actually, let me think about this differently. The inner minimization:

This is a non-convex quadratic optimization problem (since Q is not necessarily negative semidefinite). But it's a well-studied problem.

For the outer problem (max over λ), we need to optimize over the simplex {λ ≥ 0, Σλ = 1}.

This is a min-max problem that's hard in general. But we can use a **vice-optimization** approach:

2. Solve the inner QP
3. Update λ: put more weight on the lags with highest c_k
4. Repeat

This is essentially a **column generation** or **cutting plane** approach in disguise.

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def q_times_h(h, lam, n):
    """Compute Q @ h where Q = sum_k lam[k] * A_k."""
    result = np.zeros(n)
    n_lags = len(lam)
    for k in range(n_lags):
        w = lam[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        m = abs(lag)
        if m >= n:
            continue
        if lag == 0:
            result += w * h
        elif lag > 0:
            result[:n-m] += w * h[m:]
        else:
            result[m:] += w * h[:n-m]
    return result

def vice_inner(h, n, lam, inner_iters=200):
    """Solve inner QP: max h^T Q h s.t. 0<=h<=1, sum(h)=n/2."""
    eta = 0.25
    for it in range(inner_iters):
        qh = q_times_h(h, lam, n)
        h_new = h + eta * 2.0 * qh
        h_new = project_feasible(h_new, n)
        if np.max(np.abs(h_new - h)) < 1e-10:
            break
        h = h_new
    return h

def vice_optimize(h, n, deadline, max_outer=30):
    """Vice-optimization algorithm."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    
    n_lags = 2 * n - 1
    lam = np.ones(n_lags) / n_lags
    
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        
        # Inner: maximize h^T Q h
        h_curr = vice_inner(h_curr, n, lam, inner_iters=100)
        
        # Evaluate
        c = compute_all_lags(h_curr, n)
        c_max = c.max()
        if c_max < best_c:
            best_c = c_max
            best_h = h_curr.copy()
        
        # Update weights (focusing on worst lags)
        temp = 1.0 + 2.0 * outer
        lam = np.exp(temp * c)
        lam = lam / lam.sum()
    
    return best_h, best_c

def smooth_gd(h, n, deadline, alpha_list):
    """Smoothed gradient descent with momentum."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        vel = np.zeros(n)
        mu = 0.9
        eta = 0.5
        stall = 0
        
        for it in range(20000):
            if time.time() > deadline:
                break
            
            c = compute_all_lags(h_curr, n)
            c_max = c.max()
            if c_max < best_c - 1e-10:
                best_c = c_max
                best_h = h_curr.copy()
                stall = 0
            else:
                stall += 1
            if stall > 3000:
                break
            
            shifted = alpha * (c - c_max)
            w = np.exp(shifted - shifted.max())
            ws = w.sum()
            if ws < 1e-300:
                break
            w /= ws
            
            active = np.where(w > 1e-8)[0]
            if len(active) == 0:
                active = np.array([np.argmax(c)])
            
            grad = np.zeros(n)
            for idx in active:
                lag = idx - (n - 1)
                grad += w[idx] * lag_gradient(h_curr, n, lag)
            
            vel = mu * vel - eta * grad
            h_new = h_curr + vel
            h_new = project_feasible(h_new, n)
            
            c_new = compute_c5(h_new, n)
            if c_new > c_max + 0.005:

                h_new = h_curr
                vel[:] = 0
            elif eta < 3.0:
                eta *= 1.01
            
            h_curr = h_new
    
    return best_h, best_c
```
