_G_ALPHA = 0.0
_G_TARGET = 0.0

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * (2.0 / n))

def project_feasible(v, t):
    v = np.asarray(v, dtype=float)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(50):
        m = 0.5 * (lo + hi)
        s = np.clip(v - m, 0.0, 1.0).sum()
        if s > t:
            lo = m
        else:
            hi = m
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        d = t - x.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - x
            s = r.sum()
            if s > 1e-14:
                x = x + d / s * r
        else:
            s = x.sum()
            if s > 1e-14:
                x = x + d / s * x
    return np.clip(x, 0.0, 1.0)

def to_binary(h, n):
    o = np.argsort(h)[::-1]
    y = np.zeros(int(n), dtype=float)
    m = int(n // 2)
    if m > 0:
        y[o[:m]] = 1.0
    return y

def resample_h(h, n0, n1):
    if n0 == n1:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, int(n0))
    x1 = np.linspace(0.0, 2.0, int(n1))
    return np.interp(x1, x0, h)

def make_fallback(n):
    t = np.arange(int(n), dtype=float) / float(n)
    return to_binary((np.mod(t * 0.7548776662466927, 1.0) < 0.5).astype(float), n)

def smooth_obj_grad(x, alpha):
    x = np.asarray(x, dtype=float)
    n = x.size
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, 'full') * dx
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    if s > 0.0:
        w = e / s
        f = m + (zm + math.log(s)) / alpha
    else:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        f = m
    first = np.convolve(1.0 - x, w, 'full')[n - 1:2 * n - 1]
    second = np.convolve(x[::-1], w, 'full')[2 * n - 2:3 * n - 2]
    return f, dx * (first - second)

def _f(x):
    return smooth_obj_grad(x, _G_ALPHA)[0]

def _j(x):
    return smooth_obj_grad(x, _G_ALPHA)[1]

def _ef(x):
    return x.sum() - _G_TARGET

def _ej(x):
    return np.ones(x.shape)

def slsqp_pass(x, n, alpha, maxiter, ftol):
    global _G_ALPHA, _G_TARGET
    _G_ALPHA = float(alpha)
    _G_TARGET = int(n) / 2.0
    x = project_feasible(x, int(n) / 2.0)
    bounds = [(0.0, 1.0)] * int(n)
    cons = [{'type': 'eq', 'fun': _ef, 'jac': _ej}]
    try:
        r = minimize(_f, x, method='SLSQP', jac=_j, bounds=bounds,
                     constraints=cons, options={'maxiter': int(maxiter), 'ftol': ftol, 'disp': False})
        y = r.x if r.x is not None else x
    except Exception:
        y = x
    return project_feasible(np.asarray(y, dtype=float), int(n) / 2.0)

def deriv(h, lags, i):
    n = h.size
    b = np.zeros(2 * n - 1, dtype=float)
    k = i - lags
    m = (k >= 0) & (k < n)
    b[m] += 1.0 - h[k[m]]
    k = i + lags
    m = (k >= 0) & (k < n)
    b[m] -= h[k[m]]
    return b

def quad_min(c, a, lo, hi):
    d = -0.5 * a
    if d < lo:
        d = lo
    elif d > hi:
        d = hi
    return c + a * d + d * d

def exact_delta_max(c, D, d, p1, p2, z):

    q = c[p1] + D[p1] * d + d * d
    if q > v:
        v = q
    q = c[p2] + D[p2] * d + d * d
    if q > v:
        v = q
    q = c[z] + D[z] * d - 2.0 * d * d
    if q > v:
        v = q
    return v

        if disc < 0.0:

        if r2 < dh:

    r1 = (a0 - s) * 0.25
    r2 = (a0 + s) * 0.25
    if r1 + 1e-12 >= dl or r2 - 1e-12 <= dh:
        return True, dl, dh
    return False, dl, dh

def pair_ref_descent(h, n, ms, deadline, ref):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    if n < 2:
        return h, compute_c5(h, n)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, 'full')
    cur = float(corr.max())
    bh = h.copy()
    bc = cur
    lags = np.arange(-(n - 1), n)
    z = n - 1
    for sw in range(int(ms)):
        if time.time() > deadline:
            break
        imp = False
        r = ref if sw == 0 else (ref + sw * 97) % n
        if sw % 3 == 1:
            r = int(np.argmin(np.abs(h - 0.5)))
        for i in range(n):
            idx = (i + sw * 13) % n
            if idx == r:
                continue
            if (i & 31) == 0 and time.time() > deadline:
                break
            oi = h[idx]
            orr = h[r]
            lo = max(-oi, orr - 1.0)
            hi = min(1.0 - oi, orr)
            if hi - lo <= 1e-13:
                continue
            B1 = deriv(h, lags, idx)
            B2 = deriv(h, lags, r)
            D = B1 - B2
            p1 = n - 1 + (r - idx)
            p2 = n - 1 - (r - idx)
            pos = D > 1e-14
            neg = D < -1e-14
            pos[z] = False
            neg[z] = False
            zero = ~(pos | neg)
            zero[z] = False
            zmax = float(corr[zero].max()) if zero.any() else -np.inf
            lin = corr.copy()
            lin[z] = -np.inf
            if pos.any():
                lin[pos] = corr[pos] + D[pos] * lo
            if neg.any():
                lin[neg] = corr[neg] + D[neg] * hi
            tlo = float(lin.max())
            tlo = max(tlo, quad_min(corr[p1], D[p1], lo, hi), quad_min(corr[p2], D[p2], lo, hi))
            cl = corr[z] + D[z] * lo - 2.0 * lo * lo
            ch = corr[z] + D[z] * hi - 2.0 * hi * hi
            tlo = max(tlo, cl if cl < ch else ch)
            if tlo >= cur - 1e-11:
                continue
            vlo = exact_delta_max(corr, D, lo, p1, p2, z)
            vhi = exact_delta_max(corr, D, hi, p1, p2, z)
            ds = D[z] * 0.25
            if ds < lo:
                ds = lo
            elif ds > hi:
                ds = hi
            vz = corr[z] + D[z] * ds - 2.0 * ds * ds
            hiT = max(vlo, vhi, vz)
            loT = tlo
            if hiT < loT:
                hiT = loT
            for _ in range(34):
                mid = 0.5 * (loT + hiT)
                ok, _, _ = pair_feasible(mid, corr, D, lo, hi, pos, neg, zmax, p1, p2, z)
                if ok:
                    hiT = mid
                else:
                    loT = mid
            ok, fl, fx = pair_feasible(hiT, corr, D, lo, hi, pos, neg, zmax, p1, p2, z)
            cands = [lo, hi, 0.5 * (lo + hi), ds]
            if ok:
                cands = cands + [fl, fx, 0.5 * (fl + fx)]
            disc = D[z] * D[z] - 8.0 * (hiT - corr[z])
            if disc > 0.0:
                s = math.sqrt(disc)
                rr1 = (D[z] - s) * 0.25
                rr2 = (D[z] + s) * 0.25
                if lo <= rr1 <= hi:
                    cands.append(rr1)
                if lo <= rr2 <= hi:
                    cands.append(rr2)
            for pp in (p1, p2):
                a = D[pp]
                disc = a * a - 4.0 * (corr[pp] - hiT)
                if disc > 0.0:
                    s = math.sqrt(disc)
                    rr1 = (-a - s) * 0.5
                    rr2 = (-a + s) * 0.5
                    if lo <= rr1 <= hi:
                        cands.append(rr1)
                    if lo <= rr2 <= hi:
                        cands.append(rr2)
            bd = lo
            bv = hiT
            for x in cands:
                x = float(np.clip(x, lo, hi))
                val = exact_delta_max(corr, D, x, p1, p2, z)
                if val < bv - 1e-14:
                    bv = val
                    bd = x
            if bv < cur - 1e-12:
                corr += D * bd
                corr[p1] += bd * bd
                corr[p2] += bd * bd
                corr[z] -= 2.0 * bd * bd
                h[idx] = oi + bd
                h[r] = orr - bd
                cur = float(corr.max())
                if cur < bc - 1e-14:
                    bc = cur
                    bh = h.copy()
                    imp = True
        if sw % 25 == 24 and abs(h.sum() - target) > 1e-8:
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, 'full')
            cur = float(corr.max())
            if cur < bc:
                bc = cur
                bh = h.copy()
        if not imp:
            break
    h = project_feasible(h, target)
    bh = project_feasible(bh, target)
    rh = float(np.max(np.correlate(h, 1.0 - h, 'full')))
    rb = float(np.max(np.correlate(bh, 1.0 - bh, 'full')))
    if rh < rb:
        bh, rb = h, rh
    return bh, rb * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(30.0, float(budget_s) - 15.0)
    rng = np.random.RandomState(int(seed))
    best_h = None
    try:
        try:
            h0 = np.asarray(initial_h_values, dtype=float).copy()
            n = int(h0.size)
            if n < 2:
                raise ValueError
        except Exception:
            n = 400
            h0 = make_fallback(n)
        if n > 900:
            h0 = resample_h(h0, n, 400)
            n = 400
        h0 = project_feasible(h0, n / 2.0)
        best_n = n
        best_h = h0.copy()
        best_c = compute_c5(best_h, best_n)
        refs = [int(np.argmin(np.abs(best_h - 0.5))), best_n // 2, 0, best_n - 1]

        p = t0 + 0.12 * total
        for r in refs:
            if time.time() > p - 10:
                break
            yh, ch = pair_ref_descent(best_h, best_n, 50, min(p, time.time() + 0.04 * total), r)
            if ch < best_c:
                best_h, best_c = yh, ch

        p = t0 + 0.45 * total
        for a in (100.0, 1000.0, 10000.0, 100000.0, 1000000.0):
            if time.time() > p - 30:
                break
            mi = int(max(20, min(80, 35 * 400.0 / max(best_n, 1))))
            if a > 10000.0:
                mi = int(mi * 1.5)
            y = slsqp_pass(best_h, best_n, a, mi, 1e-12)
            c = compute_c5(y, best_n)
            if c < best_c:
                best_h, best_c = y, c
            if a >= 10000.0 and time.time() < p - 10:
                yh, ch = pair_ref_descent(best_h, best_n, 8, min(p, time.time() + 12.0), refs[0])
                if ch < best_c:
                    best_h, best_c = yh, ch

        p = t0 + 0.65 * total
        for r in refs:
            if time.time() > p - 15:
                break
            yh, ch = pair_ref_descent(best_h, best_n, 70, min(p, time.time() + 0.05 * total), r)
            if ch < best_c:
                best_h, best_c = yh, ch

        p = t0 + 0.86 * total
        for nh in (300, 320, 360, 400, 440, 480, 560, 640):
            if nh < 80 or nh > 900 or abs(nh - best_n) < 8:
                continue
            if time.time() > p - 45:
                break
            hh = project_feasible(resample_h(best_h, best_n, nh), nh / 2.0)
            hh, ch = pair_ref_descent(hh, nh, 25, min(p, time.time() + 20.0), int(np.argmin(np.abs(hh - 0.5))))
            if ch < best_c:
                best_h, best_c, best_n = hh, ch, nh
            for a in (5000.0, 50000.0, 500000.0):
                if time.time() > p - 15:
                    break
                mi = int(max(20, min(50, 25 * 400.0 / nh)))
                y = slsqp_pass(hh, nh, a, mi, 1e-11)
                c = compute_c5(y, nh)
                hh = y
                if c < best_c:
                    best_h, best_c, best_n = y, c, nh
                    hh = y.copy()
                hh, ch = pair_ref_descent(hh, nh, 5, min(p, time.time() + 8.0), int(np.argmin(np.abs(hh - 0.5))))
                if ch < best_c:
                    best_h, best_c, best_n = hh, ch, nh

        end = t0 + float(budget_s) - 10.0
        amps = (0.01, 0.03, 0.08, 0.15)
        for k in range(6):
            if time.time() > end - 25.0:
                break
            amp = amps[k % 4]
            h = best_h + amp * rng.randn(best_n)
            if amp > 0.05:
                h = to_binary(h, best_n)
            h = project_feasible(h, best_n / 2.0)
            for a in (2000.0, 20000.0, 200000.0):
                if time.time() > end - 15.0:
                    break
                mi = int(max(15, min(40, 20 * 400.0 / max(best_n, 1))))
                y = slsqp_pass(h, best_n, a, mi, 1e-10)
                c = compute_c5(y, best_n)
                h = y
                if c < best_c:
                    best_h, best_c = y, c
                yh, ch = pair_ref_descent(h, best_n, 4, min(end - 5.0, time.time() + 6.0),
                                          int(np.argmin(np.abs(h - 0.5))))
                if ch < best_c:
                    best_h, best_c = yh, ch

        best_h = project_feasible(best_h, best_n / 2.0)
        best_c = compute_c5(best_h, best_n)
        return best_h, float(best_c), int(best_n)
    except Exception:
        try:
            hb = project_feasible(np.asarray(initial_h_values, dtype=float), 200.0)
            return hb, compute_c5(hb, len(hb)), len(hb)
        except Exception:
            hb = make_fallback(400)
            return hb, compute_c5(hb, 400), 400
