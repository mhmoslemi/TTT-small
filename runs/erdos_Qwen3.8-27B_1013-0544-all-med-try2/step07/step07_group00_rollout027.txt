```python
import numpy as np
import time
from scipy.optimize import minimize, linprog


def corr_data(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))


def compute_c(h, n):
    return float(np.max(corr_data(h, n)))


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float, copy=True), 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(250):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14:
                break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)


def normalize_exact(h, n):
    target = 0.5 * float(n)
    h = np.array(h, dtype=float, copy=True)
    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        idx = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(idx) == 0:
            break
        h[idx] += d / float(len(idx))
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_value(x, n, alpha, mu):
    c = corr_data(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    val = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    r = x.sum() - 0.5 * float(n)
    if mu:
        val += mu * r * r
    return float(val)


def smooth_grad(x, n, alpha, mu):
    c = corr_data(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    mid = n - 1
    for mm in range(len(c)):
        wt = w[mm]
        if wt <= 1e-15:
            continue
        k = mm - mid
        ws = wt * dx
        if k >= 0:
            nv = n - k
            g[:nv] += ws * (1.0 - x[k:])
            g[k:] += ws * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += ws * (1.0 - x[:nv])
            g[:nv] += ws * (-x[kk:])
    if mu:
        g += 2.0 * mu * (x.sum() - 0.5 * float(n))
    return g


def lbfgs_polish(h, n, deadline):
    h = project_feasible(h, n)
    best = compute_c(h, n)
    bounds = [(0.0, 1.0) for _ in range(n)]
    maxiter = 140 if n <= 520 else 80
    configs = ((50.0, 0.0), (200.0, 0.05), (1000.0, 0.5), (5000.0, 5.0))
    if n > 560:
        configs = configs[:2]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_value, h, jac=smooth_grad, args=(n, alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-10, 'maxfun': maxiter * 10,
                                    'maxls': 50})
            if getattr(res, 'x', None) is not None:
                h2 = project_feasible(res.x, n)
                c2 = compute_c(h2, n)
                if c2 < best - 1e-15:
                    best = c2
                    h = h2
        except Exception:
            pass
    return h, best


def build_gradient_matrix(h, n):
    G = np.zeros((2 * n - 1, n), dtype=float)
    mid = n - 1
    for k in range(1, n):
        nv = n - k
        pos = mid + k
        neg = mid - k
        G[pos, :nv] += 1.0 - h[k:]
        G[pos, k:] += -h[:nv]
        G[neg, k:] += 1.0 - h[:nv]
        G[neg, :nv] += -h[k:]
    G[mid, :] = 1.0 - 2.0 * h
    G *= 2.0 / float(n)
    return G


def linear_polish(h, n, deadline, max_iters=25):
    h = project_feasible(h, n)
    best = compute_c(h, n)
    step = 0.08
    for _ in range(int(max_iters)):
        if time.time() > deadline - 2:
            break
        f = corr_data(h, n)
        G = build_gradient_matrix(h, n)
        lo = -np.minimum(h, step)
        hi = np.minimum(1.0 - h, step)
        A_ub = np.hstack((G, -np.ones((G.shape[0], 1))))
        b_ub = -f
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        obj = np.zeros(n + 1)
        obj[n] = 1.0
        bounds = list(zip(lo, hi)) + [(-2.0, 2.0)]
        try:
            res = linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
        except Exception:
            break
        if not res.success:
            break
        d = res.x[:n]
        if not np.all(np.isfinite(d)):
            break
        d = np.clip(d, lo, hi)
        s = d.sum()
        if abs(s) > 1e-9:
            d -= s / float(n)
            d = np.clip(d, lo, hi)
        ba = 0.0
        bc = best
        for a in (1.0, 0.7, 0.5, 0.2, 0.05, 0.01):
            hn = np.clip(h + a * d, 0.0, 1.0)
            if abs(hn.sum() - 0.5 * float(n)) > 1e-8:
                hn = project_feasible(hn, n)
            c = compute_c(hn, n)
            if c < bc - 1e-15:
                bc = c
                ba = a
        if ba > 0.0:
            h = np.clip(h + ba * d, 0.0, 1.0)
            h = project_feasible(h, n)
            best = compute_c(h, n)
        else:
            step *= 0.5
            if step < 1e-5:
                break
    return h, best


def project_disp(v, h, step_max):
    lo = -np.minimum(h, step_max)
    hi = np.minimum(1.0 - h, step_max)
    vmin = float(v.min())
    vmax = float(v.max())
    a = vmin - step_max - 1.0
    b = vmax + step_max + 1.0
    for _ in range(70):
        lam = 0.5 * (a + b)
        d = np.clip(v - lam, lo, hi)
        if d.sum() > 0.0:
            a = lam
        else:
            b = lam
    d = np.clip(v - 0.5 * (a + b), lo, hi)
    s = d.sum()
    if abs(s) > 1e-12:
        d -= s / float(len(d))
        d = np.clip(d, lo, hi)
    return d


def projected_gradient_polish(h, n, deadline):
    h = project_feasible(h, n)
    best = compute_c(h, n)
    steps = (0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0007, 0.0003, 0.0001)
    betas = (20.0, 100.0, 500.0, 2000.0, 10000.0)
    alphas = (1.0, 0.7, 0.5, 0.3, 0.1, 0.05)
    for step in steps:
        if time.time() > deadline:
            break
        for beta in betas:
            if time.time() > deadline:
                break
            rounds = 0
            while time.time() < deadline and rounds < 30:
                r = smooth_grad(h, n, beta, 0.0)
                v = -r
                v -= v.mean()
                nm = float(np.max(np.abs(v)))
                if nm < 1e-16:
                    break
                v /= nm
                d = project_disp(v, h, step)
                if float(np.max(np.abs(d))) < 1e-15:
                    break
                ba = 0.0
                bc = best
                for a in alphas:
                    hn = np.clip(h + a * d, 0.0, 1.0)
                    if abs(hn.sum() - 0.5 * float(n)) > 1e-8:
                        hn = project_feasible(hn, n)
                    c = compute_c(hn, n)
                    if c < bc - 1e-15:
                        bc = c
                        ba = a
                if ba > 0.0:
                    h = np.clip(h + ba * d, 0.0, 1.0)
                    h = project_feasible(h, n)
                    best = compute_c(h, n)
                    rounds += 1
                else:
                    break
    return h, best


def grad_from_active(idx, wv, h, n):
    out = np.zeros(n, dtype=float)
    dx = 2.0 / float(n)
    mid = n - 1
    for a in range(len(idx)):
        w = wv[a]
        if w <= 1e-14:
            continue
        k = int(idx[a]) - mid
        ws = w * dx
        if k >= 0:
            nv = n - k
            out[:nv] += ws * (1.0 - h[k:])
            out[k:] += ws * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            out[kk:] += ws * (1.0 - h[:nv])
            out[:nv] += ws * (-h[kk:])
    return out


def transfer_eval(h, v, i, j, delta, dx):
    hi0 = h[i]
    hj0 = h[j]
    vi0 = v[i]
    vj0 = v[j]
    h[i] = hi0 + delta
    h[j] = hj0 - delta
    v[i] = vi0 - delta
    v[j] = vj0 + delta
    c = float(np.max(np.correlate(h, v, mode='full')) * dx)
    h[i] = hi0
    h[j] = hj0
    v[i] = vi0
    v[j] = vj0
    return c


def pair_search_polish(h, n, deadline):
    h = project_feasible(h, n)
    v = 1.0 - h
    best = compute_c(h, n)
    dx = 2.0 / float(n)
    top = 28
    betas = (50.0, 500.0, 5000.0)
    steps = (0.025, 0.012, 0.006, 0.003, 0.0015, 0.0007, 0.0003, 0.0001)
    fracs = (1.0, 0.5, 0.25, 0.1)
    for beta in betas:
        for step in steps:
            rounds = 0
            while time.time() < deadline and rounds < 35:
                f = corr_data(h, n)
                t = float(f.max())
                if t < best - 1e-15:
                    best = t
                near = int(np.sum(f >= t - 0.004))
                cnt = min(f.size, max(12, min(120, near + 10)))
                idx = np.argsort(f)[-cnt:]
                sh = beta * (f[idx] - t)
                mx = sh.max()
                wv = np.exp(sh - mx)
                wv /= wv.sum()
                r = grad_from_active(idx, wv, h, n)
                order = np.argsort(r)
                lows = order[:top]
                highs = order[-top:]
                bi = -1
                bj = -1
                bd = 0.0
                bc = best
                for ii in lows:
                    ii = int(ii)
                    room_i = 1.0 - h[ii]
                    if room_i <= 1e-12:
                        continue
                    for jj in highs:
                        jj = int(jj)
                        if ii == jj or h[jj] <= 1e-12:
                            continue
                        u = min(room_i, h[jj], step)
                        if u < 1e-9:
                            continue
                        if r[jj] - r[ii] <= 1e-10:
                            continue
                        for fr in fracs:
                            delta = fr * u
                            c = transfer_eval(h, v, ii, jj, delta, dx)
                            if c < bc - 1e-15:
                                bc = c
                                bi = ii
                                bj = jj
                                bd = delta
                if bi >= 0:
                    h[bi] += bd
                    h[bj] -= bd
                    v = 1.0 - h
                    best = compute_c(h, n)
                    rounds += 1
                else:
                    break
    return h, best


def phase_scan(h, n, deadline, phases):
    h = np.array(h, dtype=float, copy=True)
    best = compute_c(h, n)
    besth = h.copy()
    phases = int(max(1, phases))
    x_old = np.arange(n) / float(n)
    x_ext = np.concatenate((x_old - 1.0, x_old, x_old + 1.0))
    v_ext = np.concatenate((h, h, h))
    base = np.arange(n) / float(n)
    for p in range(1, phases + 1):
        if time.time() >= deadline:
            break
        xp = (base + p / float(phases)) % 1.0
        hn = np.interp(xp, x_ext, v_ext)
        hn = project_feasible(hn, n)
        c = compute_c(hn, n)
        if c < best - 1e-15:
            best = c
            besth = hn
    return besth, best


def rescale_periodic(h, m):
    m = int(m)
    n = len(h)
    x_old = np.arange(n) / float(n)
    x_new = np.arange(m) / float(m)
    x_ext = np.concatenate((x_old - 1.0, x_old, x_old + 1.0))
    v_ext = np.concatenate((h, h, h))
    return np.interp(x_new, x_ext, v_ext)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(8.0, float(budget_s) - 5.0)

    try:
        hv = initial_h_values
        if hv is None:
            raise ValueError
        h = np.array(hv, dtype=float, copy=True)
        n = int(len(h))
        if n < 2:
            raise ValueError
    except Exception:
        n = 500
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = 0.5 + 0.30 * np.cos(np.pi * x) + 0.18 * np.sin(2.0 * np.pi * x) \
            + 0.10 * np.cos(4.0 * np.pi * x)

    h = project_feasible(h, n)
    best_h = h
    best_n = n
    best_c = compute_c(best_h, best_n)

    if time.time() < deadline - 10:
        phases = int(min(400, max(50, best_n if best_n <= 600 else 200)))
        d = min(45.0, 0.05 * (deadline - time.time()))
        best_h, best_c = phase_scan(best_h, best_n, time.time() + d, phases)

    for new_n in (512, 600):
        if time.time() > deadline - 220:
            break
        if best_n == new_n:
            continue
        ht = rescale_periodic(best_h, new_n)
        ht = project_feasible(ht, new_n)
        if compute_c(ht, new_n) > best_c + 0.03:
            continue
        rem = deadline - time.time()
        if rem < 25:
            continue
        phases = int(min(120, max(16, int(rem * 0.02))))
        ht, ct = phase_scan(ht, new_n, min(deadline, time.time() + 15), phases)
        if ct < best_c - 1e-15:
            best_h, best_c, best_n = ht, ct, new_n

    if time.time() < deadline - 120:
        alloc = min(200.0, 0.30 * (deadline - time.time()))
        if alloc > 20:
            best_h, best_c = lbfgs_polish(best_h, best_n, time.time() + alloc)

    if time.time() < deadline - 80:
        alloc = min(70.0, 0.10 * (deadline - time.time()))
        if alloc > 15:
            best_h, best_c = linear_polish(best_h, best_n, time.time() + alloc, 22)

    if time.time() < deadline - 100:
        rem = deadline - time.time()
        phases = int(min(120, max(10, int(rem * 0.02))))
        best_h, best_c = phase_scan(best_h, best_n, min(deadline, time.time() + 25), phases)

    if time.time() < deadline - 100:
        alloc = min(320.0, 0.40 * (deadline - time.time()))
        if alloc > 15:
            best_h, best_c = projected_gradient_polish(best_h, best_n, time.time() + alloc)

    if time.time() < deadline - 40:
        rem = deadline - time.time()
        alloc = min(350.0, 0.5 * max(0.0, rem - 10))
        if alloc > 15:
            best_h, best_c = pair_search_polish(best_h, best_n, time.time() + alloc)

    if time.time() < deadline - 15:
        rem = deadline - time.time()
        if rem > 30:
            best_h, best_c = linear_polish(best_h, best_n,
                                           time.time() + min(25.0, 0.25 * rem), 8)

    if time.time() < deadline - 5:
        rem = deadline - time.time()
        if rem > 20:
            best_h, best_c = projected_gradient_polish(best_h, best_n,
                                                       time.time() + min(20.0, 0.30 * rem))

    if time.time() < deadline - 1:
        rem = deadline - time.time()
        if rem > 5:
            phases = int(min(60, max(4, int(rem))))
            best_h, best_c = phase_scan(best_h, best_n, deadline, phases)

    best_h = project_feasible(best_h, best_n)
    best_h = normalize_exact(best_h, best_n)
    final_c = compute_c(best_h, best_n)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), int(best_n)
```
