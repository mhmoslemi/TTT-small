1. Compute s = c_full * dx: O(n^2) via np.correlate
2. Compute weights w via softmax: O(n)
3. Compute grad = sum_k w[k] * (dx * ∂g(m_k)/∂h): This is O(n^2) if done naively

For step 3, let's see if we can do it efficiently.
grad[i] = dx * sum_k w[k] * ∂g(m_k)/∂h[i]

For m_k ≥ 0: contributes (1-h[i-m_k])*I(m_k ≤ i ≤ n-1) + (-h[i+m_k])*I(0 ≤ i ≤ n-1-m_k)
For m_k < 0 (let m_k = -p): contributes (1-h[i+p])*I(0 ≤ i ≤ n-1-p) + (-h[i-p])*I(p ≤ i ≤ n-1)

This is like a weighted sum over all lags, which is essentially a convolution-like operation.

```python
def compute_c5_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')  # length 2n-1
    s = c * dx  # scaled
    m = s.max()
    shifted = alpha * (s - m)
    weights = np.exp(shifted - shifted.max())
    weights /= weights.sum()
    val = m + np.log(weights.sum()) / alpha  # = m + log(sum(exp)) / alpha
    
    # Gradient
    # grad[i] = sum_k weights[k] * dx * dg(m_k)/dh[i]
    # where m_k = k - (n-1)
    grad = np.zeros(n)
    for k in range(2*n-1):
        m = k - (n-1)  # lag
        wk = dx * weights[k]
        if m >= 0:
            # (1-h[i-m]) for i in [m, n-1]
            idx = np.arange(m, n)
            grad[idx] += wk * (1 - h[idx - m])
            # -h[i+m] for i in [0, n-1-m]
            idx = np.arange(0, n-m)
            grad[idx] += wk * (-h[idx + m])
        else:
            p = -m
            # (1-h[i+p]) for i in [0, n-1-p]
            idx = np.arange(0, n-p)
            grad[idx] += wk * (1 - h[idx + p])
            # -h[i-p] for i in [p, n-1]
            idx = np.arange(p, n)
            grad[idx] += wk * (-h[idx - p])
    
    return val, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_max_and_grad(h, n, alpha):
    """Compute smooth max of correlation and its gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    s = c * dx
    m_val = s.max()
    shifted = alpha * (s - m_val)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    log_sum = max_s + np.log(exp_s.sum())
    val = m_val + log_sum / alpha
    p = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    # For lags m = 0, 1, ..., n-1 (k = m + n - 1)
    for m in range(n):
        wk = dx * p[m + n - 1]
        if wk < 1e-15:
            continue
        grad[m:n] += wk * (1.0 - h[0:n-m])
        grad[0:n-m] += wk * (-h[m:n])
    # For lags m = -1, -2, ..., -(n-1) (k = n-1-m, i.e., k = n-2, n-3, ...)
    for pp in range(1, n):
        wk = dx * p[n - 1 - pp]
        if wk < 1e-15:
            continue
        grad[0:n-pp] += wk * (1.0 - h[pp:n])
        grad[pp:n] += wk * (-h[0:n-pp])
    
    return val, grad

def objective_and_grad(x, n, alpha, lam):
    """Objective with penalty for sum constraint."""
    val, grad = smooth_max_and_grad(x, n, alpha)
    pen = lam * (x.sum() - n / 2.0) ** 2
    grad_pen = 2.0 * lam * (x.sum() - n / 2.0)
    return val + pen, grad + grad_pen * np.ones(n)

def lbfgs_optimize(h, n, alpha, maxiter=500, lam=1e8):
    """Run L-BFGS-B optimization."""
    bounds = [(0.0, 1.0)] * n
    res = minimize(lambda x: objective_and_grad(x, n, alpha, lam)[0],
                   h, method='L-BFGS-B', jac=lambda x: objective_and_grad(x, n, alpha, lam)[1],
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return project_feasible(res.x, n)

def greed_binary(h_init, n, deadline, max_sweeps=200):
    """Greedy binary local search."""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, max_iters=2000000):
    """Simulated annealing on binary solutions."""
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = 0.01
    cool = 0.99998
    scale = 5000.0
    it = 0
    while it < max_iters:
        if time.time() > deadline:
            break
        it += 1
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c
```

```python
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, max_iters=2000000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].copy()
    zeros_arr = np.where(h == 0)[0].copy()
    T = 0.01
    cool = 0.99998
    scale = 5000.0
    it = 0
    while it < max_iters:
        if time.time() > deadline:
            break
        it += 1
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def lfgs_sweep(h, n, deadline, alpha_list, lam=1e6, maxiter=300):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alpha_list:
        if time.time() > deadline - 1:
            break
        h_new = lbfgs_run(best_h, n, alpha, maxiter=maxiter, lam=lam)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    deadline = t0 + total
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: L-BFGS-B with increasing alpha
    p1_end = t0 + 0.20 * total
    alphas1 = [50, 200, 1000, 5000, 50000, 500000]
    best_h, best_c = lfgs_sweep(best_h, n, p1_end, alphas1, lam=1e6, maxiter=400)
    # Phase 2: Greedy binary
    p2_end = t0 + 0.30 * total
    hb, cb = greed_binary(best_h, n, p2_end, max_sweeps=80)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 3: Re-optimize continuous from binary
    p3_end = t0 + 0.42 * total
    alphas3 = [10000, 100000, 1000000, 10000000]
    best_h, best_c = lfgs_sweep(best_h, n, p3_end, alphas3, lam=1e6, maxiter=300)
    # Phase 4: Annealing
    p4_end = t0 + 0.60 * total
    ha, ca = anneal_binary(best_h, n, p4_end, seed=seed, max_iters=3000000)
    if ca < best_c:
        best_c = ca
        best_h = ha.copy()
    # Phase 5: Re-optimize from annealed
    p5_end = t0 + 0.72 * total
    alphas5 = [50000, 500000, 5000000, 50000000]
    best_h, best_c = lfgs_sweep(best_h, n, p5_end, alphas5, lam=1e7, maxiter=300)
    # Phase 6: Perturbation restarts
    p6_end = t0 + 0.88 * total
    rng = np.random.RandomState(seed + 7)
    for trial in range(8):
        if time.time() > p6_end - 2:
            break
        noise_scale = 0.01 + 0.02 * trial / 8.0
        h_try = best_h + noise_scale * rng.randn(n)
        h_try = project_feasible(h_try, n)
        h_try, c_try = lfgs_sweep(h_try, n, min(time.time() + 15, p6_end - 1),
                                   [50000, 500000, 5000000], lam=1e6, maxiter=200)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
        hb2, cb2 = greed_binary(h_try, n, min(time.time() + 8, p6_end - 1), max_sweeps=40)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
    # Phase 7: Try lower resolution
    p7_end = t0 + 0.95 * total
    for nh in [200, 300]:
        if time.time() > p7_end - 10:
            break
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, ch = lfgs_sweep(hh, nh, min(time.time() + 12, p7_end - 5),
                            [10000, 100000, 1000000], lam=1e6, maxiter=200)
        hb3, cb3 = greed_binary(hh, nh, min(time.time() + 5, p7_end - 3), max_sweeps=30)
        if cb3 < ch:
            ch = cb3
            hh = hb3
        if ch < best_c:
            best_c = ch
            best_h = hh.copy()
            n = nh
    # Final polish
    if time.time() < deadline - 1:
        alphas_final = [10000000, 100000000]
        best_h, best_c = lfgs_sweep(best_h, n, deadline - 0.5, alphas_final, lam=1e8, maxiter=500)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
