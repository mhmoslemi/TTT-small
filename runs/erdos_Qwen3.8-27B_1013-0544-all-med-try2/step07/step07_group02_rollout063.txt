    corr = np.correlate(x, 1-x, mode='full') * dx
    m = corr.max()

    p = np.exp(s - sm)
    p /= p.sum()
    # Now compute gradient

    # For coordinate i: grad[i] = dx * sum_k p[k] * [(1-x[i+k])*1_ - x[i-k]*1_]

def adam_smooth_max(h, n, alpha, max_iter=2000, lr=0.01, deadline=None):
    dx = 2.0 / n
    m = 2 * n - 1
    h = h.astype(float).copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(1, max_iter + 1):
        if deadline and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        cm = corr.max()
        s = alpha * (corr - cm)
        sm = s.max()
        p = np.exp(s - sm)
        p /= p.sum()
        # Compute gradient
        grad = np.zeros(n)
        for k in range(m):
            lag = k - (n - 1)
            pk = p[k] * dx
            if pk < 1e-16:
                continue
            if lag >= 0:
                grad[:n - lag] += pk * (1 - h[lag:])
                grad[lag:] -= pk * h[:n - lag]
            else:
                il = -lag
                grad[il:] += pk * (1 - h[:n - il])
                grad[:n - il] -= pk * h[il:]
        # Adam update
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad ** 2
        m_hat = m_v / (1 - beta1 ** t)
        v_hat = v_v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        # Project
        h_new = np.clip(h_new, 0, 1)
        # enforce sum
        diff = n / 2.0 - h_new.sum()
        if abs(diff) > 1e-10:
            # distribute evenly among non-saturated
            if diff > 0:
                room = 1.0 - h_new
                s_room = room.sum()
                if s_room > 1e-14:
                    h_new += diff / s_room * room
            else:
                s_h = h_new.sum()
                if s_h > 1e-14:
                    h_new += diff / s_h * h_new
            h_new = np.clip(h_new, 0, 1)
        h = h_new
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

            ones.append(i)
            zeros.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0
    # Phase 1: Coordinate descent
    p1 = t0 + 0.10 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=200, deadline=p1)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()
    # Phase 2: Adam smooth max (multiple alphas)
    p2 = t0 + 0.30 * total
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() > p2 - 5:
            break
        ha, ca = adam_smooth_max(best_h, n, alpha, max_iter=2000, lr=0.005, deadline=p2 - 3)
        ha = project_feasible(ha, n)
        ca = compute_c5(ha, n)
        if ca < best_c:
            best_c, best_h = ca, ha.copy()
    # Phase 3: CD after Adam
    p3 = t0 + 0.40 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=200, deadline=p3)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()
    # Phase 4: Binary 2-swap
    p4 = t0 + 0.52 * total
    hb, cb = binary_2swap(best_h, n, p4, max_iter=200000, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
        hc, _ = coord_descent(best_h, n, max_sweeps=150, deadline=t0 + 0.58 * total)
        hc = project_feasible(hc, n)
        cc = compute_c5(hc, n)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()
    # Phase 5: SLSQP polish
    p5 = t0 + 0.72 * total
    h = best_h.copy()
    for alpha in [10000, 100000, 1e6, 1e7, 5e7, 2e8, 1e9, 5e9]:
        if time.time() > p5 - 5:
            break
        h = slsqp_optimize(h, n, alpha, maxiter=3000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 6: Final CD
    p6 = t0 + 0.82 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=200, deadline=p6)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()
    # Phase 7: Multi-resolution
    p7 = t0 + 0.93 * total
    for nh in [300, 350, 450, 500, 600]:
        if time.time() > p7 - 8:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, _ = coord_descent(hh, nh, max_sweeps=80, deadline=p7 - 6)
        hh = project_feasible(hh, nh)
        ch = compute_c5(hh, nh)
        for alpha in [100000, 1e6, 1e7]:
            if time.time() > p7 - 3:
                break
            ha, _ = adam_smooth_max(hh, nh, alpha, max_iter=1000, lr=0.004, deadline=p7 - 2)
            ha = project_feasible(ha, nh)
            ch2 = compute_c5(ha, nh)
            if ch2 < ch:
                hh, ch = ha, ch2
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh
    # Phase 8: Small perturbations + quick optimization
    if time.time() < dl - 5:
        for trial in range(10):
            if time.time() > dl - 2:
                break
            h = best_h + 0.002 * rng.randn(n)
            h = project_feasible(h, n)
            hc, _ = coord_descent(h, n, max_sweeps=80, deadline=dl - 1)
            hc = project_feasible(hc, n)
            cc = compute_c5(hc, n)
            if cc < best_c:
                best_c, best_h = cc, hc.copy()
                if time.time() < dl - 3:
                    h2 = slsqp_optimize(best_h, n, 1e9, maxiter=2000)
                    c2 = compute_c5(h2, n)
                    if c2 < best_c:
                        best_c, best_h = c2, h2.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
