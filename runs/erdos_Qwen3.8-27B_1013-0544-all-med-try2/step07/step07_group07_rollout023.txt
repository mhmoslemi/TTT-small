def project_feasible_fast(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Binary search for lambda such that sum(clip(y - lambda, 0, 1)) = target
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2, 0, 1)
    # Fine-tune to get exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Distribute among non-b saturated points
        free = (h > 0) & (h < 1)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
    return h

def project_feasible(y, n):

        h = np.clip(y - mid, 0.0, 1.0)

        if s > target:
            lo = mid
        else:
            hi = mid

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.sum(c**p) ** (1.0/p))

def _compute_lag_gradient(h_curr, n, lag):
    """Gradient of the k-th lag correlation (without dx factor)."""
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        # c_0 = sum h[i] * (1-h[i]) = sum(h) - sum(h^2)
        grad[:] = 1.0 - 2.0 * h_curr
        return grad
    if lag > 0:
        # c_lag = sum_{i=0}^{n-1-m} h[i+l] * (1-h[i])
        # d/dh[j]: if j in first n-m positions, contributes -h[j+m]
        #           if j in last n-m positions (j>=m), contributes (1-h[j-m])
        grad[:n-m] += -h_curr[m:]  # j is "first" position
        grad[m:] += (1.0 - h_curr[:n-m])  # j is "second" position
    else:
        # lag < 0, m = -lag
        # c_{-m} = sum_{i=m}^{n-1} h[i] * (1-h[i-m])

        #           if j in last n-m positions (j>=m), contributes -h[j-m]
        grad[:n-m] += (1.0 - h_curr[m:])
        grad[m:] += -h_curr[:n-m]
    return grad

def outer_approx_step(h_curr, n, trust):
    """Linearized outer approximation step."""
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _compute_lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.PDLP)
    except Exception:
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=50):

    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):

        h = np.clip(y - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(y - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h_curr
    elif lag > 0:
        grad[:n-m] += -h_curr[m:]
        grad[m:] += (1.0 - h_curr[:n-m])
    else:
        grad[:n-m] += (1.0 - h_curr[m:])
        grad[m:] += -h_curr[:n-m]
    return grad

    mx = shifted.max()

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _compute_lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.PDLP)
    except Exception:
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=40):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.1
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-10:
            break
        h_curr = h_new
        trust = max(0.005, trust * 0.9)
    return best_h, best_c

def gradient_local_search(h, n, deadline, max_rounds=500):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        c_max = c.max()
        active = np.where(c >= c_max - 0.005)[0]
        if len(active) > 20:
            active = np.argsort(c)[-20:]
        na = len(active)
        grad = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            grad += _compute_lag_gradient(h, n, lag) * dx / na
        i_best = int(np.argmax(grad))
        j_best = int(np.argmin(grad))
        if h[i_best] < 1e-6 or h[j_best] > 1.0 - 1e-6:
            sr = np.argsort(-grad)
            sa = np.argsort(grad)
            found = False
            for ii in sr[:20]:
                for jj in sa[:20]:
                    if h[ii] > 1e-4 and h[jj] < 1.0 - 1e-4:
                        i_best, j_best, found = ii, jj, True
                        break
                if found:
                    break
            if not found:
                break
        max_d = min(h[i_best], 1.0 - h[j_best])
        best_delta = 0.0
        best_val = best_c
        for frac in [0.1, 0.2, 0.35, 0.5, 0.7, 1.0]:
            d = max_d * frac
            ht = h.copy()
            ht[i_best] -= d
            ht[j_best] += d
            cv = compute_c5(ht, n)
            if cv < best_val:
                best_val = cv
                best_delta = d
        if best_delta > 0:
            h[i_best] -= best_delta
            h[j_best] += best_delta
            h = project_feasible(h, n)
            best_c = best_val
        else:
            break
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if time.time() < deadline - 50:
        oo_dl = time.time() + min(50, (deadline - time.time()) * 0.30)
        h_oo, c_oo = outer_approx(best_h, n, oo_dl)
        if c_oo < best_c:
            best_c, best_h = c_oo, h_oo
    for alpha in [5000, 20000, 100000]:
        if time.time() > deadline - 20:
            break
        h_new = slsqp_optimize(best_h, n, alpha, maxiter=2000, ftol=1e-14)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    if time.time() < deadline - 15:
        h_ml, c_ml = gradient_local_search(best_h, n, time.time() + min(15, (deadline - time.time()) * 0.4))
        if c_ml < best_c:
            best_c, best_h = c_ml, h_ml
    if time.time() < deadline - 3:
        h_new = slsqp_optimize(best_h, n, 500000, maxiter=3000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = min(350, (deadline - t0) * 0.38)
        best_h, best_c = optimize_at_n(h_init, n_init, t0 + alloc, rng)
    n_candidates = [200, 250, 350, 400]
    for n_try in n_candidates:
        if time.time() > deadline - 70:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        alloc = min(70, (deadline - time.time()) * 0.18)
        if alloc < 30:
            break
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        n_restarts = max(1, int((deadline - time.time() - 30) / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.012 * (1.0 + 0.4 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 30) * 0.35)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_optimize(best_h, n, 2000000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
        h_ml, c_ml = gradient_local_search(best_h, n, deadline - 1, max_rounds=800)
        if c_ml < best_c:
            best_c, best_h = c_ml, h_ml
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = compute_c5(best_h, 300)
    return best_h, best_c, len(best_h)
