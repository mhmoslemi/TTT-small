```python
def project_onto_box_sum(y, s, n):
    """Project y onto {h: 0<=h<=1, sum(h)=s}"""
    # Bisection on lambda
    lo, hi = y.min() - s, y.max() + s
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        if h.sum() > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2, 0, 1)
    return h
```

```python
    lo, hi = -1.0, 1.0

            lo = mid
        else:
            hi = mid

    diff = target - h_proj.sum()
    if abs(diff) > 1e-10:

            room = 1.0 - h_proj
            idx = np.where(room > 1e-10)[0]

                h_proj[idx] += diff / len(idx)

            if len(idx) > 0:
                h_proj[idx] += diff / len(idx)
```

def nesterov_optimize(h, n, deadline, alpha_init=500, alpha_end=50000):
    """Nesterov accelerated gradient for the smooth objective."""
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    
    dx = 2.0 / n
    max_iter = 2000
    beta = 0.9  # momentum
    
    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        # Adaptive alpha
        alpha = alpha_init * (alpha_end / alpha_init) ** (t / max_iter)
        
        # Compute gradient
        c = np.correlate(h_curr, 1 - h_curr, mode='full') * dx
        c_max = c.max()
        exp_c = np.exp(alpha * (c - c_max))
        Z = exp_c.sum()
        w = exp_c / Z
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for idx in range(2 * n - 1):
            wk = w[idx]
            if wk < 1e-14:
                continue
            s = idx - (n - 1)
            # First part
            j_lo = max(0, -s)
            j_hi = min(n - 1, n - 1 - s)
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += wk * (1.0 - h_curr[j_lo+s:j_hi+1+s])
            # Second part
            j_lo2 = max(0, s)
            j_hi2 = min(n - 1, n - 1 + s)
            if j_lo2 <= j_hi2:
                grad[j_lo2:j_hi2+1] -= wk * h_curr[j_lo2-s:j_hi2+1-s]
        grad *= dx
        
        # Nesterov step
        momentum = beta * (h_curr - h_prev)
        step = 1.0 / (2.0 * alpha)  # approximate 1/L
        y = h_curr + momentum
        h_new = y - step * grad
        h_new = project_feasible(h_new, n)
        
        h_prev = h_curr
        h_curr = h_new
        
        c_new = compute_c5(h_curr, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_curr.copy()
    
    return best_h, best_c
