    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    return float(_soft_max_val(c, alpha))

def slsqp_constraint(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_jac(x, n):
    return np.ones(n)

def run_lbfgs(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    mus = [5.0, 10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 100000.0]
    for i in range(len(alphas)):
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(penalty_obj, h, jac=penalty_grad,
                          args=(n, alphas[i], mus[i]), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_iters * 10,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run_slsqp(h, n, deadline, alpha=500000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_constraint, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def gradient_pair_search(h, n, deadline, top_k=20, step_size=0.003):
    best_c = compute_c5(h, n)
    alphas = [100, 500, 2000, 10000, 50000]
    for alpha in alphas:
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            # Find top-k most negative (want to increase) and most positive (want to decrease)
            neg_idx = np.argsort(g)[:top_k]
            pos_idx = np.argsort(g)[-top_k:][::-1]
            for i in neg_idx:
                if time.time() > deadline:
                    return h, best_c
                for j in pos_idx:
                    if i == j:
                        continue
                    # Try increasing h[i], decreasing h[j]
                    move = step_size * (g[j] - g[i])
                    if abs(move) < 1e-10:
                        continue
                    if move > 0 and h[i] + move <= 1.0 and h[j] - move >= 0.0:
                        h[i] += move; h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= move; h[j] += move
                    if move < 0 and h[i] + move >= 0.0 and h[j] - move <= 1.0:
                        h[i] += move; h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= move; h[j] += move
            if not improved:
                break
    return h, best_c

def local_pair_search(h, n, deadline, deltas, radius=30):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def momentum_descent(h, n, deadline, alpha=5000, lr=0.02, momentum=0.85):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    for _ in range(10000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            v *= 0.5
            lr *= 0.7
            if lr < 1e-10:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h300 = np.array(initial_h_values, dtype=float).copy()
        n300 = len(h300)
    except Exception:
        h300 = np.full(300, 0.5)
        n300 = 300

    h300 = project_feasible(h300, n300)
    best_c = compute_c5(h300, n300)
    best_h = h300.copy()

    # Phase 1: Quick refinement on n=300
    if time.time() < deadline - 600:
        t1 = min(time.time() + 120, deadline - 600)
        h300, c300 = run_lbfgs(h300, n300, t1, max_iters=150)
        if c300 < best_c:
            best_c = c300
            best_h = h300.copy()

    # Phase 2: Interpolate to n=600 and optimize
    n600 = 600
    x_old = np.linspace(0, 2, n300)
    x_new = np.linspace(0, 2, n600)
    h600 = np.interp(x_new, x_old, h300)
    h600 = project_feasible(h600, n600)
    c600 = compute_c5(h600, n600)

    # Phase 2a: L-BFGS-B on n=600
    if time.time() < deadline - 350:
        t2a = min(time.time() + 250, deadline - 350)
        h600, c600 = run_lbfgs(h600, n600, t2a, max_iters=150)

    # Phase 2b: Momentum descent on n=600
    if time.time() < deadline - 200:
        t2b = min(time.time() + 80, deadline - 200)
        h600, c600 = momentum_descent(h600, n600, t2b, alpha=5000, lr=0.01, momentum=0.9)

    # Phase 2c: SLSQP on n=600
    if time.time() < deadline - 150:
        t2c = min(time.time() + 80, deadline - 150)
        h600, c600 = run_slsqp(h600, n600, t2c, alpha=500000, maxiter=1000)

    # Phase 2d: Gradient-guided pair search on n=600
    if time.time() < deadline - 80:
        t2d = min(time.time() + 100, deadline - 80)
        h600, c600 = gradient_pair_search(h600, n600, t2d, top_k=15, step_size=0.002)

    # Phase 2e: Local pair search on n=600
    if time.time() < deadline - 40:
        t2e = min(time.time() + 80, deadline - 40)
        h600, c600 = local_pair_search(h600, n600, t2e,
                                        [0.005, 0.002, 0.001, 0.0005], radius=25)

    # Phase 2f: Final L-BFGS-B polish on n=600
    if time.time() < deadline - 10:
        h600, c600 = run_lbfgs(h600, n600, deadline - 5, max_iters=100)

    # Compare and pick best
    if c600 < best_c:
        best_h = h600
        best_c = c600
        best_n = n600
    else:
        best_n = n300

    # Final check on n300 if it was the best
    if best_n == n300 and time.time() < deadline - 10:
        h300, c300 = run_lbfgs(h300, n300, deadline - 5, max_iters=100)
        if c300 < best_c:
            best_c = c300
            best_h = h300

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
