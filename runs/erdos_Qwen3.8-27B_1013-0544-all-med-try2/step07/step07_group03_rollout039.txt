import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def subgradient_step(h, n, lr, top_k=1):
    """One subgradient step using top-k of the correlation peaks."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Get top-k peaks
    top_idx = np.argsort(corr)[-top_k:]
    s_vals = top_idx - (n - 1)  # actual shifts
    grad = np.zeros(n)
    for s in s_vals:
        # ∂c[n-1+s]/∂h[i] = (1-h[i+s]) * 1{0≤i+s<n} - h[i-s] * 1{0≤i-s<n}
        i_lo1 = max(0, -s)
        i_hi1 = min(n, n - s)
        if i_lo1 < i_hi1:
            idx1 = np.arange(i_lo1, i_hi1)
            grad[idx1] += (1.0 - h[idx1 + s])
        i_lo2 = max(0, s)
        i_hi2 = min(n, n + s)
        if i_lo2 < i_hi2:
            idx2 = np.arange(i_lo2, i_hi2)
            grad[idx2] -= h[idx2 - s]
    grad *= dx / top_k
    h_new = h - lr * grad
    return project_feasible(h_new, n)

def softmax_gradient_step(h, n, alpha, lr):
    """One gradient step using soft-max smoothing."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Soft max weights
    c_max = corr.max()
    weights = np.exp(alpha * (corr - c_max))
    weights /= weights.sum()
    
    # Compute gradient: weighted sum of individual gradients
    grad = np.zeros(n)
    # Only process significant weights for speed
    significant = weights > 1e-8
    s_vals = np.where(significant)[0] - (n - 1)
    w_vals = weights[significant]
    
    for idx in range(len(s_vals)):
        s = s_vals[idx]
        w = w_vals[idx]
        i_lo1 = max(0, -s)
        i_hi1 = min(n, n - s)
        if i_lo1 < i_hi1:
            idx1 = np.arange(i_lo1, i_hi1)
            grad[idx1] += w * (1.0 - h[idx1 + s])
        i_lo2 = max(0, s)
        i_hi2 = min(n, n + s)
        if i_lo2 < i_hi2:
            idx2 = np.arange(i_lo2, i_hi2)
            grad[idx2] -= w * h[idx2 - s]
    
    grad *= dx
    h_new = h - lr * grad
    return project_feasible(h_new, n)

def binary_swap_search(h, n, deadline, max_sweeps=100, patience=20):
    """Greedy pairwise swap local search for binary h."""
    if (h > 0.5).sum() != n // 2:
        h = to_binary(h, n).copy()
    else:
        h = (h > 0.5).astype(float).copy()
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for zi in range(len(zeros)):
            if time.time() > deadline:
                break
            i = zeros[zi]
            for oj in range(len(ones)):
                j = ones[oj]
                # Compute delta for swapping h[i]:0->1, h[j]:1->0
                delta = corr.copy()
                # h[i] changes from 0 to 1
                # Effect on corr: for each lag, the contribution of position i changes
                # corr[k] includes h[i]*(1-h[i+k-n+1]) term
                # Old: 0, New: (1-h[i+k-n+1])
                # So delta[k] += (1-h[i+k-n+1]) for valid k
                s_range = np.arange(-(n-1), n)
                # Position i contributes to corr at index n-1+s where s = shift
                # corr[n-1+s] includes h[i]*(1-h[i+s])
                # New contribution: 1*(1-h[i+s]) for valid i+s
                valid = (i + s_range >= 0) & (i + s_range < n)
                delta[valid + (n-1)] += (1.0 - h[i + s_range[valid]])
                
                # Position j changes from 1 to 0
                # corr[n-1+s] includes h[j]*(1-h[j+s])
                # Old: (1-h[j+s]), New: 0
                # delta[k] -= (1-h[j+s]) for valid j+s
                valid_j = (j + s_range >= 0) & (j + s_range < n)
                delta[valid_j + (n-1)] -= (1.0 - h[j + s_range[valid_j]])
                
                # Also, position i appears in the (1-h) factor:
                # corr[n-1+s] includes h[j-s]*(1-h[i]) when j-s = i, i.e., s = j-i
                # Wait, more generally: corr[n-1+s] = sum_t h[t]*(1-h[t+s])
                # h[i] appears as h[t] when t=i: contributes (1-h[i+s]) [already handled above]
                # h[i] appears as h[t+s] when t+s=i, i.e., t=i-s: contributes h[i-s]*(1-h[i])
                # Old: h[i-s]*(1-0) = h[i-s], New: h[i-s]*(1-1) = 0
                # So delta -= h[i-s] for valid i-s
                valid_is = (i - s_range >= 0) & (i - s_range < n)
                delta[valid_is + (n-1)] -= h[i - s_range[valid_is]]
                
                # h[j] appears as h[t+s] when t+s=j, i.e., t=j-s: contributes h[j-s]*(1-h[j])
                # Old: h[j-s]*(1-1) = 0, New: h[j-s]*(1-0) = h[j-s]
                # So delta += h[j-s] for valid j-s
                valid_js = (j - s_range >= 0) & (j - s_range < n)
                delta[valid_js + (n-1)] += h[j - s_range[valid_js]]
                
                # Check if this improves
                new_c = (corr + delta).max() * dx
                if new_c < cur_c - 1e-15:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr += delta
                    cur_c = new_c
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > patience:
                    break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42, n_iters=5000000, T0=0.01, T_end=1e-8):
    """Simulated annealing for binary swaps."""
    rng = np.random.RandomState(seed)
    h = to_binary(h_init, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    ones_arr = np.where(h == 1.0)[0].copy()
    zeros_arr = np.where(h == 0.0)[0].copy()
    
    n_total = n_iters
    cool_rate = (T_end / T0) ** (1.0 / max(n_total, 1))
    T = T0
    
    for it in range(n_total):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        
        # Compute delta (same as in binary_swap_search)
        s_range = np.arange(-(n-1), n)
        delta = np.zeros(2*n-1)
        
        valid_i = (i + s_range >= 0) & (i + s_range < n)
        delta[valid_i + (n-1)] += (1.0 - h[i + s_range[valid_i]])
        
        valid_j = (j + s_range >= 0) & (j + s_range < n)
        delta[valid_j + (n-1)] -= (1.0 - h[j + s_range[valid_j]])
        
        valid_is = (i - s_range >= 0) & (i - s_range < n)
        delta[valid_is + (n-1)] -= h[i - s_range[valid_is]]
        
        valid_js = (j - s_range >= 0) & (j - s_range < n)
        delta[valid_js + (n-1)] += h[j - s_range[valid_js]]
        
        new_c = (corr + delta).max() * dx
        delta_c = new_c - cur_c
        
        if delta_c <= 0 or (T > T_end and rng.random() < np.exp(-delta_c / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = new_c
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        
        T *= cool_rate
    
    return best_h, best_c

The change comes from:
1. Terms where t = i: h'[i](1-h'[i+s]) - h[i](1-h[i+s]) = 1*(1-h[i+s]) - 0 = (1-h[i+s])
   (but h[i+s] might have changed if i+s = j)
2. Terms where t = j: h'[j](1-h'[j+s]) - h[j](1-h[j+s]) = 0 - 1*(1-h[j+s]) = -(1-h[j+s])
   (but h[j+s] might have changed if j+s = i)
3. Terms where t+s = i (i.e., t = i-s): h'[t](1-h'[i]) - h[t](1-h[i]) = h[i-s](1-1) - h[i-s](1-0) = -h[i-s]
   (but h[i-s] might have changed if i-s = j)
