def enforce_sum(h, n):
    target = 0.5 * n
    for _ in range(20):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-13:
            return h
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(free):
            return h
        if diff > 0.0:
            w = np.zeros(n)
            w[free] = 1.0 - h[free]
        else:
            w = np.zeros(n)
            w[free] = h[free]
        sw = float(w.sum())
        if sw <= 0.0:
            return h
        h = h + diff * w / sw
        h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = target - s
    if abs(diff) > 1e-15:
        free = np.flatnonzero((h > 1e-15) & (h < 1.0 - 1e-15))
        if free.size:
            idx = int(free[int(np.argmax(h[free] * (1.0 - h[free])))]
            h[idx] = float(h[idx] + diff)
            if h[idx] < 0.0:
                h[idx] = 0.0
            elif h[idx] > 1.0:
                h[idx] = 1.0
    return h


def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).ravel().copy()
    if y.size > n:
        y = y[:n]
    elif y.size < n:
        y = np.concatenate((y, np.full(n - y.size, 0.5)))
    target = 0.5 * n
    if y.size == 0:
        return y
    ymin = float(np.min(y))
    ymax = float(np.max(y))
    if ymin >= -1e-12 and ymax <= 1.0 + 1e-12:
        h = np.clip(y, 0.0, 1.0)
        if abs(float(h.sum()) - target) < 1e-8:
            return enforce_sum(h, n)
    low = ymin - 1.0
    high = ymax + 1.0
    for _ in range(70):
        mid = 0.5 * (low + high)
        val = float(np.clip(y - mid, 0.0, 1.0).sum())
        if val > target:
            low = mid
        else:
            high = mid
    h = np.clip(y - 0.5 * (low + high), 0.0, 1.0)
    return enforce_sum(h, n)


def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def smooth_objective(x, n, alpha):
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(np.max(c))
    alpha = float(alpha)
    if alpha < 1e-6:
        return m
    z = alpha * (c - m)
    z -= float(np.max(z))
    e = np.exp(z)
    return m + float(np.log(np.sum(e))) / alpha


def smooth_jacobian(x, n, alpha):
    dx = 2.0 / float(n)
    raw = np.correlate(x, 1.0 - x, mode='full')
    c = raw * dx
    m = float(np.max(c))
    alpha = float(alpha)
    if alpha < 1e-4:
        w = np.zeros(len(c))
        active = np.flatnonzero(c >= m - 1e-8)
        if active.size == 0:
            active = np.flatnonzero(c == m)
        w[active] = 1.0 / float(active.size)
    else:
        z = alpha * (c - m)
        z -= float(np.max(z))
        e = np.exp(z)
        e /= float(np.sum(e))
        w = e
    g = np.zeros(n)
    offset = n - 1
    idxs = np.flatnonzero(w > 1e-10)
    for idx in idxs:
        wt = float(w[idx])
        tau = int(idx) - offset
        if tau < 0:
            s = -tau
            ls = n - s
            if ls > 0:
                g[:ls] += wt
                g[:ls] -= wt * x[s:]
                g[s:] -= wt * x[:ls]
        elif tau > 0:
            s = tau
            ls = n - s
            if ls > 0:
                g[s:] += wt
                g[:ls] -= wt * x[s:]
                g[s:] -= wt * x[:ls]
        else:
            g += wt * (1.0 - 2.0 * x)
    return dx * g


def _sum_c(x, n):
    return float(x.sum() - 0.5 * n)


def _sum_j(x, n):
    return np.ones(n, dtype=float)


def slsqp_smooth(x, n, alpha, maxiter):
    x0 = project_box_sum(np.asarray(x, dtype=float).ravel(), n)
    if maxiter <= 0:
        return x0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(
            smooth_objective, x0, jac=smooth_jacobian, args=(n, alpha),
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-12}
        )
        raw = res.x
        if raw is not None:
            raw = np.asarray(raw, dtype=float)
            if np.all(np.isfinite(raw)):
                return project_box_sum(raw, n)
    except Exception:
        pass
    return x0


def slsqp_sequence(h, n, deadline, alphas, maxiter):
    h = project_box_sum(np.asarray(h, dtype=float).ravel(), n)
    best_c = compute_c5(h, n)
    h_best = h.copy()
    for a in alphas:
        if time.time() > deadline - 5:
            break
        ht = slsqp_smooth(h, n, a, maxiter)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h_best = ht.copy()
            h = ht
        elif c < best_c + 1e-6:
            h = ht
    return h_best, best_c


def aug_objective(x, n, alpha, la, rho):
    target = 0.5 * n
    err = float(x.sum() - target)
    return smooth_objective(x, n, alpha) + la * err + 0.5 * rho * err * err


def aug_jacobian(x, n, alpha, la, rho):
    g = smooth_jacobian(x, n, alpha)
    target = 0.5 * n
    err = float(x.sum() - target)
    return g + (la + rho * err) * np.ones(n, dtype=float)


def lbfgs_augmented_update(x, n, alpha, la, rho, maxiter):
    x0 = project_box_sum(np.asarray(x, dtype=float).ravel(), n)
    out = x0
    new_la = la
    if maxiter <= 0:
        return out, new_la
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            aug_objective, x0, jac=aug_jacobian, args=(n, alpha, la, rho),
            method='L-BFGS-B', bounds=bounds,
            options={
                'maxiter': int(maxiter),
                'ftol': 1e-14,
                'gtol': 1e-9,
                'maxfun': int(max(1000, maxiter * 30)),
                'maxls': 40
            }
        )
        raw = res.x
        if raw is not None:
            raw = np.asarray(raw, dtype=float)
            if np.all(np.isfinite(raw)):
                target = 0.5 * n
                err = float(raw.sum() - target)
                new_la = la + rho * err
                if not np.isfinite(new_la):
                    new_la = la
                if new_la > 1e6:
                    new_la = 1e6
                elif new_la < -1e6:
                    new_la = -1e6
                out = project_box_sum(raw, n)
    except Exception:
        pass
    return out, new_la


def lbfgs_sequence(h, n, deadline, alphas, rho_start, rounds, maxiter):
    h = project_box_sum(np.asarray(h, dtype=float).ravel(), n)
    best_c = compute_c5(h, n)
    h_best = h.copy()
    h_work = h.copy()
    for a in alphas:
        if time.time() > deadline - 5:
            break
        la = 0.0
        rho = float(rho_start)
        mi = int(maxiter)
        rem = deadline - time.time()
        if rem < 30:
            mi = max(10, int(mi * 0.30))
        elif rem < 60:
            mi = max(20, int(mi * 0.60))
        last_c = None
        for _ in range(rounds):
            if time.time() > deadline - 5:
                break
            h_work, la = lbfgs_augmented_update(h_work, n, a, la, rho, mi)
            last_c = compute_c5(h_work, n)
            if last_c < best_c:
                best_c = last_c
                h_best = h_work.copy()
            rho *= 4.0
            if rho > 1e7:
                break
        if last_c is not None and last_c > best_c + 5e-3:
            h_work = h_best.copy()
    return h_best, best_c


def projected_gradient_polish(h, n, deadline, alpha=100000.0):

                    best_c = c
                    h_best = h.copy()
                    ok = True
                    break
                trial *= 0.5
            if not ok:
                break
            lr = trial
    return h_best, best_c


def pair_transfer_cost(t, h, n, i, j):
    oi = h[i]
    oj = h[j]
    h[i] = oi + t
    h[j] = oj - t
    val = compute_c5(h, n)
    h[i] = oi
    h[j] = oj
    return val


def gradient_pair_search(h, n, deadline, iterations=8, max_pairs=60):
    h = project_box_sum(np.asarray(h, dtype=float).ravel(), n)
    best = compute_c5(h, n)
    tested = set()
    for _ in range(int(iterations)):
        if time.time() > deadline - 2:
            break
        g = smooth_jacobian(h, n, 200000.0)
        g0 = g - float(np.mean(g))
        if float(np.max(np.abs(g0))) < 1e-11:
            break
        lo_idx = np.argsort(g0)[:12]
        hi_idx = np.argsort(g0)[-12:][::-1]
        improved = False
        cnt = 0
        for i in lo_idx:
            i = int(i)
            for j in hi_idx:
                j = int(j)
                cnt += 1
                if cnt > max_pairs:
                    break
                if i == j:
                    continue
                key = (i, j)
                if key in tested or (j, i) in tested:
                    continue
                lo_t = max(-float(h[i]), float(h[j]) - 1.0)
                hi_t = min(1.0 - float(h[i]), float(h[j]))
                if hi_t <= lo_t + 1e-10:
                    continue
                if g0[i] >= g0[j]:
                    continue
                tested.add(key)
                tested.add((j, i))
                try:
                    res = minimize_scalar(
                        pair_transfer_cost,
                        args=(h, n, i, j),
                        bounds=(lo_t, hi_t),
                        options={'xatol': 1e-7, 'maxiter': 30}
                    )
                except Exception:
                    continue
                if res.fun < best - 1e-14 and res.x is not None:
                    tx = float(res.x)
                    h[i] += tx
                    h[j] -= tx
                    h = project_box_sum(h, n)
                    best = compute_c5(h, n)
                    improved = True
                    break
            if improved:
                break
            if cnt > max_pairs:
                break
    return h, best


def local_polish(h, n, deadline, alpha=100000.0):

def resample(h, ns, nd):
    h = np.asarray(h, dtype=float).ravel()
    if ns == nd:
        return project_box_sum(h, nd)
    xs = (np.arange(ns) + 0.5) * (2.0 / float(ns))
    xd = (np.arange(nd) + 0.5) * (2.0 / float(nd))
    h2 = np.interp(xd, xs, h, left=0.0, right=0.0)
    return project_box_sum(h2, nd)


def noise_candidate(base, n, scale, rng):
    h0 = resample(base, len(base), n)
    nh = rng.normal(0.0, scale, n)
    nh -= float(nh.mean())
    return project_box_sum(h0 + nh, n)


def symmetrize(h, n):
    h = np.asarray(h, dtype=float).ravel()
    h = 0.5 * (h + h[::-1])
    return project_box_sum(h, n)

        hS, cS = slsqp_sequence(best_h, best_n, deadline - 5, [50000, 200000], 70)
        if cS < best_c:
            best_c = cS
            best_h = hS.copy()

    if time.time() < deadline - 10:
        hL, cL = local_polish(best_h, best_n, deadline - 3, 300000.0)
        if cL < best_c:
            best_c = cL
            best_h = hL.copy()

    best_h = project_box_sum(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
