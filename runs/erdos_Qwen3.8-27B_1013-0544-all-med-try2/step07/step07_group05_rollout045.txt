import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def sum_constraint(x, n):
    return float(x.sum() - n / 2.0)

def sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    """Simulated annealing for the overlap problem."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    current_h = h.copy()
    
    # Adaptive parameters
    T = 0.0005  # Initial temperature (scale of C5 differences)
    cooling = 0.9997
    min_T = 1e-7
    
    # Measure step time
    t0 = time.time()
    _ = compute_c5(current_h, n)
    step_time = time.time() - t0
    
    n_steps = 0
    no_improve = 0
    
    while time.time() < deadline and T > min_T:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Adaptive delta: larger when T is high
        delta = rng.uniform(0.005, 0.05 * (T / 0.0005 + 0.5))
        delta = min(delta, 0.5)
        
        # Try transfer
        sign = 1 if rng.random() < 0.5 else -1
        di = sign * delta
        dj = -sign * delta
        
        new_hi = current_h[i] + di
        new_hj = current_h[j] + dj
        
        if new_hi < 0 or new_hi > 1 or new_hj < 0 or new_hj > 1:
            # Try smaller delta
            max_d = min(abs(current_h[i] - (1 if di > 0 else 0)),
                        abs(current_h[j] - (0 if dj < 0 else 1)))
            if max_d < 1e-6:
                continue
            delta = rng.uniform(0.001, max_d * 0.9)
            di = sign * delta
            dj = -sign * delta
            new_hi = current_h[i] + di
            new_hj = current_h[j] + dj
            if new_hi < 0 or new_hi > 1 or new_hj < 0 or new_hj > 1:
                continue
        
        current_h[i] = new_hi
        current_h[j] = new_hj
        new_c = compute_c5(current_h, n)
        de = new_c - current_c
        
        if de < 0 or rng.random() < np.exp(-de / T):
            current_c = new_c
            if new_c < best_c - 1e-14:
                best_c = new_c
                best_h = current_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            current_h[i] = new_hi - di
            current_h[j] = new_hj - dj
        
        T *= cooling
        n_steps += 1
        
        # Occasionally reheat if stuck
        if no_improve > 5000 and T > 0.0001:
            T = min(T * 2, 0.001)
            # Reset current to best
            current_h = best_h.copy()
            current_c = best_c
            no_improve = 0
    
    return best_h, best_c

def pair_search(h, n, deadline, deltas):

    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):

                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def quick_project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        free = (h > 0.005) & (h < 0.995)
        nf = free.sum()
        if nf > 0:
            h[free] += diff / nf
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def sum_constraint(x, n):
    return float(x.sum() - n / 2.0)

def sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def sa_optimize(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_h = h.copy()
    current_c = best_c
    T = 0.002
    min_T = 1e-7
    no_improve = 0
    best_T = T

    while time.time() < deadline:
        stuck_factor = 1.0 + min(float(no_improve) / 3000.0, 4.0)
        n_pert = int(rng.randint(1, max(2, 2 + 2 * stuck_factor)))
        n_pert = min(n_pert, n)
        delta_scale = min(0.03 * stuck_factor * (0.5 + T / 0.002), 0.4)

        indices = rng.choice(n, size=n_pert, replace=False)
        deltas = rng.uniform(-delta_scale, delta_scale, size=n_pert)
        deltas -= deltas.mean()

        new_h = current_h.copy()
        new_h[indices] += deltas
        new_h = quick_project(new_h, n)
        new_c = compute_c5(new_h, n)
        de = new_c - current_c

        accept = de < 0
        if not accept and de < 0.05:
            accept = rng.random() < np.exp(-de / max(T, 1e-10))

        if accept:
            current_h = new_h
            current_c = new_c
            if new_c < best_c - 1e-15:
                best_c = new_c
                best_h = new_h.copy()
                no_improve = 0
                best_T = T
            else:
                no_improve += 1
        else:
            no_improve += 1

        T *= 0.9997
        if no_improve > 15000 and T > min_T * 5:
            T = max(T * 3.0, best_T * 0.8)
            current_h = best_h.copy()
            current_c = best_c
            no_improve = 0

    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break
    return h, best_c
