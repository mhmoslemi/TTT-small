    n = len(h)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_c_vec(h, n, k_idx):
    """Gradient of correlation entry k_idx w.r.t. h."""
    s = k_idx - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        g[-s:] += (1.0 - h[:n+s])
        g[:n+s] -= h[-s:]
    return g

def subgradient_descent(h, n, deadline, eta0=0.05, decay=0.9995, M=5):
    """Subgradient descent on the max of top-M correlation entries."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    eta = eta0
    
    t_start = time.time()

        g /= M

def informed_sa(h, n, deadline, rng, T0=5e-5, cooling=0.9997):
    """Simulated annealing with informed moves."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    h = h.copy()
    T = T0
    
    while time.time() < deadline and T > 1e-9:
        # Compute current worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
        s_star = k_star - (n - 1)
        
        # Pick indices biased toward the critical region
        # Points that contribute most to the worst shift
        # For shift s_star, contribution at position i is h[i]*(1-h[i+s_star])
        # Pick i from the high-contribution positions
        if s_star >= 0:
            contrib = h[:n-s_star] * (1 - h[s_star:])
            valid_range = (0, n - s_star)
        else:
            contrib = h[-s_star:] * (1 - h[:n+s_star])
            valid_range = (-s_star, n)
        
        # Pick i weighted by contribution (or uniformly as fallback)
        r = rng.random()
        if r < 0.5 and contrib.sum() > 1e-10:
            probs = contrib / contrib.sum()
            i_local = rng.choice(len(contrib), p=probs)
            i = i_local if s_star >= 0 else i_local + (-s_star)
        else:
            i = rng.randint(n)
        
        j = rng.randint(n)
        if i == j:
            j = (j + 1) % n
        
        # Determine delta
        delta = 0.005 + 0.02 * (T / T0)  # adaptive: larger at high T
        if rng.random() < 0.5:
            if h[i] + delta > 1.0 or h[j] - delta < 0.0:
                continue
            h[i] += delta; h[j] -= delta
        else:
            if h[i] - delta < 0.0 or h[j] + delta > 1.0:
                continue
            h[i] -= delta; h[j] += delta
        
        new_c = compute_c5(h, n)
        dE = new_c - current_c
        
        if dE < 0 or rng.random() < np.exp(-dE / T):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # Revert
            if h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
                h[i] -= delta; h[j] += delta
            else:
                h[i] += delta; h[j] -= delta

    return best_h, best_c

        alloc = min(60, (deadline - time.time()) * 0.35)
        hs, cs = informed_sa(h, n, time.time() + alloc, rng)
        if cs < best_c:
            best_c = cs
            h = hs
    if time.time() < deadline - 5:
        ht = slsqp(h, n, 500000, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    if time.time() < deadline - 2:
        if n <= 150:
            h, best_c = pair_search_full(h, n, deadline - 1,
                                         [0.01, 0.005, 0.002, 0.001, 0.0005])
        else:
            h, best_c = near_pair_search(h, n, deadline - 1,
                                         [0.01, 0.005, 0.002, 0.001, 0.0005], radius=50)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize initial construction with full pipeline
    alloc1 = min(280, (deadline - t0) * 0.30)
    h_opt, c_opt = optimize_single(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt, c_opt

    # Phase 2: Try a coarser grid for different structure
    n_coarse = max(80, n_init - 70)
    if time.time() > deadline - 120:
        pass
    else:
        x_init = np.linspace(0, 2, n_init)
        x_coarse = np.linspace(0, 2, n_coarse)
        h_coarse = np.interp(x_coarse, x_init, h_init)
        h_coarse = project_feasible(h_coarse, n_coarse)
        alloc2 = min(80, (deadline - time.time()) * 0.12)
        if alloc2 > 30:
            h_opt, c_opt = optimize_single(h_coarse, n_coarse, time.time() + alloc2, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 3: Try a finer grid
    n_fine = n_init + 50
    if time.time() < deadline - 100:
        x_init = np.linspace(0, 2, n_init)
        x_fine = np.linspace(0, 2, n_fine)
        h_fine = np.interp(x_fine, x_init, h_init)
        h_fine = project_feasible(h_fine, n_fine)
        alloc3 = min(100, (deadline - time.time()) * 0.15)
        if alloc3 > 30:
            h_opt, c_opt = optimize_single(h_fine, n_fine, time.time() + alloc3, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Multi-restart with structured perturbation
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 50
        n_restarts = max(3, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scale = 0.02 + 0.05 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 30) * 0.3)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_single(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Extended subgradient + SA on best
    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        alloc_sg = min(30, (deadline - time.time()) * 0.3)
        hs, cs = subgrad_descent(best_h, n, time.time() + alloc_sg, M=3, eta0=0.02)
        if cs < best_c:
            best_c = cs
            best_h = hs
        if time.time() < deadline - 10:
            hs, cs = informed_sa(best_h, n, deadline - 8, rng, T0=2e-5, cooling=0.9999)
            if cs < best_c:
                best_c = cs
                best_h = hs

    # Phase 6: Final SLSQP + pair search
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        ht = slsqp(best_h, n, 1000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 3:
            if n <= 150:
                best_h, best_c = pair_search_full(best_h, n, deadline - 1,
                                                  [0.001, 0.0005, 0.0002, 0.0001])
            else:
                best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                                  [0.001, 0.0005, 0.0002], radius=40)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
