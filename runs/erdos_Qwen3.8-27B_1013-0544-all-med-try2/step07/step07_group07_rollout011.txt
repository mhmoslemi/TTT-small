import numpy as np
import time


def project_feasible(x, n):
    target = 0.5 * float(n)
    x = np.asarray(x, dtype=float).copy()
    if n == 0:
        return x
    s0 = float(x.sum())
    xmax = float(np.max(x))
    xmin = float(np.min(x))
    lo = min(0.0, target - s0, 1.0 - xmax)
    hi = max(0.0, target - s0, -xmin)
    if hi - lo < 1e-12:
        lo -= 1.0
        hi += 1.0
    for _ in range(20):
        if float(np.clip(x + lo, 0.0, 1.0).sum()) <= target:
            break
        lo -= (hi - lo if hi > lo else 1.0) + 1.0
    for _ in range(20):
        if float(np.clip(x + hi, 0.0, 1.0).sum()) >= target:
            break
        hi += (hi - lo if hi > lo else 1.0) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(x + mid, 0.0, 1.0).sum()) < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(y.sum())
    if abs(diff) > 1e-13:
        mask = (y > 1e-7) & (y < 1.0 - 1e-7)
        if np.any(mask):
            idxs = np.where(mask)[0]
            val = y[idxs] * (1.0 - y[idxs])
            idx = int(idxs[int(np.argmax(val))])
            y[idx] = float(np.clip(y[idx] + diff, 0.0, 1.0))
        else:
            y = np.clip(y + diff / float(n), 0.0, 1.0)
    return y


def resample_h(h_src, n):
    h_src = np.asarray(h_src, dtype=float)
    m = int(len(h_src))
    if m == int(n):
        return h_src.copy()
    if m == 1:
        return np.full(int(n), float(h_src[0]))
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, int(n), endpoint=False)
    return np.interp(x_new, x_old, h_src)


def sine_start(n, seed):
    rng = np.random.RandomState(int(seed))
    x = (np.arange(int(n)) + 0.5) * (2.0 / float(n))
    f = 1.0 + 2.0 * float(rng.rand())
    ph = 2.0 * np.pi * float(rng.rand())
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * f * x + ph) + 0.25 * np.sin(4.0 * np.pi * f * x + 2.0 * ph)
    return project_feasible(h, n)


def make_start(n, ref_h, seed):
    if ref_h is not None:
        h = resample_h(ref_h, n)
        return project_feasible(h, n)
    return sine_start(n, seed)


def corr_full(h):
    n = int(len(h))
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))


def smooth_value(c, alpha):
    m = float(c.max())
    z = float(alpha) * (c - m)
    z = z - float(z.max())
    s = float(np.exp(z).sum())
    if s <= 0.0:
        return m
    return float(m + np.log(s) / float(alpha))


def softmax_weights(c, alpha):
    m = float(c.max())
    z = float(alpha) * (c - m)
    z = z - float(z.max())
    w = np.exp(z)
    s = float(w.sum())
    if s <= 0.0:
        w = np.zeros_like(c, dtype=float)
        w[int(np.argmax(c))] = 1.0
        return w
    return w / s


def lag_gradients(h, n):
    L = 2 * int(n) - 1
    G = np.zeros((int(n), L), dtype=float)
    G[:, n - 1] = 1.0 - 2.0 * h
    for m in range(1, n):
        gp = G[:, n - 1 + m]
        gp[:n - m] -= h[m:]
        gp[m:] += 1.0 - h[:n - m]
        gn = G[:, n - 1 - m]
        gn[:n - m] += 1.0 - h[m:]
        gn[m:] -= h[:n - m]
    return G


def eval_pair_at_s(c, a, s, i0, i1, i2):
    if float(s) == 0.0:
        return float(c.max())
    tmp = c + a * float(s)
    q = float(s) * float(s)
    tmp[i0] -= 2.0 * q
    tmp[i1] += q
    if i2 != i1:
        tmp[i2] += q
    return float(tmp.max())


def pair_line_search(c, G, h, p, q, n, grid=10, refine=10):
    smax = float(min(float(h[p]), 1.0 - float(h[q])))
    if smax <= 1e-11:
        return None
    a = G[:, int(q)] - G[:, int(p)]
    d = abs(int(p) - int(q))
    if d == 0:
        return None
    i0 = int(n) - 1
    i1 = i0 + d
    i2 = i0 - d

    svals = [0.0, smax]
    for k in range(1, grid + 1):
        svals.append(smax * float(k) / float(grid))

    sm1 = -0.5 * float(a[i1])
    if 1e-12 < sm1 < smax - 1e-12:
        svals.append(sm1)
    sm2 = -0.5 * float(a[i2])
    if 1e-12 < sm2 < smax - 1e-12:
        svals.append(sm2)

    top_count = min(8, int(c.size))
    top_idx = np.argsort(c)[-top_count:]
    roots_added = 0
    core = ((1.0, i1), (1.0, i2), (-2.0, i0))
    for jj in top_idx:
        j = int(jj)
        if j == i0:
            bj = -2.0
        elif j == i1 or j == i2:
            bj = 1.0
        else:
            bj = 0.0
        for bk, ik in core:
            if ik == j:
                continue
            A = bk - bj
            B = float(a[ik]) - float(a[j])
            C = float(c[ik]) - float(c[j])
            roots = []
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    roots.append(-C / B)
            else:
                D = B * B - 4.0 * A * C
                if D >= -1e-12:
                    if D < 0.0:
                        D = 0.0
                    sd = float(np.sqrt(D))
                    roots.append((-B + sd) / (2.0 * A))
                    roots.append((-B - sd) / (2.0 * A))
            for r in roots:
                if np.isfinite(r) and 1e-11 < r < smax - 1e-11:
                    svals.append(float(r))
                    roots_added += 1
                    if roots_added > 32:
                        break
            if roots_added > 32:
                break
        if roots_added > 32:
            break

    cmax0 = float(c.max())
    best_s = 0.0
    best_f = cmax0
    for s in svals:
        f = eval_pair_at_s(c, a, s, i0, i1, i2)
        if f < best_f - 1e-15:
            best_f = f
            best_s = s

    if best_s > 1e-9 and best_s < smax - 1e-9:
        span = min(0.35 * smax, 0.35 * float(best_s), 0.35 * (smax - float(best_s)))
        if span < 1e-8:
            span = max(1e-8, smax / 20.0)
        lo = max(0.0, best_s - span)
        hi = min(smax, best_s + span)
        invphi = 0.6180339887498949
        x1 = lo + (1.0 - invphi) * (hi - lo)
        x2 = lo + invphi * (hi - lo)
        f1 = eval_pair_at_s(c, a, x1, i0, i1, i2)
        f2 = eval_pair_at_s(c, a, x2, i0, i1, i2)
        for _ in range(refine):
            if f1 < f2:
                hi = x2
                x2 = x1
                f2 = f1
                x1 = lo + (1.0 - invphi) * (hi - lo)
                f1 = eval_pair_at_s(c, a, x1, i0, i1, i2)
            else:
                lo = x1
                x1 = x2
                f1 = f2
                x2 = lo + invphi * (hi - lo)
                f2 = eval_pair_at_s(c, a, x2, i0, i1, i2)
        for s in (lo, hi, x1, x2, best_s):
            f = eval_pair_at_s(c, a, s, i0, i1, i2)
            if f < best_f - 1e-15:
                best_f = f
                best_s = s

    if best_s <= 1e-12:
        return None
    return float(best_s), float(best_f)


def generate_candidates(h, Gs, c, n):
    eps = 1e-11
    cands = []
    seen = set()

    hi_limit = 15
    lo_limit = 15
    hi_idx = []
    for idx in np.argsort(Gs)[::-1]:

    for idx in np.argsort(Gs):
        if h[idx] < 1.0 - eps:

                break

    for p in hi_idx:
        hp = float(h[p])
        if hp <= eps:
            continue
        for q in lo_idx:
            if p == q:
                continue
            room = float(1.0 - h[q])
            if room <= eps:
                continue
            s = hp if hp < room else room
            score = float(Gs[p] - Gs[q])
            if score <= 1e-8:
                continue
            key = (p, q)
            if key not in seen:
                seen.add(key)
                cands.append((score * min(s, 0.08), p, q))

    top_lags = np.argsort(c)[-10:]
    for idx in top_lags:
        m = abs(int(idx) - (n - 1))
        if m == 0 or m >= n:
            continue
        cnt = n - m
        take = min(cnt, 15)

        s1 = Gs[:cnt] - Gs[m:]
        order = np.argsort(s1)[-take:][::-1]
        for ii in order:
            i = int(ii)
            score = float(s1[i])
            if score <= 1e-8:
                continue
            p = i
            q = i + m
            s = min(float(h[p]), float(1.0 - h[q]))
            if s <= eps:
                continue
            key = (p, q)
            if key not in seen:
                seen.add(key)
                cands.append((score * min(s, 0.08), p, q))

        s2 = Gs[m:] - Gs[:cnt]
        order = np.argsort(s2)[-take:][::-1]
        for ii in order:
            i = int(ii)
            score = float(s2[i])
            if score <= 1e-8:
                continue
            p = i + m
            q = i

            if s <= eps:
                continue
            key = (p, q)
            if key not in seen:
                seen.add(key)
                cands.append((score * min(s, 0.08), p, q))

    if not cands:
        hi2 = []
        for idx in np.argsort(Gs)[::-1][:8]:
            if h[idx] > eps:
                hi2.append(int(idx))
        lo2 = []
        for idx in np.argsort(Gs)[:8]:

                key = (p, q)
                if key not in seen:
                    seen.add(key)
                    cands.append((max(0.0, score) * min(s, 0.08) + 1e-6, p, q))

    return cands[:44]


def gradient_step_once(h, n, c, Gs, alpha, base=0.10):
    Gs0 = Gs - float(np.mean(Gs))
    norm = float(np.linalg.norm(Gs0))
    if norm < 1e-12:
        return None
    cmax0 = float(c.max())
    facs = (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125)
    for fac in facs:
        y = project_feasible(h - base * fac * Gs0, n)
        if float(np.linalg.norm(y - h)) < 1e-9:
            continue
        cy = corr_full(y)
        c1 = float(cy.max())
        if c1 < cmax0 - 1e-14:
            return y
    return None


def coordinate_refine(h, n, end, alpha=150.0, cand_count=24):
    h = project_feasible(h, n).copy()
    c = corr_full(h)
    cmax = float(c.max())
    best_h = h.copy()
    best_c = cmax
    stall = 0
    it = 0
    accepts_since_proj = 0

    while time.time() < end and it < 3000:
        it += 1
        c = corr_full(h)
        cmax = float(c.max())
        if cmax < best_c - 1e-15:
            best_c = cmax
            best_h = h.copy()

        w = softmax_weights(c, alpha)
        G = lag_gradients(h, n)
        Gs = G.dot(w)
        cands = generate_candidates(h, Gs, c, n)

        best_move = None
        best_f = cmax
        test_count = min(cand_count, len(cands))
        for ii in range(test_count):
            _, p, q = cands[ii]
            res = pair_line_search(c, G, h, p, q, n)
            if res is not None:
                f = res[1]
                if f < best_f - 1e-15:
                    best_f = f
                    best_move = (int(p), int(q), float(res[0]))

        if best_move is not None and best_move[2] > 1e-10:
            p, q, s = best_move
            old_p = float(h[p])
            old_q = float(h[q])
            h[p] = old_p - s
            h[q] = old_q + s
            c2 = corr_full(h)
            c2m = float(c2.max())
            if c2m > cmax + 1e-10:
                h[p] = old_p
                h[q] = old_q
            else:
                if c2m < best_c - 1e-15:
                    best_c = c2m
                    best_h = h.copy()
                accepts_since_proj += 1
                if accepts_since_proj >= 25:
                    h = project_feasible(h, n)
                    c = corr_full(h)
                    cmax = float(c.max())
                    if cmax < best_c - 1e-15:
                        best_c = cmax
                        best_h = h.copy()
                    accepts_since_proj = 0
                stall = 0
                continue

        y = gradient_step_once(h, n, c, Gs, alpha)
        if y is not None:
            cy = corr_full(y)
            cym = float(cy.max())
            if cym < cmax - 1e-14:
                h = y
                c = cy
                cmax = cym
                if cym < best_c - 1e-15:
                    best_c = cym
                    best_h = h.copy()
                stall = 0
                continue

        if stall >= 8 and stall % 4 == 0:
            state = (1234567 + it * 9176 + int(round(cmax * 1234567))) & 0x7fffffff
            rr = np.random.RandomState(state)
            scale = 0.010 + 0.006 * float(stall)
            hp = project_feasible(h + scale * rr.normal(0.0, 1.0, n), n)
            c_hp = corr_full(hp)
            c_hpmax = float(c_hp.max())
            if c_hpmax < best_c + 0.005:
                h = hp
                c = c_hp
                cmax = c_hpmax
                stall = 0
                continue

        stall += 1
        if stall >= 25:
            break

    h = project_feasible(h, n)
    c = corr_full(h)
    cmax = float(c.max())
    if cmax < best_c:
        best_c = cmax
        best_h = h.copy()
    return best_h, best_c


def refine_solution(h, n, end):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = float(corr_full(h).max())
    alphas = (50.0, 200.0, 600.0, 1800.0)
    for i, a in enumerate(alphas):
        rem = float(end - time.time())
        if rem < 10.0:
            break
        stage_end = time.time() + rem / float(len(alphas) - i)
        h2, c2 = coordinate_refine(h, n, stage_end, alpha=a, cand_count=24)
        if c2 < best_c:
            best_c = float(c2)
            best_h = h2.copy()
        h = best_h
    return best_h, best_c
