import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    # Project to [0,1]^n with sum = n/2
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def smooth_max_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    exp_shifted = np.exp(shifted)
    logsum = m + np.log(exp_shifted.sum()) / alpha
    p = exp_shifted / exp_shifted.sum()
    # Compute gradient
    grad = np.zeros(n)
    m_range = np.arange(-(n-1), n)
    for idx_m in range(2*n-1):
        if p[idx_m] < 1e-10:

        t_lo = max(0, i_lo + s_val)
        t_hi = min(n-1, i_hi + s_val)

        t_lo2 = max(0, i_lo, -s_val)

        if t_lo2 <= t_hi2:
            ts = np.arange(t_lo2, t_hi2+1)
            grad[ts] -= pm * h[ts + s_val]
    return logsum, grad

    def obj(x):

        m = c.max()
        shifted = alpha * (c - m)
        return m + np.log(np.exp(shifted).sum()) / alpha + 100.0 * (x.sum() - target)**2
    
    def grad(x):

        g += 200.0 * (x.sum() - target)
        return g

    return project_feasible(res.x, n)

def binary_local_search(h_init, n, deadline, max_sweeps=100):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    m_range = np.arange(-(n-1), n)
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])

        best_delta_c = 0
        best_pair = None
        # Try all pairs
        for i in zeros_list:

                imn = i - m_range

                delta[msk] -= h[imn[msk]]
                jm = j + m_range

                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
