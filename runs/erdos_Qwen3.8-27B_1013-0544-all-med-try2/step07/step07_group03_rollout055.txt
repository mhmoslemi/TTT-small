def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break

            if s < 1e-14:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def sm_obj_grad(h, alpha, dx):
    n = len(h)
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    e = np.exp(shifted - sm)
    w = e / e.sum()
    f = m + (sm + np.log(e.sum())) / alpha
    grad = np.zeros(n)
    for l in range(-(n - 1), n):
        c = w[n - 1 + l] * dx
        if l >= 0:
            grad[:n - l] += c * (1.0 - h[l:])
            grad[l:] -= c * h[:n - l]
        else:
            ll = -l
            grad[ll:] += c * (1.0 - h[:n - ll])
            grad[:n - ll] -= c * h[ll:]
    return f, grad

def cont_opt(h, n, alpha, maxiter, deadline, pen=1e5):
    dx = 2.0 / n
    def fun(x):
        f, g = sm_obj_grad(x, alpha, dx)
        s = x.sum() - n / 2.0
        return f + pen * s * s, g + 2.0 * pen * s
    bounds = [(0.0, 1.0)] * n

    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        best_swap = None
        best_nc = cur_c
        for i in zeros:

            for j in ones:
                d = d_template.copy()
                im = i + m_range
