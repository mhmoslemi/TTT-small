def slsqp_step(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
        return m + np.log(np.sum(np.exp(s))) / alpha
    def obj_jac(x):
        # Gradient of smooth max
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        if mx > 500:
            weights = np.exp(s - mx) / np.sum(np.exp(s - mx))
        else:
            weights = np.exp(s) / np.sum(np.exp(s))
        # dc/dx for each lag
        grad = np.zeros(n)
        for k in range(2*n-1):
            lag = k - (n-1)
            if lag == 0:
                g = 1 - 2*x
            elif lag > 0 and lag < n:
                g = np.zeros(n)
                g[:n-lag] -= x[lag:]
                g[lag:] += (1-x[:n-lag])
            elif lag < 0 and abs(lag) < n:
                g = np.zeros(n)
                m2 = abs(lag)
                g[:n-m2] += (1-x[m2:])
                g[m2:] -= x[:n-m2]
            else:
                continue
            grad += weights[k] * dx * g
        return grad
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda: np.ones(n)}]
    try:

            if tot < 1e-15:

    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        w = np.exp(s - mx)
        w /= w.sum()
    else:
        w = np.exp(s)
        w /= w.sum()

    for k in range(2 * n - 1):

            g = np.zeros(n)
            g[:n-m2] += (1.0 - x[m2:])
            g[m2:] -= x[:n-m2]
            grad += w[k] * dx * g

            best_c = c_new
            best_h = h_new
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = min(250, max(60, (deadline - t0) * 0.25))
        best_h, best_c = optimize_at_n(h_init, n_init, t0 + alloc, rng)
    res_list = [200, 250, 300, 350, 400]
    for n_try in res_list:
        if time.time() > deadline - 120:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 70:
            break
        alloc = min(110, remaining * 0.3)
        if best_h is not None and len(best_h) != n_try:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if best_h is not None and time.time() < deadline - 100:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            scales = [0.008, 0.015, 0.025, 0.04, 0.06]
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 40) * 0.3)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_new = slsqp_opt(best_h, n, 2000000, maxiter=1500)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h_new, c_new = local_search(best_h, n, deadline - 3)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
