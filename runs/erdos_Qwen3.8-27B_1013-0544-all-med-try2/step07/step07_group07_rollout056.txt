```python
            if tot < 1e-14:

    m = values.max()
    shifted = alpha * (values - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def subgradient_max(h, n, alpha=10000):
    """Compute subgradient of max_k c_k(h) using softmax approximation."""
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    # gradient of c_k w.r.t. h
    # c_k = dx * sum_m h[m] * (1 - h[m+k])
    # d(c_k)/d(h[i]) = dx * [(1-h[i+k]) - h[i-k]] (with boundary handling)

        if weights[k_idx] < 1e-10:
            continue
        lag = k_idx - (n-1)
        g = np.zeros(n)
        if lag == 0:
            g[:] = 1.0 - 2.0*h
        elif lag > 0:
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] += -h[:n-lag]
        else:

            g[:n-lag_abs] += (1.0 - h[lag_abs:])
            g[lag_abs:] += -h[:n-lag_abs]

    return grad

def nesterov_projected(h, n, deadline, alpha=10000):
    """Nesterov accelerated projected gradient for min max_k c_k."""
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    t_k = 1.0
    step = 0.001
    
    for it in range(5000):
        if time.time() > deadline:
            break
        # Nesterov extrapolation
        t_next = (1 + np.sqrt(1 + 4*t_k*t_k)) / 2
        beta = (t_k - 1) / t_next
        h_extrap = h_curr + beta * (h_curr - h_prev)
        h_extrap = project_feasible(h_extrap, n)

        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h_prev = h_curr
        h_curr = h_new
        t_k = t_next
        step *= 0.999  # slow decay
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= (1 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_max(vals, alpha):
    m = vals.max()
    s = alpha * (vals - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def slsqp_min(h, n, alpha, maxiter=5000, ftol=1e-16):
    dx = 2.0/n
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        return smooth_max(c, alpha)
    def obj_grad(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        if mx > 500:
            w = np.exp(s - mx)
        else:
            w = np.exp(s)
        w /= w.sum()
        # grad of c[lag] w.r.t. x
        grad = np.zeros(n)
        for ki in range(2*n-1):
            wk = w[ki]
            if wk < 1e-12:
                continue
            lag = ki - (n-1)
            if lag == 0:
                grad += wk * dx * (1 - 2*x)
            elif lag > 0:
                g = np.zeros(n)
                g[lag:] += (1-x[:n-lag])
                g[:n-lag] -= x[lag:]
                grad += wk * dx * g
            else:
                la = -lag
                g = np.zeros(n)
                g[:n-la] += (1-x[la:])
                g[la:] -= x[:n-la]
                grad += wk * dx * g
        return grad
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
```

```python
def local_search_fast(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for step in steps:
        if time.time() > deadline:
            break
        for _ in range(30):  # max 30 improvements per step size
            if time.time() > deadline:
                break
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n-1)
            if lag == 0:
                break
            m = abs(lag)
            
            # Find the best pair to transfer
            if lag > 0:
                # src = m_idx + m, dst = m_idx
                # Want h[src] > h[dst] and large difference
                diffs = h[m:] - h[:n-m]
                # Only consider positive diffs (h[src] > h[dst])
                valid = np.where(diffs > 0.001)[0]
                if len(valid) == 0:
                    break
                # Pick the one with largest diff
                best_idx = valid[np.argmax(diffs[valid])]
                src = best_idx + m
                dst = best_idx
            else:
                # src = m_idx - m, dst = m_idx, m_idx from m to n-1
                diffs = h[:n-m] - h[m:]
                valid = np.where(diffs > 0.001)[0]
                if len(valid) == 0:
                    break
                best_idx = valid[np.argmax(diffs[valid])]
                src = best_idx
                dst = best_idx + m
            
            delta = min(step, (h[src]-h[dst])*0.5, h[src], 1.0-h[dst])
            if delta < 1e-10:
                break
            h[src] -= delta
            h[dst] += delta
            new_c = compute_c5(h, n)
            if new_c >= best_c - 1e-15:
                h[src] += delta
                h[dst] -= delta
                break
    
    return h, best_c
```
