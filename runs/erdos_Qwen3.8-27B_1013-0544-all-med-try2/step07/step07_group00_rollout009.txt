def project_feasible(h, n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() == 0:
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 0) & (h < 1)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_max = float(c.max())
    shifted = alpha * (c - c_max)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        s = m - (n - 1)
        wm = w[m] * dx
        if s >= 0:
            nv = n - s
            grad[:nv] += wm * (1.0 - h[s:])
            grad[s:] += wm * (-h[:nv])
        else:
            ss = -s
            nv = n - ss
            grad[ss:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[ss:])
    return c_max, w, grad

def _obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    val = m + (sh.max() + np.log(np.sum(np.exp(sh)))) / alpha
    s = x.sum() - n / 2.0
    val += mu * s * s
    return float(val)

def _jac(x, n, alpha, mu):
    _, _, g = smooth_max_and_grad(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_run(h, n, deadline, alpha_list, mu_list):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alpha_list):
        if time.time() > deadline - 5:
            break
        remaining = deadline - time.time() - 5
        if remaining < 5:
            break
        mu = mu_list[i] if i < len(mu_list) else 0.0
        maxiter = min(800, max(20, int(remaining * 8)))
        try:
            res = minimize(_obj, best_h, jac=_jac, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': maxiter * 50, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def adam_run(h, n, deadline, alpha, lr_init=0.05):

    m_vec = np.zeros(n)
    v_vec = np.zeros(n)

    lr = lr_init

        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * grad * grad
        h_new = h - lr_t * m_vec / (np.sqrt(v_vec) + eps)
        h_new = project_feasible(h_new, n)

def coord_polish(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(3):

                break
