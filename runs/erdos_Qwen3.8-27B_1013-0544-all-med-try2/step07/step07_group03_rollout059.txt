    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for sweep in range(n_sweeps):

                break

            #   i=j: h[j]*(1-h[j-s_k]) -> coeff of h[j] is (1-h[j-s_k])*dx
            #   i=j+s_k (if valid): h[j+s_k]*(1-h[j]) -> coeff of h[j] is -h[j+s_k]*dx

        # g_k(x) = sum_i h'[i](1-h'[i-s_k])*dx where h'[j]=x, rest same
        # = g_k(original) + delta
        # delta for k: dx * [(x - h[j])*(1-h[j-s_k]) - h[j+s_k]*(x-h[j])]  for s_k != 0
        # for s_k = 0: dx * [(x-x²) - (h[j]-h[j]²)]
        delta = np.zeros(2 * n - 1)
        dh = x - h[j]
        for k in range(2 * n - 1):
            s_k = k - (n - 1)
            if s_k == 0:
                delta[k] = dh * (1 - x - h[j]) * dx
            else:
                jm = j - s_k
                jp = j + s_k
                h_mv = h[jm] if 0 <= jm < n else 0.0
                h_pv = h[jp] if 0 <= jp < n else 0.0
                delta[k] = dh * (1 - h_mv - h_pv) * dx
        new_g = g + delta
        val = new_g.max()
        if val < best_val:
            best_val = val
            best_x = x
    return best_x

def coordinate_descent(h, n, deadline, max_sweeps=5000, order='random', seed=42):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    n_iter = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if order == 'random':
            perm = rng.permutation(n)
        else:
            perm = range(n)
        for j in perm:
            if time.time() > deadline:
                break
            g = np.correlate(h, 1 - h, mode='full') * dx
            x_new = solve_1d_subproblem(h, j, n, g)
            if abs(x_new - h[j]) > 1e-12:
                h[j] = x_new
        c5 = compute_c5(h, n)
        if c5 < best_c - 1e-12:
            best_c = c5
            best_h = h.copy()
        n_iter += 1
    return best_h, best_c

def pg_smooth(h, n, alphas, max_iters, deadline):
    """Projected gradient with log-sum-exp smoothing."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        h_cur = h.copy()
        prev_f = compute_c5(h_cur, n)
        lr = 2.0 / (alpha * dx)
        for it in range(max_iters):
            if time.time() > deadline:
                return best_h, best_c
            g = np.correlate(h_cur, 1 - h_cur, mode='full') * dx
            m = g.max()
            shifted = alpha * (g - m)
            max_s = shifted.max()
            exp_s = np.exp(shifted - max_s)
            weights = exp_s / exp_s.sum()
            # Gradient
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                w = weights[k]
                if w < 1e-10:
                    continue
                s = k - (n - 1)
                jm = np.arange(n) - s
                vm = (jm >= 0) & (jm < n)
                jp = np.arange(n) + s
                vp = (jp >= 0) & (jp < n)
                c = np.zeros(n)
                c[vm] = 1.0 - h_cur[jm[vm]]
                c[vp] -= h_cur[jp[vp]]
                grad += w * c * dx
            # Backtracking
            for bt in range(15):
                h_try = project_feasible(h_cur - lr * grad, n)
                f_try = compute_c5(h_try, n)
                if f_try <= prev_f + 1e-12:
                    break
                lr *= 0.5
            else:
                break
            if f_try < best_c:
                best_c = f_try
                best_h = h_try.copy()
            if abs(f_try - prev_f) < 1e-14:
                break
            h_cur = h_try
            prev_f = f_try
            lr = min(lr * 1.1, 2.0 / (alpha * dx))
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_time = budget_s - 10
    deadline = t0 + total_time
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(400, 0.5)
    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Coordinate descent from initial (40% of time)
    p1 = t0 + 0.35 * total_time
    h1, c1 = coordinate_descent(h0, n, p1, max_sweeps=3000, order='random', seed=seed)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()
    
    # Phase 2: Projected gradient refinement (25%)
    p2 = t0 + 0.60 * total_time
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    h2, c2 = pg_smooth(best_h, n, alphas, 300, p2)
    if c2 < best_c:
        best_c, best_h = c2, h2.copy()
    
    # Phase 3: More coordinate descent from refined (20%)
    p3 = t0 + 0.80 * total_time
    h3, c3 = coordinate_descent(best_h, n, p3, max_sweeps=5000, order='random', seed=seed + 1)
    if c3 < best_c:
        best_c, best_h = c3, h3.copy()
    
    # Phase 4: Try binary + continuous (10%)
    p4 = t0 + 0.92 * total_time
    hb = to_binary(best_h, n)
    h4, c4 = coordinate_descent(hb, n, p4, max_sweeps=2000, order='random', seed=seed + 2)
    if c4 < best_c:
        best_c, best_h = c4, h4.copy()
    
    # Phase 5: Final polish with PG (8%)
    p5 = deadline - 2
    alphas_final = [50000, 200000, 1000000]
    h5, c5 = pg_smooth(best_h, n, alphas_final, 200, p5)
    if c5 < best_c:
        best_c, best_h = c5, h5.copy()
    
    # Final projection
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
