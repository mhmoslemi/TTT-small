import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def c5_obj_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    target = n / 2.0
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    exp_s = np.exp(shifted - shifted.max())
    w = exp_s / exp_s.sum()
    f = m + np.log(exp_s.sum()) / alpha + penalty * (x.sum() - target) ** 2
    
    grad = np.zeros(n)
    lags = np.arange(-(n - 1), n)
    for t in range(2 * n - 1):
        l = lags[t]
        wt = w[t] * dx
        if wt < 1e-18:
            continue
        lo1 = max(0, -l)
        hi1 = min(n, n - l)
        if hi1 > lo1:
            grad[lo1:hi1] += wt * (1.0 - x[lo1 + l:hi1 + l])
        lo2 = max(0, l)
        hi2 = min(n, n + l)
        if hi2 > lo2:
            grad[lo2:hi2] -= wt * x[lo2 - l:hi2 - l]
    grad += 2.0 * penalty * (x.sum() - target)
    return f, grad

        return f

        return g

                      bounds=bounds,

                              'maxfun': maxiter * 10})

    h_b[idx[:n // 2]] = 1.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def c5_obj_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    target = n / 2.0
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    exp_s = np.exp(shifted - shifted.max())
    w = exp_s / exp_s.sum()
    f = m + np.log(exp_s.sum()) / alpha + penalty * (x.sum() - target) ** 2
    grad = np.zeros(n)
    lags = np.arange(-(n - 1), n)
    for t in range(2 * n - 1):
        l = lags[t]
        wt = w[t] * dx
        if wt < 1e-18:
            continue
        lo1 = max(0, -l)
        hi1 = min(n, n - l)
        if hi1 > lo1:
            grad[lo1:hi1] += wt * (1.0 - x[lo1 + l:hi1 + l])
        lo2 = max(0, l)
        hi2 = min(n, n + l)
        if hi2 > lo2:
            grad[lo2:hi2] -= wt * x[lo2 - l:hi2 - l]
    grad += 2.0 * penalty * (x.sum() - target)
    return f, grad

def lbfgs_step(h_init, n, alpha, penalty, maxiter, deadline):
    def obj_func(x):
        f, _ = c5_obj_grad(x, n, alpha, penalty)
        return f
    def grad_func(x):
        _, g = c5_obj_grad(x, n, alpha, penalty)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h_init, jac=grad_func, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': maxiter * 15, 'maxls': 30})
        out = project_feasible(res.x, n)
        return out
    except Exception:
        return project_feasible(h_init, n)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        found = False
        for i in zeros_list:
            if time.time() > deadline:
                return best_h, best_c
            best_j = -1
            best_nc = cur_c - 1e-15
            for j in ones_list:

                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range

                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx

            if best_j >= 0:

                corr += delta
                cur_c = corr.max() * dx
                ones_list.remove(best_j)

                    best_c = cur_c
