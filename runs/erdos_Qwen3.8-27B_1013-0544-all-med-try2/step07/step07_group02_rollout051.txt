def coord_descent(h, n, max_sweeps=500, deadline=None):

    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range

            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9

            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(50):

                    continue
                lb_x = 0.0
                if neg_B.any():

                ub_x = 1.0
                if pos_B.any():

                if lb_x <= ub_x + 1e-12:

            lb_x = 0.0

    p2 = t0 + 0.40 * total
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000, 100000000]
    h2 = best_h.copy()
    for a in alphas:
        if time.time() > p2 - 10:
            break
        h2 = slsqp_smooth(h2, n, a, 4000, 1e-16)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 3: Coord descent polish
    p3 = t0 + 0.50 * total
    h3, _ = coord_descent(best_h, n, max_sweeps=200, deadline=p3)
    h3 = project_feasible(h3, n)
    c3 = compute_c5(h3, n)
    if c3 < best_c:
        best_c, best_h = c3, h3.copy()

    # Phase 4: High-alpha SLSQP
    p4 = t0 + 0.70 * total
    for a in [50000000, 200000000, 1000000000]:
        if time.time() > p4 - 8:
            break
        h4 = slsqp_smooth(best_h, n, a, 6000, 1e-18)
        c4 = compute_c5(h4, n)
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()

    # Phase 5: Diverse restarts with varying noise
    p5 = t0 + 0.92 * total
    noise_levels = [0.001, 0.003, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3]
    for nl in noise_levels:
        if time.time() > p5 - 10:
            break
        h_r = best_h + nl * rng.randn(n)
        h_r = project_feasible(h_r, n)
        h_r, _ = coord_descent(h_r, n, max_sweeps=50, deadline=p5 - 8)
        h_r = project_feasible(h_r, n)
        for a in [100000, 1000000, 10000000, 100000000]:
            if time.time() > p5 - 4:
                break
            h_r = slsqp_smooth(h_r, n, a, 2000, 1e-15)
            c_r = compute_c5(h_r, n)
            if c_r < best_c:
                best_c, best_h = c_r, h_r.copy()

    # Phase 6: Try different resolution
    if time.time() < dl - 25:
        for n_try in [600, 300]:
            if time.time() > dl - 20:
                break
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, n_try)
            h_t = np.interp(xn, xo, best_h)
            h_t = project_feasible(h_t, n_try)
            h_t, _ = coord_descent(h_t, n_try, max_sweeps=60, deadline=dl - 18)
            h_t = project_feasible(h_t, n_try)
            for a in [50000, 500000, 5000000, 50000000]:
                if time.time() > dl - 5:
                    break
                h_t = slsqp_smooth(h_t, n_try, a, 2000, 1e-15)
                c_t = compute_c5(h_t, n_try)
                if c_t < best_c:
                    best_c, best_h, n = c_t, h_t.copy(), n_try

    # Phase 7: Final final SLSQP
    if time.time() < dl - 3:
        h_f = slsqp_smooth(best_h, n, 500000000, 5000, 1e-18)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c, best_h = c_f, h_f.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
