    T_min = 0.002
    cooling = 0.9995
    while T > T_min and time.time() < deadline:
        for _ in range(50):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or (i in idx) == (j in idx):
                continue
            idx.discard(i)
            idx.add(j)
            h[i] = 0.0
            h[j] = 1.0
            new_c = compute_c5(h, n)
            if new_c < current_c or rng.random() < np.exp((current_c - new_c) / T):
                current_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                idx.discard(j)
                idx.add(i)
                h[j] = 0.0
                h[i] = 1.0
        T *= cooling
    return best_h, best_c

def coord_descent(h, n, deadline, radius=25):
    best_c = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]
    for step in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            g = smooth_max_grad(h, n, 5000)
            order = np.argsort(g)
            for idx_i in range(n):
                if time.time() > deadline:
                    return h, best_c
                i = order[idx_i]
                lo = max(0, idx_i - radius)
                hi = min(n, idx_i + radius + 1)
                for j_idx in range(lo, hi):
                    j = order[j_idx]
                    if i == j or abs(g[i] - g[j]) < 1e-10:
                        continue
                    if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                        h[i] += step
                        h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= step
                            h[j] += step
                    if time.time() > deadline:
                        return h, best_c
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 400
        h_init = np.full(n_init, 0.5)

    best_h = waterfill_project(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS-B on parent
    t1 = min(t0 + budget_s * 0.28, deadline - budget_s * 0.45)
    best_h, best_c = lbfgs_optimize(best_h, best_n, t1)

    # Phase 2: Try other grid sizes
    for n_try in [350, 450, 600, 700]:
        if time.time() > deadline - budget_s * 0.30:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = waterfill_project(h_interp, n_try)
        alloc = min(70, (deadline - time.time() - 80) * 0.4)
        if alloc < 20:
            continue
        h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Phase 3: Binary annealing + refinement
    if time.time() < deadline - budget_s * 0.20:
        alloc = min(budget_s * 0.12, deadline - time.time() - 100)
        if alloc > 25:
            h_bin, c_bin = binary_anneal(best_n, time.time() + alloc, rng)
            if c_bin < best_c + 0.01:
                h_ref = waterfill_project(h_bin, best_n)
                h_ref, c_ref = lbfgs_optimize(h_ref, best_n,
                                             min(time.time() + 80, deadline - 40))
                if c_ref < best_c:
                    best_c = c_ref
                    best_h = h_ref

    # Phase 4: Diverse restarts
    for r in range(4):
        if time.time() > deadline - 50:
            break
        noise_scale = 0.08 * (1 - r * 0.15)
        h_r = best_h + rng.randn(best_n) * noise_scale * (1.0 - 2.0 * np.abs(best_h - 0.5))
        h_r = waterfill_project(h_r, best_n)
        alloc = min(45, (deadline - time.time() - 30) * 0.35)
        if alloc < 12:
            break
        h_r, c_r = lbfgs_optimize(h_r, best_n, time.time() + alloc)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r

    # Phase 5: Coordinate descent polish
    if time.time() < deadline - 10:
        best_h, best_c = coord_descent(best_h, best_n, min(time.time() + 35, deadline - 5))

    # Phase 6: Final high-alpha L-BFGS-B
    if time.time() < deadline - 3:
        best_h, best_c = lbfgs_optimize(best_h, best_n, deadline - 1)

    best_h = waterfill_project(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
