        alloc = min(80, (deadline - time.time()) * 0.12)
        if alloc < 25:
            continue
        h_new, c_new = sa_optimize(h_interp, n_try, time.time() + alloc, seed=seed + n_try)
        if time.time() < deadline - 80:
            ht = slsqp_polish(h_new, n_try, 20000, maxiter=400)
            c2 = compute_c5(ht, n_try)
            if c2 < c_new:
                h_new, c_new = ht, c2
        if time.time() < deadline - 55:
            h_new, c_new = pair_search(h_new, n_try, deadline - 55,
                                       [0.001, 0.0005], radius=40)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 6: Final SA on best solution
    n = len(best_h)
    if time.time() < deadline - 40:
        alloc = min(40, (deadline - time.time()) - 30)
        if alloc > 15:
            h2, c2 = sa_optimize(best_h, n, time.time() + alloc, seed=seed + 999)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 7: Final SLSQP + pair search
    if time.time() < deadline - 10:
        ht = slsqp_polish(best_h, n, 50000, maxiter=300)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    if time.time() < deadline - 5:
        best_h, best_c = pair_search(best_h, n, deadline - 3,
                                     [0.0005, 0.0002, 0.0001], radius=30)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
