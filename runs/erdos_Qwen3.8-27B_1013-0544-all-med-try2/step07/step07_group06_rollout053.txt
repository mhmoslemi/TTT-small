```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            if tot > 0:
                h *= (target / tot)
            else:
                h.fill(0.5)
    return np.clip(h, 0.0, 1.0)

def exact_pair_step(h, n, i, j, n_evals=7):
    """Optimize pair (i,j) with fixed sum using 1D search."""
    s = h[i] + h[j]
    t_min = max(0.0, s - 1.0)
    t_max = min(1.0, s)
    if t_min >= t_max - 1e-15:
        return False, 0.0
    
    old_i = h[i]
    old_j = h[j]
    
    # Evaluate at s
    # Use a set of candidate points
    if n_evals <= 4:
        candidates = np.array([t_min, t_max, (t_min+t_max)/2, old_i])
    else:
        # Log-spaced or uniform
        candidates = np.linspace(t_min, t_max, n_evals)
        # Also add the current value
        candidates = np.concatenate([candidates, [old_i]])
        candidates = np.unique(candidates)
    
    best_t = old_i
    best_c = compute_c5(h, n)
    
    for t in candidates:
        h[i] = t
        h[j] = s - t
        c = compute_c5(h, n)
        if c < best_c - 1e-15:
            best_c = c
            best_t = t
    
    h[i] = best_t
    h[j] = s - best_t
    
    improved = best_t != old_i
    return improved, best_c

def pair_pass(h, n, n_pairs, deadline, rng=None):
    """Multiple random pair optimization steps."""
    if rng is None:
        rng = np.random.RandomState()
    best_c = compute_c5(h, n)
    improved_count = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        imp, c = exact_pair_step(h, n, i, j)
        if imp:
            best_c = c
            improved_count += 1
    return h, best_c, improved_count

def smooth_max_obj(h, n, alpha):
    c = np.correlate(h, 1-h, mode='full') * (2.0/n)
    m = c.max()
    sh = alpha * (c - m)
    return m + np.log(np.sum(np.exp(sh - sh.max()))) / alpha + sh.max()/alpha

def smooth_max_grad(h, n, alpha):
    c = np.correlate(h, 1-h, mode='full') * (2.0/n)
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2*n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad
```

```python
def binary_swap_pass(h, n, deadline, rng=None):
    """For 0-1 (or near 0-1) h, try all swaps and accept improvements."""
    if rng is None:
        rng = np.random.RandomState()
    best_c = compute_c5(h, n)
    zeros = np.where(h < 0.5)[0]
    ones = np.where(h >= 0.5)[0]
    n_swaps = 0
    while time.time() < deadline:
        improved = False
        # Try random swaps
        for _ in range(20000):  # number of swap attempts per round
            if time.time() > deadline:
                return h, best_c
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            # Swap: h[i] 1-h[i], h[j] 1-h[j]
            # For exact 0-1: h[i] becomes 1, h[j] becomes 0
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                improved = True
                n_swaps += 1
            else:
                h[i] = 1.0 - h[i]
                h[j] = 1.0 - h[j]
        if not improved:
            break
    return h, best_c
```
