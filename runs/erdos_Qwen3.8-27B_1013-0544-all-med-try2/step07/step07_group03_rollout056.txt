    """Compute gradient of the smooth C5 objective."""
    dx = 2.0 / n
    
    # Compute all C values
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    C = corr * dx  # C_m for each lag
    
    C_max = C.max()
    
    # Softmax weights
    shifted = alpha * (C - C_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 0)  # values >= 0 after shift, but we subtract max
    # Actually shifted <= 0 always since C_m <= C_max
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # All underflowed, use one-hot on max
        w = np.zeros(len(C))
        w[C == C_max] = 1.0 / (C == C_max).sum()
    else:
        w /= w_sum
    
    # Now compute gradient: df/dh[j] = sum_m w_m * dC_m/dh[j]
    # dC_m/dh[j] = dx * [(1-h[j-m]) if 0<=j-m<n else 0] + [h[j+m] if 0<=j+m<n else 0]
    # where m = lag_index - (n-1)
    
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)  # m values
    
    for mi, m in enumerate(lags):
        wm = w[mi]
        if wm < 1e-15:
            continue
        # For each j, dC_m/dh[j] = dx * [(1-h[j-m]) if 0<=j-m<n] + [h[j+m] if 0<=j+m<n]
        # j ranges from 0 to n-1
        # j-m in [0,n-1] means j in [m, m+n-1], intersect [0,n-1]
        # j+m in [0,n-1] means j in [-m, n-1-m], intersect [0,n-1]
        
        # First term: (1-h[j-m]) for valid j
        if m >= 0:
            j_start, j_end = m, n
        else:
            j_start, j_end = 0, n + m
        if j_end > j_start:
            grad[j_start:j_end] += wm * dx * (1.0 - h[j_start - m:j_end - m])
        
        # Second term: h[j+m] for valid j
        if m >= 0:
            j_start2, j_end2 = 0, n - m
        else:
            j_start2, j_end2 = -m, n
        if j_end2 > j_start2:
            grad[j_start2:j_end2] += wm * dx * h[j_start2 + m:j_end2 + m]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def _eq_fun(x):
    return x.sum() - x.shape[0] / 2.0

def _eq_jac(x):
    return np.ones(x.shape[0])

def c5_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    Cmax = C.max()
    shifted = alpha * (C - Cmax)
    shifted = np.minimum(shifted, 0.0)
    val = Cmax + np.log(np.exp(shifted).sum()) / alpha
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(C))
        w[np.argmax(C)] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    n1 = n - 1
    for mi in range(len(C)):
        wm = w[mi]
        if wm < 1e-14:
            continue
        m = mi - n1
        jlo = max(0, m)
        jhi = min(n, m + n)
        if jhi > jlo:
            grad[jlo:jhi] += wm * (1.0 - h[jlo - m:jhi - m])
        jlo2 = max(0, -m)
        jhi2 = min(n, n - m)
        if jhi2 > jlo2:
            grad[jlo2:jhi2] -= wm * h[jlo2 + m:jhi2 + m]
    grad *= dx
    return val, grad

def slsqp_opt(h, n, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    def fun(x):
        return c5_obj_grad(x, n, alpha)[0]
    def jac(x):
        return c5_obj_grad(x, n, alpha)[1]
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b
