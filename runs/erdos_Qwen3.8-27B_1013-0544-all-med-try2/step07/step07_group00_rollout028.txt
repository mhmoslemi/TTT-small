```python
import numpy as np
from scipy.optimize import minimize
import time

class _Budget(Exception):
    pass

_CTX = {}
_CB_STATE = None

def _get_ctx(n):
    n = int(n)
    c = _CTX.get(n)
    if c is not None:
        return c
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    valid_idx = np.concatenate((np.arange(n), np.arange(N - n + 1, N)))
    lag_vals = np.concatenate((np.arange(n), -np.arange(n - 1, 0, -1)))
    c = {
        'n': n,
        'N': N,
        'valid_idx': valid_idx,
        'lag_vals': lag_vals,
        'target': 0.5 * float(n),
        'bounds': [(0.0, 1.0)] * n,
    }
    _CTX[n] = c
    return c

def _c5_vals(h, ctx):
    N = ctx['N']
    fh = np.fft.rfft(h, N)
    fb = np.fft.rfft(1.0 - h, N)
    return np.fft.irfft(fh.conj() * fb, N)[ctx['valid_idx']]

def compute_c5(h, n):
    n = int(n)
    if not np.all(np.isfinite(h)):
        return float('inf')
    ctx = _get_ctx(n)
    vals = _c5_vals(h, ctx)
    m = float(np.max(vals))
    if m < 0.0:
        m = 0.0
    return m * (2.0 / float(n))

def compute_c5_direct(h, n):
    n = int(n)
    if n <= 0:
        return float('inf')
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / float(n)))

def finalize_sum(h, n):
    target = 0.5 * float(n)
    for _ in range(10):
        res = target - float(h.sum())
        if abs(res) < 1e-13:
            break
        mask = (h > 1e-7) & (h < 1.0 - 1e-7)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += res / float(cnt)
        elif res > 0.0:
            i = int(np.argmin(h))
            v = h[i] + res
            if v <= 1.0:
                h[i] = v
        else:
            i = int(np.argmax(h))
            v = h[i] + res
            if v >= 0.0:
                h[i] = v
    return np.clip(h, 0.0, 1.0)

def project_feasible(y, n):
    n = int(n)
    y = np.asarray(y, dtype=float).reshape(-1)
    if y.size != n:
        tmp = np.full(n, 0.5, dtype=float)
        m = min(y.size, n)
        tmp[:m] = y[:m]
        y = tmp
    if not np.all(np.isfinite(y)):
        y = np.full(n, 0.5, dtype=float)
    target = 0.5 * float(n)
    hc = np.clip(y, 0.0, 1.0)
    if abs(float(hc.sum()) - target) < 1e-10 and np.all(y >= -1e-9) and np.all(y <= 1.0 + 1e-9):
        return finalize_sum(hc, n)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if abs(val - target) < 1e-13:
            lo = hi = mid
            break
        if val > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    return finalize_sum(h, n)

def _obj_unpen(x, ctx, alpha, mu):
    vals = _c5_vals(x, ctx)
    m = float(np.max(vals))
    if m < 0.0:
        m = 0.0
    if alpha <= 0.0:
        obj = m
    else:
        z = alpha * (vals - m)
        zmax = float(np.max(z))
        wsum = float(np.exp(z - zmax).sum())
        obj = m + (zmax + np.log(wsum)) / alpha
    if mu != 0.0:
        s = float(x.sum()) - ctx['target']
        obj += mu * s * s
    return float(obj)

def _grad_unpen(x, ctx, alpha, mu):
    n = ctx['n']
    N = ctx['N']
    vals = _c5_vals(x, ctx)
    m = float(np.max(vals))
    if m < 0.0:
        m = 0.0
    if alpha <= 0.0:
        w = np.ones_like(vals) / float(vals.size)
    else:
        z = alpha * (vals - m)
        zmax = float(np.max(z))
        w = np.exp(z - zmax)
        w /= w.sum()
    active = w > 1e-12
    cnt = int(active.sum())
    g = np.zeros(n, dtype=float)
    if cnt <= 48:
        lags = ctx['lag_vals']
        for ai in np.nonzero(active)[0]:
            k = int(lags[ai])
            ww = float(w[ai])
            if k >= 0:
                a = n - k
                if a > 0:
                    g[k:k + a] += ww * (1.0 - x[:a])
                    g[:a] -= ww * x[k:k + a]
            else:
                a2 = -k
                a = n - a2
                if a > 0:
                    g[:a] += ww * (1.0 - x[a2:n])
                    g[a2:n] -= ww * x[:a]
    else:
        W = np.zeros(N, dtype=float)
        W[ctx['valid_idx']] = w
        W_rev = np.empty(N, dtype=float)
        W_rev[0] = W[0]
        if N > 1:
            W_rev[1:] = W[:0:-1]
        b = 1.0 - x
        g1 = np.fft.irfft(np.fft.rfft(W, N) * np.fft.rfft(b, N), N)[:n]
        g2 = np.fft.irfft(np.fft.rfft(W_rev, N) * np.fft.rfft(x, N), N)[:n]
        g = g1 - g2
    if mu != 0.0:
        g += 2.0 * mu * (float(x.sum()) - ctx['target'])
    return g

def _obj_pen_timed(x, ctx, alpha, mu, deadline):
    if time.time() > deadline:
        raise _Budget()
    return _obj_unpen(x, ctx, alpha, mu)

def _grad_pen_timed(x, ctx, alpha, mu, deadline):
    if time.time() > deadline:
        raise _Budget()
    return _grad_unpen(x, ctx, alpha, mu)

def _slsqp_eq(x, n):
    return float(x.sum()) - 0.5 * float(n)

def _slsqp_jac(x, n):
    return np.ones((1, int(n)), dtype=float)

def _lbfgs_callback(Xk):
    global _CB_STATE
    st = _CB_STATE
    if st is None:
        return
    st['count'] += 1
    ev = 10 if st['n'] <= 800 else 25
    if st['count'] % ev == 0:
        try:
            n = int(st['n'])
            h = project_feasible(np.asarray(Xk, dtype=float), n)
            c = compute_c5(h, n)
            if np.isfinite(c) and c < st['c']:
                st['h'] = h
                st['c'] = c
        except Exception:
            pass

def lbfgs_call(h, n, ctx, alpha, mu, deadline, maxiter):
    global _CB_STATE
    n = int(n)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if maxiter <= 0 or not np.isfinite(best_c):
        return best_h, best_c
    state = {'h': best_h, 'c': best_c, 'count': 0, 'n': n}
    _CB_STATE = state
    try:
        options = {
            'maxiter': int(maxiter),
            'ftol': 1e-15,
            'gtol': 1e-11,
            'maxfun': max(200, int(maxiter) * 40),
            'maxls': 50,
            'callback': _lbfgs_callback,
        }
        res = minimize(
            _obj_pen_timed, h,
            args=(ctx, alpha, mu, deadline),
            jac=_grad_pen_timed,
            method='L-BFGS-B',
            bounds=ctx['bounds'],
            options=options
        )
        if np.isfinite(state['c']) and state['c'] < best_c:
            best_h = state['h']
            best_c = state['c']
        if hasattr(res, 'x') and res.x is not None:
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_h = h2
                best_c = c2
    except Exception:
        if np.isfinite(state['c']) and state['c'] < best_c:
            best_h = state['h']
            best_c = state['c']
    finally:
        _CB_STATE = None
    best_h = project_feasible(best_h, n)
    c = compute_c5(best_h, n)
    if np.isfinite(c):
        best_c = c
    return best_h, best_c

def slsqp_call(h, n, ctx, alpha, deadline, maxiter=200):
    n = int(n)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if not np.isfinite(best_c):
        return best_h, best_c
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_jac, 'args': (n,)}]
    try:
        res = minimize(
            _obj_pen_timed, h,
            args=(ctx, alpha, 0.0, deadline),
            jac=_grad_pen_timed,
            method='SLSQP',
            bounds=ctx['bounds'],
            constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-13, 'disp': 0}
        )
        h2 = project_feasible(res.x, n)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    except Exception:
        pass
    best_h = project_feasible(best_h, n)
    c = compute_c5(best_h, n)
    if np.isfinite(c):
        best_c = c
    return best_h, best_c

def optimize_start(h0, n, deadline, rng):
    n = int(n)
    h = project_feasible(h0, n)
    best_c = compute_c5(h, n)
    ctx = _get_ctx(n)
    if n <= 650:
        configs = [
            (1.0, 1.0, 60), (3.0, 10.0, 80), (10.0, 50.0, 100),
            (30.0, 100.0, 120), (100.0, 300.0, 140), (300.0, 1000.0, 140),
            (1000.0, 3000.0, 150), (3000.0, 5000.0, 120),
        ]
    else:
        configs = [
            (1.0, 1.0, 40), (5.0, 20.0, 60), (20.0, 100.0, 80),
            (80.0, 300.0, 100), (300.0, 1000.0, 110),
            (1000.0, 3000.0, 110), (3000.0, 5000.0, 90),
        ]
    for idx in range(len(configs)):
        remaining = deadline - time.time()
        if remaining < 15.0:
            break
        rem_count = len(configs) - idx
        call_budget = min(120.0, max(8.0, (remaining - 4.0) / float(rem_count)))
        if call_budget < 8.0:
            break
        alpha, mu, mit = configs[idx]
        maxiter = int(min(mit, max(30, call_budget * 3.0)))
        h2, c2 = lbfgs_call(h, n, ctx, alpha, mu, time.time() + call_budget, maxiter)
        if c2 < best_c:
            best_c = c2
            h = h2
    return h, best_c

def interpolate_to(h_old, n_old, n_new):
    n_old = int(n_old)
    n_new = int(n_new)
    if n_old == n_new:
        return np.asarray(h_old, dtype=float).copy()
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h = np.interp(xn, xo, np.asarray(h_old, dtype=float))
    return project_feasible(h, n_new)

def fill_to_target(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float).copy()
    target = n // 2 if n % 2 == 0 else int(0.5 * n)
    hb = np.where(h > 0.5, 1.0, 0.0)
    cur = int(hb.sum())
    if cur < target:
        need = target - cur
        zeros = np.where(hb == 0)[0]
        if need > 0 and zeros.size > 0:
            hb[zeros[:need]] = 1.0
    elif cur > target:
        need = cur - target
        ones = np.where(hb == 1)[0]
        if need > 0:
            hb[ones[:need]] = 0.0
    if int(hb.sum()) != target:
        if int(hb.sum()) < target:
            zeros = np.where(hb == 0)[0]
            need = int(target - hb.sum())
            if zeros.size:
                hb[zeros[:need]] = 1.0
        else:
            ones = np.where(hb == 1)[0]
            need = int(hb.sum() - target)
            if ones.size:
                hb[ones[:need]] = 0.0
    return hb

def random_binary_start(n, rng):
    n = int(n)
    half = n // 2
    h = np.zeros(n, dtype=float)
    if half > 0:
        idx = rng.choice(n, size=half, replace=False)
        h[idx] = 1.0
    return fill_to_target(h, n)

def golden_ratio_start(n, rng):
    n = int(n)
    half = n // 2
    h = np.zeros(n, dtype=float)
    if half > 0:
        phi = 0.5 * (1.0 + np.sqrt(5.0))
        pos = np.floor(np.arange(half) * phi + rng.uniform(0.0, 1.0)).astype(np.int64) % n
        h[pos] = 1.0
    return fill_to_target(h, n)

def fourier_start(n, rng, freq, amp):
    n = int(n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    ph = rng.uniform(0.0, 2.0 * np.pi)
    h = 0.5 + amp * np.cos(np.pi * float(freq) * x + ph)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)

def block_start(n, rng):
    n = int(n)
    runs = int(rng.randint(4, 9))
    lengths = rng.dirichlet(np.ones(runs)) * float(n)
    h = np.zeros(n, dtype=float)
    pos = 0
    val = 0.0
    for L in lengths.astype(int):
        end = min(n, pos + int(L))
        if end > pos:
            h[pos:end] = val
        pos = end
        val = 1.0 - val
    if pos < n:
        h[pos:n] = val
    return fill_to_target(h, n)

def make_symmetric(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float).reshape(-1)
    hs = 0.5 * (h + h[::-1])
    return project_feasible(hs, n)

def binarize(h, n):
    n = int(n)
    if n % 2 != 0:
        return np.asarray(h, dtype=float).copy()
    target = n // 2
    hb = np.where(np.asarray(h, dtype=float) > 0.5, 1.0, 0.0)
    cur = int(hb.sum())
    if cur < target:
        zeros = np.where(hb == 0)[0]
        need = target - cur
        if zeros.size:
            order = np.argsort(np.abs(h[zeros] - 0.5))
            hb[zeros[order[:need]]] = 1.0
    elif cur > target:
        ones = np.where(hb == 1)[0]
        need = cur - target
        if ones.size:
            order = np.argsort(np.abs(h[ones] - 0.5))
            hb[ones[order[:need]]] = 0.0
    if int(hb.sum()) < target:
        zeros = np.where(hb == 0)[0]
        need = int(target - hb.sum())
        if zeros.size:
            hb[zeros[:need]] = 1.0
    elif int(hb.sum()) > target:
        ones = np.where(hb == 1)[0]
        need = int(hb.sum() - target)
        if ones.size:
            hb[ones[:need]] = 0.0
    return hb

def gradient_pair_polish(h, n, deadline, rng, alphas=(50.0, 200.0, 800.0, 3000.0)):
    n = int(n)
    h = project_feasible(h, n)
    ctx = _get_ctx(n)
    best_c = compute_c5(h, n)
    for alpha in alphas:
        step = 0.1
        plateau = 0
        while time.time() < deadline - 1.0 and step > 1e-5:
            g = _grad_unpen(h, ctx, alpha, 0.0)
            if not np.all(np.isfinite(g)):
                break
            order = np.argsort(g)
            negs = order[:10]
            poss = order[-10:][::-1]
            pairs = []
            for a in range(min(3, len(negs))):
                for b in range(min(3, len(poss))):
                    pairs.append((int(negs[a]), int(poss[b])))
            if len(negs) > 0 and len(poss) > 0:
                pairs.append((int(negs[rng.randint(len(negs))]), int(poss[rng.randint(len(poss))))))
                pairs.append((int(negs[rng.randint(len(negs))]), int(poss[rng.randint(len(poss))))))
            improved = False
            for neg, pos in pairs:
                if neg == pos:
                    continue
                delta = min(step, float(h[neg]), float(1.0 - h[pos]))
                if delta <= 1e-12:
                    continue
                h[neg] += delta
                h[pos] -= delta
                c = compute_c5(h, n)
                h[neg] -= delta
                h[pos] += delta
                if np.isfinite(c) and c < best_c - 1e-13:
                    h[neg] += delta
                    h[pos] -= delta
                    best_c = c
                    improved = True
                    break
            if improved:
                step = min(0.12, step * 1.25)
                plateau = 0
            else:
                step *= 0.5
                plateau += 1
                if plateau > 8:
                    break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline = t0 + budget - 4.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).reshape(-1).copy()
        n0 = int(h0.size)
        if n0 <= 0:
            h0 = np.full(512, 0.5, dtype=float)
            n0 = 512
        if n0 > 1000:
            h0 = interpolate_to(h0, n0, 1000)
            n0 = 1000
    except Exception:
        h0 = np.full(512, 0.5, dtype=float)
        n0 = 512

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_n = int(n0)

    if deadline - time.time() < 30.0:
        best_h = project_feasible(best_h, best_n)
        best_c = compute_c5_direct(best_h, best_n)
        if not np.isfinite(best_c):
            best_c = compute_c5(best_h, best_n)
        return best_h, float(best_c), int(best_n)

    soft_deadline = deadline - min(160.0, 0.15 * budget)

    if time.time() < soft_deadline:
        alloc1 = min(360.0, 0.34 * (soft_deadline - time.time()))
        if alloc1 > 35.0:
            h1, c1 = optimize_start(best_h, best_n, time.time() + alloc1, rng)
            if c1 < best_c:
                best_c = c1
                best_h = h1

    for nt in (640, 800, 1000):
        if time.time() + 80.0 > soft_deadline:
            break
        if nt <= best_n:
            continue
        remaining = soft_deadline - time.time()
        alloc = min(180.0, 0.28 * remaining)
        if alloc < 55.0:
            break
        h_interp = interpolate_to(best_h, best_n, nt)
        h2, c2 = optimize_start(h_interp, nt, time.time() + alloc, rng)
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = int(nt)

    nd = best_n if best_n % 2 == 0 else best_n + 1
    if nd < 640:
        nd = 640
    if nd > 1000:
        nd = 1000

    h_div_base = best_h if best_n == nd else interpolate_to(best_h, best_n, nd)
    starts = [
        golden_ratio_start(nd, rng),
        random_binary_start(nd, rng),
        fourier_start(nd, rng, 2, 0.75),
        block_start(nd, rng),
        make_symmetric(h_div_base, nd),
    ]
    for s in starts:
        if time.time() + 45.0 > soft_deadline:
            break
        remaining = soft_deadline - time.time()
        alloc = min(80.0, 0.18 * remaining)
        if alloc < 30.0:
            break
        h3, c3 = optimize_start(s, nd, time.time() + alloc, rng)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = int(nd)

    if best_n % 2 == 0:
        hb = binarize(best_h, best_n)
        cb = compute_c5(hb, best_n)
        if cb < best_c:
            best_h = hb
            best_c = cb

    ctxb = _get_ctx(best_n)
    if time.time() < deadline - 70.0 and (best_n <= 850 or deadline - time.time() > 260.0):
        maxit = 450 if best_n <= 600 else 220
        t_sls = min(time.time() + 55.0, deadline - 45.0)
        h4, c4 = slsqp_call(best_h, best_n, ctxb, 1000.0, t_sls, maxit)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    if best_n % 2 == 0:
        hb = binarize(best_h, best_n)
        cb = compute_c5(hb, best_n)
        if cb < best_c:
            best_h = hb
            best_c = cb

    if time.time() < deadline - 5.0:
        h5, c5v = gradient_pair_polish(best_h, best_n, deadline - 2.0, rng)
        if c5v < best_c:
            best_c = c5v
            best_h = h5

    if best_n % 2 == 0:
        hb = binarize(best_h, best_n)
        cb = compute_c5(hb, best_n)
        if cb < best_c:
            best_h = hb
            best_c = cb

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5_direct(best_h, best_n)
    if not np.isfinite(best_c):
        best_c = compute_c5(best_h, best_n)
    if not np.isfinite(best_c):
        best_c = 1.0

    return best_h, float(best_c), int(best_n)
```
