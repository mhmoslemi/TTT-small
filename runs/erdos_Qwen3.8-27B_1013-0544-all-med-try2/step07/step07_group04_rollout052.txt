def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_box_sum(p, target):
    """Project p onto {h: 0<=h<=1, sum(h)=target} using bisection on lambda."""
    # Find lambda such that sum(clip(p - lambda, 0, 1)) = target
    lo = p.min() - 1.0
    hi = p.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(p - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) / 2.0, 0.0, 1.0)

        f_val = corr[k_star]

    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_box_sum(p, target):
    lo, hi = p.min() - 1.0, p.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(p - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_hwc(h, n, max_iter, deadline, mixed_alpha=0.5):
    """Hybrid subgradient + Frank-Wolfe step."""
    dx = 2.0 / n
    target = n / 2.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
        f_val = corr[k_star] * dx
        if f_val < best_c:
            best_c = f_val
            best_h = h.copy()
        
        shift = k_star - (n - 1)
        g = np.zeros(n)
        jp = np.arange(n) + shift
        vp = (jp >= 0) & (jp < n)
        g[vp] += (1.0 - h[jp[vp]])
        jm = np.arange(n) - shift
        vm = (jm >= 0) & (jm < n)
        g[vm] -= h[jm[vm]]
        
        # FW direction: best vertex
        sorted_idx = np.argsort(g)
        v = np.zeros(n)
        v[sorted_idx[:int(target)]] = 1.0
        
        # Mixed step: combine subgradient and FW
        eta = 0.5 / np.sqrt(it + 1)
        t_fw = 2.0 / (it + 2)
        
        # Try both and pick the better
        h_sub = project_box_sum(h - eta * g, target)
        h_fw = (1.0 - t_fw) * h + t_fw * v
        h_fw = project_box_sum(h_fw, target)
        
        c_sub = compute_c5(h_sub, n)
        c_fw = compute_c5(h_fw, n)
        
        if c_sub <= c_fw:
            h = h_sub
        else:
            h = h_fw
    
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=500, seed=42):
    """Binary local search with subgradient-guided ordering."""
    target = int(n // 2)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    
    # Convert to binary
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    if len(ones) > target:
        excess = rng.choice(ones, len(ones) - target, replace=False)
        h_b[excess] = 0.0
    elif len(ones) < target:
        needed = target - len(ones)
        h_b[rng.choice(zeros, needed, replace=False)] = 1.0
    
    corr = np.correlate(h_b, 1 - h_b, mode='full')
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h_b.copy()
    
    ones_set = set(np.where(h_b == 1)[0].tolist())
    zeros_list = list(np.where(h_b == 0)[0])
    m_range = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Use subgradient to order candidates
        k_star = np.argmax(corr)
        shift = k_star - (n - 1)
        g = np.zeros(n)
        jp = np.arange(n) + shift
        vp = (jp >= 0) & (jp < n)
        g[vp] += (1.0 - h_b[jp[vp]])
        jm = np.arange(n) - shift
        vm = (jm >= 0) & (jm < n)
        g[vm] -= h_b[jm[vm]]
        
        # For zeros: try those with most negative g first (most "wanting to be 1")
        z_sorted = sorted(zeros_list, key=lambda x: g[x])
        for i in z_sorted:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            o_list = list(ones_set)
            for j in o_list:
                im = i + m_range
                imn = i - m_range
                jm_arr = j + m_range
                jmn = j - m_range
                delta = np.zeros(2 * n - 1)
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h_b[im[msk]]
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h_b[imn[msk]]
                msk = (jm_arr >= 0) & (jm_arr < n)
                delta[msk] -= 1.0 - h_b[jm_arr[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h_b[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                h_b[i], h_b[best_j] = 1.0, 0.0
                corr = np.correlate(h_b, 1 - h_b, mode='full')
                cur_c = corr.max() * dx
                ones_set.discard(best_j)
                ones_set.add(i)
                zeros_list.remove(i)
                zeros_list.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h_b.copy()
                break
        if not improved:
            break
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h
