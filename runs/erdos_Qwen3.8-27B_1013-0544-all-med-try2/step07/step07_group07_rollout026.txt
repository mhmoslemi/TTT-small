from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if lag > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += (1.0 - h[:n-m])
    else:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def binary_local_search(h, n, deadline):
    """Local search restricted to binary swaps."""
    # First, create a binary approximation
    h_bin = (h > 0.5).astype(float)
    # Fix the sum
    num_ones = int(h_bin.sum())
    if num_ones > n // 2:
        # Turn off some ones (those closest to 0.5)
        prob = h_bin.copy()
        order = np.argsort(prob)
        excess = num_ones - n // 2
        for idx in order[:excess]:
            if prob[idx] == 1.0:
                prob[idx] = 0.0
    elif num_ones < n // 2:
        # Turn on some zeros
        order = np.argsort(-prob)
        deficit = n // 2 - num_ones
        for idx in order[:deficit]:
            if prob[idx] == 0.0:
                prob[idx] = 1.0
    
    h_bin = project_feasible(prob, n)
    best_c = compute_c5(h_bin, n)
    
    # Binary swap local search
    no_improve = 0
    while time.time() < deadline and no_improve < 100:
        c = compute_all_lags(h_bin, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        
        # Find best binary swap
        best_gain = 0
        best_swap = None
        if lag > 0:
            pairs = [(i, i + m) for i in range(n - m)]
        else:
            pairs = [(i + m, i) for i in range(n - m)]
        
        for (i, j) in pairs:
            if h_bin[i] == h_bin[j]:
                continue
            # Swap: h_bin[i] and h_bin[j] exchange values
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            new_c = compute_c5(h_bin, n)
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            gain = best_c - new_c
            if gain > best_gain:
                best_gain = gain
                best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            best_c = best_c - best_gain
            no_improve = 0
        else:
            no_improve += 1
    
    return h_bin, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if lag > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += (1.0 - h[:n-m])
    else:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def topk_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    active = c[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_topk(h, n, top_idx, alpha, maxiter=2000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(topk_smooth_obj, h, args=(n, top_idx, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def fast_local_search(h, n, deadline, top_k=30, patience=40):
    h = h.copy()
    best_c = compute_c5(h, n)
    no_improve = 0
    rng = np.random.RandomState(0)
    while time.time() < deadline and no_improve < patience:
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        g = lag_gradient(h, n, lag)
        m = abs(lag)
        if lag > 0:
            pairs = [(i, i + m) for i in range(n - m)]
        else:
            pairs = [(i + m, i) for i in range(n - m)]
        if not pairs:
            break
        pd = np.array([g[j] - g[i] for i, j in pairs])
        sorted_idx = np.argsort(pd)[:top_k]
        candidates = [pairs[idx] for idx in sorted_idx]
        n_rand = min(10, len(pairs))
        for _ in range(n_rand):
            candidates.append(pairs[rng.randint(len(pairs))])
        best_move = None
        best_new_c = best_c
        for (i, j) in candidates:
            if abs(h[i] - h[j]) < 1e-8:
                continue
            direction = -1 if h[i] > h[j] else 1
            src = i if direction == -1 else j
            dst = j if direction == -1 else i
            gap = abs(h[src] - h[dst])
            for frac in [0.5, 0.25, 0.1, 0.05, 0.01]:
                delta = gap * frac
                if delta < 1e-10:
                    continue
                h[src] -= delta
                h[dst] += delta
                new_c = compute_c5(h, n)
                h[src] += delta
                h[dst] -= delta
                if new_c < best_new_c:
                    best_new_c = new_c
                    best_move = (src, dst, delta)
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0, 1)
            best_c = best_new_c
            no_improve = 0
        else:
            no_improve += 1
    return h, best_c

def binary_local_search(h, n, deadline):
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        order = np.argsort(h_bin)
        for idx in order:
            if h_bin[idx] == 1.0:
                h_bin[idx] = 0.0
                num_ones -= 1
            if num_ones <= target:
                break
    elif num_ones < target:
        order = np.argsort(-h_bin)
        for idx in order:
            if h_bin[idx] == 0.0:
                h_bin[idx] = 1.0
                num_ones += 1
            if num_ones >= target:
                break
    best_c = compute_c5(h_bin, n)
    no_improve = 0
    while time.time() < deadline and no_improve < 80:
        c = compute_all_lags(h_bin, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        if lag > 0:
            pairs = [(i, i + m) for i in range(n - m)]
        else:
            pairs = [(i + m, i) for i in range(n - m)]
        best_gain = 0
        best_swap = None
        for (i, j) in pairs:
            if h_bin[i] == h_bin[j]:
                continue
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            new_c = compute_c5(h_bin, n)
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            gain = best_c - new_c
            if gain > best_gain:
                best_gain = gain
                best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            best_c -= best_gain
            no_improve = 0
        else:
            no_improve += 1
    return h_bin, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if time.time() < deadline - 20:
        h_new = slsqp(best_h, n, 50000, maxiter=1500, ftol=1e-14)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 10:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-5:]
        h_new = slsqp_topk(best_h, n, top_idx, 20000, maxiter=1500, ftol=1e-14)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 3:
        best_h, best_c = fast_local_search(best_h, n, min(deadline - 1, time.time() + (deadline - time.time()) * 0.6), top_k=30, patience=30)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
