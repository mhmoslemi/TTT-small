def smooth_g_value(g, theta):
    if theta <= 0: return float(g.max())
    gmax = float(g.max())
    z = theta * (g - gmax)
    # z max 0; sum exp
    s = np.exp(z).sum()
    return gmax + float(np.log(s) / theta)

import numpy as np
import time
from scipy.optimize import minimize

_TIMEOUT=0.0

def corr_values(h,n):
    return np.correlate(h, 1.0-h, mode='full')*(2.0/n)

def c5_value(h,n):
    return float(np.max(corr_values(h,n)))

def project(v,n,target=None):
    if target is None: target=0.5*n
    v=np.array(v,dtype=float,copy=True)
    np.clip(v,0.0,1.0,out=v)
    s=float(v.sum())
    if abs(s-target)<1e-13:
        return v
    lo=float(v.min())-1.0
    hi=float(v.max())+1.0
    for _ in range(45):
        mid=(lo+hi)*0.5
        t=v-mid
        np.clip(t,0.0,1.0,out=t)
        s=float(t.sum())
        if s>target: lo=mid
        else: hi=mid
    lam=(lo+hi)*0.5
    t=v-lam
    np.clip(t,0.0,1.0,out=t)
    diff=target-float(t.sum())
    if abs(diff)>1e-14:
        if diff>0:
            mask=t<1.0-1e-12
            if mask.any(): t[mask]+=diff/mask.sum()
        else:
            mask=t>1e-12
            if mask.any(): t[mask]+=diff/mask.sum()
    np.clip(t,0.0,1.0,out=t)
    return t

def smooth_value(g,theta):
    gm=float(np.max(g))
    if theta<=1e-12: return gm
    z=theta*(g-gm)
    s=float(np.exp(z).sum())
    if s<=0 or not np.isfinite(s): return gm
    return float(gm + np.log(s)/theta)

def softmax_w(g,theta):
    if theta<=1e-12:
       w=np.full(g.shape,1.0/g.size)
    else:
       gm=float(np.max(g))
       z=theta*(g-gm)
       e=np.exp(z)
       s=float(e.sum())
       if s<=0 or not np.isfinite(s):
          w=np.zeros(g.shape); w[int(np.argmax(g))]=1.0
       else:
          w=e/s
    return w[::-1].copy()

def fft_stuff(n):
    L=1
    m=3*n-2
    while L<m: L<<=1
    ar=np.arange(n)
    return L, 2*n-2-ar, 2*n-1-ar, n-ar

def gradient_u(h,w,n,L,idxC,idxHi,idxLo):
    pref=np.empty(2*n,dtype=float)
    pref[0]=0.0
    np.cumsum(w,out=pref[1:])
    A1=pref[idxHi]-pref[idxLo]
    S=w+w[::-1]
    Fh=np.fft.rfft(h[::-1],L)
    Fs=np.fft.rfft(S,L)
    c=np.fft.irfft(Fh*Fs,L)
    return A1-c[idxC]

def pgrad(h,n,theta,deadline,max_iter,base_eta):
    h=project(h,n)
    g=corr_values(h,n)
    C=float(np.max(g))
    F=smooth_value(g,theta)
    L,idxC,idxHi,idxLo=fft_stuff(n)
    base=base_eta; stall=0
    for _ in range(max_iter):
       if time.time()>deadline: break
       w=softmax_w(g,theta)
       grad=gradient_u(h,w,n,L,idxC,idxHi,idxLo)
       scale=float(np.sqrt(np.dot(grad,grad)))
       if not np.isfinite(scale) or scale<1e-14: break
       eta=base/scale
       accept=False; k=25
       for k in range(25):
           trial=project(h-eta*grad,n)
           gt=corr_values(trial,n)
           Ft=smooth_value(gt,theta)
           Ct=float(np.max(gt))
           if Ft < F-1e-12 or (Ct < C-1e-12 and Ft <= F+1e-8):
               h=trial; g=gt; F=Ft; C=Ct; accept=True; break
           eta*=0.5
           if eta<1e-10: break
       if not accept:
           base*=0.5; stall+=1
           if base<1e-7 or stall>100: break
       else:
           stall=0
           if k==0: base=min(base*1.3,15.0)
           elif k<=2: base=min(base*1.05,15.0)
    return h,C

def to_binary(h,n):
    if n%2: return None
    k=n//2
    if k==0: return np.zeros(n)
    order=np.argpartition(h,k-1)[::-1]
    out=np.zeros(n); out[order[:k]]=1.0; return out

def bdelta(i,j,h,rb,n,delta):
    delta.fill(0.0)
    r=rb[::-1]
    delta[i:i+n]+=r
    delta[j:j+n]-=r
    p=n-1-i; q=n-1-j
    delta[p:p+n]-=h
    delta[q:q+n]+=h
    delta[n-1]-=2.0
    delta[n-1+i-j]+=1.0
    delta[n-1+j-i]+=1.0

def binary_search(h0,n,deadline,seed,max_iters=300000):
    h=to_binary(h0,n)
    if h is None: return None,None
    rb=1.0-h
    corr=np.correlate(h,1.0-h,mode='full')
    cur=int(np.rint(float(np.max(corr))))
    best_h=h.copy(); best=cur
    ones=np.where(h==1.0)[0]; zeros=np.where(h==0.0)[0]
    if ones.size==0 or zeros.size==0: return best_h,best*(2.0/n)
    rng=np.random.RandomState(seed)
    delta=np.zeros(2*n-1)
    trial=np.empty_like(corr)
    T=1.0
    cool=(1e-4)**(1.0/max(1,max_iters))
    dx=2.0/n
    for it in range(max_iters):
        if time.time()>deadline: break
        ai=rng.randint(zeros.size); bi=rng.randint(ones.size)
        i=int(zeros[ai]); j=int(ones[bi])
        bdelta(i,j,h,rb,n,delta)
        np.add(corr,delta,out=trial)
        new_max=float(np.max(trial))
        new_int=int(np.rint(new_max))
        diff=new_int-cur
        acc=False
        if diff<0: acc=True
        elif diff==0: acc=True
        elif T>1e-8 and diff<=10:
            if rng.random() < np.exp(-diff/T): acc=True
        if acc:
            h[i]=1.0; rb[i]=0.0; h[j]=0.0; rb[j]=1.0
            corr,trial=trial,corr
            cur=new_int
            zeros[ai]=j; ones[bi]=i
            if cur<best:
                best=cur; best_h=h.copy()
        T*=cool
    return best_h,best*dx

def sum_eq(x):
    return float(x.sum()-0.5*x.size)

def slsqp_callback(xk):
    if time.time()>_TIMEOUT:
        raise TimeoutError

def slsqp_obj(x,theta,dx,L,idxC,idxHi,idxLo):
    return smooth_value(corr_values(x,x.size),theta)

def slsqp_jac(x,theta,dx,L,idxC,idxHi,idxLo):
    n=x.size
    g=corr_values(x,n)
    w=softmax_w(g,theta)
    grad=gradient_u(x,w,n,L,idxC,idxHi,idxLo)
    return dx*grad

def slsqp(h,n,theta,deadline,maxiter):
    global _TIMEOUT
    h=project(h,n)
    dx=2.0/n
    L,idxC,idxHi,idxLo=fft_stuff(n)
    _TIMEOUT=deadline-2.0
    bounds=[(0.0,1.0)]*n
    try:
       res=minimize(slsqp_obj,h,args=(theta,dx,L,idxC,idxHi,idxLo),
                    jac=slsqp_jac,bounds=bounds,method='SLSQP',
                    constraints=[{'type':'eq','fun':sum_eq}],
                    options={'maxiter':maxiter,'ftol':1e-12,'disp':0},
                    callback=slsqp_callback)
       h2=project(res.x if hasattr(res,'x') else h,n)
       return h2,c5_value(h2,n)
    except Exception:
       return h,c5_value(h,n)

    else:

import numpy as np
import time
from scipy.optimize import minimize

_TIMEOUT = 0.0


def corr_values(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def c5_value(h, n):
    return float(np.max(corr_values(h, n)))


def project_feasible(v, n, target=None):
    if target is None:
        target = 0.5 * n
    v = np.array(v, dtype=float, copy=True)
    np.clip(v, 0.0, 1.0, out=v)
    s = float(v.sum())
    if abs(s - target) < 1e-13:
        return v
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        t = v - mid
        np.clip(t, 0.0, 1.0, out=t)
        s = float(t.sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    t = v - lam
    np.clip(t, 0.0, 1.0, out=t)
    diff = target - float(t.sum())
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = t < 1.0 - 1e-12
            if np.any(mask):
                t[mask] += diff / float(np.sum(mask))
        else:
            mask = t > 1e-12
            if np.any(mask):
                t[mask] += diff / float(np.sum(mask))
    np.clip(t, 0.0, 1.0, out=t)
    return t


def finalize_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.any(mask):
            idx = int(np.flatnonzero(mask)[0])
            h[idx] += diff
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        h = project_feasible(h, n)
    return h


def smooth_value(g, theta):
    gm = float(np.max(g))
    if theta <= 1e-12:
        return gm
    z = theta * (g - gm)
    s = float(np.exp(z).sum())
    if s <= 0.0 or not np.isfinite(s):
        return gm
    return float(gm + np.log(s) / theta)


def softmax_weights(g, theta):
    if theta <= 1e-12:
        w = np.full(g.shape, 1.0 / float(g.size))
    else:
        gm = float(np.max(g))
        z = theta * (g - gm)
        e = np.exp(z)
        s = float(e.sum())
        if s <= 0.0 or not np.isfinite(s):
            w = np.zeros(g.shape)
            w[int(np.argmax(g))] = 1.0
        else:
            w = e / s
    return w[::-1].copy()


def fft_stuff(n):
    L = 1
    m = 3 * n - 2
    while L < m:
        L <<= 1
    ar = np.arange(n)
    return L, 2 * n - 2 - ar, 2 * n - 1 - ar, n - ar


def gradient_unit(h, w, n, L, idxC, idxHi, idxLo):
    pref = np.empty(2 * n, dtype=float)
    pref[0] = 0.0
    np.cumsum(w, out=pref[1:])
    A1 = pref[idxHi] - pref[idxLo]
    S = w + w[::-1]
    Fh = np.fft.rfft(h[::-1], L)
    Fs = np.fft.rfft(S, L)
    c = np.fft.irfft(Fh * Fs, L)
    return A1 - c[idxC]


def polished_gradient(h, n, theta, deadline, max_iter, base_eta):

            break
        eta = base / scale
        accepted = False

def to_binary(h, n):
    if n % 2 != 0:
        return None
    k = n // 2
    if k == 0:
        return np.zeros(n, dtype=float)
    order = np.argpartition(h, k - 1)[::-1]
    out = np.zeros(n, dtype=float)
    out[order[:k]] = 1.0
    return out


def binary_delta_into(i, j, h, rb, n, delta):
    delta.fill(0.0)
    r = rb[::-1]
    delta[i:i + n] += r
    delta[j:j + n] -= r
    p = n - 1 - i
    q = n - 1 - j
    delta[p:p + n] -= h
    delta[q:q + n] += h
    delta[n - 1] -= 2.0
    delta[n - 1 + i - j] += 1.0
    delta[n - 1 + j - i] += 1.0


def binary_search(h0, n, deadline, seed, max_iters=250000):
    h = to_binary(h0, n)
    if h is None:
        return None, None
    rb = 1.0 - h
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur = int(np.rint(float(np.max(corr))))
    best_h = h.copy()
    best = cur
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if ones.size == 0 or zeros.size == 0:
        return best_h, best * (2.0 / n)
    rng = np.random.RandomState(seed)
    delta = np.zeros(2 * n - 1)
    trial = np.empty_like(corr)
    T = 1.0
    cool = (1e-4) ** (1.0 / max(1, max_iters))
    last_best = 0
    dx = 2.0 / n
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ai = int(rng.randint(zeros.size))
        bi = int(rng.randint(ones.size))
        i = int(zeros[ai])
        j = int(ones[bi])
        binary_delta_into(i, j, h, rb, n, delta)
        np.add(corr, delta, out=trial)
        new_int = int(np.rint(float(np.max(trial))))
        diff = new_int - cur
        acc = False
        if diff < 0:
            acc = True
        elif diff == 0:
            acc = True
        elif T > 1e-8 and diff <= 10:
            if rng.random() < np.exp(-diff / T):
                acc = True
        if acc:
            h[i] = 1.0
            rb[i] = 0.0
            h[j] = 0.0
            rb[j] = 1.0
            corr, trial = trial, corr
            cur = new_int
            zeros[ai] = j
            ones[bi] = i
            if cur < best:
                best = cur
                best_h = h.copy()
                last_best = it
        T *= cool
        if T < 0.02 and it - last_best > 80000:
            break
    return best_h, best * dx


def mix_eval(t, h1, dh, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

    g = corr_values(x, n)
    w = softmax_weights(g, theta)

        bh, bc, wh, wc = polished_gradient(best_h, n, 300000.0, deadline - 0.5, 60000, 0.6)
        if bc < best_c:
            best_c = bc
            best_h = bh.copy()

    final_h = finalize_feasible(best_h, n)
    final_c = c5_value(final_h, n)
    if final_c <= best_c + 1e-12 and final_c < best_c:
        best_c = final_c
    elif final_h is not None and best_h is not None:
        final_h = project_feasible(best_h, n)
        final_c = c5_value(final_h, n)
    return final_h, float(final_c), n
