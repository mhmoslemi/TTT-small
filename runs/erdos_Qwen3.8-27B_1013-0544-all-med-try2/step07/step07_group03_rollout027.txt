```python
def smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')  # length 2n-1
    # c[m] corresponds to shift k = m - (n-1)
    shifted = alpha * (c - c.max())
    p = np.exp(shifted - shifted.max())
    p /= p.sum()
    
    grad = np.zeros(n)
    # Loop over shifts with significant weight
    for m in range(2*n-1):
        if p[m] < 1e-12:
            continue
        k = m - (n-1)
        if k >= 0:
            kk = k
            grad[:n-kk] += p[m] * (1 - h[kk:])
            grad[kk:] -= p[m] * h[:n-kk]
        else:
            kk = -k
            grad[kk:] += p[m] * (1 - h[:n-kk])
            grad[:n-kk] -= p[m] * h[kk:]
    return grad * dx
```
