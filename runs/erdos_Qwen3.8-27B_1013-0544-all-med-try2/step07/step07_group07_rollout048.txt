import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        e = np.exp(shifted - mx)
        return float(m + (mx + np.log(np.sum(e))) / alpha)
    e = np.exp(shifted)
    return float(m + np.log(np.sum(e)) / alpha)

def smooth_obj(x, n, alpha):
    return smooth_max(compute_all_lags(x, n), alpha)

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m >= n:
        return grad
    elif lag > 0:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] += (-h[:n-m])
    else:
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] += (-h[m:])
    return grad * (2.0 / n)

def slp_step(h_curr, n, trust, top_k):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    top_idx = np.argsort(c)[-top_k:]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.norm(cp.abs(h_var - h_curr), 1) <= trust * n]
    for k in top_idx:
        lag = k - (n - 1)
        grad = _lag_grad(h_curr, n, lag)
        cons.append(c[k] + np.dot(grad, (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            return h_curr.copy(), c.max()
    if h_var.value is None:
        return h_curr.copy(), c.max()
    return project_feasible(h_var.value, n), c.max()

def slp_optimize(h, n, deadline, max_iters=60):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    top_k = 5
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        h_new, c_lin = slp_step(h_curr, n, trust, top_k)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-9:
            break
        if c_new > c_lin + 0.002:
            trust *= 0.5
        else:
            trust = max(0.01, min(0.3, trust * 1.1))
        h_curr = h_new
    return best_h, best_c

def nesterov_pg(h, n, deadline, alpha=5000, momentum=0.9):
    h_curr = h.copy()
    y = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(best_h, n)

        lam = (1 + t) / (1 + t + 1)
        y_next = project_feasible(y_new, n)

            best_c = c_check
            best_h = y_next.copy()
