import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _compute_lag_gradient(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h_curr
    elif lag > 0:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] += (-h_curr[:n - m])
    else:
        grad[:n - m] += (-h_curr[m:])
        grad[m:] += (1.0 - h_curr[:n - m])
    return grad

def gradient_search(h, n, deadline, top_m=3, max_iter=50000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    step = 0.02
    momentum = 0.5
    v = np.zeros(n)
    stall_count = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_m:]
        
        g = np.zeros(n)
        for idx in top_idx:
            k = idx - (n - 1)
            g += _compute_lag_gradient(h, n, k)
        
        g -= np.mean(g)
        v = momentum * v - step * g
        h_new = project_feasible(h + v, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:

                stall_count = 0
                if step < 1e-7:

def direct_push_search(h, n, deadline, top_m=3, push_amount=0.02, top_pairs=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    while time.time() < deadline:
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_m:]
        improved = False
        
        for idx in top_idx:
            k = idx - (n - 1)
            if k == 0:
                # c[0] = sum(h*(1-h)), minimize by pushing towards 0 or 1
                # Push positions with h near 0.5 towards 0 or 1
                mid = np.abs(h - 0.5)
                worst = np.argsort(mid)[:top_pairs]
                for i in worst:
                    if h[i] > 0.5:
                        delta = min(push_amount, (h[i] - 0.5) * 0.5)
                        h[i] -= delta
                        # Add to a low position
                        low_idx = np.argmin(h)
                        h[low_idx] += delta
                    else:
                        delta = min(push_amount, (0.5 - h[i]) * 0.5)
                        h[i] += delta
                        high_idx = np.argmax(h)
                        h[high_idx] -= delta
            elif k > 0:
                contrib = h[:n-k] * (1 - h[k:])
                worst = np.argsort(contrib)[-top_pairs:]
                for i in worst:
                    j = i + k
                    delta = min(push_amount, h[i] * 0.4, 1.0 - h[j])
                    if delta < 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
            else:
                m = -k
                contrib = h[m:] * (1 - h[:n-m])
                worst = np.argsort(contrib)[-top_pairs:]
                for i_rel in worst:
                    i = i_rel + m
                    j = i_rel
                    delta = min(push_amount, h[i] * 0.4, 1.0 - h[j])
                    if delta < 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
        
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
            improved = True
        if not improved:
            push_amount *= 0.7
            if push_amount < 1e-6:
                break
    
    return best_h, best_c

def _compute_lag_gradient_cvx(h_curr, n, lag):
    # Same as _compute_lag_gradient but returns the linearization coefficients
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h_curr
    elif lag > 0:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] += (-h_curr[:n - m])
    else:
        grad[:n - m] += (-h_curr[m:])
        grad[m:] += (1.0 - h_curr[:n - m])
    return grad

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.norm(cp.abs(h_var - h_curr), np.inf) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad = _compute_lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            try:
                prob.solve(solver=cp.PDLP)
            except Exception:
                return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=200):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.1

            break
        h_curr = h_new
        if c_new > best_c + 1e-6:
            trust *= 0.5
        else:

    return best_h, best_c

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    pert = scale * np.sin(freq * np.pi * x + phase)
    return project_feasible(h + pert, n)

def step_perturb(h, n, rng, scale):

    pert = np.zeros(n)
    n_half = n // 2
    pert[perm[:n_half]] = scale
    pert[perm[n_half:]] = -scale
    return project_feasible(h + pert, n)

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        mask = hp < 1.0 if diff > 0 else hp > 0.0
        cnt = mask.sum()
        if cnt > 0:
            hp[mask] += diff / cnt
    return np.clip(hp, 0.0, 1.0)

def lag_gradient(h, n, lag):
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n-m] += (1.0 - h[m:])
        g[m:] += (-h[:n-m])
    else:
        g[:n-m] += (-h[m:])
        g[m:] += (1.0 - h[:n-m])
    return g

def gradient_search(h, n, deadline, top_m=3, max_iter=100000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.01
    v = np.zeros(n)
    mom = 0.7
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for idx in top:
            g += lag_gradient(h, n, idx - (n - 1))
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        v = mom * v - step * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            h = h_new
            best_c = c_new
            best_h = h.copy()
            stall = 0
            step = min(step * 1.05, 0.05)
        else:
            stall += 1
            v *= 0.3
            if stall > 15:
                step *= 0.6
                stall = 0
                if step < 1e-8:
                    top_m = min(top_m + 2, 12)
                    step = 0.005
        if stall > 100:
            break
    return best_h, best_c

def push_search(h, n, deadline, top_m=3, amount=0.015, pairs=40):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    while time.time() < deadline:
        c = compute_all_lags(h, n)
        top = np.argsort(c)[-top_m:]
        changed = False
        for idx in top:
            k = idx - (n - 1)
            if k == 0:
                mid = np.abs(h - 0.5)
                worst = np.argsort(mid)[:pairs]
                for i in worst:
                    d = min(amount, abs(h[i] - 0.5) * 0.4)
                    if d < 1e-10:
                        continue
                    if h[i] > 0.5:
                        h[i] -= d
                        h[np.argmin(h)] += d
                    else:
                        h[i] += d
                        h[np.argmax(h)] -= d
            elif k > 0:
                contrib = h[:n-k] * (1.0 - h[k:])
                worst = np.argsort(contrib)[-pairs:]
                for i in worst:
                    j = i + k
                    d = min(amount, h[i] * 0.3, 1.0 - h[j])
                    if d < 1e-10:

                    h[i] -= d

            best_h = h.copy()

            amount *= 0.6
            if amount < 1e-7:

def oa_step(h_curr, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [cp.sum(hv) == n / 2.0, hv >= 0, hv <= 1,
            cp.abs(hv - h_curr) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        g = lag_gradient(h_curr, n, lag)
        cons.append(c[k] + dx * (g @ (hv - h_curr)) <= tv)
    prob = cp.Problem(cp.Minimize(tv), cons)
    solved = False
    for solver in [cp.GLOP, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            solved = True
            break
        except Exception:
            continue
    if not solved or prob.status not in ['optimal', 'optimal_inaccurate'] or hv.value is None:
        return h_curr.copy()
    return project_feasible(hv.value, n)
