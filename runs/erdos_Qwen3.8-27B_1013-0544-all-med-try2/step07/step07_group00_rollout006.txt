def compute_c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):

            if tot < 1e-14:

def simulated_annealing(h, n, deadline, t0=0.5, t1=0.01, iterations=5000):
    """Simulated annealing with structured moves."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(int(time.time() * 1000) % 10000)
    
    for it in range(iterations):
        if time.time() > deadline:
            return best_h, best_c
        
        # Temperature schedule
        frac = it / iterations
        T = t0 * (t1 / t0) ** frac
        
        # Random move: pick two positions and transfer mass
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        s = rng.choice([0.05, 0.1, 0.15, 0.2])
        sign = rng.choice([-1, 1])
        
        h[i] += sign * s
        h[j] -= sign * s
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        
        delta = c_new - best_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # Reject
            h[i] -= sign * s
            h[j] += sign * s
            h = project_feasible(h, n)
    
    return best_h, best_c
