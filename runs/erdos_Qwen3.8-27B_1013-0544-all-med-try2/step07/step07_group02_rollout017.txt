    h = h.astype(float).copy()

    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):

            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]

                ub_x = 1.0
                if pos_B.any():

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    raw_c = np.correlate(h, 1 - h, mode='full')
    c = raw_c * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    es = np.exp(shifted - sm)
    Z = es.sum()
    w = es / Z
    val = m + (sm + np.log(Z)) / alpha
    
    n_cor = len(raw_c)
    grad = np.zeros(n)
    for k in range(n_cor):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        j_lo = max(0, -lag)
        j_hi = min(n, n - lag)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += wk * (1.0 - h[j_lo + lag:j_hi + lag])

        j_lo2 = max(j_lo2, 0)
        j_hi2 = min(n, n + lag)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= wk * h[j_lo2 - lag:j_hi2 - lag]
    return val, grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    raw_c = np.correlate(h, 1 - h, mode='full')
    c = raw_c * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    es = np.exp(shifted - sm)
    Z = es.sum()
    w = es / Z
    val = m + (sm + np.log(Z)) / alpha
    n_cor = len(raw_c)
    grad = np.zeros(n)
    for k in range(n_cor):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag < 0:
            j0, j1 = 0, n + lag
            if j0 < j1:
                grad[j0:j1] += wk * (1.0 - h[-lag:])
        else:
            j0, j1 = lag, n
            if j0 < j1:
                grad[j0:j1] += wk * (1.0 - h[:n - lag])
        if lag >= 0:
            j0, j1 = 0, n - lag
            if j0 < j1:
                grad[j0:j1] -= wk * h[lag:]
        else:
            j0, j1 = -lag, n
            if j0 < j1:
                grad[j0:j1] -= wk * h[:n + lag]
    return val, grad * dx

def lbfgs_step(h, n, alpha, mu, maxiter):
    target = n / 2.0
    def obj(x):
        val, _ = smooth_max_and_grad(x, n, alpha)
        return val + mu * (x.sum() - target) ** 2
    def gfun(x):
        _, g = smooth_max_and_grad(x, n, alpha)
        g += 2.0 * mu * (x.sum() - target)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=gfun, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 50})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h

def coord_descent(h, n, max_sweeps=100, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            msk1 = (k1 >= 0) & (k1 < n)
            B[msk1] += (1.0 - h[k1[msk1]])
            k2 = i + m_range
            msk2 = (k2 >= 0) & (k2 < n)
            B[msk2] -= h[k2[msk2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(42):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def refine(h, n, alpha_list, mu, maxiter_each, deadline):
    h = h.copy()
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        h = lbfgs_step(h, n, alpha, mu, maxiter_each)
        h = project_feasible(h, n)
    return h
