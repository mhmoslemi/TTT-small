```python
def fast_move(corr, h, n, i, j, delta, dx):
    """Apply move: h[i]+=delta, h[j]-=delta. Update corr in place.
    Returns new max of corr."""
    b = 1.0 - h
    m = np.arange(2*n - 1)
    
    # Compute the 4 index arrays
    idx1 = i - m + n - 1  # b[i-m+n-1]
    idx2 = j - m + n - 1
    idx3 = i + m - n + 1  # a[i+m-n+1]
    idx4 = j + m - n + 1
    
    # Valid masks
    m1 = (idx1 >= 0) & (idx1 < n)
    m2 = (idx2 >= 0) & (idx2 < n)
    m3 = (idx3 >= 0) & (idx3 < n)
    m4 = (idx4 >= 0) & (idx4 < n)
    
    # Delta corr
    dc = np.zeros(2*n-1)
    dc[m1] += b[idx1[m1]]
    dc[m2] -= b[idx2[m2]]
    dc[m3] -= h[idx3[m3]]
    dc[m4] += h[idx4[m4]]
    dc *= delta * dx
    
    # Check if improvement
    new_max = np.max(corr + dc)
    
    # Apply
    corr += dc
    h[i] += delta
    h[j] -= delta
    
    return new_max
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def fast_local_search(h, n, deadline, rng, initial_step=0.05, min_step=0.0001,
                      step_decay=0.95, no_improve_patience=2000):
    """Fast greedy local search with incremental correlation updates."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    current_max = float(corr.max())
    best_max = current_max
    best_h = h.copy()
    
    step = initial_step
    no_improve = 0
    m_arr = np.arange(2 * n - 1)
    
    while time.time() < deadline:
        # Generate a random move
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Determine valid delta range
        if rng.rand() < 0.5:
            # h[i] += delta, h[j] -= delta
            delta = rng.uniform(0, step)
            if h[i] + delta > 1.0 or h[j] - delta < 0.0:
                continue
        else:
            # h[i] -= delta, h[j] += delta
            delta = rng.uniform(0, step)
            if h[i] - delta < 0.0 or h[j] + delta > 1.0:
                continue
            delta = -delta
        
        # Compute delta correlation (O(n) vectorized)
        b = 1.0 - h
        idx1 = i - m_arr + n - 1
        idx2 = j - m_arr + n - 1
        idx3 = i + m_arr - n + 1
        idx4 = j + m_arr - n + 1
        
        dc = np.zeros(2*n-1)
        mask1 = (idx1 >= 0) & (idx1 < n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        mask3 = (idx3 >= 0) & (idx3 < n)
        mask4 = (idx4 >= 0) & (idx4 < n)
        
        if mask1.any():
            dc[mask1] += b[idx1[mask1]]
        if mask2.any():
            dc[mask2] -= b[idx2[mask2]]
        if mask3.any():
            dc[mask3] -= h[idx3[mask3]]
        if mask4.any():
            dc[mask4] += h[idx4[mask4]]
        
        dc *= delta * dx
        
        # Check improvement
        new_max = float(np.max(corr + dc))
        if new_max < current_max - 1e-15:
            corr += dc
            h[i] += delta
            h[j] -= delta
            current_max = new_max
            no_improve = 0
            if current_max < best_max - 1e-15:
                best_max = current_max
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > no_improve_patience:
                no_improve = 0
                step *= step_decay
                if step < min_step:
                    step = min_step
    
    # Re-verify
    h_final = project_feasible(best_h, n)
    c_final = compute_c5(h_final, n)
    return h_final, c_final

def run(seed=42, budget_s=1100, **kwargs):

    deadline = t0 + budget_s - 3

    best_h = h.copy()

    # Phase 2: SLSQP with high alpha
    if time.time() < deadline - 60:
        alloc2 = min(80, (deadline - time.time()) * 0.3)
        h_opt = slsqp(h, n, 50000, maxiter=2000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_h = h_opt
            best_c = c_opt
    
    # Phase 3: More fast local search from best
    if time.time() < deadline - 30:
        t_phase3_end = deadline - 10
        h, c = fast_local_search(best_h, n, t_phase3_end, rng,

                                 step_decay=0.95, no_improve_patience=3000)
        if c < best_c:

            best_c = c
    
    # Phase 4: Final polish

        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_h = h_opt
            best_c = c_opt
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
