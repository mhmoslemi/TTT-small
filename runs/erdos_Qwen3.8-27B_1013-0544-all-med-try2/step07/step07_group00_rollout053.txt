        h_opt, c_opt = optimize_lbfgs(h_p, best_n, time.time() + alloc * 0.6, max_iters=80)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Try different n values
    for n_try in [300, 400, 600, 800]:
        if time.time() > deadline - 150:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(50, (deadline - time.time() - 100) * 0.45)
        if alloc < 20:
            continue
        h_t, c_t = optimize_lbfgs(h_interp, n_try, time.time() + alloc * 0.7, max_iters=150)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            best_n = n_try

    # Phase 5: Final aggressive optimization
    if time.time() < deadline - 40:
        t5 = min(time.time() + 60, deadline - 30)
        best_h, best_c = optimize_lbfgs(best_h, best_n, t5, max_iters=500)

    # Phase 6: Fast polish
    if time.time() < deadline - 15:
        t6 = min(time.time() + 30, deadline - 10)
        best_h, best_c = fast_polish(best_h, best_n, t6)

    # Phase 7: Very fine local search
    if time.time() < deadline - 5:
        t7 = min(time.time() + 15, deadline - 3)
        best_h, best_c = coordinate_polish(best_h, best_n, t7)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
