import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def fw_step(h, n, g):
    """Frank-Wolfe step: find vertex of hypersimplex minimizing g·v"""
    idx = np.argsort(g)
    v = np.zeros(n)
    v[idx[:n//2]] = 1.0
    return v

def fw_optimize(h, n, deadline, max_iters=5000):
    """Frank-Wolfe optimization with smooth max gradient"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [1000, 10000, 100000, 1000000]
    for alpha in alphas:
        if time.time() > deadline:
            break
        x = best_h.copy()
        for t in range(max_iters):
            if time.time() > deadline - 5:
                break
            g = grad_smooth_max(x, n, alpha)
            v = fw_step(x, n, g)
            gamma = 2.0 / (t + 2.0)
            x_new = (1 - gamma) * x + gamma * v
            x_new = np.clip(x_new, 0.0, 1.0)
            # Fix sum constraint
            x_new = project_feasible(x_new, n)
            x = x_new
            c = compute_c5(x, n)
            if c < best_c:
                best_c = c
                best_h = x.copy()
    return best_h, best_c

def _penalty_objective(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_gradient(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_run(h, n, deadline, alphas=None, max_iters=300):
    if alphas is None:
        alphas = [500, 5000, 50000, 500000, 2000000, 10000000]
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 10:

        mu = 1.0 + i * 10.0
        tr = deadline - time.time() - 10
        if tr < 10:

        try:
            res = minimize(_penalty_objective, h, jac=_penalty_gradient,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': max_iters, 'ftol': 1e-18,
                                  'gtol': 1e-16, 'maxfun': max_iters * 30,
                                  'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:

def subgradient_kick(h, n, deadline, num_steps=200):
    """Take subgradient steps for the hard max to escape local issues"""
    best_h = h.copy()
    best_c = compute_c5(h, n)

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha + 1e8 * (x.sum() - n/2.0)**2)

    g = grad_smooth_max(x, n, alpha)
    g += 2e8 * (x.sum() - n/2.0)
    return g

            break

                         args=(n, alpha), method='L-BFGS-B',

                                  'gtol': 1e-16, 'maxfun': max_iters*30,
                                  'maxls': 50})

            c_new = compute_c5(h_new, n)
            if c_new < best_c:

                h = h_new

            pass
