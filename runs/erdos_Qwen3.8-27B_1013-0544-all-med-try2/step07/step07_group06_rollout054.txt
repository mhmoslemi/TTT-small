import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def adam_optimize(h, n, deadline, alpha_init=100.0, alpha_final=100000.0, lr_init=0.02):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    max_steps = 15000
    no_improve = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        t += 1
        frac = min(t / max_steps, 1.0)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        lr_cur = lr_init * (1.0 + 0.3 * t) ** (-0.5)
        g = fast_grad_smooth_max(h, n, alpha)
        m_arr = beta1 * m_arr + (1 - beta1) * g
        v_arr = beta2 * v_arr + (1 - beta2) * g ** 2
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        step_vec = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step_vec, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:

            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
            # Adaptive learning rate reduction
            if no_improve > 100:
                lr_init *= 0.8
                no_improve = 0
    return best_h, best_c

def lbfgs_phase(h, n, deadline, max_iters=300):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(50, 1.0), (200, 5.0), (1000, 20.0),
               (5000, 100.0), (20000, 500.0),
               (100000, 2000.0), (500000, 10000.0)]

        if time.time() > deadline - 1:
            break

        def gr(x):

            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:

            if c_new < best_c - 1e-15:

                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=None):

                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:

                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha, threshold=1e-10):
    c = np.correlate(h, 1 - h, mode='full')

def structured_init(n, block_sizes=None):
    """Create structured initial guess with alternating blocks."""
    if block_sizes is None:
        block_sizes = [3, 3, 5, 5, 7]
    h = np.zeros(n)
    pos = 0
    bit = 1
    bi = 0
    while pos < n:
        bs = block_sizes[bi % len(block_sizes)]
        end = min(pos + bs, n)
        h[pos:end] = float(bit)
        pos = end
        bit = 1 - bit
        bi += 1
    return project_feasible(h, n)
