    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return float(val), c

def smooth_max_grad(h, n, alpha):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n

    for m_idx in range(2 * n - 1):

            continue
        k = m_idx - (n - 1)

        if k >= 0:

            grad[k:n] += wm * (-h[0:nv])

            grad[kk:n] += wm * (1.0 - h[0:nv])

    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

        if w[m_idx] < 1e-14:

        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    v = smooth_max_val(x, n, alpha)
    v += mu * (x.sum() - n / 2.0) ** 2
    return v

def grad_pen(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    return smooth_max_val(x, n, alpha)

def grad_eq(x, n, alpha):
    return smooth_max_grad(x, n, alpha)

def con_fun(x, n):
    return x.sum() - n / 2.0

def con_jac(x, n):
    return np.ones(n)

def run_lbfgs(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = (200, 800, 3000, 12000, 50000, 200000, 800000)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        mu = max(10.0, alpha * 0.05)
        try:
            res = minimize(obj_pen, h, jac=grad_pen,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 800, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 40})

            c_new = compute_c5(h_new, n)
            if c_new < best_c:

def run_slsqp(h, n, deadline, alpha=200000, maxiter=2000):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
