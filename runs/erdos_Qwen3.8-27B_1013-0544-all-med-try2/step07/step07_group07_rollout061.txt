c[k] = sum_{valid i} (1/2)(1-1/2) dx = (1/4) * (number of valid pairs) * dx

The number of valid pairs for lag k is (n-|k|), so c[k] = (1/4)(n-|k|) * (2/n) = (1/2)(1-|k|/n).
The maximum is at k=0: c[0] = 1/2.

So the uniform solution gives C5 = 0.5. We need to do much better.

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _grad_single_lag(h, n, lag):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif 0 < lag < n:
        g[:n-lag] = (1.0 - h[lag:]) * dx
        g[lag:] = -h[:n-lag] * dx
    elif -n < lag < 0:
        m = -lag
        g[m:] = (1.0 - h[:n-m]) * dx
        g[:n-m] = -h[m:] * dx
    return g

        c_a = c[active_idx]
    else:
        c_a = c

    shifted = alpha * (c_a - m)
    mx = shifted.max()
    if mx > 500:
        e = np.exp(shifted - mx)
        se = mx + np.log(e.sum())
    elif mx < -500:
        idx = np.argmax(shifted)
        e = np.zeros_like(shifted)
        e[idx] = 1.0
        se = 0.0

        e = np.exp(shifted)
        se = np.log(e.sum())

        return smooth_max_fg(x, n, alpha, penalty, active_idx)[0]
```

```python
def balance_lags(h, n, top_m=3, max_iters=50):
    """Try to equalize the top lags by small adjustments."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_iters):
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-top_m:]
        vals = c[top_idx]
        target = vals.mean()
        # Compute gradients for each active lag
        grads = np.stack([_grad_single_lag(h, n, k-(n-1)) for k in top_idx])
        # Solve: find delta that minimizes sum of (c_k + grad_k·delta - target)^2
        # subject to sum(delta) = 0 (to maintain constraint)
        # This is a small QP in n variables, but we can use a simple approach
        # d(c_k)/dt = grad_k · delta
        # Want: c_k + grad_k·delta ≈ target for all k
        # Approximate: grad_k·delta ≈ target - c_k
        # This is an overdetermined system. Use least squares.
        # A @ delta = b where A = grads (m x n), b = target - vals
        A = grads  # shape (m, n)
        b = target - vals
        # Add constraint: sum(delta) = 0
        # Use Lagrange multiplier: minimize ||A@delta - b||^2 + lambda*(sum(delta))^2
        # Or just add a row of 1s to A and 0 to b with large weight
        weight = 10000
        A_ext = np.vstack([A, weight * np.ones((1, n))])
        b_ext = np.concatenate([b, [0.0]])
        delta, _, _, _ = np.linalg.lstsq(A_ext, b_ext, rcond=None)
        # Scale delta to be small
        scale = min(0.1, 0.5 / (np.max(np.abs(delta)) + 1e-10))
        delta *= scale
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            scale *= 0.1
            h_new = project_feasible(h + scale * delta, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
            else:
                break
    return h, best_c
```
