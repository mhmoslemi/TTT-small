def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

            break

                break

                # Compute the mass in the block

                target_mass = block.sum()

            return smooth_max_top(x, n, alpha, top_idx)
    else:
        def obj(x):
            return smooth_max_obj(x, n, alpha)
    try:

                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_all_lags(h, n):

        if abs(diff) < 1e-15:

            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final intensive polish on best solution
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        # Very high alpha SLSQP
        h_new = slsqp_optimize(best_h, n, 1000000, maxiter=8000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        # Binary push
        if time.time() < deadline - 10:
            best_h, best_c = binary_push(best_h, n, deadline - 10)
        # Active set with very large top
        if time.time() < deadline - 10:
            best_h, best_c = active_set_refine(best_h, n, deadline - 10, top_count=30, max_rounds=5)
        # Final pair search with tiny steps
        if time.time() < deadline - 5:
            best_h, best_c = pair_local_search(best_h, n, deadline - 5)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
