import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj(x, n, alpha, mu, topk, beta):
    c = np.correlate(x, 1 - x, mode='full')
    dx = 2.0 / n
    sh = alpha * (c - c.max())
    mx = sh.max()
    sm = (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = x.sum() - n / 2.0
    val = sm * dx + mu * s * s
    if beta > 0:
        val += beta * np.sum(x * (1 - x))
    return float(val)

def _grad(x, n, alpha, mu, topk, beta):
    c = np.correlate(x, 1 - x, mode='full')
    dx = 2.0 / n
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    if topk is not None and topk < len(w):
        idx = np.argpartition(-w, min(topk - 1, len(w) - 1))[:topk]
        w2 = np.zeros_like(w)
        w2[idx] = w[idx]
        s2 = w2.sum()
        if s2 > 0:
            w2 /= s2
        w = w2
    grad = np.zeros(n)
    active = w > 1e-12
    for m in np.where(active)[0]:
        wm = w[m] * dx
        kk = m - (n - 1)
        if kk >= 0:
            nv = n - kk
            grad[:nv] += wm * (1.0 - x[kk:])
            grad[kk:] += wm * (-x[:nv])
        else:
            kk2 = -kk
            nv = n - kk2
            grad[kk2:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk2:])
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    if beta > 0:
        grad += 2.0 * beta * (x - 0.5)
    return grad

def lbfgs_stage(h, n, alpha, mu, topk, beta, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    args = (n, alpha, mu, topk, beta)
    try:
        res = minimize(_obj, h, jac=_grad, args=args, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-19,
                                'gtol': 1e-17, 'maxfun': maxiter * 30})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def multi_alpha_lbfgs(h, n, deadline, topk=80):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    stages = [
        (30, 1.0, 0.0, 150),
        (100, 5.0, 0.0, 200),
        (500, 20.0, 0.0, 250),
        (2000, 100.0, 0.0, 250),
        (8000, 500.0, 0.0, 300),
        (30000, 2000.0, 0.0, 350),
        (100000, 8000.0, 0.0, 400),

    for alpha, mu, beta, mi in stages:
        if time.time() > deadline:
            break
        tr = deadline - time.time()
        if tr < 8:

        h = h_new

    rng = np.random.RandomState(seed)
    h = np.zeros(n)
