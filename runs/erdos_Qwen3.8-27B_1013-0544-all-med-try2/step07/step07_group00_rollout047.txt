from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    mx = c.max()
    sh = alpha * (c - mx)
    amx = sh.max()
    w = np.exp(sh - amx)
    w /= w.sum()
    val = mx + (amx + np.log(np.sum(np.exp(sh - amx)))) / alpha
    # gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return val, grad

def lbfgs_optimize(h, alpha, mu, maxiter=500):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        v, _ = smooth_max_and_grad(x, alpha)
        v += mu * (x.sum() - n/2.0)**2
        return v
    
    def grad(x):
        _, g = smooth_max_and_grad(x, alpha)
        g += 2.0 * mu * (x.sum() - n/2.0)
        return g
    
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})

    return h_new, compute_c5(h_new)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    mx = c.max()
    sh = alpha * (c - mx)
    amx = sh.max()
    w = np.exp(sh - amx)
    w /= w.sum()
    val = mx + (amx + np.log(w.sum())) / alpha
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return val, grad

def lbfgs_step(h, alpha, mu, maxiter=300):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        v, _ = smooth_max_grad(x, alpha)
        return float(v + mu * (x.sum() - n / 2.0) ** 2)

    def _grad(x):
        _, g = smooth_max_grad(x, alpha)
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g

    try:
        res = minimize(_obj, h, jac=_grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': maxiter * 40, 'maxls': 80})
        h_new = project_feasible(res.x)
        return h_new, compute_c5(h_new)
    except Exception:
        return h.copy(), compute_c5(h)

def slsqp_step(h, alpha, maxiter=2000):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        v, _ = smooth_max_grad(x, alpha)
        return float(v)

    def _grad(x):
        _, g = smooth_max_grad(x, alpha)
        return g

    def _eq(x):
        return float(x.sum() - n / 2.0)

    def _eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eq_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x)
        return h_new, compute_c5(h_new)
    except Exception:
        return h.copy(), compute_c5(h)

def pair_search(h, deadline, radius=50):

        while improved and time.time() < deadline:

def binary_project(h):
    n = len(h)
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1][:n // 2]
    h_b[idx] = 1.0
    return h_b
