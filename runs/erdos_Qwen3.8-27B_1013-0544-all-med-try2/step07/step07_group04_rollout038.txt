import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(g, alpha):
    m = g.max()
    shifted = np.clip(alpha * (g - m), -400, 400)
    return m + np.log(np.exp(shifted).sum()) / alpha

def slsqp_phase(h, n, deadline, alphas, maxiter=500):
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            return smooth_max(g, alpha)
        def con_fun(x):
            return x.sum() - target
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fun}],
                          options={'maxiter': maxiter, 'ftol': 1e-10})
            hp = project_feasible(res.x, n)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()
        except Exception:
            pass
    return best_h, best_c

def binary_greedy(h, n, deadline, max_sweeps=1000):
    dx = 2.0 / n
    target = n // 2
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_bin[idx[:target]] = 1.0
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h_bin, 1-h_bin, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h_bin.copy()
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in sorted(zeros):
            for j in sorted(ones):
                delta = np.zeros(2*n-1)
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h_bin[im[msk]]
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h_bin[imn[msk]]
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h_bin[jm[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h_bin[jmn[msk]]
                delta[n-1] -= 2.0

                if 0 <= k1 < 2*n-1: delta[k1] += 1.0

            nc = (corr + delta).max() * dx
            d = nc - cur_c
            if d < best_delta:
