def project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    n = len(v)
    if n == 0: return v
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
       mid=(lo+hi)*0.5
       s=float(np.clip(v-mid,0.0,1.0).sum())
       if s > target: lo=mid
       else: hi=mid
    u=(lo+hi)*0.5
    out=np.clip(v-u,0.0,1.0)
    diff=target-out.sum()
    if diff > 1e-15:
       idx=np.argsort(1.0-out)
       for i in idx:
          if diff <= 0: break
          room=1.0-out[i]
          add=min(room,diff)
          out[i]+=add; diff-=add
    elif diff < -1e-15:
       idx=np.argsort(out)
       for i in idx:
          if diff >= 0: break
          room=out[i]
          sub=min(room,-diff)
          out[i]-=sub; diff+=sub
    return out

from scipy.optimize import minimize
import time


def c_values(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def c5(h):
    return float(np.max(c_values(h)))


def project_box_sum(v, target):
    v = np.asarray(v, dtype=float).ravel()
    n = len(v)
    target = float(target)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= n:
        return np.ones(n, dtype=float)

    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid

    u = 0.5 * (lo + hi)
    out = np.clip(v - u, 0.0, 1.0)
    diff = target - float(out.sum())

    if diff > 1e-14:
        idx = np.argsort(1.0 - out)
        for i in idx:
            if diff <= 0.0:
                break
            room = 1.0 - out[i]
            if room <= 0.0:
                continue
            add = room if room < diff else diff
            out[i] += add
            diff -= add
    elif diff < -1e-14:
        idx = np.argsort(out)
        for i in idx:
            if diff >= 0.0:
                break
            room = out[i]
            if room <= 0.0:
                continue
            sub = room if room < -diff else -diff
            out[i] -= sub
            diff += sub
    return out


def smooth_weights(c, beta):
    m = float(c.max())
    z = float(beta) * (c - m)
    keep = z > -40.0
    w = np.zeros_like(c, dtype=float)
    if not np.any(keep):
        keep = (z == float(z.max()))
        w[keep] = 1.0 / float(c.size)
        return w
    z2 = z[keep] - float(z[keep].max())
    e = np.exp(z2)
    s = float(e.sum())
    if s <= 0.0:
        keep = (c == m)
        w[keep] = 1.0 / float(c.size)
        return w
    w[keep] = e / s
    return w


def grad_weighted(h, n, weights):
    g = np.zeros(n, dtype=float)
    if len(weights) == 0:
        return g
    for m in range(2 * n - 1):
        wm = float(weights[m])
        if abs(wm) < 1e-18:
            continue
        s = m - (n - 1)
        if s >= 0:
            nv = n - s
            if nv > 0:
                g[s:] += wm * (1.0 - h[:nv])
                g[:nv] -= wm * h[s:]
        else:
            kk = -s
            nv = n - kk
            if nv > 0:
                g[:nv] += wm * (1.0 - h[kk:])
                g[kk:] -= wm * h[:nv]
    return g


def aug_pen_obj(x, n, rho, mu=0.0):
    if not np.all(np.isfinite(x)):
        return 1e100
    h = x[:n]
    t = float(x[n])
    c = c_values(h)
    v = np.maximum(c - t, 0.0)
    val = float(t) + 0.5 * float(rho) * float(np.dot(v, v))
    if mu > 0.0:
        diff = float(h.sum() - 0.5 * n)
        val += float(mu) * diff * diff
    return val


def aug_pen_grad(x, n, rho, mu=0.0):
    if not np.all(np.isfinite(x)):
        return np.zeros(n + 1, dtype=float)
    h = x[:n]
    t = float(x[n])
    c = c_values(h)
    v = np.maximum(c - t, 0.0)
    g = grad_weighted(h, n, float(rho) * v)
    if mu > 0.0:
        diff = float(h.sum() - 0.5 * n)
        g += 2.0 * float(mu) * diff
    gt = 1.0 - float(rho) * float(v.sum())
    out = np.empty(n + 1, dtype=float)
    out[:n] = g
    out[n] = gt
    return out


def sum_constr(x, n):
    return np.array([float(x[:n].sum() - 0.5 * n)])


def sum_jac(x, n):
    m = len(x)
    j = np.zeros((1, m), dtype=float)
    if n < m:
        j[0, :n] = 1.0
    return j


def lbfgs_aug(h, n, deadline, best_c, maxiter=200):
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_c = c5(h)
    hcur = h.copy()

    configs = [
        (10.0, 0.1),
        (50.0, 1.0),
        (100.0, 1.0),
        (500.0, 10.0),
        (1000.0, 10.0),
        (5000.0, 100.0),
        (10000.0, 100.0),
        (50000.0, 1000.0),
        (100000.0, 1000.0),
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    for rho, mu in configs:
        if time.time() > deadline - 3.0:
            break
        c_cur = c5(hcur)
        t_start = max(0.0, c_cur - 1e-3)
        x0 = np.concatenate((hcur, np.array([t_start])))
        try:
            res = minimize(
                aug_pen_obj,
                x0,
                jac=aug_pen_grad,
                args=(n, float(rho), float(mu)),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": int(maxiter),
                    "ftol": 1e-15,
                    "gtol": 1e-12,
                    "maxfun": int(max(100, 20 * maxiter)),
                },
            )
            xr = res.x
        except Exception:
            xr = x0

        if xr is None or not np.all(np.isfinite(xr)):
            xr = x0
        hproj = project_box_sum(xr[:n], 0.5 * n)
        cproj = c5(hproj)
        if cproj < best_c - 1e-15:
            best_c = cproj
            best_h = hproj.copy()
        if cproj <= best_c + 1e-4:
            hcur = hproj
        else:
            hcur = best_h.copy()

    return best_h, best_c


def slsqp_polish(h, n, deadline, rho=5000.0, maxiter=800):
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_c = c5(h)
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{"type": "eq", "fun": sum_constr, "jac": sum_jac, "args": (n,)}]

    for rho_try in (float(rho), 10.0 * float(rho)):
        if time.time() > deadline - 2.0:
            break
        c_cur = c5(best_h)
        x0 = np.concatenate((best_h, np.array([max(0.0, c_cur - 1e-3)])))
        try:
            res = minimize(
                aug_pen_obj,
                x0,
                jac=aug_pen_grad,
                args=(n, rho_try, 0.0),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={
                    "maxiter": int(maxiter),
                    "ftol": 1e-15,
                    "maxfun": int(max(100, 10 * maxiter)),
                },
            )
            xr = res.x
        except Exception:
            xr = x0

        if xr is None or not np.all(np.isfinite(xr)):
            xr = x0
        h2 = project_box_sum(xr[:n], 0.5 * n)
        c2 = c5(h2)
        if c2 < best_c - 1e-15:
            best_c = c2
            best_h = h2.copy()

    return best_h, best_c


def active_polish(h, n, deadline):
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_c = c5(h)
    steps = (0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001)

    for beta in (5000.0, 20000.0, 80000.0):
        if time.time() > deadline - 1.0:
            break
        hcur = h.copy()
        stagnant = 0
        for _ in range(500):
            if time.time() > deadline - 1.0:
                break
            c = c_values(hcur)
            C = float(c.max())
            if C < best_c - 1e-15:
                best_c = C
                best_h = hcur.copy()

            g = grad_weighted(hcur, n, smooth_weights(c, beta))
            gn = float(np.linalg.norm(g))
            if gn < 1e-14:
                break
            p = -g / gn

            best_cc = C
            best_h2 = None
            for st in steps:
                h2 = project_box_sum(hcur + st * p, 0.5 * n)
                cc = c5(h2)
                if cc < best_cc - 1e-15:
                    best_cc = cc
                    best_h2 = h2

            if best_h2 is not None:
                hcur = best_h2
                stagnant = 0
            else:
                stagnant += 1
                if stagnant > 2:
                    break
        h = best_h

    return best_h, best_c


def eval_pair_move(h, a, b, d):
    ha = h[a] + d
    hb = h[b] - d
    if ha < -1e-12 or ha > 1.0 + 1e-12 or hb < -1e-12 or hb > 1.0 + 1e-12:
        return None
    if ha < 0.0:
        ha = 0.0
    elif ha > 1.0:
        ha = 1.0
    if hb < 0.0:
        hb = 0.0
    elif hb > 1.0:
        hb = 1.0
    hv = h.copy()
    hv[a] = ha
    hv[b] = hb
    return c5(hv), hv


def local_search(h, n, deadline, rng):
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_c = c5(h)
    fracs = (1.0, 0.5, 0.25, 0.1)
    K = 10
    R = 50
    no_improve = 0

    while time.time() < deadline - 1.0:
        c = c_values(h)
        C = float(c.max())
        if C < best_c - 1e-15:
            best_c = C
            best_h = h.copy()

        g = grad_weighted(h, n, smooth_weights(c, 8000.0))
        lows = np.argsort(g)[:K]
        highs = np.argsort(g)[::-1][:K]

        best_cc = best_c
        best_hc = None

        for a in lows:
            for b in highs:
                if a == b:
                    continue
                dmax = min(h[b], 1.0 - h[a])
                if dmax <= 1e-9:
                    continue
                for f in fracs:
                    d = dmax * f
                    if d <= 1e-12:
                        continue
                    res = eval_pair_move(h, a, b, d)
                    if res is None:
                        continue
                    cc, hv = res
                    if cc < best_cc - 1e-15:
                        best_cc = cc
                        best_hc = hv

        if n >= 2:
            pairs = rng.randint(0, n, size=(R, 2))
            for ii in range(R):
                a = int(pairs[ii, 0])
                b = int(pairs[ii, 1])
                if a == b:
                    continue
                if g[a] <= g[b]:
                    aa, bb = a, b
                else:
                    aa, bb = b, a
                dmax = min(h[bb], 1.0 - h[aa])
                if dmax <= 1e-9:
                    continue
                for f in fracs:
                    d = dmax * f
                    if d <= 1e-12:
                        continue
                    res = eval_pair_move(h, aa, bb, d)
                    if res is None:
                        continue
                    cc, hv = res
                    if cc < best_cc - 1e-15:
                        best_cc = cc
                        best_hc = hv

        if best_hc is not None:
            h = best_hc
            best_c = best_cc
            best_h = h.copy()
            no_improve = 0
            continue

        no_improve += 1
        if no_improve > 2:
            improved = False
            for _ in range(40):
                if time.time() > deadline - 1.0:
                    break
                if rng.rand() < 0.25:
                    h2 = project_box_sum(h + 0.005 * rng.randn(n), 0.5 * n)
                else:
                    h2 = h.copy()
                    l = int(rng.randint(2, min(6, max(2, n // 4)) + 1))
                    if l >= n:
                        continue
                    i = int(rng.randint(0, n - l))
                    j = int(rng.randint(0, n - l))
                    if abs(i - j) < l:
                        continue
                    h2[i:i + l], h2[j:j + l] = h2[j:j + l], h2[i:i + l]

                cc = c5(h2)
                if cc < best_c - 1e-15:
                    h = h2
                    best_c = cc
                    best_h = h.copy()
                    improved = True
                    break
            if not improved:
                break

    return best_h, best_c


def interpolate_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x0 = 2.0 * (np.arange(n_from) + 0.5) / n_from
    x1 = 2.0 * (np.arange(n_to) + 0.5) / n_to
    y = np.interp(x1, x0, h)
    return project_box_sum(y, 0.5 * n_to)


def optimize_size(h_from, n_from, n_to, deadline, rng):
    h = interpolate_to(h_from, n_from, n_to)
    c = c5(h)
    rem = deadline - time.time()
    if rem > 25.0:
        t1 = time.time() + 0.55 * rem
        if t1 > deadline:
            t1 = deadline - 1.0
        h, c = lbfgs_aug(h, n_to, t1, c, maxiter=100)

    rem = deadline - time.time()
    if rem > 15.0:
        t2 = time.time() + 0.60 * rem
        if t2 > deadline:
            t2 = deadline - 1.0
        h, c = local_search(h, n_to, t2, rng)

    return h, c


def suggested_ns(n0):
    out = []
    cands = [
        int(n0),
        int(n0) - 1,
        int(n0) + 1,
        int(n0) - 25,
        int(n0) + 25,
        int(n0) // 2,
        int((3 * n0) // 2),
        200,
        300,
        400,
        500,
        600,
        700,
    ]
    for m in cands:
        if 50 <= m <= 1000 and m not in out:
            out.append(m)
    return out[:9]


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = float(budget_s)
    if B > 30.0:
        deadline = t0 + B - 5.0
    else:
        deadline = t0 + max(5.0, B - 1.0)
    if deadline > t0 + B - 0.2:
        deadline = t0 + B - 0.2

    rng = np.random.RandomState(seed)

    h = None
    try:
        h = np.asarray(initial_h_values, dtype=float).ravel()
    except Exception:
        h = None

    if h is None or len(h) == 0:
        n = 300
        r = rng.rand(n)
        h = np.zeros(n, dtype=float)
        h[np.argsort(r)[n // 2:]] = 1.0
    else:
        n = len(h)
        if n > 1000:
            h = interpolate_to(h, n, 600)
            n = 600
        if n < 50:
            h = interpolate_to(h, n, 100)
            n = 100

    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_c = c5(h)
    best_n = n

    if time.time() < deadline - 60.0:
        t1 = min(deadline - 500.0, time.time() + 300.0)
        if t1 < time.time() + 20.0:
            t1 = time.time() + max(10.0, min(120.0, (deadline - time.time()) * 0.4))
        if time.time() < t1:
            best_h, best_c = lbfgs_aug(best_h, best_n, t1, best_c, maxiter=250)

    if time.time() < deadline - 100.0:
        t2 = min(deadline - 350.0, time.time() + 70.0)
        if t2 < time.time() + 10.0:
            t2 = time.time() + min(40.0, (deadline - time.time()) * 0.4)
        if time.time() < t2:
            best_h, best_c = slsqp_polish(best_h, best_n, t2, rho=5000.0, maxiter=800)

    if time.time() < deadline - 80.0:
        t3 = min(deadline - 220.0, time.time() + 70.0)
        if t3 < time.time() + 5.0:
            t3 = time.time() + min(20.0, (deadline - time.time()) * 0.4)
        if time.time() < t3:
            best_h, best_c = active_polish(best_h, best_n, t3)

    if time.time() < deadline - 70.0:
        t4 = min(deadline - 120.0, time.time() + 160.0)
        if t4 < time.time() + 5.0:
            t4 = time.time() + min(40.0, (deadline - time.time()) * 0.5)
        if time.time() < t4:
            best_h, best_c = local_search(best_h, best_n, t4, rng)

    if best_n % 2 == 0 and time.time() < deadline - 200.0:
        hq = np.zeros(best_n, dtype=float)
        idx = np.argsort(best_h)
        hq[idx[best_n // 2:]] = 1.0
        hq = project_box_sum(hq, 0.5 * best_n)
        t5 = min(deadline - 180.0, time.time() + 60.0)
        if t5 < time.time() + 5.0:
            t5 = time.time() + min(20.0, (deadline - time.time()) * 0.3)
        if time.time() < t5:
            hq, cq = local_search(hq, best_n, t5, rng)
            if cq < best_c - 1e-15:
                best_c = cq
                best_h = hq.copy()

    if time.time() < deadline - 100.0:
        ns = [m for m in suggested_ns(best_n) if m != best_n]
        for i, m in enumerate(ns):
            left = deadline - time.time() - 30.0
            if left < 25.0:
                break
            alloc = min(70.0, left / max(1, len(ns) - i))
            if alloc < 25.0:
                break
            h_try, c_try = optimize_size(best_h, best_n, m, time.time() + alloc, rng)
            if c_try < best_c - 1e-15:
                best_c = c_try
                best_h = h_try.copy()
                best_n = int(m)

    if time.time() < deadline - 15.0:
        tf = min(deadline - 5.0, time.time() + 40.0)
        if time.time() < tf:
            best_h, best_c = active_polish(best_h, best_n, tf)

    if time.time() < deadline - 2.0:
        best_h, best_c = local_search(best_h, best_n, deadline - 2.0, rng)

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = c5(best_h)
    return best_h, float(best_c), int(best_n)
