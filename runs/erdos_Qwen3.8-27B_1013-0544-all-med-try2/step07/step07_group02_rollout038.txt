def fast_coord_descent(h, n, max_sweeps=500, deadline=None, tol=1e-10):
    """
    Fast coordinate descent using the structure of the bilinear form.
    For each coordinate i, compute B[k] = change in c[k] per unit change in h[i].
    Then find optimal h[i] by minimizing max_k (c[k] + B[k]*(x - h[i])).
    """
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        improved_total = 0.0
        for i in range(n):
            # Compute B[k] = dc[k]/dh[i]
            # c[k] = sum_j h[j+k]*(1-h[j])
            # dc[k]/dh[i] = (1-h[i-k]) if i-k valid (from term h[i] as j+k)
            #              + (-h[i+k]) if i+k valid (from term (1-h[i]) as (1-h[j]))
            B = np.zeros(2 * n - 1)
            # Contribution from h[i] appearing as h[j+k] where j+k=i, j=i-k
            j_val = i - m_range  # j = i - k for each k
            mask = (j_val >= 0) & (j_val < n)
            B[mask] += (1.0 - h[j_val[mask]])
            # Contribution from h[i] appearing in (1-h[j]) where j=i
            k_val = i + m_range  # k such that i+k is valid
            mask2 = (i + m_range >= 0) & (i + m_range < n)

            # Let d = x - h[i], so d in [-h[i], 1-h[i]]
            # g(d) = max_k (corr[k] + B[k]*d)
            # This is convex in d. Find minimum.
            
            hi_d = 1.0 - h[i]
            lo_d = -h[i]
            
            # Binary search on the subgradient
            # g(d) is convex, so g'(d) is non-decreasing
            # g'(d) = max_{k in argmax} B[k]
            # We want where g'(d) >= 0 and g'(d-) <= 0
            
            for _ in range(50):
                mid = (lo_d + hi_d) / 2.0
                vals = corr + B * mid
                max_idx = np.where(vals == vals.max())[0]

                sg_lo = B[max_idx].min()
                sg_hi = B[max_idx].max()
                if sg_lo >= 0:
                    hi_d = mid
                elif sg_hi <= 0:
                    lo_d = mid
                else:

                    break

                h[i] = x_opt
                improved_total += abs(x_opt - h[i])

            break
    
    return h, corr

    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=200, deadline=None, n_bisect=50):

    h = h.astype(float).copy()
    corr = corr_vector(h, n)
    
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        max_change = 0.0
        for i in range(n):
            B = np.zeros(2 * n - 1)
            j_vals = i - m_range
            mask1 = (j_vals >= 0) & (j_vals < n)
            B[mask1] += (1.0 - h[j_vals[mask1]])
            k_vals = i + m_range
            mask2 = (k_vals >= 0) & (k_vals < n)
            B[mask2] -= h[k_vals[mask2]]
            
            # Binary search for optimal offset d = x - h[i] in [-h[i], 1-h[i]]
            lo = -h[i]
            hi = 1.0 - h[i]
            for _ in range(n_bisect):
                mid = (lo + hi) / 2.0
                vals = corr + B * mid
                mx = vals.max()
                at_mx = vals == mx
                sg_lo = B[at_mx].min()
                sg_hi = B[at_mx].max()

                    hi = mid
                elif sg_hi <= 0:
                    lo = mid
                else:
                    break

                max_change = max(max_change, abs(x_opt - h[i]))
                h[i] = x_opt

        m = g.max()
        shifted = alpha * (g - m)
        # numerically stable log-sum-exp
        shifted_max = shifted.max()
        sm_val = shifted_max + np.log(np.exp(shifted - shifted_max).sum()) / alpha
        return m + sm_val / alpha

        # = cross-correlation of active (as a kernel) with h

            s1 = 0.0
            s2 = 0.0
            for k in k_shifts:
                j = i - k
                if 0 <= j < n:
                    s1 += (1.0 - h[j])
                j2 = i + k
                if 0 <= j2 < n:
                    s2 += h[j2]

        c_val = corr.max() * dx
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        
        # Decay step size
        step *= 0.999

            if s < 1e-15:

    h = h.astype(float).copy()
    corr = corr_vec(h)

            jv = i - m_range
            m1 = (jv >= 0) & (jv < n)

            kv = i + m_range

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def corr_vec(h):
    return np.correlate(h, 1 - h, mode='full').astype(float)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=100, deadline=None, n_bisect=35):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = corr_vec(h)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        max_change = 0.0
        for i in range(n):
            B = np.zeros(2 * n - 1)
            jv = i - m_range
            m1 = (jv >= 0) & (jv < n)
            B[m1] += (1.0 - h[jv[m1]])
            kv = i + m_range
            m2 = (kv >= 0) & (kv < n)
            B[m2] -= h[kv[m2]]
            lo = -h[i]
            hi = 1.0 - h[i]
            for _ in range(n_bisect):
                mid = (lo + hi) / 2.0
                vals = corr + B * mid
                mx = vals.max()
                at_mx = vals >= mx - 1e-15
                sg_lo = B[at_mx].min()
                sg_hi = B[at_mx].max()
                if sg_lo >= 0:
                    hi = mid
                elif sg_hi <= 0:
                    lo = mid
                else:
                    break
            d_opt = (lo + hi) / 2.0
            x_opt = np.clip(h[i] + d_opt, 0.0, 1.0)

                corr += B * delta
                h[i] = x_opt
                max_change = max(max_change, abs(delta))
        if max_change < 1e-13:

    return h, corr

    dx = 2.0 / n

        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sh_max = shifted.max()
        lse = sh_max + np.log(np.exp(shifted - sh_max).sum())
        return m + lse / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def pnorm_obj(h, n, p, maxiter=2000):
    """Minimize p-norm of correlation vector using SLSQP."""
    dx = 2.0 / n

        g = np.correlate(x, 1.0 - x, mode='full') * dx
        g_pos = np.maximum(g, 1e-30)
        return np.sum(g_pos ** p) ** (1.0 / p)
    bounds = [(0.0, 1.0)] * n

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
