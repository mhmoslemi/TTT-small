```
def project_box_mean(y, target):
    y=np.asarray(y,dtype=float)
    lo=float(np.min(y))-1.0
    hi=float(np.max(y))+1.0
    for _ in range(70):
       mid=(lo+hi)*0.5
       s=float(np.clip(y-mid,0.0,1.0).sum())
       if s > target: lo=mid
       else: hi=mid
    lam=(lo+hi)*0.5
    h=np.clip(y-lam,0.0,1.0)
    d=target-h.sum()
    # small correction
    for _ in range(50):
       if abs(d)<1e-14: break
       if d>0:
          room=1.0-h; t=room.sum()
          if t<1e-15: break
          h += (d/t)*room
       else:
          t=h.sum()
          if t<1e-15: break
          h += (d/t)*h
    return np.clip(h,0.0,1.0)
```

```python
import numpy as np
import time
from scipy.optimize import minimize

LBFGS_CONFIGS = (
    (100.0, 1e-3, 100),
    (500.0, 0.01, 150),
    (2000.0, 0.1, 150),
    (10000.0, 1.0, 120),
    (50000.0, 10.0, 80),
)
SMALL_LBFGS_CONFIGS = (
    (1000.0, 0.1, 80),
    (10000.0, 1.0, 60),

def project_box_mean(y, target):
    y = np.asarray(y, dtype=float)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(y - lam, 0.0, 1.0)


def corr_counts(h):
    return np.correlate(h, 1.0 - h, mode='full')


def max_count(h):
    return float(np.max(corr_counts(h)))


def c5_from_max_count(count, n):
    return float(count * (2.0 / float(n)))


def c5_count(h):
    return c5_from_max_count(max_count(h), len(h))


def soft_weights(c, alpha):
    m = float(np.max(c))
    x = alpha * (c - m)
    x = x - float(np.max(x))
    e = np.exp(x)
    s = float(np.sum(e))
    if s <= 0.0:
        return np.full(c.shape, 1.0 / float(c.size), dtype=float)
    return e / s


def smooth_max(c, alpha):
    m = float(np.max(c))
    x = alpha * (c - m)
    x = x - float(np.max(x))
    e = np.exp(x)
    s = float(np.sum(e))
    return m + float(np.log(max(s, 1e-300))) / alpha


def grad_counts(h, w, n):
    g = np.zeros(n, dtype=float)
    z = n - 1
    tol = 1e-10
    w0 = float(w[z])
    if w0 > tol:
        g += w0 * (1.0 - 2.0 * h)
    for k in range(1, n):
        nv = n - k
        wl = float(w[z - k])
        if wl > tol:
            g[0:nv] += wl * (1.0 - h[k:n])
            g[k:n] -= wl * h[0:nv]
        wr = float(w[z + k])
        if wr > tol:
            g[k:n] += wr * (1.0 - h[0:nv])
            g[0:nv] -= wr * h[k:n]
    return g


def penalty_obj(x, n, alpha, mu):
    c = corr_counts(x)
    val = smooth_max(c, alpha)
    s = float(np.sum(x) - 0.5 * float(n))
    return val + mu * s * s


def penalty_jac(x, n, alpha, mu):
    n = int(x.size)
    c = corr_counts(x)
    w = soft_weights(c, alpha)
    g = grad_counts(x, w, n)
    s = float(np.sum(x) - 0.5 * float(n))
    g += 2.0 * mu * s
    return g


def run_lbfgs(h, n, deadline, configs):
    h = project_box_mean(h, 0.5 * float(n))
    best_c = c5_count(h)
    bounds = [(0.0, 1.0)] * int(n)
    for alpha, mu, maxiter in configs:
        if time.time() > deadline - 3.0:
            break
        try:
            res = minimize(
                penalty_obj,
                h,
                args=(int(n), alpha, mu),
                jac=penalty_jac,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': int(maxiter), 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50},
            )
            h2 = project_box_mean(res.x, 0.5 * float(n))
            c2 = c5_count(h2)
            if c2 < best_c and np.all(np.isfinite(h2)):
                best_c = c2
                h = h2
        except Exception:
            pass
    return h, best_c


def smooth_obj(x, n, alpha):
    return smooth_max(corr_counts(x), alpha)


def smooth_jac(x, n, alpha):
    n = int(x.size)
    c = corr_counts(x)
    w = soft_weights(c, alpha)
    return grad_counts(x, w, n)


def sum_eq(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def sum_jac(x, n):
    return np.ones((1, int(x.size)), dtype=float)


def run_slsqp(h, n, alpha, deadline, maxiter):
    h = project_box_mean(h, 0.5 * float(n))
    best_c = c5_count(h)
    if time.time() > deadline - 5.0:
        return h, best_c
    bounds = [(0.0, 1.0)] * int(n)
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac, 'args': (int(n),)}]
    try:
        res = minimize(
            smooth_obj,
            h,
            args=(int(n), alpha),
            jac=smooth_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-12, 'maxls': 50},
        )
        h2 = project_box_mean(res.x, 0.5 * float(n))
        c2 = c5_count(h2)
        if c2 < best_c and np.all(np.isfinite(h2)):
            return h2, c2
    except Exception:
        pass
    return h, best_c

    for u in ts:
        if abs(t - u) < 1e-9:

            add_unique_ts(ts, -neg * f)
    if pos > 0.2:
        for t in (0.05, 0.1, 0.2, 0.4):
            if t < pos:
                add_unique_ts(ts, t)

    best_t = 0.0
    best = best_count
    for t in ts[:14]:
        if t > 0.0:
            t = min(t, pos)
        elif t < 0.0:
            t = max(t, -neg)
        if abs(t) < 1e-8:
            continue
        h[dst] += t
        h[src] -= t
        cnt = max_count(h)
        h[dst] -= t
        h[src] += t
        if cnt < best - 1e-12:
            best = cnt
            best_t = t

    if best_t == 0.0:
        return None

    base = best_t
    extra = []
    for f in (0.15, 0.25, 0.5, 0.75, 1.25, 1.5, 2.0):
        add_unique_ts(extra, base * f)
    step = 0.3 * abs(base)
    for d in (-2.0 * step, -step, step, 2.0 * step):
        add_unique_ts(extra, base + d)

    for t in extra:
        if t > 0.0:
            if t > pos:
                continue
        elif t < 0.0:
            if t < -neg:
                continue
        else:
            continue
        if abs(t) < 1e-9:
            continue
        h[dst] += t
        h[src] -= t
        cnt = max_count(h)
        h[dst] -= t
        h[src] += t
        if cnt < best - 1e-12:
            best = cnt
            best_t = t

    if best_t == 0.0 or best >= best_count:
        return None
    h[dst] += best_t
    h[src] -= best_t
    return float(best)


def polish_pairs(h, n, deadline, alphas, top_pairs):
    h = project_box_mean(h, 0.5 * float(n))
    best_count = max_count(h)
    best_c = c5_from_max_count(best_count, n)

    for alpha in alphas:
        stall = 0
        while time.time() < deadline - 1.0 and stall < 5:
            c = corr_counts(h)
            cur = float(np.max(c))
            if cur < best_count:
                best_count = cur
                best_c = c5_from_max_count(best_count, n)

            w = soft_weights(c, alpha)
            g = grad_counts(h, w, n)
            order = np.argsort(g)
            K = min(35, max(8, int(n) // 2))
            lows = order[:K]
            highs = order[-K:]

            cands = []
            for a in highs:
                a = int(a)
                ga = float(g[a])
                if float(h[a]) <= 1e-9:
                    continue
                for b in lows:
                    b = int(b)
                    if b == a:
                        continue
                    gb = float(g[b])
                    if ga <= gb:
                        continue
                    lim = min(float(h[a]), float(1.0 - h[b]))
                    if lim < 1e-6:
                        continue
                    cands.append((ga - gb, a, b))

            cands.sort(reverse=True)
            improved = False
            for _, s, d in cands[:int(top_pairs)]:
                if time.time() > deadline - 1.0:
                    break
                res = pair_line(h, s, d, best_count)
                if res is not None:
                    best_count = res
                    best_c = c5_from_max_count(best_count, n)
                    improved = True
            if improved:
                stall = 0
            else:
                stall += 1
    return h, best_c


def resample_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, int(n_old), endpoint=False)
    x_new = np.linspace(0.0, 2.0, int(n_new), endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_box_mean(h_new, 0.5 * float(n_new))


def random_start(n, rng, amp=0.55):
    n = int(n)
    x = np.arange(n)
    p = np.zeros(n, dtype=float)
    for m in range(1, 6):
        p += (amp / (float(m) ** 0.75)) * np.cos(2.0 * np.pi * float(m) * x / float(n) + 2.0 * np.pi * rng.random())
    if n % 2 == 0 and rng.random() < 0.5:
        h = np.zeros(n, dtype=float)
        idx = np.argsort(p)
        h[idx[-n // 2:]] = 1.0
        return h
    h = 0.5 * (1.0 + p)
    return project_box_mean(h, 0.5 * float(n))


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(int(seed))

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n0 = 300
        h = random_start(n0, rng, amp=0.6)

    n = int(h.size)
    h = project_box_mean(h, 0.5 * float(n))
    best_h = h.copy()
    best_c = c5_count(h)
    best_n = n

    end = min(t0 + 110.0, deadline - 760.0)
    if end > time.time() + 5.0:
        h2, c2 = polish_pairs(best_h, best_n, end, (5000.0, 1000.0, 100.0), 60)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            best_n = int(h2.size)
        h = best_h.copy()

    end = min(t0 + 430.0, deadline - 520.0)
    if end > time.time() + 5.0:
        h2, c2 = run_lbfgs(h, best_n, end, LBFGS_CONFIGS)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            best_n = int(h2.size)
        h = best_h.copy()

    if time.time() < deadline - 360.0:
        t_end = min(time.time() + 70.0, deadline - 320.0)
        h2, c2 = run_slsqp(h, best_n, 5000.0, t_end, 80)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            best_n = int(h2.size)
            h = h2.copy()

    if time.time() < deadline - 300.0:
        t_end = min(time.time() + 60.0, deadline - 300.0)
        h2, c2 = run_slsqp(h, best_n, 15000.0, t_end, 70)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            best_n = int(h2.size)
            h = h2.copy()

    end = min(t0 + 820.0, deadline - 210.0)
    if end > time.time() + 5.0:
        h2, c2 = polish_pairs(h, best_n, end, (20000.0, 5000.0, 1000.0, 100.0), 90)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            best_n = int(h2.size)
        h = best_h.copy()

    if time.time() < deadline - 160.0:
        for n_try in (240, 200, 400, 600):
            remain = deadline - time.time()
            if remain < 110.0:
                break
            alloc = min(80.0, remain - 80.0)
            if alloc < 25.0:
                break
            ht = resample_h(best_h, best_n, n_try)
            t1 = time.time() + max(20.0, 0.45 * alloc)
            t1 = min(t1, deadline - 90.0)
            ht, c1 = polish_pairs(ht, n_try, t1, (5000.0, 1000.0, 100.0), 50)
            if c1 < best_c:
                best_c = c1
                best_h = ht.copy()
                best_n = n_try
            if time.time() < deadline - 90.0:
                t2 = min(time.time() + max(15.0, 0.25 * alloc), deadline - 80.0)
                ht, c2 = run_lbfgs(ht, n_try, t2, SMALL_LBFGS_CONFIGS)
                if c2 < best_c:
                    best_c = c2
                    best_h = ht.copy()
                    best_n = n_try
            if time.time() < deadline - 75.0:
                t3 = min(time.time() + max(15.0, 0.2 * alloc), deadline - 70.0)
                ht, c3 = polish_pairs(ht, n_try, t3, (20000.0, 5000.0), 45)
                if c3 < best_c:
                    best_c = c3
                    best_h = ht.copy()
                    best_n = n_try

    if time.time() < deadline - 90.0:
        n_use = int(best_n)
        for _ in range(3):
            remain = deadline - time.time()
            if remain < 70.0:
                break
            alloc = min(40.0, remain - 70.0)
            if alloc < 12.0:
                break
            hr = random_start(n_use, rng, amp=0.6)
            tr = time.time() + max(10.0, 0.8 * alloc)
            tr = min(tr, deadline - 70.0)
            hr, cr = polish_pairs(hr, n_use, tr, (5000.0, 1000.0, 100.0), 40)
            if cr < best_c:
                best_c = cr
                best_h = hr.copy()
                best_n = n_use

    if time.time() < deadline - 2.0:
        h = best_h.copy()
        h2, c2 = polish_pairs(h, best_n, deadline - 1.0, (50000.0, 20000.0, 5000.0), 70)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    best_h = project_box_mean(best_h, 0.5 * float(len(best_h)))
    best_c = c5_count(best_h)
    return best_h, float(best_c), int(len(best_h))
```
