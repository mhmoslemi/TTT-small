```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(len(x))

def obj_lse(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    shift = c - c.max()
    mx = shift.max()
    if mx > 500:
        return float(c.max() + (mx + np.log(np.sum(np.exp(shift - mx)))) / alpha)
    return float(c.max() + np.log(np.sum(np.exp(shift))) / alpha)

def slsqp(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_descent(h, n, deadline, rng, lr=0.001, n_iters=None):
    """Subgradient descent on the max of correlation"""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    while time.time() < deadline:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        m_star = max_idx - (n - 1)
        
        # Compute subgradient of c[m_star] w.r.t. h
        grad = np.zeros(n)
        if m_star >= 0:
            valid1 = np.arange(n - m_star)
            grad[valid1] += (1.0 - h[valid1 + m_star])
            valid2 = np.arange(m_star, n)
            grad[valid2] -= h[valid2 - m_star]
        else:
            m = -m_star
            valid1 = np.arange(m, n)
            grad[valid1] += (1.0 - h[valid1 - m])
            valid2 = np.arange(n - m)
            grad[valid2] -= h[valid2 + m]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        # Move in negative gradient direction (but preserve sum)
        # Project gradient to sum-zero subspace
        grad_mean = grad.mean()
        grad_centered = grad - grad_mean
        
        h_new = h - lr * grad_centered
        h_new = project_feasible(h_new, n)
        
        new_c = compute_c5(h_new, n)
        if new_c < best_c - 1e-15:
            best_c = new_c
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller step
            h_new = h - lr * 0.1 * grad_centered
            h_new = project_feasible(h_new, n)
            new_c = compute_c5(h_new, n)
            if new_c < best_c - 1e-15:
                best_c = new_c
                best_h = h_new.copy()
                h = h_new
            lr *= 0.95
            if lr < 1e-8:
                lr = 0.001  # reset
    
    return best_h, best_c

def sa_fine(h, n, deadline, rng):
    """Fine-grained SA"""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    cur_c = best_c
    T = 0.002
    step = 0.005
    no_improve = 0
    patience = 100000

            di, dj = -d, d

        nj = h[j] + dj

        h[i] = ni
```

```python
def subgradient(h, n, max_idx):
    m = max_idx - (n - 1)
    grad = np.zeros(n)
    if m >= 0:
        # For p=0,...,n-1:
        # Term 1: (1-h[p-m]) if 0 <= p-m <= n-1-m, i.e., m <= p <= n-1
        # Term 2: -h[p+m] if 0 <= p+m <= n-1, i.e., 0 <= p <= n-1-m
        for p in range(n):
            if m <= p <= n - 1:
                grad[p] += (1.0 - h[p - m])
            if 0 <= p <= n - 1 - m:
                grad[p] -= h[p + m]
    else:

        for p in range(n):
            if 0 <= p <= n - 1 - m2:
                grad[p] += (1.0 - h[p + m2])
            if m2 <= p <= n - 1:
                grad[p] -= h[p - m2]
    return grad
```

```python
def subgradient_vec(h, n, max_idx):
    m = max_idx - (n - 1)
    grad = np.zeros(n)
    if m >= 0:
        # Term 1: (1-h[p-m]) for p in [m, n-1]
        idx1 = np.arange(m, n)
        grad[idx1] += (1.0 - h[idx1 - m])
        # Term 2: -h[p+m] for p in [0, n-1-m]
        idx2 = np.arange(0, n - m)
        grad[idx2] -= h[idx2 + m]
    else:
        m2 = -m
        # Term 1: (1-h[p+m2]) for p in [0, n-1-m2]
        idx1 = np.arange(0, n - m2)
        grad[idx1] += (1.0 - h[idx1 + m2])
        # Term 2: -h[p-m2] for p in [m2, n-1]
        idx2 = np.arange(m2, n)
        grad[idx2] -= h[idx2 - m2]
    return grad
```

```python
def multi_subgradient(h, n, tol=0.01):

    max_val = corr.max()
    active = np.where(corr > max_val - tol)[0]

        grad += subgradient_vec(h, n, idx)
    grad /= len(active)
```

Now, the key issue with subgradient descent is that it's slow to converge and oscillates near the optimum. For a problem where we need an improvement of 0.00007, we need a more targeted approach.

**Better idea: Use the subgradient to guide the pair search**

Instead of trying all pairs, use the subgradient to determine:
- Which positions should decrease (negative gradient)
- Which positions should increase (positive gradient)
- Only try pairs (i, j) where grad[i] < 0 and grad[j] > 0

This should be much more efficient than the exhaustive pair search.

```python
def guided_pair_search(h, n, deadline, steps):
    best_c = compute_c5(h, n)
    for d in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            corr = np.correlate(h, 1-h, mode='full')
            max_val = corr.max()
            active = np.where(corr > max_val - 0.01)[0]
            
            grad = np.zeros(n)
            for idx in active:
                grad += subgradient_vec(h, n, idx)
            grad /= len(active)
            grad -= grad.mean()  # center
            
            dec_pos = np.where(grad < -0.01)[0]
            inc_pos = np.where(grad > 0.01)[0]
            
            for i in dec_pos:
                if time.time() > deadline:
                    break
                for j in inc_pos:
                    if i == j:
                        continue
                    if h[i] - d >= 0 and h[j] + d <= 1:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                    if h[i] + d <= 1 and h[j] - d >= 0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
    return h, best_c
```

This should be much faster than the exhaustive pair search because we only try a subset of pairs.

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(len(x))

def obj_lse(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    shift = c - c.max()
    mx = shift.max()
    if mx > 500:
        return float(c.max() + (mx + np.log(np.sum(np.exp(shift - mx)))) / alpha)
    return float(c.max() + np.log(np.sum(np.exp(shift))) / alpha)
