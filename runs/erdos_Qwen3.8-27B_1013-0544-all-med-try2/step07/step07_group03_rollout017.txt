The projection is the key challenge. Projecting to the intersection of a box and a hyperplane.

The projection of a point y onto {x : 0 <= x_i <= 1, sum(x) = n/2} can be done by:
- Start with x = clip(y, 0, 1)
- Iteratively adjust: if sum(x) > n/2, reduce; if sum(x) < n/2, increase
- The optimal way is to find a threshold lambda such that x_i = clip(y_i - lambda, 0, 1) and sum(x_i) = n/2
- This is a 1D root-finding problem (binary search on lambda)

    # f is decreasing in lambda
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(y - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0, 1)

def projected_gradient_descent(h, n, deadline, top_k=10, step_init=0.01):
    cur_corr = compute_correlation(h, n)
    cur_c = cur_corr.max() * (2.0/n)
    best_h = h.copy()
    best_c = cur_c
    step = step_init
    
    while time.time() < deadline:
        # Find top-k active lags
        top_idx = np.argsort(cur_corr)[-top_k:]
        lags = top_idx - (n - 1)
        
        # Compute gradient
        grad = np.zeros(n)
        for lag in lags:
            grad += gradient_at_lag(h, n, int(lag))
        grad *= (2.0 / n) / len(lags)  # normalize
        
        # Projected gradient step with backtracking
        for _ in range(20):
            new_h = project_feasible(h - step * grad, n)
            new_corr = compute_correlation(new_h, n)
            new_c = new_corr.max() * (2.0 / n)
            if new_c < cur_c:
                h = new_h
                cur_corr = new_corr
                cur_c = new_c
                step *= 1.1  # increase step slightly
                break
            else:
                step *= 0.5
        else:
            # Couldn't improve with this gradient, try smaller top_k or restart
            break
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:

    return np.clip(y - (lo + hi) / 2, 0.0, 1.0)

def gradient_at_lag(h, n, lag):

Wait, actually computing the full gradient IS O(n²) because there are 2n-1 lags and each takes O(n). So total is O(n²) which is the same as the correlation computation. So I might as well just compute the full gradient directly.

But the issue is: if I use the full gradient (all lags equally weighted), the max function isn't well-approximated. The subgradient of the max should only use the active lags (argmax). Using all lags gives the gradient of the SUM of all correlations, not the max.

Let me also think about the binary approach more carefully. For binary h (values 0 or 1), the correlation at lag m is:
sum_i h[i]*(1-h[i+m]) = |A_m| * 1 - (# of positions where h[i]=1 and h[i+m]=1)

where A_m is the set of valid i. Actually for binary:
h[i]*(1-h[i+m]) = 1 if h[i]=1 and h[i+m]=0, else 0.

So corr[m] = number of i where h[i]=1 and h[i+m]=0 (for valid i).

This equals: (number of 1s in the overlap region) - (number of positions where both h[i]=1 and h[i+m]=1)

For the full range (lag=0): corr[0] = sum h[i]*(1-h[i]) = 0 for binary.

For lag != 0: it's the number of "1->0" transitions at that displacement.

So for binary, maximizing the minimum "separation" is related to avoiding close 1s.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def grad_at_lag(h, n, lag):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] += 1.0 - h[lag:]
        g[lag:] -= h[:n - lag]
    else:
        m = -lag
        g[m:] += 1.0 - h[:n - m]
        g[:n - m] -= h[m:]
    return g

def projected_subgrad(h, n, deadline, top_k=5, step_init=0.03):
    dx = 2.0 / n
    cur_corr = compute_corr(h, n)
    cur_c = cur_corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    step = step_init
    no_improve = 0
    while time.time() < deadline:
        top_idx = np.argsort(cur_corr)[-top_k:]
        grad = np.zeros(n)
        for li in top_idx:
            grad += grad_at_lag(h, n, int(li - (n - 1)))
        grad *= dx / top_k
        improved = False
        for _ in range(40):
            nh = project_feasible(h - step * grad, n)
            nc = compute_corr(nh, n)
            nv = nc.max() * dx
            if nv < cur_c - 1e-14:
                h, cur_corr, cur_c = nh, nc, nv
                step = min(step * 1.08, 0.5)
                improved = True
                no_improve = 0
                break
            step *= 0.6
        if not improved:
            if top_k < 50:
                top_k = min(top_k + 5, 50)
                step = step_init * 0.7
            else:
                break
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        no_improve += 1
        if no_improve > 500:
            break
    return best_h, best_c

def binary_swap_search(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    corr = compute_corr(h, n)
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        moved = False
        for i in list(zeros):
            if time.time() > deadline:
                break
            for j in list(ones):
                d = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                d[msk] += 1.0 - h[im[msk]]
                jm = j + m_range
                msk2 = (jm >= 0) & (jm < n)
                d[msk2] -= 1.0 - h[jm[msk2]]
                imn = i - m_range
                msk3 = (imn >= 0) & (imn < n)
                d[msk3] -= h[imn[msk3]]
                jmn = j - m_range
                msk4 = (jmn >= 0) & (jmn < n)
                d[msk4] += h[jmn[msk4]]
                d[n - 1] -= 2.0
                d[n - 1 + j - i] += 1.0
                d[n - 1 + i - j] += 1.0
                nc = (corr + d).max() * dx
                if nc < cur_c - 1e-14:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur_c = nc
                    ones.discard(j); zeros.discard(i)
                    ones.add(i); zeros.add(j)
                    moved = True
                    break
            if moved:
                break
        if not moved:
            break
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
    return best_h, best_c

def softmax_subgrad(h, n, deadline, temp_init=0.002, temp_min=0.0001):
    dx = 2.0 / n
    cur_corr = compute_corr(h, n)
    cur_c = cur_corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    temp = temp_init
    step = 0.01
    for _ in range(20000):
        if time.time() > deadline:
            break
        max_c = cur_corr.max()
        w = np.exp((cur_corr - max_c) / temp)
        w /= w.sum()
        active_idx = np.where(w > w.max() * 0.01)[0]
        grad = np.zeros(n)
        for li in active_idx:
            grad += w[li] * grad_at_lag(h, n, int(li - (n - 1)))
        grad *= dx
        improved = False
        for _ in range(30):
            nh = project_feasible(h - step * grad, n)
            nc = compute_corr(nh, n)
            nv = nc.max() * dx
            if nv < cur_c - 1e-15:
                h, cur_corr, cur_c = nh, nc, nv
                step = min(step * 1.05, 0.3)
                improved = True
                break
            step *= 0.6
        if not improved:
            temp = max(temp * 0.8, temp_min)
            step = 0.01
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        temp = max(temp * 0.999, temp_min)
    return best_h, best_c

def slsqp_polish(h, n, alpha, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + np.log(np.exp(shifted - sm).sum()) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
