            x_curr = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_curr, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(60, (deadline - time.time() - 30) * 0.4)
            if alloc < 25:
                break
            hd = do_slsqp(h_interp, n_try, 20000, maxiter=1000)
            cd = compute_c5(hd, n_try)
            if time.time() < deadline - 15:
                if n_try <= 300:
                    hd, cd = pair_search(hd, n_try, min(deadline - 5, time.time() + 40),
                                         [0.01, 0.005, 0.002, 0.001, 0.0005])
                else:
                    hd, cd = near_pair_search(hd, n_try, min(deadline - 5, time.time() + 40),
                                              [0.01, 0.005, 0.002, 0.001], radius=50)
            if cd < best_c:
                best_c = cd
                best_h = hd

    # Final: one more SLSQP + pair on absolute best
    remaining = deadline - time.time()
    if remaining > 15:
        nn = len(best_h)
        hd = do_slsqp(best_h, nn, 1000000, maxiter=2000)
        cd = compute_c5(hd, nn)
        if cd < best_c:
            best_c = cd
            best_h = hd
        if remaining > 8 and nn <= 300:
            best_h, best_c = pair_search(best_h, nn, deadline - 1,
                                         [0.0005, 0.0002, 0.0001])

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
