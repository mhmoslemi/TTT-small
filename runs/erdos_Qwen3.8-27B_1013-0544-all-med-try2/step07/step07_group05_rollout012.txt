        return h
    lo = float(h.min()) - 1.0

    m=float(c.max())

       w=np.full(c.size,1.0/c.size)
       f=float(c.mean())

def slsqp_step(h,n,a,iters):

    b=[(0.0,1.0)]*n

    except Exception:
       x=h
    return project(x,n)

import numpy as np
import time
from scipy.optimize import minimize

def c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / float(n)))

def project(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    t = 0.5 * float(n)
    if abs(float(h.sum()) - t) < 1e-13:
        return h
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(45):
        m = 0.5 * (lo + hi)
        s = float(np.clip(h - m, 0.0, 1.0).sum())
        if s > t:
            lo = m
        else:
            hi = m
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    d = t - float(h.sum())
    if abs(d) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[int(np.argmax(mask))] += d
        else:
            h += d / float(n)
    return np.clip(h, 0.0, 1.0)

def soft_fg(h, n, a):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(c.max())
    if a <= 1e-8:
        w = np.full(c.size, 1.0 / float(c.size))
        f = float(c.mean())
    else:
        z = np.clip(a * (c - m), -745.0, 700.0)
        e = np.exp(z)
        ss = float(e.sum())
        if ss <= 0.0:
            e = np.zeros(c.size)
            e[int(np.argmax(c))] = 1.0
            ss = 1.0
        w = e / ss
        f = float(m + np.log(ss) / a)
    g = np.zeros(n)
    w0 = w[n - 1] * dx
    if w0 > 1e-15:
        g += w0 * (1.0 - 2.0 * h)
    for s in range(1, n):
        ln = n - s
        wpd = w[n - 1 + s] * dx
        wnd = w[n - 1 - s] * dx
        if wpd > 1e-15 or wnd > 1e-15:
            left = h[:ln]
            right = h[s:]
            g[:ln] += wnd - (wpd + wnd) * right
            g[s:] += wpd - (wpd + wnd) * left
    return f, g

def pen_f(x, n, a, p):
    f, g = soft_fg(x, n, a)
    s = float(x.sum() - 0.5 * n)
    return f + p * s * s

def pen_g(x, n, a, p):
    f, g = soft_fg(x, n, a)
    s = float(x.sum() - 0.5 * n)
    if s != 0.0:
        g = g + 2.0 * p * s
    return g

def lbfgs_step(h, n, a, p, iters):

    b = [(0.0, 1.0)] * n
    try:
        res = minimize(pen_f, h, args=(n, a, p), jac=pen_g, method='L-BFGS-B',
                       bounds=b, options={'maxiter': int(iters), 'ftol': 1e-13})
        x = res.x if hasattr(res, 'x') else h
    except Exception:
        x = h
    return project(x, n)

def drive(h, n, a, p, deadline):

    besth = h.copy()

        bh, bc = polish(st, nb, te, 1)
        if bc < bestc:
            besth, bestc = bh, bc

    p5 = min(deadline - 80.0, time.time() + max(15.0, 0.08 * B))
    if time.time() + 10.0 < p5:
        bh, bc = polish(besth, nb, p5, 2)
        if bc < bestc:
            besth, bestc = bh, bc

    if time.time() + 40.0 < deadline:
        maxev = 200000 if nb <= 80 else (90000 if nb <= 150 else 25000)
        te = min(deadline - 60.0, time.time() + max(10.0, 0.05 * B))
        ph, pc = pair_search(besth, nb, te, rad=0 if nb <= 80 else 35, maxev=maxev)
        if pc < bestc:
            besth, bestc = ph, pc

    if time.time() + 55.0 < deadline:
        ht = slsqp_step(besth, nb, 800.0, 60)
        cc = c5(ht)
        if cc < bestc:
            besth, bestc = ht, cc
        if time.time() + 35.0 < deadline:
            ht = slsqp_step(besth, nb, 3000.0, 40)
            cc = c5(ht)
            if cc < bestc:
                besth, bestc = ht, cc

    nb = len(besth)
    besth = project(besth, nb)
    bestc = c5(besth)
    return besth, float(bestc), int(nb)
