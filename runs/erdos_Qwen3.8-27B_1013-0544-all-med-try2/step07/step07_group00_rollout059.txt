```python
    best_c = compute_c5(h, n)
    dx = 2.0/n
    for r in range(max_rounds):
        if deadline and time.time() > deadline:
            break
        improved = False
        corr = np.correlate(h, 1-h, mode='full') * dx
        for i in range(n):
            if deadline and time.time() > deadline:
                break
            # Compute slopes: how corr[m] changes when h[i] increases by 1
            # h[i] appears in corr as:
            # - As first factor h[i] in pairs where i is the "h" index
            # - As second factor h[i] in pairs where i is the "1-h" index
            slopes = np.zeros(2*n-1)
            # h[i] as first factor: corr[m] += 1*(1-h[j]) where i and j=i-k are paired
            # In np.correlate(h, 1-h, 'full'):
            # result[m] = sum over n of h[n] * (1-h)[m-n]
            # h[i] contributes to result[m] when n=i: result[m] += h[i]*(1-h[m-i])
            # valid when 0 <= m-i < n, i.e., i <= m < i+n

            # h[i] as second factor: (1-h)[i] contributes
            # result[m] += h[n]*(1-h[i]) when m-n=i, i.e., n=m-i
            # valid when 0 <= m-i < n and 0 <= i < n (always true)

            # change in (1-h[i]) when h[i] changes by delta: -delta

            for m in range(i, i+n):
                slopes[m] += -h[m-i] * dx

            # Find minimum using ternary search or by finding where subgradient = 0

            lo = -h[i]  # h[i] can decrease by at most h[i]
            hi = 1.0 - h[i]  # h[i] can increase by at most 1-h[i]
```

```python
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return float(val)

def _smooth_max_grad(x, n, alpha):
    return grad_smooth_max(x, n, alpha)

def _eq_con(x, n):
    return float(x.sum() - n / 2.0)

def _eq_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, max_iters=500):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        tr = deadline - time.time() - 5
        if tr < 15:
            break
        mu = alpha * 0.1
        try:
            def obj(x):
                return _smooth_max_obj(x, n, alpha) + mu * (x.sum() - n/2.0)**2
            def grad(x):
                return _smooth_max_grad(x, n, alpha) + 2.0*mu*(x.sum()-n/2.0)*np.ones(n)
            res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18, 'gtol': 1e-16,
                                   'maxfun': max_iters*50, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def fista(h, n, deadline, lr=0.01, beta=0.9, max_iters=20000):
    # FISTA with smooth max objective and projection
    best_c = compute_c5(h, n)
    best_h = h.copy()
    x = h.copy()
    y = h.copy()
    t = 1.0
    alpha = 10000  # smooth max parameter
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 1000 == 0:
            # Adapt alpha and lr
            c_now = compute_c5(y, n)
            if c_now < best_c:
                best_c = c_now
                best_h = y.copy()
            # Increase alpha occasionally
            alpha = min(alpha * 1.5, 1000000)
        
        c = compute_c5_vec(y, n)
        c_shift = alpha * (c - c.max())
        mx = c_shift.max()
        w = np.exp(c_shift - mx)
        w /= w.sum()
        
        # Compute gradient of smooth max
        dx = 2.0 / n
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            if w[m] < 1e-14:
                continue
            k = m - (n - 1)
            wm = w[m] * dx
            if k >= 0:
                nv = n - k
                grad[:nv] += wm * (1.0 - y[k:])
                grad[k:] += wm * (-y[:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += wm * (1.0 - y[:nv])
                grad[:nv] += wm * (-y[kk:])
        
        # Also add gradient for sum constraint (penalty)
        s = y.sum() - n / 2.0
        grad += 2.0 * 10.0 * s * np.ones(n)
        
        # FISTA step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - y)
        y = np.clip(y, 0.0, 1.0)
        
        # Gradient step
        x = y - lr * grad
        x = np.clip(x, 0.0, 1.0)
        
        # Project to satisfy sum constraint
        x = project_feasible(x, n)
        t = t_new
        
        if it % 500 == 0:
            c_now = compute_c5(x, n)
            if c_now < best_c:
                best_c = c_now
                best_h = x.copy()
    
    c_final = compute_c5(best_h, n)
    return best_h, c_final

def block_move_search(h, n, deadline, block_size=5, n_trials=10000):
    # Try random block moves
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(123)
    dx = 2.0 / n
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        # Pick a random block
        i = rng.randint(0, n - block_size)
        # Pick a random direction in the block (sum-preserving)
        d = rng.randn(block_size)
        d -= d.mean()  # Make it sum to 0
        d = d / (np.max(np.abs(d)) + 1e-15)
        
        # Try both directions and a few step sizes
        for step in [0.01, 0.005, 0.002]:
            for sign in [1, -1]:
                delta = sign * step * d
                # Check bounds
                new_block = best_h[i:i+block_size] + delta
                if np.all(new_block >= 0) and np.all(new_block <= 1):
                    best_h[i:i+block_size] = new_block
                    # Renormalize
                    best_h = project_feasible(best_h, n)
                    c = compute_c5(best_h, n)
                    if c < best_c:
                        best_c = c
                    else:
                        best_h[i:i+block_size] = best_h[i:i+block_size] - delta
                        best_h = project_feasible(best_h, n)
    return best_h, best_c

def slsqp_polish(h, n, deadline, alpha=1000000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_max_obj, h, jac=_smooth_max_grad,
                      args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-17})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def simulated_annealing(h, n, deadline, rng, T_init=0.01, T_min=0.001, cool=0.999):
```

**Fast coordinate descent**: For each coordinate i, compute the optimal value by evaluating C₅ at a few candidate values (current, current±step, and a few more). This is O(n) per evaluation (just update the affected correlation entries).

Actually, let me think about this more carefully. When h[i] changes by δ:
- corr[m] changes by δ * slope_i[m] for m in [i, i+n-1] (the n lags that involve coordinate i)
- All other corr[m] are unchanged

So to compute the new C₅:
1. Keep track of the current corr vector
2. Update corr[i:i+n] += δ * slope_i
3. Find max(corr)

This is O(n) per step! Much faster than O(n²).

Let me implement this:

```python
def fast_coord_descend(h, n, deadline, steps=100000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    corr = compute_c5_vec(h, n)
    dx = 2.0/n
    rng = np.random.RandomState(42)
    
    for it in range(steps):
        if time.time() > deadline:
            break
        # Pick random coordinate (or sweep)
        i = it % n  # sweep through coordinates
        # Compute slope for coordinate i
        slopes = np.zeros(2*n-1)
        for m in range(i, min(i+n, 2*n-1)):
            j1 = m - i  # index in (1-h) for first factor
            j2 = m - i  # index in h for second factor
            # h[i] as first factor: contributes (1-h[j1]) to corr[m]
            if 0 <= j1 < n:
                slopes[m] += (1.0 - h[j1]) * dx
            # -h[i] as second factor (in 1-h[i]): contributes -h[j2] to corr[m]
            if 0 <= j2 < n:
                slopes[m] += -h[j2] * dx
        
        # Try a few step sizes
        best_delta = 0
        best_local_c = corr.max()
        for ds in [0.01, 0.005, -0.01, -0.005, 0.002, -0.002, 0.02, -0.02]:
            new_val = h[i] + ds
            if new_val < 0 or new_val > 1:
                continue
            new_c = np.max(corr + ds * slopes)
            if new_c < best_local_c:
                best_local_c = new_c
                best_delta = ds
        
        if best_delta != 0:
            h[i] += best_delta
            corr[i:min(i+n, 2*n-1)] += best_delta * slopes[i:min(i+n, 2*n-1)]
            # Renormalize (adjust sum)
            diff = n/2.0 - h.sum()
            if abs(diff) > 1e-10:
                # Simple: adjust the most "free" coordinate
                h = project_feasible(h, n)
                corr = compute_c5_vec(h, n)
            
            c = corr.max()
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    return best_h, best_c
```

Wait, there's an issue. When I do project_feasible, it changes multiple coordinates, so I need to recompute the entire corr vector. This breaks the incremental update. Let me handle this differently.

Actually, the sum constraint is the issue. If I change h[i] by δ, the sum changes by δ. To maintain the sum, I need to change another coordinate j by -δ. So really, it's a 2-coordinate move.

Let me handle it as: change h[i] by δ and h[j] by -δ (for some j). Then:
- corr changes by δ * slope_i - δ * slope_j
- The new corr = old_corr + δ * (slope_i - slope_j)
- New C₅ = max(old_corr + δ * (slope_i - slope_j))

This is still O(n) per evaluation!

```python
def fast_pair_descend(h, n, deadline, max_sweeps=50):

    best_h = h.copy()

                # 2. As -h[i] in (1-h[m-i]) when m-i = i, i.e., m = 2i: 
                #    Actually no, h[i] appears in (1-h[k]) when k=i, which affects corr[m] for m where m-n = i (i.e., m = n+i) 
                #    Wait, I'm confusing myself.

        # corr[m] = sum over valid (a,b) where a-b = m-(n-1) of h[a] * (1-h[b])
        # If a = i: corr[m] includes h[i]*(1-h[i-(m-(n-1))]) = h[i]*(1-h[i-m+n-1])
        #   valid when 0 <= i-m+n-1 < n, i.e., n-1-i <= m <= n-1+i
        #   slope_i[m] += (1-h[i-m+n-1]) for m in [n-1-i, n-1+i]

        #   valid when 0 <= m-n+1+i < n, i.e., n-1-i <= m <= n-1+i
        #   slope_i[m] += -h[m-n+1+i]

        #   When b=i, a = i + (m-(n-1)) = i + m - n + 1

        for j in range(n):

            lo_j = max(0, n-1-j)

            for m in range(lo_j, hi_j+1):
                a_idx = j + m - n + 1
                b_idx = n - 1 + j - m
                slope_j[m] = (1.0 - h[b_idx]) - h[a_idx]
            slope_j *= dx
            
            # Try moving mass from j to i (delta > 0 means h[i] increases, h[j] decreases)
            # New corr = corr + delta*(slope_i - slope_j)
            # Minimize max(new_corr) over delta in [-(h[i]), min(1-h[i], h[j])]
            # Wait: delta > 0 means h[i] += delta, h[j] -= delta
            # Constraint: 0 <= h[i]+delta <= 1 and 0 <= h[j]-delta <= 1
            # So delta in [-h[i], 1-h[i]] ∩ [-h[j], 1-h[j]] ... 
            # Actually delta can be negative too (h[i] decreases, h[j] increases)
            # delta in [max(-h[i], h[j]-1), min(1-h[i], h[j])]
            
            lo_d = max(-h[i], h[j] - 1.0)
            hi_d = min(1.0 - h[i], h[j])
            if lo_d >= hi_d:
                continue
            
            # Minimize max(corr + delta * (slope_i - slope_j)) over delta in [lo_d, hi_d]
            diff = slope_i - slope_j
            # This is minimizing max_m (corr[m] + delta * diff[m])
            # = minimizing max_m (corr[m] + delta * diff[m])
            # Convex in delta. Use ternary search.
            a, b = lo_d, hi_d
            for _ in range(40):
                m1 = a + (b-a)/3
                m2 = b - (b-a)/3
                f1 = np.max(corr + m1 * diff)
                f2 = np.max(corr + m2 * diff)
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            best_delta = (a+b)/2
            
            if abs(best_delta) > 1e-8:
                new_c = np.max(corr + best_delta * diff)
                if new_c < best_c - 1e-14:
                    h[i] += best_delta
                    h[j] -= best_delta
                    corr += best_delta * diff
                    best_c = new_c
                    improved = True
    
    # Final projection
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    total = 2 * n - 1
    for m in range(total):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += -wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += -wm * h[kk:]
    return grad

def _sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha + mu * (x.sum() - n/2.0)**2)

def _sm_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    s = x.sum() - n/2.0
    g += 2.0 * mu * s
    return g

def lbfgs_opt(h, n, deadline, alphas=None):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    for alpha in alphas:
        if time.time() > deadline - 8:
            break
        tr = deadline - time.time() - 4
        if tr < 12:
            break
        mu = max(alpha * 0.05, 1.0)
        iters = min(400, max(50, int(tr * 8)))
        try:
            res = minimize(_sm_obj, h, jac=_sm_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': iters, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': iters*40, 'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def momentum_opt(h, n, deadline, alpha_init=2000, max_iters=80000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    x = h.copy()
    v = np.ones(n) * 1e-8
    beta = 0.9
    lr = 0.003
    alpha = alpha_init
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 5000 == 0:
            alpha = min(alpha * 2.0, 5000000)
            c_now = compute_c5(x, n)
            if c_now < best_c:
                best_c = c_now
                best_h = x.copy()
            lr = max(lr * 0.95, 0.0005)
        g = grad_smooth_max(x, n, alpha)
        s = x.sum() - n / 2.0
        g += 5.0 * s
        v = beta * v + (1 - beta) * g
        x = x - lr * v
        x = project_feasible(x, n)
    c_f = compute_c5(best_h, n)
    return best_h, c_f

def slope_for_coord(h, n, i):
    total = 2 * n - 1
    dx = 2.0 / n
    sl = np.zeros(total)
    lo = max(0, n - 1 - i)
    hi = min(total - 1, n - 1 + i)
    m_range = np.arange(lo, hi + 1)
    a_idx = np.clip(i + m_range - n + 1, 0, n - 1)
    b_idx = np.clip(n - 1 + i - m_range, 0, n - 1)
    sl[lo:hi + 1] = ((1.0 - h[b_idx]) - h[a_idx]) * dx
    return sl

def fast_pair_cd(h, n, deadline, window=15, max_sweeps=80):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    corr = compute_c5_vec(h, n)
    total = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            slope_i = slope_for_coord(h, n, i)
            j_lo = (i - window) % n
            j_hi = (i + window) % n
            if j_lo < j_hi:
                js = list(range(j_lo, j_hi + 1))
            else:
                js = list(range(j_lo, n)) + list(range(0, j_hi + 1))
            for j in js:
                if i == j:
                    continue
                if time.time() > deadline - 5:
                    break
                slope_j = slope_for_coord(h, n, j)
                diff = slope_i - slope_j
                lo_d = max(-h[i], h[j] - 1.0)
                hi_d = min(1.0 - h[i], h[j])
                if hi_d - lo_d < 1e-10:
                    continue
                a, b = lo_d, hi_d
                for _ in range(35):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    f1 = np.max(corr + m1 * diff)
                    f2 = np.max(corr + m2 * diff)
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                bd = (a + b) / 2.0
                if abs(bd) > 1e-9:
                    nc = np.max(corr + bd * diff)
                    if nc < best_c - 1e-14:
                        h[i] += bd
                        h[j] -= bd
                        corr += bd * diff
                        best_c = nc
                        improved = True
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def slsqp_polish(h, n, deadline, alpha=2000000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x, n=n: x.sum() - n/2.0,
             'jac': lambda x, n=n: np.ones(n)}]
    try:
        res = minimize(lambda x: _sm_obj(x, n, alpha, 0), h,
                      jac=lambda x: _sm_grad(x, n, alpha, 0),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-17})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)
```
