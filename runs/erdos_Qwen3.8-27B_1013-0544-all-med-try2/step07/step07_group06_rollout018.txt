    dx = 2.0 / n
    active = np.abs(corr - corr.max()) < (corr.max() - corr.min()) * 0.1 + 1e-10
    if active.sum() == 0:
        active = corr == corr.max()
    
    g_avg = G[active, :].mean(axis=0)  # average gradient at active shifts

    # We want to maximize this, subject to d ≤ min(1-h[i], h[j])
    # The improvement is approximately g_avg[i]*d - g_avg[j]*d = (g_avg[i]-g_avg[j])*d
    # Max d = min(1-h[i], h[j])
    
    # Score[i,j] = (g_avg[i] - g_avg[j]) * min(1-h[i], h[j])
    # We want the max positive score

    # This is not separable in a simple outer product
    
    # Let's just find the best i (to increase) and best j (to decrease)
    # For increasing i: score = g_avg[i] * (1-h[i])
    # For decreasing j: score = -g_avg[j] * h[j]
    # Total score = g_avg[i]*(1-h[i]) - g_avg[j]*h[j] = score_inc[i] + score_dec[j]
    
    score_inc = g_avg * (1 - h)
    score_dec = -g_avg * h
    
    best_i = score_inc.argmax()
    best_j = score_dec.argmax()
    
    if best_i == best_j:
        # Try second best
        second_i = np.argsort(-score_inc)[1]
        if score_inc[second_i] + score_dec[best_j] > score_inc[best_i] + score_dec[np.argsort(-score_dec)[1]]:
            best_i = second_i
        else:
            best_j = np.argsort(-score_dec)[1]
    
    d_max = min(1 - h[best_i], h[best_j])
    if d_max <= 1e-15:
        return None
    
    # Now refine with exact optimization
    g = G[:, best_i] - G[:, best_j]

    """Project onto {0<=h<=1, sum(h)=n/2} using bisection on lambda"""
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def grad_matrix(h, n):
    """G[m, i] = d(corr[m])/d(h[i])"""
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    idx = np.arange(n)

        if v2.any():
            G[m, v2] -= h[idx[v2] - s]

            break

            break

        for i in interior:
            if time.time() > deadline:
                break
            for target in [0.0, 1.0]:

                if abs(delta) < 1e-10:

def smooth_max_obj(x, n, alpha):
    corr = compute_corr(x, n)
    m = corr.max()
    if m < 1e-15:
        return 0.0
    sh = alpha * (corr - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return val

def smooth_max_grad(x, n, alpha):
    corr = compute_corr(x, n)
    m = corr.max()

    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    G = grad_matrix(x, n)
    return G.T @ w  # shape (n,)

        else:
            d_lo = d_hi = d_mid
            break
    return (d_lo + d_hi) / 2

def find_candidates(corr, G, h, n, K=80):
    mx = corr.max()
    rng = max(mx - corr.min(), 1e-10)
    active = corr >= mx - 0.08 * rng
    if active.sum() == 0:
        active = corr == mx
    g_avg = G[active].mean(axis=0)
    score_inc = g_avg * (1.0 - h)
    score_dec = -g_avg * h
    top_i = np.argsort(-score_inc)[:min(K, n)]
    top_j = np.argsort(-score_dec)[:min(K, n)]
    candidates = []
    for i in top_i:
        for j in top_j:
            if i == j:
                continue
            dm = min(1.0 - h[i], h[j])
            if dm > 1e-12:
                candidates.append((i, j, dm))
            if len(candidates) >= K * 3:
                break
        if len(candidates) >= K * 3:
            break
    return candidates

def optimize_pairs(h, n, deadline, K=80, max_sweeps=500):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    no_improve = 0
    rng = np.random.RandomState(42)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = compute_corr(h, n)
        G = grad_matrix(h, n)
        candidates = find_candidates(corr, G, h, n, K)
        if not candidates:
            break
        improved = False
        tried = set()
        for (i, j, dm) in candidates:
            if time.time() > deadline:
                break
            key = (i, j)
            if key in tried:
                continue
            tried.add(key)
            g_diff = G[:, i] - G[:, j]
            d_opt = exact_pair_d(corr, g_diff, 0.0, dm)
            if abs(d_opt) < 1e-13:
                continue
            new_h = h.copy()
            new_h[i] += d_opt
            new_h[j] -= d_opt
            new_h = project_feasible(new_h, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-15:
                best_c = new_c
                best_h = new_h.copy()
                h = new_h
                corr = compute_corr(h, n)
                G = grad_matrix(h, n)
                improved = True
                break
        if not improved:
            for (i, j, dm) in candidates:
                if time.time() > deadline:
                    break
                dm_rev = min(1.0 - h[j], h[i])
                if dm_rev < 1e-12:
                    continue
                g_diff = G[:, j] - G[:, i]
                d_opt = exact_pair_d(corr, g_diff, 0.0, dm_rev)
                if abs(d_opt) < 1e-13:
                    continue
                new_h = h.copy()
                new_h[j] += d_opt
                new_h[i] -= d_opt
                new_h = project_feasible(new_h, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c - 1e-15:
                    best_c = new_c
                    best_h = new_h.copy()
                    h = new_h
                    improved = True
                    break
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                if rng.random() < 0.5:
                    idx = rng.randint(n)
                    pert = h.copy()
                    pert[idx] = np.clip(pert[idx] + rng.uniform(-0.1, 0.1), 0, 1)
                    pert = project_feasible(pert, n)
                    c_p = compute_c5(pert, n)
                    if c_p < best_c + 0.001:
                        h = pert
                        no_improve = 0
                if no_improve >= 10:
                    break
        else:
            no_improve = 0
    return best_h, best_c

def smooth_max_obj(x, n, alpha):
    corr = compute_corr(x, n)
    m = corr.max()
    sh = alpha * (corr - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(x, n, alpha):
    corr = compute_corr(x, n)
    m = corr.max()
    sh = alpha * (corr - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    G = grad_matrix(x, n)
    return G.T @ w

def lbfgs_optimize(h, n, deadline, alphas=(200, 2000, 20000, 200000)):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    target_sum = n / 2.0
    for alpha in alphas:
        if time.time() > deadline:
            break
        mu = 50.0
        def obj(x):
            return smooth_max_obj(x, n, alpha) + mu * (x.sum() - target_sum) ** 2
        def gr(x):
            return smooth_max_grad(x, n, alpha) + 2.0 * mu * (x.sum() - target_sum)
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-18, 'gtol': 1e-12})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def binary_push(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for iteration in range(150):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 0.02) & (h < 0.98))[0]
        if len(interior) == 0:
            break
        for i in interior:
            if time.time() > deadline:
                break
            for target in [0.0, 1.0]:
                delta = target - h[i]
                if abs(delta) < 1e-10:
                    continue
                if delta > 0:
                    mask = (h > 0) & (np.arange(n) != i)
                else:
                    mask = (h < 1) & (np.arange(n) != i)
                if not mask.any():
                    continue
                new_h = h.copy()
                new_h[i] = target
                new_h[mask] -= delta / mask.sum()
                new_h = project_feasible(new_h, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c - 1e-15:
                    best_c = new_c
                    best_h = new_h.copy()
                    h = new_h
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 300
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    # Phase 1: L-BFGS smooth max
    t1 = min(t0 + 200, deadline - 500)
    h1, c1 = lbfgs_optimize(best_h, n, t1)
    if c1 < best_c:
        best_c, best_h = c1, h1
    # Phase 2: Exact pair optimization (main workhorse)
    t2 = min(time.time() + 350, deadline - 250)
    h2, c2 = optimize_pairs(best_h, n, t2, K=80, max_sweeps=800)
    if c2 < best_c:
        best_c, best_h = c2, h2
    # Phase 3: Binary push
    if time.time() < deadline - 150:
        t3 = min(time.time() + 100, deadline - 150)
        h3, c3 = binary_push(best_h, n, t3)
        if c3 < best_c:
            best_c, best_h = c3, h3
    # Phase 4: Pair optimization after binary push
    if time.time() < deadline - 80:
        t4 = min(time.time() + 80, deadline - 80)
        h4, c4 = optimize_pairs(best_h, n, t4, K=60, max_sweeps=400)
        if c4 < best_c:
            best_c, best_h = c4, h4
    # Phase 5: L-BFGS polish
    if time.time() < deadline - 60:
        h5, c5 = lbfgs_optimize(best_h, len(best_h), min(time.time() + 40, deadline - 30), alphas=(10000, 100000))
        if c5 < best_c:
            best_c, best_h = c5, h5
    # Phase 6: Try other n values
    if time.time() < deadline - 40:
        for n_try in [200, 250, 400, 500]:
            if time.time() > deadline - 20:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            alloc = min(25, max(10, deadline - time.time() - 15))
            h_t, c_t = optimize_pairs(h_interp, n_try, time.time() + alloc, K=50, max_sweeps=200)
            if c_t < best_c:
                best_c, best_h = c_t, h_t
    # Phase 7: Final pair polish
    if time.time() < deadline - 5:
        nf = len(best_h)
        h7, c7 = optimize_pairs(best_h, nf, deadline - 3, K=50, max_sweeps=300)
        if c7 < best_c:
            best_c, best_h = c7, h7
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
