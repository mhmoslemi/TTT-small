```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
```

```python
                    a2 = rng.choice(np.delete(ones, np.where(ones == a1)[0]))
                    b1 = rng.choice(zeros)

                    c_new = compute_c5(h, n)

                        best_c = c_new
                        found_improvement = True

                        zeros = np.where(h == 0)[0]
                        patience = 0
                    else:
                        h[a1] = 1; h[a2] = 1; h[b1] = 0; h[b2] = 0
    
    return h, best_c
```

```python
    h = (h_init > 0.5).astype(float)
    num_ones = int(h.sum())
    target = n // 2
    
    if num_ones > target:
        ones_idx = np.where(h == 1)[0]
        to_remove = rng.choice(ones_idx, size=num_ones - target, replace=False)
        h[to_remove] = 0
    elif num_ones < target:
        zeros_idx = np.where(h == 0)[0]
        to_add = rng.choice(zeros_idx, size=target - num_ones, replace=False)
        h[to_add] = 1
    
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        
        # Systematic sweep: try all pairs but in random order
        pairs = []
        for a in ones:
            for b in zeros:
                pairs.append((a, b))
        rng.shuffle(pairs)
        
        for a, b in pairs:
            if time.time() > deadline:
                return h, best_c
            
            # Check if a and b are still in the right state
            if h[a] != 1 or h[b] != 0:
                continue
            
            h[a] = 0
            h[b] = 1
            c = compute_c5(h, n)

                improved = True
                # Update lists
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                break  # Restart sweep with new state
            else:
                h[a] = 1
                h[b] = 0
        
        if not improved:
            break
```

```python
def binary_local_search_guided(h_init, n, rng, deadline):
    h = (h_init > 0.5).astype(float)
    # ... adjust to target ...

    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        
        # Find active shifts
        tol = 0.005
        active = np.where(c_vec >= c_max - tol)[0]
        
        # Compute gradient at active shifts
        g = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            g += grad_of_shift(h, n, k)
        g /= len(active)
        
        # Positions to decrease: h=1 and g > 0
        dec_candidates = np.where((h == 1) & (g > 0))[0]
        # Positions to increase: h=0 and g < 0
        inc_candidates = np.where((h == 0) & (g < 0))[0]
        
        if len(dec_candidates) == 0 or len(inc_candidates) == 0:
            break
        
        # Try the most promising pairs first
        # Sort by gradient magnitude
        dec_candidates = dec_candidates[np.argsort(-g[dec_candidates])]
        inc_candidates = inc_candidates[np.argsort(g[inc_candidates])]
        
        improved = False
        # Try top candidates
        for a in dec_candidates[:50]:
            if time.time() > deadline:
                return h, best_c
            for b in inc_candidates[:50]:
                h[a] = 0
                h[b] = 1
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                else:
                    h[a] = 1
                    h[b] = 0
            if improved:
                break

            # Try random pairs as fallback
            for _ in range(1000):

                a = rng.choice(np.where(h == 1)[0])
                b = rng.choice(np.where(h == 0)[0])
```
