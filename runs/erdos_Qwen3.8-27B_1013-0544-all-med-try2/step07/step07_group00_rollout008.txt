            if tot < 1e-14:

    """Fit a Chebyshev polynomial of degree K to h, return smoothed version."""
    x = np.linspace(-1, 1, n)
    # Chebyshev nodes for least squares
    c = np.zeros(K + 1)

        if k > 0:

        h_smooth += coeff * np.cos(k * np.arccos(x))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def lbfgs_optimize(h, n, deadline, alpha_start=100, alpha_max=5000000, n_alphas=7):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = np.geomspace(alpha_start, alpha_max, n_alphas)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        remaining = deadline - time.time() - 5
        if remaining < 8:
            break
        max_iters = min(int(remaining * 12), 400)
        if max_iters < 10:
            break
        obj = lambda x: smooth_max_obj(x, n, alpha)
        jac = lambda x: smooth_max_grad(x, n, alpha)
        try:
            res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-17,
                                   'gtol': 1e-15, 'maxfun': max_iters * 25,
                                   'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def chebyshev_smooth(h, n, K):
    x = np.linspace(-1, 1, n)
    theta = np.arccos(x)
    h_smooth = np.zeros(n)
    for k in range(K + 1):
        coeff = np.mean(h * np.cos(k * theta))
        if k > 0:
            coeff *= 2.0
        else:
            coeff *= 0.5
        h_smooth += coeff * np.cos(k * theta)
    return project_feasible(h_smooth, n)
