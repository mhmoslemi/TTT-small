import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    g = corr * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    e = np.exp(shifted - sm)
    e_sum = e.sum()
    w = e / e_sum
    val = m + (sm + np.log(e_sum)) / alpha
    
    shifts = np.arange(-(n-1), n)
    ii = np.arange(n)
    idx_pos = ii[:, None] + shifts[None, :]
    idx_neg = ii[:, None] - shifts[None, :]
    mask_pos = (idx_pos >= 0) & (idx_pos < n)
    mask_neg = (idx_neg >= 0) & (idx_neg < n)
    x_clipped_pos = np.clip(idx_pos, 0, n-1)
    x_clipped_neg = np.clip(idx_neg, 0, n-1)
    x_pos = np.where(mask_pos, x[x_clipped_pos], 0.0)
    x_neg = np.where(mask_neg, x[x_clipped_neg], 0.0)
    grad = 1.0 - x_pos @ w - x_neg @ w
    grad *= dx
    return val, grad

def slsqp_opt(h, n, alpha, maxiter, ftol, deadline=None):
    def obj(x):
        return smooth_obj_and_grad(x, n, alpha)[0]
    def grad(x):
        return smooth_obj_and_grad(x, n, alpha)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        result = project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        result = project_feasible(h, n)
    return result
