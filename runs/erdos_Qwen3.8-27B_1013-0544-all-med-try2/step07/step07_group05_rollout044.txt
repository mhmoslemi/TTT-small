def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: SA exploration on initial grid
    if time.time() < deadline - 80:
        sa_end = time.time() + min(250, (deadline - time.time()) * 0.30)
        h_sa, c_sa = sa_optimize(best_h, n_init, sa_end, rng,
                                 T0=5e-4, cool_rate=0.99997)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa

    # Phase 2: Refine (SLSQP + pair search)
    if time.time() < deadline - 40:
        best_h, best_c = refine(best_h, n_init, time.time() + min(120, (deadline - time.time()) * 0.25), rng)

    # Phase 3: ILS - iterated local search with random restarts
    if time.time() < deadline - 50:
        remaining = deadline - time.time() - 40
        n_trials = max(2, int(remaining / 60))
        for k in range(n_trials):
            if time.time() > deadline - 40:
                break
            scale = 0.02 + 0.06 * (k / max(1, n_trials - 1))
            hp = best_h + rng.normal(0, scale, n_init)
            hp = project_feasible(hp, n_init)
            alloc = min(55, (deadline - time.time() - 30) * 0.4)
            if alloc < 15:
                break
            h_c, c_c = refine(hp, n_init, time.time() + alloc, rng)
            if c_c < best_c:
                best_c = c_c
                best_h = h_c

    # Phase 4: Try other grid sizes
    x_grid = np.linspace(0, 2, n_init)
    for n_try in [300, 200, 150]:
        if time.time() > deadline - 70:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_grid, best_h)
        h_interp = project_feasible(h_interp, n_try)
        c_interp = compute_c5(h_interp, n_try)
        if c_interp >= best_c and c_interp > best_c + 0.001:
            continue
        alloc = min(80, (deadline - time.time()) * 0.25)
        if alloc < 30:
            break
        # SA on new grid
        h_new, c_new = sa_optimize(h_interp, n_try, time.time() + alloc * 0.4,
                                   rng, T0=3e-4, cool_rate=0.99998)
        if c_new > c_interp:
            h_new, c_new = h_interp, c_interp
        # Refine
        h_new, c_new = refine(h_new, n_try, time.time() + alloc * 0.5, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 5: Final intensive refinement
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        # Multiple SLSQP passes
        for alpha in [50000, 200000, 500000]:
            if time.time() > deadline - 5:
                break
            ht = slsqp(best_h, n, alpha, maxiter=3000)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht
        # Final pair search with fine steps
        if time.time() < deadline - 2:
            if n <= 300:
                best_h, best_c = pair_search(best_h, n, deadline - 1,
                                             [0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
            else:
                best_h, best_c = pair_search(best_h, n, deadline - 1,
                                             [0.0005, 0.0002, 0.0001, 0.00005], radius=50)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
