        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        w = np.exp(sh - mx)
    else:
        w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-15:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] -= wm * h[0:nv]
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] -= wm * h[kk:n]
    return grad

def pen_obj(x, n, alpha, mu):
    val = smooth_max_obj(x, n, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def pen_grad(x, n, alpha, mu):

    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def gradient_pair(h, n, deadline, step=0.003):
    best_c = compute_c5(h, n)
    for alpha in [200, 2000, 20000, 200000]:
        if time.time() >= deadline:
            break
        for _ in range(200):
            if time.time() >= deadline:
                return h, best_c
            improved = False
            g = smooth_max_grad(h, n, alpha)
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, min(n, i + 80)):
                    move = -step * (g[i] - g[j])
                    if abs(move) < 1e-12:
                        continue
                    if 0 <= h[i] + move <= 1 and 0 <= h[j] - move <= 1:
                        h[i] += move
                        h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS on original grid (short)
    t1 = min(t0 + 80, deadline - 600)
    if t1 > t0 + 20:
        best_h, best_c = lbfgs_optimize(best_h, n, t1, max_iters=200)

    # Phase 2: Upscale to 2n
    n2 = 2 * n
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n2)
    h2 = np.interp(x_new, x_old, best_h)
    h2 = project_feasible(h2, n2)
    c2 = compute_c5(h2, n2)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: L-BFGS on n2
    t3 = min(time.time() + 120, deadline - 450)
    if t3 > time.time() + 20:
        best_h, best_c = lbfgs_optimize(best_h, n2, t3, max_iters=200)

    # Phase 4: Momentum gradient descent on n2 (main workhorse)
    t4 = min(time.time() + 350, deadline - 250)
    if t4 > time.time() + 30:
        h4, c4 = momentum_gd(best_h, n2, t4)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Another L-BFGS pass on n2
    t5 = min(time.time() + 80, deadline - 150)
    if t5 > time.time() + 15:
        best_h, best_c = lbfgs_optimize(best_h, n2, t5, max_iters=150)

    # Phase 6: Gradient-guided pair search
    t6 = min(time.time() + 100, deadline - 60)
    if t6 > time.time() + 15:
        best_h, best_c = gradient_pair(best_h, n2, t6, step=0.002)

    # Phase 7: Standard pair search (fine deltas)
    if time.time() < deadline - 10:
        best_h, best_c = pair_search(best_h, n2, deadline,
                                     [0.005, 0.002, 0.001, 0.0005, 0.0002],
                                     radius=30)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
