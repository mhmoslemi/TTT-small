def project_feasible(h, n):

            if s < 1e-15:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    exp_sh = np.exp(sh - sm)
    Z = exp_sh.sum()
    w = exp_sh / Z
    obj = m + (sm + np.log(Z)) / alpha + mu * (h.sum() - n / 2.0) ** 2
    w_cum = np.concatenate([[0.0], np.cumsum(w)])
    S = w_cum[n:2 * n] - w_cum[0:n]
    conv1 = np.convolve(w, h, mode='full')[n - 1:2 * n - 1]
    conv2 = np.convolve(w[::-1], h, mode='full')[n - 1:2 * n - 1]
    grad = dx * (S - conv1 - conv2)
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return obj, grad

                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-14:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, cur_c

def multi_resolution_optimize(h_best, n_best, c_best, deadline, rng):
    for nh in [200, 300, 350, 400]:
        if time.time() > deadline - 60:
            break
        if nh == n_best:
            continue
        if nh % 2 != 0:
            continue
        xo = np.linspace(0, 2, n_best, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = np.interp(xn, xo, (h_best + 1.0 / n_best), period=2) 
        hh = np.clip(hh, 0, 1)
        hh = project_feasible(hh, nh)
        for alpha in [5000, 50000]:
            if time.time() > deadline - 30:
                break
            hh = lbfgs_optimize(hh, alpha, 100.0, 500, deadline - 30)
            hh = project_feasible(hh, nh)
            ch = compute_c5(hh, nh)
            if ch < c_best:
                c_best = ch
                h_best = hh.copy()
                n_best = nh
        hb, cb = binary_refine(hh, nh, deadline - 20, max_sweeps=1000, seed=99)
        if cb < c_best:
            c_best = cb
            h_best = hb.copy()
            n_best = nh
    return h_best, c_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        n = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n = 250
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_best = n

    p1 = t0 + 0.12 * (budget_s - 3)
    for alpha in [100, 500, 2000, 10000, 50000]:
        if time.time() > p1:
            break
        for restart in range(2):
            if time.time() > p1:
                break
            perturb = 0.03 * rng.randn(n) if restart > 0 else 0
            h_init = project_feasible(best_h + perturb, n)
            for mu in [50.0, 500.0]:
                if time.time() > p1:
                    break
                h_new = lbfgs_optimize(h_init, alpha, mu, 800, p1)
                h_new = project_feasible(h_new, n)
                c = compute_c5(h_new, n)
                if c < best_c:
                    best_c = c
                    best_h = h_new.copy()
                h_init = h_new.copy()

    p2 = t0 + 0.30 * (budget_s - 3)
    hb, cb = binary_refine(best_h, n_best, p2, max_sweeps=3000, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    p3 = t0 + 0.50 * (budget_s - 3)
    for alpha in [20000, 100000, 500000, 2000000]:
        if time.time() > p3:
            break
        h_pert = project_feasible(best_h + 0.04 * rng.randn(n_best), n_best)
        for mu in [100.0, 1000.0]:
            if time.time() > p3:
                break
            h_new = lbfgs_optimize(h_pert, alpha, mu, 1000, p3)
            h_new = project_feasible(h_new, n_best)
            c = compute_c5(h_new, n_best)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
                h_pert = h_new.copy()

    p4 = t0 + 0.70 * (budget_s - 3)
    hb, cb = binary_refine(best_h, n_best, p4, max_sweeps=5000, seed=seed + 1)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    p5 = t0 + 0.85 * (budget_s - 3)
    best_h, best_c, n_best = multi_resolution_optimize(best_h, n_best, best_c, p5, rng)

    if time.time() < dl - 10:
        h_pert = project_feasible(best_h + 0.02 * rng.randn(n_best), n_best)
        h_new = lbfgs_optimize(h_pert, 5000000, 5000.0, 2000, dl - 8)
        h_new = project_feasible(h_new, n_best)
        c = compute_c5(h_new, n_best)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()

    hb, cb = binary_refine(best_h, n_best, dl, max_sweeps=3000, seed=seed + 2)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    return best_h, best_c, n_best
