def sa_optimize(h, n, deadline, rng, t0=None):
    """Simulated annealing with structured moves."""
    if t0 is None:
        t0 = 1e-4
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    cur_c = best_c
    T = t0
    cooling = 0.995
    x = np.linspace(0, 2, n)
    
    while time.time() < deadline:
        # Choose move type
        mt = rng.random()
        h_new = cur_h.copy()
        
        if mt < 0.4:
            # Pair transfer
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.uniform(0.005, 0.1)
            sign = 1 if rng.random() < 0.5 else -1
            h_new[i] += sign * delta
            h_new[j] -= sign * delta
        elif mt < 0.7:
            # Block transfer
            w = rng.randint(2, max(3, n // 10))
            i = rng.randint(0, max(1, n - w))
            j = rng.randint(0, max(1, n - w))
            if abs(i - j) < w:
                continue
            delta = rng.uniform(0.001, 0.05)
            h_new[i:i+w] += delta
            h_new[j:j+w] -= delta
        else:
            # Smooth perturbation (sine wave)
            freq = rng.randint(1, max(2, n // 15))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.01, 0.15)
            h_new += amp * np.sin(freq * np.pi * x + phase)
        
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        delta_c = c_new - cur_c
        if delta_c < 0 or rng.random() < np.exp(-delta_c / T):
            cur_h = h_new
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        
        T *= cooling
        if T < 1e-8:
            T = 1e-8
    
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    """Local pair search with optional radius limit."""
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    j_range = range(i + 1, n)

                    hi = min(n, i + radius + 1)
                    j_range = range(lo, hi)
                for j in j_range:

                    # Try h[i]+=d, h[j]-=d

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jacobian(x, n):
    return np.ones(n)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 40:
        active = sh > (mx - 40)
        return float(m + (mx + np.log(max(float(active.sum()), 1.0))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jacobian, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def sa_core(h_init, n, deadline, rng):
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    cur_h = h_init.copy()
    cur_c = best_c
    T = 2e-4
    cooling = 0.9997
    x = np.linspace(0, 2, n)
    step_scale = 0.08
    min_temp = 5e-8
    while time.time() < deadline and T > min_temp:
        mt = rng.random()
        h_new = cur_h.copy()
        if mt < 0.35:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                T *= cooling
                continue
            d = rng.uniform(0.002, step_scale)
            sgn = 1.0 if rng.random() < 0.5 else -1.0
            h_new[i] += sgn * d
            h_new[j] -= sgn * d
        elif mt < 0.6:
            w = rng.randint(2, max(3, n // 8))
            i = rng.randint(0, max(1, n - w))
            j = rng.randint(0, max(1, n - w))
            if abs(i - j) < w + 1:
                T *= cooling
                continue
            d = rng.uniform(0.0005, step_scale / w)
            h_new[i:i+w] += d
            h_new[j:j+w] -= d
        else:
            freq = rng.randint(1, max(2, n // 12))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.005, 0.08)
            h_new += amp * np.sin(freq * np.pi * x + phase)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        dc = c_new - cur_c
        if dc < 0 or rng.random() < np.exp(-dc / T):
            cur_h = h_new
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        T *= cooling
        step_scale *= 0.99999
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    j_lo, j_hi = i + 1, n
                else:
                    j_lo = max(0, i - radius)
                    j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def try_grid(h_best, n_best, n_try, deadline, rng):

    x_old = np.linspace(0, 2, n_best)
