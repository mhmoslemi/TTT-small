        best_c, best_h = cb, hb.copy()
    h = best_h.copy()
    for alpha in [1e5, 1e6, 1e7, 1e8, 1e9]:
        if time.time() > p3 - 2:
            break
        h = slsqp_opt(h, n, alpha, 2000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: High-alpha SLSQP for sharp max
    p4 = t0 + 0.60 * total
    h = best_h.copy()
    for alpha in [1e8, 5e8, 2e9, 1e10, 5e10, 2e11, 1e12]:
        if time.time() > p4 - 3:
            break
        h = slsqp_opt(h, n, alpha, 5000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: Coordinate descent again
    p5 = t0 + 0.72 * total
    hc, cc = coord_descent(best_h, n, p5, max_sweeps=200)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 6: Multi-resolution
    p6 = t0 + 0.85 * total
    for nh in [300, 500, 600, 800]:
        if time.time() > p6 - 8:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        for alpha in [1e6, 1e8, 1e10]:
            if time.time() > p6 - 4:
                break
            hh = slsqp_opt(hh, nh, alpha, 1500, 1e-16)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh

    # Phase 7: Noise restarts
    p7 = t0 + 0.95 * total
    for trial in range(12):
        if time.time() > p7 - 1:
            break
        ns = 0.002 + 0.001 * trial
        h = best_h + ns * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [1e8, 1e10, 1e12]:
            if time.time() > p7 - 0.5:
                break
            h = slsqp_opt(h, n, alpha, 3000, 1e-18)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Final coordinate descent polish
    if time.time() < dl - 2:
        hc, cc = coord_descent(best_h, n, dl - 1, max_sweeps=100)
        hc = project_feasible(hc, n)
        cc = compute_c5(hc, n)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
