import numpy as np
from scipy.optimize import minimize

def project_box_sum(y,target):
    y=np.asarray(y,dtype=float); n=y.size
    if n==0: return y
    if target<=0: return np.zeros(n)
    if target>=n: return np.ones(n)
    lo=float(y.min())-1.0; hi=float(y.max())+1.0
    for _ in range(50):
        mid=0.5*(lo+hi)
        s=float(np.clip(y-mid,0.0,1.0).sum())
        if s>target: lo=mid
        else: hi=mid
    return np.clip(y-0.5*(lo+hi),0.0,1.0)

def fix_sum(h,target):
    h=np.asarray(h,dtype=float); h=np.clip(h,0.0,1.0)
    diff=target-float(h.sum())
    if abs(diff)>1e-14:
        if diff>0:
            order=np.argsort(1.0-h)[::-1]
        else:
            order=np.argsort(h)
        for idx in order:
            if idx>=h.size: break
            if diff>0:
                room=1.0-h[idx]
                if room<=0: continue
                a=min(room,diff)
            else:
                room=h[idx]
                if room<=0: continue
                a=-min(room,-diff)
            h[idx]+=a
            diff=target-float(h.sum())
            if abs(diff)<=1e-15: break
    return h

def compute_c_h(h):
    n=h.size; S=float(h.sum())
    P=np.empty(n+1); P[0]=0.0; np.cumsum(h,out=P[1:])
    L=1
    while L<2*n-1: L <<= 1
    F=np.fft.rfft(h,L)
    H=np.fft.irfft(F*np.conj(F),L)[:n]
    c=np.empty(2*n-1)
    if n>1:
        q=np.arange(1,n)
        c[:n-1]=(P[n-q]-H[1:])[::-1]
        c[n:]=S-P[q]-H[1:]
    c[n-1]=S-H[0]
    c[c<0.0]=0.0
    return c

    m=float(c.max()); s=alpha*(c-m); sm=s.max()

import numpy as np
from scipy.optimize import minimize
import time

def project_box_sum(y, target):
    y = np.asarray(y, dtype=float)
    n = y.size
    if n == 0:
        return y
    if target <= 0.0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def fix_sum(h, target):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        if diff > 0.0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)
        for idx in order:
            if idx >= h.size:
                break
            if diff > 0.0:
                room = 1.0 - h[idx]
                if room <= 0.0:
                    continue
                a = min(room, diff)
            else:
                room = h[idx]
                if room <= 0.0:
                    continue
                a = -min(room, -diff)
            h[idx] += a
            diff = target - float(h.sum())
            if abs(diff) <= 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def sym_h(v, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = v
    h[m:] = v[::-1]
    return h

def sym_from_h(h, n):
    m = n // 2
    v = 0.5 * (h[:m] + h[n - 1:m - 1:-1])
    v = project_box_sum(v, n / 4.0)
    return fix_sum(v, n / 4.0)

def compute_c_h(h):
    n = h.size
    S = float(h.sum())
    P = np.empty(n + 1)
    P[0] = 0.0
    np.cumsum(h, out=P[1:])
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    F = np.fft.rfft(h, L)
    H = np.fft.irfft(F * np.conj(F), L)[:n]
    c = np.empty(2 * n - 1)
    if n > 1:
        q = np.arange(1, n)
        c[:n - 1] = (P[n - q] - H[1:])[::-1]
        c[n:] = S - P[q] - H[1:]
    c[n - 1] = S - H[0]
    c[c < 0.0] = 0.0
    return c

def smooth_value_only_full(h, alpha):
    c = compute_c_h(h)
    m = float(c.max())
    s = alpha * (c - m)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def smooth_value_grad_full(h, alpha):
    c = compute_c_h(h)
    n = h.size
    m = float(c.max())
    s = alpha * (c - m)
    sm = s.max()
    val = m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
    w = np.exp(s - sm)
    w /= w.sum()
    wl = w[:n - 1][::-1]
    wr = w[n:]
    R = np.zeros(n)
    R[1:] = wl + wr
    t1 = np.convolve(R, h, mode='full')[:n]
    t2 = np.convolve(R, h[::-1], mode='full')[:n][::-1]
    cw = np.empty(n)
    cw[0] = 0.0
    cw[1:] = np.cumsum(wl)
    pr = cw[n - 1 - np.arange(n)]
    cw2 = np.empty(n)
    cw2[0] = 0.0
    cw2[1:] = np.cumsum(wr)
    su = cw2[-1] - cw2
    g = pr - su - t1 - t2 + w[n - 1] * (1.0 - 2.0 * h)
    return val, g

def fast_c5(h):
    n = h.size
    dx = 2.0 / n
    if n <= 256:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
    return float(np.max(compute_c_h(h))) * dx

def exact_c5(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def obj_sym(v, n, alpha):
    return smooth_value_only_full(sym_h(v, n), alpha)

def grad_sym(v, n, alpha):
    _, g = smooth_value_grad_full(sym_h(v, n), alpha)
    m = n // 2
    return g[:m] + g[n - 1:m - 1:-1]

def eq_sym(v, n):
    return float(v.sum() - n / 4.0)

def eq_sym_jac(v, n):
    return np.ones(n // 2)

def pen_obj_sym(v, n, alpha, mu):
    s = float(v.sum() - n / 4.0)
    return obj_sym(v, n, alpha) + mu * s * s

def pen_grad_sym(v, n, alpha, mu):
    s = float(v.sum() - n / 4.0)
    return grad_sym(v, n, alpha) + 2.0 * mu * s

def obj_full(h, n, alpha):
    return smooth_value_only_full(h, alpha)

def grad_full(h, n, alpha):
    return smooth_value_grad_full(h, alpha)[1]

def pen_obj_full(h, n, alpha, mu):
    s = float(h.sum() - n / 2.0)
    return obj_full(h, n, alpha) + mu * s * s

def pen_grad_full(h, n, alpha, mu):
    s = float(h.sum() - n / 2.0)
    return grad_full(h, n, alpha) + 2.0 * mu * s

def slsqp_sym(x, n, alpha, maxiter):
    m = n // 2
    x = fix_sum(project_box_sum(x, n / 4.0), n / 4.0)
    cons = [{'type': 'eq', 'fun': eq_sym, 'jac': eq_sym_jac, 'args': (n,)}]
    try:
        res = minimize(obj_sym, x, args=(n, alpha), jac=grad_sym, method='SLSQP',
                       bounds=[(0.0, 1.0)] * m, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return fix_sum(project_box_sum(res.x, n / 4.0), n / 4.0)
    except Exception:
        return x

def lbfgs_sym(x, n, alpha, mu, maxiter):
    x = fix_sum(project_box_sum(x, n / 4.0), n / 4.0)
    try:
        res = minimize(pen_obj_sym, x, args=(n, alpha, mu), jac=pen_grad_sym,
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * (n // 2),
                       options={'maxiter': maxiter, 'ftol': 1e-13,
                                'gtol': 1e-10, 'maxfun': maxiter * 25, 'maxls': 50})
        return fix_sum(project_box_sum(res.x, n / 4.0), n / 4.0)
    except Exception:
        return x

def lbfgs_full(x, n, alpha, mu, maxiter):
    target = n / 2.0
    x = fix_sum(project_box_sum(x, target), target)
    try:
        res = minimize(pen_obj_full, x, args=(n, alpha, mu), jac=pen_grad_full,

                       options={'maxiter': maxiter, 'ftol': 1e-13,
                                'gtol': 1e-10, 'maxfun': maxiter * 25, 'maxls': 50})
        return fix_sum(project_box_sum(res.x, target), target)
    except Exception:
        return x

def optimize_sym(v, n, stop_time):

    best = exact_c5(sym_h(v, n))
    vb = v.copy()
    m = n // 2
    mi = 35 if m >= 200 else (65 if m >= 100 else 90)

def interpolate_to_n(h, n0, n1):
    if n0 == n1:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    y = np.interp(x1, x0, h)
    return fix_sum(project_box_sum(y, n1 / 2.0), n1 / 2.0)

    h = np.asarray(h, dtype=float)

def plan_stop(deadline, alloc, reserve=30.0):
    now = time.time()
    rem = deadline - now - reserve
    if rem <= 1.0:
        return now
    return now + min(float(alloc), rem)

def c5_transfer(h, n, i, j, d):
    a = h[i]
    b = h[j]
    h[i] = a + d
    h[j] = b - d
    val = fast_c5(h)
    h[i] = a
    h[j] = b
    return val

def golden_pair(h, n, i, j, hi, best, it=8):
    if hi <= 1e-12:
        return 0.0, best
    a = 0.0
    b = hi
    gr = 0.6180339887498949
    x1 = b - gr * (b - a)
    x2 = a + gr * (b - a)
    f1 = c5_transfer(h, n, i, j, x1)
    f2 = c5_transfer(h, n, i, j, x2)
    fb = c5_transfer(h, n, i, j, b)
    if fb <= f1 and fb <= f2:
        if fb < best:
            return b, fb
        return 0.0, best
    for _ in range(it):
        if f1 <= f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = b - gr * (b - a)
            f1 = c5_transfer(h, n, i, j, x1)
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + gr * (b - a)
            f2 = c5_transfer(h, n, i, j, x2)
    d = 0.5 * (a + b)
    f = c5_transfer(h, n, i, j, d)
    if fb < f:
        d = b
        f = fb
    if f < best:
        return d, f
    return 0.0, best

    if nstart > 800:
        nstart = 600
    for kind in (0, 1, 2):
        stop = plan_stop(deadline, min(80.0, 0.10 * b), 90.0)
        if stop <= time.time():
            break
        v = make_sym_start(nstart, rng, kind)
        v1, _ = optimize_sym(v, nstart, stop)
        best_h, best_c, best_n = update_best(sym_h(v1, nstart), best_c, nstart, best_h, best_c, best_n)

    if time.time() < deadline - 5.0 and best_h is not None:
        h1, c1 = pair_polish_full(best_h, best_n, deadline - 2.0, rng, 6 if best_n <= 520 else 4)
        best_h, best_c, best_n = update_best(h1, c1, best_n, best_h, best_c, best_n)

    if best_h is None:
        best_n = 400
        v = make_sym_start(best_n, rng, 0)
        best_h = sym_h(v, best_n)
        best_c = exact_c5(best_h)

    best_h = fix_sum(project_box_sum(best_h, best_n / 2.0), best_n / 2.0)
    best_c = exact_c5(best_h)
    return best_h, float(best_c), int(best_n)
