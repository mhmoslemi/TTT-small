import numpy as np
import time
from scipy.optimize import minimize

_SLQ_N = 0
_SLQ_ALPHA = 1.0


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_simple(h, n):
    x = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(300):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = x < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            x[mask] += diff / cnt
        else:
            mask = x > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            x[mask] += diff / cnt
    return np.clip(x, 0.0, 1.0)


def project_feasible(h, n):
    x = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    if x.size == 0:
        return x
    low = float(np.min(x)) - 1.0
    high = float(np.max(x)) + 1.0
    for _ in range(90):
        mid = 0.5 * (low + high)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s < target:
            high = mid
        else:
            low = mid
    lam = 0.5 * (low + high)
    y = np.clip(x - lam, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        idx = int(np.argmax(1.0 - y)) if diff > 0 else int(np.argmax(y))
        y[idx] += diff
        y = np.clip(y, 0.0, 1.0)
        diff2 = target - y.sum()
        if abs(diff2) > 1e-14:
            y += diff2 / n
            y = np.clip(y, 0.0, 1.0)
    return y


def binarize_top(h, n):
    N = int(n) // 2
    h = np.asarray(h, dtype=float)
    y = np.zeros(n, dtype=np.int32)
    if N > 0:
        idx = np.argpartition(-h, N - 1)[:N]
        y[idx] = 1
    return y


def golden_binary(n):
    N = n // 2
    v = np.remainder(np.arange(n) * 0.6180339887498949, 1.0)
    h = np.zeros(n, dtype=np.int32)
    if N > 0:
        idx = np.argpartition(-v, N - 1)[:N]
        h[idx] = 1
    return h


def compute_V(h, n):
    center = n - 1
    L = 2 * n - 1
    q = np.arange(L, dtype=np.int64) - center
    V = np.zeros((L, n), dtype=np.int32)
    for p in range(n):
        ip = p + q
        m1 = (ip >= 0) & (ip < n)
        iq = p - q
        m2 = (iq >= 0) & (iq < n)
        vp = np.zeros(L, dtype=np.int32)
        vp[m1] = 1 - h[ip[m1]]
        vp[m2] -= h[iq[m2]]
        vp[center] = 0
        V[:, p] = vp
    return V


def binary_sweeps(h, n, deadline, seed, max_sweeps):
    h = np.asarray(h, dtype=np.int32).copy()
    center = n - 1
    corr = np.correlate(h, 1 - h, mode="full").astype(np.int32)
    cur = int(corr.max())
    rng = np.random.RandomState(seed)
    cols = None

    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if zeros.size < 1 or ones.size < 1:
            break

        V = compute_V(h, n)
        one_order = ones[rng.permutation(ones.size)]
        K = int(one_order.size)
        if cols is None or cols.size != K:
            cols = np.arange(K, dtype=np.int64)

        best_m = cur
        best_i = -1
        best_j = -1
        stopped = False

        for i in zeros:
            if time.time() > deadline:
                stopped = True
                break
            base = corr + V[:, i]
            M = base[:, None] - V[:, one_order]
            d = one_order - i
            M[center + d, cols] += 1
            M[center - d, cols] += 1
            mj = M.max(axis=0)
            k = int(np.argmin(mj))
            m = int(mj[k])
            if m < best_m:
                best_m = m
                best_i = int(i)
                best_j = int(one_order[k])

        if stopped:
            if best_i >= 0:
                h[best_i] = 1
                h[best_j] = 0
                corr = np.correlate(h, 1 - h, mode="full").astype(np.int32)
                cur = int(corr.max())
            break

        if best_i < 0:
            break

        h[best_i] = 1
        h[best_j] = 0
        corr = np.correlate(h, 1 - h, mode="full").astype(np.int32)
        cur = int(corr.max())

    return h, cur


def binary_optim(h0, n, deadline, seed, restarts=6, max_sweeps=60):
    hb = binarize_top(h0, n)
    best_h = hb.copy()
    best_c = float(np.max(np.correlate(hb, 1 - hb, mode="full")) * (2.0 / n))

    if time.time() < deadline - 2.0:
        h2, c2 = binary_sweeps(hb, n, deadline, seed, max_sweeps)
        c2f = float(c2) * (2.0 / n)
        if c2f < best_c:
            best_h = h2
            best_c = c2f

    rng = np.random.RandomState(seed)
    for r in range(restarts):
        if time.time() > deadline - 5.0:
            break
        h = best_h.copy()
        k = int(rng.randint(2, max(3, min(16, 2 + r * r))))
        for _ in range(k):
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            a = int(ones[rng.randint(ones.size)])
            b = int(zeros[rng.randint(zeros.size)])
            h[a], h[b] = 0, 1

        h2, c2 = binary_sweeps(h, n, deadline, seed + 100 + r, 30)
        c2f = float(c2) * (2.0 / n)
        if c2f < best_c:
            best_h = h2
            best_c = c2f

    return best_h, best_c


def slsqp_f(x):
    n = _SLQ_N
    alpha = _SLQ_ALPHA
    c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    z = alpha * c
    z = z - z.max()
    return float((z.max() + np.log(np.exp(z).sum())) / alpha)


def slsqp_g(x):
    n = _SLQ_N
    alpha = _SLQ_ALPHA
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    z = alpha * c
    z = z - z.max()
    e = np.exp(z)
    w = e / e.sum()
    ws = w * dx

    g = np.zeros(n, dtype=float)
    center = n - 1
    g[center] += ws[center] * (1.0 - 2.0 * x)

    for p in range(1, n):
        wp = ws[center + p]
        if wp:
            g[:n - p] += wp * (1.0 - x[p:])
            g[p:] -= wp * x[:n - p]
        wn = ws[center - p]
        if wn:
            g[p:] += wn * (1.0 - x[:n - p])
            g[:p] -= wn * x[p:]

    return g


def slsqp_c(x):
    return x.sum() - 0.5 * _SLQ_N


def slsqp_smooth(h, n, alpha, maxiter):
    global _SLQ_N, _SLQ_ALPHA
    _SLQ_N = int(n)
    _SLQ_ALPHA = float(alpha)

    x0 = project_simple(h, n)
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))
    cons = ({"type": "eq", "fun": slsqp_c},)

    try:
        res = minimize(
            slsqp_f,
            x0,
            jac=slsqp_g,
            bounds=bounds,
            constraints=cons,
            method="SLSQP",
            options={"maxiter": maxiter, "ftol": 1e-12, "disp": False},
        )
        return project_feasible(res.x, n)
    except Exception:
        return x0


def slsqp_sequence(h, n, deadline, seed, alphas):
    best = np.asarray(h, dtype=float).copy()
    best_c = compute_c5(best, n)
    rng = np.random.RandomState(seed)
    maxiter = 180 if n <= 300 else 110

    for ai, alpha in enumerate(alphas):
        if time.time() > deadline - 5.0:
            break
        x0 = project_simple(best, n)
        noise = 0.0
        if ai == 0:
            noise = 0.025
        elif ai == 1:
            noise = 0.008
        if noise > 0.0:
            x0 = project_simple(x0 + noise * rng.randn(n), n)

        y = slsqp_smooth(x0, n, alpha, maxiter)
        c = compute_c5(y, n)
        if c < best_c:
            best = y
            best_c = c

    return best, best_c


def resample_h(h, n_old, n):
    if n_old == n:
        return project_feasible(h.copy(), n)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n)
    return project_feasible(np.interp(x_new, x_old, h), n)
