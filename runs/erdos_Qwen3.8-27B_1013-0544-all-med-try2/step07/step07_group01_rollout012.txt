import numpy as np
from scipy.optimize import minimize
import time

def proj_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_cvec(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_cvec(h, n)))

def softmax_weights(u, alpha):
    m = float(u.max())
    s = np.minimum(alpha * (u - m), 0.0)
    w = np.exp(s)
    sw = float(w.sum())
    if sw <= 0.0:
        w = np.zeros_like(u)
        w[int(np.argmax(u))] = 1.0
    else:
        w /= sw
    return w

def grad_from_weights(h, w):
    n = len(h)
    c = n - 1
    g = np.zeros(n)
    w0 = float(w[c])
    if abs(w0) > 0.0:
        g += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        a = float(w[c - d])
        b = float(w[c + d])
        if a == 0.0 and b == 0.0:
            continue
        seg = n - d
        left = h[:seg]
        right = h[d:]
        if a != 0.0:
            g[:seg] += a * (1.0 - right)
            g[d:] += -a * left
        if b != 0.0:
            g[:seg] += -b * right
            g[d:] += b * (1.0 - left)
    return g

def smooth_obj(h, n, alpha):
    u = compute_cvec(h, n)
    m = float(u.max())
    s = np.minimum(alpha * (u - m), 0.0)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def smooth_grad(h, n, alpha):
    u = compute_cvec(h, n)
    return grad_from_weights(h, softmax_weights(u, alpha))

def lbfgs_obj_fn(x, n, alpha, mu):
    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x)):
        return 1e100
    return smooth_obj(x, n, alpha) + mu * float((x.sum() - n / 2.0) ** 2)

def lbfgs_jac_fn(x, n, alpha, mu):
    x = np.asarray(x, dtype=float)
    return smooth_grad(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0)

def lbfgs_phase(h, n, deadline, alphas, maxiter):
    h = proj_feasible(h, n)
    best = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 1:
            break
        try:
            res = minimize(lbfgs_obj_fn, h, args=(n, alpha, 1e3), jac=lbfgs_jac_fn,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                                    'maxfun': maxiter * 5, 'maxls': 30})
            x = proj_feasible(res.x, n)
            c = compute_c5(x, n)
            if c < best - 1e-15:
                h, best = x, c
        except Exception:
            pass
    return h, best

def slsqp_obj(x, n, alpha):
    return smooth_obj(x, n, alpha)

def slsqp_jac(x, n, alpha):
    return smooth_grad(x, n, alpha)

def slsqp_eq(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_eq_jac(x, n):
    return np.ones(len(x))

def slsqp_phase(h, n, deadline, alpha, maxiter):
    h = proj_feasible(h, n)
    best = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), jac=slsqp_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        x = proj_feasible(res.x, n)
        c = compute_c5(x, n)
        if c < best - 1e-15:
            return x, c
    except Exception:
        pass
    return h, best

def make_pair_lin(h, i, j, n, dx):
    lin = np.zeros(2 * n - 1)
    b = 1.0 - h
    lin[i:n + i] += b[::-1]
    lin[n - 1 - i:2 * n - 1 - i] += -h
    lin[j:n + j] -= b[::-1]
    lin[n - 1 - j:2 * n - 1 - j] += h
    lin *= dx
    return lin

                    bv = best
                    if hi > 1e-14:
                        for f in fracs:
                            d = f * hi
                            if abs(d) < 1e-16:
                                continue
                            tmp = U + d * lin
                            dd = d * d * dx
                            if D:
                                tmp[c - D] += dd
                                tmp[c + D] += dd
                            tmp[c] -= 2.0 * dd
                            val = float(tmp.max())
                            if val < bv - 1e-14:
                                bv, bd = val, d
                    if lo < -1e-14:
                        for f in fracs:
                            d = f * lo
                            if abs(d) < 1e-16:
                                continue
                            tmp = U + d * lin
                            dd = d * d * dx
                            if D:
                                tmp[c - D] += dd
                                tmp[c + D] += dd
                            tmp[c] -= 2.0 * dd
                            val = float(tmp.max())
                            if val < bv - 1e-14:
                                bv, bd = val, d
                    if bd != 0.0 and bv < best - 1e-14:
                        h[i] += bd
                        h[j] -= bd
                        h[i] = float(np.clip(h[i], 0.0, 1.0))
                        h[j] = float(np.clip(h[j], 0.0, 1.0))
                        U += bd * lin
                        dd = bd * bd * dx
                        if D:
                            U[c - D] += dd
                            U[c + D] += dd
                        U[c] -= 2.0 * dd
                        U = np.maximum(U, 0.0)
                        best = bv
                        no_improve = 0
                        improved_now = True
                        accepts += 1
                        if accepts % 50 == 0:
                            hs = 0.5 * (h + h[::-1])
                            cs = compute_c5(hs, n)
                            if cs < best - 1e-14:
                                h, best, U, accepts = hs, cs, compute_cvec(hs, n), 0
                        break
                if not improved_now:
                    no_improve += tries
    U = compute_cvec(h, n)
    best = float(U.max())
    return h, best

def sym_if_better(h, n, best_c):
    h2 = 0.5 * (h + h[::-1])
    c2 = compute_c5(h2, n)
    if c2 < best_c - 1e-15:
        return h2, c2
    return h, best_c

def binarize(h, n):
    b = np.zeros(n)
    if n % 2 == 0:
        idx = np.argsort(h)
        b[idx[n // 2:]] = 1.0
    else:
        b = (h >= 0.5).astype(float)
        b = proj_feasible(b, n)
    return b

def random_binary(n, rng):
    b = np.zeros(n)
    if n % 2 == 0:
        idx = rng.choice(n, n // 2, replace=False)
        b[idx] = 1.0
    else:
        b = proj_feasible((rng.rand(n) < 0.5).astype(float), n)
    return b

def resample_h(h, n_old, n_new):
    if n_new == n_old:
        return proj_feasible(h, n_new)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return proj_feasible(np.interp(x_new, x_old, h), n_new)

def optimize_candidate(h, n, deadline, best_c, strength, rng):
    h = proj_feasible(h, n)
    c = compute_c5(h, n)
    h, c = sym_if_better(h, n, c)
    if time.time() >= deadline:
        return h, c
    if strength == 1:
        if time.time() < deadline - 5:
            h, c = lbfgs_phase(h, n, deadline, (50, 300, 2000, 10000, 50000), 60)
            h, c = sym_if_better(h, n, c)
        if time.time() < deadline - 70 and n <= 320:
            h, c = slsqp_phase(h, n, min(deadline - 10, time.time() + 60), 50000, 80)
            h, c = sym_if_better(h, n, c)
        h, c = pair_refine(h, n, deadline - 2,
                           (0.05, 0.02, 0.008, 0.003, 0.001, 0.0004),
                           (200, 1000, 5000, 20000), rng, 8, 12, 500)
        h, c = sym_if_better(h, n, c)
    elif strength == 2:
        big_dl = min(deadline - 60, time.time() + 90)
        if time.time() < big_dl:
            h, c = pair_refine(h, n, big_dl,
                               (1.0, 0.5, 0.2, 0.08),
                               (300, 1000, 5000), rng, 6, 8, 300)
            h, c = sym_if_better(h, n, c)
        if time.time() < deadline - 35 and n <= 500:
            h, c = lbfgs_phase(h, n, deadline - 25, (500, 5000), 35)
            h, c = sym_if_better(h, n, c)
        h, c = pair_refine(h, n, deadline - 2,
                           (0.05, 0.02, 0.008, 0.003, 0.001),
                           (1000, 5000), rng, 8, 12, 400)
        h, c = sym_if_better(h, n, c)
    else:
        if time.time() < deadline - 20 and n <= 500:
            h, c = lbfgs_phase(h, n, deadline - 20, (200, 2000, 10000), 30)
            h, c = sym_if_better(h, n, c)
        h, c = pair_refine(h, n, deadline - 2,
                           (0.02, 0.008, 0.003, 0.001, 0.0004),
                           (1000, 5000), rng, 8, 12, 400)
        h, c = sym_if_better(h, n, c)
    if time.time() < deadline - 6:
        h, c = pair_refine(h, n, deadline - 2,
                           (0.002, 0.0008, 0.0003),
                           (5000, 20000), rng, 8, 10, 200)
        h, c = sym_if_better(h, n, c)
    h = proj_feasible(h, n)
    c = compute_c5(h, n)
    h, c = sym_if_better(h, n, c)
    return h, c

def attempt_candidate(h, n, alloc_s, best_c, strength, rng):
    alloc_s = float(alloc_s)
    if alloc_s < 25 or n < 2:
        return h, best_c, False
    dl = time.time() + alloc_s
    h2, c2 = optimize_candidate(h, n, dl, best_c, strength, rng)
    if c2 < best_c - 1e-15:
        return h2, c2, True
    return h, best_c, False

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(60.0, budget_s - 5.0)
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = int(len(h_init))
    except Exception:
        n_init = 300
        h_init = np.full(n_init, 0.5)
    if n_init < 20:
        n_init = 300
        h_init = np.full(n_init, 0.5)

    main_deadline = t0 + 0.70 * max(1.0, budget_s - 5.0)
    main_deadline = min(main_deadline, deadline - 200.0)
    main_deadline = max(main_deadline, t0 + 60.0)

    best_h, best_c = optimize_candidate(h_init, n_init, main_deadline, np.inf, 1, rng)
    best_n = int(len(best_h))

    rem = deadline - time.time()
    if rem > 160:
        hc = 0.5 * (h_init + h_init[::-1]).astype(float)
        best_h, best_c, ok = attempt_candidate(hc, n_init, min(90.0, rem - 90.0), best_c, 0, rng)
        if ok:
            best_n = int(len(best_h))

    rem = deadline - time.time()
    if best_n % 2 == 0 and rem > 160:
        b = binarize(best_h, best_n)
        best_h, best_c, ok = attempt_candidate(b, best_n, min(120.0, rem - 90.0), best_c, 2, rng)
        if ok:
            best_n = int(len(best_h))

    rem = deadline - time.time()
    for scale in (0.05, 0.02, 0.008):
        if rem < 110:
            break
        alloc = min(70.0, rem - 80.0)
        if alloc < 30:
            break
        hc = best_h + rng.normal(0.0, scale, best_n)
        best_h, best_c, ok = attempt_candidate(hc, best_n, alloc, best_c, 0, rng)
        if ok:
            best_n = int(len(best_h))
        rem = deadline - time.time()

    rem = deadline - time.time()
    for nt in (150, 200, 450):
        if rem < 140:
            break
        if nt == best_n or nt % 2 != 0 or nt > 800:
            continue
        alloc = min(80.0, rem - 80.0)
        if alloc < 35:
            break
        hc = resample_h(best_h, best_n, nt)
        best_h, best_c, ok = attempt_candidate(hc, nt, alloc, best_c, 0, rng)
        if ok:
            best_n = nt
        rem = deadline - time.time()

    rem = deadline - time.time()
    if rem > 120:
        n_rb = best_n if best_n % 2 == 0 else 300
        rb = random_binary(n_rb, rng)
        alloc = min(70.0, rem - 80.0)
        if alloc > 30:
            best_h, best_c, ok = attempt_candidate(rb, n_rb, alloc, best_c, 2, rng)
            if ok:
                best_n = n_rb

    if time.time() < deadline - 8:
        h_f = best_h.copy()
        if time.time() < deadline - 50 and best_n <= 500:
            h_f, c_f = lbfgs_phase(h_f, best_n, min(deadline - 30, time.time() + 30),
                                    (5000, 20000), 25)
            if c_f < best_c:
                best_h, best_c = h_f, c_f
        h_f = best_h.copy()
        h_f, c_f = pair_refine(h_f, best_n, deadline - 2,
                               (0.003, 0.001, 0.0004, 0.00015),
                               (5000, 20000, 80000), rng, 8, 10, 300)
        h_f, c_f = sym_if_better(h_f, best_n, c_f)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = proj_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    best_h, best_c = sym_if_better(best_h, best_n, best_c)
    best_h = proj_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(len(best_h))
