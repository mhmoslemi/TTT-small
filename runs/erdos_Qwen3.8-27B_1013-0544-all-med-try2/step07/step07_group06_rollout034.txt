import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = compute_corr(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_corr(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [50.0, 200.0, 1000.0, 5000.0, 20000.0, 100000.0, 500000.0, 2000000.0]

            return smooth_max_val(x, n, alpha)

            return grad_smooth_max(x, n, alpha)
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': max_iters * 30, 'maxls': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, 5000)
            order = np.argsort(-np.abs(g))
            for i in order:
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: L-BFGS-B with alpha schedule
    t1 = min(t0 + 150, deadline - 500)
    if t1 > t0 + 10:
        h1, c1 = lbfgs_phase(best_h, n, t1, max_iters=250)
        if c1 < best_c:
            best_c, best_h = c1, h1
    
    # Phase 2: Block coordinate descent (random blocks)
    t2 = min(time.time() + 300, deadline - 350)
    if t2 > time.time() + 30:
        h2, c2 = block_cd(best_h, n, t2, K=10, alpha=10000, max_blocks=12000)
        if c2 < best_c:
            best_c, best_h = c2, h2
    
    # Phase 3: Gradient-guided block CD
    t3 = min(time.time() + 250, deadline - 200)
    if t3 > time.time() + 30:
        h3, c3 = grad_block_cd(best_h, n, t3, K=12, alpha=50000, max_blocks=8000)
        if c3 < best_c:
            best_c, best_h = c3, h3
    
    # Phase 4: More block CD with different params
    t4 = min(time.time() + 150, deadline - 100)
    if t4 > time.time() + 20:
        h4, c4 = block_cd(best_h, n, t4, K=8, alpha=50000, max_blocks=6000)
        if c4 < best_c:
            best_c, best_h = c4, h4
        h4b, c4b = grad_block_cd(best_h, n, deadline - 100, K=15, alpha=200000, max_blocks=4000)
        if c4b < best_c:
            best_c, best_h = c4b, h4b
    
    # Phase 5: Pair search polish
    if time.time() < deadline - 30:
        h5, c5 = pair_search(best_h, n, deadline - 10,
                             [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                             radius=40)
        if c5 < best_c:
            best_c, best_h = c5, h5
    
    # Phase 6: Try different n
    if time.time() < deadline - 20:
        for n_try in [250, 400]:
            if time.time() > deadline - 10:
                break
            alloc = deadline - time.time() - 8
            if alloc < 20:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.3, max_iters=150)
            if time.time() < deadline - 10:
                h_t, c_t = block_cd(h_t, n_try, min(time.time() + alloc * 0.5, deadline - 10),
                                     K=10, alpha=50000, max_blocks=2000)
            if c_t < best_c:
                best_c, best_h = c_t, h_t
    
    # Final polish
    if time.time() < deadline - 2:
        h_f, c_f = pair_search(best_h, len(best_h), deadline - 1,
                               [0.001, 0.0005, 0.0002, 0.0001], radius=30)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
