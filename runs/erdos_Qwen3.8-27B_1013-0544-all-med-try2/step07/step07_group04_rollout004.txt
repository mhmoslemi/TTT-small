def smooth_value(c,beta):
    z = beta * c

import numpy as np
import time
from scipy.optimize import minimize

def corr_cvector(h):
    n = int(h.size)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def eval_max(h):
    return float(np.max(corr_cvector(h)))

def smooth_value(c, beta):
    z = beta * c
    zmax = float(z.max())
    ez = np.exp(z - zmax)
    s = float(ez.sum())
    if s <= 0.0 or not np.isfinite(s):
        return zmax / beta
    return (zmax + float(np.log(s))) / beta

def smooth_weights(c, beta):
    z = beta * c
    zmax = float(z.max())
    ez = np.exp(z - zmax)
    s = float(ez.sum())
    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros(c.size)
        p[int(np.argmax(c))] = 1.0
    else:
        p = ez / s
    l = (zmax + float(np.log(s))) / beta
    return l, p

def weighted_gradient(h, w):
    n = int(h.size)
    oh = 1.0 - h
    a = np.convolve(w, oh, mode='full')
    b = np.convolve(h, w[::-1], mode='full')
    dx = 2.0 / n
    return dx * (a[n - 1:3 * n - 2] - b[n - 1:3 * n - 2])

def project_feasible(v, n):
    T = 0.5 * n
    v = np.asarray(v, dtype=float)
    if v.size != n:
        v = np.resize(v, n)
    h = np.clip(v, 0.0, 1.0)
    r = T - float(h.sum())
    if abs(r) > 1e-7:
        vmax = float(np.max(np.maximum(v, 0.0)))
        vpos = float(np.max(np.maximum(1.0 - v, 0.0)))
        lo = -vmax - 1.0
        hi = vpos + 1.0
        for _ in range(35):
            mid = 0.5 * (lo + hi)
            s = float(np.clip(v + mid, 0.0, 1.0).sum())
            if s > T:
                hi = mid
            else:
                lo = mid
        h = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
        r = T - float(h.sum())
    for _ in range(25):
        if abs(r) < 1e-13:
            break
        mask = h < 1.0 if r > 0.0 else h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += r / cnt
        h = np.clip(h, 0.0, 1.0)
        r = T - float(h.sum())
    for _ in range(50):
        if abs(r) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += r / cnt
        h = np.clip(h, 0.0, 1.0)
        r = T - float(h.sum())
    return h

def slsqp_eq(x,n):

def slsqp_polish(h,n,beta,maxiter):

import numpy as np
import time
from scipy.optimize import minimize

def corr_cvector(h):
    n = int(h.size)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def eval_max(h):
    return float(np.max(corr_cvector(h)))

def smooth_value(c, beta):
    z = beta * c
    zmax = float(z.max())
    ez = np.exp(z - zmax)
    s = float(ez.sum())
    if s <= 0.0 or not np.isfinite(s):
        return zmax / beta
    return (zmax + float(np.log(s))) / beta

def smooth_weights(c, beta):
    z = beta * c
    zmax = float(z.max())
    ez = np.exp(z - zmax)
    s = float(ez.sum())
    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros(c.size)
        p[int(np.argmax(c))] = 1.0
    else:
        p = ez / s
    l = (zmax + float(np.log(s))) / beta
    return l, p

def weighted_gradient(h, w):
    n = int(h.size)
    oh = 1.0 - h
    a = np.convolve(w, oh, mode='full')
    b = np.convolve(h, w[::-1], mode='full')
    dx = 2.0 / n
    return dx * (a[n - 1:3 * n - 2] - b[n - 1:3 * n - 2])

def project_feasible(v, n):
    T = 0.5 * n
    v = np.asarray(v, dtype=float).reshape(-1)
    if v.size != n:
        v = np.resize(v, n)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(v, 0.0, 1.0)
    r = T - float(h.sum())
    if abs(r) > 1e-7:
        vmax = float(np.max(np.maximum(v, 0.0)))
        vpos = float(np.max(np.maximum(1.0 - v, 0.0)))
        lo = -vmax - 1.0
        hi = vpos + 1.0
        for _ in range(35):
            mid = 0.5 * (lo + hi)
            s = float(np.clip(v + mid, 0.0, 1.0).sum())
            if s > T:
                hi = mid
            else:
                lo = mid
        h = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
        r = T - float(h.sum())
    for _ in range(25):
        if abs(r) < 1e-13:
            break
        mask = h < 1.0 if r > 0.0 else h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += r / cnt
        h = np.clip(h, 0.0, 1.0)
        r = T - float(h.sum())
    for _ in range(50):
        if abs(r) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += r / cnt
        h = np.clip(h, 0.0, 1.0)
        r = T - float(h.sum())
    if abs(r) > 1e-15:
        if r > 0:
            j = int(np.argmax(h))
            add = min(r, 1.0 - h[j])
            if add > 0.0:
                h[j] += add
        else:
            j = int(np.argmin(h))
            sub = min(-r, h[j])
            if sub > 0.0:
                h[j] -= sub
    return h

def binary_feasible(h, n):
    T = 0.5 * n
    if abs(T - round(T)) > 1e-8:
        return None
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:int(T)]] = 1.0
    return b

def interpolate_h(h, n0, n1):
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n1) + 0.5) * (2.0 / n1)
    return project_feasible(np.interp(x1, x0, h), int(n1))

    if max_seconds <= 0.0:
        return h, eval_max(h)
    deadline = time.time() + float(max_seconds)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx

    best_c = cmax
    l, p = smooth_weights(c, beta)
    if beta < 80:

    elif beta < 400:
        eta = 0.03
    elif beta < 2000:
        eta = 0.016
    else:
        eta = 0.009

            c = np.correlate(h, 1.0 - h, mode='full') * dx
            cmax = float(c.max())

def optimize_schedule(h, n, limit, seed, betas):
    hbest = project_feasible(h, n).copy()
    cbest = eval_max(hbest)
    cur = hbest.copy()
    b = binary_feasible(cur, n)
    if b is not None:
        cb = eval_max(b)
        if cb < cbest - 1e-12:
            cbest = cb
            hbest = b.copy()
            cur = hbest.copy()
    weights = [0.08, 0.10, 0.14, 0.18, 0.22, 0.16, 0.12]
    for i, beta in enumerate(betas):
        rem = limit - time.time()
        if rem < 12.0:
            break
        w = weights[i] if i < len(weights) else weights[-1]
        tail = 0.0
        for j in range(i, len(betas)):
            tail += weights[j] if j < len(weights) else weights[-1]
        if tail <= 0.0:
            tail = 1.0
        alloc = rem * w / tail
        if alloc < 6.0:
            alloc = min(rem, 6.0)
        alloc = min(alloc, rem)
        hloc, cloc = pg_refine(cur, n, float(beta), alloc, seed * 100000 + i * 13 + 7)
        if cloc < cbest - 1e-12:
            cbest = cloc
            hbest = hloc.copy()
        if cloc <= cbest + 2e-4:
            cur = hloc.copy()
        else:
            cur = hbest.copy()
    return hbest, cbest

def slsqp_obj(x, n, beta):
    return smooth_value(corr_cvector(x), beta)

def slsqp_grad(x, n, beta):
    c = corr_cvector(x)
    _, p = smooth_weights(c, beta)
    return weighted_gradient(x, p)

def slsqp_eq(x, n):
    return float(np.sum(x) - 0.5 * n)

def slsqp_polish(h, n, beta, maxiter):
    h = project_feasible(h, n)
    c0 = eval_max(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'args': (n,)}]
    try:
        res = minimize(
            slsqp_obj, h, method='SLSQP', jac=slsqp_grad,
            bounds=bounds, constraints=cons, args=(n, beta),
            options={'maxiter': int(maxiter), 'ftol': 1e-14, 'iprint': 0}
        )
        h2 = project_feasible(np.asarray(res.x, dtype=float), n)
        if not np.all(np.isfinite(h2)):
            return h, c0
        c2 = eval_max(h2)
        if c2 < c0:
            return h2, c2
        return h, c0
    except Exception:
        return h, c0
