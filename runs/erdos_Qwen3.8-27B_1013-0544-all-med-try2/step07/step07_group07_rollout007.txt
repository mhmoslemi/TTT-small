```python
import time
import numpy as np
from scipy.optimize import minimize


def compute_lags(h):
    n = len(h)
    if n == 0:
        return np.zeros(1)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def compute_c5(h):
    c = compute_lags(h)
    if c.size == 0:
        return 0.0
    return float(np.max(c))


def project_feasible(y):
    y = np.asarray(y, dtype=float).copy()
    n = len(y)
    target = 0.5 * n
    if not np.all(np.isfinite(y)):
        y = np.full(n, 0.5)
    if (np.all(y >= -1e-10) and np.all(y <= 1.0 + 1e-10)
            and abs(float(np.sum(y)) - target) < 1e-9):
        return np.clip(y, 0.0, 1.0)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    s = float(np.sum(h))
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0.0:
            room = 1.0 - h
            tot = float(np.sum(room))
            if tot > 1e-15:
                h += (diff / tot) * room
        else:
            tot = float(np.sum(h))
            if tot > 1e-15:
                h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def smooth_value(c, alpha):
    if c.size == 0:
        return 0.0
    m = float(np.max(c))
    sh = alpha * (c - m)
    mx = float(np.max(sh))
    if mx > 500.0:
        return m + (mx + float(np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return m + float(np.log(np.sum(np.exp(sh)))) / alpha


def smooth_weights(c, alpha):
    m = float(np.max(c))
    sh = alpha * (c - m)
    mx = float(np.max(sh))
    p = np.exp(sh - mx)
    s = float(np.sum(p))
    if s <= 0.0:
        p = np.zeros_like(c)
        p[0] = 1.0
    else:
        p /= s
    return p


def weighted_gradient_full(h, w):
    n = len(h)
    dx = 2.0 / n
    off = n - 1
    grad = np.zeros(n)
    eps = 1e-16
    wk = float(w[off])
    if wk > eps:
        grad += wk * dx * (1.0 - 2.0 * h)
    for m in range(1, n):
        wp = float(w[off - m])
        wn = float(w[off + m])
        if wp <= eps and wn <= eps:
            continue
        a = n - m
        if wp > eps:
            s = wp * dx
            grad[:a] += s * (1.0 - h[m:])
            grad[m:] -= s * h[:a]
        if wn > eps:
            s = wn * dx
            grad[:a] -= s * h[m:]
            grad[m:] += s * (1.0 - h[:a])
    return grad


def weighted_gradient_indices(h, idx, w):
    n = len(h)
    dx = 2.0 / n
    off = n - 1
    grad = np.zeros(n)
    eps = 1e-16
    for ii, kk in enumerate(idx):
        wk = float(w[ii])
        if wk <= eps:
            continue
        lag = int(kk) - off
        m = abs(lag)
        scale = wk * dx
        a = n - m
        if m == 0:
            grad += scale * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:a] -= scale * h[m:]
            grad[m:] += scale * (1.0 - h[:a])
        else:
            grad[:a] += scale * (1.0 - h[m:])
            grad[m:] -= scale * h[:a]
    return grad


def evaluate_smooth(h, alpha):
    c = compute_lags(h)
    f = smooth_value(c, alpha)
    w = smooth_weights(c, alpha)
    g = weighted_gradient_full(h, w)
    return f, g, c


def _sum_obj(x, n):
    return float(np.sum(x) - 0.5 * n)


def _sum_jac(x, n):
    return np.ones(len(x))


def _smooth_obj_idx(x, n, idx, alpha):
    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x)):
        return 1e100
    c = compute_lags(x)
    if idx is None:
        vals = c
    else:
        vals = c[idx]
    return smooth_value(vals, alpha)


def _smooth_jac_idx(x, n, idx, alpha):
    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x)):
        return np.zeros(len(x))
    c = compute_lags(x)
    if idx is None:
        w = smooth_weights(c, alpha)
        return weighted_gradient_full(x, w)
    w = smooth_weights(c[idx], alpha)
    return weighted_gradient_indices(x, idx, w)


def slsqp_smooth(h, n, alpha, idx, maxiter):
    h0 = project_feasible(h)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_obj, 'jac': _sum_jac, 'args': (n,)}]
        res = minimize(
            _smooth_obj_idx,
            h0,
            args=(n, idx, alpha),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            jac=_smooth_jac_idx,
            options={'maxiter': int(maxiter), 'ftol': 1e-12, 'disp': False}
        )
        if res.x is None:
            return h0
        return project_feasible(res.x)
    except Exception:
        return h0


def select_top_idx(c, count):
    count = int(count)
    if count <= 0:
        return np.empty(0, dtype=int)
    count = min(count, len(c))
    return np.argsort(c)[-count:]


def smooth_descent(h, deadline, alphas, max_iters=3000):
    best = project_feasible(np.asarray(h, dtype=float).copy())
    best_c = compute_c5(best)
    carry = best.copy()
    carry_c = best_c
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        hcur = carry.copy()
        f, g, c = evaluate_smooth(hcur, alpha)
        exact0 = float(np.max(c))
        no_improve = 0
        for _ in range(int(max_iters)):
            if time.time() > deadline - 3:
                break
            if exact0 > best_c + 1e-9:
                break
            gnorm = float(np.linalg.norm(g))
            if gnorm < 1e-12:
                break
            step = min(100.0, 0.05 / max(gnorm, 1e-14))
            step = max(step, 1e-6)
            accepted = False
            for _ in range(12):
                trial = project_feasible(hcur - step * g)
                ct = compute_lags(trial)
                ft = smooth_value(ct, alpha)
                et = float(np.max(ct))
                if (ft <= f - 1e-14 and et <= exact0 + 1e-11) or et < exact0 - 1e-12:
                    hcur = trial
                    c = ct
                    f = ft
                    exact0 = et
                    accepted = True
                    break
                step *= 0.5
            if not accepted:
                break
            if exact0 < best_c - 1e-13:
                best_c = exact0
                best = hcur.copy()
                no_improve = 0
            else:
                no_improve += 1
            f, g, c = evaluate_smooth(hcur, alpha)
            exact0 = float(np.max(c))
            if no_improve > 350:
                break
        if exact0 <= carry_c + 1e-10:
            carry = hcur
            carry_c = exact0
        else:
            carry = best.copy()
            carry_c = best_c
    return best, best_c

                    if s == t:
                        continue
                    delta = min(d, h[s], 1.0 - h[t]) * 0.5
                    if delta < 1e-10:
                        continue
                    trial = h.copy()
                    trial[s] -= delta
                    trial[t] += delta
                    nc = compute_c5(trial)
                    if nc < best_c - 1e-13:
                        h = trial
                        best_c = nc
                        improved = True
                        break
    return h, best_c


def resample_h(h, ndx):
    n = len(h)
    if ndx == n:
        return project_feasible(h)
    xs = np.linspace(0.0, 2.0, n)
    xd = np.linspace(0.0, 2.0, ndx)
    y = np.interp(xd, xs, h)
    return project_feasible(y)


def improve_variants(h):
    cands = [h, 1.0 - h, h[::-1], 1.0 - h[::-1]]
    n = len(h)
    if n >= 5:
        sm = np.convolve(h, [0.25, 0.5, 0.25], mode='same')
        cands.append(sm)
        cands.append(1.0 - sm)
    best = None
    bc = float('inf')
    for v in cands:
        vv = project_feasible(np.asarray(v, dtype=float).copy())
        cc = compute_c5(vv)
        if best is None or cc < bc:
            best = vv
            bc = cc
    return best, bc


def optimize_point(h, deadline, use_slsqp=False, light=False):
    n = len(h)
    h = project_feasible(np.asarray(h, dtype=float).copy())
    c = compute_c5(h)
    if light:
        alphas = [2000, 10000, 50000]
        max_iters = 1100
    else:
        alphas = [800, 2500, 10000, 50000]
        max_iters = 2200
    if time.time() < deadline - 5:
        h2, c2 = smooth_descent(h, deadline, alphas, max_iters)
        if c2 < c - 1e-14:
            h, c = h2, c2
    if use_slsqp:
        if time.time() < deadline - (80 if not light else 45):
            cg = compute_lags(h)
            idx = select_top_idx(cg, 30)
            mi = (70 if not light else 25) if n <= 380 else (45 if not light else 15)
            h2 = slsqp_smooth(h, n, 35000, idx, mi)
            c2 = compute_c5(h2)
            if c2 < c - 1e-14:
                h, c = h2, c2
        if time.time() < deadline - (55 if not light else 30):
            mi = (30 if not light else 12) if n <= 380 else (18 if not light else 8)
            h2 = slsqp_smooth(h, n, 15000, None, mi)
            c2 = compute_c5(h2)
            if c2 < c - 1e-14:
                h, c = h2, c2
    if time.time() < deadline - 2:
        h2, c2 = local_refine(h, deadline)
        if c2 < c - 1e-14:
            h, c = h2, c2
    return h, c


def make_noisy(h, rng, scale):
    n = len(h)
    x = np.linspace(0.0, 2.0, n)
    f = int(1 + rng.randint(0, 5))
    phase = float(rng.uniform(0.0, 2.0 * np.pi))
    noise = scale * np.sin(np.pi * f * x + phase)
    noise = noise + 0.25 * scale * rng.normal(0.0, 1.0, n)
    return project_feasible(h + noise)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(280, 0.5)

    best_h, best_c = improve_variants(h_init)
    n_best = len(best_h)

    remaining = deadline - t0
    base_alloc = min(450.0, max(90.0, 0.34 * remaining))
    base_deadline = t0 + base_alloc
    best_h, best_c = optimize_point(best_h, base_deadline, use_slsqp=True, light=False)
    n_best = len(best_h)

    h2 = project_feasible(1.0 - best_h)
    c2 = compute_c5(h2)
    if c2 < best_c - 1e-14:
        best_h, best_c = h2, c2
    h2 = project_feasible(best_h[::-1])
    c2 = compute_c5(h2)
    if c2 < best_c - 1e-14:
        best_h, best_c = h2, c2
    n_best = len(best_h)

    if time.time() < deadline - 120:
        raw = []
        for off in (-80, -60, -40, -20, 20, 40, 80):
            raw.append(len(best_h) + off)
        for m in (160, 200, 240, 280, 320, 360, 420):
            raw.append(m)
        seen = set()
        scored = []
        for m in raw:
            if m in seen:
                continue
            seen.add(m)
            if 140 <= m <= 900 and m % 2 == 0 and m != len(best_h):
                scored.append((abs(m - len(best_h)), m, m))
        scored.sort()
        cand_ns = [x[2] for x in scored]
        idx_c = 0
        while idx_c < len(cand_ns) and time.time() < deadline - 75:
            nn = cand_ns[idx_c]
            idx_c += 1
            left = deadline - time.time()
            per = (left - 55.0) / max(1, len(cand_ns) - idx_c)
            alloc = max(25.0, min(85.0, per))
            if time.time() + alloc > deadline - 40:
                alloc = max(18.0, deadline - 45.0 - time.time())
            if alloc < 18:
                break
            h_int = resample_h(best_h, nn)
            h2, c2 = optimize_point(h_int, time.time() + alloc, use_slsqp=False, light=True)
            if c2 < best_c - 1e-14 or (abs(c2 - best_c) <= 1e-14 and nn < len(best_h)):
                best_h, best_c = h2, c2

    if time.time() < deadline - 110:
        scales = [0.018, 0.04, 0.075]
        for sc in scales:
            if time.time() > deadline - 60:
                break
            hp = make_noisy(best_h, rng, sc)
            alloc = min(45.0, max(20.0, (deadline - time.time() - 45.0)))
            h2, c2 = optimize_point(hp, time.time() + alloc, use_slsqp=False, light=True)
            if c2 < best_c - 1e-14:
                best_h, best_c = h2, c2

    h2 = project_feasible(1.0 - best_h)
    c2 = compute_c5(h2)
    if c2 < best_c - 1e-14:
        best_h, best_c = h2, c2
    h2 = project_feasible(best_h[::-1])
    c2 = compute_c5(h2)
    if c2 < best_c - 1e-14:
        best_h, best_c = h2, c2

    if time.time() < deadline - 35:
        n = len(best_h)
        cg = compute_lags(best_h)
        idx = select_top_idx(cg, 30)
        h2 = slsqp_smooth(best_h, n, 100000, idx, 28 if n <= 380 else 16)
        c2 = compute_c5(h2)
        if c2 < best_c - 1e-14:
            best_h, best_c = h2, c2

    if time.time() < deadline - 8:
        h2, c2 = smooth_descent(best_h, deadline - 0.5, [60000, 200000], 1400)
        if c2 < best_c - 1e-14:
            best_h, best_c = h2, c2

    if time.time() < deadline - 1:
        h2, c2 = local_refine(best_h, deadline)
        if c2 < best_c - 1e-14:
            best_h, best_c = h2, c2

    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
```
