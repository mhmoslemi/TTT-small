Better approach: **Pairwise coordinate descent**
- Pick two coordinates j and k
- Optimize (h[j], h[k]) jointly subject to h[j] + h[k] = h[j]_old + h[k]_old

- The objective becomes max_m (c_m + (t-h[j]_old)*a_m_j + (s-t-h[k]_old)*a_m_k)

  = max_m (b_m + t * d_m) where b_m and d_m depend on the pair

This is again a 1D convex optimization! And it automatically preserves the sum constraint.

The algorithm:
1. Compute correlation
2. For each pair (j, k), find optimal t
3. The best pair gives the largest improvement
4. Repeat

def pairwise_descent(h, n, max_sweeps, deadline, seed=42):
    dx = 2.0 / n
    h = h.copy()
    rng = np.random.RandomState(seed)
    k_vals = np.arange(2*n-1) - (n-1)
    
    current_c = np.correlate(h, 1-h, mode='full').copy()
    best_c = current_c.max() * dx
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        # Precompute a_j for all j (derivative of correlation w.r.t. h[j])
        # a_j[m] = (1-h[j-k]) - h[j+k] for valid indices
        # This is O(n²) total
        
        improved = False
        # Try random pairs
        n_one = np.where(h > 0.5)[0]
        n_zero = np.where(h < 0.5)[0]
        
        if len(n_one) == 0 or len(n_zero) == 0:
            # All same value, no improvement possible
            break
        
        # Try a subset of pairs
        n_pairs = min(1000, len(n_one) * len(n_zero))
        pairs = []
        for _ in range(n_pairs):
            j = rng.choice(n_one)
            k = rng.choice(n_zero)
            pairs.append((j, k))
        
        best_improvement = 0
        best_pair = None
        best_t = None
        
        for (j, k) in pairs:
            if j == k:
                continue
            hj = h[j]
            hk = h[k]
            s = hj + hk  # must be preserved
            
            # a_j[m] = derivative of c[m] w.r.t. h[j]
            # a_k[m] = derivative of c[m] w.r.t. h[k]
            
            # For coordinate j:
            idx1_j = j - k_vals
            valid1_j = (idx1_j >= 0) & (idx1_j < n)
            a_j = np.zeros(2*n-1)
            a_j[valid1_j] += (1 - h[idx1_j[valid1_j]])
            idx2_j = j + k_vals
            valid2_j = (idx2_j >= 0) & (idx2_j < n)
            a_j[valid2_j] -= h[idx2_j[valid2_j]]
            
            # For coordinate k:
            idx1_k = k - k_vals
            valid1_k = (idx1_k >= 0) & (idx1_k < n)
            a_k = np.zeros(2*n-1)
            a_k[valid1_k] += (1 - h[idx1_k[valid1_k]])
            idx2_k = k + k_vals
            valid2_k = (idx2_k >= 0) & (idx2_k < n)
            a_k[valid2_k] -= h[idx2_k[valid2_k]]
            
            # New h[j] = t, h[k] = s - t
            # Change in c[m] = (t-hj)*a_j[m] + (s-t-hk)*a_k[m] = (t-hj)*a_j[m] + (hj+t-hk-hk+... 
            # Wait: s - t - hk = hj + hk - t - hk = hj - t
            # So change = (t-hj)*a_j[m] + (hj-t)*a_k[m] = (t-hj)*(a_j[m] - a_k[m])
            
            d = a_j - a_k  # direction
            # New c[m] = current_c[m] + (t-hj)*d[m]
            # Minimize max_m (current_c[m] + (t-hj)*d[m]) over t in [max(0,s-1), min(1,s)]
            
            # Let u = t - hj, so u in [max(0,s-1)-hj, min(1,s)-hj]
            # = [max(0, hk-1), min(1, hk+hj)-hj]
            # = [max(0, hk-1), min(1-hj, hk)]
            # Since 0≤hj≤1 and 0≤hk≤1: s-1 = hj+hk-1, and min(1,s) = min(1, hj+hk)
            # u_min = max(0, s-1) - hj = max(0, hk-1) = 0 (since hk≤1)
            # u_max = min(1, s) - hj = min(1-hj, hk)
            # So u in [0, min(1-hj, hk)]
            
            u_max = min(1.0 - hj, hk)
            if u_max < 1e-12:
                continue
            
            # Minimize max_m (current_c[m] + u*d[m]) over u in [0, u_max]
            # This is 1D convex optimization
            
            # Try u=0, u=u_max, and the crossing of top-2
            c0 = current_c
            c_max = current_c + u_max * d
            im0 = np.argmax(c0)
            im_mx = np.argmax(c_max)
            
            best_u = 0.0
            best_val = c0.max()
            
            v_max = c_max.max()
            if v_max < best_val:
                best_val = v_max
                best_u = u_max
            
            if im0 != im_mx and abs(d[im0] - d[im_mx]) > 1e-12:
                u_cross = (c0[im_mx] - c0[im0]) / (d[im0] - d[im_mx])
                if 0 < u_cross < u_max:
                    v_cross = (c0 + u_cross * d).max()
                    if v_cross < best_val:
                        best_val = v_cross
                        best_u = u_cross
            
            improvement = c0.max() - best_val
            if improvement > best_improvement:
                best_improvement = improvement
                best_pair = (j, k)
                best_t = hj + best_u
        
        if best_pair is not None and best_improvement > 1e-15:
            j, k = best_pair
            s = h[j] + h[k]
            h[j] = best_t
            h[k] = s - best_t
            h = np.clip(h, 0, 1)
            current_c = np.correlate(h, 1-h, mode='full')
            if current_c.max() * dx < best_c:
                best_c = current_c.max() * dx
            improved = True
        
        if not improved:
            break
    
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=100000):

        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:

                       constraints=cons,

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        h = h + diff / n
    return np.clip(h, 0.0, 1.0)

def compute_A(h, n):
    L = 2 * n - 1
    k_vals = np.arange(L) - (n - 1)
    A = np.zeros((n, L))
    for j in range(n):
        i1 = j - k_vals
        v1 = (i1 >= 0) & (i1 < n)
        A[j, v1] += (1.0 - h[i1[v1]])
        i2 = j + k_vals
        v2 = (i2 >= 0) & (i2 < n)
        A[j, v2] -= h[i2[v2]]
    return A

def find_best_u(c, d, u_max):
    if u_max < 1e-12:
        return 0.0, c.max()
    c0max = c.max()
    c_up = c + u_max * d
    up_val = c_up.max()
    best_u = 0.0
    best_val = c0max
    if up_val < best_val - 1e-14:

        best_u = u_max
    im0 = np.argmax(c)
    imu = np.argmax(c_up)
    if im0 != imu and abs(d[im0] - d[imu]) > 1e-12:
        uc = (c[imu] - c[im0]) / (d[im0] - d[imu])
        if 0 < uc < u_max:
            vc = (c + uc * d).max()
            if vc < best_val - 1e-14:
                best_val = vc
                best_u = uc
    return best_u, best_val

            idx = np.arange(n)
            rng.shuffle(idx)
            ones = idx[:n // 2]

        for _ in range(n_try):

            hk = h[k]

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_round(h, n):
    target = n // 2
    hb = np.zeros(n)
    hb[np.argsort(h)[::-1][:target]] = 1.0
    return hb
