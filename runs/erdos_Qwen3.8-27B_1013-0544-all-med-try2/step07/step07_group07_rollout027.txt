```python
def compute_all_lags(h, n):

def project_feasible(h, n):
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = np.maximum(1.0 - h, 1e-15)
            s = room.sum()
            if s < 1e-15: break
            h += diff * room / s
        else:
            mass = np.maximum(h, 1e-15)
            s = mass.sum()
            if s < 1e-15: break
            h -= abs(diff) * mass / s
    return np.clip(h, 0.0, 1.0)

def _lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m >= n: return g
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n-m] -= h[m:]
        g[m:] += 1.0 - h[:n-m]
    else:
        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]
    return g

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0/n)
    m = c.max()
    s = alpha * (c - m)
    if s.max() > 500:
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sum_fun(x, n):
    return float(np.sum(x) - n/2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=5000):
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=[(0,1)]*n,

                      options={'maxiter':maxiter,'ftol':1e-16})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def lp_step(h_curr, n, trust, ncuts=150):
    dx = 2.0/n
    c = compute_all_lags(h_curr, n)
    idx = np.argsort(c)[::-1][:ncuts]
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [cp.sum(hv)==n/2.0, hv>=0, hv<=1, cp.norm(hv-h_curr,2)<=trust]
    for k in idx:
        lag = k - (n-1)
        g = _lag_grad(h_curr, n, lag)
        cons.append(g @ hv <= tv + g @ h_curr - c[k])
    p = cp.Problem(cp.Minimize(tv), cons)
    for solver in [cp.GLOP, cp.CBC]:
        try:
            p.solve(solver=solver)
            break
        except:
            continue
    if p.status not in ['optimal','optimal_inaccurate'] or hv.value is None:
        return h_curr.copy()
    return project_feasible(hv.value, n)

def lp_loop(h, n, deadline, maxit=40):

            trust *= 0.5
            if trust < 1e-5: break

            h_c = h_n
            trust = min(0.8, trust * 1.15)
    return best_h, best_c

def fast_ls(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)

                        else:
                            h[j] += delta; h[i] -= delta
    return h, best_c
```

```python
def focused_ls(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0/n
    for d in [0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() > deadline: break
        for _ in range(200):  # max sweeps
            if time.time() > deadline: break
            c = np.correlate(h, 1-h, mode='full')
            k_star = int(np.argmax(c))
            lag = k_star - (n-1)
            m = abs(lag)
            if m == 0: break
            # Find top contributing pairs
            if lag > 0:
                contrib = h[m:] * (1 - h[:n-m])  # h[a+m]*(1-h[a])
            else:
                contrib = h[:n-m] * (1 - h[m:])  # h[a]*(1-h[a+m])
            # Try modifying top pairs
            top_pairs = np.argsort(contrib)[::-1][:50]
            improved = False
            for idx in top_pairs:
                if time.time() > deadline: break
                if lag > 0:
                    a, b = idx, idx + m
                    # Want to reduce h[b]*(1-h[a]): decrease h[b] or increase h[a]
                    if h[b] > 0.01:
                        delta = min(d, h[b]*0.3, 1.0-h[a])
                        if delta < 1e-9: continue
                        h[b] -= delta; h[a] += delta
                        cn = compute_c5(h, n)
                        if cn < best_c - 1e-14:
                            best_c = cn; improved = True
                        else:
                            h[b] += delta; h[a] -= delta
                else:
                    a, b = idx, idx + m
                    # Want to reduce h[a]*(1-h[b]): decrease h[a] or increase h[b]
                    if h[a] > 0.01:
                        delta = min(d, h[a]*0.3, 1.0-h[b])
                        if delta < 1e-9: continue
                        h[a] -= delta; h[b] += delta
                        cn = compute_c5(h, n)
                        if cn < best_c - 1e-14:
                            best_c = cn; improved = True
                        else:
                            h[a] += delta; h[b] -= delta
            if not improved: break
    return h, best_c
```

Actually wait, there's a subtlety. When lag > 0, the correlation at that lag is:

To reduce R, we want to decrease h[a+m] or increase h[a] for pairs with large contribution. The "contribution" of pair (a, a+m) is h[a+m]*(1-h[a]).

If we decrease h[b] (where b=a+m) and increase h[a], the contribution of this specific pair decreases. But it might increase contributions to OTHER lags. That's why we check the full C5.

But also, moving mass from b to a might help or hurt other lags. The focused approach only looks at the current max lag, which is a greedy strategy. It should work well in practice.
