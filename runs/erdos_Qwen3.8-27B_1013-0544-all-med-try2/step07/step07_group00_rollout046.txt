import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo = h.min() - 0.5
    hi = h.max() + 0.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, n, alpha, top_k=50):
    c = compute_c5_vec(h, n)
    m = c.max()
    nc = len(c)
    if top_k < nc:
        idx = np.argsort(c)[-top_k:]
        mask = np.zeros(nc, dtype=bool)
        mask[idx] = True
    else:
        mask = np.ones(nc, dtype=bool)
    sh = np.full(nc, -np.inf)
    sh[mask] = alpha * (c[mask] - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w /= wsum
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(nc):
        if w[mi] < 1e-15:
            continue
        k = mi - (n - 1)
        wm = w[mi] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:k + nv])
            grad[k:] += -wm * h[:nv]
        else:
            kp = -k
            nv = n - kp
            grad[kp:] += wm * (1.0 - h[:nv])
            grad[:nv] += -wm * h[kp:]
    return grad

def nesterov_proximal(h, n, deadline, alpha0=200, top_k0=30, momentum0=0.9):
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    h_prev = h.copy()
    mu = momentum0
    alpha = alpha0
    top_k = top_k0
    step = 0.005
    for it in range(20000):
        if time.time() > deadline:
            break
        h_y = h_cur + mu * (h_cur - h_prev)
        g = smooth_max_grad(h_y, n, alpha, top_k)
        h_new = project_feasible(h_y - step * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h_prev = h_cur.copy()
            h_cur = h_new
            if it % 500 == 0 and alpha < 500000:
                alpha *= 1.3
                top_k = min(2 * n - 1, top_k + 5)
                step *= 0.95
        else:
            step *= 0.8
            if step < 1e-6:
                step = 0.002
                h_prev = h_cur.copy()
    return h_cur, best_c

def slsqp_solve(h, n, deadline, alpha=500000, maxiter=5000):
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    def eq(x):
        return float(x.sum() - n / 2.0)
    def eq_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def lbfgs_solve(h, n, deadline, alpha=5000, mu=1.0, maxiter=2000):
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        val += mu * (x.sum() - n / 2.0) ** 2
        return float(val)
    def grad(x):
        g = smooth_max_grad(x, n, alpha, top_k=2 * n - 1)
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-17, 'gtol': 1e-14,
                                'maxfun': maxiter * 20, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def pair_polish_analytic(h, n, deadline, step=0.001):
    best_c = compute_c5(h, n)
    c = compute_c5_vec(h, n)
    # Find active lags
    top_lags = np.argsort(c)[-20:]
    step_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for st in step_list:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_c5_vec(h, n)
            for lag in top_lags:
                if time.time() > deadline:
                    return h, best_c
                k = int(lag) - (n - 1)
                # Compute gradient for this specific lag
                if k >= 0:
                    nv = n - k
                    g_up = (1.0 - h[k:k + nv])
                    g_dn = -h[:nv].copy()
                    grad_k = np.zeros(n)
                    grad_k[:nv] += g_up
                    grad_k[k:] += g_dn
                else:
                    kp = -k
                    nv = n - kp
                    grad_k = np.zeros(n)
                    grad_k[kp:] += (1.0 - h[:nv])
                    grad_k[:nv] += -h[kp:]
                # Find best pair: increase where grad_k is most negative, decrease where most positive
                order = np.argsort(grad_k)
                for ii in range(min(5, n)):
                    i = order[ii]
                    if grad_k[i] >= 0:
                        break
                    for jj in range(n - 1, max(n - 6, n // 2), -1):
                        j = order[jj]
                        if grad_k[j] <= 0:
                            break
                        # Try h[i] += st, h[j] -= st
                        if h[i] + st <= 1.0 and h[j] - st >= 0.0:
                            h[i] += st
                            h[j] -= st
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                improved = True
                            else:
                                h[i] -= st
                                h[j] += st
                        if h[i] - st >= 0.0 and h[j] + st <= 1.0:
                            h[i] -= st
                            h[j] += st
                            c_new = compute_c5(h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                improved = True
                            else:
                                h[i] += st
                                h[j] -= st
    return h, best_c

def random_start(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def golden_start(n):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.floor(np.arange(n // 2) * phi).astype(int) % n
    h[positions] = 1.0
    return project_feasible(h, n)

def fourier_start(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for f in range(1, 8):
        h += rng.uniform(0.15, 0.3) * np.cos(2 * np.pi * f * x / 2.0 + rng.uniform(0, 2 * np.pi))
    return project_feasible(h, n)
