        best_h, c_tmp = slsqp_optimize(best_h, 300, min(time.time() + 60, deadline - 470),
                                        alpha=200000, maxiter=800)
        if c_tmp < best_c:
            best_c = c_tmp

    # Phase 2: Pair search on n=300
    if time.time() < deadline - 400:
        best_h, best_c = grad_pair_search(best_h, 300, min(time.time() + 80, deadline - 390),
                                           step=0.003)
        if time.time() < deadline - 380:
            best_h, best_c = local_pair_search(best_h, 300, min(time.time() + 60, deadline - 370),
                                                deltas=[0.005, 0.002, 0.001, 0.0005], radius=40)

    # Phase 3: Upsample to n=600 and optimize
    if time.time() < deadline - 350:
        h600 = upsample(best_h, 300, 600)
        c600 = compute_c5(h600, 600)
        t3 = min(time.time() + 300, deadline - 100)
        h600, c600 = lbfgs_optimize(h600, 600, t3,
                                      alphas=[200, 1000, 5000, 20000, 100000, 500000],
                                      max_iters=150)
        if time.time() < deadline - 80:
            h600, c600 = slsqp_optimize(h600, 600, min(time.time() + 50, deadline - 70),
                                         alpha=500000, maxiter=600)
        if time.time() < deadline - 60:
            h600, c600 = grad_pair_search(h600, 600, min(time.time() + 50, deadline - 50),
                                           step=0.002, alphas=[500, 5000, 50000])
        if c600 < best_c:
            best_c = c600
            best_h = h600
            best_n = 600

    # Phase 4: Try n=400 as intermediate
    if time.time() < deadline - 80 and best_n == 300:
        h400 = upsample(best_h, 300, 400)
        t4 = min(time.time() + 60, deadline - 40)
        h400, c400 = lbfgs_optimize(h400, 400, t4,
                                      alphas=[500, 5000, 50000], max_iters=150)
        if time.time() < deadline - 30:
            h400, c400 = grad_pair_search(h400, 400, min(time.time() + 30, deadline - 20),
                                           step=0.002, alphas=[1000, 10000])
        if c400 < best_c:
            best_c = c400
            best_h = h400
            best_n = 400

    # Phase 5: Final refinement on best
    if time.time() < deadline - 5:
        if time.time() < deadline - 3:
            best_h, best_c = grad_pair_search(best_h, best_n, deadline - 2,
                                               step=0.001, alphas=[5000, 50000])

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
