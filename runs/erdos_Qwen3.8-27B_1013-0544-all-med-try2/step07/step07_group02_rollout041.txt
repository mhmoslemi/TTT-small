```python
def compute_gradient(h, n, alpha):
    """Compute gradient of smoothed max correlation objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx
    m = g.max()
    shifted = alpha * (g - m)
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    
    # Now compute gradient
    # w[k] is the softmax weight for the k-th entry of the full correlation
    # Full correlation entry k corresponds to lag = k - (n-1)
    
    grad = np.zeros(n)
    for j in range(n):
        # For each lag l = -(n-1)..(n-1), weight is w[l + n - 1]
        # ∂corr(lag)/∂h[j] = (1-h[j+lag])*[0≤j+lag<n] - h[j-lag]*[0≤j-lag<n]
        total = 0.0
        for lag in range(-(n-1), n):
            wk = w[lag + n - 1]
            if wk < 1e-15:
                continue
            jp = j + lag
            jm = j - lag
            if 0 <= jp < n:
                total += wk * (1.0 - h[jp])
            if 0 <= jm < n:
                total -= wk * h[jm]
        grad[j] = dx * total
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_onto_box_sum(y, n):
    """Euclidean projection onto {h: 0<=h<=1, sum(h)=n/2} via bisection on lambda."""
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-13:
            break
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def smoothed_objective(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    max_s = shifted.max()
    return m + (shifted.max() - max_s + np.log(np.exp(shifted - max_s).sum())) / alpha

def smoothed_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx
    m = g.max()
    shifted = alpha * (g - m)
    max_s = shifted.max()
    w = np.exp(shifted - max_s)
    w /= w.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for j in range(n):
        # T1: sum w[k] * (1-h[j+lag]) where lag = k-(n-1), valid if 0<=j+lag<n
        # j+lag = j+k-(n-1), valid range: (n-1)-j <= k <= n-1+j, clipped to [0, 2n-2]
        klo = max(0, (n-1) - j)
        khi = min(2*n-2, (n-1) + j)
        idx = np.arange(klo, khi+1)
        h_pos = j + idx - (n-1)
        grad[j] += w[idx].dot(1.0 - h[h_pos])
        
        # T2: sum w[k] * h[j-lag] where j-lag = j-k+(n-1), valid if 0<=j-lag<n
        # j-k+(n-1) in [0,n-1]: k in [j-(n-1), j+n-1], clipped to [0, 2n-2]
        klo2 = max(0, j - (n-1))
        khi2 = min(2*n-2, j + n - 1)
        idx2 = np.arange(klo2, khi2+1)
        h_pos2 = j - idx2 + (n-1)
        grad[j] -= w[idx2].dot(h[h_pos2])
    
    grad *= dx
    return grad

def projected_gradient(h, n, alpha, max_iters, deadline):
    """Run projected gradient method with backtracking line search."""
    h = h.copy()
    f_curr = smoothed_objective(h, n, alpha)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad = smoothed_gradient(h, n, alpha)
        
        # Backtracking line search
        step = 1.0
        improved = False
        for _ in range(50):
            h_try = h - step * grad
            h_try = project_onto_box_sum(h_try, n)
            f_try = smoothed_objective(h_try, n, alpha)
            if f_try < f_curr - 1e-12:
                h = h_try
                f_curr = f_try
                improved = True
                break
            step *= 0.5
        if not improved:
            break
    
    return h

def binary_greedy(h_init, n, deadline, max_sweeps=200):
    """Binary greedy: swap 1s and 0s to reduce max correlation."""
    dx = 2.0 / n
    # Convert to binary
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:n//2]] = 1.0
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    m_range = np.arange(-(n-1), n)
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros):
            best_j = -1
            best_nc = cur_c
            for j in list(ones):
                # Compute change in correlation if we swap h[i]=1, h[j]=0
                delta = np.zeros(2*n-1)
                # Effect of setting h[i]=1 (was 0):
                # corr changes by: (1-h[i+lag]) for valid i+lag, minus h[i-lag] for valid i-lag
                # Wait, this is the same as the gradient computation for a single unit change
                for lag in range(-(n-1), n):
                    k = lag + n - 1
                    jp = i + lag
                    jm = i - lag
                    if 0 <= jp < n:
                        delta[k] += (1.0 - h[jp])
                    if 0 <= jm < n:
                        delta[k] -= h[jm]
                # Effect of setting h[j]=0 (was 1): negation of above for position j
                for lag in range(-(n-1), n):
                    k = lag + n - 1
                    jp = j + lag
                    jm = j - lag
                    if 0 <= jp < n:
                        delta[k] -= (1.0 - h[jp])
                    if 0 <= jm < n:
                        delta[k] += h[jm]
                
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i] = 1.0
                h[best_j] = 0.0
                corr += delta
                ones.remove(best_j)
                zeros.remove(i)
                ones.add(i)
                zeros.add(best_j)
                cur_c = best_nc
                improved = True
                break
        if not improved:
            break
    
    return h, cur_c
```

    sh = alpha * (corr - m)
    ms = sh.max()
    return m + (ms - ms + np.log(np.exp(sh - ms).sum())) / alpha

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx
    m = g.max()
    sh = alpha * (g - m)
    ms = sh.max()
    w = np.exp(sh - ms)
    w /= w.sum()
    
    grad = np.zeros(n)
    for j in range(n):
        klo = max(0, (n-1) - j)
        khi = min(2*n-2, (n-1) + j)
        idx = np.arange(klo, khi+1)
        grad[j] += w[idx].dot(1.0 - h[j + idx - (n-1)])
        
        klo2 = max(0, j - (n-1))
        khi2 = min(2*n-2, j + n - 1)
        idx2 = np.arange(klo2, khi2+1)
        grad[j] -= w[idx2].dot(h[j - idx2 + (n-1)])
    
    return grad * dx

def proj_grad(h, n, alpha, max_iters, deadline):
    h = h.copy()
    f_curr = smooth_obj(h, n, alpha)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad = smooth_grad(h, n, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (alpha + 1)  # initial step size
        for _ in range(40):
            h_try = project_onto_box_sum(h - step * grad, n)
            f_try = smooth_obj(h_try, n, alpha)
            if f_try < f_curr - 1e-14:
                h = h_try
                f_curr = f_try
                break
            step *= 0.5
        else:
            break
    return h
