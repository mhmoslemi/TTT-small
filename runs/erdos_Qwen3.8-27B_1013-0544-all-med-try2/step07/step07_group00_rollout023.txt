def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr <= 1e-15:
                break
            h += (d / tr) * room
        else:
            tr = h.sum()
            if tr <= 1e-15:
                break
            h += (d / tr) * h
        h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.argsort(1.0 - h)[::-1]
            for i in idx:
                if d <= 1e-15: break
                a = min(1.0 - h[i], d)
                h[i] += a; d -= a

            idx = np.argsort(h)
            for i in idx:
                if d >= -1e-15: break
                a = min(h[i], -d)
                h[i] -= a; d += a

import numpy as np
from scipy.optimize import minimize
import time

_deadline_global = 0.0


def _callback(x):
    return time.time() > _deadline_global


def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr <= 1e-15:
                break
            h += (d / tr) * room
        else:
            tr = h.sum()
            if tr <= 1e-15:
                break
            h += (d / tr) * h
        h = np.clip(h, 0.0, 1.0)

    d = target - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.argsort(1.0 - h)[::-1]
            for i in idx:
                if d <= 1e-15:
                    break
                a = min(1.0 - h[i], d)
                h[i] += a
                d -= a
        else:
            idx = np.argsort(h)
            for i in idx:
                if d >= -1e-15:
                    break
                a = min(h[i], -d)
                h[i] -= a
                d += a
    return np.clip(h, 0.0, 1.0)


def c5_vec(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def c5(h, n):
    return float(np.max(c5_vec(h, n)))


def eps_for_c(c):
    if c.size <= 1:
        return 1e-5
    k = int(0.95 * (c.size - 1))
    return max(1e-5, float(c.max() - np.partition(c, k)[k]))


def smooth_objective(x, n, alpha, mu, active):
    c = c5_vec(x, n)
    arr = c if active is None else c[active]
    b = float(arr.max())
    z = alpha * (arr - b)
    z = z - z.max()
    val = b + np.log(np.sum(np.exp(z))) / alpha
    if mu:
        val += mu * (x.sum() - 0.5 * n) ** 2
    return float(val)


def smooth_gradient(x, n, alpha, mu, active):
    c = c5_vec(x, n)
    arr = c if active is None else c[active]
    b = float(arr.max())
    z = alpha * (arr - b)
    z = z - z.max()
    w = np.exp(z)
    w /= w.sum()

    g = np.zeros(n)
    dx = 2.0 / n
    m = n - 1
    idx_iter = range(len(c)) if active is None else active

    for idx, wi in zip(idx_iter, w):
        if wi <= 1e-14:
            continue
        k = idx - m
        wm = wi * dx
        if k == 0:
            g += wm * (1.0 - 2.0 * x)
        elif k > 0:
            nv = n - k
            g[k:] += wm * (1.0 - x[:nv])
            g[:nv] -= wm * x[k:]
        else:
            d = -k
            nv = n - d
            g[:nv] += wm * (1.0 - x[d:])
            g[d:] -= wm * x[:nv]

    if mu:
        g += 2.0 * mu * (x.sum() - 0.5 * n)
    return g


def _eqfun(x, n):
    return float(x.sum() - 0.5 * n)


def _eqjac(x, n):
    return np.ones_like(x)


def choose_active(c, topk, eps, max_count):
    m = len(c)
    topk = min(max(1, topk), m)
    max_count = min(max_count, m)

    order = np.argsort(c)[::-1]
    act = order[:topk]
    inside = np.where(c >= c.max() - eps)[0]
    if inside.size > topk:
        inside = inside[np.argsort(c[inside])[::-1][:topk]]
    act = np.unique(np.concatenate((act, inside)))

    if act.size < max_count:
        step = max(1, m // max(1, max_count - act.size))
        act = np.unique(np.concatenate((act, np.arange(0, m, step))))

    if act.size > max_count:
        act = act[np.argsort(c[act])[::-1][:max_count]]
    return act


def lbfgs_run(h, n, alpha, mu, active, maxiter, run_deadline, maxfun):
    global _deadline_global
    h = np.asarray(h, dtype=float)
    if not np.all(np.isfinite(h)):
        return h, c5(h, n)

    _deadline_global = run_deadline
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            smooth_objective,
            h,
            args=(n, alpha, mu, active),
            method="L-BFGS-B",
            jac=smooth_gradient,
            bounds=bounds,
            callback=_callback,
            options={"maxiter": maxiter, "maxfun": maxfun,
                     "ftol": 1e-16, "gtol": 1e-14, "maxls": 30},
        )
    except Exception:
        return h, c5(h, n)

    if res is None or not np.all(np.isfinite(res.x)):
        return h, c5(h, n)

    hp = project_feasible(res.x, n)
    return hp, c5(hp, n)


def slsqp_run(h, n, alpha, active, maxiter):
    h = np.asarray(h, dtype=float)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _eqfun, "jac": _eqjac, "args": (n,)}]
    try:
        res = minimize(
            smooth_objective,
            h,
            args=(n, alpha, 0.0, active),
            method="SLSQP",
            jac=smooth_gradient,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-12},
        )
    except Exception:
        return h, c5(h, n)

    if res is None or not np.all(np.isfinite(res.x)):
        return h, c5(h, n)

    hp = project_feasible(res.x, n)
    return hp, c5(hp, n)

def interpolate_h(h, no, nn):
    if nn == no:
        return h.copy()
    xo = np.linspace(0.0, 2.0, no, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    return np.interp(xn, xo, h)


def best_transform(h, n):
    h0 = project_feasible(h, n)
    best = h0.copy()
    bc = c5(h0, n)
    bases = [h0, 1.0 - h0, h0[::-1], 1.0 - h0[::-1]]
    shifts = (0, 1, 2, 3, 5, 10, 25, 50, 100, 125)
    for base in bases:
        for s in shifts:
            cand = np.roll(base, s)
            cand = project_feasible(cand, n)
            c = c5(cand, n)
            if c + 1e-14 < bc:
                bc = c
                best = cand.copy()
    return best, bc


def default_start(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in (1, 2, 3):
        h += 0.2 * np.cos(np.pi * k * x)
    return project_feasible(h, n)

        hf, cc = slsqp_run(best_h, best_n, 50000.0, act, 45 if best_n <= 600 else 25)
        if cc + 1e-14 < best_c:
            best_h, best_c = hf, cc

    for nt in (400, 300, 600, 750):
        if time.time() > deadline - 70.0:
            break
        nn = max(80, min(int(nt), int(best_n * 1.6) + 1))
        if nn == best_n:
            continue
        hi = project_feasible(interpolate_h(best_h, best_n, nn), nn)
        alloc = min(70.0, (deadline - time.time() - 40.0) * 0.45)
        if alloc < 25.0:
            continue
        hh, cc = active_phase(hi, nn, time.time() + alloc, rng)
        if cc + 1e-14 < best_c:
            hh, cc = best_transform(hh, nn)
            if cc + 1e-14 < best_c:
                best_h, best_c, best_n = hh, cc, nn

    if time.time() < deadline - 20.0:
        best_h, best_c = best_transform(best_h, best_n)

    if time.time() < deadline - 2.0:
        cvec = c5_vec(best_h, best_n)
        act = choose_active(cvec, 40, eps_for_c(cvec), 60)
        hf, cc = slsqp_run(best_h, best_n, 100000.0, act, 20)
        if cc + 1e-14 < best_c:
            best_h, best_c = hf, cc

    best_h = project_feasible(best_h, best_n)
    best_c = c5(best_h, best_n)
    return best_h, best_c, int(best_n)
