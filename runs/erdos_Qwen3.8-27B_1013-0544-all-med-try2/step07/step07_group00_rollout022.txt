import numpy as np
from scipy.optimize import minimize
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def fft_conv(A, B, N):
    return np.fft.irfft(np.fft.rfft(A, N) * np.fft.rfft(B, N), N)


def corr_raw(h, N):
    n = len(h)
    return fft_conv(h[::-1], 1.0 - h, N)[:2 * n - 1]


def c5_exact(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(h, n):
    x = np.asarray(h, dtype=float)
    T = 0.5 * n
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(120):
        d = T - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (d / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (d / s) * h
    for _ in range(8):
        d = T - h.sum()
        if d == 0.0:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if not mask.any():
            break
        idx = int(np.flatnonzero(mask)[0])
        h[idx] += d
        if h[idx] < 0.0 or h[idx] > 1.0:
            break
    return np.clip(h, 0.0, 1.0)


def obj_penalty(x, n, N1, dx, alpha, mu):
    if not np.all(np.isfinite(x)):
        return 1e100
    c = corr_raw(x, N1) * dx
    m = c.max()
    z = alpha * (c - m)
    mx = z.max()
    e = np.exp(z - mx)
    s = x.sum() - 0.5 * n
    return float(m + (mx + np.log(e.sum())) / alpha + mu * s * s)


def grad_from_weights(B, x, n, N2):
    center = n - 1
    t1 = fft_conv(B[::-1], 1.0 - x, N2)
    t3 = fft_conv(B, x, N2)
    return t1[center:center + n] - t3[center:center + n]


def grad_penalty(x, n, N1, N2, dx, alpha, mu):
    if not np.all(np.isfinite(x)):
        return np.zeros(n)
    c = corr_raw(x, N1) * dx
    m = c.max()
    z = alpha * (c - m)
    mx = z.max()
    e = np.exp(z - mx)
    w = e / e.sum() * dx
    g = grad_from_weights(w, x, n, N2)
    g += 2.0 * mu * (x.sum() - 0.5 * n)
    return g


def optimize_lbfgs(h0, n, deadline):
    cur_h = project_feasible(np.asarray(h0, dtype=float).copy(), n)
    best_h = cur_h.copy()
    cur_c = best_c = c5_exact(cur_h)
    if time.time() > deadline - 15.0:
        return best_h, best_c, cur_h

    dx = 2.0 / n
    N1 = next_pow2(2 * n - 1)
    N2 = next_pow2(3 * n - 2)
    bounds = [(0.0, 1.0) for _ in range(n)]
    cfgs = [
        (20.0, 1.0), (80.0, 1.0), (300.0, 5.0), (1200.0, 10.0),
        (5000.0, 30.0), (20000.0, 100.0), (80000.0, 500.0),
        (300000.0, 2000.0)
    ]
    cap = 700 if n <= 600 else 350
    est = 0.005 if n <= 600 else 0.012

    for alpha, mu in cfgs:
        rem = deadline - time.time()
        if rem < 20.0:
            break
        maxfun = int(min(cap, max(30, rem / (est * 4.0))))
        if maxfun < 30:
            break
        try:
            res = minimize(
                obj_penalty, cur_h, jac=grad_penalty,
                args=(n, N1, N2, dx, alpha, mu),
                method="L-BFGS-B", bounds=bounds,
                options={
                    "maxiter": maxfun, "maxfun": maxfun,
                    "ftol": 1e-16, "gtol": 1e-12,
                    "maxcor": 30, "maxls": 30
                }
            )
        except Exception:
            continue
        if not hasattr(res, "x") or not np.all(np.isfinite(res.x)):
            continue
        h2 = project_feasible(res.x, n)
        c2 = c5_exact(h2)
        if not np.isfinite(c2):
            continue
        if c2 < best_c:
            best_h = h2.copy()
            best_c = float(c2)
        if c2 <= cur_c + 0.005:
            cur_h = h2
            cur_c = float(c2)
    return best_h, best_c, cur_h


def resample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(np.asarray(h, dtype=float).copy(), n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, left=float(h[0]), right=float(h[-1]))
    return project_feasible(h_new, n_new)


def golden_start(n):
    h = np.zeros(n)
    half = n // 2
    phi = 0.5 * (1.0 + np.sqrt(5.0))
    idx = (np.arange(half) * phi).astype(int) % n
    h[idx] = 1.0
    return project_feasible(h, n)


def lse_weights(c, dx, alpha):
    m = c.max()
    z = alpha * (c - m)
    z -= z.max()
    e = np.exp(z)
    s = e.sum()
    if s <= 0.0:
        e = np.full_like(c, 1.0)
        s = e.sum()
    return e / s * dx


def coord_effect(h, r, n):
    L = 2 * n - 1
    center = n - 1
    e = np.zeros(L)
    s1 = center - r
    s2 = r
    e[s1:s1 + n] += 1.0 - h
    e[s2:s2 + n] += -h[::-1]
    return e


def make_effects(h, idxs, n):
    d = {}
    for r in idxs:
        rr = int(r)
        d[rr] = coord_effect(h, rr, n)
    return d


def pair_try(h, c, eplus, eminus, src, dst, center, n, best_raw):
    if src == dst:
        return None
    maxd = min(float(h[src]), 1.0 - float(h[dst]))
    if maxd <= 1e-10:
        return None
    s = eplus - eminus
    raw = (maxd, 0.5 * maxd, 0.25 * maxd, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001)
    cands = []
    for d in raw:
        if d > 1e-9 and d <= maxd + 1e-12:
            if not cands or abs(d - cands[-1]) > 1e-12:
                cands.append(d)

    idx1 = center + (src - dst)
    idx2 = center + (dst - src)

    bd = 0.0
    bmax = best_raw
    for d in cands:
        y = c + d * s
        d2 = d * d
        y[center] -= 2.0 * d2
        y[idx1] += d2
        y[idx2] += d2
        ym = float(np.max(y))
        if ym < bmax:
            bmax = ym
            bd = d

    if bd <= 0.0 or bmax >= best_raw - 1e-11:
        return None

    bd2 = bd * bd
    c += bd * s
    c[center] -= 2.0 * bd2
    c[idx1] += bd2
    c[idx2] += bd2
    h[dst] = min(1.0, float(h[dst]) + bd)
    h[src] = max(0.0, float(h[src]) - bd)
    return float(np.max(c))


def pair_polish(h, n, deadline):
    dx = 2.0 / n
    center = n - 1
    N1 = next_pow2(2 * n - 1)
    N2 = next_pow2(3 * n - 2)
    c = corr_raw(h, N1)
    best_raw = float(np.max(c))
    M = min(40, max(4, n // 10))
    alpha = 20000.0
    acc = 0
    while time.time() < deadline - 2.0:
        if deadline - time.time() < 4.0:
            break
        B = lse_weights(c, dx, alpha)
        g = grad_from_weights(B, h, n, N2)
        order = np.argsort(g)
        low = order[:M]
        high = order[-M:][::-1]
        effs = make_effects(h, np.unique(np.concatenate((low, high))), n)
        improved = False
        for i in high:
            ii = int(i)
            if h[ii] <= 1e-11:
                continue
            gi = float(g[ii])
            for j in low:
                if time.time() > deadline - 2.0:
                    break
                jj = int(j)
                if ii == jj:
                    continue
                gj = float(g[jj])
                if gj - gi > -1e-9:
                    break
                if h[jj] >= 1.0 - 1e-11:
                    continue
                newm = pair_try(h, c, effs[jj], effs[ii], ii, jj, center, n, best_raw)
                if newm is not None:
                    best_raw = newm
                    acc += 1
                    if acc >= 20:
                        c = corr_raw(h, N1)
                        best_raw = float(np.max(c))
                        acc = 0
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h
