Also, I just realized: the `np.correlate` function computes the correlation of the full arrays. For `mode='full'`, the output length is 2n-1. The value at index m corresponds to the sum:
Σ_i h[i] * (1-h)[i + m - (n-1)]

where the sum is over all i where both indices are valid.

So for m = n-1 (the center), it's Σ_i h[i] * (1-h)[i] = Σ_i h[i] - Σ_i h[i]².
This is the "zero-shift" correlation, which equals ∫h(x)(1-h(x))dx.

For the max to be small, we need ALL shifts to have small correlation. The zero-shift is Σ h[i](1-h[i]) which is maximized when h[i] = 1/2 (giving 1/4 per element, total n/4 * dx = 2/4 = 0.5) and minimized when h[i] ∈ {0,1} (giving 0 per element).

import numpy as np
from scipy.optimize import minimize
import time

def c5val(h, n):
    return float(np.max(np.correlate(h, 1 - h, 'full')) * 2.0 / n)

def c5vec(h, n):
    return np.correlate(h, 1 - h, 'full') * 2.0 / n

def proj(h, n):
    h = np.clip(h.copy(), 0, 1)
    t = n * 0.5
    for _ in range(200):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (d / s) * h
    return np.clip(h, 0, 1)

def sgrad(h, n, a):
    c = c5vec(h, n)
    s = a * (c - c.max())
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    w *= 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m]
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])
    return g

def fobj(x, n, a, mu):
    c = c5vec(x, n)
    m = c.max()
    s = a * (c - m)
    mx = s.max()
    return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / a + mu * (x.sum() - n * 0.5) ** 2)

def fjac(x, n, a, mu):
    g = sgrad(x, n, a)
    g += 2.0 * mu * (x.sum() - n * 0.5)
    return g

def slsqp_fun(x, n):
    return float(x.sum() - n * 0.5)

def slsqp_jac(x, n):
    return np.ones(n)

def run_lbfgs(h, n, dl, mi=200):
    h = proj(h, n)
    bc = c5val(h, n)
    bd = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    for a in alphas:
        if time.time() > dl - 2:
            break
        try:
            r = minimize(fobj, h, jac=fjac, args=(n, a, 1e-4),
                         method='L-BFGS-B', bounds=bd,
                         options={'maxiter': mi, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': mi * 5})
            hn = proj(r.x, n)
            cn = c5val(hn, n)
            if cn < bc - 1e-15:
                bc, h = cn, hn
        except Exception:
            pass
    return h, bc

def run_slsqp(h, n, dl, a=100000, mi=1500):
    bd = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_fun, 'jac': slsqp_jac, 'args': (n,)}]
    try:
        r = minimize(fobj, h, args=(n, a, 1e-10), method='SLSQP',
                     bounds=bd, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-14})
        hn = proj(r.x, n)
        cn = c5val(hn, n)
        return hn, cn
    except Exception:
        return h.copy(), c5val(h, n)

def grad_pair(h, n, dl, step=0.003, radius=80):
    h = proj(h, n)
    bc = c5val(h, n)
    for a in [500, 2000, 10000]:
        while time.time() < dl:
            imp = False
            g = sgrad(h, n, a)
            for i in range(n):
                if time.time() > dl:
                    return h, bc
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    diff = g[i] - g[j]
                    if abs(diff) < 1e-10:
                        continue
                    mv = step * np.sign(diff)
                    vi = h[i] - mv
                    vj = h[j] + mv
                    if 0 <= vi <= 1 and 0 <= vj <= 1:
                        h[i], h[j] = vi, vj
                        cn = c5val(h, n)
                        if cn < bc - 1e-14:
                            bc, imp = cn, True
                        else:
                            h[i], h[j] = vi + mv, vj - mv
            if not imp:
                break
    return h, bc

def pair_search(h, n, dl, deltas, radius=50):

                            h[i] += d
