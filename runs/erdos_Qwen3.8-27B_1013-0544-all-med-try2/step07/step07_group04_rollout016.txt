import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_sum(a, target):
    """Project a onto {0<=a<=1, sum(a)=target} using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(a - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(a - (lo + hi) / 2, 0, 1)

def lse_grad(h, alpha, dx):
    """Compute gradient of LSE-smoothed max correlation."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = corr.max()
    exp_c = np.exp(alpha * (corr - m))
    w = exp_c / exp_c.sum()  # softmax weights
    grad = np.zeros(n)
    # For each k, shift = k - n + 1
    for k in range(2 * n - 1):
        if w[k] < 1e-8:
            continue
        shift = k - n + 1
        wk = alpha * w[k]
        # Term 1: (1 - h[j+shift]) for j where 0<=j+shift<n
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 + shift:hi1 + shift])
        # Term 2: -h[j-shift] for j where 0<=j-shift<n
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * h[lo2 - shift:hi2 - shift]
    return grad * dx, float((m + np.log(exp_c.sum()) / alpha) * dx)

def nesterov_lse(h, n, alphas, n_iter, deadline, init_lr=0.01, momentum=0.9):
    """Nesterov-like accelerated LSE descent."""
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    h_prev = h.copy()
    
    for alpha in alphas:
        n_steps = n_iter
        for t in range(n_steps):
            if time.time() > deadline:
                return best_h, best_c
            # Nesterov look-ahead
            h_nest = h_curr + momentum * (h_curr - h_prev)
            h_nest = np.clip(h_nest, 0, 1)
            
            # Compute gradient at Nesterov point
            grad, obj = lse_grad(h_nest, alpha, dx)
            
            # Adaptive step size
            lr = init_lr / (1 + t * 0.01)
            
            # Update
            h_prev = h_curr.copy()
            h_curr = h_nest - lr * grad
            h_curr = project_sum(h_curr, target)
            
            # Check actual objective
            if t % 50 == 0:
                c = compute_c5(h_curr, n)
                if c < best_c:
                    best_c = c
                    best_h = h_curr.copy()
    
    c = compute_c5(h_curr, n)
    if c < best_c:
        best_c = c
        best_h = h_curr.copy()
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_search_guided(h_init, n, deadline, max_sweeps=500, seed=42):
    """Binary local search with guided pair selection."""
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()
    
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_list = sorted(zeros)
        for i in z_list:
            if time.time() > deadline:
                break
            o_list = sorted(ones)
            for j in o_list:
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                delta = np.zeros(2 * n - 1)
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_sum(a, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(a - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(a - (lo + hi) / 2, 0, 1)

def lse_grad(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = corr.max()
    exp_c = np.exp(alpha * (corr - m))
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-8:
            continue
        shift = k - n + 1
        wk = alpha * w[k]
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 + shift:hi1 + shift])
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * h[lo2 - shift:hi2 - shift]
    obj = (m + np.log(exp_c.sum()) / alpha) * dx
    return grad * dx, obj

def nesterov_lse(h, n, alpha, n_iter, deadline, init_lr=0.3, momentum=0.85):
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    h_prev = h.copy()
    for t in range(n_iter):
        if time.time() > deadline:
            break
        h_nest = h_curr + momentum * (h_curr - h_prev)
        h_nest = np.clip(h_nest, 0, 1)
        grad, obj = lse_grad(h_nest, alpha, dx)
        lr = init_lr / (1.0 + t * 0.005 + alpha * 1e-5)
        h_prev = h_curr.copy()
        h_curr = h_nest - lr * grad
        h_curr = project_sum(h_curr, target)
        if t % 100 == 0:
            c = compute_c5(h_curr, n)
            if c < best_c:
                best_c = c
                best_h = h_curr.copy()
    c = compute_c5(h_curr, n)
    if c < best_c:
        best_c = c
        best_h = h_curr.copy()
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h_init, n, deadline, max_sweeps=500, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in sorted(zeros):
            if time.time() > deadline:
                break
            for j in sorted(ones):
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                delta = np.zeros(2 * n - 1)
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def coordinate_descent(h, n, deadline, n_passes=50):
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    for p in range(n_passes):
        if time.time() > deadline:
            break
        changed = False
        for j in range(n):
            if time.time() > deadline:
                break
            save = h[j]
            alpha_vals = np.zeros(2 * n - 1)
            beta_vals = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                shift = k - n + 1
                a = 1.0 if 0 <= j + shift < n else 0.0
                b = 1.0 if 0 <= j - shift < n else 0.0
                hjs = h[j + shift] if 0 <= j + shift < n else 0.0
                hjs2 = h[j - shift] if 0 <= j - shift < n else 0.0
                alpha_vals[k] = a * (1.0 - hjs) - b * hjs2
                if a == 0 and b == 0:
                    beta_vals[k] = 0.0
                else:
                    beta_vals[k] = 0.0
            h[j] = 0.0
            corr0 = np.correlate(h, 1 - h, mode='full')
            h[j] = save
            for k in range(2 * n - 1):
                shift = k - n + 1
                a = 1.0 if 0 <= j + shift < n else 0.0
                b = 1.0 if 0 <= j - shift < n else 0.0
                hjs = h[j + shift] if 0 <= j + shift < n else 0.0
                hjs2 = h[j - shift] if 0 <= j - shift < n else 0.0
                alpha_vals[k] = a * (1.0 - hjs) - b * hjs2
                beta_vals[k] = corr0[k]
            candidates = [0.0, 1.0, save]
            for k1 in range(2 * n - 1):
                for k2 in range(k1 + 1, 2 * n - 1):
                    denom = alpha_vals[k1] - alpha_vals[k2]
                    if abs(denom) > 1e-12:
                        tc = (beta_vals[k2] - beta_vals[k1]) / denom
                        if 0.0 <= tc <= 1.0:
                            candidates.append(tc)
            best_v = save
            best_val = np.inf
            for v in candidates:
                v = np.clip(v, 0.0, 1.0)
                h[j] = v
                val = np.max(beta_vals + alpha_vals * v) * dx
                if val < best_val:
                    best_val = val
                    best_v = v
            h[j] = best_v
            if abs(save - best_v) > 1e-10:
                changed = True
        if not changed:
            break
    return project_feasible(h, n)

        h2 = slsqp_smooth(h2, n, 500000, 2000, dl - 10)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
    if time.time() < dl - 3:
        hb, cb = binary_local_search(best_h, n, dl, max_sweeps=400, seed=seed + 7)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
