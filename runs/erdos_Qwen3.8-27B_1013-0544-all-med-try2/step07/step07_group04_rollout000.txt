        if lag > 0:

            vals[a] = pref[L] - np.dot(hc[:L], hc[lag:])
        elif lag < 0:
            r = -lag; L = n - r
            vals[a] = (total - pref[r]) - np.dot(hc[r:], hc[:L])
        else:
            vals[a] = total - np.dot(hc, hc)
    return vals

def active_ineq(v):
    n = _AS['n']
    t = float(v[n])
    return t - _AS['dx'] * active_c_values(v[:n], _AS['lags'])

def active_ineq_jac(v):
    n = _AS['n']; lags = _AS['lags']; m = len(lags)
    J = np.zeros((m, n + 1)); J[:, n] = 1.0
    dx = _AS['dx']
    hc = np.clip(v[:n], 0.0, 1.0)
    for a, lag in enumerate(lags):
        gc = np.zeros(n)
        if lag > 0:
            L = n - lag
            gc[:L] = 1.0 - hc[lag:]
            gc[lag:] -= hc[:L]
        elif lag < 0:
            r = -lag; L = n - r
            gc[r:] = 1.0 - hc[:L]
            gc[:L] -= hc[r:]
        else:
            gc = 1.0 - 2.0 * hc
        J[a, :n] = -dx * gc
    return J

def active_polish(h, n, deadline, rng, K0=None):
    target = 0.5 * n
    dx = 2.0 / n
    h = project_box_sum(h, target)
    cvec = compute_c_fft(h)
    best_c = float(cvec.max()) * dx
    best_h = h.copy()
    cur_h = h.copy(); cur_cvec = cvec.copy()
    cur_c = best_c
    if K0 is None:
        K0 = min(25, max(8, n // 6))
    K = K0
    stall = 0
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': active_eq, 'jac': active_eq_jac},
            {'type': 'ineq', 'fun': active_ineq, 'jac': active_ineq_jac}]
    while True:
        rem = deadline - time.time()
        if rem < 35.0:
            break
        cur_c = float(cur_cvec.max()) * dx
        if cur_c < best_c - 1e-12:
            best_c = cur_c; best_h = cur_h.copy()
        lags = select_lags(cur_cvec, n, K)
        _AS['n'] = n; _AS['target'] = target; _AS['lags'] = lags; _AS['dx'] = dx
        x0 = np.empty(n + 1); x0[:n] = cur_h; x0[n] = cur_c + 1e-7
        ineq = active_ineq(x0)
        if ineq.min() < -1e-8:
            x0[n] += -float(ineq.min()) + 1e-9
        maxit = 160 if rem > 240 else (100 if rem > 120 else 50)
        try:
            res = minimize(active_objective, x0, jac=active_objective_jac, method='SLSQP',
                           bounds=bounds, constraints=cons, options={'maxiter': maxit, 'ftol': 1e-11})
        except Exception:
            break
        if not np.all(np.isfinite(res.x)):
            break
        h2 = project_box_sum(res.x[:n], target)
        c2 = compute_c_fft(h2)
        C2 = float(c2.max()) * dx
        if C2 < best_c - 1e-11:
            best_c = C2; best_h = h2.copy(); stall = 0
        else:
            stall += 1
        if C2 <= best_c + 1e-5:
            cur_h = h2; cur_cvec = c2; cur_c = C2
        else:
            cur_h = best_h.copy(); cur_cvec = compute_c_fft(best_h); cur_c = best_c
        if stall >= 3:
            if K < 55:
                K = min(55, K + 8)
            cur_h = project_box_sum(cur_h + 0.02 * rng.randn(n), target)
            cur_cvec = compute_c_fft(cur_h)
            cur_c = float(cur_cvec.max()) * dx
            stall = 0
    return best_h, best_c

def resample(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, n_from, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n_to, endpoint=False)
    return project_box_sum(np.interp(x1, x0, h), 0.5 * n_to)

def base_starts(n, rng):
    out = []
    h = np.zeros(n)
    idx = rng.permutation(n)
    k = n // 2
    h[idx[:k]] = 1.0
    if n % 2:
        h[idx[k]] = 0.5
    out.append(h)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    for f in (3, 5, 8, 13):
        hh = 0.5 + 0.49 * np.sin(2.0 * np.pi * f * x + rng.uniform(0.0, 6.28))
        out.append(project_box_sum(hh, 0.5 * n))
    return out

import numpy as np
import time
from scipy.optimize import minimize

_AS = {'n': 0, 'target': 0.0, 'lags': (), 'dx': 1.0}

def project_box_sum(y, s):
    y = np.asarray(y, dtype=float)
    if y.size == 0:
        return y
    lo = y.min() - 1.0 - 1e-12
    hi = y.max() + 1.0 + 1e-12
    mid = 0.5 * (lo + hi)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(y - mid, 0.0, 1.0)
        if h.sum() > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - mid, 0.0, 1.0)
    diff = s - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0.0:
            room = 1.0 - h
            free = room > 1e-12
            if free.any():
                cap = room[free].sum()
                if cap > 1e-15:
                    h[free] += min(diff, cap) * room[free] / cap
        else:
            free = h > 1e-12
            if free.any():
                cap = h[free].sum()
                if cap > 1e-15:
                    h[free] -= min(-diff, cap) * h[free] / cap
    return np.clip(h, 0.0, 1.0)

def fft_len(n):
    N = 1
    while N < 2 * n:
        N <<= 1
    return N

def compute_c_fft(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = h.size
    c = np.empty(2 * n - 1)
    N = fft_len(n)
    hp = np.zeros(N)
    hp[:n] = h
    H = np.fft.rfft(hp)
    auto = np.fft.irfft(H * H.conj(), n=N)
    pref = np.empty(n + 1)
    pref[0] = 0.0
    pref[1:] = np.cumsum(h)
    total = pref[-1]
    center = n - 1
    pos = np.arange(n)
    c[center + pos] = pref[n - pos] - auto[pos]
    neg = np.arange(1, n)
    c[center - neg] = (total - pref[neg]) - auto[N - neg]
    return c

def select_lags(c, n, K):
    K = min(K, int(c.size))
    order = np.argsort(c)[::-1][:K]
    center = n - 1
    return [int(p - center) for p in order]

def active_objective(v):
    return float(v[_AS['n']])

def active_objective_jac(v):
    n = _AS['n']
    j = np.zeros(n + 1)
    j[-1] = 1.0
    return j

def active_eq(v):
    return float(np.sum(v[:_AS['n']]) - _AS['target'])

def active_eq_jac(v):
    n = _AS['n']
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def active_c_values(h, lags):
    n = h.size
    hc = np.clip(h, 0.0, 1.0)
    vals = np.empty(len(lags))
    pref = np.empty(n + 1)
    pref[0] = 0.0
    pref[1:] = np.cumsum(hc)
    total = pref[-1]
    for a, lag in enumerate(lags):
        if lag > 0:
            L = n - lag
            vals[a] = pref[L] - np.dot(hc[:L], hc[lag:])
        elif lag < 0:
            r = -lag
            L = n - r
            vals[a] = (total - pref[r]) - np.dot(hc[r:], hc[:L])
        else:
            vals[a] = total - np.dot(hc, hc)
    return vals

def active_ineq(v):
    n = _AS['n']
    t = float(v[n])
    return t - _AS['dx'] * active_c_values(v[:n], _AS['lags'])

def active_ineq_jac(v):
    n = _AS['n']
    lags = _AS['lags']
    m = len(lags)
    J = np.zeros((m, n + 1))
    J[:, n] = 1.0
    dx = _AS['dx']
    hc = np.clip(v[:n], 0.0, 1.0)
    for a, lag in enumerate(lags):
        gc = np.zeros(n)
        if lag > 0:
            L = n - lag
            gc[:L] = 1.0 - hc[lag:]
            gc[lag:] -= hc[:L]
        elif lag < 0:
            r = -lag
            L = n - r
            gc[r:] = 1.0 - hc[:L]
            gc[:L] -= hc[r:]
        else:
            gc = 1.0 - 2.0 * hc
        J[a, :n] = -dx * gc
    return J

def active_polish(h, n, deadline, rng, K0=None):
    target = 0.5 * n
    dx = 2.0 / n
    h = project_box_sum(h, target)
    cvec = compute_c_fft(h)
    best_c = float(cvec.max()) * dx
    best_h = h.copy()
    cur_h = h.copy()
    cur_cvec = cvec.copy()
    cur_c = best_c
    if K0 is None:
        K0 = min(25, max(8, n // 6))
    K = K0
    stall = 0
    mem = set()
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': active_eq, 'jac': active_eq_jac},
        {'type': 'ineq', 'fun': active_ineq, 'jac': active_ineq_jac},
    ]
    while True:
        rem = deadline - time.time()
        if rem < 35.0:
            break
        cur_c = float(cur_cvec.max()) * dx
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = cur_h.copy()
        top = select_lags(cur_cvec, n, K)
        mem.update(top)
        if len(mem) > 90:
            mem = set(top)
        lags = list(top)
        cap = min(60, max(35, K + 20))
        for lag in mem:
            if len(lags) >= cap:
                break
            if lag not in lags:
                lags.append(lag)
        _AS['n'] = n
        _AS['target'] = target
        _AS['lags'] = lags
        _AS['dx'] = dx
        x0 = np.empty(n + 1)
        x0[:n] = cur_h
        x0[n] = cur_c + 1e-7
        ineq = active_ineq(x0)
        if ineq.min() < -1e-8:
            x0[n] += -float(ineq.min()) + 1e-9
        if rem > 400:
            maxit = 180
        elif rem > 200:
            maxit = 110
        elif rem > 90:
            maxit = 55
        else:
            maxit = 25
        try:
            res = minimize(
                active_objective, x0, jac=active_objective_jac,
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': maxit, 'ftol': 1e-11, 'disp': False}
            )
        except Exception:
            break
        if not np.all(np.isfinite(res.x)):
            break
        h2 = project_box_sum(res.x[:n], target)
        c2 = compute_c_fft(h2)
        C2 = float(c2.max()) * dx
        if C2 < best_c - 1e-11:
            best_c = C2
            best_h = h2.copy()
            stall = 0
        else:
            stall += 1
        if C2 <= best_c + 1e-5:
            cur_h = h2
            cur_cvec = c2
            cur_c = C2
        else:
            cur_h = best_h.copy()
            cur_cvec = compute_c_fft(best_h)
            cur_c = best_c
        if stall >= 3:
            if K < 55:
                K = min(55, K + 8)
            cur_h = project_box_sum(cur_h + 0.02 * rng.randn(n), target)
            cur_cvec = compute_c_fft(cur_h)
            cur_c = float(cur_cvec.max()) * dx
            stall = 0
    return best_h, best_c

def resample(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, n_from, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n_to, endpoint=False)
    return project_box_sum(np.interp(x1, x0, h), 0.5 * n_to)

    for nn, frac in phases:
        alloc = frac / total_frac * usable
        if time.time() + 20.0 > t_end:
            break
        deadline = min(t_end - 2.0, time.time() + alloc)

    return best_h, float(c_final), int(best_n)
