def corr_dx(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def c5(h, n):
    c = corr_dx(h, n)
    return float(np.max(c))

def project_box_sum(y, target):
    y = np.array(y, dtype=float, copy=True)
    target = float(target)
    n = y.size
    if n == 0:
        return y
    if (np.all(y >= 0.0) and np.all(y <= 1.0) and abs(y.sum() - target) < 1e-12):
        return y
    lo = float(np.min(y) - 1.0)
    hi = float(np.max(y) + 1.0)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += (diff / s) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(x, n, alpha):
    cd = corr_dx(x, n)
    m = float(np.max(cd))
    z = alpha * (cd - m)
    zm = float(np.max(z))
    ex = np.exp(z - zm)
    return m + float(np.log(ex.sum())) / alpha

def lse_grad(x, n, alpha):
    cd = corr_dx(x, n)
    m = float(np.max(cd))
    z = alpha * (cd - m)
    zm = float(np.max(z))
    ex = np.exp(z - zm)
    w = ex / ex.sum()
    cent = n - 1
    A = np.convolve(1.0 - x, w, mode='full')[cent:cent+n]
    B = np.convolve(x, w[::-1], mode='full')[cent:cent+n]
    return (2.0 / n) * (A - B)

def lse_slsqp(h, n, alpha, maxiter):
    h = project_box_sum(h, 0.5 * n)
    c0 = c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_h, 'jac': sum_jac_h, 'args': (n,)}]
    try:
        res = minimize(lse_obj, h, jac=lse_grad, args=(n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        h1 = project_box_sum(res.x, 0.5 * n)
        c1 = c5(h1, n)
        if c1 < c0 - 1e-13:
            return h1, c1
    except Exception:
        pass
    return h, c0

def c_lags(h, n, active):
    dx = 2.0 / n
    c0 = n - 1
    out = np.empty(len(active), dtype=float)
    one_minus = 1.0 - h
    for r, m in enumerate(active):

            out[r] = dx * np.dot(h[k:], one_minus[:n-k])
        else:
            rneg = -k
            out[r] = dx * np.dot(h[:n-rneg], one_minus[rneg:])

def c_lag_jac(h, n, m):
    dx = 2.0 / n
    c0 = n - 1
    k = int(m) - c0
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n-k]
        g[:n-k] -= h[k:]
    else:
        r = -k
        g[:n-r] += 1.0 - h[r:]
        g[r:] -= h[:n-r]
    return dx * g

def subgrad_refine(h, n, deadline, max_iters=120):
    target = 0.5 * n
    h = project_box_sum(h, target)
    best = c5(h, n)
    hbest = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            return hbest, best
        c = corr_dx(h, n)
        m = float(np.max(c))
        idx = np.where(c >= m - 1e-6)[0]
        if idx.size > 60:
            idx = np.argsort(c)[-60:]
        g = np.zeros(n)
        for mi in idx:
            g += c_lag_jac(h, n, int(mi))
        g = g / max(1, idx.size)
        g -= np.mean(g)
        norm = float(np.linalg.norm(g))
        if norm < 1e-12:
            break
        d = -g / norm
        found = False
        for step in (0.05, 0.02, 0.005, 0.001, 0.0002):
            h1 = project_box_sum(h + step * d, target)
            cm = c5(h1, n)
            if cm < best - 1e-14:
                h = h1
                hbest = h1.copy()
                best = cm
                found = True
                break
        if not found:
            break
    return hbest, best

    dx=2.0/n; p=n-1; om=1.0-h; out=np.empty(len(active))
    for r,m in enumerate(active):
       k=int(m)-p

       else:

def c_lag_jac(h,n,m):

import time
import numpy as np
from scipy.optimize import minimize


def corr_dx(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def c5(h, n):
    return float(np.max(corr_dx(h, n)))


def project_box_sum(y, target):
    y = np.array(y, dtype=float, copy=True)
    target = float(target)
    if y.size == 0:
        return y
    lo = float(np.min(y) - 1.0)
    hi = float(np.max(y) + 1.0)
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += (diff / s) * h
        h = np.clip(h, 0.0, 1.0)
    return h


def sum_eq(x, n):
    return float(np.sum(x[:n]) - 0.5 * n)


def sum_jac(x, n):
    j = np.zeros(len(x))
    if n > 0:
        j[:n] = 1.0
    return j


def lse_obj(x, n, a):
    c = corr_dx(x, n)
    m = float(np.max(c))
    z = a * (c - m)
    z -= float(np.max(z))
    e = np.exp(z)
    return m + float(np.log(e.sum())) / a


def lse_grad(x, n, a):
    c = corr_dx(x, n)
    m = float(np.max(c))
    z = a * (c - m)
    z -= float(np.max(z))
    e = np.exp(z)
    w = e / e.sum()
    p = n - 1
    A = np.convolve(1.0 - x, w, mode='full')[p:p + n]
    B = np.convolve(x, w[::-1], mode='full')[p:p + n]
    return (2.0 / n) * (A - B)


def lse_slsqp(h, n, a, mi):
    h = project_box_sum(h, 0.5 * n)
    c0 = c5(h, n)
    try:
        res = minimize(lse_obj, h, jac=lse_grad, args=(n, a), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac, 'args': (n,)}],
                       options={'maxiter': mi, 'ftol': 1e-10})
        h1 = project_box_sum(res.x, 0.5 * n)
        c1 = c5(h1, n)
        if c1 < c0 - 1e-13:
            return h1, c1
    except Exception:
        pass
    return h, c0


def c_lags(h, n, active):
    dx = 2.0 / n
    p = n - 1
    om = 1.0 - h
    out = np.empty(len(active), dtype=float)
    for r, m in enumerate(active):
        k = int(m) - p
        if k >= 0:
            out[r] = dx * np.dot(h[k:], om[:n - k])
        else:
            kk = -k
            out[r] = dx * np.dot(h[:n - kk], om[kk:])
    return out


def c_lag_jac(h, n, m):
    dx = 2.0 / n
    p = n - 1
    k = int(m) - p
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    else:
        kk = -k
        g[:n - kk] += 1.0 - h[kk:]
        g[kk:] -= h[:n - kk]
    return dx * g


def active_inq(x, n, active):
    return x[n] - c_lags(x[:n], n, active)


def active_jac(x, n, active):
    h = x[:n]
    J = np.zeros((len(active), n + 1))
    for r, m in enumerate(active):
        J[r, :n] -= c_lag_jac(h, n, m)
        J[r, n] = 1.0
    return J


def T_obj(x, n):
    return float(x[n])


def T_jac(x, n):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g


def select_active(cb, cc, base, ma, gap):
    if ma <= 0:
        return []
    if cc is None:
        cc = cb
    cc = np.maximum(cb, cc)
    m = float(np.max(cc))
    top = np.argsort(cc)[-ma:].tolist()
    rel = np.where(cc >= m - gap)[0].tolist()
    cand = set(top)
    cand.update(rel)
    if base is not None:
        cand.update(int(i) for i in base)
    if len(cand) > ma:
        arr = np.fromiter(cand, dtype=int)
        order = np.argsort(-cc[arr])
        cand = arr[order[:ma]].tolist()
    else:
        cand = list(cand)
    cand.sort()
    return cand


def active_refine(h, n, end_time, ma=40, mout=6):
    tgt = 0.5 * n
    h = project_box_sum(h, tgt)
    cb = corr_dx(h, n)
    best = float(np.max(cb))
    act = select_active(cb, None, None, ma, 0.005)
    bd = [(0.0, 1.0)] * n
    for _ in range(mout):
        reserve = 70 if n > 450 else 50
        if time.time() > end_time - reserve:
            break
        rem = end_time - time.time()
        if n <= 350:
            mi = int(min(60, max(25, rem / 6.0)))
        else:
            mi = int(min(35, max(20, rem / 8.0)))
        if mi < 20:
            break
        x0 = np.empty(n + 1)
        x0[:n] = h
        x0[n] = best + 1e-8
        cons = [
            {'type': 'eq', 'fun': sum_eq, 'jac': sum_jac, 'args': (n,)},
            {'type': 'ineq', 'fun': active_inq, 'jac': active_jac, 'args': (n, tuple(act))}
        ]
        try:
            res = minimize(T_obj, x0, jac=T_jac, args=(n,), method='SLSQP',
                           bounds=bd + [(0.0, 2.0)], constraints=cons,
                           options={'maxiter': mi, 'ftol': 1e-10})
            h1 = project_box_sum(res.x[:n], tgt)
            c1 = corr_dx(h1, n)
            cm = float(np.max(c1))
            ct = float(res.x[n])
        except Exception:
            h1 = h.copy()
            c1 = cb.copy()
            cm = best
            ct = best
        imp = cm < best - 1e-13
        if imp:
            h, best, cb = h1, cm, c1
        cc = c1 if cm <= best + 0.02 else None
        cb_sel = cb if imp else corr_dx(h, n)
        nas = select_active(cb_sel, cc, act, ma, 0.002)
        if cc is not None and np.any(c1 - ct > 1e-8):
            tv = np.argsort(c1 - ct)[-min(6, ma):].tolist()
            for i in tv:
                if i not in nas:
                    nas.append(i)
            if len(nas) > ma:
                arr = np.fromiter(nas, dtype=int)
                cc2 = np.maximum(cb_sel, cc)
                order = np.argsort(-cc2[arr])
                nas = arr[order[:ma]].tolist()
                nas.sort()
        if not imp and set(nas) == set(act):
            break
        act = nas
    h = project_box_sum(h, tgt)
    return h, c5(h, n)


def subgrad(h, n, end_time, it=60):
    tgt = 0.5 * n
    h = project_box_sum(h, tgt)
    best = c5(h, n)
    hb = h.copy()
    for _ in range(it):
        if time.time() > end_time:
            return hb, best
        c = corr_dx(h, n)
        m = float(np.max(c))
        idx = np.where(c >= m - 1e-6)[0]
        if idx.size > 50:
            idx = np.argsort(c)[-50:]
        g = np.zeros(n)
        for ii in idx:
            g += c_lag_jac(h, n, int(ii))
        g /= max(1, len(idx))
        g -= float(np.mean(g))
        norm = float(np.linalg.norm(g))
        if norm < 1e-12:
            break
        d = -g / norm
        ok = False
        for s in (0.05, 0.02, 0.005, 0.001, 2e-4):
            h1 = project_box_sum(h + s * d, tgt)
            cm = c5(h1, n)
            if cm < best - 1e-14:
                h, hb, best = h1, h1.copy(), cm
                ok = True
                break
        if not ok:
            break
    return hb, best


def default_h(n, seed=0):
    x = np.linspace(0.0, 2.0, n)
    return np.clip(0.5 + 0.42 * np.sin(3.1 * x) + 0.06 * np.sin(7.7 * x + 0.13 * seed), 0.0, 1.0)


def optimize_n(h, n, end_time, rng=None, ma=None):
    if ma is None:
        ma = 50 if n <= 350 else 30 if n <= 600 else 18
    h = project_box_sum(h, 0.5 * n)
    best = c5(h, n)
    if n <= 700 and time.time() < end_time - 70:
        if n <= 350:
            alphas = (300, 1500, 8000)
        elif n <= 600:
            alphas = (800, 5000)
        else:
            alphas = (2500,)
        for a in alphas:
            if time.time() > end_time - 80:
                break
            rem = end_time - time.time()
            if n <= 350:
                mi = int(min(35, max(18, rem / 6.0)))
            else:
                mi = int(min(25, max(12, rem / 8.0)))
            if mi < 12:
                break
            h, best = lse_slsqp(h, n, a, mi)
    if time.time() < end_time - 70:
        h, best = active_refine(h, n, end_time, ma, 6)
    if n <= 650 and time.time() < end_time - 35:
        a = 20000 if n <= 350 else 10000
        rem = end_time - time.time()
        mi = int(min(30, max(12, rem / 8.0)))
        if mi >= 12:
            h, best = lse_slsqp(h, n, a, mi)
    if n <= 650 and time.time() < end_time - 15:
        h, best = subgrad(h, n, end_time - 2, 60)
    if n <= 350 and time.time() < end_time - 35:
        h, best = active_refine(h, n, end_time, min(70, ma + 10), 3)
    h = project_box_sum(h, 0.5 * n)
    return h, c5(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    db = float(budget_s)
    end_all = t0 + db - 5.0
    rng = np.random.RandomState(seed)

    try:
        hbest = np.array(initial_h_values, dtype=float).copy()
        nbest = len(hbest)
    except Exception:
        nbest = 250
        hbest = default_h(nbest, seed)

    hbest = project_box_sum(hbest, 0.5 * nbest)
    bestc = c5(hbest, nbest)

    base_end = min(end_all - 60.0, t0 + max(50.0, 0.45 * db))
    if base_end > t0 + 25:
        h, c = optimize_n(hbest, nbest, base_end, rng, 55 if nbest <= 350 else 35)
        if c < bestc - 1e-12:
            hbest, bestc, nbest = h, c, nbest

    for add in (50, 150, 300, 500):
        nn = nbest + add
        if nn < 80 or nn >= 950 or nn == len(hbest):
            continue
        rem = end_all - time.time()
        if rem < 170:
            break
        xold = np.linspace(0.0, 2.0, len(hbest))
        xnew = np.linspace(0.0, 2.0, nn)
        hi = np.interp(xnew, xold, hbest)
        alloc = min(260.0, 0.42 * rem)
        end = min(end_all - 70.0, time.time() + alloc)
        if end - time.time() < 50:
            break
        ma = 50 if nn <= 350 else 30 if nn <= 600 else 18
        h, c = optimize_n(hi, nn, end, rng, ma)
        if c < bestc - 1e-12:
            hbest, bestc, nbest = h, c, nn

    if nbest <= 650:
        for r in range(3):
            rem = end_all - time.time()
            if rem < 90:
                break
            hp = project_box_sum(hbest + rng.normal(0.0, 0.02 + 0.05 * r, nbest), 0.5 * nbest)
            alloc = min(120.0, 0.40 * rem)
            end = min(end_all - 50.0, time.time() + alloc)
            if end - time.time() < 40:
                break
            h, c = optimize_n(hp, nbest, end, rng, 45 if nbest <= 350 else 25)
            if c < bestc - 1e-12:
                hbest, bestc = h, c

    if nbest <= 400 and time.time() < end_all - 25:
        h, c = subgrad(hbest, nbest, end_all - 2, 70)
        if c < bestc - 1e-12:
            hbest, bestc = h, c

    if nbest <= 350 and time.time() < end_all - 35:
        h, c = active_refine(hbest, nbest, end_all, 55, 3)
        if c < bestc - 1e-12:
            hbest, bestc = h, c

    hbest = project_box_sum(hbest, 0.5 * nbest)
    bestc = c5(hbest, nbest)
    return hbest, bestc, nbest
