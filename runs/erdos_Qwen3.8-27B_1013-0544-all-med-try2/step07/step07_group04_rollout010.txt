import time
import numpy as np
from scipy.optimize import minimize


def compute_c5_float(h):
    n = len(h)
    hf = np.asarray(h, dtype=float)
    corr = np.correlate(hf, 1.0 - hf, mode="full")
    return float(np.max(corr) * (2.0 / n))


def project_sum(u, target):
    u = np.asarray(u, dtype=float)
    if u.size == 0:
        return u
    target = float(target)
    u = np.clip(u, 0.0, 1.0)
    if abs(u.sum() - target) < 1e-10:
        return u
    lo = float(np.min(u)) - 1.0
    hi = float(np.max(u)) + 1.0
    for _ in range(70):
        lam = 0.5 * (lo + hi)
        v = np.clip(u - lam, 0.0, 1.0)
        if v.sum() > target:
            lo = lam
        else:
            hi = lam
    return np.clip(u - 0.5 * (lo + hi), 0.0, 1.0)


def smooth_values_and_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    a = alpha * f
    m = float(np.max(a))
    e = np.exp(a - m)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        val = 0.0
        p = np.zeros(len(f), dtype=float)
    else:
        val = (m + float(np.log(se))) / alpha
        p = e / se
    grad = np.zeros(n, dtype=float)
    w = p * dx
    act = np.flatnonzero(w > 1e-20)
    for pos in act:
        ww = float(w[pos])
        s = int(pos) - (n - 1)
        if s >= 0:
            hb = h[:n - s]
            ha = h[s:n]
            grad[s:n] += ww * (1.0 - hb)
            grad[:n - s] -= ww * ha
        else:
            r = -s
            ha = h[:n - r]
            hb = h[r:n]
            grad[:n - r] += ww * (1.0 - hb)
            grad[r:n] -= ww * ha
    return float(val), grad


def h_obj(x, alpha, n, mu, target):
    val, grad = smooth_values_and_grad(x, alpha, n)
    diff = float(np.sum(x)) - target
    if diff != 0.0:
        val = val + 0.5 * mu * diff * diff
        grad = grad + mu * diff
    return float(val), grad


def h_penalty_min(h0, n, alpha, maxiter, mu):
    n = int(n)
    target = 0.5 * n
    h = np.clip(np.asarray(h0, dtype=float), 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n
    maxfun = max(2 * maxiter, 20)
    try:
        res = minimize(
            h_obj,
            h,
            args=(alpha, n, mu, target),
            jac=True,
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": maxiter, "maxfun": maxfun, "ftol": 1e-12, "gtol": 1e-6},
        )
        h = np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return project_sum(h, target)


def z_from_h(h, scale, zb):
    hc = np.clip(np.asarray(h, dtype=float), 1e-9, 1.0 - 1e-9)
    z = scale * np.log(hc / (1.0 - hc))
    z = z - float(np.mean(z))
    return np.clip(z, -zb, zb)


def solve_beta(z, scale, target):
    z = np.asarray(z, dtype=float)
    lo = float(np.min(z)) - 45.0 * scale
    hi = float(np.max(z)) + 45.0 * scale
    for _ in range(55):
        beta = 0.5 * (lo + hi)
        y = np.clip((z - beta) / scale, -50.0, 50.0)
        s = 1.0 / (1.0 + np.exp(-y))
        if s.sum() > target:
            lo = beta
        else:
            hi = beta
    return 0.5 * (lo + hi)


def sigmoid_obj(z, alpha, scale, n):
    target = 0.5 * n
    beta = solve_beta(z, scale, target)
    y = np.clip((z - beta) / scale, -50.0, 50.0)
    h = 1.0 / (1.0 + np.exp(-y))
    val, gradh = smooth_values_and_grad(h, alpha, n)
    b = h * (1.0 - h)
    B = float(np.sum(b))
    if B < 1e-18:
        grad = np.zeros_like(z)
    else:
        D = float(np.dot(b, gradh))
        grad = (b / scale) * (gradh - (b / B) * D)
        gm = float(np.mean(grad))
        grad = grad - gm
    return float(val), grad


def sigmoid_minimize(h0, n, scale, alpha, maxiter):
    n = int(n)
    zb = 45.0
    z = z_from_h(h0, scale, zb)
    bounds = [(-zb, zb)] * n
    maxfun = max(2 * maxiter, 20)
    try:
        res = minimize(
            sigmoid_obj,
            z,
            args=(alpha, scale, n),
            jac=True,
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": maxiter, "maxfun": maxfun, "ftol": 1e-12, "gtol": 1e-6},
        )
        z = res.x
    except Exception:
        pass
    beta = solve_beta(z, scale, 0.5 * n)
    y = np.clip((z - beta) / scale, -50.0, 50.0)
    h = 1.0 / (1.0 + np.exp(-y))
    return project_sum(h, 0.5 * n)


def resample_h(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    m = len(h)
    if m == n:
        return project_sum(h, 0.5 * n)
    xo = np.linspace(0.0, 2.0, m, endpoint=False)
    xn = np.linspace(0.0, 2.0, n, endpoint=False)
    y = np.interp(xn, xo, h)
    return project_sum(y, 0.5 * n)


def make_random_seed(n, seed):
    n = int(n)
    rng = np.random.RandomState(int(seed))
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    y = np.zeros(n, dtype=float)
    for _ in range(6):
        freq = rng.uniform(0.5, 3.0)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.25, 1.0)
        y = y + amp * np.sin(2.0 * np.pi * freq * x + ph)
    h = project_sum(0.5 + 0.35 * y, 0.5 * n)
    return h


def bin_from_float(hf, n, rng):
    n = int(n)
    target = n // 2
    key = np.asarray(hf, dtype=float).copy()
    key = key + 1e-12 * rng.random(n)
    idx = np.argsort(key)[::-1]
    h = np.zeros(n, dtype=np.int32)
    h[idx[:target]] = 1
    return h


def bin_max_count(h):
    hh = np.asarray(h, dtype=np.int32)
    return int(np.correlate(hh, 1 - hh, mode="full").max())


def bin_candidates(h, f, top_idx, ones, zeros, rng, batch_size):
    n = len(h)
    cands = []
    seen = set()
    nr = max(20, batch_size // 6)
    for _ in range(nr):
        a = int(ones[rng.randint(len(ones))])
        b = int(zeros[rng.randint(len(zeros))])
        key = a * n + b
        if key not in seen:
            seen.add(key)
            cands.append((a, b))
    space = batch_size - len(cands)
    if space > 0 and len(top_idx) > 0:
        nd = max(1, min(5, space // max(1, len(top_idx))))
        for pos in top_idx:
            if len(cands) >= batch_size:
                break
            pos = int(pos)
            s = pos - (n - 1)
            if s == 0:
                continue
            if s >= 0:
                a_arr = np.arange(s, n)
            else:
                a_arr = np.arange(0, n + s)
            if a_arr.size == 0:
                continue
            b_arr = a_arr - s
            cond = (h[a_arr] == 1) & (h[b_arr] == 0)
            idx = np.flatnonzero(cond)
            if idx.size == 0:
                continue
            m = min(int(idx.size), nd)
            if m == int(idx.size):
                chosen = idx
            else:
                chosen = idx[rng.choice(int(idx.size), m, replace=False)]
            for ii in chosen:
                a = int(a_arr[ii])
                b = int(b_arr[ii])
                key = a * n + b
                if key not in seen:
                    seen.add(key)
                    cands.append((a, b))
    attempts = 0
    while len(cands) < batch_size and attempts < max(20, 5 * batch_size):
        attempts += 1
        a = int(ones[rng.randint(len(ones))])
        b = int(zeros[rng.randint(len(zeros))])
        key = a * n + b
        if key not in seen:
            seen.add(key)
            cands.append((a, b))
    return cands


def binary_search(h_float, n, seed, deadline, batch_size=None):
    n = int(n)
    if batch_size is None:
        if n <= 150:
            batch_size = 350
        elif n <= 300:
            batch_size = 200
        else:
            batch_size = 110
    rng = np.random.RandomState(int(seed))
    h = bin_from_float(h_float, n, rng)
    comp = 1 - h
    cur_count = bin_max_count(h)
    best_h = h.copy()
    best_count = cur_count
    Tc = 1.0
    stagnation = 0

    while time.time() < deadline:
        if cur_count > best_count + 12 or stagnation > 90:
            if rng.random() < 0.4:
                h = best_h.copy()
                cur_count = best_count
            else:
                h = best_h.copy()
                ones = np.flatnonzero(h)
                zeros = np.flatnonzero(h == 0)
                for _ in range(rng.randint(1, 6)):
                    if ones.size == 0 or zeros.size == 0:
                        break
                    a = int(rng.choice(ones))
                    b = int(rng.choice(zeros))
                    h[a], h[b] = h[b], h[a]
            comp = 1 - h
            cur_count = bin_max_count(h)
            stagnation = 0
            Tc = max(0.05, min(3.0, Tc * 1.2))

        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break

        f = np.correlate(h, comp, mode="full")
        maxc = int(f.max())
        cur_count = maxc
        top_count = min(18, f.size)
        if top_count <= 0:
            top_idx = np.zeros(0, dtype=int)
        elif top_count == 1:
            top_idx = np.array([int(np.argmax(f))], dtype=int)
        else:
            top_idx = np.argpartition(-f, top_count - 1)[:top_count].astype(int)

        cands = bin_candidates(h, f, top_idx, ones, zeros, rng, batch_size)
        worsen_limit = 2 + int(Tc)
        if worsen_limit > 5:
            worsen_limit = 5
        best_eval = cur_count + worsen_limit
        best_move = None

        for a, b in cands:
            a = int(a)
            b = int(b)
            if h[a] != 1 or h[b] != 0:
                continue
            h[a] = 0
            h[b] = 1
            comp[a] = 1
            comp[b] = 0
            cint = int(np.correlate(h, comp, mode="full").max())
            h[a] = 1
            h[b] = 0
            comp[a] = 0
            comp[b] = 1
            if cint < best_eval:
                best_eval = cint
                best_move = (a, b)

        if best_move is not None:
            old = cur_count
            if best_eval < cur_count:
                acc = True
            elif best_eval == cur_count:
                acc = rng.random() < (0.5 if stagnation > 30 else 0.2)
            else:
                acc = rng.random() < np.exp(-float(best_eval - cur_count) / max(1e-3, Tc))
            if acc:
                a, b = best_move
                h[a] = 0
                h[b] = 1
                comp[a] = 1
                comp[b] = 0
                cur_count = best_eval
                if cur_count < best_count:
                    best_count = cur_count
                    best_h = h.copy()
                    stagnation = 0
                elif cur_count < old:
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                stagnation += 1
            Tc = max(0.05, min(3.0, Tc * 0.98))
        else:
            stagnation += 5
            Tc = max(0.05, Tc * 0.95)

    return best_h, float(best_count) * (2.0 / n)

        h_final = project_sum(h_best, 0.5 * int(n_best))
        c_final = compute_c5_float(h_final)
        return h_final, c_final, int(n_best)

    except Exception:
        try:
            h = np.asarray(initial_h_values, dtype=float).copy()
            n = int(len(h))
        except Exception:
            n = 250
            h = np.full(n, 0.5)
        h = project_sum(h, 0.5 * n)
        return h, compute_c5_float(h), int(n)
