    #              - h[i+m] if 0<=i<=n-1-m else 0

        c_m = 0.0
        for j in range(n - m):
            c_m += h_prev[j+m] * (1 - h_prev[j])
        c_m *= (2.0 / n)
        
        for i in range(n):
            g = 0.0
            if m <= i:  # h[i] appears as h[j+m] with j=i-m
                g += (1 - h_prev[i-m])
            if i <= n - 1 - m:  # h[i] appears in (1-h[j]) with j=i
                g -= h_prev[i+m]
            A[m, i] = g * (2.0/n)
        
        rhs[m] = t - c_m + A[m] @ h_prev
    
    # QP: min ||h - h_old||^2 s.t. A@h <= rhs, 0<=h<=1, sum(h)=n/2
    h_var = cp.Variable(n)
    obj = cp.Minimize(cp.sum_squares(h_var - h_old))
    constraints = [A @ h_var <= rhs, h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iter=1000, abstol=1e-8, reltol=1e-8)
        if h_var.value is not None:
            return np.clip(h_var.value, 0, 1)
    except:
        pass
    
    # Fallback: projected gradient on the QP
    h = h_old.copy()
    for it in range(max_iter):
        # Gradient of objective: 2*(h - h_old)
        # Plus penalty for constraint violations
        grad = 2.0 * (h - h_old)
        c_check = A @ h - rhs
        active = np.maximum(0, c_check)
        grad += active @ A.T
        h -= 0.5 * grad  # step size
        h = np.clip(h, 0, 1)
        # Project onto sum constraint
        diff = n/2 - h.sum()
        h += diff / n
        h = np.clip(h, 0, 1)
    
    return project_feasible(h, n)

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent_pass(h, n, mu=50.0, iters=3):
    """Multiple passes of coordinate descent with exact 1D minimization."""
    h = h.copy()
    best_c = compute_c5(h, n)
    total_lags = 2 * n - 1
    
    for _ in range(iters):
        improved = False
        for i in range(n):
            # Compute current correlation
            c_all = np.correlate(h, 1 - h, mode='full') * (2.0/n)
            
            # Compute E[k] for coordinate i
            h_old = h[i]
            E = np.zeros(total_lags)
            for k in range(total_lags):
                L = k - (n - 1)
                e = 0.0
                ipL = i + L
                imL = i - L
                if 0 <= ipL < n:
                    e += (1 - h[ipL])
                if 0 <= imL < n:
                    e -= h[imL]
                E[k] = e * (2.0 / n)
            
            sum_change = h.sum() - n/2
            
            # Minimize f(x) = max_k(c_all[k] + (x-h_old)*E[k]) + mu*(sum_change + x - h_old)^2
            # over x in [0,1]
            a, b = 0.0, 1.0
            for _ in range(60):
                m1 = a + (b - a) / 3
                m2 = b - (b - a) / 3
                f1 = (c_all + (m1 - h_old) * E).max() + mu * (sum_change + m1 - h_old)**2
                f2 = (c_all + (m2 - h_old) * E).max() + mu * (sum_change + m2 - h_old)**2
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            x_opt = (a + b) / 2
            
            if abs(x_opt - h_old) > 1e-10:
                h[i] = x_opt
                improved = True
        
        h = project_feasible(h, n)
        c_now = compute_c5(h, n)
        if c_now < best_c:
            best_c = c_now
        elif not improved:
            break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_descent(h, n, mu=100.0, sweeps=5, ternary_iters=60):
    """Coordinate descent with exact 1D minimization."""
    h = h.copy()
    best_c = compute_c5(h, n)
    total_lags = 2 * n - 1
    dx = 2.0 / n
    c_all = np.correlate(h, 1 - h, mode='full') * dx
    L_vec = np.arange(-(n-1), n)
    
    for sweep in range(sweeps):
        for i in range(n):
            h_old = h[i]
            
            # Compute E vectorized
            ipL = i + L_vec
            imL = i - L_vec
            valid_ip = (ipL >= 0) & (ipL < n)
            valid_im = (imL >= 0) & (imL < n)
            E = np.zeros(total_lags)
            if np.any(valid_ip):
                E[valid_ip] += (1.0 - h[ipL[valid_ip]])
            if np.any(valid_im):
                E[valid_im] -= h[imL[valid_im]]
            E *= dx
            
            sum_dev = h.sum() - n / 2.0
            
            # Ternary search for optimal x in [0,1]
            a, b = 0.0, 1.0
            for _ in range(ternary_iters):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = (c_all + (m1 - h_old) * E).max() + mu * (sum_dev + m1 - h_old)**2
                f2 = (c_all + (m2 - h_old) * E).max() + mu * (sum_dev + m2 - h_old)**2
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            x_opt = (a + b) / 2.0
            
            c_all += (x_opt - h_old) * E
            h[i] = x_opt
        
        h = project_feasible(h, n)
        c_all = np.correlate(h, 1 - h, mode='full') * dx
        c_now = compute_c5(h, n)
        if c_now < best_c:
            best_c = c_now
    
    return h, best_c

def fine_pair_search(h, n, deadline, deltas):

    best_c = compute_c5(h, n)
    for d in deltas:

                if time.time() > deadline:
                    break
                for j in range(n):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0 and h[i] + h[j] == h[i] + h[j]:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                if time.time() > deadline:
                    break
            if not improved:
                break
    return h, best_c
