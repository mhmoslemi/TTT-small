        scale = 0.15 if restart == 0 else 0.05 * (restart + 1)
        theta = rng.normal(0, scale, 2 * K)
        lr = 0.002
        b1, b2, eps = 0.9, 0.999, 1e-8
        m_buf = np.zeros(2 * K)
        v_buf = np.zeros(2 * K)
        t = 0
        for alpha in [50, 200, 1000, 5000, 20000, 80000]:
            for _ in range(3000):
                if time.time() > deadline:
                    break
                h = fourier_h(theta, x, K)
                h = np.clip(h, 0, 1)
                c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
                cs = c - c.max()
                obj = c.max() + np.log(np.sum(np.exp(alpha * cs))) / alpha
                # Gradient via finite differences on theta (cheap since 2K dims)
                grad = np.zeros(2 * K)
                fd = 1e-6
                for i in range(2 * K):
                    theta[i] += fd
                    h2 = np.clip(fourier_h(theta, x, K), 0, 1)
                    c2 = np.correlate(h2, 1 - h2, mode='full') * (2.0 / n)
                    cs2 = c2 - c2.max()
                    obj2 = c2.max() + np.log(np.sum(np.exp(alpha * cs2))) / alpha
                    theta[i] -= fd
                    grad[i] = (obj2 - obj) / fd
                t += 1
                m_buf = b1 * m_buf + (1 - b1) * grad
                v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
                m_hat = m_buf / (1 - b1 ** t)
                v_hat = v_buf / (1 - b2 ** t)
                theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
                if t % 2000 == 0:
                    lr *= 0.7
            # Evaluate
            h_final = np.clip(fourier_h(theta, x, K), 0, 1)
            h_final = project_feasible(h_final, n)
            c = compute_c5(h_final)
            if c < best_c:
                best_c = c
                best_h = h_final.copy()
    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius):

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(150, 0.5)
        n_init = 150

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c = compute_c5(h_init)

    # Phase 1: Adam on initial grid
    alloc1 = min(250, (deadline - t0) * 0.25)
    h_opt, c_opt = adam_optimize(h_init, n_init,
                                  [50, 200, 1000, 5000, 20000, 80000],
                                  3000, t0 + alloc1)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    # Phase 2: Fourier parameterization for global search
    if time.time() < deadline - 100:
        for n_try in [n_init, 200, 300]:
            if time.time() > deadline - 60:
                break
            x = np.linspace(0, 2, n_try, endpoint=False)
            alloc_f = min(120, (deadline - time.time()) * 0.2)
            if alloc_f < 30:
                break
            h_f, c_f = fourier_optimize(x, n_try, K=15, deadline=time.time() + alloc_f,
                                        seed=seed + n_try)
            if h_f is not None and c_f < best_c:
                best_c = c_f
                best_h = h_f

    # Phase 3: Try different n via interpolation + Adam
    for n_try in [100, 150, 200, 350]:
        if time.time() > deadline - 80:
            break
        x_src = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_src, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(100, (deadline - time.time()) * 0.18)
        if alloc < 20:
            break
        h_opt, c_opt = adam_optimize(h_interp, n_try,
                                      [200, 1000, 5000, 20000, 80000],
                                      2500, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Multi-restart perturbation
    n = len(best_h)
    if time.time() < deadline - 50:
        remaining = deadline - time.time() - 25
        n_restarts = max(3, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.02 + 0.05 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 15) * 0.3)
            if alloc < 10:
                break
            h_opt, c_opt = adam_optimize(hp, n, [500, 5000, 30000], 2000,
                                          time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Pair search fine-tuning
    n = len(best_h)
    if time.time() < deadline - 8:
        if n <= 160:
            best_h, best_c = pair_search(best_h, n, deadline - 3,
                                         [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 3,
                                              [0.005, 0.002, 0.001, 0.0005, 0.0002],
                                              radius=35)

    # Phase 6: Final high-alpha Adam polish
    if time.time() < deadline - 4:
        n = len(best_h)
        h_opt, c_opt = adam_optimize(best_h, n, [50000, 200000, 500000],
                                      3000, time.time() + (deadline - time.time() - 2))
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final pair search polish
    n = len(best_h)
    if time.time() < deadline - 2 and n <= 160:
        best_h, best_c = pair_search(best_h, n, deadline - 1,
                                     [0.0005, 0.0002, 0.0001, 0.00005])

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
