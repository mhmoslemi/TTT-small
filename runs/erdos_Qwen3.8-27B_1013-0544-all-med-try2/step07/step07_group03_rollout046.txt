import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_obj_and_grad(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    logsum = sm + np.log(exps.sum())
    penalty = lam * (x.sum() - n / 2.0) ** 2
    obj = m + logsum / alpha + penalty
    softmax = exps / exps.sum()
    a = 1.0 - x
    b = x
    rev_b = b[::-1]
    s = softmax
    T1 = np.array([np.dot(a, s[n-1-j:2*n-1-j]) for j in range(n)]) * dx
    T2 = np.array([np.dot(rev_b, s[j:j+n]) for j in range(n)]) * dx
    grad = T1 - T2
    if lam > 0:
        grad += 2.0 * lam * (x.sum() - n / 2.0)
    return obj, grad

def lbfgs_smooth(h, n, alpha, maxiter, ftol, lam_init=0.1, max_dual=50):
    x = h.copy()
    lam = lam_init
    for _ in range(max_dual):

                       method='L-BFGS-B', jac=True,

        lam = max(lam * 2.0, 1e-6)
        if lam > 1e8:
            break
        x = project_feasible(x, n)
    return x, smooth_obj_and_grad(x, n, alpha, 0.0)[0] if False else compute_c5(x, n)

    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])

        improved = False

                break

                if i == j:

                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range

                delta[n - 1] -= 2.0

                    delta[k2] += 1.0

                    corr += delta
