import numpy as np
import time

STRICT_TOL = 1e-11
DIFF_TOL = 1e-12
IMPROVE_TOL = 1e-11
FRACS = (0.5, 0.25, 0.75, 0.9, 0.98)
SCALES = (0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25)


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (d / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (d / s) * h
        h = np.clip(h, 0.0, 1.0)
    e = target - h.sum()
    for _ in range(5):
        if abs(e) < 1e-14:
            break
        idx = int(np.argmax(1.0 - h)) if e > 0.0 else int(np.argmax(h))
        h[idx] += e
        e = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return project_feasible(h_old, n_new)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    if n_new == 1:
        x_new = np.array([1.0])
    else:
        x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return project_feasible(h_new, n_new)


def compute_D(h, n):
    base = n - 1
    D = np.empty(2 * n - 1, dtype=float)
    for d in range(-(n - 1), n):
        if d >= 0:
            D[base + d] = np.dot(h[:n - d], 1.0 - h[d:])
        else:
            m = -d
            D[base - d] = np.dot(h[m:], 1.0 - h[:n - m])
    return D


def compute_G(h, n):
    L = 2 * n - 1
    base = n - 1
    G = np.zeros((n, L), dtype=float)
    for d in range(-(n - 1), n):
        i = base + d
        if d >= 0:
            G[:n - d, i] = 1.0 - h[d:]
            if d == 0:
                G[:, i] -= h
            else:
                G[d:, i] -= h[:n - d]
        else:
            m = -d
            G[m:, i] = 1.0 - h[:n - m]
            G[:n - m, i] -= h[m:]
    return G


def init_state(h, n):
    return compute_D(h, n), compute_G(h, n)


def apply_h_change(G, u, delta_h, n, base, rows):
    G[rows, base + (u - rows)] -= delta_h
    G[rows, base + (rows - u)] -= delta_h


def try_add_pair(pairs, seen, s, t, h, n):
    if s == t:
        return
    if h[s] <= 1e-8 or h[t] >= 1.0 - 1e-8:
        return
    key = s * n + t
    if key not in seen:
        seen.add(key)
        pairs.append((s, t))

    seen = set()
    L = 2 * n - 1

def trial_move(D, G, tmax, base, s, t, dmax):
    if dmax <= 1e-9:
        return None
    c1 = base + t - s
    c2 = base + s - t
    diff = G[t] - G[s]
    sl = np.maximum(tmax - D, 0.0)
    strict = sl <= STRICT_TOL
    if strict.any() and not np.all(diff[strict] < -DIFF_TOL):
        return None

    ub = np.inf
    pos = diff > 1e-14
    if np.any(pos):
        cand = sl[pos] / diff[pos]
        if cand.size:
            ub = min(ub, float(np.min(cand)))

    a = float(diff[c1])
    val = float(sl[c1])
    if val <= 1e-14:
        u = 0.0
    else:
        u = 0.5 * (-a + float(np.sqrt(a * a + 4.0 * val)))
    if u < ub:
        ub = u

    a = float(diff[c2])
    val = float(sl[c2])
    if val <= 1e-14:
        u = 0.0
    else:
        u = 0.5 * (-a + float(np.sqrt(a * a + 4.0 * val)))
    if u < ub:
        ub = u

    if not np.isfinite(ub):
        ub = dmax
    if ub > dmax:
        ub = dmax
    if ub <= 1e-10:
        return None

    best_delta = 0.0
    best_cm = tmax
    for fr in FRACS:
        delta = fr * ub
        if delta <= 1e-13:
            continue
        Dc = D + delta * diff
        Dc[c1] += delta * delta
        Dc[c2] += delta * delta
        cm = float(Dc.max())
        if cm < best_cm:
            best_cm = cm
            best_delta = delta
    if best_delta <= 0.0 or best_cm >= tmax - IMPROVE_TOL:
        return None
    return best_delta, best_cm, diff


def subgradient_step(h, n, deadline, rng):
    D, G = init_state(h, n)
    tmax = float(D.max())
    if time.time() > deadline:
        return h, tmax, False
    L = 2 * n - 1
    top = np.argsort(D)[::-1][:5]
    for st in range(3):
        if time.time() > deadline - 1:
            break
        if st == 0:
            w = np.ones(len(top)) / len(top)
        else:
            w = np.abs(rng.normal(1.0, 0.5, len(top)))
            sw = w.sum()
            if sw < 1e-300:
                w = np.ones(len(top)) / len(top)
            else:
                w = w / sw
        grad = np.zeros(n, dtype=float)
        for k in range(len(top)):
            grad += w[k] * G[:, top[k]]
        dirv = -grad
        dirv -= dirv.mean()
        if np.max(np.abs(dirv)) < 1e-12:
            break
        alpha = 1.0
        pos = dirv > 1e-14
        if pos.any():
            alpha = min(alpha, float(np.min((1.0 - h[pos]) / dirv[pos])))
        neg = dirv < -1e-14
        if neg.any():
            alpha = min(alpha, float(np.min(h[neg] / (-dirv[neg]))))
        if not np.isfinite(alpha) or alpha <= 1e-13:
            break
        for fr in (0.4, 0.2, 0.1, 0.05, 0.02):
            a = fr * alpha
            hc = h + a * dirv
            hc = project_feasible(hc, n)
            Dc = compute_D(hc, n)
            cm = float(Dc.max())
            if cm < tmax - 1e-10:
                return hc, cm, True
    return h, tmax, False


def polish(h, n, deadline, rng, allow_perturb=True):
    h = project_feasible(h, n).copy()
    D, G = init_state(h, n)
    tmax = float(D.max())
    best_h = h.copy()
    best_D = tmax
    base = n - 1
    rows = np.arange(n)
    accept = 0
    no_imp = 0
    last_best = time.time()
    pf_idx = 0

    while time.time() < deadline:
        pairs = generate_pairs(h, D, G, tmax, n, rng,
                               extra_random=200 + 120 * no_imp,
                               expand=no_imp >= 2)
        improved = False
        for s, t in pairs:
            if time.time() > deadline:
                break
            dmax = h[s] if h[s] < 1.0 - h[t] else 1.0 - h[t]
            res = trial_move(D, G, tmax, base, s, t, dmax)
            if res is None:
                continue
            delta, new_tmax, diff = res
            D += delta * diff
            c1 = base + t - s
            c2 = base + s - t
            sq = delta * delta
            D[c1] += sq
            D[c2] += sq
            h[s] -= delta
            h[t] += delta
            apply_h_change(G, s, -delta, n, base, rows)
            apply_h_change(G, t, delta, n, base, rows)
            tmax = float(D.max())
            if tmax < best_D - IMPROVE_TOL:
                best_D = tmax
                best_h = h.copy()
                last_best = time.time()
            improved = True
            accept += 1
            if accept % 100 == 0:
                h = project_feasible(h, n)
                D, G = init_state(h, n)
                tmax = float(D.max())
                if tmax < best_D - IMPROVE_TOL:
                    best_D = tmax
                    best_h = h.copy()
                    last_best = time.time()

        if improved:
            no_imp = 0
            continue
        no_imp += 1

        if no_imp == 2 and time.time() < deadline - 12 and time.time() - last_best > 5:
            h2, tm2, ok = subgradient_step(h, n, deadline - 2, rng)
            if ok:
                D, G = init_state(h2, n)
                tmax = tm2
                no_imp = 0
                if tmax < best_D - IMPROVE_TOL:
                    best_D = tmax
                    best_h = h.copy()
                    last_best = time.time()
                continue

        if not allow_perturb:
            if no_imp >= 3:
                break
            continue

        if time.time() - last_best > 25:
            scale = SCALES[min(pf_idx, len(SCALES) - 1)]
            mode = pf_idx % 4
            if mode == 0:
                hp = best_h + rng.normal(0.0, scale, n)
            elif mode == 1:
                hp = 1.0 - best_h + rng.normal(0.0, 0.01, n)
            elif mode == 2:
                hp = best_h[::-1] + rng.normal(0.0, 0.01, n)
            else:
                shift = int(rng.randint(max(1, n - 1)))
                hp = np.roll(best_h, shift) + rng.normal(0.0, 0.01, n)
            hp = project_feasible(hp, n)
            h = hp.copy()
            D, G = init_state(h, n)
            tmax = float(D.max())
            if tmax < best_D - IMPROVE_TOL:
                best_D = tmax
                best_h = h.copy()
                last_best = time.time()
            no_imp = 0
            pf_idx += 1
            continue

        break

    best_h = project_feasible(best_h, n)
    best_c = eval_c5(best_h)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(1.0, float(budget_s) - 3.0)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = float("inf")
    best_n = 300

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=float).copy()
        if parent.size == 0:
            parent = None
    except Exception:
        parent = None

    remaining = deadline - t0
    if parent is not None:
        n_p = len(parent)
        if n_p > 1000:
            h0 = resample_h(parent, 300)
            n_p = 300
        else:
            h0 = project_feasible(parent, n_p)
        best_h = h0
        best_c = eval_c5(h0)
        best_n = n_p

        primary = min(0.55 * budget_s, max(60.0, remaining - 250.0))
        primary = min(primary, remaining - 5.0)
        if primary > 30.0:
            h1, c1 = polish(best_h, best_n, t0 + primary, rng, True)
            if c1 < best_c:
                best_h, best_c, best_n = h1, c1, best_n

    if best_h is None:
        n = 300
        x = np.linspace(0.0, 2.0, n)
        h = 0.5 + 0.25 * np.cos(np.pi * (x - 1.0)) + 0.1 * np.cos(2.0 * np.pi * (x - 1.0))
        h = project_feasible(h, n)
        alloc = min(0.58 * budget_s, deadline - t0 - 220.0)
        alloc = max(30.0, alloc)
        h1, c1 = polish(h, n, t0 + alloc, rng, True)
        best_h, best_c, best_n = h1, c1, n

    if best_h is not None:
        alts = []
        for cand in (best_n + 1, best_n - 1, best_n + 10, best_n - 10,
                     best_n + 20, best_n - 20, best_n + 40):
            if 60 <= cand <= 1000 and cand not in alts:
                alts.append(cand)
        tried = 0
        for nt in alts:
            rem = deadline - time.time()
            if rem < 130.0 or tried >= 2:
                break
            alloc = min(160.0, rem - 90.0)
            if alloc < 60.0:
                break
            h_start = resample_h(best_h, nt)
            h2, c2 = polish(h_start, nt, time.time() + alloc, rng, True)
            if c2 < best_c:
                best_h, best_c, best_n = h2, c2, nt
            tried += 1

    if best_h is not None:
        rem = deadline - time.time() - 2.0
        if rem > 25.0:
            h3, c3 = polish(best_h, best_n, time.time() + rem - 2.0, rng, False)
            if c3 < best_c:
                best_h, best_c = h3, c3

    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = eval_c5(best_h)
        best_n = 300

    best_h = project_feasible(best_h, best_n)
    best_c = eval_c5(best_h)
    return best_h, best_c, best_n
