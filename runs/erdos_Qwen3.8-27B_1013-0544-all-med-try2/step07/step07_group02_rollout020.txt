        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        dj = n - 1 + (j - i)
        di = n - 1 + (i - j)
        if dj >= 0 and dj < 2 * n - 1:
            delta[dj] += 1.0
        if di >= 0 and di < 2 * n - 1:
            delta[di] += 1.0
        nc = (corr + delta).max() * dx
        accept = nc < cur_c or (T > 1e-7 and rng.random() < np.exp(-(nc - cur_c) / T))
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            j_pos = ones_idx.index(j)
            ones_idx[j_pos] = i
            i_pos = zeros_idx.index(i)
            zeros_idx[i_pos] = j
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        e = np.exp(shifted - sm)
        return m + (sm + np.log(e.sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 15
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Binary simulated annealing
    p1 = t0 + 0.06 * total
    hb, cb = binary_simulated_annealing(h0, n, p1, max_iter=40000)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Coordinate descent (randomized order)
    p2 = t0 + 0.18 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=400, deadline=p2, rng=True)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 3: SLSQP with gradual alpha ramp (main polish)
    p3 = t0 + 0.58 * total
    alphas = [10, 50, 200, 1000, 5000, 20000, 100000, 500000, 2000000,
              10000000, 50000000, 200000000, 1000000000, 5000000000]
    h = best_h.copy()
    for k, alpha in enumerate(alphas):
        if time.time() > p3 - 10:
            break
        iters = 3000 if alpha < 10000 else 5000
        h = slsqp_smooth(h, n, alpha, iters, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Coordinate descent from smoothed result
    p4 = t0 + 0.68 * total
    hc2, _ = coord_descent(best_h, n, max_sweeps=300, deadline=p4, rng=True)
    hc2 = project_feasible(hc2, n)
    cc2 = compute_c5(hc2, n)
    if cc2 < best_c:
        best_c, best_h = cc2, hc2.copy()

    # Phase 5: High-precision SLSQP
    p5 = t0 + 0.80 * total
    h = best_h.copy()
    for alpha in [10000000, 100000000, 1000000000, 10000000000, 100000000000]:
        if time.time() > p5 - 5:
            break
        h = slsqp_smooth(h, n, alpha, 6000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 6: Multi-resolution exploration
    p6 = t0 + 0.92 * total
    resolutions = [250, 300, 350, 450, 500, 600]
    for nh in resolutions:
        if time.time() > p6 - 25:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, _ = coord_descent(hh, nh, max_sweeps=80, deadline=p6 - 20, rng=True)
        hh = project_feasible(hh, nh)
        ch = compute_c5(hh, nh)
        for alpha in [100000, 5000000, 50000000, 500000000]:
            if time.time() > p6 - 10:
                break
            hh = slsqp_smooth(hh, nh, alpha, 2000, 1e-16)
            ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh

    # Phase 7: Final perturbation restarts
    if time.time() < dl - 12:
        scales = [0.001, 0.002, 0.004, 0.008, 0.015, 0.03]
        for scale in scales:
            if time.time() > dl - 3:
                break
            h = best_h + scale * rng.randn(n)
            h = project_feasible(h, n)
            for alpha in [50000000, 500000000, 5000000000]:
                if time.time() > dl - 2:
                    break
                h = slsqp_smooth(h, n, alpha, 4000, 1e-18)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
