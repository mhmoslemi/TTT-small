def project_feasible(x, n=None):
    if n is None: n=len(x)
    x=np.asarray(x,dtype=float).reshape(-1)
    if x.size != n:
       x = np.resize(x,n)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    target=0.5*n
    if x.size==0: return x
    if float(np.min(x)) >= -1e-12 and float(np.max(x)) <= 1.0+1e-12:
       h=np.clip(x,0.0,1.0)
       return _adjust_sum(h, target)
    lo=float(np.min(x))-1.0; hi=float(np.max(x))+1.0
    # ensure sum bounds:
    # At lo sum maybe <=target? increase lo? Actually lo lower -> sum higher. If sum_at(lo) <= target, decrement lo.
    for _ in range(10):
       s=np.sum(np.clip(x-lo,0.0,1.0))
       if s >= target: break
       lo -= 1.0
    for _ in range(10):
       s=np.sum(np.clip(x-hi,0.0,1.0))
       if s <= target: break
       hi += 1.0
    for _ in range(50):
       mid=0.5*(lo+hi)
       s=np.sum(np.clip(x-mid,0.0,1.0))
       if s > target: lo=mid
       else: hi=mid
    h=np.clip(x-0.5*(lo+hi),0.0,1.0)

def grad_step(h,n,deadline,lr=0.05,tol=1e-8,k=40,momentum=0.8):
    h=project_feasible(h,n); C=compute_c5(h,n); v=np.zeros(n); stale=0; lr_cur=lr

       c=compute_corr(h,n); w=top_weights(c,tol,k)

       g -= float(np.mean(g))

       if scale < 1e-14: break

import numpy as np
import time
from scipy.optimize import linprog


def _adjust_sum(h, target):
    for _ in range(12):
        diff = target - float(np.sum(h))
        if abs(diff) <= 1e-13:
            break
        if diff > 0:
            mask = (h > 1e-12) & (h < 1.0 - 1e-12)
            if not mask.any():
                mask = (h < 0.5)
            if not mask.any():
                mask = (h < 1.0 - 1e-12)
        else:
            mask = (h > 1e-12) & (h < 1.0 - 1e-12)
            if not mask.any():
                mask = (h > 0.5)
            if not mask.any():
                mask = (h > 1e-12)
        if not mask.any():
            break
        h = h.copy()
        h[mask] += diff / float(np.count_nonzero(mask))
        h = np.clip(h, 0.0, 1.0)
    return h


def project_feasible(x, n=None):
    if n is None:
        n = len(x)
    n = int(n)
    x = np.asarray(x, dtype=float).reshape(-1)
    if x.size != n:
        x = np.resize(x, n)
    if x.size == 0:
        return x
    if not np.all(np.isfinite(x)):
        x = np.where(np.isfinite(x), x, 0.5)
    target = 0.5 * n
    if float(np.min(x)) >= -1e-12 and float(np.max(x)) <= 1.0 + 1e-12:
        h = np.clip(x, 0.0, 1.0)
        return _adjust_sum(h, target)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(20):
        if float(np.sum(np.clip(x - lo, 0.0, 1.0))) >= target:
            break
        lo -= 1.0
    for _ in range(20):
        if float(np.sum(np.clip(x - hi, 0.0, 1.0))) <= target:
            break
        hi += 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    return _adjust_sum(h, target)


def compute_corr(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h, n):
    return float(np.max(compute_corr(h, n)))


def top_weights(c, tol, k):
    if k < 1:
        k = 1
    m = float(np.max(c))
    idx = np.nonzero(c >= m - tol)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))])
    if idx.size > k:
        idx = idx[np.argsort(c[idx])[-k:]]
    w = np.zeros(c.shape, dtype=float)
    w[idx] = 1.0 / float(idx.size)
    return w


def active_gradient(h, n, m):
    g = np.zeros(n, dtype=float)
    L = int(m) - (n - 1)
    if L >= 0:
        r = L
        nv = n - r
        if nv > 0:
            g[r:] += 1.0 - h[:nv]
            g[:nv] -= h[r:]
    else:
        r = -L
        nv = n - r
        if nv > 0:
            g[:nv] += 1.0 - h[r:]
            g[r:] -= h[:nv]
    return g * (2.0 / float(n))


def grad_weights(h, n, w):
    g = np.zeros(n, dtype=float)
    nz = np.nonzero(w)[0]
    for m in nz:
        g += w[m] * active_gradient(h, n, int(m))
    return g


def slp_polish(h, n, deadline, q=500, rho_init=0.12):
    h = project_feasible(h, n)
    C = compute_c5(h, n)
    rho = float(rho_init)
    fail = 0
    max_len = 2 * n - 1
    q = max(1, min(int(q), max_len))
    target_sum = 0.5 * n

    while time.time() < deadline - 1.0:
        if deadline - time.time() < 3.0:
            break
        c = compute_corr(h, n)
        C = float(np.max(c))
        if q < c.size:
            idx = np.argpartition(c, q - 1)[-q:]
        else:
            idx = np.arange(c.size)
        f_act = c[idx]
        G = np.empty((int(idx.size), n), dtype=float)
        for a, mm in enumerate(idx):
            G[a, :] = active_gradient(h, n, int(mm))
        A = np.empty((f_act.size, n + 1), dtype=float)
        A[:, :n] = G
        A[:, -1] = -1.0
        b = -f_act
        Aeq = np.zeros((1, n + 1), dtype=float)
        Aeq[0, :n] = 1.0
        beq = np.zeros(1, dtype=float)
        cobj = np.zeros(n + 1, dtype=float)
        cobj[-1] = 1.0
        bounds = np.empty((n + 1, 2), dtype=float)
        bounds[:n, 0] = -np.minimum(rho, h)
        bounds[:n, 1] = np.minimum(rho, 1.0 - h)
        bounds[-1, 0] = 0.0
        bounds[-1, 1] = max(C + rho + 1.0, 1.0)

        try:
            res = linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq, bounds=bounds, method="highs")
        except Exception:
            res = None

        improved = False
        if res is not None and getattr(res, "success", False):
            d = np.asarray(res.x[:n], dtype=float)
            d -= float(np.mean(d))
            if float(np.abs(d).max()) > 1e-10:
                best_c = C
                best_h = h
                for a in (1.0, 0.8, 0.6, 0.4, 0.25, 0.125, 0.0625, 0.03125):
                    hc = h + a * d
                    if (float(np.min(hc)) < -1e-9 or float(np.max(hc)) > 1.0 + 1e-9 or
                            abs(float(np.sum(hc)) - target_sum) > 1e-8):
                        hc = project_feasible(hc, n)
                    else:
                        hc = np.clip(hc, 0.0, 1.0)
                        hc = _adjust_sum(hc, target_sum)
                    cc = compute_c5(hc, n)
                    if cc < best_c - 1e-13:
                        best_c = cc
                        best_h = hc
                if best_c < C:
                    h = best_h
                    C = best_c
                    improved = True
                    fail = 0
                    if rho < 0.2:
                        rho = min(0.2, rho * 1.12)

        if not improved:
            fail += 1
            if fail >= 2 and time.time() + 8.0 < deadline:
                kh, kc = grad_step(h, n, time.time() + 8.0, lr=0.06, tol=1e-7, k=30)
                if kc < C - 1e-12:
                    h = kh
                    C = kc
                    fail = 0
                    rho = min(0.18, rho * 1.1)
            rho *= 0.6
            if fail >= 5 or rho < 1e-5:
                break
    return h, C


def grad_step(h, n, deadline, lr=0.08, tol=1e-8, k=40, momentum=0.8):
    h = project_feasible(h, n)
    C = compute_c5(h, n)
    v = np.zeros(n, dtype=float)
    stale = 0
    lr_cur = float(lr)

    while time.time() < deadline - 1.0:
        c = compute_corr(h, n)
        w = top_weights(c, tol, k)
        g = grad_weights(h, n, w)
        g -= float(np.mean(g))
        scale = float(np.max(np.abs(g)))
        if scale < 1e-14:
            break
        v = momentum * v - g
        vmax = float(np.max(np.abs(v)))
        if vmax < 1e-14:
            v = -g
            vmax = scale
        step = lr_cur / vmax
        found = False
        for a in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            hc = project_feasible(h + a * step * v, n)
            cc = compute_c5(hc, n)
            if cc < C - 1e-13:
                h = hc
                C = cc
                stale = 0
                found = True
                break
        if not found:
            stale += 1
            lr_cur *= 0.7
            v *= 0.5
            if stale > 25 or lr_cur < 1e-5:
                break
    return h, C


def pair_polish(h, n, deadline, max_pairs=30):
    h = project_feasible(h, n)
    C = compute_c5(h, n)
    rounds = 0
    while time.time() < deadline - 2.0 and rounds < 25:
        rounds += 1
        c = compute_corr(h, n)
        w = top_weights(c, 1e-7, 30)
        g = grad_weights(h, n, w)
        g -= float(np.mean(g))
        order = np.argsort(g)
        improved = False
        for a in range(min(int(max_pairs), n // 2)):
            if time.time() > deadline - 2.0:
                break
            src = int(order[-1 - a])
            dst = int(order[a])
            if src == dst:
                continue
            maxd = min(float(h[src]), float(1.0 - h[dst]))
            if maxd <= 1e-10:
                continue
            best_c = C
            best_h = None
            for frac in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
                d = frac * maxd
                hn = h.copy()
                hn[src] -= d
                hn[dst] += d
                hc = project_feasible(hn, n)
                cc = compute_c5(hc, n)
                if cc < best_c - 1e-13:
                    best_c = cc
                    best_h = hc
            if best_h is not None:
                h = best_h
                C = best_c
                improved = True
        if not improved:
            break
    return h, C


def resample_to(h, n_new):
    n_old = len(h)
    n_new = int(n_new)
    if n_new == n_old:
        return project_feasible(h, n_new)
    idx = np.minimum((np.arange(n_new) * n_old) // n_new, n_old - 1)
    return project_feasible(h[idx], n_new)


def refine_candidate(h, n, deadline):
    n = int(n)
    h = project_feasible(h, n)
    C = compute_c5(h, n)
    rem = deadline - time.time()
    if rem <= 5.0:
        return h, C

    q1 = min(800 if n <= 500 else 500, 2 * n - 1)
    h, C = slp_polish(h, n, time.time() + 0.5 * rem, q=q1, rho_init=0.12)

    rem = deadline - time.time()
    if rem > 12.0:
        h2, c2 = grad_step(h, n, time.time() + min(rem * 0.22, 150.0), lr=0.08, tol=1e-7, k=50)
        if c2 < C:
            h, C = h2, c2

    rem = deadline - time.time()
    if rem > 12.0:
        q2 = min(400 if n <= 500 else 300, 2 * n - 1)
        h3, c3 = slp_polish(h, n, time.time() + 0.65 * rem, q=q2, rho_init=0.03)
        if c3 < C:
            h, C = h3, c3

    rem = deadline - time.time()
    if rem > 6.0:
        h4, c4 = pair_polish(h, n, time.time() + 0.85 * rem, max_pairs=35)
        if c4 < C:

    return h, C
