        best_c, best_h = c4, h4
    # Phase 5: Higher resolution attempts
    if time.time() < deadline - 90:
        for n_new in [400, 500, 600]:
            if time.time() > deadline - 60:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_new)
            h_ni = np.interp(x_new, x_old, best_h)
            h_ni = project_feasible(h_ni, n_new)
            t_a = min(deadline - 40, time.time() + 40)
            h_a, c_a = iter_active(h_ni, n_new, t_a)
            t_b = min(deadline - 25, t_a + 20)
            h_b, c_b = bin_local_search(h_a, n_new, t_b)
            t_c = min(deadline - 10, t_b + 15)
            h_c, c_c = iter_active(h_b, n_new, t_c)
            h_best = min([(c_a, h_a), (c_b, h_b), (c_c, h_c)], key=lambda t: t[0])
            if h_best[0] < best_c:
                best_c, best_h = h_best[0], h_best[1]
    # Phase 6: Final refinement
    n_final = len(best_h)
    if time.time() < deadline - 5:
        h_f, c_f = iter_active(best_h, n_final, deadline - 1)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
