```python
import numpy as np
import time
from scipy.optimize import minimize

_CACHE = {'last': None}

def eval_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return 1.0
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            h[free] += diff / free.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def project_box_sum(v, n):
    target = 0.5 * n
    hc = np.clip(v, 0.0, 1.0)
    if abs(hc.sum() - target) < 1e-12:
        return hc
    lo = -np.max(v) - 1.0
    hi = -np.min(v) + 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        if np.clip(v + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)

def sum_con(x, n):
    return float(np.sum(x) - 0.5 * n)

def sum_con_jac(x, n):
    return np.ones(len(x))

def smooth_c_weights(c, alpha, K):
    c = np.asarray(c, dtype=float)
    x = float(np.max(c))
    K = int(K) if K is not None else c.size
    if K <= 0:
        K = 1
    if K >= c.size:
        idx = np.arange(c.size)
        vals = c
    else:
        idx = np.argpartition(c, c.size - K)[-K:]
        vals = c[idx]
    z = alpha * (vals - x)
    p = np.exp(z)
    s = p.sum()
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(len(idx))
        p[int(np.argmax(vals))] = 1.0
        s = 1.0
    w = p / s
    f = x + np.log(s) / alpha
    return f, w, idx

def smooth_grad_from_weights(h, n, w, idx):
    g = np.zeros(n)
    center = n - 1
    for wi, lag in zip(w, idx):
        if wi <= 1e-13:
            continue
        if lag == center:
            g += wi * (1.0 - 2.0 * h)
        elif lag < center:
            m = center - lag
            g[:n - m] += wi * (1.0 - h[m:])
            g[m:] -= wi * h[:n - m]
        else:
            m = lag - center
            g[m:] += wi * (1.0 - h[:n - m])
            g[:n - m] -= wi * h[m:]
    return g

def _smooth_infos(h, n, alpha, K):
    c = np.correlate(h, 1.0 - h, mode='full')
    key = (int(n), float(alpha), int(K))
    last = _CACHE.get('last')
    if last is not None and last['key'] == key and np.array_equal(last['h'], h):
        return last['data']
    f, w, idx = smooth_c_weights(c, alpha, K)
    _CACHE['last'] = {'key': key, 'h': h.copy(), 'data': (f, w, idx)}
    return f, w, idx

def top_logsumexp_fun(h, n, alpha, K):
    return float(_smooth_infos(h, n, alpha, K)[0])

def top_logsumexp_grad(h, n, alpha, K):
    _, w, idx = _smooth_infos(h, n, alpha, K)
    return smooth_grad_from_weights(h, n, w, idx)

def slsqp_top(h, n, alpha, K, maxiter):
    h = project_feasible(h, n)
    if maxiter <= 0:
        return h, eval_c5(h)
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(top_logsumexp_fun, h, args=(n, alpha, K), jac=top_logsumexp_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': int(maxiter), 'ftol': 1e-14})
        h1 = project_feasible(res.x, n)
    except Exception:
        h1 = h
    return h1, eval_c5(h1)

def projected_grad_refine(h, n, deadline, alpha, K, max_iter):
    h = project_feasible(h, n).copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    x = float(np.max(c))
    f, _, _ = smooth_c_weights(c, alpha, K)
    best_h = h.copy()
    best_x = x
    step = 0.02
    no = 0
    for _ in range(int(max_iter)):
        if time.time() > deadline - 1.0:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        x = float(np.max(c))
        if x < best_x - 1e-14:
            best_x = x
            best_h = h.copy()
        f, w, idx = smooth_c_weights(c, alpha, K)
        g = smooth_grad_from_weights(h, n, w, idx)
        g -= np.mean(g)
        gn = float(np.linalg.norm(g))
        if not np.isfinite(gn) or gn < 1e-14:
            break
        d = -g / gn
        accepted = False
        for fac in (1.0, 0.5, 0.25, 0.1, 0.04):
            hn = project_box_sum(h + fac * step * d, n)
            cn = np.correlate(hn, 1.0 - hn, mode='full')
            xn = float(np.max(cn))
            if xn < x - 1e-13:
                h = hn
                x = xn
                accepted = True
                break
            fn, _, _ = smooth_c_weights(cn, alpha, K)
            if fn < f - 1e-11 and xn < x + 1e-10:
                h = hn
                x = xn
                f = fn
                accepted = True
                break
        if accepted:
            step = min(0.12, step * 1.15)
            no = 0
        else:
            step *= 0.5
            no += 1
            if step < 1e-8 or no > 35:
                break
    best_h = project_feasible(best_h, n)
    return best_h, best_x * (2.0 / n)

def improve_by_permutations(h, n, deadline):
    best_h = h.copy()
    best_c = eval_c5(h)
    step = max(1, (n + 15) // 16)
    for s in range(0, n, step):
        if time.time() > deadline:
            break
        for arr in (np.roll(h, s), np.roll(h[::-1], s)):
            c = eval_c5(arr)
            if c < best_c - 1e-15:
                best_c = c
                best_h = arr.copy()
    return best_h, best_c

def resample_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def default_h(n):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.35 * np.cos(np.pi * x) + 0.15 * np.cos(2.0 * np.pi * x)
    return project_feasible(h, n)

def binary_polish(h, n, deadline, rng, max_iter):
    if n % 2 != 0:
        return h, eval_c5(h)
    scores = h + 1e-4 * rng.normal(0.0, 1.0, n)
    order = np.argsort(-scores)
    hb = np.zeros(n, dtype=float)
    hb[order[:n // 2]] = 1.0
    c = eval_c5(hb)
    best = hb.copy()
    best_c = c
    ones = list(np.nonzero(hb)[0])
    zeros = list(np.nonzero(hb < 0.5)[0])
    non = 0
    for _ in range(int(max_iter)):
        if time.time() > deadline:
            break
        if not ones or not zeros:
            break
        ii = int(rng.randint(len(ones)))
        jj = int(rng.randint(len(zeros)))
        i = ones[ii]
        j = zeros[jj]
        hb[i] = 0.0
        hb[j] = 1.0
        c2 = eval_c5(hb)
        if c2 < best_c - 1e-15:
            best_c = c2
            best = hb.copy()
            non = 0
            ones[ii] = j
            zeros[jj] = i
        else:
            hb[i] = 1.0
            hb[j] = 0.0
            non += 1
            if non > 12000:
                break
    return best, best_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c = eval_c5(h)

    h1, c1 = improve_by_permutations(best_h, n, min(deadline, time.time() + 8.0))
    if c1 < best_c:
        best_h, best_c = h1, c1

    if time.time() < deadline - 18.0:
        h1, c1 = projected_grad_refine(best_h, n, min(deadline - 18.0, time.time() + 35.0), 60, 50, 260)
        if c1 < best_c:
            best_h, best_c = h1, c1

    if n <= 650 and time.time() < deadline - 45.0:
        for alp, kk, mx in ((20, 150, 90), (100, 60, 90), (500, 25, 70), (2500, 10, 55)):
            if time.time() > deadline - 32.0:
                break
            h1, c1 = slsqp_top(best_h, n, alp, kk, mx)
            if c1 < best_c:
                best_h, best_c = h1, c1

    for alp, kk, it in ((150, 30, 260), (1200, 10, 180)):
        if time.time() > deadline - 22.0:
            break
        h1, c1 = projected_grad_refine(best_h, n, min(deadline - 12.0, time.time() + 32.0), alp, kk, it)
        if c1 < best_c:
            best_h, best_c = h1, c1

    for sc in (0.02, 0.06):
        if time.time() > deadline - 22.0:
            break
        hp = project_feasible(best_h + rng.normal(0.0, sc, n), n)
        h1, c1 = projected_grad_refine(hp, n, min(deadline - 12.0, time.time() + 25.0), 500, 20, 180)
        if c1 < best_c:
            best_h, best_c = h1, c1

    if n % 2 == 0 and n <= 600 and time.time() < deadline - 10.0:
        hb, cb = binary_polish(best_h, n, min(deadline - 2.0, time.time() + 22.0), rng, 25000)
        if cb < best_c:
            best_h, best_c = hb, cb

    best_h = project_feasible(best_h, n)
    best_c = eval_c5(best_h)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline = t0 + budget - 6.0
    rng = np.random.RandomState(seed)

    try:
        h_init = np.asarray(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    if h_init is None or n_init < 16:
        h_init = default_h(300)
        n_init = 300

    h_init = project_feasible(h_init, n_init)
    parent_end = min(deadline, t0 + max(150.0, 0.46 * budget))
    best_h, best_c = optimize_n(h_init, n_init, parent_end, rng)

    n_options = [200, 240, 300, 360, 480, 250, 600, 400, 700, 800, 150, 500]
    for nn in n_options:
        rem = deadline - time.time() - 35.0
        if rem < 70.0:
            break
        if nn == len(best_h):
            continue
        alloc = min(120.0, rem * 0.42)
        if alloc < 40.0:
            continue
        h_start = resample_to_n(best_h, nn)
        h1, c1 = optimize_n(h_start, nn, time.time() + alloc, rng)
        if c1 < best_c:
            best_h, best_c = h1, c1

    n_best = len(best_h)
    if time.time() < deadline - 20.0:
        h1, c1 = projected_grad_refine(best_h, n_best, deadline - 2.0, 2500, 8, 3500)
        if c1 < best_c:
            best_h, best_c = h1, c1

    if n_best <= 650 and time.time() < deadline - 45.0:
        h1, c1 = slsqp_top(best_h, n_best, 800, 14, 420)
        if c1 < best_c:
            best_h, best_c = h1, c1
    if n_best <= 650 and time.time() < deadline - 25.0:
        h1, c1 = slsqp_top(best_h, n_best, 5000, 8, 260)
        if c1 < best_c:
            best_h, best_c = h1, c1

    if time.time() < deadline - 2.0:
        best_h = project_feasible(best_h, len(best_h))
        best_c = eval_c5(best_h)

    return best_h, best_c, int(len(best_h))
```
