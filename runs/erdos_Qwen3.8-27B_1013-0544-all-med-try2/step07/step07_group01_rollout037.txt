    best_c = compute_c5(h, n)
    dx = 2.0 / n

        if time.time() > deadline:
            break
        improved = False
        for pos in range(n):
            if time.time() > deadline:
                return h, best_c
            v = h[pos]
            
            # Compute A_m for all m
            A = np.zeros(2*n - 1)
            for m in range(2*n - 1):
                k = m - (n - 1)
                # h[pos] as "h" part: need pos+k in [0, n-1]
                if 0 <= pos + k <= n - 1:
                    A[m] += (1.0 - h[pos + k])
                # h[pos] as "g" part (1-h): need pos-k in [0, n-1]
                if 0 <= pos - k <= n - 1:

            S = h.sum() - v  # sum of all other elements

            # over v' in [0, 1]

                vals = A * vp + B
                return float(np.max(vals) * dx + mu * (S + vp - n/2)**2)

            res = minimize_scalar(obj_1d, bounds=(0, 1), method='bounded',

            h[pos] = v_new

            if c < best_c - 1e-12:

                improved = True
        if not improved:
            break
    
    # Project to feasible
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    cs = alpha * (c - c.max())
    mx = cs.max()
    w = np.exp(cs - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    cs = alpha * (c - m)
    mx = cs.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(cs - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(cs))) / alpha
    return val + mu * (x.sum() - n / 2.0) ** 2

def _smooth_max_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def coord_descent(h, n, deadline, mu=1.0, max_sweeps=80):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    full = np.correlate(h, 1 - h, mode='full')
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for pos in range(n):
            if time.time() > deadline:
                return h, best_c
            v = h[pos]
            A = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                k = m - (n - 1)
                if 0 <= pos + k <= n - 1:
                    A[m] += 1.0 - h[pos + k]
                if 0 <= pos - k <= n - 1:
                    A[m] -= h[pos - k]
            B = full - A * v
            S = h.sum() - v
            target_sum = n / 2.0
            
            def obj1d(vp):
                return float(np.max(A * vp + B) * dx + mu * (S + vp - target_sum) ** 2)
            
            res = minimize_scalar(obj1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            v_new = float(res.x)
            delta = v_new - v
            if abs(delta) > 1e-12:
                h[pos] = v_new
                full += A * delta
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def lbfgs_opt(h, n, deadline, alphas=None, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [(100, 5.0), (500, 10.0), (2000, 50.0),
                  (10000, 200.0), (50000, 1000.0), (200000, 5000.0)]
    for alpha, mu in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_smooth_max_obj, h, jac=_smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, deltas=None, radius=50):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    cs = alpha * (c - c.max())
    mx = cs.max()
    w = np.exp(cs - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    cs = alpha * (c - m)
    mx = cs.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(cs - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(cs))) / alpha
    return val + mu * (x.sum() - n / 2.0) ** 2

def _smooth_max_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _coord_1d_obj(vp, A, B, S, dx, mu, target_sum):
    return float(np.max(A * vp + B) * dx + mu * (S + vp - target_sum) ** 2)

def coord_descent(h, n, deadline, mu=1.0, max_sweeps=60):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    full = np.correlate(h, 1 - h, mode='full')
    k_arr = np.arange(-(n - 1), n)
    target_sum = n / 2.0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for pos in range(n):
            if time.time() > deadline:
                return h, best_c
            v = h[pos]
            idx1 = pos + k_arr
            mask1 = (idx1 >= 0) & (idx1 < n)
            idx2 = pos - k_arr
            mask2 = (idx2 >= 0) & (idx2 < n)
            A = np.zeros(2 * n - 1)
            A[mask1] += 1.0 - h[idx1[mask1]]
            A[mask2] -= h[idx2[mask2]]
            B = full - A * v
            S = h.sum() - v
            res = minimize_scalar(_coord_1d_obj, bounds=(0.0, 1.0), method='bounded',
                                  args=(A, B, S, dx, mu, target_sum),
                                  options={'xatol': 1e-10})
            v_new = float(res.x)
            delta = v_new - v
            if abs(delta) > 1e-12:
                h[pos] = v_new
                full += A * delta
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def lbfgs_opt(h, n, deadline, alphas=None, max_iters=150):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [(100, 5.0), (500, 10.0), (2000, 50.0),
                  (10000, 200.0), (50000, 1000.0), (200000, 5000.0)]
    for alpha, mu in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_smooth_max_obj, h, jac=_smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, deltas=None, radius=40):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def optimize_single(h, n, deadline):
    h, c = coord_descent(h, n, min(deadline - 60, time.time() + 120), mu=1.0, max_sweeps=50)
    if time.time() < deadline - 30:
        h, c = lbfgs_opt(h, n, min(deadline - 30, time.time() + 80), max_iters=150)
    if time.time() < deadline - 10:
        h, c = coord_descent(h, n, min(deadline - 10, time.time() + 60), mu=0.5, max_sweeps=40)
    if time.time() < deadline - 3:
        h, c = pair_search(h, n, deadline - 2, [0.002, 0.001, 0.0005, 0.0002], radius=30)
    return h, c
