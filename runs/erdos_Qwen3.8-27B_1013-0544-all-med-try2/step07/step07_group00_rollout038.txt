```python
import numpy as np
from scipy.optimize import minimize
import time

def fft_correlate(a, b):
    """Compute np.correlate(a, b, 'full') using FFT"""
    n = len(a)
    m = len(b)
    N = n + m - 1
    N_fft = 1 << (N - 1).bit_length()
    A = np.fft.rfft(a, N_fft)
    B = np.fft.rfft(b, N_fft)
    result = np.fft.irfft(A * np.conj(B), N_fft)[:N]
    return result

def compute_c5_vec_fast(h, n):
    """Compute correlation values using FFT"""
    dx = 2.0 / n
    c = fft_correlate(h, 1.0 - h)
    return c * dx

def compute_c5_fast(h, n):
    """Compute C5 value"""
    return float(np.max(compute_c5_vec_fast(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_fast(h, w, n):
    """
    Compute gradient of smoothed max of correlation values.
    h: array of length n
    w: softmax weights, length 2n-1
    Returns: gradient array of length n
    """
    # P[i] = sum of w[k] for k in range where 0 <= i-k+n-1 < n
    # i.e., i-n+1 <= k <= i, clipped to [0, 2n-2]
    W_cum = np.zeros(2*n)
    W_cum[1:] = np.cumsum(w)
    lo = np.maximum(0, np.arange(n) - n + 1)
    hi = np.minimum(np.arange(n), 2*n - 2)
    P = W_cum[hi + 1] - W_cum[lo]
    
    # S1[i] = convolve(h, w, 'full')[i+n-1]
    N1 = 1 << (3*n - 2).bit_length()
    conv1 = np.fft.irfft(np.fft.rfft(h, N1) * np.fft.rfft(w, N1), N1)[:3*n-1]
    S1 = conv1[n-1:2*n-1]

    conv2 = np.fft.irfft(np.fft.rfft(h, N1) * np.fft.rfft(w_flipped, N1), N1)[:3*n-1]
    S2 = conv2[n-1:2*n-1]
    
    return P - S1 - S2

def obj_and_grad(x, n, alpha):
    """Objective and gradient for L-BFGS-B"""
    c = compute_c5_vec_fast(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    logsumexp = mx + np.log(np.sum(np.exp(shifted - mx)))
    obj = m + logsumexp / alpha
    
    w = np.exp(shifted - mx)
    w /= w.sum()
    
    grad = compute_grad_fast(x, w, n)
    return obj, grad

def smooth_max_obj(x, n, alpha, mu):
    """Penalized objective for L-BFGS-B"""
    c = compute_c5_vec_fast(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    """Gradient of penalized objective"""
    c = compute_c5_vec_fast(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    grad = compute_grad_fast(x, w, n)
    grad += 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    return grad

def lbfgs_optimize(h, n, deadline, max_alphas=8):
    """L-BFGS-B optimization with increasing alpha"""
    best_c = compute_c5_fast(h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    mus = [1.0, 5.0, 20.0, 100.0, 500.0, 2000.0, 10000.0, 50000.0]
    
    for i in range(min(max_alphas, len(alphas))):
        if time.time() > deadline - 10:
            break
        remaining = deadline - time.time() - 10
        if remaining < 5:
            break
        alpha = alphas[i]
        mu = mus[i]
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': 300, 'ftol': 1e-18,
                                  'gtol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5_fast(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def coordinate_descent(h, n, deadline):
    """Coordinate descent with FFT-accelerated evaluation"""
    best_c = compute_c5_fast(h, n)
    dx = 2.0 / n
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for step in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                if h[i] + step <= 1.0:
                    # Find best j to pair with i
                    best_j = -1
                    best_c_try = best_c
                    for j in range(n):
                        if i == j:
                            continue
                        if h[j] - step < 0.0:
                            continue
                        h[i] += step
                        h[j] -= step
                        c_try = compute_c5_fast(h, n)
                        h[i] -= step
                        h[j] += step
                        if c_try < best_c_try - 1e-15:
                            best_c_try = c_try
                            best_j = j
                    if best_j >= 0:
                        h[i] += step
                        h[best_j] -= step
                        best_c = best_c_try
                        improved = True
    return h, best_c
```

```python
def compute_grad_fast(h, w, n):
    # P[i] = cumsum(w)[min(2n-2, i+n-1)+1] - cumsum(w)[max(0, i)]
    W_cum = np.zeros(2*n)
    W_cum[1:] = np.cumsum(w)
    P = W_cum[np.minimum(2*n-2, np.arange(n) + n - 1) + 1] - W_cum[np.arange(n)]
    
    # S1[i] = convolve(h, w, 'full')[i+n-1]
    N1 = 1 << (3*n - 2).bit_length()
    conv1 = np.fft.irfft(np.fft.rfft(h, N1) * np.fft.rfft(w, N1), N1)[:3*n-1]
    S1 = conv1[n-1:2*n-1]
    
    # S2[i] = convolve(h, w_flipped, 'full')[n-1+i]
    w_flipped = w[::-1]
    conv2 = np.fft.irfft(np.fft.rfft(h, N1) * np.fft.rfft(w_flipped, N1), N1)[:3*n-1]
    S2 = conv2[n-1:2*n-1]
    
    return P - S1 - S2
```
