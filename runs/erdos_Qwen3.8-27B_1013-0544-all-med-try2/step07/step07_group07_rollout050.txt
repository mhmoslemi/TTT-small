                    h[j] += delta

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _lag_grad(h_curr, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h_curr
        return grad
    if m >= n:
        return grad
    if lag > 0:
        grad[:n - m] += (1.0 - h_curr[m:])
        grad[m:] -= h_curr[:n - m]
    else:
        grad[:n - m] -= h_curr[m:]
        grad[m:] += (1.0 - h_curr[:n - m])
    return grad

def lp_step(h_curr, n, trust, n_cuts):
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
    cons.append(h_var >= h_curr - trust)
    cons.append(h_var <= h_curr + trust)
    idx = np.argsort(c)[-n_cuts:]
    for k in idx:
        lag = k - (n - 1)
        g = _lag_grad(h_curr, n, lag)
        cons.append(c[k] + dx * (g @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.GLOP, cp.CBC, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy(), False
    if h_var.value is None:
        return h_curr.copy(), False
    return project_feasible(h_var.value, n), True
