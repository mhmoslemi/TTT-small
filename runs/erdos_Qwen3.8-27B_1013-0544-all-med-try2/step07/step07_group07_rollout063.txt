def compute_overlaps(h, n):
    """Compute overlap at all lags. Returns array of length 2n-1 indexed as np.correlate would."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def compute_c5(h, n):
    return float(np.max(compute_overlaps(h, n)))

def binarize(h, n):
    """Convert to binary, maintaining exactly n/2 ones."""
    h_bin = (h >= 0.5).astype(float)
    n_ones = h_bin.sum()
    target = n / 2
    if n_ones > target:
        # Need to flip some 1s to 0s
        idx_ones = np.where(h_bin == 1)[0]
        # Flip the ones with lowest h value
        order = np.argsort(h[idx_ones])
        for k in range(int(n_ones - target)):
            h_bin[idx_ones[order[k]]] = 0
    elif n_ones < target:
        idx_zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx_zeros])
        for k in range(int(target - n_ones)):
            h_bin[idx_zeros[order[k]]] = 1
    return h_bin

def binary_overlaps(h, n):
    """Compute overlaps for binary h. Returns array indexed by lag (n-1-m) to (n-1+m)."""
    # overlap at offset d (d can be -(n-1) to n-1)
    # Using the convention: overlap(d) = sum_i h[i]*(1-h[i+d]) for valid i
    # This is np.correlate(h, 1-h, mode='full') for binary h
    ones = (h == 1).astype(int)
    zeros = (h == 0).astype(int)
    # For each d > 0: overlap(d) = sum_{i=0}^{n-1-d} ones[i] * zeros[i+d]
    # This is the cross-correlation of ones and zeros
    # np.correlate(ones, zeros, mode='full')[n-1+d] = sum_i ones[i] * zeros[i+d] for d >= 0
    # Actually np.correlate(a, b)[k] = sum_i a[i+k-(len(b)-1)] * b[i]
    # Let me just use the direct computation for clarity
    n_ov = np.zeros(2*n-1)
    for d in range(1, n):
        # overlap(d) = sum_{i=0}^{n-1-d} ones[i] * zeros[i+d]
        n_ov[n-1+d] = np.sum(ones[:n-d] * zeros[d:])
        # overlap(-d) = sum_{i=0}^{n-1-d} ones[i+d] * zeros[i]
        n_ov[n-1-d] = np.sum(ones[d:] * zeros[:n-d])
    n_ov[n-1] = 0  # lag 0 is always 0 for binary
    return n_ov

    # Δoverlap(d) = I(h[i-d]=1, i-d>=0) - I(h[i+d]=0, i+d<n) + I(h[j+d]=0, j+d<n) - I(h[j-d]=1, j-d>=0)

        if j + d < n and h[j+d] == 1: delta_nd -= 1
        new_ov[n-1-d] += delta_nd

    # I(h[i-d]=1, i-d>=0) for each d
    i_minus_d = i - d

    term3 = np.zeros(n-1)

    term4 = np.zeros(n-1)

    term7[valid_jmd] = 1.0 - h[j_minus_d[valid_jmd]]
    
    term8 = np.zeros(n-1)
    term8[valid_jpd] = h[j_plus_d[valid_jpd]]
    
    delta_neg = term5 - term6 + term7 - term8  # Δoverlap(-d) for d=1,...,n-1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_overlap_arr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
    return np.clip(hp, 0.0, 1.0)

def binarize(h, n):
    hb = np.zeros(n)
    order = np.argsort(-h)
    hb[order[:n // 2]] = 1.0
    return hb

def binary_overlap_counts(h, n):
    """Return array of overlap counts indexed 0..2n-2 (lag = idx - (n-1))."""
    ov = np.zeros(2 * n - 1, dtype=np.int32)
    for d in range(1, n):
        ov[n - 1 + d] = np.sum(h[:n - d] * (1.0 - h[d:]))
        ov[n - 1 - d] = np.sum(h[d:] * (1.0 - h[:n - d]))
    return ov

def swap_delta_vectors(h, i, j, n):
    """Compute delta overlap for all lags when h[i]:1->0, h[j]:0->1. Returns (delta_neg, delta_pos) for lags 1..n-1."""
    d = np.arange(1, n)
    ipd = i + d; ipd_v = ipd < n
    imd = i - d; imd_v = imd >= 0
    jpd = j + d; jpd_v = jpd < n
    jmd = j - d; jmd_v = jmd >= 0
    t1 = np.zeros(n - 1); t1[imd_v] = h[imd[imd_v]]
    t2 = np.zeros(n - 1); t2[ipd_v] = 1.0 - h[ipd[ipd_v]]
    t3 = np.zeros(n - 1); t3[jpd_v] = 1.0 - h[jpd[jpd_v]]
    t4 = np.zeros(n - 1); t4[jmd_v] = h[jmd[jmd_v]]
    delta_pos = t1 - t2 + t3 - t4
    t5 = np.zeros(n - 1); t5[ipd_v] = h[ipd[ipd_v]]
    t6 = np.zeros(n - 1); t6[imd_v] = 1.0 - h[imd[imd_v]]
    t7 = np.zeros(n - 1); t7[jmd_v] = 1.0 - h[jmd[jmd_v]]
    t8 = np.zeros(n - 1); t8[jpd_v] = h[jpd[jpd_v]]
    delta_neg = t5 - t6 + t7 - t8
    return delta_neg, delta_pos

def binary_hill_climb(h, n, deadline, max_rounds=5000):
    """Best-improvement binary local search with bottleneck targeting."""
    h = h.copy()
    ov = binary_overlap_counts(h, n).astype(np.float64)
    best_c = ov.max() * 2.0 / n
    no_improve = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        m = ov.max()
        if m < 1e-9:
            break
        # Find bottleneck lags
        bn_idx = np.where(ov >= m - 1e-9)[0]
        best_improve = 0.0
        best_swap = None
        # Try targeted swaps: reduce bottleneck
        for bi in bn_idx:
            lag = bi - (n - 1)
            if lag == 0:
                continue
            d = abs(lag)
            if lag > 0:
                src = np.where((h[:n-d] == 1) & (h[d:] == 0))[0]
                tgt = np.where((h[:n-d] == 0) & (h[d:] == 1))[0]
            else:
                src = np.where((h[d:] == 1) & (h[:n-d] == 0))[0]
                tgt = np.where((h[d:] == 0) & (h[:n-d] == 1))[0]
            # Try all src-tgt pairs (capped for speed)
            if len(src) * len(tgt) > 20000:
                idx_s = rng_sample_indices(len(src), 200)
                idx_t = rng_sample_indices(len(tgt), 200)
                src = src[idx_s]; tgt = tgt[idx_t]
            for si in src:
                if lag > 0:
                    i = si
                else:
                    i = si + d
                for tj in tgt:
                    if lag > 0:
                        j = tj
                    else:
                        j = tj
                    # Quick check: does this reduce the bottleneck?
                    if lag > 0:
                        new_bottleneck = ov[b.i] - 1 + (1 if (j < n - d and h[j + d] == 0) else 0)
                    else:
                        new_bottleneck = ov[b.i] - 1 + (1 if (j + d < n and h[j + d] == 0) else 0)
                    if new_bottleneck >= m:
                        continue
                    # Full evaluation
                    dn, dp = swap_delta_vectors(h, i, j, n)
                    new_ov = ov.copy()
                    new_ov[1:n] += dn
                    new_ov[n+1:] += dp
                    new_max = new_ov.max()
                    improve = m - new_max
                    if improve > best_improve:
                        best_improve = improve
                        best_swap = (i, j, new_ov)
        if best_swap is not None:
            i, j, new_ov = best_swap
            h[i] = 0.0; h[j] = 1.0
            ov = new_ov
            new_c = ov.max() * 2.0 / n
            if new_c < best_c:
                best_c = new_c
            no_improve = 0
        else:

def rng_sample_indices(n, k):
    if n <= k:
        return np.arange(n)
    return np.random.choice(n, k, replace=False)

def continuous_refine(h, n, deadline):
    """Refine using SLSQP with smooth max objective."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    def smooth_obj(x, alpha):
        c = compute_overlap_arr(x, n)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 300:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def sum_con(x):
        return float(x.sum() - n / 2.0)
    
    def sum_con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    for alpha in [2000, 10000, 50000]:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(alpha,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def gradient_descent(h, n, deadline, lr_init=0.01, max_steps=5000):
    """Custom gradient descent with active-set tracking."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        c = compute_overlap_arr(h, n)
        m = c.max()
        # Softmax weights for active set
        alpha = 200.0
        w = np.exp(alpha * (c - m))
        w /= w.sum()
        # Gradient of weighted sum of overlaps
        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            if w[k_idx] < 1e-10:
                continue
            lag = k_idx - (n - 1)
            dx = 2.0 / n
            if lag == 0:
                grad += w[k_idx] * dx * (1.0 - 2.0 * h)
            elif lag > 0:
                grad[:n - lag] += w[k_idx] * dx * (1.0 - h[lag:])
                grad[lag:] += w[k_idx] * dx * (-h[:n - lag])
            else:
                d = -lag
                grad[d:] += w[k_idx] * dx * (1.0 - h[:n - d])
                grad[:n - d] += w[k_idx] * dx * (-h[d:])
        # Project gradient
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return best_h, best_c

def grad_descent_step(h, n, alpha):
    c = compute_overlap_arr(h, n)
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
