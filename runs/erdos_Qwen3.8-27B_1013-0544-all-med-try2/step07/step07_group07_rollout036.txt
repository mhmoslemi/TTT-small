def slsqp_full(h, n, alpha=100000, maxiter=4000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        r = minimize(_smooth_obj_full, h, args=(n, alpha),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > deadline:
            break
        prog = True
        while prog:
            if time.time() > deadline:
                break
            prog = False
            c = compute_all_lags(h, n)
            ks = int(np.argmax(c))
            lag = ks - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                break
            if lag > 0:
                pairs = h[:n-m] * (1 - h[m:])
            else:
                pairs = h[m:] * (1 - h[:n-m])
            order = np.argsort(pairs)[::-1][:20]
            for i in order:
                if time.time() > deadline:
                    break
                j = i + m
                hs = h.copy()
                if lag > 0:
                    if h[i] > h[j]:
                        dt = min(d, (h[i]-h[j])*0.4, h[i]*0.5, (1-h[j])*0.5)
                        if dt < 1e-9:
                            continue
                        h[i] -= dt
                        h[j] += dt
                    else:
                        dt = min(d, (h[j]-h[i])*0.4, h[j]*0.5, (1-h[i])*0.5)
                        if dt < 1e-9:
                            continue
                        h[i] += dt
                        h[j] -= dt
                else:
                    if h[j] > h[i]:
                        dt = min(d, (h[j]-h[i])*0.4, h[j]*0.5, (1-h[i])*0.5)
                        if dt < 1e-9:
                            continue
                        h[i] += dt
                        h[j] -= dt
                    else:
                        dt = min(d, (h[i]-h[j])*0.4, h[i]*0.5, (1-h[j])*0.5)
                        if dt < 1e-9:
                            continue
                        h[i] -= dt
                        h[j] += dt
                cn = compute_c5(h, n)
                if cn < best_c - 1e-14:
                    best_c = cn
                    prog = True
                else:
                    h[:] = hs
    return h, best_c

def optimize(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lp_end = min(deadline - 50, time.time() + 250)
    h, c = lp_loop(h, n, lp_end, top_k=20)
    if c < best_c:
        best_c = c
        best_h = h
    if time.time() < deadline - 40:
        h2 = slsqp_solve(best_h, n, top_m=15, alpha=50000)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    if time.time() < deadline - 30:
        h2 = slsqp_solve(best_h, n, top_m=10, alpha=200000)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    if time.time() < deadline - 25:
        h2 = lp_loop(best_h, n, min(deadline - 15, time.time() + 80), top_k=15)
        if compute_c5(h2, n) < best_c:
            best_c = compute_c5(h2, n)
            best_h = h2
    if time.time() < deadline - 5:
        best_h, best_c = local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(250, 0.5)
        n0 = 250
    best_h = None
    best_c = float('inf')
    alloc1 = min(450, (deadline - t0) * 0.5)
    h_opt, c_opt = optimize(h0, n0, t0 + alloc1)
    best_h, best_c = h_opt, c_opt
    for n_try in [200, 250, 350, 400]:
        if time.time() > deadline - 130:
            break
        if n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 60
        if remaining < 80:
            break
        alloc = min(130, remaining * 0.4)
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_start = np.interp(x_new, x_best, best_h)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if best_h is not None and time.time() < deadline - 70:
        n = len(best_h)
        restart_count = 0
        while time.time() < deadline - 35 and restart_count < 15:
            scale = 0.005 + 0.008 * rng.random()
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(55, deadline - time.time() - 25)
            if alloc < 25:
                break
            h_opt, c_opt = optimize(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            restart_count += 1
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        h2 = slsqp_full(best_h, n, alpha=500000, maxiter=5000)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2
        best_h, best_c = local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(h0, n0)
        best_c = compute_c5(best_h, n0)
    return best_h, best_c, len(best_h)
