```
def project_feasible(h,n):
    h=np.asarray(h,dtype=float).copy()
    h=np.clip(h,0.0,1.0)
    target=float(n)/2.0
    for _ in range(5):
       s=float(h.sum()); diff=target-s
       if abs(diff)<1e-13: break
       if diff>0:
          room=float((1.0-h).sum())
          if room<1e-15: break
          h += (diff/room)*(1.0-h)
       else:
          if s<1e-15: break
          h += (diff/s)*h
    h=np.clip(h,0,1)
    diff=target-h.sum()
    if abs(diff)>1e-10:
       idxs=np.where((h>1e-10)&(h<1-1e-10))[0]
       if idxs.size:
          for idx in idxs:
             if diff>0:
                add=min(diff,1.0-h[idx])
                h[idx]+=add; diff-=add
             else:
                sub=min(-diff,h[idx])
                h[idx]-=sub; diff+=sub
             if abs(diff)<1e-13: break
       else:
          # last resort: adjust any with room
          if diff>0:
             idx=int(np.argmax(1-h)); h[idx]=min(1,h[idx]+diff)
          else:
             idx=int(np.argmax(h)); h[idx]=max(0,h[idx]+diff)
    return h
```

```python
import numpy as np
import time
from scipy.optimize import minimize


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(6):
        d = target - float(h.sum())
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            r = float((1.0 - h).sum())
            if r > 1e-15:
                h += (d / r) * (1.0 - h)
        else:
            s = float(h.sum())
            if s > 1e-15:
                h += (d / s) * h
    h = np.clip(h, 0.0, 1.0)
    d = target - float(h.sum())
    if abs(d) > 1e-10:
        idx = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        for i in idx:
            if d > 0.0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h


def c5_val(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def smooth_value_raw(c, alpha):
    m = float(np.max(c))
    if alpha <= 1e-8:
        return m
    sh = alpha * (c - m)
    sh = sh - float(np.max(sh))
    return m + float(np.log(np.sum(np.exp(sh)))) / alpha


def softmax_weights(c, alpha):
    m = float(np.max(c))
    if alpha <= 1e-8:
        w = np.full(c.shape, 1.0 / c.size)
        return w
    sh = alpha * (c - m)
    sh = sh - float(np.max(sh))
    w = np.exp(sh)
    s = float(w.sum())
    if s <= 1e-300:
        w = np.zeros(c.shape)
        w[c.argmax()] = 1.0
    else:
        w /= s
    return w


def grad_from_weights(h, w, n):
    w = np.asarray(w, dtype=float)
    conv_a = np.convolve(1.0 - h, w, mode="full")
    conv_b = np.convolve(h, w[::-1], mode="full")
    s = n - 1
    return conv_a[s:s + n] - conv_b[s:s + n]


def lbf_objective(x, n, alpha, mu):
    x = np.asarray(x, dtype=float)
    if np.any(np.isnan(x)):
        return 1e100
    raw = np.correlate(x, 1.0 - x, mode="full")
    s = x.sum() - 0.5 * float(n)
    return float(smooth_value_raw(raw, alpha) + mu * s * s)


def lbf_gradient(x, n, alpha, mu):
    x = np.asarray(x, dtype=float)
    raw = np.correlate(x, 1.0 - x, mode="full")
    w = softmax_weights(raw, alpha)
    g = grad_from_weights(x, w, n)
    s = x.sum() - 0.5 * float(n)
    g = g + 2.0 * mu * s
    return g


def lbfbs_optimize(h0, n, deadline):
    h = project_feasible(h0, n).copy()
    best_c = c5_val(h, n)
    bounds = [(0.0, 1.0)] * n
    scale = max(0.5, float(n) / 500.0)
    configs = (
        (2.0, 1.0),
        (10.0, 3.0),
        (40.0, 5.0),
        (150.0, 10.0),
        (600.0, 30.0),
        (2500.0, 100.0),
        (8000.0, 300.0),
    )
    for a_base, mu_base in configs:
        if time.time() > deadline - 1.5:
            break
        alpha = a_base * scale
        mu = mu_base * scale
        try:
            res = minimize(
                lbf_objective, h, jac=lbf_gradient, args=(n, alpha, mu),
                method="L-BFGS-B", bounds=bounds,
                options={"maxiter": 70, "maxfun": 170,
                         "ftol": 1e-12, "gtol": 1e-4, "maxls": 20}
            )
            h_new = project_feasible(res.x, n)
            c_new = c5_val(h_new, n)
            if c_new < best_c - 1e-13:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c


def planned_end(deadline, max_alloc, reserve, min_alloc=12.0):
    left = deadline - time.time()
    alloc = min(float(max_alloc), max(0.0, left - float(reserve)))
    if alloc > float(min_alloc):
        return time.time() + alloc
    return None


def pair_search(h0, n, rng, deadline, binary=False,
                fracs=None, alphas=None, cand_limit=None, max_stall=5):
    if fracs is None:
        fracs = (1.0,) if binary else (0.2, 0.5, 0.8, 1.0)
    if alphas is None:
        alphas = (20.0, 100.0, 500.0, 2500.0, 8000.0)
    if cand_limit is None:
        cand_limit = 1400 if n <= 600 else 650

    cur = project_feasible(h0, n).copy()
    dx = 2.0 / float(n)
    raw = np.correlate(cur, 1.0 - cur, mode="full")
    best_raw = float(np.max(raw))
    best_c = best_raw * dx

    p = max(6, min(36, int(np.sqrt(float(cand_limit) / max(1, len(fracs))))))
    scale = max(0.5, float(n) / 500.0)
    stall = 0
    rounds = 0

    while time.time() < deadline - 1.0 and stall < max_stall and rounds < 3000:
        rounds += 1
        raw = np.correlate(cur, 1.0 - cur, mode="full")
        best_raw = float(np.max(raw))
        improved = False

        for al_base in alphas:
            if time.time() > deadline - 1.0:
                break
            alpha = al_base * scale
            w = softmax_weights(raw, alpha)
            g = grad_from_weights(cur, w, n)
            g = np.where(np.isfinite(g), g, 0.0)

            low = np.argsort(g)[:p]
            high = np.argsort(g)[::-1][:p]
            tmp = cur.copy()
            best_mv = None
            best_loc = best_raw
            tested = 0

            for i in low:
                if tested >= cand_limit:
                    break
                hi = float(cur[i])
                if binary:
                    if hi > 0.5:
                        continue
                else:
                    if hi >= 1.0 - 1e-12:
                        continue

                for j in high:
                    if tested >= cand_limit:
                        break
                    if i == j:
                        continue
                    lo = float(cur[j])
                    if binary:
                        if lo < 0.5:
                            continue
                    else:
                        if lo <= 1e-12:
                            continue

                    mass = min(lo, 1.0 - hi)
                    if mass <= 1e-10:
                        continue

                    for fr in fracs:
                        d = fr * mass
                        if d < 1e-9:
                            continue
                        tmp[i] = hi + d
                        tmp[j] = lo - d
                        if -1e-10 <= tmp[i] <= 1.0 + 1e-10 and -1e-10 <= tmp[j] <= 1.0 + 1e-10:
                            rr = float(np.max(np.correlate(tmp, 1.0 - tmp, mode="full")))
                            if rr < best_loc - 1e-9:
                                best_loc = rr
                                best_mv = (i, j, d)
                        tmp[i] = hi
                        tmp[j] = lo
                        tested += 1
                        if tested >= cand_limit:
                            break

            if best_mv is not None:
                i, j, d = best_mv
                cur[i] += d
                cur[j] -= d
                if binary:
                    cur[i] = 1.0 if cur[i] > 0.5 else 0.0
                    cur[j] = 1.0 if cur[j] > 0.5 else 0.0
                cur = project_feasible(cur, n)
                best_raw = best_loc
                best_c = best_raw * dx
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1

    return cur, best_c


def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def golden_binary(n, rng):
    half = n // 2
    phi = 1.618033988749895
    off = rng.randint(0, n)
    pos = np.unique((np.arange(half) * phi + off).astype(int) % n)
    if pos.size < half:
        rem = np.setdiff1d(np.arange(n), pos)
        if rem.size:
            pos = np.concatenate((pos, rng.choice(rem, half - pos.size, replace=False)))
    h = np.zeros(n)
    h[pos[:half]] = 1.0
    return h


def fourier_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for f in range(1, 10):
        amp = rng.uniform(0.05, 0.25)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(np.pi * f * x + ph)
    return project_feasible(h, n)


def round_binary(h, n, rng):
    hh = np.asarray(h, dtype=float) + rng.normal(0.0, 1e-5, n)
    idx = np.argsort(hh)[::-1][:n // 2]
    b = np.zeros(n)
    b[idx] = 1.0
    return b


def resample_and_optimize(h0, n0, n2, rng, deadline):
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x1, x0, h0) + rng.normal(0.0, 0.012, n2)
    h2 = project_feasible(h2, n2)
    c2 = c5_val(h2, n2)

    left = deadline - time.time()
    if left > 60.0:
        end = min(time.time() + min(45.0, 0.45 * left), deadline - 25.0)
        if end > time.time() + 12.0:
            h2, c2 = lbfbs_optimize(h2, n2, end)

    left = deadline - time.time()
    if left > 30.0:
        end = min(time.time() + min(45.0, 0.50 * left), deadline - 2.0)
        if end > time.time() + 10.0:
            h2, c2 = pair_search(h2, n2, rng, end, max_stall=3)

    return h2, c5_val(project_feasible(h2, n2), n2)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(20.0, float(budget_s)) - 2.0
    rng = np.random.RandomState(int(seed))

    try:
        p = np.asarray(initial_h_values, dtype=float)
    except Exception:
        p = None

    if p is None or p.size == 0:
        n0 = 500
        h0 = fourier_start(n0, rng)
    else:
        if p.size % 2 != 0 or p.size < 100 or p.size > 1200:
            nt = 500 if (p.size < 100 or p.size % 2 != 0) else 1000
            x0 = np.linspace(0.0, 2.0, p.size, endpoint=False)
            x1 = np.linspace(0.0, 2.0, nt, endpoint=False)
            p = np.interp(x1, x0, p)
        n0 = p.size
        h0 = project_feasible(p, n0)

    best_h = h0
    best_n = n0
    best_c = c5_val(best_h, best_n)

    if deadline - time.time() < 25.0:
        return best_h, best_c, best_n

    end = planned_end(deadline, 240.0, 520.0)
    if end is not None:
        hh, cc = lbfbs_optimize(best_h, best_n, end)
        if cc < best_c:
            best_c, best_h = cc, hh

    end = planned_end(deadline, 260.0, 400.0)
    if end is not None:
        hh, cc = pair_search(best_h, best_n, rng, end, max_stall=5)
        if cc < best_c:
            best_c, best_h = cc, hh

    if deadline - time.time() > 300.0:
        hb = round_binary(best_h, best_n, rng)
        end = planned_end(deadline, 140.0, 290.0, min_alloc=18.0)
        if end is not None:
            hb, cb = pair_search(
                hb, best_n, rng, end, binary=True, fracs=(1.0,),
                alphas=(50.0, 200.0, 800.0, 3000.0),
                cand_limit=900 if best_n <= 600 else 500, max_stall=4
            )
            if cb < best_c:
                best_c, best_h = cb, hb

        end = planned_end(deadline, 70.0, 180.0, min_alloc=15.0)
        if end is not None:
            hh, cc = lbfbs_optimize(best_h, best_n, end)
            if cc < best_c:
                best_c, best_h = cc, hh

        end = planned_end(deadline, 70.0, 100.0, min_alloc=12.0)
        if end is not None:
            hh, cc = pair_search(best_h, best_n, rng, end,
                                 fracs=(0.1, 0.3, 0.7), max_stall=3)
            if cc < best_c:
                best_c, best_h = cc, hh

    starts = []
    try:
        starts.append(golden_binary(best_n, rng))
    except Exception:
        pass
    starts.append(random_binary(best_n, rng))
    starts.append(fourier_start(best_n, rng))

    for s in starts:
        if deadline - time.time() < 130.0:
            break
        end = planned_end(deadline, 40.0, 120.0, min_alloc=12.0)
        if end is not None:
            hh, cc = lbfbs_optimize(s, best_n, end)
        else:
            hh, cc = s, c5_val(s, best_n)

        end = planned_end(deadline, 50.0, 50.0, min_alloc=12.0)
        if end is not None:
            hh, cc = pair_search(
                hh, best_n, rng, end, max_stall=3,
                cand_limit=600 if best_n <= 600 else 300
            )
        else:
            hh = project_feasible(hh, best_n)
            cc = c5_val(hh, best_n)

        if cc < best_c:
            best_c, best_h = cc, hh

    if deadline - time.time() > 130.0:
        hp = project_feasible(best_h + rng.normal(0.0, 0.04, best_n), best_n)
        end = planned_end(deadline, 45.0, 70.0, min_alloc=12.0)
        if end is not None:
            hp, cc = lbfbs_optimize(hp, best_n, end)
        else:
            cc = c5_val(hp, best_n)

        end = planned_end(deadline, 45.0, 20.0, min_alloc=10.0)
        if end is not None:
            hp, cc = pair_search(hp, best_n, rng, end, max_stall=2)

        if cc < best_c:
            best_c, best_h = cc, hp

    for n2 in (800, 1000):
        if best_n >= n2:
            continue
        if deadline - time.time() < 180.0:
            break
        h2, c2 = resample_and_optimize(best_h, best_n, n2, rng, deadline)
        if c2 < best_c - 1e-12:
            best_c, best_h, best_n = c2, h2, n2
        if deadline - time.time() < 120.0:
            break

    left = deadline - time.time()
    if left > 50.0:
        end = planned_end(deadline, 30.0, 10.0, min_alloc=8.0)
        if end is not None:
            hh, cc = lbfbs_optimize(best_h, best_n, end)
            if cc < best_c:
                best_c, best_h = cc, hh

        left = deadline - time.time()
        if left > 20.0:
            hh, cc = pair_search(
                best_h, best_n, rng, min(time.time() + max(5.0, left - 1.0), deadline - 1.0),
                fracs=(0.01, 0.05, 0.2, 0.5, 1.0), max_stall=2,
                cand_limit=700 if best_n <= 600 else 350
            )
            if cc < best_c:
                best_c, best_h = cc, hh

    best_h = project_feasible(best_h, best_n)
    best_c = float(c5_val(best_h, best_n))
    return best_h, best_c, best_n
```
