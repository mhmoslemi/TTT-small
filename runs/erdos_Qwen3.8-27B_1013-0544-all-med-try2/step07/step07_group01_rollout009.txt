def compute_D(h,n):
    m=2*n-1
    D=np.zeros((n,m),float)
    D[:,n-1] = 1.0 - 2.0*h
    for k in range(1,n):
        L=n-k
        mp=n-1+k; mn=n-1-k
        D[0:L,mp] += 1.0 - h[k:k+L]
        D[k:n,mp] += -h[0:L]
        D[k:n,mn] += 1.0 - h[0:L]
        D[0:L,mn] += -h[k:k+L]

    h = np.clip(np.asarray(h, float), 0.0, 1.0).copy()
    for _ in range(300):
        d = t - h.sum()
        if abs(d) < 1e-13: return h
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15: break
            h += d / s * r
        else:
            s = h.sum()
            if s < 1e-15: break

import numpy as np
import time
from scipy.optimize import minimize

SAMPLE_TS = (0.0, 0.005, 0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.65, 0.8, 0.9, 0.975, 1.0)


def project_feasible(h, n):
    t = n / 2.0
    h = np.clip(np.asarray(h, float), 0.0, 1.0).copy()
    for _ in range(300):
        d = t - h.sum()
        if abs(d) < 1e-13:
            return h
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (d / s) * h
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if d > 0:
        for i in np.argsort(1.0 - h)[::-1]:
            a = min(d, 1.0 - h[i])
            h[i] += a
            d -= a
            if d <= 1e-15:
                break
    elif d < 0:
        d = -d
        for i in np.argsort(h)[::-1]:
            a = min(d, h[i])
            h[i] -= a
            d -= a
            if d <= 1e-15:
                break
    return h


def c_counts(h):
    return np.correlate(h, 1.0 - h, mode="full")


def c5_from_c(c, n):
    return float(c.max()) * (2.0 / float(n))


def compute_D(h, n):
    m = 2 * n - 1
    D = np.zeros((n, m), dtype=float)
    D[:, n - 1] = 1.0 - 2.0 * h
    for k in range(1, n):
        L = n - k
        a = n - 1 + k
        b = n - 1 - k
        D[:L, a] += 1.0 - h[k:]
        D[k:, a] -= h[:L]
        D[k:, b] += 1.0 - h[:L]
        D[:L, b] -= h[k:]
    return D


def softmax_weights(c, alpha):
    x = alpha * (c - c.max())
    x -= x.max()
    w = np.exp(x)
    s = w.sum()
    if s < 1e-300:
        return np.full(c.shape, 1.0 / c.size)
    return w / s


def pair_f_exact(c, diff, work, m0, mp, mn, d, delta):
    np.multiply(diff, delta, out=work)
    work += c
    dd = delta * delta
    work[m0] -= 2.0 * dd
    if d:
        work[mn] += dd
        work[mp] += dd
    return float(work.max())


def best_delta_work(c, diff, work, m0, mp, mn, d, lo, hi, iters=14):
    if hi <= lo + 1e-14:
        return 0.0, float(c.max())
    M0 = float(c.max())
    best = M0
    bestd = 0.0
    span = hi - lo
    for tt in SAMPLE_TS:
        de = lo + tt * span
        val = pair_f_exact(c, diff, work, m0, mp, mn, d, de)
        if val < best - 1e-14:
            best = val
            bestd = de
    if best >= M0 - 1e-12:
        return 0.0, M0
    rad = max(0.2 * span, abs(bestd) * 0.25)
    a = max(lo, bestd - rad)
    b = min(hi, bestd + rad)
    if b - a < 1e-8:
        for de in (lo, 0.5 * (lo + hi), hi):
            val = pair_f_exact(c, diff, work, m0, mp, mn, d, de)
            if val < best:
                best = val
                bestd = de
        return bestd, best
    ph = 0.6180339887498949
    c1 = b - ph * (b - a)
    c2 = a + ph * (b - a)
    f1 = pair_f_exact(c, diff, work, m0, mp, mn, d, c1)
    f2 = pair_f_exact(c, diff, work, m0, mp, mn, d, c2)
    if f1 < best:
        best = f1
        bestd = c1
    if f2 < best:
        best = f2
        bestd = c2
    for _ in range(max(0, iters)):
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - ph * (b - a)
            f1 = pair_f_exact(c, diff, work, m0, mp, mn, d, c1)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + ph * (b - a)
            f2 = pair_f_exact(c, diff, work, m0, mp, mn, d, c2)
        if f1 < best:
            best = f1
            bestd = c1
        if f2 < best:
            best = f2
            bestd = c2
    de = 0.5 * (a + b)
    val = pair_f_exact(c, diff, work, m0, mp, mn, d, de)
    if val < best:
        best = val
        bestd = de
    return bestd, best


def random_shake(h, n, scale, rng):
    z = rng.normal(0.0, 1.0, n)
    z -= z.mean()
    return project_feasible(h + scale * z, n)


def binary_start(n, rng):
    h = np.zeros(n, dtype=float)
    k = int(round(n / 2.0))
    if 0 < k < n:
        h[rng.choice(n, k, replace=False)] = 1.0
    return project_feasible(h, n)


def sine_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    amp = 0.35 + 0.15 * rng.rand()
    h = 0.5 + amp * np.sin(np.pi * x + 2.0 * rng.rand())
    return project_feasible(h, n)


def local_pair_opt(h, n, deadline, seed=0, top_pairs=700, line_iters=14, shake=True):
    rng = np.random.RandomState(seed)
    h = project_feasible(h, n).copy()
    c = c_counts(h)
    best_h = h.copy()
    best_c = c.copy()
    best_M = float(c.max())
    m = 2 * n - 1
    work = np.empty(m, dtype=float)
    diff = np.empty(m, dtype=float)
    stag = 0
    shakes = 0

    while time.time() < deadline:
        D = compute_D(h, n)
        M = float(c.max())
        alpha = 1000.0 * (1.0 + 0.2 * min(stag, 8))
        alpha = min(alpha, 200000.0)
        w = softmax_weights(c, alpha)
        g = D.T @ w

        iu = np.triu_indices(n, 1)
        if iu[0].size == 0:
            break
        sc = np.abs(g[iu[0]] - g[iu[1]])
        p = min(int(top_pairs), int(sc.size))
        if p <= 0:
            break
        if p < sc.size:
            idx = np.argpartition(sc, -p)[-p:]
        else:
            idx = np.arange(sc.size)
        if p > 1:
            idx = idx[np.argsort(sc[idx])[::-1]]
        pi = iu[0][idx]
        pj = iu[1][idx]

        if stag and p < sc.size:
            rc = min(200, max(10, p // 4))
            if rc > 0:
                ri = rng.randint(0, n, rc)
                rj = rng.randint(0, n, rc)
                mk = ri != rj
                pi = np.concatenate((pi, ri[mk]))
                pj = np.concatenate((pj, rj[mk]))

        improved = False
        for pos in range(pi.size):
            if time.time() > deadline:
                break
            i = int(pi[pos])
            j = int(pj[pos])
            if i == j:
                continue
            np.subtract(D[i], D[j], out=diff)
            d = abs(i - j)
            m0 = n - 1
            mp = n - 1 + d
            mn = n - 1 - d
            lo = max(-float(h[i]), float(h[j] - 1.0))
            hi = min(1.0 - float(h[i]), float(h[j]))
            if hi - lo < 1e-10:
                continue
            de, val = best_delta_work(c, diff, work, m0, mp, mn, d, lo, hi, int(line_iters))
            if de != 0.0 and val < M - 1e-12:
                h2 = h.copy()
                h2[i] += de
                h2[j] -= de
                h2 = project_feasible(h2, n)
                c2 = c_counts(h2)
                M2 = float(c2.max())
                if M2 < M - 1e-13:
                    h = h2
                    c = c2
                    M = M2
                    D = compute_D(h, n)
                    improved = True
                    if M2 < best_M - 1e-14:
                        best_M = M2
                        best_h = h.copy()
                        best_c = c.copy()

        if improved:
            stag = 0
            shakes = 0
        else:
            stag += 1
            if shake and shakes < 3 and stag >= 2:
                h = random_shake(h, n, 0.01 * (2.0 ** shakes), rng)
                c = c_counts(h)
                shakes += 1
                stag = 0
            elif stag > 5:
                break

    best_h = project_feasible(best_h, n)
    best_c = c_counts(best_h)
    return best_h, best_c


def soft_obj(x, n, alpha):
    c = c_counts(x)
    m = float(c.max())
    sh = alpha * (c - m)
    sh -= sh.max()
    return m + float(np.log(np.sum(np.exp(sh)))) / alpha


def soft_grad(x, n, alpha):
    c = c_counts(x)
    m = float(c.max())
    sh = alpha * (c - m)
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    D = compute_D(x, n)
    return D.T @ w


def pen_obj(x, n, alpha, mu):
    return soft_obj(x, n, alpha) + mu * (float(x.sum()) - 0.5 * n) ** 2


def pen_grad(x, n, alpha, mu):
    return soft_grad(x, n, alpha) + 2.0 * mu * (float(x.sum()) - 0.5 * n)


def smooth_lbfgs(h, n, deadline, alpha=10000.0, mu=10.0, maxiter=35):
    h = project_feasible(h, n).copy()
    best_c = c_counts(h)
    best_M = float(best_c.max())
    maxiter = int(max(1, min(60, maxiter)))
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            pen_obj,
            h,
            jac=pen_grad,
            args=(n, alpha, mu),
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": maxiter, "ftol": 1e-14, "gtol": 1e-12,
                     "maxfun": max(10, 10 * maxiter), "maxls": 40},
        )
        x = getattr(res, "x", None)
        if x is not None and np.all(np.isfinite(x)):
            h2 = project_feasible(x, n)
            c2 = c_counts(h2)
            M2 = float(c2.max())
            if M2 < best_M - 1e-13:
                return h2, c2
    except Exception:
        pass
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 4.0
    rng = np.random.RandomState(seed)

    have_parent = True
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
        if n < 2:
            raise ValueError
    except Exception:
        have_parent = False
        n = 300
        h = sine_start(n, rng)

    h = project_feasible(h, n)
    c = c_counts(h)
    best_h = h.copy()
    best_obj = c5_from_c(c, n)

    if not have_parent and time.time() < deadline - 100.0:
        h2, c2 = smooth_lbfgs(h, n, deadline - 100.0, alpha=10000.0, mu=10.0, maxiter=60)
        o2 = c5_from_c(c2, n)
        if o2 < best_obj - 1e-15:
            best_obj = o2
            best_h = h2.copy()

    if time.time() < deadline - 120.0:
        t1 = min(deadline - 100.0, time.time() + 350.0)
        h1, c1 = local_pair_opt(h, n, t1, seed=seed + 1, top_pairs=750,
                                line_iters=14, shake=have_parent)
        o1 = c5_from_c(c1, n)
        if o1 < best_obj - 1e-15:
            best_obj = o1
            best_h = h1.copy()

    if time.time() < deadline - 90.0:
        rem = deadline - time.time()
        smooth_s = min(35.0, rem - 70.0)
        if smooth_s > 10.0:
            h2, c2 = smooth_lbfgs(best_h, n, time.time() + smooth_s,
                                  alpha=20000.0, mu=10.0,
                                  maxiter=int(min(35, max(10, smooth_s * 0.8))))
            o2 = c5_from_c(c2, n)
            if o2 < best_obj - 1e-15:
                best_obj = o2
                best_h = h2.copy()
            if time.time() < deadline - 50.0:
                t3 = min(deadline - 30.0, time.time() + 110.0)
                h3, c3 = local_pair_opt(h2, n, t3, seed=seed + 2, top_pairs=500,
                                        line_iters=12, shake=True)
                o3 = c5_from_c(c3, n)
                if o3 < best_obj - 1e-15:
                    best_obj = o3
                    best_h = h3.copy()

    candidates = []
    if deadline - time.time() > 90.0:
        candidates.append(random_shake(best_h, n, 0.015, rng))
        candidates.append(random_shake(best_h, n, 0.04, rng))
        candidates.append(binary_start(n, rng))

    for idx, hc in enumerate(candidates):
        rem = deadline - time.time()
        if rem < 55.0:
            break
        alloc = min(70.0, rem - 40.0)
        if alloc < 25.0:
            break
        if idx >= 2 and rem > 110.0:
            hs, cs = smooth_lbfgs(hc, n, time.time() + min(12.0, alloc * 0.3),
                                  alpha=10000.0, mu=10.0, maxiter=15)
            hc = hs
        ho, co = local_pair_opt(hc, n, time.time() + alloc, seed=seed + 101 + idx,
                                top_pairs=350, line_iters=10, shake=True)
        oo = c5_from_c(co, n)
        if oo < best_obj - 1e-15:
            best_obj = oo
            best_h = ho.copy()

    if time.time() < deadline - 2.0:
        hz, cz = local_pair_opt(best_h, n, deadline - 2.0, seed=seed + 999,
                                top_pairs=1000, line_iters=10, shake=False)
        oz = c5_from_c(cz, n)
        if oz < best_obj:
            best_obj = oz
            best_h = hz.copy()

    best_h = project_feasible(best_h, n)
    best_obj = c5_from_c(c_counts(best_h), n)
    return best_h, best_obj, n
