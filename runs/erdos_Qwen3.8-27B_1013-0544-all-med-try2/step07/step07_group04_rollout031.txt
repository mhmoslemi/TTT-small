11. **Alternative: Subgradient / proximal method**

    Since we want to minimize max_k c_k, and each c_k is a quadratic function of h (bilinear), we can use a subgradient approach:
    
    - At each iteration, compute c = correlate(h, 1-h)
    - Find k* = argmax c
    - Compute the subgradient of c_{k*} with respect to h
    - Take a step in the negative subgradient direction
    - Project back to the feasible set

    The subgradient of c_{k*} = ∑_i h[i]*(1-h[i+k*]) with respect to h[m] is:
    - From first factor: (1-h[m+k*]) (if m+k* is valid)
    - From second factor: -h[m-k*] (if m-k* is valid)
    
    So grad[m] = (1-h[m+k*]) - h[m-k*] (with out-of-range terms as 0)
    
    Step: h_new = h - α * grad
    Project: clip to [0,1], adjust sum to N/2.

    - Take a step in -grad direction
    - Project

from scipy.optimize import linprog
import numpy as np

def minimax_lp(h, n, deadline, n_iter=1000):
    """Iterative linearization of the minimax problem."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        cur_max = c.max()
        
        # Find worst shift
        k_star = np.argmax(c)
        s = k_star - (n - 1)
        
        # Gradient of c[k_star] w.r.t. h
        grad = np.zeros(n)
        if s >= 0:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] -= h[:n-s]
        else:
            ss = -s
            grad[ss:] += (1.0 - h[:n-ss])
            grad[:ss] -= h[ss:]
        
        # Linearize: c[k_star](h_new) ≈ cur_max + grad · (h_new - h)
        # Want to minimize t s.t. cur_max + grad·(h_new - h) ≤ t
        # i.e., grad·h_new ≤ t - cur_max + grad·h
        # RHS = t - cur_max + c[k_star] = t (since grad·h = c[k_star] for the linear part... wait no)
        # Actually grad·h = ∑ grad[m]*h[m] which is NOT equal to c[k_star].
        # c[k_star] = ∑ h[i]*(1-h[i+s])
        # grad[m] = (1-h[m+s]) - h[m-s]
        # grad·h = ∑ [(1-h[m+s]) - h[m-s]] * h[m] = ∑ h[m] - ∑ h[m]h[m+s] - ∑ h[m-s]h[m]
        # This is not simply c[k_star].
        
        # Let me just use the linear constraint directly:
        # c[k_star](h) ≤ t
        # Linear approximation at h: c[k_star](h) ≈ c[k_star](h) + grad·(h_new - h)
        # = cur_max + grad·h_new - grad·h
        # So: grad·h_new ≤ t - cur_max + grad·h
        
        rhs_const = -cur_max + np.dot(grad, h)
        
        # LP: min t
        # s.t. grad·h_new ≤ t + rhs_const  →  grad·h_new - t ≤ rhs_const
        #      0 ≤ h_new ≤ 1
        #      sum(h_new) = n/2
        #
        # Variables: h_new (n vars) + t (1 var) = n+1 vars
        # 
        # Objective: minimize t → c = [0, ..., 0, 1]
        # 
        # Inequality: grad·h_new - t ≤ rhs_const
        #   A_ub = [grad, -1], b_ub = [rhs_const]
        # 
        # Also need c[k] ≤ t for other top shifts for stability
        # Let's add constraints for top 5 shifts
        top_idx = np.argsort(c)[::-1][:5]
        A_ub_rows = []
        b_ub_vals = []
        for idx in top_idx:
            s_i = idx - (n - 1)
            g_i = np.zeros(n)
            if s_i >= 0:
                g_i[:n-s_i] += (1.0 - h[s_i:])
                g_i[s_i:] -= h[:n-s_i]
            else:
                ss_i = -s_i
                g_i[ss_i:] += (1.0 - h[:n-ss_i])
                g_i[:ss_i] -= h[ss_i:]
            rhs_i = -c[idx] + np.dot(g_i, h)
            A_ub_rows.append(np.concatenate([g_i, [-1.0]]))
            b_ub_vals.append(rhs_i)
        
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        
        # sum(h_new) = n/2: equality constraint
        A_eq = np.hstack([np.ones(n), [0.0]])
        b_eq = np.array([n / 2.0])
        
        # Bounds: 0 ≤ h_i ≤ 1, t ≥ 0
        bounds = [(0, 1)] * n + [(0, None)]
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        
        if res.status == 0:
            h_new = res.x[:n]
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
        else:
            # LP failed, try a small subgradient step
            h = project_feasible(h - 0.01 * grad, n)
    
    return best_h, best_c

2. For that shift, identify all "bad" positions (i where h[i]=1, h[i+k*]=0)
3. Try to fix each bad position by either:
   a. Setting h[i+k*] = 1 (and some other h[j] = 0)
   b. Setting h[i] = 0 (and some other h[k] = 1)
4. Only consider the most promising candidates

This reduces the search from O(n²) to O(n * n_bad) where n_bad is much smaller than n.

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def shift_grad(h, n, s):
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        ss = -s
        grad[ss:] += (1.0 - h[:n - ss])
        grad[:ss] -= h[ss:]
    return grad

def lp_minimax(h, n, deadline, max_iters=3000, top_k=8):
    bx = h.copy()
    best_c = compute_c5(bx, n)
    best_h = bx.copy()
    cut_indices = []
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(bx, 1 - bx, mode='full')
        top = np.argsort(c)[::-1][:top_k]
        new_cuts = [int(t) for t in top if int(t) not in cut_indices]
        cut_indices.extend(new_cuts)
        if len(cut_indices) > 30:
            cut_indices = [int(t) for t in np.argsort(c)[::-1][:20]]
        A_rows = []
        b_vals = []
        for ci in cut_indices:
            s = ci - (n - 1)
            g = shift_grad(bx, n, s)
            rhs = -c[ci] + np.dot(g, bx)
            A_rows.append(np.concatenate([g, [-1.0]]))
            b_vals.append(rhs)
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        A_eq = np.hstack([np.ones(n), [0.0]])
        b_eq = np.array([n / 2.0])
        bounds = [(0, 1)] * n + [(0, None)]
        obj = np.zeros(n + 1)
        obj[-1] = 1.0
        try:
            res = linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                break
            h_new = np.clip(res.x[:n], 0, 1)
            h_new = project_feasible(h_new, n)
            cn = compute_c5(h_new, n)
            if cn < best_c:
                best_c = cn
                best_h = h_new.copy()
            if np.max(np.abs(h_new - bx)) < 1e-10:
                break
            bx = h_new
        except Exception:
            break
    return best_h, best_c

def subgrad_method(h, n, deadline, top_k=15, n_iter=50000):
    bx = h.copy()
    best_c = compute_c5(bx, n)
    best_h = bx.copy()
    lr = 1.0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = np.correlate(bx, 1 - bx, mode='full')
        top = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for t in top:
            s = int(t) - (n - 1)
            grad += shift_grad(bx, n, s)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        bx = bx - (lr / gn) * grad
        bx = project_feasible(bx, n)
        cn = compute_c5(bx, n)
        if cn < best_c:
            best_c = cn
            best_h = bx.copy()
            lr = min(lr * 1.03, n * 0.3)
        else:
            lr = max(lr * 0.97, 0.05)
    return best_h, best_c

def binary_search(h_init, n, deadline, max_sweeps=2000, seed=42, n_restarts=4):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            pb = rng.randint(5, 15)
            for _ in range(pb):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                a = rng.choice(oi)
                b = rng.choice(zi)
                h[a], h[b] = 0.0, 1.0
        corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
        cur_c = corr.max() * dx
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        for _ in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            worst_k = int(np.argmax(corr))
            worst_s = worst_k - (n - 1)
            if worst_s >= 0:
                bad_ones = [i for i in range(n - worst_s) if h[i] == 1 and h[i + worst_s] == 0]
            elif worst_s < 0:
                ss = -worst_s
                bad_ones = [i for i in range(ss, n) if h[i] == 1 and h[i - ss] == 0]
            else:
                bad_ones = ones[:]
            rng.shuffle(bad_ones)
            zh = zeros[:]
            rng.shuffle(zh)
            tried = set()
            for i in bad_ones:
                if i >= n or h[i] != 1:
                    continue
                oi = i
                for j in zh:
                    if j >= n or h[j] != 0 or (oi, j) in tried:
                        continue
                    tried.add((oi, j))
                    im = oi + m_range
                    imn = oi - m_range
                    jm = j + m_range
                    jmn = j - m_range
                    delta = np.zeros(2 * n - 1)
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    msk = (imn >= 0) & (imn < n)
                    delta[msk] -= h[imn[msk]]
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - oi)
                    if 0 <= k1 < 2 * n - 1:
                        delta[k1] += 1.0
                    k2 = n - 1 + (oi - j)
                    if 0 <= k2 < 2 * n - 1:
                        delta[k2] += 1.0
                    nm = (corr + delta).max() * dx
                    if nm < cur_c - 1e-15:
                        h[oi], h[j] = 0.0, 1.0
                        corr += delta
                        cur_c = nm
                        ones.remove(oi)
                        ones.append(j)
                        zeros.remove(j)
                        zeros.append(oi)
                        improved = True
                        break
            if improved:
                ones = list(np.where(h == 1)[0])
                zeros = list(np.where(h == 0)[0])
            else:
                for j in zeros:
                    for oi in ones:
                        im = oi + m_range
                        imn = oi - m_range
                        jm = j + m_range
                        jmn = j - m_range
                        delta = np.zeros(2 * n - 1)
                        msk = (im >= 0) & (im < n)
                        delta[msk] += 1.0 - h[im[msk]]
                        msk = (imn >= 0) & (imn < n)
                        delta[msk] -= h[imn[msk]]
                        msk = (jm >= 0) & (jm < n)
                        delta[msk] -= 1.0 - h[jm[msk]]
                        msk = (jmn >= 0) & (jmn < n)
                        delta[msk] += h[jmn[msk]]
                        delta[n - 1] -= 2.0
                        k1 = n - 1 + (j - oi)
                        if 0 <= k1 < 2 * n - 1:
                            delta[k1] += 1.0
                        k2 = n - 1 + (oi - j)
                        if 0 <= k2 < 2 * n - 1:
                            delta[k2] += 1.0
                        nm = (corr + delta).max() * dx
                        if nm < cur_c - 1e-15:
                            h[oi], h[j] = 0.0, 1.0
                            corr += delta
                            cur_c = nm
                            ones = list(np.where(h == 1)[0])
                            zeros = list(np.where(h == 0)[0])
                            improved = True
                            break
                    if improved:
                        break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
