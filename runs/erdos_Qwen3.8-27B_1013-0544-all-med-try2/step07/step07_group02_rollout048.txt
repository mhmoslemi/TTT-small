    free_mask = (h > free_thresh) & (h < 1 - free_thresh)

    fixed_sum = h[~free_mask].sum()
    free_target = n / 2.0 - fixed_sum

        h_new[free_idx] = v

    c5 = compute_c5(h5, n)
    if c5 < best_c:
        best_c, best_h = c5, h5.copy()

    # Phase 6: Very tight active set SLSQP (tiny free set)
    p6 = t0 + 0.80 * total
    alphas_tight = [500000, 5000000, 50000000, 500000000]
    h6 = active_set_slsqp(best_h, n, 0.02, alphas_tight, p6)
    c6 = compute_c5(h6, n)
    if c6 < best_c:
        best_c, best_h = c6, h6.copy()

    # Phase 7: Full SLSQP for final polish
    p7 = t0 + 0.90 * total
    h7 = full_slsqp(best_h, n, [1000000, 10000000], p7)
    c7 = compute_c5(h7, n)
    if c7 < best_c:
        best_c, best_h = c7, h7.copy()

    # Phase 8: Final coordinate descent + tight active set
    p8 = dl
    h8, _ = coord_descent(best_h, n, max_sweeps=150, deadline=p8 - 10)
    h8 = project_feasible(h8, n)
    c8 = compute_c5(h8, n)
    if c8 < best_c:
        best_c, best_h = c8, h8.copy()
    if time.time() < dl - 20:
        h8b = active_set_slsqp(best_h, n, 0.01, [10000000, 100000000], dl - 5)
        c8b = compute_c5(h8b, n)
        if c8b < best_c:
            best_c, best_h = c8b, h8b.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
