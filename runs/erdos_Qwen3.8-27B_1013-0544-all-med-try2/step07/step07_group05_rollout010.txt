import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_corr(h, n, p):
    r = np.arange(-(n - 1), n)
    idx1 = p + r
    v1 = (idx1 >= 0) & (idx1 < n)
    t1 = np.zeros(2 * n - 1)
    t1[v1] = 1.0 - h[idx1[v1]]
    idx2 = p - r
    v2 = (idx2 >= 0) & (idx2 < n)
    t2 = np.zeros(2 * n - 1)
    t2[v2] = -h[idx2[v2]]
    return t1 + t2

def smooth_max_obj(x, n, alpha):
    c = compute_corr(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def find_best_d(c, delta, d_lo, d_hi, top_k):
    d_act = delta[top_k]
    c_act = c[top_k]
    cands = [d_lo, d_hi, 0.0]
    m = len(top_k)
    for a in range(m):
        for b in range(a + 1, m):
            dd = d_act[a] - d_act[b]
            if abs(dd) > 1e-13:
                dk = (c_act[b] - c_act[a]) / dd
                if d_lo < dk < d_hi:
                    cands.append(dk)
    best_d = 0.0
    best_v = np.max(c)
    for dc in cands:
        v = np.max(c + dc * delta)
        if v < best_v - 1e-15:
            best_v = v
            best_d = dc
    return best_d

def pair_search(h, n, deadline, top_m=8):
    best_c = compute_c5(h, n)
    n_corr = 2 * n - 1
    for sweep in range(500):
        if time.time() > deadline:
            break
        c = compute_corr(h, n)
        best_c = c.max()
        top_k = np.argsort(-c)[:top_m]
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            gi = grad_corr(h, n, i)
            for j in range(i + 1, n):
                if time.time() > deadline:
                    return h, best_c
                gj = grad_corr(h, n, j)
                delta = gi - gj
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_lo >= d_hi - 1e-12:
                    continue
                best_d = find_best_d(c, delta, d_lo, d_hi, top_k)
                if abs(best_d) > 1e-12:
                    h[i] += best_d
                    h[j] -= best_d
                    c_new = compute_corr(h, n)
                    c_max = c_new.max()
                    if c_max < best_c - 1e-13:
                        best_c = c_max
                        c = c_new
                        improved = True
                        gi = grad_corr(h, n, i)
                        top_k = np.argsort(-c)[:top_m]
                    else:
                        h[i] -= best_d
                        h[j] += best_d
        if not improved:
            break
    return h, best_c

def optimize_grid(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000]
    for a in alphas:
        if time.time() > deadline - 60:
            break

        c = compute_c5(ht, n)

            h = ht

            h = ht
