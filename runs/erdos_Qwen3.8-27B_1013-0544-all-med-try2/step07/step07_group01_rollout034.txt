def critical_shift_grad(h, n, c, threshold):
    """Compute gradient of c[k] only for shifts k where c[k] >= threshold"""
    grad = np.zeros(n)
    for k in range(2*n-1):
        if c[k] < threshold:
            continue
        kk = k - (n-1)
        if kk >= 0:
            # c[kk] = sum_{i=0}^{n-1-kk} h[i]*(1-h[i+kk])

                grad[i+kk] += (1.0 - h[i])

        h_try, c_try = lbfgs_refine(h_try, n, min(time.time() + 15, deadline), alpha=200000)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Annealed momentum GD (main optimization)
    if time.time() < deadline - 500:
        t1 = min(time.time() + 350, deadline - 500)
        best_h, best_c = anneal_gd(best_h, n_init, t1, alpha_start=50, alpha_end=200000, lr=0.004)

    # Phase 2: L-BFGS-B with high alpha
    if time.time() < deadline - 250:
        t2 = min(time.time() + 80, deadline - 250)
        h2, c2 = lbfgs_refine(best_h, n_init, t2, alpha=200000)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Gradient-guided pair search
    if time.time() < deadline - 150:
        t3 = min(time.time() + 80, deadline - 150)
        best_h, best_c = gradient_pair_search(best_h, n_init, t3)

    # Phase 4: Standard pair search with fine steps
    if time.time() < deadline - 80:
        t4 = min(time.time() + 100, deadline - 80)
        best_h, best_c = pair_search(best_h, n_init, t4, radius=100)

    # Phase 5: Multi-resolution exploration
    for n_try in [250, 400, 500]:
        if time.time() > deadline - 60:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(50, deadline - time.time() - 20)
        if alloc < 25:
            break
        h_t, c_t = anneal_gd(h_interp, n_try, time.time() + alloc * 0.6,
                              alpha_start=100, alpha_end=100000, lr=0.003)
        if time.time() < deadline - 10:
            h_t, c_t = lbfgs_refine(h_t, n_try, time.time() + min(20, deadline - time.time() - 5), alpha=200000)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t

    # Phase 6: Final refinement loop
    if time.time() < deadline - 10:
        nf = len(best_h)
        best_h, best_c = lbfgs_refine(best_h, nf, deadline - 10, alpha=500000)

    # Phase 7: Final pair search polish
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = pair_search(best_h, nf, deadline - 2, radius=80)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
