        wm = w[m_idx] * dx

            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def _sacf(x, n):
    return float(x.sum() - n / 2.0)

def _sacj(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, max_iters=400):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 0.5), (500, 2.0), (2000, 5.0),
               (10000, 20.0), (50000, 100.0), (200000, 500.0), (1000000, 2000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 1:
            break
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': max_iters * 15,
                                   'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=200000, maxiter=800):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sacf, 'jac': _sacj, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                      args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_descent(h, n, deadline, lr_init=0.005, alpha_init=500.0):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    alpha = alpha_init
    for _ in range(80000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, alpha, 0.0)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        step_lr = lr
        improved = False
        for _ in range(50):
            h_new = project_feasible(h - step_lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step_lr *= 0.5
        if not improved:
            lr *= 0.5
            alpha *= 1.3
        if lr < 1e-13:
            break
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n0 = 300

    best_h = project_feasible(h_init, n0)
    best_c = compute_c5(best_h, n0)

    # Phase 1: Optimize on original grid
    t1 = min(t0 + budget_s * 0.30, deadline - 250)
    best_h, best_c = lbfgs_optimize(best_h, n0, t1, max_iters=300)

    if time.time() < deadline - 200:
        t2 = min(time.time() + 80, deadline - 200)
        h2, c2 = slsqp_optimize(best_h, n0, t2, alpha=300000, maxiter=600)
        if c2 < best_c:
            best_c = c2; best_h = h2

    if time.time() < deadline - 150:
        t3 = min(time.time() + 50, deadline - 150)
        h3, c3 = grad_descent(best_h, n0, t3, lr_init=0.004, alpha_init=500.0)
        if c3 < best_c:
            best_c = c3; best_h = h3

    if time.time() < deadline - 100:
        t4 = min(time.time() + 50, deadline - 100)
        best_h, best_c = pair_search(best_h, n0, t4,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005], radius=35)

    # Phase 2: Try random perturbation restart on original grid
    if time.time() < deadline - 200:
        h_pert = best_h + 0.02 * rng.randn(n0)
        h_pert = project_feasible(h_pert, n0)
        t5 = min(time.time() + 60, deadline - 200)
        hp, cp = lbfgs_optimize(h_pert, n0, t5, max_iters=200)
        if cp < best_c:
            best_c = cp; best_h = hp

    # Phase 3: Upscale to 2x grid
    n1 = n0 * 2
    x_old = np.linspace(0, 2, n0)
    x_new = np.linspace(0, 2, n1)
    h_up = np.interp(x_new, x_old, best_h)
    h_up = project_feasible(h_up, n1)
    c_up = compute_c5(h_up, n1)

    t6 = min(time.time() + budget_s * 0.30, deadline - 150)
    h_up, c_up = lbfgs_optimize(h_up, n1, t6, max_iters=200)
    if c_up < best_c:
        best_c = c_up; best_h = h_up

    if time.time() < deadline - 100:
        t7 = min(time.time() + 70, deadline - 100)
        h7, c7 = slsqp_optimize(best_h, len(best_h), t7, alpha=500000, maxiter=400)
        if c7 < best_c:
            best_c = c7; best_h = h7

    if time.time() < deadline - 60:
        t8 = min(time.time() + 40, deadline - 60)
        h8, c8 = grad_descent(best_h, len(best_h), t8, lr_init=0.002, alpha_init=2000.0)
        if c8 < best_c:
            best_c = c8; best_h = h8

    # Phase 4: Downsample 2x back to 1x and re-optimize
    if len(best_h) > n0 and time.time() < deadline - 80:
        x_hi = np.linspace(0, 2, len(best_h))
        x_lo = np.linspace(0, 2, n0)
        h_down = np.interp(x_lo, x_hi, best_h)
        h_down = project_feasible(h_down, n0)
        t9 = min(time.time() + 40, deadline - 80)
        hd, cd = lbfgs_optimize(h_down, n0, t9, max_iters=200)
        if cd < best_c:
            best_c = cd; best_h = hd

    # Phase 5: Final refinement
    if time.time() < deadline - 10:
        nf = len(best_h)
        if time.time() < deadline - 8:
            best_h, best_c = lbfgs_optimize(best_h, nf, deadline - 8, max_iters=100)
        if time.time() < deadline - 3 and nf <= 400:
            best_h, best_c = pair_search(best_h, nf, deadline - 2,
                                         [0.005, 0.002, 0.001, 0.0005], radius=25)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
