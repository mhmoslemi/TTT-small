```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx), int(np.argmax(c))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n):
    """Compute the subgradient of C5 (up to the dx factor which we handle separately)"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    m_star = int(np.argmax(corr))
    # m_star is index in [0, 2n-2], corresponding to shift m = m_star - (n-1)
    m = m_star - (n - 1)
    # c[m] = sum_i h[i+m] * (1-h[i]) for valid i
    # ∂c[m]/∂h[j] = (1-h[j-m]) * 1_{0≤j-m<n} - h[j+m] * 1_{0≤j+m<n}
    grad = np.zeros(n)
    jm = np.arange(n) + m  # j + m for each j
    jmm = np.arange(n) - m  # j - m for each j
    mask_jm = (jm >= 0) & (jm < n)
    mask_jmm = (jmm >= 0) & (jmm < n)
    grad[mask_jmm] += (1 - h[jmm[mask_jmm]])
    grad[mask_jm] -= h[jm[mask_jm]]
    return grad, m_star, corr

def nesterov_subgrad(h, n, max_iter, deadline, init_lr=0.1):
    """Nesterov accelerated subgradient descent"""
    dx = 2.0 / n
    best_c = compute_c5(h, n)[0]
    best_h = h.copy()
    v = np.zeros(n)
    h_curr = h.copy()
    h_para = h.copy()  # parallel iterate for Nesterov
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Nesterov: evaluate at lookahead point
        h_eval = h_para + v
        h_eval = np.clip(h_eval, 0, 1)
        h_eval = project_feasible(h_eval, n)
        
        # Compute subgradient
        grad, m_star, corr = subgradient_step(h_eval, n)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        
        # Polyak step size
        c_eval = np.max(corr) * dx
        # Use a decaying step size
        lr = init_lr / (np.sqrt(it + 1))
        
        v = v - lr * grad / g_norm
        h_para = np.clip(h_para + v, 0, 1)
        h_para = project_feasible(h_para, n)
        
        # Evaluate
        c_curr = compute_c5(h_para, n)[0]
        if c_curr < best_c - 1e-10:
            best_c = c_curr
            best_h = h_para.copy()
    
    return best_h, best_c
```

```python
            # After: h[im] * (1-1) = 0

        else:

            # After: h[jm] * (1-0) = h[jm]
            delta += h[jm]
    
    return delta
```

```python
    return x.sum() - _target_sum

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def lse_gradient(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    weights = np.exp(sh - sm)
    weights /= weights.sum()
    # grad = sum_k weights[k] * dx * d(c_k)/dx[j]
    # c_k has shift m_k = k - (n-1)
    shifts = np.arange(2*n-1) - (n-1)  # shift for each correlation index
    grad = np.zeros(n)
    # For each k, ∂c_k/∂x[j] = (1-x[j+m_k])*1_{valid} - x[j-m_k]*1_{valid}
    # Sum over k with weights
    # Vectorize: for each j, sum_k weights[k] * [(1-x[j+m_k])*1_{0<=j+m_k<n} - x[j-m_k]*1_{0<=j-m_k<n}]
    # This is O(n * 2n) = O(n^2)
    for k in range(2*n-1):
        if weights[k] < 1e-15:
            continue
        mk = shifts[k]
        wk = weights[k] * dx
        # (1-x[j+mk]) for valid j+mk
        jm = np.arange(n) + mk
        mask = (jm >= 0) & (jm < n)
        if mask.any():
            grad[mask] += wk * (1.0 - x[jm[mask]])
        jmm = np.arange(n) - mk
        mask2 = (jmm >= 0) & (jmm < n)
        if mask2.any():
            grad[mask2] -= wk * x[jmm[mask2]]
    return grad

                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
```

For a fixed i (zero position), I want to find the best j (one position) to swap with. The delta for pair (i, j) at shift m is:

Δc[m] = (contribution from i) + (contribution from j) + (cross terms)

Contribution from i (setting h[i] from 0 to 1):

Contribution from j (setting h[j] from 1 to 0):
= -(1-h[j+m])*1_{valid(j+m)} + h[j-m]*1_{valid(j-m)}

Cross terms (when the two changes interact):
- If i+m = j: the (1-h[i+m]) term should use h[j]_after = 0, so (1-0)=1 instead of (1-1)=0. Extra: +1.
  Wait, I already have (1-h[i+m]) where i+m=j, so (1-h[j]) = (1-1) = 0. But after the swap, h[j]=0, so it should be (1-0)=1. So the correction is +1.

- If j+m = i: the -(1-h[j+m]) term should use h[i]_after = 1, so -(1-1)=0 instead of -(1-0)=-1. Extra: +1.

So:
Δc[m] = f_i(m) + g_j(m) + cross(i,j,m)

where:
f_i(m) = (1-h[i+m])*1_{valid(i+m)} - h[i-m]*1_{valid(i-m)}
g_j(m) = -(1-h[j+m])*1_{valid(j+m)} + h[j-m]*1_{valid(j-m)}
cross(i,j,m) = 1_{i+m=j} + 1_{i-m=j} + 1_{j+m=i} + 1_{j-m=i}

Then for step 4, for each j, I need max(corr + f_i + G[j_idx, :] + cross[j_idx, :]). This is O(n) per j, or O(n²/2) for all j.

So the total is O(n²) per i (dominated by step 4), and O(n³) total. Same complexity.

The only way to make it faster is to avoid checking all j for each i. The K-set trick helps here: for a given i, if I can determine that no j will give an improvement (based on the K-set), I can skip i entirely.

Alternatively, I can use a **greedy approach with limited search**: for each i, only check the "most promising" j values (e.g., those that reduce the max correlation the most).

Let me think about a different angle: **use the full delta vector and find the best swap via matrix operations**.

For all pairs (i, j), the delta vector is:
Δ[i,j] = f_i + g_j + cross[i,j]

where f_i and g_j are vectors of length 2n-1, and cross[i,j] is sparse.

The new max for pair (i,j) is: max(corr + Δ[i,j]) = max(corr + f_i + g_j + cross[i,j])

If I ignore the cross term (which is at most 4), I can approximate:
max(corr + f_i + g_j) = max over m of (corr[m] + f_i[m] + g_j[m])

For a fixed i, this is: max over j of [max over m of (corr[m] + f_i[m] + g_j[m])]

Let W[m] = corr[m] + f_i[m] (fixed for given i).
Then I need: min over j of max over m of (W[m] + g_j[m])

This is a "minimax" over j. If I precompute for each m the value (W[m] + g_j[m]) for all j, I get a matrix M of size (2n-1) × (n/2). The max over m for each j gives a vector of length n/2. The minimum of this vector gives the best j.

Computing M: M[m, j] = W[m] + g_j[m] = W[m] + (-(1-h[j+m])*1_{valid} + h[j-m]*1_{valid})

For all m and j: this is a (2n-1) × (n/2) matrix. Computing it:
- W is a (2n-1) vector, broadcast to all j: O(n²)
- g_j[m] for all j, m: O(n²)

Then M = W[:,None] + G (where G is the g matrix): O(n²)
max over m: O(n * n/2) = O(n²/2)
argmin: O(n/2)

Total per i: O(n²)
Total for all i: O(n³)

Still O(n³). But with good NumPy vectorization, the constant is much smaller than the pure Python loops.

For n=250: O(n³) = 15.6M operations, but vectorized as matrix ops (250 × 125 matrix), each taking ~1μs, so ~250μs per i, and 62.5ms per full sweep. That's much better than the Python loop version!

Let me implement this vectorized approach:

```python
def vectorized_binary_sweep(h, corr, n, shifts, deadline):
    """One sweep of binary local search, vectorized."""
    dx = 2.0 / n
    cur_c = corr.max() * dx
    ones_list = np.where(h == 1)[0]
    zeros_list = np.where(h == 0)[0]
    n_ones = len(ones_list)
    n_zeros = len(zeros_list)
    
    # Precompute g_j[m] for all ones j and all shifts m
    # G[j_idx, m_idx] = -(1-h[j+m])*1_{valid(j+m)} + h[j-m]*1_{valid(j-m)}
    G = np.zeros((n_ones, 2*n-1))
    for ji, j in enumerate(ones_list):
        jm = j + shifts
        mask_jm = (jm >= 0) & (jm < n)
        if mask_jm.any():
            G[ji, mask_jm] -= (1.0 - h[jm[mask_jm]])
        jmm = j - shifts
        mask_jmm = (jmm >= 0) & (jmm < n)
        if mask_jmm.any():
            G[ji, mask_jmm] += h[jmm[mask_jmm]]
    
    improved = False
    for i in zeros_list:
        if time.time() > deadline:
            break
        # Compute f_i[m]
        fi = np.zeros(2*n-1)
        jm = i + shifts
        mask_jm = (jm >= 0) & (jm < n)
        if mask_jm.any():
            fi[mask_jm] += (1.0 - h[jm[mask_jm]])
        jmm = i - shifts
        mask_jmm = (jmm >= 0) & (jmm < n)
        if mask_jmm.any():
            fi[mask_jmm] -= h[jmm[mask_jm]]  # bug: should be h[jmm[mask_jmm]]
        
        # W = corr + fi
        W = corr + fi
        # M = W[:, None] + G  (shape: (2n-1, n_ones))
        M = W[:, None] + G
        # For each j, the new max is max(M[:, j])
        new_maxes = M.max(axis=0)  # shape: (n_ones,)
        new_cs = new_maxes * dx
        
        # Find best j
        best_ji = np.argmin(new_cs)
        if new_cs[best_ji] < cur_c - 1e-14:
            j = ones_list[best_ji]
            # Apply cross-term correction
            # Check if cross terms matter
            cross = 0
            if i + (j - i) == j and 0 <= 0 < 2*n-1:  # i+m=j where m=j-i
                # cross term at m = j-i
                pass  # already handled in G? Let me check.
            # Actually, the cross terms: when i+m=j, the fi term uses (1-h[j])=0
            # but should use (1-h[j]_after)=1. So fi at m=j-i is off by 1.
            # Similarly for other cross terms.
            # Let me just do a full recomputation for the accepted pair.
            
            # Full delta
            dw = np.zeros(2*n-1)
            for idx_m, mk in enumerate(shifts):
                d = 0.0
                jm2 = i + mk
                if 0 <= jm2 < n:
                    d += 1.0 if jm2 == j else (1.0 - h[jm2])
                jmm2 = i - mk
                if 0 <= jmm2 < n:
                    d += -1.0 if jmm2 == j else (-h[jmm2])
                jm3 = j + mk
                if 0 <= jm3 < n:
                    d += -1.0 if jm3 == i else (-(1.0 - h[jm3]))
                jmm3 = j - mk
                if 0 <= jmm3 < n:
                    d += 1.0 if jmm3 == i else h[jmm3]
                dw[idx_m] = d
            
            new_corr = corr + dw
            nc = new_corr.max() * dx
            if nc < cur_c - 1e-14:
                h[i] = 1.0
                h[j] = 0.0
                corr = new_corr
                cur_c = nc
                # Recompute G (since h changed)
                ones_list = np.where(h == 1)[0]
                n_ones = len(ones_list)
                G = np.zeros((n_ones, 2*n-1))
                for ji2, j2 in enumerate(ones_list):
                    jm4 = j2 + shifts
                    mask4 = (jm4 >= 0) & (jm4 < n)
                    if mask4.any():
                        G[ji2, mask4] -= (1.0 - h[jm4[mask4]])
                    jmm4 = j2 - shifts
                    mask5 = (jmm4 >= 0) & (jmm4 < n)
                    if mask5.any():
                        G[ji2, mask5] += h[jmm4[mask5]]
                improved = True
                break
    
    return h, corr, cur_c, improved
```
