def project_box_sum(h, target):
    h = np.array(h, dtype=float).copy()

        if x.sum() > target:

        else:
            hi = mid
    x = np.clip(h - hi, 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-14:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)

        row.fill(0)
        row[p:n + p] += ommh
        row[n - 1 - p:2 * n - 1 - p] -= h
    return B


def update_Bmat_coord(B, n, x, delta_h):
    off = n - 1
    rows = np.arange(n)
    val = -int(delta_h)
    B[rows, off + x - rows] += val
    B[rows, off - x + rows] += val

                continue
            base[:] = B16[z, :]
            base += c16
            base -= B16[j]
            base[:, off] -= 2
            base[:, off + j - z] += 1
            base[:, off + z - j] += 1
            nm = base.max(axis=1)
            pos = int(np.argmin(nm))
            val = int(nm[pos])
            if val < best_new:
                best_new = val
                best_move = (j, int(z[pos]))
                if val <= Cm - 2:
                    early = True
                    break

        if best_move is not None:
            j, i = best_move
            delta = Bmat[i].astype(np.int16) - Bmat[j].astype(np.int16)
            delta[off] -= 2
            delta[off + j - i] += 1
            delta[off + i - j] += 1
            c = (c16 + delta).astype(np.int16)
            Cm = int(c.max())
            h[i] = 1
            h[j] = 0
            update_Bmat_coord(Bmat, n, i, 1)
            update_Bmat_coord(Bmat, n, j, -1)
            if j in ones:
                ones.remove(j)
            if i in zeros:
                zeros.remove(i)
            ones.append(i)
            zeros.append(j)
            moves += 1
            no_improve = 0
            if Cm < best_Cm:
                best_Cm = Cm
                best_h = h.copy()
            if best_Cm * dx < TARGET_C - 1e-12:
                return best_h.astype(float), float(best_Cm * dx), best_Cm
        else:
            no_improve += 1
            if no_improve == 1 and kick_budget > 0 and len(ones) and len(zeros):
                j = int(ones[rng.randint(len(ones))])
                if h[j] == 1:
                    z = np.asarray(zeros, dtype=np.int64)
                    if z.size:
                        B16 = Bmat.astype(np.int16)
                        c16 = c.astype(np.int16)
                        base = np.empty((z.size, L), dtype=np.int16)
                        base[:] = B16[z, :]
                        base += c16
                        base -= B16[j]
                        base[:, off] -= 2
                        base[:, off + j - z] += 1
                        base[:, off + z - j] += 1
                        nm = base.max(axis=1)
                        pos = int(np.argmin(nm))
                        if int(nm[pos]) <= Cm + 1:
                            i = int(z[pos])
                            c = base[pos].copy().astype(np.int16)
                            Cm = int(c.max())
                            h[i] = 1
                            h[j] = 0
                            update_Bmat_coord(Bmat, n, i, 1)
                            update_Bmat_coord(Bmat, n, j, -1)
                            if j in ones:
                                ones.remove(j)
                            if i in zeros:
                                zeros.remove(i)
                            ones.append(i)
                            zeros.append(j)
                            kick_budget -= 1
                            no_improve = 0
    return best_h.astype(float), float(best_Cm * dx), best_Cm


def coord_interval_ok(B, c, neg, pos, a, t):
    lb = 0.0
    if np.any(neg):
        v = a + float(np.max((t - c[neg]) / B[neg]))
        if v > lb:
            lb = v
    ub = 1.0
    if np.any(pos):
        v = a + float(np.min((t - c[pos]) / B[pos]))
        if v < ub:
            ub = v
    return lb <= ub + 1e-10


def coordinate_opt_x(B, c, a, M, tol=1e-6):
    pos = B > 1e-12
    neg = B < -1e-12
    zero = ~(pos | neg)
    zmax = float(c[zero].max()) if np.any(zero) else -1e300
    t_hi = M - tol
    if zmax >= t_hi:
        return None
    if not coord_interval_ok(B, c, neg, pos, a, t_hi):
        return None
    lo = zmax - 1e-7 if zmax > -1e299 else -1.0
    hi = t_hi
    for _ in range(18):
        mid = 0.5 * (lo + hi)
        if coord_interval_ok(B, c, neg, pos, a, mid):
            hi = mid
        else:
            lo = mid
    t = hi + 1e-9
    lb = 0.0
    if np.any(neg):
        lb = max(lb, float(a + np.max((t - c[neg]) / B[neg])))
    ub = 1.0
    if np.any(pos):
        ub = min(ub, float(a + np.min((t - c[pos]) / B[pos])))
    x = 0.5 * (lb + ub)
    if x < 0.0:
        x = 0.0
    if x > 1.0:
        x = 1.0
    return float(x)


def coord_descent(h, n, deadline, max_sweeps=60, seed=1):
    h = project_box_sum(np.asarray(h, dtype=float), 0.5 * n)
    c = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = float(c.max())
    best_h = h.copy()
    best_C = M * (2.0 / n)
    plus = (1.0 - h[::-1]).copy()
    B = np.empty(2 * n - 1, dtype=float)
    rng = np.random.RandomState(seed)

    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            a = float(h[i])
            B.fill(0.0)
            B[i:n + i] += plus
            B[n - 1 - i:2 * n - 1 - i] -= h
            x = coordinate_opt_x(B, c, a, M, tol=1e-6)
            if x is not None and abs(x - a) > 1e-12:
                c += B * (x - a)
                h[i] = x
                plus[n - 1 - i] = 1.0 - x
                M = float(c.max())
        hp = project_box_sum(h, 0.5 * n)
        cp = np.correlate(hp, 1.0 - hp, mode="full")
        Cp = float(cp.max()) * (2.0 / n)
        if Cp < best_C - 1e-12:
            best_C = Cp
            best_h = hp.copy()
        h = hp
        c = cp.astype(float)
        M = float(c.max())
        plus = (1.0 - h[::-1]).copy()
    return best_h, best_C


def smooth_F_and_w(x, n, alpha):
    x = np.asarray(x, dtype=float)
    corr = np.correlate(x, 1.0 - x, mode="full")
    g = (2.0 / n) * corr
    m = float(g.max())
    L = 2 * n - 1
    if alpha <= 1e-8:
        w = np.full(L, 1.0 / L, dtype=float)
        return m, w, corr
    e = np.exp(-alpha * (m - g))
    s = float(e.sum())
    if s <= 0.0:
        w = np.zeros(L, dtype=float)
        w[int(np.argmax(g))] = 1.0
    else:
        w = e / s
    F = m + float(np.log(s)) / alpha
    return F, w, corr


def smooth_gradient(w, h, n):
    off = n - 1
    dx = 2.0 / n
    A = 1.0 - np.asarray(h, dtype=float)
    T1 = np.convolve(w, A, mode="full")[off:off + n]
    Wr = w[::-1]
    T2 = np.convolve(Wr, A * 0.0 + np.asarray(h, dtype=float), mode="full")[off:off + n]
    return dx * (T1 - T2)


def smooth_value(x, n, alpha):
    return smooth_F_and_w(x, n, alpha)[0]


def smooth_jac(x, n, alpha):
    _, w, _ = smooth_F_and_w(x, n, alpha)
    return smooth_gradient(w, x, n)


def sum_con_fun(x, n):
    return float(np.sum(x)) - 0.5 * n


def sum_con_jac(x, n):
    return np.ones(int(n), dtype=float)


def slsqp_sequence(h, n, alphas, deadline, maxiter=160):
    x = project_box_sum(np.asarray(h, dtype=float), 0.5 * n)
    cval, _ = compute_c5(x, n)
    best_h = x.copy()
    best_c = cval
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con_fun, "jac": sum_con_jac, "args": (n,)}]

    for alpha in alphas:
        if time.time() > deadline - 25:
            break
        try:
            res = minimize(
                smooth_value,
                x,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                jac=smooth_jac,
                args=(n, alpha),
                options={"maxiter": maxiter, "ftol": 1e-13},
            )
        except Exception:
            break
        if not np.all(np.isfinite(res.x)):
            break
        x = project_box_sum(res.x, 0.5 * n)
        cval, _ = compute_c5(x, n)
        if cval < best_c - 1e-12:
            best_c = cval
            best_h = x.copy()
    return best_h, best_c


def sigmoid(z):
    z = np.asarray(z, dtype=float)
    out = np.empty_like(z)
    pos = z >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-z[pos]))
    ez = np.exp(z[~pos])
    out[~pos] = ez / (1.0 + ez)
    return out


def h_to_z(h):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 1e-4, 1.0 - 1e-4)
    return np.log(h / (1.0 - h))


def z_objective(z, n, alpha, mu):
    h = sigmoid(z)
    F, _, _ = smooth_F_and_w(h, n, alpha)
    s = float(h.sum()) - 0.5 * n
    return float(F) + float(mu) * s * s


def z_jacobian(z, n, alpha, mu):
    h = sigmoid(z)
    _, w, _ = smooth_F_and_w(h, n, alpha)
    gh = smooth_gradient(w, h, n)
    s = float(h.sum()) - 0.5 * n
    return gh + (2.0 * mu * s) * h * (1.0 - h)


def lfsb_sequence(h, n, alphas, deadline, mu=500.0, maxiter=120):
    z = h_to_z(np.asarray(h, dtype=float))
    bounds = [(-10.0, 10.0)] * n
    best_h = project_box_sum(h, 0.5 * n)
    best_c, _ = compute_c5(best_h, n)

    for alpha in alphas:
        if time.time() > deadline - 25:
            break
        try:
            res = minimize(
                z_objective,
                z,
                method="L-BFGS-B",
                bounds=bounds,
                jac=z_jacobian,
                args=(n, alpha, mu),
                options={"maxiter": maxiter, "ftol": 1e-12, "gtol": 1e-8, "maxls": 20},
            )
        except Exception:
            break
        if not np.all(np.isfinite(res.x)):
            break
        z = np.clip(res.x, -10.0, 10.0)
        hp = project_box_sum(sigmoid(z), 0.5 * n)
        cval, _ = compute_c5(hp, n)
        if cval < best_c - 1e-12:
            best_c = cval
            best_h = hp.copy()
    return best_h, best_c


def symmetrize(h, n):
    return project_box_sum(0.5 * (np.asarray(h, dtype=float) + 1.0 - np.asarray(h, dtype=float)[::-1]), 0.5 * n)


def better_candidate(hc, cval, nval, best_h, best_c, best_n):
    if cval < best_c - 1e-12:
        return project_box_sum(hc, 0.5 * nval), float(cval), int(nval)
    return best_h, best_c, best_n


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_stop = t0 + max(1.0, float(budget_s))
    deadline = hard_stop - 6.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h0))
    except Exception:
        n = 400
        xx = np.linspace(0.0, 2.0, n)
        h0 = 0.5 + 0.5 * np.sin(2.0 * np.pi * xx)

    if n < 4:
        n = 400
        xx = np.linspace(0.0, 2.0, n)
        h0 = 0.5 + 0.5 * np.sin(2.0 * np.pi * xx)
    if n % 2:
        n -= 1
    h0 = project_box_sum(h0, 0.5 * n)
    best_c, _ = compute_c5(h0, n)
    best_h = h0.copy()

    rem0 = max(1.0, deadline - t0)
    dl_bin = t0 + 0.22 * rem0
    hb, cb, _ = binary_local_search(h0, n, min(dl_bin, deadline), seed=seed + 1)
    best_h, best_c, n = better_candidate(hb, cb, n, best_h, best_c, n)
    bin_h = hb.copy()

    dl_bin2 = t0 + 0.31 * rem0
    if time.time() < dl_bin2 - 20:
        h2 = project_box_sum(h0 + 0.035 * rng.randn(n), 0.5 * n)
        hb2, cb2, _ = binary_local_search(h2, n, min(dl_bin2, deadline), seed=seed + 2)
        best_h, best_c, n = better_candidate(hb2, cb2, n, best_h, best_c, n)
        if cb2 < cb:
            bin_h = hb2.copy()

    dl_coord1 = t0 + 0.45 * rem0
    hc, cc = coord_descent(best_h, n, min(dl_coord1, deadline), max_sweeps=55, seed=seed + 3)
    best_h, best_c, n = better_candidate(hc, cc, n, best_h, best_c, n)

    hs = symmetrize(best_h, n)
    cvals, _ = compute_c5(hs, n)
    best_h, best_c, n = better_candidate(hs, cvals, n, best_h, best_c, n)

    hmx = project_box_sum(0.5 * (best_h + np.asarray(bin_h, dtype=float)), 0.5 * n)
    cmx, _ = compute_c5(hmx, n)
    best_h, best_c, n = better_candidate(hmx, cmx, n, best_h, best_c, n)

    dl_lr = t0 + 0.78 * rem0
    alphas_lr = [50, 200, 1000, 5000, 25000, 100000, 500000]
    hl, cl = lfsb_sequence(best_h, n, alphas_lr, min(dl_lr, deadline), mu=500.0, maxiter=120)
    best_h, best_c, n = better_candidate(hl, cl, n, best_h, best_c, n)

    dl_coord2 = t0 + 0.88 * rem0
    hc2, cc2 = coord_descent(best_h, n, min(dl_coord2, deadline), max_sweeps=35, seed=seed + 4)
    best_h, best_c, n = better_candidate(hc2, cc2, n, best_h, best_c, n)

    dl_sls = t0 + 0.975 * rem0
    if time.time() < dl_sls - 35:
        alphas_sl = [50000, 250000, 1000000]
        hs2, cs2 = slsqp_sequence(best_h, n, alphas_sl, min(dl_sls, deadline), maxiter=140)
        best_h, best_c, n = better_candidate(hs2, cs2, n, best_h, best_c, n)

    if time.time() < deadline - 80 and n >= 120:
        for nh in [320, 288]:
            if nh % 2:
                nh -= 1
            if time.time() > deadline - 55 or nh < 40:
                break
            xo = np.linspace(0.0, 2.0, n)
            xn = np.linspace(0.0, 2.0, nh)
            hh = project_box_sum(np.interp(xn, xo, best_h), 0.5 * nh)
            hbh, cbh, _ = binary_local_search(hh, nh, min(deadline, time.time() + 35), seed=seed + 57 + nh)
            best_h, best_c, n = better_candidate(hbh, cbh, nh, best_h, best_c, n)
            if time.time() < deadline - 25:
                hcr, ccr = coord_descent(best_h, n, min(deadline, time.time() + 12), max_sweeps=18, seed=seed + 71 + nh)
                best_h, best_c, n = better_candidate(hcr, ccr, n, best_h, best_c, n)

    best_h = project_box_sum(best_h, 0.5 * n)
    best_c, _ = compute_c5(best_h, n)
    return best_h, float(best_c), int(n)
