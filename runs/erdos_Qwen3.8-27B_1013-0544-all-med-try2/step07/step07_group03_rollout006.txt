    def expit(x):

        pos = x >= 0
        out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))

        out[~pos] = ex / (1.0 + ex)

def project_box_sum(y, target):
    y = np.asarray(y, dtype=float).copy()
    n = y.size
    if n == 0:
        return y
    target = float(target)
    if target < 0.0: target = 0.0
    if target > float(n): target = float(n)
    y = np.nan_to_num(y, nan=0.0, posinf=5.0, neginf=-5.0)
    lo = -float(np.max(y))
    hi = 1.0 - float(np.min(y))

        order = np.argsort(room)[::-1] if err > 0 else np.argsort(room)

def compute_c5(h, n):

    if n <= 0:
        return 0.0

    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full')

def to_binary(h):
    n = h.size
    h = np.asarray(h, dtype=float)
    b = np.zeros(n, dtype=float)
    if n > 0:
        idx = np.argsort(h)[::-1]
        b[idx[:n // 2]] = 1.0
    return b

def interpolate_h(h, n, m):
    n = int(n); m = int(m)
    if m % 2 == 1: m -= 1
    if m <= 0: return np.zeros(0)
    if n == m:
        return np.asarray(h, dtype=float).copy()
    xo = (np.arange(n) + 0.5) * (2.0 / float(n))
    xn = (np.arange(m) + 0.5) * (2.0 / float(m))
    return np.interp(xn, xo, np.asarray(h, dtype=float))

def smooth_value(h, alpha, dx, ones):
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(q))
    alpha = float(alpha)
    if alpha <= 1.0:
        return m
    z = alpha * (q - m)
    zmax = float(np.max(z))
    s = float(np.sum(np.exp(z - zmax)))
    if s <= 0.0:
        return m
    return m + (zmax + float(np.log(s))) / alpha

def smooth_grad(h, alpha, dx, ones):
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(q))
    alpha = float(alpha)
    if alpha <= 1.0:
        # use plain max gradient: choose max lag
        r = int(np.argmax(q))
        p = np.zeros(q.size)
        p[r] = 1.0
    else:
        z = alpha * (q - m)
        zmax = float(np.max(z))
        w = np.exp(z - zmax)
        s = float(np.sum(w))
        p = w / s if s > 0 else w
    n = h.size
    c = n - 1
    P = p
    R = P[::-1]
    A = np.convolve(ones, P, mode='full')[c:c + n]
    B = np.convolve(h, R, mode='full')[c:c + n]
    C = np.convolve(h, P, mode='full')[c:c + n]
    return dx * (A - B - C)

def h_to_v(h, vmax):
    a = np.clip(np.asarray(h, dtype=float), 1e-12, 1.0 - 1e-12)
    v = np.log(a / (1.0 - a))
    return np.clip(v, -float(vmax), float(vmax))

def h_phi_from_v(v, st):
    vc = np.asarray(v, dtype=float)
    cv = st.get('cv')
    if cv is not None and cv.shape == vc.shape:

        st['cv'] = vc.copy(); st['ch'] = h; st['cp'] = np.zeros(n)

    st['cp'] = phi
    return h, phi

def _sp_obj(v):
    st = _SP
    h, phi = h_phi_from_v(v, st)
    val = smooth_value(h, st['alpha'], st['dx'], st['ones'])
    reg = st['reg']
    if reg != 0.0:
        val += reg * float(np.dot(v, v))
    return val

def _sp_grad(v):
    st = _SP
    h, phi = h_phi_from_v(v, st)
    g = smooth_grad(h, st['alpha'], st['dx'], st['ones'])
    wsum = float(np.sum(phi))
    if wsum <= 1e-30:
        grad = np.zeros_like(v, dtype=float)
    else:
        wg = float(np.dot(phi, g))
        grad = phi * (g - wg / wsum)
    reg = st['reg']
    if reg != 0.0:
        grad = grad + 2.0 * reg * np.asarray(v, dtype=float)
    return grad

def project_box_sum(y, target):
    y = np.asarray(y, dtype=float).copy()
    n = y.size

        return y

    if target < 0.0:

def smooth_value(h, alpha, dx, ones):
    q = np.correlate(h, 1.0 - h, mode='full') * dx

    alpha = float(alpha)

    s = float(np.sum(np.exp(z - zmax)))

        return m
    return m + (zmax + float(np.log(s))) / alpha

def smooth_grad(h, alpha, dx, ones):
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(q))
    alpha = float(alpha)
    if alpha <= 1.0:
        p = np.zeros(q.size, dtype=float)
