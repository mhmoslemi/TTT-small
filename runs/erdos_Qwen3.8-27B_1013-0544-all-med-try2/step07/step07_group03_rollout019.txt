import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def subgradient_polish(h, n, steps=3000, lr=0.03, deadline=None):
    dx = 2.0 / n
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for s in range(steps):
        if deadline is not None and time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, mode='full') * dx
        m = g.max()

        w = np.exp(shifted - shifted.max())
        w /= w.sum()
        grad = np.zeros(n)
        for idx in range(2 * n - 1):

            k = idx - (n - 1)

        best_c, best_h = cp, hp.copy()
    # Phase 5: Anneal for better binary base, then re-smooth
    p5_dl = t0 + 0.82 * total
    ha, ca = anneal_binary(best_h, n, p5_dl, seed=seed + 7, n_iters=3000000)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()
    h = best_h.copy()
    for alpha in [10000, 200000, 2000000]:
        if time.time() > p5_dl - 2:
            break
        h = slsqp_optimize(h, n, alpha, 800, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 6: Try alternate resolution
    p6_dl = t0 + 0.93 * total
    for nh in [350, 450]:
        if time.time() > p6_dl - 15:
            break
        if nh % 2 != 0:
            continue
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        for alpha in [5000, 200000]:
            if time.time() > p6_dl - 10:
                break
            hh = slsqp_optimize(hh, nh, alpha, 600, 1e-15)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh
        hp2, cp2 = subgradient_polish(hh, nh, steps=1500, lr=0.03, deadline=p6_dl - 5)
        if cp2 < best_c:
            best_c, best_h, n = cp2, hp2.copy(), nh
    # Phase 7: Final polish
    p7_dl = dl - 2
    h = best_h.copy()
    for alpha in [5000000, 50000000]:
        if time.time() > p7_dl - 2:
            break
        h = slsqp_optimize(h, n, alpha, 800, 1e-17)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    hp, cp = subgradient_polish(best_h, n, steps=2000, lr=0.02, deadline=p7_dl)
    if cp < best_c:
        best_c, best_h = cp, hp.copy()
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
