import time
import numpy as np
from scipy.optimize import minimize

G_N = 300
G_M = 150
G_ALPHA = 1.0
G_MU = 1.0
G_TARGET = 150.0
PRE = {}
EMP = np.empty(0, dtype=int)

def proj(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    t = 0.5 * n
    for _ in range(100):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (d / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (d / s) * h
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.any():
            h[m[0]] += d
            h = np.clip(h, 0.0, 1.0)
    return h

def cf(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def cv(h, n):
    return float(np.max(cf(h, n)))

def sm(c, al):
    m = np.max(c)
    s = al * (c - m)
    smx = np.max(s)
    return float(m + (smx + np.log(np.sum(np.exp(s - smx)))) / al)

def gfc(c, h, n, al):
    cm = np.max(c)
    s = al * (c - cm)
    smx = np.max(s)
    w = np.exp(s - smx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    nm = n - 1
    for idx in range(2 * n - 1):
        wm = w[idx]
        if wm < 1e-13:
            continue
        wm *= dx
        k = idx - nm
        if k >= 0:
            nv = n - k
            if nv:
                g[:nv] += wm * (1.0 - h[k:n])
                g[k:n] -= wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            if nv:
                g[kk:n] += wm * (1.0 - h[:nv])
                g[:nv] -= wm * h[kk:n]
    return g

def abuild(a, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = a
    h[m:] = 1.0 - a
    return h

def full_obj(x):
    h = np.clip(x, 0.0, 1.0)
    return sm(cf(h, G_N), G_ALPHA) + G_MU * (h.sum() - G_TARGET) ** 2

def full_jac(x):
    h = np.clip(x, 0.0, 1.0)
    g = gfc(cf(h, G_N), h, G_N, G_ALPHA)
    return g + 2.0 * G_MU * (h.sum() - G_TARGET)

def anti_obj(x):
    h = abuild(np.clip(x, 0.0, 1.0), G_N)
    return sm(cf(h, G_N), G_ALPHA)

def anti_jac(x):
    a = np.clip(x, 0.0, 1.0)
    h = abuild(a, G_N)
    g = gfc(cf(h, G_N), h, G_N, G_ALPHA)
    return g[:G_M] - g[G_M:]

def prep(n):
    if n in PRE:
        return PRE[n]
    bi = [EMP] * n
    bo = [EMP] * n
    si = [EMP] * n
    so = [EMP] * n
    for i in range(n):
        if i + 1 < n:
            bo[i] = np.arange(i + 1, n)
            bi[i] = np.arange(n, 2 * n - 1 - i)
        if i:
            so[i] = np.arange(i)
            si[i] = np.arange(n - 1 - i, n - 1)
    PRE[n] = (bi, bo, si, so)
    return PRE[n]

def applyd(c, h, i, d, pr, dx, n):
    old = h[i]
    bi, bo, si, so = pr[i]
    if bi.size:
        c[bi] += d * dx * (1.0 - h[bo])
    if si.size:
        c[si] -= d * dx * h[so]
    new = old + d
    c[n - 1] += dx * (new - new * new - old + old * old)
    h[i] = new

def evp(c, h, i, di, j, dj, pr, n, dx):
    oc = c.copy()
    oi = h[i]
    oj = h[j]
    applyd(c, h, i, di, pr, dx, n)
    applyd(c, h, j, dj, pr, dx, n)
    v = float(np.max(c))
    h[i] = oi
    h[j] = oj
    c[:] = oc
    return v

    h = abuild(a, n)
    dx = 2.0 / n

def anti_project(h):
    n = len(h)
    m = n // 2
    a = 0.5 * (h[:m] + 1.0 - h[m:])
    return np.clip(a, 0.0, 1.0)

def sym_start(n, seed):
    m = n // 2
    amp = 1.0 / np.sqrt(3.0)
    k = m // 2
    b = np.full(m, -amp)
    b[:k] = amp
    if m % 2:
        b[-1] = 0.0
    rng = np.random.default_rng(seed + 17)
    b = rng.permutation(b)
    return 0.5 * (1.0 + b)

def ramp_start(n):
    m = n // 2
    b = np.linspace(-1.0, 1.0, m)
    return 0.5 + 0.5 * b

def blk_start(n):
    m = n // 2
    amp = 0.5 / np.sqrt(3.0)
    return np.concatenate((np.full(m // 2, 0.5 - amp), np.full(m - m // 2, 0.5 + amp)))

    m = n // 2
    x = np.arange(m)
