    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = c.max()
    shifted = (c - m) / T
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    log_sum = max_sh + np.log(exp_sh.sum())
    lse_val = m + T * log_sum
    w = exp_sh / exp_sh.sum()  # length 2n-1, w[k+n-1] corresponds to shift k
    
    grad = np.zeros(n)
    # For each k, w[k+n-1] * (1-h[j+k]) contributes to grad[j] for valid j
    # and w[k+n-1] * h[j-k] is subtracted
    
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        # Term 1: wk * (1-h[j+k]) for j where 0 <= j+k < n
        # j ranges from max(0, -k) to min(n-1, n-1-k)
        j_start = max(0, -k)
        j_end = min(n, n - k)
        if j_end > j_start:
            grad[j_start:j_end] += wk * (1.0 - h[j_start+k:j_end+k])
        # Term 2: -wk * h[j-k] for j where 0 <= j-k < n
        # j ranges from max(0, k) to min(n-1, n-1+k)
        j_start2 = max(0, k)
        j_end2 = min(n, n + k)
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= wk * h[j_start2-k:j_end2-k]
    
    grad *= dx
    return lse_val, grad

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_objective_and_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = c.max()
    shifted = (c - m) / T
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    log_sum = max_sh + np.log(exp_sh.sum())
    lse_val = m + T * log_sum
    w = exp_sh / exp_sh.sum()
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        js1 = max(0, -k)
        je1 = min(n, n - k)
        if je1 > js1:
            grad[js1:je1] += wk * (1.0 - h[js1+k:je1+k])
        js2 = max(0, k)
        je2 = min(n, n + k)
        if je2 > js2:
            grad[js2:je2] -= wk * h[js2-k:je2-k]
    grad *= dx
    return lse_val, grad

def projected_gradient(h, n, T, max_iter, deadline, lr_init=0.01):
    """Projected gradient descent with backtracking on LSE approximation."""
    best_h = h.copy()
    best_val = lse_objective_and_grad(h, n, T)[0]
    lr = lr_init
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        val, grad = lse_objective_and_grad(h, n, T)
        
        # Backtracking line search
        for _ in range(50):
            new_h = project_feasible(h - lr * grad, n)
            new_val = lse_objective_and_grad(new_h, n, T)[0]
            if new_val < val - 1e-12:
                h = new_h
                break
            lr *= 0.5
        else:
            break
            
        new_val = lse_objective_and_grad(h, n, T)[0]
        if new_val < best_val:
            best_val = new_val
            best_h = h.copy()
        lr = min(lr * 1.1, lr_init)  # slowly increase learning rate
    
    return best_h, best_val

A faster approach: for each "hot" shift k* (the one that achieves the max), we want to find a swap that reduces c[k*]. The condition for a swap (i:0→1, j:1→0) to reduce c[k*] is:
Δc[k*] < 0
(1-h[i+k*]) - h[i-k*] - (1-h[j+k*]) + h[j-k*] < 0

This depends on the values at positions i+k*, i-k*, j+k*, j-k*. We can precompute for each i (zero position) the "cost" of making it 1, and for each j (one position) the "cost" of making it 0, for the critical shift k*.

Let me define:
- cost1[i] = (1-h[i+k*]) - h[i-k*] (with boundary handling: if i+k* is out of range, treat h as 0; same for i-k*)
- cost0[j] = (1-h[j+k*]) - h[j-k*]

Then Δc[k*] = cost1[i] - cost0[j], and we want cost1[i] < cost0[j].

The best swap for shift k* would be: minimize cost1[i] - cost0[j] = min_i cost1[i] - max_j cost0[j].
If this is negative, we have an improving swap.

But we also need to make sure the swap doesn't increase other correlations above the current max. This is the hard part.

A practical approach: just try the best swap for the critical shift, and check if it actually reduces the overall max. If not, try the next best, etc.

This would be O(n log n) per swap attempt (to find min cost1 and max cost0), and we only need to check O(n) shifts to verify. So O(n²) per accepted swap, and we hope to accept enough swaps.

def binary_local_search(h, n, deadline, max_iter=10000):
    """Fast binary local search focusing on the critical shift."""
    dx = 2.0 / n
    h = h.copy()
    # Ensure binary
    target = n // 2
    # Round to binary keeping n/2 ones
    sorted_idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[sorted_idx[:target]] = 1.0
    h = h_b
    
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    cur_c = corr.max() * dx
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Find the critical shifts
        k_star = np.argmax(corr)  # 0 to 2n-2
        k = k_star - (n - 1)  # actual shift: -(n-1) to (n-1)
        
        # Find best swap for this shift
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # cost1[i] = (1-h[i+k]) - h[i-k] for i in zeros
        # cost0[j] = (1-h[j+k]) - h[j-k] for j in ones
        # (with boundary: h[out_of_range] = 0)
        
        best_delta = np.inf
        best_i = -1
        best_j = -1
        
        for i in zeros:
            c1 = 0.0
            if 0 <= i+k < n:
                c1 += (1.0 - h[i+k])
            if 0 <= i-k < n:
                c1 -= h[i-k]
            for j in ones:
                c0 = 0.0
                if 0 <= j+k < n:
                    c0 += (1.0 - h[j+k])
                if 0 <= j-k < n:
                    c0 -= h[j-k]
                delta = c1 - c0
                if delta < best_delta:
                    best_delta = delta
                    best_i = i
                    best_j = j
                    if delta < -0.5:  # good enough
                        break
            if best_delta < -0.5:
                break
        
        if best_delta >= -1e-10:
            break  # no improving swap for this shift
        
        # Apply swap and check
        h[best_i] = 1.0
        h[best_j] = 0.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        if new_corr.max() < corr.max() - 1e-12:
            corr = new_corr
            cur_c = corr.max() * dx
        else:
            # Revert
            h[best_i] = 0.0
            h[best_j] = 1.0
            # Try a random swap as perturbation
            if it % 100 == 0 and len(ones) > 0 and len(zeros) > 0:
                ri = zeros[np.random.randint(len(zeros))]
                rj = ones[np.random.randint(len(ones))]
                h[ri] = 1.0
                h[rj] = 0.0
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                if new_corr.max() < corr.max() * 1.05:  # accept slightly worse
                    corr = new_corr
                    cur_c = corr.max() * dx
                else:
                    h[ri] = 0.0
                    h[rj] = 1.0
    
    return h, cur_c

    c = corr * dx

    sh = (c - m) / T

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_val_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = c.max()
    sh = (c - m) / T
    sh_max = sh.max()
    e = np.exp(sh - sh_max)
    lse = m + T * (sh_max + np.log(e.sum()))
    w = e / e.sum()
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        k = ki - (n - 1)
        wk = w[ki]
        if wk < 1e-14:
            continue
        a = max(0, -k)
        b = min(n, n - k)
        if b > a:
            grad[a:b] += wk * (1.0 - h[a + k:b + k])
        a2 = max(0, k)
        b2 = min(n, n + k)
        if b2 > a2:
            grad[a2:b2] -= wk * h[a2 - k:b2 - k]
    grad *= dx
    return lse, grad

def grad_optimize(h, n, T, deadline, lr0=0.5, momentum=0.9):
    h = h.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    val, _ = lse_val_grad(h, n, T)
    best_val = val
    lr = lr0
    stall = 0
    while time.time() < deadline:
        lse, grad = lse_val_grad(h, n, T)
        v = momentum * v - lr * grad
        new_h = project_feasible(h + v, n)
        new_lse, _ = lse_val_grad(new_h, n, T)
        if new_lse < lse - 1e-15:
            h = new_h
            stall = 0
            if new_lse < best_val:
                best_val = new_lse
                best_h = h.copy()
        else:
            lr *= 0.5
            v *= 0.5
            stall += 1
            if stall > 30:
                break
            if lr < 1e-12:
                break
    return best_h, best_val

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

    rng = np.random.RandomState(seed)
    h = to_binary(h_init, n)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    consecutive_fail = 0
    it = 0
    while time.time() < deadline:
        it += 1
        top_ki = np.argsort(corr)[::-1][:n_top_shifts]
        found = False
        for ki in top_ki:
            k = int(ki) - (n - 1)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:

            best_d = np.inf
            bi = -1
            bj = -1
            for i in zeros:

                for j in ones:

                    d = c1 - c0

                if best_d <= -3:

                    found = True

                    break

                    h[bi] = 0.0
                    h[bj] = 1.0
        if not found:

                # Perturb: random swaps

                    i = zeros[rng.randint(len(zeros))]

                consecutive_fail = 0
