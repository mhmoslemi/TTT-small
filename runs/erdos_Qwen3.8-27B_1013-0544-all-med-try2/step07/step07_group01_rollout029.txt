    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()

                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break

    If alpha is None, use exact subgradient (gradient of the max term)."""
    c = compute_c5_vec(h, n)

    return np.clip(h, 0.0, 1.0)

def grad_c5(h, n, alpha):
    c = compute_c5_vec(h, n)
    dx = 2.0 / n
    grad = np.zeros(n)
    if alpha is None:
        m_max = c.max()
        w = (c >= m_max - 1e-12).astype(float)
        w /= w.sum()
    else:
        sh = alpha * (c - c.max())
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    g = grad_c5(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def cons_fun(x, n):
    return float(x.sum() - n / 2.0)

def cons_jac(x, n):
    return np.ones(len(x))

def lbfgs_optimize(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 10.0), (500, 50.0), (2000, 100.0), (10000, 500.0),
               (50000, 2000.0), (200000, 5000.0), (1000000, 20000.0),
               (5000000, 100000.0), (20000000, 500000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        rem = deadline - time.time() - 3
        if rem < 8:
            break
        per = min(70, rem / max(1, len(configs)))
        try:
            res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters * 15,
                                   'maxls': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def subgradient_descent(h, n, deadline, lr_init=0.004):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    for _ in range(80000):
        if time.time() > deadline:
            break
        g = grad_c5(h, n, alpha=None)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        improved = False
        step = lr
        for _ in range(25):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.8
        if lr < 1e-14:
            break
    return best_h, best_c

def gradient_pair_search(h, n, deadline, step_size=0.003):
    best_c = compute_c5(h, n)
    alphas = [50, 200, 1000, 5000, 20000]
    for alpha in alphas:
        while time.time() < deadline:
            improved = False
            g = grad_c5(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    move = -step_size * (g[i] - g[j])
                    if abs(move) < 1e-12:
                        continue
                    if 0.0 <= h[i] + move <= 1.0 and 0.0 <= h[j] - move <= 1.0:
                        h[i] += move; h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= move; h[j] += move
            if not improved:
                break
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def topk_focused_descent(h, n, deadline, k=5, lr_init=0.003):
    """Subgradient using only top-k active constraints for sharper direction."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    for _ in range(30000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        dx = 2.0 / n
        idx = np.argpartition(-c, k)[:k]
        idx = idx[np.argsort(-c[idx])]
        m_max = c.max()
        w = np.zeros(2 * n - 1)
        for ii in idx:
            w[ii] = 1.0 / k
        grad = np.zeros(n)
        for m in idx:
            kk = m - (n - 1)
            wm = w[m] * dx
            if kk >= 0:
                nv = n - kk
                grad[0:nv] += wm * (1.0 - h[kk:n])
                grad[kk:n] += wm * (-h[0:nv])
            else:
                jk = -kk
                nv = n - jk
                grad[jk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[jk:n])
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm
        improved = False
        step = lr
        for _ in range(25):
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.8
        if lr < 1e-14:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B with aggressive alpha ramp
    t1 = min(t0 + 180, deadline - 400)
    best_h, best_c = lbfgs_optimize(best_h, n_init, t1, max_iters=250)

    # Phase 2: SLSQP at high alpha
    if time.time() < deadline - 280:
        t2 = min(time.time() + 80, deadline - 280)
        for a in [200000, 1000000, 5000000]:
            if time.time() > t2:
                break
            h2, c2 = slsqp_optimize(best_h, n_init, min(time.time() + 35, t2), alpha=a, maxiter=1500)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Top-k focused subgradient
    if time.time() < deadline - 200:
        t3 = min(time.time() + 50, deadline - 200)
        h3, c3 = topk_focused_descent(best_h, n_init, t3, k=3, lr_init=0.003)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Full subgradient
    if time.time() < deadline - 150:
        t4 = min(time.time() + 40, deadline - 150)
        h4, c4 = subgradient_descent(best_h, n_init, t4, lr_init=0.002)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Gradient-guided pair search
    if time.time() < deadline - 100:
        t5 = min(time.time() + 60, deadline - 100)
        best_h, best_c = gradient_pair_search(best_h, n_init, t5, step_size=0.002)

    # Phase 6: L-BFGS very high alpha
    if time.time() < deadline - 80:
        t6 = min(time.time() + 40, deadline - 80)
        bounds = [(0.0, 1.0)] * n_init
        for alpha in [10000000, 50000000]:
            if time.time() > t6:
                break
            try:
                res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                              args=(n_init, alpha, 100000.0), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 150, 'ftol': 1e-18,
                                       'gtol': 1e-16, 'maxfun': 3000, 'maxls': 80})
                h_new = project_feasible(res.x, n_init)
                c_new = compute_c5(h_new, n_init)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 7: Pair search fine-tuning
    if time.time() < deadline - 40:
        t7 = min(time.time() + 50, deadline - 40)
        best_h, best_c = pair_search(best_h, n_init, t7,
                                     [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                     radius=40)

    # Phase 8: Try n=400 upscaling
    if time.time() < deadline - 60:
        n_try = 400
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(45, deadline - time.time() - 30)
        if alloc > 20:
            h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc * 0.6, max_iters=150)
            if time.time() < deadline - 15:
                h_t, c_t = pair_search(h_t, n_try, deadline - 10,
                                       [0.002, 0.001, 0.0005, 0.0002], radius=35)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n_init = n_try

    # Phase 9: Final subgradient polish
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = topk_focused_descent(best_h, nf, deadline - 2, k=3, lr_init=0.001)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
