def project_feasible(h, n):

            if tot < 1e-15:

    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: L-BFGS-B on parent
    t1 = min(t0 + 180, deadline - 500)
    best_h, best_c = lbfgs_optimize(best_h, best_n, t1, max_iters=600)

    # Phase 2: SLSQP polish
    if time.time() < deadline - 400:
        t2 = min(time.time() + 55, deadline - 400)
        h, c = slsqp_optimize(best_h, best_n, t2, alpha=5000000, maxiter=6000)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 3: Gradient-directed steepest
    if time.time() < deadline - 300:
        t3 = min(time.time() + 50, deadline - 300)
        h, c = grad_steepest(best_h, best_n, t3, k_top=50)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 4: Try n=650
    if time.time() < deadline - 250:
        n_new = 650
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_new)
        alloc = min(130, (deadline - time.time() - 150) * 0.55)
        if alloc > 30:
            h_t, c_t = lbfgs_optimize(h_interp, n_new, time.time() + alloc, max_iters=400)
            if time.time() < deadline - 100:
                t4 = min(time.time() + 35, deadline - 100)
                h_t, c_t = slsqp_optimize(h_t, n_new, t4, alpha=5000000, maxiter=4000)
            if c_t < best_c:
                best_c, best_h, best_n = c_t, h_t, n_new

    # Phase 5: Try n=400 (coarser, may find better shape)
    if time.time() < deadline - 200:
        n_coarse = 400
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
        h_coarse = np.interp(x_coarse, x_old, best_h)
        h_coarse = project_feasible(h_coarse, n_coarse)
        alloc = min(80, (deadline - time.time() - 100) * 0.4)
        if alloc > 20:
            h_t, c_t = lbfgs_optimize(h_coarse, n_coarse, time.time() + alloc, max_iters=400)
            if c_t < best_c:
                best_c, best_h, best_n = c_t, h_t, n_coarse

    # Phase 6: Diverse restarts at best_n
    starts = [fourier_flat_start(best_n, rng),
              golden_start(best_n),
              perturbed_start(best_h, best_n, rng, 0.02),
              perturbed_start(best_h, best_n, rng, 0.05)]
    for s in starts:
        if time.time() > deadline - 70:
            break
        alloc = min(20, (deadline - time.time() - 40) * 0.45)
        if alloc < 8:
            break
        h_s, c_s = lbfgs_optimize(s, best_n, time.time() + alloc, max_iters=300)
        if c_s < best_c:
            best_c, best_h = c_s, h_s

    # Phase 7: Intensive local pair search
    if time.time() < deadline - 15:
        best_h, best_c = local_pair_search(best_h, best_n, deadline - 10, radius=100)

    # Phase 8: Global fine pair search
    if time.time() < deadline - 3:
        best_h, best_c = global_pair_fine(best_h, best_n, deadline - 1)

    # Phase 9: Final SLSQP squeeze
    if time.time() < deadline - 2:
        h_f, c_f = slsqp_optimize(best_h, best_n, deadline - 1, alpha=10000000, maxiter=2000)
        if c_f < best_c:
            best_c, best_h = c_f, h_f

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
