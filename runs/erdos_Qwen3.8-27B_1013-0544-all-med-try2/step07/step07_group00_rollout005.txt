def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smmax(h, n, alpha):
    """Gradient of smooth max of overlap integrals."""
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def obj_smmax(x, n, alpha, mu):

    m = c.max()

    mx = sh.max()

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

            return float(np.sum(x[k:] * (1 - x[:nv])) * dx)
        else:
            kk = -k
            nv = n - kk
            return float(np.sum(x[:nv] * (1 - x[kk:])) * dx)

            g[k:] += (-x[:nv]) * dx
        else:

            nv = n - kk
            g[kk:] += (1 - x[:nv]) * dx
            g[:nv] += (-x[kk:]) * dx
        return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smmax(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def obj_smmax(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_obj(x, n, alpha, mu):
    g = grad_smmax(x, n, alpha)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def eq_fun(x, n):
    return float(x.sum() - n / 2.0)

def eq_jac(x, n):
    return np.ones(n)

def single_shift_obj(x, n, k, mu):
    dx = 2.0 / n
    if k >= 0:
        nv = n - k
        val = float(np.sum(x[k:] * (1 - x[:nv])) * dx)
    else:
        kk = -k
        nv = n - kk
        val = float(np.sum(x[:nv] * (1 - x[kk:])) * dx)
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def single_shift_grad(x, n, k, mu):
    g = np.zeros(n)
    dx = 2.0 / n
    if k >= 0:
        nv = n - k
        g[:nv] += (1.0 - x[k:]) * dx
        g[k:] += (-x[:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        g[kk:] += (1.0 - x[:nv]) * dx
        g[:nv] += (-x[kk:]) * dx
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def alt_shift_min(h, n, deadline, n_rounds=3, top_k=4):

    bounds = [(0.0, 1.0)] * n

                except Exception:

        # Re-sort and continue
    return h, best_c

def slsqp_opt(h, n, deadline, alpha=500000):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    def sobj(x):
        cx = compute_c5_vec(x, n)
        mxv = cx.max()
        sx = alpha * (cx - mxv)
        smx = sx.max()
        return mxv + (smx + np.log(np.sum(np.exp(sx - smx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    iters = max(100, min(4000, int((deadline - time.time()) * 40)))
    try:
        res = minimize(sobj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': iters, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def pair_polish(h, n, deadline, radius=40):
    best_c = compute_c5(h, n)
    for d in [0.003, 0.001, 0.0004, 0.0001]:
        any_imp = True
        while any_imp and time.time() < deadline:
            any_imp = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; any_imp = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; any_imp = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c
