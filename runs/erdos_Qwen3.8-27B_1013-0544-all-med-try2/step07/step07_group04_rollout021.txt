import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_onto_set(p, S):
    """Project p onto {h: 0<=h_i<=1, sum(h)=S} via bisection on shift."""
    n = len(p)
    lo = p.min() - 1.0
    hi = p.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(p - mid, 0.0, 1.0)
        s = h.sum()
        if abs(s - S) < 1e-15:
            break
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_gradient(h, n, alpha):
    """Compute gradient of smooth max of correlations."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    nm1 = n - 1
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-12:
            continue
        t = m_idx - nm1
        # First term: (1-h[j-t]) for j in [max(0,t), min(n, n+t))
        js = max(0, t)
        je = min(n, n + t)
        if js < je:
            h_s = js - t
            h_e = je - t
            grad[js:je] += w * (1.0 - h[h_s:h_e])
        # Second term: -h[j+t] for j in [max(0,-t), min(n, n-t))
        js2 = max(0, -t)
        je2 = min(n, n - t)
        if js2 < je2:
            h_s2 = js2 + t
            h_e2 = je2 + t
            grad[js2:je2] += w * (-h[h_s2:h_e2])
    
    return grad, float(m)

def projected_gradient(h0, n, alpha, maxiter, deadline, lr_init=0.01):
    """Projected gradient descent with backtracking on smooth max."""
    S = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    stagnant = 0
    
    for it in range(maxiter):
        if time.time() > deadline:
            break
        grad, cur_c = smooth_gradient(h, n, alpha)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Backtracking line search
        for _ in range(30):
            h_new = project_onto_set(h - lr * grad, S)
            c_new = compute_c5(h_new, n)
            if c_new < cur_c + 1e-14:
                break
            lr *= 0.5
        else:
            lr = max(lr, 1e-10)
            h_new = project_onto_set(h - lr * grad, S)
        
        h = h_new
        if stagnant > 50:
            lr = lr_init
            stagnant = 0
    
    return best_h, best_c

def binary_local_search(h_init, n, deadline, max_sweeps=500, n_restarts=3, seed=42):
    """Binary local search with flip operations."""
    target = n // 2
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    rng = np.random.RandomState(seed)
    
    # Start from best binary approximation
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:target]] = 1.0
    
    best_c = np.inf
    best_h = h.copy()
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            # Random perturbation
            for _ in range(rng.randint(2, 6)):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    h[a], h[b] = 0.0, 1.0
            # Project to exact binary
            h = (h >= 0.5).astype(float)
            # Fix count
            cnt = h.sum()
            if cnt > target:
                oi = np.where(h == 1)[0]
                to_flip = rng.choice(oi, int(cnt - target), replace=False)
                h[to_flip] = 0
            elif cnt < target:
                zi = np.where(h == 0)[0]
                to_flip = rng.choice(zi, int(target - cnt), replace=False)
                h[to_flip] = 1
        
        corr = np.correlate(h, 1 - h, mode='full').copy()
        cur_c = corr.max() * dx
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            # Shuffle for diversity
            rng.shuffle(zeros)
            for i in zeros:
                if time.time() > deadline:
                    break
                rng.shuffle(ones)
                for j in ones:
                    # Compute delta in correlation
                    im = i + m_range
                    imn = i - m_range
                    jm = j + m_range
                    jmn = j - m_range
                    delta = np.zeros(2 * n - 1)
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    msk = (imn >= 0) & (imn < n)
                    delta[msk] -= h[imn[msk]]
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - i)
                    if 0 <= k1 < 2*n-1:
                        delta[k1] += 1.0
                    k2 = n - 1 + (i - j)
                    if 0 <= k2 < 2*n-1:
                        delta[k2] += 1.0
                    new_corr = corr + delta
                    nc = new_corr.max() * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr = new_corr
                        cur_c = nc
                        ones.remove(j)
                        ones.append(i)
                        zeros.remove(i)
                        zeros.append(j)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_onto_set(p, S):
    n = len(p)
    lo = p.min() - 1.0
    hi = p.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(p - mid, 0.0, 1.0)
        s = h.sum()
        if abs(s - S) < 1e-15:
            break
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    sh = sh - sh.max()
    ev = np.exp(sh)
    w = ev / ev.sum()
    grad = np.zeros(n)
    nm1 = n - 1
    for mi in range(2 * n - 1):
        wi = w[mi]
        if wi < 1e-12:
            continue
        t = mi - nm1
        js = max(0, t)
        je = min(n, n + t)
        if js < je:
            grad[js:je] += wi * (1.0 - h[js - t:je - t])
        js2 = max(0, -t)
        je2 = min(n, n - t)
        if js2 < je2:
            grad[js2:je2] += wi * (-h[js2 + t:je2 + t])
    return grad, float(m)

def proj_grad(h0, n, alpha, maxiter, deadline, lr_init=None):
    S = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init if lr_init else 2.0 / alpha
    stall = 0
    for it in range(maxiter):
        if time.time() > deadline:
            break
        grad, cur_c = smooth_grad(h, n, alpha)
        if cur_c < best_c - 1e-16:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 80:
            h += 0.02 * np.random.RandomState(it).randn(n)
            h = project_onto_set(h, S)
            lr = lr_init if lr_init else 2.0 / alpha
            stall = 0
        for _ in range(40):
            hn = project_onto_set(h - lr * grad, S)
            cn = compute_c5(hn, n)
            if cn <= cur_c + 1e-15:
                break
            lr *= 0.5
        h = project_onto_set(h - lr * grad, S)
    return best_h, best_c

def binary_search(h_init, n, deadline, max_sweeps=500, n_restarts=3, seed=42):
    target = n // 2
    dx = 2.0 / n
    mr = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:target]] = 1.0
    best_c = np.inf
    best_h = h.copy()
    for rest in range(n_restarts):
        if time.time() > deadline:
            break
        if rest > 0:
            for _ in range(rng.randint(2, 8)):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a, b = rng.choice(oi), rng.choice(zi)
                    h[a], h[b] = 0.0, 1.0
            h = (h >= 0.5).astype(float)
            cnt = int(h.sum())
            if cnt > target:
                oi = np.where(h == 1)[0]
                h[rng.choice(oi, cnt - target, replace=False)] = 0
            elif cnt < target:
                zi = np.where(h == 0)[0]
                h[rng.choice(zi, target - cnt, replace=False)] = 1
        corr = np.correlate(h, 1 - h, mode='full').copy()
        cur_c = corr.max() * dx
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        for _ in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            rng.shuffle(zeros)
            for i in zeros:
                if time.time() > deadline:
                    break
                rng.shuffle(ones)
                for j in ones:
                    im = i + mr
                    imn = i - mr
                    jm = j + mr
                    jmn = j - mr
                    delta = np.zeros(2 * n - 1)
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    msk = (imn >= 0) & (imn < n)
                    delta[msk] -= h[imn[msk]]
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - i)
                    if 0 <= k1 < 2 * n - 1:
                        delta[k1] += 1.0
                    k2 = n - 1 + (i - j)
                    if 0 <= k2 < 2 * n - 1:
                        delta[k2] += 1.0
                    nc = (corr + delta).max() * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr = corr + delta
                        cur_c = nc
                        ones.remove(j)
                        ones.append(i)
                        zeros.remove(i)
                        zeros.append(j)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
