def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

    best_c = compute_c5(h, n)
    for step in range(num_steps):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (h.sum() - n/2.0)**2
    return val

def smooth_max_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])

    v_arr = np.zeros(n)

            break
        t += 1

Let me try a different approach for the local search: **projected gradient descent with backtracking line search**.

At each step:
1. Compute the (smooth) gradient g
2. Project -g onto the tangent space of the feasible set (i.e., sum of changes = 0, and 0 ≤ h + step ≤ 1)
3. Take a step in that direction with optimal step size

The projection onto the feasible tangent space is:
- Start with s = -g
- Normalize: s = s - mean(s) (to ensure sum = 0)
- For coordinates where h[i] + t*s[i] would violate [0,1], clip s[i] appropriately
- This is like a "proximal" step

This is essentially what Adam does with the projection, but with a proper line search.

def projected_grad_step(h, n, g, max_step=0.1):
    """Take a projected gradient step with line search."""
    s = -g.copy()
    s -= s.mean()  # Ensure zero sum
    # Find max valid step
    t_max = max_step
    mask_pos = s > 1e-12
    mask_neg = s < -1e-12
    if mask_pos.any():
        t_max = min(t_max, np.min((1 - h[mask_pos]) / s[mask_pos]))
    if mask_neg.any():
        t_max = min(t_max, np.min(-h[mask_neg] / s[mask_neg]))
    t_max = max(t_max, 1e-10)
    # Backtracking line search
    c_best = compute_c5(h, n)
    t = t_max
    for _ in range(20):
        h_new = h + t * s
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < c_best - 1e-16:
            return h_new, c_new
        t *= 0.5
        if t < 1e-10:
            break
    return h, c_best

            if abs(delta) < 1e-8:
                continue
            mask = (h > abs(delta) / (n - 1)) & (h < 1.0 - abs(delta) / (n - 1))
            mask[i] = False
            if not mask.any():
                mask = np.ones(n, dtype=bool)
                mask[i] = False
            if not mask.any():
                continue
            h_new = h.copy()
            h_new[i] = target
            h_new[mask] -= np.sign(delta) * abs(delta) / mask.sum()
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c - 1e-15:
                best_c = c
                h = h_new
                improved = True
        if not improved:
            break
    return h, best_c

def anneal_phase(h, n, deadline, T_init=0.0005, T_final=0.00001, max_iters=3000):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.default_rng(123)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = it / max_iters
        T = T_init * (T_final / T_init) ** frac
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = 0.003 * rng.uniform(0.5, 1.5) * (1.0 if rng.random() < 0.5 else -1.0)
        if h[i] + delta < 0 or h[i] + delta > 1:
            continue
        if h[j] - delta < 0 or h[j] - delta > 1:
            continue
        h[i] += delta
        h[j] -= delta
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-16:
            best_c = c
            best_h = h.copy()
        elif c > best_c + 1e-16:
            if rng.random() > np.exp(-(c - best_c) / T):
                h[i] -= delta
                h[j] += delta
                h = project_feasible(h, n)
    return best_h, best_c

def refine_pairs(h, n, deadline, deltas):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for jj in [(i + 1) % n, (i - 1) % n, (i + 2) % n, (i - 2) % n,
                           (i + 3) % n, (i + n // 4) % n, (i + n // 2) % n, (i + 3 * n // 4) % n]:
                    if i == jj:
                        continue
                    if h[i] + d <= 1.0 and h[jj] - d >= 0.0:
                        h[i] += d
                        h[jj] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[jj] += d
                    if h[i] - d >= 0.0 and h[jj] + d <= 1.0:
                        h[i] -= d
                        h[jj] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[jj] -= d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: L-BFGS-B moderate alpha
    t1 = min(t0 + 120, deadline - 580)
    best_h, best_c = lbfgs_phase(best_h, n_init, t1, [50, 200, 1000, 5000])
    # Phase 2: Adam main optimization
    if time.time() < deadline - 380:
        t2 = min(time.time() + 280, deadline - 380)
        h_a, c_a = adam_phase(best_h, n_init, t2, alpha_init=20, alpha_final=50000, lr_init=0.012)
        if c_a < best_c:
            best_c, best_h = c_a, h_a
    # Phase 3: Projected gradient refinement
    if time.time() < deadline - 250:
        t3 = min(time.time() + 100, deadline - 250)
        h3, c3 = proj_grad_phase(best_h, n_init, t3, alpha=1000)
        if c3 < best_c:
            best_c, best_h = c3, h3
    # Phase 4: Binary push
    if time.time() < deadline - 180:
        t4 = min(time.time() + 70, deadline - 180)
        h4, c4 = binary_push(best_h, n_init, t4)
        if c4 < best_c:
            best_c, best_h = c4, h4
    # Phase 5: Simulated annealing
    if time.time() < deadline - 100:
        t5 = min(time.time() + 70, deadline - 100)
        h5, c5v = anneal_phase(best_h, n_init, t5)
        if c5v < best_c:
            best_c, best_h = c5v, h5
    # Phase 6: Pair refinement
    if time.time() < deadline - 60:
        t6 = min(time.time() + 50, deadline - 60)
        best_h, best_c = refine_pairs(best_h, n_init, t6,
                                      [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
    # Phase 7: Try other n
    if time.time() < deadline - 50:
        for n_try in [250, 400]:
            if time.time() > deadline - 35:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            alloc = min(25, deadline - time.time() - 25)
            if alloc < 12:
                break
            h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.4, [200, 2000, 10000])
            if time.time() < deadline - 12:
                h_t, c_t = binary_push(h_t, n_try, min(time.time() + 10, deadline - 8))
            if c_t < best_c:
                best_c, best_h = c_t, h_t
    # Phase 8: Final polish
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = refine_pairs(best_h, nf, deadline - 2,
                                      [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
