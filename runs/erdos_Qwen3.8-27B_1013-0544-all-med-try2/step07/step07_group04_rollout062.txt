def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_func(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    scaled = alpha * corr * dx
    sm = scaled.max()
    val = (sm + np.log(np.exp(scaled - sm).sum())) / alpha
    return val

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    scaled = alpha * corr * dx
    sm = scaled.max()
    softmax = np.exp(scaled - sm)
    softmax /= softmax.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = softmax[k] * dx
        if w < 1e-16:
            continue
        j = k - (n - 1)
        i_min = max(0, -j)
        i_max = min(n - 1, n - 1 - j)
        if i_min <= i_max:
            grad[i_min:i_max + 1] += w * (1.0 - h[i_min:i_max + 1 + j])
        i_min2 = max(0, j)
        i_max2 = min(n - 1, n - 1 + j)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2 + 1] -= w * h[i_min2:i_max2 + 1 - j]
    return grad

def slsqp_analytic(h, n, alpha, maxiter, ftol):
    global _N_GLOBAL
    _N_GLOBAL = n
    
    def _obj(x):
        return smooth_max_func(x, n, alpha)
    
    def _jac(x):
        return smooth_max_grad(x, n, alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    
    try:
        res = minimize(_obj, h, jac=_jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

This is related to the concept of a "perfect difference set" or a "BIBD" (balanced incomplete block design).

For a perfect difference set of order (v, k, λ), every nonzero difference d = x-y (x,y ∈ S, x≠y) occurs exactly λ times. This gives very uniform autocorrelation.

But perfect difference sets are rare and don't exist for all parameters. For v=250, k=125, we'd need λ such that k(k-1) = λ(v-1), so 125*124 = λ*249, giving λ = 15500/249 ≈ 62.25, which is not an integer. So a perfect difference set doesn't exist for these parameters.

A BIBD(250, 125, λ) with the above λ doesn't exist either.

So we can't get a perfectly uniform autocorrelation. The best we can do is minimize the maximum deviation.

Another approach: use a Fourier-based construction. If h is the indicator of a set S, then the autocorrelation can be expressed in terms of the Fourier transform of h. Minimizing the maximum autocorrelation is related to minimizing the maximum of the "aperiodic autocorrelation" of the sequence.

The Lempel-Greenberger construction or the Welch construction might give good initial solutions.

    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h

def smooth_max_func(x):
    n = _N_G
    alpha = _ALPHA_G
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    scaled = alpha * corr * dx
    sm = scaled.max()
    return (sm + np.log(np.exp(scaled - sm).sum())) / alpha

def smooth_max_grad(x):
    n = _N_G
    alpha = _ALPHA_G
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    scaled = alpha * corr * dx
    sm = scaled.max()
    softmax = np.exp(scaled - sm)
    softmax /= softmax.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = softmax[k] * dx
        if w < 1e-16:

        j = k - (n - 1)

        i_max2 = min(n - 1, n - 1 + j)
        if i_min2 <= i_max2:

import numpy as np
from scipy.optimize import minimize
import time

_N_G = 0
_ALPHA_G = 10000.0

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_func(x):
    n = _N_G
    alpha = _ALPHA_G
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    scaled = alpha * corr * dx
    sm = scaled.max()
    return (sm + np.log(np.exp(scaled - sm).sum())) / alpha

def smooth_max_grad(x):
    n = _N_G
    alpha = _ALPHA_G
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    scaled = alpha * corr * dx
    sm = scaled.max()
    softmax = np.exp(scaled - sm)
    softmax /= softmax.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = softmax[k] * dx
        if w < 1e-16:
            continue
        j = k - (n - 1)
        i_min = max(0, -j)
        i_max = min(n - 1, n - 1 - j)
        if i_min <= i_max:
            grad[i_min:i_max + 1] += w * (1.0 - x[i_min:i_max + 1 + j])
        i_min2 = max(0, j)
        i_max2 = min(n - 1, n - 1 + j)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2 + 1] -= w * x[i_min2:i_max2 + 1 - j]
    return grad

def _cons_fun(x):
    return x.sum() - _N_G / 2.0

def _cons_jac(x):
    return np.ones_like(x)

def slsqp_analytic(h, n, alpha, maxiter):
    global _N_G, _ALPHA_G
    _N_G = n
    _ALPHA_G = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(smooth_max_func, h, jac=smooth_max_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def try_swap(h, corr, p, q, n):
    dx = 2.0 / n
    delta = np.zeros(2 * n - 1)
    for j in range(-(n - 1), n):
        if j == 0:
            continue
        k = j + n - 1
        total = 0.0
        seen = set()
        for i in (p, q, p - j, q - j):
            if i in seen or i < 0 or i >= n or i + j < 0 or i + j >= n:
                continue
            seen.add(i)
            hi = h[i]
            hij = h[i + j]
            old = hi * (1.0 - hij)
            hi_new = 1.0 if i == p else (0.0 if i == q else hi)
            ij = i + j
            ij_new = 1.0 if ij == p else (0.0 if ij == q else hij)
            new = hi_new * (1.0 - ij_new)
            total += (new - old)
        delta[k] = total
    new_corr = corr + delta
    return new_corr, new_corr.max() * dx

def best_improvement_binary(h_init, n, deadline, max_rounds=30000):
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if not ones_list or not zeros_list:
            break
        best_new_max = cur_c
        best_p = -1
        best_q = -1
        best_corr = None
        for p in zeros_list:
            for q in ones_list:
                new_corr, nc = try_swap(h, corr, p, q, n)
                if nc < best_new_max - 1e-14:
                    best_new_max = nc
                    best_p = p
                    best_q = q
                    best_corr = new_corr
        if best_p < 0:
            break
        h[best_p] = 1.0
        h[best_q] = 0.0
        corr = best_corr
        cur_c = best_new_max
        ones_list.remove(best_q)
        ones_list.append(best_p)
        zeros_list.remove(best_p)
        zeros_list.append(best_q)
    return h, cur_c

def multi_restart_binary(h_init, n, deadline, rng, n_restarts=8):
    dx = 2.0 / n
    target = n // 2
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)
    # Binary-ify
    idx = np.argsort(best_h)[::-1]
    best_h = np.zeros(n)
    best_h[idx[:target]] = 1.0
    best_c = compute_c5(best_h, n)
    
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        if r == 0:
            start = best_h.copy()
        else:
            start = best_h.copy()
            n_flips = rng.randint(2, 6)
            ones = np.where(start == 1)[0]
            zeros = np.where(start == 0)[0]
            for _ in range(n_flips):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                a = rng.choice(ones)
                b = rng.choice(zeros)
                start[a], start[b] = start[b], start[a]
                ones = np.where(start == 1)[0]
                zeros = np.where(start == 0)[0]
        sub_dl = min(deadline, time.time() + max(10, (deadline - time.time()) / (n_restarts - r)))
        hb, cb = best_improvement_binary(start, n, sub_dl, max_rounds=10000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    return best_h, best_c
