def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    """Compute gradient of smooth max approximation of C5."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    # Shifted for numerical stability
    a = alpha * (c - m)
    w = np.exp(a)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
    else:
        w /= w_sum
    
    # For each significant weight, compute gradient of c_k
    g = np.zeros(n)
    m_range = np.arange(-(n-1), n)
    
    # Only process weights above threshold for efficiency
    thresh = max(1e-10, 1.0 / (n * 10))
    active = w > thresh
    if not active.any():
        active[c.argmax()] = True
    
    for idx in np.where(active)[0]:
        wi = w[idx]
        k = m_range[idx]
        # d c_k / d h[i] = (1-h[i+k]) * 1[i+k in [0,n)) - h[i-k] * 1[i-k in [0,n))

        imk = np.arange(n) - k

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    a = alpha * (c - m)
    w = np.exp(a)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    m_range = np.arange(-(n - 1), n)
    thresh = 1.0 / (len(w) * 50)
    active = w > thresh
    if not active.any():
        active[c.argmax()] = True
    idxs = np.where(active)[0]
    for idx in idxs:
        wi = w[idx]
        k = int(m_range[idx])
        if k >= 0:
            mask1 = np.arange(n - k)
            g[mask1] += wi * (1.0 - h[mask1 + k])
            mask2 = np.arange(k, n)
            g[mask2] -= wi * h[mask2 - k]
        else:
            kk = -k
            mask2 = np.arange(n - kk)
            g[mask2] -= wi * h[mask2 + kk]
            mask1 = np.arange(kk, n)
            g[mask1] += wi * (1.0 - h[mask1 - kk])
    return g, m

def projected_smooth_optimize(h, n, alpha, max_steps=5000, deadline=None):

    return best_h, best_val

def coord_descent(h, n, max_sweeps=200, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            val0 = np.max(corr)
            val1 = np.max(corr + B)
            lo = min(val0, val1) - 1e-9
            hi = max(val0, val1) + 1e-9
            for _ in range(35):
                mid = (lo + hi) / 2.0
                if zero_max > mid + 1e-14:
                    lo = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi = mid
                else:
                    lo = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, np.max((hi - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, np.min((hi - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-10:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
                improved = True
        if not improved:
            break
    return h, corr

def binary_swap(h, n, deadline, max_sweeps=100):

            best_nc = cur_c

                m_range = np.arange(-(n - 1), n)

                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                if j > i:
                    delta[n - 1 + (j - i)] += 1.0
                    delta[n - 1 + (i - j)] += 1.0
                elif j < i:
                    delta[n - 1 + (j - i)] += 1.0

                nc = (corr + delta).max() * dx

                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                improved = True
        if not improved:
