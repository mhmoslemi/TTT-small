def build_sym(v, n):
    m = n // 2
    v = np.asarray(v, dtype=float).ravel()
    if v.size < m:
        v = np.resize(v, m)
    v = np.clip(v[:m], 0.0, 1.0)
    h = np.empty(n, dtype=float)
    h[:m] = v
    if n % 2 == 0:
        h[m:] = 1.0 - v[::-1]
    else:
        h[m] = 0.5
        h[m + 1:] = 1.0 - v[::-1]
    return h


def sym_from_full(h, n):
    m = n // 2
    h = np.asarray(h, dtype=float).ravel()
    if n % 2 == 0:
        v = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    else:

def lag_values(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def compute_c5(h, n):
    return float(np.max(lag_values(h, n)))


def soft_obj_from_g(g, alpha):
    g = np.asarray(g, dtype=float)
    m = float(np.max(g))
    if alpha <= 0.0 or not np.isfinite(m):
        return m
    z = alpha * (g - m)
    zm = float(np.max(z))
    z = z - zm
    z = np.clip(z, -700.0, 0.0)
    s = float(np.sum(np.exp(z)))
    if s <= 0.0 or not np.isfinite(s):
        return m
    return m + (zm + math.log(s)) / alpha


def softmax_p(g, alpha):
    g = np.asarray(g, dtype=float)
    m = float(np.max(g))

        p[int(np.argmax(g))] = 1.0

    return grad

def swap_delta(i, j, n, h, ar):
    delta = np.zeros(2 * n - 1, dtype=float)

    idxA = n - 1 + i - ar
    addA = 1.0 - h
    addA = addA.copy()
    addA[j] += h[j]
    delta[idxA] += addA

    idxB = n - 1 + ar - i
    addB = -h
    addB = addB.copy()
    addB[j] += h[j]
    delta[idxB] += addB

    idxC = n - 1 + j - ar
    delta[idxC] += h - 1.0

    idxD = n - 1 + ar - j
    addD = h.copy()
    addD[i] = 0.0
    delta[idxD] += addD

    delta[n - 1] -= 2.0
    return delta

            v = (rng.rand(n // 2) < 0.5).astype(float)
            h0 = build_sym(v, n)
        else:
            h0 = np.zeros(n, dtype=float)
            h0[rng.permutation(n)[:target]] = 1.0

        corr = np.correlate(h0, 1.0 - h0, mode="full").astype(float)
        cur = float(np.max(corr))
        h_cur = h0.copy()
        corr_cur = corr.copy()

        if cur < best_count:
            best_count = cur
            best_h = h_cur.copy()
            best_c = cur * dx

        for _ in range(int(max_sweeps)):
            if time.time() > deadline:
                break
            ones = np.where(h_cur == 1.0)[0].tolist()
            zeros = np.where(h_cur == 0.0)[0].tolist()
            rng.shuffle(ones)
            rng.shuffle(zeros)
            improved = False

            for i in zeros:
                if time.time() > deadline:
                    break
                if h_cur[i] != 0.0:
                    continue
                for j in ones:
                    if time.time() > deadline:
                        break
                    if h_cur[j] != 1.0:
                        continue
                    delta = swap_delta(i, j, n, h_cur, ar)
                    new_corr = corr_cur + delta
                    nm = float(np.max(new_corr))
                    if nm < cur - 1e-8:
                        h_cur[i] = 1.0
                        h_cur[j] = 0.0
                        corr_cur = new_corr
                        cur = nm
                        try:
                            ones.remove(j)
                            ones.append(i)
                            zeros.remove(i)
                            zeros.append(j)
                        except Exception:
                            pass
                        if cur < best_count:
                            best_count = cur
                            best_h = h_cur.copy()
                            best_c = cur * dx
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break

    if best_h is None:
        return None, np.inf
    return best_h, float(best_c)


def run(seed=42, budget_s=1100, **kwargs):
    try:
        t0 = time.time()
        dl = t0 + max(10.0, float(budget_s)) - 8.0
        rng = np.random.RandomState(seed)

        try:
            arr0 = np.asarray(initial_h_values, dtype=float).ravel().copy()
            if arr0.size == 0:
                raise ValueError
            n_src = len(arr0)
        except Exception:
            n_src = 250
            vv = 0.5 + 0.45 * np.cos(np.linspace(0.0, math.pi, 125))
            arr0 = build_sym(vv, 250)

        if n_src > 500:
            arr = interp_full(arr0, n_src, 500)
            n0 = 500
        elif n_src < 100:
            arr = interp_full(arr0, n_src, 250)
            n0 = 250
        else:
            arr = arr0.copy()
            n0 = n_src

        if n0 % 2:
            arr = interp_full(arr, n0, n0 + 1)
            n0 += 1

        if n0 < 2:
            arr = build_sym(0.5 + 0.45 * np.cos(np.linspace(0.0, math.pi, 125)), 250)
            n0 = 250

        arr = project_feasible(arr, n0)
        best_n = int(n0)
        best_h = arr.copy()
        best_c = compute_c5(best_h, best_n)

        vs = sym_from_full(best_h, best_n)
        hs = build_sym(vs, best_n)
        cs = compute_c5(hs, best_n)
        if better_candidate(cs, best_n, best_c, best_n):
            best_h = hs.copy()
            best_c = float(cs)

        if dl - t0 < 20.0:
            return best_h, float(best_c), best_n

        total = max(1.0, dl - t0)
        p_main = t0 + 0.40 * total
        p_rand = t0 + 0.55 * total
        p_full = t0 + 0.65 * total
        p_bin = t0 + 0.85 * total
        p_mult = dl - 25.0

        if time.time() < p_main:
            h_m, c_m = sym_polish_loop(sym_from_full(best_h, best_n), best_n, p_main, True)
            if better_candidate(c_m, best_n, best_c, best_n):
                best_h = h_m.copy()
                best_c = float(c_m)

        if time.time() < p_rand:
            for rn in (250, 300):
                if time.time() >= p_rand - 2.0:
                    break
                for kind in range(5):
                    if time.time() >= p_rand - 2.0:
                        break
                    v0 = random_sym_v(rn, rng, kind)
                    h_r, c_r = sym_polish_loop(v0, rn, p_rand, False)
                    if better_candidate(c_r, rn, best_c, best_n):
                        best_h = h_r.copy()
                        best_c = float(c_r)
                        best_n = int(rn)

        if best_n <= 350 and time.time() < min(p_full, dl - 60.0):
            h_f, c_f = full_polish_sequence(best_h, best_n, min(p_full, dl - 60.0))
            if better_candidate(c_f, best_n, best_c, best_n):
                best_h = h_f.copy()
                best_c = float(c_f)

        if time.time() < min(p_bin, dl - 80.0):
            bn = best_n if (best_n % 2 == 0 and best_n <= 300) else 250
            hstart = interp_full(best_h, best_n, bn) if bn != best_n else best_h
            hstart = project_feasible(hstart, bn)
            deadline_bin = min(p_bin, dl - 80.0)
            hb, cb = binary_local(hstart, bn, deadline_bin, seed, 3, 15)
            if hb is not None:
                if better_candidate(cb, bn, best_c, best_n):
                    best_h = hb.copy()
                    best_c = float(cb)
                    best_n = int(bn)
                if time.time() < dl - 50.0 and cb < best_c + 0.01:
                    h_b, c_b = sym_polish_loop(sym_from_full(hb, bn), bn, dl - 50.0, False)
                    if better_candidate(c_b, bn, best_c, best_n):
                        best_h = h_b.copy()
                        best_c = float(c_b)
                        best_n = int(bn)

        if time.time() < p_mult:
            for nh in (200, 250, 300, 350):
                if time.time() >= p_mult - 5.0:
                    break
                if nh == best_n:
                    continue
                vh = interp_sym_from_full(best_h, best_n, nh)
                h_n, c_n = sym_polish_loop(vh, nh, p_mult, False)
                if better_candidate(c_n, nh, best_c, best_n):
                    best_h = h_n.copy()
                    best_c = float(c_n)
                    best_n = int(nh)

        if time.time() < dl - 5.0:
            vf = sym_from_full(best_h, best_n)
            v2 = lbfgsb_sym(vf, best_n, 1000000.0, 900)
            h2 = build_sym(v2, best_n)
            c2 = compute_c5(h2, best_n)
            if better_candidate(c2, best_n, best_c, best_n):
                best_h = h2.copy()
                best_c = float(c2)

            v2 = slsqp_sym(v2, best_n, 1000000.0, 300)
            h2 = build_sym(v2, best_n)
            c2 = compute_c5(h2, best_n)
            if better_candidate(c2, best_n, best_c, best_n):
                best_h = h2.copy()
                best_c = float(c2)

        if best_n <= 350 and time.time() < dl - 30.0:
            h2 = slsqp_full(best_h, best_n, 100000.0, 250)
            c2 = compute_c5(h2, best_n)
            if better_candidate(c2, best_n, best_c, best_n):
                best_h = h2.copy()
                best_c = float(c2)

        if time.time() < dl - 12.0:
            h2 = projected_gradient(best_h, 50000.0, 600, dl - 3.0)
            c2 = compute_c5(h2, best_n)
            if better_candidate(c2, best_n, best_c, best_n):
                best_h = h2.copy()
                best_c = float(c2)

        if abs(float(np.sum(best_h)) - 0.5 * float(best_n)) > 1e-8:
            best_h = project_feasible(best_h, best_n)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c = compute_c5(best_h, best_n)
        return best_h, float(best_c), int(best_n)

    except Exception:
        try:
            arr = np.asarray(initial_h_values, dtype=float).ravel().copy()
            n = len(arr)
            if n % 2:
                n += 1
                arr = interp_full(arr, len(arr) - 1, n)
            if n < 2:
                n = 250
                arr = build_sym(0.5 + 0.45 * np.cos(np.linspace(0.0, math.pi, 125)), 250)
            arr = project_feasible(arr, n)
            return arr, float(compute_c5(arr, n)), int(n)
        except Exception:
            v = 0.5 + 0.45 * np.cos(np.linspace(0.0, math.pi, 125))
            h = build_sym(v, 250)
            return h, float(compute_c5(h, 250)), 250
