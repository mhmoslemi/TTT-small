import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_c = c.max()
    shifted = alpha * (c - max_c)
    exp_sh = np.exp(shifted)
    sum_exp = exp_sh.sum()
    F = max_c + np.log(sum_exp) / alpha
    S = exp_sh / sum_exp
    omh = 1.0 - h
    grad = np.zeros(n)
    for m in range(2*n-1):
        s = S[m]
        if s < 1e-14:
            continue
        lag = m - (n-1)
        if lag >= 0:
            grad[:n-lag] += s * omh[lag:]
            grad[lag:] -= s * h[:n-lag]
        else:
            ll = -lag
            grad[ll:] += s * omh[:n-ll]
            grad[:n-ll] -= s * h[ll:]
    return F, grad * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()

            break
        if diff > 0:

            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

        F, g = compute_obj_grad(h, n, alpha)
        g = g - g.mean()
        m_state = b1 * m_state + (1.0 - b1) * g
        v_state = b2 * v_state + (1.0 - b2) * g * g
        mhat = m_state / (1.0 - b1 ** t)
        vhat = v_state / (1.0 - b2 ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        h = h - step
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - h.sum()
        if diff > 1e-10:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-10:
                h += diff / s * room
        elif diff < -1e-10:
            s = h.sum()
            if s > 1e-10:
                h += diff / s * h
        h = np.clip(h, 0.0, 1.0)
        if it % 2000 == 0:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    c = compute_c5(h, n)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 6
    dl = t0 + total

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)

    # Phase 1: Adam with increasing alpha (main workhorse)
    p1 = t0 + 0.30 * total
    h = best_h.copy()
    alphas_main = [50, 200, 1000, 5000, 20000, 100000, 500000]
    n_alpha = len(alphas_main)
    for ai, alpha in enumerate(alphas_main):
        remain = (p1 - time.time()) / (n_alpha - ai)
        ddl = min(time.time() + remain, p1 - 2)
        h, c = adam_loop(h, n, alpha, ddl, lr=0.004)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 2: Very high alpha polish on current best
    p2 = t0 + 0.45 * total
    h = best_h.copy()
    for alpha in [2e6, 1e7, 5e7]:
        if time.time() > p2 - 5:
            break
        remain = (p2 - time.time()) / 3.0
        h, c = adam_loop(h, n, alpha, min(time.time() + remain, p2 - 2), lr=0.002)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 3: Noise restarts
    p3 = t0 + 0.60 * total
    for trial in range(4):
        if time.time() > p3 - 10:
            break
        noise_scale = 0.008 + 0.004 * trial
        h = best_h + noise_scale * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [10000, 200000, 5e6]:
            if time.time() > p3 - 5:
                break
            remain = (p3 - time.time()) / 3.0
            h, c = adam_loop(h, n, alpha, min(time.time() + remain, p3 - 2), lr=0.003)
            if c < best_c:
                best_c, best_h = c, h

    # Phase 4: Binary rounding + re-smooth
    p4 = t0 + 0.75 * total
    hb = to_binary(best_h, n)
    cb = compute_c5(hb, n)
    for alpha in [5000, 100000, 2e6, 2e7]:
        if time.time() > p4 - 5:
            break
        remain = (p4 - time.time()) / 4.0
        hb2, cb2 = adam_loop(hb, n, alpha, min(time.time() + remain, p4 - 2), lr=0.004)
        if cb2 < best_c:
            best_c, best_h = cb2, hb2

    # Phase 5: Higher resolution
    p5 = t0 + 0.90 * total
    for nh in [500, 600]:
        if time.time() > p5 - 30:
            break
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = np.interp(xn, xo, best_h, left=0, right=0)
        hh = project_feasible(hh, nh)
        for alpha in [2000, 50000, 1e6, 5e6]:
            if time.time() > p5 - 20:
                break
            remain = (p5 - time.time()) / 4.0
            hh, ch = adam_loop(hh, nh, alpha, min(time.time() + remain, p5 - 5), lr=0.004)
            if ch < best_c:
                best_c, best_h, n = ch, hh, nh

    # Phase 6: Final ultra-high-alpha polish
    if time.time() < dl - 8:
        for alpha in [5e7, 5e8, 5e9]:
            if time.time() > dl - 4:
                break
            remain = (dl - time.time()) / 3.0
            h, c = adam_loop(best_h, n, alpha, min(time.time() + remain, dl - 2), lr=0.001)
            if c < best_c:
                best_c, best_h = c, h

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
