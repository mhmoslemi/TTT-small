            if tot < 1e-15:

            if tot < 1e-15:

    best_c = compute_c5(h, n)
    best_h = h.copy()
    mu = 50.0
    for alpha in [50, 200, 1000, 5000, 20000]:
        if time.time() > deadline - 5:
            break
        res = minimize(
            smooth_obj, h, args=(n, alpha, mu),
            jac=smooth_grad, method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-8}
        )
        h = project_feasible(res.x, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def pair_search(h, n, deadline, rng, max_iter=200000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        improved = True
        count = 0
        while improved and time.time() < deadline and count < max_iter:
            improved = False
            count += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for sign in [1, -1]:
                ni = h[i] + sign * d
                nj = h[j] - sign * d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i] = ni
                    h[j] = nj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] -= sign * d
                        h[j] += sign * d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B on initial
    alloc = min(180, (deadline - t0) * 0.25)
    h_opt, c_opt = optimize_lbfgsb(h_init, n_init, t0 + alloc, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt
    # Top-k gradient polish
    if time.time() < deadline - 30:
        h_opt, c_opt = topk_grad_step(best_h, len(best_h), k=8, lr=0.003, steps=100)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Pair search
    if time.time() < deadline - 50:
        n = len(best_h)
        h_opt, c_opt = pair_search(best_h, n, min(deadline - 50, time.time() + 120), rng, max_iter=80000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Try other n values
    x_best = np.linspace(0, 2, len(best_h))
    for n_try in [80, 120, 150, 200]:
        if time.time() > deadline - 80:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_best, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(80, (deadline - time.time()) * 0.2)
        if alloc < 25:
            break
        h_opt, c_opt = optimize_lbfgsb(h_interp, n_try, time.time() + alloc, rng)
        h_opt, c_opt = topk_grad_step(h_opt, n_try, k=5, lr=0.003, steps=80)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Multi-restart with diverse initializations
    if time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(2, min(6, int(remaining / 50)))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.03 + 0.12 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 15) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_lbfgsb(hp, n, time.time() + alloc, rng)
            h_opt, c_opt = topk_grad_step(h_opt, n, k=5, lr=0.002, steps=60)
            if time.time() < deadline - 10:
                h_opt, c_opt = pair_search(h_opt, n, min(deadline - 10, time.time() + 30), rng, max_iter=30000)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Final polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_opt, c_opt = optimize_lbfgsb(best_h, n, deadline - 8, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        if time.time() < deadline - 5:
            h_opt, c_opt = topk_grad_step(best_h, n, k=10, lr=0.002, steps=200)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
        if time.time() < deadline - 3:
            best_h, best_c = pair_search(best_h, n, deadline - 1, rng, max_iter=50000)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
