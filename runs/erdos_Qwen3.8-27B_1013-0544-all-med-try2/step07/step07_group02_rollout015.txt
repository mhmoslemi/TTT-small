    for k in top_k:

        k1 = np.arange(n) + m
        mask = (k1 >= 0) & (k1 < n)
        grad[mask] += (1.0 - h[k1[mask]])
        k2 = np.arange(n) - m
        mask2 = (k2 >= 0) & (k2 < n)
        grad[mask2] -= h[k2[mask2]]
    grad *= dx / n_active
    h_new = h - eta * grad
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 12
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Coordinate descent to reach local optimum
    p1 = t0 + 0.18 * total
    h1 = coord_descent(best_h, n, max_sweeps=1000, deadline=p1)
    h1 = project_feasible(h1, n)
    c1 = compute_c5(h1, n)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    # Phase 2: SLSQP with gradual alpha increase and interleaved coord descent
    p2 = t0 + 0.62 * total
    alphas = [10, 30, 100, 300, 1000, 3000, 10000, 30000, 100000,
              300000, 1000000, 3000000, 10000000, 30000000, 100000000,
              300000000, 1000000000, 3000000000]
    h = best_h.copy()
    for idx, alpha in enumerate(alphas):
        if time.time() > p2 - 4:
            break
        maxit = 3000 if alpha < 100000 else 2000
        h = slsqp_optimize(h, n, alpha, maxiter=maxit, ftol=1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # Interleaved coord descent every few alpha steps
        if idx % 3 == 2 and time.time() < p2 - 10:
            cd_dl = min(p2 - 8, time.time() + 8)
            h = coord_descent(h, n, max_sweeps=80, deadline=cd_dl)
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 3: Subgradient steps to escape potential flat spots
    p3 = t0 + 0.72 * total
    h = best_h.copy()
    for _ in range(200):
        if time.time() > p3:
            break
        h_new = subgradient_step(h, n, n_active=5, eta=0.015)
        c_new = compute_c5(h_new, n)
        if c_new < compute_c5(h, n) - 1e-12:
            h = h_new
        else:
            h_new = subgradient_step(h, n, n_active=3, eta=0.008)
            c_new2 = compute_c5(h_new, n)
            if c_new2 < compute_c5(h, n) - 1e-12:
                h = h_new
            else:
                break
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 4: Final coordinate descent
    p4 = t0 + 0.82 * total
    h4 = coord_descent(best_h, n, max_sweeps=500, deadline=p4)
    h4 = project_feasible(h4, n)
    c4 = compute_c5(h4, n)
    if c4 < best_c:
        best_c, best_h = c4, h4.copy()

    # Phase 5: Final SLSQP with very high alpha
    p5 = t0 + 0.93 * total
    for alpha in [500000000, 2000000000, 10000000000]:
        if time.time() > p5 - 3:
            break
        h = slsqp_optimize(best_h, n, alpha, maxiter=4000, ftol=1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 6: Final polish - coord descent + tiny SLSQP
    if time.time() < dl - 6:
        h = coord_descent(best_h, n, max_sweeps=200, deadline=dl - 4)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if time.time() < dl - 2:
            h = slsqp_optimize(best_h, n, 5000000000, maxiter=2000, ftol=1e-18)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 7: Quick perturbation search
    if time.time() < dl - 3:
        for scale in [0.001, 0.0005]:
            if time.time() > dl - 2:
                break
            h_p = best_h + scale * rng.randn(n)
            h_p = project_feasible(h_p, n)
            h_p = coord_descent(h_p, n, max_sweeps=100, deadline=dl - 1)
            h_p = project_feasible(h_p, n)
            c_p = compute_c5(h_p, n)
            if c_p < best_c:
                best_c, best_h = c_p, h_p.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
