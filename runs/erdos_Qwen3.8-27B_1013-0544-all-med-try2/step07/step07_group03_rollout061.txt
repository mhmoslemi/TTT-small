    for k in range(n):

        if wk < 1e-18:

        grad[:n-k] += wk * (1.0 - h[k:])

        grad[kk:] += wk * (1.0 - h[:n-kk])
        # grad[0:n-kk] -= wk * h[kk:n]
        grad[:n-kk] -= wk * h[kk:]
    
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def soft_weights(corr_dx, alpha):
    """Compute softmax weights for the smooth max"""
    m = corr_dx.max()
    s = alpha * (corr_dx - m)
    w = np.exp(s)
    w /= w.sum()
    return w

def compute_grad_vec(h, weights, n):
    """Vectorized gradient computation"""
    dx = 2.0 / n
    grad = np.zeros(n)
    offset = n - 1
    for k in range(n):
        wk = weights[offset + k]
        if wk < 1e-18:
            continue
        grad[:n-k] += wk * (1.0 - h[k:])
        grad[k:] -= wk * h[:n-k]
    for kk in range(1, n):
        wk = weights[offset - kk]
        if wk < 1e-18:
            continue
        grad[kk:] += wk * (1.0 - h[:n-kk])
        grad[:n-kk] -= wk * h[kk:]
    return grad * dx

def subgradient_optimize(h, n, alpha_start, alpha_end, n_steps, lr, momentum, deadline):
    """Subgradient method with momentum and adaptive alpha"""
    h = h.copy()
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    corr_dx = corr * (2.0 / n)
    best_c = corr_dx.max()
    velocity = np.zeros(n)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Adaptive alpha
        progress = step / max(n_steps - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        # Compute weights
        w = soft_weights(corr_dx, alpha)
        
        # Compute gradient
        grad = compute_grad_vec(h, w, n)
        
        # Momentum update
        velocity = momentum * velocity - lr * grad
        h = h + velocity
        
        # Project
        h = project_feasible(h, n)
        
        # Update correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        corr_dx = corr * (2.0 / n)
        c = corr_dx.max()
        
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    offset = n - 1
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        s = alpha * (corr - m)
        logsum = np.log(np.exp(s).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greed_binary(h_init, n, deadline, max_sweeps=200):
    """Binary local search via swaps"""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:n // 2]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_delta = 1e15
            best_j = -1
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_delta:
                    best_delta = nc
                    best_j = j
            if best_j >= 0 and best_delta < cur_c - 1e-15:
                # Apply swap
                i, j = i, best_j
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range

                jm = j + m_range

                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range

                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = best_delta
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def soft_weights(corr_dx, alpha):
    m = corr_dx.max()
    s = alpha * (corr_dx - m)
    w = np.exp(s)
    ss = w.sum()
    if ss < 1e-300:
        w = np.zeros(len(w))
        idx = np.argmax(corr_dx)
        w[idx] = 1.0
    else:
        w /= ss
    return w

def compute_grad_vec(h, weights, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    offset = n - 1
    for k in range(n):
        wk = weights[offset + k]
        if wk < 1e-18:
            continue
        grad[:n-k] += wk * (1.0 - h[k:])
        grad[k:] -= wk * h[:n-k]
    for kk in range(1, n):
        wk = weights[offset - kk]
        if wk < 1e-18:
            continue
        grad[kk:] += wk * (1.0 - h[:n-kk])
        grad[:n-kk] -= wk * h[kk:]
    return grad * dx

            break
        batch = min(50, n_steps - step)

            if time.time() > deadline:

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        s = alpha * (corr - m)
        logsum = np.log(np.exp(s).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
