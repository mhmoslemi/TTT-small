    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    lse = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    return lse + penalty_w * (h.sum() - n / 2.0) ** 2

    def obj(x):
        return smooth_max_objective(x, alpha, penalty_w)
