def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).reshape(-1), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (d / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (d / s) * h
    d = target - h.sum()
    if abs(d) > 1e-12:
        for idx in np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]:
            if abs(d) < 1e-12:
                break
            st = d
            u = h[idx] + st
            if u < 0.0:
                st = -h[idx]
            elif u > 1.0:
                st = 1.0 - h[idx]
            h[idx] += st
            d = target - h.sum()
    return np.clip(h, 0.0, 1.0)

def better_solution(c, n, best_c, best_n):
    if c < best_c - 1e-12:

    return False

def qab(h, n):
    o = 1.0 - h

    B = np.zeros(n)
    for d in range(n):
        m = n - d
        A[d] = float(np.dot(h[:m], o[d:]))
        if d:
            B[d] = float(np.dot(o[:m], h[d:]))
    return A, B

def active_weights(A, B, n, C):

        wa = np.zeros(n)
        wa[0] = 1.0
        return wa, np.zeros(n)

def active_gradient(h, wa, wb, n):

            G[d:] -= a * h[:-d]

def sum_c(x, n):
    return float(np.sum(x) - 0.5 * n)

def sum_j(x, n):
    return np.ones(n)

def soft_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = float(c.max())
    z = np.clip(alpha * (c - m), -40.0, 40.0)
    zmax = float(z.max())
    return float(m + (zmax + math.log(float(np.sum(np.exp(z - zmax))))) / alpha)

def resample(h, m):

            alloc = min(170.0, max(35.0, 0.25 * rem))
        else:
            alloc = min(130.0, max(25.0, 0.20 * rem))
        alloc = min(alloc, rem - 18.0)
        if alloc < 10.0:
            break
        try:
            hh, cc = optimize_start(starts[idx], time.time() + alloc, rng, idx == 0 and hp is not None)
            cc = compute_c5(hh)
            if better_solution(cc, len(hh), best_c, best_n):
                best_c = cc
                best_h = hh
                best_n = len(hh)
        except Exception:
            pass
    if best_h is not None and time.time() < deadline - 12.0:
        try:
            ff, cc = active_descent(best_h, len(best_h), deadline - 3.0, rng)
            if better_solution(cc, len(ff), best_c, best_n):
                best_c = cc
                best_h = ff
                best_n = len(ff)
        except Exception:
            pass
    if best_h is not None and best_n % 2 == 0 and time.time() < deadline - 8.0:
        try:
            bb = binary_round(best_h, best_n)
            cb = compute_c5(bb)
            if cb < best_c + 0.02:
                lb, cb = binary_local(bb, best_n, deadline - 1.0)
                if cb < best_c - 1e-12:
                    best_c = cb
                    best_h = lb
        except Exception:
            pass
    if best_h is None:
        best_h = project_feasible(np.full(100, 0.5), 100)
        best_c = compute_c5(best_h)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h)
    return best_h, float(best_c), len(best_h)
