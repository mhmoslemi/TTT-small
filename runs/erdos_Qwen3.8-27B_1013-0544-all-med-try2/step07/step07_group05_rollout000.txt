def softmax_counts(terms,alpha):
  m=terms.max()
  if alpha<=0: p=np.full(len(terms),1/len(terms)); return p
  z=alpha*(terms-m)
  e=np.exp(z)
  s=e.sum()

def corr_terms(h):
    n=int(len(h)); total=float(h.sum())
    N=fft_len(n)
    fh=np.fft.rfft(h,N)
    P=np.fft.irfft(fh*np.conjugate(fh),N)[:n]

        terms[n:]=total-pref[1:n]-P1
    np.maximum(terms,0.0,out=terms)
    return terms

def grad_from_p(h,p):

import numpy as np
import time
from scipy.optimize import minimize

def make_even(m):
    m = int(round(float(m)))
    if m < 2:
        m = 2
    return m if m % 2 == 0 else m + 1

def _pb(y, t):
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        s = float(np.clip(y - m, 0.0, 1.0).sum())
        if s > t:
            lo = m
        else:
            hi = m
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(6):
        d = t - float(h.sum())
        if abs(d) < 1e-13:
            break
        f = (h < 1.0) if d > 0 else (h > 0.0)
        m = int(f.sum())
        if m == 0:
            break
        h[f] += d / m
        h = np.clip(h, 0.0, 1.0)
    return h

def project(y, t):
    h = np.clip(y, 0.0, 1.0)
    d = t - float(h.sum())
    if abs(d) > 1e-13:
        f = (h < 1.0) if d > 0 else (h > 0.0)
        m = int(f.sum())
        if m > 0:
            h[f] += d / m
            h = np.clip(h, 0.0, 1.0)
            if abs(t - float(h.sum())) > 1e-12:
                h = _pb(y, t)
    return h

def fft_len(n):
    m = 2 * n - 1
    return 1 << (m - 1).bit_length()

def corr_terms(h):
    n = int(len(h))
    total = float(h.sum())
    N = fft_len(n)
    fh = np.fft.rfft(h, N)
    P = np.fft.irfft(fh * np.conjugate(fh), N)[:n]
    P = np.maximum(P.real, 0.0)
    pref = np.concatenate(([0.0], np.cumsum(h)))
    terms = np.empty(2 * n - 1)
    terms[0] = total - P[0]
    if n > 1:
        P1 = P[1:]
        terms[1:n] = pref[n - 1:0:-1] - P1
        terms[n:] = total - pref[1:n] - P1
    np.maximum(terms, 0.0, out=terms)
    return terms

def maxcnt(h):
    return float(corr_terms(h).max())

def cnt_c5(c, n):
    return float(c * 2.0 / n)

def lse_from_terms(terms, alpha):
    m = float(terms.max())
    if alpha <= 0.0:
        p = np.full_like(terms, 1.0 / len(terms))
        return m, p
    e = np.exp(alpha * (terms - m))
    s = float(e.sum())
    if s <= 0.0:
        p = np.zeros_like(terms)
        p[int(np.argmax(terms))] = 1.0
        return m, p
    return m + np.log(s) / alpha, e / s

def grad_from_p(h, p):
    n = int(len(h))
    L = n - 1
    if L <= 0:
        return np.zeros(n)
    a = p[n:]
    b = p[1:n]
    p0 = float(p[0])
    A = np.zeros(n)
    A[1:] = np.cumsum(a)
    cumb = np.concatenate(([0.0], np.cumsum(b)))
    B = cumb[L:0:-1].copy()
    W = a + b
    N = fft_len(n)
    fh = np.fft.rfft(h, N)
    K = np.zeros(N)
    K[1:n] = W
    K[N - L:N] = W[::-1]
    G = np.fft.irfft(fh * np.fft.rfft(K, N), N)[:n]
    return A + B + p0 - 2.0 * p0 * h - G

def fgrad(h, alpha):
    terms = corr_terms(h)
    f, p = lse_from_terms(terms, alpha)
    return f, grad_from_p(h, p), float(terms.max())

def obj_lbfgs(x, n, alpha, lam):
    f, g, _ = fgrad(x, alpha)
    err = float(x.sum()) - 0.5 * n
    return f + 0.5 * lam * err * err, g + lam * err

def lbfgs(h, n, alpha, lam, maxit):
    try:
        res = minimize(
            obj_lbfgs, h, args=(n, alpha, lam),
            method="L-BFGS-B", bounds=[(0.0, 1.0)] * n, jac=True,
            options={"maxiter": maxit, "maxfun": 10 * maxit,
                     "ftol": 1e-12, "gtol": 1e-4, "maxls": 20}
        )
        return project(res.x, 0.5 * n)
    except Exception:
        return h.copy()

def active_moves(h, cnt, end, rng):
    n = len(h)
    terms = corr_terms(h)
    top = np.argpartition(terms, -3)[-3:]
    for k in top:
        if k == 0:
            continue
        if k >= n:
            m = int(k - n + 1)
            vals = h[m:] * (1.0 - h[:-m])
        else:
            m = int(k)
            vals = h[:n - m] * (1.0 - h[m:])

            continue
        cand = list(np.argsort(vals)[::-1][:4])

                return h, cnt

                ht = h.copy()
                ht[src] -= dd
                ht[dst] += dd
                ct = maxcnt(ht)
                if ct < cnt - 1e-12:

    return h, cnt
