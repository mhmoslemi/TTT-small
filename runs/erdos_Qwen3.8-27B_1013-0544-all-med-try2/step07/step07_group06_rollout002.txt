def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h,0,1)
    target = n/2.0
    for _ in range(100):
        s=h.sum()
        diff=target-s
        if abs(diff)<1e-13: break
        if diff>0:
            room=1.0-h
            tr=room.sum()
            if tr<1e-15: break
            h += diff*room/tr
        else:
            ts=h.sum()
            if ts<1e-15: break
            h -= (-diff)*h/ts
    h=np.clip(h,0,1)
    r=target-h.sum()
    if r>1e-15:
       idx=int(np.argmax(1.0-h)); h[idx]=min(1.0,h[idx]+r)
    elif r<-1e-15:
       idx=int(np.argmax(h)); h[idx]=max(0.0,h[idx]+r)
    return h
```

import numpy as np
import time


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-15:
                break
            h += diff * room / tr
        else:
            ts = h.sum()
            if ts < 1e-15:
                break
            h -= (-diff) * h / ts
    h = np.clip(h, 0.0, 1.0)
    r = target - h.sum()
    if r > 1e-14:
        i = int(np.argmax(1.0 - h))
        h[i] = min(1.0, h[i] + r)
    elif r < -1e-14:
        i = int(np.argmax(h))
        h[i] = max(0.0, h[i] + r)
    return h


def cost_vec(h):
    return np.correlate(h, 1.0 - h, mode="full")


def objective(h):
    return float(np.max(cost_vec(h)) * (2.0 / len(h)))


def make_weights(c0, top):
    top = int(min(max(1, int(top)), len(c0)))
    w = np.zeros(len(c0), dtype=float)
    w[np.argsort(c0)[-top:]] = 1.0 / top
    return w


def weighted_gradient(h, n, weights):
    g = np.zeros(n, dtype=float)
    idxs = np.nonzero(weights > 0.0)[0]
    for idx in idxs:
        w = float(weights[idx])
        k = int(idx) - (n - 1)
        if k == 0:
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            nv = n - k
            g[:nv] -= w * h[k:]
            g[k:] += w * (1.0 - h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[:nv] += w * (1.0 - h[kk:])
            g[kk:] -= w * h[:nv]
    return g


def shift_gradient(h, n, k):
    g = np.zeros(n, dtype=float)
    if k == 0:
        return 1.0 - 2.0 * h
    if k > 0:
        nv = n - k
        g[:nv] -= h[k:]
        g[k:] += 1.0 - h[:nv]
    else:
        kk = -k
        nv = n - kk
        g[:nv] += 1.0 - h[kk:]
        g[kk:] -= h[:nv]
    return g


def score_pair(i, j, h, g, seen, pairs, scores, mult=1.0):
    if i == j:
        return
    if i > j:
        i, j = j, i
    if (i, j) in seen:
        return
    seen.add((i, j))
    cap = min(1.0 - float(h[i]), float(h[j])) + min(float(h[i]), 1.0 - float(h[j]))
    scores.append(float(abs(g[i] - g[j]) * cap * mult))
    pairs.append((i, j))


def pair_linear_coeffs(h, n, i, j):
    tt = np.arange(n, dtype=int)
    A = np.zeros(2 * n - 1, dtype=float)
    A[n - 1 + i - tt] += 1.0 - h[tt]
    A[n - 1 + tt - i] -= h[tt]
    A[n - 1 + j - tt] -= 1.0 - h[tt]
    A[n - 1 + tt - j] += h[tt]
    return A


def pair_f_scalar(c0, A, t, sidx, p1, p2):
    v = c0 + A * t
    u = t * t
    v[sidx] -= 2.0 * u
    v[p1] += u
    v[p2] += u
    return float(np.max(v))


def pair_line_candidate(h, c0, i, j, grid_points=24, refine_iters=12):
    n = len(h)
    dx = 2.0 / n
    base = float(np.max(c0) * dx)
    if i == j:
        return 0.0, base
    lo = max(-float(h[i]), float(h[j]) - 1.0)
    hi = min(1.0 - float(h[i]), float(h[j]))
    if hi - lo < 1e-14:
        return 0.0, base
    A = pair_linear_coeffs(h, n, i, j)
    sidx = n - 1
    p1 = n - 1 + i - j
    p2 = n - 1 + j - i
    G = int(max(8, grid_points))
    tt = np.linspace(lo, hi, G)
    tt2 = tt * tt
    v = c0[None, :] + A[None, :] * tt[:, None]
    v[:, sidx] -= 2.0 * tt2
    v[:, p1] += tt2
    v[:, p2] += tt2
    vals = np.max(v, axis=1)
    pos = int(np.argmin(vals))
    best_t = float(tt[pos])
    best_f = float(vals[pos] * dx)
    for t0 in (0.0, lo, hi):
        f0 = pair_f_scalar(c0, A, t0, sidx, p1, p2) * dx
        if f0 < best_f:
            best_f = f0
            best_t = float(t0)
    if hi - lo > 1e-12:
        if pos == 0:
            a, b = lo, 0.5 * (tt[0] + tt[1])
        elif pos == G - 1:
            a, b = 0.5 * (tt[-1] + tt[-2]), hi
        else:
            a, b = tt[pos - 1], tt[pos + 1]
        if b - a > 1e-13:
            for _ in range(int(refine_iters)):
                m1 = a + (b - a) * 0.3819660112501051
                m2 = a + (b - a) * 0.6180339887498949
                f1 = pair_f_scalar(c0, A, m1, sidx, p1, p2)
                f2 = pair_f_scalar(c0, A, m2, sidx, p1, p2)
                if f1 <= f2:
                    b = m1
                else:
                    a = m2
            tmid = 0.5 * (a + b)
            fmid = pair_f_scalar(c0, A, tmid, sidx, p1, p2) * dx
            if fmid < best_f:
                best_f = fmid
                best_t = float(tmid)
    best_t = float(np.clip(best_t, lo, hi))
    return best_t, float(best_f)


def pair_polish(h, n, rng, deadline, grid_points=24, top_eval=50, random_pairs=20, K=12):

    best_h = h.copy()
    dx = 2.0 / n
    stall = 0
    accepted = 0
    sweep = 0
    while time.time() < deadline:
        c0 = cost_vec(h)
        cur = float(np.max(c0) * dx)
        if cur < best_c:

            best_h = h.copy()
        g = weighted_gradient(h, n, make_weights(c0, 12))
        g -= g.mean()
        seen = set()
        pairs = []
        scores = []
        Kuse = int(min(max(2, K + (3 if stall > 1 else 0)), n // 2))
        order = np.argsort(g)
        low = order[:Kuse]
        high = order[-Kuse:]
        for a in low:
            for b in high:
                score_pair(int(a), int(b), h, g, seen, pairs, scores, 1.0)
        for idx in np.argsort(c0)[-4:]:
            gk = shift_gradient(h, n, int(idx) - (n - 1))
            ok = np.argsort(gk)
            l = ok[:max(2, Kuse // 2)]
            hh = ok[-max(2, Kuse // 2):]
            for a in l:
                for b in hh:
                    score_pair(int(a), int(b), h, gk, seen, pairs, scores, 2.0)
        for _ in range(random_pairs):
            a = int(rng.integers(n))

        if not pairs:
            return best_h, best_c
        arr = list(zip(scores, pairs))
        arr.sort(reverse=True)
        eval_limit = int(top_eval)
        if stall >= 2:
            eval_limit = int(min(220, top_eval + 60 + 20 * stall))
        if sweep % 9 == 8:
            eval_limit = max(eval_limit, int(min(250, 3 * top_eval)))
        selected = arr[:eval_limit]
        best_pair = None
        best_pair_val = cur + 1.0
        for _, (i, j) in selected:
            if time.time() >= deadline:
                break
            t, _ = pair_line_candidate(h, c0, i, j, grid_points=grid_points)
            if abs(t) < 1e-14:
                continue
            hn = h.copy()
            hn[i] += t
            hn[j] -= t
            hn = np.clip(hn, 0.0, 1.0)
            ce = objective(hn)
            if ce < best_pair_val:
                best_pair_val = ce
                best_pair = (i, j, t, hn)
        accept_improve = best_pair is not None and best_pair_val < cur - 1e-13
        anneal_limit = 0.0

def binary_push(h, n, rng, deadline):
    h = project_feasible(np.array(h, dtype=float).copy(), n)
    cur = objective(h)
    best_h = h.copy()
    for _ in range(80):
        if time.time() >= deadline:
            break
        improved = False
        for i in np.argsort(np.minimum(h, 1.0 - h)):
            if time.time() >= deadline:
                break
            x = float(h[i])
            if x < 0.25:
                d = -x
                tgt = 0.0
            elif x > 0.75:
                d = 1.0 - x
                tgt = 1.0
            else:
                continue
            if abs(d) < 1e-12:
                continue
            m = (h > d) if d > 0 else (h < -d)
            m[i] = False
            if not m.any():
                m = (h > 1e-12) & (h < 1.0 - 1e-12)
                m[i] = False
            if not m.any():
                continue
            hn = h.copy()
            hn[i] = tgt
            hn[m] -= d / float(m.sum())
            hn = project_feasible(hn, n)
            c = objective(hn)
            if c < cur - 1e-13:
                h = hn
                cur = c
                best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, cur


def transfer_to_n(h, n):
    old = len(h)
    if old == n:
        return project_feasible(np.array(h, dtype=float).copy(), n)
    x = np.linspace(0.0, 2.0, old)
    y = np.linspace(0.0, 2.0, n)
    return project_feasible(np.interp(y, x, h), n)
