def slsqp_refine(h, n, alpha, maxiter=2000, ftol=1e-14):
    """Single SLSQP call with given alpha."""
    dx = 2.0 / n

        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    bounds = [(0.0, 1.0)] * n

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return h

        grad = np.zeros(n)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):

            break

            if s < 1e-15:
                break
            h = h + diff / s * h

def coord_descent(h, n, max_sweeps=200, deadline=None, bits=50):

    bounds = [(0.0, 1.0)] * n
