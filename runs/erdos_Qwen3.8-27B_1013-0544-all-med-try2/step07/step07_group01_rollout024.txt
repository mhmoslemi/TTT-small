            h[i] = c2; h[j] = s - c2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_corr(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val + mu * (x.sum() - n / 2.0) ** 2)

def smooth_grad(x, n, alpha, mu):
    c = compute_corr(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        wm = w[k_idx] * dx_val
        if wm < 1e-14:

        p_lo = max(0, -k)
        p_hi = min(n - 1, n - 1 - k)
        grad[p_lo:p_hi+1] += wm * (1.0 - x[p_lo+k:p_hi+k+1])
        grad[p_lo+k:p_hi+k+1] += wm * (-x[p_lo:p_hi+1])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    h[i] = c1; h[j] = s - c1
    f1 = compute_c5(h, n)
    h[i] = c2; h[j] = s - c2
    f2 = compute_c5(h, n)
    for _ in range(max_evals - 2):
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            h[i] = c1; h[j] = s - c1
            f1 = compute_c5(h, n)
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            h[i] = c2; h[j] = s - c2
            f2 = compute_c5(h, n)
    best_t = t0
    best_f = c0
    for t in [t0, (a + b) * 0.5, max(lo, min(hi, (a+b)*0.3)), max(lo, min(hi, (a+b)*0.7))]:
        t = max(lo, min(hi, t))
        h[i] = t; h[j] = s - t
        f = compute_c5(h, n)
        if f < best_f:
            best_f = f
            best_t = t
    h[i] = best_t; h[j] = s - best_t
    return best_f < c0 - 1e-15

def full_sweep(h, n, radius, max_evals=30):
    improved = 0
    for i in range(n):
        for j in range(i + 1, min(n, i + radius + 1)):
            if pair_ls(h, n, i, j, max_evals):
                improved += 1
    return improved

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        h = np.full(250, 0.5)
        n = 250
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: L-BFGS-B (100s)
    t_p1 = min(t_start + 100, deadline - 700)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 1.0, 100), (500, 5.0, 100), (2000, 20.0, 150),
               (10000, 100.0, 200), (50000, 500.0, 200), (300000, 2000.0, 200)]
    for alpha, mu, maxit in configs:
        if time.time() > t_p1:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18, 'maxfun': maxit * 5})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    # Phase 2: Full sweep pair line search with decreasing radius (700s)
    t_p2_end = deadline - 150
    radii = [30, 25, 20, 15, 10, 8, 5, 3, 2, 1]
    for radius in radii:
        if time.time() > t_p2_end:
            break
        max_sweeps = max(1, int((t_p2_end - time.time()) / 30))
        for sw in range(max_sweeps):
            if time.time() > t_p2_end:
                break
            improved = full_sweep(h, n, radius, max_evals=30)
            c_now = compute_c5(h, n)
            if c_now < best_c - 1e-15:
                best_c = c_now
                best_h = h.copy()
            if improved == 0:
                break
    
    # Phase 3: Random + gradient-guided pair search (100s)
    if time.time() < deadline - 50:
        t_p3 = min(time.time() + 80, deadline - 50)
        no_improve = 0
        while time.time() < t_p3 and no_improve < 20:
            c_before = compute_c5(h, n)
            g = smooth_grad(h, n, 5000.0, 0.0)
            improved = 0
            # Top gradient pairs
            idx = np.argsort(-np.abs(g))
            for ii in range(min(60, n)):
                i = idx[ii]
                for j in [i-1, i+1, i-2, i+2, i-5, i+5]:
                    if 0 <= j < n and j != i:
                        if pair_ls(h, n, i, j, max_evals=25):
                            improved += 1
            # Random pairs
            for _ in range(200):
                i, j = rng.randint(n), rng.randint(n)
                if i != j and pair_ls(h, n, i, j, max_evals=20):
                    improved += 1
            c_now = compute_c5(h, n)
            if c_now < best_c - 1e-15:
                best_c = c_now
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
                # Perturb to escape
                h += rng.randn(n) * 0.002
                h = project_feasible(h, n)
    
    # Phase 4: Try slightly different n (50s)
    if time.time() < deadline - 20:
        for n_try in [320, 280, 350]:
            if n_try == n or time.time() > deadline - 15:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            ht = np.interp(x_new, x_old, best_h) + rng.randn(n_try) * 0.005
            ht = project_feasible(ht, n_try)
            try:
                res = minimize(smooth_obj, ht, jac=smooth_grad, args=(n_try, 50000, 500.0),
                              method='L-BFGS-B', bounds=[(0.0,1.0)]*n_try,
                              options={'maxiter': 80, 'ftol': 1e-16})
                ht = project_feasible(res.x, n_try)
            except Exception:
                pass
            if time.time() < deadline - 10:
                full_sweep(ht, n_try, 3, max_evals=20)
                full_sweep(ht, n_try, 2, max_evals=20)
            c_try = compute_c5(ht, n_try)
            if c_try < best_c:
                best_c = c_try
                best_h = ht.copy()
                n = n_try
    
    best_h = project_feasible(best_h, len(best_h))
    final_c = compute_c5(best_h, len(best_h))
    if final_c > best_c:
        best_h = project_feasible(best_h, len(best_h))
        final_c = compute_c5(best_h, len(best_h))
    return best_h, final_c, len(best_h)
