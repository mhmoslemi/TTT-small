    best_h = h.copy()
    best_c = cur_c
    ones = [int(x) for x in np.where(h == 1)[0]]
    zeros = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            for j in ones:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def anneal_binary(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = np.zeros(n)
    h[np.argsort(h_init)[::-1][:n // 2]] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_arr = np.where(h == 1)[0].astype(int)
    zeros_arr = np.where(h == 0)[0].astype(int)
    T = 0.01
    cool = 0.99997
    scale = 2000.0
    while time.time() < deadline:
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i, j = int(zeros_arr[ai]), int(ones_arr[bi])
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def slsqp_polish(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + sm / alpha + np.log(np.exp(shifted - sm).sum()) / alpha
    def grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        exps = np.exp(shifted - sm)
        w = exps / exps.sum()
        gr = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            L = k - (n - 1)
            if L >= 0:
                m1 = np.arange(n - L)
                gr[m1] += wk * (1.0 - x[m1 + L])
                m2 = np.arange(L, n)
                gr[m2] -= wk * x[m2 - L]
            else:
                L2 = -L
                m1 = np.arange(L2, n)
                gr[m1] += wk * (1.0 - x[m1 - L2])
                m2 = np.arange(n - L2)
                gr[m2] -= wk * x[m2 + L2]
        return gr
    cons_fun = lambda x: x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: binary local search
    p1 = t0 + 0.06 * budget_s
    hb, cb = binary_search(h0, n, p1, max_sweeps=400)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 2: PG continuous optimization main
    p2 = t0 + 0.40 * budget_s
    h = best_h.copy()
    alphas = [50, 100, 300, 1000, 3000, 10000, 30000, 100000, 300000, 1000000, 3000000]
    for alpha in alphas:
        if time.time() > p2 - 10:
            break
        h, f = pgd_opt(h, n, alpha, 8000, p2 - 10)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 3: SLSQP polish
    p3 = t0 + 0.52 * budget_s
    h = best_h.copy()
    for alpha in [50000, 500000, 5000000, 50000000]:
        if time.time() > p3 - 5:
            break
        h = slsqp_polish(h, n, alpha, 800, p3 - 5)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 4: scatter + re-optimize
    p4 = t0 + 0.82 * budget_s
    for trial in range(30):
        if time.time() > p4 - 20:
            break
        scale = 0.01 + 0.04 * rng.random()
        h = best_h + scale * rng.randn(n)
        h = project(h, n)
        hb, cb = binary_search(h, n, min(time.time() + 8, p4 - 20), max_sweeps=80)
        for alpha in [5000, 50000, 500000, 5000000]:
            if time.time() > p4 - 18:
                break
            hb, _ = pgd_opt(hb, n, alpha, 3000, p4 - 18)
            c = compute_c5(hb, n)
            if c < best_c:
                best_c, best_h = c, hb.copy()
    # Phase 5: final PG polish
    h = best_h.copy()
    for alpha in [10000000, 50000000, 200000000]:
        if time.time() > deadline - 3:
            break
        h, _ = pgd_opt(h, n, alpha, 5000, deadline - 3)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    best_h = project(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
