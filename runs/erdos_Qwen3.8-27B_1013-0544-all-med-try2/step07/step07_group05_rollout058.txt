    corr = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    max_sh = sh.max()
    lse = max_sh + np.log(np.sum(np.exp(sh - max_sh)))
    return m + lse / alpha

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj_func(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    max_sh = sh.max()
    lse = max_sh + np.log(np.sum(np.exp(sh - max_sh)))
    return float(m + lse / alpha)

def _sum_con(x, n):
    return float(x.sum() - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha=50000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_func, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def sa_explore(h_start, n, deadline, rng, T0=0.04, cooling=0.9999):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    T = T0
    step_count = 0

    while time.time() < deadline:
        h_old = h.copy()
        r = rng.random()

        if r < 0.5:
            bs = int(rng.randint(1, min(16, n // 4)))
            i = rng.randint(0, n - bs)
            old = h[i:i+bs].copy()
            scale = 0.08 + 0.25 * (T / T0)
            noise = rng.normal(0, scale, bs)
            new = np.clip(old + noise, 0, 1)
            diff = old.sum() - new.sum()
            if abs(diff) > 1e-12:
                if diff > 0:
                    room = 1 - new
                    if room.sum() < 1e-12:
                        step_count += 1
                        T *= cooling
                        continue
                    new += diff * room / room.sum()
                else:
                    if new.sum() < 1e-12:
                        step_count += 1
                        T *= cooling
                        continue
                    new += diff * new / new.sum()
            new = np.clip(new, 0, 1)
            h[i:i+bs] = new
        elif r < 0.8:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                step_count += 1
                T *= cooling
                continue
            delta = rng.uniform(-0.2, 0.2)
            if h[i] + delta > 1.0 or h[j] - delta < 0.0:
                step_count += 1
                T *= cooling
                continue
            h[i] += delta
            h[j] -= delta
        else:
            bs = int(rng.randint(3, max(4, n // 5)))
            i = rng.randint(0, n - bs)
            old = h[i:i+bs].copy()
            new = np.clip(old + rng.normal(0, 0.5, bs), 0, 1)
            diff = old.sum() - new.sum()
            if diff > 1e-12:
                room = 1 - new
                if room.sum() > 1e-12:
                    new += diff * room / room.sum()
            elif diff < -1e-12:
                if new.sum() > 1e-12:
                    new += diff * new / new.sum()
            new = np.clip(new, 0, 1)
            h[i:i+bs] = new

        new_c = compute_c5(h, n)
        delta_e = new_c - cur_c
        if delta_e < 0 or (T > 1e-10 and rng.random() < np.exp(-min(delta_e / T, 400))):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h = h_old

        T *= cooling
        step_count += 1
        if step_count % 20000 == 0 and T < 0.004:
            T = 0.015

    return best_h, best_c

def basin_hopping(h_start, n, deadline, rng):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = 0.006
    stall = 0

    while time.time() < deadline - 10:
        scale = 0.04 + 0.3 * min(float(stall) / 12.0, 1.0)
        h_pert = h + rng.normal(0, scale, n)
        h_pert = project_feasible(h_pert, n)

        h_local = slsqp_optimize(h_pert, n, alpha=50000, maxiter=800)
        c_local = compute_c5(h_local, n)
        c_cur = compute_c5(h, n)

        delta = c_local - c_cur
        if delta < 0:
            h = h_local
            stall = 0
        elif T > 1e-10 and rng.random() < np.exp(-min(delta / T, 400)):
            h = h_local
            stall += 1
        else:
            stall += 1

        if c_local < best_c:
            best_c = c_local
            best_h = h_local.copy()
            stall = 0

        T *= 0.92
        if T < 0.0008:
            T = 0.006

    return best_h, best_c

def pair_polish(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c
