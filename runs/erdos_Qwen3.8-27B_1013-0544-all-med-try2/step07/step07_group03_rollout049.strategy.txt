            new_h[start_idx:start_idx+window_size] = np.clip(patch, 0.0, 1.0)
        
        # Project to ensure sum and bounds
        new_h = project_feasible(new_h, n)
        if abs(new_h.sum() - n/2) < original_sum * 0.8:
            candidates.append(new_h)
    
    return candidates
