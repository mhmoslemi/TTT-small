import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    ac = alpha * corr * dx
    amax = ac.max()
    exp_ac = np.exp(ac - amax)
    p_sum = exp_ac.sum()
    obj = (amax + np.log(p_sum)) / alpha
    p = exp_ac / p_sum
    a = 1.0 - h[::-1]
    first = np.correlate(p, a, mode='valid')
    second = np.correlate(p, h, mode='valid')
    grad = dx * (first - second[::-1])
    return obj, grad

def slsqp_run(h, n, alpha, maxiter, deadline):
    target = n / 2.0
    def obj_fun(x):
        return smooth_obj_grad(x, n, alpha)[0]
    def grad_fun(x):
        return smooth_obj_grad(x, n, alpha)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def lbfgsb_run(h, n, alpha, penalty, maxiter, deadline):
    target = n / 2.0
    def obj_fun(x):
        v, g = smooth_obj_grad(x, n, alpha)
        d = x.sum() - target
        return v + 0.5 * penalty * d * d
    def grad_fun(x):
        v, g = smooth_obj_grad(x, n, alpha)
        d = x.sum() - target
        return g + penalty * d
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    corr = np.correlate(hb, 1.0 - hb, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = hb.copy()
    rng = np.random.RandomState(seed)
    m_arr = np.arange(2 * n - 1)
    shift = m_arr - (n - 1)
    stall = 0
    for it in range(50000):
        if time.time() > deadline:

        improved = False
        n_try = 300
        for _ in range(n_try):

            j = rng.randint(n)
            if hb[i] == 0.0 and hb[j] == 1.0:
                delta = np.zeros(2 * n - 1)
                mk = (i - shift >= 0) & (i - shift < n)
                delta[mk] += 1.0 - hb[i - shift[mk]]
                mk = (i + shift >= 0) & (i + shift < n)
                delta[mk] -= hb[i + shift[mk]]
                mk = (j - shift >= 0) & (j - shift < n)

                mk = (j + shift >= 0) & (j + shift < n)
