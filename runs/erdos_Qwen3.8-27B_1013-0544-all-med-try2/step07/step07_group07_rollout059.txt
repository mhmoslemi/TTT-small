def compute_c5(h, n):
    dx = 2.0 / n

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha, top_m):
    c = compute_all_lags(x, n)
    if top_m is not None and top_m < len(c):
        idx = np.argsort(c)[-top_m:]
        c = c[idx]
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha, top_m):
    c = compute_all_lags(x, n)
    if top_m is not None and top_m < len(c):
        idx = np.argsort(c)[-top_m:]
        c = c[idx]
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sum_eq(x, n):
    return float(x.sum() - n / 2.0)

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000, ftol=1e-15, top_m=None):
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha, top_m), method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m < n:
        ne = n - m
        if lag > 0:
            g[:ne] += 1.0 - h[m:]
            g[m:] -= h[:ne]
        else:
            g[:ne] -= h[m:]
            g[m:] += 1.0 - h[:ne]
    return g

def proj_gradient(h, n, alpha, lr0, n_iter, deadline, top_m=15):

    best_h, best_c = h_c.copy(), compute_c5(h_c, n)
    lr = lr0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_c, n)
        idx = np.argsort(c)[-top_m:]
        ct = c[idx]
        w = np.exp(alpha * (ct - ct.max()))
        w /= w.sum()
        g = np.zeros(n)
        for i, ki in enumerate(idx):
            g += w[i] * lag_grad(h_c, n, int(ki) - (n - 1))
        h_new = project_feasible(h_c - lr * g, n)
        cn = compute_c5(h_new, n)
        if cn < best_c - 1e-16:
            best_c, best_h = cn, h_new.copy()
            lr = min(lr * 1.05, 0.08)
        else:
            lr *= 0.6
        h_c = h_new
    return best_h, best_c

def outer_approx(h, n, deadline, max_it=40):

def bin_round(h, n):
