    block2 = h[start2:start2+block_size]
    
    # Transfer from block2 to block1
    max_to_1 = min(step, (1.0 - block1).max(), block2.min())
    max_to_2 = min(step, block1.min(), (1.0 - block2).max())
    
    if rng.random() < 0.5:
        d = max_to_1 * rng.uniform(0.3, 1.0)
        if d < 1e-8:
            return False, 0
        # Distribute d across block1 (add) and block2 (subtract)
        # Proportional to available room
        room1 = 1.0 - block1
        room2 = block2
        total_room1 = room1.sum()
        total_room2 = room2.sum()
        if total_room1 < 1e-10 or total_room2 < 1e-10:
            return False, 0
        h[start1:start1+block_size] += d * room1 / total_room1
        h[start2:start2+block_size] -= d * room2 / total_room2
    else:
        d = max_to_2 * rng.uniform(0.3, 1.0)
        if d < 1e-8:
            return False, 0
        room1 = block1
        room2 = 1.0 - block2
        total_room1 = room1.sum()
        total_room2 = room2.sum()
        if total_room1 < 1e-10 or total_room2 < 1e-10:
            return False, 0
        h[start1:start1+block_size] -= d * room1 / total_room1
        h[start2:start2+block_size] += d * room2 / total_room2
    
    return True, d

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slsqp(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    def cfun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': cfun}]
    try:
        res = minimize(lse_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def sa_optimize(h_start, n, deadline, rng, T0=0.025):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_cur = best_c
    T = T0
    step = 0.06
    no_improve = 0
    step_count = 0
    accept_count = 0

    while time.time() < deadline:
        if step_count > 300:
            rate = accept_count / max(1, step_count)
            if rate > 0.6:
                step = min(0.25, step * 1.12)
            elif rate < 0.15:
                step = max(0.003, step * 0.85)
            step_count = 0
            accept_count = 0

        if no_improve > 25000:
            h = best_h.copy()
            c_cur = best_c
            T = max(T * 0.65, 0.0008)
            step = max(0.02, step * 0.7)
            no_improve = 0

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue

        pos = rng.random() < 0.5
        if pos:
            d_max = min(step, 1.0 - h[i], h[j])
        else:
            d_max = min(step, h[i], 1.0 - h[j])

        if d_max < 1e-9:
            continue
        d = d_max * rng.uniform(0.05, 1.0)

        si, sj = h[i], h[j]
        if pos:
            h[i] += d
            h[j] -= d
        else:
            h[i] -= d
            h[j] += d

        c_new = compute_c5(h, n)
        delta = c_new - c_cur

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_cur = c_new
            accept_count += 1
            step_count += 1
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = si, sj
            no_improve += 1
            step_count += 1

        T *= 0.99992

    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair(h, n, deadline, deltas, radius=45):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break
    return h, best_c
