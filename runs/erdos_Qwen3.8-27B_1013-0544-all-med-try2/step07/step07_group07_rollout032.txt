def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

            if tot < 1e-14:

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp(h, n, alpha=50000, maxiter=3000):

def lag_gradient(h, n, shift):
    """Compute gradient of c_shift with respect to h."""
    dx = 2.0 / n
    g = np.zeros(n)
    if shift == 0:
        g[:] = (1 - 2*h) * dx
    elif shift > 0:
        g[:n-shift] += (1 - h[shift:]) * dx
        g[shift:] -= h[:n-shift] * dx
    else:
        m = -shift
        g[m:] += (1 - h[:n-m]) * dx
        g[:n-m] -= h[m:] * dx
    return g

def projected_gradient(h, n, deadline, max_iters=2000, eps=1e-4):
    """Projected subgradient descent on max overlap."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        c_max = c.max()
        
        # Active set
        active = np.where(c >= c_max - eps)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Compute subgradient (average of active gradients)
        shifts = active - (n - 1)
        g_sum = np.zeros(n)
        for s in shifts:
            g_sum += lag_gradient(h, n, int(s))
        g_avg = g_sum / len(active)
        
        # Project to sum-zero
        d = g_avg - np.mean(g_avg)
        
        # Step in direction -d (to decrease the max)
        neg_d = -d
        # Find max feasible step
        pos_mask = neg_d > 1e-15
        neg_mask = neg_d < -1e-15
        if pos_mask.any():
            max_pos = np.min((1 - h[pos_mask]) / neg_d[pos_mask])
        else:
            max_pos = np.inf
        if neg_mask.any():
            max_neg = np.min(h[neg_mask] / (-neg_d[neg_mask]))
        else:
            max_neg = np.inf
        max_step = max(1e-10, min(max_pos, max_neg) * 0.9)
        
        # Try a few step sizes with backtracking
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = max_step * frac
            h_new = h - step * d
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                improved = True
                break
        
        if not improved:
            # Try smaller eps or break
            eps *= 0.5
            if eps < 1e-8:
                break
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sc(x, n):
    return float(np.sum(x) - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha=50000, maxiter=4000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, shift):
    dx = 2.0 / n
    g = np.zeros(n)
    if shift == 0:
        g[:] = (1 - 2 * h) * dx
    elif shift > 0:
        g[:n - shift] += (1 - h[shift:]) * dx
        g[shift:] -= h[:n - shift] * dx
    else:
        m = -shift
        g[m:] += (1 - h[:n - m]) * dx
        g[:n - m] -= h[m:] * dx
    return g
