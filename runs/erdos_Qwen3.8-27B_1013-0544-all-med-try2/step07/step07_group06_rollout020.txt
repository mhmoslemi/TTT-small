import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    if mx > 500:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = np.exp(c_shift - mx)
        w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_opt(h, n, deadline, alpha=10000, maxit=200, mu=100.0):
    h = h.copy()
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_value(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
    def gr(x):
        g = grad_smooth_max(x, n, alpha)
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    best_c = compute_c5(h, n)
    try:
        res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-12,
                               'maxfun': maxit * 15, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h = h_new
            best_c = c_new
    except Exception:
        pass
    return h, best_c

def mirror_descend(h, n, deadline, alpha_init=100, alpha_final=500000, lr_init=0.1):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    max_steps = 3000
    for step in range(max_steps):
        if time.time() > deadline:
            break
        frac = min(step / max_steps, 1.0)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        lr = lr_init * (1.0 + 0.3 * step) ** (-0.5)
        g = grad_smooth_max(h, n, alpha)
        # Multiplicative update
        h_new = h * np.exp(-lr * g)
        h_new = np.clip(h_new, 1e-10, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller step
            lr2 = lr * 0.5
            h_try = h * np.exp(-lr2 * g)
            h_try = np.clip(h_try, 1e-10, 1.0)
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-16:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try
            else:
                lr_init *= 0.9
    return best_h, best_c

def gradient_pair_search(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, 5000)
            # For each i, find best j to transfer to/from
            # g[i] > 0 means increasing h[i] increases objective → we want to decrease h[i]
            # g[i] < 0 means increasing h[i] decreases objective → we want to increase h[i]
            order = np.argsort(g)  # most negative first (want to increase these)
            for idx in range(n):
                if time.time() > deadline:
                    return h, best_c
                i = order[idx]
                if g[i] >= -1e-10:
                    break  # no more points we want to increase
                # Find j to take from (highest gradient, i.e., we want to decrease)
                for jdx in range(len(order)-1, max(len(order)-1-20, -1), -1):
                    j = order[jdx]
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                        break  # only try one j per i per pass
            if not improved:
                break
    return h, best_c

def full_pair_search(h, n, deadline, deltas, radius=100):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:

                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:

def binary_push(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(100):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            if 0.05 < h[i] < 0.95:
                continue
            target = 0.0 if h[i] < 0.5 else 1.0
            delta = target - h[i]
            if abs(delta) < 1e-10:
                continue
            candidates = np.where((h > 0) & (h < 1) & (np.arange(n) != i))[0]
            if len(candidates) == 0:
                continue
            # Spread the delta among candidates
            h_new = h.copy()
            h_new[i] = target
            # Distribute to candidates with room
            room = np.zeros(n)
            room[i] = 0
            if delta > 0:
                room = 1.0 - h_new
                room[i] = 0
            else:
                room = h_new
                room[i] = 0
            tot = room.sum()
            if tot < abs(delta) * 0.99:
                continue
            h_new += (delta / tot) * room
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c - 1e-15:
                best_c = c
                h = h_new
                improved = True
        if not improved:
            break
    return h, best_c

def make_periodic(n, period, duty):
    """Create a periodic pattern with given period and duty cycle."""
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    return project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    if mx > 500:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = np.exp(c_shift - mx)
        w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_opt(h, n, deadline, alpha=10000, maxit=200, mu=100.0):

    best_c = compute_c5(h, n)
    best_x = h.copy()
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_value(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
    def gr(x):
        g = grad_smooth_max(x, n, alpha)
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-12,
                               'maxfun': maxit * 15, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_x = h_new
    except Exception:
        pass
    return best_x, best_c
