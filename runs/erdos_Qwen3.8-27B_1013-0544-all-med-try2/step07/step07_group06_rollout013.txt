```
def build_grad_matrix(h,n):
    L=2*n-1
    G=np.zeros((L,n),dtype=float)
    shift=n-1
    for m in range(L):
        k=m-shift
        if k>=0:
```

```python
import numpy as np
import time
from scipy.optimize import minimize, linprog, differential_evolution

_STATE = {"n": 300}

def compute_c5_vec(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    T = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    d = T - h.sum()
    if abs(d) < 1e-13:
        return h
    for _ in range(30):
        d = T - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (d / s) * h
    h = np.clip(h, 0.0, 1.0)
    d = T - h.sum()
    if abs(d) > 1e-12:
        if d > 0.0:
            r = 1.0 - h
            s = r.sum()
            if s > 1e-15:
                h += (d / s) * r
        else:
            s = h.sum()
            if s > 1e-15:
                h += (d / s) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_weights(w, h, n):
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    shift = n - 1
    for m in range(2 * n - 1):
        wi = w[m]
        if wi < 1e-18:
            continue
        k = m - shift
        wm = wi * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] -= wm * h[:nv]
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] -= wm * h[kk:]
    return g

def build_grad_matrix(h, n):
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=float)
    shift = n - 1
    for m in range(L):
        k = m - shift
        if k >= 0:
            nv = n - k
            G[m, :nv] += 1.0 - h[k:]
            G[m, k:] -= h[:nv]
        else:
            kk = -k
            nv = n - kk
            G[m, kk:] += 1.0 - h[:nv]
            G[m, :nv] -= h[kk:]
    return G

def lp_phase(h, n, deadline, step_max=0.25, max_iters=2000):
    h = project_feasible(h, n)
    best = compute_c5(h, n)
    step = step_max
    for it in range(max_iters):
        if time.time() > deadline - 1.0:
            break
        C = compute_c5_vec(h, n)
        smax = float(np.max(C))
        if it % 4 == 3:
            tol = 0.12 * max(1e-12, smax - float(np.min(C)))
            rows = np.where(C >= smax - tol)[0]
            if rows.size < 6:
                rows = np.arange(len(C))
        else:
            rows = np.arange(len(C))
        G = build_grad_matrix(h, n) * (2.0 / float(n))
        sub = G[rows]
        Ls = sub.shape[0]
        A = np.empty((Ls, n + 1), dtype=float)
        A[:, :n] = sub
        A[:, n] = -1.0
        b = -C[rows]
        Aeq = np.zeros((1, n + 1), dtype=float)
        Aeq[0, :n] = 1.0
        cobj = np.zeros(n + 1, dtype=float)
        cobj[n] = 1.0
        lo = -np.minimum(h, step)
        hi = np.minimum(1.0 - h, step)
        bnds = list(zip(lo, hi))
        bnds.append((0.0, np.inf))
        try:
            res = linprog(cobj, A_ub=A, b_ub=b, b_eq=[0.0], A_eq=Aeq,
                          bounds=bnds, options={"presolve": True})
        except Exception:
            break
        if not res.success:
            step *= 0.5
            if step < 1e-5:
                break
            continue
        delta = res.x[:-1]
        if not np.all(np.isfinite(delta)) or not np.any(np.abs(delta) > 1e-14):
            step *= 0.5
            if step < 1e-5:
                break
            continue
        improved = False
        for lam in (1.0, 0.9, 0.75, 0.6, 0.4, 0.25, 0.125, 0.0625):
            h2 = project_feasible(h + lam * delta, n)
            c2 = compute_c5(h2, n)
            if c2 < best - 1e-11:
                h = h2
                best = c2
                improved = True
                break
        if improved:
            step = min(step_max, step * 1.35)
        else:
            step *= 0.5
            if step < 1e-5:
                break
    return h, best

def pair_polish(h, n, deadline, seed=1):

                md = min(1.0 - h[i], h[j])

def de_objective(x):
    n = _STATE["n"]
    x = np.clip(x, 0.0, 1.0)
    c = float(np.max(compute_c5_vec(x, n)))
    s = float(x.sum()) - 0.5 * n
    return c + 50.0 * s * s

def de_phase(h, n, deadline, seed):
    h = project_feasible(h, n)
    _STATE["n"] = n
    bounds = [(0.0, 1.0)] * n
    try:
        res = differential_evolution(
            de_objective,
            bounds,
            x0=h,
            popsize=16,
            maxiter=35,
            seed=seed,
            tol=1e-5,
            workers=1,
            polish=False
        )
        h2 = project_feasible(res.x, n)
    except Exception:
        return h, compute_c5(h, n)
    t2 = min(deadline - 8.0, time.time() + 22.0)
    h2, c2 = lp_phase(h2, n, t2, step_max=0.12, max_iters=80)
    if time.time() < deadline - 5.0:
        h2, c2 = pair_polish(h2, n, deadline - 4.0, seed=seed + 1)
    return h2, c2

def random_perturb(h, n, scale, rng):
    x = h.copy()
    m = min(80, n // 2)
    i1 = rng.integers(0, n, m)
    j1 = rng.integers(0, n, m)
    for i, j in zip(i1, j1):
        if i != j:
            d = scale * rng.uniform(0.1, 1.0)
            x[i] += d
            x[j] -= d
    return project_feasible(x, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    try:
        h_best = np.asarray(initial_h_values, dtype=float).copy()
        n_best = len(h_best)
    except Exception:
        n_best = 240
        h_best = np.full(n_best, 0.5, dtype=float)

    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)

    rem = deadline - time.time()
    if rem > 520.0:
        h2, c2 = lp_phase(h_best, n_best, time.time() + min(360.0, rem - 520.0),
                          step_max=0.25, max_iters=3000)
        if c2 < best_c:
            h_best, best_c = h2, c2

    rem = deadline - time.time()
    if rem > 100.0:
        h2, c2 = pair_polish(h_best, n_best, time.time() + min(55.0, rem - 100.0),
                             seed=seed + 1)
        if c2 < best_c:
            h_best, best_c = h2, c2

    rem = deadline - time.time()
    while rem > 170.0:
        progressed = False
        for nt in (80, 130, 180):
            rem = deadline - time.time()
            if rem < 170.0:
                break
            if nt < 60 or nt > 990:
                continue
            xo = np.linspace(0.0, 2.0, n_best)
            xn = np.linspace(0.0, 2.0, nt)
            x0 = project_feasible(np.interp(xn, xo, h_best), nt)
            h2, c2 = de_phase(x0, nt, time.time() + min(75.0, rem - 95.0), seed=seed + nt)
            if c2 < best_c:
                h_best, best_c, n_best = h2, c2, nt
                progressed = True
        if not progressed:
            break

    rem = deadline - time.time()
    if rem > 140.0:
        for sc in (0.018, 0.06, 0.13):
            rem = deadline - time.time()
            if rem < 90.0:
                break
            alloc = min(45.0, rem - 75.0)
            x = random_perturb(h_best, n_best, sc, rng)
            h2, c2 = lp_phase(x, n_best, time.time() + alloc,
                              step_max=0.12, max_iters=400)
            if time.time() < deadline - 25.0:
                h2, c2 = pair_polish(h2, n_best, deadline - 20.0, seed=seed + 17)
            if c2 < best_c:
                h_best, best_c = h2, c2

    for fac in (0.7, 1.45, 2.0):
        rem = deadline - time.time()
        if rem < 90.0:
            break
        nt = int(round(n_best * fac))
        if nt % 2:
            nt += 1
        if nt < 70 or nt > 1000 or nt == n_best:
            continue
        xo = np.linspace(0.0, 2.0, n_best)
        xn = np.linspace(0.0, 2.0, nt)
        x0 = project_feasible(np.interp(xn, xo, h_best), nt)
        h2, c2 = lp_phase(x0, nt, time.time() + min(38.0, rem - 70.0),
                          step_max=0.15, max_iters=80)
        if time.time() < deadline - 20.0:
            h2, c2 = pair_polish(h2, nt, deadline - 15.0, seed=seed + nt)
        if c2 < best_c:
            h_best, best_c, n_best = h2, c2, nt

    if time.time() < deadline - 18.0:
        h2, c2 = lp_phase(h_best, n_best, deadline - 14.0,
                          step_max=0.07, max_iters=250)
        if c2 < best_c:
            h_best, best_c = h2, c2

    if time.time() < deadline - 3.0:
        h2, c2 = pair_polish(h_best, n_best, deadline - 2.0, seed=seed + 91)
        if c2 < best_c:
            h_best, best_c = h2, c2

    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)
    return h_best, best_c, n_best
```
