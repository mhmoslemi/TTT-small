import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def corr_array(h, n):
    return np.correlate(h, 1 - h, mode='full')

def grad_smooth_max(h, n, alpha, active_mask=None):
    """Compute gradient of smooth max of correlations"""
    c = corr_array(h, n)
    if active_mask is not None:
        c_masked = np.where(active_mask, c, -np.inf)
    else:
        c_masked = c
    shifted = c_masked - c_masked.max()
    weights = np.exp(alpha * shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    weights /= wsum
    # Compute weighted sum of gradients
    m_range = np.arange(-(n-1), n)
    g = np.zeros(n)
    for idx in range(len(weights)):
        if weights[idx] < 1e-15:
            continue
        k = m_range[idx]
        # grad_k[j] = (1-h[j+k]) * 1_{0<=j+k<n} - h[j-k] * 1_{0<=j-k<n}
        # Vectorized:
        # Term 1: (1-h[j+k]) for j such that 0<=j+k<n
        #   j+k in [0,n-1] => j in [-k, n-1-k], intersect [0,n-1]
        #   If k>=0: j in [0, n-1-k]
        #   If k<0: j in [-k, n-1]
        # Term 2: -h[j-k] for j such that 0<=j-k<n
        #   j-k in [0,n-1] => j in [k, n-1+k], intersect [0,n-1]
        #   If k>=0: j in [k, n-1]
        #   If k<0: j in [0, n-1+k]
        
        if k >= 0:
            # Term 1: j in [0, n-1-k], value (1-h[j+k])
            g[0:n-k] += weights[idx] * (1 - h[k:n])
            # Term 2: j in [k, n-1], value -h[j-k]
            g[k:n] += weights[idx] * (-h[0:n-k])
        else:
            # k < 0, let kk = -k
            kk = -k
            # Term 1: j in [kk, n-1], value (1-h[j+k]) = (1-h[j-kk])
            g[kk:n] += weights[idx] * (1 - h[0:n-kk])
            # Term 2: j in [0, n-1-kk], value -h[j-k] = -h[j+kk]
            g[0:n-kk] += weights[idx] * (-h[kk:n])
    return g

def project_hypersimplex(x, target):
    """Project x onto {0 <= h <= 1, sum(h) = target} using bisection"""
    n = len(x)
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(80):
        tau = (lo + hi) / 2
        s = np.clip(x - tau, 0, 1).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2
    return np.clip(x - tau, 0, 1)

def nesterov_smooth_max(h, n, alpha, max_iter, deadline, step_init=0.01):
    """Nesterov accelerated projected gradient on smooth max of correlations"""
    h = h.copy()
    h_prev = h.copy()
    m = len(h) // 2  # momentum
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    c_arr = corr_array(h, n)
    max_c = c_arr.max()
    # Active mask: top 20% of shifts
    threshold = np.percentile(c_arr, 80)
    active_mask = c_arr >= threshold
    
    step = step_init
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Nesterov extrapolation
        mu = (1 + np.sqrt(1 + 4 * m * m)) / 2  # or use (m-1)/(m+2)
        y = h + (m - 1) / (m + 2) * (h - h_prev)
        
        # Compute gradient at y
        g = grad_smooth_max(y, n, alpha, active_mask)
        
        # Step
        z = y - step * g
        
        # Project
        h_new = project_hypersimplex(z, target)
        
        # Evaluate
        c_new = compute_c5(h_new, n)
        
        # Adaptive step size
        if c_new > compute_c5(h, n) + 1e-12:
            step *= 0.5
        else:
            step = min(step * 1.1, 0.1)
        
        h_prev = h
        h = h_new
        m += 1
        
        # Update active mask periodically
        if it % 50 == 0:
            c_arr = corr_array(h, n)
            threshold = np.percentile(c_arr, 80)
            active_mask = c_arr >= threshold
        
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_hypersimplex(np.clip(res.x, 0, 1), n/2.0)
    except Exception:
        return h

def coord_descent(h, n, max_sweeps=500, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(45):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))

    dx = 2.0 / n
    c_arr = np.correlate(h, 1-h, mode='full')

        c = np.correlate(x, 1-x, mode='full')
        return np.argsort(c)[::-1][:top_m]

    top_idx = np.argsort(c_arr)[::-1][:top_m]
    
    def obj(x):
        c = np.correlate(x, 1-x, mode='full')[top_idx] * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n/2.0
    
    bounds = [(0.0, 1.0)] * n

    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return np.clip(res.x, 0, 1)
    except Exception:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_hypersimplex(x, target):
    n = len(x)
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(80):
        tau = (lo + hi) / 2.0
        s = np.clip(x - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth_max(h, n, alpha, active_mask):
    c = np.correlate(h, 1 - h, mode='full')
    c_work = np.where(active_mask, c, -np.inf)
    cm = c_work.max()
    weights = np.exp(alpha * (c_work - cm))
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    weights /= wsum
    m_range = np.arange(-(n - 1), n)
    g = np.zeros(n)
    aid = np.where(active_mask)[0]
    for ii in aid:
        w = weights[ii]
        if w < 1e-12:
            continue
        k = m_range[ii]
        if k >= 0:
            g[0:n-k] += w * (1.0 - h[k:n])
            g[k:n] -= w * h[0:n-k]
        else:
            kk = -k
            g[kk:n] += w * (1.0 - h[0:n-kk])
            g[0:n-kk] -= w * h[kk:n]
    return g

def nesterov_smooth(h, n, alpha, deadline, max_iter=20000, pct=75):
    target = n / 2.0
    h = h.copy()
    h_prev = h.copy()
    c_arr = np.correlate(h, 1 - h, mode='full')
    thr = np.percentile(c_arr, pct)
    active = c_arr >= thr
    step = 0.005
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        mu = it / (it + 3.0)
        y = h + mu * (h - h_prev)
        g = grad_smooth_max(y, n, alpha, active)
        # Backtracking line search
        f_cur = compute_c5(h, n)
        s = step
        for _b in range(25):
            z = y - s * g
            h_new = project_hypersimplex(z, target)
            f_new = compute_c5(h_new, n)
            if f_new <= f_cur + 1e-15:
                break
            s *= 0.5
        else:
            h_new = h
            f_new = f_cur
        h_prev = h
        h = h_new
        if f_new < best_c:
            best_c = f_new
            best_h = h.copy()
        if it % 100 == 0:
            c_arr = np.correlate(h, 1 - h, mode='full')
            thr = np.percentile(c_arr, pct)
            active = c_arr >= thr
        step = s
        step = min(step * 1.02, 0.05)
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_hypersimplex(np.clip(res.x, 0, 1), n / 2.0)
    except Exception:
        return h

def coord_descent(h, n, max_sweeps=500, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            m1 = (k1 >= 0) & (k1 < n)
            B[m1] += (1.0 - h[k1[m1]])
            k2 = i + m_range
            m2 = (k2 >= 0) & (k2 < n)
            B[m2] -= h[k2[m2]]
            v0 = np.max(corr - B * h[i])
            v1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(v0, v1) + 1e-9
            lo_v = min(v0, v1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zmax = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(45):
                mid = (lo_v + hi_v) / 2.0
                if zmax > mid + 1e-14:
                    lo_v = mid
                    continue
                lb = 0.0
                if neg_B.any():
                    lb = max(lb, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub = 1.0
                if pos_B.any():
                    ub = min(ub, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb <= ub + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb = 0.0
            if neg_B.any():
                lb = max(lb, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub = 1.0
            if pos_B.any():
                ub = min(ub, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb + ub) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr
