import numpy as np
import time
from scipy.optimize import minimize

def corr_full(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))

def cval_full(h, n):
    return float(np.max(corr_full(h, n)))

def project_target(x, target):
    y = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    for _ in range(100):
        d = target - float(np.sum(y))
        if abs(d) < 1e-14:
            break
        if d > 0.0:
            r = 1.0 - y
            s = float(np.sum(r))
            if s < 1e-15:
                break
            y += (d / s) * r
        else:
            s = float(np.sum(y))
            if s < 1e-15:
                break
            y += (d / s) * y
    y = np.clip(y, 0.0, 1.0)
    d = target - float(np.sum(y))
    if d > 0.0:
        i = int(np.argmax(1.0 - y))
        y[i] = min(1.0, y[i] + d)
    elif d < 0.0:
        i = int(np.argmax(y))
        y[i] = max(0.0, y[i] + d)
    return y

def project_left(y):
    return project_target(y, 0.5 * float(y.size))

def project_full(h):
    return project_target(h, 0.5 * float(h.size))

def ext_left(y, n):
    h = np.empty(n, dtype=float)
    m = n // 2
    h[:m] = y
    h[m:] = y[::-1]
    return h

def val_left(y, n):
    return cval_full(ext_left(y, n), n)

def smooth_max_c(c, alpha):
    m = float(np.max(c))
    if alpha <= 1e-12 or not np.isfinite(m):
        return m
    z = alpha * (c - m)
    zm = float(np.max(z))
    e = np.exp(z - zm)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return m
    return float(m + (zm + np.log(s)) / alpha)

def smooth_weights_c(c, alpha):
    c = np.asarray(c, dtype=float)
    w = np.zeros(c.size, dtype=float)
    m = int(np.argmax(c))
    w[m] = 1.0
    if alpha <= 1e-12:
        return w
    mx = float(c.max())
    z = alpha * (c - mx)
    zm = float(np.max(z))
    e = np.exp(z - zm)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return w
    return e / s

def grad_smooth_full(h, n, w):
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    for k in range(2 * n - 1):
        wk = float(w[k])
        if wk <= 1e-14:
            continue
        lag = k - (n - 1)
        wm = wk * dx
        if lag >= 0:
            nv = n - lag
            if nv > 0:
                g[0:nv] += wm * (1.0 - h[lag:n])
                g[lag:n] += wm * (-h[0:nv])
        else:
            lag = -lag
            nv = n - lag
            if nv > 0:
                g[lag:n] += wm * (1.0 - h[0:nv])
                g[0:nv] += wm * (-h[lag:n])
    return g

def obj_left(y, n, alpha, mu):
    m = n // 2
    if y.size != m:
        y = project_left(np.resize(y, m))
    h = ext_left(y, n)
    c = corr_full(h, n)
    val = smooth_max_c(c, alpha)
    s = float(np.sum(y)) - 0.5 * float(m)
    return float(val + mu * s * s)

def grad_left(y, n, alpha, mu):
    m = n // 2
    if y.size != m:
        y = project_left(np.resize(y, m))
    h = ext_left(y, n)
    c = corr_full(h, n)
    w = smooth_weights_c(c, alpha)
    g = grad_smooth_full(h, n, w)
    gy = g[:m] + g[m:][::-1]
    s = float(np.sum(y)) - 0.5 * float(m)
    if mu != 0.0:
        gy += 2.0 * mu * s
    return gy

def cons_left(y, n):
    return float(np.sum(y) - 0.5 * float(y.size))

def jac_left(y, n):
    return np.ones(y.size, dtype=float)

def pair_search_left(y, n, deadline, radius=30, max_scans=2, deltas=None):
    if deltas is None:
        deltas = (0.004, 0.002, 0.001, 0.0005)
    y = project_left(y)
    best = val_left(y, n)
    m = y.size
    radius = max(4, int(radius))
    for d in deltas:
        for _ in range(max_scans):
            if time.time() > deadline:
                return y, best
            improved = False
            for i in range(m):
                if time.time() > deadline:
                    return y, best
                lo = max(0, i - radius)
                hi = min(m, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    ai = y[i]
                    aj = y[j]
                    if ai + d <= 1.0 and aj - d >= 0.0:
                        y[i] = ai + d
                        y[j] = aj - d
                        c = val_left(y, n)
                        if c < best - 1e-14:
                            best = c
                            improved = True
                        else:
                            y[i] = ai
                            y[j] = aj
                    ai = y[i]
                    aj = y[j]
                    if ai - d >= 0.0 and aj + d <= 1.0:
                        y[i] = ai - d
                        y[j] = aj + d
                        c = val_left(y, n)
                        if c < best - 1e-14:
                            best = c
                            improved = True
                        else:
                            y[i] = ai
                            y[j] = aj
            if not improved:
                break
    return project_left(y), best

                    y2 = project_left(r.x)

                                      radius=max(8, int(32 * f)),
                                      max_scans=2)
            if c2 < best_c:
                best_y = y2
                best_c = c2
    else:
        if time.time() < deadline - 3.0 and n <= 500:
            y2, c2 = pair_search_left(best_y, n, deadline - 2.0,
                                      radius=max(6, int(18 * f)),
                                      max_scans=1, deltas=(0.002, 0.001, 0.0005))
            if c2 < best_c:
                best_y = y2
                best_c = c2
    return ext_left(best_y, n), best_c, best_y

def make_cos_starts(n):
    m = n // 2
    dx = 2.0 / float(n)
    x = (np.arange(m, dtype=float) + 0.5) * dx
    out = []
    for a in (0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40):
        y = 0.5 + a * np.cos(np.pi * x)
        out.append(project_left(y))
    for a in (0.25, 0.30, 0.35):
        for b in (-0.10, -0.05, 0.05, 0.10):
            y = 0.5 + a * np.cos(np.pi * x) + b * np.cos(2.0 * np.pi * x)
            out.append(project_left(y))
    for a in (0.25, 0.30):
        for b in (-0.08, 0.08):
            y = 0.5 + a * np.cos(np.pi * x) + b * np.cos(3.0 * np.pi * x)
            out.append(project_left(y))
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s) - 6.0)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel()
    except Exception:
        h0 = np.full(300, 0.5)
    if h0.size < 2 or not np.all(np.isfinite(h0)):
        h0 = np.full(300, 0.5)

    if h0.size % 2 == 1 or h0.size < 100 or h0.size > 600:
        n0 = 300
    else:
        n0 = h0.size
    if h0.size != n0:
        xo = (np.arange(h0.size, dtype=float) + 0.5) * (2.0 / float(h0.size))
        xn = (np.arange(n0, dtype=float) + 0.5) * (2.0 / float(n0))
        h0 = np.interp(xn, xo, h0)
    h0 = project_full(h0, n0)

    best_h = h0.copy()
    best_n = n0
    best_c = cval_full(best_h, best_n)
    best_sym = False

    m0 = n0 // 2
    hs = 0.5 * (h0 + h0[::-1])
    ys = project_left(hs[:m0])
    cs = val_left(ys, n0)
    if cs < best_c - 1e-12:
        best_h = ext_left(ys, n0)
        best_c = cs
        best_sym = True

    end1 = min(deadline - 220.0, t0 + 430.0)
    if cs < best_c + 0.005 and end1 > time.time() + 25.0:
        full, c, ys = optimize_left(ys, n0, end1, quick=False)
        if c < best_c - 1e-12:
            best_h = full
            best_n = n0
            best_c = c
            best_sym = True

    end2 = min(deadline - 180.0, time.time() + 150.0)
    if end2 > time.time() + 30.0 and n0 <= 600:
        c1, y1 = 1e9, None
        c2, y2 = 1e9, None
        for yx in make_cos_starts(n0):
            cx = val_left(yx, n0)
            if cx < c1:
                c2, y2 = c1, y1
                c1, y1 = cx, yx
            elif cx < c2:
                c2, y2 = cx, yx
        for cval, ystart in ((c1, y1), (c2, y2)):
            if ystart is None:
                continue
            sub = min(end2, time.time() + 55.0)
            if sub > time.time() + 8.0:
                full, c, _ = optimize_left(ystart, n0, sub, quick=True)
                if c < best_c - 1e-12:
                    best_h = full
                    best_n = n0
                    best_c = c
                    best_sym = True

    if time.time() < deadline - 90.0:
        cands = []
        if best_n > 200:
            cands.append(200)
        if best_n > 150:
            cands.append(150)
        if best_n > 128:
            cands.append(128)
        for tn in cands:
            if tn < 80 or tn >= best_n:
                continue
            if time.time() > deadline - 70.0:
                break
            xo = (np.arange(best_n, dtype=float) + 0.5) * (2.0 / float(best_n))
            xn = (np.arange(tn, dtype=float) + 0.5) * (2.0 / float(tn))
            hi = np.interp(xn, xo, best_h)
            hsx = 0.5 * (hi + hi[::-1])
            yx = project_left(hsx[:tn // 2])
            c0 = val_left(yx, tn)
            if c0 < best_c + 0.003:
                sub = min(deadline - 45.0, time.time() + 90.0)
                if sub > time.time() + 20.0:
                    full, c, _ = optimize_left(yx, tn, sub, quick=True)
                    if c < best_c - 1e-12:
                        best_h = full
                        best_n = tn
                        best_c = c
                        best_sym = True

    if time.time() < deadline - 120.0:
        if best_n < 300:
            up = 300
        elif best_n < 600:
            up = best_n * 2
        else:
            up = best_n
        if up > best_n and up <= 900:
            xo = (np.arange(best_n, dtype=float) + 0.5) * (2.0 / float(best_n))
            xn = (np.arange(up, dtype=float) + 0.5) * (2.0 / float(up))
            hi = np.interp(xn, xo, best_h)
            hsx = 0.5 * (hi + hi[::-1])
            yx = project_left(hsx[:up // 2])
            c0 = val_left(yx, up)
            if c0 < best_c + 0.005:
                sub = min(deadline - 35.0, time.time() + 260.0)
                if sub > time.time() + 35.0:
                    full, c, _ = optimize_left(yx, up, sub, quick=False)
                    if c < best_c - 1e-12:
                        best_h = full
                        best_n = up
                        best_c = c
                        best_sym = True

    if best_sym and best_n % 2 == 0 and best_n <= 700 and time.time() < deadline - 5.0:
        yx = project_left(best_h[:best_n // 2])
        yx, c = pair_search_left(yx, best_n, deadline - 2.0,
                                 radius=16, max_scans=1,
                                 deltas=(0.0005, 0.0002))
        if c < best_c - 1e-12:
            best_h = ext_left(yx, best_n)
            best_c = c

    if best_sym and best_n % 2 == 0:
        hx = ext_left(project_left(best_h[:best_n // 2]), best_n)
    else:
        hx = project_full(best_h, best_n)
    cx = cval_full(hx, best_n)
    return hx, cx, best_n
