def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def coord_descent(h, n, max_sweeps=500, deadline=None, rng=None, shuffle=False):
    m_range = np.arange(-(n - 1), n)

        if deadline and time.time() > deadline:
            break
        if shuffle and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for i in order:
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
