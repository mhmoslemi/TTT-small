```python
def compute_subgrad(h, n, k_star_idx):
    """Compute gradient of c[k_star_idx] w.r.t. h"""
    lag = k_star_idx - (n - 1)
    grad = np.zeros(n)
    # First term: (1-h[p+lag]) for p where 0≤p+lag<n
    if lag >= 0:
        # p in [0, n-1-lag]
        p_range = range(0, n - lag)
        grad[:n-lag] += (1.0 - h[lag:n])
    else:
        # p in [-lag, n-1]
        p_start = -lag
        grad[p_start:] += (1.0 - h[:n-p_start])

    if lag >= 0:
        # p in [lag, n-1]
        grad[lag:] -= h[:n-lag]
    else:
        # p in [0, n+lag-1]
        p_end = n + lag
        grad[:p_end] -= h[-lag:]
    return grad
```

```python
def compute_gradient(h, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    
    # Compute weighted gradient
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)  # lag for each k
    
    for k in range(len(w)):
        if w[k] < 1e-10:
            continue
        lag = lags[k]
        # First term: w[k] * (1-h[p+lag]) for valid p
        if lag >= 0:
            grad[:n-lag] += w[k] * (1.0 - h[lag:])
        else:
            grad[-lag:] += w[k] * (1.0 - h[:n+lag])
        # Second term: -w[k] * h[p-lag] for valid p
        if lag >= 0:
            grad[lag:] -= w[k] * h[:n-lag]
        else:
            grad[:n+lag] -= w[k] * h[-lag:]
    
    # Penalty gradient
    grad += 2.0 * penalty * (h.sum() - n/2.0)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h.values, 0.0, 1.0) if hasattr(h, 'values') else np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_max_obj(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    sm = shifted.max()
    logsum = sm + np.log(np.exp(shifted - sm).sum())
    return c_max + logsum / alpha + penalty * (x.sum() - n / 2.0) ** 2

def smooth_max_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    g = c * dx
    g_max = g.max()
    shifted = alpha * (g - g_max)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w_sum = w.sum()
    w /= w_sum
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-12:
            continue
        wk = w[k]
        lag = int(lags[k])
        if lag >= 0:
            grad[:n - lag] += wk * (1.0 - x[lag:])
            grad[lag:] -= wk * x[:n - lag]
        else:
            grad[-lag:] += wk * (1.0 - x[:n + lag])
            grad[:n + lag] -= wk * x[-lag:]
    grad *= dx
    grad += 2.0 * penalty * (x.sum() - n / 2.0)
    return grad

def lbfgs_step(h_init, n, alpha, penalty, maxiter):
    def obj(x):
        return smooth_max_obj(x, n, alpha, penalty)
    def grad(x):
        return smooth_max_grad(x, n, alpha, penalty)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': maxiter * 5, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

            stall = 0
```
