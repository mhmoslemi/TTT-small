import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=1000, deadline=None, random_order=False):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    rng = np.random.RandomState(12345)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        order = rng.permutation(n) if random_order else range(n)
        for i in order:
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            # Binary search for optimal value
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            lo_v = min(val0, val1) - 1e-9
            hi_v = max(val0, val1) + 1e-9
            for _ in range(40):
                mid = (lo_v + hi_v) / 2.0
                # Check feasibility: exists x in [0,1] such that max(corr + (x-h[i])*B) <= mid
                pos_B = B > 1e-14
                neg_B = B < -1e-14
                zero_B = ~(pos_B | neg_B)
                if zero_B.any() and corr[zero_B].max() > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:

            # Find optimal x

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        log_sum = (shifted.max() + np.log(np.exp(shifted - shifted.max()).sum()))
        return m + log_sum / alpha
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        w = np.exp(shifted - sm)
        w /= w.sum()
        # gradient of logsumexp / alpha = sum_k w[k] * dc_k/dx
        # dc_k/dx[i] = (1-x[i+k]) - x[i-k] (with boundary handling)
        # For shift k (correlation index n-1+k):
        # dc/dx[i] = (1-x[i+k])*1[i+k valid] - x[i-k]*1[i-k valid]
        grad = np.zeros(n)
        for i in range(n):

            idx = n - 1 + valid_k

            # Term 2: -x[i-k] for valid k  
            valid_k2 = np.arange(i - (n-1), i + 1)  # k values where i-k in [0,n)
            idx2 = n - 1 + valid_k2
            grad[i] -= w[idx2] @ x[i - valid_k2]
        return grad * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)

import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=1000, deadline=None, seed=42):
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    # Precompute masks for each coordinate
    masks_pos = []  # masks_pos[i]: boolean mask for i+m_range valid
    masks_neg = []  # masks_neg[i]: boolean mask for i-m_range valid
    for i in range(n):
        k1 = i - m_range
        m1 = (k1 >= 0) & (k1 < n)
        masks_pos.append(m1)
        k2 = i + m_range
        m2 = (k2 >= 0) & (k2 < n)
        masks_neg.append(m2)

            break
        order = rng.permutation(n)
        for i in order:
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            m1 = masks_pos[i]
            B[m1] += (1.0 - h[k1[m1]])
            k2 = i + m_range
            m2 = masks_neg[i]
            B[m2] -= h[k2[m2]]
            
            hi = h[i]
            # Binary search for optimal maximum value
            v0 = np.max(corr - B * hi)
            v1 = np.max(corr + B * (1.0 - hi))
            lo_v = min(v0, v1) - 1e-10
            hi_v = max(v0, v1) + 1e-10

            delta = x_opt - h[i]

                h[i] = x_opt

def slsqp_smooth(h, n, alpha, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        w = np.exp(shifted - sm)
        w /= w.sum()
        # Gradient: grad[i] = dx * sum_k w[n-1+k] * [(1-x[i+k])*1[valid] - x[i-k]*1[valid]]
        grad = np.zeros(n)
        for i in range(n):
            # Term 1: sum over k where 0<=i+k<n: w[n-1+k] * (1-x[i+k])
            ks = np.arange(-i, n - i)
            idx = n - 1 + ks
            grad[i] += np.dot(w[idx], 1.0 - x[i + ks])
            # Term 2: -sum over k where 0<=i-k<n: w[n-1+k] * x[i-k]
            ks2 = np.arange(i - (n-1), i + 1)
            idx2 = n - 1 + ks2
            grad[i] -= np.dot(w[idx2], x[i - ks2])
        return grad * dx
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: ones.copy()}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgradient_polish(h, n, iters=500, deadline=None):
    dx = 2.0 / n
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    m_range = np.arange(-(n - 1), n)
    lr = 0.01
    for it in range(iters):
        if deadline and time.time() > deadline:
            break
        k_star = np.argmax(corr)
        k_shift = k_star - (n - 1)
        # Subgradient for coordinate i: (1-h[i+k_shift]) - h[i-k_shift] (with boundary)
        grad = np.zeros(n)
        for i in range(n):
            ipk = i + k_shift
            if 0 <= ipk < n:
                grad[i] += (1.0 - h[ipk])
            imk = i - k_shift
            if 0 <= imk < n:
                grad[i] -= h[imk]
        # Update
        h_old = h.copy()
        h = h - lr * grad * dx
        h = project_feasible(h, n)
        # Recompute correlation
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
        # Adaptive step size
        if corr.max() < np.max(np.correlate(h_old, 1 - h_old, mode='full')):
            lr = min(lr * 1.1, 0.1)
        else:
            lr = max(lr * 0.7, 0.001)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=500, deadline=None, seed=42, bs_iters=34):
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    prev_max = corr.max()
    stale = 0
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            msk1 = (k1 >= 0) & (k1 < n)
            B[msk1] += (1.0 - h[k1[msk1]])
            k2 = i + m_range
            msk2 = (k2 >= 0) & (k2 < n)
            B[msk2] -= h[k2[msk2]]
            hi_val = h[i]
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            v0 = np.max(corr - B * hi_val)
            v1 = np.max(corr + B * (1.0 - hi_val))
            lo_v = min(v0, v1) - 1e-10
            hi_v = max(v0, v1) + 1e-10
            for _ in range(bs_iters):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-13:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, hi_val + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, hi_val + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-11:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, hi_val + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, hi_val + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            delta = x_opt - h[i]
            if abs(delta) > 1e-14:
                corr += B * delta
                h[i] = x_opt
        cur_max = corr.max()
        if cur_max < prev_max - 1e-9:
            prev_max = cur_max
            stale = 0
        else:
            stale += 1
            if stale > 80:
                break
    return h, corr

def slsqp_smooth(h, n, alpha, maxiter=1500, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        w = np.exp(shifted - sm)
        w /= w.sum()
        grad = np.zeros(n)
        for i in range(n):
            ks = np.arange(-i, n - i)
            idx = n - 1 + ks
            grad[i] += np.dot(w[idx], 1.0 - x[i + ks])
            ks2 = np.arange(i - (n - 1), i + 1)
            idx2 = n - 1 + ks2
            grad[i] -= np.dot(w[idx2], x[i - ks2])
        return grad * dx
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: ones.copy()}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgradient_polish(h, n, iters=300, deadline=None):
    dx = 2.0 / n
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    m_range = np.arange(-(n - 1), n)
    lr = 0.005
    best_c = corr.max()
    best_h = h.copy()
    for it in range(iters):
        if deadline and time.time() > deadline:
            break
        k_star = int(np.argmax(corr))
        k_shift = k_star - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            ipk = i + k_shift
            if 0 <= ipk < n:
                grad[i] += (1.0 - h[ipk])
            imk = i - k_shift
            if 0 <= imk < n:
                grad[i] -= h[imk]
        h_new = h - lr * grad * dx
        h_new = project_feasible(h_new, n)
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        if corr_new.max() < best_c:
            best_c = corr_new.max()
            best_h = h_new.copy()
            h = h_new
            corr = corr_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr = max(lr * 0.8, 0.0005)
    return best_h
