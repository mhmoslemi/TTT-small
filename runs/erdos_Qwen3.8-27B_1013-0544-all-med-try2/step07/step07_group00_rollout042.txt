    h = np.clip(h.copy(), 0.0, 1.0)

            if tot < 1e-14:

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    s = x.sum() - n / 2.0

    m = c.max()

    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()

    best_c = compute_c5(h, n)
    radius = 80

            break

            hi = min(n, i + radius + 1)

def local_search(h, n, deadline, radius=40, n_trials=11):
    best = compute_c5(h, n)
    rng = np.random.RandomState(7)
    while time.time() < deadline:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                return h, best
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if i == j:
                    continue
                # Screen forward direction
                md1 = min(1.0 - h[i], h[j])
                if md1 >= 1e-8:
                    dt = md1 * 0.3
                    h[i] += dt; h[j] -= dt
                    ct = compute_c5(h, n)
                    h[i] -= dt; h[j] += dt
                    if ct < best:
                        d, cn = pair_ternary(h, n, i, j, 1, n_trials)
                        if d > 1e-9 and cn < best - 1e-15:
                            if 1:
                                h[i] += d; h[j] -= d
                            best = cn
                            improved = True
                    else:
                        # Screen reverse direction
                        md2 = min(h[i], 1.0 - h[j])
                        if md2 >= 1e-8:
                            dt2 = md2 * 0.3
                            h[i] -= dt2; h[j] += dt2
                            ct2 = compute_c5(h, n)
                            h[i] += dt2; h[j] -= dt2
                            if ct2 < best:
                                d, cn = pair_ternary(h, n, i, j, -1, n_trials)
                                if d > 1e-9 and cn < best - 1e-15:
                                    h[i] -= d; h[j] += d
                                    best = cn
                                    improved = True
        if not improved:
            break
    return h, best

def spectral_smooth_grad(h, n):
    """Gradient toward flattening autocorrelation (reduces variance of R_k)."""
    c = compute_c5_all(h, n)
    m = c.mean()
    # Weight lags that are above the mean more heavily
    weights = np.maximum(c - m, 0)
    weights = weights ** 2
    if weights.sum() < 1e-14:
        return np.zeros(n)
    # Compute gradient of sum(weight_k * c_k)
    dx = 2.0 / n
    g = np.zeros(n)
    for mi in range(2 * n - 1):
        if weights[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        wm = weights[mi] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])
    return g

def projected_subgrad(h, n, deadline, n_steps=50000, lr0=0.005):
    """Subgradient step toward flattening the overlap profile."""
    best = compute_c5(h, n)
    for t in range(n_steps):
        if time.time() > deadline:
            break
        g = spectral_smooth_grad(h, n)
        s = h.sum() - n / 2.0
        g += 0.1 * s
        lr = lr0 / (1.0 + 0.01 * t)
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best:
            best = c_new
            h = h_new
        else:
            lr0 *= 0.95
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = project_feasible(np.full(n, 0.5), n)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS with sweeping alpha
    t1 = min(t0 + 180, deadline - 350)
    h, c = lbfgs_optimize(h, n, t1, alphas=[200, 2000, 20000, 200000, 2000000])
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 2: Subgradient smoothing (flatten the profile)
    t2 = min(t1 + 60, deadline - 250)
    if time.time() < t2:
        h2, c2 = projected_subgrad(h, n, t2, n_steps=20000, lr0=0.003)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 3: L-BFGS again after smoothing
    t3 = min(t2 + 120, deadline - 200)
    h, c = lbfgs_optimize(h, n, t3, alphas=[50000, 500000, 5000000])
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 4: Intensive local search with ternary search
    t4 = min(t3 + 350, deadline - 80)
    h, c = local_search(h, n, t4, radius=40, n_trials=11)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 5: Final L-BFGS polish with very high alpha
    t5 = min(t4 + 80, deadline - 30)
    if time.time() < t5:
        h, c = lbfgs_optimize(h, n, t5, alphas=[10000000, 50000000], maxiter=200)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 6: Final local search with larger radius
    t6 = deadline - 1
    if time.time() < t6:
        h, c = local_search(h, n, t6, radius=70, n_trials=12)
        if c < best_c:
            best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
