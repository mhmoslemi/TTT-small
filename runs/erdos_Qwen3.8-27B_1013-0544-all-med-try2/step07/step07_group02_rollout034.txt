Let me try a different angle: **use a penalty-based approach with cvxpy**.

Actually, the problem min_h max_k g_k(h) can be formulated as:
min t
s.t. g_k(h) ≤ t for all k
     0 ≤ h ≤ 1
     sum(h) = n/2

Each constraint g_k(h) ≤ t is a quadratic constraint. This is not a standard SOCP or SDP, but it is a QCQP. cvxpy might not handle this directly since the quadratic forms are not necessarily convex.

Let me check: g_k(h) = Σ_i h[i] * (1 - h[i+k]) * dx = Σ_i h[i]*dx - Σ_i h[i]*h[i+k]*dx

The second term Σ_i h[i]*h[i+k]*dx is a quadratic form. Its matrix has entries 0.5*dx at positions (i, i+k) and (i+k, i). This matrix is NOT positive semi-definite in general (it's a circulant-like matrix with -1 on the k-th diagonal).

So this is a non-convex QCQP. cvxpy won't help directly.

Let me think about another approach: **alternating direction method of multipliers (ADMM)**.

We can write: min_h max_k g_k(h) as:
min_{h, t} t s.t. g_k(h) ≤ t

Using Lagrangian relaxation: L(h, t, λ) = t + Σ_k λ_k (g_k(h) - t)

For fixed λ, minimize over h, t:
- w.r.t. t: 1 - Σ_k λ_k = 0, so we need Σ_k λ_k = 1 (λ is on the simplex)
- w.r.t. h: minimize Σ_k λ_k g_k(h) subject to 0 ≤ h ≤ 1, sum(h) = n/2

The inner problem is: minimize a quadratic function subject to box and sum constraints. This is a QP!

The quadratic function is: Σ_k λ_k [Σ_i h[i] - Σ_i h[i]h[i+k]] * dx
= (Σ_k λ_k) Σ_i h[i] * dx - Σ_k λ_k Σ_i h[i]h[i+k] * dx
= Σ_i h[i] * dx - Σ_{i,k} λ_k h[i]h[i+k] * dx (using Σ_k λ_k = 1)

The Hessian is: -dx * (matrix with entries from Σ_k λ_k h[i]h[i+k])

This is a quadratic form with a specific band structure. For a given λ, this is a QP that scipy can solve.

But the overall approach (dual ascent on λ) might converge slowly.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-18):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def power_mean_optimize(h, n, p, maxiter=3000, ftol=1e-18):
    """Use power mean as smooth max approximation."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        gm = g - g.max()
        pmean = np.exp(np.logaddexp.reduce(gm) * 0)  # not quite right
        # Actually: (sum g_i^p)^(1/p) = exp((1/p) * log(sum exp(p * g_i)))
        # = exp((1/p) * logsumexp(p * g))
        # = max(g) + (1/p) * logsumexp(p*(g - max(g)))
        m = g.max()
        pg = p * (g - m)
        ll = np.log(np.exp(pg).sum())
        return m + ll / p
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, n, max_sweeps=200, deadline=None, bisect_iters=35):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            zero_B = (np.abs(B) < 1e-14)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(bisect_iters):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                pos_B = B > 1e-14
                neg_B = B < -1e-14
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_local_search(h_b, n, deadline, max_sweeps=200):
    """Local search on binary solutions: try flipping pairs."""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = h_b.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            # Try the best j
            best_j = -1
            best_nc = cur_c
            for j in ones:
                delta = np.zeros(2 * n - 1)
                # h[i] goes 0->1, h[j] goes 1->0
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones.remove(best_j)
                zeros.remove(i)
                ones.append(i)
                zeros.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-18):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, n, max_sweeps=200, deadline=None, bisect_iters=35):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            zero_B = (np.abs(B) < 1e-14)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(bisect_iters):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                pos_B = B > 1e-14
                neg_B = B < -1e-14
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_local_search(h_b, n, deadline, max_sweeps=100):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = h_b.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            best_delta = None
            for j in ones:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                if j - i < n:
                    delta[n - 1 + (j - i)] += 1.0
                if i - j > -(n):
                    delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
                    best_delta = delta
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                corr += best_delta
                cur_c = best_nc
                ones.remove(best_j)
                zeros.remove(i)
                ones.append(i)
                zeros.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c
