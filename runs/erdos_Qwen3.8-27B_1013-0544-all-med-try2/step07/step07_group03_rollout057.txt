import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * (corr - corr.max())
    p = np.exp(shifted)
    p /= p.sum()
    p_rev = p[::-1].copy()
    c1 = np.correlate(p, h, mode='full')
    c2 = np.correlate(p_rev, h, mode='full')
    idx = 2*(n-1) - np.arange(n)
    grad = (1.0 - c1[idx] - c2[idx]) * dx
    return grad

def pgd_optimize(h, n, alpha, n_iter, lr, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        grad = smooth_gradient(h, n, alpha)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        if not np.all(np.isfinite(h_new)):
            break
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c
