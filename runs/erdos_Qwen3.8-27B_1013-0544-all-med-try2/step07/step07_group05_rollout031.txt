```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_c(h, n):
    """Get the full correlation values times dx."""
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def smooth_obj(h, n, alpha):
    c = get_c(h, n)
    m = c.max()
    s = alpha * (c - m)
    ms = s.max()
    if ms > 500:
        return float(m + (ms + np.log(np.sum(np.exp(s - ms)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def smooth_grad(h, n, alpha):
    c = get_c(h, n)
    m = c.max()
    s = alpha * (c - m)
    ms = s.max()
    if ms > 500:
        w = np.exp(s - ms)
    else:
        w = np.exp(s)
    w /= w.sum()
    
    dx = 2.0 / n
    n_lags = 2 * n - 1
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    
    for m_idx in range(n_lags):
        s_val = m_idx - n + 1
        w_val = w[m_idx]
        if w_val < 1e-14:
            continue
        lo1 = max(0, -s_val)
        hi1 = min(n, n - s_val)
        if lo1 < hi1:
            term1[lo1:hi1] += w_val * (1.0 - h[lo1+s_val:hi1+s_val])
        lo2 = max(0, s_val)
        hi2 = min(n, n + s_val)
        if lo2 < hi2:
            term2[lo2:hi2] += w_val * h[lo2-s_val:hi2-s_val]
    
    return (term1 - term2) * dx

    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_obj(x, n, alpha)
    
    def grad(x):
        x = np.clip(x, 0, 1)
        return smooth_grad(x, n, alpha)
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
    return project_feasible(res.x, n)

def coordinate_descent(h, n, deadline, max_sweeps=100):
    """Cyclic coordinate descent with 1D optimization."""
    best_c = compute_c5(h, n)
    h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            # Compute gradient contributions for coordinate j
            c = get_c(h, n)  # (2n-1,)
            dx = 2.0 / n
            g = np.zeros(2 * n - 1)
            for m_idx in range(2 * n - 1):
                s_val = m_idx - n + 1
                # dC[m]/dh[j] = (1 - h[j+s]) - h[j-s]
                val = 0.0
                idx1 = j + s_val
                if 0 <= idx1 < n:
                    val += (1.0 - h[idx1])
                idx2 = j - s_val
                if 0 <= idx2 < n:
                    val -= h[idx2]
                g[m_idx] = val * dx
            
            # Now find delta to minimize max(c + delta * g)
            # subject to 0 <= h[j] + delta <= 1
            lo = -h[j]
            hi = 1.0 - h[j]
            
            # Ternary search on [lo, hi]
            a, b = lo, hi
            for _ in range(30):
                m1 = a + (b - a) / 3
                m2 = b - (b - a) / 3
                f1 = (c + m1 * g).max()
                f2 = (c + m2 * g).max()
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            delta = (a + b) / 2
            h[j] = np.clip(h[j] + delta, 0, 1)
        
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            improved = True
        else:
            if not improved:
                break
    
    return project_feasible(h, n), best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def get_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    c = get_c(h, n)
    m = c.max()
    s = alpha * (c - m)
    ms = s.max()
    if ms > 500:
        return float(m + (ms + np.log(np.sum(np.exp(s - ms)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def smooth_grad(h, n, alpha):
    c = get_c(h, n)
    m = c.max()
    s = alpha * (c - m)
    ms = s.max()
    if ms > 500:
        w = np.exp(s - ms)
    else:
        w = np.exp(s)
    w /= w.sum()
    
    dx = 2.0 / n
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    
    for m_idx in range(2 * n - 1):
        s_val = m_idx - n + 1
        w_val = w[m_idx]
        if w_val < 1e-14:
            continue
        lo1 = max(0, -s_val)
        hi1 = min(n, n - s_val)
        if lo1 < hi1:
            term1[lo1:hi1] += w_val * (1.0 - h[lo1+s_val:hi1+s_val])
        lo2 = max(0, s_val)
        hi2 = min(n, n + s_val)
        if lo2 < hi2:
            term2[lo2:hi2] += w_val * h[lo2-s_val:hi2-s_val]
    
    return (term1 - term2) * dx

def linear_oracle(g, n):
    target = int(n / 2)
    idx = np.argsort(g)
    s = np.zeros(n)
    s[idx[:target]] = 1.0
    return s

def frank_wolfe(h, n, alpha, deadline):
    h = h.copy()
    for it in range(10000):
        if time.time() > deadline:
            break
        g = smooth_grad(h, n, alpha)
        s = linear_oracle(g, n)
        # Step size: backtracking
        f_cur = smooth_obj(h, n, alpha)
        t = 1.0
        best_t = 2.0 / (it + 2)  # fallback
        best_f = smooth_obj(h + best_t * (s - h), n, alpha)
        for _ in range(20):
            h_try = h + t * (s - h)
            if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                t *= 0.5
                continue
            f_try = smooth_obj(h_try, n, alpha)
            if f_try <= f_cur - 1e-4 * t * np.dot(g, s - h):
                best_t = t
                best_f = f_try
                break
            t *= 0.5
            if t < 1e-8:
                break
        h = h + best_t * (s - h)
        h = project_feasible(h, n)
    return h

def lbfgs_optimize(h, n, alpha, maxiter, deadline):
    def obj(x):
        return smooth_obj(np.clip(x, 0, 1), n, alpha)
    def grad(x):
        return smooth_grad(np.clip(x, 0, 1), n, alpha)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def get_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    c = get_c(h, n)
    m = c.max()
    s = alpha * (c - m)
    ms = s.max()
    if ms > 500:
        return float(m + (ms + np.log(np.sum(np.exp(s - ms)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def smooth_grad(h, n, alpha):
    c = get_c(h, n)
    m = c.max()
    s = alpha * (c - m)
    ms = s.max()
    if ms > 500:
        w = np.exp(s - ms)
    else:
        w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    for m_idx in range(2 * n - 1):
        sv = m_idx - n + 1
        wv = w[m_idx]
        if wv < 1e-14:
            continue
        lo1 = max(0, -sv)
        hi1 = min(n, n - sv)
        if lo1 < hi1:
            term1[lo1:hi1] += wv * (1.0 - h[lo1 + sv:hi1 + sv])
        lo2 = max(0, sv)
        hi2 = min(n, n + sv)
        if lo2 < hi2:
            term2[lo2:hi2] += wv * h[lo2 - sv:hi2 - sv]
    return (term1 - term2) * dx

def lbfgs_optimize(h, n, alpha, maxiter, deadline):
    def obj(x):
        return smooth_obj(np.clip(x, 0, 1), n, alpha)
    def grad(x):
        return smooth_grad(np.clip(x, 0, 1), n, alpha)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12,
                                               'factr': 1e10})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def frank_wolfe(h, n, alpha, max_iter, deadline):
    h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = smooth_grad(h, n, alpha)
        target = int(n // 2)
        idx = np.argsort(g)
        s = np.zeros(n)
        s[idx[:target]] = 1.0
        f_cur = smooth_obj(h, n, alpha)
        direction = s - h
        t = 1.0
        best_t = 2.0 / (it + 2)
        for _ in range(25):
            h_try = h + t * direction
            if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                t *= 0.5
                continue
            f_try = smooth_obj(h_try, n, alpha)
            if f_try <= f_cur - 1e-4 * t * np.dot(g, direction):
                best_t = t
                break
            t *= 0.5
            if t < 1e-9:
                break
        h = h + best_t * direction
        h = project_feasible(h, n)

def subgrad_kick(h, n, n_steps=2000):
    h = h.copy()
    h_prev = h.copy()
    for t in range(1, n_steps + 1):
        beta = (t - 1) / (t + 2)

    return h

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n = 100
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)

    # Phase 1: L-BFGS-B with increasing alpha
    for alpha in [300, 1000, 5000, 20000, 100000]:
        if time.time() > deadline - 80:
            break
        alloc = min(100, (deadline - time.time()) * 0.25)
        if alloc < 15:
            break
        h_opt = lbfgs_optimize(best_h, n, alpha, maxiter=400, deadline=t0 + alloc)
        c = compute_c5(h_opt, n)
        if c < best_c:
            best_c = c
            best_h = h_opt.copy()

    # Phase 2: Frank-Wolfe exploration
    if time.time() < deadline - 60:
        fw_dl = time.time() + min(100, (deadline - time.time()) * 0.2)
        for alpha in [500, 2000, 10000]:
            if time.time() > fw_dl:
                break
            h_fw = frank_wolfe(best_h, n, alpha, max_iter=3000, deadline=fw_dl)
            c = compute_c5(h_fw, n)
            if c < best_c:
                best_c = c
                best_h = h_fw.copy()
    if time.time() < deadline - 50:
        for alpha in [200000, 1000000]:
            if time.time() > deadline - 40:
                break
            h_opt = lbfgs_optimize(best_h, n, alpha, maxiter=300, deadline=None)
            c = compute_c5(h_opt, n)
            if c < best_c:
                best_c = c
                best_h = h_opt.copy()

    # Phase 3: Subgradient kicks from multiple perturbations
    if time.time() < deadline - 50:
        remaining = deadline - time.time() - 30
        n_kicks = max(3, int(remaining / 40))
        for r in range(n_kicks):
            if time.time() > deadline - 30:
                break
            scale = 0.02 + 0.06 * (r / max(1, n_kicks - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            # Subgradient kick
            h_kick = subgrad_kick(hp, n, n_steps=1000)
            # Then L-BFGS-B refinement
            h_opt = lbfgs_optimize(h_kick, n, 50000, maxiter=300, deadline=None)
            c = compute_c5(h_opt, n)
            if c < best_c:
                best_c = c
                best_h = h_opt.copy()

    # Phase 4: Pair search / near-pair search
    if time.time() < deadline - 5:
        if n <= 160:
            best_h, best_c = pair_search(best_h, n, deadline - 2,
                                         [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 2,
                                              [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                                              radius=60)

    # Phase 5: Final L-BFGS-B polish + pair search
    if time.time() < deadline - 3:
        h_opt = lbfgs_optimize(best_h, n, 2000000, maxiter=500, deadline=None)
        c = compute_c5(h_opt, n)
        if c < best_c:
            best_c = c
            best_h = h_opt.copy()
        if time.time() < deadline - 1:
            if n <= 160:
                best_h, best_c = pair_search(best_h, n, deadline - 0.5,
                                             [0.0005, 0.0002, 0.0001, 0.00005])
            else:
                best_h, best_c = near_pair_search(best_h, n, deadline - 0.5,
                                                  [0.0005, 0.0002, 0.0001], radius=40)

    # Phase 6: Try a different n (interpolate)
    if time.time() < deadline - 30:
        for n_try in [200, 300]:
            if time.time() > deadline - 20:
                break
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            h_opt = lbfgs_optimize(h_interp, n_try, 10000, maxiter=200, deadline=None)
            h_opt = lbfgs_optimize(h_opt, n_try, 100000, maxiter=200, deadline=None)
            c = compute_c5(h_opt, n_try)
            if c < best_c:
                best_c = c
                best_h = h_opt.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
