```
def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-13: break
        if d > 0:
            mask = h < 1.0
            if not mask.any(): break
            room = 1.0 - h
            s = room[mask].sum()
            if s <= 0: break
            h[mask] += d * (room[mask] / s)
        else:
            mask = h > 0.0
            if not mask.any(): break
            mass = h[mask].sum()
            if mass <= 0: break
            h[mask] += d * (h[mask] / mass)
    h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment using least fractional
    d = target - h.sum()
    if abs(d) > 1e-12:
       if d > 0:
          order = np.argsort(1.0 - h)
       else:
          order = np.argsort(h)
       for idx in order:
          if d > 0:
             add = min(1.0 - h[idx], d)
             h[idx] += add; d -= add
             if d <= 1e-14: break
          else:
             sub = min(h[idx], -d)
             h[idx] -= sub; d += sub
             if d >= -1e-14: break
    return h
```

```python
import numpy as np
import math
import time
from scipy.optimize import minimize

_G_N = 0
_G_ALPHA = 1.0


def compute_c5(h, n):
    x = np.asarray(h, dtype=float)
    c = np.correlate(x, 1.0 - x, mode="full")
    return float(np.max(c) * (2.0 / n))


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h
            s = room[mask].sum()
            if s <= 0.0:
                break
            h[mask] += d * room[mask] / s
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            mass = h[mask].sum()
            if mass <= 0.0:
                break
            h[mask] += d * h[mask] / mass
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0.0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if d > 0.0:
                add = min(1.0 - h[idx], d)
                h[idx] += add
                d -= add
            else:
                sub = min(h[idx], -d)
                h[idx] -= sub
                d += sub
            if abs(d) < 1e-14:
                break
    return h


def downsample_h(parent, n, noise, rng):
    n0 = len(parent)
    if n0 == n:
        h = np.asarray(parent, dtype=float).copy()
    else:
        x0 = np.linspace(0.0, 2.0, n0)
        xn = np.linspace(0.0, 2.0, n)
        h = np.interp(xn, x0, np.asarray(parent, dtype=float))
    if noise > 0.0 and rng is not None:
        h = h + noise * rng.normal(0.0, 1.0, n)
    return project_feasible(h, n)


def threshold_binary(h, n):
    vals = np.asarray(h, dtype=float)
    order = np.argsort(vals, kind="stable")[::-1]
    out = np.zeros(n, dtype=np.int16)
    out[order[: n // 2]] = 1
    return out


def make_binary_start(hc, n, variant, rng):
    t = n // 2
    if variant == 0:
        vals = hc + 1e-9 * rng.rand(n)
        return threshold_binary(vals, n)
    if variant == 1:
        vals = hc + 0.10 * rng.normal(0.0, 1.0, n)
        return threshold_binary(vals, n)
    if variant == 2:
        alpha = 0.6180339887498949
        phase = (np.arange(n) * alpha) % 1.0
        vals = 0.45 * hc + 0.55 * (1.0 - phase)
        return threshold_binary(vals, n)
    if variant == 3:
        idx = rng.permutation(n)
        out = np.zeros(n, dtype=np.int16)
        out[idx[:t]] = 1
        return out
    alpha = math.sqrt(2.0) - 1.0
    phase = (np.arange(n) * alpha) % 1.0
    vals = 0.35 * hc + 0.35 * (1.0 - phase) + 0.30 * rng.normal(0.0, 1.0, n)
    return threshold_binary(vals, n)


def smooth_obj(x):
    n = _G_N
    c = np.correlate(x, 1.0 - x, mode="full")[::-1]
    m = float(np.max(c))
    z = _G_ALPHA * (c - m)
    z -= np.max(z)
    return m + float(np.log(np.exp(z).sum())) / _G_ALPHA


def smooth_cons(x):
    return float(np.sum(x) - 0.5 * _G_N)


def continuous_refine(h, n, deadline, alphas):
    global _G_N, _G_ALPHA
    h = project_feasible(np.asarray(h, dtype=float), n)
    best = h.copy()
    bestc = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": smooth_cons}]
    if n <= 160:
        maxiter = 1200
    elif n <= 240:
        maxiter = 600
    elif n <= 320:
        maxiter = 400
    else:
        maxiter = 250
    for alpha in alphas:
        if time.time() > deadline - 2.0:
            break
        _G_N = n
        _G_ALPHA = float(alpha)
        try:
            res = minimize(
                smooth_obj,
                h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": maxiter, "ftol": 1e-10},
            )
            newh = np.asarray(res.x, dtype=float)
        except Exception:
            break
        if not np.all(np.isfinite(newh)):
            break
        newh = project_feasible(newh, n)
        nc = compute_c5(newh, n)
        curc = compute_c5(h, n)
        if nc < curc:
            h = newh
        if nc < bestc:
            bestc = nc
            best = newh.copy()
    return best, bestc


def build_state(H):
    n = int(len(H))
    L = 2 * n - 1
    sh = n - 1
    H = np.asarray(H, dtype=np.int16).copy()
    om = (1 - H).astype(np.int16)
    c = np.correlate(H, om, mode="full")[::-1].astype(np.int32, copy=False)

    S = np.zeros((n, L), dtype=np.int32)
    Sflat = S.reshape(-1)
    rev = H[::-1]
    for p in range(n):
        base = p * L
        Sflat[base + sh - p : base + sh - p + n] += om
        Sflat[base + p : base + p + n] -= rev

    rows = np.arange(n, dtype=np.int64)
    offs = rows * L
    zeros = np.where(H == 0)[0].astype(np.int64)
    ones = np.where(H == 1)[0].astype(np.int64)
    pos_zero = np.full(n, -1, dtype=np.int64)
    pos_one = np.full(n, -1, dtype=np.int64)
    pos_zero[zeros] = np.arange(zeros.size, dtype=np.int64)
    pos_one[ones] = np.arange(ones.size, dtype=np.int64)

    half = max(1, max(zeros.size, ones.size))
    return {
        "n": n,
        "L": L,
        "sh": sh,
        "H": H,
        "c": c,
        "S": S,
        "Sflat": Sflat,
        "rows": rows,
        "offs": offs,
        "zeros": zeros,
        "ones": ones,
        "pos_zero": pos_zero,
        "pos_one": pos_one,
        "basebuf": np.empty(L, dtype=np.int32),
        "matbuf": np.empty((half, L), dtype=np.int32),
        "Sonebuf": np.empty((half, L), dtype=np.int32),
        "tmp": np.empty(L, dtype=np.int32),
        "idxm": np.arange(half, dtype=np.int64),
    }


def best_pair_score(st):
    c = st["c"]
    S = st["S"]
    sh = st["sh"]
    zeros = st["zeros"]
    ones = st["ones"]
    if zeros.size == 0 or ones.size == 0:
        return -1, -1, int(c.max())

    cur = int(c.max())
    m = int(ones.size)
    row_idx = st["idxm"][:m]
    Sone = st["Sonebuf"]
    mat = st["matbuf"]
    base = st["basebuf"]
    Sone[:m, :] = S[ones]

    best_score = cur + 1
    best_i = -1
    best_j = -1

    for z in zeros:
        z = int(z)
        np.add(c, S[z], out=base)
        np.subtract(base[None, :], Sone[:m, :], out=mat[:m, :])
        mat[:m, sh] -= 2
        mat[row_idx, (ones - z) + sh] += 1
        mat[row_idx, (z - ones) + sh] += 1
        score = mat[:m, :].max(axis=1)
        local = int(score.min())
        if local < best_score:
            pos = int(score.argmin())
            best_score = local
            best_i = z
            best_j = int(ones[pos])
    return best_i, best_j, best_score


def pair_score(st, i, j):
    i = int(i)
    j = int(j)
    tmp = st["tmp"]
    np.subtract(st["c"], st["S"][j], out=tmp)
    tmp += st["S"][i]
    sh = st["sh"]
    m = int(tmp.max())
    v1 = int(tmp[sh + j - i] + 1)
    v2 = int(tmp[sh + i - j] + 1)
    v0 = int(tmp[sh] - 2)
    return max(m, v1, v2, v0)


def apply_pair(st, i, j):
    i = int(i)
    j = int(j)
    sh = st["sh"]
    S = st["S"]
    Sflat = st["Sflat"]
    c = st["c"]

    c += S[i] - S[j]
    c[sh] -= 2
    c[sh + j - i] += 1
    c[sh + i - j] += 1

    rows = st["rows"]
    offs = st["offs"]

    idx = offs + (sh + i - rows)
    Sflat[idx] -= 1
    idx = offs + (sh + rows - i)
    Sflat[idx] -= 1

    idx = offs + (sh + j - rows)
    Sflat[idx] += 1
    idx = offs + (sh + rows - j)
    Sflat[idx] += 1

    H = st["H"]
    H[i] = 1
    H[j] = 0

    zl = st["zeros"]
    ol = st["ones"]
    pz = st["pos_zero"]
    po = st["pos_one"]
    zi = int(pz[i])
    oj = int(po[j])
    zl[zi] = j
    pz[j] = zi
    pz[i] = -1
    ol[oj] = i
    po[i] = oj
    po[j] = -1

    return int(c.max())


def do_descent(st, deadline, max_moves):
    moves = 0
    cur = int(st["c"].max())
    while time.time() < deadline and moves < max_moves:
        i, j, score = best_pair_score(st)
        if i < 0 or score >= cur:
            break
        cur = apply_pair(st, i, j)
        moves += 1
        if moves % 1000 == 0:
            st = build_state(st["H"])
            cur = int(st["c"].max())
    return st, moves, cur


def binary_search(n, H0, deadline, rng):
    target = n // 2
    H = np.asarray(H0, dtype=np.int16).copy()
    s = int(H.sum())
    if s != target:
        if s > target:
            cand = np.where(H == 1)[0]
            H[cand[: s - target]] = 0
        elif s < target:
            cand = np.where(H == 0)[0]
            H[cand[: target - s]] = 1

    st = build_state(H)
    cur = int(st["c"].max())
    best_H = H.copy()
    best_count = cur

    if n <= 200:
        init_cap = 5000
    elif n <= 320:
        init_cap = 1200
    else:
        init_cap = 400
    st, _, cur = do_descent(st, deadline, init_cap)
    if cur < best_count:
        best_count = cur
        best_H = st["H"].copy()

    last_imp = time.time()
    no_imp = 0

    while time.time() < deadline - 1.0:
        if cur > best_count + 10 or time.time() - last_imp > 45.0 or rng.rand() < 0.10:
            st = build_state(best_H)
            cur = best_count
            no_imp = 0
            for _ in range(50):
                if time.time() > deadline - 1.0:
                    break
                if st["zeros"].size == 0 or st["ones"].size == 0:
                    break
                i = int(st["zeros"][rng.randint(st["zeros"].size)])
                j = int(st["ones"][rng.randint(st["ones"].size)])
                score = pair_score(st, i, j)
                if score <= cur or (score - cur <= 3 and rng.random() < 0.25):
                    cur = apply_pair(st, i, j)
                    if cur < best_count:
                        best_count = cur
                        best_H = st["H"].copy()
                        last_imp = time.time()

        T = 0.4
        if cur > best_count:
            T = min(5.0, 0.6 + 0.25 * (cur - best_count))

        end_t = time.time() + 4.0
        if n <= 200:
            max_it = 30000
        elif n <= 320:
            max_it = 12000
        else:
            max_it = 5000
        it = 0
        accepts = 0

        while time.time() < end_t and it < max_it:
            it += 1
            if st["zeros"].size == 0 or st["ones"].size == 0:
                break
            i = int(st["zeros"][rng.randint(st["zeros"].size)])
            j = int(st["ones"][rng.randint(st["ones"].size)])
            score = pair_score(st, i, j)
            d = score - cur
            accept = d < 0 or (d <= 8 and T > 1e-5 and rng.random() < math.exp(-float(d) / T))
            if accept:
                cur = apply_pair(st, i, j)
                accepts += 1
                if cur < best_count:
                    best_count = cur
                    best_H = st["H"].copy()
                    last_imp = time.time()
                    no_imp = 0
                else:
                    no_imp += 1
                if accepts and accepts % 5000 == 0:
                    st = build_state(st["H"])
                    cur = int(st["c"].max())
                T = max(0.05, T * 0.997)
            else:
                no_imp += 1
                if no_imp > 5000:
                    st = build_state(best_H)
                    cur = best_count
                    no_imp = 0
                    T = 1.0
                    break

        if time.time() < deadline - 1.0:
            desc_cap = 200 if n <= 200 else 80
            st, _, cur = do_descent(st, min(deadline - 1.0, time.time() + 3.0), desc_cap)
            if cur < best_count:
                best_count = cur
                best_H = st["H"].copy()
                last_imp = time.time()

        if time.time() - last_imp > 90.0:
            break

    best_c = best_count * (2.0 / float(n))
    return best_H.astype(float), best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
        parent_n = int(len(parent))
    except Exception:
        parent_n = 400
        parent = np.full(parent_n, 0.5, dtype=float)

    if parent_n <= 0:
        parent_n = 400
        parent = np.full(parent_n, 0.5, dtype=float)

    best_h = None
    best_c = 1.0
    best_n = 0

    if parent_n % 2 == 0 and 64 <= parent_n <= 512:
        hp = project_feasible(parent, parent_n)
        cp = compute_c5(hp, parent_n)
        best_h = hp
        best_c = cp
        best_n = parent_n

    ns = []
    for m in (128, 160, 200, 240, 280, 320, 360, 400):
        if m % 2 == 0 and m not in ns:
            ns.append(m)
    if parent_n % 2 == 0 and 64 <= parent_n <= 512 and parent_n not in ns:
        ns.append(parent_n)
    ns.sort()

    scenarios = []
    for n in ns:
        nv = 5 if n <= 160 else 4 if n <= 240 else 3
        for v in range(nv):
            scenarios.append((n, v))

    hc_cache = {}
    polish_reserve = 160.0
    sc_deadline = deadline - polish_reserve

    for idx, (n, v) in enumerate(scenarios):
        if time.time() > sc_deadline:
            break
        rem = max(1, len(scenarios) - idx)
        slot = (sc_deadline - time.time()) / float(rem)
        slot = max(3.0, min(60.0, slot))

        if n not in hc_cache:
            hc = downsample_h(parent, n, 0.0, rng)
            hc_cache[n] = hc
            c0 = compute_c5(hc, n)
            if c0 < best_c:
                best_c = c0
                best_h = hc.copy()
                best_n = n
        else:
            hc = hc_cache[n]

        H0 = make_binary_start(hc, n, v, rng)
        H1, c1 = binary_search(n, H0, time.time() + slot, rng)
        if c1 < best_c:
            best_c = c1
            best_h = H1.copy()
            best_n = int(n)

    starts = []
    if best_h is not None:
        starts.append((best_h, int(best_n)))
    if best_n in hc_cache:
        starts.append((hc_cache[best_n], int(best_n)))
    if parent_n in hc_cache:
        starts.append((hc_cache[parent_n], int(parent_n)))

    seen = set()
    uniq = []
    for h0, n0 in starts:
        if n0 not in seen:
            seen.add(n0)
            uniq.append((h0, n0))

    pol_deadline = deadline - 3.0
    if time.time() < pol_deadline and uniq:
        nstarts = len(uniq)
        for pi, (h0, n0) in enumerate(uniq):
            limit = pol_deadline - (pol_deadline - time.time()) * (nstarts - pi - 1) / max(1, nstarts)
            if time.time() > limit - 2.0:
                continue
            if n0 <= 200:
                alphas = (20, 100, 500)
            else:
                alphas = (20, 100, 300)
            h1, c1 = continuous_refine(h0, n0, limit, alphas)
            if c1 < best_c:
                best_c = c1
                best_h = h1.copy()
                best_n = int(n0)

    if best_h is None:
        n = 200
        H0 = make_binary_start(np.full(n, 0.5, dtype=float), n, 3, rng)
        h1, c1 = binary_search(n, H0, deadline - 2.0, rng)
        best_h = h1
        best_c = c1
        best_n = n

    best_h = project_feasible(np.asarray(best_h, dtype=float), int(best_n))
    best_c = compute_c5(best_h, int(best_n))
    return best_h, best_c, int(best_n)
```
