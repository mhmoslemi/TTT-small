import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        room = 1.0 - h
        if diff > 0:
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_objective(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    # For each lag k (index in c is k + n - 1):
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def lbfgs_optimize(h, n, deadline, alpha=100000, maxiter=500):
    def obj(x):
        return smooth_max_objective(x, n, alpha)
    def grad(x):
        return smooth_max_grad(x, n, alpha)
    
    bounds = [(0.0, 1.0)] * n
    try:

                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-16, 'maxfun': maxiter * 20})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except:
        return h.copy(), compute_c5(h, n)

def local_search(h, n, deadline, max_rounds=20):

    best_h = h.copy()
    
    for alpha in [1000, 5000, 50000, 200000]:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005]

                    i = order[i_pos]

                                c = compute_c5(best_h, n)

def diverse_start(n, rng, method='binary'):
    if method == 'binary':
        h = np.zeros(n)
        half = n // 2
        idx = rng.choice(n, size=half, replace=False)

    elif method == 'fourier':

            phase = rng.uniform(0, 2 * np.pi)

    elif method == 'golden':

        phi = (1.0 + np.sqrt(5.0)) / 2.0

                    best_h[i] -= step
                    best_h[j] += step
        
        if not improved:
            # Try smaller steps
            for i_pos in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                i = order[i_pos]
                for j_pos in range(i_pos + 1, min(n, i_pos + 30)):
                    j = order[j_pos]
                    if g[i] - g[j] > -1e-10:
                        continue
                    step = min(0.001, (1 - best_h[i]), best_h[j])
                    if step < 1e-7:
                        continue
                    best_h[i] += step
                    best_h[j] -= step
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] -= step
                        best_h[j] += step
            if not improved:
                break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-13:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def lbfgs_opt(h, n, deadline, alpha=50000, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h, jac=lambda x: smooth_max_grad(x, n, alpha),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except:
        return h.copy(), compute_c5(h, n)

def gradient_local_search(h, n, deadline, window=50):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps_to_try = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for step in steps_to_try:
        any_improved = True
        while any_improved and time.time() < deadline:
            any_improved = False
            g = smooth_max_grad(best_h, n, 10000)
            order = np.argsort(g)
            for i_pos in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                i = order[i_pos]
                for j_pos in range(i_pos + 1, min(n, i_pos + window + 1)):
                    j = order[j_pos]
                    if g[j] - g[i] < 1e-10:
                        continue
                    if best_h[i] + step <= 1.0 and best_h[j] - step >= 0.0:
                        best_h[i] += step
                        best_h[j] -= step
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            any_improved = True
                        else:
                            best_h[i] -= step
                            best_h[j] += step
    return best_h, best_c

def diverse_start(n, rng, method):
    if method == 'binary':
        h = np.zeros(n)
        half = n // 2
        idx = rng.choice(n, size=half, replace=False)
        h[idx] = 1.0
    elif method == 'golden':
        h = np.zeros(n)
        half = n // 2
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        positions = np.floor(np.arange(half) * phi).astype(int) % n
        h[positions] = 1.0
    elif method == 'fourier':
        h = np.full(n, 0.5)
        x = np.linspace(0, 2, n, endpoint=False)
        for freq in range(1, 8):
            amp = rng.uniform(0.05, 0.25) / freq
            phase = rng.uniform(0, 2 * np.pi)
            h += amp * np.cos(2 * np.pi * freq * x + phase)
    elif method == 'banded':
        h = np.zeros(n)
        block = max(1, n // 8)
        for b in range(8):
            s = b * block
            e = min(n, s + block)
            if b % 2 == 0:
                h[s:e] = 1.0
            else:
                h[s:e] = 0.5
    else:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
    return project_feasible(h, n)

def coarse_to_fine(h_fine, n_fine, n_coarse, deadline):
    t = time.time()
    x_fine = np.linspace(0, 2, n_fine, endpoint=False)
    x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
    h_c = np.interp(x_coarse, x_fine, h_fine)
    h_c = project_feasible(h_c, n_coarse)
    t_alloc = min(60, (deadline - t) * 0.4)
    h_c, c_c = lbfgs_opt(h_c, n_coarse, t + t_alloc, alpha=10000, maxiter=150)
    h_c, c_c = gradient_local_search(h_c, n_coarse, t + t_alloc + 30, window=30)
    x_fine2 = np.linspace(0, 2, n_fine, endpoint=False)
    h_up = np.interp(x_fine2, x_coarse, h_c)
    h_up = project_feasible(h_up, n_fine)
    return h_up
