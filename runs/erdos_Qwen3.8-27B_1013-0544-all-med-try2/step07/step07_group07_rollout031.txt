            h_new, c_new = multi_alpha_slsqp(hp, n, t_end, [5000, 50000])
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Final polish
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        h_new = slsqp_opt(best_h, n, 500000, maxiter=4000, ftol=1e-17)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new; best_h = h_new
        best_h, best_c = local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(300, 0.5), 300)
        best_c = compute_c5(best_h, 300)

    return best_h, best_c, len(best_h)
