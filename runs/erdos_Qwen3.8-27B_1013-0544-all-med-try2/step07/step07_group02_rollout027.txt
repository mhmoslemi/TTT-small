def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj_and_grad(h, n, alpha, pen_w=1e8):
    """Compute soft-max objective and gradient with constraint penalty."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    log_sum_exp = sm + np.log(exp_s.sum())
    soft_max = m + log_sum_exp / alpha
    weights = exp_s / exp_s.sum()
    
    # Gradient computation
    lags = np.arange(2 * n - 1) - (n - 1)
    # Only keep significant weights
    if alpha > 50:
        sig = weights > 1e-12
    else:
        sig = np.ones(2*n-1, dtype=bool)
    
    idx_sig = np.where(sig)[0]
    grad = np.full(n, dx)
    
    for idx in idx_sig:
        lag = lags[idx]
        w = weights[idx]
        if lag == 0:
            grad[:] -= dx * w * h * 2.0
        else:
            # T1: h[j+lag] for valid j
            js = max(0, -lag)
            je = min(n, n - lag)
            if js < je:
                grad[js:je] -= dx * w * h[js+lag:je+lag]
            # T2: h[j-lag] for valid j
            js2 = max(0, lag)
            je2 = min(n, n + lag)
            if js2 < je2:
                grad[js2:je2] -= dx * w * h[js2-lag:je2-lag]
    
    # Constraint penalty
    viol = h.sum() - n / 2.0
    obj = soft_max + pen_w * viol * viol
    grad += 2.0 * pen_w * viol
    
    return obj, grad
