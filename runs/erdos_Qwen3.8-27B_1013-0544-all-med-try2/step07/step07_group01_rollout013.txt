    c = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
    m = np.max(c)
    sh = alpha*(c-m)
    mx = np.max(sh)
    val = m + (mx + np.log(np.sum(np.exp(sh-mx))))/alpha
    s = np.sum(x) - 0.5*n

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    if abs(float(h.sum()) - target) < 1e-14:
        return h

    lo = float(h.min())
    hi = float(h.max())
    mid = 0.5 * (lo + hi)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    f = np.clip(h - mid, 0.0, 1.0)
    r = target - float(f.sum())
    if abs(r) > 1e-13:
        for _ in range(200):
            mask = (f > 1e-12) & (f < 1.0 - 1e-12)
            cnt = int(mask.sum())
            if cnt <= 0:

            f[mask] += r / cnt

            r = target - float(f.sum())
            if abs(r) < 1e-15:
                break
    return f


def compute_c_full(h, n):
    return np.correlate(h, 1.0 - h, mode="full")[::-1] * (2.0 / float(n))


def compute_c5(h, n):
    return float(np.max(compute_c_full(h, n)))


def default_h(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.5 * np.cos(np.pi * x)
    return project_feasible(h, n)


def update_c(c, h, p, delta, dx):
    n = len(h)
    a = delta * dx
    s = n - 1 - int(p)
    c[s:s + n] += a
    c[s:s + n] -= a * h
    c[n - 1:n + int(p)] += -a * h[int(p)::-1]
    c[n - 1] -= delta * delta * dx
    h[int(p)] += delta


def try_single(c, h, i, j, d, dx, best, tol):
    if i == j or d <= 1e-9:
        return best, False
    ct = c.copy()
    oi = float(h[i])
    oj = float(h[j])
    update_c(c, h, i, d, dx)
    update_c(c, h, j, -d, dx)
    val = float(np.max(c))
    if val < best - tol:
        return val, True
    c[:] = ct
    h[i] = oi
    h[j] = oj
    return best, False


def grad_active(c, h, n):
    k = min(150, c.size)
    idx = np.argsort(c, kind="mergesort")[::-1][:k]
    weights = 1.0 / (np.arange(1, k + 1, dtype=float))
    off = n - 1
    dx = 2.0 / float(n)
    g = np.zeros(n)

    for tt, wt in zip(idx, weights):
        t = int(tt)
        if wt <= 1e-16:
            continue
        s = t - off
        if s >= 0:
            r = s
            lev = n - r
            g[0:lev] += wt * (1.0 - h[r:n])
            g[r:n] += -wt * h[0:lev]
        else:
            r = -s
            lev = n - r
            g[r:n] += wt * (1.0 - h[0:lev])
            g[0:lev] += -wt * h[r:n]
    g *= dx
    return g


def proj_grad_step(h, c, n, dx, max_attempts=2):
    best = float(np.max(c))
    taus = (0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001)

    for _ in range(max_attempts):
        g = grad_active(c, h, n)
        d = g - float(np.mean(g))
        norm = float(np.linalg.norm(d))
        if norm < 1e-14:
            break
        d = -d / norm
        done = False
        for tau in taus:
            trial = project_feasible(h + tau * d, n)
            tc = compute_c_full(trial, n)
            tv = float(np.max(tc))
            if tv < best - 1e-12:
                h = trial
                c = tc
                best = tv
                done = True
                break
        if not done:
            break
    return h, c, best


def local_search(h, n, deadline, seed=1):
    h = project_feasible(h, n).copy()
    c = compute_c_full(h, n)
    dx = 2.0 / float(n)
    best = float(np.max(c))

    safe_h = h.copy()
    safe_c = c.copy()
    safe_best = best

    rng = np.random.RandomState(seed)
    scales = (1.0e-1, 5.0e-2, 2.5e-2, 1.25e-2, 6.25e-3, 3.125e-3,
              1.5625e-3, 7.8125e-4, 3.90625e-4, 2.0e-4, 1.0e-4,
              5.0e-5, 2.5e-5, 1.25e-5, 6.0e-6, 3.0e-6, 1.5e-6)
    scale_idx = 0
    no_scale = 0
    restarts = 0
    pass_count = 0
    tol = 1e-12
    low_count = 20 if n >= 500 else 24

    last_verify = time.time()
    last_improve = time.time()

    while time.time() < deadline:
        now = time.time()
        if now - last_verify > 3.0 or now - last_improve > 20.0:
            h = project_feasible(h, n)
            c = compute_c_full(h, n)
            actual = float(np.max(c))
            if actual < best:
                best = actual
            if actual < safe_best - 1e-12:
                safe_best = actual
                safe_h = h.copy()
                safe_c = c.copy()
                last_improve = time.time()
            elif actual > safe_best + 1e-7:
                h = safe_h.copy()
                c = safe_c.copy()
                best = safe_best
            last_verify = now

        pass_count += 1
        if pass_count % 25 == 0 and time.time() < deadline - 5:
            h2, c2, b2 = proj_grad_step(h, c, n, dx, max_attempts=2)
            if b2 < best - tol:
                h = h2
                c = c2
                best = b2
                last_improve = time.time()
                no_scale = 0

        g = grad_active(c, h, n)
        gp = g - float(np.mean(g))
        order = np.argsort(gp, kind="mergesort")
        low = order[:low_count]
        high = order[-low_count:]

        scale = scales[min(scale_idx, len(scales) - 1)]
        improved = False
        acc = 0

        for i in low:
            if time.time() >= deadline:
                break
            for j in high:
                if i == j:
                    continue
                dmax = min(float(h[j]), 1.0 - float(h[i]))
                if dmax <= 1e-9:
                    continue
                half = 0.5 * dmax
                d = scale if scale < half else half
                if d > 1e-9:
                    nb, ok = try_single(c, h, i, j, d, dx, best, tol)
                    if ok:
                        best = nb
                        improved = True
                        last_improve = time.time()
                        no_scale = 0
                        acc += 1
                        if acc > 22:
                            break
            if acc > 22:
                break

        if acc < 28 and time.time() < deadline - 1:
            for _ in range(80):
                if time.time() >= deadline:
                    break
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                if gp[i] <= gp[j]:
                    recv, giver = i, j
                else:
                    recv, giver = j, i
                dmax = min(float(h[giver]), 1.0 - float(h[recv]))
                if dmax <= 1e-9:
                    continue
                half = 0.5 * dmax
                half_scale = 0.5 * scale
                d = half_scale if half_scale < half else half
                if d > 1e-9:
                    nb, ok = try_single(c, h, recv, giver, d, dx, best, tol)
                    if ok:
                        best = nb
                        improved = True
                        last_improve = time.time()
                        no_scale = 0
                        acc += 1
                        if acc > 34:
                            break

        if improved:
            no_scale = 0
        else:
            no_scale += 1
            if no_scale >= 2:
                scale_idx += 1
                no_scale = 0
                if scale_idx >= len(scales):
                    scale_idx = len(scales) - 1
                    if restarts >= 6 or time.time() > deadline - 25:
                        break

                    h = project_feasible(h, n)
                    c = compute_c_full(h, n)
                    actual = float(np.max(c))
                    if actual < safe_best:
                        safe_best = actual
                        safe_h = h.copy()
                        safe_c = c.copy()

                    h = safe_h.copy()
                    for _ in range(3):
                        a = int(rng.randint(n))
                        b = int(rng.randint(n))
                        if a == b:
                            continue
                        d = 0.02 * float(rng.rand())
                        if rng.rand() < 0.5:
                            recv, giver = a, b
                        else:
                            recv, giver = b, a
                        d = min(d, float(h[giver]), 1.0 - float(h[recv]))
                        if d > 1e-9:
                            h[recv] += d
                            h[giver] -= d

                    h = project_feasible(h, n)
                    c = compute_c_full(h, n)
                    actual = float(np.max(c))
                    if actual > safe_best + 4e-3:
                        h = safe_h.copy()
                        c = safe_c.copy()
                        best = safe_best
                    else:
                        best = actual
                    last_improve = time.time()
                    last_verify = time.time()
                    scale_idx = 4
                    no_scale = 0
                    restarts += 1
                    continue

    h = project_feasible(h, n)
    c = compute_c_full(h, n)
    actual = float(np.max(c))
    if actual < safe_best:
        safe_best = actual
        safe_h = h.copy()
    return safe_h, safe_best


def lbfgs_obj(x, n, alpha, mu):
    x = np.asarray(x, dtype=float)
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full")[::-1] * dx
    m = float(np.max(c))
    sh = alpha * (c - m)
    mx = float(np.max(sh))
    val = m + (mx + float(np.log(np.sum(np.exp(sh - mx))))) / alpha
    s = float(np.sum(x)) - 0.5 * float(n)
    val += mu * s * s
    return float(val)


def lbfgs_grad(x, n, alpha, mu):
    x = np.asarray(x, dtype=float)
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full")[::-1] * dx
    m = float(np.max(c))
    sh = alpha * (c - m)
    mx = float(np.max(sh))
    e = np.exp(sh - mx)
    w = e / float(np.sum(e))

    off = n - 1
    g = np.zeros(n)
    for t, wt in enumerate(w):
        if wt <= 1e-15:
            continue
        s = int(t) - off
        if s >= 0:
            r = s
            lev = n - r
            g[0:lev] += wt * (1.0 - x[r:n])
            g[r:n] += -wt * x[0:lev]
        else:
            r = -s
            lev = n - r
            g[r:n] += wt * (1.0 - x[0:lev])
            g[0:lev] += -wt * x[r:n]
    g *= dx
    g += 2.0 * mu * (float(np.sum(x)) - 0.5 * float(n))
    return g


def lbfgs_polish(h, n, deadline):
    h = project_feasible(h, n)
    best = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    maxiter = 220 if n <= 400 else 110
    configs = ((100.0, 1e-2), (1000.0, 1e-1), (5000.0, 1.0), (20000.0, 10.0))

    for alpha, mu in configs:
        if time.time() > deadline - 4:
            break
        try:
            res = minimize(
                lbfgs_obj, h, jac=lbfgs_grad,
                args=(n, alpha, mu), method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": maxiter, "ftol": 1e-16,
                         "gtol": 1e-12, "maxfun": maxiter * 6}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                h2 = project_feasible(res.x, n)
                c2 = compute_c5(h2, n)
                if np.isfinite(c2) and c2 < best - 1e-14:
                    h = h2
                    best = c2
        except Exception:
            pass
    return h, best


def interp_to_n(h, n0, n1):
    if n0 == n1:
        return np.asarray(h, dtype=float).copy()
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    return np.interp(x1, x0, np.asarray(h, dtype=float), left=float(h[0]), right=float(h[-1]))


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline = t0 + max(5.0, budget - 5.0)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 300
        h = default_h(n)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    best_n = n

    if not np.isfinite(best_c):
        best_h = default_h(n)
        best_c = compute_c5(best_h, n)

    if budget < 180.0:
        h2, c2 = local_search(best_h, best_n, deadline, seed=int(seed) + 1)
        bb = project_feasible(h2, best_n)
        return bb, compute_c5(bb, best_n), best_n

    # Phase A: polish current resolution.
    tA = t0 + 0.24 * budget
    if tA > deadline - 0.40 * budget:
        tA = deadline - 0.40 * budget
    if best_n <= 400 and budget > 450.0 and time.time() < tA - 70.0:
        tL = min(time.time() + 0.06 * budget, tA - 70.0)
        if tL > time.time() + 15.0:
            hh, cc = lbfgs_polish(best_h, best_n, tL)
            if np.isfinite(cc) and cc < best_c:
                best_h = project_feasible(hh, best_n)
                best_c = cc
    if tA > time.time() + 20.0:
        hh, cc = local_search(best_h, best_n, tA, seed=int(seed) + 1)
        if np.isfinite(cc) and cc < best_c:
            best_h = project_feasible(hh, best_n)
            best_c = cc
        elif np.isfinite(cc):
            best_h = project_feasible(hh, best_n)
            best_c = min(best_c, compute_c5(best_h, best_n))

    # Phase B: move to 600 samples if below.
    if best_n < 600 and time.time() < deadline - 0.32 * budget:
        tB = t0 + 0.70 * budget
        if tB > deadline - 40.0:
            tB = deadline - 40.0
        if tB > time.time() + 30.0:
            hb = project_feasible(interp_to_n(best_h, best_n, 600), 600)
            hb, cb = local_search(hb, 600, tB, seed=int(seed) + 2)
            if np.isfinite(cb) and cb < best_c:
                best_h = project_feasible(hb, 600)
                best_c = cb
                best_n = 600

    # Phase C: final refinement at 900 samples if time remains.
    if best_n < 900 and time.time() < deadline - 35.0:
        hc = project_feasible(interp_to_n(best_h, best_n, 900), 900)
        hc, cc = local_search(hc, 900, deadline - 2.0, seed=int(seed) + 3)
        if np.isfinite(cc) and cc < best_c:
            best_h = project_feasible(hc, 900)
            best_c = cc
            best_n = 900

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
