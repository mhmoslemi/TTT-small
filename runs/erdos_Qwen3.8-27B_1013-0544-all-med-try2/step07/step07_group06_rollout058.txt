def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * 2.0 / n
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-14:
        grad += w0 * dx * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = w[n - 1 + k] * dx
        wn = w[n - 1 - k] * dx
        if wp > 1e-14:
            grad[:n-k] += wp * (1.0 - h[k:])
            grad[k:] += wp * (-h[:n-k])
        if wn > 1e-14:
            grad[k:] += wn * (1.0 - h[:n-k])
            grad[:n-k] += wn * (-h[k:])
    return val, grad

def gd_refine(h, n, deadline, max_alpha=5000000):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 20.0
    step = 0.03
    stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        if it % 50 == 0:
            alpha = min(alpha * 1.8, max_alpha)
        val, g = smooth_max_val_grad(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g_dir = g / gn
        improved = False
        for ls in range(12):
            h_new = project_feasible(h - step * g_dir, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                improved = True
                stall = 0
                break
            step *= 0.6
        if not improved:
            stall += 1
            step = max(step, 0.001)
            if stall % 20 == 0:
                step = 0.03
        if stall > 100:
            break
    return best_h, best_c

def swap_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        for _ in range(100):
            if time.time() > deadline:
                return h, best_c
            improved = False
            val, g = smooth_max_val_grad(h, n, 100000)
            order = np.argsort(-np.abs(g))
            for i in order[:80]:
                if time.time() > deadline:
                    return h, best_c
                if g[i] > 0.0005:
                    if h[i] < d:
                        continue
                    cands = np.where((g < -0.0005) & (h <= 1.0 - d))[0]
                    if len(cands) == 0:
                        continue
                    j = cands[np.argmin(g[cands])]
                elif g[i] < -0.0005:
                    if h[i] > 1.0 - d:
                        continue
                    cands = np.where((g > 0.0005) & (h >= d))[0]
                    if len(cands) == 0:
                        continue
                    j = cands[np.argmax(g[cands])]
                    h[i] += d
                    h[j] -= d
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
                    continue
                else:
                    continue
                h[i] -= d
                h[j] += d
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved = True
                else:
                    h[i] += d
                    h[j] -= d
            if not improved:
                break
    return h, best_c

def binary_push(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(60):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            if h[i] < 0.03 or h[i] > 0.97:
                continue
            target = 0.0 if h[i] < 0.5 else 1.0
            delta = target - h[i]
            mask = (h > abs(delta)) & (h < 1.0 - abs(delta))
            mask[i] = False
            if not mask.any():
                mask = (h > 0.01) & (h < 0.99)
                mask[i] = False
            if not mask.any():
                continue
            h_new = h.copy()
            h_new[i] = target
            h_new[mask] -= delta / mask.sum()
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c - 1e-14:
                best_c = c
                h = h_new
                improved = True
        if not improved:
            break
    return h, best_c

def refine_pipeline(h, n, deadline):
    h = project_feasible(h.copy(), n)
    c = compute_c5(h, n)
    # Phase 1: GD
    t1 = min(time.time() + (deadline - time.time()) * 0.4, deadline - (deadline - time.time()) * 0.4)
    h1, c1 = gd_refine(h, n, t1)
    if c1 < c:
        c, h = c1, h1
    # Phase 2: Swap
    if time.time() < deadline - 60:
        t2 = min(time.time() + 80, deadline - 60)
        h2, c2 = swap_search(h, n, t2)
        if c2 < c:
            c, h = c2, h2
    # Phase 3: Binary push
    if time.time() < deadline - 40:
        t3 = min(time.time() + 50, deadline - 40)
        h3, c3 = binary_push(h, n, t3)
        if c3 < c:
            c, h = c3, h3
    # Phase 4: GD again
    if time.time() < deadline - 30:
        t4 = min(time.time() + 40, deadline - 30)
        h4, c4 = gd_refine(h, n, t4)
        if c4 < c:
            c, h = c4, h4
    # Phase 5: Final swap
    if time.time() < deadline - 10:
        h5, c5 = swap_search(h, n, deadline - 5)
        if c5 < c:
            c, h = c5, h5
    return h, c
