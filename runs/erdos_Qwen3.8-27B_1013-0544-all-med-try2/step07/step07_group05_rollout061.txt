Actually, one more idea that might be effective: **iterative minimization of the worst overlap**. 

Algorithm:
1. Compute all overlaps c_k
2. Let k* = argmax c_k
3. Solve: min_h c_{k*}(h) subject to sum(h)=n/2, 0<=h<=1, and c_k(h) <= c_{k*}(h) + ε for all k
4. Repeat

Step 3 is a quadratic program (since c_{k*} is bilinear, not linear). So it's still non-convex.

But if I linearize: at the current h, c_{k*}(h + δ) ≈ c_{k*}(h) + ∇c_{k*}(h) · δ. Then:
min δ: ∇c_{k*}(h) · δ
s.t. sum(δ) = 0, -h <= δ <= 1-h, and ∇c_k(h)·δ <= ε for all k

This is a linear program! I could solve it with scipy.optimize.linprog.

```python
def lp_minmax_step(h, n, t):
    """One LP step to reduce the maximum overlap."""
    from scipy.optimize import linprog
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    k_star = int(np.argmax(corr))
    m = k_star - (n - 1)
    
    # Linearize c around h
    # c_k(h + d) ≈ c_k(h) + grad_k(h) · d
    # We want to minimize grad_{k*} · d subject to:
    #   sum(d) = 0
    #   -h_i <= d_i <= 1-h_i
    #   grad_k(h) · d <= (c_{k*} - c_k + eps) for all k (keep others from increasing)
    
    # C matrix: grad for each k
    # c_k = sum_i h[i](1-h[i+m_k]) where m_k = k-(n-1)
    # grad_k[i] = (1-h[i+m_k])*[0<=i+m_k<n] - h[i-m_k]*[0<=i-m_k<n]
    
    # This is a 499 x 250 matrix. Building it is O(n^2) per step.
    # For n=250, that's 125000 ops. Fine.
    
    # Actually, let's just do the top few active constraints to keep the LP small.
    top_k = np.argsort(corr)[-10:]  # top 10 largest overlaps
    
    # Build gradient matrix for top_k shifts
    A = np.zeros((len(top_k), n))
    for idx, k in enumerate(top_k):
        m_k = k - (n - 1)
        for i in range(n):
            g = 0.0
            if 0 <= i + m_k < n:
                g += (1.0 - h[i + m_k])
            if 0 <= i - m_k < n:
                g -= h[i - m_k]
            A[idx, i] = g
    
    # Objective: minimize A[0] · d (reduce the worst)
    c_obj = A[0]
    
    # Constraints:
    # A[i] · d <= A[0] · h... no, we want A[i]·d <= slack_i where slack_i = c_{k_i} - c_{k_0} + eps
    b_slack = corr[top_k] - corr[k_star] + 0.001  # allow slight increase for others
    
    # Actually: we want new_c_k <= old_c_{k*} for all k
    # new_c_k ≈ old_c_k + A[i]·d
    # old_c_k + A[i]·d <= old_c_{k*}
    # A[i]·d <= old_c_{k*} - old_c_k
    b_max = corr[k_star] - corr[top_k]  # = -b_slack
    
    # Equality: sum(d) = 0
    A_eq = np.ones(n).reshape(1, -1)
    b_eq = np.zeros(1)
    
    # Bounds: -h_i <= d_i <= 1-h_i
    bounds = [(-h[i], 1-h[i]) for i in range(n)]
    
    res = linprog(c_obj, A_ub=A, b_ub=b_max, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_new = project_feasible(h + res.x, n)
    else:
        h_new = h.copy()
    
    return h_new
```

```python
def build_grad_matrix(h, n, shifts):
    """Build gradient matrix for given shifts."""
    A = np.zeros((len(shifts), n))
    for idx, m in enumerate(shifts):
        row = np.zeros(n)
        lo1 = max(0, m)
        hi1 = min(n, n + m)
        if hi1 > lo1:
            row[lo1:hi1] += (1.0 - h[lo1-m:hi1-m])
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if hi2 > lo2:
            row[lo2:hi2] -= h[lo2+m:hi2+m]
        A[idx] = row
    return A
```

This is O(n) per shift, O(n * |shifts|) total. For |shifts|=10 and n=250, that's 2500 ops. Fast.

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slsqp(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_shift(h, n, m):
    grad = np.zeros(n)
    lo1 = max(0, m)
    hi1 = min(n, n + m)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 - m:hi1 - m])
    lo2 = max(0, -m)
    hi2 = min(n, n - m)
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 + m:hi2 + m]
    return grad

def subgrad_minmax(h, n, t_end, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.004
    no_improve = 0

    for it in range(200000):
        if time.time() > t_end:
            break
        c = compute_all_overlaps(h, n)
        k_star = int(np.argmax(c))
        m = k_star - (n - 1)
        if m == 0:
            continue
        grad = grad_shift(h, n, m)
        h = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
            no_improve = 0
            eta = min(eta * 1.02, 0.04)
        else:
            no_improve += 1
            if no_improve > 200:
                eta *= 0.9
            else:
                eta *= 0.995
    return best_h, best_c

def sa_optimize(h, n, t_end, rng, T_init=0.015):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    T = T_init
    cooling = 0.9998

    for it in range(500000):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if rng.rand() < 0.5:
            max_d = min(h[i], 1.0 - h[j])
            if max_d < 1e-12:
                continue
            d = rng.uniform(0.001, min(0.08, max_d))
            old_i, old_j = h[i], h[j]
            h[i] -= d
            h[j] += d
        else:
            max_d = min(1.0 - h[i], h[j])
            if max_d < 1e-12:
                continue
            d = rng.uniform(0.001, min(0.08, max_d))
            old_i, old_j = h[i], h[j]
            h[i] += d
            h[j] -= d

        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-min(delta / T, 60))):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = old_i
            h[j] = old_j
        T *= cooling
    return best_h, best_c

def pair_search(h, n, t_end, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < t_end:
            improved = False
            for i in range(n):
                if time.time() > t_end:
                    return h, best_c
                if radius is None:
                    lo = i + 1
                    hi = n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c
