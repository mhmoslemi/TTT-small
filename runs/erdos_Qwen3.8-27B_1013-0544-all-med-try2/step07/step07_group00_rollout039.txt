import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s < target:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (target - s) / tot * room
        else:
            if s < 1e-14:
                break
            h *= target / s
    return np.clip(h, 0.0, 1.0)

def subgrad_step(h, n, alpha, step_size):
    """Take one subgradient step with smoothing parameter alpha"""
    c = compute_c5_vec(h, n)
    a = alpha * c
    mx = a.max()
    w = np.exp(a - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    offset = n - 1
    for m in range(2 * n - 1):
        if w[m] < 1e-10:
            continue
        k = m - offset
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            if nv <= 0:
                continue
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv <= 0:
                continue
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return h - step_size * grad, c.max()

def block_cd(h, n, deadline, block_size=10):

    best_c = compute_c5(h, n)
    n_blocks = n // block_size

    for iteration in range(200):

            break
        improved = False

            if time.time() > deadline:
                return h, best_c
            lo = b * block_size
            hi = lo + block_size

                hi = n
            # Try small perturbations of this block

                h_try = h.copy()
                block = h_try[lo:hi]
                # Random perturbation

                h_try = project_feasible(h_try, n)

                if c_try < best_c - 1e-12:
                    best_c = c_try
                    h = h_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c

    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    beta = 0.9
    # Polyak-like step size
    # We need a lower bound on c*: use 0.25 as theoretical lower bound
    c_lb = 0.25
    h_curr = h.copy()
    c_curr = best_c
    for it in range(iters):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h_curr, n)
        a = alpha * c
        mx = a.max()
        w = np.exp(a - mx)
        w /= w.sum()
        dx = 2.0 / n

    return best_h, best_c

def binary_search_local(h, n, deadline, radius=60):
    """Local search on binary rounding"""
    best_c = compute_c5(h, n)
    # Create binary version
    h_bin = (h >= 0.5).astype(float)
    # Ensure exactly n//2 ones
    ones = np.where(h_bin == 1)[0]
    if len(ones) > n // 2:
        # Remove excess ones (those closest to 0.5)
        excess = len(ones) - n // 2
        h_temp = h.copy()
        h_temp[ones] = h[ones]
        # Keep the ones with highest h values
        order = np.argsort(-h[ones])
        remove = ones[order[excess:]]
        h_bin[remove] = 0
    elif len(ones) < n // 2:
        zeros = np.where(h_bin == 0)[0]
        deficit = n // 2 - len(ones)
        order = np.argsort(-h[zeros])
        add = zeros[order[:deficit]]
        h_bin[add] = 1
    
    # Local search: try swapping a 1 and a 0
    h_ls = h_bin.copy()
    best_ls_c = compute_c5(h_ls, n)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones_pos = np.where(h_ls == 1)[0]
        zeros_pos = np.where(h_ls == 0)[0]
        # Shuffle for diversity
        rng = np.random.RandomState(hash((n, len(ones_pos))) % 2**31)
        rng.shuffle(ones_pos)
        rng.shuffle(zeros_pos)
        for i in ones_pos[:500]:
            if time.time() > deadline:
                break
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            cands = zeros_pos[(zeros_pos >= lo) & (zeros_pos <= hi)]
            for j in cands:
                h_ls[i] = 0
                h_ls[j] = 1
                c = compute_c5(h_ls, n)
                if c < best_ls_c - 1e-12:
                    best_ls_c = c
                    improved = True
                    break
                else:
                    h_ls[i] = 1
                    h_ls[j] = 0
            if improved:
                zeros_pos = np.where(h_ls == 0)[0]
                ones_pos = np.where(h_ls == 1)[0]
                if improved:
                    break
        if not improved:
            break
    
    # Relax from binary
    h_relaxed = h_ls.copy()

One more idea: **coordinate descent with exact line search**. At each step, pick one coordinate h[i] and optimize it exactly (given all other coordinates fixed). The objective as a function of h[i] is:
max_k (sum_{j≠i} h[j](1-h[j+k]) + h[i](1-h[i+k]) - h[i-k]h[i] (if valid)) * dx

This is the max of linear functions in h[i], so the optimum is at h[i]=0, h[i]=1, or at the intersection of two active constraints. We can find this exactly!

Wait, that's not quite right. The term h[i](1-h[i+k]) is bilinear, not linear in h[i] (since 0<=h<=1, but h[i+k] is fixed). Actually, h[i](1-h[i+k]) = h[i] - h[i]*h[i+k]. With h[i+k] fixed, this is linear in h[i]. Similarly, -h[i-k]*h[i] is linear in h[i]. So each c_k is linear in h[i] when all other variables are fixed.

The max of linear functions is a convex piecewise-linear function. Its minimum over [0,1] occurs at one of the breakpoints or at 0 or 1.

The breakpoints occur where two c_k values are equal. For coordinate descent, I can:
1. Fix all variables except h[i]
2. For each k, compute c_k as a linear function of h[i]: c_k = a_k * h[i] + b_k
3. The objective is max_k (a_k * h[i] + b_k)
4. Find the minimum of this convex PWL function over [0,1]

This gives an exact coordinate descent step! And it's fast: O(n) per coordinate (compute a_k, b_k for all k).

For n=500, optimizing all 500 coordinates takes O(n²) = 250,000 operations per sweep. Each sweep might take 1ms. So 1000 sweeps = 1 second. Very fast!
