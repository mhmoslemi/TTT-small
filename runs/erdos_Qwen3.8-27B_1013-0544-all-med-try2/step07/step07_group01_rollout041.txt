def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

            tot = room.sum()
            if tot < 1e-15: break

        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h

    c = compute_c5_vec(h, n)

    else:

    bounds = [(0.0, 1.0)] * n
    configs = [(100, 10.0), (500, 20.0), (2000, 100.0), (10000, 500.0),
               (50000, 2000.0), (200000, 10000.0), (1000000, 50000.0),
               (5000000, 200000.0), (20000000, 500000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2: break
        try:
            res = minimize(_smax_obj, h, jac=_smax_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': max_iters * 12, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def momentum_gd(h, n, deadline, lr_init=0.002, alpha_init=500.0, mom=0.85):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    alpha = alpha_init
    for _ in range(200000):
        if time.time() > deadline: break
        g = _smax_grad(h, n, alpha, 0.0)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-16: break
        g = g / g_norm
        v = mom * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        lr *= 0.9999
    return best_h, best_c

def pair_search_local(h, n, deadline, deltas, radius=80):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline: return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j: continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved: break
    return h, best_c

def pair_search_global(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline: return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved: break
    return h, best_c

def sa_binary(h, n, deadline, n_steps=300000):
    rng = np.random.RandomState(777)
    h_bin = (h > 0.5).astype(float)
    total = h_bin.sum()
    if total > n / 2:
        ones = np.where(h_bin == 1)[0]
        to_rm = rng.choice(ones, int(total - n/2), replace=False)
        h_bin[to_rm] = 0
    elif total < n / 2:
        zeros = np.where(h_bin == 0)[0]
        to_add = rng.choice(zeros, int(n/2 - total), replace=False)
        h_bin[to_add] = 1
    h_bin = project_feasible(h_bin, n)
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    cur_c = best_c
    T = 0.03
    T_min = 0.000005
    decay = (T_min / T) ** (1.0 / max(n_steps, 1))
    ones_pos = list(np.where(h_bin == 1)[0].tolist())
    zeros_pos = list(np.where(h_bin == 0)[0].tolist())
    for step in range(n_steps):
        if time.time() > deadline: break
        ii = rng.randint(len(ones_pos))
        jj = rng.randint(len(zeros_pos))
        i = ones_pos[ii]
        j = zeros_pos[jj]
        h_bin[i] = 0; h_bin[j] = 1
        c_new = compute_c5(h_bin, n)
        delta = c_new - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_bin.copy()
            ones_pos[ii] = j
            zeros_pos[jj] = i
        else:
            h_bin[i] = 1; h_bin[j] = 0
        T = max(T * decay, T_min)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: L-BFGS-B
    t1 = min(t0 + 200, deadline - 600)
    best_h, best_c = lbfgs_phase(best_h, n, t1, max_iters=200)
    # Phase 2: Momentum GD
    if time.time() < deadline - 400:
        t2 = min(time.time() + 80, deadline - 400)
        h2, c2 = momentum_gd(best_h, n, t2, lr_init=0.003, alpha_init=2000.0, mom=0.85)
        if c2 < best_c: best_c = c2; best_h = h2
    # Phase 3: Binary SA (alternative path)
    if time.time() < deadline - 300:
        t3 = min(time.time() + 80, deadline - 300)
        h3, c3 = sa_binary(best_h, n, t3, n_steps=200000)
        if c3 < best_c: best_c = c3; best_h = h3
    # Phase 4: L-BFGS-B again on SA result if improved
    if time.time() < deadline - 250:
        t4 = min(time.time() + 60, deadline - 250)
        h4, c4 = lbfgs_phase(best_h, n, t4, max_iters=150)
        if c4 < best_c: best_c = c4; best_h = h4
    # Phase 5: Fine pair search local
    if time.time() < deadline - 150:
        t5 = min(time.time() + 120, deadline - 150)
        best_h, best_c = pair_search_local(best_h, n, t5,
            [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005], radius=60)
    # Phase 6: Global pair search (coarser deltas)
    if time.time() < deadline - 80:
        t6 = min(time.time() + 60, deadline - 80)
        best_h, best_c = pair_search_global(best_h, n, t6, [0.01, 0.005, 0.002])
    # Phase 7: Try n=200, n=400
    if time.time() < deadline - 60:
        for n_try in [200, 400]:
            if time.time() > deadline - 40: break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_t = np.interp(x_new, x_old, best_h)
            h_t = project_feasible(h_t, n_try)
            alloc = min(50, deadline - time.time() - 20)
            if alloc < 15: break
            h_t, c_t = lbfgs_phase(h_t, n_try, time.time() + alloc * 0.5, max_iters=100)
            if time.time() < deadline - 15:
                h_t, c_t = pair_search_local(h_t, n_try, deadline - 10,
                    [0.01, 0.005, 0.001, 0.0005], radius=40)
            if c_t < best_c:
                best_c = c_t; best_h = h_t; n = n_try
    # Phase 8: Final polish
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = pair_search_local(best_h, nf, deadline - 1,
            [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002], radius=40)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
