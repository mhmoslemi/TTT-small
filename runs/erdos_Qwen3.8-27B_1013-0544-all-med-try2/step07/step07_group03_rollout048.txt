        best_c, best_h = ca, ha.copy()

    # Phase 2: L-BFGS-B continuous optimization (40% time)
    p2 = t0 + 0.52 * total
    h = best_h.copy()
    alphas = [500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for alpha in alphas:
        if time.time() > p2 - 5:
            break
        h = lbfgs_smooth(h, n, alpha, 500, p2)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: SLSQP refinement with high alpha (20% time)
    p3 = t0 + 0.72 * total
    h = best_h.copy()
    for alpha in [5000000, 20000000, 100000000]:
        if time.time() > p3 - 5:
            break
        h = slsqp_smooth(h, n, alpha, 2000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Binary anneal from continuous, re-optimize (15% time)
    p4 = t0 + 0.87 * total
    ha2, ca2 = anneal_binary(best_h, n, t0 + 0.80 * total, seed=seed + 1, max_iters=800000)
    if ca2 < best_c:
        best_c, best_h = ca2, ha2.copy()
    h = best_h.copy()
    for alpha in [100000, 2000000, 50000000]:
        if time.time() > p4 - 3:
            break
        h = lbfgs_smooth(h, n, alpha, 300, p4)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: Try other resolutions (20% time)
    p5 = dl - 2
    for nh in [300, 500, 600, 200]:
        if time.time() > p5 - 20:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, ch = greedy_binary(hh, nh, t0 + 0.90 * total, max_sweeps=50)
        for alpha in [10000, 500000, 5000000]:
            if time.time() > p5 - 15:
                break
            hh = lbfgs_smooth(hh, nh, alpha, 300, p5)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh

    # Final polish
    if time.time() < dl - 1:
        h = best_h + 0.003 * rng.randn(n)
        h = project_feasible(h, n)
        h = slsqp_smooth(h, n, 100000000, 1000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
