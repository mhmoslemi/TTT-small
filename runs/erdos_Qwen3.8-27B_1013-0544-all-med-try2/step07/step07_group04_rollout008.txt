```python
import numpy as np
import time


def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def max_count(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(6):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = float(room.sum())
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = float(h.sum())
            if s < 1e-14:
                break
            h += (diff / s) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-15:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = int(np.argmin(h))
            add = min(-diff, h[idx])
            h[idx] -= add
    return h

    if h.size == 0:

    old_n = h.size

            n = 2
    if n > 350:
        n = 300
    if h.size != n:
        xo = np.linspace(0.0, 2.0, h.size, endpoint=False)
        xn = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(xn, xo, h)
    else:
        h = h[:n]
    h = project_feasible(h, n)
    return h, n


def binarize_scores(scores, n, rng, noise=0.0):
    s = np.asarray(scores, dtype=float)
    if s.size != n:
        xo = np.linspace(0.0, 2.0, s.size, endpoint=False)
        xn = np.linspace(0.0, 2.0, n, endpoint=False)
        s = np.interp(xn, xo, s)
    else:
        s = s[:n]
    if noise > 0.0:
        s = s + noise * rng.randn(n)
    target = n // 2
    order = np.argsort(-s)
    h = np.zeros(n, dtype=float)
    h[order[:target]] = 1.0
    return h


def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def build_L(h, n):
    k = 2 * n - 1
    L = np.zeros((n, k), dtype=float)
    o = 1.0 - h
    for d in range(1, n):
        s = n - d
        c = n - 1 + d
        L[:s, c] += o[d:]
        L[d:, c] -= h[:s]
        c = n - 1 - d
        L[d:, c] += o[:s]
        L[:s, c] -= h[d:]
    return L


def greedy_construct(n, reverse=False, seed=0):
    rng = np.random.RandomState(seed)
    target = n // 2
    if reverse:
        h = np.ones(n, dtype=float)
    else:
        h = np.zeros(n, dtype=float)
    F = np.zeros(2 * n - 1, dtype=float)
    for _ in range(target):
        L = build_L(h, n)
        if reverse:
            vals = np.max(F[None, :] - L, axis=1)
        else:
            vals = np.max(F[None, :] + L, axis=1)
        mv = float(vals.min())
        opts = np.where(vals <= mv + 0.49)[0]
        p = int(opts[rng.randint(opts.size)])
        if reverse:
            F -= L[p]
            h[p] = 0.0
        else:
            F += L[p]
            h[p] = 1.0
        F = np.rint(F)
    return h


def binary_local_hill(h0, n, deadline, seed=0, max_kicks=5, early_count=-1e9):
    rng = np.random.RandomState(seed)
    h0 = np.asarray(h0, dtype=float)
    if (
        np.all((h0 >= 0.0) & (h0 <= 1.0))
        and np.allclose(h0, np.round(h0), atol=1e-8)
        and abs(float(h0.sum()) - n / 2.0) < 0.25
    ):
        h = np.round(h0).astype(float)
    else:
        h = binarize_scores(h0, n, rng, noise=0.0)
    F = np.rint(np.correlate(h, 1.0 - h, mode="full"))
    T = float(F.max())
    best_h = h.copy()
    best_T = T
    kicks = 0

    while time.time() < deadline:
        improved = False
        neutral = 0
        while time.time() < deadline:
            if best_T <= early_count:
                break
            zeros = np.where(h < 0.5)[0]
            ones = np.where(h >= 0.5)[0]
            if zeros.size == 0 or ones.size == 0:
                break
            L = build_L(h, n)
            Z = zeros.size
            O = ones.size
            Add = L[zeros]
            Rem = -L[ones]
            rows = np.arange(O)
            base = np.empty((O, 2 * n - 1), dtype=float)
            best_c = np.inf
            bi = -1
            bj = -1
            for a in range(Z):
                np.add(Rem, Add[a], out=base)
                base += F
                zi = int(zeros[a])
                idx1 = n - 1 + (ones - zi)
                idx2 = n - 1 + (zi - ones)
                base[rows, idx1] += 1.0
                base[rows, idx2] += 1.0
                cand = base.max(axis=1)
                jj = int(np.argmin(cand))
                if cand[jj] < best_c:
                    best_c = float(cand[jj])
                    bi = a
                    bj = jj
            if bi < 0:
                break
            accept = False
            if best_c < T - 1e-7:
                accept = True
            elif best_c <= T + 1e-7 and neutral < 300 and rng.rand() < 0.15:
                accept = True
                neutral += 1
            if not accept:
                break
            zi = int(zeros[bi])
            oj = int(ones[bj])
            F = F + Add[bi] + Rem[bj]
            idx1 = n - 1 + (oj - zi)
            idx2 = n - 1 + (zi - oj)
            F[idx1] += 1.0
            F[idx2] += 1.0
            F = np.rint(F)
            h[zi] = 1.0
            h[oj] = 0.0
            T = float(F.max())
            if T < best_T - 1e-7:
                best_T = T
                best_h = h.copy()
            improved = True
            if best_T <= early_count:
                break
        if time.time() >= deadline or best_T <= early_count:
            break
        if not improved:
            kicks += 1
            if kicks > max_kicks:
                break
            h = best_h.copy()
            kk = int(rng.randint(1, 4))
            for _ in range(kk):
                zs = np.where(h < 0.5)[0]
                os_ = np.where(h >= 0.5)[0]
                if zs.size == 0 or os_.size == 0:
                    break
                a = int(rng.choice(zs))
                b = int(rng.choice(os_))
                h[a], h[b] = h[b], h[a]
            F = np.rint(np.correlate(h, 1.0 - h, mode="full"))
            T = float(F.max())
            if T > best_T + 4.0:
                h = binarize_scores(best_h + 0.5 * rng.randn(n), n, rng, 0.05)
                F = np.rint(np.correlate(h, 1.0 - h, mode="full"))
                T = float(F.max())
    return best_h, best_T * (2.0 / n)


def smooth_gradient(h, n, w):
    g = np.zeros(n, dtype=float)
    o = 1.0 - h
    center = n - 1
    wc = float(w[center])
    if wc > 0.0:
        g += wc * (1.0 - 2.0 * h)
    nz = np.where(w > 1e-12)[0]
    for m in nz:
        mi = int(m)
        wi = float(w[mi])
        if wi == 0.0 or mi == center:
            continue
        if mi > center:
            d = mi - center
            s = n - d
            g[:s] += wi * o[d:]
            g[d:] -= wi * h[:s]
        else:
            d = center - mi
            s = n - d
            g[d:] += wi * o[:s]
            g[:s] -= wi * h[d:]
    return g


def continuous_polish(h0, n, deadline, seed=0):
    rng = np.random.RandomState(seed)
    h = project_feasible(h0, n)
    best_h = h.copy()
    c = np.correlate(h, 1.0 - h, mode="full")
    C = float(c.max())
    best_cnt = C
    if time.time() >= deadline:
        return best_h, best_cnt * (2.0 / n)
    lr = 0.05
    rem = max(1.0, deadline - time.time())
    taus = [(10.0, 0.15), (3.0, 0.20), (1.0, 0.25), (0.3, 0.15), (0.1, 0.12), (0.03, 0.13)]
    for tau, fr in taus:
        if time.time() >= deadline:
            break
        phase_end = min(deadline, time.time() + fr * rem)
        stuck = 0
        while time.time() < phase_end:
            c = np.correlate(h, 1.0 - h, mode="full")
            C = float(c.max())
            if C < best_cnt - 1e-12:
                best_cnt = C
                best_h = h.copy()
            z = (c - C) / tau
            z = np.minimum(z, 0.0)
            w = np.exp(z)
            s = float(w.sum())
            if s <= 1e-300 or not np.isfinite(s):
                idx = np.where(c >= C - tau)[0]
                w = np.zeros_like(c)
                if idx.size == 0:
                    idx = np.argmax(c)
                    w[idx] = 1.0
                else:
                    w[idx] = 1.0
                    w /= float(idx.size)
            else:
                w /= s
            grad = smooth_gradient(h, n, w)
            gm = float(np.max(np.abs(grad)))
            if gm < 1e-12:
                break
            step = -lr * grad
            sm = float(np.max(np.abs(step)))
            if sm > 0.12:
                step *= 0.12 / sm
            accepted = False
            for _ in range(12):
                h1 = project_feasible(h + step, n)
                cnt1 = float(np.max(np.correlate(h1, 1.0 - h1, mode="full")))
                if cnt1 < C - 1e-12:
                    h = h1
                    C = cnt1
                    if C < best_cnt:
                        best_cnt = C
                        best_h = h.copy()
                    lr = min(0.2, lr * 1.08)
                    accepted = True
                    break
                step *= 0.5
            if not accepted:
                best_tr = None
                best_tr_cnt = C
                dirn = grad.copy()
                dn = float(np.max(np.abs(dirn)))
                if dn > 0.0:
                    dirn *= -1.0 / dn
                for _ in range(5):
                    h1 = project_feasible(h + 0.025 * dirn + 0.003 * rng.randn(n), n)
                    cnt1 = float(np.max(np.correlate(h1, 1.0 - h1, mode="full")))
                    if cnt1 < best_tr_cnt:
                        best_tr_cnt = cnt1
                        best_tr = h1
                if best_tr is not None:
                    h = best_tr
                    C = best_tr_cnt
                    if C < best_cnt:
                        best_cnt = C
                        best_h = h.copy()
                lr = max(1e-5, lr * 0.6)
                stuck += 1
                if stuck > 120:
                    break
    best_h = project_feasible(best_h, n)
    cnt = max_count(best_h, n)
    if cnt < best_cnt:
        best_cnt = cnt
    return best_h, best_cnt * (2.0 / n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    hard = t0 + total - 4.0
    target_c = 0.3808 - 1e-12

    try:
        hp = np.asarray(initial_h_values, dtype=float)
    except Exception:
        hp = np.full(250, 0.5)
    hp, n = prepare_vector(hp)

    best_h = hp.copy()
    best_c = compute_c5(best_h, n)
    target_count = target_c * n / 2.0
    rng = np.random.RandomState(seed)

    if time.time() < hard and hard - time.time() > 80.0:
        end1 = min(t0 + 0.09 * total, hard - 150.0)
        if end1 > time.time() + 10.0:
            hc, cc = continuous_polish(hp, n, end1, seed=seed)
            if cc < best_c:
                best_c = cc
                best_h = hc.copy()

    bin_dead = min(t0 + 0.55 * total, hard - 160.0)
    if best_c >= target_c and time.time() < bin_dead:
        target_count = target_c * n / 2.0
        hpar = binarize_scores(hp, n, rng, 0.0)
        end = min(bin_dead, time.time() + 150.0)
        if end > time.time() + 10.0:
            hb, cb = binary_local_hill(hpar, n, end, seed + 1, 5, target_count)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

        if best_c >= target_c and time.time() + 70.0 < bin_dead:
            hg = greedy_construct(n, reverse=False, seed=seed + 3)
            end = min(bin_dead, time.time() + 110.0)
            hb, cb = binary_local_hill(hg, n, end, seed + 3, 5, target_count)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

        if best_c >= target_c and time.time() + 70.0 < bin_dead:
            hg = greedy_construct(n, reverse=True, seed=seed + 4)
            end = min(bin_dead, time.time() + 110.0)
            hb, cb = binary_local_hill(hg, n, end, seed + 4, 5, target_count)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

        if best_c >= target_c and time.time() + 40.0 < bin_dead:
           endimento = min(bin_dead, time.time() + 90.0)
            hinv = 1.0 - hp
            hb, cb = binary_local_hill(hinv, n, endimento, seed + 5, 5, target_count)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

        for r in range(5):
            if best_c < target_c or time.time() + 15.0 >= bin_dead:
                break
            hrand = random_binary(n, seed + 50 + r)
            end = min(bin_dead, time.time() + 30.0)
            hb, cb = binary_local_hill(hrand, n, end, seed + 50 + r, 4, target_count)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

        for r in range(3):
            if best_c < target_c or time.time() + 15.0 >= bin_dead:
                break
            hnoise = binarize_scores(hp + 0.25 * rng.randn(n), n, rng, 0.0)
            end = min(bin_dead, time.time() + 35.0)
            hb, cb = binary_local_hill(hnoise, n, end, seed + 100 + r, 4, target_count)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

        if best_c >= target_c and time.time() + 20.0 < bin_dead:
            hstart = binarize_scores(best_h, n, rng, 0.0)
            hb, cb = binary_local_hill(hstart, n, bin_dead, seed + 200, 4, target_c * n / 2.0)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

    if best_c >= target_c and time.time() < hard - 30.0:
        remaining = hard - time.time()
        end2 = min(hard - 60.0, time.time() + 0.65 * remaining)
        hc, cc = continuous_polish(best_h, n, end2, seed + 300)
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()
        if best_c >= target_c and time.time() < hard - 35.0:
            hc, cc = continuous_polish(hp, n, hard - 20.0, seed + 400)
            if cc < best_c:
                best_c = cc
                best_h = hc.copy()

    if best_c >= target_c and time.time() < hard - 220.0 and n < 350:
        n2 = n + 50
        if n2 % 2 == 1:
            n2 += 1
        if n2 <= 350 and n2 != n:
            xo = np.linspace(0.0, 2.0, n, endpoint=False)
            xn = np.linspace(0.0, 2.0, n2, endpoint=False)
            h2 = project_feasible(np.interp(xn, xo, best_h), n2)
            end = min(hard - 90.0, time.time() + 130.0)
            if end > time.time() + 20.0:
                h2c, c2 = continuous_polish(h2, n2, end, seed + 700)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2c.copy()
                    n = n2
                if best_c >= target_c and time.time() < hard - 45.0:
                    tc2 = target_c * n / 2.0
                    h2b = binarize_scores(best_h, n, rng, 0.0)
                    endb = min(hard - 15.0, time.time() + 45.0)
                    hb, cb = binary_local_hill(h2b, n, endb, seed + 800, 2, tc2)
                    if cb < best_c:
                        best_c = cb
                        best_h = hb.copy()

    if best_c >= target_c and time.time() < hard - 20.0:
        target_count = target_c * n / 2.0
        hstart = binarize_scores(best_h, n, rng, 0.0)
        hb, cb = binary_local_hill(hstart, n, hard - 8.0, seed + 900, 3, target_count)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        if best_c >= target_c and time.time() < hard - 4.0:
            hc, cc = continuous_polish(best_h, n, hard - 1.0, seed + 950)
            if cc < best_c:
                best_c = cc
                best_h = hc.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
