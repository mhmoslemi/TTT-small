But wait - the subgradient method for a max function is known to be slow. It converges at rate O(1/sqrt(T)) and can oscillate. A better approach might be to use the "averaged" subgradient (like in Nesterov's method) or to use the smooth approximation (log-sum-exp) gradient.

The gradient of the log-sum-exp approximation is:
grad_soft[i] = sum_k softmax(c_k) * ∂c_k/∂h[i]

This is a weighted average of the subgradients, with weights given by the softmax of the correlation values. When alpha is large, this concentrates on the max; when small, it's more diffuse.

This gradient is smooth and well-defined, so we can use better optimization methods (like L-BFGS or even simple gradient descent with backtracking line search).

Let me implement this:

```python
def smooth_gradient(h, n, alpha):
    """Compute gradient of the log-sum-exp approximation of C5."""
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * dx
    m = c.max()
    logits = alpha * (c - m)
    # Softmax weights
    logits_shift = logits - logits.max()
    w = np.exp(logits_shift)
    w /= w.sum()
    
    # Gradient: weighted sum of individual gradients
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        s = k - (n - 1)
        for i in range(n):
            g = 0.0
            ip = i + s
            im = i - s
            if 0 <= ip < n:
                g += (1.0 - h[ip])
            if 0 <= im < n:
                g -= h[im]
            grad[i] += w[k] * g * dx
    return grad, c.max() + np.log(w.sum()) / alpha
```

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            mask = np.ones(n, dtype=bool)
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def soft_max_grad(x, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(x, 1.0 - x, mode='full')
    c = c_raw * dx
    m = c.max()
    logits = alpha * (c - m)
    logits_shift = logits - logits.max()
    w = np.exp(logits_shift)
    ws = w.sum()
    if ws == 0:
        return np.zeros(n)
    w /= ws
    
    grad = np.zeros(n)
    nn = 2 * n - 1
    for k in range(nn):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        wk = w[k] * dx
        for i in range(n):
            g = 0.0
            ip = i + s
            im = i - s
            if 0 <= ip < n:
                g += (1.0 - x[ip])
            if 0 <= im < n:
                g -= x[im]
            if g != 0.0:
                grad[i] += wk * g
    return grad

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(soft_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_descent_smooth(h, n, deadline, alpha=5000):
    """Gradient descent on the smooth approximation with backtracking."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    lr = 0.001
    
    while time.time() < deadline:
        g = soft_max_grad(x, n, alpha)
        # Backtracking line search
        for contraction in range(20):
            t = lr / (2.0 ** contraction)
            x_new = np.clip(x - t * g, 0.0, 1.0)
            # Fix sum
            diff = n / 2.0 - x_new.sum()
            if abs(diff) > 1e-10:
                mask = (x_new > 0) & (x_new < 1)
                if mask.sum() > 0:
                    x_new[mask] += diff / mask.sum()
                    x_new = np.clip(x_new, 0.0, 1.0)
            c_new = compute_c5(x_new, n)
            c_old = compute_c5(x, n)
            if c_new < c_old - 1e-14:
                x = x_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = x.copy()
                break
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    
    return best_h, best_c

    m = c.max()

    ws = w.sum()
    w /= ws
    
    # For each shift k (0 to 2n-2), shift s = k - (n-1)
    # ∂c(s)/∂h[i] = (1-h[i+s]) - h[i-s] with bounds

    #   grad_s[i] = (1-h[i+s]) if 0<=i+s<n, plus -h[i-s] if 0<=i-s<n

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def soft_max_grad(x, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(x, 1.0 - x, mode='full')
    c = c_raw * dx
    m = c.max()
    logits = alpha * (c - m)
    logits_shift = logits - logits.max()
    w = np.exp(logits_shift)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        wk = w[k]
        if s > 0:
            grad[s:] += wk * (1.0 - x[:n - s])
            grad[:n - s] += wk * (-x[s:])
        elif s < 0:
            s2 = -s
            grad[:n - s2] += wk * (1.0 - x[s2:])
            grad[s2:] += wk * (-x[:n - s2])
        else:
            grad[:] += wk * (1.0 - 2.0 * x)
    grad *= dx
    return grad

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(soft_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_search(h, n, deadline, alpha=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    lr = 0.05
    stall = 0
    while time.time() < deadline:
        g = soft_max_grad(x, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g = g / gn
        improved = False
        for _ in range(30):
            xn = np.clip(x - lr * g, 0.0, 1.0)
            diff = n / 2.0 - xn.sum()
            if abs(diff) > 1e-10:
                mask = (xn > 0) & (xn < 1)
                if mask.sum() > 0:
                    xn[mask] += diff / mask.sum()
                    xn = np.clip(xn, 0.0, 1.0)
            cn = compute_c5(xn, n)
            co = compute_c5(x, n)
            if cn < co - 1e-15:
                x = xn
                if cn < best_c:
                    best_c = cn
                    best_h = x.copy()
                    stall = 0
                lr = min(lr * 1.05, 0.5)
                improved = True
                break
            else:
                lr *= 0.7
        if not improved:
            lr = 0.01
            stall += 1
            if stall > 200:
                break
    return best_h, best_c

def simulated_annealing(h, n, deadline, rng, t0=0.005, t_min=1e-7):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    T = t0
    cur = h.copy()
    n_iter = 0
    no_improve = 0
    while time.time() < deadline and T > t_min:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        scale = 0.005 + 0.05 * T / t0
        delta = scale * rng.uniform(0.3, 1.0)
        if rng.rand() < 0.5:
            if cur[i] + delta > 1.0 or cur[j] - delta < 0.0:
                continue
            cur[i] += delta
            cur[j] -= delta
            revert = (-delta, delta, i, j)
        else:
            if cur[i] - delta < 0.0 or cur[j] + delta > 1.0:
                continue
            cur[i] -= delta
            cur[j] += delta
            revert = (delta, -delta, i, j)
        new_c = compute_c5(cur, n)
        de = new_c - cur_c
        if de < 0 or rng.rand() < np.exp(-de / max(T, 1e-10)):
            cur_c = new_c
            n_iter += 1
            if new_c < best_c:
                best_c = new_c
                best_h = cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            did, dj, ii, jj = revert
            cur[ii] += did
            cur[jj] += dj
            no_improve += 1
        T *= 0.9997
        if no_improve > 50000:
            T = max(T * 5.0, t_min)
            no_improve = 0
            cur = best_h.copy()
            cur_c = best_c
    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
    return h, best_c
