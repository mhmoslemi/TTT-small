    dx = 2.0 / n
    target = n // 2
    h = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h[idx[:target]] = 1.0
    
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max()
    best_c = cur_c
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Precompute flip_in for all zeros
        # flip_in[j][k] = (1-h[j-k])*1_{0<=j-k<n} + (-h[j+k])*1_{0<=j+k<n}
        # For each k (0 to 2n-2), representing lag k-(n-1):
        #   term1: 1-h[j-(k-(n-1))] if 0<=j-(k-(n-1))<n
        #   term2: -h[j+(k-(n-1))] if 0<=j+(k-(n-1))<n

        # flip_in has shape (n_zeros, 2n-1)
        k_vals = np.arange(2*n-1) - (n-1)  # actual lag values: -(n-1) to n-1
        flip_in = np.zeros((len(zeros), 2*n-1))
        for idx_j, j in enumerate(zeros):
            # term1: 1-h[j-k] for valid j-k
            idx1 = j - k_vals
            mask1 = (idx1 >= 0) & (idx1 < n)
            flip_in[idx_j, mask1] += (1 - h[idx1[mask1]])
            # term2: -h[j+k] for valid j+k
            idx2 = j + k_vals
            mask2 = (idx2 >= 0) & (idx2 < n)
            flip_in[idx_j, mask2] += (-h[idx2[mask2]])
        
        # Similarly for flip_out for all ones
        flip_out = np.zeros((len(ones), 2*n-1))
        for idx_i, i in enumerate(ones):
            idx1 = i - k_vals
            mask1 = (idx1 >= 0) & (idx1 < n)
            flip_out[idx_i, mask1] += (h[idx1[mask1]] - 1)
            idx2 = i + k_vals
            mask2 = (idx2 >= 0) & (idx2 < n)
            flip_out[idx_i, mask2] += h[idx2[mask2]]
        
        # Now find best swap
        best_improvement = 0
        best_ii, best_jj = -1, -1
        
        for ii in range(len(ones)):
            # new_corr for all j when swapping ones[ii] and zeros[j]
            new_corrs = corr[None, :] + flip_out[ii][None, :] + flip_in  # (n_zeros, 2n-1)
            new_maxes = new_corrs.max(axis=1)
            j_best = new_maxes.argmin()
            if new_maxes[j_best] < cur_c - best_improvement:
                best_improvement = cur_c - new_maxes[j_best]
                best_ii = ii
                best_jj = j_best
        
        if best_ii == -1:
            break  # No improving swap found
        
        # Apply best swap
        i = ones[best_ii]
        j = zeros[best_jj]
        h[i] = 0
        h[j] = 1
        corr = corr + flip_out[best_ii] + flip_in[best_jj]
        cur_c = corr.max()
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        h = np.clip(h, 0.0, 1.0)
        if diff > 0:
            room = 1.0 - h

            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    # Use dx-scaled corr for the objective
    corr_x = corr * dx
    m = corr_x.max()
    sh = alpha * (corr_x - m)
    sm = sh.max()
    p = np.exp(sh - sm)
    p_sum = p.sum()
    p = p / p_sum  # softmax weights, shape (2n-1,)
    
    # Objective value
    obj = m + (sm + np.log(p_sum)) / alpha
    
    # Gradient: sum_k p_k * d(corr_k * dx)/dx_j
    # = dx * sum_k p_k * [(1-x[j-k]) * 1_{0<=j-k<n} - x[j+k] * 1_{0<=j+k<n}]
    # We can compute this as:
    # grad[j] = dx * (sum_m p[j-m] * (1-x[m]) - sum_m p[m-j] * x[m])
    # where the sums are over m=0..n-1
    
    # Using the padding trick:
    # p is indexed 0..2n-2, corresponding to lags -(n-1)..n-1
    # For grad[j]: first term = sum_{m=0}^{n-1} p[j-m+(n-1)] * (1-x[m])
    #              second term = sum_{m=0}^{n-1} p[m-j+(n-1)] * x[m]
    
    # This is a cross-correlation!
    # first[j] = (p conv (1-x))[j + n - 1] ... let me think
    # Actually: sum_m p[j-m+n-1] * (1-x[m]) for m=0..n-1
    # Let's define P as p (length 2n-1)
    # first[j] = sum_{m=0}^{n-1} P[j-m+n-1] * (1-x[m])
    # This is the cross-correlation of P and (1-x) evaluated at position j+n-1
    # np.correlate(P, 1-x, 'full')[j+n-1] = sum_n P[n+(j+n-1)] * (1-x)[n] ... no that's not right
    
    # Let me just do it directly for n=250
    grad = np.zeros(n)
    omx = 1 - x  # one minus x
    P = p  # length 2n-1
    offset = n - 1
    for j in range(n):
        # First term: sum_{m=0}^{n-1} P[j-m+offset] * omx[m]
        # j-m+offset ranges from j-offset to j+offset = j-n+1 to j+n-1
        # Since offset=n-1: ranges from j-(n-1) to j+(n-1)
        # All within [0, 2n-2] for j in [0, n-1] ✓
        indices1 = j + offset - np.arange(n)  # = j+n-1-m for m=0..n-1
        # indices1 goes from j+n-1 down to j
        # All in [0, 2n-2] ✓
        first = np.dot(P[indices1], omx)
        
        # Second term: sum_{m=0}^{n-1} P[m-j+offset] * x[m]
        # m-j+offset ranges from offset-j to n-1-j+offset = n-1-j to 2n-2-j
        # For j in [0,n-1]: range is [n-1-j, 2n-2-j], all in [0, 2n-2] ✓
        indices2 = np.arange(n) - j + offset  # = m-j+n-1 for m=0..n-1
        second = np.dot(P[indices2], x)
        
        grad[j] = dx * (first - second)
    
    return obj, grad

    target = n / 2.0

            if s < 1e-14:

    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    sm = sh.max()
    exp_sh = np.exp(sh - sm)
    p_sum = exp_sh.sum()
    p = exp_sh / p_sum
    obj = m + (sm + np.log(p_sum)) / alpha

    offset = n - 1
    omx = 1 - x
    grad = np.zeros(n)

        idx2 = np.arange(n) - j + offset
        grad[j] = dx * (np.dot(p[idx1], omx) - np.dot(p[idx2], x))
    return obj, grad

        mask1 = (idx1 >= 0) & (idx1 < n)
        t1 = np.where(mask1, 1.0 - h_pad[np.clip(idx1, 0, n - 1)], 0.0)
        idx2 = zeros[:, None] + k_range[None, :]
        mask2 = (idx2 >= 0) & (idx2 < n)
        t2 = np.where(mask2, -h_pad[np.clip(idx2, 0, n - 1)], 0.0)
        flip_in = t1 + t2
        idx1o = ones[:, None] - k_range[None, :]
        mask1o = (idx1o >= 0) & (idx1o < n)
        t1o = np.where(mask1o, h_pad[np.clip(idx1o, 0, n - 1)] - 1.0, 0.0)
        idx2o = ones[:, None] + k_range[None, :]
        mask2o = (idx2o >= 0) & (idx2o < n)
        t2o = np.where(mask2o, h_pad[np.clip(idx2o, 0, n - 1)], 0.0)
        flip_out = t1o + t2o
        # Find best swap (vectorized over all pairs)
        new_corrs = corr[None, None, :] + flip_out[:, None, :] + flip_in[None, :, :]
        new_maxes = new_corrs.max(axis=2)
        flat_idx = new_maxes.argmin()
        best_io, best_iz = np.unravel_index(flat_idx, new_maxes.shape)
        best_val = new_maxes[best_io, best_iz]
        if best_val >= cur_c - 1e-14:
            break
        i = ones[best_io]
        j = zeros[best_iz]
        h[i] = 0.0
        h[j] = 1.0
        h_pad[:] = h
        corr = corr + flip_out[best_io] + flip_in[best_iz]
        cur_c = corr.max()
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP with alpha annealing
    p1 = t0 + 0.08 * (budget_s - 5)
    alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000]
    for alpha in alphas:
        if time.time() > p1:
            break
        h = slsqp_optimize(best_h.copy(), n, alpha, 3000, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: Main binary search
    p2 = t0 + 0.55 * (budget_s - 5)
    hb, cb = vectorized_binary_search(best_h, n, p2, max_sweeps=50000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 3: SLSQP polish from binary
    p3 = t0 + 0.65 * (budget_s - 5)
    if time.time() < p3:
        h = best_h + 0.05 * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [50000, 200000, 1000000, 5000000]:
            if time.time() > p3:
                break
            h = slsqp_optimize(h, n, alpha, 2000, p3)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    # Phase 4: Binary restarts with perturbations
    p4 = deadline - 10
    for restart in range(20):
        if time.time() > p4:
            break
        if restart % 4 == 0:
            h_start = best_h.copy()
        else:
            h_start = best_h.copy()
            h_bin = np.zeros(n)
            h_bin[np.argsort(h_start)[::-1][:n // 2]] = 1.0
            for _ in range(rng.randint(2, 6)):
                oi = np.where(h_bin == 1)[0]
                zi = np.where(h_bin == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    h_bin[a], h_bin[b] = h_bin[b], h_bin[a]
            h_start = h_bin
        hb, cb = vectorized_binary_search(h_start, n, min(time.time() + 30, p4), max_sweeps=20000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Phase 5: Final SLSQP
    if time.time() < deadline - 3:
        h = slsqp_optimize(best_h.copy(), n, 10000000, 2000, deadline - 2)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
