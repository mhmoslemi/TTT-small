import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _conv(a, b):
    m = len(a) + len(b) - 1
    nfft = _next_pow2(m)
    fa = np.fft.rfft(a, nfft)
    fb = np.fft.rfft(b, nfft)
    return np.fft.irfft(fa * fb, nfft)[:m]


def corr_counts_fft(x):
    n = len(x)
    m = 2 * n - 1
    nfft = _next_pow2(m)
    hx = np.fft.rfft(x, nfft)
    sx = np.fft.rfft(1.0 - x, nfft)
    pos = np.fft.irfft(np.conj(hx) * sx, nfft)[:n]
    neg = np.fft.irfft(hx * np.conj(sx), nfft)[:n]
    c = np.empty(m, dtype=float)
    c[n - 1:] = pos
    c[:n - 1] = neg[1:][::-1]
    return c


def exact_score(x):
    n = len(x)
    return float(np.max(np.correlate(x, 1.0 - x, mode="full")) * (2.0 / n))


def project_feasible(x, n):
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(30):
        s = x.sum()
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0.0:
            room = 1.0 - x
            tot = room.sum()
            if tot <= 1e-14:
                break
            x = x + d * room / tot
        else:
            tot = x.sum()
            if tot <= 1e-14:
                break
            x = x + d * x / tot
    x = np.clip(x, 0.0, 1.0)
    d = target - x.sum()
    if abs(d) > 1e-10:
        if d > 0.0:
            order = np.argsort(1.0 - x)[::-1]
        else:
            order = np.argsort(x)[::-1]
        for idx in order:
            if abs(d) <= 1e-12:
                break
            if d > 0.0:
                add = min(1.0 - x[idx], d)
                if add > 0.0:
                    x[idx] += add
                    d -= add
            else:
                sub = min(x[idx], -d)
                if sub > 0.0:
                    x[idx] -= sub
                    d += sub
    return x


def softmax_value_weights(c, alpha):
    c = np.asarray(c, dtype=float)
    m = float(np.max(c))
    if alpha <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    z = alpha * (c - m)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    w = e / s
    val = m + float(np.log(s)) / alpha
    return val, w


def grad_from_weights(x, w, n, dx):
    if n <= 1:
        return np.zeros(n, dtype=float)
    m = 2 * n - 1
    w0 = float(w[n - 1])
    A = w[n:m]
    B = w[0:n - 1][::-1]
    K = np.zeros(n, dtype=float)
    K[1:] = A + B
    c1 = _conv(x, K)
    c2 = _conv(x[::-1], K)
    hterm = c1[:n] + c2[:n][::-1]
    s_plus = np.zeros(n, dtype=float)
    if n > 1:
        s_plus[:n - 1] = np.cumsum(A)[::-1]
    s_minus = np.zeros(n, dtype=float)
    if n > 1:
        s_minus[1:] = np.cumsum(B)
    raw = w0 * (1.0 - 2.0 * x) + s_plus + s_minus - hterm
    return dx * raw


def _lbfgs_soft_fun(x, n, alpha, mu):
    if not np.all(np.isfinite(x)):
        return 1e100
    dx = 2.0 / n
    c = corr_counts_fft(x) * dx
    val, w = softmax_value_weights(c, alpha)
    s = x.sum() - 0.5 * n
    return float(val + 0.5 * mu * s * s)


def _lbfgs_soft_grad(x, n, alpha, mu):
    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)
    dx = 2.0 / n
    c = corr_counts_fft(x) * dx
    val, w = softmax_value_weights(c, alpha)
    g = grad_from_weights(x, w, n, dx)
    s = x.sum() - 0.5 * n
    g += mu * s
    return g


def _topk_indices(c, k):
    m = len(c)
    k = int(k)
    if k <= 0:
        k = 1
    if k >= m:
        return np.arange(m)
    return np.argpartition(c, -k)[-k:]


def _topk_fun(x, n, k, mu):
    if not np.all(np.isfinite(x)):
        return 1e100
    dx = 2.0 / n
    c = corr_counts_fft(x) * dx
    idx = _topk_indices(c, k)
    val = float(np.mean(c[idx]))
    s = x.sum() - 0.5 * n
    return float(val + 0.5 * mu * s * s)


def _topk_grad(x, n, k, mu):
    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)
    dx = 2.0 / n
    c = corr_counts_fft(x) * dx
    idx = _topk_indices(c, k)
    w = np.zeros(2 * n - 1, dtype=float)
    w[idx] = 1.0 / float(len(idx))
    g = grad_from_weights(x, w, n, dx)
    s = x.sum() - 0.5 * n
    g += mu * s
    return g


def make_soft_stages(n, maxiter):
    stages = []
    params = (
        (4.0, 1.0),
        (8.0, 2.0),
        (16.0, 5.0),
        (32.0, 10.0),
        (64.0, 20.0),
        (128.0, 50.0),
        (256.0, 100.0),
        (512.0, 200.0),
        (1024.0, 500.0),
    )
    for a, mu in params:
        stages.append(((n, a, mu), maxiter))
    return stages


def make_topk_stages(n, maxiter):
    stages = []
    params = (
        (12, 2.0),
        (8, 5.0),
        (5, 10.0),
        (2, 30.0),
        (1, 60.0),
    )
    for k, mu in params:
        stages.append(((n, k, mu), maxiter))
    return stages


def _lbfgs_run(fun, grad, x, n, deadline, stages):
    best = project_feasible(x, n)
    best_c = exact_score(best)
    current = best
    bounds = [(0.0, 1.0)] * n
    for args, maxiter in stages:
        if time.time() > deadline - 2.0:
            break
        rem = deadline - time.time()
        if rem < 8.0:
            break
        cur_c = exact_score(current)
        it = int(maxiter * min(1.0, rem / 180.0))
        it = max(30, min(int(maxiter), it))
        try:
            res = minimize(
                fun,
                current.copy(),
                jac=grad,
                args=args,
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": it,
                    "ftol": 1e-12,
                    "gtol": 1e-10,
                    "maxfun": max(100, 20 * it),
                    "maxls": 40,
                },
            )
            if getattr(res, "x", None) is not None and np.all(np.isfinite(res.x)):
                cand = project_feasible(res.x, n)
                cand_c = exact_score(cand)
                if cand_c < best_c:
                    best = cand
                    best_c = cand_c
                if cand_c < cur_c + 0.002:
                    current = cand
                else:
                    current = best
        except Exception:
            current = best
    return best, best_c


def upsample_h(old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    y = np.interp(x_new, x_old, old)
    return project_feasible(y, n_new)


def start_golden(n):
    phi = 0.5 * (1.0 + np.sqrt(5.0))
    frac = (np.arange(n, dtype=float) * phi) % 1.0
    idx = np.argsort(frac)[-(n // 2):]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    if n % 2 == 1:
        h[0] = 0.5
    return project_feasible(h, n)


def start_alternating(n):
    h = ((np.arange(n) % 2) == 0).astype(float)
    if n % 2 == 1:
        h[0] = 0.5
    return project_feasible(h, n)


def start_fourier(n, rng):
    h = np.full(n, 0.5, dtype=float)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    for f in range(1, 6):
        amp = rng.uniform(0.05, 0.22)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(2.0 * np.pi * f * x / 2.0 + phase)
    return project_feasible(h, n)


def _select_low(vals, idx, k):
    if len(idx) == 0:
        return np.array([], dtype=int)
    if k >= len(idx):
        return idx.copy()
    return idx[np.argsort(vals)[:k]]


def _select_high(vals, idx, k):
    if len(idx) == 0:
        return np.array([], dtype=int)
    if k >= len(idx):
        return idx[np.argsort(vals)[::-1]].copy()
    return idx[np.argsort(vals)[-k:][::-1]]


def pair_polish(x, n, deadline, rng):
    best = project_feasible(x, n)
    best_c = exact_score(best)
    dx = 2.0 / n
    guide_k = 5
    inc_count = 35
    dec_count = 35
    max_trials = 250
    no_gain = 0

    while time.time() < deadline - 1.5:
        c = corr_counts_fft(best) * dx
        idx = _topk_indices(c, guide_k)
        w = np.zeros(2 * n - 1, dtype=float)
        w[idx] = 1.0 / float(len(idx))
        g = grad_from_weights(best, w, n, dx)

        inc = np.where(best < 1.0 - 1e-10)[0]
        dec = np.where(best > 1e-10)[0]
        if len(inc) == 0 or len(dec) == 0:
            break

        inc_sel = _select_low(g, inc, inc_count)
        dec_sel = _select_high(g, dec, dec_count)

        cand = []
        for i in inc_sel:
            mdi = 1.0 - best[i]
            if mdi <= 1e-10:
                continue
            gi = g[i]
            for j in dec_sel:
                mdj = best[j]
                if mdj <= 1e-10:
                    continue
                d = min(mdi, mdj)
                gain = (g[j] - gi) * d
                if gain > 1e-12:
                    cand.append((gain, i, j, d))

        cand.sort(reverse=True)
        improved = False
        trials = 0

        for gain, i, j, d in cand:
            if trials >= max_trials:
                break
            for frac in (1.0, 0.25, 0.05):
                dtry = d * frac
                if dtry <= 1e-9:
                    continue
                trials += 1
                test = best.copy()
                test[i] += dtry
                test[j] -= dtry
                np.clip(test, 0.0, 1.0, out=test)
                cval = exact_score(test)
                if cval < best_c - 1e-15:
                    new = project_feasible(test, n)
                    cnew = exact_score(new)
                    if cnew < best_c:
                        best = new
                        best_c = cnew
                        improved = True
                        break
                if trials >= max_trials:
                    break
            if improved:
                break

        if improved:
            no_gain = 0
            continue

        random_ok = False
        for _ in range(30):
            i = int(rng.choice(inc))
            j = int(rng.choice(dec))
            d = min(1.0 - best[i], best[j], 0.05)
            if d <= 1e-10:
                continue
            test = best.copy()
            test[i] += d
            test[j] -= d
            np.clip(test, 0.0, 1.0, out=test)
            if exact_score(test) < best_c:
                new = project_feasible(test, n)
                cnew = exact_score(new)
                if cnew < best_c:
                    best = new
                    best_c = cnew
                    random_ok = True
                    break

        if not random_ok:
            no_gain += 1
            if no_gain >= 4:
                break

    return best, best_c
