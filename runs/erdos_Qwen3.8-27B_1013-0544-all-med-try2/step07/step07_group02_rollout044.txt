    dx = 2.0 / n

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()

            break

            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
