This is:
sum_k b_k * exp(alpha*(a_k + b_k*x)) = 0

sum_k b_k * exp(v_k) * exp(u_k * x) = 0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def coord_descent(h, n, alpha, n_sweeps, deadline):
    """Coordinate descent on log-sum-exp smoothed objective."""
    c = compute_corr(h, n)  # scaled correlation values (length 2n-1)
    m_range = np.arange(-(n-1), n)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            # Compute d_k = change in c_k when h[i] goes from 0 to 1
            # d_k = (1-h[i+k]) - h[i-k] for valid indices
            h_i = h[i]
            d = np.zeros(2*n-1)
            # h[i+k] valid when 0 <= i+k < n, i.e., -i <= k < n-i
            # The shift k in our correlation corresponds to index n-1+k
            # d[n-1+k] = (1-h[i+k]) - h[i-k]
            
            # For shift s: d at position n-1+s
            # (1-h[i+s]) for i+s in [0,n-1]
            lo_s = max(0, -i)  # i+s >= 0 => s >= -i
            hi_s = min(n-1, n-1-i)  # i+s < n => s < n-i, so s <= n-1-i
            
            for s in range(lo_s, hi_s + 1):
                d[n-1+s] += (1.0 - h[i+s])
            
            # -h[i-s] for i-s in [0,n-1]
            lo_s2 = max(0, i-(n-1))  # i-s >= 0 => s <= i
            hi_s2 = min(n-1, i)  # i-s <= n-1 => s >= i-(n-1)
            # Wait: i-s in [0, n-1] means 0 <= i-s < n, so i-n+1 <= s <= i
            lo_s2 = max(0, i-n+1)
            hi_s2 = min(n-1, i)
            
            for s in range(lo_s2, hi_s2 + 1):
                d[n-1+s] -= h[i-s]
            
            # Now c_k(x) = c_k + (x - h_i) * d_k
            # Minimize logsumexp(alpha * c_k(x)) over x in [0, 1]
            # = logsumexp(alpha * c + alpha*(x-h_i)*d)
            # Let a = alpha * c, b = alpha * d
            # Minimize logsumexp(a + (x-h_i)*b)
            # Derivative: sum exp(a + (x-h_i)*b) * b / sum exp(a + (x-h_i)*b)
            # = weighted average of b's
            
            # Use bisection
            a = alpha * c
            b = alpha * d
            
            def f_prime(x):
                offset = (x - h_i) * b
                vals = a + offset
                max_v = vals.max()
                exps = np.exp(vals - max_v)
                return float(np.dot(b, exps) / exps.sum())
            
            # f is convex, f'(x) is monotonically increasing
            # Find where f'(x) = 0
            x_lo, x_hi = 0.0, 1.0
            fp_lo = f_prime(x_lo)
            fp_hi = f_prime(x_hi)

                x_star = x_hi

                    if fp_mid > 0:
                        x_hi = x_mid
                    else:
                        x_lo = x_mid
                x_star = 0.5 * (x_lo + x_hi)
            
            # Update
            delta = x_star - h_i
            if abs(delta) > 1e-15:
                c += delta * d
                h[i] = x_star
    
    # Project
    h = project_feasible(h, n)
    return h

from scipy.optimize import minimize
import time

def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):

        if abs(diff) < 1e-14:

            if s < 1e-15:
                break
            h += diff / s * h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def coord_descent_step(h, n, alpha, c, i):
    """Update coordinate i using bisection. Returns new h[i] and delta c."""
    h_i = h[i]
    # Compute d = change in c when h[i] goes from 0 to 1
    # d[n-1+s] = (1-h[i+s]) - h[i-s]
    d = np.zeros(2 * n - 1)
    
    # (1-h[i+s]) terms: valid when 0 <= i+s < n
    lo = max(0, -i)
    hi = min(n - 1, n - 1 - i)
    for s in range(lo, hi + 1):
        d[n - 1 + s] += (1.0 - h[i + s])
    
    # -h[i-s] terms: valid when 0 <= i-s < n
    lo2 = max(0, i - n + 1)
    hi2 = min(n - 1, i)
    for s in range(lo2, hi2 + 1):
        d[n - 1 + s] -= h[i - s]
    
    # Bisection to find optimal x in [0, 1]
    a = alpha * c
    b = alpha * d
    
    def fp(x):
        vals = a + (x - h_i) * b
        mv = vals.max()
        e = np.exp(vals - mv)
        return float(np.dot(b, e) / e.sum())
    
    fp_lo = fp(0.0)
    fp_hi = fp(1.0)
    if fp_lo >= 0:
        x_star = 0.0
    elif fp_hi <= 0:
        x_star = 1.0
    else:
        xlo, xhi = 0.0, 1.0
        for _ in range(50):
            xm = 0.5 * (xlo + xhi)
            if fp(xm) > 0:
                xhi = xm
            else:
                xlo = xm
        x_star = 0.5 * (xlo + xhi)
    
    return x_star, (x_star - h_i) * d

    c = compute_corr(h, n)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            h[i] = np.clip(h[i], 0.0, 1.0)
            x_star, dc = coord_descent_step(h, n, alpha, c, i)
            c += dc
            h[i] = x_star
    h = project_feasible(h, n)
    return h

def binary_local_search(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:

                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 1.0, 0.0
                # Recompute delta for the actual swap
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]  # h[i] is now 1, but we need old values
                # Actually, let's just recompute

                cur_c = corr.max() * dx
                ones_list.remove(best_j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(best_j)
                improved = True
                break
        if not improved:

        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
