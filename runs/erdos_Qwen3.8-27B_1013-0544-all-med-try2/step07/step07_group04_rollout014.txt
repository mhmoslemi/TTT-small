```python
def compute_grad(h, w, n):
    """Compute gradient of soft-max objective. w is the soft-max weight vector (length 2n-1)."""
    grad = np.zeros(n)
    for i in range(n):
        # Valid lags: 0 <= i+lag < n and 0 <= i-lag < n
        # i+lag in [0, n-1] => lag in [-i, n-1-i]
        # i-lag in [0, n-1] => lag in [i-(n-1), i]
        lag_min = max(-i, i - (n-1))
        lag_max = min(n-1-i, i)
        if lag_max < lag_min:
            continue
        lags = np.arange(lag_min, lag_max + 1)
        wk = w[lags + n - 1]
        grad[i] = np.sum(wk * (1 - h[i + lags] - h[i - lags]))
```

```python
    grad = np.zeros(n)
    lag = np.arange(-(n-1), n)
    for i in range(n):
        # First term: sum over valid lag of w[lag+n-1] * (1-h[i+lag])
        lag1 = np.arange(max(-(n-1), -i), min(n-1, n-1-i) + 1)
        if len(lag1) > 0:
            grad[i] += np.sum(w[lag1 + n - 1] * (1 - h[i + lag1]))
        # Second term: -sum over valid lag of w[lag+n-1] * h[i-lag]
        lag2 = np.arange(max(-(n-1), i-(n-1)), min(n-1, i) + 1)
        if len(lag2) > 0:
            grad[i] -= np.sum(w[lag2 + n - 1] * h[i - lag2])
    
    return grad, c, m
```
