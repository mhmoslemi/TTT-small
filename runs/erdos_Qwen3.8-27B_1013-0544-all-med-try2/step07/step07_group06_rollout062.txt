def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room

            if tot < 1e-15:

    return c.max() + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def grad_smooth_max(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def coord_descent(h, n, deadline, mu_init=500.0, t_n=50, max_sweeps=8000):
    h = h.copy()
    t_grid = np.linspace(0.0, 1.0, t_n)
    n_lags = 2 * n - 1
    target = n / 2.0
    zero_idx = n - 1
    f_cur = np.correlate(h, 1 - h, mode='full').copy()
    sum_h = h.sum()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    mu = mu_init
    rng = np.random.default_rng(42)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved_any = False
        order = rng.permutation(n)
        for i in order:
            hi = h[i]
            S = sum_h - hi
            a = np.zeros(n_lags)
            a[n - 1 - i:2 * n - 1 - i] += (1.0 - h)
            a[i:i + n] += -h[::-1]
            deltas = t_grid - hi
            F_check = f_cur[:, None] + a[:, None] * deltas[None, :]
            F_check[zero_idx, :] = f_cur[zero_idx] + deltas * (1.0 - t_grid - hi)
            g_vals = F_check.max(axis=0) + mu * (S + t_grid)
            best_t = t_grid[np.argmin(g_vals)]
            new_delta = best_t - hi
            if abs(new_delta) > 1e-10:
                f_cur += a * new_delta
                f_cur[zero_idx] -= new_delta * new_delta
                h[i] = best_t
                sum_h += new_delta
                improved_any = True
        sum_h = h.sum()
        if abs(sum_h - target) > 2.0:
            mu = min(mu * 2.0, 50000.0)
        elif abs(sum_h - target) < 0.05:
            mu = max(mu * 0.5, 50.0)
        if sweep % 10 == 0 or not improved_any:
            c_cur = compute_c5(h, n)
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
        if not improved_any:
            break
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def lbfgs_phase(h, n, deadline, max_iters=300):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 10.0), (5000, 100.0), (50000, 1000.0), (500000, 5000.0), (5000000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        def obj(x):
            return smooth_max_value(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
        def gr(x):
            g = grad_smooth_max(x, n, alpha)
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': max_iters * 20, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: Coordinate descent (primary method)
    t1 = min(t0 + budget_s * 0.55, deadline - 300)
    h_cd, c_cd = coord_descent(best_h, n_init, t1, mu_init=500.0, t_n=50, max_sweeps=10000)
    if c_cd < best_c:
        best_c, best_h = c_cd, h_cd
    # Phase 2: Finer coordinate descent refinement
    if time.time() < deadline - 200:
        t2 = min(time.time() + 120, deadline - 200)
        h_cd2, c_cd2 = coord_descent(best_h, n_init, t2, mu_init=2000.0, t_n=100, max_sweeps=5000)
        if c_cd2 < best_c:
            best_c, best_h = c_cd2, h_cd2
    # Phase 3: L-BFGS-B polish
    if time.time() < deadline - 120:
        t3 = min(time.time() + 80, deadline - 120)
        h_l, c_l = lbfgs_phase(best_h, n_init, t3, max_iters=200)
        if c_l < best_c:
            best_c, best_h = c_l, h_l
    # Phase 4: Pair search
    if time.time() < deadline - 40:
        t4 = deadline - 10
        best_h, best_c = pair_search(best_h, n_init, t4,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                     radius=40)
    # Phase 5: Try other n values briefly
    if time.time() < deadline - 30:
        for n_try in [200, 400]:
            if time.time() > deadline - 15:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            alloc = deadline - time.time() - 10
            if alloc < 20:
                break
            h_t, c_t = coord_descent(h_interp, n_try, time.time() + alloc * 0.6, mu_init=500, t_n=50, max_sweeps=2000)
            if c_t < best_c:
                c_t2, _ = compute_c5(h_t, n_try), 0
                best_c, best_h = c_t, h_t
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
