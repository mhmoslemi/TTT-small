```python
        valid_sum[j] = lam_cs[hi_idx] - lam_cs[lo_idx]
    
    return dx * (valid_sum - A - B)
```

Wait, there's an issue. The objective F(h) = max_k f_k(h) is what we want to minimize. The dual problem is max_λ min_h ∑_k λ_k f_k(h). The primal-dual algorithm alternates:
- Primal: minimize ∑_k λ_k f_k(h) (using gradient descent)
- Dual: maximize ∑_k λ_k f_k(h) over λ (using gradient ascent on the dual)

The dual update should be: λ_k ← λ_k * exp(η * f_k(h)) (multiplicative update) or λ ← softmax(f/τ).

But there's a subtlety: in the primal step, we're minimizing ∑_k λ_k f_k(h), not F(h) = max_k f_k(h). The primal-dual gap gives us the bound:
F(h) ≥ ∑_k λ_k f_k(h)

So the actual C₅ is always ≥ the surrogate ∑_k λ_k f_k(h). We need to track the actual C₅ = max_k f_k(h) separately.

Let me also think about whether the primal step should use the gradient of the surrogate or the subgradient of the max.

If we use the subgradient of F(h) = max_k f_k(h), the subgradient is ∇f_{k*}(h) where k* = argmax k. This is simpler but might be less smooth.

If we use the gradient of the surrogate ∑_k λ_k f_k(h), it's smoother but requires dual updates.

I think the surrogate approach is better because:
1. The gradient is smoother (weighted average of all subgradients)
2. The dual update naturally focuses on the "active" constraints
3. It's a standard primal-dual scheme

v ← β*v - α*grad

This is Nesterov-like acceleration.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient_fast(h, lam, dx):
    """Vectorized gradient computation."""
    n = len(h)
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(-(n-1), n)[None, :]
    h_idx_A = j_idx + k_idx
    h_idx_B = j_idx - k_idx
    mask_A = (h_idx_A >= 0) & (h_idx_A < n)
    mask_B = (h_idx_B >= 0) & (h_idx_B < n)
    H_A = np.where(mask_A, h[np.clip(h_idx_A, 0, n-1)] * mask_A, 0.0)
    H_B = np.where(mask_B, h[np.clip(h_idx_B, 0, n-1)] * mask_B, 0.0)
    A = H_A @ lam
    B = H_B @ lam
    valid_sum = mask_A.astype(np.float64) @ lam
    return dx * (valid_sum - A - B)

def primal_dual_step(h, n, lam, alpha, momentum_state, beta=0.9):
    """One primal-dual step."""
    dx = 2.0 / n
    grad = compute_gradient_fast(h, lam, dx)
    v = momentum_state[0]
    v_new = beta * v - alpha * grad
    h_new = h + v_new
    h_new = project_feasible(h_new, n)
    return h_new, v_new

def primal_dual_optimize(h, n, deadline, rng, max_iterations=5000):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    
    # Initialize dual
    c_all = compute_c5_all(h, n)
    temp = 0.05
    exponents = c_all / temp
    exponents -= exponents.max()
    lam = np.exp(exponents)
    lam /= lam.sum()
    
    alpha = 0.01
    v = np.zeros(n)
    stall = 0
    best_stall = 0
    iters = 0
    
    while time.time() < deadline and iters < max_iterations:
        iters += 1
        
        # Primal step with backtracking line search
        improved = False
        a = alpha
        for ls in range(30):
            h_trial, v_new = primal_dual_step(h, n, lam, a, [v])
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-12:
                best_c = c_trial
                h = h_trial
                v = v_new
                improved = True
                break
            a *= 0.6
            if a < 1e-8:
                break
        
        if not improved:
            stall += 1
            if stall > 100:
                # Reset momentum and try larger step
                v = np.zeros(n)
                alpha = min(0.05, alpha * 2)
                stall = 0
                if best_stall > 200:
                    break
        else:
            stall = 0
            alpha = min(0.05, alpha * 1.05)
        
        best_stall = max(best_stall, stall)
        
        # Dual update (every 5 iterations to save time)
        if iters % 5 == 0:
            c_all = compute_c5_all(h, n)
            exponents = c_all / temp
            exponents -= exponents.max()
            lam_new = np.exp(exponents)
            lam_new /= lam_new.sum()
            lam = 0.7 * lam + 0.3 * lam_new
            temp *= 0.995
        
        # Periodically re-concentrate dual on argmax
        if iters % 50 == 0:
            c_all = compute_c5_all(h, n)
            k_max = np.argmax(c_all)
            lam_focus = np.zeros(2*n-1)
            lam_focus[k_max] = 1.0
            lam = 0.5 * lam + 0.5 * lam_focus
    
    return h, best_c

def slsqp_refine(h, n, alpha_smooth, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        sh = alpha_smooth * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha_smooth)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha_smooth)
    
    def obj_grad(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        sh = alpha_smooth * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        # grad of sum_k w_k * c_k
        # c_k = sum_i h[i]*(1-h[i+k])*dx (approx)
        # Use the same gradient computation
        return compute_gradient_fast(x, w * (len(w)), dx)  # approximate
    
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or res.fun < obj(h):
            return project_feasible(res.x, n)
    except:
        pass
    return h

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    j_range = range(i+1, n)
                else:
                    lo = max(0, i-radius)
                    hi = min(n, i+radius+1)
                    j_range = range(lo, hi)
                for j in j_range:
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break
    return h, best_c

def optimize_h(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    
    # Phase 1: Primal-dual
    t1 = time.time()
    alloc_pd = min(300, (deadline - t1) * 0.4)
    h_pd, c_pd = primal_dual_optimize(h, n, t1 + alloc_pd, rng)
    if c_pd < best_c:
        best_c = c_pd
        h = h_pd
    
    # Phase 2: SLSQP with various smoothing
    for a in [1000, 5000, 20000, 100000]:
        if time.time() > deadline - 20:
            break
        ht = slsqp_refine(h, n, a, maxiter=1000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    
    # Phase 3: Pair search
    if time.time() < deadline - 5:
        if n <= 150:
            h, best_c = pair_search(h, n, deadline - 3,
```
