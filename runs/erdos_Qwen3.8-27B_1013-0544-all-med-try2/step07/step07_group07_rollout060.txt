def project_feasible(h, n):

            tot = room.sum()

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

```python
import numpy as np
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(h, n, alpha):
    """Compute smooth max objective and gradient."""
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    c = c_raw * dx
    m = c.max()
    sh = alpha * (c - m)
    if sh.max() > 500:
        e = np.exp(sh - sh.max())
    else:
        e = np.exp(sh)
    w = e / e.sum()
    val = m + np.log(e.sum()) / alpha
    
    # Gradient computation
    S_plus = np.zeros(n)
    cumw = np.concatenate(([0], np.cumsum(w)))
    for j in range(n):
        lo = max(0, n - 1 - j)
        hi = min(2 * n - 2, n - 1 + j)
        S_plus[j] = cumw[hi + 1] - cumw[lo]
    
    A = np.zeros(n)
    for i in range(2 * n - 1):
        lag = i - (n - 1)
        if lag >= 0:
            A[:n - lag] += w[i] * h[lag:]
        else:
            A[-lag:] += w[i] * h[:n + lag]
    
    B = np.zeros(n)
    for i in range(2 * n - 1):
        lag = (n - 1) - i
        if lag >= 0:
            B[:n - lag] += w[i] * h[lag:]
        else:
            B[-lag:] += w[i] * h[:n + lag]
    
    grad = dx * (S_plus - A - B)
    return val, grad

def adam_optimize(h, n, deadline, alphas):
    """Adam optimizer with adaptive alpha schedule."""
    h_curr = project_feasible(h.copy(), n)
    best_h = h_curr.copy()
    best_c = compute_c5(best_h, n)
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        m_v = np.zeros(n)
        v_v = np.zeros(n)
        t_step = 0
        lr = 0.01
        
        while time.time() < deadline - 5:
            t_step += 1
            val, grad = obj_and_grad(h_curr, n, alpha)
            m_v = 0.9 * m_v + 0.1 * grad
            v_v = 0.999 * v_v + 0.001 * grad**2
            m_hat = m_v / (1 - 0.9**t_step)
            v_hat = v_v / (1 - 0.999**t_step)
            h_curr = h_curr - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            h_curr = project_feasible(h_curr, n)
            
            c_curr = compute_c5(h_curr, n)
            if c_curr < best_c - 1e-12:
                best_c = c_curr
                best_h = h_curr.copy()
            
            # Reduce learning rate over time
            lr = 0.01 / (1 + 0.01 * t_step)
            
            if t_step > 5000:
                break
    return best_h, best_c
