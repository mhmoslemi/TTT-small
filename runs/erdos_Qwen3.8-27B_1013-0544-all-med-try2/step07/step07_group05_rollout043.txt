def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=50, tol=1e-13, order=None):
    h = project_feasible(h.copy(), n)
    c_curr = compute_corr(h)
    best_c = c_curr.max()
    dx = 2.0 / n
    s_range = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        total_change = 0.0
        indices = order if order is not None else range(n)
        for i0 in indices:
            valid1 = (i0 - s_range >= 0) & (i0 - s_range < n)
            valid2 = (i0 + s_range >= 0) & (i0 + s_range < n)
            a = np.zeros(2*n-1)
            a[valid1] += 1.0 - h[i0 - s_range[valid1]]
            a[valid2] -= h[i0 + s_range[valid2]]
            
            lo = -h[i0]
            hi = 1.0 - h[i0]
            
            # Grid search for coarse minimum
            grid = np.linspace(lo, hi, 64)
            vals = np.max(c_curr[None, :] + a[None, :] * grid[:, None], axis=1)
            idx = np.argmin(vals)
            d_best = grid[idx]
            
            # Bisection on subgradient
            for _ in range(40):
                f_at_d = c_curr + a * d_best
                max_val = f_at_d.max()
                active = f_at_d >= max_val - 1e-13
                min_slope = a[active].min()
                max_slope = a[active].max()
                if min_slope <= 0.0 and max_slope >= 0.0:
                    break
                elif max_slope < 0.0:
                    lo = d_best
                    d_best = (lo + hi) / 2.0
                else:
                    hi = d_best
                    d_best = (lo + hi) / 2.0
            
            new_val = np.clip(d_best + h[i0], 0.0, 1.0)
            delta = new_val - h[i0]
            if abs(delta) > 1e-15:
                h[i0] = new_val
                c_curr += a * delta
                total_change += abs(delta)
        
        cur_max = c_curr.max()
        if cur_max < best_c:
            best_c = cur_max
        if total_change < tol:
            break
    
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_smooth(h, n, alpha, maxiter=2000):

    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = compute_corr(x) * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        return m + np.log(np.sum(np.exp(sh))) / alpha

        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)

        # grad of max part + grad of log-sum-exp part
        # d/dx log(exp(alpha*(c-m))) ... this is tricky with the max
        # Let's just use numerical gradient or skip gradient
        return None
    
    def sum_con_eq(x):
        return x.sum() - n / 2.0
    
    def sum_con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': sum_con_eq, 'jac': sum_con_jac}]
    
    try:

def pair_search(h, n, deadline, deltas, radius=None):

    if radius is None:

    for d in deltas:

                    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=50, tol=1e-14):
    h = project_feasible(h.copy(), n)
    c_curr = compute_corr(h)
    s_range = np.arange(-(n - 1), n)
    ns = 2 * n - 1
    for sweep in range(max_sweeps):
        total_change = 0.0
        for i0 in range(n):
            v1 = (i0 - s_range >= 0) & (i0 - s_range < n)
            v2 = (i0 + s_range >= 0) & (i0 + s_range < n)
            a = np.zeros(ns)
            a[v1] += 1.0 - h[i0 - s_range[v1]]
            a[v2] -= h[i0 + s_range[v2]]
            lo = -h[i0]
            hi = 1.0 - h[i0]
            if hi - lo < 1e-15:
                continue
            grid = np.linspace(lo, hi, 64)
            vals = np.max(c_curr[None, :] + a[None, :] * grid[:, None], axis=1)
            idx = int(np.argmin(vals))
            d_best = grid[idx]
            for _ in range(50):
                f_at_d = c_curr + a * d_best
                mx = f_at_d.max()
                active = f_at_d >= mx - 1e-13
                mn_s = a[active].min()
                mx_s = a[active].max()
                if mn_s <= 0.0 and mx_s >= 0.0:
                    break
                if mx_s < 0.0:
                    lo = d_best
                    d_best = (lo + hi) / 2.0
                else:
                    hi = d_best
                    d_best = (lo + hi) / 2.0
            new_val = float(np.clip(d_best + h[i0], 0.0, 1.0))
            delta = new_val - h[i0]
            if abs(delta) > 1e-15:
                h[i0] = new_val
                c_curr += a * delta
                total_change += abs(delta)
        if sweep % 10 == 9:
            c_curr = compute_corr(h)
        if total_change < tol:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_smooth(x, n, alpha, maxiter=1500):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def obj(z):
        c = compute_corr(z) * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sc(z):
        return float(z.sum() - n / 2.0)
    def scj(z):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(obj, x, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return x.copy()

def pair_search_fast(h, n, deadline, deltas, radius=None):
    h = h.copy()
    c_curr = compute_corr(h)
    best_c = c_curr.max()
    dx = 2.0 / n
    s_range = np.arange(-(n - 1), n)
    ns = 2 * n - 1
    if radius is None:
        radius = n // 2
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c * dx
                v1i = (i - s_range >= 0) & (i - s_range < n)
                v2i = (i + s_range >= 0) & (i + s_range < n)
                a_i = np.zeros(ns)
                a_i[v1i] += 1.0 - h[i - s_range[v1i]]
                a_i[v2i] -= h[i + s_range[v2i]]
                lo_j = max(0, i - radius)
                hi_j = min(n, i + radius + 1)
                for j in range(lo_j, hi_j):
                    if i == j:
                        continue
                    v1j = (j - s_range >= 0) & (j - s_range < n)
                    v2j = (j + s_range >= 0) & (j + s_range < n)
                    a_j = np.zeros(ns)
                    a_j[v1j] += 1.0 - h[j - s_range[v1j]]
                    a_j[v2j] -= h[j + s_range[v2j]]
                    delta_c = d * (a_i - a_j)
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        c_curr += delta_c
                        c = c_curr.max()
                        if c < best_c - 1e-14:
                            best_c = c
                            h[i] += d
                            h[j] -= d
                            improved = True
                        else:
                            c_curr -= delta_c
                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        c_curr -= delta_c
                        c = c_curr.max()
                        if c < best_c - 1e-14:
                            best_c = c
                            h[i] -= d
                            h[j] += d
                            improved = True
                        else:
                            c_curr += delta_c
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def optimize_block(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    if time.time() < deadline - 5:
        h, c = coord_descent(h, n, max_sweeps=40)
        if c < best_c:
            best_c = c
    for alpha in [2000, 10000, 50000, 200000]:
        if time.time() > deadline - 8:
            break
        ht = slsqp_smooth(h, n, alpha, maxiter=1200)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
        if time.time() < deadline - 5:
            h, c2 = coord_descent(h, n, max_sweeps=20)
            if c2 < best_c:
                best_c = c2
    if time.time() < deadline - 4:
        ht = slsqp_smooth(h, n, 800000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
        h, c2 = coord_descent(h, n, max_sweeps=15)
        if c2 < best_c:
            best_c = c2
            h = ht
    return h, best_c

    t0 = time.time()

    deadline = t0 + budget_s - 5
