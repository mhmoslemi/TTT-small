```python
def compute_B(h, n):
    s = np.arange(-(n-1), n)  # shape (2n-1,)
    i = np.arange(n)  # shape (n,)
    S = s[:, None]  # (2n-1, 1)
    I = i[None, :]  # (1, n)
    # B[k, i] = (1-h[i-s[k]]) * [0<=i-s[k]<n] - h[i+s[k]] * [0<=i+s[k]<n]
    idx1 = I - S  # (2n-1, n), value is i - s[k]
    idx2 = I + S  # (2n-1, n), value is i + s[k]
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    B = np.where(mask1, 1.0 - h[idx1], 0.0) - np.where(mask2, h[idx2], 0.0)
    return B
```

```python
def project_feasible(h, n):
    target = n / 2.0

        if abs(diff) < 1e-14:

            s = room.sum()

def compute_B(h, n):

    s = np.arange(-(n-1), n)

def slp_optimize(h, n, deadline, max_iter=200, init_radius=0.5):
    """Sequential Linear Programming optimization"""
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * (2.0/n)
    best_h = h.copy()
    best_c = cur_c
    radius = init_radius
    stagnant = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 20 == 0:
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            c = corr.max() * (2.0/n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                stagnant = 0
            elif c >= best_c - 1e-10:
                stagnant += 1
                if stagnant > 3:
                    radius = max(0.05, radius * 0.7)
            else:
                stagnant = 0
        
        h_new, step = slp_step(h, n, corr, radius)
        if step < 1e-6:
            break
        # Accept or reject
        h = h_new
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
        c = corr.max() * (2.0/n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            # Increase radius if improving
            radius = min(1.0, radius * 1.1)
        elif c > best_c + 1e-8:
            # Rejected, reduce radius
            radius = max(0.05, radius * 0.5)
            h = best_h  # revert
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
    
    return best_h, best_c

def coord_descent(h, n, max_sweeps=200, deadline=None):
    """Coordinate descent with exact 1D minimax solution"""
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            hi_v = np.max(corr - B * h[i]) + 1e-12
            lo_v = np.min(np.maximum(corr - B * h[i], corr + B * (1.0 - h[i]))) - 1e-12
            # Binary search for optimal t
            for _ in range(50):
                mid = (lo_v + hi_v) / 2.0
                lb_x = h[i]
                ub_x = h[i]
                pos_B = B > 1e-14
                neg_B = B < -1e-14
                zero_B = ~(pos_B | neg_B)
                if zero_B.any() and corr[zero_B].max() > mid + 1e-14:
                    lo_v = mid
                    continue
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= 1.0 and ub_x >= 0.0:
                    hi_v = mid
                else:
                    lo_v = mid
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-14:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
```

```python
def slp_step_linprog(h, n, corr, radius):
    dx = 2.0 / n
    B = compute_B(h, n)  # (2n-1, n)
    
    # Variables: x = [h_0, ..., h_{n-1}, t]

    c[n] = 1.0  # coefficient of t

    A2 = np.hstack([np.eye(n), np.zeros((n, 1))])
    b2 = h + radius
    
    # 3) Trust region: -h_var[i] <= -h[i] + radius (n constraints)
    A3 = np.hstack([-np.eye(n), np.zeros((n, 1))])
    b3 = -h + radius
    
    A_ub = np.vstack([A1, A2, A3])
    b_ub = np.concatenate([b1, b2, b3])
    
    # Equality: sum(h_var) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = [(0, 1)] * n + [(0, None)]

    h_new = res.x[:n]
    step = np.max(np.abs(h_new - h))
    return h_new, step
```

```python
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
```
