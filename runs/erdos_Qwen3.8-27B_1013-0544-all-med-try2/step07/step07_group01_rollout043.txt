    best_c = corr.max() * dx
    m = np.arange(2 * n - 1)
    k = m - (n - 1)

            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
