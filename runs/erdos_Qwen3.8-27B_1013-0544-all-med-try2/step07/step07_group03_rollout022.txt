    shifted = alpha * (corr - m)
    sm = shifted.max()

    logsum = sm + np.log(exp_s.sum())

    # offsets[k] = k - (n-1) for k = 0..2n-2
    # grad[j] = sum_k p[k] * [(1-h[j+off_k])*1_{valid} - h[j-off_k]*1_{valid}]

    # mask1[j, k] = 1 if 0 <= j + offsets[k] < n
    # mask2[j, k] = 1 if 0 <= j - offsets[k] < n
    j_idx = np.arange(n)[:, None]  # (n, 1)
    offsets = np.arange(-(n-1), n)[None, :]  # (1, 2n-1)
    
    mask1 = (j_idx + offsets >= 0) & (j_idx + offsets < n)  # (n, 2n-1)
    mask2 = (j_idx - offsets >= 0) & (j_idx - offsets < n)  # (n, 2n-1)
    
    # First term: for each j, sum_k p[k] * (1-h[j+offset_k]) * mask1[j,k]
    # h at positions j+offset_k: this is a gather operation
    # indices1[j, k] = j + offsets[k] where mask1[j,k]
    idx1 = j_idx + offsets  # (n, 2n-1)
    # Clip to valid range for safe indexing
    idx1_clipped = np.clip(idx1, 0, n-1)
    h_at_1 = h[idx1_clipped]  # (n, 2n-1)
    term1 = (mask1 * (1.0 - h_at_1) * p[None, :]).sum(axis=1)
    
    # Second term: for each j, sum_k p[k] * h[j-offset_k] * mask2[j,k]

    h_at_2 = h[idx2_clipped]
    term2 = (mask2 * h_at_2 * p[None, :]).sum(axis=1)
    
    grad = (term1 - term2) * dx
    
    if penalty > 0:
        s = h.sum() - n / 2.0

            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 3: SLSQP for tighter constraints at high alpha
    p3 = t0 + 0.70 * total
    for alpha in [10000, 50000, 200000]:
        if time.time() > p3 - 10:
            break
        h = slsqp_opt(best_h, n, alpha, 3000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Binary annealing
    p4 = t0 + 0.85 * total
    ha, ca = anneal_binary(best_h, n, p4, seed=seed + 1)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()

    # Phase 5: Final polish with SLSQP + L-BFGS-B
    for alpha in [50000, 200000, 1000000]:
        if time.time() > dl - 5:
            break
        h = slsqp_opt(best_h, n, alpha, 2000, 1e-17)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if time.time() > dl - 3:
            break
        h = lbgfs_opt(h, n, alpha, 2000.0, 5000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Try a different resolution
    if time.time() < dl - 30:
        nh = n + 50
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        for alpha in [10000, 50000, 200000]:
            if time.time() > dl - 10:
                break
            hh = lbgfs_opt(hh, nh, alpha, 1000.0, 5000)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
