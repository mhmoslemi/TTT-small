    ds = si - sj  # change in S per unit delta

            best_delta = d

    C = compute_c5_vec(h, n)
    G = compute_grad_matrix(h, n)
    dx = 2.0 / n
    D = dx * G
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        C = compute_c5_vec(h, n)
        alpha = 500.0
        sh = alpha * (C - C.max())
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= (w.sum() + 1e-30)
        grad = D.T @ w
        order = np.argsort(-np.abs(grad))
        improved = False
        M = min(top_m, n)
        for ii in range(M):
            i = order[ii]
            if h[i] < 1e-8 or h[i] > 1 - 1e-8:
                continue
            lo_j = max(0, i - radius)
            hi_j = min(n, i + radius + 1)
            if hi_j - lo_j < 2:
                lo_j, hi_j = 0, n
            idx = np.arange(lo_j, hi_j)
            idx = idx[idx != i]
            if len(idx) == 0:
                continue
            if grad[i] > 0:
                j = idx[np.argmin(grad[idx])]
            else:
                j = idx[np.argmax(grad[idx])]
            delta = line_search_pair(C, h, n, i, j, D)
            if abs(delta) > 1e-12:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    C = compute_c5_vec(h, n)
                    G = compute_grad_matrix(h, n)
                    D = dx * G
                    improved = True
                    no_improve = 0
                else:
                    h[i] -= delta
                    h[j] += delta
                    h = np.clip(h, 0.0, 1.0)
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
    return h, best_c

def binary_push(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(100):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            if 0.005 < h[i] < 0.995:
                continue
            target = 0.0 if h[i] < 0.5 else 1.0
            delta = target - h[i]
            if abs(delta) < 1e-12:
                continue
            mask = (np.abs(h - 0.5) > abs(delta)) & (h > 0) & (h < 1)
            mask[i] = False
            cnt = mask.sum()
            if cnt == 0:
                continue
            h_new = h.copy()
            h_new[i] = target
            h_new[mask] -= delta / cnt
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c - 1e-15:
                best_c = c
                h = h_new
                improved = True
        if not improved:
            break
    return h, best_c

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in [(2000, 10.0), (20000, 50.0), (200000, 200.0)]:
        if time.time() > deadline - 1:
            break
        def obj(x):
            return smooth_max_val(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2
        def gr(x):
            g = grad_smooth(x, n, alpha)
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': max_iters * 15, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: L-BFGS baseline
    t1 = min(t0 + 80, deadline - 600)
    if t1 > t0 + 10:
        best_h, best_c = lbfgs_phase(best_h, n, t1, max_iters=150)
    # Phase 2: Adam global search
    if time.time() < deadline - 500:
        t2 = min(time.time() + 150, deadline - 500)
        h_a, c_a = adam_phase(best_h, n, t2, alpha_init=30, alpha_end=50000, lr0=0.02)
        if c_a < best_c:
            best_c, best_h = c_a, h_a
    # Phase 3: Line-search pair descent (main phase)
    if time.time() < deadline - 300:
        t3 = min(time.time() + 400, deadline - 300)
        best_h, best_c = pair_descent(best_h, n, t3, max_sweeps=50000, radius=80, top_m=200)
    # Phase 4: Binary push
    if time.time() < deadline - 150:
        t4 = min(time.time() + 80, deadline - 150)
        h_b, c_b = binary_push(best_h, n, t4)
        if c_b < best_c:
            best_c, best_h = c_b, h_b
    # Phase 5: More pair descent after binary push
    if time.time() < deadline - 100:
        t5 = min(time.time() + 120, deadline - 100)
        best_h, best_c = pair_descent(best_h, n, t5, max_sweeps=50000, radius=120, top_m=250)
    # Phase 6: Fine pair descent with full range
    if time.time() < deadline - 50:
        t6 = min(time.time() + 50, deadline - 50)
        best_h, best_c = pair_descent(best_h, n, t6, max_sweeps=50000, radius=n, top_m=n)
    # Phase 7: Final L-BFGS polish
    if time.time() < deadline - 10:
        best_h, best_c = lbfgs_phase(best_h, n, deadline - 3, max_iters=100)
    # Final check
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
