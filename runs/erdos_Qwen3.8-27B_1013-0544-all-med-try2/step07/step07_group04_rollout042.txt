```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)
```

For the projection, the simple iterative approach might not be optimal. A better approach for projecting onto {0≤h≤1, sum(h)=t} is:

1. Start with h_clipped = clip(h, 0, 1)
2. If sum(h_clipped) = t, done.
3. If sum(h_clipped) > t: need to reduce. Find threshold τ such that sum(max(0, min(1, h - τ))) = t. This is a 1D root-finding problem.
4. If sum(h_clipped) < t: need to increase. Find threshold τ such that sum(max(0, min(1, h - τ))) = t.

Actually, the exact projection onto {0≤h≤1, sum(h)=t} is:
- Sort h values
- Find the threshold τ such that sum(clip(h - τ, 0, 1)) = t
- The projection is clip(h - τ, 0, 1)

This can be done with a binary search on τ.

Let me implement this:

```python
def project_box_sum(h, t):
    """Project h onto {0<=h<=1, sum(h)=t}"""
    # Binary search on the regularization parameter
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(h, t):
    """Project h onto {0<=h<=1, sum(h)=t} using binary search on threshold."""
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max(c, alpha):
    """Compute smooth max approximation and its gradient weights."""
    shifted = alpha * c - alpha * c.max()
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    val = c.max() + np.log(exp_vals.sum()) / alpha
    return val, w

def compute_subgrad(h, w, n):
    """Compute subgradient of smooth max objective.
    w: weights over shifts, length 2n-1, indexed by m-(n-1) for m in [-(n-1), n-1]
    Returns gradient of sum_m w[m] * c_m (without dx factor).
    """
    grad = np.zeros(n)
    omh = 1.0 - h
    for m in range(-(n-1), n):
        wm = w[m + n - 1]
        if wm < 1e-14:
            continue
        # Term 1: wm * (1-h[i+m]) for valid i+m
        if m >= 0:
            if m < n:
                grad[:n-m] += wm * omh[m:]
        else:
            if -m < n:
                grad[-m:] += wm * omh[:n+m]
        # Term 2: -wm * h[i-m] for valid i-m
        if m >= 0:
            if m < n:
                grad[m:] -= wm * h[:n-m]
        else:
            if -m < n:
                grad[:n+m] -= wm * h[-m:]
    return grad

def subgradient_descent(h, n, alpha, lr, n_iter, deadline):
    """Run subgradient descent with projection."""
    h = h.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_shifted = c - c.max()
        w = np.exp(alpha * c_shifted)
        w /= w.sum()
        grad = compute_subgrad(h, w, n)
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, n / 2.0)
        h = h_new
    return h

    """Simulated annealing for binary h."""

    target = n // 2

    idx = np.argsort(h0)[::-1]

    cur_c = corr.max()
    dx = 2.0 / n
    
    best_h = h.copy()
    best_c = cur_c
    
    # Temperature schedule: geometric cooling
    T_start = 0.5
    T_end = 0.01
    # We'll do steps until deadline
    
    t = time.time()
    elapsed_first = deadline - t
    # Estimate steps we can do
    # Each step: pick 2 random indices, swap, compute new corr (numpy correlate ~5μs for n=250)
    # Plus overhead ~5μs, so ~10μs per step

    if total_est < 1000:
        total_est = 1000
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() < deadline:
        # Temperature
        progress = step / total_est
        T = T_start * (T_end / T_start) ** progress
        
        # Pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        # Try swap
        h[i], h[j] = 1.0, 0.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_c = corr.max()
        delta = new_c - cur_c
        
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            # Accept

            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # Reject
            h[i], h[j] = 0.0, 1.0
        
        step += 1
        if step % 10000 == 0 and rng.rand() < 0.01:
            # Periodically recompute (shouldn't need to, but safety)
            pass
    
    return best_h, best_c * dx
```

```python
    omh = 1.0 - h
    for m in range(-(n - 1), n):

    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
```
