def binary_push(h, n, deadline):
    """Push values towards 0 or 1."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(100):
        if time.time() > deadline:
            break
        improved = False
        # Sort by "distance from binary"
        dists = np.minimum(h, 1 - h)
        order = np.argsort(dists)  # closest to binary first
        for i in order:
            if time.time() > deadline:
                return h, best_c
            if 0.05 < h[i] < 0.95:
                continue
            target = 0.0 if h[i] < 0.5 else 1.0
            delta = target - h[i]
            if abs(delta) < 1e-10:
                continue
            # Find where to put the mass
            mask = (h > abs(delta)) & (h < 1.0 - abs(delta))
            mask[i] = False
            if not mask.any():
                mask = (h > 0) & (h < 1)
                mask[i] = False
                if not mask.any():
                    continue
            h_new = h.copy()
            h_new[i] = target
            h_new[mask] -= delta / mask.sum()
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c - 1e-15:
                best_c = c
                h = h_new
                improved = True
        if not improved:
            break
    return h, best_c

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)

    dx = 2.0 / n

            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def lbfgs_optimize(h, n, deadline, alphas=(50, 200, 1000, 5000, 50000, 200000, 1000000)):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        remaining = deadline - time.time()
        maxit = max(50, min(500, int(remaining * 2)))
        mu = 1.0  # small penalty for constraint (the projection handles it)
        def obj(x):
            return smooth_max_value(x, n, alpha)
        def gr(x):
            return grad_smooth_max(x, n, alpha)
        try:
            res = minimize(obj, h, jac=gr, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxit * 30, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def coordinate_descent(h, n, deadline):
    """Direct coordinate descent targeting the worst shift."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    n_full = 2 * n - 1
    
    for iteration in range(2000):
        if time.time() > deadline:
            break
        
        c = compute_c5_vec(h, n)
        k_star = c.argmax() - (n - 1)
        
        # Gradient of the worst shift f_{k_star}
        g = np.zeros(n)
        if k_star >= 0:
            nv = n - k_star
            g[0:nv] = (1.0 - h[k_star:n])
            g[k_star:n] = -h[0:nv]
        else:
            kk = -k_star
            nv = n - kk
            g[kk:n] = (1.0 - h[0:nv])
            g[0:nv] = -h[kk:n]
        g *= dx
        
        # Find the best coordinate to decrease (most positive gradient = most to lose)
        # and the best to increase (most negative gradient = most to gain)
        # Subject to: we need to keep sum(h) = n/2
        
        # Simple: find i with max g[i] (decrease h[i]) and j with min g[j] (increase h[j])
        i_dec = np.argmax(g)
        j_inc = np.argmin(g)
        
        if i_dec == j_inc:
            break
        
        # Try to move mass from i_dec to j_inc
        # We want to decrease h[i_dec] and increase h[j_inc]
        # Try different amounts
        possible = min(h[i_dec], 1.0 - h[j_inc])
        if possible < 1e-10:
            # Try other pairs
            sorted_g = np.argsort(g)
            found = False
            for j_inc2 in sorted_g[:10]:
                for i_dec2 in sorted_g[-10:]:
                    if i_dec2 == j_inc2:
                        continue
                    possible = min(h[i_dec2], 1.0 - h[j_inc2])
                    if possible < 1e-10:
                        continue
                    for frac in [1.0, 0.5, 0.25, 0.1]:
                        d = possible * frac
                        h_try = h.copy()
                        h_try[i_dec2] -= d
                        h_try[j_inc2] += d
                        h_try = project_feasible(h_try, n)
                        c_try = compute_c5(h_try, n)
                        if c_try < best_c - 1e-15:
                            best_c = c_try
                            h = h_try
                            found = True
                            break
                    if found:
                        break
                if found:
                    break
            if not found:
                break
            continue
        
        moved = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            d = possible * frac
            h_try = h.copy()
            h_try[i_dec] -= d
            h_try[j_inc] += d
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:
                best_c = c_try
                h = h_try
                moved = True
                break
        if not moved:
            # Try gradient step
            for lr in [0.1, 0.05, 0.02, 0.01]:
                h_try = project_feasible(h - lr * g, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-15:
                    best_c = c_try
                    h = h_try
                    moved = True
                    break
            if not moved:
                break
    
    return h, best_c

def adam_optimize(h, n, deadline, alpha_init=30.0, alpha_final=500000.0, lr_init=0.01):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    max_steps = 10000
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        t += 1
        frac = min(t / max_steps, 1.0)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        lr_cur = lr_init * (1.0 + 0.5 * t) ** (-0.5)
        g = grad_smooth_max(h, n, alpha)
        m_arr = beta1 * m_arr + (1 - beta1) * g
        v_arr = beta2 * v_arr + (1 - beta2) * g ** 2
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        step_vec = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step_vec, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=None, grad_guided=False, alpha=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            if grad_guided:
                g = grad_smooth_max(h, n, alpha)
                order = np.argsort(-np.abs(g))
            else:
                order = np.arange(n)
            for i in order:
                if time.time() > deadline:
                    return h, best_c
