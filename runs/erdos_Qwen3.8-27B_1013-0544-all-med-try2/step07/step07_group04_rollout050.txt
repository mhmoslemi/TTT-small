import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def get_corr_delta_two(h, n, i, j):
    """Efficient computation of correlation change when h[i]:0->1, h[j]:1->0."""
    m_range = np.arange(-(n-1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def binary_sa(h_init, n, deadline, t0_init=0.02, cool_rate=0.9997, max_rounds=200, seed=42):
    """Simulated annealing for binary local search."""
    rng = np.random.RandomState(seed)
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    T = t0_init
    round_num = 0
    
    while time.time() < deadline and T > 1e-6:
        round_num += 1
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)
        
        delta = get_corr_delta_two(h, n, i, j)
        new_corr = corr + delta
        new_c = new_corr.max() * dx
        
        if new_c < cur_c:
            # Accept (improvement)
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = new_c
            ones = np.delete(ones, np.where(ones == j)[0][0])
            zeros = np.delete(zeros, np.where(zeros == i)[0][0])
            ones = np.append(ones, i)
            zeros = np.append(zeros, j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # Maybe accept (worse)
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                h[i], h[j] = 1.0, 0.0
                corr = new_corr
                cur_c = new_c
                ones = np.delete(ones, np.where(ones == j)[0][0])
                zeros = np.delete(zeros, np.where(zeros == i)[0][0])
                ones = np.append(ones, i)
                zeros = np.append(zeros, j)
        
        T *= cool_rate
    
    return best_h, best_c

def binary_greedy(h_init, n, deadline, max_sweeps=500, seed=42):
    """Greedy binary local search with first-improvement."""
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    m_range = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_list = sorted(zeros)
        for i in z_list:
            if time.time() > deadline:
                break
            o_list = sorted(ones)
            for j in o_list:
                delta = get_corr_delta_two(h, n, i, j)
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def get_corr_delta_two(h, n, i, j):
    m_range = np.arange(-(n-1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def binary_sa(h_init, n, deadline, t0_init=0.015, cool_rate=0.9998, seed=42):
    rng = np.random.RandomState(seed)
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    T = t0_init
    while time.time() < deadline and T > 1e-7:
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = zeros_list[rng.randint(len(zeros_list))]
        j = ones_list[rng.randint(len(ones_list))]
        delta = get_corr_delta_two(h, n, i, j)
        new_corr = corr + delta
        new_c = new_corr.max() * dx
        if new_c < cur_c:
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = new_c
            ones_list.remove(j)
            zeros_list.remove(i)
            ones_list.append(i)
            zeros_list.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                h[i], h[j] = 1.0, 0.0
                corr = new_corr
                cur_c = new_c
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
        T *= cool_rate
    return best_h, best_c

def binary_greedy(h_init, n, deadline, max_sweeps=500, seed=42):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_list = sorted(zeros)
        for i in z_list:
            if time.time() > deadline:
                break
            o_list = sorted(ones)
            for j in o_list:
                delta = get_corr_delta_two(h, n, i, j)
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + delta
                    cur_c = nc
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
