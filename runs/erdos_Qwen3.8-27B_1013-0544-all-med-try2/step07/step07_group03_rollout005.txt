def project_feasible(h,n):
    h=np.asarray(h,dtype=float)
    target=0.5*n
    lo=h.min()-1.0
    hi=h.max()+1.0
    for _ in range(70):
        tau=0.5*(lo+hi)
        s=np.clip(h-tau,0.0,1.0).sum()
        if s>target: lo=tau
        else: hi=tau
    x=np.clip(h-tau,0.0,1.0)
    for _ in range(10):
        r=target-x.sum()
        if abs(r)<1e-14: break
        mask=(x>1e-12)&(x<1.0-1e-12)
        if mask.sum()==0:
            # shift slightly from one edge
            if r>0:
                idx=np.argmin(x)
                x[idx]=min(1.0,x[idx]+r)
            else:
                idx=np.argmax(x)
                x[idx]=max(0.0,x[idx]+r)
            continue
        x[mask]+=r/mask.sum()
        x=np.clip(x,0.0,1.0)
    # final exact-ish
    r=target-x.sum()
    if abs(r)>0:
        mask=(x>1e-15)&(x<1.0-1e-15)
        if mask.sum()>0:
            x[mask]+=r/mask.sum()
            x=np.clip(x,0.0,1.0)
    return x

def soft_compute(h,alpha):

    h=np.asarray(h,dtype=float)
    if alpha < 1.0: alpha=1.0
    if _SX is not None and _SX.size==h.size and _SA==alpha and np.array_equal(h,_SX):
        return _SV,_SG
    n=h.size
    dx=2.0/n
    c=np.correlate(h,1.0-h,mode='full')*dx
    m=float(c.max())
    sh=alpha*(c-m)
    e=np.exp(sh)
    s=e.sum()
    w=e/s
    g=np.zeros(n)
    g += w[n-1]*(1.0-2.0*h)*dx
    # Precompute W for s=1..n-1

            g[:-s] += sc*(1.0 - h[s:])
            g[s:] += sc*(1.0 - h[:-s])
    val = m + float(np.log(s)/alpha)
    _SX = h.copy(); _SA=float(alpha); _SV=float(val); _SG=g
    return _SV,_SG

```python
import numpy as np
import time
from scipy.optimize import minimize

_SX = None
_SA = None
_SV = None
_SG = None
PAIR_F = (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625, 0.001953125)
PAIR_A = (0.0005, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1)


def _fix_sum(x, target):
    for _ in range(10):
        r = target - x.sum()
        if abs(r) < 1e-14:
            break
        m = (x > 1e-12) & (x < 1.0 - 1e-12)
        if m.sum() > 0:
            x[m] += r / m.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            if r > 0:
                i = int(np.argmin(x))
                x[i] = min(1.0, x[i] + r)
            else:
                i = int(np.argmax(x))
                x[i] = max(0.0, x[i] + r)
    return x


def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    target = 0.5 * n
    x = np.clip(h, 0.0, 1.0)
    if abs(x.sum() - target) <= 1e-10:
        return _fix_sum(x.copy(), target)
    lo = min(float(x.min()) - 1.0, -1.0)
    hi = max(float(x.max()) + 1.0, 1.0)
    for _ in range(70):
        tau = 0.5 * (lo + hi)
        s = np.clip(h - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    x = np.clip(h - tau, 0.0, 1.0)
    return _fix_sum(x, target)


def exact_c5(h, n=None):
    if n is None:
        n = int(len(h))
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)


def to_binary(h, n):
    if n % 2:
        return project_feasible(h, n)
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb


def transform_sym(h, kind):
    if kind == 0:
        return 0.5 * (h + 1.0 - h[::-1])
    if kind == 1:
        return 0.5 * (h + h[::-1])
    return h.copy()


def resample_h(h, nh):
    n = len(h)
    if n == nh:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(nh) + 0.5) / nh
    return np.interp(xn, xo, h)


def soft_compute(h, alpha):
    global _SX, _SA, _SV, _SG
    h = np.asarray(h, dtype=float)
    if float(alpha) < 1.0:
        alpha = 1.0
    if _SX is not None and _SA == alpha and h.size == _SX.size and np.array_equal(h, _SX):
        return _SV, _SG
    n = h.size
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = e.sum()
    w = e / s
    g = np.zeros(n)
    g += w[n - 1] * (1.0 - 2.0 * h) * dx
    wl = w[:n - 1][::-1]
    wr = w[n:]
    for k in range(1, n):
        a = wl[k - 1]
        b = wr[k - 1]
        if a > 1e-300 or b > 1e-300:
            g[:-k] += dx * (a * (1.0 - h[k:]) - b * h[k:])
            g[k:] += dx * (b * (1.0 - h[:-k]) - a * h[:-k])
    val = m + float(np.log(s) / alpha)
    _SX = h.copy()
    _SA = alpha
    _SV = val
    _SG = g
    return _SV, _SG


def soft_obj(x, alpha):
    return soft_compute(x, alpha)[0]


def soft_grad(x, alpha):
    return soft_compute(x, alpha)[1]


def sum_eq(x):
    return float(x.sum() - 0.5 * len(x))


def slsqp_smooth(h, n, alpha, maxiter, ftol):
    if maxiter <= 0:
        return h
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': sum_eq},)
    try:
        res = minimize(soft_obj, h, method='SLSQP', jac=soft_grad, bounds=bounds,
                       constraints=cons, args=(alpha,),
                       options={'maxiter': int(maxiter), 'ftol': ftol, 'iprint': 0})
        if hasattr(res, 'x') and res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def safe_iters(base, est, deadline):
    rem = deadline - time.time()
    if rem <= 0.0 or est <= 0.0:
        return 0
    return int(min(base, rem / est))


def derivative_vector(h, p, n):
    L = 2 * n - 1
    c = np.zeros(L)
    c[n - 1] = 1.0 - 2.0 * h[p]
    if p < n - 1:
        s = np.arange(1, n - p)
        li = n - 1 - s
        ri = n - 1 + s
        c[li] += 1.0 - h[p + s]
        c[ri] -= h[p + s]
    if p > 0:
        s = np.arange(1, p + 1)
        li = n - 1 - s
        ri = n - 1 + s
        c[li] -= h[p - s]
        c[ri] += 1.0 - h[p - s]
    return c


def pair_polish(h, n, deadline, K=8, max_sweeps=1000):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur = float(np.max(corr) * dx)
    best_h = h.copy()
    best_c = cur
    L = 2 * n - 1
    tmp = np.empty(L)
    stagnant = 0
    for sweep in range(max_sweeps):
        if time.time() >= deadline:
            break
        if sweep and sweep % 20 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
            cur = float(np.max(corr) * dx)
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
        _, g = soft_compute(h, 2000.0)
        top = max(12, 3 * K)
        top = min(top, n)
        inc_ids = np.argsort(g)[:top]
        dec_ids = np.argsort(g)[-top:][::-1]
        inc = []
        dec = []
        for ii in inc_ids:
            ii = int(ii)
            if h[ii] < 1.0 - 1e-9:
                inc.append(ii)
                if len(inc) >= K:
                    break
        for jj in dec_ids:
            jj = int(jj)
            if h[jj] > 1e-9:
                dec.append(jj)
                if len(dec) >= K:
                    break
        if not inc or not dec:
            break
        vI = [derivative_vector(h, i, n) for i in inc]
        vJ = [derivative_vector(h, j, n) for j in dec]
        improved = False
        for ii in range(len(inc)):
            if time.time() >= deadline:
                break
            i = inc[ii]
            ci = vI[ii]
            for jj in range(len(dec)):
                if (jj & 15) == 0 and time.time() >= deadline:
                    break
                j = dec[jj]
                if i == j:
                    continue
                maxd = min(h[j], 1.0 - h[i])
                if maxd <= 1e-9:
                    continue
                a = ci - vJ[jj]
                k1 = n - 1 + (i - j)
                k2 = n - 1 + (j - i)
                best_d = 0.0
                best_val = cur
                for f in PAIR_F:
                    d = f * maxd
                    if d <= 1e-10 or d > maxd + 1e-12:
                        continue
                    d = min(d, maxd)
                    np.add(corr, d * a, out=tmp)
                    d2 = d * d
                    tmp[n - 1] -= 2.0 * d2
                    tmp[k1] += d2
                    if k2 != k1:
                        tmp[k2] += d2
                    val = float(np.max(tmp) * dx)
                    if val < best_val - 1e-14:
                        best_val = val
                        best_d = d
                for d in PAIR_A:
                    if d > maxd + 1e-12:
                        continue
                    if d <= 1e-10:
                        continue
                    d = min(d, maxd)
                    np.add(corr, d * a, out=tmp)
                    d2 = d * d
                    tmp[n - 1] -= 2.0 * d2
                    tmp[k1] += d2
                    if k2 != k1:
                        tmp[k2] += d2
                    val = float(np.max(tmp) * dx)
                    if val < best_val - 1e-14:
                        best_val = val
                        best_d = d
                if best_d > 0.0:
                    h[i] = min(1.0, h[i] + best_d)
                    h[j] = max(0.0, h[j] - best_d)
                    delta = best_d * a
                    d2 = best_d * best_d
                    delta[n - 1] -= 2.0 * d2
                    delta[k1] += d2
                    if k2 != k1:
                        delta[k2] += d2
                    corr += delta
                    cur = best_val
                    improved = True
                    if cur < best_c:
                        best_c = cur
                        best_h = h.copy()
        if not improved:
            stagnant += 1
            if stagnant >= 3:
                break
        else:
            stagnant = 0
    c = exact_c5(best_h, n)
    if c < best_c:
        best_c = c
    return best_h, best_c


def polish_h(h, n, deadline, mode=0):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = exact_c5(h, n)
    for kind in (0, 1):
        hs = project_feasible(transform_sym(h, kind), n)
        c = exact_c5(hs, n)
        if c < best_c:
            best_h, best_c, h = hs.copy(), c, hs.copy()
    sl_specs = (
        (20.0, 80, 1e-12),
        (200.0, 120, 1e-12),
        (2000.0, 150, 1e-12),
        (20000.0, 200, 1e-13),
        (200000.0, 250, 1e-14),
    )
    use = 5 if mode == 0 else (3 if mode == 1 else 2)
    nf = max(1.0, float(n) / 400.0)
    est = 0.10 * nf * nf
    for i in range(use):
        if time.time() >= deadline:
            break
        a, base, ft = sl_specs[i]
        it = safe_iters(base, est, deadline)
        if it < 20:
            break
        h2 = slsqp_smooth(best_h, n, a, it, ft)
        c2 = exact_c5(h2, n)
        if c2 < best_c:
            best_c, best_h = c2, h2
        if mode == 0 and i == 2 and time.time() < deadline - 30:
            hp, cp = pair_polish(best_h, n, min(deadline, time.time() + 20.0), K=8, max_sweeps=2000)
            if cp < best_c:
                best_c, best_h = cp, hp
    K = 12 if mode == 0 else 8
    ms = 100000 if mode == 0 else (3000 if mode == 1 else 300)
    if time.time() < deadline - 1:
        hp, cp = pair_polish(best_h, n, deadline, K=K, max_sweeps=ms)
        if cp < best_c:
            best_c, best_h = cp, hp
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + float(budget_s) - 25.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.full(400, 0.5)
    n0 = int(len(h0))
    if n0 <= 0:
        n0 = 400
        h0 = np.full(n0, 0.5)
    best_h = project_feasible(h0, n0)
    best_c = exact_c5(best_h, n0)
    best_n = n0

    for kind in (0, 1):
        hs = project_feasible(transform_sym(h0, kind), n0)
        c = exact_c5(hs, n0)
        if c < best_c:
            best_c, best_h = c, hs.copy()

    if best_n % 2 == 0:
        hb = to_binary(best_h, best_n)
        c = exact_c5(hb, best_n)
        if c < best_c:
            best_c, best_h = c, hb.copy()

    T = max(1.0, dl - t0)
    main_dl = t0 + 0.70 * T
    best_h, best_c = polish_h(best_h, best_n, main_dl, mode=0)

    if time.time() < dl - 90:
        alt_dl = min(dl - 70, t0 + 0.84 * T)
        for scale in (0.02, 0.05):
            if time.time() >= alt_dl - 20:
                break
            ht = project_feasible(best_h + scale * rng.randn(best_n), best_n)
            dlim = min(alt_dl, time.time() + max(20.0, 0.06 * T))
            ht, ct = polish_h(ht, best_n, dlim, mode=1)
            if ct < best_c:
                best_c, best_h = ct, ht

    if time.time() < dl - 100:
        res_dl = min(dl - 40, t0 + 0.92 * T)
        old_n = best_n
        for nh in (max(120, old_n // 2), max(160, old_n * 3 // 4), min(900, old_n + 80)):
            if time.time() >= res_dl - 20:
                break
            nh = int(nh)
            if nh % 2:
                nh += 1
            if nh <= 60 or nh > 1000 or nh == best_n:
                continue
            ht = project_feasible(resample_h(best_h, nh), nh)
            dlim = min(res_dl, time.time() + max(25.0, 0.06 * T))
            ht, ct = polish_h(ht, nh, dlim, mode=1)
            if ct < best_c:
                best_c, best_h, best_n = ct, ht, nh

    if time.time() < dl - 5:
        best_h, best_c = pair_polish(best_h, best_n, dl - 5, K=12, max_sweeps=100000)

    best_h = project_feasible(best_h, best_n)
    best_c = exact_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
```
