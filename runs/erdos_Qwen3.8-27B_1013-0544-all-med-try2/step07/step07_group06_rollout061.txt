One more consideration: the binary search for d_opt. The function f(d) = max_m(c_base[m] + d * slopes[m]) is convex piecewise linear. Its subgradient at any point is the set of slopes of the active (maximizing) linear functions. At the minimum, 0 must be in the subgradient (or at a boundary).

Binary search: 
- At d=mid, compute which m are active
- If all active have slope < 0: f is decreasing at mid, so optimum is to the right → lo = mid
- If all active have slope > 0: f is increasing at mid, so optimum is to the left → hi = mid
- If some active have slope < 0 and some > 0: mid is optimal (or within a flat region) → done

This converges in O(log(1/eps) * n) time, which is O(n) for practical purposes.

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()

    target = n / 2.0

            if time.time() > deadline:
                return h, best_c
            old_i, old_j = h[i], h[j]
            d_lo = max(-old_i, old_j - 1.0)
            d_hi = min(1.0 - old_i, old_j)
            if d_hi - d_lo < 1e-12:
                continue
            slopes = D[i] - D[j]
            lo_d, hi_d = d_lo, d_hi
            for _ in range(30):
                mid_d = (lo_d + hi_d) / 2
                val = c_base + mid_d * slopes
                vmax = val.max()
                active = (val >= vmax - 1e-14)
                sg_min = slopes[active].min()
                sg_max = slopes[active].max()
                if sg_min <= 0 <= sg_max:
                    break
                elif sg_max < 0:
                    lo_d = mid_d
                else:
                    hi_d = mid_d
            d_opt = np.clip((lo_d + hi_d) / 2, d_lo, d_hi)
            if abs(d_opt) < 1e-13:
                continue
            h[i] = old_i + d_opt
            h[j] = old_j - d_opt
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                c_base = compute_c5_vec(h, n)
                D = compute_D(h, n)
            else:
                h[i] = old_i
                h[j] = old_j
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.default_rng(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B
    t1 = min(t0 + 80, deadline - 500)
    best_h, best_c = lbfgs_phase(best_h, n, t1, max_iters=200)

    # Phase 2: Paired exact descent (main workhorse)
    if time.time() < deadline - 350:
        t2 = min(time.time() + 250, deadline - 350)
        h2, c2 = paired_exact_descent(best_h, n, t2, top_coords=80, max_sweeps=3000)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: L-BFGS-B polish
    if time.time() < deadline - 250:
        t3 = min(time.time() + 60, deadline - 250)
        h3, c3 = lbfgs_phase(best_h, n, t3, max_iters=150)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: More paired descent
    if time.time() < deadline - 180:
        t4 = min(time.time() + 150, deadline - 180)
        h4, c4 = paired_exact_descent(best_h, n, t4, top_coords=100, max_sweeps=3000)
        if c4 < best_c:
            best_c, best_h = c4, h4

    # Phase 5: Full pair sweep for thoroughness
    if time.time() < deadline - 100:
        t5 = min(time.time() + 60, deadline - 100)
        h5, c5 = full_pair_sweep(best_h, n, t5)
        if c5 < best_c:
            best_c, best_h = c5, h5

    # Phase 6: Try different n values
    if time.time() < deadline - 80:
        for n_try in [250, 350, 400]:
            if time.time() > deadline - 50:
                break
            x_old = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0 / len(best_h)
            x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            alloc = min(30, deadline - time.time() - 20)
            if alloc < 10:
                break
            ht, ct = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.4, max_iters=150)
            if time.time() < deadline - 15:
                ht, ct = paired_exact_descent(ht, n_try, min(time.time() + 15, deadline - 10), top_coords=60, max_sweeps=500)
            if ct < best_c:
                best_c, best_h = ct, ht

    # Phase 7: Final paired descent
    if time.time() < deadline - 10:
        nf = len(best_h)
        hf, cf = paired_exact_descent(best_h, nf, deadline - 5, top_coords=100, max_sweeps=2000)
        if cf < best_c:
            best_c, best_h = cf, hf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
