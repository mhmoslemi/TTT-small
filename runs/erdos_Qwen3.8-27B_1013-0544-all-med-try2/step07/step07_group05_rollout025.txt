def adam_optimize(h, n, deadline, alpha_start=100, alpha_end=50000, n_iter_max=5000):
    h = h.copy()
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_c = compute_c5(h, n)
    best_h = h.copy()
    patience = 0
    max_patience = 200
    
    while t < n_iter_max and time.time() < deadline:
        # Adaptive alpha
        progress = min(1.0, t / (n_iter_max * 0.8))
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        # Compute gradient
        grad = smooth_max_grad(h, n, alpha)
        
        # ADAM update
        t += 1
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad ** 2
        m_hat_hat = m_hat / (1 - beta1 ** t)
        v_hat_hat = v_hat / (1 - beta2 ** t)
        
        h_new = h - lr * m_hat_hat / (np.sqrt(v_hat_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        h = h_new
        if patience > max_patience:
            # Reset learning rate
            lr *= 0.5
            patience = 0
            if lr < 1e-7:
                break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (d / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (d / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    if sh.max() > 500:
        return float(m + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    return float(x.sum() - n / 2.0)

    return np.ones(n)

                for _ in range(18):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    temp[:corr_len] = corr
                    temp[li:hi] += m1 * si
                    temp[lj:hj] -= m1 * sj
                    v1 = temp.max()
                    temp[:corr_len] = corr
                    temp[li:hi] += m2 * si
                    temp[lj:hj] -= m2 * sj
                    v2 = temp.max()
                    if v1 > v2:
                        a = m1
                    else:
                        b = m2
                d_opt = (a + b) / 2.0
                if abs(d_opt) < 1e-10:
                    continue
                temp[:corr_len] = corr
                temp[li:hi] += d_opt * si
                temp[lj:hj] -= d_opt * sj
                c_val = float(temp.max())
                if c_val < best_c - min_gain:
                    h[i] += d_opt
                    h[j] -= d_opt
                    corr = temp.copy()
                    best_c = c_val
                    improved_any = True
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def fixed_pair_search(h, n, deadline, deltas):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    alphas = [500, 3000, 15000, 75000, 300000, 1500000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        ht = slsqp_opt(h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    if time.time() < deadline - 5:
        h, best_c = smart_pair_search(h, n, deadline - 2)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100
    best_h = None
    best_c = float('inf')
    n_candidates = sorted(set([n_init] + [150, 200, 250, 300]))
    n_candidates = [n for n in n_candidates if 50 <= n <= 400]
    w = [2.5 if n == n_init else 1.0 for n in n_candidates]
    tot_w = sum(w)
    avail = deadline - t0 - 20
    for idx, n_try in enumerate(n_candidates):
        if time.time() > deadline - 15:
            break
        alloc = (w[idx] / tot_w) * avail
        alloc = min(alloc, deadline - time.time() - 8)
        if alloc < 20:
            continue
        sub_dl = time.time() + alloc
        if n_try == n_init:
            h_start = h_init.copy()
        else:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, h_init)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, sub_dl)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        best_h, best_c = smart_pair_search(best_h, n, deadline - 3, min_gain=5e-14)
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        ht = slsqp_opt(best_h, n, 5000000, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    if best_h is not None and time.time() < deadline - 2:
        n = len(best_h)
        if n <= 250:
            best_h, best_c = fixed_pair_search(best_h, n, deadline - 1,
                [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
