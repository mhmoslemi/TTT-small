def local_refine(h, n, deadline, max_passes=200):

                        else:
                            h[i] += d; h[j] -= d

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def _sum_jacobian(x, n):
    return np.ones(len(x))

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def slsqp_refine(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jacobian, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    C = np.correlate(h, 1 - h, mode='full')

            improved = False

                for j in range(i + 1, n):

                        # Incremental update

                        saved_j = h[j].copy()
                        h[i] += d
                        h[j] -= d

```python
def projected_gradient(h, n, deadline, alpha=10000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.01
    
    while time.time() < deadline:
        # Compute smooth objective and gradient
        corr = np.correlate(h, 1 - h, mode='full')
        c_all = corr * dx
        cm = c_all.max()
        # Log-sum-exp weighted gradient
        e = np.exp(alpha * (c_all - cm))
        w = e / e.sum()
        
        # Gradient = weighted sum of individual gradients
        # g = sum_k w_k * grad(c_k)
        # For efficiency, only compute for top few w_k
        top_idx = np.argsort(w)[-20:]
        w_top = w[top_idx]
        w_top /= w_top.sum()
        
        grad = np.zeros(n)
        for t in range(len(top_idx)):
            k = top_idx[t] - (n - 1)
            # grad of c_k w.r.t. h
            if k >= 0:
                grad[:n-k] += w_top[t] * (1 - h[k:])
                grad[k:] -= w_top[t] * h[:n-k]
            else:
                kp = -k
                grad[kp:] += w_top[t] * (1 - h[:n-kp])
                grad[:n-kp] -= w_top[t] * h[kp:]
        grad *= dx
        
        # Projected step
        h_new = np.clip(h - lr * grad, 0, 1)
        # Project onto sum constraint
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.1, 0.5)
        else:
            lr *= 0.8
            if lr < 1e-6:
                lr = 0.01
                alpha = min(alpha * 2, 1000000)
    
    return best_h, best_c
```
