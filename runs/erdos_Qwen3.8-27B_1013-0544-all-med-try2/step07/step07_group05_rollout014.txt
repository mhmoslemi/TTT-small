def pair_perturb(h, n, deadline, radius=30):
    """Try small pairwise adjustments for nearby indices."""
    corr = compute_corr(h, n)
    best_c = float(corr.max())
    k = np.arange(2 * n - 1)
    
    deltas = [0.01, 0.005, 0.001, 0.0001]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi_idx = min(n, i + radius + 1)
                for j in range(lo, hi_idx):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        # Update corr
                        # h[i] changes by +d, h[j] changes by -d
                        # Effect on corr: d * (slopes for i) - d * (slopes for j)
                        # But this is approximate... let me just recompute
                        h[i] += d
                        h[j] -= d
                        c = float(compute_corr(h, n).max())
                        if c < best_c - 1e-14:
                            best_c = c
                            corr = compute_corr(h, n)
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
