        if abs(diff) < 1e-14:

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h = np.clip(h + (diff / tot) * room, 0, 1)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h * (target / tot), 0, 1)

    shifted = alpha * (c - c.max())
    mx = shifted.max()
    exps = np.exp(shifted - mx)
    val = c.max() + (mx + np.log(exps.sum())) / alpha
    w = exps / exps.sum()

    grad = np.zeros(n)

                grad[:nv] += wm * (-h[kk:])

    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    for alpha in alphas:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def c5_vec(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0, 1)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-15:
                break
            h = np.clip(h + (d / t) * r, 0, 1)
        else:
            t = h.sum()
            if t < 1e-15:
                break
            h = np.clip(h * (target / t), 0, 1)
    return h

def sm_grad(h, n, a):
    c = c5_vec(h, n)
    s = a * (c - c.max())
    mx = s.max()
    e = np.exp(s - mx)
    w = e / e.sum()
    v = c.max() + np.log(e.sum()) / a
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(len(w)):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        cf = w[m] * a * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                g[:nv] += cf * (1.0 - h[k:])
                g[k:] += cf * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                g[kk:] += cf * (1.0 - h[:nv])
                g[:nv] += cf * (-h[kk:])
    return v, g, w

def lbfgs_opt(h, n, deadline, alpha_list, max_iters=300):
    h = project(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline:
            break

        if res is not None:
            h_new = project(res, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
    return best_h, best_c

def _lbfgs_single(h, n, alpha, max_iters, bounds):
    def obj(x):
        v, _ = sm_grad(x, n, alpha)
        return v
    def jac(x):
        _, g, _ = sm_grad(x, n, alpha)
        return g
    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iters, 'ftol': 1e-18, 'gtol': 1e-14})
        return res.x
    except Exception:
        return None

def adam_opt(h, n, deadline, alpha_list, lr0=0.01, steps_per_alpha=2000):
    h = project(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    for alpha in alpha_list:

        if time.time() > deadline:
            break

        t += 1
    c = compute_c5(h, n)
    if c < best_c:

    return best_h, best_c

def _adam_inner(h, n, state, t0, alpha, lr0, b1, b2, eps, steps, deadline):
    if isinstance(state, np.ndarray):
        m_arr = state.copy()
    else:
        m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = t0 if isinstance(t0, int) else 0

        if time.time() > deadline:
            break
        t += 1

        m_arr = b1 * m_arr + (1 - b1) * g
        v_arr = b2 * v_arr + (1 - b2) * g * g
        mc = m_arr / (1 - b1 ** t)
        vc = v_arr / (1 - b2 ** t)
        h = project(h - lr * mc / (np.sqrt(vc) + eps), n)
    return h

def grad_pair_search(h, n, deadline, deltas, top_k=30):

def binary_push(h, n, deadline, threshold=0.4):

    best_c = compute_c5(h, n)
    for _ in range(300):

            if time.time() > deadline:
                return h, best_c
            if h[i] < threshold:
                target = 0.0
            elif h[i] > 1 - threshold:
                target = 1.0
            else:
                continue
            delta = target - h[i]
            if abs(delta) < 1e-12:
                continue
            mask = np.zeros(n, dtype=bool)
            if delta > 0:
                mask = (h > delta) & (h < 1.0) & (h < 0.5)
            else:
                mask = (h < -delta) & (h > 0.5)
            mask[i] = False
            if not mask.any():

                mask[i] = False

                continue
            cnt = mask.sum()
            h_new = h.copy()
            h_new[i] = target
            h_new[mask] -= delta / cnt
            h_new = project(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c - 1e-15:
                best_c = c
                h = h_new
                improved = True
        if not improved:
            break
    return h, best_c

def adam_direct(h, n, deadline, alpha, lr, max_steps):
