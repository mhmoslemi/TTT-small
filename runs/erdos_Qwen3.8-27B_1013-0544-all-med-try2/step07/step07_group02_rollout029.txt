def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgradient_perturb(h, n, scale):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    active = corr >= m - 0.01 * m  # near-active constraints
    # Compute subgradient
    m_range = np.arange(-(n - 1), n)
    sg = np.zeros(n)
    for i in range(n):
        B = np.zeros(2 * n - 1)
        k1 = i - m_range
        mask1 = (k1 >= 0) & (k1 < n)
        B[mask1] += (1.0 - h[k1[mask1]])
        k2 = i + m_range
        mask2 = (k2 >= 0) & (k2 < n)
        B[mask2] -= h[k2[mask2]]
        sg[i] = (B[active] * dx).mean() if active.any() else 0
    h_new = h - scale * sg
    return project_feasible(h_new, n)

def smooth_grad(h, n, dx, alpha):
    C = np.correlate(h, 1-h, mode='full')  # length 2n-1
    s = alpha * C * dx
    s_max = s.max()
    exp_s = np.exp(s - s_max)
    p = exp_s / exp_s.sum()
    
    # p is indexed 0 to 2n-2, corresponding to shifts -(n-1) to (n-1)
    # shift[k] = k - (n-1)
    shifts = np.arange(-(n-1), n)
    
    grad = np.zeros(n)
    for j in range(n):
        # (1 - h[j + shift]) for valid j+shift in [0, n-1]
        idx1 = j + shifts
        mask1 = (idx1 >= 0) & (idx1 < n)
        term1 = np.sum(p * (1 - h[idx1[mask1]]) * mask1)
        
        # h[j - shift] for valid j-shift in [0, n-1]
        idx2 = j - shifts
        mask2 = (idx2 >= 0) & (idx2 < n)

        # Actually: Σ_k p[k] * h[j - shift[k]] where shift[k] = k-(n-1)
        # = Σ_k p[k] * h[j - k + n - 1]
        # j - k + n - 1 must be in [0, n-1]
        idx2 = j - np.arange(2*n-1) + (n-1)
        mask2 = (idx2 >= 0) & (idx2 < n)
        term2 = np.sum(p[mask2] * h[idx2[mask2]])
        
        grad[j] = dx * (term1 - term2)
    
    return grad
