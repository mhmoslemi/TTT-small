        try:

                           options={'maxiter': int(max_inner), 'ftol': 1e-12, 'gtol': 1e-8, 'disp': False})
            y = res.x if res.x is not None else x
        except Exception:
            y = x
        y = np.clip(y, 0.0, 1.0)
        g = float(y.sum() - target)
        if abs(g) < 1e-7:
            x = y
            break
        _G_LAM = _G_LAM + _G_RHO * g
        if _G_LAM > 1e8: _G_LAM = 1e8

        x = project_feasible(y, n)

    v=np.asarray(v,dtype=float); t=0.5*n
    lo=v.min()-1.0; hi=v.max()+1.0
    for _ in range(70):

       else: h+=d/n
       np.clip(h,0.0,1.0,out=h)
    return h

def project_feasible(v, n):
    v = np.asarray(v, dtype=float)
    t = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(70):
        m = 0.5 * (lo + hi)
        s = float(np.clip(v - m, 0.0, 1.0).sum())
        if s > t:
            lo = m
        else:
            hi = m
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(8):
        d = t - float(h.sum())
        if abs(d) < 1e-12:
            break
        msk = (h > 1e-9) & (h < 1.0 - 1e-9)
        if msk.any():
            h[msk] += d / float(msk.sum())
        else:
            h += d / float(n)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def eval_c5(h, n):
    if h.size == 0:
        return float("inf")
    c = np.correlate(h, 1.0 - h, mode="full")
    if not np.all(np.isfinite(c)):
        return float("inf")
    return float(np.max(c) * (2.0 / n))


def gradient_correlations(x, w, n):
    sl = slice(n - 1, n - 1 + n)
    A = np.convolve(w, np.ones(n), mode="full")[sl]
    B = np.convolve(w, x, mode="full")[sl]
    C = np.convolve(w[::-1], x, mode="full")[sl]
    return A - B - C


def _get_smooth(x, alpha, n):
    global _CACHE
    if _CACHE is not None:
        px, pn, pa, pf, pg = _CACHE
        if pn == n and pa == alpha and np.array_equal(px, x):
            return pf, pg

    c = np.correlate(x, 1.0 - x, mode="full")
    if not np.all(np.isfinite(c)):
        return float("inf"), np.zeros(n, dtype=float)

    cmax = float(np.max(c))
    m = c.size
    if alpha <= 1e-3:
        weights = np.full(m, 1.0 / m)
        extra = float(np.mean(c) - cmax)
    else:
        sh = alpha * (c - cmax)
        e = np.exp(sh)
        s = float(e.sum())
        if s <= 1e-300:
            weights = np.zeros(m)
            weights[int(np.argmax(c))] = 1.0
            extra = 0.0
        else:
            weights = e / s
            extra = float(np.log(s / m) / alpha)

    fval = cmax + extra
    g = gradient_correlations(x, weights, n)
    _CACHE = (x, n, alpha, fval, g)
    return fval, g


def fun_smooth(x):
    return _get_smooth(x, _G_ALPHA, _G_N)[0]


def jac_smooth(x):
    return _get_smooth(x, _G_ALPHA, _G_N)[1]


def constraint_eq(x):
    return float(x.sum()) - 0.5 * _G_N


def constraint_eq_jac(x):
    return np.ones(_G_N)


def fun_pen(x):
    n = _G_N
    f, _ = _get_smooth(x, _G_ALPHA, n)
    g = float(x.sum()) - 0.5 * n
    return f + 0.5 * _G_RHO * g * g


def jac_pen(x):
    n = _G_N
    _, g = _get_smooth(x, _G_ALPHA, n)
    q = float(x.sum()) - 0.5 * n
    return g + (_G_RHO * q) * np.ones_like(x)


def base_inner(n):
    if n <= 300:
        return 380
    if n <= 600:
        return 230
    return 150


def optimize_lbfgs(x0, n, alpha, deadline, max_inner=300, max_outer=2):
    global _G_N, _G_ALPHA, _G_RHO
    x = project_feasible(x0, n)
    _G_N = n
    _G_ALPHA = float(alpha)
    _G_RHO = 100.0
    target = 0.5 * n
    bounds = [(0.0, 1.0)] * n
    max_inner = int(max_inner)

    for _ in range(int(max_outer)):
        if time.time() > deadline:
            break
        try:
            res = minimize(fun_pen, x, method="L-BFGS-B", bounds=bounds, jac=jac_pen,
                           options={"maxiter": max_inner, "ftol": 1e-12, "gtol": 1e-8,
                                    "disp": False})
            y = res.x if res.x is not None else x
        except Exception:
            y = x
        y = np.clip(y, 0.0, 1.0)
        g = float(y.sum()) - target
        if abs(g) < 1e-7:
            x = y
            break
        _G_RHO = min(_G_RHO * 6.0, 1e8)
        x = project_feasible(y, n)
    return project_feasible(x, n)


def optimize_slsqp(x0, n, alpha, deadline, maxiter=80):
    global _G_N, _G_ALPHA
    if time.time() > deadline:
        return project_feasible(x0, n)
    x = project_feasible(x0, n)
    _G_N = n
    _G_ALPHA = float(alpha)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac},)
    try:
        res = minimize(fun_smooth, x, method="SLSQP", bounds=bounds,
                       constraints=cons, jac=jac_smooth,
                       options={"maxiter": int(maxiter), "ftol": 1e-12, "iprint": 0})
        y = res.x if res.x is not None else x
    except Exception:
        y = x
    return project_feasible(y, n)


def binary_start(h, n):
    if n % 2 != 0:
        return project_feasible(h, n)
    target = n // 2
    b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    b[idx[:target]] = 1.0
    return project_feasible(b, n)


def upsample(h, n, n2):
    x1 = np.linspace(0.0, 2.0, n, endpoint=False)
    x2 = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x2, x1, h)
    return project_feasible(h2, n2)


def run(seed=42, budget_s=1100, **kwargs):
    global _CACHE
    B = float(budget_s)
    t0 = time.time()
    usable = max(30.0, B - 20.0)
    hard = t0 + B - 8.0
    rng = np.random.RandomState(seed)
    _CACHE = None

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    if n0 < 10:
        n0, h0 = 250, np.full(250, 0.5)
    h0 = project_feasible(h0, n0)

    best_h = h0.copy()
    best_n = n0
    best_c = eval_c5(best_h, best_n)
    if not np.isfinite(best_c):
        best_h = project_feasible(np.full(n0, 0.5), n0)
        best_c = eval_c5(best_h, best_n)

    work = best_h.copy()
    ph1 = t0 + 0.28 * usable
    ph2 = t0 + 0.54 * usable
    ph3 = t0 + 0.78 * usable

    alphas = [30.0, 120.0, 500.0, 2000.0, 8000.0]
    for alpha in alphas:
        if time.time() > ph1 - 10.0:
            break
        cand = optimize_lbfgs(work, best_n, alpha, ph1, base_inner(best_n), 2)
        c = eval_c5(cand, best_n)
        if np.isfinite(c) and c < best_c:
            best_c = c
            best_h = cand.copy()
            work = cand.copy()
        else:
            work = cand if (np.isfinite(c) and c < best_c + 5e-5) else best_h.copy()

        if best_n % 2 == 0:
            bb = binary_start(work, best_n)
            cb = eval_c5(bb, best_n)
            if np.isfinite(cb) and cb < best_c:
                best_c = cb
                best_h = bb.copy()
                work = bb.copy()

        if best_n <= 300 and time.time() < ph1 - 14.0:
            cand2 = optimize_slsqp(work, best_n, max(alpha, 500.0), ph1 - 6.0, 70)
            c2 = eval_c5(cand2, best_n)
            if np.isfinite(c2) and c2 < best_c:
                best_c = c2
                best_h = cand2.copy()
                work = cand2.copy()

    if best_n < 500 and time.time() < hard - 120.0:
        n2 = 500
        h2 = upsample(best_h, best_n, n2)
        c2 = eval_c5(h2, n2)
        work2 = h2
        if np.isfinite(c2) and c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            best_n = n2
            work2 = best_h.copy()

        for alpha in [60.0, 250.0, 1000.0, 4000.0]:
            if time.time() > ph2 - 10.0:
                break
            cand = optimize_lbfgs(work2, n2, alpha, ph2, base_inner(n2), 2)
            c = eval_c5(cand, n2)
            if np.isfinite(c) and c < best_c:
                best_c = c
                best_h = cand.copy()
                best_n = n2
                work2 = cand.copy()
            else:
                work2 = best_h.copy() if best_n == n2 else work2

            if n2 <= 300 and time.time() < ph2 - 20.0:
                cand2 = optimize_slsqp(work2, n2, max(alpha, 800.0), ph2 - 8.0, 70)
                c2 = eval_c5(cand2, n2)
                if np.isfinite(c2) and c2 < best_c:
                    best_c = c2
                    best_h = cand2.copy()
                    best_n = n2
                    work2 = cand2.copy()

        if n2 % 2 == 0:
            bb = binary_start(work2, n2)
            cb = eval_c5(bb, n2)
            if np.isfinite(cb) and cb < best_c:
                best_c = cb
                best_h = bb.copy()
                best_n = n2

    n_cur = best_n
    ph4 = hard - 10.0

    for mi in range(8):
        if time.time() > ph3 - 15.0:
            break
        if mi == 7:
            x = 0.5 * (best_h + best_h[::-1])
        else:
            s = int((mi + 1) * n_cur / 8.0)
            x = 0.5 * (best_h + np.roll(best_h, s))
        x = project_feasible(x, n_cur)
        alpha = [200.0, 800.0, 3000.0, 1200.0][mi % 4]
        cand = optimize_lbfgs(x, n_cur, alpha, ph3, base_inner(n_cur), 2)
        c = eval_c5(cand, n_cur)
        if np.isfinite(c) and c < best_c:
            best_c = c
            best_h = cand.copy()
        if n_cur % 2 == 0:
            bb = binary_start(cand, n_cur)
            cb = eval_c5(bb, n_cur)
            if np.isfinite(cb) and cb < best_c:
                best_c = cb
                best_h = bb.copy()

    pert_alphas = [250.0, 1000.0, 4000.0]
    scales = [0.02, 0.05, 0.10, 0.03]
    pc = 0
    while time.time() < ph3 - 8.0 and pc < 36:
        x = best_h + scales[pc % 4] * rng.standard_normal(n_cur)
        x = project_feasible(x, n_cur)
        alpha = pert_alphas[pc % 3]
        cand = optimize_lbfgs(x, n_cur, alpha, ph3, base_inner(n_cur), 2)
        c = eval_c5(cand, n_cur)
        if np.isfinite(c) and c < best_c:
            best_c = c
            best_h = cand.copy()
        if n_cur % 2 == 0:
            bb = binary_start(cand, n_cur)
            cb = eval_c5(bb, n_cur)
            if np.isfinite(cb) and cb < best_c:
                best_c = cb
                best_h = bb.copy()
        pc += 1
        if pc % 9 == 0 and n_cur <= 300 and time.time() < ph3 - 18.0:
            cand2 = optimize_slsqp(best_h, n_cur, 1500.0, ph3 - 8.0, 70)
            c2 = eval_c5(cand2, n_cur)
            if np.isfinite(c2) and c2 < best_c:
                best_c = c2
                best_h = cand2.copy()

    work = best_h.copy()
    for alpha in [400.0, 2000.0, 8000.0]:
        if time.time() > ph4 - 15.0:
            break
        cand = optimize_lbfgs(work, n_cur, alpha, ph4, base_inner(n_cur), 2)
        c = eval_c5(cand, n_cur)
        if np.isfinite(c) and c < best_c:
            best_c = c
            best_h = cand.copy()
        work = cand if (np.isfinite(c) and c < best_c + 3e-5) else best_h.copy()
        if n_cur <= 300 and time.time() < ph4 - 18.0:
            cand2 = optimize_slsqp(work, n_cur, max(alpha, 1500.0), ph4 - 8.0, 80)
            c2 = eval_c5(cand2, n_cur)
            if np.isfinite(c2) and c2 < best_c:
                best_c = c2
                best_h = cand2.copy()

    if n_cur % 2 == 0:
        bb = binary_start(best_h, n_cur)
        cb = eval_c5(bb, n_cur)
        if np.isfinite(cb) and cb < best_c:
            best_c = cb
            best_h = bb.copy()
        x = project_feasible(best_h + 0.01 * rng.standard_normal(n_cur), n_cur)
        bb = binary_start(x, n_cur)
        cb = eval_c5(bb, n_cur)
        if np.isfinite(cb) and cb < best_c:
            best_c = cb
            best_h = bb.copy()

    best_h = project_feasible(best_h, n_cur)
    best_c = eval_c5(best_h, n_cur)
    return best_h, float(best_c), int(n_cur)
