def corr_gradient(h,A,info):
    n=info['n']; off=info['off']; L=info['L']
    grad=np.zeros(n,float)
    pos=A[off:off+n]
    diff=np.zeros(n+1,float)
    if pos.size:
       diff[0]+=pos.sum()
       diff[1:n+1]-=pos[::-1]

       neg=A[off-1:off-1-n:-1]

import time
import numpy as np
from scipy.optimize import minimize, linprog

_info_cache = {}
_FRACS = (0.1, 0.25, 0.5, 0.75)

def c5_of(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def corr_of(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def project_feasible(h):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    target = 0.5 * n
    if n == 0:
        return h
    h = np.clip(h, 0.0, 1.0)
    lo = float(np.min(h) - 1.0)
    hi = float(np.max(h) + 1.0)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(60):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = out < 1.0 - 1e-15
            if not np.any(mask):
                break
            room = (1.0 - out)[mask]
            rsum = float(room.sum())
            if rsum <= 1e-15:
                break
            out[mask] += (diff / rsum) * room
        else:
            mask = out > 1e-15
            if not np.any(mask):
                break
            val = out[mask]
            vsum = float(val.sum())
            if vsum <= 1e-15:
                break
            out[mask] -= (-diff / vsum) * val
    out = np.clip(out, 0.0, 1.0)
    diff = target - float(np.sum(out))
    if abs(diff) > 1e-15:
        idx = np.where((out > 1e-14) & (out < 1.0 - 1e-14))[0]
        if idx.size:
            out[idx[0]] = float(np.clip(out[idx[0]] + diff, 0.0, 1.0))
        else:
            idx = np.where(out < 1.0 if diff > 0 else out > 0.0)[0]
            if idx.size:
                out[idx[0]] = float(np.clip(out[idx[0]] + diff, 0.0, 1.0))
    return out

def to_binary(h, n):
    out = np.zeros(n, dtype=float)
    k = n // 2
    if k > 0:
        idx = np.argsort(np.asarray(h, dtype=float))[::-1][:k]
        out[idx] = 1.0
    return out

def interp_h(h, m):
    h = np.asarray(h, dtype=float)
    n = h.size
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return np.interp(xn, xo, h)

def make_info(n):
    if n in _info_cache:
        return _info_cache[n]
    off = n - 1
    L = 2 * n - 1
    fj = []
    fp = []
    fl = []
    sj = []
    sp = []
    sl = []
    for idx in range(L):
        d = idx - off
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            j = np.arange(m, dtype=np.int32)
            p = np.arange(d, d + m, dtype=np.int32)
            li = np.full(m, idx, dtype=np.int32)
            fj.append(j)
            fp.append(p)
            fl.append(li)
            j2 = np.arange(d, n, dtype=np.int32)
            p2 = np.arange(m, dtype=np.int32)
            sj.append(j2)
            sp.append(p2)
            sl.append(li)
        else:
            e = -d
            m = n - e
            if m <= 0:
                continue
            j = np.arange(e, n, dtype=np.int32)
            p = np.arange(m, dtype=np.int32)
            li = np.full(m, idx, dtype=np.int32)
            fj.append(j)
            fp.append(p)
            fl.append(li)
            j2 = np.arange(m, dtype=np.int32)
            p2 = np.arange(e, e + m, dtype=np.int32)
            sj.append(j2)
            sp.append(p2)
            sl.append(li)
    inf = {
        'n': n,
        'off': off,
        'L': L,
        'fj': np.concatenate(fj).astype(np.int32),
        'fp': np.concatenate(fp).astype(np.int32),
        'fl': np.concatenate(fl).astype(np.int32),
        'sj': np.concatenate(sj).astype(np.int32),
        'sp': np.concatenate(sp).astype(np.int32),
        'sl': np.concatenate(sl).astype(np.int32),
    }
    _info_cache[n] = inf
    return inf

def corr_gradient(h, A, inf):
    n = inf['n']
    off = inf['off']
    grad = np.zeros(n, dtype=float)
    pos = A[off:off + n]
    diff = np.zeros(n + 1, dtype=float)
    if pos.size:
        diff[0] += float(pos.sum())
        diff[1:n + 1] -= pos[::-1]
    if n > 1:
        neg = A[off - 1:off - 1 - n:-1]
        if neg.size:
            diff[1:n] += neg
            diff[n] -= float(neg.sum())
    grad += np.cumsum(diff)[:n]
    w = -(h[inf['fp']] * A[inf['fl']])
    grad += np.bincount(inf['fj'], weights=w, minlength=n)
    w = -(h[inf['sp']] * A[inf['sl']])
    grad += np.bincount(inf['sj'], weights=w, minlength=n)
    return grad

def smooth_obj(x, alpha):
    x = np.asarray(x, dtype=float)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(c))
    z = alpha * (c - m)
    return m + float(np.log(np.exp(z).sum()) / alpha)

def smooth_grad(x, alpha):
    x = np.asarray(x, dtype=float)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(c))
    z = alpha * (c - m)
    e = np.exp(z)
    p = e / float(e.sum())
    return corr_gradient(x, p, make_info(x.size))

def eq_fun(x):
    return float(np.sum(x) - 0.5 * np.asarray(x).size)

def eq_jac(x):
    return np.ones(np.asarray(x).size, dtype=float)

def slsqp_alpha(h, alpha, maxit):
    h = project_feasible(h)
    n = h.size
    maxit = int(maxit)
    if n == 0 or maxit <= 0:
        return h
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},)
    try:
        res = minimize(
            smooth_obj,
            h,
            method='SLSQP',
            jac=smooth_grad,
            bounds=bounds,
            constraints=cons,
            args=(float(alpha),),
            options={'maxiter': maxit, 'ftol': 1e-12, 'disp': False}
        )
        out = np.asarray(res.x, dtype=float)
    except Exception:
        return h
    return project_feasible(np.clip(out, 0.0, 1.0))

def shift_grad(h, idx, n):
    off = n - 1
    d = int(idx) - off
    g = np.zeros(n, dtype=float)
    if d >= 0:
        m = n - d
        if m > 0:
            g[:m] += 1.0 - h[d:d + m]
            g[d:] -= h[:m]
    else:
        e = -d
        m = n - e
        if m > 0:
            g[e:e + m] += 1.0 - h[:m]
            g[:m] -= h[e:e + m]
    return g

def lp_direction(h, c, active, max_step):
    n = h.size
    t = float(np.max(c))
    rows = []
    b = []
    for idx in active:
        d = shift_grad(h, idx, n)
        row = np.empty(n + 1, dtype=float)
        row[:n] = d
        row[n] = -1.0
        rows.append(row)
        b.append(-float(c[idx]))
    if not rows:
        return None
    A = np.vstack(rows)
    b = np.asarray(b, dtype=float)
    bounds = []
    for i in range(n):
        lo = float(max(-h[i], -max_step))
        hi = float(min(1.0 - h[i], max_step))
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((t - 1000.0, t + 100.0))
    cobj = np.zeros(n + 1, dtype=float)
    cobj[n] = 1.0
    Aeq = np.ones((1, n), dtype=float)
    beq = np.zeros(1, dtype=float)
    try:
        sol = linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq, bounds=bounds)
    except Exception:
        try:
            sol = linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq, bounds=bounds, method='highs')
        except Exception:
            return None
    if sol is None or not sol.success:
        return None
    v = np.asarray(sol.x[:n], dtype=float)
    if np.linalg.norm(v) < 1e-12:
        return None
    return v

def lp_refine(h, deadline, max_rounds=250):
    n = h.size
    c = corr_of(h)
    cur = float(c.max())
    for ms in (0.20, 0.10, 0.05, 0.02):
        rounds = 0
        while time.time() < deadline and rounds < max_rounds:
            order = np.argsort(c)[::-1]
            active = order[:35]
            active = active[c[active] > cur - 1.0]
            if active.size < 5:
                active = order[:5]
            v = lp_direction(h, c, active, ms)
            if v is None:
                break
            improved = False
            for a in (1.0, 0.6, 0.4, 0.25, 0.12):
                hn = h + a * v
                if np.any(hn < -1e-12) or np.any(hn > 1.0 + 1e-12):
                    hn = project_feasible(np.clip(hn, 0.0, 1.0))
                cn = corr_of(hn)
                val = float(cn.max())
                if val < cur - 1e-10:
                    h = hn
                    c = cn
                    cur = val
                    improved = True
                    break
            rounds += 1
            if not improved:
                break
    return h, c

def pair_lin(h, i, j, off):
    n = h.size
    L = 2 * n - 1
    lin = np.zeros(L, dtype=float)
    nmh = 1.0 - h
    lin[off - i:off + n - i] += nmh
    lin[off + i - n + 1:off + i + 1] -= h
    lin[off - j:off + n - j] -= nmh
    lin[off + j - n + 1:off + j + 1] += h
    return lin

def pair_search(h, n, deadline, rng, max_trials=80000):
    h = np.asarray(h, dtype=float).copy()
    c = corr_of(h)
    cur = float(c.max())
    off = n - 1
    active = np.array([], dtype=int)
    no_improve = 0
    for it in range(max_trials):
        if time.time() > deadline:
            break
        if it % 200 == 0 or no_improve > 1200:
            c = corr_of(h)
            cur = float(c.max())
            order = np.argsort(c)[::-1]
            active = order[:80]
            active = active[c[active] > cur - 1.0]
            if active.size == 0:
                active = order[:20]
            no_improve = 0
        if active.size and rng.rand() < 0.68:
            idx = int(active[int(rng.randint(active.size))])
            d = idx - off
            if d >= 0:
                m = n - d
                p = int(rng.randint(m))
                i = p
                j = p + d
            else:
                e = -d
                m = n - e
                p = int(rng.randint(m))
                i = e + p
                j = p
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
        hi = float(h[i])
        hj = float(h[j])
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if high - low <= 1e-12:
            continue
        lin = pair_lin(h, i, j, off)
        cands = [high, low, 0.5 * (high + low)]
        pos_span = high
        neg_span = -low
        for f in _FRACS:
            if pos_span > 1e-12:
                cands.append(f * pos_span)
            if neg_span > 1e-12:
                cands.append(-f * neg_span)
        best_a = 0.0
        best_val = cur
        best_nc = None
        for a in cands:
            a = float(a)
            if abs(a) < 1e-14:
                continue
            nc = c + a * lin
            nc[off] -= 2.0 * a * a
            nc[off + j - i] += a * a
            nc[off + i - j] += a * a
            val = float(nc.max())
            if val < best_val - 1e-12:
                best_val = val
                best_a = a
                best_nc = nc
        if best_nc is not None:
            h[i] = float(np.clip(h[i] + best_a, 0.0, 1.0))
            h[j] = float(np.clip(h[j] - best_a, 0.0, 1.0))
            c = best_nc
            cur = best_val
            no_improve = 0
        else:
            no_improve += 1
    return h, c

            best_h = hp.copy()
            best_c5 = cand_c5

    if time.time() < lim4:
        hp, cp = lp_refine(best_h.copy(), lim4, 300)
        hp = project_feasible(hp)
        cp = corr_of(hp)
        cand_c5 = float(cp.max()) * (2.0 / best_n)
        if cand_c5 < best_c5:
            best_h = hp.copy()
            best_c5 = cand_c5

    hp = best_h.copy()
    camb = best_c5
    for alpha in (50000.0, 250000.0, 800000.0):
        if time.time() > lim5 - 15.0:
            break
        hh = slsqp_alpha(hp, alpha, 140)
        hh = project_feasible(hh)
        hh_c5 = c5_of(hh)
        if hh_c5 < camb - 1e-15:
            hp = hh
            camb = hh_c5
    if camb < best_c5:
        best_h = hp
        best_c5 = camb

    other_ns = []
    if best_n < 500:
        other_ns.append(500)
    if best_n < 700:
        other_ns.append(700)
    if best_n < 320:
        other_ns.append(320)

    for nh in other_ns:
        if time.time() > lim6 - 35.0:
            break
        hh = project_feasible(interp_h(best_h, nh))
        sub_dead = min(lim6 - 25.0, time.time() + 45.0)
        if time.time() < sub_dead:
            hh, _ = pair_search(hh, nh, sub_dead, rng, 25000)
        hh = project_feasible(hh)
        if time.time() < lim6 - 15.0:
            hh = slsqp_alpha(hh, 60000.0, 90)
            hh = project_feasible(hh)
        hc = c5_of(hh)
        if hc < best_c5:
            best_h = hh.copy()
            best_c5 = hc
            best_n = nh

    restarts = 0
    while time.time() < deadline - 6.0 and restarts < 6:
        base = best_h
        if restarts % 3 != 2 and best_n >= 64:
            noise = 0.015 * rng.randn(best_n)
            cand = project_feasible(base + noise)
        else:
            cand = to_binary(base, best_n) if best_n % 2 == 0 else base.copy()
        sub_dead = min(deadline - 6.0, time.time() + 35.0)
        if time.time() < sub_dead:
            cand, _ = pair_search(cand, best_n, sub_dead, rng, 20000)
        cand = project_feasible(cand)
        if time.time() < deadline - 3.0:
            cand = slsqp_alpha(cand, 120000.0, 70)
            cand = project_feasible(cand)
        cc = c5_of(cand)
        if cc < best_c5:
            best_h = cand.copy()
            best_c5 = cc
        restarts += 1

    best_h = project_feasible(best_h)
    best_c5 = c5_of(best_h)
    return best_h, float(best_c5), int(best_n)
