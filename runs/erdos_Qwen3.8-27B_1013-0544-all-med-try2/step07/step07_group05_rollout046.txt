def project_feasible(h, n):

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return float(m + lse / alpha)

def _cons_fun(x, n):
    return x.sum() - n / 2.0

def _cons_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def sa_optimize(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    cur_c = best_c
    t = 0.005
    target_rate = 0.35
    accept = 0
    total = 0
    batch = 300

    while time.time() < deadline:
        for _ in range(batch):
            if time.time() >= deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if rng.rand() < 0.5:
                max_d = min(1.0 - cur_h[i], cur_h[j])
                if max_d < 1e-8:
                    continue
                d = min(rng.uniform(0.001, 0.2), max_d)
                cur_h[i] += d
                cur_h[j] -= d
            else:
                max_d = min(cur_h[i], 1.0 - cur_h[j])
                if max_d < 1e-8:
                    continue
                d = min(rng.uniform(0.001, 0.2), max_d)
                cur_h[i] -= d
                cur_h[j] += d
            cur_h = np.clip(cur_h, 0.0, 1.0)
            diff = n / 2.0 - cur_h.sum()
            if abs(diff) > 1e-12:
                cur_h += diff / n
            new_c = compute_c5(cur_h, n)
            total += 1
            if new_c <= cur_c:
                cur_c = new_c
                accept += 1
                if new_c < best_c:
                    best_c = new_c
                    best_h = cur_h.copy()
            else:

                    cur_c = new_c

            if rate > target_rate + 0.08:

                t = min(0.05, t * 1.4)
            accept = 0
            total = 0
        if t < 0.00008:

    return best_h, best_c

def near_pair_search(h, n, deadline, deltas, radius=60):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:

                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c
