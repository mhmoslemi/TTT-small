import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_onto_slice(h, n):
    target = n / 2.0
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    ac = alpha * (c - m)
    ac_s = ac - ac.max()
    exp_ac = np.exp(ac_s)
    Z = exp_ac.sum()
    p = exp_ac / Z
    obj = m + (ac.max() + np.log(Z)) / alpha
    
    idx = np.arange(n)
    omh = 1.0 - h
    M1 = idx[None, :] - idx[:, None] + (n - 1)
    M2 = idx[:, None] - idx[None, :] + (n - 1)
    T1 = (p[M1] * omh[None, :]).sum(axis=1)
    T2 = (p[M2] * h[None, :]).sum(axis=1)
    grad = dx * (T1 - T2)
    return obj, grad

def projected_gradient(h, n, alpha, lr, iters, deadline=None):
    h = h.copy()
    idx = np.arange(n)
    M1 = idx[None, :] - idx[:, None] + (n - 1)
    M2 = idx[:, None] - idx[None, :] + (n - 1)
    dx = 2.0 / n
    best_obj = compute_c5(h, n)
    for it in range(iters):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        ac = alpha * (c - m)
        ac_s = ac - ac.max()
        exp_ac = np.exp(ac_s)
        Z = exp_ac.sum()
        p = exp_ac / Z
        omh = 1.0 - h
        T1 = (p[M1] * omh[None, :]).sum(axis=1)
        T2 = (p[M2] * h[None, :]).sum(axis=1)
        grad = dx * (T1 - T2)
        
        # Backtracking line search
        step = lr
        obj0 = compute_c5(h, n)
        for _ in range(20):
            h_new = project_onto_slice(h - step * grad, n)
            obj_new = compute_c5(h_new, n)
            if obj_new < obj0:
                h = h_new
                break
            step *= 0.5
        else:
            break
    return h

def coord_descent(h, n, max_sweeps=200, deadline=None):
    # Same as before but simplified
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(40):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h
