   def project_feasible(h, n):
       """Force sum(h)=n/2 and clips to [0,1]"""
       target = n/2
       h_feas = h.copy()
       diff = target - np.sum(h_feas)
       room = np.clip(1-h_feas, 0, 1e9)  # Projection space
       if diff==0: return h_feas
       s_val = np.sum(room)
       if abs(diff) < 1e-10 * s_val: return h_feas
       h_feas += diff/s_val * room
       return project_feasible(h_feas.clip(0,1), n)
