import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=500, deadline=None, random_order=False, rng=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    order = np.arange(n)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        if random_order and rng is not None:
            rng.shuffle(order)
        for i in order:
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range

def slsqp_smooth(h, n, alpha, maxiter, ftol, deadline=None):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        if sm > 400:
            return m + len(g) / alpha
        ex = np.exp(shifted - sm)
        return m + (sm + np.log(ex.sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h

def greedy_binary(h_init, n, deadline, max_sweeps=300):

                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)

                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:

                delta = np.zeros(2 * n - 1)

                msk = (im >= 0) & (im < n)

    return best_h, best_c
