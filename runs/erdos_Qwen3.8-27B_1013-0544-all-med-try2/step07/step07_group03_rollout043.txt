from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n//2]] = 1.0
    return hb

def obj_and_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    w = exp_s / exp_s.sum()
    f = m + (sm + np.log(exp_s.sum())) / alpha
    
    # Add penalty for sum constraint
    s = x.sum() - n / 2.0
    f += penalty * s * s
    
    # Gradient
    j_idx = np.arange(n).reshape(n, 1)
    lags = np.arange(-(n-1), n).reshape(1, 2*n-1)
    valid1 = (j_idx + lags >= 0) & (j_idx + lags < n)
    idx1 = np.clip(j_idx + lags, 0, n-1)
    x1 = x[idx1]
    T1 = (w * (1.0 - x1) * valid1).sum(axis=1)
    
    valid2 = (j_idx - lags >= 0) & (j_idx - lags < n)
    idx2 = np.clip(j_idx - lags, 0, n-1)
    x2 = x[idx2]
    T2 = (w * x2 * valid2).sum(axis=1)
    
    grad = dx * (T1 - T2) + 2.0 * penalty * s
    return f, grad

def lbfgs_smooth(h, n, alpha, penalty, maxiter=1000, time_limit=None):
    dx = 2.0 / n
    def fun(x):
        return obj_and_grad(x, n, alpha, penalty)[0]
    def g_fun(x):
        return obj_and_grad(x, n, alpha, penalty)[1]
    
    bounds = [(0.0, 1.0)] * n
    t_start = time.time()
    try:
        res = minimize(fun, h, jac=g_fun, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h_init, n, deadline, n_random_tries=50000, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_set = set(int(x) for x in np.where(h == 1)[0])
    zeros_set = set(int(x) for x in np.where(h == 0)[0])
    
    for _ in range(n_random_tries):
        if time.time() > deadline:
            break
        i = zeros_set[rng.randint(len(zeros_set))]  # random zero
        j = ones_set[rng.randint(len(ones_set))]    # random one
        if i == j:
            continue
        
        # Compute delta
        delta = np.zeros(2*n-1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += (1.0 - h[im[msk]])
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= (1.0 - h[jm[msk]])
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n-1] -= 2.0
        k1 = n-1 + (j-i)

        k2 = n-1 + (i-j)

            best_c, best_h = c, h.copy()

    # Phase 3: Binary search after smoothing
    p3 = min(t0 + 320, dl - 80)
    hb, cb = greed_binary(best_h, n, p3, max_sweeps=200)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 4: Annealing for structural improvement
    p4 = min(t0 + 500, dl - 60)
    ha, ca = anneal_binary(best_h, n, p4, seed=seed, T0=0.015, cool=0.9997)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()

    # Phase 5: High-alpha L-BFGS from best binary
    p5 = min(t0 + 650, dl - 50)
    h = to_binary(best_h, n).copy()
    for a in [100000, 500000, 2000000]:
        if time.time() > p5 - 8:
            break
        h = lbfgs_opt(h, n, a, 500.0, maxiter=300)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 6: Multi-resolution
    p6 = min(t0 + 900, dl - 40)
    for nh in [500, 300, 600]:
        if time.time() > p6 - 25:
            break
        if nh % 2 != 0:
            nh += 1
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        p_sub = min(dl - 20, time.time() + 25)
        hq, cq = greed_binary(hh, nh, p_sub, max_sweeps=60)
        if cq < compute_c5(hh, nh):
            hh = hq
        for a in [5000, 50000, 500000]:
            if time.time() > dl - 15:
                break
            hh = lbfgs_opt(hh, nh, a, 500.0, maxiter=200)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh

    # Phase 7: Final alcohol - short greeds + noise
    final_dl = dl - 2
    if time.time() < final_dl:
        hf, cf = greed_binary(best_h, n, final_dl, max_sweeps=100)
        if cf < best_c:
            best_c, best_h = cf, hf.copy()
    # Noise restart
    if time.time() < final_dl - 10:
        h = best_h + 0.003 * rng.randn(n)
        h = project_feasible(h, n)
        for a in [5000000, 20000000]:
            if time.time() > final_dl - 3:
                break
            h = lbfgs_opt(h, n, a, 500.0, maxiter=200)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
