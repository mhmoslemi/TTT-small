import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, deadline, max_passes=200, anneal_temp=0.0):
    dx = 2.0 / n
    result = np.correlate(h, 1-h, 'full').astype(float)
    h = h.copy().astype(float)
    best_h = h.copy()
    best_max = result.max()
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        any_change = False
        for i in range(n):
            # Compute d[m] = ∂result[m]/∂h[i]
            d = np.zeros(2*n-1)
            # Term 1: for m in [max(0,i-n+1), i], d[m] = 1-h[i-m]
            lo1 = max(0, i - n + 1)
            hi1 = i
            if hi1 >= lo1 and hi1 < 2*n-1:
                # h[i-m] for m=lo1..hi1 => h[i-hi1..i-lo1] reversed
                d[lo1:hi1+1] = 1.0 - h[i-hi1:i-lo1+1][::-1]
            # Term 2: for m in [0, min(n-1-i, 2n-2)], d[m] += -h[i+m]
            hi2 = min(n-1-i, 2*n-2)
            if hi2 >= 0:
                d[0:hi2+1] -= h[i:i+hi2+1]
            
            c = result - d * h[i]
            
            # Bisection to find optimal h[i] in [0,1]
            lo_v, hi_v = 0.0, 1.0
            for _ in range(25):
                mid = (lo_v + hi_v) / 2.0
                vals = c + d * mid
                max_idx = np.argmax(vals)
                if d[max_idx] > 1e-14:
                    hi_v = mid
                elif d[max_idx] < -1e-14:
                    lo_v = mid
                else:
                    lo_v = hi_v = mid
                    break
            new_val = (lo_v + hi_v) / 2.0
            
            # Optional annealing
            if anneal_temp > 1e-10:
                old_max = result.max()
                new_max_candidate = (c + d * new_val).max()
                if new_max_candidate > old_max:
                    if np.random.random() < np.exp(-(new_max_candidate - old_max) / anneal_temp):
                        pass  # accept
                    else:
                        continue  # reject
                # If new_val == old_val, no change needed
                if abs(new_val - h[i]) < 1e-12:
                    continue
            
            if abs(new_val - h[i]) > 1e-12:
                result += d * (new_val - h[i])
                h[i] = new_val
                any_change = True
                new_max = result.max()
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
        
        if not any_change:
            break
    
    return best_h, best_max

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

            if s < 1e-14:

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):

        m = g.max()

        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

    d = np.zeros(2*n-1)

    # L(k) = max(0, n-1-k), R(k) = min(n-1, 2n-2-k)
    # i in [L(k), R(k)] means:
    #   i >= n-1-k → k >= n-1-i
    #   i <= 2n-2-k → k <= 2n-2-i
    #   k >= 0, k <= 2n-2

    lo1 = max(0, n-1-i)

                    best_h = h.copy()
        
        if not any_change:

        rng = np.random.RandomState(42)
    h = h.copy().astype(float)
    result = np.correlate(h, 1-h, 'full').astype(float)
    dx = 2.0 / n
    best_h = h.copy()
    best_max = result.max()
    
    indices = np.arange(n)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        if p > 0:
            rng.shuffle(indices)
        any_change = False
        for idx_i in range(n - 1):
            i = indices[idx_i]
            j = indices[idx_i + 1]
            
            # Sum constraint: h[i] + h[j] = c
            c = h[i] + h[j]
            if c < 1e-14 or c > 2 - 1e-14:
                continue  # Can't move both in the same direction
            
            d_i = compute_deriv(h, i, n)
            d_j = compute_deriv(h, j, n)
            dd = d_i - d_j  # combined derivative w.r.t. t = h[i]

            # The linear part is already in dd[n-1], the quadratic is -2*δ_i²
            
            old_i = h[i]
            lo_t = max(0.0, c - 1.0)
            hi_t = min(1.0, c)
            
            if hi_t - lo_t < 1e-12:
                continue
            
            for _ in range(25):
                mid = (lo_t + hi_t) / 2.0
                delta = mid - old_i
                new_r = result + dd * delta
                new_r[n-1] -= 2.0 * delta * delta
                new_max = new_r.max()
                max_idx = np.argmax(new_r)
                deriv = dd[max_idx] - (4.0 * delta if max_idx == n-1 else 0.0)
                # Wait: derivative of -2*delta² w.r.t. t is -4*delta
                if deriv > 1e-12:

```python
def compute_deriv_vectorized(h, i, n):
    d = np.zeros(2*n-1)
    # Contribution 1: k in [max(0, n-1-i), min(2n-2, 2n-2-i)]
    lo1 = max(0, n-1-i)
    hi1 = min(2*n-2, 2*n-2-i)
    if hi1 >= lo1:
        # d[k] += 1 - h[i+k-n+1] for k in [lo1, hi1]
        # i+k-n+1 for k=lo1: i+lo1-n+1
        # i+k-n+1 for k=hi1: i+hi1-n+1
        start = i + lo1 - n + 1
        end = i + hi1 - n + 1 + 1
        d[lo1:hi1+1] += 1.0 - h[start:end]
    
    # Contribution 2: k in [max(0, i), min(2n-2, i+n-1)]
    lo2 = max(0, i)
    hi2 = min(2*n-2, i+n-1)
    if hi2 >= lo2:
        # d[k] -= h[i-k+n-1] for k in [lo2, hi2]
        # i-k+n-1 for k=lo2: i-lo2+n-1
        # i-k+n-1 for k=hi2: i-hi2+n-1
        # As k increases, i-k+n-1 decreases
        start = i - lo2 + n - 1
        end = i - hi2 + n - 1
        d[lo2:hi2+1] -= h[end:start+1][::-1]
    
    return d
```

This uses numpy slicing which is O(1) in Python (the actual work is O(n) but vectorized in C). Much faster!

Let me verify the slicing:
For contribution 1: k goes from lo1 to hi1. The index into h is i+k-n+1.
- At k=lo1: idx = i+lo1-n+1
- At k=hi1: idx = i+hi1-n+1
So h[start:end] where start = i+lo1-n+1, end = i+hi1-n+1+1. ✓

For contribution 2: k goes from lo2 to hi2. The index into h is i-k+n-1.
- At k=lo2: idx = i-lo2+n-1
- At k=hi2: idx = i-hi2+n-1
Since k increases, idx decreases. So we need h[i-lo2+n-1], h[i-lo2+n-1 - 1], ..., h[i-hi2+n-1].
This is h[i-hi2+n-1 : i-lo2+n-1 + 1][::-1].
start_rev = i-hi2+n-1, end_rev = i-lo2+n-1+1
d[lo2:hi2+1] -= h[start_rev:end_rev][::-1] ✓
