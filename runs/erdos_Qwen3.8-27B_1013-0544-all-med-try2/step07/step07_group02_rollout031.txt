                if pos_B.any():

                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-14:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def slsqp_smooth(h, n, alpha, maxiter=5000, ftol=1e-18):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        M = g.max()
        shifted = alpha * (g - M)
        sm = shifted.max()
        return M + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def binary_greedy_swap(h, n, deadline):
    """Try all single swaps (move a 1 from one position to another) to improve."""
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    dx = 2.0 / n
    cur_c = corr.max() * dx
    ones = list(np.where(h == 1)[0])
    for _ in range(50):
        if time.time() > deadline:
            break
        improved = False
        best_delta = None
        best_pair = None
        best_nc = cur_c
        for j in ones:
            for i in range(n):
                if h[i] == 1:
                    continue
                # Try swap: h[j]→0, h[i]→1
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                if i != j:
                    delta[n - 1 + (j - i)] += 1.0
                    delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_delta = delta
                    best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i], h[j] = 1.0, 0.0
        corr += best_delta
        cur_c = best_nc
        ones.remove(j)
        ones.append(i)
    return h, cur_c

One key improvement: in the coordinate descent, I can use a **conjugate direction** or **Gauss-Southwell** rule to select coordinates in a smarter order.

Another improvement: instead of binary search with 40 iterations, I can use an analytical solution.

For the 1D minimax problem:
min_x max_k (c_k + B_k * x)

where c_k = corr[k] - B[k]*h[i] (after recentering)

The optimal x is where the upper envelope of the lines {c_k + B_k * x} is minimized over x ∈ [0,1].

The upper envelope is a piecewise linear convex function. Its minimum occurs at a "kink" point where two lines cross, or at a boundary (x=0 or x=1).

The kink points are at x = (c_j - c_k) / (B_k - B_j) for pairs (j,k) with B_k ≠ B_j.

To find the minimum, we can:
1. Sort the lines by slope (B_k)
2. Build the upper envelope (convex hull in dual space)
3. Find the minimum of the envelope

This is the "linear programming in 1D" problem, solvable in O(n log n) time.

But for n=400 (2n-1 = 799 lines), the bisection with 40 iterations (each O(n) for the max) is O(40n) = O(32000), while the analytical solution is O(n log n) = O(8000). Not a huge difference.

Let me instead focus on a different angle: **using the structure at the optimum**.

At the optimum, the set of "active" shifts (those achieving the max correlation) has a special property: they form a minimal "covering" of the constraint space. The dimension of the active set determines the "tightness" of the solution.

**New idea: Use a "barrier" or "logarithmic" penalty**

Instead of softplus, use:
f(h) = -sum_k log(-corr_k + C) for C > max(corr_k)

This is the barrier method for the constraint corr_k ≤ C. As C approaches the optimal value, the solution approaches the true optimum.

But this requires C > max(corr_k), which means we need to start with a valid C and decrease it.

**Bisection on C with barrier method:**
1. Start with C_current = current best
2. Decrease C by a small amount
3. Solve the barrier problem: min -sum log(C - corr_k) s.t. 0≤h≤1, sum(h)=n/2
4. If feasible, update C and repeat
5. If infeasible, increase C and try again

This is essentially a "linear search on C" combined with a smooth optimization.

**Eliminating the equality constraint:**
Let h[n-1] = n/2 - sum_{i=0}^{n-2} h[i]

Then the problem becomes:
min softplus_max(corr(h_1,...,h_{n-1}, n/2 - sum_{i<n-1} h_i), alpha)
s.t. 0 ≤ h[i] ≤ 1 for i < n-1
     0 ≤ n/2 - sum_{i<n-1} h[i] ≤ 1 (i.e., n/2-1 ≤ sum_{i<n-1} h[i] ≤ n/2)

The last constraint is a single linear constraint on the reduced variables. With L-BFGS-B, I'd need to handle this... actually L-BFGS-B only handles bounds, not general linear constraints.

Alternative: use a penalty for the sum constraint:
min softplus_max(...) + mu * (sum(h) - n/2)^2

Increase mu over time. This is the "augmented Lagrangian" approach and works well in practice.

```python
def lbfgs_smooth(h, n, alpha, mu, maxiter=1000):
    dx = 2.0 / n
    m = 2 * n - 1
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        M = g.max()
        shifted = alpha * (g - M)
        sm = shifted.max()
        soft = M + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
        penalty = mu * (x.sum() - n/2.0) ** 2
        return soft + penalty
    
    def jac(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        M = g.max()
        shifted = alpha * (g - M)
        sm = shifted.max()
        w = np.exp(shifted - sm)
        w /= w.sum()
        # Gradient of softplus: sum_k w[k] * dx * B[k, :]
        # B[k, i] = dc[k]/dh[i]
        grad = np.zeros(n)
        for i in range(n):
            m_range = np.arange(-(n-1), n)
            B = np.zeros(m)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += 1.0 - x[k1[mask1]]
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= x[k2[mask2]]
            grad[i] = np.dot(w, B) * dx
        # Add penalty gradient
        grad += 2 * mu * (x.sum() - n/2.0)
        return grad
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_feasible(np.clip(res.x, 0, 1), n)
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def coord_descent(h, n, max_sweeps=500, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    m = 2 * n - 1
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(m)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += 1.0 - h[k1[mask1]]
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            hi_v = max(np.max(corr - B * h[i]), np.max(corr + B * (1.0 - h[i]))) + 1e-9
            lo_v = min(np.max(corr - B * h[i]), np.max(corr + B * (1.0 - h[i]))) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(40):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
