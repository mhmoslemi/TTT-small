import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=200, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_max = corr[~(pos_B | neg_B)].max()
            for _ in range(40):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def lp_linearization(h_prev, n, prev_c):
    """Solve LP: min t s.t. linearized c_k <= t, 0<=h<=1, sum(h)=n/2"""
    m_range = np.arange(-(n-1), n)
    # Compute gradients of each corr[k] w.r.t. h[i]
    # corr[k] for k = 0..2n-2, shift = k - (n-1)
    # ∂corr[k]/∂h[i] = (1-h[i+shift]) if valid - h[i-shift] if valid
    # where shift = k - (n-1)
    
    # For efficiency, compute B[i] for each i:
    # B[k, i] = ∂corr[k]/∂h[i]
    # = (1 - h[i + (k-(n-1))]) * 1[i+(k-(n-1)) in [0,n-1]] 
    #   - h[i - (k-(n-1))] * 1[i-(k-(n-1)) in [0,n-1]]
    
    # This is O(n^2) to compute but we can vectorize.
    # Actually for the LP, we need: A[h] <= t + b where A is (2n-1) x n
    # A[k, i] = ∂corr[k]/∂h[i] evaluated at h_prev
    # b[k] = corr_prev[k] - A[k,:] @ h_prev
    
    # Let's compute A efficiently.
    # For shift s = k - (n-1):
    # A[k,i] = (1 - h[i+s]) * 1[0<=i+s<=n-1] - h[i-s] * 1[0<=i-s<=n-1]
    
    # This is like a difference of shifted versions of (1-h) and h.
    # Let me compute it as: for each s, the vector A[ns, :] where ns = n-1+s
    # A[ns, i] = (1-h[i+s]) if i+s in [0,n-1] else 0, minus h[i-s] if i-s in [0,n-1] else 0
    
    # I'll compute this as two shifted arrays added.
    # shift_plus: (1-h) shifts: A[i] += (1-h[i+s]) for valid i
    # shift_minus: h shifts: A[i] -= h[i-s] for valid i
    
    # Actually, let me just compute the matrix directly for n=400.
    # A is (2n-1) x n = 799 x 400. That's fine.
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        s = k - (n-1)
        # A[k, i] += (1 - h_prev[i+s]) for i where 0 <= i+s <= n-1
        if s >= 0:
            # i+s <= n-1 => i <= n-1-s, and i >= 0
            idx = np.arange(0, n-s)
            A[k, idx] += (1.0 - h_prev[idx + s])
        else:
            # i+s >= 0 => i >= -s, and i <= n-1
            idx = np.arange(-s, n)
            A[k, idx] += (1.0 - h_prev[idx + s])
        
        # A[k, i] -= h_prev[i-s] for i where 0 <= i-s <= n-1
        if s <= 0:
            # i-s >= 0 => i >= s (but s<=0 so i>=0 always), i-s <= n-1 => i <= n-1+s
            if n-1+s >= 0:
                idx = np.arange(0, n+s+1)
                A[k, idx] -= h_prev[idx - s]
        else:
            # i-s >= 0 => i >= s, i-s <= n-1 => i <= n-1+s (but n-1+s > n-1 so i <= n-1)
            idx = np.arange(s, n)
            A[k, idx] -= h_prev[idx - s]
    
    # b[k] = prev_c[k] - A[k,:] @ h_prev
    b = prev_c - A @ h_prev
    
    # LP: min t
    # s.t. A[k,:] @ h - t <= b[k] for all k
    #      0 <= h <= 1
    #      sum(h) = n/2
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    for k in range(2*n-1):
        constraints.append(A[k] @ h_var - t_var <= b[k])
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.GLOP)
    except:
        try:
            prob.solve(solver=cp.CBC)
        except:
            prob.solve(solver=cp.CVXOPT)
    
    if h_var.value is not None:
        return np.clip(h_var.value, 0, 1), t_var.value
    return h_prev, prev_c.max() * (2.0/n)

def build_B(h, n):
    m_range = np.arange(-(n - 1), n)
    B = np.zeros((n, 2 * n - 1))
    for i in range(n):
        k1 = i - m_range
        mask1 = (k1 >= 0) & (k1 < n)
        B[i, mask1] += (1.0 - h[k1[mask1]])
        k2 = i + m_range
        mask2 = (k2 >= 0) & (k2 < n)
        B[i, mask2] -= h[k2[mask2]]
    return B

def update_B(B, h, n, pi, pj, di, dj):
    """Update B after changing h[pi] by di and h[pj] by dj"""
    for i in range(n):
        k1 = n - 1 + pi - i
        if 0 <= k1 <= 2 * n - 2:
            B[i, k1] -= di
        k2 = n - 1 + i - pi
        if 0 <= k2 <= 2 * n - 2:
            B[i, k2] -= di
        k3 = n - 1 + pj - i
        if 0 <= k3 <= 2 * n - 2:
            B[i, k3] -= dj
        k4 = n - 1 + i - pj
        if 0 <= k4 <= 2 * n - 2:
            B[i, k4] -= dj

def binary_best_improve(h, n, deadline, max_iter=300):
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = corr.max()
    B = build_B(h, n)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        best_improve = 0.0
        best_i_idx = -1
        best_j = -1
        for ii, i in enumerate(zeros):
            Ci = corr + B[i]
            D = Ci[None, :] - B[ones, :]
            costs = D.max(axis=1)
            min_cost = costs.min()
            improve = cur_max - min_cost
            if improve > best_improve:
                best_improve = improve
                best_i_idx = ii
                best_j = ones[costs.argmin()]
        if best_i_idx < 0:
            break
        i = zeros[best_i_idx]
        j = best_j
        corr += B[i] - B[j]
        h[i] = 1.0
        h[j] = 0.0
        cur_max = corr.max()
        update_B(B, h, n, i, j, 1.0, -1.0)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return h, corr

def coord_descent(h, n, max_sweeps=500, deadline=None):
    m_range = np.arange(-(n - 1), n)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(n):
            B = np.zeros(2 * n - 1)
            k1 = i - m_range
            mask1 = (k1 >= 0) & (k1 < n)
            B[mask1] += (1.0 - h[k1[mask1]])
            k2 = i + m_range
            mask2 = (k2 >= 0) & (k2 < n)
            B[mask2] -= h[k2[mask2]]
            val0 = np.max(corr - B * h[i])
            val1 = np.max(corr + B * (1.0 - h[i]))
            hi_v = max(val0, val1) + 1e-9
            lo_v = min(val0, val1) - 1e-9
            pos_B = B > 1e-14
            neg_B = B < -1e-14
            zero_B = ~(pos_B | neg_B)
            zero_max = corr[zero_B].max() if zero_B.any() else -np.inf
            for _ in range(45):
                mid = (lo_v + hi_v) / 2.0
                if zero_max > mid + 1e-14:
                    lo_v = mid
                    continue
                lb_x = 0.0
                if neg_B.any():
                    lb_x = max(lb_x, h[i] + np.max((mid - corr[neg_B]) / B[neg_B]))
                ub_x = 1.0
                if pos_B.any():
                    ub_x = min(ub_x, h[i] + np.min((mid - corr[pos_B]) / B[pos_B]))
                if lb_x <= ub_x + 1e-12:
                    hi_v = mid
                else:
                    lo_v = mid
            lb_x = 0.0
            if neg_B.any():
                lb_x = max(lb_x, h[i] + np.max((hi_v - corr[neg_B]) / B[neg_B]))
            ub_x = 1.0
            if pos_B.any():
                ub_x = min(ub_x, h[i] + np.min((hi_v - corr[pos_B]) / B[pos_B]))
            x_opt = np.clip((lb_x + ub_x) / 2.0, 0.0, 1.0)
            if abs(x_opt - h[i]) > 1e-15:
                corr += B * (x_opt - h[i])
                h[i] = x_opt
    return h, corr

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary best-improvement (new approach)
    p1 = t0 + 0.12 * total
    hb = to_binary(best_h, n)
    hb, _ = binary_best_improve(hb, n, p1, max_iter=300)
    cb = compute_c5(hb, n)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Coordinate descent from best
    p2 = t0 + 0.30 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=3000, deadline=p2)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 3: SLSQP progressive smoothing
    p3 = t0 + 0.55 * total
    h = best_h.copy()
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000,
              10000000, 50000000, 200000000, 1000000000]
    for alpha in alphas:
        if time.time() > p3 - 5:
            break
        h = slsqp_smooth(h, n, alpha, 5000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Coordinate descent again
    p4 = t0 + 0.70 * total
    hc, _ = coord_descent(best_h, n, max_sweeps=3000, deadline=p4)
    hc = project_feasible(hc, n)
    cc = compute_c5(hc, n)
    if cc < best_c:
        best_c, best_h = cc, hc.copy()

    # Phase 5: Binary best-improvement again on current best
    p5 = t0 + 0.78 * total
    hb = to_binary(best_h, n)
    hb, _ = binary_best_improve(hb, n, p5, max_iter=200)
    cb = compute_c5(hb, n)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
        hc, _ = coord_descent(best_h, n, max_sweeps=500, deadline=p5 + 0.02 * total)
        hc = project_feasible(hc, n)
        cc = compute_c5(hc, n)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()

    # Phase 6: More SLSQP from latest
    p6 = t0 + 0.85 * total
    h = best_h.copy()
    for alpha in [5000000, 50000000, 500000000, 5000000000]:
        if time.time() > p6 - 3:
            break
        h = slsqp_smooth(h, n, alpha, 5000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 7: Perturbation restarts
    p7 = t0 + 0.95 * total
    for trial in range(5):
        if time.time() > p7 - 8:
            break
        scale = 0.02 / (trial + 1)
        h = best_h + scale * rng.randn(n)
        h = project_feasible(h, n)
        hc, _ = coord_descent(h, n, max_sweeps=800, deadline=p7 - 5)
        hc = project_feasible(hc, n)
        cc = compute_c5(hc, n)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()
        for alpha in [50000000, 500000000]:
            if time.time() > p7 - 2:
                break
            hc = slsqp_smooth(hc, n, alpha, 3000, 1e-18)
            cc = compute_c5(hc, n)
            if cc < best_c:
                best_c, best_h = cc, hc.copy()

    # Phase 8: Try other resolutions
    if time.time() < dl - 25:
        for nh in [200, 300, 500, 600]:
            if time.time() > dl - 20:
                break
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, nh)
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            hh, _ = coord_descent(hh, nh, max_sweeps=1000, deadline=dl - 15)
            hh = project_feasible(hh, nh)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh
            if time.time() < dl - 12:
                hh = slsqp_smooth(hh, nh, 50000000, 3000, 1e-18)
                hh = project_feasible(hh, nh)
                ch = compute_c5(hh, nh)
                if ch < best_c:
                    best_c, best_h, n = ch, hh.copy(), nh

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
