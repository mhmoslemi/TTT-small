    lp_calls=0
    prev_best=best_c
    for step_scale in steps:
        if time.time() > deadline: break
        gap=0.25
        stall=0
        perturb=0
        improvements=0
        for it in range(700):
            if time.time() > deadline-1: break
            cvec=np.correlate(h,1.0-h,'full')
            cur=float(cvec.max())
            if cur < best_c - 1e-12:
                best_c=cur; best_h=h.copy()
            idx,w=active_weights(cvec,gap,min(30,2*n-1))
            grad=weighted_grad(h,idx,w,n)
            grad -= float(np.mean(grad))
            nrm=float(np.linalg.norm(grad))
            moved=False
            if nrm > 1e-12:
                delta=-grad/nrm*step_scale
                hs1,c1=line_search(h,delta,S,n,cur)
                if hs1 is not None:
                    h=hs1; cur=c1; moved=True
            if moved:
                improvements += 1; stall=0
            else:
                stall += 1
                if stall > 12:
                    if lp_calls < 12 and time.time() < deadline-3:
                        delta=lp_direction(h,cvec,n,max(1e-5,step_scale*1.5),14)
                        if delta is not None:

                    if stall > 25 and perturb < 20 and time.time() < deadline-3:

import time
import numpy as np
from scipy.optimize import linprog


def _init_lag_convention():
    h = np.array([1.0, 1.0, 0.0, 0.0])
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c[4]) > 0.5


_LAG_POS_RIGHT = _init_lag_convention()


def lag_from_index(ii, n):
    off = int(ii) - (n - 1)
    if _LAG_POS_RIGHT:
        return off
    return -off


def project(y, S):
    y = np.asarray(y, dtype=float)
    y = np.nan_to_num(y, nan=0.0, posinf=1.0, neginf=0.0)
    if y.size == 0:
        return y

    if float(np.min(y)) >= -1e-8 and float(np.max(y)) <= 1.0 + 1e-8:
        x = np.clip(y, 0.0, 1.0)
        for _ in range(3):
            diff = S - float(x.sum())
            if abs(diff) < 1e-12:
                break
            if diff > 0.0:
                mask = x < 1.0 - 1e-12
            else:
                mask = x > 1e-12
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            x[mask] += diff / float(cnt)
            np.clip(x, 0.0, 1.0, out=x)
        if abs(S - float(x.sum())) <= 1e-8:
            return x

    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - float(x.sum())
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            x[mask] += diff / float(cnt)
        else:
            x += diff / float(x.size)
        np.clip(x, 0.0, 1.0, out=x)
    return x


def bound_c(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / float(n)))


def active_weights(cvec, gap, max_active):
    cmax = float(np.max(cvec))
    idx = np.flatnonzero(cvec >= cmax - gap)
    if idx.size == 0:
        idx = np.array([int(np.argmax(cvec))])
    if idx.size > max_active:
        idx = idx[np.argsort(cvec[idx])[::-1]][:max_active]
    else:
        idx = idx[np.argsort(cvec[idx])[::-1]]
    g = cmax - cvec[idx]
    scale = max(gap, 1e-9)
    w = np.exp(-4.0 * g / scale)
    s = float(w.sum())
    if s <= 1e-300:
        w = np.full(idx.size, 1.0 / float(idx.size))
    else:
        w = w / s
    return idx, w


def weighted_grad(h, idx, w, n):
    grad = np.zeros(n)
    center = n - 1
    for a in range(idx.size):
        ww = float(w[a])
        if ww == 0.0:
            continue
        m = lag_from_index(idx[a], n)
        if m == 0:
            grad += ww * (1.0 - 2.0 * h)
        else:
            d = abs(m)
            ln = n - d
            if m > 0:
                grad[:ln] += ww * (1.0 - h[d:])
                grad[d:] -= ww * h[:ln]
            else:
                grad[d:] += ww * (1.0 - h[:ln])
                grad[:ln] -= ww * h[d:]
    return grad


def line_search(h, delta, S, n, c0):
    best_c = float(c0)
    best_h = None
    for a in (1.0, 0.6, 0.4, 0.25, 0.125, 0.0625):
        h1 = project(h + a * delta, S)
        c1 = float(np.max(np.correlate(h1, 1.0 - h1, mode='full')))
        if c1 < best_c:
            best_c = c1
            best_h = h1
    if best_h is not None and best_c < float(c0) - 1e-10:
        return best_h, best_c
    return None, None


def lp_direction(h, cvec, n, max_step, K=18):
    order = np.argsort(cvec)[::-1]
    if order.size == 0:
        return None
    cmax = float(np.max(cvec))
    rows = []
    b = []
    for ii in order[:K]:
        m = lag_from_index(ii, n)
        g = np.zeros(n)
        if m == 0:
            g = 1.0 - 2.0 * h
        else:
            d = abs(m)
            ln = n - d
            if m > 0:
                g[:ln] += 1.0 - h[d:]
                g[d:] -= h[:ln]
            else:
                g[d:] += 1.0 - h[:ln]
                g[:ln] -= h[d:]
        if float(np.max(np.abs(g))) < 1e-14:
            continue
        row = np.empty(n + 1)
        row[0] = -1.0
        row[1:] = g
        rows.append(row)
        b.append(-float(cvec[ii]))
    if not rows:
        return None

    A = np.vstack(rows)
    b = np.array(b)
    bounds = [(cmax - 100.0, cmax + 1e-9)]
    for i in range(n):
        xi = float(h[i])
        lo = max(-max_step, -xi - 1e-12)
        hi = min(max_step, 1.0 - xi - 1e-12)
        if lo > hi:
            if xi <= 0.0 or xi >= 1.0:
                lo = 0.0
                hi = 0.0
            else:
                lo = min(-xi, max_step)
                hi = min(1.0 - xi, max_step)
                if lo > hi:
                    lo = hi = 0.0
        bounds.append((float(lo), float(hi)))

    Aeq = np.ones((1, n + 1))
    Aeq[0, 0] = 0.0
    c = np.zeros(n + 1)
    c[0] = 1.0
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=[0.0],
                      bounds=bounds, options={'disp': False})
    except Exception:
        return None
    if not res.success:
        return None
    delta = res.x[1:]
    if float(np.linalg.norm(delta)) < 1e-8:
        return None
    return delta


def smooth_noise(n, rng):
    x = rng.normal(size=n)
    k = 9
    y = np.convolve(x, np.ones(k) / float(k), mode='same')
    y -= float(np.mean(y))
    sd = float(np.std(y))
    if sd > 1e-12:
        y /= sd
    return y


def resample_h(h, n):
    m = int(h.size)
    if m == n:
        return h.copy()
    xo = np.linspace(0.0, 2.0, m)
    xn = np.linspace(0.0, 2.0, n)
    return np.interp(xn, xo, h)


def rotation_start(n, alpha, beta):
    x = (np.arange(n, dtype=float) * alpha) % 1.0
    h = (x < beta).astype(float)
    return project(h, 0.5 * float(n))


def random_fourier_start(n, rng):
    x = np.linspace(0.0, 2.0, n)
    h = np.full(n, 0.5)
    for f in range(1, 7):
        amp = 0.25 / np.sqrt(float(f))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h = h + amp * np.sin(np.pi * f * x + ph)
    return project(h, 0.5 * float(n))


def make_binary(h, n):
    target = n // 2
    b = np.zeros(n)
    if target > 0:
        idx = np.argpartition(h, n - target)[n - target:]
        b[idx] = 1.0
    return b


def active_polish(h0, n, deadline, rng):
    S = 0.5 * float(n)
    h = project(h0, S)
    cvec = np.correlate(h, 1.0 - h, mode='full')
    cur = float(cvec.max())
    best_h = h.copy()
    best_c = cur
    steps = (0.05, 0.02, 0.01, 0.004, 0.0015, 0.0007, 0.00025)
    lp_calls = 0
    prev_best = best_c

    for step_scale in steps:
        if time.time() > deadline:
            break
        gap = 0.25
        stall = 0
        perturb = 0
        improvements = 0
        for _ in range(700):
            if time.time() > deadline - 1.0:
                break
            cvec = np.correlate(h, 1.0 - h, mode='full')
            cur = float(cvec.max())
            if cur < best_c - 1e-12:
                best_c = cur
                best_h = h.copy()

            idx, w = active_weights(cvec, gap, min(80, 2 * n - 1))
            grad = weighted_grad(h, idx, w, n)
            grad -= float(np.mean(grad))
            nrm = float(np.linalg.norm(grad))
            moved = False

            if nrm > 1e-12:
                delta = -grad / nrm * step_scale
                hs1, c1 = line_search(h, delta, S, n, cur)
                if hs1 is not None:
                    h = hs1
                    cur = c1
                    moved = True

            if moved:
                improvements += 1
                stall = 0
                continue

            stall += 1
            if stall > 12 and lp_calls < 6 and time.time() < deadline - 3.0:
                delta = lp_direction(h, cvec, n, max(1e-5, step_scale * 1.5), 18)
                if delta is not None:
                    lp_calls += 1
                    hs1, c1 = line_search(h, delta, S, n, cur)
                    if hs1 is not None:
                        h = hs1
                        cur = c1
                        moved = True

            if not moved:
                if stall > 25 and perturb < 20 and time.time() < deadline - 3.0:
                    eps = min(0.03, max(0.001, step_scale * 0.5))
                    h = project(best_h + eps * smooth_noise(n, rng), S)
                    cur = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                    gap = min(5.0, gap * 1.8)
                    perturb += 1
                    stall = 0
                elif stall > 80:
                    break

        if best_c >= prev_best - 1e-12 and step_scale <= 0.004:
            break
        prev_best = best_c

    best_h = project(best_h, S)
    bc = float(np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * (2.0 / float(n)))
    return best_h, bc

    for nn in (n, 300):
        for alpha in (alpha_phi, 0.5):
            h0 = rotation_start(nn, alpha, 0.5)
            h1 = project(h0 + 0.01 * smooth_noise(nn, rng), 0.5 * float(nn))
            raw_tasks.append((float(bound_c(h1, nn)), idx, h1, nn))
            idx += 1

    h0 = random_fourier_start(n, rng)
    raw_tasks.append((float(bound_c(h0, n)), idx, h0, n))
    idx += 1

    h0 = make_binary(best_h, n)
    if n % 2 == 0:
        raw_tasks.append((float(bound_c(h0, n)), idx, h0, n))
        idx += 1

    raw_tasks.sort()
    tasks = []
    for item in raw_tasks:
        tasks.append((item[2], item[3]))
    tasks = tasks[:8]

    for ti, job in enumerate(tasks):
        h0, nn = job
        if time.time() > dl - 30.0:
            break
        left = dl - time.time()
        if left < 25.0:
            break
        remaining = max(1, len(tasks) - ti)
        sub = max(15.0, min(60.0, (left / float(remaining)) * 0.55))
        cand_dl = min(dl - 20.0, time.time() + sub)
        hh, cc = active_polish(h0, nn, cand_dl, rng)
        if cc < best_c:
            best_h = hh.copy()
            best_c = cc
            n = int(best_h.size)

    if time.time() < dl - 5.0:
        hh, cc = active_polish(best_h, len(best_h), dl - 3.0, rng)
        if cc < best_c:
            best_h = hh.copy()
            best_c = cc
            n = int(best_h.size)

    best_h = project(best_h, 0.5 * float(len(best_h)))
    best_c = float(bound_c(best_h, len(best_h)))
    return best_h, best_c, int(len(best_h))
