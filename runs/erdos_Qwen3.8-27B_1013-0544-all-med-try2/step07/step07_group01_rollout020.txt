import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sm_val(c, alpha):
    mc = c.max()
    sc = alpha * (c - mc)
    if sc.max() > 500:
        return float(mc + (sc.max() + np.log(np.sum(np.exp(sc - sc.max())))) / alpha)
    return float(mc + np.log(np.sum(np.exp(sc))) / alpha)

def _sm_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    mc = c.max()
    sc = alpha * (c - mc)
    mx = sc.max()
    w = np.exp(sc - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _obj_pen(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = _sm_val(c, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def _grad_pen(x, n, alpha, mu):
    g = _sm_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _obj_cons(x, n, alpha):
    return _sm_val(compute_c5_vec(x, n), alpha)

def _grad_cons(x, n, alpha):
    return _sm_grad(x, n, alpha)

def _con_eq(x, n):
    return x.sum() - n / 2.0

def _con_eq_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alpha, mu, deadline, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    t_start = time.time()
    res = minimize(_obj_pen, h, jac=_grad_pen, args=(n, alpha, mu),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                            'maxfun': maxiter * 20, 'maxls': 40})
    h_new = project_feasible(res.x, n)
    return h_new, compute_c5(h_new, n)

def slsqp_opt(h, n, alpha, deadline, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac, 'args': (n,)}]
    res = minimize(_obj_cons, h, jac=_grad_cons, args=(n, alpha),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    h_new = project_feasible(res.x, n)
    return h_new, compute_c5(h_new, n)

def nesterov_gd(h, n, deadline, alpha_seq, lr=0.01, beta=0.9):
    h = project_feasible(h, n)
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in alpha_seq:
        if time.time() > deadline:
            break
        for _ in range(300):
            if time.time() > deadline:
                break
            h_nd = np.clip(h + beta * v, 0.0, 1.0)
            g = _sm_grad(h_nd, n, alpha)
            v = beta * v - lr * g
            h_new = project_feasible(h + v, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            old_c = compute_c5(h, n)
            if c_new > old_c + 0.5 / alpha:
                v = 0.5 * v
                # don't update h
            else:
                h = h_new
    return best_h, best_c

def grad_local_search(h, n, deadline, steps=(0.01, 0.005, 0.002, 0.001, 0.0005)):
    """Gradient-guided local search: only try moves that the gradient suggests."""
    best_c = compute_c5(h, n)
    alpha = 5e5
    for step in steps:
        if time.time() > deadline:
            break
        while time.time() < deadline:
            improved = False
            g = _sm_grad(h, n, alpha)
            # Find the top and bottom gradient positions
            # Move mass from high positive gradient to high negative gradient
            idx_sorted = np.argsort(g)
            # Try a subset of promising pairs
            for trial in range(100):
                if time.time() > deadline:
                    return h, best_c
                i = idx_sorted[np.random.randint(len(idx_sorted))]
                j = idx_sorted[np.random.randint(len(idx_sorted))]
                if i == j:
                    continue
                d = step if g[i] > g[j] else -step
                if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
            if not improved:
                break
    return h, best_c

def _obj_pen(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    return _sm_val(c, alpha) + mu * (x.sum() - n / 2.0) ** 2

def _grad_pen(x, n, alpha, mu):
    return _sm_grad(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0)

def _obj_cons(x, n, alpha):
    return _sm_val(compute_c5_vec(x, n), alpha)

def _grad_cons(x, n, alpha):
    return _sm_grad(x, n, alpha)

def _con_eq(x, n):
    return x.sum() - n / 2.0

def _con_eq_jac(x, n):
    return np.ones(n)

def lbfgs_opt(h, n, alpha, mu, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_pen, h, jac=_grad_pen, args=(n, alpha, mu),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                            'maxfun': maxiter * 25, 'maxls': 40})
    h_new = project_feasible(res.x, n)
    return h_new, compute_c5(h_new, n)

def slsqp_opt(h, n, alpha, maxiter=800):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac, 'args': (n,)}]
    res = minimize(_obj_cons, h, jac=_grad_cons, args=(n, alpha),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    h_new = project_feasible(res.x, n)
    return h_new, compute_c5(h_new, n)

def nesterov_anneal(h, n, deadline, alpha_seq, lr, beta, T_start, T_end, rng):
    h = project_feasible(h, n)
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_cur = best_c
    for alpha in alpha_seq:
        if time.time() > deadline:
            break
        n_steps = 2000
        for step in range(n_steps):
            if time.time() > deadline:
                break
            T = T_start * (T_end / T_start) ** (step / max(1, n_steps))
            h_nd = np.clip(h + beta * v, 0.0, 1.0)
            g = _sm_grad(h_nd, n, alpha)
            v = beta * v - lr * g
            h_prop = project_feasible(h + v, n)
            c_prop = compute_c5(h_prop, n)
            if c_prop <= c_cur or rng.random() < np.exp(-min(50.0, (c_prop - c_cur) / max(T, 1e-8))):
                h = h_prop
                c_cur = c_prop
                if c_prop < best_c:
                    best_c = c_prop
                    best_h = h_prop.copy()
    return best_h, best_c

def fast_local_search(h, n, deadline, rng):
    best_c = compute_c5(h, n)
    alpha = 2e6
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for step in steps:
        if time.time() > deadline:
            break
        while time.time() < deadline:
            improved = False
            g = _sm_grad(h, n, alpha)
            n_tries = min(5000, int((deadline - time.time()) * 200))
            for _ in range(max(1, n_tries)):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = step if g[i] > g[j] else -step
                if 0.0 <= h[i] + d <= 1.0 and 0.0 <= h[j] - d <= 1.0:
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
            if not improved:
                break
    return h, best_c

def optimize_from_start(h0, n, deadline, rng):
    h = project_feasible(h0, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    # Phase 1: L-BFGS-B with increasing alpha
    alphas_lbfgs = [10, 50, 200, 1000, 5000, 20000]
    for a in alphas_lbfgs:
        if time.time() > deadline - 120:
            break
        h_new, c_new = lbfgs_opt(h, n, a, mu=0.1, maxiter=250)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    # Phase 2: Nesterov + annealing
    if time.time() < deadline - 80:
        a_seq = np.geomspace(50000, 5e6, 15)
        h_new, c_new = nesterov_anneal(h, n, min(time.time() + 80, deadline - 50),
                                        a_seq, lr=0.008, beta=0.85,
                                        T_start=0.005, T_end=1e-5, rng=rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
    # Phase 3: SLSQP polish
    if time.time() < deadline - 30:
        for a in [200000, 1e6]:
            if time.time() > deadline - 20:
                break
            h_new, c_new = slsqp_opt(h, n, a, maxiter=600)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
    # Phase 4: Local search
    if time.time() < deadline - 5:
        h_new, c_new = fast_local_search(h, n, deadline - 2, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n = len(h_parent)
    except Exception:
        n = 300
        h_parent = np.full(n, 0.5)

    # Build diverse starting points
    starts = [h_parent]
    # Random perturbation
    starts.append(np.clip(np.full(n, 0.5) + 0.3 * rng.randn(n), 0, 1))
    # Block pattern: alternating blocks
    block_starts = []
    bs = max(1, n // 10)
    bp = np.zeros(n)
    for b in range(0, n, bs * 2):
        bp[b:b+bs] = 1.0
    starts.append(bp)
    # Sinusoidal
    x = np.linspace(0, 2, n)
    starts.append(np.clip(0.5 + 0.4 * np.sin(2 * np.pi * x), 0, 1))

    # Allocate time per start
    n_starts = len(starts)
    time_per_start = (deadline - t0) * 0.7 / n_starts
    final_time = (deadline - t0) * 0.3

    best_h = None
    best_c = 1.0
    best_n = n

    # Optimize from each start
    for s_idx in range(n_starts):
        start_deadline = t0 + time_per_start * (s_idx + 1) + (deadline - t0) * 0.05
        if time.time() > deadline - 30:
            break
        h_s = project_feasible(starts[s_idx], n)
        h_opt, c_opt = optimize_from_start(h_s, n, start_deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            best_n = n

    # Final refinement on best solution
    remaining = deadline - time.time() - 5
    if remaining > 30 and best_h is not None:
        # Try one more SLSQP + local search
        if remaining > 50:
            h_f, c_f = slsqp_opt(best_h, best_n, 2e6, maxiter=1000)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f.copy()
        if time.time() < deadline - 3:
            h_f, c_f = fast_local_search(best_h, best_n, deadline - 2, rng)
            if c_f < best_c:
                best_c = c_f
                best_h = h_f.copy()

    # Try a different n if time allows
    remaining = deadline - time.time() - 5
    if remaining > 120 and best_h is not None:
        n2 = best_n + 50
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n2)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n2)
        h_opt2, c_opt2 = optimize_from_start(h_interp, n2, min(time.time() + remaining - 10, deadline - 5), rng)
        if c_opt2 < best_c:
            best_c = c_opt2
            best_h = h_opt2.copy()
            best_n = n2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
