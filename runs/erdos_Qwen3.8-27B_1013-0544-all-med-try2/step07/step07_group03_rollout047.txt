    corr = np.correlate(h, 1-h, mode='full') * dx
    # Get top M lags

    max_val = corr[top_indices[0]]

        c = np.correlate(x, 1-x, mode='full') * dx
        return c.max()

            c = np.correlate(x, 1-x, mode='full')

def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]

    h_b[idx[:n // 2]] = 1.0

def smooth_max_obj_and_grad(h, n, alpha):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')  # shape (2n-1,)
