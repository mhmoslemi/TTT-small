                if not mask.any():
                    continue

                h_new[i] = target

                mask[i] = False
                if not mask.any():
                    continue

                h_new[i] = target
                h_new[mask] += (-delta) / mask.sum()
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
                improved = True
        if not improved:
            break

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 50:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def smooth_max_grad(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag in range(2 * n - 1):
        if w[lag] < 1e-15:
            continue
        k = lag - (n - 1)
        wm = w[lag] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

                    continue
                j = dests[np.argmin(g[dests])]
                max_d = min(h[i], 1.0 - h[j])
                hi_i, hi_j = h[i], h[j]
                lo, hi = 0.0, max_d
                best_d = 0.0
                for _ in range(14):
                    mid = (lo + hi) / 2.0
                    h[i] = hi_i - mid
                    h[j] = hi_j + mid
                    ct = compute_c5(h, n)
                    if ct < best_c:
                        best_c = ct
                        best_d = mid
                        lo = mid
                    else:
                        hi = mid
                if best_d > 1e-12:
                    h[i] = hi_i - best_d
                    h[j] = hi_j + best_d
                    improved = True
                else:
                    h[i] = hi_i
                    h[j] = hi_j
            if h[i] < 1.0 - 1e-10:
                srcs = np.where(h > 1e-10)[0]
                if len(srcs) == 0:
                    continue
                j = srcs[np.argmax(g[srcs])]
                max_d = min(1.0 - h[i], h[j])
                hi_i, hi_j = h[i], h[j]
                lo, hi = 0.0, max_d
                best_d = 0.0
                for _ in range(14):
                    mid = (lo + hi) / 2.0
                    h[i] = hi_i + mid
                    h[j] = hi_j - mid
                    ct = compute_c5(h, n)
                    if ct < best_c:
                        best_c = ct
                        best_d = mid
                        lo = mid
                    else:
                        hi = mid
                if best_d > 1e-12:
                    h[i] = hi_i + best_d
                    h[j] = hi_j - best_d
                    improved = True
                else:
                    h[i] = hi_i
                    h[j] = hi_j
        if not improved:
            break
    return h, best_c

def binary_swap_search(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    bin_h = np.zeros(n)
    # Threshold to binary
    thresh = 0.5
    bin_h = (h > thresh).astype(float)
    # Fix sum
    target = n // 2
    ones = np.where(bin_h == 1)[0]
    zeros = np.where(bin_h == 0)[0]
    if len(ones) > target:
        excess = len(ones) - target
        for idx in ones[:excess]:
            bin_h[idx] = 0
    elif len(ones) < target:
        deficit = target - len(ones)
        for idx in zeros[:deficit]:
            bin_h[idx] = 1
    c_bin = compute_c5(bin_h, n)
    if c_bin < best_c:
        best_c = c_bin
        h = bin_h.copy()
    # Local search: swap 1s with 0s
    while time.time() < deadline:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        g = smooth_max_grad(h, n, 50000)
        one_order = np.argsort(g[ones])[::-1][:100]
        zero_order = np.argsort(g[zeros])[:100]
        for oi in one_order:
            if time.time() > deadline:
                return h, best_c
            for zi in zero_order[:30]:
                i, j = ones[oi], zeros[zi]
                h[i] = 0
                h[j] = 1
                ct = compute_c5(h, n)
                if ct < best_c - 1e-15:
                    best_c = ct
                    improved = True
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    g = smooth_max_grad(h, n, 50000)
                    break
                else:
                    h[i] = 1
                    h[j] = 0
        if not improved:
            break
    return h, best_c

def quantize_push(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(100):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            if 0.1 < h[i] < 0.9:
                continue
            target = 0.0 if h[i] < 0.5 else 1.0
            delta = target - h[i]
            if abs(delta) < 1e-12:
                continue
            if delta > 0:
                mask = (h > delta) & (h < 1.0 - delta * 1.1)
                mask[i] = False
            else:
                mask = (h < -delta) & (h > delta * 1.1)
                mask[i] = False
            if not mask.any():
                continue
            h_new = h.copy()
            h_new[i] = target
            h_new[mask] += delta / mask.sum()
            h_new = project_feasible(h_new, n)
            ct = compute_c5(h_new, n)
            if ct < best_c - 1e-15:
                best_c = ct
                h = h_new
                improved = True
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)
    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    # Phase 1: L-BFGS-B aggressive
    t1 = min(t0 + 280, deadline - 500)
    h, best_c = lbfgs_phase(h, n, t1)
    best_h = h.copy()
    # Phase 2: Adam
    if time.time() < deadline - 350:
        t2 = min(time.time() + 130, deadline - 350)
        ha, ca = adam_phase(h, n, t2, alpha=500000, lr=0.02)
        if ca < best_c:
            best_c, best_h = ca, ha
    # Phase 3: Binary swap search
    if time.time() < deadline - 200:
        t3 = min(time.time() + 80, deadline - 200)
        hb, cb = binary_swap_search(best_h, n, t3)
        if cb < best_c:
            best_c, best_h = cb, hb
    # Phase 4: Second L-BFGS-B round
    if time.time() < deadline - 150:
        t4 = min(time.time() + 70, deadline - 150)
        h4, c4 = lbfgs_phase(best_h, n, t4)
        if c4 < best_c:
            best_c, best_h = c4, h4
    # Phase 5: Pair bisect
    if time.time() < deadline - 100:
        t5 = min(time.time() + 60, deadline - 100)
        hp, cp = pair_bisect(best_h, n, t5, top_k=200)
        if cp < best_c:
            best_c, best_h = cp, hp
    # Phase 6: Quantize
    if time.time() < deadline - 60:
        t6 = min(time.time() + 30, deadline - 60)
        hq, cq = quantize_push(best_h, n, t6)
        if cq < best_c:
            best_c, best_h = cq, hq
    # Phase 7: Another binary swap if quantize helped
    if time.time() < deadline - 50:
        t7 = min(time.time() + 30, deadline - 50)
        hb2, cb2 = binary_swap_search(best_h, n, t7)
        if cb2 < best_c:
            best_c, best_h = cb2, hb2
    # Phase 8: Final pair bisect
    if time.time() < deadline - 10:
        hf, cf = pair_bisect(best_h, n, deadline - 3, top_k=250)
        if cf < best_c:
            best_c, best_h = cf, hf
    # Phase 9: Try other resolutions
    if time.time() < deadline - 30:
        for n_try in [250, 400, 200, 500, 350]:
            if time.time() > deadline - 15:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
            alloc = min(20, max(5, deadline - time.time() - 10))
            if alloc < 8:
                break
            h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.5)
            if time.time() < deadline - 8:
                h_t, c_t = pair_bisect(h_t, n_try, deadline - 6, top_k=120)
            if c_t < best_c:
                best_c = c_t
                x_t = np.linspace(0, 2, n_try)
                x_n = np.linspace(0, 2, n)
                best_h = project_feasible(np.interp(x_n, x_t, h_t), n)
                best_c = compute_c5(best_h, n)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
