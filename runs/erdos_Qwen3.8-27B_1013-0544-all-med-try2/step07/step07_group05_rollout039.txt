def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    if sm > 500:
        return float(m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def obj_anti(x, n, alpha):
    nh = n // 2
    h = np.empty(n)
    h[:nh] = x
    h[nh:] = 1.0 - x
    return obj_lse(h, n, alpha)

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(len(x))

def slsqp_full(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def slsqp_anti(a, n, alpha, maxiter):

        return h

def grad_c5(h, n):
    """Compute subgradient of C5 w.r.t. h at the active lag."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    k_max = int(c.argmax())
    lag = k_max - (n - 1)
    grad = np.zeros(n)
    idx1 = np.arange(max(0, -lag), min(n, n - lag))
    grad[idx1] += (1.0 - h[idx1 + lag])
    idx2 = np.arange(max(0, lag), min(n, n + lag))
    grad[idx2] -= h[idx2 - lag]
    return grad * dx

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)

def directed_pair_search(h, n, deadline, deltas):

    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            g = grad_c5(h, n)
            # Sort indices by gradient: most positive first (decrease), most negative first (increase)
            order = np.argsort(g)
            # Try pairs: (high grad -> decrease, low grad -> increase)
            n_try = min(n, 80)
            inc_idx = order[:n_try]  # most negative gradient -> increase
            dec_idx = order[n-n_try:]  # most positive gradient -> decrease
            for i in dec_idx:
                if time.time() > deadline:
                    break
                for j in inc_idx:
                    if i == j:
                        continue
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            # Also try reverse: (low grad -> increase, high grad -> decrease)
            for i in inc_idx:
                if time.time() > deadline:
                    break
                for j in dec_idx:
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
