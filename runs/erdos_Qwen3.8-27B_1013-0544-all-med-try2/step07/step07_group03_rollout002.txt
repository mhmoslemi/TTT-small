def project_feasible(h, target=None):
    h = np.asarray(h, dtype=float)
    n = h.size
    if target is None: target=0.5*n
    if h.size==0: return h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(30):
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15: break
            h = h + (diff / s) * h
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            return h
    # bisection fallback
    lo, hi = -1.0, 1.0
    for _ in range(50):
       mid=0.5*(lo+hi)
       val=np.clip(h-mid,0,1).sum()
       if val > target: lo=mid
       else: hi=mid
    h=np.clip(h-0.5*(lo+hi),0,1)
    diff=target-h.sum()
    if abs(diff)>1e-12:
       if diff>0: j=int(np.argmax(1.0-h)); h[j]=min(1.0, h[j]+diff)
       else: j=int(np.argmax(h)); h[j]=max(0.0, h[j]+diff)
    return h
```

```python
import numpy as np
import time
from scipy.optimize import minimize

_FAR = {}
_OPT = {'alpha': 1.0, 'target': 0.0, 'mu': 0.0, 'one': None}

def _pw2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def _fft_cache(n):
    if n not in _FAR:
        N = _pw2(3 * n + 5)
        L = n - 1
        idx = (np.arange(2 * n - 1) - L) % N
        _FAR[n] = (N, idx)
    return _FAR[n]

def fft_grad(h, w):
    n = h.size
    N, idx = _fft_cache(n)
    W = np.zeros(N)
    W[idx] = w
    fW = np.fft.rfft(W, N)
    g1 = np.fft.irfft(np.fft.rfft(1.0 - h, N) * np.conj(fW), N)
    g2 = np.fft.irfft(np.fft.rfft(h, N) * fW, N)
    g = (g1 - g2)[:n]
    if not np.all(np.isfinite(g)):
        g = np.zeros(n)
    return g

def smooth_val(h, alpha):
    h = np.clip(h, 0.0, 1.0)
    vals = np.correlate(h, 1.0 - h, 'full')
    if vals.size == 0:
        return 0.0, np.zeros_like(h)
    if not np.isfinite(vals.min()):
        return 1e3, np.zeros_like(h)
    m = float(vals.max())
    sh = np.minimum(alpha * (vals - m), 0.0)
    e = np.exp(sh)
    s = e.sum()
    if s <= 0.0 or not np.isfinite(s):
        return m, np.zeros_like(h)
    w = e / s
    g = fft_grad(h, w)
    return m + np.log(s) / alpha, g

def c5_of(h):
    n = h.size
    if n == 0:
        return 1.0
    v = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(v) * 2.0 / n)

def project_feasible(h, target=None):
    h = np.asarray(h, float)
    n = h.size
    if target is None:
        target = 0.5 * n
    if n == 0:
        return h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(30):
        if diff > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if diff > 1e-10:
        j = int(np.argmax(1.0 - h))
        h[j] = min(1.0, h[j] + diff)
    elif diff < -1e-10:
        j = int(np.argmax(h))
        h[j] = max(0.0, h[j] + diff)
    return h

def to_binary(h):
    n = h.size
    k = n // 2
    b = np.zeros(n)
    if k > 0:
        idx = np.argsort(h)[::-1][:k]
        b[idx] = 1.0
    return b

def resample(h, nn):
    n = h.size
    if n == nn:
        return h.copy()
    xo = (np.arange(n) + 0.5) * 2.0 / n
    xin = (np.arange(nn) + 0.5) * 2.0 / nn
    return project_feasible(np.interp(xin, xo, h))

def _set_opt(n, alpha, mu=0.0):
    _OPT['alpha'] = alpha
    _OPT['target'] = 0.5 * n
    _OPT['mu'] = mu
    if _OPT['one'] is None or _OPT['one'].size != n:
        _OPT['one'] = np.ones(n)

def pen_obj(x):
    h = np.clip(x, 0.0, 1.0)
    f, g = smooth_val(h, _OPT['alpha'])
    err = h.sum() - _OPT['target']
    if _OPT['mu'] != 0.0:
        f += 0.5 * _OPT['mu'] * err * err
        g = g + _OPT['mu'] * err * _OPT['one']
    if not np.isfinite(f):
        f = 1e3
        g = np.zeros_like(h)
    return f, g

def slq_obj(x):
    h = np.clip(x, 0.0, 1.0)
    return smooth_val(h, _OPT['alpha'])

def slq_eq(x):
    return _OPT['target'] - x.sum()

def slq_jac(x):
    return -_OPT['one']

def lbfgs_step(h, alpha, mu, maxiter, ftol=1e-14):
    h = project_feasible(h)
    n = h.size
    _set_opt(n, alpha, mu)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(pen_obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        y = res.x
    except Exception:
        return h
    return project_feasible(y)

def slsqp_step(h, alpha, maxiter, ftol=1e-13):
    h = project_feasible(h)
    n = h.size
    _set_opt(n, alpha, 0.0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slq_eq, 'jac': slq_jac}]
    try:
        res = minimize(slq_obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        y = res.x
    except Exception:
        return h
    return project_feasible(y)

def polish(h, deadline, alphas, maxiter_lb=120, maxiter_sq=40, do_sq=False,
           mu_base=30.0, mu_rate=0.015):
    best = h.copy()
    bestc = c5_of(best)
    if not np.isfinite(bestc):
        bestc = 1e9
    for a in alphas:
        if time.time() > deadline:
            break
        x = lbfgs_step(best, a, mu_base + mu_rate * a, maxiter_lb)
        c = c5_of(x)
        if np.isfinite(c) and c < bestc - 1e-15:
            best, bestc = x.copy(), c
        if do_sq and time.time() < deadline - 2.0:
            x = slsqp_step(best, a, maxiter_sq)
            c = c5_of(x)
            if np.isfinite(c) and c < bestc - 1e-15:
                best, bestc = x.copy(), c
    return best, bestc

def perturb(h, rng, scale, do_bin, nsw):
    n = h.size
    if do_bin:
        x = to_binary(h)
    else:
        x = h.copy()
    if scale > 0.0:
        x = x + scale * rng.standard_normal(n)
    if nsw > 0:
        a = rng.randint(0, n, nsw)
        b = rng.randint(0, n, nsw)
        m = a != b
        if np.any(m):
            a = a[m]
            b = b[m]
            t = x[a].copy()
            x[a] = x[b]
            x[b] = t
    return project_feasible(x)

def random_binary(h, rng, scale):
    n = h.size
    b = np.zeros(n)
    k = n // 2
    if k > 0:
        idx = rng.choice(n, k, replace=False)
        b[idx] = 1.0
    if scale > 0.0:
        b = b + scale * rng.standard_normal(n)
    return project_feasible(b)

def diverse_search(h0, c0, rng, deadline):
    n = h0.size
    best = h0.copy()
    bestc = c0 if np.isfinite(c0) else 1e9
    for cand in (1.0 - best, best[::-1]):
        if time.time() > deadline:
            break
        x, c = polish(cand, min(time.time() + 30.0, deadline), [100.0, 600.0], 90, 40, False)
        if c < bestc:
            best, bestc = x.copy(), c
    it = 0
    scales = (0.008, 0.020, 0.050, 0.100, 0.200)
    while time.time() < deadline:
        it += 1
        if it % 9 == 0:
            cand = random_binary(best, rng, 0.04)
        elif it % 5 == 0:
            cand = project_feasible(to_binary(best) + 0.04 * rng.standard_normal(n))
        else:
            cand = perturb(best, rng, scales[it % 5], rng.random() < 0.2, 3)
        x, c = polish(cand, min(time.time() + 15.0, deadline), [120.0, 700.0], 70, 0, False)
        if c < bestc:
            best, bestc = x.copy(), c
            x, c = polish(x, min(time.time() + 40.0, deadline), [300.0, 1500.0], 100, 40, True)
            if c < bestc:
                best, bestc = x.copy(), c
    return best, bestc

def active_step(h, deadline, step=0.04, maxiter=2500):
    best = h.copy()
    bestc = c5_of(best)
    cur = bestc
    if not np.isfinite(bestc):
        return best, bestc
    no = 0
    for _ in range(maxiter):
        if time.time() > deadline:
            break
        vals = np.correlate(h, 1.0 - h, 'full')
        m = float(vals.max())
        mask = vals >= m - 0.02
        if not np.any(mask):
            mask[int(np.argmax(vals))] = True
        w = np.zeros_like(vals)
        w[mask] = 1.0 / np.count_nonzero(mask)
        g = fft_grad(h, w)
        acc = False
        for fac in (1.0, 0.5, 0.25):
            y = project_feasible(h - step * fac * g)
            cy = c5_of(y)
            if np.isfinite(cy) and cy < cur - 1e-15:
                h, cur, acc = y, cy, True
                if cur < bestc:
                    bestc, best = cur, y.copy()
                break
        if acc:
            step = min(0.30, step * 1.08)
            no = 0
        else:
            step *= 0.5
            if step < 1e-8:
                break
            no += 1
            if no > 80:
                break
    return best, bestc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(5.0, float(budget_s) - 10.0)
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None or h0.size < 10 or not np.all(np.isfinite(h0)):
        n0 = 400
        h0 = np.zeros(n0)
        idx = rng.choice(n0, n0 // 2, replace=False)
        h0[idx] = 1.0

    h0 = project_feasible(h0)
    n = int(h0.size)
    if n > 800:
        n = 500
        h0 = resample(h0, n)

    best_h = h0.copy()
    best_c = c5_of(best_h)
    if not np.isfinite(best_c):
        best_c = 1e9
        best_h = project_feasible(h0)

    dl1 = t0 + 0.12 * total
    if time.time() < dl1 - 3.0:
        best_h, best_c = polish(best_h, dl1, [50.0, 200.0, 700.0, 2500.0], 120, 40, True)

    dl2 = t0 + 0.27 * total
    for nh in (320, 256, 480, 560):
        if time.time() > dl2 - 5.0:
            break
        if nh == n or nh < 80 or nh > 700:
            continue
        hh = resample(best_h, nh)
        tl2 = min(time.time() + 45.0, dl2 - 3.0)
        if time.time() >= tl2:
            continue
        hl, cl = polish(hh, tl2, [80.0, 400.0, 1600.0], 100, 35, True)
        if np.isfinite(cl) and cl < best_c:
            best_h, best_c, n = hl.copy(), cl, nh

    dl3 = t0 + 0.80 * total
    if time.time() < dl3 - 3.0:
        best_h, best_c = diverse_search(best_h, best_c, rng, dl3)

    n = int(best_h.size)
    if time.time() < dl - 8.0:
        best_h, best_c = polish(best_h, dl, [800.0, 3000.0, 10000.0], 100, 50, True)

    if time.time() < dl - 2.0:
        best_h, best_c = active_step(best_h, dl - 1.0)

    best_h = project_feasible(best_h)
    best_c = c5_of(best_h)
    if not np.isfinite(best_c):
        best_h = h0.copy()
        best_c = c5_of(best_h)

    return best_h, float(best_c), int(best_h.size)
