                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def binary_projection(h, n):
    """Project to nearest binary sequence with n/2 ones."""
    indices = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[indices[:n//2]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, deadline):
    """Local search on binary sequence: swap 0s and 1s."""
    best_c = compute_c5(h_bin, n)
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    for d in [1]:  # binary swaps only
        improved_round = True
        while improved_round and time.time() < deadline:
            improved_round = False
            for i in ones.copy():
                if time.time() > deadline:
                    break
                for j in zeros.copy():
                    if h_bin[i] != 1.0 or h_bin[j] != 0.0:
                        continue
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
                    c = compute_c5(h_bin, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved_round = True
                    else:
                        h_bin[i] = 1.0
                        h_bin[j] = 0.0
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
    return h_bin, best_c
