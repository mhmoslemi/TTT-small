def make_precomp(n):
    offsets = np.arange(-(n-1), n, dtype=int)
    rows = np.arange(n)[:, None]
    pos1 = (rows + offsets[None, :]).astype(int)
    pos2 = (rows - offsets[None, :]).astype(int)
    mask1 = (pos1 >= 0) & (pos1 < n)
    mask2 = (pos2 >= 0) & (pos2 < n)
    return pos1, mask1, pos2, mask2

def lbfgs_opt(h_in, n, alpha, mu, maxiter, deadline, precomp=None):
    dx = 2.0 / n
    target = n / 2.0
    if precomp is None:
        precomp = make_precomp(n)
    pos1, mask1, pos2, mask2 = precomp
    ncorr = 2 * n - 1
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        sh = alpha * (corr - m)
        ms = sh.max()
        return m + (ms + np.log(np.exp(sh - ms).sum())) / alpha + 0.5 * mu * (x.sum() - target)**2
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        sh = alpha * (corr - m)
        ms = sh.max()
        exps = np.exp(sh - ms)
        w = exps / exps.sum()
        h1 = np.zeros((n, ncorr))
        h1[mask1] = x[pos1[mask1]]
        h2 = np.zeros((n, ncorr))
        h2[mask2] = x[pos2[mask2]]
        g_sm = dx * (w[None, :] * (1.0 - h1 - h2)).sum(axis=1)
        return g_sm + mu * (x.sum() - target)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_in, method='L-BFGS-B', bounds=bounds, jac=grad,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h_in

def binary_optimize(h_in, n, deadline, max_sweeps=2000, seed=42):
    """Efficient binary local search."""
    target = n // 2
    dx = 2.0 / n
    ncorr = 2 * n - 1
    
    # Start from binary approximation
    idx = np.argsort(h_in)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    g = np.correlate(h, 1 - h, mode='full') * dx
    cur_c = g.max()
    
    # Precompute V: V[p, k] = h[p+k_s] + h[p-k_s] where k_s = k - (n-1)
    def compute_V(h):
        V = np.zeros((n, ncorr))
        for k in range(ncorr):
            ks = k - (n - 1)
            pos_f = np.arange(n) + ks
            pos_b = np.arange(n) - ks
            mf = (pos_f >= 0) & (pos_f < n)
            mb = (pos_b >= 0) & (pos_b < n)
            V[mf, k] = h[pos_f[mf]]
            V[mb, k] += h[pos_b[mb]]
        return V
    
    V = compute_V(h)
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    for restart in range(5):
        if time.time() > deadline:
            break
        if restart > 0:
            # Perturb: random swaps
            for _ in range(rng.randint(3, 10)):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    a = rng.choice(ones)
                    b = rng.choice(zeros)
                    h[a], h[b] = 0.0, 1.0
            g = np.correlate(h, 1 - h, mode='full') * dx
            cur_c = g.max()
            V = compute_V(h)
        
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            # Find best swap: minimize max(g + dx*(V[j] - V[i]))
            # Vectorized: compute for all (i,j) pairs
            # diff[i,j,k] = g[k] + dx*(V[j,k] - V[i,k])
            # We want: min over (i,j) of max_k diff[i,j,k]
            
            # Memory: len(zeros) * len(ones) * ncorr could be large
            # For n=250: 125 * 125 * 499 = 7.8M floats = 62MB. OK.
            
            # Compute: for each i in zeros, j in ones:
            # new_max[i,j] = max_k (g[k] + dx*(V[j,k] - V[i,k]))
            
            # = max_k (g[k] - dx*V[i,k] + dx*V[j,k])
            # Let A[i,k] = g[k] - dx*V[i,k]  (shape: nz x ncorr)
            # Let B[j,k] = dx*V[j,k]         (shape: no x ncorr)
            # new_max[i,j] = max_k (A[i,k] + B[j,k])
            
            nz = len(zeros)
            no = len(ones)
            
            A = g[None, :] - dx * V[zeros, :]  # (nz, ncorr)
            B = dx * V[ones, :]  # (no, ncorr)
            
            # new_max[i,j] = max_k (A[i,k] + B[j,k])
            # This is a "max-plus" product, which is O(nz * no * ncorr)
            # For n=250: 125*125*499 ≈ 7.8M. With numpy:
            
            # We can do: A_expanded = A[:, None, :] + B[None, :, :]  # (nz, no, ncorr)
            # new_max = A_expanded.max(axis=2)
            # But this uses nz*no*ncorr memory = 7.8M * 8 = 62MB. OK for n=250.
            
            # Actually, let's be smarter. We only care about swaps that improve.
            # A swap improves iff max_k (g[k] + dx*(V[j,k] - V[i,k])) < cur_c
            # i.e., for all k: dx*V[j,k] < cur_c - g[k] + dx*V[i,k]
            # Let margin[k] = cur_c - g[k] (all >= 0)
            # Condition: V[j,k] < margin[k]/dx + V[i,k] for all k
            
            # For the BEST improvement, we want to minimize max_k(g[k] + dx*(V[j,k]-V[i,k]))
            # which is the same as minimizing max_k (g[k] - dx*V[i,k] + dx*V[j,k])
            
            # Let's just do the full computation for n=250
            if nz * no * ncorr < 50_000_000:
                # Full computation
                AM = A[:, None, :]  # (nz, 1, ncorr)
                BM = B[None, :, :]  # (1, no, ncorr)
                new_max = (AM + BM).max(axis=2)  # (nz, no)
            else:
                # Fallback: iterate over zeros
                new_max = np.full((nz, no), np.inf)
                for ii in range(nz):
                    new_max[ii, :] = (A[ii][None, :] + B).max(axis=1)
            
            # Find the best swap
            flat_idx = np.argmin(new_max)
            ii, jj = np.unravel_index(flat_idx, (nz, no))
            
            if new_max[ii, jj] < cur_c - 1e-15:
                i = zeros[ii]
                j = ones[jj]
                h[i], h[j] = 1.0, 0.0
                # Update g
                g = g + dx * (V[j, :] - V[i, :])
                cur_c = g.max()
                # Update V
                # V[p,k] changes where p+k_s = i or p-k_s = i (h[i] went 0->1: +1)
                # and where p+k_s = j or p-k_s = j (h[j] went 1->0: -1)
                _update_V(V, h, n, i, j, +1, -1)
            else:
                break
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c

def _update_V(V, h, n, i, j, delta_i, delta_j):

    # and when p+k_s = j or p-k_s = j (contribution of h[j])

            A = g[None, :] - dx * V[zeros_idx, :]  # (nz, ncorr)
            B = dx * V[ones_idx, :]  # (no, ncorr)
            
            # new_max[i,j] = max_k (A[i,k] + B[j,k])
            # For memory efficiency with n=250:
            # A is (125, 499), B is (125, 499)
            # Full: (125, 125, 499) = 7.8M elements = 62MB. OK.
            
            new_max = (A[:, None, :] + B[None, :, :]).max(axis=2)
            
            flat = np.argmin(new_max)
            ii, jj = np.unravel_index(flat, new_max.shape)
            
            if new_max[ii, jj] < cur_c - 1e-14:
                i, j = zeros_idx[ii], ones_idx[jj]
                g = g + dx * (V[j, :] - V[i, :])
                cur_c = g.max()
                h[i], h[j] = 1.0, 0.0
                update_V(V, i, j, 1.0, -1.0)
            else:
                break
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c
