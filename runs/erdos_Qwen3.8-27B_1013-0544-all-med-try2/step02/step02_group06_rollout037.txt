import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            if d > 0:
                h[int(np.argmax(h))] = min(1.0, h.max() + d)
            else:
                h[int(np.argmin(h))] = max(0.0, h.min() + d)
            continue
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(x - m, -30.0, 0.0)
    return m + np.log(np.sum(np.exp(beta * x_shift))) / beta

def lag_grad(h, n, s):
    g = np.zeros(n)
    for j in range(n):
        if 0 <= j + s < n:
            g[j] += (1.0 - h[j + s])
        if 0 <= j - s < n:
            g[j] += (-h[j - s])
    return g

def optimize_candidate(h_init, n, ts, bnds, budget_s, t0, min_remaining=50):
    """Optimize a single candidate, return (h, c5)."""
    h = project_box_sum(h_init, ts)
    c5 = compute_c5(h)
    dx = 2.0 / n

    def obj(h, beta, lam):
        c = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(c, beta) * dx + lam * (h.sum() - ts) ** 2

    def rem():
        return budget_s - (time.time() - t0)

    # Progressive smooth_max annealing
    for beta in [30, 100, 300, 1000, 3000]:
        if rem() < min_remaining:
            break
        lam = beta * 4
        try:
            r = minimize(obj, h, args=(beta, lam),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 2000, 'ftol': 1e-17, 'gtol': 1e-15})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < c5:
                h, c5 = h_new, c
        except Exception:
            pass

    # Worst-lag gradient with line search
    for it in range(8000):
        if rem() < min_remaining:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        g = lag_grad(h, n, s)
        improved = False
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001]:
            h_try = project_box_sum(h - alpha * g, ts)
            c_try = compute_c5(h_try)
            if c_try < c5 - 1e-15:
                h, c5 = h_try, c_try
                improved = True
                break
        if not improved:
            # Try top-3 combined
            top3 = np.argsort(corr)[-3:]
            g3 = np.zeros(n)
            for kk in top3:
                si = int(kk) - (n - 1)
                g3 += lag_grad(h, n, si) * 0.4
            for alpha in [0.01, 0.005, 0.002]:
                h_try = project_box_sum(h - alpha * g3, ts)
                c_try = compute_c5(h_try)
                if c_try < c5 - 1e-15:
                    h, c5 = h_try, c_try
                    improved = True
                    break
            if not improved:
                h += np.random.randn(n) * 0.004
                h = project_box_sum(h, ts)

    # High beta polish
    for beta in [5000, 15000]:
        if rem() < min_remaining // 3:
            break
        try:
            r = minimize(obj, h, args=(beta, beta * 5),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 8000, 'ftol': 1e-18, 'gtol': 1e-16})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < c5:
                h, c5 = h_new, c
        except Exception:
            pass

    return h, c5


def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
    n = 70
    ts = n / 2.0

    def rem():
        return budget_s - (time.time() - t0)

    # Initial candidates
    candidates = []
    if 'initial_h_values' in globals() and initial_h_values is not None:

    # Binary random starts
    for _ in range(4):
        h_bin = np.zeros(n)

        candidates.append(h_bin)

    # Sinusoidal starts
    x = np.linspace(0, 2, n, endpoint=False)
    for w, amp in [(3 * np.pi, 0.3), (5 * np.pi, 0.25), (7 * np.pi, 0.2)]:
        for ph in [0, np.pi / 2]:
            h_s = 0.5 + amp * np.sin(w * x + ph)
            candidates.append(project_box_sum(np.clip(h_s, 0, 1), ts))

    # Allocate time: first candidate (initial) gets more time
    best_h = candidates[0]
    best_c5 = np.inf

    # Optimize each candidate with decreasing time budget
    time_alloc = [200, 120, 100, 80, 70, 60, 50, 40, 35, 30, 25, 20]
    for i, cand in enumerate(candidates):

            break
        alloc = time_alloc[min(i, len(time_alloc) - 1)]

        if c_cand < best_c5:

                                         t0 + rem() + 100, t0, min_remaining=30)

    if rem() > 40:
        dx = 2.0 / n
        try:
            def obj_slsqp(h):

            h_new = project_box_sum(r.x, ts)

        except Exception:

            r = minimize(obj_r, h_r, method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 1500, 'ftol': 1e-17})
            h_new = project_box_sum(r.x, ts)

                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Final worst-lag polish
    for it in range(3000):
        if rem() < 5:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        s = int(np.argmax(corr)) - (n - 1)
        g = lag_grad(best_h, n, s)
        improved = False
        for alpha in [0.005, 0.002, 0.001, 0.0005, 0.0002]:

            c_try = compute_c5(h_try)

                improved = True
                break
        if not improved and it % 500 == 0:
            best_h += np.random.randn(n) * 0.002
            best_h = project_box_sum(best_h, ts)

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
