    bnds = [(0.0, 1.0)] * n
    for beta in [100, 300, 800, 2000, 5000]:
        if budget_s - (time.time() - t0) < 60:
            break
        try:
            r = minimize(obj_func, best_h, args=(n, ts, beta, 3000.0),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 3000, 'ftol': 1e-16})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 3: Upsample to 2n and optimize
    n2 = 2 * n
    ts2 = n2 / 2.0
    x_lo = (np.arange(n) + 0.5) * 2.0 / n
    x_hi = (np.arange(n2) + 0.5) * 2.0 / n2
    h_up = np.interp(x_hi, x_lo, best_h)
    h_up = project_box_sum(h_up, ts2)
    c_up = compute_c5(h_up)

    if c_up < best_c5 and budget_s - (time.time() - t0) > 120:
        bnds2 = [(0.0, 1.0)] * n2
        for beta in [300, 1000, 3000]:
            if budget_s - (time.time() - t0) < 60:
                break
            try:
                r = minimize(obj_func, h_up, args=(n2, ts2, beta, 3000.0),
                             method='L-BFGS-B', bounds=bnds2,
                             options={'maxiter': 3000, 'ftol': 1e-16})
                h_new = project_box_sum(r.x, ts2)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 4: Diverse restarts
    nc = len(best_h)
    tsc = nc / 2.0
    bndsc = [(0.0, 1.0)] * nc
    restart = 0
    while budget_s - (time.time() - t0) > 100:
        restart += 1
        h = best_h.copy()
        strat = restart % 7

        if strat == 0:
            h += np.random.randn(nc) * 0.04
        elif strat == 1:
            h = (h > 0.5).astype(float)
            while h.sum() > tsc:
                idx = np.where(h == 1)[0]
                h[np.random.choice(idx)] = 0
            while h.sum() < tsc:
                idx = np.where(h == 0)[0]
                h[np.random.choice(idx)] = 1
            h += np.random.randn(nc) * 0.03
        elif strat == 2:
            w = np.random.uniform(2.0, 8.0)
            ph = np.random.uniform(0.0, 2.0 * np.pi)
            h += 0.04 * np.sin(w * np.arange(nc) / nc * 2.0 * np.pi + ph)
        elif strat == 3:
            g, _ = subgradient_worst(h, nc)
            h -= 0.01 * g
            h += np.random.randn(nc) * 0.01
        elif strat == 4:
            blen = np.random.randint(3, max(4, nc // 6))
            b1 = np.random.randint(0, nc - blen)
            b2 = np.random.randint(0, nc - blen)
            if abs(b1 - b2) >= blen:
                tmp = h[b1:b1+blen].copy()
                h[b1:b1+blen] = h[b2:b2+blen]
                h[b2:b2+blen] = tmp
        elif strat == 5:
            # Multi-lag subgradient
            corr = np.correlate(h, 1.0 - h, mode="full")
            top_k = np.argsort(corr)[-3:]
            for kk in top_k:
                ks = int(kk) - (nc - 1)
                g = np.zeros(nc)
                for jj in range(nc):
                    v = 0.0
                    i1 = jj + ks
                    i2 = jj - ks
                    if 0 <= i1 < nc:
                        v += (1.0 - h[i1])
                    if 0 <= i2 < nc:
                        v -= h[i2]
                    g[jj] = v
                h -= 0.005 * g
        else:
            # Quantize to 5 levels and perturb
            h = np.round(h * 4.0) / 4.0
            h += np.random.randn(nc) * 0.015

        h = project_box_sum(h, tsc)

        if budget_s - (time.time() - t0) > 60:
            try:
                r = minimize(obj_func, h, args=(nc, tsc, 1000, 2000.0),
                             method='L-BFGS-B', bounds=bndsc,
                             options={'maxiter': 800, 'ftol': 1e-15})
                h_new = project_box_sum(r.x, tsc)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
                    nc = len(best_h)
                    tsc = nc / 2.0
                    bndsc = [(0.0, 1.0)] * nc
            except Exception:
                pass

    # Phase 5: Final high-precision polish
    nf = len(best_h)
    tsf = nf / 2.0
    bndsf = [(0.0, 1.0)] * nf
    for beta in [8000, 15000, 30000]:
        if budget_s - (time.time() - t0) < 12:
            break
        try:
            r = minimize(obj_func, best_h, args=(nf, tsf, beta, 8000.0),
                         method='L-BFGS-B', bounds=bndsf,
                         options={'maxiter': 20000, 'ftol': 1e-18})
            h_new = project_box_sum(r.x, tsf)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
