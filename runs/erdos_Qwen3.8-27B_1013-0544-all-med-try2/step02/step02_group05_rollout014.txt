import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(c, beta):
    m = np.max(c)
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta

def project_exact(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    m = np.max(c)
    w = np.exp(beta * (c - m))
    w /= w.sum()
    grad = np.zeros(n)
    omt = 1.0 - h
    nm1 = n - 1
    for j in range(n):
        k_lo = max(-nm1, j - nm1)
        k_hi = min(nm1, j)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            grad[j] += np.dot(w[ks + nm1], omt[j - ks])
        k_lo = max(-nm1, -j)
        k_hi = min(nm1, nm1 - j)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            grad[j] -= np.dot(w[ks + nm1], h[j + ks])
    return grad * dx

def proj_grad_descent(h, target, beta, n_iter, lr, deadline):
    n = len(h)
    h = h.copy()
    cur_val = smooth_max_val(np.correlate(h, 1.0 - h, 'full'), beta) * (2.0 / n)
    best_h = h.copy()
    best_val = cur_val
    stale = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        g = smooth_grad(h, beta)
        step = lr
        improved = False
        for _ in range(15):
            h_new = project_exact(h - step * g, target)
            v = smooth_max_val(np.correlate(h_new, 1.0 - h_new, 'full'), beta) * (2.0 / n)
            if v < cur_val:
                h = h_new
                cur_val = v
                improved = True
                if v < best_val:
                    best_val = v
                    best_h = h.copy()
                    stale = 0
                break
            step *= 0.5
        if not improved:
            stale += 1
            lr *= 0.7
            if stale > 200:
                break
    return best_h, best_val

def interp_to_resolution(h_src, n_new):
    n_src = len(h_src)
    if n_src == n_new:
        return h_src.copy()
    xo = np.linspace(0, 2, n_src, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, xo, h_src, period=2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = 0.5 + 0.3 * np.sin(np.linspace(0, 4 * np.pi, 70, endpoint=False))
        h_parent = project_exact(h_parent, 35.0)

    best_h = h_parent.copy()
    best_c5 = compute_c5(h_parent)
    best_n = len(h_parent)

    def slsqp_pass(h0, n, target, max_iter, beta):
        bounds = Bounds(np.zeros(n), np.ones(n))
        def cf(x):
            return np.sum(x) - target
        cons = [{'type': 'eq', 'fun': cf, 'jac': lambda x: np.ones(n)}]
        def obj(x):
            return smooth_max_val(np.correlate(x, 1.0 - x, 'full'), beta) * (2.0 / n)
        try:
            res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
            return project_exact(res.x, target)
        except Exception:
            return h0

    # Phase 1: SLSQP at parent resolution
    n70 = len(h_parent)
    t70_target = n70 / 2.0
    if time.time() < deadline - 300:
        for beta in [60, 120]:
            if time.time() > deadline - 300:
                break
            h_sl = slsqp_pass(best_h, n70, t70_target, 300, beta)
            c5 = compute_c5(h_sl)
            if c5 < best_c5:
                best_h, best_c5 = h_sl.copy(), c5

    # Phase 2: Projected gradient polishing at parent resolution
    betas_pg = [40, 60, 100, 150, 250, 400]
    for beta in betas_pg:
        if time.time() > deadline - 200:
            break
        h_pg, _ = proj_grad_descent(best_h, t70_target, beta, 50000, 0.05, min(time.time() + 80, deadline - 100))
        c5 = compute_c5(h_pg)
        if c5 < best_c5:
            best_h, best_c5 = h_pg.copy(), c5

    # Phase 3: Try other resolutions
    for n_alt in [60, 80, 90, 110]:
        if time.time() > deadline - 100:
            break
        target_alt = n_alt / 2.0
        h_alt = interp_to_resolution(best_h, n_alt)
        h_alt = project_exact(h_alt, target_alt)
        # Quick SLSQP
        if time.time() < deadline - 80:
            h_alt = slsqp_pass(h_alt, n_alt, target_alt, 150, 80)
            c5 = compute_c5(h_alt)
            if c5 < best_c5:
                best_h, best_c5, best_n = h_alt.copy(), c5, n_alt
        # Projected gradient
        for beta in [60, 120, 200]:
            if time.time() > deadline - 60:
                break
            h_alt, _ = proj_grad_descent(h_alt, target_alt, beta, 30000, 0.04, min(time.time() + 40, deadline - 40))
            c5 = compute_c5(h_alt)
            if c5 < best_c5:
                best_h, best_c5, best_n = h_alt.copy(), c5, n_alt

    # Phase 4: Diverse restarts at best resolution
    if time.time() < deadline - 80:
        n_best = best_n
        t_best = n_best / 2.0
        for restart in range(8):
            if time.time() > deadline - 60:
                break
            scale = 0.08 * (1.0 + 0.3 * np.sin(restart * 1.7))
            h_r = project_exact(best_h + np.random.randn(n_best) * scale, t_best)
            for beta in [50, 100, 200]:
                if time.time() > deadline - 50:
                    break
                h_r, _ = proj_grad_descent(h_r, t_best, beta, 20000, 0.04, min(time.time() + 30, deadline - 40))
                c5 = compute_c5(h_r)
                if c5 < best_c5:
                    best_h, best_c5 = h_r.copy(), c5

    # Phase 5: Final high-beta polish
    n_f = best_n
    t_f = n_f / 2.0
    for beta in [300, 500, 700]:
        if time.time() > deadline - 10:
            break
        best_h, _ = proj_grad_descent(best_h, t_f, beta, 80000, 0.01, min(time.time() + 40, deadline - 5))
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5

    return best_h, best_c5, best_n
