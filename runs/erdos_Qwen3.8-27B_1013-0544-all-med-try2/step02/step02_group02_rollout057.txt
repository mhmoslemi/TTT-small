    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

    target_sum = n / 2.0

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:

            if total < 1e-14:

    shift = k_star - (n - 1)
    grad = np.zeros(n)
    # Term 1: (1-h[i+shift]) for valid i+shift

    if len(idx1) > 0:
        grad[idx1] += (1.0 - h[idx1 + shift])
    # Term 2: -h[i-shift] for valid i-shift
    idx2 = np.arange(max(0, shift), min(n, n + shift))
    if len(idx2) > 0:

    return grad

        if gn < 1e-12:

        direction = -grad / gn

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def compute_all_grads(h, n):
    """Compute gradients of all correlation values w.r.t. h"""
    n_corr = 2 * n - 1
    grads = np.zeros((n_corr, n))
    for k in range(n_corr):
        shift = k - (n - 1)
        i1_lo, i1_hi = max(0, -shift), min(n, n - shift)
        if i1_hi > i1_lo:
            idx = np.arange(i1_lo, i1_hi)
            grads[k, idx] += (1.0 - h[idx + shift])
        i2_lo, i2_hi = max(0, shift), min(n, n + shift)
        if i2_hi > i2_lo:
            idx = np.arange(i2_lo, i2_hi)
            grads[k, idx] -= h[idx - shift]
    return grads

def bundle_iteration(h, n):

    grads = compute_all_grads(h, n)

    b_ub = np.zeros(n_corr)
    for k in range(n_corr):
        g = grads[k]
        c_k = c_vals[k]
