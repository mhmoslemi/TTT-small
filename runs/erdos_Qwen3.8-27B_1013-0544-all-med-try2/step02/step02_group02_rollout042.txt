        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_pen(h, n, alpha, pen):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        f = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        f = m + np.log(np.sum(np.exp(sh))) / alpha
    s = np.sum(h) - n / 2.0
    return float(f + pen * s * s)

def obj_pen_grad(h, n, alpha, pen):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        e = np.exp(sh - mx)
    else:
        e = np.exp(sh)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-18:
            continue
        l = k - (n - 1)
        if l >= 0:
            jo = l
            if jo == 0:
                g += wk * (1.0 - 2.0 * h)
            else:
                g[jo:] += wk * (1.0 - h[:n-jo])
                g[:n-jo] += wk * (-h[jo:])
        else:
            jo = -l
            g[:n-jo] += wk * (1.0 - h[jo:])
            g[jo:] += wk * (-h[:n-jo])
    s = np.sum(h) - n / 2.0
    g += 2.0
