    return m + np.log(np.sum(np.exp(shifted))) / beta

    m = np.max(x)
    shifted = beta * (x - m)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    weights /= weights.sum()
    return weights

    h = h.copy()

        s = np.clip(h + mid, 0, 1).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid

    n = len(h)
    grad = np.zeros(n)
    p = abs(lag)
    if p == 0:
        grad[:] = 1.0 - 2.0 * h

        # grad[i] = -h[i-p] for i in [p, n-1]
        grad[:n-p] += (1.0 - h[p:])
        grad[p:] -= h[:n-p]
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(vals, beta):
    m = np.max(vals)
    shifted = np.clip(beta * (vals - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(vals, beta):
    m = np.max(vals)
    shifted = np.clip(beta * (vals - m), -500, 500)
    w = np.exp(shifted)
    w /= w.sum()
    return w

def corr_grad(h, lag):
    n = len(h)
    grad = np.zeros(n)
    p = abs(lag)
    if p == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag >= 0:
        grad[:n-p] += (1.0 - h[p:])
        grad[p:] -= h[:n-p]
    else:
        grad[p:] += (1.0 - h[:n-p])
        grad[:n-p] -= h[p:]
    return grad

def obj_full(h, n, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, beta) * dx

def grad_full(h, n, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for ki in range(len(corr)):
        if w[ki] < 1e-12:
            continue
        lag = ki - (n - 1)
        grad += w[ki] * corr_grad(h, lag)
    return grad * dx

def obj_top(h, n, dx, beta, top_idx):

        bounds = Bounds(np.zeros(n_points), np.ones(n_points))
        penalty = 1e5

        if len(h_parent) == n_points:
            h0 = h_parent.copy()
        else:
            xo = np.linspace(0, 2, len(h_parent), endpoint=False)
            xn = np.linspace(0, 2, n_points, endpoint=False)
            h0 = np.interp(xn, xo, h_parent, period=2.0)
        h0 = project_h(h0, target_sum)

        curr_h = h0.copy()
        curr_c5 = compute_c5(h0)

        # Phase 1: Full smooth_max with analytical gradient, increasing beta
        for beta in [25, 50, 100, 200, 400]:
            if time.time() > deadline - 15:
                break
            try:
                res = minimize(obj_full_pen, h0,
                               args=(n_points, dx, beta, target_sum, penalty),
                               method='SLSQP', bounds=bounds, jac=True,
                               options={'maxiter': 400, 'ftol': 1e-16})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < curr_c5:
                    curr_h, curr_c5 = h_new.copy(), c5
                    h0 = h_new.copy()
            except Exception:
                pass

        # Phase 2: Top-lag focused
        if time.time() < deadline - 30:
            corr = np.correlate(curr_h, 1.0 - curr_h, mode="full")
            k_top = max(4, n_points // 12)
            top_idx = np.argsort(corr)[::-1][:k_top]

            for beta in [50, 100, 200, 400, 800]:
                if time.time() > deadline - 15:
                    break
                try:
                    res = minimize(obj_top_pen, curr_h,
                                   args=(n_points, dx, beta, target_sum, penalty, top_idx),
                                   method='SLSQP', bounds=bounds, jac=True,
                                   options={'maxiter': 600, 'ftol': 1e-16})
                    h_new = project_h(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < curr_c5:
                        curr_h, curr_c5 = h_new.copy(), c5
                    else:
                        break
                except Exception:
                    pass

            # Re-identify top lags and expand
            if time.time() < deadline - 10:
                corr2 = np.correlate(curr_h, 1.0 - curr_h, mode="full")
                top_idx2 = np.argsort(corr2)[::-1][:k_top + 3]
                try:
                    res = minimize(obj_top_pen, curr_h,
                                   args=(n_points, dx, 500, target_sum, penalty, top_idx2),
                                   method='SLSQP', bounds=bounds, jac=True,
                                   options={'maxiter': 800, 'ftol': 1e-17})
                    h_new = project_h(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < curr_c5:
                        curr_h, curr_c5 = h_new.copy(), c5
                except Exception:
                    pass

        if curr_c5 < best_c5:
            best_h, best_c5, best_n = curr_h.copy(), curr_c5, n_points

    # Final high-precision polish
    if best_h is not None and time.time() < start_time + budget_s - 10:
        n_f = best_n
        ts = n_f / 2.0
        dx_f = 2.0 / n_f
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        corr_f = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_f = np.argsort(corr_f)[::-1][:8]

        for beta in [600, 1200]:
            if time.time() > start_time + budget_s - 5:
                break
            try:
                res = minimize(obj_top_pen, best_h,
                               args=(n_f, dx_f, beta, ts, 1e6, top_f),
                               method='SLSQP', bounds=bounds_f, jac=True,
                               options={'maxiter': 5000, 'ftol': 1e-18})
                h_new = project_h(res.x, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    return best_h, best_c5, best_n
