from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def smooth_max_weights(x, beta):
    """Return softmax weights for smooth max"""
    shifted = beta * (x - np.max(x))
    e = np.exp(shifted)
    return e / e.sum()

def corr_gradient(h, n):
    """Compute gradient of all lags. Returns (2n-1, n) array."""
    grads = np.zeros((2*n-1, n))
    j = np.arange(n)
    for m in range(2*n-1):
        k = m - (n-1)
        if k >= 0:
            # I_1: k ≤ j ≤ n-1
            mask1 = (j >= k) & (j <= n-1)
            # I_2: 0 ≤ j ≤ n-1-k
            mask2 = (j >= 0) & (j <= n-1-k)
            # (1-h[j-k]) for mask1
            idx1 = j[mask1] - k
            grads[m, mask1] = (1.0 - h[idx1])
            # -h[j+k] for mask2
            idx2 = j[mask2] + k
            grads[m, mask2] += -h[idx2]
        else:
            # I_1: 0 ≤ j ≤ n-1+k
            mask1 = (j >= 0) & (j <= n-1+k)
            # I_2: -k ≤ j ≤ n-1+k
            mask2 = (j >= -k) & (j <= n-1+k)
            idx1 = j[mask1] - k
            grads[m, mask1] = (1.0 - h[idx1])
            idx2 = j[mask2] + k
            grads[m, mask2] += -h[idx2]
    return grads

def project_to_box_sum(h, target_sum):
    """Project h onto {x: 0 ≤ x ≤ 1, sum(x) = target_sum} via bisection."""
    h = np.asarray(h, dtype=float)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fix small numerical error
    diff = target_sum - projected.sum()
    if abs(diff) > 1e-12:
        # Add uniformly to non-boundary elements
        free = (projected > 1e-10) & (projected < 1 - 1e-10)
        if free.sum() > 0:
            projected[free] += diff / free.sum()
            projected = np.clip(projected, 0.0, 1.0)
    return projected

    corr = get_corr(h)

    grads = corr_gradient(h, n)  # (2n-1, n)
    # Gradient = dx * sum_k weights[k] * grads[k]
    g = (weights @ grads).astype(float) * dx
    return g
