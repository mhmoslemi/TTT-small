def project_h(h, target_sum):
    esum = np.sum(h)
    diff = target_sum - esum
    h_clipped = np.clip(h, 0.0, 1.0, out=h.copy())
    if abs(diff) < 1e-6:
        return h_clipped
    non_zero = (h_clipped > 1e-6) & (h_clipped < 1.0-1e-6)
    if not np.any(non_zero):
        return h_clipped
    h_adjust = h_clipped[non_zero]
    h_adjust += diff * (non_zero.astype(float).sum() / (h_adjust.sum() if abs(h_adjust.sum()) > 0 else 1))
    return np.clip(h_adjust, 0.0, 1.0)

    if time_left() < 30:  # 30 seconds per attempt
        break
        
    # Apply diversification if stuck
    stuck_count = 0
    if best_c5_val == prev_best:
        stuck_count += 1
        if stuck_count >= 3:  # Restart required
            h_init = np.random.uniform(0.3, 0.7, n_points)
            h_init = project_h(h_init, target_sum)
            best_h = h_init.astype(float)
            prev_best = compute_c5(best_h)
            stuck_count = 0
            continue
    
    # Apply perturbation
    h_perturbed = best_h.copy()
    if attempt % 3 == 0:
        # Clone perturbation
        center = np.argmax(np.abs(h_perturbed))
        radius = min(n_points//5, 30)
        left, right = max(0, center-radius)//2, min(n_points, center+radius)//2
        h_perturbed[left,right] += 0.1 * np.random.randn(1)
    elif attempt % 3 == 1:
        # Core perturbation
        for _ in range(3):
            i, j = np.random.choice(n_points,2,replace=False)
            h_perturbed[i] += 0.08 * np.random.randn()
            h_perturbed[j] -= 0.08 * np.random.randn()
    else:
        # Boundary adjustment
        h_perturbed[0:10] *= 0.8 + 0.4*np.random.rand(10)
        h_perturbed[-10:] *= 0.8 + 0.4*np.random.rand(10)
    
    h_perturbed = project_h(h_perturbed, target_sum)
    
    # Refine with SLSQP
    try:
        def obj_func(x):
            corr = np.correlate(x, (1-x), mode='full')
            return smooth_max(corr, min(beta_val*2, 2000)) * dx
        
        res = minimize(obj_func, h_perturbed, method='SLSQP',
                        bounds=[(0,1) for _ in range(n_points)],
                        constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                        options={'maxiter': 300, 'ftol': 1e-10})
        
        new_h = res.x if res.success else h_perturbed
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5_val:
            best_c5_val = new_c5
            best_h = new_h
            prev_best = best_c5_val
    
    except:
        continue
    
    # Early termination check
    if time_left() < 20:
        break
