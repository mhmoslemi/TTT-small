```python
import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_free = free.sum()
        if n_free == 0:
            # All at bounds, can't project perfectly
            break
        h[free] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, beta):
    """Compute smooth max of overlaps and gradient"""
    n = len(h)
    dx = 2.0 / n
    overlaps = get_overlaps(h)
    m = np.max(overlaps)
    shifts = np.clip(beta * (overlaps - m), -300, 300)
    exps = np.exp(shifts)
    Z = np.sum(exps)
    weights = exps / Z
    smooth_val = m + np.log(Z / len(overlaps)) / beta
    
    # Gradient: sum_k w_k * d(overlap_k)/dh
    # d(overlap_k)/dh[j] = dx * [(1-h[j+k])*1{j+k<n} - h[j-k]*1{j-k>=0}]
    grad = np.zeros(n)
    # Only use top lags for efficiency
    sorted_idx = np.argsort(overlaps)[::-1]
    for idx in sorted_idx[:15]:
        w = weights[idx]
        if w < 1e-6:
            break
        lag = int(idx) - (n - 1)
        k = abs(lag)
        if k == 0:
            lg = 1.0 - 2.0 * h
        else:
            lg = np.zeros(n)
            lg[:n-k] += (1.0 - h[k:])
            lg[k:] -= h[:n-k]
        grad += w * lg
    grad *= dx
    return smooth_val, grad
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(overlaps)
    shifts = np.clip(beta * (overlaps - m), -300, 300)
    exps = np.exp(shifts)
    Z = np.sum(exps)
    weights = exps / Z
    val = m + np.log(Z / len(overlaps)) / beta
    grad = np.zeros(n)
    top = np.argsort(overlaps)[::-1]
    for idx in top[:25]:
        w = weights[int(idx)]
        if w < 1e-7:
            break
        lag = int(idx) - (n - 1)
        k = abs(lag)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
    grad *= dx
    return val, grad


def lbfgs_optimize(h_cur, target, bounds, beta, max_iter=400):
    def obj_grad(x):
        v, g = smooth_max_grad(x, beta)
        cv = np.sum(x) - target
        v += 5000.0 * cv * cv
        g = g + 10000.0 * cv * np.ones_like(x)
        return v, g
    res = minimize(obj_grad, h_cur, method='L-BFGS-B', jac=True,
                   bounds=bounds, options={'maxiter': max_iter, 'gtol': 1e-9})
    return project_feasible(res.x, target)


def slsqp_optimize(h_cur, target, bounds, n_pts, beta=800):
    def s_obj(x):
        v, _ = smooth_max_grad(x, beta)
        return v
    def s_jac(x):
        _, g = smooth_max_grad(x, beta)
        return g
    def s_eq(x):
        return np.sum(x) - target
    def s_eq_j(x):
        return np.ones(n_pts)
    cons = [{'type': 'eq', 'fun': s_eq, 'jac': s_eq_j}]
    res = minimize(s_obj, h_cur, method='SLSQP', jac=s_jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 3000, 'ftol': 1e-14})
    return project_feasible(res.x, target)


def coord_descent(h_cur, target, n_pts, max_iters, rng):
    c5 = compute_c5(h_cur)
    deltas = [0.04, -0.04, 0.02, -0.02, 0.08, -0.08, 0.01, -0.01]
    for it in range(max_iters):
        j = it % n_pts
        best_c = c5
        best_h = h_cur[j]
        for dv in deltas:
            h_t = h_cur.copy()
            h_t[j] = h_cur[j] + dv
            if h_t[j] < 0.0 or h_t[j] > 1.0:
                continue
            h_t = project_feasible(h_t, target)
            c = compute_c5(h_t)
            if c < best_c - 1e-15:
                best_c = c
                best_h = h_cur.copy()
                best_h[j] = h_t[j]
                best_h = project_feasible(best_h, target)
        if best_c < c5:
            h_cur = best_h
            c5 = best_c
    return h_cur, c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 140
        x = np.linspace(0, 2, n0)
        h0 = 0.5 + 0.3 * np.sin(np.pi * x) + 0.15 * np.sin(2 * np.pi * x)
        h0 = project_feasible(h0, n0 / 2)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Resolutions to try
    n_parent = len(h0)
    n_list = [n_parent]
    if n_parent * 2 <= 560:
        n_list.append(n_parent * 2)

    # Initializations to try
    def make_inits(n_pts, target):
        inits = []
        # From current best (interpolated)
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_pts)
        inits.append(project_feasible(np.interp(x_new, x_old, best_h), target))
        # Block pattern
        h_blk = np.zeros(n_pts)
        nb = max(3, n_pts // 20)
        bs = n_pts // nb
        for i in range(nb):
            h_blk[i * bs:(i + 1) * bs] = 0.7 if i % 2 == 0 else 0.3
        if nb * bs < n_pts:
            h_blk[nb * bs:] = 0.3
        inits.append(project_feasible(h_blk, target))
        # Random perturbation of best
        h_rnd = best_h.copy()
        if len(h_rnd) != n_pts:
            x_old2 = np.linspace(0, 2, len(h_rnd))
            h_rnd = np.interp(x_new, x_old2, h_rnd)
        h_rnd += rng.uniform(-0.1, 0.1, n_pts)
        inits.append(project_feasible(h_rnd, target))
        return inits

    for n_pts in n_list:
        if tl() < 40:
            break
        target = n_pts / 2.0
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
        inits = make_inits(n_pts, target)

        for h_init in inits:
            if tl() < 25:
                break
            h_cur = h_init.copy()
            c5_cur = compute_c5(h_cur)
            if c5_cur < best_c5:
                best_h, best_c5 = h_cur.copy(), c5_cur

            # L-BFGS-B with increasing beta
            for beta in [50, 200, 600]:
                if tl() < 10:
                    break
                try:
                    h_new = lbfgs_optimize(h_cur, target, bounds, beta)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur:
                        h_cur, c5_cur = h_new, c5_new
                        if c5_cur < best_c5:
                            best_h, best_c5 = h_cur.copy(), c5_cur
                except Exception:
                    pass

            # SLSQP polish
            if tl() > 15:
                try:
                    h_new = slsqp_optimize(h_cur, target, bounds, n_pts)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur:
                        h_cur, c5_cur = h_new, c5_new
                        if c5_cur < best_c5:
                            best_h, best_c5 = h_cur.copy(), c5_cur
                except Exception:
                    pass

            # Coordinate descent
            if tl() > 20:
                max_cd = min(8000, int(tl() * 800))
                h_cur, c5_cur = coord_descent(h_cur, target, n_pts, max_cd, rng)
                if c5_cur < best_c5:
                    best_h, best_c5 = h_cur.copy(), c5_cur

    # Final SLSQP on best
    if tl() > 20:
        n_f = len(best_h)
        target_f = n_f / 2.0
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        try:
            h_new = slsqp_optimize(best_h, target_f, bounds_f, n_f, beta=1000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
        except Exception:
            pass

    # Final CD polish
    if tl() > 15:
        n_f = len(best_h)
        target_f = n_f / 2.0
        max_cd = min(6000, int(tl() * 600))
        best_h, best_c5 = coord_descent(best_h, target_f, n_f, max_cd, rng)

    return best_h, best_c5, len(best_h)
```
