        bs = np.random.randint(0, n - blen)
        hp[bs:bs + blen] = 0.5 + 0.3 * np.random.randn(blen)
        starts.append(project_box_sum(np.clip(hp, 0, 1), ts))

    for hp in starts:
        if tl() < 35:
            break
        for p in [100, 400]:
            if tl() < 15:
                break
            def obj_r(h):
                c = np.correlate(h, 1.0 - h, mode="full")
                return smooth_max_hybrid(c, p) * (2.0 / n) + 1000.0 * (h.sum() - ts) ** 2
            try:
                r = minimize(obj_r, hp, method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 400, 'ftol': 1e-15})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 6: Final coordinate descent
    if tl() > 30:
        h = _coordinate_descent(best_h.copy(), n, ts, max_iter=150)
        c = compute_c5(h)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 7: Ultra-precise final polish
    if tl() > 20:
        for p in [2000, 5000]:
            if tl() < 10:
                break
            def obj_final(h):
                c = np.correlate(h, 1.0 - h, mode="full")
                return smooth_max_hybrid(c, p) * (2.0 / n) + 10000.0 * (h.sum() - ts) ** 2
            try:
                r = minimize(obj_final, best_h, method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 3000, 'ftol': 1e-18})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 8: Tiny subgradient steps
    if tl() > 10:
        h = best_h.copy()
        step = 0.003
        for it in range(80):
            if tl() < 3:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            k_star = int(np.argmax(corr)) - (n - 1)
            g = _subgrad_for_lag(h, n, k_star)
            g -= g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-10:
                break
            h = project_box_sum(h - step * g / gn, ts)
            c = compute_c5(h)
            if c < best_c5:
                best_h, best_c5 = h.copy(), c
            step *= 0.995

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
