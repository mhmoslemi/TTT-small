            j = (i + 1) % n

            s = target - other_sum

            best_v = h[i]
            best_obj = float('inf')

            grid = np.linspace(s_lo, s_hi, 30)
            objs = []
            for v in grid:
                h[i] = v
                h[j] = s - v
                corr = np.correlate(h, 1.0 - h, mode="full")
                obj = np.max(corr) * (2.0 / n)
                objs.append(obj)
                if obj < best_obj:
                    best_obj = obj
                    best_v = v
            
            # Refine around best grid point
            idx = np.argmin(objs)
            lo = grid[max(0, idx-1)]
            hi = grid[min(len(grid)-1, idx+1)]
            
            h[i] = best_v  # restore
            h[j] = s - best_v
            
            # Fine grid refinement
            fine_grid = np.linspace(lo, hi, 20)
            for v in fine_grid:
                h[i] = v
                h[j] = s - v
                corr = np.correlate(h, 1.0 - h, mode="full")
                obj = np.max(corr) * (2.0 / n)

                    best_obj = obj
                    best_v = v

            h[j] = s - best_v

    """Find min_v max_j (a[j] - b[j]*v) over v in [v_lo, v_hi].
    Returns (v_opt, f_opt)."""
    # Binary search on t
    # For given t, feasible v satisfies:
    #   For b[j] > 0: v >= (a[j] - t) / b[j]
    #   For b[j] < 0: v <= (a[j] - t) / b[j]
    #   For b[j] == 0: a[j] <= t
    #   v_lo <= v <= v_hi

    # Actually, f(v_lo) and f(v_hi) give upper bounds on t.
    t_lo = min(np.min(a) - np.max(np.abs(b)) * (v_hi - v_lo), 0)  # very safe lower bound
    t_hi = min(np.max(a - b * v_lo), np.max(a - b * v_hi))  # upper bound

                            optimize_pair(h_cd, i, j, target, n)
                    c5 = compute_c5(h_cd)
                    if c5 < best_c5:
                        best_h, best_c5, best_n = h_cd.copy(), c5, n_points

            # Phase 4: SLSQP polish
            if time_left() > 20 and best_n == n_points:
                for beta in [100, 200, 400]:
                    if time_left() < 15:
                        break
                    h_s = slsqp_polish(best_h.copy(), target, beta, maxiter=500)
                    c5 = compute_c5(h_s)
                    if c5 < best_c5:
                        best_h, best_c5 = h_s, c5
                        best_n = n_points

    # Final ultra-precise polish
    if best_h is not None and time_left() > 10:
        n_f = best_n
        target_f = n_f / 2.0
        for beta in [500, 800]:
            if time_left() < 8:
                break
            h_s = slsqp_polish(best_h.copy(), target_f, beta, maxiter=1000)
            c5 = compute_c5(h_s)
            if c5 < best_c5:
                best_h, best_c5 = h_s, c5
        # One more round of coordinate descent
        if time_left() > 5:
            best_h = coord_descent(best_h, target_f, n_passes=30, rng=rng)
            c5 = compute_c5(best_h)
            if c5 > best_c5:
                # revert
                pass
            else:
                best_c5 = c5

    if best_h is None and parent is not None:
        best_h = parent
        best_c5 = compute_c5(best_h)
        best_n = len(parent)

    return best_h, best_c5, best_n
