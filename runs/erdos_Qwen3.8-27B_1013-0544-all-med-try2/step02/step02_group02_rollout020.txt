    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-14:

            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def eff_vec(h, p, n):
    """Effect vector for position p: how c changes when h[p] changes by 1 unit."""
    nc = 2 * n - 1
    ef = np.zeros(nc)
    mid = n - 1
    # Forward: h[p] appears as h[i] in c[k] when i=p, k valid
    # c[k] = sum_i h[i] * (1-h[i+k-mid]), so h[p] contributes (1-h[p+k-mid])
    k_lo = max(0, mid - p)
    k_hi = min(nc - 1, mid + (n - 1 - p))
    if k_lo <= k_hi:
        idx = p + np.arange(k_lo, k_hi + 1) - mid
        ef[k_lo:k_hi + 1] += (1.0 - h[idx])
    # Backward: h[p] appears as h[i+k-mid] when i+k-mid=p
    # c[k] has term h[i]*(1-h[p]) where i+k-mid=p, i.e., i=p-k+mid
    # Contribution: h[p-k+mid] * (-1) for the change in (1-h[p])
    k_lo = max(0, p)
    k_hi = min(nc - 1, p + mid)
    if k_lo <= k_hi:
        idx = p - np.arange(k_lo, k_hi + 1) + mid
        ef[k_lo:k_hi + 1] -= h[idx]
    return ef

        s_hi = ef[active].max()

        else:

    best_h = h.copy()

    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

        if c5 < best_c5 - 1e-15:

    dx = 2.0 / n
    def obj(hh):
        c = np.correlate(hh, 1 - hh, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 700:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def sum_con(hh):
        return np.sum(hh) - n / 2.0
    
    def sum_con_jac(hh):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

   c[k] changes by δ * (1 - h[p+k-(n-1)]) for valid k (i.e., 0 ≤ p+k-(n-1) < n)
   → k ranges from (n-1)-p to (n-1)+(n-1-p)

   c[k] has term h[i] * (1-h[p]) = h[i] - h[i]*h[p]

   Valid when 0 ≤ p-k+(n-1) < n → k ranges from p to p+(n-1)
