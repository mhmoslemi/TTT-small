def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(80):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def make_binary(h, n):
    h = (h > 0.5).astype(float)
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n_ones - target, replace=False)] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target - n_ones, replace=False)] = 1
    return h

def thue_morse(n):
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t
