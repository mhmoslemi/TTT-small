import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smoothmax_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    return (m + np.log(np.sum(exps)) / beta) * dx


def smoothmax_grad(h, beta, wthresh=1e-7):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / np.sum(exps)
    val = (m + np.log(np.sum(exps)) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < wthresh:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n - s] += wk * (1.0 - h[s:n])
            grad[s:] -= wk * h[:n - s]
        else:
            s2 = -s
            grad[s2:] += wk * (1.0 - h[:n - s2])
            grad[:n - s2] -= wk * h[s2:]
    grad *= dx
    return val, grad


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-13:
            free = (h > 1e-9) & (h < 1.0 - 1e-9)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_optimize(h, beta, maxiter):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(x):
        return smoothmax_val(x, beta)

    def objg(x):
        return smoothmax_grad(x, beta)[1]

    cons = {'type': 'eq', 'fun': lambda x: x.sum() - target,
            'jac': lambda x: np.ones(n)}
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=[cons],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def local_perturb_optimize(h_best, seed_val, budget_remaining):
    n = len(h_best)
    rng = np.random.RandomState(seed_val)
    best = h_best.copy()
    best_c = compute_c5(best)
    t_start = time.time()
    for attempt in range(20):
        if time.time() - t_start > budget_remaining * 0.8:
            break
        scale = 0.08 / (1.0 + 0.3 * attempt)
        h_pert = best + scale * rng.randn(n)
        h_pert = project_feasible(h_pert, n / 2.0)
        h_new = slsqp_optimize(h_pert, 200, maxiter=250)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best, best_c = h_new, c_new
    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)

    # Phase 1: Annealed SLSQP on initial resolution
    betas_phase1 = [40, 100, 250, 600]
    for b in betas_phase1:
        if tl() < 80:
            break
        h_try = slsqp_optimize(h_best, b, maxiter=500)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            h_best, best_c5 = h_try, c_try

    # Phase 2: Higher resolution with annealed optimization
    for n_new in [280, 350, 450]:
        if tl() < 250:
            break
        h_interp = interp_to_n(h_best, n_new)
        for b in [60, 200, 500]:
            if tl() < 120:
                break
            h_try = slsqp_optimize(h_interp, b, maxiter=400)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                h_best, best_c5 = h_try, c_try
                h_interp = h_best.copy()

    # Phase 3: Perturbation + local search to escape local optima
    if tl() > 200:
        h_try, c_try = local_perturb_optimize(h_best, seed, tl() * 0.35)
        if c_try < best_c5:
            h_best, best_c5 = h_try, c_try

    # Phase 4: More perturbations with different scales
    if tl() > 150:
        h_try, c_try = local_perturb_optimize(h_best, seed + 1000, tl() * 0.25)
        if c_try < best_c5:
            h_best, best_c5 = h_try, c_try

    # Phase 5: Final high-beta polish
    if tl() > 60:
        h_final = slsqp_optimize(h_best, 1000, maxiter=800)
        c_final = compute_c5(h_final)
        if c_final < best_c5:
            h_best, best_c5 = h_final, c_final

    # Phase 6: One more round at higher resolution if time permits
    if tl() > 100:
        n_cur = len(h_best)
        n_up = min(600, n_cur + 100)
        h_up = interp_to_n(h_best, n_up)
        h_up = slsqp_optimize(h_up, 300, maxiter=400)
        c_up = compute_c5(h_up)
        if c_up < best_c5:
            h_best, best_c5 = h_up, c_up
        if tl() > 40:
            h_up2 = slsqp_optimize(h_best, 800, maxiter=500)
            c_up2 = compute_c5(h_up2)
            if c_up2 < best_c5:
                h_best, best_c5 = h_up2, c_up2

    # Final validation
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
