    corr = compute_corr_full(x)
    m = np.max(corr)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = compute_corr_full(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    max_sh = np.max(shifted)
    exps = np.exp(shifted - max_sh)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + (max_sh + np.log(sw)) / beta) * dx
    j = np.arange(n)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            # j-s range: j from s to n-1, contributes (1-x[j-s])
            indices = np.arange(s, n)
            grad[indices] += wk * (1.0 - x[indices - s])
            # j+s range: j from 0 to n-1-s, contributes -x[j+s]
            indices2 = np.arange(0, n - s)
            grad[indices2] += wk * (-x[indices2 + s])
        else:
            ss = -s
            # j-s = j+ss range: j from 0 to n-1-ss
            indices = np.arange(0, n - ss)
            grad[indices] += wk * (1.0 - x[indices + ss])
            # j+s = j-ss range: j from ss to n-1
            indices2 = np.arange(ss, n)
            grad[indices2] += wk * (-x[indices2 - ss])
    grad *= dx
    return val, grad

def slsqp_solve(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        v, _ = smooth_max_val_grad(x, beta)
        return v
    def objg(x):
        _, g = smooth_max_val_grad(x, beta)
        return g
    
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def nesterov_subgrad(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        y = x if t == 0 else x + ((t-1)/(t+2)) * (x - x_prev)
        y = project_feasible(y, target)
        corr = compute_corr_full(y)
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            if s >= 0:
                idx = np.arange(s, n)
                grad[idx] += (1.0 - y[idx - s])
                idx2 = np.arange(0, n - s)
                grad[idx2] += (-y[idx2 + s])
            else:
                ss = -s
                idx = np.arange(0, n - ss)
                grad[idx] += (1.0 - y[idx + ss])
                idx2 = np.arange(ss, n)
                grad[idx2] += (-y[idx2 - ss])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def coord_descent(h, n_sweeps=3):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    for sweep in range(n_sweeps):
        for j in range(n):
            # Compute alphas, betas for this j
            alphas = np.zeros(2*n-1)
            betas = np.zeros(2*n-1)
            hj = h[j]
            for s in range(-(n-1), n):
                k = s + n - 1
                if 0 <= j+s < n:
                    if s == 0:
                        alphas[k] += 1.0
                        betas[k] += -1.0
                    else:
                        alphas[k] += (1.0 - h[j+s])
                if 0 <= j-s < n and s != 0:
                    alphas[k] += (-h[j-s])
            corr_cur = compute_corr_full(h)
            gamma = corr_cur - alphas * hj - betas * hj**2
            # Minimize over t in [0,1]
            t_grid = np.linspace(0, 1, 200)
            corr_t = betas[:, None]*t_grid[None,:]**2 + alphas[:, None]*t_grid[None,:] + gamma[:, None]
            vals = np.max(corr_t, axis=0) * dx
            best_idx = np.argmin(vals)
            # Fine grid
            lo = max(0, t_grid[max(0, best_idx-3)])
            hi = min(1, t_grid[min(199, best_idx+3)])
            t_fine = np.linspace(lo, hi, 50)
            corr_f = betas[:, None]*t_fine[None,:]**2 + alphas[:, None]*t_fine[None,:] + gamma[:, None]
            vals_f = np.max(corr_f, axis=0) * dx
            h[j] = t_fine[np.argmin(vals_f)]
            h = project_feasible(h, target)
    return h

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)
