- y_k = x_k + (t_{k-1} - 1)/t_k * (x_k - x_{k-1})
- x_{k+1} = prox(y_k - lr * grad(y_k))
- t_{k+1} = (1 + sqrt(1+4*t_k^2))/2

Let me fix that. Also, I should be careful: the subgradient of max_k I_k is the subgradient of the active I_k. With Nesterov, we evaluate at the lookahead point y.

The optimal solution needs to be a balance: some structure to reduce the overlap, but not so much that some lags become bad.

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def con_fun(h, ts, nn):
    return h.sum() - ts


def con_jac(h, ts, nn):
    return np.ones(nn)


def obj_smooth(h, dv, bv):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, bv) * dv


def obj_weighted(h, w, n_val):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.dot(w, corr) * (2.0 / n_val))
