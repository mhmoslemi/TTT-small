    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break

        else:
            mask = h > 0.0

        if cnt == 0:

        h = np.clip(h, 0.0, 1.0)

def get_subgradient(h, k, n):
    """Subgradient of lag k overlap with respect to h."""
    grad = np.zeros(n)
    if k >= 0:
        for i in range(n - k):
            grad[i] += (1.0 - h[i + k])
            grad[i + k] -= h[i]
    else:
        k = -k
        for i in range(n - k):
            grad[i + k] += (1.0 - h[i])
            grad[i] -= h[i + k]
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_at_h(h, k, n):
    grad = np.zeros(n)
    if k >= 0:
        idx = np.arange(n - k)
        grad[idx] += (1.0 - h[idx + k])
        grad[idx + k] -= h[idx]
    else:
        kk = -k
        idx = np.arange(n - kk)
        grad[idx + kk] += (1.0 - h[idx])
        grad[idx] -= h[idx + kk]
    return grad

    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

    return int(np.max(corr)) * dx

def binary_local_search(h_bin, n, max_iter=100000, time_limit=None, t0=None):
    """Local search: swap 1↔0 pairs to reduce max corr."""
    h = h_bin.astype(float).copy()
    n_ones = int(h.sum())
    assert n_ones == n // 2
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    dx = 2.0 / n
    max_val = int(np.max(np.correlate(h, 1.0-h, mode='full')))

            if time.time() - t0 > time_limit:
                break

        found_improvement = False
        for _ in range(50):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            # Tentative swap
            h[i], h[j] = 0, 1
            new_max = int(np.max(np.correlate(h, 1.0-h, mode='full')))
            h[j], h[i] = 1, 0
            if new_max < max_val:
                # Accept
                h[i], h[j] = 0, 1
                ones.remove(i)
                ones.append(j)
                zeros.append(i)
                zeros.remove(j)
                max_val = new_max
                found_improvement = True
                break
        if not found_improvement:
            # Try kicked move: accept a slightly worse solution
            best_kick = None
            best_kick_val = max_val + 1  # Only accept if at most 1 worse
            for _ in range(100):
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                h[i], h[j] = 0, 1
                new_max = int(np.max(np.correlate(h, 1.0-h, mode='full')))
                h[j], h[i] = 1, 0
                if new_max < best_kick_val:
                    best_kick_val = new_max
                    best_kick = (i, j)
            if best_kick is not None and np.random.rand() < 0.3:
                i, j = best_kick
                h[i], h[j] = 0, 1
                ones.remove(i)
                ones.append(j)
                zeros.append(i)
                zeros.remove(j)
                max_val = best_kick_val
    
    return h, max_val * dx

Actually, let me optimize this. Instead of computing the full correlation for each trial swap, I can compute the change in the maximum more efficiently.

When we swap h[i] from 1→0 and h[j] from 0→1:
- For each lag k, the correlation changes by:
  - Remove contribution of position i (was 1, now 0): the term h[i]*(1-h[i+k]) removed = 1*(1-h[i+k]) = 1-h[i+k]
  - Add contribution of position j (was 0, now 1): the term h[j]*(1-h[j+k]) added = 1*(1-h[j+k]) = 1-h[j+k]
  - Also, position i appears as the "shifted" position in lag -k: term h[i-k]*(1-h[i]) changes from h[i-k]*0 to h[i-k]*1
  - Position j appears as the "shifted" position: term h[j-k]*(1-h[j]) changes from h[j-k]*1 to h[j-k]*0

This is getting complicated. Let me just use the full correlation for simplicity, since n=70 is small.

Key insight for binary search: Instead of random swaps, use **informed swaps** based on which lags are worst.

If the worst lag is k*, then we want to reduce the number of (i, i+k*) pairs where h[i]=1, h[i+k*]=0. A good swap would be:
- Find a position i where h[i]=1, h[i+k*]=0 (contributing to the worst lag)
- Swap it with position j where h[j]=0, h[j+k*]=1 (this would reduce the worst lag)

After the swap: position i becomes 0 (removing one bad pair at lag k*), position j becomes 1 (adding one good pair at lag k*). Net change in lag k* overlap: -2.

But this might increase overlap at other lags. Need to check.
