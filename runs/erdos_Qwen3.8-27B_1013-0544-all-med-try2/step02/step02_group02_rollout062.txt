            t_hi = min(1.0 - h[i], h[j])
            if t_hi - t_lo < 1e-13:
                continue
            gi = grad_col(h, i, n, lags)
            gj = grad_col(h, j, n, lags)
            g = gi - gj
            # Bisection
            lo, hi = t_lo, t_hi
            for _ in range(20):
                mid = 0.5 * (lo + hi)
                k_star = int(np.argmax(C + mid * g))
                sg = g[k_star]
                if sg > 1e-9:
                    hi = mid
                elif sg < -1e-9:
                    lo = mid
                else:
                    break
            t_opt = 0.5 * (lo + hi)
            if t_opt < t_lo: t_opt = t_lo
            if t_opt > t_hi: t_opt = t_hi
            if abs(t_opt) < 1e-12:
                continue
            # First-order check
            fo_max = float(np.max(C + t_opt * g))
            if fo_max >= best_c5 - 1e-10:
                continue
            # Exact check
            h[i] += t_opt
            h[j] -= t_opt
            C_new = compute_correlations(h, n)
            new_max = float(np.max(C_new))
            if new_max < best_c5 - 1e-12:
                C = C_new
                best_c5 = new_max
                n_imp += 1
            else:
                h[i] -= t_opt
                h[j] += t_opt
        if n_imp == 0:
            break
    return h, C, best_c5

def slsqp_polish(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    lags = np.arange(-(n-1), n)
    
    def obj(h_flat):
        c = compute_correlations(h_flat, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 700:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh)))) / alpha + m - m  # just m + logsumexp/alpha
        # Actually: m + log(sum(exp(alpha*(c-m))))/alpha
        # = m + log(sum(exp(alpha*c)) / exp(alpha*m))/alpha
        # = m + (log(sum(exp(alpha*c))) - alpha*m)/alpha
        # = m + log(sum(exp(alpha*c)))/alpha - m
        # = log(sum(exp(alpha*c)))/alpha
        # So it's just the log-sum-exp of alpha*c divided by alpha!
    
    # Let me redo the objective properly
    def obj2(h_flat):
        c = compute_correlations(h_flat, n)
        ac = alpha * c
        mx = ac.max()
        if mx > 700:
            return float((mx + np.log(np.sum(np.exp(ac - mx)))) / alpha)
        return float(np.log(np.sum(np.exp(ac)))) / alpha
    
    def grad_obj(h_flat):
        c = compute_correlations(h_flat, n)
        ac = alpha * c
        mx = ac.max()
        if mx > 700:
            se = np.exp(ac - mx)
            w = se / se.sum()
        else:
            se = np.exp(ac)
            w = se / se.sum()
        # Gradient of (1/alpha) * log(sum(exp(alpha*c)))
        # = sum_k w[k] * dc[k]/dh
        grad = np.zeros(n)
        for m in range(n):
            g = grad_col(h_flat, m, n, lags)
            grad[m] = np.dot(w, g)
        return grad
    
    def sum_con(h_flat):
        return np.sum(h_flat) - n / 2.0
    def sum_con_jac(h_flat):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    c_before = compute_c5(h, n)
    try:
        res = minimize(obj2, h, jac=grad_obj, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new, n) < c_before:
            return h_new
    except Exception:
        pass
    return h
