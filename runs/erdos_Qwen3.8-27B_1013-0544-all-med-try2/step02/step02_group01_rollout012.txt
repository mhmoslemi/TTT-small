def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    """Binary coordinate descent: for each i, try setting h[i]=0 or h[i]=1,

    n = len(h)
    target = n / 2.0

    h = np.round(h)
    # Ensure sum = target
    # If too many ones, flip some to zero; if too many zeros, flip some to one
    while h.sum() > target:

        best_i = -1
        best_c = compute_c5(h)
        ones = np.where(h == 1)[0]
        for i in ones:
            h[i] = 0
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_i = i
            h[i] = 1
        if best_i >= 0:

        else:

    while h.sum() < target:

            h[i] = 1

                best_i = i
            h[i] = 0
        if best_i >= 0:
            h[best_i] = 1

Similarly, h_old[i+s] uses the old value unless i+s = j, in which case h_old[i+s] = h_old[j] = 0.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        ms = (j - s >= 0) & (j - s < n)
        mp = (j + s >= 0) & (j + s < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[j[ms] - s])
        if mp.any():
            grad[mp] += wk * (-h[j[mp] + s])
    grad *= dx
    return val, grad

def slsqp_refine(h, maxiter=500, beta=200):
    n = len(h)
    target = n / 2.0
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)
