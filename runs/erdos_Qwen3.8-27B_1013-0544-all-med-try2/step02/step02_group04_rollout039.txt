    starts = [best_h.copy()]
    t_arr = np.arange(n_points)
    for freq in [2, 3, 5]:
        h_s = 0.5 + 0.15 * np.sin(2 * np.pi * freq * t_arr / n_points + rng.uniform(0, 6.28))
        starts.append(project_feasible(h_s, target))
    h_sym = 0.5 + 0.12 * np.cos(4 * np.pi * t_arr / n_points)
    starts.append(project_feasible(h_sym, target))
    h_mir = 0.5 + 0.1 * np.sin(np.pi * t_arr / n_points)
    starts.append(project_feasible(h_mir, target))

    betas = [20, 60, 150, 400, 1000, 2500, 5000]

    # Multi-start optimization
    for si in range(len(starts)):
        if time_left() < 80:
            break
        h_cur = starts[si].copy()
        for beta in betas:
            if time_left() < 40:
                break
            h_opt = slsqp_phase(h_cur, target, beta, n_points, max_iter=4000)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            h_cur = h_opt

    # Gradient-based refinement with projected steps
    if time_left() > 60:
        for beta in [1000, 3000]:
            if time_left() < 40:
                break
            lr = 0.8 / (beta * n_points)
            h_pg = project_grad_step(best_h, target, beta, lr, n_steps=3000)
            c5 = compute_c5(h_pg)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_pg.copy()

    # Simulated annealing with local refinement
    if time_left() > 80:
        cur_h = best_h.copy()
        cur_c5 = best_c5
        n_sa = 3000
        for it in range(n_sa):
            if time_left() < 30:
                break
            temp = 0.0005 * np.exp(-it / (n_sa * 0.3)) + 1e-5
            scale = 0.02 * np.exp(-it / (n_sa * 0.5)) + 0.003
            h_pert = cur_h + rng.randn(n_points) * scale
            h_pert = project_feasible(h_pert, target)
            # Quick local refine
            h_loc = slsqp_phase(h_pert, target, 800, n_points, max_iter=1000)
            c5_loc = compute_c5(h_loc)
            accept = c5_loc < cur_c5 or rng.random() < np.exp(-(c5_loc - cur_c5) / temp)
            if accept:
                cur_h = h_loc
                cur_c5 = c5_loc
            if c5_loc < best_c5:
                best_c5 = c5_loc
                best_h = h_loc.copy()

    # Targeted perturbation: attack the max lag directly
    if time_left() > 50:
        c = corr_of(best_h)
        active_lag = np.argmax(c)
        lag = active_lag - (n_points - 1)
        # Reduce correlation at this lag via gradient step
        for trial in range(20):
            if time_left() < 20:
                break
            h_try = best_h.copy()
            # Gradient of specific lag correlation
            g_spec = np.zeros(n_points)
            j = lag
            if j >= 0:
                length = n_points - j
                g_spec[j:] = (1.0 - h_try[:length])
                g_spec[:length] -= h_try[j:]
            else:
                jj = -j
                length = n_points - jj
                g_spec[:length] = (1.0 - h_try[jj:])
                g_spec[jj:] -= h_try[:length]
            # Subtract mean to respect sum constraint
            g_spec -= np.mean(g_spec)
            for lr_try in [0.001, 0.005, 0.01, 0.02]:
                h_cand = project_feasible(h_try - lr_try * g_spec, target)
                c5_cand = compute_c5(h_cand)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = h_cand.copy()
                    h_try = h_cand.copy()
                    break

    # Final high-precision SLSQP polish
    if time_left() > 20:
        for beta in [5000, 10000, 20000]:
            if time_left() < 10:
                break
            h_fin = slsqp_phase(best_h, target, beta, n_points, max_iter=8000)
            c5 = compute_c5(h_fin)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_fin.copy()

    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
