If I think of it as a graph where edges connect i to i+k, then the "overlap" at shift k is the number of edges going from a 1-node to a 0-node. We want to minimize the maximum edge count over all k.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def make_golden_h(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = set()
    for i in range(20 * n):
        p = int((i * phi * n) % n)
        if p not in positions:
            positions.add(p)
            if len(positions) == n // 2:
                break
    while len(positions) < n // 2:
        positions.add(rng.randint(n))
    h[list(positions)] = 1.0
    return h

def ls_best(h, time_limit, rng):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0.0
        best_i, best_j = -1, -1
        for i in ones:
            if time.time() - start >= time_limit:
                return h, c5
            for j in zeros:
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                nc5 = compute_c5(h)
                delta = c5 - nc5
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
                h[i] = 1.0
                h[j] = 0.0
        if best_i < 0:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
        c5 -= best_delta
    return h, c5

def ls_random(h, time_limit, rng, n_samples=500):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for _ in range(n_samples):
            if time.time() - start >= time_limit:
                return h, c5
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            nc5 = compute_c5(h)
            if nc5 < c5:
                c5 = nc5
                improved = True
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                break
            h[i] = 1.0
            h[j] = 0.0
        if not improved:
            break
    return h, c5

def sa_search(h, time_limit, rng):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = min(1.0, elapsed / max(time_limit, 1e-9))
        T = 0.008 * (1.0 - frac) + 0.0003
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        nc5 = compute_c5(h)
        delta = nc5 - c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            c5 = nc5
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h, c5
