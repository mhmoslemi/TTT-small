    """Project v onto {x: sum(x) = target_sum, 0 <= x <= 1}"""
    # Bisection on alpha: sum(clip(v - alpha, 0, 1)) = target_sum

    for T in temps:
        remaining = budget_s - 80 - (time.time() - t0)
        if remaining < 3:
            break
        h_new, c5_new = proj_grad_descent(best_h, n, target_sum, T, 0.5, 5000, remaining)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Local search for fine-tuning
    remaining = budget_s - 40 - (time.time() - t0)
    if remaining > 5:
        h_new, c5_new = local_search(best_h, n, target_sum, min(remaining, 60))
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Random restarts
    remaining = budget_s - 30 - (time.time() - t0)
    restart_count = 0
    while remaining > 8 and restart_count < 30:
        h_pert = project_onto_constraint(best_h + rng.randn(n) * 0.05, target_sum)
        h_new, c5_new = proj_grad_descent(h_pert, n, target_sum, 15.0, 0.3, 2000, min(15, remaining - 5))
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        restart_count += 1
        remaining = budget_s - 30 - (time.time() - t0)

    # Phase 4: Grid refinement
    remaining = budget_s - 10 - (time.time() - t0)
    if remaining > 20:
        for factor in [2, 4]:
            n_new = n * factor
            if n_new > 400:
                break
            rem = budget_s - 10 - (time.time() - t0)
            if rem < 15:
                break
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.arange(n_new) * (2.0 / n_new)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_onto_constraint(h_up, n_new / 2.0)
            for T in [50, 20, 10, 5, 3]:
                rem = budget_s - 10 - (time.time() - t0)
                if rem < 5:
                    break
                h_new, c5_new = proj_grad_descent(h_up, n_new, n_new / 2.0, T, 0.3, 3000, rem - 5)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    n = n_new
                    target_sum = n / 2.0
                    h_up = best_h.copy()

    # Final local search on best solution
    remaining = budget_s - 1 - (time.time() - t0)
    if remaining > 2:
        h_new, c5_new = local_search(best_h, n, target_sum, remaining)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
