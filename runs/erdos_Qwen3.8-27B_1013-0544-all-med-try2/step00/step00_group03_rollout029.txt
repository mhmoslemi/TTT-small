    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_grad_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad
