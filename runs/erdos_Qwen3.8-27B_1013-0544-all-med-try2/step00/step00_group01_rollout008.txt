    # (with appropriate bounds)

    # - Decreasing h at positions i+k_star where h[i] is small
    # - Increasing h at positions i where h[i+k_star] is large

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def upsample(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()

    new_dx = 2.0 / new_n
    new_x = (np.arange(new_n) + 0.5) * new_dx
    h_new = np.interp(new_x, old_x, h)
    h_new = np.clip(h_new, 0.0, 1.0)

        diff = target - s
        if abs(diff) < 1e-14:

            if tr < 1e-15:
                break

            tr = h_new.sum()
            if tr < 1e-15:
