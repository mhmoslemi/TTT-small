```python
import numpy as np
import time
from numpy.lib.stride_tricks import sliding_window_view

def project_onto_set(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c), c

def subgradient_step(h, weights, n):
    M = sliding_window_view(weights, n)
    return M.sum(axis=1) - 2.0 * (M.T @ h)

def optimize_phase(h, n, dx, target_sum, max_iters, seed, initial_alpha=0.05, initial_t=10.0):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5, _ = compute_c5(h, dx)
    alpha = initial_alpha
    t = initial_t
    no_improve = 0
    
    for it in range(max_iters):
        c_max, c = compute_c5(h, dx)
        if c_max < best_c5 - 1e-8:
            best_c5 = c_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            h = best_h + rng.randn(n) * 0.02
            h = project_onto_set(h, target_sum)
            no_improve = 0
            alpha = max(alpha * 0.5, 0.01)
            continue
        
        c_shifted = c - c_max
        weights = np.exp(t * c_shifted)
        ws = np.sum(weights)
        weights /= ws
        
        grad = subgradient_step(h, weights, n)
        h = h - alpha * grad
        h = project_onto_set(h, target_sum)
        
        if c_max < best_c5:
            alpha = min(alpha * 1.01, 0.3)
            t = min(t * 1.005, 50.0)
        else:
            alpha *= 0.997
            if alpha < 0.005:
                alpha = 0.005
    
    return best_h, best_c5

def coordinate_refine(h, n, dx, target_sum, max_iters=5000):
    """Try single-coordinate flips to improve."""
    best_c5, _ = compute_c5(h, dx)
    h = h.copy()
    improved = True
    it = 0
    while improved and it < max_iters:
        improved = False
        it += 1
        c_max, c = compute_c5(h, dx)
        # Find the worst shift
        k_star = np.argmax(c)
        # Try flipping each coordinate
        for m in range(n):
            for delta in [-0.1, -0.05, 0.05, 0.1]:
                h_new = h.copy()
                h_new[m] = np.clip(h[m] + delta, 0, 1)
                # Project
                h_new = project_onto_set(h_new, target_sum)
                c_new, _ = compute_c5(h_new, dx)
                if c_new < best_c5 - 1e-8:
                    h = h_new
                    best_c5 = c_new
                    improved = True
                    break
            if improved:
                break
    return h, best_c5
```

```python
import numpy as np
import time
from numpy.lib.stride_tricks import sliding_window_view

def project_onto_set(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}."""
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h, dx):
    """Return (max_c, full_c) for the overlap function."""
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c), c

def weighted_grad(h, weights, n):
    """Compute gradient of sum_j w_j * c_j w.r.t. h."""
    M = sliding_window_view(weights, n)
    return M.sum(axis=1) - 2.0 * (M.T @ h)

def subgradient_optimize(h, n, dx, target_sum, max_time, seed):
    """Main subgradient optimization loop."""
    rng = np.random.RandomState(seed)
    start = time.time()
    best_h = h.copy()
    best_c5, _ = compute_c5(h, dx)
    alpha = 0.05
    t = 12.0
    no_improve = 0
    momentum = np.zeros(n)
    beta = 0.7

    while time.time() - start < max_time:
        c_max, c = compute_c5(h, dx)
        if c_max < best_c5 - 1e-9:
            best_c5 = c_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400:
            h = best_h + rng.randn(n) * (0.02 + 0.03 * (no_improve // 1000))
            h = project_onto_set(h, target_sum)
            no_improve = 0
            alpha = 0.05
            t = 12.0
            momentum[:] = 0.0
            continue

        # Smooth max weights
        c_shifted = c - c_max
        weights = np.exp(t * c_shifted)
        ws = np.sum(weights)
        weights /= ws

        # Gradient
        grad = weighted_grad(h, weights, n)
        momentum = beta * momentum + (1.0 - beta) * grad
        h = h - alpha * momentum
        h = project_onto_set(h, target_sum)

        # Adaptive step size and temperature
        if c_max < best_c5:
            alpha = min(alpha * 1.01, 0.25)
            t = min(t * 1.003, 60.0)
        else:
            alpha = max(alpha * 0.998, 0.005)
            if no_improve > 200:
                t = max(t * 0.99, 5.0)

    return best_h, best_c5

def try_binarize(h, target_sum, dx):

    if diff > 0:
        zeros = np.where(h_bin == 0)[0]
        order = zeros[np.argsort(-h[zeros])]
        for i in range(int(np.round(diff))):

    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        order = ones[np.argsort(h[ones])]
        for i in range(int(np.round(-diff))):
            if i < len(order):
                h_bin[order[i]] = 0.0
    h_bin = project_onto_set(h_bin, target_sum)
    return h_bin
```
