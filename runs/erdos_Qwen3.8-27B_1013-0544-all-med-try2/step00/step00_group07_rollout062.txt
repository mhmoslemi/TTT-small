    dx = 2.0 / n

    k_star = np.argmax(corr)

            h_sa[i], h_sa[j] = 1.0, 0.0
        temp *= 0.99995

    # Phase 2: Continuous optimization (homotopy with L-BFGS-B)
    h_cur = project(best_h.copy(), n)
    c5_cur = compute_c5(h_cur, n)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur.copy()

    h_opt = h_cur.copy()
    betas = [5, 10, 20, 50, 100, 200, 500, 1000]
    for beta in betas:
        if time.time() - t0 > budget_s - 40:
            break
        pen = max(5000.0, float(beta) * 500.0)
        try:
            res = minimize(
                penalized_obj, h_opt,
                args=(n, beta, pen),
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-16}
            )
            h_new = project(res.x, n)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_opt = h_new
        except Exception:
            pass

    # Phase 3: SLSQP refinement with direct constraint
    if time.time() - t0 < budget_s - 30:
        h_slsqp = project(best_h.copy(), n)
        for beta in [50, 200, 1000]:
            if time.time() - t0 > budget_s - 15:
                break
            try:
                res = minimize(
                    smooth_max_obj, h_slsqp,
                    args=(n, beta),
                    method='SLSQP',
                    bounds=[(0.0, 1.0)] * n,
                    constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}],
                    options={'maxiter': 300, 'ftol': 1e-14}
                )
                h_new = project(res.x, n)
                c5_new = compute_c5(h_new, n)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_slsqp = h_new
            except Exception:
                pass

    # Phase 4: Random restarts from best
    for _ in range(10):
        if time.time() - t0 > budget_s - 15:
            break
        h_r = best_h + rng.normal(0.0, 0.05, n)
        h_r = project(h_r, n)
        for beta, pen in [(20, 5000.0), (200, 50000.0), (1000, 200000.0)]:
            if time.time() - t0 > budget_s - 8:
                break
            try:
                res = minimize(
                    penalized_obj, h_r,
                    args=(n, beta, pen),
                    method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * n,
                    options={'maxiter': 1000, 'ftol': 1e-16}
                )
                h_new = project(res.x, n)
                c5_new = compute_c5(h_new, n)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_r = h_new
            except Exception:
                pass

    # Phase 5: Try larger n via interpolation
    if time.time() - t0 < budget_s - 30:
        for n_try in [160, 320]:
            if time.time() - t0 > budget_s - 15:
                break
            x_old = np.linspace(0.0, 2.0, n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project(h_interp, n_try)
            for beta, pen in [(20, 5000.0), (200, 50000.0)]:
                if time.time() - t0 > budget_s - 10:
                    break
                try:
                    res = minimize(
                        penalized_obj, h_interp,
                        args=(n_try, beta, pen),
                        method='L-BFGS-B',
                        bounds=[(0.0, 1.0)] * n_try,
                        options={'maxiter': 500, 'ftol': 1e-15}
                    )
                    h_new = project(res.x, n_try)
                    c5_new = compute_c5(h_new, n_try)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        n = n_try
                    h_interp = h_new
                except Exception:
                    break

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
