import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def greedy_polish(h, time_limit=10.0):
    n = len(h)
    t_start = time.time()
    cur_c5 = compute_c5(h)
    for _ in range(20):
        if time.time() - t_start > time_limit:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for ai in ones:
            if time.time() - t_start > time_limit:
                return h, cur_c5
            for bj in zeros:
                h[ai] = 0.0
                h[bj] = 1.0
                nc = compute_c5(h)
                if nc < cur_c5:
                    cur_c5 = nc
                    improved = True
                else:
                    h[ai] = 1.0
                    h[bj] = 0.0
        if not improved:
            break
    return h, cur_c5

def sa_optimize(h, rng, time_limit):
    n = len(h)
    t_start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T0, Tend = 0.1, 0.001
    while True:
        elapsed = time.time() - t_start
        if elapsed >= time_limit:
            break
        T = T0 * (Tend / T0) ** (elapsed / max(time_limit, 1e-9))
        save = h.copy()
        if rng.random() < 0.7:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]

            bj = zeros[rng.randint(len(zeros))]

            h[bj] = 1.0
        else:
            L = rng.randint(2, min(20, n // 3))
            p = rng.randint(0, n - L)
            h[p:p + L] = h[p:p + L][::-1]
        nc = compute_c5(h)
        d = nc - cur_c5
        if d <= 0 or rng.random() < np.exp(-d / max(T, 1e-10)):
            cur_c5 = nc
            if cur_c5 < best_c5:

    elif no < target:
        arr = np.where(h == 0)[0]
        rng.shuffle(arr)

    x_new = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
