    return np.max(corr)

def lse_max(values, t):
    m = np.max(values)
    shifted = t * (values - m)
    # Clip to prevent overflow
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / t

def obj_fn(x, n, t, mu):
    h = x
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    val = lse_max(corr, t)
    cv = np.sum(h) - n / 2.0
    return val + mu * cv * cv

def project_constraint(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def make_starts(n, rng, best_h, init):
    starts = []
    starts.append(np.ones(n) * 0.5)
    for _ in range(4):
        h = rng.random(n)
        starts.append(project_constraint(h, n))
    for bl in [2, 3, 4, 5, 7]:
        h = np.zeros(n)
        for i in range(n):
            if (i // bl) % 2 == 0:
                h[i] = 1.0
        starts.append(project_constraint(h, n))
    if best_h is not None and len(best_h) != n:
        x_old = (np.arange(len(best_h)) + 0.5) * 2.0 / len(best_h)
        x_new = (np.arange(n) + 0.5) * 2.0 / n
        h_i = np.interp(x_new, x_old, best_h)
        starts.append(project_constraint(h_i, n))
    if init is not None and n == len(init):
        starts.append(init.copy())
    # Quasi-periodic starts
    for p in [7, 9, 11, 13]:
        h = np.zeros(n)
        for i in range(n):
            frac = (i % p) / p
            if 0.1 < frac < 0.6:
                h[i] = 1.0
        starts.append(project_constraint(h, n))
    return starts

def refine(h, n, t0, budget_left):
    x = h.copy()
    t = t0
    while t < 2000 and budget_left() > 5:
        mu = 500.0
        res = minimize(
            obj_fn, x, args=(n, t, mu),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': 400, 'ftol': 1e-14, 'gtol': 1e-10}
        )
        x_new = project_constraint(res.x, n)
        c5_new = compute_c5(x_new)
        c5_old = compute_c5(x)
        if c5_new <= c5_old + 1e-12:
            x = x_new
        t = int(t * 2.5)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    init = None
    
    def elapsed():
        return time.time() - t_start
    
    def budget_left():
        return budget_s - elapsed()
    
    try:
        init = np.array(initial_h_values, dtype=float)
        n0 = len(init)
        init = project_constraint(init, n0)
        c5_init = compute_c5(init)
        best_h = init.copy()
        best_c5 = c5_init
    except Exception:
        n0 = 45
        best_h = np.ones(n0) * 0.5
        best_c5 = compute_c5(best_h)
    
    n_list = sorted(set([n0, 30, 40, 50, 60, 80, 100, 120]))
    
    for n in n_list:
        if budget_left() < 40:
            break
        
        starts = make_starts(n, rng, best_h, init)
        # Deduplicate
        unique = [starts[0]]
        for s in starts[1:]:
            if all(np.linalg.norm(s - u) > 0.05 for u in unique):
                unique.append(s)
        
        for s in unique:
            if budget_left() < 10:
                break
            # Quick eval of starting point
            c5_s = compute_c5(s)
            # Only refine if promising or it's our best
            if c5_s < best_c5 + 0.05:
                x = refine(s, n, 5, budget_left)
                c5_x = compute_c5(x)
                if c5_x < best_c5:
                    best_c5 = c5_x
                    best_h = x.copy()
    
    # Final high-precision refinement
    if best_h is not None and budget_left() > 20:
        n = len(best_h)
        x = best_h.copy()
        for t_val in [200, 500, 1000, 2000, 5000]:
            if budget_left() < 3:
                break
            mu = 5000.0
            res = minimize(
                obj_fn, x, args=(n, t_val, mu),
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 600, 'ftol': 1e-15, 'gtol': 1e-12}
            )
            x_new = project_constraint(res.x, n)
            c5_new = compute_c5(x_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = x_new.copy()
            x = x_new
    
    # Coordinate descent / rounding local search
    if best_h is not None and budget_left() > 10:
        n = len(best_h)
        h_loc = best_h.copy()
        improved = True
        max_rounds = 3
        for rnd in range(max_rounds):
            if budget_left() < 5 or not improved:
                break
            improved = False
            for _ in range(n * 2):
                if budget_left() < 3:
                    break
                idx = rng.randint(n)
                h_try = h_loc.copy()
                if h_try[idx] >= 0.5:
                    h_try[idx] = 1.0
                else:
                    h_try[idx] = 0.0
                h_try = project_constraint(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h_loc = h_try.copy()
                    improved = True
                elif c5_try <= best_c5 + 1e-10:
                    h_loc = h_try.copy()
    
    # Ensure final constraint satisfaction
    if best_h is None:
        best_h = init if init is not None else np.ones(45) * 0.5
    n_final = len(best_h)
    best_h = project_constraint(best_h, n_final)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_final
