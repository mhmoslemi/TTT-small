def eval_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')

    """Best-improvement single-swap sweep. Returns (new_c5, improved)."""
    current_c5 = eval_c5(h, dx)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    best_val = current_c5
    best_i = -1
