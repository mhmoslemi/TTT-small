- When multiple lags are tied, the gradient is a subgradient
- The function is piecewise smooth
- In practice, picking one of the tied lags and descending works okay

But a better approach might be to use a smooth max:
f(h) = (1/tau) * log(sum_k exp(tau * corr[k] * dx))

For large tau, this approximates the max. The gradient is:
grad f = (1/sum_k exp(...)) * sum_k exp(tau * corr[k] * dx) * grad(corr[k] * dx)

This is like a weighted average of the gradients, with weights proportional to how close each lag is to the max.

```python
def smooth_max_obj(h, dx, tau, penalty):

    vals = corr * dx
    # Smooth max
    max_val = np.max(vals)
    exp_vals = np.exp(tau * (vals - max_val))
    smooth_max = max_val + np.log(np.sum(exp_vals)) / tau
    integral = np.sum(h) * dx - 1
    return smooth_max + penalty * integral**2

def smooth_max_grad(h, dx, tau, penalty):

    max_val = np.max(vals)
    exp_vals = np.exp(tau * (vals - max_val))

    n = len(h)
```

```python
                    best_c5, best_h = c5_cur, h_start.copy()

        h_init = np.array(best_h[:n] if len(best_h) == n else proj_constraint(
            np.interp(np.linspace(0, 2, n, endpoint=False),
                      np.linspace(0, 2, len(best_h), endpoint=False), best_h), S))

    # Final polish on the best solution
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h, best_c5 = polish_erdos(best_h, dx_final, n_final / 2.0, max_iter=5000, time_limit=30)

    return (best_h, best_c5, n_final)
```
