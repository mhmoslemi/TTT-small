import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def greedy_search(h, n, max_iters=5000):
    h = h.copy()
    c5 = compute_c5(h, n)
    for _ in range(max_iters):
        best_val = c5
        bi, bj = -1, -1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            for j in zeros:
                h[i] = 0; h[j] = 1
                v = compute_c5(h, n)
                h[i] = 1; h[j] = 0
                if v < best_val - 1e-14:
                    best_val = v; bi, bj = i, j
        if bi < 0:
            break
        h[bi] = 0; h[bj] = 1
        c5 = best_val
    return h, c5

def greedy_2swap(h, n, max_iters=15):
    if n > 60:
        return h.copy(), compute_c5(h, n)
    h = h.copy()
    c5 = compute_c5(h, n)
    for _ in range(max_iters):
        best_val = c5
        bm = None
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for a in range(len(ones)):
            for b in range(a + 1, len(ones)):
                for c in range(len(zeros)):
                    for d in range(c + 1, len(zeros)):
                        i1, i2, j1, j2 = ones[a], ones[b], zeros[c], zeros[d]
                        h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                        v = compute_c5(h, n)
                        h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
                        if v < best_val - 1e-14:
                            best_val = v; bm = (i1, i2, j1, j2)
        if bm is None:
            break
        i1, i2, j1, j2 = bm
        h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
        c5 = best_val
    return h, c5

def sa_search(h, n, max_time, seed=42):
    rng = np.random.RandomState(seed)
    T = 0.04
    T_min = 0.0002
    cool = 0.99999
    h = h.copy()
    c5 = compute_c5(h, n)
    bh = h.copy()
    bc5 = c5
    ts = time.time()
    it = 0
    stuck = 0
    while time.time() - ts < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0; h[j] = 1
        nc = compute_c5(h, n)
        d = nc - c5
        if d <= 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            c5 = nc
            if c5 < bc5 - 1e-14:
                bc5 = c5
                bh = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            h[i] = 1; h[j] = 0
            stuck += 1
        it += 1
        T = max(T * cool, T_min)
        if stuck > 30000:
            T = max(T, 0.02)
            stuck = 0
            for _ in range(3):
                o2 = np.where(h == 1)[0]
                z2 = np.where(h == 0)[0]
                ii = int(rng.choice(o2))
                jj = int(rng.choice(z2))
                h[ii] = 0; h[jj] = 1
            c5 = compute_c5(h, n)
    return bh, bc5

def resample_val(h, no, nn, seed=0):
    if no == nn:
        return h.copy()
    rng = np.random.RandomState(seed)
    hn = np.zeros(nn, dtype=np.float64)

        hn[i] = h[oi]
    tgt = nn // 2
    cur = int(np.round(np.sum(hn)))
    if cur > tgt:
        ex = cur - tgt
        o = np.where(hn == 1)[0].copy()
        rng.shuffle(o)

            hn[o[k]] = 0
    elif cur < tgt:
        df = tgt - cur
        z = np.where(hn == 0)[0].copy()
        rng.shuffle(z)
        for k in range(min(df, len(z))):
            hn[z[k]] = 1
    return hn
