import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def soft_max(x, t):
    m = np.max(x)
    if t < 1e-8:
        return m
    return m + t * np.log(np.sum(np.exp((x - m) / t)))

def project_constraint(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def fast_project(h, n):
    diff = n * 0.5 - np.sum(h)
    return np.clip(h + diff / n, 0.0, 1.0)

def optimize_fixed_n(h_start, n, rng, t0, budget_s, phase_budget):
    bounds = [(0.0, 1.0)] * n
    target = n * 0.5
    dx = 2.0 / n
    best_h = project_constraint(h_start, n)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    def remaining():
        return min(phase_budget, tl())

    # Phase A: SLSQP annealing
    for t in [1.0, 0.1, 0.01, 0.001]:
        if remaining() < 30:
            break
        def obj_a(h):
            return soft_max(np.correlate(h, 1.0 - h, mode="full") * dx, t)
        res = minimize(obj_a, best_h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                      options={'maxiter': 2000, 'ftol': 1e-14})
        c5 = compute_c5(res.x)
        if c5 < best_c5:
            best_c5, best_h = c5, res.x.copy()

    # Phase B: Random restarts
    max_restarts = int(remaining() / 0.08)
    max_restarts = min(max_restarts, 2000)
    for i in range(max_restarts):

        m = i % 7

        elif m == 3:

        elif m == 4:

            f2 = rng.randint(2, 7)
            a1 = rng.uniform(0.1, 0.3)
            a2 = rng.uniform(0.05, 0.15)
            p1, p2 = rng.uniform(0, 6.283), rng.uniform(0, 6.283)
            hs = project_constraint(best_h + a1 * np.sin(2*np.pi*f1*x + p1) + a2 * np.sin(2*np.pi*f2*x + p2), n)
        elif m == 5:
            hs = fast_project(best_h + 0.2 * rng.randn(n), n)
        else:
            # Two-block
            split = rng.randint(n // 3, 2 * n // 3)
            hs = np.zeros(n)
            hs[:split] = 1.0

                break

            h_start = project_constraint(h_start, n)

        h_opt, c5_opt = optimize_fixed_n(h_start, n, rng, t0, budget_s, phase_budget)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n

    # Final polish on best resolution
    if time.time() - t0 < budget_s - 30 and len(best_h) == best_n:
        h_polish, c5_polish = optimize_fixed_n(best_h, best_n, rng, t0, budget_s,
                                               int(time.time() - t0) if False else max(30, int(budget_s - (time.time() - t0))))
        if c5_polish < best_c5:
            best_c5 = c5_polish
            best_h = h_polish
