import time

def compute_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    """Compute all overlap values."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_star = np.argmax(corr)
    k = m_star - (n - 1)
    grad = np.zeros(n)

        # overlap = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i]) * dx
        for i in range(n - k):
            grad[i+k] += (1.0 - h[i]) * dx
            grad[i] -= h[i+k] * dx
    else:

        # overlap = sum_{i=0}^{n-1-k_abs} h[i]*(1-h[i+k_abs]) * dx
        for i in range(n - k_abs):
            grad[i] += (1.0 - h[i+k_abs]) * dx
            grad[i+k_abs] -= h[i] * dx

    val = m + tau * np.log(np.sum(exps))
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if weights[m_idx] < 1e-8:
            continue
        k = m_idx - (n - 1)
        grad += weights[m_idx] * grad_at_lag(h, k)
    return val, grad

def optimize_smooth(h0, n, time_limit, rng, tau_start=0.05):
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_val = compute_c5(h)
    start = time.time()
    lr = 0.01
    tau = tau_start
    
    while time.time() - start < time_limit:
        # Decaying tau
        if tau > 0.003:
            tau = max(0.003, tau * 0.999)
        
        val, grad = smooth_max_obj_grad(h, tau)
        
        # Backtracking line search
        alpha = lr
        for _ in range(15):
            h_trial = project_feasible(h - alpha * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_val:
                h = h_trial
                best_val = c5_trial
                best_h = h.copy()
                break
            alpha *= 0.5
        
        # Adapt learning rate
        if best_val < compute_c5(h):
            lr = min(lr * 1.05, 0.1)
        else:
            lr = max(lr * 0.9, 0.0001)
    
    return best_h, best_val

def coordinate_search(h0, n, time_limit, rng):
    h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(h)
    start = time.time()
    step = 0.05
    
    while time.time() - start < time_limit:
        improved = False
        # Try all pairs (i, j): transfer amount from h[i] to h[j]
        # This is O(n^2) per sweep
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                if h[i] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                # Try transferring 'step' from i to j
                transfer = min(step, h[i], 1.0 - h[j])
                if transfer < 1e-10:
                    continue
                h[i] -= transfer
                h[j] += transfer
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                else:
                    h[i] += transfer
                    h[j] -= transfer
        if not improved:
            step *= 0.7
            if step < 0.001:
                break
        elif step < 0.05:
            step = min(step * 1.2, 0.05)
    
    return h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_at_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        if k == 0:
            grad = (1.0 - 2.0 * h) * dx
        else:
            grad[k:] = (1.0 - h[:n-k]) * dx
            grad[:n-k] -= h[k:] * dx
    else:
        ka = -k
        grad[:n-ka] = (1.0 - h[ka:]) * dx
        grad[ka:] -= h[:n-ka] * dx
    return grad

def projected_grad_step(h, n, lr, rng, n_samples=10):
    """One step of projected subgradient with random sampling of active constraints."""
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(overlaps)
    
    # Find all lags that are close to the max
    active = np.where(overlaps > max_val - 0.005)[0]
    
    # Randomly sample from active constraints
    if len(active) > n_samples:
        sampled = rng.choice(active, size=n_samples, replace=False)
    else:
        sampled = active
    
    # Average gradient
    grad = np.zeros(n)
    for m_idx in sampled:
        k = m_idx - (n - 1)
        grad += grad_at_lag(h, k)
    grad /= len(sampled)
    
    # Take step
    h_new = project_feasible(h - lr * grad, n)
    return h_new, max_val
