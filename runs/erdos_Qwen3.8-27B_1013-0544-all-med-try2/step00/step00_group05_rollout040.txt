import numpy as np
import time
from scipy.optimize import minimize_scalar

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def repair_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(20):
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            # Need to increase sum
            for i in range(len(h)):
                if h[i] < 1.0:
                    adj = min(diff, 1.0 - h[i])
                    h[i] += adj
                    diff -= adj
                    if diff < 1e-10:
                        break
        else:
            for i in range(len(h)):
                if h[i] > 0.0:
                    adj = max(diff, -h[i])
                    h[i] += adj
                    diff -= adj
                    if diff > -1e-10:
                        break
    return h

def coordinate_descent(h, target_sum, time_limit, start_time, lam=5.0):
    n = len(h)
    h = repair_sum(h.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for sweep in range(10000):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for i in range(n):
            old = h[i]
            def obj(v):
                h[i] = v
                s = h.sum()
                return compute_c5(h) + lam * (s - target_sum) ** 2
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9, 'maxiter': 25})
            h[i] = res.x
            if abs(h[i] - old) > 1e-6:
                improved = True
        h = repair_sum(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def make_block(n):

    h[:n//2] = 1.0
    return h

    return np.full(n, 0.5)

def make_alternating(n):

        h[-1] = 0.0
    # Fix sum: need sum = n/2
    # With alternating, sum = ceil(n/2) for odd n, n/2 for even n
    return h

def minimize_1d(f, lo, hi, tol=1e-9, max_iter=50):
    """Minimize convex f on [lo, hi] using golden section search."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2

from scipy.optimize import minimize_scalar

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * (2.0 / n)

def repair_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            rooms = 1.0 - h
            tr = rooms.sum()
            if tr < 1e-12:
                break
            h = h + rooms * (diff / tr)
        else:
            ta = h.sum()
            if ta < 1e-12:
                break
            h = h - h * (-diff / ta)
        h = np.clip(h, 0.0, 1.0)
    return h

    def __init__(self, h, idx):
        self.h = h
        self.idx = idx
    def __call__(self, v):
        self.h[self.idx] = v
        return compute_c5(self.h)
