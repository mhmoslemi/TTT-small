def update_corr(corr, h, i, j, n):
    """Update correlation array after swapping h[i] (1→0) and h[j] (0→1)."""
    k = np.arange(2*n-1) - (n-1)  # lags from -(n-1) to n-1
    # h[i+k] for valid i+k
    ik = i + k
    valid_ik = (ik >= 0) & (ik < n)
    h_ik = np.where(valid_ik, h[ik.clip(0, n-1)], 0)
    # (1-h[i-k]) for valid i-k
    ik2 = i - k
    valid_ik2 = (ik2 >= 0) & (ik2 < n)
    one_minus_h_ik2 = np.where(valid_ik2, 1 - h[ik2.clip(0, n-1)], 0)
    # h[j+k] for valid j+k
    jk = j + k
    valid_jk = (jk >= 0) & (jk < n)
    h_jk = np.where(valid_jk, h[jk.clip(0, n-1)], 0)
    # (1-h[j-k]) for valid j-k
    jk2 = j - k
    valid_jk2 = (jk2 >= 0) & (jk2 < n)

Let me re-derive. Before the swap:
- h[i] = 1, h[j] = 0

After the swap:
- h[i] = 0, h[j] = 1

The change in c[m] (which sums h[n'+k](1-h[n']) over valid n') is:

Terms where n' = i: 
Before: h[i+k](1-h[i]) = h[i+k]*0 = 0
After: h'[i+k](1-h'[i]) = h'[i+k]*1 = h'[i+k]
But h' differs from h only at positions i and j. So h'[i+k] = h[i+k] unless i+k = j (in which case h'[i+k] = 1) or i+k = i (in which case h'[i+k] = 0).
This is getting complicated. Let me just use the simpler formula:

Δc[m] = (new term at n'=i) - (old term at n'=i) + (new term at n'=j) - (old term at n'=j)

Actually, the cleanest way is:

c[m] = sum_{n'} h[n'+k] * (1-h[n'])

The only terms that change are those where n' = i, n' = j, n'+k = i, or n'+k = j.

For n' = i (k = m-n+1):
Old: h[i+k] * (1-h[i]) = h[i+k] * 0 = 0
New: h'[i+k] * (1-h'[i]) = h'[i+k] * 1

For n' = j:
Old: h[j+k] * (1-h[j]) = h[j+k] * 1 = h[j+k]
New: h'[j+k] * (1-h'[j]) = h'[j+k] * 0 = 0

For n'+k = i, i.e., n' = i-k:
Old: h[i] * (1-h[i-k]) = 1 * (1-h[i-k])
New: h'[i] * (1-h'[i-k]) = 0 * (1-h'[i-k]) = 0

Old: h[j] * (1-h[j-k]) = 0 * (1-h[j-k]) = 0
New: h'[j] * (1-h'[j-k]) = 1 * (1-h'[j-k])

- h'[i+k] = h[i+k] unless i+k ∈ {i, j}. If i+k=i (k=0), h'[i+k]=0. If i+k=j, h'[i+k]=1.

def compute_c5_raw(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr)

def compute_c5(h):
    n = len(h)
    return compute_c5_raw(h) * 2.0 / n

def generate_initial(n, parent_h):
    n_parent = len(parent_h)
    x_parent = np.linspace(0, 2, n_parent + 1)
    h_ext = np.append(parent_h, parent_h[0])
    x_new = np.linspace(0, 2, n)
    h_interp = np.interp(x_new, x_parent, h_ext)
    n_ones = n // 2
    idx = np.argsort(-h_interp)
    h = np.zeros(n)
    h[idx[:n_ones]] = 1
    return h
