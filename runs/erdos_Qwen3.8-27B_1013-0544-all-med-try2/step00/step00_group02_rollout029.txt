import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(c)

def smooth_obj(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(c)
    return m + np.log(np.sum(np.exp(c - m))) / beta

def smooth_grad(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx  # length 2n-1
    m = np.max(c)
    p = np.exp(beta * (c - m))
    p_sum = p.sum()
    p = p / p_sum  # probability weights
    
    grad = np.zeros(n)
    for j in range(n):
        for s in range(2 * n - 1):
            k = s - (n - 1)
            i = j - k
            if 0 <= i <= n - 1:
                grad[j] += p[s] * (1.0 - h[i])
            if 0 <= j + k <= n - 1:
                grad[j] -= p[s] * h[j + k]
    return grad

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def local_search(h, n, max_time):
    start = time.time()
    best = h.copy()
    best_c5 = compute_c5(best)
    for sweep in range(1000):
        if time.time() - start > max_time:
            break
        improved = False
        for i in range(n):
            for delta in [0.1, 0.05, 0.02, 0.01]:
                for sign in [1, -1]:
                    h_try = best.copy()
                    h_try[i] = np.clip(best[i] + sign * delta, 0, 1)
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best = h_try
                        best_c5 = c5_try
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    def tl():
        return budget_s - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    bounds = [(0.0, 1.0)] * n
    
    # Stage 1: SLSQP with analytical gradient, increasing beta
    for beta in [20, 50, 100, 200, 500]:
        if tl() < 30:
            break
        def obj(h):
            return smooth_obj(h, n, beta)
        def grad(h):
            return smooth_grad(h, n, beta)
        def con(h):
            return np.sum(h) - n / 2.0
        
        res = minimize(obj, best_h, jac=grad, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con}],
                      options={'maxiter': 5000, 'ftol': 1e-16})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Stage 2: Projected gradient descent
    if tl() > 60:
        h_curr = best_h.copy()
        beta = 200
        for iteration in range(2000):
            if tl() < 5:
                break
            g = smooth_grad(h_curr, n, beta)
            # Projected gradient step
            lr = 0.01
            h_try = h_curr - lr * g

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def smooth_obj(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(c)
    return m + np.log(np.sum(np.exp(c - m))) / beta

def smooth_grad(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(c)
    p = np.exp(beta * (c - m))
    p /= p.sum()
    
    grad = np.zeros(n)
    for j in range(n):
        for s in range(2 * n - 1):
            k = s - (n - 1)
            i = j - k
            if 0 <= i <= n - 1:
                grad[j] += dx * p[s] * (1.0 - h[i])
            if 0 <= j + k <= n - 1:
                grad[j] -= dx * p[s] * h[j + k]
    return grad

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
