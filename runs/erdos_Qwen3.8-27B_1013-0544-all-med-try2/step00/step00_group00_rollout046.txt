import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * dx

def smooth_max_obj(h, n, m, mu):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    values = corr * dx
    max_val = np.max(values)
    shifted = m * (values - max_val)
    shifted = np.clip(shifted, -50.0, 0.0)
    log_sum = max_val + np.log(np.sum(np.exp(shifted))) / m
    penalty = mu * (np.sum(h) - n / 2.0) ** 2
    return log_sum + penalty

def project_to_constraint(h, n):
