            grad[js2] += -h[js2 - ss]
    return grad * dx / top_k


def optimize_adam(h0, n_points, target_sum, n_iters, start_alpha, end_alpha, base_lr, rng, time_limit, start_time):
    """Adam optimization with annealed log-sum-exp temperature."""
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        frac = it / max(n_iters - 1, 1)
        alpha = start_alpha * (end_alpha / start_alpha) ** frac
        lr = base_lr * (1.0 - 0.4 * frac)
        _, grad = lse_obj_grad(h0, alpha)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h0 = h0 - lr * m_hat / (np.sqrt(v_hat) + eps)
        h0 = project_feasible(h0, target_sum)
        c5 = compute_c5(h0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h0.copy()
            m *= 0.5
            v *= 0.5
    return best_h, best_c5


def try_binary_restart(rng, n_points, target_sum, time_limit, start_time):
    """Create a random binary sequence and optimize briefly."""
    one_positions = rng.choice(n_points, size=int(target_sum), replace=False)
    h = np.zeros(n_points)
    h[one_positions] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    global_start = time.time()
    rng = np.random.RandomState(seed)
    n_points = 70
    target_sum = n_points / 2.0

    # Get initial values
    h_init = None
    try:
        if initial_h_values is not None:
            h_init = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    if h_init is None:
        h_init = np.full(n_points, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Adam with annealed temperature (60% budget)
    phase1_time = budget_s * 0.60
    h1, c5_1 = optimize_adam(
        h_init.copy(), n_points, target_sum,
        n_iters=50000, start_alpha=3.0, end_alpha=0.3,
        base_lr=0.3, rng=rng,
        time_limit=phase1_time, start_time=global_start
    )
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Subgradient refinement on hard max (20% budget)
    phase2_time = global_start + budget_s * 0.80
    h2 = best_h.copy()
    n_refine = 50000
    for it in range(n_refine):
        if time.time() > phase2_time:
            break
        grad = subgrad_topk(h2, top_k=3)
        lr = 0.03 * (1.0 - it / max(n_refine, 1)) + 0.003
        h2 = project_feasible(h2 - lr * grad, target_sum)
        c5 = compute_c5(h2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2.copy()

    # Phase 3: Try a few binary restarts with short optimization (20% budget)
    phase3_time = budget_s - 3
    for restart in range(6):
        if time.time() > phase3_time:
            break
        h_br = try_binary_restart(rng, n_points, target_sum, phase3_time, global_start)
        c5_br = compute_c5(h_br)
        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br.copy()
        # Short Adam optimization from this restart
        time_left = phase3_time - time.time()
        if time_left > 5:
            h_r, c5_r = optimize_adam(
                h_br.copy(), n_points, target_sum,
                n_iters=5000, start_alpha=2.0, end_alpha=0.3,
                base_lr=0.2, rng=rng,
                time_limit=time_left, start_time=time.time()
            )
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Also try optimizing from initial with different parameters
    if time.time() < global_start + budget_s - 5:
        time_left = global_start + budget_s - 5 - time.time()
        h_alt, c5_alt = optimize_adam(
            h_init.copy(), n_points, target_sum,
            n_iters=20000, start_alpha=1.0, end_alpha=0.1,
            base_lr=0.15, rng=rng,
            time_limit=time_left, start_time=time.time()
        )
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()

    return (best_h, best_c5, n_points)
