    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def repair(h, n):

    target = n / 2.0
    s = np.sum(h)
    if s > 0:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        # Fix sum after clipping
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:

            else:

                    remaining += dec

        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return res.x
    except:
        return h

    """Coordinate descent optimization"""
    rng = np.random.RandomState(0)
    target = n / 2.0
    h = h.copy()
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            # Try perturbing h[j]
            best_val = h[j]
            best_obj = c5_smooth(h, temp)
            
            # Try a few candidate values
            candidates = [0.0, 1.0, best_val + 0.1, best_val - 0.1,
                         best_val + 0.05, best_val - 0.05]
            candidates = [max(0.0, min(1.0, c)) for c in candidates]
            
            for c in candidates:
                if abs(c - h[j]) < 1e-6:
                    continue
                # Adjust another coordinate to maintain sum
                h[j] = c
                diff = target - np.sum(h)
                if abs(diff) > 1e-6:
                    # Find a coordinate to adjust

                        adj = np.argmax(1.0 - h)
                    else:
                        adj = np.argmin(h)
                    if adj == j:

                    h[adj] = np.clip(h[adj] + diff, 0.0, 1.0)

                    best_val = c

            diff = target - np.sum(h)
            if abs(diff) > 1e-6:

            break

    for period, amp in [(0.4, 0.45), (0.6, 0.45), (0.5, 0.5), (2.0/3, 0.4), (0.3, 0.5)]:
        x = np.arange(n) * 2.0 / n
        h_p = 0.5 + amp * np.sin(2 * np.pi * x / period)
        starts.append(repair(h_p, n))
    # binary balanced
    if n % 2 == 0:
        h_b = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h_b[idx] = 1.0
        starts.append(h_b)
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    h0 = np.array(initial_h_values, dtype=float)
    best_h = h0.copy()
    best_n = len(h0)
    best_c5 = c5_exact(h0)

    # Phase 1: SLSQP with annealed temperature on parent
    for temp in [0.1, 0.05, 0.02, 0.01, 0.005]:
        if tl() < 80:
            break
        h_opt = slsqp_opt(best_h, best_n, temp=temp, maxiter=1000)
        c5v = c5_exact(h_opt)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_opt

    # Phase 2: Nelder-Mead on best
    if tl() > 120:
        try:
            res = minimize(lambda x: c5_exact(x), best_h, method='Nelder-Mead',
                         options={'maxiter': 20000, 'xatol': 1e-10, 'fatol': 1e-12,
                                  'adaptive': True})
            c5v = c5_exact(res.x)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = repair(res.x, best_n)
        except Exception:
            pass

    # Phase 3: Coordinate descent refinement
    if tl() > 60:
        h_cd = coord_descend(best_h, best_n, sweeps=50, temp=0.005)
        c5v = c5_exact(h_cd)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_cd

    # Phase 4: Try different n values
    n_tries = sorted(set([best_n, 50, 70, 100, 120, 150, 200, 250, 300]))
    for n_t in n_tries:
        if tl() < 50:
            break
        if n_t == best_n:
            continue
        starts = make_starts(n_t, seed + n_t, best_h, best_n)
        for h_s in starts[:4]:  # limit starts per n
            if tl() < 25:
                break
            h_opt = slsqp_opt(h_s, n_t, temp=0.03, maxiter=500)
            c5v = c5_exact(h_opt)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_opt
                best_n = n_t
                if tl() > 40:
                    h_opt2 = slsqp_opt(h_opt, n_t, temp=0.01, maxiter=800)
                    c5v2 = c5_exact(h_opt2)
                    if c5v2 < best_c5:
                        best_c5 = c5v2
                        best_h = h_opt2

    # Phase 5: Final refinement on best
    if tl() > 40:
        for temp in [0.005, 0.003]:
            if tl() < 20:
                break
            h_opt = slsqp_opt(best_h, best_n, temp=temp, maxiter=1500)
            c5v = c5_exact(h_opt)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_opt

    # Final repair and verification
    best_h = repair(best_h, best_n)
    best_c5 = c5_exact(best_h)

    return (best_h, best_c5, best_n)
