(1-h[j-s_k]) * [j-s_k valid] + (-h[j+s_k]) * [j+s_k valid]

So for each k, the gradient is determined by the shift s_k. I can compute all of them at once if I'm clever, but for n=47 it's fine to just loop.

```python
def grad_smooth_max(h, T=50.0):
    """Compute gradient of smoothed max of correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)  # array of length 2n-1
    c = corr * T
    c -= c.max()  # for numerical stability
    weights = np.exp(c)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        s = k - (n - 1)
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        nr = np.arange(lo, hi + 1)
        j1 = nr + s
        j2 = nr
        grad[j1] += weights[k] * (1.0 - h[j2]) * dx
        grad[j2] += weights[k] * (-h[j1]) * dx
    return grad
```

```python
def project_feasible(h, target):

    for _ in range(200):

        free = (h > 1e-10) & (h < 1.0 - 1e-10)

                idx = int(np.argmax(h))
                h[idx] = max(0.0, h[idx] + d)
```
