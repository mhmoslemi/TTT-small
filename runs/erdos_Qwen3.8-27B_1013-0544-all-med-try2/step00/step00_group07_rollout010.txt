    vals = corr * dx
    m = np.max(vals)
    a = alpha * (vals - m)
    weights = np.exp(a) / np.sum(np.exp(a))  # shape (2n-1,)
    
    # Gradient
    grad = np.zeros(n)
    for i in range(n):
        # First term: sum_s weights[s] * (1-h[i+s]) for valid i+s
        # Second term: sum_s weights[s] * h[i-s] for valid i-s

                grad[i] += weights[si] * (1.0 - h[i + shift])
            if 0 <= i - shift < n:
                grad[i] -= weights[si] * h[i - shift]

    return grad
