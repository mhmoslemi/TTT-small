def compute_c5(h):

def interpolate_to_n(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    # Renormalize
    target_sum = new_n / 2.0
    current_sum = np.sum(h_new)
    h_new = h_new * (target_sum / current_sum)
    h_new = np.clip(h_new, 0, 1)
    # Fix sum (clipping might have changed it)
    diff = target_sum - np.sum(h_new)
    if diff > 0:
        # Add to elements with room
        rooms = 1.0 - h_new
        idx = np.argsort(rooms)[::-1]
        for i in idx:
            add = min(diff, rooms[i])
            h_new[i] += add
            diff -= add

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def sa_optimize(h, T_init, T_min, cool, deadline, rng, delta_range=(0.01, 0.1)):
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h)
    best_h_local = h.copy()
    best_c5_local = c5_cur
    T = T_init
    stagnant = 0
    dlo, dhi = delta_range

    while time.time() < deadline:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue

        delta = float(rng.uniform(dlo, dhi))

        if rng.random() < 0.5:
            src, dst = j, i
        else:
            src, dst = i, j

        max_t = h[src] if h[src] < 1.0 - h[dst] else 1.0 - h[dst]
        if max_t <= 1e-12:
            continue

        actual = delta if delta < max_t else max_t
        old_src = h[src]
        old_dst = h[dst]

        h[src] = old_src - actual
        h[dst] = old_dst + actual

        c5_new = compute_c5(h)
        d = c5_new - c5_cur

        if d < 0.0 or (T > T_min and rng.random() < np.exp(-d / T)):
            c5_cur = c5_new
            stagnant += 1
            if c5_new < best_c5_local:
                best_c5_local = c5_new
                best_h_local = h.copy()
                stagnant = 0
        else:
            h[src] = old_src
            h[dst] = old_dst

        T = T * cool
        if T < T_min:
            T = T_min
        if stagnant > 50000:
            T = T * 2.0
            if T < 0.003:
                T = 0.003
            stagnant = 0

    return best_h_local, best_c5_local

def interp_normalize(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    target = new_n / 2.0
    s = h_new.sum()
    if s > 1e-10:
        h_new *= (target / s)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            m = h_new < 1.0
        else:
            m = h_new > 0.0
        cnt = int(m.sum())
        if cnt > 0:
            h_new[m] += diff / cnt
    return np.clip(h_new, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    ihv = globals().get('initial_h_values')
    if ihv is not None:
        h_init = np.array(ihv, dtype=np.float64).copy()
    else:
        h_init = np.full(47, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    n0 = len(h_init)

    # Phase 1: SA at initial resolution (35% of budget)
    dl1 = t0 + budget_s * 0.35
    h1, c5_1 = sa_optimize(h_init, 0.004, 1e-8, 0.999995, dl1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Higher resolution (30% of budget)
    if time.time() < deadline - 100:
        n1 = min(n0 * 2, 200)
        h2_init = interp_normalize(best_h, n1)
        dl2 = t0 + budget_s * 0.68
        rng2 = np.random.default_rng(seed + 12345)
        h2, c5_2 = sa_optimize(h2_init, 0.003, 1e-8, 0.999995, dl2, rng2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 3: Fine-tune with smaller delta range (remaining budget)
    if time.time() < deadline - 10:
        rng3 = np.random.default_rng(seed + 67890)
        h3, c5_3 = sa_optimize(best_h.copy(), 0.0015, 1e-9, 0.999999, deadline, rng3, delta_range=(0.002, 0.04))
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3

    # Final validation: ensure sum constraint
    n_final = len(best_h)
    target_sum = n_final / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-8:
        if diff > 0:
            m = best_h < 1.0
        else:
            m = best_h > 0.0
        cnt = int(m.sum())
        if cnt > 0:
            best_h[m] += diff / cnt
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n_final
