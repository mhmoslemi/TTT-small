import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def project_feasible(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1} using water-filling."""
    h = np.clip(h, 0, 1)
    for _ in range(100):
        diff = h.sum() - target_sum
        if abs(diff) < 1e-12:
            break
        # Find which elements can be adjusted
        if diff > 0:
            # Need to decrease; can decrease elements > 0
            mask = h > 0
            if not mask.any():
                break
            adj = diff / mask.sum()
            h[mask] -= adj
            h = np.clip(h, 0, 1)
        else:
            # Need to increase; can increase elements < 1
            mask = h < 1
            if not mask.any():
                break
            adj = -diff / mask.sum()
            h[mask] += adj
            h = np.clip(h, 0, 1)
    return h

def soft_max_obj_and_grad(h, eps=0.01):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Soft max
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / eps)
    sum_exp = np.sum(exp_c)
    f = c_max + eps * np.log(sum_exp)
    p = exp_c / sum_exp  # softmax probabilities
    
    # Gradient
    grad = np.zeros(n)
    for j in range(n):
        for k in range(2 * n - 1):
            # ∂c_k/∂h_j
            m = k - n + 1
            g = 0.0
            if 0 <= j + m < n:
                g += (1.0 - h[j + m])
            if 0 <= j - m < n:
                g -= h[j - m]
            grad[j] += p[k] * g * dx
    return f, grad

    n = len(initial_h_values)

        """Projected gradient descent with backtracking."""
        h = h.copy()
        best_local_c5 = compute_c5(h)
        best_local_h = h.copy()

                best_local_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

_g_target = 0.0

def _sum_eq(x):
    return x.sum() - _g_target

def _slsqp_obj(x):
    return softmax_obj_grad(x, 0.003)[0]

def compute_c5(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = h.sum() - target_sum
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-12
            if not mask.any():
                break
            h[mask] -= diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            h[mask] += (-diff) / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    ec = np.exp((c - cm) / eps)
    se = np.sum(ec)
    f = cm + eps * np.log(se)
    p = ec / se
    grad = np.zeros(n)
    for j in range(n):
        dmin1 = max(-j, -(n - 1))
        dmax1 = min(n - 1 - j, n - 1)
        d1 = np.arange(dmin1, dmax1 + 1)
        dmin2 = max(j - n + 1, -(n - 1))
        dmax2 = min(j, n - 1)
        d2 = np.arange(dmin2, dmax2 + 1)
        grad[j] = np.dot(p[d1 + n - 1], 1.0 - h[j + d1]) - np.dot(p[d2 + n - 1], h[j - d2])
    grad *= dx
    return f, grad

def pgd_optimize(h, n_iter, eps, target_sum, max_time):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.05
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > max_time:
            break
        f, grad = softmax_obj_grad(h, eps)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        for _ in range(50):
            h_new = project_feasible(h - step * grad, target_sum)
            f_new, _ = softmax_obj_grad(h_new, eps)
            if f_new <= f * (1.0 - 1e-5) or step < 1e-12:
                break
            step *= 0.5
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            step = min(step * 1.15, 0.2)
        elif it > 200:
            step *= 0.7
    return best_h, best_c5

def slsqp_polish(h, n, target_sum, t_start, budget_s):
    if budget_s - (time.time() - t_start) < 30:
        return h, compute_c5(h)
    _g_target = target_sum
    cons = [{'type': 'eq', 'fun': _sum_eq}]
    bnds = [(0, 1)] * n
    try:
        res = minimize(_slsqp_obj, h.copy(), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': 400, 'ftol': 1e-12})
        if hasattr(res, 'x'):
            hn = project_feasible(res.x, target_sum)
            return hn, compute_c5(hn)
    except Exception:
        pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    global _g_target
    t_start = time.time()
    rng = np.random.RandomState(seed)

    h0 = project_feasible(np.array(initial_h_values, dtype=float).copy(), target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def tleft():
        return budget_s - (time.time() - t_start)

    # Phase 1: PGD from initial point
    t1 = min(80, tleft() - 30)
    if t1 > 10:
        h1, c51 = pgd_optimize(h0, 40000, 0.01, target_sum, t1)
        if c51 < best_c5:

    # Phase 2: SLSQP polish
    h2, c52 = slsqp_polish(best_h, n, target_sum, t_start, budget_s)
    if c52 < best_c5:

    # Phase 3: Sharper soft-max refinement
    if tleft() > 60:

        if c53 < best_c5:

        h3, c53 = slsqp_polish(best_h, n, target_sum, t_start, budget_s)
        if c53 < best_c5:
            best_h, best_c5 = h3, c53

    # Phase 4: Multiple restarts with diverse initializations
    while tleft() > 50:
        r = rng.random()
        if r < 0.25:
            h_rand = (rng.rand(n) < 0.5).astype(float)
            h_rand = project_feasible(h_rand, target_sum)
        elif r < 0.45:
            h_rand = project_feasible(rng.uniform(0, 1, n), target_sum)
        elif r < 0.65:
            h_rand = project_feasible(best_h + 0.2 * rng.randn(n), target_sum)
        elif r < 0.8:
            h_rand = project_feasible(best_h + 0.4 * rng.randn(n), target_sum)
        else:
            stride = rng.randint(2, 8)
            h_rand = np.zeros(n)
            h_rand[::stride] = 1.0
            h_rand = project_feasible(h_rand, target_sum)

        t_per = max(5, min(25, (tleft() - 30) // 3))
        h_opt, c5_opt = pgd_optimize(h_rand, 6000, 0.01, target_sum, t_per)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
