    """Compute gradient of c_k with respect to h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # c_k = sum_i h[i+k-(n-1)] * (1-h[i]) * dx
    # For valid i: 0 <= i < n and 0 <= i+k-(n-1) < n
    # i ranges from max(0, n-1-k) to min(n-1, n-1+k)
    i_lo = max(0, n - 1 - k)
    i_hi = min(n - 1, n - 1 + k)
    if i_lo > i_hi:
        return grad
    i_vals = np.arange(i_lo, i_hi + 1)
    a_idx = i_vals + k - (n - 1)  # indices into h for the "a" part
    # grad[j] gets contribution from:
    # - (1-h[i]) when j = a_idx[i] = i + k - (n-1)
    # - (-h[a_idx[i]]) when j = i
    for i, (ii, ai) in enumerate(zip(i_vals, a_idx)):
        grad[ai] += (1 - h[ii])
        grad[ii] += -h[ai]
    return grad * dx
