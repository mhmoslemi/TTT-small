import numpy as np
import time

def project_feasible(g, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(200):
        mid = 0.5*(lo + hi)
        if np.clip(g - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - 0.5*(lo+hi), 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0/n
    return float((np.correlate(h, 1-h, mode='full') * dx).max())

def soft_obj_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    vals = corr * dx
    M = vals.max()
    ev = np.exp(tau*(vals - M))
    S = ev.sum()
    w = ev / S
    obj = M + np.log(S)/tau
    grad = np.zeros(n)
    for idx in range(len(w)):
        s = idx - (n-1)
        wi = w[idx]*dx
        a1, b1 = max(0, -s), min(n, n-s)
        if a1 < b1:
            grad[a1:b1] += wi*(1.0 - h[a1+s:b1+s])
        a2, b2 = max(0, s), min(n, n+s)
        if a2 < b2:
            grad[a2:b2] -= wi*h[a2-s:b2-s]
    return obj, grad

def pgd_optimize(h0, n, deadline, rng):
    dx = 2.0/n
    h = project_feasible(h0, n)
    best_h, best_c5 = h.copy(), compute_c5(h)
    tau, lr, stall = 5.0, 1.0, 0
    while time.time() < deadline:
        _, grad = soft_obj_grad(h, dx, tau)
        noise = 0.005*(1+stall/50.0)*rng.randn(n)
        h = project_feasible(h - lr*grad + noise, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h, stall = c5, h.copy(), 0
        else:
            stall += 1
            lr *= 0.93

            tau = min(tau*1.5, 500.0)

        if stall > 400:
