from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 value for given h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_grad(h):
    """Compute C5 value and gradient with respect to h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    c5 = corr[max_idx] * dx
    
    # Gradient of corr[max_idx] with respect to h
    # corr[max_idx] = sum_i h[i] * (1-h[i+shift]) where shift = max_idx - (n-1)
    shift = max_idx - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        # sum_{i=0}^{n-1-shift} h[i] * (1-h[i+shift])
        for i in range(n - shift):
            grad[i] += dx * (1 - h[i+shift])
            grad[i+shift] += -dx * h[i]

        # sum_{i=s}^{n-1} h[i] * (1-h[i-s])

            grad[i] += dx * (1 - h[i-s])
            grad[i-s] += -dx * h[i]

def project_feasible(y, target_sum):
    """Project y onto {x: 0 <= x[i] <= 1, sum(x) = target_sum}"""
    lo, hi = -1.0, 1.0  # range for lambda
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2, 0, 1)
    return x

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def compute_multi_gradient(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_indices = np.argsort(corr)[::-1][:top_k]
    max_corr = corr[top_indices[0]]
    min_top = corr[top_indices[-1]]
    if max_corr - min_top < 1e-12:
        weights = np.ones(top_k) / top_k

        weights = np.exp(30.0 * (corr[top_indices] - max_corr))
        weights /= weights.sum()
    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        s = idx - (n - 1)
        i_low = max(0, -s)
        i_high = min(n - 1, n - 1 - s)

def project_feasible(y, target_sum):

            lo = mid
        else:
            hi = mid
