import numpy as np

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 47  # Start with the initial sample count
    
    # Initial solution
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = h / sum_h * (n_points / 2)
    
    # Function to compute C5
    def compute_c5(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = h_values.reshape(-1, 1)
        ones_minus_h = 1 - h
        # Compute the correlation
        corr = signal.correlate(h, ones_minus_h, mode="full")
        # Take the max value and multiply by dx
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Function to optimize using gradient descent
    def optimize_c5(h_initial, max_iter=100, learning_rate=0.01, tol=1e-5):
        h = h_initial.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        for i in range(max_iter):
            # Compute the gradient
            n_points = len(h)
            dx = 2.0 / n_points
            h = h.reshape(-1, 1)
            ones_minus_h = 1 - h
            corr = signal.correlate(h, ones_minus_h, mode="full")
            max_corr_idx = np.argmax(corr)
            # Find the shift that contributes most to max_corr
            shift = max_corr_idx - (n_points - 1)
            if shift < 0:
                shift = 0
            elif shift >= n_points:
                shift = n_points - 1
            # Extract the relevant part of the correlation
            corr_shift = corr[shift:shift + n_points]
            # Compute the derivative
            grad = np.zeros_like(h)
            for j in range(n_points):
                # The gradient for h[j] is -sum(ones_minus_h[k - j]) over k where h[k] is involved
                # Approximate the derivative using finite differences
                h_plus = h.copy()
                h_plus[j] += 1e-5
                c5_plus = compute_c5(h_plus)
                h_minus = h.copy()
                h_minus[j] -= 1e-5
                c5_minus = compute_c5(h_minus)
                grad[j] = (c5_plus - c5_minus) / (2 * 1e-5)
            # Update the weights
            h = h - learning_rate * grad
            # Ensure bounds
            h = np.clip(h, 0, 1)
            # Normalize to maintain integral constraint
            h = h * (n_points / 2) / np.sum(h)
            # Update best solution
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            # Early stopping
            if abs(current_c5 - best_c5) < tol:
                break
        return best_h, best_c5
    
    # Initial optimization with the initial solution
    optimized_h, c5_bound = optimize_c5(h)
    
    # Try increasing the number of points if within time budget
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        n_points = 100  # Try with 100 points
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        optimized_h, c5_bound = optimize_c5(h)
    
    # Final check
    if time.time() - start_time > budget_s - 1:
        return optimized_h.tolist(), c5_bound, n_points
    
    return optimized_h.tolist(), c5_bound, n_points
