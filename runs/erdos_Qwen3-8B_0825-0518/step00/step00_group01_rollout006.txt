    dx = 2.0 / n_points
    sum_h = sum(initial_h_values)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h_values = [h * (n_points / 2) / sum_h for h in initial_h_values]
    
    def compute_c5(h):
        h = np.array(h)
        h_pad = np.zeros_like(h)
        h_pad[-len(h):] = h
        corr = np.correlate(h_pad, 1 - h, mode='full')
        max_overlap = np.max(corr * dx)
        return max_overlap
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds(0, 1)
    
    # Initial optimization attempt
    result = minimize(
        objective,
        initial_h_values,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Further refinement using a local optimization
    result = minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Use a genetic algorithm for further exploration
    def genetic_algorithm(pop_size=30, generations=20):
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = np.array([h * (n_points / 2) / np.sum(h) for h in population])

            parents = np.random.choice(population.shape[0], size=pop_size, p=1 / fitness)

            offspring = []

            population = np.array(offspring)
