```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Placeholder, assuming this is available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial guess
    n_points = len(initial_h_values)
    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    
    # Try refining with a smaller n_points for faster optimization
    n_points = min(n_points, 100)
    h = h_initial[:n_points]
    
    # Ensure the integral condition is met
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    
    # Define the optimization problem
    h_var = cp.Variable(n_points, nonneg=True)
    objective = cp.Minimize(cp.max(cp.sum(cp.multiply(h_var, 1 - cp.roll(h_var, 1)), axis=0)))
    constraints = [
        cp.sum(h_var) == n_points / 2,
        0 <= h_var,
        h_var <= 1
    ]
    
    # Solve the problem
    prob = cp.Problem(objective, constraints)
    try:
        result = prob.solve(solver=cp.ECOS, verbose=False, max_iter=1000)
        if result is not None:
            h_opt = np.round(h_var.value, 5)
            c5_bound = evaluate_erdos_solution(h_opt)
        else:
            h_opt = h
            c5_bound = evaluate_erdos_solution(h)
    except:
        h_opt = h
        c5_bound = evaluate_erdos_solution(h)
    
    # Refine further with a local optimization
    def objective_func(h_values):
        h_values = np.clip(h_values, 0, 1)
        scale = (n_points / 2) / np.sum(h_values)
        h_values = h_values * scale
        return evaluate_erdos_solution(h_values)
    
    result = opt.minimize(
        objective_func,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)]
    )
    
    h_opt = np.clip(result.x, 0, 1)
    scale = (n_points / 2) / np.sum(h_opt)
    h_opt = h_opt * scale
    c5_bound = evaluate_erdos_solution(h_opt)
    
    # Ensure we don't exceed the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # Try a few more iterations if time allows
        for _ in range(3):
            result = opt.minimize(
                objective_func,
                h_opt,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)]
            )
            h_opt = np.clip(result.x, 0, 1)
            scale = (n_points / 2) / np.sum(h_opt)
            h_opt = h_opt * scale
            c5_bound = evaluate_erdos_solution(h_opt)
    
    return (h_opt, c5_bound, n_points)
```
