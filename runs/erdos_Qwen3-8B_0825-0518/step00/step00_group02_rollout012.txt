import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(45) * 2 - 1  # Placeholder, actual initial construction should be used

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for faster evaluation

    # Initialize with the provided initial_h_values
    if len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 2 - 1  # Random initial guess

    # Ensure sum(h) * dx = 1
    dx = 2.0 / n_points
    h = h / (np.sum(h) * dx) * (n_points / 2)

    # Evaluate initial solution
    c5 = evaluate_erdos_solution(h)
    best_c5 = c5
    best_h = h.copy()

    # Function to evaluate the overlap integral for a given h
    def evaluate(h):
        nonlocal best_c5, best_h
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Optimization function with gradient descent
    def optimize_h(h):
        # Objective function to minimize
        def objective(h_vec):
            h = h_vec.reshape(-1)
            dx = 2.0 / len(h)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

        # Constraints
        def constraints(h_vec):
            h = h_vec.reshape(-1)
            return np.sum(h) - (len(h) / 2)

        # Optimization
        options = {'maxiter': 1000, 'ftol': 1e-6}
        result = optimize.minimize(
            objective,
            h.flatten(),
            method='L-BFGS-B',
            jac=None,
            bounds=[(0, 1) for _ in range(len(h))],
            constraints={'type': 'eq', 'fun': constraints},
            options=options
        )
        return result.x.reshape(-1)

    # Main loop to refine solution
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        h = optimize_h(h)
        current_c5 = evaluate(h)

        # If we can increase n_points and find a better solution, do so
        if len(h) < 200 and current_c5 < best_c5:
            n_points = 200
            dx = 2.0 / n_points
            h = np.random.rand(n_points) * 2 - 1
            h = h / (np.sum(h) * dx) * (n_points / 2)
            current_c5 = evaluate(h)

    # Final check and return the best solution found
    if best_c5 < 0.380876:
        return (best_h, best_c5, len(best_h))
    else:
        return (best_h, best_c5, len(best_h))
