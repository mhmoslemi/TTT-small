import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 45
    h_initial = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h_initial = np.sum(h_initial) * dx
    assert np.isclose(sum_h_initial, 1.0), "Initial h does not satisfy the integral constraint"
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h, dtype=float)
        h_extended = np.concatenate([h, np.zeros_like(h)])
        corr = np.correlate(h_extended, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Function to evaluate the overlap integral for all shifts
    def compute_all_correlations(h):
        h = np.array(h, dtype=float)
        h_extended = np.concatenate([h, np.zeros_like(h)])
        corr = np.correlate(h_extended, 1 - h, mode="full")
        return corr * dx
    
    # Optimization using CVXPY
    n_points = 45
    h = cp.Variable(n_points)
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]
    obj = cp.Minimize(0)  # We will evaluate C5 in the loop
    prob = cp.Problem(obj, constraints)
    
    # Initial guess
    h_initial = initial_h_values.copy()
    best_h = h_initial
    best_c5 = compute_c5(h_initial)
    
    # Optimization loop
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        
        # Solve the problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            result = prob.solve(solver=cp.SCS, verbose=False)
            if result is not None and cp.value(result) is not None:
                h_candidate = np.round(h.value, 5)
                c5 = compute_c5(h_candidate)
                if c5 < best_c5:
                    best_h = h_candidate
                    best_c5 = c5
        except:
            pass
        
        # Perturb the solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h_initial = np.clip(h_initial + perturbation, 0, 1)
    
    # Evaluate final best solution
    final_c5 = compute_c5(best_h)
    n_points = len(best_h)
    
    # Return the result
    return (best_h, final_c5, n_points)
