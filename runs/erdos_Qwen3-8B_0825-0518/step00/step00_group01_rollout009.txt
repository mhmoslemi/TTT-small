import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.concatenate([np.zeros(n_points), h[:-1]])
    correlate = signal.correlate(h, 1 - h, mode="full")
    c5 = np.max(correlate) * dx
    return c5

def initial_h_values():
    return np.load('initial_h_values.npy')

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Start with 100 points, can increase if needed
    dx = 2.0 / n_points
    initial_h = initial_h_values()[:n_points]  # Use initial values if available
    
    # Objective: minimize max correlation
    def objective(h):
        h = np.array(h)
        shifted = np.concatenate([np.zeros(n_points), h[:-1]])
        correlate = signal.correlate(h, 1 - h, mode="full")
        return np.max(correlate) * dx
    
    def objective_with_gradient(h):
        h = np.array(h)
        shifted = np.concatenate([np.zeros(n_points), h[:-1]])
        correlate = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlate) * dx
        # Estimate gradient (approximate with finite differences)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            c5_plus = objective(h_plus)
            grad[i] = (c5_plus - c5) / 1e-5
        return c5, grad
    
    # Optimization using gradient descent with line search
    h = initial_h.copy()
    learning_rate = 0.01
    max_iterations = 1000
    tolerance = 1e-6
    best_h = h.copy()
    best_c5 = objective(h)
    
    for iteration in range(max_iterations):
        # Enforce bounds
        h = np.clip(h, 0, 1)
        # Compute objective and gradient
        c5, grad = objective_with_gradient(h)
        # Update h
        h -= learning_rate * grad
        # Enforce sum constraint
        sum_h = np.sum(h)
        if sum_h != n_points / 2:
            h = h / sum_h * (n_points / 2)
        # Check convergence
        if abs(c5 - best_c5) < tolerance:
            break
        best_c5 = c5
        best_h = h.copy()
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Try a more sophisticated optimization with CVXPY
    # Define variables
    h_var = cp.Variable(n_points)
    # Objective: minimize max correlation
    # This is non-convex and hard to optimize with CVXPY, so we use a convex relaxation
    # Approximate with maximum over all shifts
    max_shift = 0
    for shift in range(1, n_points):
        # Compute correlation for this shift
        corr = cp.sum(h_var * (1 - h_var[shift:]))
        max_shift = cp.Max(max_shift, corr)
    obj = max_shift * dx
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2,
        0 <= h_var,
        h_var <= 1
    ]
    # Solve
    try:
        problem = cp.Problem(cp.Minimize(obj), constraints)
        problem.solve(solver=cp.SCIP)
        if problem.status == 'optimal':
            h = np.array(h_var.value)
    except:
        pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h)
    return h, final_c5, n_points
