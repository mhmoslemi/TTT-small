import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points_list = [100, 200, 300, 400, 500, 600, 700, 800, 900, 1000]
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)

        def objective_func(h_values):
            nonlocal best_c5
            h = h_values
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            h_pad = np.concatenate([h, np.zeros_like(h)])
            shifted = np.zeros_like(h_pad)
            for k in range(1, len(h_pad) - 1):
                shifted[k] = np.sum(h_pad[:len(h_pad) - k] * (1 - h_pad[k:]))
            c5 = np.max(shifted) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n_points
            return c5

        # Initial optimization with gradient descent
        h_opt = np.copy(h)
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            shifted = np.zeros_like(h_opt)
            for k in range(1, len(h_opt) - 1):
                shifted[k] = np.sum(h_opt[:len(h_opt) - k] * (1 - h_opt[k:]))
            c5 = np.max(shifted) * dx
            grad = np.zeros_like(h_opt)
            for i in range(len(h_opt)):
                h_test = h_opt.copy()
                h_test[i] += 1e-4
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                shifted_test = np.zeros_like(h_test)
                for k in range(1, len(h_test) - 1):
                    shifted_test[k] = np.sum(h_test[:len(h_test) - k] * (1 - h_test[k:]))
                c5_test = np.max(shifted_test) * dx
                grad[i] = (c5_test - c5) / 1e-4
            h_opt -= 0.1 * grad
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)

        # Evaluate the best h so far
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        shifted = np.zeros_like(h_opt)
        for k in range(1, len(h_opt) - 1):
            shifted[k] = np.sum(h_opt[:len(h_opt) - k] * (1 - h_opt[k:]))
        c5 = np.max(shifted) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            best_n = n_points

    return best_h, best_c5, best_n
