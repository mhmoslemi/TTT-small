def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = np.zeros(1)
    best_n_points = 0
    n_points = 120  # Higher resolution for better accuracy

    # Initial guess from provided initial_h_values
    initial_h = initial_h_values.copy()
    # Resample to n_points if needed
    if len(initial_h) != n_points:
        # Interpolate the initial_h to n_points samples
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)

    def objective(h_values):
        return evaluate_erdos_solution(h_values)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constraint_func(h_values):
        return np.sum(h_values) * (2.0 / n_points) - 1

    constraint = {'type': 'eq', 'fun': constraint_func}

    # Use more advanced optimization with multiple restarts
    np.random.seed(seed)
    for _ in range(3):  # Multiple restarts to escape local minima
        # Create a perturbed initial guess
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_initial = initial_h + perturbation
        perturbed_initial = np.clip(perturbed_initial, 0.0, 1.0)
        perturbed_initial = perturbed_initial / np.sum(perturbed_initial) * (n_points / 2)

        result = optimize.minimize(
            objective,
            perturbed_initial,
            bounds=bounds,
            constraints=constraint,
            method='SLSQP',
            options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
        )

        current_h = result.x
        current_c5 = evaluate_erdos_solution(current_h)
        # Ensure the constraint is satisfied
        if np.abs(np.sum(current_h) * (2.0 / n_points) - 1) > 1e-6:
            current_h = current_h / np.sum(current_h) * (n_points / 2)
            current_c5 = evaluate_erdos_solution(current_h)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
            best_n_points = n_points

        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s - 1:
            break

    return (best_h, best_c5, best_n_points)
