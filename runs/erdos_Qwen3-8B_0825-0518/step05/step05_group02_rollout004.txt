def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 120  # Starting with 120 points for balance between accuracy and speed
    
    # Function to generate initial guess with random perturbations
    def generate_initial_guess(n_points):
        h = np.zeros(n_points)
        h[:len(initial_h_values)] = initial_h_values[:n_points]
        h = h / np.sum(h) * (n_points / 2)
        h += np.random.uniform(-0.1, 0.1, size=n_points)  # Small random perturbations
        h = np.clip(h, 0, 1)  # Ensure values are within [0, 1]
        return h

    # Function to evaluate the C_5 value for a given h
    def objective(h_values):
        return evaluate_erdos_solution(h_values)

    # Function to enforce integral constraint
    def constraint_func(h_values):
        return np.sum(h_values) * (2.0 / n_points) - 1.0

    # Define bounds for each point
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Setup optimization problem with SLSQP method
    def optimize_h(initial_guess):
        constraint = {'type': 'eq', 'fun': constraint_func}
        result = optimize.minimize(
            objective,
            initial_guess,
            bounds=bounds,
            constraints=constraint,
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        return result.x

    # Run with multiple initializations to avoid local optima
    best_h = None
    best_c5 = float('inf')

    # Try multiple initializations
    for i in range(5):
        initial_guess = generate_initial_guess(n_points)
        optimized_h = optimize_h(initial_guess)
        current_c5 = evaluate_erdos_solution(optimized_h)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h

        # Check for time budget
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 1:
            break

    # Ensure constraint is satisfied
    if np.abs(np.sum(best_h) * (2.0 / n_points) - 1) > 1e-6:
        best_h = best_h / np.sum(best_h) * (n_points / 2)

    return (best_h, best_c5, n_points)
