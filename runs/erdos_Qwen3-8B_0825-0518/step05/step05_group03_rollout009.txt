def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = np.zeros(1)
    best_n_points = 0

    def generate_initial_h(n_points, seed):
        np.random.seed(seed)
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        return h

    n_point_options = [80, 100, 120]  # Different resolutions to balance accuracy and speed
    num_initials = 3  # Number of different initial configurations for diversity

    for n_points in n_point_options:
        dx = 2.0 / n_points
        initial_configs = [generate_initial_h(n_points, seed + i) for i in range(num_initials)]

        def objective(h_values):
            return evaluate_erdos_solution(h_values)

        bounds = [(0.0, 1.0) for _ in range(n_points)]

        def constraint_func(h_values):
            return np.sum(h_values) * dx - 1

        constraint = {'type': 'eq', 'fun': constraint_func}

        for initial_config in initial_configs:
            # Use L-BFGS-B with adaptive settings for better convergence
            result = optimize.minimize(
                objective,
                initial_config,
                bounds=bounds,
                constraints=constraint,
                method='L-BFGS-B',
                options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
            )

            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h)
            if np.abs(np.sum(current_h) * dx - 1) > 1e-6:
                current_h = current_h / np.sum(current_h) * (n_points / 2)
                current_c5 = evaluate_erdos_solution(current_h)

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                best_n_points = n_points

            elapsed_time = time.time() - start_time
            if elapsed_time >= budget_s - 1:
                break

    return (best_h, best_c5, best_n_points)
