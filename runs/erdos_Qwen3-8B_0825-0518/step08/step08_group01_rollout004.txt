        return evaluate_erdos_solution(h_values)

        return np.sum(h_values) - n_points / 2

    def optimize_with_bounds(h_values):
        result = minimize(
            evaluate,
            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='SLSQP',

            # Evaluate fitness
            fitness = [evaluate(h) for h in population]

            next_gen = [best_population] * elite_size
            
            # Select parents
            parents = np.random.choice(len(population), size=population_size - elite_size, replace=False)
            parents = [population[i] for i in parents]
            
            # Crossover
            for i in range(0, len(parents), 2):
                if i + 1 < len(parents):

                    child1 = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)

            # Mutation

            for h in next_gen:

                    tabu_list.pop(0)
                tabu_list.append(h)
            
            # Filter tabu solutions
            filtered_next_gen = []
            for h in next_gen:
                if not any(np.allclose(h, t, atol=1e-5) for t in tabu_list):

            population = next_gen

    best_result = optimize_with_bounds(ts_result)

        if current_c5 < best_c5:
            best_result = result
            best_c5 = current_c5
    
    return (best_result, best_c5, n_points)
