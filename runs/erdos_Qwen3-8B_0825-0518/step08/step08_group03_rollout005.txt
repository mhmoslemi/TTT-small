def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    
    # Start with a moderate number of points for speed
    n_points = 100
    dx = 2.0 / n_points
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess from given values, normalized properly
    initial_guess = initial_h_values.copy() / np.sum(initial_h_values) * (n_points / 2)
    
    # Objective: minimize the maximum overlap
    def objective(h_values):
        # Enforce constraint
        if np.abs(np.sum(h_values) - n_points / 2) > 1e-8:
            h_values = h_values / np.sum(h_values) * (n_points / 2)
        return evaluate_erdos_solution(h_values)
    
    # Function to compute the overlap
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return corr.max()
    
    # Global optimization with differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=30,
        mutation=(0.5, 1),
        recombination=0.6,

    )

    best_c5 = evaluate_erdos_solution(best_h)
    
    # Refinement stage with SLSQP for better convergence
    # Define constraints for optimization
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    
    # Local optimization
    result = optimize.minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints=constraint,
        method='SLSQP',
        options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
    )
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Perturb and refine with multiple restarts
    for _ in range(3):

        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        new_guess = best_h + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)

            objective,
            new_guess,
            bounds=bounds,
            constraints=constraint,
            method='SLSQP',
            options={'maxiter': 30, 'ftol': 1e-8, 'disp': False}
        )

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    # Check remaining time
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
