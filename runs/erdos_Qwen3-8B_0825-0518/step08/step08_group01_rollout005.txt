    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)

        return np.sum(h_values) - n_points / 2

    def optimize_with_bounds(h_values):
        result = minimize(

            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}

    def genetic_algorithm():
        population_size = 100

        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]

            # Evaluate fitness
            fitness = [evaluate(h) for h in population]

            # Keep best to pass to next generation
            next_gen = [best_population] * elite_size
            
            # Select parents
            parents = np.random.choice(len(population), size=population_size - elite_size, replace=False)
            parents = [population[i] for i in parents]

            for i in range(len(parents) - 1, 0, -1):
                a, b = np.random.choice(len(parents), 2, replace=False)
                if fitness[a] < fitness[b]:
                    parents[i] = parents[a]
                else:
                    parents[i] = parents[b]
            
            # Crossover
            for i in range(0, len(parents), 2):
                if i + 1 < len(parents):
                    parent1 = parents[i]
                    parent2 = parents[i + 1]

                    next_gen.append(child2)
            
            # Adaptive mutation based on fitness
            for i in range(len(next_gen)):
                for j in range(n_points):

            population = next_gen
            best_population = population[0] if evaluate(population[0]) < evaluate(best_population) else best_population

    # Local optimization using SLSQP
    best_result = optimize_with_bounds(ga_result)

    for _ in range(5):

        if current_c5 < best_c5:
            best_result = result
            best_c5 = current_c5
    
    return (best_result, best_c5, n_points)
