def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    
    # Try with different n_points: 100, 150, 200
    n_points_list = [200, 150, 100]
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 1:
            break
        
        dx = 2.0 / n_points
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Generate a better initial guess based on a wave-like pattern
        initial_guess = np.zeros(n_points)
        for i in range(n_points):
            angle = (i / n_points) * 2 * np.pi
            initial_guess[i] = 0.5 + 0.5 * np.cos(2 * angle)
        
        # Scale initial guess to satisfy the integral condition
        initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
        
        def evaluate(h_values):
            """Compute the overlap integral for the current h_values."""
            return evaluate_erdos_solution(h_values)
        
        def constraint_sum(h_values):
            """Constraint to ensure the integral of h over [0,2] is 1."""
            return np.sum(h_values) - n_points / 2
        
        # Use differential evolution for global search
        def objective_function(x):
            return evaluate(x)
        
        # Run differential evolution to find a good starting point
        result = differential_evolution(
            objective_function,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        
        # Local optimization using SLSQP
        result = minimize(
            evaluate,
            result.x,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        
        current_c5 = evaluate(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
            best_n = n_points
        
        # Try a few perturbations and refine with SLSQP
        for _ in range(5):
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)

            new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
            result = minimize(
                evaluate,
                new_guess,
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                method='SLSQP',
                options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
            )

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n = n_points
    
    return (best_h, best_c5, best_n)
