def run(seed=42, budget_s=60, **kwargs):

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h_values):
        if np.abs(np.sum(h_values) - n_points / 2) > 1e-6:
            h_values = h_values / np.sum(h_values) * (n_points / 2)
        return evaluate_erdos_solution(h_values)

    def local_optimization(guess):
        options = {'maxiter': 100, 'ftol': 1e-8, 'disp': False}

            objective,

            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},

            options=options

    best_h, best_c5 = local_optimization(initial_h)
