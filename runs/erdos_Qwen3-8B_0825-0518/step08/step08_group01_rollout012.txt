    dx = 2.0 / n_points
    
    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    def optimize_with_bounds(h_values):
        result = minimize(
            evaluate,
            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}

    def differential_evolution():
        population_size = 50

        crossover_probability = 0.7

        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]

            fitness = [evaluate(h) for h in population]

            next_gen = [best_population] * elite_size
            
            # Generate new population
            for i in range(population_size - elite_size):
                # Select three distinct random individuals
                a, b, c = np.random.choice(len(population), 3, replace=False)

                trial_fitness = evaluate(trial)
                if trial_fitness < fitness[i]:

                    next_gen.append(population[i])
            
            population = next_gen

            best_fitness = min(best_fitness, evaluate(best_population))

    de_result = differential_evolution()

    best_result = optimize_with_bounds(de_result)

    # Early stop if time is up
    elapsed_time = time.time() - start_time

    for _ in range(5):

        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)

            best_result = result
            best_c5 = current_c5
    
    return (best_result, best_c5, n_points)
