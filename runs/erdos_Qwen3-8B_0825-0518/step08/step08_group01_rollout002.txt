    dx = 2.0 / n_points
    
    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def optimize_with_bounds(h_values):
        result = minimize(

            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}

    def differential_evolution():
        population_size = 50

        # Initialize population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h / np.sum(h) * (n_points / 2) for h in population]
        
        best_population = population[0]

            # Evaluate fitness
            fitness = [evaluate(h) for h in population]

            next_gen = [best_population] * elite_size
            
            # Select parents
            parents = np.random.choice(len(population), size=population_size - elite_size, replace=False)
            parents = [population[i] for i in parents]
            
            # Crossover and mutation
            for i in range(len(parents)):

                indices = np.random.choice(len(parents), size=3, replace=False)

                mutant = p1 + mutation_factor * (p2 - p3)

                trial = np.copy(p1)

                        trial[j] = mutant[j]

                # Replace if better
                if evaluate(trial) < evaluate(population[i]):

            population = next_gen

        return best_population

    de_result = differential_evolution()
    
    # Local optimization using SLSQP
    best_result = optimize_with_bounds(de_result)

            best_result = result
            best_c5 = current_c5
    
    return (best_result, best_c5, n_points)
