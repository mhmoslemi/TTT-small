def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 200  # Higher resolution for better accuracy
    dx = 2.0 / n_points
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess based on provided starting pattern
    initial_h = np.zeros(n_points)
    pattern = initial_h_values
    pattern = np.repeat(pattern, n_points // len(pattern))
    initial_h[:len(pattern)] = pattern[:n_points]
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    # Objective function to minimize the max overlap
    def objective(h_values):
        # Normalize to ensure constraint is satisfied
        h_values = h_values / np.sum(h_values) * (n_points / 2)
        # Evaluate overlap integral
        overlap = correlate(h_values, 1 - h_values, mode="full")
        return np.max(overlap) * dx
    
    # Global optimization with differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=50,
        mutation=(0.5, 1),
        recombination=0.6,
        disp=False,
        polish=True,
        tol=1e-8,
        seed=seed
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Local optimization with SLSQP
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    options = {'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    result = minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        method='SLSQP',
        options=options
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Multi-start refinement with perturbations
    for _ in range(10):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        new_guess = best_h + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
        result = minimize(
            objective,
            new_guess,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            method='SLSQP',
            options=options
        )
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    # Final check within time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
