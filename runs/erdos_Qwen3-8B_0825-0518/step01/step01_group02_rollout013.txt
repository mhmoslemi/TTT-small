import numpy as np
import scipy.optimize as optimize
import time

from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):

    initial_h_values = np.array([0.4925, 0.5833, 0.6458, 0.6865, 0.7135, 0.7342, 0.7495, 0.7602, 0.7673, 0.7713,

                                 0.193, 0.168, 0.143, 0.118, 0.093, 0.068, 0.043, 0.018, 0.0])

    # Set up the initial parameters
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    time_limit = budget_s
    start_time = time.time()

    def compute_c5(h):
        h = np.array(h)
        shifted_h = np.concatenate([h[-n_points + 1:], h], axis=0)
        one_minus_h = 1 - h
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return np.max(corr) * dx

    # Try initial point
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    # Try uniform point
    uniform_h = np.full(n_points, 0.5)
    c5_uniform = compute_c5(uniform_h)
    if c5_uniform < best_c5:
        best_c5 = c5_uniform
        best_h = uniform_h.copy()

    # Try alternating point
    alt_h = np.zeros(n_points)
    for i in range(n_points):
        alt_h[i] = 0.5 if i % 2 == 0 else 0.0
    c5_alt = compute_c5(alt_h)
    if c5_alt < best_c5:
        best_c5 = c5_alt
        best_h = alt_h.copy()

    # Try to optimize using L-BFGS-B
    def objective_func(x):
        return compute_c5(x)

    # Optimization loop
    while time.time() - start_time < time_limit:
        result = optimize.minimize(
            objective_func,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)]

        c5_new = compute_c5(new_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h.copy()

    # Return the best solution found
    return best_h, best_c5, n_points
