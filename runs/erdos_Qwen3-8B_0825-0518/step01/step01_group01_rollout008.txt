import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
from scipy.integrate import simps
from scipy.optimize import minimize
from scipy.interpolate import interp1d

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode='full')) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    import time
    start_time = time.time()

    # Parameters for the search
    n_points = 100  # Start with 100 samples for speed
    dx = 2.0 / n_points
    max_iterations = 100
    tolerance = 1e-5

    # Initial guess
    h_initial = initial_h_values[:n_points].copy()
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2.0)  # Ensure integral is 1

    # Objective function to minimize the max overlap
    def objective_function(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0)  # Ensure integral is 1
        overlap = signal.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

    # Constraint: integral of h must be 1
    def constraint(h):
        h = h.reshape(-1)
        return np.sum(h) - n_points / 2.0

    # Bounds for h
    bounds = [(0, 1) for _ in range(n_points)]

    # Use L-BFGS-B for optimization
    result = minimize(objective_function, h_initial, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint, 'jac': lambda h: np.zeros(n_points)})

    # Check if we have a valid solution
    if result.success:
        h_opt = result.x
        c5_value = evaluate_erdos_solution(h_opt)
    else:
        h_opt = h_initial
        c5_value = evaluate_erdos_solution(h_opt)

    # Try a few more iterations with different initial points
    for _ in range(3):
        # Perturb the initial guess
        h_perturbed = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2.0)

        result = minimize(objective_function, h_perturbed, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint, 'jac': lambda h: np.zeros(n_points)})

        if result.success:
            h_opt = result.x
            c5_value = evaluate_erdos_solution(h_opt)
        else:
            h_opt = h_perturbed
            c5_value = evaluate_erdos_solution(h_opt)

        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break

    # Final evaluation
    c5_value = np.max(signal.correlate(h_opt, 1 - h_opt, mode='full')) * dx
    h_values = h_opt

    return (h_values, c5_value, n_points)
