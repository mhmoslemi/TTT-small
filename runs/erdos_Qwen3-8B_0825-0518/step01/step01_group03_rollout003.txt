import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = signal.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

        constraint = cp.sum(h_var) == n / 2

        # Since the integral is equivalent to the convolution
        
        # Reformulate objective function to be a maximization of the max overlap
        # We minimize the negative of the max overlap
        # But since it is a convex problem, we can't directly minimize the max
        # So use a convex relaxation
        # Use the trick where we introduce a variable for the max value
        # We minimize the maximum overlap by introducing a variable t and constraints
        t = cp.Variable()
        constraints = [constraint, t >= 0]
