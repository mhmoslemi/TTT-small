import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions and variables
evaluate_erdos_solution = None
initial_h_values = None

def run(seed=42, budget_s=60, **kwargs):
    # Define parameter for the number of samples
    n_points = 100  # Start with 100 samples, which is under the 1k limit
    dx = 2.0 / n_points
    max_iterations = 100  # Limit the number of iterations to stay within time budget
    best_c5 = float('inf')
    best_h = None

    # Function to compute overlap integral
    def compute_c5(h):
        h = np.array(h)
        h_expanded = np.zeros_like(h)
        h_expanded[:len(h)] = h
        h_expanded[len(h):] = h
        h_expanded = h_expanded[:2 * len(h)]
        corr = np.correlate(h_expanded, 1 - h, mode="full")
        return np.max(corr) * dx

    # Initial guess: use the provided initial_h_values, or uniform distribution if not available
    if initial_h_values is not None:
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_points]
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

    # Optimization loop
    start_time = time.time()
    for _ in range(max_iterations):
        # Define optimization variables
        h = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h) == n_points / 2]

        # Objective: minimize the max overlap integral
        # To handle max, we minimize each possible correlation
        # However, this is computationally heavy. Instead, use a heuristic
        # We'll use an iterative approach with gradient descent
        h_value = h_initial.copy()
        for _ in range(10):  # 10 iterations per optimization step
            # Compute the overlap integral
            h_expanded = np.concatenate([h_value, h_value])
            corr = np.correlate(h_expanded, 1 - h_value, mode="full")
            c5 = np.max(corr) * dx

            # Update h_value using gradient descent
            # We minimize the max overlap integral by adjusting h_value
            # We use a simple gradient descent with penalty for the max correlation
            penalty = 100 * (c5 - best_c5)
            grad = np.zeros(n_points)
            for i in range(n_points):
                for k in range(n_points):
                    # Approximate gradient with finite differences
                    h_test = h_value.copy()
                    h_test[i] += 1e-6
                    c5_test = compute_c5(h_test)
                    grad[i] += (c5_test - c5) / 1e-6
            h_value -= 0.01 * grad

            # Ensure h_value stays within [0,1] and sum to n_points / 2
            h_value = np.clip(h_value, 0, 1)
            h_value = h_value / np.sum(h_value) * (n_points / 2)

            # Update best solution
            current_c5 = compute_c5(h_value)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_value.copy()

        # Check if time is up
        if time.time() - start_time > budget_s - 1:
            break

    return best_h, best_c5, n_points
