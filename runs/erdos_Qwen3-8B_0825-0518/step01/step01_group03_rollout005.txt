def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_shifted = np.zeros((n_points, n_points))
    for i in range(n_points):
        h_shifted[i, :] = h
    corr = np.zeros(n_points)
    for k in range(n_points):
        corr[k] = np.sum(h * (1 - h_shifted[k, :])) * dx
    return np.max(corr)
