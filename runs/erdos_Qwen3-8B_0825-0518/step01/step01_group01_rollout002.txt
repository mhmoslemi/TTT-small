        # Initialize with a random solution

        h = h.tolist()
        # Optimization
        t_start = time.time()
        while time.time() - t_start < budget_s:
            # Compute overlap for all possible shifts
            h = np.array(h)
            shifted_h = np.zeros((n_points, n_points))
            for k in range(1, n_points):
                shifted_h[k:] = h[:-k]
            overlap = np.dot(h, 1 - shifted_h)
            max_overlap = np.max(overlap) * dx
            # Objective: minimize max_overlap
            # Use L-BFGS-B optimization
            def objective(h_flat):
                h = h_flat.copy()
                h = h / np.sum(h) * (n_points / 2)
                shifted_h = np.zeros((n_points, n_points))
                for k in range(1, n_points):
                    shifted_h[k:] = h[:-k]
                overlap = np.dot(h, 1 - shifted_h)
                return np.max(overlap) * dx

            def constraint(h_flat):
                return np.sum(h_flat) - n_points / 2

            bounds = [(0, 1) for _ in range(n_points)]
            result = opt.minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},

                best_c5 = current_c5

                best_n_points = n_points

            if np.abs(current_c5 - best_c5) < 1e-6:
