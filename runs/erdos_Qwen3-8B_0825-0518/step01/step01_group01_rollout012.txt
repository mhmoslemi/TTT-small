import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assuming initial_h_values.npy is provided

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for faster evaluation
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        return evaluate_erdos_solution(h)
    
    # Function to optimize h for a given n_points
    def optimize_h(n_points):
        dx = 2.0 / n_points
        h_initial = np.random.rand(n_points) * 2 - 1  # Initial guess
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)  # Ensure sum is n_points / 2
        h_initial = np.clip(h_initial, 0, 1)
        
        # Bounds for each h[i]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function: minimize the maximum overlap
        def objective(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)  # Normalize
            return evaluate_overlap(h)
        
        # Use L-BFGS-B for optimization
        result = optimize.minimize(objective, h_initial, bounds=bounds, method='L-BFGS-B', tol=1e-6)
        return result.x, result.fun
    
    # Try different n_points
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Run optimization with current n_points
        h_opt, c5 = optimize_h(n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
    
    # Try refining the best solution
    if best_h is not None and time.time() - start_time < budget_s - 1:
        n_points_refine = 600
        dx = 2.0 / n_points_refine
        h_refine = np.interp(np.linspace(0, 2, n_points_refine), np.linspace(0, 2, len(best_h)), best_h)
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points_refine / 2) / np.sum(h_refine)
        h_refine = np.clip(h_refine, 0, 1)
        
        # Bounds for each h[i]
        bounds = [(0, 1) for _ in range(n_points_refine)]
        
        # Objective function
        def objective(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points_refine / 2) / np.sum(h)
            return evaluate_overlap(h)
        
        # Run optimization
        result = optimize.minimize(objective, h_refine, bounds=bounds, method='L-BFGS-B', tol=1e-6)
        h_final = result.x
        final_c5 = result.fun
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final
    
    return best_h, best_c5, len(best_h)
