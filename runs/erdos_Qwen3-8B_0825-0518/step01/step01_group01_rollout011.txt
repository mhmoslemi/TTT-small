```python
import numpy as np
import scipy.optimize as opt
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial number of points
    n_points = 100  # Start with a smaller number for faster evaluation
    dx = 2.0 / n_points
    
    # Initial guess from provided initial_h_values
    h_initial = np.array(initial_h_values[:n_points])  # Ensure it's of size n_points
    
    # Function to compute the max overlap C_5
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        # Normalize h to ensure the integral is 1
        h_norm = h / (np.sum(h) * dx)
        # Compute the overlap using convolution
        overlap = np.correlate(h_norm, 1 - h_norm, mode="full")
        # Multiply by dx to get the correct integral scale
        c5 = np.max(overlap * dx)
        return c5
    
    # Optimization function with L-BFGS-B to minimize C5
    def objective(h):
        return compute_c5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Bounds for h: 0 <= h[i] <= 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimize
    result = opt.minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=cons)
    
    # Extract the best solution
    best_h = result.x
    
    # Compute the best C5 value
    best_c5 = compute_c5(best_h)
    
    # Print status
    elapsed_time = time.time() - start_time
    print(f"Optimized with n_points = {n_points}, C5 = {best_c5:.6f}, elapsed: {elapsed_time:.2f}s")
    
    # If we have time, try increasing n_points for better accuracy
    if elapsed_time < budget_s - 10:
        n_points = 200
        dx = 2.0 / n_points
        # Adjust initial guess to fit new n_points (interpolate or resize)
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        # Recompute with new n_points
        result = opt.minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=cons)
        best_h = result.x
        best_c5 = compute_c5(best_h)
        elapsed_time = time.time() - start_time
        print(f"Optimized with n_points = {n_points}, C5 = {best_c5:.6f}, elapsed: {elapsed_time:.2f}s")
    
    # Return the result
    return (best_h.tolist(), best_c5, n_points)
```
