import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Define initial n_points
    n_points = 47  # Start with the initial construction
    h_values = initial_h_values.copy()
    
    # Optimize the function
    start_time = time.time()
    
    # Define the function to minimize the maximum overlap
    def objective_function(h_values):
        # Reshape to 1D array
        h = h_values.reshape(-1)
        dx = 2.0 / len(h)
        
        # Compute the overlap integral for each k
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        
        # Return the maximum overlap
        return np.max(overlap)
    
    # Define constraints
    def constraints(h_values):
        # Reshape to 1D array
        h = h_values.reshape(-1)
        return [np.sum(h) - len(h) / 2]  # Ensure the integral is 1
    
    # Optimization settings
    n_points = 50  # Try with a slightly larger n_points for more accuracy
    
    # Define the optimization variables
    h = cp.Variable(n_points)
    
    # Objective: minimize the maximum overlap
    dx = 2.0 / n_points
    h_vec = h.value if h.value is not None else h
    overlap = np.correlate(h_vec, 1 - h_vec, mode="full") * dx
    max_overlap = cp.Max(overlap)
    
    # Formulate the optimization problem
    prob = cp.Problem(cp.Minimize(max_overlap), [cp.sum(h) == n_points / 2, 0 <= h, h <= 1])
    
    # Solve the problem
    prob.solve(solver=cp.ECOS, verbose=False)
    
    # Extract the result
    h_values = h.value
    c5_bound = np.max(np.correlate(h_values, 1 - h_values, mode="full") * dx)
    
    # Ensure time constraint is respected
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print(f"Timeout reached. Returning best solution so far.")
    
    return h_values, c5_bound, n_points
