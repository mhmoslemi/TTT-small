```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.signal import correlate
from scipy.sparse import diags
from scipy.sparse.linalg import lsqr
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Helper functions

def calculate_c5(h, dx):
    h_flat = np.concatenate([h, h])
    h_flipped = np.flip(h)
    overlap = correlate(h_flat, h_flipped, mode='full')
    overlap = overlap[1:-1]  # Remove redundant elements
    max_overlap = np.max(overlap)
    return max_overlap * dx

def optimize_h(n_points, seed=42):
    np.random.seed(seed)
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    return h

def objective(h_flat, dx, n_points):
    h = h_flat[:n_points]
    return calculate_c5(h, dx)

def constraint_sum(h_flat, dx, n_points):
    h = h_flat[:n_points]
    return np.sum(h) - n_points / 2

def update_h(h_flat, dx, n_points, step, alpha):
    h = h_flat[:n_points]
    h += step * alpha * np.random.randn(n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return np.concatenate([h, np.zeros(n_points)])

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = h[:n_points]
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = calculate_c5(h, dx)
    
    for i in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        c5_perturbed = calculate_c5(h_perturbed, dx)
        if c5_perturbed < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_perturbed

        # Gradient descent with line search
        def objective_func(x):
            return calculate_c5(x, dx)
        
        def constraint_func(x):
            return np.sum(x) - n_points / 2
        
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        
        result = opt.minimize(objective_func, h, method='L-BFGS-B', bounds=bounds, constraints=cons)
        h = result.x
        c5 = calculate_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    return best_h, best_c5, n_points
```
