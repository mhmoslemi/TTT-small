def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 100  # Moderate number of points to balance speed and accuracy
    dx = 2.0 / n_points
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        return evaluate_erdos_solution(h_values)
    
    # Function to enforce the integral constraint
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Bounds: each h_i must be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Construct an initial guess based on the provided initial_h_values
    initial_guess = initial_h_values / np.sum(initial_h_values) * (n_points / 2)
    
    # Optimization options
    options = {
        'maxiter': 150,
        'ftol': 1e-8,
        'disp': False
    }
    
    # Optimize for the given initial guess
    result = optimize.minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        method='SLSQP',
        options=options
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Normalize the solution if the constraint is not satisfied
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    # Ensure we do not exceed the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    # Perform multiple independent optimizations starting from different structured guesses
    num_starts = 5
    best_c5_value = best_c5
    best_h_values = best_h
    
    for _ in range(num_starts):
        # Construct structured random starting point
        structured_guess = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                structured_guess[i] = np.random.uniform(0.6, 1.0)
            elif i % 5 == 1 or i % 5 == 2:
                structured_guess[i] = np.random.uniform(0.3, 0.7)
            else:
                structured_guess[i] = np.random.uniform(0.0, 0.3)
        structured_guess = structured_guess / np.sum(structured_guess) * (n_points / 2)
        
        result = optimize.minimize(
            objective,
            structured_guess,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='SLSQP',
            options=options
        )
        
        current_h = result.x
        current_c5 = evaluate_erdos_solution(current_h)
        
        if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
            current_h = current_h / np.sum(current_h) * (n_points / 2)
        
        if current_c5 < best_c5_value:
            best_c5_value = current_c5
            best_h_values = current_h
    
    return (best_h_values, best_c5_value, n_points)
