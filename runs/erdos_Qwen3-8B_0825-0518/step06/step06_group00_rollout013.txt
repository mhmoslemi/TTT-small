import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * 2.0 / len(h)

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 50  # Use fewer points for faster evaluation
    dx = 2.0 / n_points
    
    def evaluate_overlap(h_values):
        return evaluate_erdos_solution(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Start from the initial provided structure
    initial_guess = initial_h_values[:n_points]
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Use a more robust optimization strategy with multiple starts
    options = {
        'maxiter': 150,
        'ftol': 1e-8,
        'disp': False
    }
    
    # Initial optimization
    result = optimize.minimize(
        evaluate_overlap,
        initial_guess,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        method='L-BFGS-B',
        options=options
    )
    
    best_h = result.x
    best_c5 = evaluate_overlap(best_h)
    
    # Normalize if constraint not met
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    # Track the best solution
    best_c5_value = best_c5
    best_h_values = best_h
    
    # Perform multiple random restarts
    num_starts = 5
    for _ in range(num_starts):
        # Construct a structured random starting point
        structured_guess = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                structured_guess[i] = np.random.uniform(0.6, 1.0)
            elif i % 5 == 1 or i % 5 == 2:
                structured_guess[i] = np.random.uniform(0.3, 0.7)
            else:
                structured_guess[i] = np.random.uniform(0.0, 0.3)
        structured_guess = structured_guess / np.sum(structured_guess) * (n_points / 2)
        
        result = optimize.minimize(
            evaluate_overlap,
            structured_guess,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='L-BFGS-B',
            options=options
        )
        
        current_h = result.x
        current_c5 = evaluate_overlap(current_h)
        
        if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
            current_h = current_h / np.sum(current_h) * (n_points / 2)
        
        if current_c5 < best_c5_value:
            best_c5_value = current_c5
            best_h_values = current_h
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h_values, best_c5_value, n_points)
    
    # Use genetic algorithm for global search
    from scipy.optimize import differential_evolution
    
    def objective(h_values):
        return evaluate_overlap(h_values)
    
    # Use a larger population for better exploration
    result_de = differential_evolution(
        objective,
        bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        polish=True,
        popsize=15,
        maxiter=50,
        mutation=(0.5, 1),
        recombination='blend',
        disp=False
    )
    
    de_h = result_de.x
    de_c5 = evaluate_overlap(de_h)
    
    if np.abs(np.sum(de_h) - n_points / 2) > 1e-6:
        de_h = de_h / np.sum(de_h) * (n_points / 2)
    
    if de_c5 < best_c5_value:
        best_c5_value = de_c5
        best_h_values = de_h
    
    return (best_h_values, best_c5_value, n_points)
