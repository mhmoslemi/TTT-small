def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 100  # Moderate resolution for balance between speed and accuracy
    
    dx = 2.0 / n_points
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Objective function
    def objective(h_values):
        # Ensure constraint is met
        if np.abs(np.sum(h_values) - n_points / 2) > 1e-6:
            h_values = h_values / np.sum(h_values) * (n_points / 2)
        return evaluate_erdos_solution(h_values)
    
    # Construct a structured initial guess based on the current best
    initial_guess = np.array(initial_h_values)
    initial_guess = initial_guess[:n_points]
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    best_h = initial_guess
    best_c5 = evaluate_erdos_solution(best_h)
    
    # First stage: Global optimization with DE
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=50,
        mutation=(0.5, 1),
        recombination=0.6,
        disp=False,
        polish=True,
        tol=1e-8,
        seed=seed
    )
    current_h = result.x
    current_c5 = evaluate_erdos_solution(current_h)
    if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
        current_h = current_h / np.sum(current_h) * (n_points / 2)
        current_c5 = evaluate_erdos_solution(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h
    
    # Second stage: Local refinement with SLSQP
    options = {'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    result = optimize.minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
        method='SLSQP',
        options=options
    )
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Third stage: Multi-start refinement with perturbations
    for _ in range(5):
        # Perturb and normalize
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        new_guess = best_h + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
        result = optimize.minimize(
            objective,
            new_guess,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
            method='SLSQP',
            options=options
        )
        current_h = result.x
        current_c5 = evaluate_erdos_solution(current_h)
        if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
            current_h = current_h / np.sum(current_h) * (n_points / 2)
            current_c5 = evaluate_erdos_solution(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    # Final check within time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
