def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 150  # Good balance between precision and speed
    dx = 2.0 / n_points
    
    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess from given initial_h_values
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    # Function to perform local optimization
    def local_optimize(h_values):
        result = minimize(
            evaluate,
            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        return result.x
    
    # Function to generate a random perturbation
    def perturb(h_values, scale=0.1):
        perturbation = np.random.uniform(-scale, scale, size=n_points)
        new_h = h_values + perturbation
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = new_h / np.sum(new_h) * (n_points / 2)
        return new_h
    
    # Function to perform a single step of global search
    def global_search_step(best_h):
        # Use differential evolution to explore a small neighborhood
        result = differential_evolution(
            lambda x: evaluate(x),
            bounds,
            strategy='best1bin',
            popsize=10,
            maxiter=10,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        return result.x
    
    # Initial optimization
    best_h = local_optimize(h_initial)
    best_c5 = evaluate(best_h)
    
    # Use differential evolution for global search
    de_result = differential_evolution(
        lambda x: evaluate(x),

        strategy='best1bin',
        popsize=20,
        maxiter=40,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    best_h = de_result.x
    best_c5 = evaluate(best_h)
    
    # Early stop if time is up
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)

    for _ in range(20):
        # Generate a new perturbed guess
        new_h = perturb(best_h)
        new_c5 = evaluate(new_h)
        
        # Local optimization on the perturbed guess
        new_h_opt = local_optimize(new_h)
        new_c5_opt = evaluate(new_h_opt)
        
        # Update best solution
        if new_c5_opt < best_c5:
            best_h = new_h_opt
            best_c5 = new_c5_opt
    
    # Additional refinement with local optimization
    best_h = local_optimize(best_h)
    best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
