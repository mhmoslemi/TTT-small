def run(seed=42, budget_s=60, **kwargs):

    n_points = 100  # Moderate resolution for balance between speed and accuracy

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Constraint: integral of h over [0,2] is 1
    def constraint_func(h):
        return np.sum(h) - n_points / 2

    # Objective function to minimize the maximum overlap
    def objective(h_values):
        # Normalize to ensure the constraint is met
        if np.abs(np.sum(h_values) - n_points / 2) > 1e-6:
            h_values = h_values / np.sum(h_values) * (n_points / 2)

        overlap = np.correlate(h_values, 1 - h_values, mode="full")
        return max(overlap) * dx

    for _ in range(5):
