def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 100  # Balanced choice for efficiency and quality

    dx = 2.0 / n_points

    # Initial guess
    initial_h_values_normalized = initial_h_values.copy() / np.sum(initial_h_values) * (n_points / 2)
    initial_h = initial_h_values_normalized
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(initial_h)

    # Define the constraint function to maintain integral condition
    def constraint_func(h):
        return np.sum(h) - n_points / 2

    # Objective function: minimize the maximum overlap
    def objective(h_values):
        # Normalize to ensure the integral constraint is satisfied
        if np.abs(np.sum(h_values) - n_points / 2) > 1e-6:
            h_values = h_values / np.sum(h_values) * (n_points / 2)
        # Compute the maximum overlap
        overlap = np.correlate(h_values, 1 - h_values, mode="full")
        return max(overlap) * dx

    # Use differential evolution for global search
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=50,
        mutation=(0.5, 1),
        recombination=0.6,
        disp=False,
        polish=True,
        tol=1e-8,
        seed=seed
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Apply SLSQP for local refinement
    constraints = {'type': 'eq', 'fun': constraint_func}
    options = {'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    result = optimize.minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options=options
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Multi-start refinement with perturbations
    for _ in range(5):
        perturbation = np.random.normal(0, 0.05, size=n_points)
        new_guess = best_h + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
        result = optimize.minimize(
            objective,
            new_guess,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options=options
        )
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)
