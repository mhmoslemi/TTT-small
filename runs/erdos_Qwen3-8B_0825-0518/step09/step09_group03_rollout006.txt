evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * 2.0 / len(h)

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 200  # Higher resolution for better accuracy
    
    dx = 2.0 / n_points
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Objective function
    def objective(h_values):
        # Ensure constraint is met
        if np.abs(np.sum(h_values) - n_points / 2) > 1e-6:
            h_values = h_values / np.sum(h_values) * (n_points / 2)
        return evaluate_erdos_solution(h_values)
    
    # Initial guess from provided initial_h_values
    initial_guess = np.resize(initial_h_values, n_points)
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    best_h = initial_guess
    best_c5 = evaluate_erdos_solution(best_h)
    
    # First stage: Differential Evolution for global optimization
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=50,
        mutation=(0.5, 1),
        recombination=0.6,
        disp=False,
        polish=True,
        tol=1e-8,
        seed=seed
    )
    current_h = result.x
    current_c5 = evaluate_erdos_solution(current_h)
    if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
        current_h = current_h / np.sum(current_h) * (n_points / 2)
        current_c5 = evaluate_erdos_solution(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h
    
    # Second stage: Local refinement with SLSQP
    options = {'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    result = optimize.minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
        method='SLSQP',
        options=options
    )
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Third stage: Spectral Optimization for better convergence
    def spectral_objective(h_values):
        h_values = np.clip(h_values, 0.0, 1.0)
        if np.abs(np.sum(h_values) - n_points / 2) > 1e-6:
            h_values = h_values / np.sum(h_values) * (n_points / 2)
        return evaluate_erdos_solution(h_values)
    
    # Build sparse matrix for correlation
    def build_correlation_matrix(h_values):
        n = len(h_values)
        corr_matrix = np.zeros((n, n))
        for i in range(n):
            for k in range(n):
                corr_matrix[i, k] = np.sum(h_values[i:] * (1 - h_values[:n - i]))
        return corr_matrix
    
    # Eigenvalues for optimization
    def spectral_optimize(h_values):
        A = build_correlation_matrix(h_values)
        A_sparse = csr_matrix(A)
        eigvals = eigs(A_sparse, k=1, which='SM')[0]
        return eigvals
    
    # Local refinement with spectral optimization
    result = optimize.minimize(
        spectral_optimize,
        best_h,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
        method='L-BFGS-B',
        options=options
    )
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Final check within time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
