def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 180  # Increased to 180 for better accuracy
    dx = 2.0 / n_points

    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess from the provided initial_h_values, scaled to n_points
    initial_h = np.copy(initial_h_values[:n_points])
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    h_initial = initial_h

    initial_c5 = evaluate(h_initial)

    def optimize_with_bounds(h_values):
        result = minimize(
            evaluate,
            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        return result.x

    def objective_function(x):
        h = x
        return evaluate(h)

    def differential_evolution_search():
        result = differential_evolution(
            objective_function,
            bounds,
            strategy='best1bin',
            popsize=30,
            maxiter=50,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        return result.x

    de_result = differential_evolution_search()

    best_result = optimize_with_bounds(de_result)
    best_c5 = evaluate(best_result)

    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_result, best_c5, n_points)

    # Refinement with local optimization
    for _ in range(30):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        new_guess = best_result + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
        result = optimize_with_bounds(new_guess)
        current_c5 = evaluate(result)
        if current_c5 < best_c5:
            best_result = result
            best_c5 = current_c5

    # Additional refinement with a global search
    def global_search():
        result = differential_evolution(
            objective_function,
            bounds,
            strategy='best1bin',
            popsize=30,
            maxiter=20,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + 1
        )
        return result.x

    global_result = global_search()
    global_result = optimize_with_bounds(global_result)
    global_c5 = evaluate(global_result)

    if global_c5 < best_c5:
        best_result = global_result
        best_c5 = global_c5

    return (best_result, best_c5, n_points)
