def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 200  # 200 points for better resolution, still under 1k
    dx = 2.0 / n_points
    
    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = np.copy(initial_h_values[:n_points])
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    def optimize_with_bounds(h_values):
        result = minimize(
            evaluate,
            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        return result.x
    
    def differential_evolution_search():
        result = differential_evolution(
            lambda x: evaluate(x),
            bounds,
            strategy='best1bin',
            popsize=20,
            maxiter=50,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        return result.x
    
    # Initial optimization
    best_result = optimize_with_bounds(initial_guess)
    best_c5 = evaluate(best_result)
    
    # Run differential evolution search
    de_result = differential_evolution_search()
    de_result = np.clip(de_result, 0.0, 1.0)
    de_result = de_result / np.sum(de_result) * (n_points / 2)
    current_c5 = evaluate(de_result)
    
    if current_c5 < best_c5:
        best_result = de_result
        best_c5 = current_c5
    
    # Early stopping check
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_result, best_c5, n_points)
    
    # Local refinements
    for _ in range(20):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        new_guess = best_result + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
        result = optimize_with_bounds(new_guess)
        current_c5 = evaluate(result)
        if current_c5 < best_c5:
            best_result = result
            best_c5 = current_c5
    
    return (best_result, best_c5, n_points)
