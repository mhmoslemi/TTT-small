def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 150  # Balanced approach between accuracy and computational feasibility
    dx = 2.0 / n_points

    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initialize from the given initial_h_values, scaled and trimmed to fit n_points
    initial_h = initial_h_values[:n_points]
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    initial_h = np.clip(initial_h, 0.0, 1.0)

    # Objective function for optimization
    def objective_function(x):
        return evaluate(x)

    # Strategy 1: Differential Evolution for global search
    def differential_evolution_optimize():
        result = differential_evolution(
            objective_function,
            bounds,
            strategy='best1bin',
            popsize=20,
            maxiter=50,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + 1
        )
        return result.x

    # Strategy 2: Sequential Least Squares Programming for local refinement
    def slsqp_refine(h_values):
        result = minimize(
            evaluate,
            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False, 'eps': 1e-8}
        )
        return result.x

    # Strategy 3: Local Perturbations with SLSQP
    def perturb_and_refine(h_values, iterations=30):
        best_h = h_values
        best_c5 = evaluate(h_values)
        for i in range(iterations):
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            new_h = best_h + perturbation
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = new_h / np.sum(new_h) * (n_points / 2)
            result = slsqp_refine(new_h)
            current_c5 = evaluate(result)
            if current_c5 < best_c5:
                best_h = result
                best_c5 = current_c5
        return best_h, best_c5

    # Begin optimization
    de_result = differential_evolution_optimize()
    best_h, best_c5 = perturb_and_refine(de_result)

    # Early stop if time is up
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 2:
        return (best_h, best_c5, n_points)

    # Additional refinements with different perturbation sizes
    for perturbation_scale in [0.05, 0.02, 0.01]:
        for _ in range(15):
            perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
            new_h = best_h + perturbation
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = new_h / np.sum(new_h) * (n_points / 2)
            result = slsqp_refine(new_h)
            current_c5 = evaluate(result)
            if current_c5 < best_c5:
                best_h = result
                best_c5 = current_c5

    return (best_h, best_c5, n_points)
