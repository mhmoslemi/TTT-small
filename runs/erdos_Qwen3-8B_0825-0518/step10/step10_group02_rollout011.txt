def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 150  # Balance between accuracy and time
    dx = 2.0 / n_points

    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Start with a refined version of the initial h_values
    initial_h = np.copy(initial_h_values[:n_points])
    initial_h = initial_h[:n_points] / np.sum(initial_h[:n_points]) * (n_points / 2)

    # Use SLSQP with constraints
    def objective_function(x):
        return evaluate(x)

    def optimize_with_bounds(h_values):
        result = minimize(
            objective_function,
            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False, 'eps': 1e-8}
        )
        return result.x

    # Initial optimization from initial_h
    best_result = optimize_with_bounds(initial_h)
    best_c5 = evaluate(best_result)

    # Early stop if time is up
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_result, best_c5, n_points)

    # Use a more aggressive differential evolution to explore the space
    def differential_evolution_search():
        result = differential_evolution(
            objective_function,
            bounds,
            strategy='best1bin',
            popsize=20,
            maxiter=100,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        return result.x

    de_result = differential_evolution_search()
    result = optimize_with_bounds(de_result)
    current_c5 = evaluate(result)
    if current_c5 < best_c5:
        best_result = result
        best_c5 = current_c5

    # Additional local refinements with random perturbations and SLSQP
    for _ in range(20):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        new_guess = best_result + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
        result = optimize_with_bounds(new_guess)
        current_c5 = evaluate(result)
        if current_c5 < best_c5:
            best_result = result
            best_c5 = current_c5

    return (best_result, best_c5, n_points)
