def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    # Use n_points = 150 for better accuracy without exceeding the budget
    n_points = 150
    dx = 2.0 / n_points

    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess based on the given initial_h_values scaled to n_points
    h_initial = initial_h_values[:n_points].copy()
    h_initial = h_initial[:n_points] / np.sum(h_initial[:n_points]) * (n_points / 2)
    h_initial = np.clip(h_initial, 0.0, 1.0)

    # Function to evaluate the overlap integral
    def objective_function(x):
        return evaluate(x)

    # Differential Evolution for global search with adaptive strategy
    def differential_evolution_search():
        result = differential_evolution(
            objective_function,
            bounds,
            strategy='best1bin',
            popsize=20,
            maxiter=50,
            tol=1e-8,
            mutation=(0.5, 1.0),
            recombination=0.8,
            seed=seed
        )
        return result.x

    # Sequential Least Squares Programming for local refinement
    def optimize_with_bounds(h_values):
        result = minimize(
            objective_function,
            h_values,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False, 'eps': 1e-8}
        )
        return result.x

    # Run differential evolution search
    de_result = differential_evolution_search()
    best_result = de_result
    best_c5 = evaluate(best_result)

    # Early stop if time is up
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_result, best_c5, n_points)

    # Further refinement with perturbations and SLSQP
    for _ in range(15):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        new_guess = best_result + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
        result = optimize_with_bounds(new_guess)
        current_c5 = evaluate(result)
        if current_c5 < best_c5:
            best_result = result
            best_c5 = current_c5
            elapsed_time = time.time() - start_time
            if elapsed_time > budget_s - 1:
                break

    return (best_result, best_c5, n_points)
