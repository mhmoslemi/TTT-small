def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 150
    dx = 2.0 / n_points

    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initialize with initial_h_values scaled to n_points
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial = h_initial[:n_points] / np.sum(h_initial[:n_points]) * (n_points / 2)

    # Define objective
    def objective(x):
        h = x.copy()
        return evaluate(h)

    # Use a more advanced optimization strategy: differential evolution + local refinement
    def differential_evolution_search():
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        return result.x

    # Refinement with SLSQP
    def optimize_with_bounds(h_values):
        result = minimize(
            objective,
            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='SLSQP',
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False, 'eps': 1e-8}
        )
        return result.x

    # Run differential evolution
    de_result = differential_evolution_search()
    best_result = optimize_with_bounds(de_result)
    best_c5 = evaluate(best_result)

    # Early stop if time is up
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_result, best_c5, n_points)

    for _ in range(10):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)

        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
        result = optimize_with_bounds(new_guess)
        current_c5 = evaluate(result)
        if current_c5 < best_c5:
            best_result = result
            best_c5 = current_c5

    # Try different starting points for further exploration
    for _ in range(5):

        current_c5 = evaluate(result)
        if current_c5 < best_c5:
            best_result = result
            best_c5 = current_c5

    return (best_result, best_c5, n_points)
