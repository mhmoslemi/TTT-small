def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 160  # 160 points for better resolution and performance
    dx = 2.0 / n_points
    
    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess based on provided initial values
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)

    initial_c5 = evaluate(h_initial)
    best_result = h_initial
    best_c5 = initial_c5
    
    # Optimizer settings
    def optimize_func(h_values):
        result = optimize.minimize(
            evaluate,
            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        return result.x
    
    # Differential evolution with tailored parameters
    def differential_evolution_search():
        result = differential_evolution(
            evaluate,
            bounds,
            strategy='best1bin',
            popsize=20,
            maxiter=60,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )
        return result.x

    de_result = differential_evolution_search()
    best_result = optimize_func(de_result)
    best_c5 = evaluate(best_result)

    if elapsed_time > budget_s - 1:
        return (best_result, best_c5, n_points)

    for _ in range(30):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)

        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
        refined_result = optimize_func(new_guess)
        refined_c5 = evaluate(refined_result)
        if refined_c5 < best_c5:
            best_result = refined_result
            best_c5 = refined_c5
    
    # Constrained optimization with CVXPY
    h = cp.Variable(n_points)
    constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]

    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, eps=1e-8)
        best_result = h.value
        best_c5 = evaluate(best_result)
    except:
        pass
    
    return (best_result, best_c5, n_points)
