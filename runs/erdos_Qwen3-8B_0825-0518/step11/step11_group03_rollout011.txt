        return np.sum(h_values) - n_points / 2

    def differential_evolution_search():
        result = differential_evolution(

            popsize=20,
            maxiter=30,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed
        )

    de_result = differential_evolution_search()
    best_result = de_result

    # Local optimization with SLSQP
    result = minimize(

        best_result,

        method='SLSQP',
        options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
    )

    best_c5 = evaluate(best_result)

    for _ in range(15):

            lambda x: evaluate(x),

            method='SLSQP',

        )

            best_result = result.x
