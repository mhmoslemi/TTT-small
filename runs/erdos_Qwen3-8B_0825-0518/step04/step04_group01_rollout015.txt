def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 120  # Increased number of points for better accuracy
    dx = 2.0 / n_points
    
    # Create an initial solution based on the provided initial_h_values
    initial_h = np.zeros(n_points)
    initial_h[:len(initial_h_values)] = initial_h_values[:n_points]
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    def objective(h_values):
        return evaluate_erdos_solution(h_values)
    
    # Define bounds for each point
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define the constraint: integral of h(x) over [0, 2] must be 1
    def constraint_func(h_values):
        return np.sum(h_values) * dx - 1
    
    constraint = {'type': 'eq', 'fun': constraint_func}
    
    # Use SLSQP for constrained optimization with a more aggressive search
    result = optimize.minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints=constraint,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure the constraint is satisfied
    if np.abs(np.sum(best_h) * dx - 1) > 1e-6:
        best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    # Track elapsed time to return before the budget runs out
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s - 1:
        return (best_h, best_c5, n_points)
    
    # Perform a local search around the best solution
    # Random perturbation
    perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
    perturbed_h = best_h + perturbation
    perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
    perturbed_h = perturbed_h / np.sum(perturbed_h) * (n_points / 2)
    
    # Evaluate perturbed solution
    perturbed_c5 = evaluate_erdos_solution(perturbed_h)
    
    # Choose better solution
    if perturbed_c5 < best_c5:
        best_h = perturbed_h
        best_c5 = perturbed_c5
    
    return (best_h, best_c5, n_points)
