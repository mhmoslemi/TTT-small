def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 120  # Increased number of points for better accuracy
    dx = 2.0 / n_points

    # Start from the initial_h_values but ensure it fits the n_points
    initial_h = np.zeros(n_points)
    initial_h[:len(initial_h_values)] = initial_h_values[:n_points]
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    def objective(h_values):
        return evaluate_erdos_solution(h_values)
    
    # Define bounds for each point
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    # Define the constraint: integral of h(x) over [0, 2] must be 1
    def constraint_func(h_values):
        return np.sum(h_values) * dx - 1
    
    constraint = {'type': 'eq', 'fun': constraint_func}

    # Initial optimization with SLSQP
    result = optimize.minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints=constraint,
        method='SLSQP',
        options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)

    if np.abs(np.sum(best_h) * dx - 1) > 1e-6:
        best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    # Track elapsed time to return before the budget runs out
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s - 1:
        return (best_h, best_c5, n_points)

    if elapsed_time < budget_s - 5:
        from scipy.optimize import differential_evolution
        
        # Refine the solution using differential evolution
        def refined_objective(h_values):
            return evaluate_erdos_solution(h_values)
        
        result_de = differential_evolution(
            refined_objective,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            popsize=15,
            mutation=(0.5, 1.0),
            recombination='blend',
            tol=1e-6,
            maxiter=20,
            constraints=[constraint],
            disp=False,
            vectorized=True
        )

        if np.abs(np.sum(best_h_de) * dx - 1) > 1e-6:

    # Final check to ensure constraint is satisfied
    if np.abs(np.sum(best_h) * dx - 1) > 1e-6:
        best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    return (best_h, best_c5, n_points)
