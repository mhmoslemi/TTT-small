        overlap = signal.correlate(h, 1 - h, mode="full")
        overlap = overlap * dx

    def optimize_h():
        nonlocal current_c5, n_points, dx, initial_h
        t0 = time.time()
        best_h = initial_h.copy()
        best_c5 = current_c5
        
        # Try smaller numbers of points for faster computation
        for n in [128, 64, 32, 16, 8, 4, 2]:

                break
            n_points = n
            dx = 2.0 / n_points
            x = np.linspace(0, 2, n_points)
            
            # Define optimization variables
            h = cp.Variable(n_points, name="h")
            constraints = [
                cp.sum(h) == n_points / 2,
                0 <= h,
                h <= 1
            ]
            
            # Define symmetry and periodicity constraints
            symmetry_constraints = []
            for i in range(n_points):
                if i != n_points // 2:
                    symmetry_constraints.append(h[i] == h[n_points - i - 1])
            
            constraints.extend(symmetry_constraints)
            
            # Compute overlap integrals
            k_values = np.arange(-n_points + 1, n_points)
            k_values = k_values[np.abs(k_values) <= n_points // 2]
            k_values = np.unique(k_values)
            
            # Minimize the maximum of the overlaps
            obj = cp.Minimize(cp.maximum(*[cp.sum(h * (1 - h.roll(k))) for k in k_values]))
            prob = cp.Problem(obj, constraints)
            
            try:
                result = prob.solve(solver=cp.SCS, eps=1e-4, verbose=False)
                if result is not None:
                    current_h = h.value
                    new_c5 = compute_c5(current_h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = current_h
                        print(f"Improved C5 to {best_c5} with {n_points} points")
            except:
                pass
        
        return best_h, best_c5, n_points
    
    try:
        h_values, c5_bound, n_points = optimize_h()
    except:
