import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate
import random

def run(seed=42, budget_s=60, **kwargs):
    # Initial construction (pre-loaded)
    initial_h_values = np.array([0.4925, 0.5833, 0.6458, 0.6865, 0.7135, 0.7342, 0.7495, 0.7602, 0.7673, 0.7713,
                                 0.7722, 0.7701, 0.7652, 0.7578, 0.7482, 0.7367, 0.7235, 0.7086, 0.6923, 0.6746,
                                 0.6559, 0.6364, 0.6162, 0.5954, 0.574, 0.5522, 0.5301, 0.5076, 0.4847, 0.4616,
                                 0.4382, 0.4145, 0.3905, 0.3663, 0.3419, 0.3173, 0.2926, 0.2677, 0.2428, 0.2179,
                                 0.193, 0.168, 0.143, 0.118, 0.093, 0.068, 0.043, 0.018, 0.0])

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    time_limit = budget_s
    start_time = time.time()

    def compute_c5(h):
        h = np.array(h)
        shifted_h = np.concatenate([h[-n_points + 1:], h], axis=0)
        one_minus_h = 1 - h
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return np.max(corr) * dx

    # Try the initial solution
    best_c5 = compute_c5(initial_h_values)
    best_h = initial_h_values.copy()
    
    # Try a few other starting points
    uniform_h = np.full(n_points, 0.5)
    c5_uniform = compute_c5(uniform_h)
    if c5_uniform < best_c5:
        best_c5 = c5_uniform
        best_h = uniform_h.copy()
    
    alt_h = np.zeros(n_points)
    for i in range(n_points):
        alt_h[i] = 0.5 if i % 2 == 0 else 0.0
    c5_alt = compute_c5(alt_h)
    if c5_alt < best_c5:
        best_c5 = c5_alt
        best_h = alt_h.copy()
    
    # Function to minimize the maximum overlap integral
    def minimize_max_overlap(h_values):
        h_values = np.array(h_values)
        shifted_h = np.concatenate([h_values[-n_points + 1:], h_values], axis=0)
        one_minus_h = 1 - h_values
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return np.max(corr) * dx

    # Use a more sophisticated optimization approach with multiple restarts
    # Define a function to perform a local optimization from a starting point
    def local_optimize(start_h, max_iter=10, tolerance=1e-6):
        result = optimize.minimize(
            minimize_max_overlap,
            start_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'ftol': tolerance, 'gtol': tolerance, 'maxiter': max_iter}
        )
        if result.success:
            return result.x
        else:
            return start_h

    # Try several different starting constructions to avoid local optima
    start_points = [best_h.copy()]
    
    # Add a few different random perturbations
    for _ in range(3):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        start_points.append(new_h.copy())
    
    # Add a checkerboard pattern
    checkerboard_h = np.zeros(n_points)
    for i in range(n_points):
        checkerboard_h[i] = 0.5 if i % 2 == 0 else 0.0
    start_points.append(checkerboard_h.copy())
    
    # Add a sinusoidal variation
    sinusoidal_h = np.zeros(n_points)
    for i in range(n_points):
        sinusoidal_h[i] = 0.5 + 0.25 * np.sin(2 * np.pi * i / n_points)
    start_points.append(sinusoidal_h.copy())

    triangle_h = np.zeros(n_points)

    start_points.append(triangle_h.copy())
    
    # Perform optimization from multiple starting points
    for start_h in start_points:
        # Run local optimization
        optimized_h = local_optimize(start_h)
        c5_candidate = compute_c5(optimized_h)
        
        # Update best solution if improved
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = optimized_h.copy()
    
    # Perform a final refinement with tighter tolerances
    result = optimize.minimize(
        minimize_max_overlap,
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 50}
    )
    
    if result.success:
        h_candidate = result.x
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
    
    # Final check for any remaining perturbations
    for _ in range(3):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        c5_new = compute_c5(new_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h.copy()
    
    # Ensure we don't exceed the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
