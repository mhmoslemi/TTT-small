def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)
    
    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)
    
    def objective(h_values):
        return evaluate(h_values)
    
    def perturb(h, magnitude=0.1):
        noise = np.random.normal(0, magnitude, len(h))
        return np.clip(h + noise, 0, 1)
    
    def generate_random_h(n_points):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    def optimize_h(n_points, initial_h):
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Define the objective function to minimize the max correlation
        # To handle the max operation, we'll use a trick with auxiliary variables
        # and constraints
        max_corr = cp.Variable()
        constraints.append(max_corr >= 0)
        
        # Evaluate the correlation for all shifts
        shifts = np.arange(1, n_points)
        for shift in shifts:
            # Compute the correlation for this shift
            # For simplicity, we compute the correlation values directly in the objective
            # This is a simplification and may not scale well for large n_points
            corr = 0
            for i in range(n_points - shift):
                corr += h[i] * (1 - h[i + shift])
            constraints.append(max_corr >= corr)
        
        # Define the problem to minimize the maximum correlation
        prob = cp.Problem(cp.Minimize(max_corr), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == 'optimal':
                return h.value
            else:
                return initial_h
        except:
            return initial_h
    
    # Start with the initial solution
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        h_candidate = generate_random_h(n_points)
        c5_candidate = evaluate(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
    
    # Optimize using a more sophisticated approach
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        h_candidate = optimize_h(n_points, best_h)
        c5_candidate = evaluate(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
    
    # Local search with perturbations
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        perturbed_h = perturb(best_h, magnitude=0.05)
        c5 = evaluate(perturbed_h)
        if c5 < best_c5:
            best_h = perturbed_h
            best_c5 = c5
    
    # Simulated annealing
    temp = 1.0
    temp_min = 0.1
    cooling_rate = 0.995
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        perturbed_h = perturb(best_h, magnitude=0.1)
        c5 = evaluate(perturbed_h)
        delta = c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            best_h = perturbed_h
            best_c5 = c5
        temp *= cooling_rate
    
    return (best_h, best_c5, n_points)
