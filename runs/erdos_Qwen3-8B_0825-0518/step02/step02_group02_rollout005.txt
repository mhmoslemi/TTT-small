import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 50
    initial_h = initial_h_values()
    initial_h = np.resize(initial_h, (n_points,))  # Ensure it's of correct length
    sum_initial = np.sum(initial_h)
    scale_factor = (n_points / 2.0) / sum_initial if sum_initial != 0 else 1.0
    initial_h = np.clip(initial_h * scale_factor, 0.0, 1.0)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    # Objective function and constraints
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    # Constraints
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2.0
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization
    result = optimize.minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}

    # Check for early exit due to time limits
    if time.time() - start_time > budget_s:
        print("Time limit reached - using best found so far.")
    
    # Best solution found
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, n_points
