import scipy.signal as signal

    def compute_c5(h):

        overlap = signal.correlate(h, 1 - h, mode="full")
        overlap = overlap * dx
        return np.max(overlap)
    
    current_c5 = compute_c5(initial_h)
    best_h = initial_h.copy()
    best_c5 = current_c5
    best_n = n_points
    
    # Use a more systematic approach with smaller n_points
    for n_points in [128, 64, 32, 16, 8, 4, 2]:
        if time.time() > time.time() + budget_s * 0.8:
            break
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Define optimization variables
        h = cp.Variable(n_points, name="h")
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        
        # Compute overlap integrals
        def compute_overlap(h_val):
            h_val = np.array(h_val)
            h_extended = np.concatenate([h_val, h_val[:n_points-1]])
            overlap = signal.correlate(h_val, 1 - h_val, mode="full")
            return overlap * dx
        
        # Use a heuristic approach with random perturbations
        for _ in range(10):
            if time.time() > time.time() + budget_s * 0.8:
                break
            # Start from initial_h or best_h
            if n_points < len(best_h):
                h_initial = np.zeros(n_points)
                h_initial = np.interp(np.linspace(0, 2, n_points), x, best_h)
            else:
                h_initial = best_h[:n_points]
            
            # Add small random perturbations
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            h_initial = h_initial + perturbation
            h_initial = np.clip(h_initial, 0, 1)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
            
            # Define objective function to minimize the maximum overlap
            obj = cp.Minimize(cp.Max(cp.multiply(cp.ones(n_points), compute_overlap(h))))
            prob = cp.Problem(obj, constraints)

                result = prob.solve(solver=cp.SCS, eps=1e-4, verbose=False)

        if best_c5 < current_c5:
            current_c5 = best_c5
            initial_h = best_h
    
    return best_h, best_c5, best_n
