def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 80  # Using 80 points for better resolution
    dx = 2.0 / n_points
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        return evaluate_erdos_solution(h_values)
    
    # Function to enforce the integral constraint
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Bounds: each h_i must be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Build initial guess based on the provided initial_h_values, refined
    def interpolate_initial(initial, n_points):
        original_indices = np.linspace(0, 2, len(initial))
        new_indices = np.linspace(0, 2, n_points)
        interp = np.interp(new_indices, original_indices, initial)
        return interp
    
    initial_h = interpolate_initial(initial_h_values, n_points)
    initial_h = np.clip(initial_h, 0.0, 1.0)
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    # Optimization options
    options = {
        'maxiter': 200,
        'ftol': 1e-8,
        'disp': False
    }
    
    # Define optimization problem using L-BFGS-B solver for better performance
    def optimize_problem(h_values):
        result = optimize.minimize(
            objective,
            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='L-BFGS-B',
            options=options
        )
        return result.x
    
    # Normalize and enforce constraint
    def normalize(h_values):
        total = np.sum(h_values)
        if total == 0:
            return np.zeros_like(h_values)
        return h_values / total * (n_points / 2)
    
    # Run optimization with multiple starting points
    num_starts = 5
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for _ in range(num_starts):
        # Create random perturbations of the initial guess
        perturbations = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_guess = initial_h + perturbations
        perturbed_guess = np.clip(perturbed_guess, 0.0, 1.0)
        perturbed_guess = normalize(perturbed_guess)
        
        result = optimize.minimize(
            objective,
            perturbed_guess,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='L-BFGS-B',
            options=options
        )
        
        current_h = normalize(result.x)
        current_c5 = evaluate_erdos_solution(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    # Ensure we do not exceed the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
