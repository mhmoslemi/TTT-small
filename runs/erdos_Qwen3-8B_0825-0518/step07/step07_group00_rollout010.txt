def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 200  # Higher resolution for better approximation
    
    dx = 2.0 / n_points
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Objective function
    def objective(h_values):
        # Ensure constraint is met
        if np.abs(np.sum(h_values) - n_points / 2) > 1e-6:
            h_values = h_values / np.sum(h_values) * (n_points / 2)
        # Compute overlap integral
        correlation = np.correlate(h_values, 1 - h_values, mode="full")
        return max(correlation) * dx
    
    # Initial guesses
    initial_guesses = [
        initial_h_values.copy() / np.sum(initial_h_values) * (n_points / 2),
        np.random.rand(n_points),
        np.zeros(n_points),
        np.ones(n_points) * 0.5,
        np.array([0.5 if i % 2 == 0 else 0.0 for i in range(n_points)]),
        np.array([0.5 if i % 3 == 0 else 0.0 for i in range(n_points)]),
        np.array([0.8 if i % 5 == 0 else 0.0 for i in range(n_points)]),
        np.array([0.5 if i % 10 == 0 else 0.0 for i in range(n_points)]),
        np.array([0.5 if i % 15 == 0 else 0.0 for i in range(n_points)]),
        np.array([0.5 if i % 20 == 0 else 0.0 for i in range(n_points)]),
    ]
    
    best_h = initial_guesses[0]
    best_c5 = objective(best_h)
    
    # Global optimization with DE
    for guess in initial_guesses:
        try:
            result = differential_evolution(
                objective,
                bounds,
                strategy='best1bin',
                popsize=15,
                maxiter=100,
                mutation=(0.7, 1.0),
                recombination=0.7,
                disp=False,
                polish=True,
                tol=1e-8,
                seed=seed
            )
            current_h = result.x
            current_c5 = objective(current_h)
            if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
                current_h = current_h / np.sum(current_h) * (n_points / 2)
                current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        except:
            pass
    
    # Local optimization with SLSQP
    options = {'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    result = optimize.minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2},
        method='SLSQP',
        options=options
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Multi-start refinement with perturbations
    for _ in range(10):
        # Perturb and normalize
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        new_guess = best_h + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
        result = optimize.minimize(
            objective,
            new_guess,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2},
            method='SLSQP',
            options=options
        )
        current_h = result.x
        current_c5 = objective(current_h)
        if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
            current_h = current_h / np.sum(current_h) * (n_points / 2)
            current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    # Final check within time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
