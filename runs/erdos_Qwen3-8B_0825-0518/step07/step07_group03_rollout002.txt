def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 60  # 60 points for efficiency and fine-grained control
    dx = 2.0 / n_points
    
    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initialize with a better guess based on the initial construction
    initial_guess = initial_h_values[:n_points].copy()
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Optimization options
    options = {
        'max_iter': 150,
        'ftol': 1e-8,
        'disp': False
    }
    
    def optimize_with_initial(guess):
        result = optimize.minimize(
            evaluate,
            guess,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='SLSQP',
            options=options
        )
        # Ensure constraint is satisfied
        if np.abs(np.sum(result.x) - n_points / 2) > 1e-6:
            result.x = result.x / np.sum(result.x) * (n_points / 2)
        return result
    
    # First optimization
    result = optimize_with_initial(initial_guess)
    best_h = result.x
    best_c5 = result.fun
    
    # Ensure we do not exceed the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    # Multi-start optimization with random perturbations
    num_starts = 15
    best_c5_value = best_c5
    best_h_values = best_h
    
    for _ in range(num_starts):
        # Generate random perturbations around the initial guess
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        new_guess = initial_guess + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
        
        result = optimize_with_initial(new_guess)
        
        current_h = result.x
        current_c5 = result.fun
        
        if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
            current_h = current_h / np.sum(current_h) * (n_points / 2)
        
        if current_c5 < best_c5_value:
            best_c5_value = current_c5
            best_h_values = current_h
    
    # Additionally, explore a few other starting points
    other_start_points = [
        np.full(n_points, 0.5),
        np.random.uniform(0, 1, n_points),
        np.array([0.7 if i % 3 == 0 else 0.3 for i in range(n_points)]),
        np.array([0.6 if i % 2 == 0 else 0.4 for i in range(n_points)]),
    ]
    
    for start in other_start_points:
        start = start / np.sum(start) * (n_points / 2)
        result = optimize_with_initial(start)
        
        current_h = result.x
        current_c5 = result.fun
        
        if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
            current_h = current_h / np.sum(current_h) * (n_points / 2)
        
        if current_c5 < best_c5_value:
            best_c5_value = current_c5
            best_h_values = current_h
    
    return (best_h_values, best_c5_value, n_points)
