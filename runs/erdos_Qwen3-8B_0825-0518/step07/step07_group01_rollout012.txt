def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 200  # Increased to 200 for better resolution
    dx = 2.0 / n_points
    
    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate initial guess from initial_h_values
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    # Optimization function
    def optimize_with_bounds(h_values):
        result = minimize(
            evaluate,
            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        return result.x
    
    # Genetic Algorithm with improved selection and diversity maintenance
    def genetic_algorithm():
        population_size = 50
        generations = 50
        mutation_rate = 0.15
        elite_size = 5
        
        # Initialize population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h / np.sum(h) * (n_points / 2) for h in population]
        
        best_population = population[0]
        best_fitness = evaluate(best_population)
        
        for gen in range(generations):
            # Evaluate fitness
            fitness = [evaluate(h) for h in population]
            best_index = np.argmin(fitness)
            best_fitness = fitness[best_index]
            
            # Keep best to pass to next generation
            next_gen = [best_population] * elite_size
            
            # Select parents with tournament selection
            parents = []
            for _ in range(population_size - elite_size):
                tournament = np.random.choice(len(population), size=5, replace=False)
                winner = tournament[np.argmin([fitness[i] for i in tournament])]
                parents.append(population[winner])
            
            # Crossover
            for i in range(0, len(parents), 2):
                if i + 1 < len(parents):
                    parent1 = parents[i]
                    parent2 = parents[i + 1]
                    child1 = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                    child2 = np.where(np.random.rand(n_points) < 0.5, parent2, parent1)
                    next_gen.append(child1)
                    next_gen.append(child2)
            
            # Mutation
            for i in range(len(next_gen)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        next_gen[i][j] = np.random.uniform(0, 1)
                        next_gen[i][j] = next_gen[i][j] / np.sum(next_gen[i]) * (n_points / 2)
            
            population = next_gen
            current_best = population[0] if evaluate(population[0]) < best_fitness else best_population
            best_population = current_best
        
        return best_population
    
    # Run the genetic algorithm with early stopping
    ga_result = genetic_algorithm()
    
    # Local optimization using SLSQP
    best_result = optimize_with_bounds(ga_result)
    best_c5 = evaluate(best_result)
    
    # Early stop if time is up
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_result, best_c5, n_points)
    
    # Additional local refinements with a more systematic perturbation
    for _ in range(10):
        perturbation = np.random.normal(0, 0.05, size=n_points)
        new_guess = best_result + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
        result = optimize_with_bounds(new_guess)
        current_c5 = evaluate(result)
        if current_c5 < best_c5:
            best_result = result
            best_c5 = current_c5
    
    # Final refinement using a different optimizer
    result = minimize(
        evaluate,
        best_result,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        method='L-BFGS-B',
        options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
    )
    final_result = result.x
    final_c5 = evaluate(final_result)
    
    if final_c5 < best_c5:
        best_result = final_result
        best_c5 = final_c5
    
    return (best_result, best_c5, n_points)
