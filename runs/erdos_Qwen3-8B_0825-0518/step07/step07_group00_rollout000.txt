def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 100  # Balance between precision and computational feasibility

    dx = 2.0 / n_points
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Objective function
    def objective(h):
        # Ensure constraint is met
        if np.abs(np.sum(h) - n_points / 2) > 1e-6:
            h = h / np.sum(h) * (n_points / 2)
        return evaluate_erdos_solution(h)

    # Initial guess based on the provided initial_h_values
    h_initial = initial_h_values.copy()
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    # Use a hybrid approach with Differential Evolution and local optimization

    # Differential Evolution for exploration
    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=50,
        mutation=(0.5, 1),
        recombination=0.6,
        disp=False,
        polish=True,
        tol=1e-8,
        seed=seed
    )
    best_h = result_de.x
    best_c5 = evaluate_erdos_solution(best_h)

    # Local optimization with SLSQP for refinement
    def constraint(h):
        return np.sum(h) - n_points / 2

    options = {'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    result_slsqp = minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        method='SLSQP',
        options=options
    )
    best_h = result_slsqp.x
    best_c5 = evaluate_erdos_solution(best_h)

    # Perturbation and refinement
    for _ in range(5):
        # Perturb and normalize
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        new_guess = best_h + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
        result = minimize(
            objective,
            new_guess,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            method='SLSQP',
            options=options
        )
        current_h = result.x
        current_c5 = evaluate_erdos_solution(current_h)
        if np.abs(np.sum(current_h) - n_points / 2) > 1e-6:
            current_h = current_h / np.sum(current_h) * (n_points / 2)
            current_c5 = evaluate_erdos_solution(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

    # Final check within time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)
