def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    
    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess: use provided initial_h_values and scale
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    # Local optimizer
    def local_optimizer(h_values):
        result = minimize(
            evaluate,
            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='SLSQP',
            options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
        )
        return result.x
    
    # Global search with differential evolution
    def global_search():
        # Define objective function for differential evolution
        def objective(x):
            h = x.copy()
            h = h / np.sum(h) * (n_points / 2)
            return evaluate(h)
        
        # Bounds for differential evolution
        bounds_evolution = [(0.0, 1.0) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds_evolution,
            strategy='best1bin',
            maxiter=50,
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False
        )
        return result.x
    
    # Run global search
    best_h = global_search()
    best_c5 = evaluate(best_h)
    
    # Local refinement
    best_h = local_optimizer(best_h)
    best_c5 = evaluate(best_h)
    
    # Early stop if time is up
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    # Additional local refinements
    for _ in range(10):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        new_guess = best_h + perturbation
        new_guess = np.clip(new_guess, 0.0, 1.0)
        new_guess = new_guess / np.sum(new_guess) * (n_points / 2)
        result = local_optimizer(new_guess)
        current_c5 = evaluate(result)
        if current_c5 < best_c5:
            best_h = result
            best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
