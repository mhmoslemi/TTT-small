def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 100  # Start with 100 points for resolution
    dx = 2.0 / n_points
    
    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to generate a new candidate
    def generate_candidate():
        # Start from initial_h_values and scale if necessary
        candidate = np.copy(initial_h_values[:n_points])
        candidate = candidate / np.sum(candidate) * (n_points / 2)
        return candidate
    
    # Function to perform local optimization
    def local_optimize(h_values):
        result = minimize(
            evaluate,
            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        return result.x
    
    # Function to perform genetic algorithm
    def genetic_algorithm():
        population_size = 50
        generations = 50
        mutation_rate = 0.1
        elite_size = 5
        
        # Initialize population with random values
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h / np.sum(h) * (n_points / 2) for h in population]
        
        best_population = population[0]
        
        for gen in range(generations):
            # Evaluate fitness
            fitness = [evaluate(h) for h in population]
            best_fitness = min(fitness)
            best_index = np.argmin(fitness)
            
            # Keep best to pass to next generation
            next_gen = [best_population] * elite_size
            
            # Select parents
            parents = np.random.choice(len(population), size=population_size - elite_size, replace=False)
            parents = [population[i] for i in parents]
            
            # Crossover
            for i in range(0, len(parents), 2):
                if i + 1 < len(parents):
                    parent1 = parents[i]
                    parent2 = parents[i + 1]
                    child1 = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                    child2 = np.where(np.random.rand(n_points) < 0.5, parent2, parent1)
                    next_gen.append(child1)
                    next_gen.append(child2)
            
            # Mutation
            for i in range(len(next_gen)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        next_gen[i][j] = np.random.uniform(0, 1)
                        next_gen[i][j] = next_gen[i][j] / np.sum(next_gen[i]) * (n_points / 2)
            
            population = next_gen
            best_population = population[0] if evaluate(population[0]) < evaluate(best_population) else best_population
        
        return best_population
    
    # Function to perform adaptive local search
    def adaptive_search():
        h = generate_candidate()
        best_h = h
        best_c5 = evaluate(h)
        
        for _ in range(100):
            # Create perturbations
            perturbations = np.random.uniform(-0.1, 0.1, size=n_points)

            h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2)
            
            # Local optimization
            h_optimized = local_optimize(h_perturbed)
            c5 = evaluate(h_optimized)
            
            if c5 < best_c5:
                best_h = h_optimized
                best_c5 = c5
        
        return best_h, best_c5
    
    # Initialize with a random candidate and adaptively improve
    h = generate_candidate()
    best_c5 = evaluate(h)

    # Run genetic algorithm
    ga_result = genetic_algorithm()
    current_c5 = evaluate(ga_result)
    
    if current_c5 < best_c5:
        best_h = ga_result
        best_c5 = current_c5
    
    # Run adaptive search
    adaptive_h, adaptive_c5 = adaptive_search()
    
    if adaptive_c5 < best_c5:
        best_h = adaptive_h
        best_c5 = adaptive_c5
    
    # Additional local refinements
    for _ in range(10):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        new_guess = best_h + perturbation

        current_c5 = evaluate(result)
        if current_c5 < best_c5:
            best_h = result
            best_c5 = current_c5
    
    # Early stop if time is up
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
