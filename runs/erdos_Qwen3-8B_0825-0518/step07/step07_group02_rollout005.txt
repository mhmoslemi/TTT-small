    start_time = time.time()

    dx = 2.0 / n_points
    
    # Define the objective function to minimize
    def objective(h_values):
        return evaluate_erdos_solution(h_values)
    
    # Constraint for the total integral
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Bounds for h_values
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initialize with the initial_h_values scaled to n_points
    initial_guess = np.zeros(n_points)
    mapping = np.linspace(0, 2, len(initial_h_values))
    idx = np.argsort(mapping)
    for i in range(n_points):
        # Find the closest initial value
        for j in range(len(mapping)):
            if idx[j] == i:
                initial_guess[i] = initial_h_values[j]
                break
    
    # Normalize to meet the integral constraint
    initial_guess = initial_guess / np.sum(initial_guess) * (n_points / 2)
    
    # Optimization options
    options = {
        'maxiter': 200,
        'ftol': 1e-8,
        'disp': False,
        'iprint': 0
    }
    
    # Define optimization problem
    def optimize_problem(h_values):
        result = optimize.minimize(
            objective,
            h_values,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='L-BFGS-B',
            options=options
        )
        return result.x
    
    # Normalize function to ensure the integral constraint is satisfied
    def normalize(h_values):
        total = np.sum(h_values)
        if total == 0:
            return np.zeros_like(h_values)
        return h_values / total * (n_points / 2)
    
    # Run optimization with multiple starting points
    num_starts = 5
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for _ in range(num_starts):
        # Create a random perturbation of the initial guess
        perturbations = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_guess = initial_guess + perturbations
        perturbed_guess = np.clip(perturbed_guess, 0.0, 1.0)
        perturbed_guess = normalize(perturbed_guess)
        
        result = optimize.minimize(
            objective,
            perturbed_guess,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            method='L-BFGS-B',
            options=options
        )
        
        current_h = normalize(result.x)
        current_c5 = evaluate_erdos_solution(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    # Ensure we do not exceed the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
