        G[row, right] -= dx * h[left]

        h_pol = _run_slsqp(best_h, n, dx, m, bounds, constraints, maxiter)
        if h_pol is not None:
            sc = exact_score(h_pol)
            if sc < best_s - 1e-12:
                best_s = sc
                best_h = h_pol.copy()
        if best_s < global_best_score - 1e-12:
            global_best_score = best_s
            global_best_h = best_h.copy()
        n_fine = 2 * n
        if n_fine <= 128:
            dx_f = 2.0 / n_fine
            m_f = n_fine / 2.0
            h_fine_base = capped_simplex_project(np.repeat(best_h, 2), m_f)
            bounds_f = [(0.0, 1.0)] * n_fine + [(0.0, 1.01)]
            constraints_f = [
                {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n_fine, dx_f)},
                {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac, 'args': (n_fine, dx_f)}
            ]
            maxiter_f = min(600, 300 + 2 * n_fine)
            starts_f = make_starts(h_fine_base, n_fine, rng)
            h_pf = capped_simplex_project(np.repeat(parent, 2), m_f)
            if not np.allclose(h_pf, h_fine_base, atol=1e-8):
                starts_f.append(h_pf)
            best_fh = h_fine_base.copy()
            best_fs = exact_score(h_fine_base)
            for start in starts_f:
                s0 = exact_score(start)
                if s0 < best_fs:
                    best_fs = s0
                    best_fh = start.copy()
                h_res = _run_slsqp(start, n_fine, dx_f, m_f, bounds_f, constraints_f, maxiter_f)
                if h_res is not None:
                    sc = exact_score(h_res)
                    if sc < best_fs - 1e-12:
                        best_fs = sc
                        best_fh = h_res.copy()
            h_pol_f = _run_slsqp(best_fh, n_fine, dx_f, m_f, bounds_f, constraints_f, maxiter_f)
            if h_pol_f is not None:
                sc = exact_score(h_pol_f)
                if sc < best_fs - 1e-12:
                    best_fs = sc
                    best_fh = h_pol_f.copy()
            if best_fs < global_best_score - 1e-12:
                global_best_score = best_fs
                global_best_h = best_fh.copy()
    final_h = capped_simplex_project(global_best_h, len(global_best_h) / 2.0)
    if not _is_feasible(final_h):
        final_h = parent
    final_score = float(exact_score(final_h))
    final_n = int(len(final_h))
    return (final_h, final_score, final_n)
