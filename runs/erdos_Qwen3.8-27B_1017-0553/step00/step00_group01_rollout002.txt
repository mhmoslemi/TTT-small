def project_capped_simplex(y, T, n):
    y = np.asarray(y, dtype=np.float64)
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo, hi = np.min(y) - 1.0, np.max(y)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > T:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    r = T - np.sum(x)
    if abs(r) > 1e-12:
        for i in range(n):
            if r > 0:
                a = min(r, 1.0 - x[i]); x[i] += a; r -= a
            else:
                a = min(-r, x[i]); x[i] -= a; r += a
            if abs(r) < 1e-14:
                break
    x = np.clip(x, 0.0, 1.0)
    if abs(np.sum(x) - T) > 1e-8:
        return np.full(n, 0.5)
    return x

def exact_score(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr_grad(h, dx, n):
    m = 2 * n - 1
    c = np.zeros(m)
    G = np.zeros((m, n))
    for k in range(m):
        d = k - (n - 1)
        il, ih = max(0, -d), min(n - 1, n - 1 - d)
        idx = np.arange(il, ih + 1)
        c[k] = dx * np.sum(h[idx + d] * (1.0 - h[idx]))
        G[k, idx + d] += dx * (1.0 - h[idx])
        G[k, idx] -= dx * h[idx + d]
    return c, G

def lsexp_obj(h, tau, dx, n):
    c, _ = compute_corr_grad(h, dx, n)
    a = np.max(c)
    return a + tau * np.log(np.sum(np.exp((c - a) / tau)))

def lsexp_grad(h, tau, dx, n):
    c, G = compute_corr_grad(h, dx, n)
    a = np.max(c)
    w = np.exp((c - a) / tau)
    w /= np.sum(w)
    return G.T @ w

def lsexp_eq(h, tau, dx, n):
    return np.sum(h) - n / 2.0

def lsexp_eq_jac(h, tau, dx, n):
    return np.ones(n)

def epi_obj(z, n, dx, T):
    return z[n]

def epi_obj_grad(z, n, dx, T):
    g = np.zeros(n + 1); g[n] = 1.0
    return g

def epi_ineq(z, n, dx, T):

def epi_ineq_jac(z, n, dx, T):

    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G; J[:, n] = 1.0
    return J

def epi_eq_jac(z, n, dx, T):
