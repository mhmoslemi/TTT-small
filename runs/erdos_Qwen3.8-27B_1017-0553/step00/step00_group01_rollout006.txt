        rng.shuffle(pairs)
        for p, q in pairs:
            lo = max(-cur[p], cur[q] - 1.0)
            hi = min(1.0 - cur[p], cur[q])
            if hi - lo < 1e-14:
                continue
            v = np.zeros(n)
            v[p] = 1.0
            v[q] = -1.0
            b = dx * (np.correlate(v, 1.0 - cur, mode='full') - np.correlate(cur, v, mode='full'))
            qc = -dx * np.correlate(v, v, mode='full')
            alpha, pred = line_search(a, b, qc, lo, hi)
            if pred >= cur_score - 1e-12:
                continue
            trial = cur.copy()
            trial[p] += alpha
            trial[q] -= alpha
            trial = project(trial, T)
            if not is_feasible(trial, T):
                continue
            s = exact_score(trial, dx)
            if s < cur_score - 1e-12:
                cur = trial
                cur_score = s
                a = lag_vec(cur, dx)
                improved = True
                if s < best_score:
                    best = cur.copy()
                    best_score = s
        if not improved:
            break
    return best, best_score

def random_escape(h, T, dx, rng, n_dirs):
    n = len(h)
    cur = h.copy()
    cur_score = exact_score(cur, dx)
    best = cur.copy()
    best_score = cur_score
    accepted = False
    for _ in range(n_dirs):
        v = rng.standard_normals(n)
        v = v - v.mean()
        if n > 1:
            v[-1] = -float(np.sum(v[:-1]))
        mx = float(np.max(np.abs(v)))
        if mx < 1e-14 or not np.all(np.isfinite(v)):
            continue
        v = v / mx
        lo = -np.inf
        hi = np.inf
        for i in range(n):
            if v[i] > 0:
                lo = max(lo, -cur[i] / v[i])
                hi = min(hi, (1.0 - cur[i]) / v[i])
            elif v[i] < 0:
                lo = max(lo, (1.0 - cur[i]) / v[i])
                hi = min(hi, -cur[i] / v[i])
        if hi - lo < 1e-14:
            continue
        a = lag_vec(cur, dx)
        b = dx * (np.correlate(v, 1.0 - cur, mode='full') - np.correlate(cur, v, mode='full'))
        qc = -dx * np.correlate(v, v, mode='full')
        alpha, pred = line_search(a, b, qc, lo, hi)
        if pred >= cur_score - 1e-12:
            continue
        trial = project(cur + alpha * v, T)
        if not is_feasible(trial, T):
            continue
        s = exact_score(trial, dx)
        if s < cur_score - 1e-12:
            cur = trial
            cur_score = s
            accepted = True
            if s < best_score:
                best = cur.copy()
                best_score = s
    return best, best_score, accepted

def gen_starts(h0, n, T, dx, rng):
    starts = [h0.copy(), h0[::-1].copy(), 1.0 - h0]
    if n > 1:
        starts.append(np.roll(h0, max(1, n // 3)).copy())
    starts.append(project(h0 + 0.04 * rng.standard_normals(n), T))
    starts.append(project(h0 + 0.12 * rng.standard_normals(n), T))
    starts.append(project(rng.random(n), T))
    idx = rng.permutation(n)
    nb = np.zeros(n)
    nf = int(np.floor(T))
    nb[idx[:nf]] = 1.0
    rem = T - nf
    if rem > 1e-10 and nf < n:
        nb[idx[nf]] = rem
    starts.append(project(nb, T))
    if n >= 2:
        f = int(rng.integers(1, min(5, n // 2 + 1)))
        ph = float(rng.random()) * 2.0 * np.pi
        x = np.arange(n, dtype=np.float64)
        starts.append(project(0.5 + 0.45 * np.cos(2.0 * np.pi * f * (x + 0.5) / n + ph), T))
    unique = []
    for s in starts:
        if not is_feasible(s, T):
            continue
        dup = False
        for u in unique:
            if float(np.max(np.abs(s - u))) < 1e-12:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try:
        s = int(seed) % (2 ** 32)
    except (ValueError, TypeError):
        s = 42
    rng = np.random.default_rng(s)
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).flatten()
        if len(h0) == 0 or not np.all(np.isfinite(h0)):
            h0 = None
    except NameError:
        pass
    except Exception:
        pass
    if h0 is not None:
        n = len(h0)
    else:
        n = kwargs.get('n_points', 47)
        if not isinstance(n, int) or n < 1 or n > 64:
            n = 47
    T = n / 2.0
    dx = 2.0 / n
    if h0 is not None and is_feasible(h0, T):
        fallback = h0.copy()
    elif h0 is not None:
        fallback = project(h0, T)
    else:
        fallback = np.full(n, 0.5)
    if n == 1:
        return (fallback, exact_score(fallback, dx), n)
    best_h = fallback.copy()
    best_score = exact_score(fallback, dx)
    skip_pairs = n > 64
    starts = gen_starts(fallback, n, T, dx, rng)
    for st in starts:
        s0 = exact_score(st, dx)
        if s0 < best_score:
            best_h = st.copy()
            best_score = s0
        if skip_pairs:
            continue
        bh, bs = pair_sweeps(st, T, dx, rng, 4)
        if bs < best_score:
            best_h = bh.copy()
            best_score = bs
        bh2, bs2, acc = random_escape(best_h, T, dx, rng, 4)
        if bs2 < best_score:
            best_h = bh2.copy()
            best_score = bs2
        if acc:
            bh3, bs3 = pair_sweeps(best_h, T, dx, rng, 2)
            if bs3 < best_score:
                best_h = bh3.copy()
                best_score = bs3
    for sigma in [0.015, 0.05, 0.15]:
        pert = project(best_h + sigma * rng.standard_normals(n), T)
        if not is_feasible(pert, T):
            continue
        ps = exact_score(pert, dx)
        if ps < best_score:
            best_h = pert.copy()
            best_score = ps
        if not skip_pairs:
            bh, bs = pair_sweeps(pert, T, dx, rng, 2)
            if bs < best_score:
                best_h = bh.copy()
                best_score = bs
            bh2, bs2, acc = random_escape(best_h, T, dx, rng, 3)
            if bs2 < best_score:
                best_h = bh2.copy()
                best_score = bs2
            if acc:
                bh3, bs3 = pair_sweeps(best_h, T, dx, rng, 1)
                if bs3 < best_score:
                    best_h = bh3.copy()
                    best_score = bs3
    best_h = project(best_h, T)
    fb = project(fallback, T)
    s_best = exact_score(best_h, dx)
    s_fb = exact_score(fb, dx)
    if s_fb < s_best:
        best_h, s_best = fb, s_fb
    if not is_feasible(best_h, T):
        best_h = np.full(n, 0.5)
        s_best = exact_score(best_h, dx)
    return (best_h, float(s_best), n)
