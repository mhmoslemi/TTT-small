import numpy as np

def capped_project(y, T):
    n = len(y)
    if not np.all(np.isfinite(y)) or np.max(np.abs(y)) > 1e12:
        return np.full(n, T / n)
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(100):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(y - mid, 0, 1)) > T:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0, 1)
    r = T - np.sum(x)
    if r > 1e-14:
        for i in range(n):
            if r <= 0:
                break
            add = min(r, 1.0 - x[i])
            x[i] += add
            r -= add
    elif r < -1e-14:
        for i in range(n):
            if r >= 0:
                break
            sub = min(-r, x[i])
            x[i] -= sub
            r += sub
    if not np.all(np.isfinite(x)):
        return np.full(n, T / n)
    return x

def exact_score(h, dx):
    if h.ndim != 1 or not np.all(np.isfinite(h)):
        return float('inf')
    n = len(h)
    if np.any(h < -1e-10) or np.any(h > 1 + 1e-10):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def grad_col(h, k, n, dx):
    d = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = k - d
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += dx * (1.0 - h[idx1[m1]])
    idx2 = k + d
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= dx * h[idx2[m2]]
    return g

def line_search(a, b, q, lo, hi):
    if hi - lo < 1e-14:
        return 0.0
    best_a = 0.0
    best_v = float(np.max(a))
    alphas = np.linspace(lo, hi, 17)
    if 0.0 >= lo and 0.0 <= hi:
        alphas = np.concatenate([alphas, [0.0]])
    alphas = np.unique(alphas)
    for alpha in alphas:
        v = float(np.max(a + b * alpha + q * alpha * alpha))
        if v < best_v:
            best_v = v
            best_a = alpha
    radius = (hi - lo) / 17.0
    center = best_a
    for _ in range(5):
        for f in [-1.0, -0.5, 0.0, 0.5, 1.0]:
            alpha = center + f * radius
            if alpha < lo or alpha > hi:
                continue
            v = float(np.max(a + b * alpha + q * alpha * alpha))
            if v < best_v:
                best_v = v
                best_a = alpha
        center = best_a
        radius *= 0.5
    return best_a

def pair_descent(h, n, dx, T, rng, max_sweeps):
    cur = h.copy()
    cur_s = exact_score(cur, dx)
    best = cur.copy()
    best_s = cur_s
    for _ in range(max_sweeps):
        improved = False
        pairs = [(p, q) for p in range(n) for q in range(p + 1, n)]
        rng.shuffle(pairs)
        for p, q in pairs:
            a = np.correlate(cur, 1.0 - cur, mode='full') * dx
            b = grad_col(cur, p, n, dx) - grad_col(cur, q, n, dx)
            qd = np.zeros(2 * n - 1)
            off = q - p
            qd[n - 1] = -2.0 * dx
            qd[n - 1 + off] = dx
            qd[n - 1 - off] = dx
            lo = max(-cur[p], cur[q] - 1.0)
            hi = min(1.0 - cur[p], cur[q])
            if hi - lo < 1e-14:
                continue
            alpha = line_search(a, b, qd, lo, hi)
            if abs(alpha) < 1e-14:
                continue
            trial = cur.copy()
            trial[p] += alpha
            trial[q] -= alpha
            trial = capped_project(trial, T)
            s = exact_score(trial, dx)
            if s < cur_s - 1e-12:
                cur = trial
                cur_s = s
                improved = True
            if s < best_s - 1e-12:
                best_s = s
                best = trial.copy()
        if not improved:
            break
    return best, best_s

def random_escape(h, n, dx, T, rng, n_dirs):
    cur = h.copy()
    best = h.copy()
    best_s = exact_score(h, dx)
    for _ in range(n_dirs):
        v = rng.standard_normal(n)
        v -= v.mean()
        if n > 1:
            v[-1] = -v[:-1].sum()
        vm = np.max(np.abs(v))
        if vm < 1e-14 or not np.all(np.isfinite(v)):
            continue
        v /= vm
        a = np.correlate(cur, 1.0 - cur, mode='full') * dx
        b = dx * (np.correlate(v, 1.0 - cur, mode='full') - np.correlate(cur, v, mode='full'))
        q = -dx * np.correlate(v, v, mode='full')
        lo, hi = -1e18, 1e18
        for i in range(n):
            if v[i] > 1e-14:
                lo = max(lo, -cur[i] / v[i])
                hi = min(hi, (1.0 - cur[i]) / v[i])
            elif v[i] < -1e-14:
                lo = max(lo, (1.0 - cur[i]) / v[i])
                hi = min(hi, -cur[i] / v[i])
        lo = max(lo, -1.0)
        hi = min(hi, 1.0)
        if hi - lo < 1e-14:
            continue
        alpha = line_search(a, b, q, lo, hi)
        if abs(alpha) < 1e-14:
            continue
        trial = capped_project(cur + alpha * v, T)
        s = exact_score(trial, dx)
        if s < best_s - 1e-12:
            best_s = s
            best = trial.copy()
            cur = trial
    return best, best_s
