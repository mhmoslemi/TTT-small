def project_waterfill(y, n):
    # y is a 1D array of length n
    # Project onto {x: 0 <= x_i <= 1, sum(x) = n/2}
    target = n / 2.0
    y = np.clip(y, -100, 100)  # limit extremes
    y = np.where(np.isfinite(y), y, 0.5)
    
    lo = np.min(y) - 1.0
    hi = np.max(y)
    
    for _ in range(100):
        tau = (lo + hi) / 2.0
        x = np.clip(y - tau, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = tau
        else:
            hi = tau
    
    x = np.clip(y - (lo+hi)/2.0, 0.0, 1.0)
    r = target - np.sum(x)
    
    # Distribute residual
    if r > 0:
        caps = 1.0 - x
        # Add to entries with most capacity
        for i in np.argsort(-caps):
            add = min(caps[i], r)
            x[i] += add
            r -= add
            if r <= 1e-15:
                break
    elif r < 0:
        r = -r
        for i in np.argsort(x):
            sub = min(x[i], r)
            x[i] -= sub
            r -= sub
            if r <= 1e-15:
                break
    
    if not is_feasible(x):
        return np.full(n, 0.5)
    return x

import numpy as np
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def is_feasible(h):
    if h.ndim != 1 or len(h) == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return False
    if abs(np.sum(h) - len(h) / 2.0) > 1e-7:
        return False
    return True

def project_waterfill(y, n):

def compute_q_and_jac(h):
    n = len(h)
    n_lags = 2 * n - 1
    q = np.zeros(n_lags)
    G = np.zeros((n_lags, n))
    for d_act in range(-(n - 1), n):
        d_idx = d_act + (n - 1)
        i_lo = max(0, -d_act)
        i_hi = min(n - 1, n - 1 - d_act)
        if i_lo > i_hi:
            continue
        i_range = np.arange(i_lo, i_hi + 1)
        j_range = i_range + d_act
        q[d_idx] = np.sum(h[j_range] * (1.0 - h[i_range]))
        if d_act == 0:
            G[d_idx, i_range] = 1.0 - 2.0 * h[i_range]
        else:
            G[d_idx, j_range] += (1.0 - h[i_range])
            G[d_idx, i_range] += (-h[j_range])
    return q, G

def epi_obj(z, n):
    return z[n]

def epi_obj_jac(z, n):
    out = np.zeros(n + 1)
    out[n] = 1.0
    return out

def epi_ineq(z, n):
    h = z[:n]
    q, _ = compute_q_and_jac(h)
    return z[n] - q

def epi_ineq_jac(z, n):

    _, G = compute_q_and_jac(h)
    out = np.zeros((2 * n - 1, n + 1))
    out[:, :n] = -G
    out[:, n] = 1.0

def epi_eq(z, n):

def run_slsqp(h_start, n):
    q0, _ = compute_q_and_jac(h_start)
    t0 = np.max(q0) + 1e-7 * max(1, n)
    z0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac, 'args': (n,)},
        {'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_jac, 'args': (n,)}
    ]
    res = minimize(epi_obj, z0, method='SLSQP', jac=epi_obj_jac,
                   bounds=bounds, constraints=cons, args=(n,),
                   maxiter=300, ftol=1e-9)
    return res

import numpy as np
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def is_feasible(h):
    if h.ndim != 1 or len(h) == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return False
    if abs(np.sum(h) - len(h) / 2.0) > 1e-7:
        return False
    return True

def project_waterfill(y, n):
    y = np.asarray(y, dtype=np.float64).copy()
    y = np.where(np.isfinite(y), y, 0.5)
    y = np.clip(y, -50.0, 50.0)
    target = n / 2.0
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - tau, 0.0, 1.0))
        if s > target:
            lo = tau
        else:
            hi = tau
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - np.sum(x)
    if r > 1e-14:
        idx = np.argsort(-(1.0 - x))
        for i in idx:
            a = min(1.0 - x[i], r)
            x[i] += a
            r -= a
            if r <= 1e-14:
                break
    elif r < -1e-14:
        r = -r
        idx = np.argsort(x)
        for i in idx:
            a = min(x[i], r)
            x[i] -= a
            r -= a
            if r <= 1e-14:
                break
    if not is_feasible(x):
        return np.full(n, 0.5)
    return x

def transfer_area(h_src, n_dst):
    a = len(h_src)
    y = np.zeros(n_dst)
    for k in range(n_dst):
        dst_lo = 2.0 * k / n_dst
        dst_hi = 2.0 * (k + 1) / n_dst
        j_start = max(0, int(np.floor(dst_lo * a / 2.0)))
        j_end = min(a - 1, int(np.ceil(dst_hi * a / 2.0)) - 1)
        if j_end < j_start:
            j_end = j_start
        total = 0.0
        for j in range(j_start, j_end + 1):
            src_lo = 2.0 * j / a
            src_hi = 2.0 * (j + 1) / a
            ov = min(dst_hi, src_hi) - max(dst_lo, src_lo)
            if ov > 0.0:
                total += h_src[j] * ov
        y[k] = (n_dst / 2.0) * total
    return y

def compute_q_and_jac(h):
    n = len(h)
    n_lags = 2 * n - 1
    q = np.zeros(n_lags)
    G = np.zeros((n_lags, n))
    for d_idx in range(n_lags):
        d_act = d_idx - (n - 1)
        i_lo = max(0, -d_act)
        i_hi = min(n - 1, n - 1 - d_act)
        if i_lo > i_hi:
            continue
        ir = np.arange(i_lo, i_hi + 1)
        jr = ir + d_act
        q[d_idx] = np.sum(h[jr] * (1.0 - h[ir]))
        if d_act == 0:
            G[d_idx, ir] = 1.0 - 2.0 * h[ir]
        else:
            G[d_idx, jr] += (1.0 - h[ir])
            G[d_idx, ir] += (-h[jr])
    return q, G

def epi_obj(z, n):
    return z[n]

def epi_obj_jac(z, n):
    out = np.zeros(n + 1)
    out[n] = 1.0
    return out

def epi_ineq(z, n):
    q, _ = compute_q_and_jac(z[:n])
    return z[n] - q

def epi_ineq_jac(z, n):
    _, G = compute_q_and_jac(z[:n])
    out = np.zeros((2 * n - 1, n + 1))
    out[:, :n] = -G
    out[:, n] = 1.0
    return out

def epi_eq(z, n):
    return np.sum(z[:n]) - n / 2.0

def epi_eq_jac(z, n):
    out = np.zeros(n + 1)
    out[:n] = 1.0

def run_slsqp(h_start, n):

        {'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_jac, 'args': (n,)}

    res = minimize(epi_obj, z0, method='SLSQP', jac=epi_obj_jac,
                   bounds=bounds, constraints=cons, args=(n,),
                   maxiter=300, ftol=1e-9)
