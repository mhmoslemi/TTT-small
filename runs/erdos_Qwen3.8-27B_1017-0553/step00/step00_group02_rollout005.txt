    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - np.sum(x)
    if abs(r) > 1e-14:
        if r > 0:
            cap = 1.0 - x
            tc = np.sum(cap)
            if tc > 1e-14:
                x += r * cap / tc
            else:
                x += r / n
        else:
            tc = np.sum(x)
            if tc > 1e-14:
                x += r * x / tc
            else:
                x += r / n
    x = np.clip(x, 0.0, 1.0)
    if not is_feasible(x, tol=1e-8):
        return np.full(n, 0.5)
    return x

def transfer_area(h_src, n_tgt):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    y = np.zeros(n_tgt)
    for k in range(n_tgt):
        k_lo = 2.0 * k / n_tgt
        k_hi = 2.0 * (k + 1) / n_tgt
        j_lo = max(0, int(np.ceil(k_lo / dx_src)) - 1)
        j_hi = min(n_src - 1, int(np.floor(k_hi / dx_src)) + 1)
        for j in range(j_lo, j_hi + 1):
            ov = min(k_hi, 2.0 * (j + 1) / n_src) - max(k_lo, 2.0 * j / n_src)
            if ov > 0:
                y[k] += h_src[j] * ov * n_tgt / 2.0
    return y

def compute_lag_vector(h):
    n = len(h)
    q = np.zeros(2 * n - 1)
    omh = 1.0 - h
    for d in range(-(n - 1), n):
        idx = d + n - 1
        if d == 0:
            i = np.arange(n)
        elif d > 0:
            i = np.arange(0, n - d)
        else:
            i = np.arange(-d, n)
        j = i + d
        q[idx] = np.dot(h[j], omh[i])
    return q

def compute_lag_jacobian(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        idx = d + n - 1
        if d == 0:
            G[idx, :] = 1.0 - 2.0 * h
        else:
            if d > 0:
                i = np.arange(0, n - d)
            else:
                i = np.arange(-d, n)
            j = i + d
            G[idx, j] += (1.0 - h[i])
            G[idx, i] -= h[j]
    return G

def obj_fn(z, n):
    return z[n]

def obj_jac(z, n):
    j = np.zeros(n + 1)
    j[n] = 1.0
    return j

def ineq_fn(z, n):
    h = z[:n]
    return z[n] - compute_lag_vector(h)

def ineq_jac(z, n):
    h = z[:n]
    G = compute_lag_jacobian(h)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def eq_fn(z, n):
    return np.sum(z[:n]) - n / 2.0

def eq_jac(z, n):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def run_slsqp(h_start, n, maxiter=300, ftol=1e-9):
    q0 = compute_lag_vector(h_start)
    t0 = np.max(q0) + 1e-7 * max(1.0, float(n))
    z0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac, 'args': (n,)},
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac, 'args': (n,)},
    ]
    try:
        res = minimize(obj_fn, z0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons, args=(n,),
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
    except Exception:
        return None
    if res is None or res.x is None or len(res.x) != n + 1:
        return None
    h_r = res.x[:n]
    if not np.all(np.isfinite(h_r)):
        return None
    return project_box_mass(h_r, n)

def run(seed=42, **kwargs):
    has_parent = False
    h_parent = None
    n0 = 45
    try:
        raw = initial_h_values  # noqa: F821
        if raw is not None:
            raw = np.array(raw, dtype=np.float64).ravel()
            if raw.size > 0 and np.all(np.isfinite(raw)):
                n0 = raw.size
                h_parent = project_box_mass(raw, n0)
                has_parent = True
    except Exception:
        pass
    if not has_parent:
        n0 = 45
        h_parent = np.full(n0, 0.5)

    best_h = h_parent.copy()
    best_score = exact_score(h_parent)
    best_n = n0

    cap = 180
    if n0 <= cap:
        resolutions = sorted(set([n0, min(2 * n0, cap), min(4 * n0, cap)]))
    else:
        resolutions = [min(n0, cap)]

    level_best = h_parent.copy()
    rng = np.random.default_rng(seed)

    for lvl, n in enumerate(resolutions):
        if lvl == 0:
            h_base = h_parent.copy()
        else:
            h_base = transfer_area(level_best, n)
            h_base = project_box_mass(h_base, n)

        base_sc = exact_score(h_base)
        if base_sc < best_score - 1e-12:
            best_score = base_sc
            best_h = h_base.copy()
            best_n = n
        level_best = h_base.copy()
        level_sc = base_sc

        starts = [h_base.copy(), h_base[::-1].copy(), 1.0 - h_base.copy(),
                  h_base + 0.04 * rng.standard_normal(n),
                  h_base + 0.15 * rng.standard_normal(n)]
        starts = [project_box_mass(s, n) for s in starts]

        for h_start in starts:
            ss = exact_score(h_start)
            if ss < level_sc - 1e-12:
                level_sc = ss
                level_best = h_start.copy()
            if ss < best_score - 1e-12:
                best_score = ss
                best_h = h_start.copy()
                best_n = n
            h_opt = run_slsqp(h_start, n)
            if h_opt is not None:
                os_ = exact_score(h_opt)
                if os_ < level_sc - 1e-12:
                    level_sc = os_
                    level_best = h_opt.copy()
                if os_ < best_score - 1e-12:
                    best_score = os_
                    best_h = h_opt.copy()
                    best_n = n

    best_h = project_box_mass(best_h, best_n)
    if not is_feasible(best_h):
        best_h = h_parent.copy()
        best_n = len(h_parent)
    return (best_h.astype(np.float64), float(exact_score(best_h)), int(best_n))
