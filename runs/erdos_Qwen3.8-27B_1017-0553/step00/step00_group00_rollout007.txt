import numpy as np

def project_capped_simplex(v, m):
    n = len(v)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > m:
            low = mid
        else:
            high = mid
    h = np.clip(v - 0.5 * (low + high), 0.0, 1.0)
    r = m - float(np.sum(h))
    if abs(r) > 5e-13:
        if r > 0:
            cap = 1.0 - h
            idx = np.argsort(-cap)
            added = 0.0
            for i in idx:
                add = min(float(cap[i]), r - added)
                if add > 0:
                    h[i] += add
                    added += add
                if added >= r:
                    break
        else:
            r = -r
            idx = np.argsort(-h)
            removed = 0.0
            for i in idx:
                rem = min(float(h[i]), r - removed)
                if rem > 0:
                    h[i] -= rem
                    removed += rem
                if removed >= r:
                    break

def compute_lags_and_grads(h):
    n = len(h)
    dx = 2.0 / n
    Q = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        left = slice(lo, hi)
        right = slice(lo + d, hi + d)
        Q[row] = dx * float(np.dot(h[left], 1.0 - h[right]))
        G[row, left] += dx * (1.0 - h[right])
        G[row, right] -= dx * h[left]
    return Q, G
