            else:
                base = np.full(n, 0.5, dtype=np.float64)
        else:
            base = project_box_mass(area_transfer(best_vec, n)) if len(best_vec) != n else best_vec.copy()
        base = project_box_mass(base)

        starts = []
        if li == 0 and parent_vec is None:
            i = np.arange(n, dtype=np.float64)
            starts.append(np.full(n, 0.5, dtype=np.float64))
            starts.append(project_box_mass(0.5 + 0.4 * np.cos(2.0 * np.pi * i / n)))
            starts.append(project_box_mass(0.5 + 0.4 * np.cos(4.0 * np.pi * i / n + 0.7)))
            starts.append(make_binary_start(n, rng))
            starts.append(project_box_mass(0.5 + 0.25 * rng.standard_normal(n)))
        else:
            if is_feasible(base):
                starts.append(base.copy())
            starts.append(project_box_mass(base + 0.05 * rng.standard_normal(n)))
            starts.append(project_box_mass(base + 0.20 * rng.standard_normal(n)))
            starts.append(make_binary_start(n, rng))
            theta = rng.uniform(0, 2.0 * np.pi)
            i = np.arange(n, dtype=np.float64)
            starts.append(project_box_mass(0.5 + 0.4 * np.cos(2.0 * np.pi * i / n + theta) + 0.10 * rng.standard_normal(n)))

        for start in starts:
            start = project_box_mass(start)
            if not is_feasible(start):
                continue
            sc = exact_c5(start)
            if sc < best_score - 1e-12:
                best_score = sc
                best_vec = start.copy()
            try:
                res_vec, res_score = optimize_start(start, temperatures, iter_limit)
                if res_vec is not None and is_feasible(res_vec) and np.isfinite(res_score):
                    if res_score < best_score - 1e-12:
                        best_score = res_score
                        best_vec = res_vec.copy()
            except Exception:
                pass

    final = project_box_mass(best_vec)
    if is_feasible(final):
        fs = exact_c5(final)
        if fs < fallback_score - 1e-12:
            return final, fs, int(len(final))
        else:
            return fallback_vec, fallback_score, int(len(fallback_vec))
    return fallback_vec, fallback_score, int(len(fallback_vec))
