import numpy as np
from scipy.optimize import minimize

def project_capped_simplex(v, m):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return np.full(0, 0.5)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    r = m - np.sum(h)
    if abs(r) > 1e-15:
        if r > 0:
            slack = 1.0 - h
            idx = np.argsort(-slack)
            for i in idx:
                if r <= 0:
                    break
                add = min(r, 1.0 - h[i])
                h[i] += add
                r -= add
        else:
            idx = np.argsort(h)
            for i in idx:
                if r >= 0:
                    break
                sub = min(-r, h[i])
                h[i] -= sub
                r += sub
    if abs(np.sum(h) - m) > 1e-10:
        return np.full(n, 0.5)
    return h

def exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or len(h) == 0 or not np.all(np.isfinite(h)):
        return np.inf
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_lags(h, n, dx):
    Q = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for idx, d in enumerate(range(-(n - 1), n)):
        lo = max(0, -d)
        hi = min(n, n - d)
        hL = h[lo:hi]
        hR = h[lo + d:hi + d]
        Q[idx] = dx * np.dot(hL, 1.0 - hR)
        G[idx, lo:hi] += dx * (1.0 - hR)
        G[idx, lo + d:hi + d] -= dx * hL
    return Q, G

def _obj(z, n, dx):
    return z[n]

def _obj_jac(z, n, dx):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _eq_fun(z, n, dx):
    return np.sum(z[:n]) - n / 2.0

def _eq_jac(z, n, dx):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def _ineq_fun(z, n, dx):
    Q, _ = compute_lags(z[:n], n, dx)
    return z[n] - Q

def _ineq_jac(z, n, dx):
    _, G = compute_lags(z[:n], n, dx)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def _validate(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return False
    return True

def _make_starts(base, n, rng):
    starts = [base.copy()]
    for sigma in [0.03, 0.08, 0.20]:
        v = base + sigma * rng.standard_normal(n)
        starts.append(project_capped_simplex(v, n / 2.0))
    v = base + 0.6 * rng.standard_normal(n)
    starts.append(project_capped_simplex(v, n / 2.0))
    for k in [1, 2, 3]:
        phase = rng.uniform(0, 2 * np.pi)
        i = np.arange(n)
        v = 0.5 + 0.35 * np.cos(2 * np.pi * k * i / n + phase)
        starts.append(project_capped_simplex(v, n / 2.0))
    return starts

def _solve_one(z0, n, dx, maxiter):
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.01)]
    cons = [
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n, dx)},
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac, 'args': (n, dx)},
    ]
    try:
        res = minimize(_obj, z0, method='SLSQP', jac=_obj_jac,
                       args=(n, dx), bounds=bounds, constraints=cons,
                       options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
        if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
            return res.x
    except Exception:
        pass
    return None

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine base from parent
    parent = globals().get('initial_h_values', None)
    base_n = None
    base_vec = None
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if 2 <= len(p) <= 999 and np.all(np.isfinite(p)):
            base_n = len(p)
            if np.all(p >= -1e-12) and np.all(p <= 1.0 + 1e-12) and abs(np.sum(p) - len(p) / 2.0) < 1e-10:
                base_vec = np.clip(p, 0.0, 1.0).copy()
                r = len(p) / 2.0 - np.sum(base_vec)
                if r > 0:
                    idx = np.argsort(1.0 - base_vec)
                    for i in idx:
                        if r <= 0:
                            break
                        add = min(r, 1.0 - base_vec[i])
                        base_vec[i] += add
                        r -= add
                elif r < 0:
                    idx = np.argsort(base_vec)
                    for i in idx:
                        if r >= 0:
                            break
                        sub = min(-r, base_vec[i])
                        base_vec[i] -= sub
                        r += sub
            else:
                base_vec = project_capped_simplex(p, len(p) / 2.0)
        elif 2 <= len(p) <= 999:
            base_n = len(p)
            base_vec = np.full(len(p), 0.5)
    if base_n is None:
        base_n = int(kwargs.get('n_points', 45))
        if base_n < 2:
            base_n = 45
        if base_n > 999:
            base_n = 45
        base_vec = np.full(base_n, 0.5)
    n = base_n
    dx = 2.0 / n
    m = n / 2.0

    # Ensure base is feasible
    if not _validate(base_vec, n):
        base_vec = project_capped_simplex(base_vec, m)
    if not _validate(base_vec, n):
        base_vec = np.full(n, 0.5)

    best_h = base_vec.copy()
    best_score = exact_score(best_h)

    def _try_candidate(h):
        nonlocal best_h, best_score
        h = project_capped_simplex(h, n / 2.0)
        if not _validate(h, n):
            return
        s = exact_score(h)
        if s < best_score - 1e-12:
            best_score = s
            best_h = h.copy()

    # SLSQP at base resolution
    if 2 <= n <= 128:

                best_h = s_vec.copy()
            t0 = min(1.01, sc + 1e-7)

        # Polish from best

            parent_fine = project_capped_simplex(parent_fine, m_f)
            if _validate(parent_fine, N):

                if pf < fine_score - 1e-12:
                    fine_score = pf
                    coarse_best_fine = parent_fine.copy()
                    if fine_score < best_score - 1e-12:
                        best_score = fine_score

        starts_f = _make_starts(coarse_best_fine, N, rng)

            if result is not None:

                h_cand = project_capped_simplex(h_cand, m_f)
                if _validate(h_cand, N):
                    s = exact_score(h_cand)
                    if s < fine_score - 1e-12:
                        fine_score = s
                        coarse_best_fine = h_cand.copy()
                        if s < best_score - 1e-12:
                            best_score = s
                            best_h = h_cand.copy()
        # Polish fine
        t0 = min(1.01, fine_score + 1e-7)
        z0 = np.concatenate([coarse_best_fine, [t0]])

        final_h = np.full(final_n, 0.5)
        final_score = exact_score(final_h)
    return final_h, float(final_score), int(final_n)
