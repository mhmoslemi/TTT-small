import numpy as np

def exact_score(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def feasible(h, n):
    if len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    m = n / 2.0
    return abs(np.sum(h) - m) <= 1e-10

def mass_repair(h, n):
    m = n / 2.0

            break
        if r > 0:
            order = np.argsort(1.0 - h)[::-1]
            for i in order:
                if r <= 0:
                    break
                amt = min(r, 1.0 - h[i])
                if amt > 0:
                    h[i] += amt
                    r -= amt
        else:
            order = np.argsort(h)[::-1]
            for i in order:
                if r >= 0:
                    break
                amt = min(-r, h[i])
                if amt > 0:
                    h[i] -= amt
                    r += amt
    return h

def cap_projection(v, n):

def get_lag_data(h, n):
    dx = 2.0 / n

        G[k, sl] += dx * (1.0 - h_r)
        G[k, sl_r] -= dx * h_l
    return q, G

    return F, grad
