def pair_descent(h, n, dx, rng, max_sweeps):
    best_h = h.copy()
    best_score = exact_score(h, n)
    a = lag_vec(h, dx)
    for sweep in range(max_sweeps):
        improved = False
        pairs = [(p, q) for p in range(n) for q in range(p+1, n)]
        rng.shuffle(pairs)
        for p, q in pairs:
            v = np.zeros(n)
            v[p] = 1.0
            v[q] = -1.0
            b = dx * (np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full'))
            qv = -dx * np.correlate(v, v, mode='full')

    return best_h, best_score

def random_escape(h, n, dx, rng, n_dir):
    best_score = exact_score(h, n)
    a = lag_vec(h, dx)
    any_accepted = False
    for _ in range(n_dir):
        v = rng.standard_normal(n)
        v -= np.mean(v)
        if n > 1:

        hi = float('inf')

def generate_starts(h0, n, dx, T, rng):
    starts = [h0.copy()]
    # Reversal
    starts.append(h0[::-1].copy())
    # Complement
    starts.append((1.0 - h0).copy())
    # Roll
    if n > 1:
        shift = max(1, n // 3)
        starts.append(np.roll(h0, shift).copy())
    # Gaussian perturbations
    for sigma in [0.04, 0.12]:
        pert = h0 + sigma * rng.standard_normal(n)
        starts.append(project(pert, T))
    # Uniform random
    starts.append(project(rng.random(n), T))
    # Near-binary
    idx = rng.permutation(n)
    nb = np.zeros(n)
    floor_T = int(np.floor(T))
    nb[idx[:floor_T]] = 1.0
    frac = T - floor_T
    if frac > 1e-10:
        nb[idx[floor_T]] = frac
    starts.append(project(nb, T))
    # Cosine
    if n >= 2:
        f = rng.integers(1, min(4, n // 2) + 1)
        phase = rng.uniform(0, 2 * pi)
        i = np.arange(n)
        cos_v = 0.5 + 0.45 * np.cos(2 * pi * f * (i + 0.5) / n + phase)
        starts.append(project(cos_v, T))
    # Deduplicate and validate
    unique = [starts[0]]
    for s in starts[1:]:
        if not np.all(np.isfinite(s)):
            continue
        dup = False
        for u in unique:
            if np.max(np.abs(s - u)) < 1e-12:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique

        return np.full(n, T / n)
    lo = float(np.min(y) - 1.0)
    hi = float(np.max(y) + 1.0)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > T:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    r = T - np.sum(x)
    for i in range(n):
        if r > 1e-14:
            add = min(r, 1.0 - x[i])
            x[i] += add
            r -= add
        elif r < -1e-14:
            sub = min(-r, x[i])
            x[i] -= sub
            r += sub
    if not np.all(np.isfinite(x)) or abs(float(np.sum(x)) - T) > 1e-10:
        return np.full(n, T / n)
    return x

def line_search(a, b, q, lo, hi):
    if hi - lo < 1e-15:
        return 0.0
    if not (np.all(np.isfinite(a)) and np.all(np.isfinite(b)) and np.all(np.isfinite(q))):
        return 0.0
    best_val = float(np.max(a))
    best_a = 0.0
    xs = np.linspace(lo, hi, 17)
    xs = np.unique(np.concatenate([xs, [0.0]]))
    vals = np.max(a[None, :] + xs[:, None] * b[None, :] + (xs[:, None] ** 2) * q[None, :], axis=1)
    vals = np.where(np.isfinite(vals), vals, np.inf)
    idx = int(np.argmin(vals))
    if vals[idx] < best_val:
        best_val = vals[idx]
        best_a = float(xs[idx])
    seg_lo = float(xs[max(0, idx - 1)])
    seg_hi = float(xs[min(len(xs) - 1, idx + 1)])
    for _ in range(5):
        if seg_hi - seg_lo < 1e-13:
            break
        xs2 = np.linspace(seg_lo, seg_hi, 5)
        vals2 = np.max(a[None, :] + xs2[:, None] * b[None, :] + (xs2[:, None] ** 2) * q[None, :], axis=1)
        vals2 = np.where(np.isfinite(vals2), vals2, np.inf)
        j = int(np.argmin(vals2))
        if vals2[j] < best_val:
            best_val = vals2[j]
            best_a = float(xs2[j])
        seg_lo = float(xs2[max(0, j - 1)])
        seg_hi = float(xs2[min(len(xs2) - 1, j + 1)])
    return best_a

def pair_descent(h, n, dx, rng, max_sweeps):
    h = h.copy()
    best_score = exact_score(h, n)
    a = lag_vec(h, dx)
    T = n / 2
    for _ in range(max_sweeps):
        improved = False
        pairs = [(p, q) for p in range(n) for q in range(p + 1, n)]
        rng.shuffle(pairs)
        for p, q in pairs:
            lo = max(-h[p], h[q] - 1.0)
            hi = min(1.0 - h[p], h[q])
            if hi - lo < 1e-14:
                continue
            v = np.zeros(n)
            v[p] = 1.0
            v[q] = -1.0
            b = dx * (np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full'))
            qv = -dx * np.correlate(v, v, mode='full')
            alpha = line_search(a, b, qv, lo, hi)
            if abs(alpha) < 1e-15:
                continue
            trial = h.copy()
            trial[p] += alpha
            trial[q] -= alpha
            trial = project(trial, T)
            sc = exact_score(trial, n)
            if sc < best_score - 1e-12:
                h = trial
                best_score = sc
                a = lag_vec(h, dx)
                improved = True
        if not improved:
            break
    return h, best_score

def random_escape(h, n, dx, rng, n_dir):
    h = h.copy()
    best_score = exact_score(h, n)
    a = lag_vec(h, dx)
    T = n / 2
    any_acc = False
    for _ in range(n_dir):
        v = rng.standard_normal(n)
        v -= np.mean(v)
        if n > 1:
            v[-1] = -np.sum(v[:-1])
        mn = np.max(np.abs(v))
        if mn < 1e-15:
            continue
        v /= mn
        lo = float('-inf')
        hi = float('inf')
        for i in range(n):
            if v[i] > 0:
                lo = max(lo, -h[i] / v[i])
                hi = min(hi, (1.0 - h[i]) / v[i])
            elif v[i] < 0:
                lo = max(lo, (1.0 - h[i]) / v[i])
                hi = min(hi, -h[i] / v[i])
        if hi - lo < 1e-14:
            continue
        b = dx * (np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full'))
        qv = -dx * np.correlate(v, v, mode='full')
        alpha = line_search(a, b, qv, lo, hi)
        if abs(alpha) < 1e-15:
            continue
        trial = project(h + alpha * v, T)
        sc = exact_score(trial, n)
        if sc < best_score - 1e-12:
            h = trial
            best_score = sc
            a = lag_vec(h, dx)
            any_acc = True
    return h, best_score, any_acc

def generate_starts(h0, n, dx, T, rng):
    starts = [h0.copy()]
    starts.append(h0[::-1].copy())
    starts.append((1.0 - h0).copy())
    if n > 1:
        starts.append(np.roll(h0, max(1, n // 3)).copy())
    for sigma in (0.04, 0.12):
        starts.append(project(h0 + sigma * rng.standard_normal(n), T))
    starts.append(project(rng.random(n), T))
    idx = rng.permutation(n)
    nb = np.zeros(n)
    fT = int(np.floor(T))
    nb[idx[:fT]] = 1.0
    frac = T - fT
    if frac > 1e-10:
        nb[idx[fT]] = frac
    starts.append(project(nb, T))
    if n >= 2:
        f = int(rng.integers(1, min(4, n // 2) + 1))
        ph = float(rng.uniform(0, 2 * pi))
        i = np.arange(n, dtype=np.float64)
        cv = 0.5 + 0.45 * np.cos(2 * pi * f * (i + 0.5) / n + ph)
        starts.append(project(cv, T))
    unique = [starts[0]]
    for s in starts[1:]:
        if not np.all(np.isfinite(s)):
            continue
        dup = False
        for u in unique:
            if np.max(np.abs(s - u)) < 1e-12:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try:
        seed = int(seed) % (2 ** 32)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
        if h0.ndim != 1 or len(h0) == 0 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        n_kw = kwargs.get('n_points', 47)
        n = 47 if not isinstance(n_kw, int) or n_kw < 2 or n_kw > 64 else n_kw
        h0 = np.full(n, 0.5)
    n = len(h0)
    dx = 2.0 / n
    T = n / 2
    fallback = project(h0.copy(), T)
    if n <= 1:
        return (fallback, exact_score(fallback, n), n)
    starts = generate_starts(fallback, n, dx, T, rng)
    best_h = fallback.copy()
    best_score = exact_score(fallback, n)
    sw = 4 if n <= 64 else 0
    for s in starts:
        h_cur = s.copy()
        if sw > 0:
            h_cur, sc = pair_descent(h_cur, n, dx, rng, sw)
        else:
            sc = exact_score(h_cur, n)
        h_cur, sc, acc = random_escape(h_cur, n, dx, rng, 4)
        if acc and sw > 0:
            h_cur, sc = pair_descent(h_cur, n, dx, rng, 2)
        if sc < best_score:
            best_h = h_cur.copy()
            best_score = sc
    for sigma in (0.015, 0.05, 0.15):
        h_try = project(best_h + sigma * rng.standard_normal(n), T)
        h_try, sc = pair_descent(h_try, n, dx, rng, 2)
        h_try, sc, acc = random_escape(h_try, n, dx, rng, 3)
        if acc:
            h_try, sc = pair_descent(h_try, n, dx, rng, 1)
        if sc < best_score:
            best_h = h_try.copy()
            best_score = sc
    final = project(best_h, T)
    final_score = exact_score(final, n)
    fb_score = exact_score(fallback, n)
    if fb_score < final_score:
        final = fallback
        final_score = fb_score
    return (final, final_score, n)
