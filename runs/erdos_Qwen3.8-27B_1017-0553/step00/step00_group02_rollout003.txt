def smooth_value(h, tau):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    r = c * (2.0 / n)
    R = np.max(r)
    e = np.exp((r - R) / tau)
    Z = np.sum(e)
    return R + tau * math.log(Z)

def optimize_start(h0, temperatures, iteration_limit, rng):
    h0 = project_box_mass(h0)
    if not is_feasible(h0):
        h0 = np.full(len(h0), 0.5)
    best_vec = h0.copy()
    best_score = exact_c5(h0)
    x = h0.copy()
    n = len(x)
    
    for tau in temperatures:
        alpha = min(max(0.5, n / 45.0), 8.0 * n / 45.0)
        alpha_max = 8.0 * n / 45.0
        stale = 0
        for it in range(iteration_limit):
            g = smooth_value_gradient(x, tau)
            if g is None:
                break
            accepted = False
            for trial in range(16):
                y = project_box_mass(x - alpha * g)
                if not is_feasible(y):
                    break
                d_step = y - x
                if np.max(np.abs(d_step)) <= 1e-8:
                    break
                slope = np.dot(g, d_step)
                if slope >= 0:
                    alpha *= 0.5
                    continue
                phi_y = smooth_value(y, tau)
                phi_x = smooth_value(x, tau)
                if phi_y <= phi_x + 1e-4 * slope + 1e-12 * max(1.0, abs(phi_x)):
                    # accept
                    exact_y = exact_c5(y)
                    if exact_y < best_score - 1e-12:
                        best_score = exact_y
                        best_vec = y.copy()
                    x = y
                    accepted = True
                    phi_decrease = phi_x - phi_y
                    if phi_decrease <= 1e-12:
                        stale += 1
                    else:
                        stale = 0
                    alpha = min(1.25 * alpha, alpha_max)
                    break
                else:
                    alpha *= 0.5
            if not accepted:
                break
            if stale >= 5:
                break
    
    # Also check final x
    if is_feasible(x):
        sc = exact_c5(x)
        if sc < best_score - 1e-12:
            best_score = sc
            best_vec = x.copy()
    
    return best_vec, best_score

    if n_src == n_target:
        return project_box_mass(h_src)
    j_idx = np.arange(n_src)
    k_idx = np.arange(n_target)
    # Source bins: [j/n_src, (j+1)/n_src)
    # Target bins: [k/n_target, (k+1)/n_target)
    j_starts = j_idx / n_src
    j_ends = (j_idx + 1) / n_src
    k_starts = k_idx / n_target
    k_ends = (k_idx + 1) / n_target
    # Overlap matrix: ov[j,k] = max(0, min(j_ends[j], k_ends[k]) - max(j_starts[j], k_starts[k]))
    # Using broadcasting
    ov = np.maximum(0, np.minimum(j_ends[:, None], k_ends[None, :]) - np.maximum(j_starts[:, None], k_starts[None, :]))
    result = ov.T @ h_src * n_target
    return project_box_mass(result)

    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return 1e10
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or len(h) == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - len(h) / 2.0) > 1e-9:
        return False
    return True

    y = np.asarray(y, dtype=np.float64)
    n = len(y)
    if n == 0 or not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    y = np.clip(y, -1e6, 1e6)
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(y - mid, 0.0, 1.0)
        if np.sum(x) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

    for i in range(n * 200):

        idx = i % n

            add = min(resid, 1.0 - x[idx])

            resid -= add
        else:
            sub = min(-resid, x[idx])
            x[idx] -= sub
            resid += sub
    if not is_feasible(x):

    return x

    result = (ov.T @ h_src) * n_target

    R = float(np.max(r))

import numpy as np
import math

def exact_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return 1e10
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def is_feasible(h):
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or len(h) == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - len(h) / 2.0) > 1e-9:
        return False
    return True

def project_box_mass(y):
    y = np.asarray(y, dtype=np.float64)
    n = len(y)
    if n == 0 or not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    y = np.clip(y, -1e6, 1e6)
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    resid = target - np.sum(x)
    for i in range(n * 200):
        if abs(resid) < 1e-14:
            break
        idx = i % n
        if resid > 0:
            a = min(resid, 1.0 - x[idx])
            x[idx] += a
            resid -= a
        else:
            a = min(-resid, x[idx])
            x[idx] -= a
            resid += a
    if not is_feasible(x):
        return np.full(n, 0.5)
    return x

def area_transfer(h_src, n_target):
    n_src = len(h_src)
    if n_src == n_target:
        return project_box_mass(h_src)
    j = np.arange(n_src)
    k = np.arange(n_target)
    ov = np.maximum(0.0, np.minimum((j + 1)[:, None] / n_src, (k + 1)[None, :] / n_target)
                    - np.maximum(j[:, None] / n_src, k[None, :] / n_target))
    return project_box_mass((ov.T @ h_src) * n_target)

def smooth_value(h, tau):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    r = c * (2.0 / n)
    R = float(np.max(r))
    e = np.exp((r - R) / tau)
    return R + tau * math.log(float(np.sum(e)))

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    r = c * dx
    R = float(np.max(r))
    e = np.exp((r - R) / tau)
    p = e / float(np.sum(e))
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        idx = d + n - 1
        w = dx * p[idx]
        if d >= 0:
            L = n - d
            if L > 0:
                grad[:L] += -h[d:d + L] * w
                grad[d:] += (1.0 - h[:L]) * w
        else:
            s = -d
            L = n - s
            if L > 0:
                grad[:L] += (1.0 - h[s:s + L]) * w
                grad[s:s + L] += -h[:L] * w
    if not np.all(np.isfinite(grad)):
        return None
    return grad

def optimize_start(h0, temperatures, iteration_limit):
    h0 = project_box_mass(h0)
    if not is_feasible(h0):
        h0 = np.full(len(h0), 0.5)
    best_vec = h0.copy()
    best_score = exact_c5(h0)
    x = h0.copy()
    n = len(x)

    a_max = 8.0 * n / 45.0
    for tau in temperatures:
        alpha = a_init
        stale = 0
        for _ in range(iteration_limit):

            if g is None:

            phi_x = smooth_value(x, tau)

                phi_y = smooth_value(y, tau)

                alpha *= 0.5

    sc = exact_c5(x)
    if sc < best_score - 1e-12:
        best_score = sc
        best_vec = x.copy()
    return best_vec, best_score
