def project_cap_simplex(y, T):
    n = len(y)
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo, hi = y.min() - 1.0, y.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0.0, 1.0).sum() > T:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    r = T - x.sum()
    for i in range(n):
        if r > 1e-14:
            a = min(r, 1.0 - x[i]); x[i] += a; r -= a
        elif r < -1e-14:
            a = min(-r, x[i]); x[i] -= a; r += a
