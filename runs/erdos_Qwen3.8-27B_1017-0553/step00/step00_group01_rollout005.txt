    best, best_score = cur.copy(), cur_score
    a = get_lag_vector(cur, dx)
    for _ in range(n_sweeps):
        imp = False
        pairs = [(p, q) for p in range(n) for q in range(p + 1, n)]
        rng.shuffle(pairs)
        for p, q in pairs:
            lo = max(-cur[p], cur[q] - 1.0)
            hi = min(1.0 - cur[p], cur[q])
            if hi - lo < 1e-14:
                continue
            v = np.zeros(n)
            v[p], v[q] = 1.0, -1.0
            b = dx * (np.correlate(v, 1.0 - cur, mode='full') - np.correlate(cur, v, mode='full'))
            qv = -dx * np.correlate(v, v, mode='full')
            alpha, pv = line_search(a, b, qv, lo, hi)
            if pv >= cur_score - 1e-12:
                continue
            trial = cur.copy()
            trial[p] += alpha
            trial[q] -= alpha
            trial = project_capped_simplex(trial, T)
            if not is_feasible(trial, n):
                continue
            s = exact_score(trial, n)
            if s < cur_score - 1e-12:
                cur, cur_score = trial, s
                a = get_lag_vector(cur, dx)
                imp = True
                if s < best_score - 1e-12:
                    best, best_score = cur.copy(), s
        if not imp:
            break
    return best, best_score

def random_escape(h, n, dx, rng, n_dir):
    T = n / 2.0
    cur, cur_score = h.copy(), exact_score(h, n)
    best, best_score = cur.copy(), cur_score
    accepted = False
    for _ in range(n_dir):
        v = rng.standard_normal(n)
        v -= np.mean(v)
        if n > 1:
            v[-1] = -np.sum(v[:-1])
        vm = np.max(np.abs(v))
        if vm < 1e-12 or not np.all(np.isfinite(v)):
            continue
        v /= vm
        fe = feasible_interval_general(v, cur)
        if fe is None:
            continue
        a = get_lag_vector(cur, dx)
        b = dx * (np.correlate(v, 1.0 - cur, mode='full') - np.correlate(cur, v, mode='full'))
        qv = -dx * np.correlate(v, v, mode='full')
        alpha, pv = line_search(a, b, qv, fe[0], fe[1])
        if pv >= cur_score - 1e-12:
            continue
        trial = project_capped_simplex(cur + alpha * v, T)
        if not is_feasible(trial, n):
            continue
        s = exact_score(trial, n)
        if s < cur_score - 1e-12:
            cur, cur_score = trial, s
            accepted = True
            if s < best_score - 1e-12:
                best, best_score = cur.copy(), s
    return best, best_score, accepted

def build_starts(parent, n, T, rng):
    starts = []
    if parent is not None:
        p = project_capped_simplex(parent.copy(), T)
        starts.append(p)
        starts.append(project_capped_simplex(1.0 - p, T))
        starts.append(project_capped_simplex(p[::-1].copy(), T))
        if n > 1:
            starts.append(project_capped_simplex(np.roll(p, max(1, n // 3)), T))
    base = np.full(n, 0.5)
    starts.append(project_capped_simplex(base + 0.04 * rng.standard_normal(n), T))
    starts.append(project_capped_simplex(base + 0.12 * rng.standard_normal(n), T))
    starts.append(project_capped_simplex(rng.uniform(0, 1, n), T))
    perm = rng.permutation(n)
    nb = np.zeros(n)
    ft = int(np.floor(T))
    nb[perm[:ft]] = 1.0
    rem = T - ft
    if rem > 1e-12 and ft < n:
        nb[perm[ft]] = rem
    starts.append(project_capped_simplex(nb, T))
    if n >= 2:
        phase = rng.uniform(0, 2 * np.pi)
        f = int(rng.integers(1, min(4, n // 2) + 1))
        x = np.arange(n, dtype=np.float64)
        starts.append(project_capped_simplex(0.5 + 0.45 * np.cos(2 * np.pi * f * (x + 0.5) / n + phase), T))
    deduped = []
    for s in starts:
        if not is_feasible(s, n):
            continue
        if not any(np.max(np.abs(s - d)) < 1e-12 for d in deduped):
            deduped.append(s)
    return deduped if deduped else [np.full(n, 0.5)]

def run(seed=42, **kwargs):
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv.copy()
    except Exception:
        pass
    if parent is not None:
        n = len(parent)
    else:
        n = kwargs.get('n_points', 47)
        if not isinstance(n, int):
            n = 47
        n = max(1, min(64, n))
    T = n / 2.0
    dx = 2.0 / n
    try:
        s = int(seed) % (2 ** 32)
    except Exception:
        s = 42
    rng = np.random.default_rng(s)
    if n == 1:
        return np.array([1.0]), 0.0, 1
    starts = build_starts(parent, n, T, rng)
    fallback = project_capped_simplex(starts[0].copy(), T)
    fb_score = exact_score(fallback, n)
    global_best, global_score = fallback.copy(), fb_score
    use_pair = n <= 64
    for st in starts:
        if use_pair:
            bh, bs = pair_descent(st, n, dx, rng, 4)
            bh, bs, acc = random_escape(bh, n, dx, rng, 4)
            if acc:
                bh, bs = pair_descent(bh, n, dx, rng, 2)
        else:
            bh, bs, _ = random_escape(st, n, dx, rng, 4)
        if bs < global_score - 1e-12:
            global_best, global_score = bh.copy(), bs
    for sigma in [0.015, 0.05, 0.15]:
        pert = project_capped_simplex(global_best + sigma * rng.standard_normal(n), T)
        if not is_feasible(pert, n):
            continue
        if use_pair:
            bh, bs = pair_descent(pert, n, dx, rng, 2)
        else:
            bh, bs = pert, exact_score(pert, n)
        bh, bs, acc = random_escape(bh, n, dx, rng, 3)
        if acc and use_pair:
            bh, bs = pair_descent(bh, n, dx, rng, 1)
        if bs < global_score - 1e-12:
            global_best, global_score = bh.copy(), bs
    final = project_capped_simplex(global_best, T)
    final_score = exact_score(final, n)
    if final_score > fb_score + 1e-10:
        final, final_score = fallback, fb_score
    return final, float(final_score), int(n)
