        for s in unique_starts:
            sc = exact_score(s)
            if sc < best_score - 1e-12:
                best_h = s.copy()
                best_score = sc
            try:
                opt_h, opt_sc = continuation_search(s, n)
                if opt_sc < best_score - 1e-12:
                    best_h = opt_h.copy()
                    best_score = opt_sc
            except Exception:
                pass
        try:
            poll_h, poll_sc = pairwise_poll(best_h, n)
            if poll_sc < best_score - 1e-12:
                best_h = poll_h.copy()
                best_score = poll_sc
        except Exception:
            pass
    if not is_feasible(best_h, n):
        proj = project_capped_simplex(best_h, n)
        if proj is not None:
            best_h = proj
        else:
            best_h = baseline.copy()
    final_score = exact_score(best_h)
    return best_h, final_score, n
