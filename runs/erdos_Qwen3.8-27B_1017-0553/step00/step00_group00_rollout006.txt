    high = float(np.max(v))

        h_fine_base = project_capped_simplex(h_fine_base, m_fine)
        h_parent_fine = None
        if parent is not None:
            p = np.asarray(parent, dtype=np.float64)
            if p.ndim == 1 and len(p) == base_n and np.all(np.isfinite(p)):
                hp = np.repeat(p, 2)
                h_parent_fine = project_capped_simplex(np.clip(hp, 0.0, 1.0), m_fine)
        h_fine_opt, s_fine = optimize_at_resolution(h_fine_base, n_fine, rng, h_parent_fine)
        if s_fine < best_score - 1e-12:
            best_score = s_fine
            best_h = h_fine_opt
            best_n = n_fine
    h_final = np.asarray(best_h, dtype=np.float64)
    n_final = len(h_final)
    h_final = project_capped_simplex(h_final, n_final / 2.0)
    score_final = exact_score(h_final)
    if not np.all(np.isfinite(h_final)) or score_final > best_score + 1e-6:
        h_final = base_vec.copy()
        n_final = len(h_final)
        score_final = exact_score(h_final)
    return (h_final, score_final, n_final)
