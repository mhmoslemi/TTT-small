        alpha = line_search(a, b_vec, q_vec, lo, hi)
        if alpha == 0.0:
            continue
        pred = float(np.max(a + alpha * b_vec + alpha * alpha * q_vec))
        if pred > score_cur - 1e-12:
            continue
        trial = project_capped_simplex(h_cur + alpha * v, n / 2.0)
        sc = exact_score(trial, dx)
        if sc < score_cur - 1e-12:
            h_cur = trial
            score_cur = sc
            any_acc = True
            if sc < best_score:
                best_score = sc
                best_h = trial.copy()
    return best_h, best_score, any_acc

def make_starts(fallback, n, rng, T):
    starts = [fallback.copy(), (1.0 - fallback).copy()]
    if n > 1:
        starts.append(fallback[::-1].copy())
        starts.append(np.roll(fallback, max(1, n // 3)).copy())
    for sd in [0.04, 0.12]:
        starts.append(fallback + sd * rng.standard_normal(n))
    starts.append(rng.uniform(0.0, 1.0, n))
    perm = rng.permutation(n)
    nb = np.zeros(n)
    nb[perm[:int(T)]] = 1.0
    frac = T - int(T)
    if frac > 0 and int(T) < n:
        nb[perm[int(T)]] = frac
    starts.append(nb)
    if n >= 2:
        f = int(rng.integers(1, min(4, n // 2) + 1))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        starts.append(0.5 + 0.45 * np.cos(2.0 * np.pi * f * (np.arange(n) + 0.5) / n + ph))
    projected = []
    for s in starts:
        s = np.asarray(s, dtype=np.float64)
        if s.shape != (n,) or not np.all(np.isfinite(s)):
            continue
        s = project_capped_simplex(s, T)
        dup = False
        for p in projected:
            if np.max(np.abs(s - p)) < 1e-12:
                dup = True
                break
        if not dup:
            projected.append(s)
    return projected

def run(seed=42, **kwargs):
    try:
        seed = int(seed) % (2 ** 32)
    except (ValueError, TypeError):
        seed = 42
    rng = np.random.default_rng(seed)
    n, dx, T, fallback = 47, 2.0 / 47, 23.5, np.full(47, 0.5)
    try:
        n_req = int(kwargs.get('n_points', 47))
        if 1 <= n_req <= 64:
            n = n_req
    except (ValueError, TypeError):
        pass
    try:
        ph = np.array(initial_h_values, dtype=np.float64).flatten()
        if ph.size > 0 and np.all(np.isfinite(ph)):
            n = ph.size
            dx = 2.0 / n
            T = n / 2.0
            if np.all(ph >= -1e-10) and np.all(ph <= 1.0 + 1e-10) and abs(np.sum(ph) - T) < 1e-10:
                fallback = np.clip(ph, 0.0, 1.0)
            else:
                fallback = project_capped_simplex(ph, T)
    except (NameError, Exception):
        pass
    if n <= 1:
        return fallback, exact_score(fallback, 2.0 / n), n
    starts = make_starts(fallback, n, rng, T)
    best_h = fallback.copy()
    best_score = exact_score(fallback, 2.0 / n)
    for s in starts:
        sc = exact_score(s, 2.0 / n)
        if sc < best_score:
            best_score = sc
            best_h = s.copy()
    if n <= 64:
        for s in starts:
            sc = exact_score(s, 2.0 / n)
            bh, bs = pair_descent(s, 2.0 / n, n, rng, 4)
            bh, bs, acc = random_escape(bh, 2.0 / n, n, rng, 4)
            if bs < best_score:
                best_score = bs
                best_h = bh.copy()
            if acc:
                bh, bs = pair_descent(bh, 2.0 / n, n, rng, 2)
                if bs < best_score:
                    best_score = bs
                    best_h = bh.copy()
        for sigma in [0.015, 0.05, 0.15]:
            pert = project_capped_simplex(best_h + sigma * rng.standard_normal(n), T)
            sc = exact_score(pert, 2.0 / n)
            if sc >= best_score:
                continue
            bh, bs = pair_descent(pert, 2.0 / n, n, rng, 2)
            bh, bs, acc = random_escape(bh, 2.0 / n, n, rng, 3)
            if acc:
                bh, bs = pair_descent(bh, 2.0 / n, n, rng, 1)
            if bs < best_score:
                best_score = bs
                best_h = bh.copy()
    final_h = project_capped_simplex(best_h, T)
    final_score = exact_score(final_h, 2.0 / n)
    fb_score = exact_score(fallback, 2.0 / n)
    if final_score >= fb_score:
        final_h, final_score = fallback.copy(), fb_score
    return final_h, final_score, n
