        if s_score < best_score - 1e-12:
            best_score = s_score
            best_h = start.copy()
        t0 = min(1.01, s_score + 1e-7)
        z0 = np.concatenate([start, [t0]])
        h_sol = try_solve(z0, base_n, dx)
        if h_sol is not None:
            sol_score = exact_score(h_sol)
            if sol_score < best_score - 1e-12:
                best_score = sol_score
                best_h = h_sol.copy()

    t0 = min(1.01, best_score + 1e-7)
    z0 = np.concatenate([best_h, [t0]])
    h_sol = try_solve(z0, base_n, dx)
    if h_sol is not None:
        sol_score = exact_score(h_sol)
        if sol_score < best_score - 1e-12:
            best_score = sol_score
            best_h = h_sol.copy()

    if base_n <= 64:
        fine_n = 2 * base_n
        fine_dx = 2.0 / fine_n
        fine_rng = np.random.default_rng(seed + 1)
        h_fine = capped_simplex_projection(np.repeat(best_h, 2), fine_n / 2.0)
        fine_best_h = h_fine.copy()
        fine_best_score = exact_score(h_fine)
        fine_starts = generate_starts(h_fine, fine_n, fine_rng)
        p_trans = capped_simplex_projection(np.repeat(h_base, 2), fine_n / 2.0)
        if not np.allclose(p_trans, fine_starts[0]):
            fine_starts.append(p_trans)
        for start in fine_starts:
            s_score = exact_score(start)
            if s_score < fine_best_score - 1e-12:
                fine_best_score = s_score
                fine_best_h = start.copy()
            t0 = min(1.01, s_score + 1e-7)
            z0 = np.concatenate([start, [t0]])
            h_sol = try_solve(z0, fine_n, fine_dx)
            if h_sol is not None:
                sol_score = exact_score(h_sol)
                if sol_score < fine_best_score - 1e-12:
                    fine_best_score = sol_score
                    fine_best_h = h_sol.copy()
        t0 = min(1.01, fine_best_score + 1e-7)
        z0 = np.concatenate([fine_best_h, [t0]])
        h_sol = try_solve(z0, fine_n, fine_dx)
        if h_sol is not None:
            sol_score = exact_score(h_sol)
            if sol_score < fine_best_score - 1e-12:
                fine_best_score = sol_score
                fine_best_h = h_sol.copy()
        if fine_best_score < best_score - 1e-12:
            best_score = fine_best_score
            best_h = fine_best_h.copy()

    best_h = np.asarray(best_h, dtype=np.float64)
    n_pts = len(best_h)
    if not np.all(np.isfinite(best_h)):
        best_h = np.full(base_n, 0.5)
        n_pts = base_n
    best_h = capped_simplex_projection(best_h, n_pts / 2.0)
    best_score = exact_score(best_h)
    return (best_h, best_score, int(n_pts))
