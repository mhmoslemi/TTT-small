import numpy as np
from scipy.optimize import minimize

def capped_simplex_project(v, m):
    """Project v onto {h: 0<=h_i<=1, sum h_i = m} via bisection on lambda."""
    n = len(v)
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(80):
        mid = 0.5 * (low + high)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > m:
            low = mid
        else:
            high = mid
    h = np.clip(v - mid, 0.0, 1.0)
    # Fix floating-point mass residual
    r = m - np.sum(h)
    if r > 0:
        idx = np.argsort(1.0 - h)[::-1]
        for i in idx:
            add = min(r, 1.0 - h[i])
            if add > 0:
                h[i] += add
                r -= add
            if abs(r) < 1e-15:
                break
    elif r < 0:
        idx = np.argsort(h)[::-1]
        for i in idx:
            rem = min(-r, h[i])
            if rem > 0:
                h[i] -= rem
                r += rem
            if abs(r) < 1e-15:
                break
    return h

def exact_score(h):
    """Compute C5 = max of full correlation h with 1-h, scaled by dx."""
    if h is None or len(h) == 0:
        return float('inf')
    h = np.asarray(h, dtype=np.float64).ravel()
    if not np.all(np.isfinite(h)):
        return float('inf')
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def lag_q_and_grad(h, n, dx):
    """Compute Q vector (length 2n-1) and gradient matrix G (shape (2n-1, n))."""
    m_lags = 2 * n - 1
    Q = np.zeros(m_lags)
    G = np.zeros((m_lags, n))
    for d in range(-(n-1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        left = slice(lo, hi)
        right = slice(lo + d, hi + d)
        Q[row] = dx * np.dot(h[left], 1.0 - h[right])
        # Gradient: partially derivative of Q_d w.r.t. h_j
        # For j in left range (j from lo to hi-1), the term h[j]*(1-h[j+d]) contributes
        # d/dh_j = (1 - h[j+d]) for j in left, and -h[j-lo] for j in right (as the "1-h" part)
        # Actually: Q_d = dx * sum_{i=lo}^{hi-1} h[i]*(1-h[i+d])
        # dQ_d/dh_j = dx * [I(j in left range)*(1 - h[j+d]) - I(j in right range)*h[j-d]]
        # j in left range means lo <= j <= hi-1
        # j in right range means lo+d <= j <= hi-1+d
        G[row, left] += dx * (1.0 - h[right])
        G[row, right] -= dx * h[left]
    return Q, G

Wait, I need to be careful about the gradient. Let me re-derive.

Q_d = dx * sum_{i=lo}^{hi-1} h[i] * (1 - h[i+d])

Take derivative w.r.t. h[j]:
- If j appears as h[i] (i.e., j = i for some i in [lo, hi-1]), contribution: dx * (1 - h[j+d])
- If j appears as h[i+d] (i.e., j = i+d for some i in [lo, hi-1], meaning j-d = i, so j-d in [lo, hi-1], meaning j in [lo+d, hi-1+d]), contribution: -dx * h[j-d]

So:
- For j in [lo, hi-1]: G[row, j] += dx * (1 - h[j+d])
- For j in [lo+d, hi-1+d]: G[row, j] -= dx * h[j-d]

In NumPy slice terms with left = slice(lo, hi) and right = slice(lo+d, hi+d):
- G[row, left] += dx * (1 - h[right])
- G[row, right] -= dx * h[left]

This is correct. When d=0, left=right=slice(0,n), so:
G[row, :] += dx*(1-h) and G[row, :] -= dx*h, giving dx*(1-2h). Correct.

import numpy as np
from scipy.optimize import minimize

def capped_simplex_project(v, m):
    n = len(v)
    low = np.min(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(80):
        mid = 0.5 * (low + high)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > m:
            low = mid
        else:
            high = mid
    h = np.clip(v - mid, 0.0, 1.0)
    r = m - np.sum(h)
    if abs(r) > 1e-15:
        if r > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if r > 0:
                add = min(r, 1.0 - h[i])
                h[i] += add
                r -= add
            else:
                rem = min(-r, h[i])
                h[i] -= rem
                r += rem
            if abs(r) < 1e-15:
                break
    return h

def exact_score(h):
    if h is None:
        return float('inf')
    h = np.asarray(h, dtype=np.float64).ravel()
    if len(h) == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def is_feasible(h, n, tol=1e-10):
    if h is None or len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(np.sum(h) - n / 2.0) > tol:
        return False
    return True

def lag_Q_G(h, n, dx):
    Q = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        left = slice(lo, hi)
        right = slice(lo + d, hi + d)
        Q[row] = dx * np.dot(h[left], 1.0 - h[right])
        G[row, left] += dx * (1.0 - h[right])
        G[row, right] -= dx * h[left]
    return Q, G

def obj_cb(z, n, dx):
    return z[n]

def obj_jac(z, n, dx):
    j = np.zeros(n + 1)
    j[n] = 1.0
    return j

def eq_cb(z, n, dx):
    return np.sum(z[:n]) - n / 2.0

def eq_jac(z, n, dx):
    j = np.ones(n + 1)
    j[-1] = 0.0
    return j

def ineq_cb(z, n, dx):
    Q, _ = lag_Q_G(z[:n], n, dx)
    return z[n] - Q

def ineq_jac(z, n, dx):
    _, G = lag_Q_G(z[:n], n, dx)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def make_starts(base, n, rng):
    starts = [base.copy()]
    for sigma in [0.03, 0.08, 0.20]:
        v = base + sigma * rng.standard_normal(n)
        starts.append(capped_simplex_project(v, n / 2.0))
    v = 0.5 + 0.6 * rng.standard_normal(n)
    starts.append(capped_simplex_project(v, n / 2.0))
    for k in [1, 2, 3]:
        phase = rng.uniform(0, 2 * np.pi)
        i = np.arange(n)
        v = 0.5 + 0.35 * np.cos(2 * np.pi * k * i / n + phase)
        starts.append(capped_simplex_project(v, n / 2.0))
    return starts

def run_slsqp(z0, n, dx):
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.01)]
    cons = [
        {'type': 'eq', 'fun': eq_cb, 'jac': eq_jac, 'args': (n, dx)},
        {'type': 'ineq', 'fun': ineq_cb, 'jac': ineq_jac, 'args': (n, dx)},
    ]
    maxit = min(600, 300 + 2 * n)
    try:
        res = minimize(obj_cb, z0, jac=obj_jac, args=(n, dx), method='SLSQP',
                       bounds=bounds, constraints=cons, options={'ftol': 1e-10,
                       'maxiter': maxit, 'disp': False})
    except Exception:
        return None
    if res is None or not hasattr(res, 'x') or len(res.x) != n + 1:
        return None
    if not np.all(np.isfinite(res.x)):
        return None
    return res.x

def optimize_resolution(h_base, n, rng, extra_start=None):
    dx = 2.0 / n
    m = n / 2.0
    starts = make_starts(h_base, n, rng)
    if extra_start is not None:
        es = capped_simplex_project(extra_start, m)
        if not np.array_equal(es, h_base):
            starts.append(es)
    best_h = h_base.copy()
    best_s = exact_score(best_h)
    for s in starts:
        sc = exact_score(s)
        if sc < best_s - 1e-12:
            best_s = sc
            best_h = s.copy()
    for s in starts:
        t0 = min(1.01, exact_score(s) + 1e-7)
        z0 = np.concatenate([s, [t0]])
        x = run_slsqp(z0, n, dx)
        if x is None:
            continue
        h_c = capped_simplex_project(x[:n], m)
        sc = exact_score(h_c)
        if sc < best_s - 1e-12:
            best_s = sc
            best_h = h_c.copy()
    # Polish from best
    t0 = min(1.01, best_s + 1e-7)
    z0 = np.concatenate([best_h, [t0]])
    x = run_slsqp(z0, n, dx)
    if x is not None:
        h_c = capped_simplex_project(x[:n], m)
        sc = exact_score(h_c)
        if sc < best_s - 1e-12:
            best_s = sc
            best_h = h_c.copy()
    return best_h, best_s

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    fallback_n = 45
    base_n = fallback_n
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if 1 <= len(p) <= 999:
            base_n = len(p)
            if np.all(np.isfinite(p)):
                if is_feasible(p, base_n):
                    h_base = p.copy()
                else:
                    h_base = capped_simplex_project(p, base_n / 2.0)
            else:
                h_base = np.full(base_n, 0.5)
        else:
            h_base = np.full(fallback_n, 0.5)
    else:
        h_base = np.full(fallback_n, 0.5)

    global_best_h = h_base.copy()
    global_best_s = exact_score(h_base)
    global_best_n = len(h_base)

    if 2 <= base_n <= 128:
        h_opt, s_opt = optimize_resolution(h_base, base_n, rng)
        if s_opt < global_best_s - 1e-12 or (abs(s_opt - global_best_s) <= 1e-12 and base_n < global_best_n):
            global_best_h = h_opt
            global_best_s = s_opt
            global_best_n = base_n

    if 2 <= base_n <= 64:
        fine_n = 2 * base_n
        if fine_n <= 128:
            h_fine = np.repeat(global_best_h, 2)
            h_fine = capped_simplex_project(h_fine, fine_n / 2.0)
            h_extra = np.repeat(h_base, 2) if h_base.shape != global_best_h.shape else None
            h_opt2, s_opt2 = optimize_resolution(h_fine, fine_n, rng, extra_start=h_extra)
            if s_opt2 < global_best_s - 1e-12 or (abs(s_opt2 - global_best_s) <= 1e-12 and fine_n < global_best_n):
                global_best_h = h_opt2
                global_best_s = s_opt2
                global_best_n = fine_n

    # Final validation
    fv = capped_simplex_project(global_best_h, global_best_n / 2.0)
    sc = exact_score(fv)
    if not np.isfinite(sc) or sc > 1.0:
        fv = h_base.copy()
        sc = exact_score(fv)
    return fv, sc, len(fv)
