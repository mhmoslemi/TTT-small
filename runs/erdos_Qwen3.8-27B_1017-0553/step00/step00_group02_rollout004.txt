        parent = None
    if parent is None:
        parent = np.full(n0, 0.5)

    gbs = exact_score(parent)
    gbh = parent.copy()
    gbn = n0

    resolutions = [n0]
    if n0 <= 90:
        for mult in [2, 4]:
            v = n0 * mult
            if v <= 180 and v not in resolutions:
                resolutions.append(v)
    else:
        if 180 not in resolutions:
            resolutions.append(180)
    resolutions = sorted(set(resolutions))

    level_best = parent.copy()
    level_best_n = n0

    for li, n in enumerate(resolutions):
        if li == 0:
            h_base = level_best.copy()
        else:
            h_base = area_transfer(level_best, n)
        lb_score = exact_score(h_base)
        if lb_score < gbs - 1e-12:
            gbs, gbh, gbn = lb_score, h_base.copy(), n
        lb = h_base.copy()

        starts = make_starts(h_base, n, rng)
        for hs in starts:
            hs = project_box_mass(hs, n)
            if not is_feasible(hs):
                continue
            s = exact_score(hs)
            if s < gbs - 1e-12:
                gbs, gbh, gbn = s, hs.copy(), n
            if s < lb_score - 1e-12:
                lb, lb_score = hs.copy(), s
            res = run_slsqp(hs, n)
            if res is not None:
                s2 = exact_score(res)
                if s2 < gbs - 1e-12:
                    gbs, gbh, gbn = s2, res.copy(), n
                if s2 < lb_score - 1e-12:
                    lb, lb_score = res.copy(), s2
        level_best = lb
        level_best_n = n

    final_h = project_box_mass(gbh, gbn)
    if is_feasible(final_h):
        final_s = exact_score(final_h)
    else:
        final_h = np.full(45, 0.5)
        final_s = exact_score(final_h)
    return final_h, final_s, len(final_h)
