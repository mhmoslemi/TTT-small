def random_escape(h, dx, n, T, rng, n_dirs=4):
    current_score = exact_score(h, dx)
    best_h, best_score = h.copy(), current_score
    a = lag_vector(h, dx)
    accepted = False
    for _ in range(n_dirs):
        v = rng.standard_normal(n)
        v -= np.mean(v)
        v[n - 1] = -np.sum(v[:n - 1])
        vmax = np.max(np.abs(v))
        if vmax < 1e-15:
            continue
        v /= vmax
        interval = feasible_alpha_interval(h, v)
        if interval is None:
            continue
        lo, hi = interval
        b = dx * (np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full'))
        qvec = -dx * np.correlate(v, v, mode='full')
        alpha, predicted = line_search(a, b, qvec, lo, hi)
        if predicted > current_score - 1e-12:
            continue
        trial = capped_simplex_project(h + alpha * v, T)
        s = exact_score(trial, dx)
        if s < current_score - 1e-12:
            h, current_score = trial, s
            a = lag_vector(h, dx)
            accepted = True
            if s < best_score:
                best_score, best_h = s, h.copy()
    return best_h, best_score, accepted

def make_starts(fallback, n, T, rng):
    starts = [fallback.copy(), fallback[::-1].copy(), (1.0 - fallback).copy()]
    if n > 1:
        starts.append(np.roll(fallback, max(1, n // 3)).copy())
    for sigma in [0.04, 0.12]:
        starts.append(capped_simplex_project(fallback + sigma * rng.standard_normal(n), T))
    starts.append(capped_simplex_project(rng.random(n), T))
    perm = rng.permutation(n)
    vb = np.zeros(n)
    ft = int(np.floor(T))
    vb[perm[:ft]] = 1.0
    frac = T - ft
    if frac > 1e-10 and ft < n:
        vb[perm[ft]] = frac
    starts.append(capped_simplex_project(vb, T))
    if n >= 2:
        phase = rng.uniform(0, 2 * np.pi)
        freq = int(rng.integers(1, max(2, min(4, n // 2) + 1)))
        t = np.arange(n)
        starts.append(capped_simplex_project(0.5 + 0.45 * np.cos(2 * np.pi * freq * (t + 0.5) / n + phase), T))
    unique = [starts[0]]
    for s in starts[1:]:
        if not any(np.max(np.abs(s - u)) < 1e-12 for u in unique):
            unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try:
        seed_val = int(seed) % (2 ** 32)
    except (ValueError, TypeError):
        seed_val = 42
    rng = np.random.default_rng(seed_val)
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, ValueError, TypeError):
        pass
    if parent is not None and parent.ndim == 1 and len(parent) > 0 and np.isfinite(parent).all():
        n = len(parent)
        dx = 2.0 / n
        T = n / 2.0
        fallback = capped_simplex_project(parent, T)
    else:
        n = 47
        dx = 2.0 / n
        T = n / 2.0
        fallback = np.full(n, 0.5)
    if n == 1:
        return (fallback, exact_score(fallback, dx), n)
    fallback_score = exact_score(fallback, dx)
    global_best_h, global_best_score = fallback.copy(), fallback_score
    do_pairs = (n <= 64)
    starts = make_starts(fallback, n, T, rng)
    for s in starts:
        sc = exact_score(s, dx)
        if sc < global_best_score:
            global_best_score, global_best_h = sc, s.copy()
    for s in starts:
        h = s.copy()
        if do_pairs:
            h, sc = pair_descent(h, dx, n, T, rng, n_sweeps=4)
            if sc < global_best_score:
                global_best_score, global_best_h = sc, h.copy()
            h, sc, acc = random_escape(h, dx, n, T, rng, n_dirs=4)
            if sc < global_best_score:
                global_best_score, global_best_h = sc, h.copy()
            if acc:
                h, sc = pair_descent(h, dx, n, T, rng, n_sweeps=2)
                if sc < global_best_score:
                    global_best_score, global_best_h = sc, h.copy()
    for sigma in [0.015, 0.05, 0.15]:
        h = capped_simplex_project(global_best_h + sigma * rng.standard_normal(n), T)
        if do_pairs:
            h, sc = pair_descent(h, dx, n, T, rng, n_sweeps=2)
        else:
            sc = exact_score(h, dx)
        h, sc, acc = random_escape(h, dx, n, T, rng, n_dirs=3)
        if acc and do_pairs:
            h, sc = pair_descent(h, dx, n, T, rng, n_sweeps=1)
        if sc < global_best_score:
            global_best_score, global_best_h = sc, h.copy()
    final_h = capped_simplex_project(global_best_h, T)
    final_score = exact_score(final_h, dx)
    if final_score < fallback_score:
        return (final_h, final_score, n)
    return (fallback, fallback_score, n)
