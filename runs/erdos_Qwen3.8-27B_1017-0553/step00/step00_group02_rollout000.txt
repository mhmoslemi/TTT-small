                total += h_src[j] * ov
        y[k] = total * n_target / 2.0
    return project_box_mass(y, n_target)

def lags_and_jacobian(h):
    n = len(h)
    q = np.zeros(2 * n - 1, dtype=np.float64)
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    for d in range(-(n - 1), n):
        di = d + (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_hi < i_lo:
            continue
        i = np.arange(i_lo, i_hi + 1)
        j = i + d
        if d == 0:
            q[di] = np.sum(h[i] * (1.0 - h[i]))
            G[di, i] = 1.0 - 2.0 * h[i]
        else:
            q[di] = np.sum(h[j] * (1.0 - h[i]))
            G[di, j] += (1.0 - h[i])
            G[di, i] += (-h[j])
    return q, G

def slsqp_obj(z, n):
    return z[n]

def slsqp_obj_jac(z, n):
    g = np.zeros(n + 1, dtype=np.float64)
    g[n] = 1.0
    return g

def slsqp_ineq(z, n):
    h = z[:n]
    q, _ = lags_and_jacobian(h)
    return z[n] - q

def slsqp_ineq_jac(z, n):
    h = z[:n]
    _, G = lags_and_jacobian(h)
    J = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def slsqp_eq(z, n):
    return np.array([z[:n].sum() - n / 2.0])

def slsqp_eq_jac(z, n):
    J = np.zeros((1, n + 1), dtype=np.float64)
    J[0, :n] = 1.0
    return J

def run(seed=42, **kwargs):
    parent = None
    try:
        gp = globals().get('initial_h_values', None)
        if gp is not None:
            gp = np.asarray(gp, dtype=np.float64)
            if gp.ndim == 1 and len(gp) > 0 and np.all(np.isfinite(gp)):
                parent = project_box_mass(gp, len(gp))
    except Exception:
        parent = None
    if parent is None:
        n0 = 45
        parent = np.full(n0, 0.5, dtype=np.float64)
    else:
        n0 = len(parent)

    best_score = evaluate_score(parent)
    best_h = parent.copy()
    best_n = n0

    if n0 <= 180:
        res_set = sorted(set([n0, min(2 * n0, 180), min(4 * n0, 180)]))
    else:
        res_set = sorted(set([n0, 180]))
    res_set = [r for r in res_set if r >= 2]

    rng = np.random.default_rng(seed)
    h_current = parent.copy()

    for level, n in enumerate(res_set):
        if level == 0:
            base = h_current.copy()
        else:
            base = area_transfer(h_current, n)

        starts = [base, base[::-1].copy(), 1.0 - base]
        starts.append(project_box_mass(base + 0.04 * rng.standard_normal(n), n))
        starts.append(project_box_mass(base + 0.15 * rng.standard_normal(n), n))

        level_best = base.copy()
        level_best_score = evaluate_score(level_best)
        if level_best_score < best_score - 1e-12:
            best_score = level_best_score
            best_h = level_best.copy()
            best_n = n

        for start in starts:
            start = project_box_mass(start, n)
            ss = evaluate_score(start)
            if ss < level_best_score - 1e-12:
                level_best_score = ss
                level_best = start.copy()
            if ss < best_score - 1e-12:
                best_score = ss
                best_h = start.copy()
                best_n = n

            q_init, _ = lags_and_jacobian(start)
            t0 = max(float(np.max(q_init)), 0.0) + 1e-7 * max(1.0, float(n))
            z0 = np.concatenate([start, [t0]])
            bounds = [(0.0, 1.0)] * n + [(0.0, None)]
            cons = [
                {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac, 'args': (n,)},
                {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)}
            ]
            try:
                res = minimize(slsqp_obj, z0, method='SLSQP',
                               jac=slsqp_obj_jac, args=(n,),
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-9, 'disp': False})
                if res is not None and hasattr(res, 'x') and res.x is not None:
                    hr = np.asarray(res.x[:n], dtype=np.float64)
                    if np.all(np.isfinite(hr)):
                        hr = project_box_mass(hr, n)
                        if is_feasible(hr):
                            rs = evaluate_score(hr)
                            if rs < level_best_score - 1e-12:
                                level_best_score = rs
                                level_best = hr.copy()
                            if rs < best_score - 1e-12:
                                best_score = rs
                                best_h = hr.copy()
                                best_n = n
            except Exception:
                pass

        h_current = level_best.copy()

    best_h = project_box_mass(best_h, best_n)
    if not is_feasible(best_h):
        best_h = parent.copy()
        best_n = len(parent)
    final_score = evaluate_score(best_h)
    return best_h.astype(np.float64), final_score, int(best_n)
