def q_and_jacobian(h):
    n = len(h)
    n_lags = 2 * n - 1
    q = np.zeros(n_lags)
    G = np.zeros((n_lags, n))
    for d in range(-(n-1), n):
        idx = d + (n - 1)  # 0-based lag index
        i_start = max(0, -d)
        i_end = min(n - 1, n - 1 - d)
        if i_start > i_end:
            continue
        i_idx = np.arange(i_start, i_end + 1)
        j_idx = i_idx + d
        if d == 0:
            q[idx] = np.sum(h[i_idx] * (1.0 - h[i_idx]))
            G[idx, i_idx] = 1.0 - 2.0 * h[i_idx]
        else:
            q[idx] = np.sum(h[j_idx] * (1.0 - h[i_idx]))
            G[idx, j_idx] += (1.0 - h[i_idx])
            G[idx, i_idx] += 0 - h[j_idx]
    return q, G

        hi_k = 2.0 * (k + 1) / m
        for j in range(a):
            lo_j = 2.0 * j / a
            hi_j = 2.0 * (j + 1) / a
            ov = min(hi_k, hi_j) - max(lo_k, lo_j)
            if ov > 0:
                y[k] += h_src[j] * ov * m / 2.0
    return y


def q_jacobian(h):
    n = len(h)
    nl = 2 * n - 1
    q = np.zeros(nl)
    G = np.zeros((nl, n))
    for d in range(-(n - 1), n):
        idx = d + n - 1
        i0 = max(0, -d)
        i1 = min(n - 1, n - 1 - d)
        ii = np.arange(i0, i1 + 1)
        jj = ii + d
        q[idx] = np.dot(h[jj], 1.0 - h[ii])
        if d == 0:
            G[idx, ii] = 1.0 - 2.0 * h[ii]
        else:
            G[idx, jj] += 1.0 - h[ii]
            G[idx, ii] -= h[jj]
    return q, G


def _obj(z, n):
    return z[n]


def _obj_jac(z, n):
    v = np.zeros(n + 1)
    v[n] = 1.0
    return v


def _ineq(z, n):
    h = z[:n]
    q, _ = q_jacobian(h)
    return z[n] - q


def _ineq_jac(z, n):
    h = z[:n]
    _, G = q_jacobian(h)
    nl = 2 * n - 1
    J = np.zeros((nl, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J


def _eq(z, n):
    return np.sum(z[:n]) - n / 2.0


def _eq_jac(z, n):
    v = np.zeros(n + 1)
    v[:n] = 1.0
    return v

        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac, 'args': (n,)},

        pass

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def is_feasible(h, tol=1e-9):
    if h.ndim != 1 or len(h) == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(np.sum(h) - len(h) / 2.0) > tol:
        return False
    return True


def waterfill_project(y, n):
    if y.shape[0] != n or not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    y = np.clip(y, -1e4, 1e4).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))

        else:
