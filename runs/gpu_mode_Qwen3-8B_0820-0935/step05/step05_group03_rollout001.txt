import torch
import triton
import triton.language as tl
import math
import sys
sys.setrecursionlimit(1000000)

def custom_kernel(data):
    """
    Implements the "outgoing" version of the TriMul operator using Triton for acceleration.

    The kernel computes a tensor of shape [bs, seq_len, seq_len, dim] through a series
    of linear projections, gating, and pairwise matrix multiplications, with masking applied.

    The implementation uses a Triton kernel to perform the expensive pairwise multiplication
    efficiently, achieving significant speedup compared to the pure PyTorch version.

    Parameters:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [bs, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [bs, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # Prepare the weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Prepare the input
    x = input_tensor.to(torch.float32)
    x = x.contiguous()

    # Apply LayerNorm
    x = x * norm_weight + norm_bias  # x = (x - mean) * weight + bias
    x = x.float()

    # Compute left and right projections
    left = x @ left_proj_weight.t()
    right = x @ right_proj_weight.t()

    # Handle mask
    mask = mask.unsqueeze(-1).to(torch.float32).contiguous()
    left = left * mask
    right = right * mask

    # Compute gates
    left_gate = (x @ left_gate_weight.t()).sigmoid()
    right_gate = (x @ right_gate_weight.t()).sigmoid()
    out_gate = (x @ out_gate_weight.t()).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Now we need to compute the TriMul operation: out = einsum('...i k d, ...j k d -> ...i j d', left, right)
    # This is a batched N x N x D tensor product for each batch

    # For the Triton kernel, we will process the batch, i-j pair, and k dimension
    # We'll split the N dimension into blocks of size BLOCK
    # The kernel will compute the output for each i, j pair and accumulate

    # Output tensor shape: [bs, seq_len, seq_len, dim]
    output = torch.zeros((bs, seq_len, seq_len, dim), device=x.device, dtype=torch.float32)

    # Triton kernel definition
    @triton.jit
    def tritonal_kernel(
        left_ptr, right_ptr, output_ptr,
        bs, seq_len, dim,
        BLOCK: tl.constexpr
    ):
        # Compute the position in the batch
        batch_idx = tl.program_id(0)
        batch_offsets = batch_idx * seq_len * seq_len * dim
        
        # Compute the position in the i-j grid
        pid = tl.program_id(1)
        num_pid = tl.cdiv(seq_len * seq_len, BLOCK)  # Each tile processes BLOCK elements
        i, j = divmod(pid, seq_len), pid % seq_len
        
        # Compute the offset in the i-j grid
        i_offsets = i * seq_len + tl.arange(0, seq_len)
        j_offsets = j * seq_len + tl.arange(0, seq_len)
        
        # Compute the offset in the k dimension
        k_offsets = tl.arange(0, seq_len)
        
        # Load the left and right tensors
        left = tl.load(left_ptr + batch_offsets + i_offsets[:, None] * seq_len * dim + k_offsets[None, :], 
                       mask=(i_offsets < seq_len) & (k_offsets < seq_len), 
                       other=0.0)
        right = tl.load(right_ptr + batch_offsets + j_offsets[:, None] * seq_len * dim + k_offsets[None, :], 
                        mask=(j_offsets < seq_len) & (k_offsets < seq_len), 
                        other=0.0)

        # Compute the product and accumulate into output
        out = tl.sum(left * right, axis=1)
        tl.store(output_ptr + batch_offsets + i_offsets[:, None] * seq_len * dim + j_offsets[None, :], 
                 out, mask=(i_offsets < seq_len) & (j_offsets < seq_len))

    # Launch the kernel
    # We will launch it in a nested grid where:
    # - The first dimension (0) corresponds to the batch
    # - The second dimension (1) corresponds to the i-j pairs
    # The kernel uses BLOCK to process each tile of the i-j grid
    # The kernel is run for all i,j pairs, and for each, the k dimension is processed
    # The kernel is run for all batches
    # The kernel uses the same layout as the input

    # We'll launch the kernel with a block size of 128 (this can be tuned for speed)
    BLOCK = 128
    num_blocks = seq_len * seq_len
    grid = (bs, num_blocks, 1)
    tritonal_kernel[grid](
        left, right, output,
        bs, seq_len, dim,
        BLOCK=BLOCK
    )

    # Apply the output gate and LayerNorm
    output = output * out_gate  # Apply the output gate
    output = output @ to_out_weight.t()  # Final projection
    output = output * to_out_norm_weight + to_out_norm_bias  # Apply LayerNorm

    return output
