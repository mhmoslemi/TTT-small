import torch
import triton
import triton.language as tl

def custom_kernel(data: torch.Tensor):
    """
    Implements the outgoing version of the TriMul operator using a Triton kernel.
    This kernel performs the forward pass, including projection, masking, gating,
    and the pairwise matrix multiplication using einsum-like operations. The output
    is computed in float32 with a Triton kernel for performance optimization.
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure all data is on the correct device
    input_tensor = input_tensor.to('cuda')
    mask = mask.to('cuda')
    
    # Load weights to device
    norm_weight = norm_weight.to('cuda')
    norm_bias = norm_bias.to('cuda')
    left_proj_weight = left_proj_weight.to('cuda')
    right_proj_weight = right_proj_weight.to('cuda')
    left_gate_weight = left_gate_weight.to('cuda')
    right_gate_weight = right_gate_weight.to('cuda')
    out_gate_weight = out_gate_weight.to('cuda')
    to_out_norm_weight = to_out_norm_weight.to('cuda')
    to_out_norm_bias = to_out_norm_bias.to('cuda')
    to_out_weight = to_out_weight.to('cuda')

    # Allocate output tensor
    output = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, dtype=torch.float32, device='cuda')

    # Step 1: LayerNorm on input
    input_norm = input_tensor
    mean = torch.sum(input_norm * norm_weight, dim=[1, 2, 3], keepdim=True) / seq_len**2
    var = torch.sum((input_norm - mean) ** 2 * norm_weight, dim=[1, 2, 3], keepdim=True) / seq_len**2
    input_norm = (input_norm - mean) * torch.rsqrt(var + 1e-5) * norm_weight + norm_bias

    # Step 2: Linear projections
    left = torch.matmul(input_norm, left_proj_weight.t())
    right = torch.matmul(input_norm, right_proj_weight.t())

    # Step 3: Apply mask (mask is [bs, seq_len, seq_len])
    mask = mask.unsqueeze(-1).to(torch.float32)
    left = left * mask
    right = right * mask

    # Step 4: Sigmoid gates (left_gate, right_gate, out_gate)
    left_gate = torch.matmul(input_norm, left_gate_weight.t()).sigmoid()
    right_gate = torch.matmul(input_norm, right_gate_weight.t()).sigmoid()
    out_gate = torch.matmul(input_norm, out_gate_weight.t()).sigmoid()

    # Step 5: Apply gates
    left = left * left_gate
    right = right * right_gate

    # Step 6: Perform einsum-like operation, which is (i, k, d) and (j, k, d) -> (i, j, d)
    # We implement this using a Triton kernel to optimize for performance
    # This is the core of the TriMul operation and is the most expensive part

    # Allocate memory for intermediate output and result
    intermediate = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, dtype=torch.float32, device='cuda')
    result = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, dtype=torch.float32, device='cuda')

    # Launch the Triton kernel for the matrix multiplication
    # We use tiling to handle large N efficiently
    # This kernel performs the matrix multiplication of left and right
    # with mask applied and using float32 precision

    # Define the Triton kernel for the matrix multiplication
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, output_ptr,
        seq_len, batch_size, hidden_dim,
        BLOCK: tl.constexpr,
        num_warps: tl.constexpr
    ):
        # Each thread block processes a block of the output matrix
        pid = tl.program_id(0)
        grid_size = triton.cdiv(seq_len * seq_len, BLOCK)
        grid_idx = pid * BLOCK

        # Each thread computes a single output element
        i = tl.arange(0, BLOCK)
        j = tl.arange(0, BLOCK)

        # Compute the indices
        # i is row index, j is column index
        i = i + grid_idx
        j = j + grid_idx

        # Check if indices are within bounds
        valid = (i < seq_len) & (j < seq_len)
        i = i * valid
        j = j * valid

        # Initialize accumulator
        acc = tl.zeros((BLOCK, BLOCK), dtype=tl.float32)

        # Iterate over the sequence dimension and compute the output
        for k in range(seq_len):
            # Load left and right values
            left_k = tl.load(left_ptr + i * seq_len + k * seq_len, mask=valid, other=0.0)
            right_k = tl.load(right_ptr + j * seq_len + k * seq_len, mask=valid, other=0.0)

            # Perform the matrix multiplication for this k
            # Compute (i, k) * (k, j) = i, j
            acc += tl.dot(left_k, right_k, axis=1)

        # Store the result
        tl.store(output_ptr + i * seq_len + j, acc, mask=valid)

    # Launch the kernel
    BLOCK = 128
    num_warps = 4
    grid = (triton.cdiv(seq_len * seq_len, BLOCK),)
    tri_mul_kernel[grid](left, right, intermediate, seq_len, batch_size, hidden_dim, BLOCK=BLOCK, num_warps=num_warps)

    # Step 7: Apply the output gate
    intermediate = intermediate * out_gate

    # Step 8: Normalize intermediate with LayerNorm
    mean = torch.sum(intermediate * to_out_norm_weight, dim=[1, 2, 3], keepdim=True) / seq_len**2
    var = torch.sum((intermediate - mean) ** 2 * to_out_norm_weight, dim=[1, 2, 3], keepdim=True) / seq_len**2
    intermediate = (intermediate - mean) * torch.rsqrt(var + 1e-5) * to_out_norm_weight + to_out_norm_bias

    # Step 9: Final linear projection
    output = torch.matmul(intermediate, to_out_weight.t())

    return output
