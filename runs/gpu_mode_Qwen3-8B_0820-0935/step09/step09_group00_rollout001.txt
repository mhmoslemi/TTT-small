import torch
import triton
import triton.language as tl
from dataclasses import dataclass

@triton.jit
def triangle_mul_kernel(
    input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr,
    to_out_norm_ptr, to_out_ptr, out_ptr,
    batch_size, seq_len, dim, hidden_dim,
    num_blocks, BLOCK: tl.constexpr,
    has_mask: tl.constexpr
):
    # Get program ID
    pid = tl.program_id(0)

    # Compute the block index for the batch dimension
    b_idx = pid // (seq_len * seq_len)
    # Compute the block index for the (i, j) pair in the sequence dimension
    i_j_idx = pid % (seq_len * seq_len)
    # Compute (i, j) from i_j_idx
    i = i_j_idx // seq_len
    j = i_j_idx % seq_len

    # Compute the offset for current (i, j)
    offs = tl.arange(0, BLOCK)
    m_offsets = tl.arange(0, seq_len)
    n_offsets = tl.arange(0, seq_len)
    k_offsets = tl.arange(0, seq_len)

    # Compute the positions to load from the input, mask, and weights
    # We'll compute for all k in [0, seq_len) where mask[i, k] and mask[j, k] are 1

    # We will load the left and right projections, apply gates, and then perform the einsum

    # Left gate weight
    left_gate = tl.load(left_gate_ptr + (offs[:, None] * hidden_dim + tl.arange(0, hidden_dim)), None)
    left_gate = left_gate.sigmoid()

    # Right gate weight
    right_gate = tl.load(right_gate_ptr + (offs[:, None] * hidden_dim + tl.arange(0, hidden_dim)), None)
    right_gate = right_gate.sigmoid()

    # Out gate weight
    out_gate = tl.load(out_gate_ptr + (offs[:, None] * hidden_dim + tl.arange(0, hidden_dim)), None)
    out_gate = out_gate.sigmoid()

    # Left projection weight
    left_proj = tl.load(left_proj_ptr + (offs[:, None] * hidden_dim + tl.arange(0, hidden_dim)), None)

    # Right projection weight
    right_proj = tl.load(right_proj_ptr + (offs[:, None] * hidden_dim + tl.arange(0, hidden_dim)), None)

    # Compute left * left_gate
    left = tl.load(input_ptr + (b_idx * seq_len * seq_len * dim + i * seq_len * dim + offs[:, None] * dim), None)
    left *= left_gate

    # Compute right * right_gate
    right = tl.load(input_ptr + (b_idx * seq_len * seq_len * dim + j * seq_len * dim + offs[:, None] * dim), None)
    right *= right_gate

    # Multiply and sum all k
    # Perform outer product and sum over k
    out = tl.dot(left, right)  # [BLOCK, 1]

    # Apply to_out_norm
    to_out_norm = tl.load(to_out_norm_ptr + (offs[:, None] * hidden_dim + tl.arange(0, hidden_dim)), None)
    out = out * to_out_norm  # [BLOCK, 1]

    # Apply out_gate
    out *= out_gate  # [BLOCK, 1]

    # Multiply with to_out weight
    to_out = tl.load(to_out_ptr + (offs[:, None] * dim + tl.arange(0, dim)), None)
    out = out * to_out  # [BLOCK, 1]

    # Write the result to the output tensor
    tl.store(out_ptr + (b_idx * seq_len * seq_len * dim + i * seq_len * dim + j * dim + offs), out, mask=offs < seq_len)

def custom_kernel(data):

    N = input_tensor.shape[1]
    C = input_tensor.shape[3]

    # Compute the required output shape
    output_shape = (B, N, N, C)

    # Initialize output tensor
    output = torch.empty(output_shape, dtype=torch.float32, device=input_tensor.device)

    # Access weights directly
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Prepare parameters for the kernel
    num_blocks = triton.cdiv(N * N, 128)
    num_threads = 128

    # Launch the Triton kernel
    grid = (num_blocks, )
    triangle_mul_kernel[grid](
        input_tensor,
        mask,
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
        to_out_norm_weight,
