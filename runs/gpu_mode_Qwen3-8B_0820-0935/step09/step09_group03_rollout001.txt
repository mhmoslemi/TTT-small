import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Optimized Triton implementation of the TriMul operator from AlphaFold3, Chai, 
    and Protenix. This kernel computes the outgoing version of TriMul, which involves:
    - Applying LayerNorm on input
    - Projection and gate operations on both left and right
    - Masking and gate multiplications
    - A matrix multiplication over the sequence dimension (i, k) and (j, k), which is the 
      core expensive operation.
    - Finally, normalization and output projection.

    The implementation is optimized for large sequences (N) and is designed to run efficiently
    on NVIDIA L40S (Ada Lovelace architecture).
    """

    input_tensor, mask, weights, config = data
    batch_size, seqlen, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Define constants
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    NUM_THREADS = 128

    # Ensure the input and mask are on the correct device and dtype
    input_tensor = input_tensor.to(torch.float32).contiguous()
    mask = mask.to(torch.float32).contiguous()

    # Access weights from the provided dictionary
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Apply LayerNorm - this is done in the forward pass, not the kernel
    norm = torch.nn.LayerNorm(dim).to(input_tensor.device)
    x = norm(input_tensor)

    # Projection operations
    left = torch.nn.functional.linear(x, left_proj_weight)  # Shape: [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask
    mask = mask.unsqueeze(-1)  # Shape: [B, N, N, 1]
    left = left * mask
    right = right * mask

    # Apply gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply gate multiplications
    left = left * left_gate
    right = right * right_gate

    # Matrix multiplication across k dimension
    # Result: [B, N, N, hidden_dim]
    # This is the most expensive part and is implemented in the Triton kernel

    # Prepare output buffer
    output = torch.zeros((batch_size, seqlen, seqlen, hidden_dim), device=input_tensor.device)

    # Kernel to compute the inner product over k dimension: i, k and j, k
    @triton.jit
    def matmul_kernel(
        left_ptr, right_ptr, out_ptr,
        batch_size, seqlen,
        hidden_dim,
        BLOCK_M: tl.constexpr,
        BLOCK_N: tl.constexpr,
        BLOCK_K: tl.constexpr,
    ):
        # We index the tensor using (batch, i, j, k)
        # Use three loops: batch, i, j

        # Compute the grid dimensions
        grid_size = triton.cdiv(seqlen, BLOCK_M) * triton.cdiv(seqlen, BLOCK_N)
        pid = tl.program_id(0)

        # Compute batch and i, j indices
        batch_id = pid // (triton.cdiv(seqlen, BLOCK_N))
        i_idx = (pid // triton.cdiv(seqlen, BLOCK_N)) % triton.cdiv(seqlen, BLOCK_N)
        j_idx = pid % triton.cdiv(seqlen, BLOCK_N)

        # Offset for the current batch and block
        batch_off = batch_id * seqlen * seqlen
        i_off = i_idx * seqlen
        j_off = j_idx * seqlen

        # Offsets for the thread block
        offs_m = tl.arange(0, BLOCK_M)
        offs_n = tl.arange(0, BLOCK_N)
        offs_k = tl.arange(0, BLOCK_K)

        # Use k as axis for accumulation
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k in range(0, seqlen, BLOCK_K):
            # Load left and right blocks
            # Left: [batch, i, k, hidden_dim] -> [i, k, hidden_dim]
            left_block = tl.load(left_ptr + batch_off + i_off + k * seqlen + offs_k, mask=offs_k < seqlen, other=0.0)
            left_block = left_block.reshape(BLOCK_K, hidden_dim)  # shape [BLOCK_K, hidden_dim]

            # Right: [batch, j, k, hidden_dim] -> [j, k, hidden_dim]
            right_block = tl.load(right_ptr + batch_off + j_off + k * seqlen + offs_k, mask=offs_k < seqlen, other=0.0)
            right_block = right_block.reshape(BLOCK_K, hidden_dim)  # shape [BLOCK_K, hidden_dim]

            for m in range(BLOCK_M):
                for n in range(BLOCK_N):
                    # Dot product of left and right blocks
                    acc[m][n] += tl.dot(left_block[offs_k][...], right_block[offs_k][...])

        # Store the result
        tl.store(out_ptr + batch_off + i_off + offs_m + j_off + offs_n, acc, mask=offs_m < BLOCK_M, offs_n < BLOCK_N)

    # Launch the kernel
    # Convert input pointers
    left_ptr = left.data_ptr()
    right_ptr = right.data_ptr()
    out_ptr = output.data_ptr()

    # Compute grid size
    grid_size = triton.cdiv(seqlen, BLOCK_M) * triton.cdiv(seqlen, BLOCK_N)
    num_blocks = triton.cdiv(grid_size, NUM_THREADS)

    # Launch the kernel
    matmul_kernel[(num_blocks,)](left_ptr, right_ptr, out_ptr, batch_size, seqlen, hidden_dim, BLOCK_M, BLOCK_N, BLOCK_K)

    # Apply final normalization and gate
    out = torch.nn.functional.linear(output, to_out_weight, to_out_norm_bias)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), to_out_norm_weight)

    # Apply the output gate
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    out = out * out_gate

    # Convert to float32 for final output
    return out.to(torch.float32)
