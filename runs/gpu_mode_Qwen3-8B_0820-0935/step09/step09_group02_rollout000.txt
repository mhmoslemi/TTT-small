import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]):
    """
    Implements the "outgoing" version of the TriMul operator using a Triton kernel.
    This kernel computes the TriMul forward pass efficiently for the given input.

    The TriMul operation is defined as:
    out[i,j] = sum_k (left[i,k] * right[j,k])
    with appropriate application of masks and gates.

    The kernel is designed to handle large sequence lengths N in a memory-efficient and
    compute-optimized manner using tiling and fusion strategies suitable for an L40S GPU.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    seq_len = input_tensor.shape[1]
    batch_size = input_tensor.shape[0]

    # Access the given weights of the model
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --- Precompute inputs (in PyTorch for simplicity) ---
    x = input_tensor
    batch_size, seq_len, _, _ = x.shape

    # Normalize the input tensor using LayerNorm
    x = x.to(torch.float32)
    x_norm = x * norm_weight + norm_bias

    # Project input for left and right
    left_proj = x_norm @ left_proj_weight.t()
    right_proj = x_norm @ right_proj_weight.t()

    # Apply mask to left and right
    mask = mask.unsqueeze(-1).to(torch.float32)
    left_proj[mask == 0] = 0
    right_proj[mask == 0] = 0

    # Compute left and right gates
    left_gate = x_norm @ left_gate_weight.t().sigmoid()
    right_gate = x_norm @ right_gate_weight.t().sigmoid()
    out_gate = x_norm @ out_gate_weight.t().sigmoid()

    # Apply gates
    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate

    # --- Triton Kernel for the TriMul operation ---
    # We'll implement the core TriMul operation (sum over k) using Triton
    # This kernel computes out[i,j] = sum_k (left[i,k] * right[j,k])

    # Define constants for tiling
    BLOCK = 128  # Block size for sequence dimension
    GROUP = 16   # Grouping for parallelism in the sequence dimension

    @triton.jit
    def triton_trimul_kernel(
        # Input pointers
        left_ptr, right_ptr, mask_ptr, 
        # Output pointer
        out_ptr,
        # Dimensions
        batch_size: tl.constexpr, seq_len: tl.constexpr, hidden_dim: tl.constexpr,
        # Tiling parameters
        BLOCK: tl.constexpr, GROUP: tl.constexpr
    ):
        # Compute the thread ID
        pid = tl.program_id(0)
        pid_m = pid // (seq_len // GROUP)  # Batch and sequence dimension
        pid_n = pid % (seq_len // GROUP)

        # Compute the offset in the batch (0-based)
        batch_idx = pid_m // (seq_len // GROUP)
        # Compute the offset in the sequence (0-based)
        seq_idx = pid_m % (seq_len // GROUP)

        # Compute the block offset in the sequence
        off_m = pid_n * GROUP + tl.arange(0, GROUP)
        off_n = seq_idx * GROUP + tl.arange(0, GROUP)

        # Offset for the left and right tensors
        left_offs = (batch_idx * seq_len + off_m[:, None]) * hidden_dim + off_n[None, :]
        right_offs = (batch_idx * seq_len + off_n[:, None]) * hidden_dim + off_m[None, :]

        # Load left and right tensors into registers or shared memory
        # We'll tile over both i and j to reduce memory access
        left = tl.load(left_ptr + left_offs, mask=off_m < seq_len, other=0.0)
        right = tl.load(right_ptr + right_offs, mask=off_n < seq_len, other=0.0)

        # Compute the result: out[i,j] = sum_k (left[i,k] * right[j,k])
        # We need to compute all combinations of i and j
        out = tl.dot(left, right, reduce=True)  # Reduces over k

        # Apply the mask (0.0 where mask is 0)
        mask = tl.load(mask_ptr + (batch_idx * seq_len + off_m[:, None]) * seq_len + off_n[None, :], other=1.0)
        out = out * mask

        # Store the result
        tl.store(out_ptr + (batch_idx * seq_len + off_m[:, None]) * seq_len + off_n[None, :], out)

    # Allocate output tensor
    output = torch.zeros((batch_size, seq_len, seq_len, hidden_dim), 
                         device=input_tensor.device, dtype=torch.float32)

    # Launch the Triton kernel
    grid = (triton.cdiv(seq_len, GROUP), )
    triton_trimul_kernel[grid](
        left_proj,
        right_proj,
        mask,
        output,
        batch_size, seq_len, hidden_dim,
        BLOCK=BLOCK, GROUP=GROUP
    )

    # Apply the output gate
    output = output * out_gate

    # Normalize the output using LayerNorm
    output_norm = output * to_out_norm_weight + to_out_norm_bias

    # Final linear projection
    output = output_norm @ to_out_weight.t()

    return output
