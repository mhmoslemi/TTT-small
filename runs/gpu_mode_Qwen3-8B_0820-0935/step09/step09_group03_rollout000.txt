import triton
import triton.language as tl
import torch

def custom_kernel(data):
    """
    Implements the outgoing TriMul operation (as used in AlphaFold3 and similar models) using Triton.
    Key steps:
    1. Apply LayerNorm to input.
    2. Project and gate input using weights and sigmoid.
    3. Perform pairwise multiplication and sum over sequence dimension (N).
    4. Apply final normalization and output projection.

    The output is returned in float32.
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Convert input tensor to float32 (if needed)
    input_tensor = input_tensor.to(torch.float32)

    # Apply LayerNorm (weights are provided)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # LayerNorm: x = (x - mean) / sqrt(var + eps) * gamma + beta
    # To implement LayerNorm in Triton, we need to compute mean and variance across last dimension
    # Then apply the scale and shift
    # Since this is not implemented here, we'll use PyTorch's implementation directly
    norm = torch.nn.LayerNorm(dim, eps=1e-5)
    norm.weight = torch.nn.Parameter(norm_weight)
    norm.bias = torch.nn.Parameter(norm_bias)
    x = norm(input_tensor)

    # Project to hidden_dim
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left = torch.matmul(x, left_proj_weight.t())
    right = torch.matmul(x, right_proj_weight.t())

    # Apply mask to left and right projections
    mask = mask.unsqueeze(-1)  # [bs, seq_len, seq_len, 1]
    left = left * mask
    right = right * mask

    # Apply gates (sigmoid)
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.matmul(x, left_gate_weight.t()).sigmoid()
    right_gate = torch.matmul(x, right_gate_weight.t()).sigmoid()
    out_gate = torch.matmul(x, out_gate_weight.t()).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Now perform TriMul operation: out[i,j] = sum_k (left[i,k] * right[j,k])
    # This is a tensor with shape [bs, seq_len, seq_len, hidden_dim]
    # To implement this efficiently in Triton, we need to perform the sum over the k dimension

    # Convert to Triton tensor
    left = left.contiguous()
    right = right.contiguous()
    output = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, device=x.device)

    # Implement the TriMul part using a Triton kernel
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        batch_size, seq_len, hidden_dim,
        BLOCK: tl.constexpr
    ):
        # Determine the current thread's position
        pid = tl.program_id(0)
        block_id = pid // (seq_len * seq_len)
        row_id = (pid // seq_len) % seq_len
        col_id = pid % seq_len

        # Offsets for the current block
        block_offsets = tl.arange(0, BLOCK)
        offsets = block_offsets + (block_id * seq_len * seq_len + row_id * seq_len + col_id)

        # Load the values for current row and column from left and right
        left_values = tl.load(left_ptr + offsets, mask=block_offsets < seq_len, other=0.0)
        right_values = tl.load(right_ptr + offsets, mask=block_offsets < seq_len, other=0.0)

        # Perform the element-wise multiplication and accumulate
        out_values = left_values * right_values

        # Store the result
        tl.store(out_ptr + offsets, out_values, mask=block_offsets < seq_len)

    # Compute the grid size
    grid_size = triton.cdiv(seq_len, 128) * seq_len * seq_len
    grid = (grid_size,)

    # Launch the kernel
    tri_mul_kernel[grid](
        left, right, output,
        batch_size, seq_len, hidden_dim,
        BLOCK=128
    )

    # Apply the final LayerNorm and output projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    to_out_norm = torch.nn.LayerNorm(hidden_dim, eps=1e-5)
    to_out_norm.weight = torch.nn.Parameter(to_out_norm_weight)
    to_out_norm.bias = torch.nn.Parameter(to_out_norm_bias)
    output = to_out_norm(output)

    # Apply the output gate
    output = output * out_gate

    # Final projection
    output = torch.matmul(output, to_out_weight.t())

    return output.to(torch.float32)
