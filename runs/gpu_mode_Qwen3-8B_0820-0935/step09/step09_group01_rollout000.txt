import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # Convert mask to a 3D tensor [bs, seq_len, seq_len], as it was previously unsqueezed
    mask = mask.unsqueeze(-1).bool()
    mask = mask.view(batch_size, seq_len, seq_len)

    # Triton kernel to perform the forward computation for all elements
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        mask_ptr, norm_weight, norm_bias, left_proj_weight, right_proj_weight,
        left_gate_weight, right_gate_weight, out_gate_weight,
        to_out_norm_weight, to_out_norm_bias, to_out_weight,
        dim, hidden_dim, seq_len, batch_size,
        BLOCK: tl.constexpr, MAX_SEQ_LEN: tl.constexpr,
    ):
        # Compute program ID
        pid = tl.program_id(0)
        # Compute thread ID within the block
        tid = tl.program_id(1)
        # Compute offset from thread ID
        off_n = tid
        # Compute the block offset for each batch
        block_batch = pid // (seq_len)
        block_i = (pid // seq_len) % seq_len
        block_j = pid % seq_len

        # Compute indices for the current block
        idx_b = block_batch
        idx_i = block_i
        idx_j = block_j

        # Compute mask and scale it
        mask_val = tl.load(mask_ptr + idx_b * seq_len * seq_len + idx_i * seq_len + idx_j, mask=tl.uchar_mask())
        mask_val = tl.astype(mask_val, tl.float32)
        mask_val = tl.where(mask_val, 1.0, 0.0)

        # Compute all indices
        offsets = tl.arange(0, BLOCK)
        m_mask = (offsets < seq_len) & (offsets + idx_i < seq_len) & (offsets + idx_j < seq_len)
        m_mask = tl.where(m_mask, 1.0, 0.0)

        # Convert indices to pointer offsets
        # left: [batch, i, k, hidden]
        # right: [batch, j, k, hidden]
        # out: [batch, i, j, hidden]

        # Initialize accumulator
        acc = tl.zeros((hidden_dim,), dtype=tl.float32)

        # Iterate over k and compute the dot product
        for k in range(0, seq_len, BLOCK):
            offset_k = k

            # Compute the offset in the left and right tensors
            offset_left = idx_b * seq_len * seq_len * hidden_dim + idx_i * seq_len * hidden_dim + offset_k * hidden_dim
            offset_right = idx_b * seq_len * seq_len * hidden_dim + idx_j * seq_len * hidden_dim + offset_k * hidden_dim

            # Load left and right values
            left_vals = tl.load(left_ptr + offset_left, mask=m_mask, other=0.0)
            right_vals = tl.load(right_ptr + offset_right, mask=m_mask, other=0.0)

            # Compute element-wise product of left and right values
            product = left_vals * right_vals
            product = product * mask_val

            # Accumulate the product
            acc += tl.dot(product, product)
            # Wait for the load and compute
            tl.dot(product, product, acc, acc)

        # Apply norm
        acc = acc * norm_weight + norm_bias

        # Apply to_out_norm and to_out
        acc = tl.dot(acc, to_out_norm_weight) + to_out_norm_bias
        acc = acc * out_gate_weight
        acc = tl.dot(acc, to_out_weight)

        # Store result
        offset_out = idx_b * seq_len * seq_len * hidden_dim + idx_i * seq_len * hidden_dim + idx_j * hidden_dim
        tl.store(out_ptr + offset_out, acc)

    # Prepare input data for the kernel
    # Convert left and right tensors to a format suitable for the kernel
    # left = [batch, seq_len, seq_len, hidden]
    # right = [batch, seq_len, seq_len, hidden]

    # Assume all tensors are in float32 and aligned
    # We'll use linear indexing and block-wise processing

    # Extract weights
    norm_weight = weights["norm.weight"].to(memory_format=torch.channels_last)
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"].to(memory_format=torch.channels_last)
    right_proj_weight = weights["right_proj.weight"].to(memory_format=torch.channels_last)
    left_gate_weight = weights["left_gate.weight"].to(memory_format=torch.channels_last)
    right_gate_weight = weights["right_gate.weight"].to(memory_format=torch.channels_last)
    out_gate_weight = weights["out_gate.weight"].to(memory_format=torch.channels_last)
    to_out_norm_weight = weights["to_out_norm.weight"].to(memory_format=torch.channels_last)
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"].to(memory_format=torch.channels_last)

    # Convert input to a format suitable for the kernel
    # Apply norm: norm(x)
    input_tensor = input_tensor.to(torch.float32)
    x_norm = torch.nn.functional.layer_norm(input_tensor, input_tensor.shape[-1], norm_weight, norm_bias)

    # Project to hidden dimension
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask
    left_masked = left * mask.type_as(left)
    right_masked = right * mask.type_as(right)

    # Apply gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Multiply by gates
    left = left_masked * left_gate
    right = right_masked * right_gate

    # Compute the output using the Triton kernel
    out_size = batch_size * seq_len * seq_len * hidden_dim
    out = torch.empty(batch_size * seq_len * seq_len * hidden_dim, dtype=torch.float32, device=input_tensor.device)

    BLOCK = 128
    MAX_SEQ_LEN = seq_len
    grid = (batch_size * seq_len * seq_len, 1)  # batch * seq_len * seq_len blocks
    tri_mul_kernel[grid](
        left,
        right,
        out,
        mask,
        norm_weight,
        norm_bias,
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
        to_out_norm_weight,
        to_out_norm_bias,
        to_out_weight,
        dim,
        hidden_dim,
        seq_len,
        batch_size,
        BLOCK=BLOCK,
        MAX_SEQ_LEN=MAX_SEQ_LEN,
    )

    # Reshape output
    out = out.view(batch_size, seq_len, seq_len, hidden_dim)

    # Apply to_out_norm and out_gate
    out = torch.nn.functional.layer_norm(out, out.shape[-1], to_out_norm_weight, to_out_norm_bias)
    out = out * out_gate

    # Apply to_out
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
