import torch
import triton
import triton.language as tl
from torch import nn, einsum

def custom_kernel(data):
    """
    Implements the outgoing version of the Triangle Multiplicative Update (TriMul) operator
    using a Triton kernel. This kernel is optimized for large sequence dimensions and
    involves pairwise multiplicative updates over the sequence dimension.

    Parameters:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [bs, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [bs, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters

    Output:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Move to device as needed
    device = input_tensor.device
    if device.type == 'cuda':
        triton_device = 'cuda'
    else:
        triton_device = 'cpu'

    # Access weights
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # Convert mask to device
    mask = mask.to(device)

    # Prepare inputs
    x = input_tensor.to(torch.float32)
    x = x.to(device)

    # Normalize input
    norm_weight = norm_weight.to(device)
    norm_bias = norm_bias.to(device)
    x = x * norm_weight + norm_bias

    # Project to hidden dimension
    left = x @ left_proj_weight
    right = x @ right_proj_weight

    # Apply mask
    mask = mask.unsqueeze(-1)
    left = left * mask
    right = right * mask

    # Apply gates
    left_gate = x @ left_gate_weight
    right_gate = x @ right_gate_weight
    out_gate = x @ out_gate_weight

    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left * left_gate
    right = right * right_gate

    # Now apply the TriMul operation: out[i,j,d] = sum_k left[i,k,d] * right[j,k,d]
    # We use a Triton kernel to compute this efficiently
    # We'll split the computation into blocks to fit in shared memory

    # Convert to device
    left = left.to(device)
    right = right.to(device)

    # Prepare output
    out_shape = (bs, seq_len, seq_len, hidden_dim)
    out = torch.zeros(out_shape, device=device)

    # Triton kernel definition for the TriMul operation
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        seq_len, batch_size, hidden_dim,
        BLOCK: tl.constexpr
    ):
        # Compute the offsets for the current thread
        pid = tl.program_id(0)
        block_pid = tl.program_id(1)
        i = pid * BLOCK + tl.arange(0, BLOCK)
        j = block_pid * BLOCK + tl.arange(0, BLOCK)

        # Apply mask to i and j indices
        i_mask = i < seq_len
        j_mask = j < seq_len

        # Load left and right into registers
        left = tl.load(left_ptr + (i[:, None] * seq_len + j[None, :]) * hidden_dim, mask=i_mask[:, None] & j_mask[None, :], other=0.0)
        right = tl.load(right_ptr + (j[:, None] * seq_len + i[None, :]) * hidden_dim, mask=j_mask[:, None] & i_mask[None, :], other=0.0)

        # Compute the pairwise product
        product = left * right

        # Sum over the k dimension
        out = tl.sum(product, axis=1)

        # Store the result
        tl.store(out_ptr + (i[:, None] * seq_len + j[None, :]) * hidden_dim, out, mask=i_mask[:, None] & j_mask[None, :])

    # Launch the kernel
    BLOCK = 128
    num_blocks = triton.cdiv(seq_len, BLOCK)
    grid = (num_blocks, num_blocks)
    tri_mul_kernel[grid](
        left, right, out,
        seq_len, bs, hidden_dim,
        BLOCK=BLOCK
    )

    # Apply the output gate
    out = out * out_gate

    # Normalize output and project to the original dimension
    out = out.to(torch.float32)
    out = out * to_out_norm_weight + to_out_norm_bias
    out = out @ to_out_weight

    return out
