import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    """
    Implements the outgoing version of the TriMul operation from AlphaFold3.
    This kernel is optimized for NVIDIA L40S using Triton for performance.
    The operation is applied on a 4D tensor and involves matrix multiplications,
    masking, and activation functions. The output is in float32 format.
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access weights
    norm_weight = weights["norm.weight"].to(input_tensor.device).contiguous()
    norm_bias = weights["norm.bias"].to(input_tensor.device).contiguous()
    left_proj_weight = weights["left_proj.weight"].to(input_tensor.device).contiguous()
    right_proj_weight = weights["right_proj.weight"].to(input_tensor.device).contiguous()
    left_gate_weight = weights["left_gate.weight"].to(input_tensor.device).contiguous()
    right_gate_weight = weights["right_gate.weight"].to(input_tensor.device).contiguous()
    out_gate_weight = weights["out_gate.weight"].to(input_tensor.device).contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].to(input_tensor.device).contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].to(input_tensor.device).contiguous()
    to_out_weight = weights["to_out.weight"].to(input_tensor.device).contiguous()

    # Output tensor shape
    output = torch.zeros((bs, seq_len, seq_len, hidden_dim), device=input_tensor.device, dtype=torch.float32)

    # Step 1: Apply LayerNorm on input
    if dim != hidden_dim:
        # Apply LayerNorm on input before projections
        x_norm = torch.nn.functional.layer_norm(
            input_tensor,
            (dim,),
            norm_weight,
            norm_bias,
            True
        )
    else:
        x_norm = input_tensor

    # Step 2: Perform projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Step 3: Apply mask (optional)
    if mask is not None:
        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Step 4: Compute gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Step 5: Apply gate functions
    left = left * left_gate
    right = right * right_gate

    # Step 6: Compute TriMul (using Triton kernel)
    # We define a Triton kernel to compute the product sum over k
    # For the output tensor: out[b, i, j, d] = sum_{k} left[b, i, k, d] * right[b, j, k, d]
    # We optimize the kernel to use shared memory and block-based tiling

    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, output_ptr,
        seq_len: tl.constexpr, bs: tl.constexpr, dim: tl.constexpr,
        BLOCK: tl.constexpr,
    ):
        pid = tl.program_id(0)
        block_idx = pid // (seq_len // BLOCK)  # block for output i
        block_idx_j = pid % (seq_len // BLOCK)  # block for output j

        # Calculate the offset for the current block
        i_offsets = tl.arange(0, BLOCK) + block_idx * BLOCK
        j_offsets = tl.arange(0, BLOCK) + block_idx_j * BLOCK

        # Initialize output block
        out = tl.zeros((BLOCK, BLOCK), dtype=tl.float32)

        # Load left and right blocks
        for k in range(seq_len):
            # Load left block
            left_k = tl.load(left_ptr + (i_offsets[:, None] * seq_len + k * seq_len * dim) + (tl.arange(0, BLOCK) * dim), 
                            mask=(i_offsets < seq_len) & (tl.arange(0, BLOCK) < dim), 
                            other=0.0)
            # Load right block
            right_k = tl.load(right_ptr + (j_offsets[None, :] * seq_len + k * seq_len * dim) + (tl.arange(0, BLOCK) * dim), 
                            mask=(j_offsets < seq_len) & (tl.arange(0, BLOCK) < dim), 
                            other=0.0)

            # Compute product and accumulate
            product = tl.reshape(left_k, (BLOCK, dim)) * tl.reshape(right_k, (dim, BLOCK))
            out += tl.trans(product)

        # Store the result
        out = tl.reshape(out, (BLOCK * BLOCK,))
        tl.store(output_ptr + (i_offsets[:, None] * seq_len + j_offsets[None, :]) * dim, out)

    # Launch the kernel
    num_blocks = (seq_len + BLOCK - 1) // BLOCK
    grid = (num_blocks * num_blocks,)
    tri_mul_kernel[grid](
        left, right, output,
        seq_len, bs, dim,
        BLOCK=128
    )

    # Step 7: Apply LayerNorm and gate to final output
    output = torch.nn.functional.layer_norm(
        output,
        (hidden_dim, ),
        to_out_norm_weight,
        to_out_norm_bias,
        True
    )
    output = output * out_gate
    output = torch.nn.functional.linear(output, to_out_weight)

    # Ensure the final output is in float32
    output = output.to(torch.float32)

    return output
