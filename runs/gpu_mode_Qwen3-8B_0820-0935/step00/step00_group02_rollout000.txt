import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Implements the TriMul operation using Triton kernels for optimized performance.
    The kernel handles the outgoing version of the TriMul operation for protein structure prediction tasks.
    It performs matrix multiplications, element-wise operations, and einsum-like operations efficiently.
    
    Inputs:
        - input_tensor: [bs, seq_len, seq_len, dim] input tensor
        - mask: [bs, seq_len, seq_len] mask tensor
        - weights: Dictionary of model weights
        - config: Dictionary containing model configuration parameters
    
    Output:
        - output: [bs, seq_len, seq_len, dim] output tensor
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, N, _ = input_tensor.shape
    seq_len = N
    batch_size = B
    
    # Access and convert all weights to float32 as expected
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    to_out_weight = weights["to_out.weight"].to(torch.float32)

    # Apply LayerNorm
    x = input_tensor.to(torch.float32)
    x = torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # Project to hidden dimensions
    _left = torch.matmul(x, left_proj_weight.t())
    _right = torch.matmul(x, right_proj_weight.t())

    # Apply mask
    mask = mask.unsqueeze(-1).to(torch.float32)
    _left *= mask
    _right *= mask

    # Apply gates
    _left_gate = torch.matmul(x, left_gate_weight.t()).sigmoid()
    _right_gate = torch.matmul(x, right_gate_weight.t()).sigmoid()
    _out_gate = torch.matmul(x, out_gate_weight.t()).sigmoid()

    # Multiply with gate values
    _left *= _left_gate
    _right *= _right_gate

    # Compute the TriMul
    # We'll implement this as a Triton kernel to avoid the O(N^3) cost of einsum in PyTorch

    # We will use a Triton kernel to compute the TriMul
    # The kernel will compute the following: (i, j, k) where i, j, k ∈ [0, N)
    # and the output is computed as sum over k: left[i][k] * right[j][k]

    # We need to make sure the kernel is aligned with the device and memory
    # Also, ensure all tensors are on the same device (we assume they are)

    # Allocate output tensor
    output = torch.zeros((batch_size, seq_len, seq_len, hidden_dim), device=x.device, dtype=torch.float32)

    # Now define the TriMul kernel
    @triton.jit
    def trit_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        seq_len: tl.constexpr,
        seq_len_n: tl.constexpr,
        hidden_dim: tl.constexpr,
        BLOCK: tl.constexpr
    ):
        # Each thread handles a (i, j) pair
        pid = tl.program_id(0)
        i = pid * BLOCK + tl.arange(0, BLOCK)
        j = tl.arange(0, seq_len)
        i = i[:, None]
        j = j[None, :]

        # Load left and right tensors
        # left[k][i] * right[k][j]
        # So for i, j we need to accumulate over k

        # We will use 2D indexing for k to avoid O(N^3) complexity
        k_range = tl.arange(0, seq_len)

        # Allocate accumulation buffer
        acc = tl.zeros((BLOCK, seq_len), dtype=tl.float32)

        # Accumulate over k
        for k in range(0, seq_len, BLOCK):
            # Load left[i][k], right[j][k]
            # Note: We are using 2D indexing, so we need to ensure alignment
            # left_ptr is stored as [batch, i, k, d], and we have [B, seq_len, seq_len, hidden_dim]
            # We are handling one (i, j) pair at a time

            # Adjust index to the correct batch dimension (0), and select i, j, k
            # Since we are handling one (i, j) pair at a time, the batch is fixed at 0
            # So offset = 0 * (seq_len * seq_len * hidden_dim) + i * seq_len * hidden_dim + j * hidden_dim + k * hidden_dim
            # However, for speed, we can use contiguous memory access

            k = k + tl.arange(0, BLOCK)
            k = k[:, None]

            # Compute address offsets
            left_off = (i * seq_len + k) * hidden_dim + j[None, :]
            right_off = (j * seq_len + k) * hidden_dim + i[None, :]

            # Load the values
            left_vals = tl.load(left_ptr + left_off, mask=(k < seq_len), other=0.0)
            right_vals = tl.load(right_ptr + right_off, mask=(k < seq_len), other=0.0)

            # Multiply and accumulate
            acc += tl.dot(left_vals, right_vals, axis=0)

        # Store result
        out_off = (i * seq_len + j) * hidden_dim + tl.arange(0, hidden_dim)
        tl.store(out_ptr + out_off, acc, mask=(i < seq_len_n))

    # Now launch the Triton kernel
    # We need to set up the grid for the kernel
    # Each thread handles one (i, j) pair, with a block size of BLOCK
    # So the grid is (seq_len, 1, 1)
    # For now, let's set BLOCK = 64 to balance between memory and compute

    # We'll use a fixed Block size for simplicity
    BLOCK = 128
    grid = (triton.cdiv(seq_len * seq_len, BLOCK),)  # Each thread handles one (i,j) pair
    trit_mul_kernel[grid](
        _left.view(batch_size, seq_len, seq_len, hidden_dim).contiguous().data_ptr(),
        _right.view(batch_size, seq_len, seq_len, hidden_dim).contiguous().data_ptr(),
        output.data_ptr(),
        seq_len,
        seq_len,
        hidden_dim,
        BLOCK
    )

    # Apply LayerNorm on the output
    out = torch.nn.functional.layer_norm(output, normalized_shape=(hidden_dim,), 
                                        weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply with out_gate
    out *= _out_gate

    # Apply final Linear projection
    out = torch.matmul(out, to_out_weight.t())

    # Ensure output is in float32
    return out.to(torch.float32)
