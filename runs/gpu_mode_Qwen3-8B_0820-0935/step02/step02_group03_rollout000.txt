import triton
import triton.language as tl
import torch
import math
from typing import Tuple, Dict, Any
import triton._C.libtriton as triton_c

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Implements the TriMul operation using a Triton kernel for optimized performance.
    This kernel is optimized for the given hardware constraints (NVIDIA L40S).
    """

    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # Ensure tensors are on the same device
    device = input_tensor.device

    # Define constants
    SEQLen = seq_len
    BS = batch_size
    DIM = dim
    HIDDEN_DIM = hidden_dim
    BLOCK_SIZE = 64  # Tuned for L40S (smaller block size for better L2 utilization)
    MAX_BLOCK = triton.cdiv(SEQLen, BLOCK_SIZE) * BS  # Max number of blocks needed

    # Initialize output tensor
    output = torch.zeros((BS, SEQLen, SEQLen, HIDDEN_DIM), device=device, dtype=torch.float32)

    # Define Triton kernel
    @triton.jit
    def triton_kernel(
        left_ptr: tl.pointer_type,
        right_ptr: tl.pointer_type,
        gate_left_ptr: tl.pointer_type,
        gate_right_ptr: tl.pointer_type,
        gate_out_ptr: tl.pointer_type,
        out_ptr: tl.pointer_type,
        mask_ptr: tl.pointer_type,
        batch: tl.constexpr,
        seq_len: tl.constexpr,
        dim: tl.constexpr,
        hidden_dim: tl.constexpr,
        BLOCK_SIZE: tl.constexpr,
        STRIDE_LEFT: tl.constexpr,
        STRIDE_RIGHT: tl.constexpr,
        STRIDE_GATE_LEFT: tl.constexpr,
        STRIDE_GATE_RIGHT: tl.constexpr,
        STRIDE_GATE_OUT: tl.constexpr,
        STRIDE_OUT: tl.constexpr,
        STRIDE_MASK: tl.constexpr
    ):
        pid = tl.program_id(0)
        pid_m = pid // (seq_len * hidden_dim)
        pid_n = (pid // hidden_dim) % seq_len
        pid_k = pid % hidden_dim
        
        # Compute offsets
        offset_m = pid_m * seq_len + pid_n
        offset_k = pid_k

        # Initialize accumulator
        accumulator = tl.zeros((BLOCK_SIZE), dtype=tl.float32)

        # Iterate over k
        for k in range(0, hidden_dim, BLOCK_SIZE):
            # Compute k indices
            k_start = k
            k_end = min(k + BLOCK_SIZE, hidden_dim)

            # Load data
            mask = tl.load(mask_ptr + offset_m * STRIDE_MASK + k_start * hidden_dim, mask=tl.arange(0, hidden_dim) < (k_end - k_start), other=0.0)
            left = tl.load(left_ptr + offset_m * STRIDE_LEFT + k_start * hidden_dim, mask=tl.arange(0, hidden_dim) < (k_end - k_start), other=0.0)
            right = tl.load(right_ptr + offset_n * STRIDE_RIGHT + k_start * hidden_dim, mask=tl.arange(0, hidden_dim) < (k_end - k_start), other=0.0)

            # Apply mask
            left = left * mask
            right = right * mask

            # Compute contributions
            product = tl.dot(left, right)
            accumulator += product

        # Apply gate functions
        gate_left = tl.load(gate_left_ptr + offset_k * STRIDE_GATE_LEFT, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0).sigmoid()
        gate_right = tl.load(gate_right_ptr + offset_k * STRIDE_GATE_RIGHT, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0).sigmoid()
        gate_out = tl.load(gate_out_ptr + offset_k * STRIDE_GATE_OUT, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0).sigmoid()

        # Combine outputs
        output = accumulator * gate_left * gate_right
        output = output * gate_out

        # Store result
        tl.store(out_ptr + offset_m * STRIDE_OUT + offset_k, output)

    # Prepare pointers
    left_ptr = input_tensor.contiguous().data_ptr()
    right_ptr = input_tensor.contiguous().data_ptr()
    gate_left_ptr = torch.zeros((BS, SEQLen, SEQLen, hidden_dim), device=device).data_ptr()
    gate_right_ptr = torch.zeros((BS, SEQLen, SEQLen, hidden_dim), device=device).data_ptr()
    gate_out_ptr = torch.zeros((BS, SEQLen, SEQLen, hidden_dim), device=device).data_ptr()
    out_ptr = output.contiguous().data_ptr()
    mask_ptr = mask.contiguous().data_ptr()
    STRIDE_LEFT = input_tensor.stride(0)
    STRIDE_RIGHT = input_tensor.stride(0)
    STRIDE_GATE_LEFT = gate_left_ptr
    STRIDE_GATE_RIGHT = gate_right_ptr
    STRIDE_GATE_OUT = gate_out_ptr
    STRIDE_OUT = output.stride(0)
    STRIDE_MASK = mask.stride(0)

    # Launch kernel
    grid = (BS * SEQLen * hidden_dim, )
    triton_kernel[grid](
        left_ptr,
        right_ptr,
        gate_left_ptr,
        gate_right_ptr,
        gate_out_ptr,
        out_ptr,
        mask_ptr,
        BS,
        SEQLen,
        DIM,
        HIDDEN_DIM,
        BLOCK_SIZE,
        STRIDE_LEFT,
        STRIDE_RIGHT,
        STRIDE_GATE_LEFT,
        STRIDE_GATE_RIGHT,
        STRIDE_GATE_OUT,
        STRIDE_OUT,
        STRIDE_MASK
    )

    return output
