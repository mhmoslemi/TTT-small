import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    """
    Implements the TriMul module using a Triton kernel for optimized performance.

    The TriMul operation applies a series of linear projections and gated operations, 
    followed by a matrix multiplication across the sequence dimension. The kernel computes
    this efficiently using Triton's programmatic control flow and memory access patterns.
    The final output is cast back to float32 as required.
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    batch_size, seqlen = input_tensor.shape[0], input_tensor.shape[1]

    # Extract weight tensors
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    to_out_weight = weights["to_out.weight"].to(torch.float32)

    # Compute the norm
    x = input_tensor.to(torch.float32)
    x = x.view(batch_size, seqlen, seqlen, dim)
    x = x * norm_weight + norm_bias

    # Apply projections
    left = torch.matmul(x, left_proj_weight.t())
    right = torch.matmul(x, right_proj_weight.t())

    # Apply mask
    mask = mask.unsqueeze(-1).to(torch.float32)
    left = left * mask
    right = right * mask

    # Apply gate projections
    left_gate = torch.matmul(x, left_gate_weight.t()).sigmoid()
    right_gate = torch.matmul(x, right_gate_weight.t()).sigmoid()
    out_gate = torch.matmul(x, out_gate_weight.t()).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute TriMul
    output = torch.zeros(batch_size, seqlen, seqlen, hidden_dim, dtype=torch.float32, device=x.device)

    # Triton kernel to perform TriMul
    @triton.jit
    def triton_trimul_kernel(
        left_ptr, right_ptr, output_ptr,
        batch_size, seqlen, hidden_dim,
        BLOCK: tl.constexpr, 
        BLOCK_K: tl.constexpr
    ):
        pid = tl.program_id(0)
        i = pid // (seqlen * seqlen)
        j = (pid // seqlen) % seqlen
        k = pid % seqlen

        # Compute index for batch, i, j, k
        offs_b = i
        offs_i = j
        offs_j = k
        offs_k = tl.arange(0, BLOCK_K)

        # Load left and right data
        left_ptrs = left_ptr + (offs_b * seqlen * seqlen * hidden_dim + offs_i * seqlen * hidden_dim + offs_k * hidden_dim)
        right_ptrs = right_ptr + (offs_b * seqlen * seqlen * hidden_dim + offs_j * seqlen * hidden_dim + offs_k * hidden_dim)

        left = tl.load(left_ptrs, shape=(BLOCK_K, hidden_dim), col=True)
        right = tl.load(right_ptrs, shape=(BLOCK_K, hidden_dim), col=True)

        # Compute the product
        acc = tl.zeros((hidden_dim,), dtype=tl.float32)
        for k_idx in range(BLOCK_K):
            acc += tl.dot(left[k_idx], right[k_idx])
        
        # Store the result
        out_indices = (offs_b * seqlen * seqlen + offs_i * seqlen + offs_j)
        tl.store(output_ptr + out_indices, acc)

    # Launch the kernel
    # We split the operation into small blocks for better memory reuse
    # We use a BLOCK size of 128 for the sequence dimension
    num_work_items = batch_size * seqlen * seqlen
    num_blocks = triton.cdiv(num_work_items, BLOCK=128)
    grid = (num_blocks,)
    triton_trimul_kernel[grid](
        left.contiguous().view(batch_size * seqlen * seqlen * hidden_dim),
        right.contiguous().view(batch_size * seqlen * seqlen * hidden_dim),
        output.contiguous().view(batch_size * seqlen * seqlen * hidden_dim),
        batch_size, seqlen, hidden_dim,
        BLOCK=128,
        BLOCK_K=128
    )

    # Apply normalization and output gate
    output = torch.matmul(output, to_out_norm_weight.t()) + to_out_norm_bias
    output = output * out_gate

    # Final projection
    output = torch.matmul(output, to_out_weight.t())

    return output.to(torch.float32)
