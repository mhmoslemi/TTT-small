import torch
import triton
import triton.language as tl
import math
from typing import Tuple, Dict

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Implements the outgoing version of the Triangle Multiplicative Update (TriMul) module from AlphaFold3.
    It processes a 4D tensor [B, N, N, C] and applies a series of linear projections, gate operations, and pairwise multiplication.

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [bs, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [bs, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters

    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Define the kernel
    @triton.jit
    def trit_mul_kernel(
        input_ptr,  # pointer to input tensor [B, N, N, D]
        mask_ptr,   # pointer to mask tensor [B, N, N]
        left_proj_ptr,
        right_proj_ptr,
        left_gate_ptr,
        right_gate_ptr,
        out_gate_ptr,
        to_out_norm_ptr,
        to_out_ptr,
        out_ptr,    # pointer to output tensor [B, N, N, D]
        B: tl.constexpr,
        N: tl.constexpr,
        dim: tl.constexpr,
        hidden_dim: tl.constexpr
    ):
        # Define block sizes
        BLOCK = 64
        num_block_m = tl.cdiv(N, BLOCK)
        num_block_n = tl.cdiv(N, BLOCK)
        num_block_k = tl.cdiv(N, BLOCK)
        # Current block index in batch
        pid_b = tl.program_id(0)
        # Current block index in M dimension
        pid_m = tl.program_id(1)
        # Current block index in N dimension
        pid_n = tl.program_id(2)
        # Local offset in block
        offs_m = pid_m * BLOCK + tl.arange(0, BLOCK)
        offs_n = pid_n * BLOCK + tl.arange(0, BLOCK)
        # Compute actual indices
        m_idx = offs_m
        n_idx = offs_n

        # Load mask
        mask_off = tl.math.floormod(m_idx, N) * N + tl.math.floormod(n_idx, N)
        mask_val = tl.load(mask_ptr + pid_b * N * N + mask_off, mask=(m_idx < N) & (n_idx < N))

        # Initialize output block
        out_block = tl.zeros((BLOCK, BLOCK, dim), dtype=tl.float32)

        # Load left and right projections
        left_proj_ptr = left_proj_ptr + pid_b * N * N * hidden_dim + (m_idx[:, None] * N + n_idx[None, :]) * hidden_dim
        right_proj_ptr = right_proj_ptr + pid_b * N * N * hidden_dim + (n_idx[:, None] * N + m_idx[None, :]) * hidden_dim

        for k in range(num_block_k):
            # Load left and right projections
            left_data = tl.load(left_proj_ptr + k * hidden_dim, mask=(m_idx < N) & (n_idx < N), other=0.0)
            right_data = tl.load(right_proj_ptr + k * hidden_dim, mask=(n_idx < N) & (m_idx < N), other=0.0)

            # Compute the product
            out_block += tl.dot(left_data, right_data, axis=1)

        # Apply gates
        # Load left gate
        left_gate_ptr = left_gate_ptr + pid_b * N * N * hidden_dim + (m_idx[:, None] * N + n_idx[None, :]) * hidden_dim
        left_gate_val = tl.load(left_gate_ptr, mask=(m_idx < N) & (n_idx < N), other=0.0)
        left_gate_val = left_gate_val.sigmoid()

        # Load right gate
        right_gate_ptr = right_gate_ptr + pid_b * N * N * hidden_dim + (n_idx[:, None] * N + m_idx[None, :]) * hidden_dim
        right_gate_val = tl.load(right_gate_ptr, mask=(n_idx < N) & (m_idx < N), other=0.0)
        right_gate_val = right_gate_val.sigmoid()

        # Apply gates
        out_block = out_block * left_gate_val * right_gate_val

        # Apply output gate
        out_gate_ptr = out_gate_ptr + pid_b * N * N * hidden_dim + (m_idx[:, None] * N + n_idx[None, :]) * hidden_dim
        out_gate_val = tl.load(out_gate_ptr, mask=(m_idx < N) & (n_idx < N), other=0.0)
        out_gate_val = out_gate_val.sigmoid()
        out_block = out_block * out_gate_val

        # Apply normalization
        to_out_norm_ptr = to_out_norm_ptr + pid_b * N * N * hidden_dim + (m_idx[:, None] * N + n_idx[None, :]) * hidden_dim
        norm_weights = tl.load(to_out_norm_ptr, mask=(m_idx < N) & (n_idx < N), other=0.0)
        out_block = out_block * norm_weights

        # Apply to_out projection
        to_out_ptr = to_out_ptr + pid_b * N * N * dim + (m_idx[:, None] * N + n_idx[None, :]) * dim
        to_out_weights = tl.load(to_out_ptr, mask=(m_idx < N) & (n_idx < N), other=0.0)
        out_block = out_block * to_out_weights

        # Store to output
        tl.store(out_ptr + pid_b * N * N * dim + (m_idx[:, None] * N + n_idx[None, :]) * dim, out_block, mask=(m_idx < N) & (n_idx < N))

    # Prepare inputs
    B = input_tensor.shape[0]
    N = input_tensor.shape[1]
    dim = input_tensor.shape[3]
    hidden_dim = config["hidden_dim"]

    # Extract weights
    norm_weight = weights["norm.weight"].to(torch.float32).contiguous()
    norm_bias = weights["norm.bias"].to(torch.float32).contiguous()
    left_proj_weight = weights["left_proj.weight"].to(torch.float32).contiguous()
    right_proj_weight = weights["right_proj.weight"].to(torch.float32).contiguous()
    left_gate_weight = weights["left_gate.weight"].to(torch.float32).contiguous()
    right_gate_weight = weights["right_gate.weight"].to(torch.float32).contiguous()
    out_gate_weight = weights["out_gate.weight"].to(torch.float32).contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32).contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32).contiguous()
    to_out_weight = weights["to_out.weight"].to(torch.float32).contiguous()

    # Prepare mask
    mask = mask.unsqueeze(-1)  # shape: [B, N, N, 1]
    mask = mask.to(torch.float32).contiguous()

    # Initialize output
    output_shape = (B, N, N, dim)
    output = torch.zeros(output_shape, device=input_tensor.device, dtype=torch.float32)

    # Launch kernel
    grid = (B, triton.cdiv(N, 64), triton.cdiv(N, 64))
    trit_mul_kernel[grid](
        input_tensor.data_ptr(),
        mask.data_ptr(),
        left_proj_weight.data_ptr(),
        right_proj_weight.data_ptr(),
        left_gate_weight.data_ptr(),
        right_gate_weight.data_ptr(),
        out_gate_weight.data_ptr(),
        to_out_norm_weight.data_ptr(),
        to_out_weight.data_ptr(),
        output.data_ptr(),
        B=B,
        N=N,
        dim=dim,
        hidden_dim=hidden_dim
    )

    return output
