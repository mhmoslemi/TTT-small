import torch
import triton
import triton.language as tl
import math

# Docstring: Implement the "outgoing" version of the TriMul operator using efficient Triton kernels
# to optimize the pair-wise operations that sum over the sequence dimension for high performance on NVIDIA L40S

@triton.jit
def tri_mul_kernel(
    input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr,
    to_out_norm_ptr, to_out_ptr, 
    batch_size, seq_len, dim, hidden_dim,
    BLOCK: tl.constexpr, 
    BLOCK_K: tl.constexpr, 
    BLOCK_I: tl.constexpr, 
    BLOCK_J: tl.constexpr,
):
    # Compute the program ID
    pid = tl.program_id(0)
    pid_m = pid // (seq_len * seq_len)
    pid_n = (pid // seq_len) % seq_len
    pid_k = pid % seq_len

    # Compute the offset for the current batch
    batch_offset = pid_m * seq_len * seq_len * dim

    # Compute the offset for the current row i
    i = pid_n
    i_offset = i * seq_len * dim

    # Compute the offset for the current row j
    j = pid_k
    j_offset = j * dim

    # Load the mask value
    mask = tl.load(mask_ptr + batch_offset + i_offset + j_offset, eviction_policy="evict_last")
    if mask == 0:
        return

    # Compute the offset for left and right projections
    left_ptr = input_ptr + batch_offset + i_offset
    right_ptr = input_ptr + batch_offset + j_offset
    left_gate_ptr = left_gate_ptr + batch_offset + i_offset
    right_gate_ptr = right_gate_ptr + batch_offset + j_offset
    out_gate_ptr = out_gate_ptr + batch_offset + i_offset
    to_out_norm_ptr = to_out_norm_ptr + batch_offset + i_offset
    to_out_ptr = to_out_ptr + batch_offset + i_offset

    # Compute the indices for the current block
    offs_k = tl.arange(0, BLOCK_K)
    offs_i = tl.arange(0, BLOCK_I)
    offs_j = tl.arange(0, BLOCK_J)

    # Load the left and right projections
    left = tl.load(left_ptr + offs_k[:, None] * seq_len * dim + offs_i[None, :], mask=offs_k[:, None] < seq_len)
    right = tl.load(right_ptr + offs_k[:, None] * seq_len * dim + offs_j[None, :], mask=offs_k[:, None] < seq_len)

    # Compute the left and right gates
    left_gate = tl.load(left_gate_ptr + offs_k[:, None] * seq_len * dim + offs_i[None, :], mask=offs_k[:, None] < seq_len)
    right_gate = tl.load(right_gate_ptr + offs_k[:, None] * seq_len * dim + offs_j[None, :], mask=offs_k[:, None] < seq_len)
    out_gate = tl.load(out_gate_ptr + offs_k[:, None] * seq_len * dim + offs_i[None, :], mask=offs_k[:, None] < seq_len)

    # Compute the left and right with gates
    left = left * left_gate
    right = right * right_gate

    # Compute the matrix multiplication
    acc = tl.dot(left, right, allow_tf32=True)

    # Multiply by the out gate
    acc = acc * out_gate

    # Apply the normalization
    norm_weight = tl.load(to_out_norm_ptr + offs_i[None, :] * hidden_dim + offs_j[None, :], mask=offs_i[None, :] < seq_len)
    norm_bias = tl.load(to_out_norm_ptr + hidden_dim + offs_i[None, :] * hidden_dim + offs_j[None, :], mask=offs_i[None, :] < seq_len)
    acc = acc * norm_weight + norm_bias

    # Apply the final linear projection
    to_out_weight = tl.load(to_out_ptr + offs_i[None, :] * hidden_dim + offs_j[None, :], mask=offs_i[None, :] < seq_len)
    out = tl.dot(acc, to_out_weight, allow_tf32=True)

    # Store the result
    out_ptr = input_ptr + batch_offset + i_offset + j_offset
    tl.store(out_ptr + offs_i[None, :] * dim + offs_j[None, :], out, mask=offs_i[None, :] < seq_len)

    # Note: The kernel is launched with the block sizes and grid size defined above
