import torch
from torch import Tensor
import triton
import triton.language as tl

def custom_kernel(data: tuple) -> Tensor:
    """
    Triton implementation of the outgoing TriMul operator from AlphaFold3.
    This kernel implements the forward pass of the TriMul module, which operates
    over a 4D tensor [bs, seq_len, seq_len, dim]. The implementation is optimized
    for the L40S hardware using Triton's tiling and parallelism capabilities.

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [bs, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [bs, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters

    Returns:
        Output tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    batch_size, seqlen, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Load weights from the dictionary
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Initialize output tensor
    output = torch.zeros((batch_size, seqlen, seqlen, dim), device=input_tensor.device)

    # Compute norms
    norm_weight = norm_weight.to(input_tensor.dtype)
    norm_bias = norm_bias.to(input_tensor.dtype)
    x = input_tensor * norm_weight + norm_bias

    # Define the Triton kernel
    @triton.jit
    def triton_trimul_kernel(
        x_ptr, mask_ptr, left_proj_weight_ptr, right_proj_weight_ptr,
        left_gate_weight_ptr, right_gate_weight_ptr, out_gate_weight_ptr,
        to_out_norm_weight_ptr, to_out_norm_bias_ptr, to_out_weight_ptr,
        out_ptr,
        N: tl.constexpr, C: tl.constexpr, HIDDEN: tl.constexpr, BLOCK: tl.constexpr
    ):
        # Compute the batch dimension
        pid = tl.program_id(0)
        batch = pid
        # Compute the sequence dimension
        pid_seq = tl.program_id(1)
        seq = pid_seq

        # Compute the offset for the current batch and sequence
        offsets = tl.arange(0, BLOCK)
        mask_offsets = tl.arange(0, BLOCK)[:, None] * N + tl.arange(0, BLOCK)[None, :]

        # Load weights
        left_proj_weight = tl.load(left_proj_weight_ptr + tl.arange(0, HIDDEN) * C + offsets[:, None], mask=offsets[:, None] < C)
        right_proj_weight = tl.load(right_proj_weight_ptr + tl.arange(0, HIDDEN) * C + offsets[:, None], mask=offsets[:, None] < C)
        left_gate_weight = tl.load(left_gate_weight_ptr + tl.arange(0, HIDDEN) * C + offsets[:, None], mask=offsets[:, None] < C)
        right_gate_weight = tl.load(right_gate_weight_ptr + tl.arange(0, HIDDEN) * C + offsets[:, None], mask=offsets[:, None] < C)
        out_gate_weight = tl.load(out_gate_weight_ptr + tl.arange(0, HIDDEN) * C + offsets[:, None], mask=offsets[:, None] < C)
        to_out_norm_weight = tl.load(to_out_norm_weight_ptr + tl.arange(0, HIDDEN) * C + offsets[:, None], mask=offsets[:, None] < C)
        to_out_norm_bias = tl.load(to_out_norm_bias_ptr + tl.arange(0, HIDDEN) * C + offsets[:, None], mask=offsets[:, None] < C)
        to_out_weight = tl.load(to_out_weight_ptr + tl.arange(0, HIDDEN) * C + offsets[:, None], mask=offsets[:, None] < C)

        # Compute the mask
        mask = tl.load(mask_ptr + batch * N * N + seq * N + mask_offsets)
        mask = mask.reshape((BLOCK, BLOCK))
        mask = mask.to(tl.float32)

        # Compute left and right projections
        left = tl.load(x_ptr + batch * N * N * C + seq * N * C + tl.arange(0, N) * C + tl.arange(0, C)[:, None], mask=offsets[:, None] < N) * mask
        right = tl.load(x_ptr + batch * N * N * C + seq * N * C + tl.arange(0, N) * C + tl.arange(0, C)[:, None], mask=offsets[:, None] < N) * mask

        # Compute gates
        left_gate = tl.load(x_ptr + batch * N * N * C + seq * N * C + tl.arange(0, N) * C + tl.arange(0, C)[:, None], mask=offsets[:, None] < N) * left_gate_weight
        left_gate = tl.sigmoid(left_gate)
        right_gate = tl.load(x_ptr + batch * N * N * C + seq * N * C + tl.arange(0, N) * C + tl.arange(0, C)[:, None], mask=offsets[:, None] < N) * right_gate_weight
        right_gate = tl.sigmoid(right_gate)
        out_gate = tl.load(x_ptr + batch * N * N * C + seq * N * C + tl.arange(0, N) * C + tl.arange(0, C)[:, None], mask=offsets[:, None] < N) * out_gate_weight
        out_gate = tl.sigmoid(out_gate)

        # Compute left and right with gates
        left = left * left_gate
        right = right * right_gate

        # Compute the TriMul operation
        out = tl.dot(left, right)

        # Normalize the output
        out = out * to_out_norm_weight + to_out_norm_bias

        # Apply the out gate
        out = out * out_gate

        # Store the result
        tl.store(out_ptr + batch * N * N * C + seq * N * C + tl.arange(0, N) * C + tl.arange(0, C)[:, None], out)

    # Launch the kernel
    grid = (batch_size, seqlen, 1)
    triton_trimul_kernel[grid](
        input_tensor,
        mask,
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
        to_out_norm_weight,
        to_out_norm_bias,
        to_out_weight,
        output,
        N=seqlen,
        C=dim,
        HIDDEN=hidden_dim,
        BLOCK=128
    )

    # Ensure the output is in float32
    return output.to(torch.float32)
