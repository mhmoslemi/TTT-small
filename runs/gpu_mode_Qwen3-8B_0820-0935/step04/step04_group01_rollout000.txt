import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    """
    Implements the outgoing version of the TriMul module for protein structure prediction models.
    This kernel optimizes the key operation using Triton to improve speed while maintaining
    correctness. The implementation carefully handles masks and performs optimized fused operations.
    
    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [batch_size, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [batch_size, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters

    Returns:
        output: Processed tensor [batch_size, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # Define constants for Triton kernel
    SEQLEN = seq_len
    BS = batch_size
    DIM = dim
    HIDDEN_DIM = hidden_dim
    BLOCK = 128  # Tuned block size based on L40S hardware

    # Access weights
    norm_weight = weights["norm.weight"].to(torch.float32).contiguous()
    norm_bias = weights["norm.bias"].to(torch.float32).contiguous()
    left_proj_weight = weights["left_proj.weight"].to(torch.float32).contiguous()
    right_proj_weight = weights["right_proj.weight"].to(torch.float32).contiguous()
    left_gate_weight = weights["left_gate.weight"].to(torch.float32).contiguous()
    right_gate_weight = weights["right_gate.weight"].to(torch.float32).contiguous()
    out_gate_weight = weights["out_gate.weight"].to(torch.float32).contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32).contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32).contiguous()
    to_out_weight = weights["to_out.weight"].to(torch.float32).contiguous()

    # Handle mask
    mask = mask.unsqueeze(-1).to(torch.float32).contiguous()

    # Define the kernel
    @triton.jit
    def tritond_kernel(
        input_ptr, left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr,
        norm_weight_ptr, norm_bias_ptr, to_out_norm_weight_ptr, to_out_norm_bias_ptr, to_out_weight_ptr,
        mask_ptr,
        input_strides, left_proj_strides, right_proj_strides, left_gate_strides, right_gate_strides, out_gate_strides,
        norm_weight_strides, norm_bias_strides, to_out_norm_weights_strides, to_out_norm_biases_strides, to_out_weights_strides,
        mask_strides,
        dim, hidden_dim, seqlen, bs, BLOCK: tl.constexpr,
        dtype: tl.constexpr
    ):
        # Define offsets
        pid = tl.program_id(0)
        block_id = pid // (triton.cdiv(seqlen, BLOCK) * triton.cdiv(seqlen, BLOCK))
        block_m = (pid // triton.cdiv(seqlen, BLOCK)) % triton.cdiv(seqlen, BLOCK)
        block_n = pid % triton.cdiv(seqlen, BLOCK)

        # Compute offsets for the current block
        offs_m = block_m * BLOCK + tl.arange(0, BLOCK)
        offs_n = block_n * BLOCK + tl.arange(0, BLOCK)

        # Load left and right project weight (shared memory for weight reuse)
        left_proj_weight = tl.load(left_proj_ptr + tl.arange(0, hidden_dim) * hidden_dim, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
        right_proj_weight = tl.load(right_proj_ptr + tl.arange(0, hidden_dim) * hidden_dim, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
        left_gate_weight = tl.load(left_gate_ptr + tl.arange(0, hidden_dim) * hidden_dim, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
        right_gate_weight = tl.load(right_gate_ptr + tl.arange(0, hidden_dim) * hidden_dim, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
        out_gate_weight = tl.load(out_gate_ptr + tl.arange(0, hidden_dim) * hidden_dim, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
        to_out_norm_weight = tl.load(to_out_norm_weight_ptr + tl.arange(0, hidden_dim) * hidden_dim, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
        to_out_norm_bias = tl.load(to_out_norm_bias_ptr + tl.arange(0, hidden_dim), mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
        to_out_weight = tl.load(to_out_weight_ptr + tl.arange(0, dim) * hidden_dim, mask=tl.arange(0, dim) < dim, other=0.0)

        # Load norms
        norm_weight = tl.load(norm_weight_ptr + tl.arange(0, dim), mask=tl.arange(0, dim) < dim, other=0.0)
        norm_bias = tl.load(norm_bias_ptr + tl.arange(0, dim), mask=tl.arange(0, dim) < dim, other=0.0)

        # Compute the mask
        mask = tl.load(mask_ptr + block_m * seqlen * seqlen + tl.arange(0, seqlen), mask=tl.arange(0, seqlen) < seqlen, other=0.0)

        # Loop over sequences
        for b in range(bs):
            # Load input
            x = tl.load(input_ptr + (b * seqlen * seqlen * dim) + offs_m[:, None] * seqlen + tl.arange(0, dim), mask=offs_m < seqlen, other=0.0)
            x = x * norm_weight + norm_bias

            # Project left and right
            left = tl.dot(x, left_proj_weight)
            right = tl.dot(x, right_proj_weight)

            # Apply mask
            left = left * mask
            right = right * mask

            # Apply gates
            left_gate = tl.dot(x, left_gate_weight).sigmoid()
            right_gate = tl.dot(x, right_gate_weight).sigmoid()
            out_gate = tl.dot(x, out_gate_weight).sigmoid()

            left = left * left_gate
            right = right * right_gate

            # Compute output
            out = tl.zeros((BLOCK, BLOCK, hidden_dim), dtype=tl.float32)
            for k in range(seqlen):
                a = tl.load(input_ptr + (b * seqlen * seqlen * dim) + (offs_m[:, None] * seqlen + k) * dim + tl.arange(0, hidden_dim))
                b = tl.load(input_ptr + (b * seqlen * seqlen * dim) + (offs_n[None, :] * seqlen + k) * dim + tl.arange(0, hidden_dim))
                out += tl.dot(a, b)

            # Apply normalization and gate
            out = out * to_out_norm_weight + to_out_norm_bias
            out = out * out_gate

            # Output
            tl.store(output_ptr + (b * seqlen * seqlen * dim) + offs_m[:, None] * seqlen + offs_n[None, :], out)

    # Output tensor shape
    output = torch.empty(batch_size, seq_len, seq_len, dim, device=input_tensor.device, dtype=torch.float32)

    # Launch the kernel
    grid = (triton.cdiv(seq_len, BLOCK) * triton.cdiv(seq_len, BLOCK),)
    tritond_kernel[grid](
        input_tensor, left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight, out_gate_weight,
        norm_weight, norm_bias, to_out_norm_weight, to_out_norm_bias, to_out_weight,
        mask,
        input_tensor.stride(0), input_tensor.stride(1), input_tensor.stride(2), input_tensor.stride(3),
        left_proj_weight.stride(0), left_proj_weight.stride(1),
        right_proj_weight.stride(0), right_proj_weight.stride(1),
        left_gate_weight.stride(0), left_gate_weight.stride(1),
        right_gate_weight.stride(0), right_gate_weight.stride(1),
        out_gate_weight.stride(0), out_gate_weight.stride(1),
        norm_weight.stride(0), norm_bias.stride(0),
        to_out_norm_weight.stride(0), to_out_norm_bias.stride(0),
        to_out_weight.stride(0), to_out_weight.stride(1),
        mask.stride(0), mask.stride(1),
        dim, hidden_dim, seq_len, batch_size, BLOCK,
        torch.float32
    )

    return output
