import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    """
    Optimized Triton kernel implementation of the TriMul operation.
    
    The TriMul operation performs a sequence of linear projections, gate activations,
    and a pairwise multiplication that computes a tensor of shape [bs, seq_len, seq_len, dim].
    This implementation uses Triton's low-level optimizations to reduce memory access
    and improve performance on NVIDIA L40S (Ada Lovelace, sm_89).
    """
    # Extract input and configuration
    input_tensor, mask, weights, config = data
    bs, seqlen, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # Check if mask is present; if not, assume all sequences are available
    mask_present = mask is not None

    # Prepare the weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Convert all to float32 for processing
    norm_weight = norm_weight.to(torch.float32)
    norm_bias = norm_bias.to(torch.float32)
    left_proj_weight = left_proj_weight.to(torch.float32)
    right_proj_weight = right_proj_weight.to(torch.float32)
    left_gate_weight = left_gate_weight.to(torch.float32)
    right_gate_weight = right_gate_weight.to(torch.float32)
    out_gate_weight = out_gate_weight.to(torch.float32)
    to_out_norm_weight = to_out_norm_weight.to(torch.float32)
    to_out_norm_bias = to_out_norm_bias.to(torch.float32)
    to_out_weight = to_out_weight.to(torch.float32)

    # Compute norms (layer normalization)
    # This could be implemented in Triton for faster execution, but for simplicity,
    # we'll compute it in PyTorch for now
    input_tensor = input_tensor.to(torch.float32)
    input_tensor = input_tensor * norm_weight + norm_bias

    # Linear projections
    left_proj = torch.matmul(input_tensor, left_proj_weight.t())
    right_proj = torch.matmul(input_tensor, right_proj_weight.t())

    # Apply mask
    if mask_present:
        mask = mask.unsqueeze(-1).to(torch.float32)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Linear gates
    left_gate = torch.matmul(input_tensor, left_gate_weight.t()).sigmoid()
    right_gate = torch.matmul(input_tensor, right_gate_weight.t()).sigmoid()
    out_gate = torch.matmul(input_tensor, out_gate_weight.t()).sigmoid()

    # Apply gate activations
    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate

    # Compute TriMul operation
    # We need to compute: out = einsum("bikd, bjdk -> bijd", left_proj, right_proj)

    # For performance, we implement this in Triton directly
    # This is the core of the TriMul kernel

    # Define the Triton kernel
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        batch_size, seq_len, dim,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        # Identify the block position in the batch, row, and column
        pid = tl.program_id(0)
        pid_m = pid // (seq_len * seq_len)
        pid_n = (pid // seq_len) % seq_len
        pid_k = pid % seq_len

        # Compute the offsets for the current block
        offs_m = pid_m * seq_len + tl.arange(0, seq_len)
        offs_n = pid_n * seq_len + tl.arange(0, seq_len)
        offs_k = pid_k * seq_len + tl.arange(0, seq_len)

        # Load the left, right, and mask tensors into registers
        # Since we have already applied the mask and gate activation in PyTorch,
        # here, we assume we're dealing with the masked, gated versions
        left = tl.load(left_ptr + offs_m[:, None] * seq_len * seq_len + offs_k[None, :] * seq_len, boundary_check=(0, 1))
        right = tl.load(right_ptr + offs_n[None, :] * seq_len + offs_k[:, None] * seq_len, boundary_check=(0, 1))

        # Compute the product and accumulate
        acc = tl.zeros((seq_len, seq_len), dtype=tl.float32)
        for k in range(0, seq_len, BLOCK_K):
            # Extract a block of left and right
            left_block = left[k:k+BLOCK_K, :]
            right_block = right[k:k+BLOCK_K, :]
            # Compute product
            dot = tl.dot(left_block, right_block)
            # Accumulate
            acc += dot

        # Store the result
        tl.store(out_ptr + offs_m[:, None] * seq_len * seq_len + offs_n[None, :], acc)

    # Prepare the output tensor
    output = torch.zeros(bs, seqlen, seqlen, dim, device=input_tensor.device, dtype=torch.float32)

    # Launch the Triton kernel
    # Determine the grid size
    grid_size = (bs * seqlen * seqlen, )
    # Launch the kernel
    tri_mul_kernel[grid_size](
        left_proj.contiguous(),
        right_proj.contiguous(),
        output.contiguous(),
        batch_size=bs,
        seq_len=seqlen,
        dim=dim,
        BLOCK_M=seqlen,
        BLOCK_N=seqlen,
        BLOCK_K=seqlen // 2
    )

    # Apply the output gate
    output = output * out_gate

    # Apply the final normalization and projection
    output = torch.matmul(output, to_out_norm_weight.t()) + to_out_norm_bias
    output = torch.matmul(output, to_out_weight.t())

    return output
