import triton
import triton.language as tl
import torch
import math
import sys
import os
import time

def custom_kernel(data):
    """
    Optimized Triton implementation of the TriMul module.
    
    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [bs, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [bs, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape

    # Convert mask to float32
    mask = mask.to(torch.float32)

    # Extract all weights
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    to_out_weight = weights["to_out.weight"].to(torch.float32)

    # Compute the output
    output = _triton_triplet_mul(
        input_tensor,
        mask,
        norm_weight,
        norm_bias,
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
        to_out_norm_weight,
        to_out_norm_bias,
        to_out_weight,
        seq_len,
        bs,
        dim,
        hidden_dim
    )

    return output

@triton.jit
def _triton_triplet_mul(
    x_ptr, mask_ptr, norm_weight, norm_bias,
    left_proj_weight, right_proj_weight,
    left_gate_weight, right_gate_weight,
    out_gate_weight, to_out_norm_weight, to_out_norm_bias,
    to_out_weight,
    N: tl.constexpr, B: tl.constexpr, dim: tl.constexpr, hidden_dim: tl.constexpr,
    BLOCK: tl.constexpr,
):
    # Each thread handles one output element
    pid = tl.program_id(0)
    i = pid // (N * N)
    j = (pid // N) % N
    k = pid % N

    # Calculate the offset in the output tensor
    offset = i * N * N + j * N + k
    o = tl.load(x_ptr + offset)

    # Compute the norm
    norm_weight = norm_weight.view(hidden_dim)
    norm_bias = norm_bias.view(hidden_dim)
    x = o + norm_bias
    x = x * norm_weight

    # Compute the projection for left and right
    left_proj = tl.load(left_proj_weight + (k * hidden_dim + tl.arange(0, hidden_dim))).view(N, hidden_dim)
    right_proj = tl.load(right_proj_weight + (k * hidden_dim + tl.arange(0, hidden_dim))).view(N, hidden_dim)
    left = x * left_proj
    right = x * right_proj

    # Apply masking
    mask = tl.load(mask_ptr + (i * N * N + j * N + k))
    left = left * mask
    right = right * mask

    # Compute gate projections
    left_gate = tl.load(left_gate_weight + (k * hidden_dim + tl.arange(0, hidden_dim))).view(N, hidden_dim)
    right_gate = tl.load(right_gate_weight + (k * hidden_dim + tl.arange(0, hidden_dim))).view(N, hidden_dim)
    out_gate = tl.load(out_gate_weight + (k * hidden_dim + tl.arange(0, hidden_dim))).view(N, hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # Compute the TriMul output
    out = tl.zeros((hidden_dim,), dtype=tl.float32)
    for k in range(N):
        left_k = left[k]
        right_k = right[k]
        out += tl.dot(left_k, right_k)

    # Apply output gate
    out = out * out_gate

    # Apply to_out_norm
    to_out_norm_weight = to_out_norm_weight.view(hidden_dim)
    to_out_norm_bias = to_out_norm_bias.view(hidden_dim)
    out = out * to_out_norm_weight + to_out_norm_bias

    # Apply to_out
    to_out_weight = to_out_weight.view(hidden_dim, dim)
    out = tl.dot(out, to_out_weight)

    # Store the result
    offset = i * N * N + j * N + k
    tl.store(x_ptr + offset, out)

def _triton_call(x, mask, norm_weight, norm_bias,
                 left_proj_weight, right_proj_weight,
                 left_gate_weight, right_gate_weight,
                 out_gate_weight, to_out_norm_weight, to_out_norm_bias,
                 to_out_weight,
                 N, B, dim, hidden_dim):
    # Ensure the number of threads is correct
    num_threads = B * N * N
    grid = (num_threads,)
    
    # Call the kernel
    _triton_triplet_mul[grid](
        x,
        mask,
        norm_weight,
        norm_bias,
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
        to_out_norm_weight,
        to_out_norm_bias,
        to_out_weight,
        N,
        B,
        dim,
        hidden_dim,
        BLOCK=4,
    )

def _triton_triplet_mul_output(x, mask, norm_weight, norm_bias,
                              left_proj_weight, right_proj_weight,
                              left_gate_weight, right_gate_weight,
                              out_gate_weight, to_out_norm_weight, to_out_norm_bias,
                              to_out_weight,
                              N, B, dim, hidden_dim):
    # Initialize the output
    out = torch.empty(B, N, N, dim, dtype=torch.float32, device=x.device)

    # Call the kernel for each batch
    for i in range(B):
        _triton_call(x[i], mask[i], norm_weight, norm_bias,
                     left_proj_weight, right_proj_weight,
                     left_gate_weight, right_gate_weight,
                     out_gate_weight, to_out_norm_weight, to_out_norm_bias,
                     to_out_weight,
                     N, 1, dim, hidden_dim)

    return out

def _triton_triple_mul(x, mask, weights, config):
    # Extract configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = x.shape

    # Extract all weights
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    to_out_weight = weights["to_out.weight"].to(torch.float32)

    # Compute the output
    out = _triton_triplet_mul_output(x, mask, norm_weight, norm_bias,
                                     left_proj_weight, right_proj_weight,
                                     left_gate_weight, right_gate_weight,
                                     out_gate_weight, to_out_norm_weight,
                                     to_out_norm_bias, to_out_weight,
                                     N, B, dim, hidden_dim)

    return out

def custom_kernel(data):
    """
    Optimized Triton implementation of the TriMul module.
    
    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [bs, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [bs, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape

    # Compute the output
    output = _triton_triple_mul(input_tensor, mask, weights, config)

    return output
