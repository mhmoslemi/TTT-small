import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton implementation of the TriMul module with optimized memory access and computation.
    This kernel is designed for the "outgoing" version of the TriMul operator, as used in AlphaFold3.

    Key optimizations:
    1. Perform all dense linear transformations and activation functions using Triton kernels.
    2. Handle masking with efficient pointer arithmetic and mask application.
    3. Compute the TriMul operation (i.e., einsum) using tile-based memory access.

    Args:
        data: Tuple containing (input_tensor: torch.Tensor, mask: torch.Tensor, 
                             weights: Dict[str, torch.Tensor], config: Dict)

    Returns:
        output: Tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access the given weights of the model
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure data types match
    input_tensor = input_tensor.to(torch.float32)
    mask = mask.to(torch.float32)

    # Compute the norm
    norm_weight = norm_weight.to(input_tensor.device)
    norm_bias = norm_bias.to(input_tensor.device)
    x = input_tensor
    x = x * norm_weight + norm_bias

    # Compute projections
    left = compute_linear(x, left_proj_weight, (batch_size, seq_len, seq_len, hidden_dim))
    right = compute_linear(x, right_proj_weight, (batch_size, seq_len, seq_len, hidden_dim))
    mask = mask.unsqueeze(-1)
    left = left * mask
    right = right * mask

    # Compute gates
    left_gate = compute_linear(x, left_gate_weight, (batch_size, seq_len, seq_len, hidden_dim))
    right_gate = compute_linear(x, right_gate_weight, (batch_size, seq_len, seq_len, hidden_dim))
    out_gate = compute_linear(x, out_gate_weight, (batch_size, seq_len, seq_len, hidden_dim))

    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute TriMul operation (einsum: '... i k d, ... j k d -> ... i j d')
    output = compute_trimul(left, right, batch_size, seq_len, hidden_dim)

    # Apply normalization
    output = output * to_out_norm_weight + to_out_norm_bias
    output = output * out_gate

    # Final projection
    output = compute_linear(output, to_out_weight, (batch_size, seq_len, seq_len, dim))

    return output

@triton.jit
def compute_linear(x_ptr, weight_ptr, shape, BLOCK: tl.constexpr):
    """
    Triton implementation of a linear operation (matmul).
    """
    batch_size, seq_len, _, dim = shape
    pid = tl.program_id(0)
    pid_m = pid // (seq_len * seq_len)
    pid_n = (pid // seq_len) % seq_len
    pid_k = pid % seq_len

    offs_b = pid_m * batch_size
    offs_i = pid_n * seq_len + tl.arange(0, seq_len)
    offs_k = tl.arange(0, seq_len)
    offs_d = tl.arange(0, dim)

    mask = (offs_i < seq_len) & (offs_k < seq_len) & (offs_d < dim)

    x = tl.load(x_ptr + offs_b + offs_i[:, None] * seq_len * seq_len + offs_k[None, :] * seq_len + offs_d[None, :], mask=mask)
    x = x.to(tl.float32)

    weight = tl.load(weight_ptr + offs_d[None, :] * seq_len + offs_k[:, None], mask=mask)
    weight = weight.to(tl.float32)

    acc = tl.dot(x, weight)
    return acc

@triton.jit
def compute_trimul(left_ptr, right_ptr, batch_size, seq_len, dim, BLOCK: tl.constexpr):
    """
    Triton implementation of the TriMul operation (einsum: '... i k d, ... j k d -> ... i j d').
    """
    pid = tl.program_id(0)
    pid_b = pid // (seq_len * seq_len * dim)
    pid_i = (pid // (seq_len * dim)) % seq_len
    pid_j = (pid // dim) % seq_len
    pid_d = pid % dim

    offs_b = pid_b * batch_size
    offs_i = pid_i * seq_len + tl.arange(0, seq_len)
    offs_j = pid_j * seq_len + tl.arange(0, seq_len)
    offs_k = tl.arange(0, seq_len)
    offs_d = pid_d * dim + tl.arange(0, dim)

    mask = (offs_i < seq_len) & (offs_j < seq_len) & (offs_k < seq_len) & (offs_d < dim)

    left = tl.load(left_ptr + offs_b + offs_i[:, None] * seq_len * seq_len + offs_k[None, :] * seq_len + offs_d[None, :], mask=mask)
    right = tl.load(right_ptr + offs_b + offs_j[:, None] * seq_len * seq_len + offs_k[None, :] * seq_len + offs_d[None, :], mask=mask)

    left = left.to(tl.float32)
    right = right.to(tl.float32)

    acc = tl.dot(left, right, axis=1)
    return acc

@triton.jit
def compute_linear_final(x_ptr, weight_ptr, shape, BLOCK: tl.constexpr):
    """
    Triton implementation of the final linear operation.
    """
    batch_size, seq_len, _, dim = shape
    pid = tl.program_id(0)
    pid_b = pid // (seq_len * seq_len)
    pid_i = (pid // seq_len) % seq_len
    pid_j = pid % seq_len
    pid_d = tl.arange(0, dim)

    offs_b = pid_b * batch_size
    offs_i = pid_i * seq_len + tl.arange(0, seq_len)
    offs_j = pid_j * seq_len + tl.arange(0, seq_len)
    offs_d = tl.arange(0, dim)

    mask = (offs_i < seq_len) & (offs_j < seq_len) & (offs_d < dim)

    x = tl.load(x_ptr + offs_b + offs_i[:, None] * seq_len + offs_j[None, :] * 1 + offs_d[None, :], mask=mask)
    x = x.to(tl.float32)

    weight = tl.load(weight_ptr + offs_d[None, :], mask=mask)
    weight = weight.to(tl.float32)

    acc = tl.dot(x, weight)
    return acc
