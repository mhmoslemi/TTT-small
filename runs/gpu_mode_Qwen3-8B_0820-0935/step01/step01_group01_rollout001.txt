        x = tl.load(x_ptr + (offs_b * seq_len * seq_len * dim + offs_i * seq_len * dim + offs_j * dim + offs_k), dtype=tl.float32)

        left = x * left_weight
        right = x * right_weight
        left_gate = x * left_gate_weight
        right_gate = x * right_gate_weight

        out = tl.dot(left, right)
        out = out * out_gate_weight
        out = tl.dot(out, to_out_norm_weight) + to_out_norm_bias
        out = out * to_out_weight
