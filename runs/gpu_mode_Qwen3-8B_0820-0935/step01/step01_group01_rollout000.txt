import torch
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Triton implementation of the TriMul module in a highly optimized manner.
    
    This kernel implements the outgoing version of the TriMul operator as described in the AlphaFold3 paper.
    It uses a combination of linear projections, gating mechanisms, and pairwise summation operations.
    The kernel is optimized for the L40S architecture with 142 SMs and 96 MB L2 cache.
    The final output is cast to float32 to match the expected output format.
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Extract weights from the dictionary
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Prepare tensor shapes and strides
    B, N, N, C = batch_size, seq_len, seq_len, dim
    num_blocks = triton.cdiv(N, 128)

    # Allocate output tensor
    output = torch.zeros((B, N, N, hidden_dim), device=input_tensor.device, dtype=torch.float32)

    # Kernel to compute TriMul
    @triton.jit
    def trit_kernel(
        input_ptr, mask_ptr, weights, output_ptr,
        B: tl.constexpr, N: tl.constexpr, C: tl.constexpr,
        W1: tl.constexpr, W2: tl.constexpr, W3: tl.constexpr, W4: tl.constexpr, W5: tl.constexpr, W6: tl.constexpr,
        W7: tl.constexpr, W8: tl.constexpr, W9: tl.constexpr, W10: tl.constexpr, W11: tl.constexpr, W12: tl.constexpr,
        BLOCK: tl.constexpr
    ):
        pid = tl.program_id(0)
        off_b = pid // num_blocks
        off_m = (pid % num_blocks) * BLOCK + tl.arange(0, BLOCK)
        off_n = tl.arange(0, BLOCK)
        off_k = tl.arange(0, BLOCK)

        # Load input and mask
        i = off_b
        m = off_m
        n = off_n
        k = off_k

        # Compute the base indices
        m_idx = i * N + m
        n_idx = i * N + n
        k_idx = i * N + k

        # Calculate the strides for input, mask, and output
        input_strides = (N * N * C, N * C, C, 1)
        mask_strides = (N * N, N, 1, 1)
        output_strides = (N * N * hidden_dim, N * hidden_dim, hidden_dim, 1)

        # Load weights
        weight1 = weights[0]
        weight2 = weights[1]
        weight3 = weights[2]
        weight4 = weights[3]
        weight5 = weights[4]
        weight6 = weights[5]
        weight7 = weights[6]
        weight8 = weights[7]
        weight9 = weights[8]
        weight10 = weights[9]
        weight11 = weights[10]
        weight12 = weights[11]

        # Initialize intermediate values
        left = tl.zeros((BLOCK, C), dtype=tl.float32)
        right = tl.zeros((BLOCK, C), dtype=tl.float32)
        left_gate = tl.zeros((BLOCK, C), dtype=tl.float32)
        right_gate = tl.zeros((BLOCK, C), dtype=tl.float32)
        out_gate = tl.zeros((BLOCK, C), dtype=tl.float32)
        acc = tl.zeros((BLOCK, hidden_dim), dtype=tl.float32)

        # Compute left and right projections
        for k in range(N):
            input_off = i * N * N * C + m * N * C + k * C
            mask_off = i * N * N + m * N + k
            left_k = tl.load(input_ptr + input_off, mask=(k < N), other=0.0)
            right_k = tl.load(input_ptr + input_off, mask=(k < N), other=0.0)
            left_k = left_k * tl.load(mask_ptr + mask_off, mask=(k < N), other=0.0)
            right_k = right_k * tl.load(mask_ptr + mask_off, mask=(k < N), other=0.0)

            left_k = tl.dot(left_k, weight1)
            right_k = tl.dot(right_k, weight2)

            left_k = left_k * tl.load(weight3, mask=(k < hidden_dim), other=0.0)
            right_k = right_k * tl.load(weight4, mask=(k < hidden_dim), other=0.0)

            left_k = left_k * tl.load(weight5, mask=(k < hidden_dim), other=0.0)
            right_k = right_k * tl.load(weight6, mask=(k < hidden_dim), other=0.0)

            left = left + left_k
            right = right + right_k

        # Compute output
        for k in range(N):
            left_k = tl.load(input_ptr + input_off, mask=(k < N), other=0.0)
            left_k = left_k * tl.load(weight7, mask=(k < hidden_dim), other=0.0)
            left_k = left_k * tl.load(weight8, mask=(k < hidden_dim), other=0.0)
            left = left + left_k

            right_k = tl.load(input_ptr + input_off, mask=(k < N), other=0.0)
            right_k = right_k * tl.load(weight9, mask=(k < hidden_dim), other=0.0)
            right_k = right_k * tl.load(weight10, mask=(k < hidden_dim), other=0.0)
            right = right + right_k

            out_k = tl.dot(left, right, weight11, weight12)
            acc = acc + out_k

        # Apply output gate
        out_gate = tl.load(weight11, mask=(k < hidden_dim), other=0.0)
        out_gate = out_gate * tl.load(weight12, mask=(k < hidden_dim), other=0.0)
        out = out * out_gate

        # Store output
        output_off = i * N * N * hidden_dim + m * N * hidden_dim + n * hidden_dim
        tl.store(output_ptr + output_off, out, mask=(m < N) & (n < N))

    num_blocks = triton.cdiv(seq_len, 128)
    grid = (num_blocks,)

    # Prepare weights for the kernel
    weights = [
        norm_weight, norm_bias,
        left_proj_weight, right_proj_weight,
        left_gate_weight, right_gate_weight,
        out_gate_weight, to_out_norm_weight, to_out_norm_bias, to_out_weight
    ]

    trit_kernel[grid](
        input_tensor,
        mask,
        weights,
        output,
        batch_size, seq_len, dim,
        norm_weight.shape[0], norm_weight.shape[1],
        left_proj_weight.shape[0], left_proj_weight.shape[1],
        left_gate_weight.shape[0], left_gate_weight.shape[1],
        out_gate_weight.shape[0], to_out_norm_weight.shape[0],
        to_out_norm_bias.shape[0], to_out_weight.shape[0],
        128
    )

    return output
