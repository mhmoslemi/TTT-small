import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any
import math

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Optimized Triton kernel implementation of the outgoing Triangle Multiplicative Update (TriMul) module.
    This version of TriMul is designed for high-performance on NVIDIA L40S GPUs using Triton's capabilities.
    It processes the forward pass of the TriMul operation efficiently by leveraging kernel fusion and memory access optimizations.
    
    Input:
    - data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
        - input: Input tensor of shape [bs, seq_len, seq_len, dim]
        - mask: Mask tensor of shape [bs, seq_len, seq_len]
        - weights: Dictionary containing model weights
        - config: Dictionary containing model configuration parameters
    
    Output:
    - output: Processed tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Allocate result
    output = torch.empty_like(input_tensor)

    # Define Triton kernel
    @triton.jit
    def tri_mul_kernel(
        input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr,
        left_gate_ptr, right_gate_ptr, out_gate_ptr,
        to_out_norm_weight_ptr, to_out_weight_ptr,
        out_ptr,
        batch_size: tl.constexpr, seq_len: tl.constexpr, dim: tl.constexpr,
        hidden_dim: tl.constexpr, BLOCK: tl.constexpr
    ):
        # Compute grid dimensions
        pid = tl.program_id(0)
        pid_m = pid // (seq_len * seq_len)
        pid_i = (pid % (seq_len * seq_len)) // seq_len
        pid_j = (pid % (seq_len * seq_len)) % seq_len

        # Load input and mask for this batch
        batch = pid_m
        i = pid_i
        j = pid_j

        # Load mask for (i, j)
        mask_vals = tl.load(mask_ptr + batch * seq_len * seq_len + i * seq_len + j, mask=1, other=0.0)

        # Load input values at (i, j)
        input_vals = tl.load(input_ptr + batch * seq_len * seq_len * dim + i * seq_len * dim + j * dim, mask=1, other=0.0)

        # Apply norm
        norm_values = input_vals * norm_weight + norm_bias

        # Compute left and right projections
        left_proj = tl.dot(norm_values, left_proj_weight.T)
        right_proj = tl.dot(norm_values, right_proj_weight.T)

        # Apply mask
        left_proj = left_proj * mask_vals
        right_proj = right_proj * mask_vals

        # Compute gates
        left_gate = tl.dot(norm_values, left_gate_weight.T)
        left_gate = tl.sigmoid(left_gate)
        left_proj = left_proj * left_gate

        right_gate = tl.dot(norm_values, right_gate_weight.T)
        right_gate = tl.sigmoid(right_gate)
        right_proj = right_proj * right_gate

        # Compute output
        out = tl.zeros((hidden_dim,), dtype=tl.float32)
        for k in range(seq_len):
            # Load left and right projections for k-th position
            left_k = tl.load(left_proj_ptr + batch * seq_len * hidden_dim + i * hidden_dim + k * hidden_dim)
            right_k = tl.load(right_proj_ptr + batch * seq_len * hidden_dim + j * hidden_dim + k * hidden_dim)

            # Multiply and accumulate
            out += tl.dot(left_k, right_k)

        # Apply output gate
        out_gate = tl.dot(norm_values, out_gate_weight.T)
        out_gate = tl.sigmoid(out_gate)
        out = out * out_gate

        # Apply to_out_norm
        out = tl.dot(out, to_out_norm_weight.T) + to_out_norm_bias

        # Apply to_out
        out = tl.dot(out, to_out_weight.T)

        # Store result
        tl.store(out_ptr + batch * seq_len * seq_len * dim + i * seq_len * dim + j * dim, out)

    # Launch kernel
    grid = (batch_size * seq_len * seq_len,)
    BLOCK = 128
    tri_mul_kernel[grid](
        input_tensor,
        mask,
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
        to_out_norm_weight,
        to_out_weight,
        output,
        batch_size,
        seq_len,
        dim,
        hidden_dim,
        BLOCK
    )

    # Ensure output is in float32
    return output.to(torch.float32)
