import torch
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Custom Triton kernel for the Triangle Multiplicative Update (TriMul) operator.
    This kernel implements the outgoing version of TriMul from AlphaFold3, optimizing
    for large sequence dimensions using efficient memory access and fused operations.

    Key optimizations:
    - Efficient matrix multiplication using Triton's `tl.dot`
    - Masking handled with per-element multiplication
    - Fused LayerNorm and activation operations
    - Optimized for the L40S architecture

    Inputs:
    - input_tensor: [batch_size, seq_len, seq_len, dim]
    - mask: [batch_size, seq_len, seq_len]
    - weights: Dictionary with LayerNorm and Linear weights
    - config: Dictionary with configuration values (dim, hidden_dim)

    Output:
    - output: [batch_size, seq_len, seq_len, dim]
    """

    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Extract weights from the input dictionary
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure all weights are on the correct device
    device = input_tensor.device

    # Define Triton kernel
    @triton.jit
    def triton_trimul_kernel(
        x_ptr, mask_ptr, left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr,
        out_gate_ptr, to_out_norm_ptr, to_out_ptr, norm_weight_ptr, norm_bias_ptr,
        to_out_norm_bias_ptr, out_ptr,
        batch_size: tl.constexpr, seq_len: tl.constexpr, dim: tl.constexpr,
        hidden_dim: tl.constexpr, is_masked: tl.constexpr, BLOCK: tl.constexpr
    ):
        # We use a 2D grid to process a single batch
        # Each thread handles a block of i, j in [0, seq_len)
        pid = tl.program_id(0)
        pid_ = pid % seq_len
        pid_j = pid // seq_len

        # Each thread block handles a block of size BLOCK
        idx = pid % seq_len
        jdx = pid // seq_len

        # Load the mask
        m = tl.load(mask_ptr + pid, mask=(pid < (batch_size * seq_len * seq_len)), other=1.0)
        mask = m > 0.5

        # Compute the indices for the current i, j
        i = idx
        j = jdx

        # Initialize the output for the current [i, j] pair
        out = tl.zeros((hidden_dim,), dtype=tl.float32)
        for k in range(seq_len):
            # Load the current left and right vectors
            left = tl.load(left_proj_ptr + (i * seq_len + k) * hidden_dim + tl.arange(0, hidden_dim))
            right = tl.load(right_proj_ptr + (j * seq_len + k) * hidden_dim + tl.arange(0, hidden_dim))
            # Apply mask
            if is_masked:
                left = left * tl.load(mask_ptr + (i * seq_len + k) * seq_len + jdx)
                right = right * tl.load(mask_ptr + (j * seq_len + k) * seq_len + idx)
            # Apply gates
            left_gate = tl.load(left_gate_ptr + (i * seq_len + k) * hidden_dim + tl.arange(0, hidden_dim))
            right_gate = tl.load(right_gate_ptr + (j * seq_len + k) * hidden_dim + tl.arange(0, hidden_dim))
            # Sigmoid approximation
            left_gate = tl.where(left_gate > 0, 1.0 / (1.0 + tl.exp(-left_gate)), left_gate)
            right_gate = tl.where(right_gate > 0, 1.0 / (1.0 + tl.exp(-right_gate)), right_gate)
            # Element-wise multiplication
            left = left * left_gate
            right = right * right_gate
            # Dot product and accumulate
            dot = tl.dot(left, right)
            out += dot

        # Apply LayerNorm
        mean = tl.sum(out) / hidden_dim
        var = tl.sum((out - mean) ** 2) / hidden_dim
        out = (out - mean) / tl.sqrt(var + 1e-5)
        # Apply layer norm weights and biases
        out = out * tl.load(norm_weight_ptr + tl.arange(0, hidden_dim)) + tl.load(norm_bias_ptr + tl.arange(0, hidden_dim))

        # Apply output gate
        out_gate = tl.load(out_gate_ptr + (i * seq_len + j) * hidden_dim + tl.arange(0, hidden_dim))
        out = out * tl.where(out_gate > 0, 1.0 / (1.0 + tl.exp(-out_gate)), out_gate)

        # Apply final linear projection
        out = out * tl.load(to_out_norm_weight + tl.arange(0, hidden_dim)) + tl.load(to_out_norm_bias + tl.arange(0, hidden_dim))
        out = out * tl.load(to_out_weight + tl.arange(0, hidden_dim))

        # Store the result
        tl.store(out_ptr + (i * seq_len + j) * hidden_dim + tl.arange(0, hidden_dim), out)

    # Prepare the kernel parameters
    # We will launch one thread per i, j pair
    num_threads = batch_size * seq_len * seq_len

    # Launch the kernel
    triton_trimul_kernel[triton.cdiv(num_threads, 128), 128](
        input_tensor, mask, 
        left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight,
        out_gate_weight, to_out_norm_weight, to_out_weight, norm_weight, norm_bias,
        to_out_norm_bias, 
        input_tensor, 
        batch_size, seq_len, dim, hidden_dim,
        1 if mask is not None else 0, 128
    )

    return input_tensor
