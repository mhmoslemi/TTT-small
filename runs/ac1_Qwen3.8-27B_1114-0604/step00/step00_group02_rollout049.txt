"""
Gradient-based optimizer to minimize f(a) = 2n*max_k conv(a,a)[k] / (sum a)^2.

Approach:
  * Fast self-convolution via real FFT (length 2n-1).
  * Exact analytic gradient with respect to each variable, using the
    observed top-k convolution peaks with exponential soft-max weights to
    approximate the max (smooth, low-noise).
  * Adam optimizer with adaptive learning rate, non-negativity clipping,
    and periodic restarts that perturb from the best incumbent to escape
    local minima.
  * Three phases over the wall-clock budget:
      Phase 1 (explore): many diverse initializations, moderate steps.
      Phase 2 (refine):  small steps + larger top-k smoothing on best.
      Phase 3 (polish):  very small steps on the global best.
  * Validity always checked with the authoritative evaluate_sequence().
"""

import numpy as np
import time


# ----------------------------------------------------------------------------
# Convolution and scoring helpers

def fast_self_conv(a):
    """Linear self-convolution of 1-D array a (length n), via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def score_from_conv(n, c, S):
    """Objective value given n, convolution c and sum S."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def topk_indices(c, k):
    """Returns (indices, values, weights) for the k largest peaks of c with
    exponential soft-max weights (heavily concentrated on the top peak)."""
    k_act = k if k < len(c) else len(c)
    if k_act <= 0:
        return np.array([]), np.array([]), np.array([])
    if k_act == 1:
        idx = np.array([int(np.argmax(c))])
        vals = c[idx]
        wts = np.array([1.0])
        return idx, vals, wts
    idx = np.argpartition(c, -k_act)[-k_act:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = float(vals[0])
    if top > 1e-12:
        temp = 0.12 * top
        wts = np.exp((vals - top) / temp)
        wts = wts / wts.sum()
    else:
        wts = np.ones(k_act) / k_act
    return idx, vals, wts


def grad_topk(a, c, k):
    """Analytic gradient of the weighted-peak surrogate.

    Returns (grad, M_weighted, S).
    For a peak at k, d conv(k)/d a[j] = 2*a[k-j] for j in
    [max(0, k-n+1), min(k, n-1)] and 0 otherwise, i.e. a reversed slice.
    """
    n = len(a)
    S = float(a.sum())
    idx, vals, wts = topk_indices(c, k)
    grad_c = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        # conv(ki) pairs a[j] with a[ki-j]; reversed slice gives a[ki-j]
        grad_c[j_start:j_end + 1] += wi * 2.0 * a[j_start:j_end + 1][::-1]
    M = float(np.dot(vals, wts))
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M / S3)
    return grad, M, S


def adam_update(a, g, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """In-place Adam step."""
    t1 = b1 ** t
    t2 = b2 ** t
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    mh = m / (1.0 - t1)
    vh = v / (1.0 - t2)
    a -= lr * mh / (np.sqrt(vh) + eps)


def make_adam_state(n):
    m = np.zeros(n)
    v = np.zeros(n)
    return m, v


def clamp(a):
    """Enforce the evaluator's domain [0, 1000] and a positive sum."""
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 1e-3:
        a = np.clip(a, 0.0, 1000.0) + 1e-3 / a.size
    return a

def init_uniform(n):
    return np.ones(n)


def init_ramp(n, alpha, rng):
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * x - 1.0)
    return np.clip(a, 0.01, 1000.0)


def init_quad(n, alpha):
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * x - 1.0) ** 2
    return np.clip(a, 0.01, 1000.0)


def init_cos(n, freq, alpha):
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * x)
    return np.clip(a, 0.01, 1000.0)


def init_blocks(n, nb, rng):
    per = n // nb
    vals = 0.4 + rng.random(nb)
    a = np.repeat(vals, per)
    if len(a) < n:
        a = np.concatenate([a, vals[:n - len(a)]])
    a = a[:n]
    return np.clip(a, 0.01, 1000.0)

    a = 0.6 + 0.4 * rng.random(n)
    if gap > 2:
        a[::gap] = 0.0
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 1e-3:
        a += 0.5
    return a

    inits = []
    inits.append(("uniform", init_uniform(n)))
    for al in (0.4, 0.8, 1.5):
        inits.append(("ramp_%s" % al, init_ramp(n, al, rng)))
    for al in (0.8, 1.5, 3.0):
        inits.append(("quad_%s" % al, init_quad(n, al)))
    for fq, al in ((1, 0.6), (2, 0.6), (3, 0.4), (5, 0.3)):
        inits.append(("cos_%d_%s" % (fq, al), init_cos(n, fq, al)))

        inits.append(("gapped_%d" % gp, init_gapped(n, gp, rng)))
    inits.append(("rand", np.clip(0.3 + rng.random(n), 0.01, 1000.0)))
    return inits
