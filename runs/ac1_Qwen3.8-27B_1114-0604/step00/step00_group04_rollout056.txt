```
def _weights(c, q, frac, uni):
    m = float(np.max(c))
    if m <= 0 or not np.isfinite(m):
        return np.full(c.shape, 1.0/c.size)
    eps = np.finfo(np.float64).tiny
    logc = np.log(c + eps)
    logm = np.log(m + eps)
    prevailing = np.exp(np.clip(q * (logc - logm), -70.0, 0.0))
    sp = float(prevailing.sum())
    if sp <= 0 or not np.isfinite(sp):
```

```python
"""
Search for nonnegative sequences minimizing
    2*n*max(convolve(a,a)) / (sum(a))**2.

Strategy:
  1. Grid-search a strong parametric family
         p_i proportional to x_i^{-1/2} * (1 + alpha*x_i + beta*x_i^2),
     over several lengths (around 1k-8k) and parameters.
  2. Use a time-boxed exponentiated-gradient local search to polish the
     best usable-length candidate.
  3. Always keep and return the best exact-scoring sequence found.
"""

import numpy as np
import time


def _normalize_nonnegative(a):
    """Return a float64 1D array with sum 1 and no negative entries, or None."""
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return None
    a = a.copy()
    bad = ~np.isfinite(a)
    if np.any(bad):
        a[bad] = 0.0
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return None
    return a / s


def _score_prob(a):
    """Score assuming a is already normalized to sum 1."""
    if a is None:
        return float("inf")
    if a.size == 0:
        return float("inf")
    c = np.convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * a.size * m)


def _score_any(a):
    """Score for arbitrary nonnegative array using the exact objective."""
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return float("inf")
    a = a.copy()
    bad = ~np.isfinite(a)
    if np.any(bad):
        a[bad] = 0.0
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return float("inf")
    c = np.convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * a.size * m / (s * s))


def make_family_sequence(n, alpha, beta, shift):
    """
    Build
        p_i ∝ ((i+shift)/n)^(-1/2) * (1 + alpha*x_i + beta*x_i^2),
    normalized to sum 1.  If the polynomial becomes nonpositive anywhere,
    clip to zero; if all-zero, return None.
    """
    if n <= 0:
        return None
    i = np.arange(n, dtype=np.float64)
    x = (i + float(shift)) / float(n)
    w = x ** (-0.5)
    poly = 1.0 + alpha * x + beta * x * x
    w *= poly
    w = np.maximum(w, 0.0)
    s = float(w.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return None
    return w / s


def _default_sequence():
    """A safe strong family member."""
    p = make_family_sequence(4096, 2.0, 0.0, 0.5)
    if p is None:
        p = np.ones(1024, dtype=np.float64)
        p /= 1024.0
    return p


def _grid_search(budget_s):
    """Cheap global-ish search over family lengths and parameters."""
    start = time.monotonic()
    best = None
    best_score = float("inf")

    simple_configs = [
        (8192, (2.0,)),
        (6144, (2.0,)),
        (4096, (1.8, 2.0, 2.2)),
        (3072, (1.8, 2.0, 2.2)),
        (2048, (1.5, 1.8, 2.0, 2.2)),
        (1024, (1.0, 1.5, 1.8, 2.0, 2.2)),
    ]

    stop = False
    for n, alphas in simple_configs:
        if stop:
            break
        for alpha in alphas:
            if time.monotonic() - start > max(1.0, budget_s * 0.55):
                stop = True
                break
            p = make_family_sequence(n, alpha, 0.0, 0.5)
            if p is None:
                continue
            sc = _score_prob(p)
            if sc < best_score:
                best = p.copy()
                best_score = sc
                print("grid simple: n=%d alpha=%.3f score=%.8f" % (n, alpha, sc))

    if time.monotonic() - start < budget_s * 0.85:
        fine_stop = False
        for n in (2048, 1024):
            if fine_stop:
                break
            for alpha in (1.7, 1.8, 1.9, 2.0, 2.1):
                if time.monotonic() - start > budget_s * 0.98:
                    fine_stop = True
                    break
                for beta in (-0.30, -0.15, 0.0, 0.15):
                    if time.monotonic() - start > budget_s:
                        fine_stop = True
                        break
                    p = make_family_sequence(n, alpha, beta, 0.5)
                    if p is None:
                        continue
                    sc = _score_prob(p)
                    if sc < best_score:
                        best = p.copy()
                        best_score = sc
                        print("grid fine: n=%d alpha=%.3f beta=%.3f score=%.8f"
                              % (n, alpha, beta, sc))

    if best is None:
        best = _default_sequence()
        best_score = _score_prob(best)

    return best, best_score


def _block_perturb(p, rng, sigma):
    """Random multiplicative perturbation on a contiguous block, renormalized."""
    n = p.size
    min_w = max(2, int(n // 200))
    max_w = max(min_w, int(n // 8))
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    q = p.copy()
    q[start:start + width] *= np.exp(rng.normal(0.0, sigma))
    s = float(q.sum())
    if s <= 0.0 or not np.isfinite(s):
        return p.copy()
    return q / s


def _weights(c, q, frac, uni):
    """
    Build nonnegative weights for active/near-active convolution positions.
    q controls concentration; uni mixes in a uniform weight over positions
    within frac of the maximum to avoid degenerate single-active gradients.
    """
    m = float(np.max(c))
    if m <= 0.0 or not np.isfinite(m):
        return np.full(c.shape, 1.0 / c.size, dtype=np.float64)

    eps = np.finfo(np.float64).tiny
    logc = np.log(c + eps)
    logm = np.log(m + eps)

    prevailing = np.exp(np.clip(q * (logc - logm), -70.0, 0.0))
    sp = float(prevailing.sum())
    if sp <= 0.0 or not np.isfinite(sp):
        prevailing = np.ones_like(c, dtype=np.float64)
        sp = float(c.size)
    prevailing = prevailing / sp

    active = c >= (1.0 - float(frac)) * m
    aw = np.zeros_like(c, dtype=np.float64)
    cnt = int(np.count_nonzero(active))

    if cnt <= 0:
        k = max(1, min(c.size, int(0.01 * c.size) + 1))
        if k >= c.size:
            aw[:] = 1.0
            cnt = c.size
        else:
            order = np.argsort(c)
            aw[order[-k:]] = 1.0
            cnt = k

    aw = aw / float(cnt)
    return (1.0 - float(uni)) * prevailing + float(uni) * aw


def _local_search(p0, time_budget_s, seed=123):
    """
    Exponentiated-gradient line search on the simplex, using a smoothed
    maximum of the convolution.  The exact score is used for accepted moves.
    """
    p = _normalize_nonnegative(p0)
    if p is None:
        return p0, float("inf")

    best = p.copy()
    best_score = _score_prob(p)
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    eta = 0.1
    q = 6.0
    frac = 0.05
    uni = 0.20
    no_improve = 0
    iteration = 0
    max_iters = 100000

    while iteration < max_iters and (time.monotonic() - start) < float(time_budget_s):
        iteration += 1

        c = np.convolve(p, p)
        w = _weights(c, q, frac, uni)

        # Gradient of weighted sum of c_k.
        g = 2.0 * np.correlate(w, p, mode="valid")
        d = g - float(np.mean(g))
        scale = float(np.max(np.abs(d)))

        if not np.isfinite(scale) or scale < 1e-18:
            break

        d = d / scale

        e1 = max(1e-4, eta * 0.5)
        e2 = max(1e-4, eta)
        e3 = min(50.0, eta * 2.0)

        if e1 >= e2:
            e1 = max(1e-5, e2 * 0.25)
        if e3 <= e2:
            e3 = min(50.0, max(2.0 * e2, 1e-3))

        improved = False
        best_try = best_score
        best_try_p = best
        best_e = e2

        for e in (e1, e2, e3):
            ex = -e * d
            mx = float(np.max(ex))
            if np.isfinite(mx):
                ex = ex - mx
            else:
                ex = np.zeros_like(ex)
            ex = np.clip(ex, -70.0, 0.0)

            p_try = p * np.exp(ex)
            s = float(p_try.sum())
            if s <= 0.0 or not np.isfinite(s):
                continue

            p_try = p_try / s
            sc = _score_prob(p_try)
            if sc < best_try:
                best_try = sc
                best_try_p = p_try
                best_e = e

        if best_try < best_score - 1e-15:
            best = best_try_p.copy()
            best_score = best_try
            p = best.copy()
            no_improve = 0
            improved = True

            if best_e == e3 and e3 < 50.0:
                eta = e3
            elif best_e == e1 and e1 > 1e-3:
                eta = e1

        if not improved:
            no_improve += 1

            if no_improve == 1 or (no_improve % 10 == 0):
                sigma = 0.04 + 0.02 * min(4.0, float(no_improve) / 10.0)
                for _ in range(8):
                    p_try = _block_perturb(p, rng, sigma)
                    sc = _score_prob(p_try)
                    if sc < best_score - 1e-15:
                        best = p_try.copy()
                        best_score = sc
                        p = best.copy()
                        no_improve = 0
                        break

            if no_improve > 40:
                if q < 16.0:
                    q *= 1.3
                elif frac < 0.20:
                    frac = min(0.20, frac * 1.4)
                elif uni < 0.50:
                    uni = min(0.50, uni + 0.10)
                elif eta < 10.0:
                    eta = min(10.0, eta * 2.0)
                elif no_improve > 300:
                    break

            if eta > 5e-4:
                eta = max(5e-4, eta * 0.75)

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Return the best nonnegative sequence found within the time budget."""
    total_start = time.monotonic()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 2400.0
    deadline = total_start + max(1.0, total_budget - 20.0)

    grid_budget = max(1.0, min(120.0, (deadline - time.monotonic()) * 0.15))
    best, best_score = _grid_search(grid_budget)

    # Include the provided reference if it is reasonably sized.
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.ndim == 1 and 32 <= h.size <= 8192:
            hp = _normalize_nonnegative(h)
            if hp is not None:
                sc = _score_prob(hp)
                if np.isfinite(sc) and sc < best_score:
                    best = hp.copy()
                    best_score = sc
                    print("height_sequence_1 normalized score=%.8f" % sc)
    except Exception:
        pass

    # Polish a candidate.  If the best candidate is very long, use a
    # fast 4096-length candidate for local search and keep the longer one
    # unless the polished shorter one beats it.
    remaining = deadline - time.monotonic()
    if remaining > 30.0:
        if best is not None and best.size <= 4096:
            local_budget = min(900.0, remaining * 0.60)
            local_start = best.copy()
        else:
            local_budget = min(600.0, remaining * 0.50)
            local_start = make_family_sequence(4096, 2.0, 0.0, 0.5)
            if local_start is None:
                local_start = _default_sequence()

        if local_start is not None and local_budget > 2.0:
            local_candidate, local_score = _local_search(
                local_start,
                local_budget,
                seed=(int(seed) if np.isfinite(seed) else 42) + 1
            )
            if np.isfinite(local_score) and local_score < best_score:
                best = local_candidate
                best_score = local_score
                print("local improved score=%.8f n=%d" % (best_score, best.size))

    if best is None or not np.isfinite(best_score):
        best = _default_sequence()
        best_score = _score_prob(best)

    # Final consistency check against the official evaluator if available.
    try:
        official = float(evaluate_sequence([float(x) for x in best]))
        if np.isfinite(official):
            best_score = official
        else:
            best = _default_sequence()
            best_score = _score_prob(best)
    except Exception:
        pass

    return [float(x) for x in best]
```
