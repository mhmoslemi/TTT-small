import cvxpy as cp

def compute_convolution(a):

def score_of(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a ** 2)

def solve_lp_direction(f, M):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)[k] <= M for all k."""
    n = len(f)
    b = cp.Variable(n)
    
    # Objective: maximize sum(b)
    objective = cp.Maximize(cp.sum(b))
    
    # Constraints
    constraints = [b >= 0]
    
    # Convolution constraints
    for k in range(2 * n - 1):
        # (f * b)[k] = sum_j f[j] * b[k-j] for valid j
        # Valid j: 0 <= j < n and 0 <= k-j < n
        # So j ranges from max(0, k-n+1) to min(n-1, k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            # (f*b)[k] = sum_{j=j_min}^{j_max} f[j] * b[k-j]

            # (f*b)[k] = sum_{i=k-j_max}^{k-j_min} f[k-i] * b[i]

            for j in range(j_min, j_max + 1):
                i = k - j
                coeffs[i] = f[j]
            constraints.append(coeffs @ b <= M)
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CLARABEL)
    
    if b.value is None:
        return None
    return np.maximum(b.value, 0)
