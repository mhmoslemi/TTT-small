"""
LP-based water-filling iterative optimization for minimizing 2n*max(a*a)/sum(a)^2.
Uses multiple starting points, sequence lengths, and LP-descent iterations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix(n, f):
    """Build sparse (2n-1) x n matrix where A[k,i] = f[k-i] if 0<=k-i<n else 0."""
    lens = np.minimum(n, 2 * n - 1 - np.arange(n))
    total = int(lens.sum())
    cols = np.repeat(np.arange(n, dtype=np.int32), lens)
    cum = np.concatenate([[0], np.cumsum(lens[:-1])])
    idx = np.repeat(np.arange(len(lens)), lens)
    offsets = np.arange(total, dtype=np.int32) - cum[idx]
    rows = cols + offsets
    data = f[offsets]
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a):
    """Solve max 1^T b s.t. conv_matrix(a) @ b <= max(a*a), b >= 0."""
    n = len(a)
    c = np.convolve(a, a)
    M = float(np.max(c))
    if M <= 0:
        return a.copy()
    A = build_conv_matrix(n, a)
    obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return np.maximum(res.x, 0.0)


def compute_score(a):
    """Compute objective 2n*max(a*a)/sum(a)^2 for numpy array a."""
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    M = float(np.max(c))
    return 2.0 * len(a) * M / (S * S)


def lp_refine(a0, max_iters, time_budget, seed):
    """Run LP-based refinement iterations starting from a0."""
    rng = np.random.default_rng(seed)
    a = np.array(a0, dtype=np.float64)
    a = np.maximum(a, 0.0)
    S = float(np.sum(a))
    if S < 1e-10:
        return a, compute_score(a)
    a = a * (1.0 / S)  # normalize to sum=1 (scale-invariant objective)

    best_a = a.copy()
    best_score = compute_score(a)
    t_step = 0.5
    start = time.monotonic()

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        M_old = float(np.max(np.convolve(a, a)))

        b = solve_lp(a)
        if b is None:
            break

        S_b = float(np.sum(b))
        if S_b < 1e-10:
            break

        b = b / S_b  # normalize b to sum=1

        # Try step sizes from large to small
        accepted = False
        for t in [t_step, 0.3, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * b
            a_new = np.maximum(a_new, 0.0)
            S_new = float(np.sum(a_new))
            if S_new < 1e-10:
                continue
            a_new = a_new / S_new
            score_new = compute_score(a_new)
            if score_new < best_score - 1e-12:
                a = a_new
                best_a = a.copy()
                best_score = score_new
                t_step = t
                accepted = True
                break
        if not accepted:
            # Try even smaller step
            a_new = (1.0 - 0.01) * a + 0.01 * b
            a_new = np.maximum(a_new, 0.0)
            S_new = float(np.sum(a_new))
            if S_new > 1e-10:
                a_new = a_new / S_new
                score_new = compute_score(a_new)
                if score_new < best_score:
                    a = a_new
                    best_a = a.copy()
                    best_score = score_new
                    t_step = 0.01
                else:
                    break
            else:
                break

        if it % 20 == 0:
            print(f"  iter={it}, score={best_score:.6f}, M={M_old:.6f}")

    return best_a, best_score


def random_perturb(a, rng, strength):
    """Apply a random multiplicative perturbation."""
    n = len(a)
    block_size = max(1, n // 50)
    n_blocks = n // block_size + 1
    pert = np.ones(n)
    for _ in range(rng.integers(1, n_blocks + 1)):
        start = int(rng.integers(0, n - block_size))
        end = min(n, start + int(rng.integers(block_size, block_size * 3)))
        pert[start:end] *= np.exp(rng.normal(0, strength))
    a_new = np.maximum(a * pert, 0.0)
    S = np.sum(a_new)
    if S < 1e-10:
        return a.copy()
    return a_new / S


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    LP-based water-filling search for minimizing 2n*max(a*a)/sum(a)^2.
    Tries multiple sequence lengths and starting patterns.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 10.0)

    best_a = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(a, 0.0)
        S = float(np.sum(a))
        if S < 0.01:
            return
        # Clip to valid range
        a = np.clip(a, 0.0, 1000.0)
        score = compute_score(a)
        if score < best_score - 1e-9:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST: score={score:.6f}, n={len(a)}")

    # Also check the existing sequence
    try:
        existing = np.asarray(height_sequence_1, dtype=np.float64)
        existing = np.clip(existing, 0.0, 1000.0)
        try_update(existing)
    except Exception:
        pass

    print(f"Initial best score: {best_score:.6f}")

    # Define search configurations: (n, init_type, time_alloc)
    configs = [
        (1024, 'alt', 300),
        (2048, 'alt', 300),
        (1024, 'cos4', 200),
        (2048, 'cos8', 200),
        (512, 'alt', 150),
        (1024, 'existing', 200),
        (4096, 'alt', 300),
        (1024, 'rand', 150),
        (2048, 'existing', 200),
        (512, 'cos4', 100),
    ]

    # Shuffle configs slightly to randomize
    for cfg_idx, (n, init_type, t_alloc) in enumerate(configs):
        if time.monotonic() > deadline:
            break

        print(f"\nConfig {cfg_idx}: n={n}, init={init_type}, budget={t_alloc}s")

        # Generate initial a
        if init_type == 'alt':
            a0 = np.ones(n)
            a0[1::2] = 0.5
        elif init_type == 'cos4':
            t = np.arange(n)
            a0 = 1.0 + 0.4 * np.cos(2 * np.pi * t / 4.0)
            a0 = np.maximum(a0, 0.01)
        elif init_type == 'cos8':
            t = np.arange(n)
            a0 = 1.0 + 0.4 * np.cos(2 * np.pi * t / 8.0)
            a0 = np.maximum(a0, 0.01)
        elif init_type == 'existing':
            try:
                ex = np.asarray(height_sequence_1, dtype=np.float64)
                if len(ex) >= n:
                    a0 = ex[:n].copy()
                else:
                    reps = int(np.ceil(n / len(ex)))
                    a0 = np.tile(ex, reps)[:n]
            except Exception:
                a0 = np.ones(n)
                a0[1::2] = 0.5
        elif init_type == 'rand':
            a0 = np.abs(rng.normal(1.0, 0.5, n)) + 0.1
        else:
            a0 = np.ones(n)

        a0 = np.clip(a0, 0.0, 1000.0)
        S0 = float(np.sum(a0))
        if S0 < 1e-10:
            continue
        a0 = a0 / S0

        try_update(a0)

        # Run LP refinement with allocated time
        result_a, result_score = lp_refine(a0, max_iters=1000, time_budget=t_alloc, seed=seed + cfg_idx)
        try_update(result_a)

        # After LP refinement, try some random perturbations to escape local minima
        if time.monotonic() < deadline:
            a_cur = result_a.copy()
            S_cur = float(np.sum(a_cur))
            if S_cur > 1e-10:
                a_cur = a_cur / S_cur
            for piter in range(50):
                if time.monotonic() > deadline:
                    break
                a_pert = random_perturb(a_cur, rng, 0.1)
                score_pert = compute_score(a_pert)
                if score_pert < compute_score(a_cur):
                    a_cur = a_pert
                    try_update(a_cur)
                elif rng.random() < 0.1:
                    # Occasionally accept worse (simulated annealing)
                    a_cur = a_pert

    # Final phase: use remaining time on the best sequence found
    if best_a is not None and time.monotonic() < deadline:
        print(f"\nFinal refinement phase for best sequence (n={len(best_a)}, score={best_score:.6f})")
        remaining = deadline - time.monotonic()
        result_a, result_score = lp_refine(best_a, max_iters=1000, time_budget=max(0, remaining * 0.9), seed=seed + 999)
        try_update(result_a)

    # Fallback
    if best_a is None:
        best_a = np.ones(1)

    # Final validation using evaluate_sequence
    best_list = [float(x) for x in best_a]
    final_score = evaluate_sequence(best_list)
    print(f"\nFinal: score={final_score:.6f} (local={best_score:.6f}), n={len(best_list)}")

    return best_list
