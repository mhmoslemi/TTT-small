import numpy as np
import time

def compute_convolution(a):
    return np.convolve(a, a)

def compute_score(a):
    n = len(a)
    c = np.convolve(a, a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)

def compute_gradient(a, top_k=5):
    """Compute gradient of f(a) = 2n * max_k c[k] / (sum a)^2 using top-k peaks."""
    n = len(a)
    c = np.convolve(a, a)
    S = a.sum()
    
    k_top = min(top_k, len(c))
    top_idx = np.argpartition(c, -k_top)[-k_top:]
    top_vals = c[top_idx]
    w = top_vals / top_vals.sum()
    
    # Weighted gradient of the "smooth max"
    grad_c = np.zeros(n)
    for k, wi in zip(top_idx, w):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            # dc[k]/da[j] = 2*a[k-j] for j in [j_start, j_end]
            # a[k-j_start : k-j_end+1] reversed gives a[k-j] for j from j_start to j_end
            grad_c[j_start:j_end+1] += wi * 2.0 * a[k - j_start : k - j_end + 1][::-1]
    
    M = float(np.dot(top_vals, w))
    grad_f = (2.0 * n) * (grad_c / (S * S) - 2.0 * M / (S * S * S))
    
    return grad_f, M, S

def adam_step(params, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m[:] = beta1 * m + (1 - beta1) * grad
    v[:] = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    params[:] = params - lr * m_hat / (np.sqrt(v_hat) + eps)
    return m, v

"""
Gradient-based optimization to minimize 2n*max(conv(a,a))/(sum a)^2.
Uses Adam optimizer with analytic gradient from top-k convolution peaks,
multiple restarts with diverse initializations, and adaptive step sizes.
"""

import numpy as np
import time

def fast_self_conv(a):
    """Compute a*a (linear convolution) using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score_from_conv(a, c):
    """Compute score given a and its self-convolution c."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * np.max(c) / (S * S)

def compute_grad_topk(a, c, k=5):
    """Compute gradient using top-k peaks of convolution."""
    n = len(a)
    S = a.sum()
    
    k_actual = min(k, len(c))
    # Get top-k indices
    if k_actual == len(c):
        top_idx = np.arange(len(c))
        top_val = c.copy()
    else:
        top_idx = np.argpartition(c, -k_actual)[-k_actual:]
        top_val = c[top_idx]
        # Sort by value descending
        order = np.argsort(top_val)[::-1]
        top_idx = top_idx[order]
        top_val = top_val[order]
    
    # Soft weights: exponential of scaled values
    # This gives more weight to the largest peaks
    max_val = top_val[0]
    if max_val > 0:
        weights = np.exp((top_val - max_val) / (max_val * 0.1 + 1e-10))
        weights /= weights.sum()
    else:
        weights = np.ones(k_actual) / k_actual
    
    # Compute gradient of weighted sum of peaks
    grad_c = np.zeros(n)
    for i in range(k_actual):
        ki = int(top_idx[i])
        wi = weights[i]
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start <= j_end:
            # grad[j] += wi * 2 * a[ki-j] for j in [j_start, j_end]
            # a[ki-j_start : ki-j_end+1][::-1] gives [a[ki-j_start], a[ki-j_start-1], ..., a[ki-j_end]]
            # which equals [a[j_end], a[j_end-1], ..., a[j_start]] = a[j_start:j_end+1][::-1]
            grad_c[j_start:j_end+1] += wi * 2.0 * a[j_start:j_end+1][::-1]
    
    M_weighted = float(np.dot(top_val, weights))
    S2 = S * S
    S3 = S2 * S
    
    # f = 2n * M / S^2
    # df/da[j] = 2n * (grad_c[j] / S^2 - 2*M / S^3)
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M_weighted / S3)
    
    return grad, M_weighted, S

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimizes 2n*max(conv(a,a))/(sum a)^2 using Adam with top-k peak gradient.
    Multiple initializations, adaptive learning rate, periodic restarts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 10.0
    
    best_score = np.inf
    best_a = None
    
    # Determine sizes to try
    # The given sequence has length 6962, but we can try different sizes
    sizes = [6962]
    if 'height_sequence_1' in dir() and height_sequence_1 is not None:
        sizes[0] = len(height_sequence_1)
    
    # Generate initializations
    def make_inits(n):
        inits = []
        
        # Uniform (baseline, score = 2)
        inits.append(('uniform', np.ones(n)))
        
        # Slightly asymmetric uniform
        x = np.linspace(0, 1, n)
        for alpha in [0.1, 0.3, 0.5, 1.0, 2.0]:
            a = 1.0 + alpha * (2*x - 1)  # linear ramp from 1-alpha to 1+alpha
            a = np.clip(a, 0.01, None)
            inits.append((f'linear_{alpha}', a))
        
        # Quadratic
        for alpha in [0.5, 1.0, 2.0]:
            a = 1.0 + alpha * (2*x - 1)**2
            a = np.clip(a, 0.01, None)
            inits.append((f'quad_{alpha}', a))
        
        # Cosine
        for alpha in [0.5, 1.0]:
            a = 1.0 + alpha * np.cos(2*np.pi*x)

            inits.append((f'cos_{alpha}', a))
