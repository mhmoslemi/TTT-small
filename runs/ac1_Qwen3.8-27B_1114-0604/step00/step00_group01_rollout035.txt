"""
Alternating minimization with log-sum-exp smoothing to minimize
the self-convolution peak: 2n * max(a*a) / (sum(a))^2.
"""

import numpy as np
from scipy.signal import fftconvolve
import time


def project_to_simplex(v):
    """Project vector v onto the probability simplex."""
    n = v.size
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u * ind > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Ensure sum is exactly 1 (numerical fix)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.ones(n) / n
    return w


def compute_conv(a):
    """Compute full self-convolution of a."""
    return fftconvolve(a, a)


def compute_score(a):
    """Compute the evaluation score (lower is better)."""
    n = a.size
    c = compute_conv(a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_c / (S * S)


def inner_step(a, beta, num_iters, base_lr):
    """
    Given fixed a, find b on simplex minimizing logsumexp(beta * (a*b)).
    Returns the best b found.
    """
    n = a.size

        lr = base_lr / (1.0 + 0.1 * i)

        b = b - lr * grad

Minimizes 2n*max(a*a)/(sum(a))^2 via alternating projected gradient descent
with log-sum-exp smoothing and annealing.
"""

import numpy as np
from scipy.signal import fftconvolve
import time


def project_to_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n <= 1:
        return np.array([max(0.0, float(v[0]))]) if n == 1 else v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u * ind > (cssv - 1.0)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 1e-15:
        w /= s
    else:
        w = np.ones(n, dtype=np.float64) / n
    return w


def fast_conv(a, b):
    """Full linear convolution using FFT."""
    return fftconvolve(a, b)


def eval_score_fast(a):
    """Fast computation of 2n*max(a*a)/(sum(a))^2."""
    n = a.size
    c = fast_conv(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 1e-6:
        return np.inf
    return 2.0 * n * max_c / (S * S)


def inner_solve(a, beta, num_iters, base_lr):
    """Minimize logsumexp(beta*(a*b)) over b in simplex via projected gradient."""
    n = a.size
    b = np.ones(n, dtype=np.float64) / n
    best_b = b.copy()
    best_val = np.inf

    for i in range(num_iters):
        c = fast_conv(a, b)
        cmax = float(c.max())
        shifted = beta * (c - cmax)
        e = np.exp(shifted)
        log_sum = np.log(e.sum()) + beta * cmax
        val = log_sum / beta

        if val < best_val:
            best_val = val
            best_b = b.copy()

        log_p = shifted - np.log(e.sum())
        p = np.exp(log_p)
        p /= p.sum()

        p_rev = p[::-1]
        grad_conv = fast_conv(a, p_rev)
        grad = grad_conv[n - 1:2 * n - 1][::-1]

        lr = base_lr / (1.0 + 0.05 * i)
        b = b - lr * grad
        b = project_to_simplex(b)

    return best_b

    n = a0.size

    S = a.sum()
    if S < 1e-6:
        a = np.ones(n, dtype=np.float64) / n

        b = inner_solve(a, beta, num_inner, base_lr)
