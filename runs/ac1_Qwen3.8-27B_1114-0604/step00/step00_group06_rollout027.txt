Let me think about the LP-based approach more carefully.

Given a current sequence a with convolution peak M = max_k b_k:
- We want to find a new sequence a' = a + t*d (for small t > 0) that reduces the peak.
- The condition is: (a' * a')_k <= M' for all k, where M' < M.
- But this is quadratic in d, so it's not a LP.

The SOTA approach instead does:
- Fix M (the current peak).
- Find g >= 0 that maximizes sum(g) subject to (a * g)_k <= M for all k.
- This IS a LP because (a * g)_k = sum_i a_i * g_{k-i} is linear in g.
- Then update: a_new = (1-t)*a + t*g/scale(g).

This is a clever approach because it keeps the problem linear.

    """Compute the score for sequence a."""
    n = len(a)
    b = np.convolve(a, a)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * np.max(b) / (S * S)

def gradient_descent_step(a, lr):
    """One step of subgradient descent."""
    n = len(a)
    b = np.convolve(a, a)
    S = np.sum(a)
    M = np.max(b)
    max_idx = np.where(b == M)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in max_idx:
        # grad[i] += 2 * a[k-i] for valid k-i

            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(max_idx)

    score_grad = 2.0 * n * grad / (S * S) - 4.0 * n * M / (S * S * S)

    n = len(a)

    S = np.sum(a)

        M = np.max(b)

The L∞ norm of (f*f) is bounded by the L1 norm of F^2: ||f*f||∞ <= ||F^2||_1.

from scipy.optimize import linprog
import time

def compute_score(a):
    n = len(a)
    b = np.convolve(a, a)
    S = np.sum(a)
    return 2.0 * n * np.max(b) / (S * S)

def subgrad_step(a, lr):
    """One step of subgradient descent on the score."""
    n = len(a)
    b = np.convolve(a, a)
    S = np.sum(a)
    M = np.max(b)
    max_idx = np.where(b == M)[0]
    
    # Subgradient of max(b) with respect to a
    # ∂b_k/∂a_i = 2*a[k-i] if 0 <= k-i < n, else 0
    grad = np.zeros(n)
    for k in max_idx:
        # vectorized: for i in range(n), if 0 <= k-i < n, grad[i] += 2*a[k-i]
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            grad[i_start:i_end+1] += 2.0 * a[j_start:j_end+1]
    grad /= len(max_idx)
    
    # Score gradient: d/dai [2n*M/S^2] = 2n*grad_i/S^2 - 4n*M/S^3
    score_grad = 2.0 * n * grad / (S*S) - 4.0 * n * M / (S**3)
    
    a_new = a - lr * score_grad
    return np.clip(a_new, 0.0, 1000.0)

def lp_direction(a, M):
    """Solve LP to find direction g that maximizes sum(g) s.t. (a*g)_k <= M, g>=0."""
    n = len(a)
    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: (a*g)_k <= M for k=0,...,2n-2
    A_ub = np.zeros((2*n-1, n))

        # (a*g)_k = sum_i a[i]*g[k-i] for valid i
        for i in range(max(0, k-n+1), min(k, n-1)+1):
            j = k - i
            A_ub[k, j] = a[i]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x if res.success else None
