    while n < x:
        n *= 2
    return n

    """Compute gradient of soft_max(conv(a,a)) w.r.t. a using FFT."""
    n = len(a)
    a_rfft = np.fft.rfft(a, N)
    b = np.fft.irfft(a_rfft * a_rfft)[:2*n-1]

    mb = np.max(beta_b)
