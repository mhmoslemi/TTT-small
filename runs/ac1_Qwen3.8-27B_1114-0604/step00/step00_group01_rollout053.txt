LP-based iterative optimization for minimizing the convolution peak ratio.
Uses scipy HiGHS solver on a reduced set of active constraints (top-K lags).
Combines LP direction finding with adaptive blending and block perturbations.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def compute_conv_peak(f):
    """Compute convolution f*f, return (max_val, sorted_indices_desc)."""
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    # Get indices sorted by convolution value (descending)
    sorted_idx = np.argsort(conv)[::-1]
    return max_val, conv, sorted_idx

def build_lp_matrix(f, active_lags, n):
    """Build sparse matrix for LP constraints.
    Row i (for lag k=active_lags[i]), column j: f[k-j] if 0 <= k-j <= n-1.
    """
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(active_lags):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            idxs = k - js
            rows.extend([i] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[idxs].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_lags), n))
    return A

    """Solve the LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
