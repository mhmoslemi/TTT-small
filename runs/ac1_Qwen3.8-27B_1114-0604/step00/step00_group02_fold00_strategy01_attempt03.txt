        parent = rng.uniform(0, 1000, n).astype(np.float64)
        population_size = 100
    
    # Normalization
    normalization_factor = np.sqrt(2*n)
    scaled_parent = parent * (normalization_factor / np.sum(parent))
    
    # Initial population
    population = []
    if 'initial_sequence' in kwargs:
        # Include provided initial sequence in mutation pool
        population.append(scaled_parent.tolist())
        for i in range(population_size-1):
            base = scaled_parent.copy()
            # Different random walk if available
            walk_step = rng.random()
            if walk_step < 0.7:
                base[start:end] = base[start:end] * (1 + rng.normal(0, 0.1, end-start+1))
            else:
                base[start:end] = base[start:end] - rng.uniform(0.05, 0.3, end-start+1)
            base = np.clip(base, 0, 1000)
            population.append(base.tolist())
    else:
        for i in range(population_size):
            base = rng.uniform(0, 1000, n).astype(np.float64)
            if rng.random() < 0.3:
                start = rng.integers(0, max(1, n//5))
                end = rng.integers(start, min(n-2, start+100))
                base[start:end] = base[start:end] * (1 + rng.normal(0, 0.2, end-start+1))
            population.append(base.tolist())
    population = [seq for seq in population if np.sum(np.abs(np.array(seq))) > 0.0001]
    
    # Best score tracking
    best_cand = population[0].copy()
    best_score = evaluate_sequence(best_cand)
    
    # Early return if already included in population
    if best_score < np.inf:
        best_cand_sequence = best_cand.copy()
        best_score_value = best_score
    
    # Main loop timing
    start_time = time.time()
    max_time = float(budget_s)
    max_evaluations = 1000
    
    # If evaluation is too expensive, we use the current best
    cand_buffer = []
    self_buffer = []
    
    # Evaluate and select process
    all_scores = [evaluate_sequence(seq) for seq in population]
    for idx, score in enumerate(all_scores):
        if np.isfinite(score) and score < best_score_value:
            best_cand_sequence = population[idx].copy()
            best_score_value = score
    
    # Initialize statistics for adaptive computation
    if np.count_nonzero(all_scores) < 10:
        # Not enough valid evaluations
        return cand_buffer or population[0]
    
    # Iterative improvement with caching
    return_sequence = best_cand_sequence.copy()
    common_descent = None
    
    # Learning through reduce, adapt, and refresh
    try:
        steps_to_improve = 100
        poll_points = 20
    except:
        # Default
        steps_to_improve = 100
        
    # Bounded random walk for primary mutation
    def random_walk():
        if rng.integers(0, 10) < steps_to_improve and poll_points > 0:
            pos = rng.integers(0, n)
            scale = 1 + rng.randint(-100, 100) / 200
            return min(max(pos * scale / n, 0), 1.2)
    
    # Caching for rapid evaluation
    cache = {}
    
    for i in range(max_evaluations):
        if time.time() - start_time > max_time:
            break
        
        sequences_to_eval = []
        for j in range(poll_points):
            # Permutation to combat local minima
            idx = rng.integers(0, population_size)
            depth = random_walk()
            perturbation = np.clip(rng.laplace(0, depth, n), 0, 1000)
            # Normalize to avoid changing dimension and to ensure fair comparison
            total = np.sum(perturbation)
            normalization_factor = np.sqrt(2*n) if total > 0 else 0.01
            normalized = perturbation * (normalization_factor / total)
            sequences_to_eval.append(normalized)
            
        # Parallel evaluation to speed up computationally heavy steps
        scores = [evaluate_sequence(seq) for seq in sequences_to_eval]
        
        # Adaptive improvement
        keep_best = []
        for idx, score_val in enumerate(scores):
            if np.isfinite(score_val):
                if score_val < best_score_value or (score_val == best_score_value and rng.random() < 0.01):
                    best_cand_sequence = sequences_to_eval[idx].copy()
                    best_score_value = score_val
                keep_best.append(True)
            else:
                keep_best.append(False)
        
        # End condition
        if np.isnan(best_score_value) or np.isinf(best_score) or time.time() - start_time > max_time - 10:
            break
            
        # Update cached best candidate
        if i < 5 or time.time() - start_time < max_time - 300:
            cand_buffer.append(best_cand_sequence)
            self_buffer.append(best_score_value)
        elif i >= 5 and (time.time() - start_time) > max_time - 300:
            # Recycling best for 30 iterations before random sample
            for rep_count in range(30):
                new_seq = best_cand_sequence.copy()
                # Random variation
                pos = rng.integers(0, n-1)
                new_seq[pos] = new_seq[pos] * (1 + rng.normal(0, 0.8))
                cand_buffer.append(new_seq)
                rep_count += 1
            
        cand_buffer.append(best_cand_sequence), self_buffer.append(best_score_value)
        
        return_sequence = best_cand_sequence if i % 20 else return_sequence
    
    # Format for evaluation using full convolution
    final_sequence = np.array(return_sequence).astype(np.float64)
    
    # Final score
    return final_sequence
