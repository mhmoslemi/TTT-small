    n = len(f)

    rows = []
    cols = []

                cols.append(j)

**Key insight**: The self-convolution C[k] = sum_i a[i]*a[k-i] can be seen as the "energy" at each lag. The sum of all C[k] is (sum a)^2. So if we want to minimize max C[k], we want to spread the energy as evenly as possible across all 2n-1 lags.

To do better, we need the sequence to have more variation while still keeping the convolution flat. This is a tension: more variation in a leads to more variation in C, but we need C to be flat.

**Another approach: iterative water-filling**

1. Start with a uniform sequence.
2. Find the lag k* with the maximum C[k].

4. Repeat.

The contribution of a[j] to C[k*] is 2*a[k*-j] (as derived above). So to reduce C[k*], we should decrease a[j] for j where a[k*-j] is large, or increase a[j] for j where a[k*-j] is small (but this increases other C[k]).

This is essentially what the LP approach does: find the best direction to move that keeps all C[k] <= M.

"""
LP-based iterative optimization for minimizing 2*n*max(self-convolution)/sum(f)^2.
Uses central binomial sequence initialization, LP constraint generation,
and local search. Target: beat 1.5027.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _central_binomial(n):
    """Generate a[k] = C(2k,k)/4^k for k=0,...,n-1."""
    a = np.zeros(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2.0*k - 1.0) / (2.0*k)
    return a


def _score(f):
    """Compute objective directly."""
    n = f.size
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(f, f)
    return 2.0 * n * c.max() / (s * s)


def _solve_lp(f, M, timeout=None):
    """Solve max sum(b) s.t. b >= 0, A(b) <= M where A is conv with f."""
    n = f.size
    if n > 5000:
        # Too large, subsample constraints
        # Only use every other constraint
        pass
    
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = (col_idx.astype(np.int64) + np.tile(np.arange(n, dtype=np.int64), n))
    vals = np.tile(f, n)
    
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n), dtype=np.float64)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def _lp_iteration(f, t=0.5, M_mult=1.0):
    """One LP-based iteration."""
    conv = np.convolve(f, f)
    M = conv.max() * M_mult
    if M <= 1e-10:
        return f.copy()
    
    b = _solve_lp(f, M)
    if b is None or not np.all(np.isfinite(b)):
        return f.copy()
    if b.max() < 1e-15:
        return f.copy()
    
    s = f.sum()
    b_sum = b.sum()
    if b_sum < 1e-10:
        return f.copy()
    b = b * (s / b_sum)
    
    new = (1.0 - t) * f + t * b

"""
LP-based iterative optimization for minimizing 2*n*max(conv(f,f))/sum(f)^2.
Uses central binomial sequence initialization (gives ~pi/2 ≈ 1.57), LP constraint
generation for direction finding, multiple M multipliers, random restarts, and
local search to push below the 1.5027 target.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _central_binomial(n):
    """Generate a[k] = C(2k,k)/4^k for k=0,...,n-1."""
    a = np.zeros(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return a


def _score(f):
    """Compute objective: 2*n*max(conv)/sum^2."""
    n = f.size
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(f, f)
    return 2.0 * n * c.max() / (s * s)


def _solve_lp(f, M):
    """Solve max sum(b) s.t. b >= 0, conv(f,b)[k] <= M for all k."""
    n = f.size
    # Build sparse convolution matrix: A[k,j] = f[k-j] if 0<=k-j<n
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = col_idx.astype(np.int64) + np.tile(np.arange(n, dtype=np.int64), n)
    vals = np.tile(f, n)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n), dtype=np.float64)

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def _lp_iteration(f, t=0.5, M_mult=1.0):
    """One LP-based iteration: find b via LP, return (1-t)*f + t*b."""
    conv = np.convolve(f, f)
    M = conv.max() * M_mult
    if M <= 1e-10:
        return f.copy()

    b = _solve_lp(f, M)
    if b is None or not np.all(np.isfinite(b)):
        return f.copy()
    if b.max() < 1e-15:
        return f.copy()

    s = f.sum()
    b_sum = b.sum()
    if b_sum < 1e-10:
        return f.copy()
    b = b * (s / b_sum)

    new = (1.0 - t) * f + t * b
    return np.clip(new, 0.0, 1000.0)


def _local_search(f, rng, deadline, best_score, max_iter=80000):

    n = f.size

        candidate = np.clip(candidate, 0.0, 1000.0)

        if np.isfinite(sc) and sc < best_score:
            best = candidate
