"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. FFT-based fast convolution
2. Gradient descent with softmax approximation of max
3. Local search with peak shaving and random perturbations
4. Multiple initializations and restarts

import numpy as np
import time

def fast_conv(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def fast_score(a):
    """Fast score computation (scale-invariant)."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * c.max() / (S * S)

def softmax_logits(c, T):
    """Compute softmax weights for approximating max."""
    cmax = c.max()
    logits = T * (c - cmax)
    w = np.exp(logits)
    w /= w.sum()
    return w

def gradient_softmax(a, T):
    """
    Gradient of softmax approx of max(conv(a,a)) w.r.t. a.
    dF/da[j] = T * sum_k w[k] * 2*a[k-j] (for 0<=k-j<n)
             = 2T * sum_i a[i] * w[j+i]
    """
    n = len(a)
    c = fast_conv(a)
    w = softmax_logits(c, T)
    # grad[j] = 2T * sum_{i=0}^{n-1} a[i] * w[j+i]
    # = 2T * np.correlate(w, a, 'valid')[j]
    grad = 2.0 * T * np.correlate(w, a, 'valid')
    return grad

    if S < 1e-15:

    return a * (target / S)

1. FFT-based fast convolution

import numpy as np

def fast_conv(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_score(a):
    """Fast score computation (scale-invariant)."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * c.max() / (S * S)


def softmax_weights(c, T):
    """Compute softmax weights for approximating max."""
    cmax = c.max()
    logits = T * (c - cmax)
    w = np.exp(logits)
    s = w.sum()
    if s < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return w / s


def gradient_softmax(a, T):
    """
    Gradient of softmax(max(conv(a,a))) w.r.t. a.
    dF/da[j] = 2T * sum_{i=0}^{n-1} a[i] * w[j+i]
    """
    n = len(a)
    c = fast_conv(a)
    w = softmax_weights(c, T)
    grad = 2.0 * T * np.correlate(w, a, 'valid')
    return grad
