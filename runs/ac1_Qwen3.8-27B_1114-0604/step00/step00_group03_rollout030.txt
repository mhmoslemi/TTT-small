```python
def compute_autocorr(f):
```

```python
    n = len(f)
```

```python
def optimize_sequence(n, f_init, alpha_start=1.0, alpha_end=50.0, lr=0.01, n_iter=1000):
    f = f_init.copy()
    f = np.maximum(f, 0)
    f = f / np.sum(f)  # normalize to sum 1
    
    best_score = np.inf
    best_f = f.copy()
    
    for it in range(n_iter):
        # Anneal alpha
        t = it / max(1, n_iter - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        
        # Compute autocorrelation
        R = np.convolve(f, f, mode='full')
        
        # Compute objective
        max_R = np.max(R)
        obj = 2 * n * max_R  # (sum f)^2 = 1 since normalized
        
        if obj < best_score:
            best_score = obj
            best_f = f.copy()
        
        # Compute softmax weights
        # Use log-sum-exp for stability
        R_max = np.max(R)
        log_w = alpha * (R - R_max) - np.log(np.sum(np.exp(alpha * (R - R_max))))
        w = np.exp(log_w)
        
        # Compute gradient
        # g[i] = 2 * sum_j w[i+j] * f[j]
        M = np.lib.stride_tricks.as_strided(w, shape=(n, n), strides=(w.strides[0], w.strides[0]))
        grad = M @ f
        grad = 2 * grad
        
        # Project gradient to maintain sum constraint
        # We want to move in the direction that reduces the objective
        # while keeping sum(f) = 1
        # Project: subtract the mean (approximately)
        grad_proj = grad - np.mean(grad)
        
        # Update
        f = f - lr * grad_proj
        f = np.maximum(f, 0)
        f = f / np.sum(f)
    
    return best_f, best_score
```
