from scipy.optimize import linprog

def lp_direction(f, M, n):
    """Solve LP to find direction g that maximizes sum(g) subject to conv(f,g) <= M."""
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g), i.e., minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: conv(f, g)[k] <= M for k = 0, ..., 2n-2
    # conv(f, g)[k] = sum_i f[i] * g[k-i] (valid indices)
    # This is a linear function of g.
    # A_ub[k, j] = f[j] if k-j is in [0, n-1], i.e., if 0 <= k-j <= n-1
    # i.e., k-(n-1) <= j <= k
    
    A_rows = []
    for k in range(2*n-1):
        row = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end + 1):
            row[j] = f[k - j]
        A_rows.append(row)
    
    A_ub = np.array(A_rows)
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

from scipy.optimize import linprog

def solve_lp_direction(f, M, K=100):
    """
    Solve reduced LP: max sum(g) s.t. g>=0, conv(f,g)[k] <= M for top K k's.
    """
    n = len(f)
    b = np.convolve(f, f)
    # Get top K indices
    top_k = np.argsort(b)[-K:]
    
    # Build constraint matrix (K x n)
    A = np.zeros((K, n))
    for idx, k in enumerate(top_k):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[idx, j] = f[k - j]
    
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None
