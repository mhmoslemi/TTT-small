    if s < 1e-10:
        return x.copy(), 0.0
    x_new /= s

def fast_score(x):
    """Compute 2n*max(conv(x,x))/(sum x)^2 efficiently."""
    n = x.size
    if n == 0:
        return np.inf
    sum_a = x.sum()
    if sum_a < 0.01:
        return np.inf
    size = 1
    target = 2 * n - 1
    while size < target:
        size <<= 1
    X = np.fft.rfft(x, size)
    c = np.fft.irfft(X * X, size)[:target]
    max_b = c.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def one_step(x, alpha):
