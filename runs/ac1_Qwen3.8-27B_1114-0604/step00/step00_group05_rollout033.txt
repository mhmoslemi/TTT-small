"""
FFT-based gradient descent for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothing of the max, temperature annealing, and simplex projection.
Multiple initializations including power-law, window functions, and random.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def _next_pow2(n):
    x = 1
    while x < n:
        x <<= 1
    return x


def _convolve_fft(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    L = _next_pow2(2 * n - 1)
    a_pad = np.zeros(L)
    a_pad[:n] = a
    A = fft(a_pad)
    C = np.real(ifft(A * A))
    return C[:2 * n - 1]


def _compute_gradient(a, T):
    """
    Compute gradient of f_T(a) = 2n * T*log(sum exp(C_k/T)) / S^2
    where C = conv(a,a), S = sum(a).
    Returns (gradient, C, S, g).
    """
    n = len(a)
    S = np.sum(a)
    C = _convolve_fft(a)
    
    # Softmax weights (numerically stable)
    C_shifted = C - np.max(C)
    exp_C = np.exp(C_shifted / T)
    w = exp_C / np.sum(exp_C)
    
    # g = T * log(sum exp(C/T))
    g = T * (np.max(C) + np.log(np.sum(exp_C)))
    
    # T_i = sum_j a_j * w_{j+i} for i=0..n-1
    # T_i = conv(a, w_rev)[2n-2-i]
    w_rev = w[::-1]
    L3 = _next_pow2(3 * n - 2)
    a_p3 = np.zeros(L3)
    a_p3[:n] = a
    wr_p3 = np.zeros(L3)
    wr_p3[:2 * n - 1] = w_rev
    conv_full = np.real(ifft(fft(a_p3) * fft(wr_p3)))
    T_vec = conv_full[n - 1:2 * n - 1][::-1]
    
    # Gradient: grad_i = 4n*(S*T_i - g) / S^3
    grad = 4.0 * n * (S * T_vec - g) / (S ** 3)
    
    return grad, C, S, g


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using clip + renormalize (approximate)."""
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s < 1e-15:
        return np.ones_like(v) / len(v)
    return v / s


def _local_search(a_init, T_start, T_end, n_iter, seed=0):
    """
    FFT-based gradient descent with temperature annealing.
    a_init: starting point (will be projected to simplex)
    T_start: initial temperature (large = smooth)
    T_end: final temperature (small = sharp)
    n_iter: total number of iterations
    Returns best sequence found and its exact score.
    """
    rng = np.random.default_rng(seed)
    a = _project_simplex(a_init.copy())
    n = len(a)
    
    best_a = a.copy()
    
    # We'll evaluate exact score periodically
    n_eval = max(1, n_iter // 50)
    
    for it in range(n_iter):
        # Anneal temperature
        frac = it / max(1, n_iter - 1)
        T = T_start * (T_end / T_start) ** frac
        
        # Compute gradient
        grad, C, S, g = _compute_gradient(a, T)
        
        # Backtracking line search
        lr = 0.01
        obj_exact = 2.0 * n * np.max(C) / (S ** 2)
        for _ in range(20):
            a_new = a - lr * grad
            a_new = _project_simplex(a_new)
            C_new = _convolve_fft(a_new)
            S_new = np.sum(a_new)
            obj_new = 2.0 * n * np.max(C_new) / (S_new ** 2)
            if obj_new < obj_exact:
                break
            lr *= 0.5
        else:
            break
        
        a = a_new
        obj_exact = obj_new
        
        # Periodically track best
        if it % n_eval == 0 or it == n_iter - 1:
            if obj_exact < 2.0 * n * np.max(_convolve_fft(best_a)) / (np.sum(best_a) ** 2):
                best_a = a.copy()
    
    return best_a


def _init_powerlaw(n, alpha):
    """Power-law initialization: a_j = (j+1)^(-alpha), normalized."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = j ** (-alpha)
    return a / np.sum(a)


def _init_hann(n):
    """Hann window initialization."""
    j = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1)))
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _init_triangle(n):
    """Triangle (Bartlett) window."""
    j = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1 - np.abs(j - mid) / (mid + 1e-10)
    a = np.maximum(a, 0)
    return a / np.sum(a)


def _init_random(n, seed):
    """Random initialization projected to simplex."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    return a / np.sum(a)


def _init_gaussian(n, sigma_frac):
    """Gaussian-shaped initialization."""
    j = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((j - mid) / (sigma_frac * n)) ** 2)
    return a / np.sum(a)
