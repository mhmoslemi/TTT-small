   a. Compute `c = fftconvolve(a, a)` using FFT (fast).
   b. Compute the objective `f = 2n * max(c) / sum(a)^2`.
   c. Compute the "pressure" at each position: how much each element contributes to the peak.
   d. Update the sequence to reduce the peak.

For step (d), I'll use a **multiplicative update** inspired by the multiplicative weight method:

For each element `a[i]`, compute its contribution to the peak convolution values:
`contrib[i] = sum_{k in K*} a[k-i]` (where `K* = argmax(c)`, and `a[j] = 0` for `j` out of range)

Then update: `a[i] = a[i] * (1 - eta * contrib[i] / max(contrib))`

This reduces elements that contribute most to the peak.

But this alone would just make the sequence go to zero. We need to also redistribute the "mass."

Better approach: **Redistribute from peak-contributing elements to valley-contributing elements.**

`a[i] = a[i] + eta * (valley_contrib[i] - peak_contrib[i])`

where `peak_contrib[i]` is the contribution to the peak and `valley_contrib[i]` is the contribution to the valleys.

Then renormalize to maintain the sum (or not, since the objective is scale-invariant).

def fft_convolve(a):
    """Compute a*a (full linear convolution) using FFT."""
    n = len(a)
    L = 1
    while L < 2*n - 1:
        L <<= 1
    Af = np.fft.rfft(a, L)
    Cf = Af * Af
    c = np.fft.irfft(Cf, L)
    return c[:2*n-1]

def objective(a):
    """Compute the objective function value."""
    n = len(a)
    c = fft_convolve(a)
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)

def subgradient(a, c, K=5):
    """Compute approximate subgradient of max(c) w.r.t. a."""
    n = len(a)
    K_star = np.argsort(c)[-K:]
    g = np.zeros(n)
    for k in K_star:
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min > i_max:
            continue
        # dc[k]/da[i] = a[k-i] for i in [i_min, i_max]

        g[i_min:i_max+1] += a[idx]
    g /= K

import numpy as np
import time

"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using soft-max gradient descent with
FFT-accelerated convolution, adaptive learning rate, multiplicative updates,
multiple diverse initializations, and coordinate-descent polishing.
"""


def _fft_conv(a, L):
    """Full linear autopower-convolution a*a via FFT, length 2n-1."""
    Af = np.fft.rfft(a, L)
    return np.fft.irfft(Af * Af, 2 * len(a) - 1, L)


def _soft_max_grad(a, c, tau, K):
    """Gradient of soft-max_tau(c) w.r.t. a, using top-K peaks."""
    n = len(a)
    cshift = c - c.max()
    p = np.exp(cshift / tau)
    psum = p.sum()
    if psum < 1e-300:
        return np.zeros(n)
    p /= psum
    if K < len(p):
        top = np.argpartition(p, -K)[-K:]
    else:
        top = np.arange(len(p))
    g = np.zeros(n)
    for k in top:
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        # g[i] += pk * a[k-i] for i in [lo, hi]
        g[lo:hi + 1] += pk * a[k - hi:k - lo + 1][::-1]
    return g


def _score_from_conv(a, c):
    """Compute objective from precomputed convolution."""
    n = len(a)
    M = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)


def _sgd_phase(a, L, n, time_budget, rng, tau_init, tau_end, eta_init,
               momentum, K_init, K_end, verbose=False, label=""):
    """Run soft-max gradient descent for a given time budget."""
    a = a.astype(np.float64).copy()
    if a.sum() < 0.01:
        a = np.ones(n) / n
    v = np.zeros(n)
    best_local = a.copy()
    best_local_score = _score_from_conv(a, _fft_conv(a, L))
    eta = eta_init
    it = 0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        c = _fft_conv(a, L)
        score = _score_from_conv(a, c)
        if not np.isfinite(score):
            a = best_local.copy()
            v = np.zeros(n)
            continue
        if score < best_local_score:
            best_local = a.copy()
            best_local_score = score

        # Adaptive parameters (linear anneal over time)
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        tau = tau_init + (tau_end - tau_init) * frac
        K = max(5, int(K_init + (K_end - K_init) * frac))

        g = _soft_max_grad(a, c, tau, K)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g = g / gnorm

        v = momentum * v - eta * g
        a_new = a + v
        a_new = np.maximum(0.0, a_new)
        S_new = a_new.sum()
        if S_new < 0.01:
            a = best_local.copy()
            v = np.zeros(n)
            eta *= 0.5
            continue
        c_new = _fft_conv(a_new, L)
        score_new = _score_from_conv(a_new, c_new)

        if score_new < score * 1.0005:
            a = a_new
            # Occasionally increase eta if making fast progress
            if it > 200 and score_new < score * 0.999:
                eta = min(eta * 1.05, eta_init * 2)
        else:
            eta = max(eta * 0.9, eta_init * 0.01)
            v *= 0.5

        it += 1
        if verbose and it % 50000 == 0:
            print(f"    {label} iter={it}: score={score:.6f} best={best_local_score:.6f} tau={tau:.3f} eta={eta:.5f}")

    return best_local, best_local_score


def _coord_descent(a, n, time_budget, rng, L):
    """Coordinate descent: try scaling individual elements."""
    a = a.astype(np.float64).copy()
    c = _fft_conv(a, L)
    score = _score_from_conv(a, c)
    best = a.copy()

            # If i != k-i, both terms change: delta*a[k-i] + a[k-i]*delta = 2*delta*a[k-i]... no.

            # If k-i == i (i.e., k=2i): delta_c[k] = delta*a[i] + a[i]*delta... no, it's just delta*a[i] (one term becomes (a[i]+delta)*a[i] instead of a[i]*a[i], so delta = a[i]*delta... wait.
            # Let me just recompute: c_new[k] - c_old[k] = sum_j (a[j]+delta_j)*a[k-j] - sum_j a[j]*a[k-j]
            # where delta_j = delta if j==i else 0.

            # But we need a[k-i] to be valid (0 <= k-i < n).

                best_score = s_new
                c = c_new
                score = s_new
                improved = True

                a[i] = a_old
                if f > 1.0:
