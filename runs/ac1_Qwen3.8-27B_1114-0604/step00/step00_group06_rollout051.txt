def fast_full_score(a):
   n = len(a)
   if n ==0: return np.inf
   size=1
   while size < 2*n-1: size <<=1
   A = np.fft.rfft(a, size)
   conv = np.fft.irfft(A*A, size)[:2*n-1]
   max_b = float(conv.max())
   s = float(a.sum())
   if s <0.01: return np.inf
   return float(2*n*max_b/(s*s))

def fft_size_for(n):
    need = 2 * int(n) - 1
    return 1 << max(0, need - 1).bit_length()


def fast_full_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.inf
    size = fft_size_for(n)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2 * n - 1]
    max_b = float(conv.max()) if conv.size else 0.0
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def full_conv_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.array([]), np.inf
    size = fft_size_for(n)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2 * n - 1]
    max_b = float(conv.max()) if conv.size else 0.0
    s = float(a.sum())
    if s < 0.01:
        return conv, np.inf
    score = float(2.0 * n * max_b / (s * s))
    return conv, score


def is_prime(n):
    n = int(n)
    if n < 2:
        return False
    if n % 2 == 0:
        return n == 2
    d = 3
    while d * d <= n:
        if n % d == 0:
            return False
        d += 2
    return True


def largest_prime_below(x):
    x = int(x)
    if x < 2:
        return 2
    n = max(2, x)
    while n >= 2:
        if is_prime(n):
            return n
        n -= 1
    return 2

def finite_field_support(N, q, variant, rng):
    N = int(N)
    q = int(q)
    if q < 2 or N < 2:
        return random_support(rng, N, min(N, 20))
    vals = np.empty(q, dtype=np.int64)
    for x in range(q):
        if variant == 0:
            y = (x * x) % q
        elif variant == 1:
            y = (x * x + x) % q
        elif variant == 2:
            y = (x * x + 2 * x + 1) % q
        else:
            y = (x * x + 3 * x + 2) % q
        vals[x] = x + q * y

def apply_array_delta(counts, sum_sq, idx, delta):
    if idx is None or idx.size == 0:
        return sum_sq
    old = counts[idx]
    new = old + delta
    counts[idx] = new
    return sum_sq + int(np.sum(new * new - old * old))


def apply_scalar_delta(counts, sum_sq, idx, delta):
    old = int(counts[int(idx)])
    new = old + int(delta)
    counts[int(idx)] = new
    return sum_sq + (new * new - old * old)

def canonicalize_support_weights(S, w, N):
    S = np.asarray(S, dtype=np.int64).ravel()
    if w is not None:
        w = np.asarray(w, dtype=np.float64).ravel()
    N = int(N)
    if S.size == 0:
        return S, w
    valid = (S >= 0) & (S < N)
    S = S[valid]
    if w is not None:
        if w.size == S.size:
            w = w[valid]
        else:
            w = None
    if S.size == 0:
        return S, w
    order = np.argsort(S)
    S = S[order]
    if w is not None:
        w = w[order]
    S2, idx = np.unique(S, return_index=True)
    if S2.size != S.size:
        S = S2
        if w is not None and w.size == idx.size:
            w = w[idx]
    return S, w

        if len(selected) >= half:
            break

    buckets = [80, 100, 120, 150, 180, 220, 260, 300, 350, 400, 450, 500, 550, 600, 700, 800]
    for b in buckets:
        if len(selected) >= max_sel:
            break
        for S, w, score in filtered:
            if abs(int(S.size) - b) <= 20:
                key = tuple(S.tolist())
                if key not in seen:
                    selected.append((S, w, score))
                    seen.add(key)
                    break

    for S, w, score in filtered:
        if len(selected) >= max_sel:
            break
        key = tuple(S.tolist())
        if key not in seen:
            selected.append((S, w, score))
            seen.add(key)

    return selected[:max_sel]


def precompute_support_kflat(S, N):
    S = np.asarray(S, dtype=np.int64).ravel()
    N = int(N)
    m = int(S.size)
    if m == 0:
        return np.empty(0, dtype=np.int64), 0
    S2 = (S[:, None] + S[None, :]).ravel()
    counts = np.bincount(S2, minlength=2 * N - 1)
    active = np.flatnonzero(counts > 0)
    g2l = np.zeros(2 * N - 1, dtype=np.int64)
    g2l[active] = np.arange(active.size, dtype=np.int64)
    Kflat = g2l[S2]
    return Kflat, int(active.size)


def safe_normalize(w):
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size == 0:
        return np.array([1.0], dtype=np.float64)
    w = np.where(np.isfinite(w), w, 0.0)
    w = np.clip(w, 0.0, None)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(w.size, 1.0 / w.size, dtype=np.float64)
    return w / s


def sparse_c_score(Kflat, L, N, w):
    w = np.asarray(w, dtype=np.float64).ravel()
    Kflat = np.asarray(Kflat, dtype=np.int64).ravel()
    L = int(L)
    N = int(N)
    if Kflat.size == 0 or L <= 0 or w.size == 0:
        return np.empty(0, dtype=np.float64), np.inf
    W = np.outer(w, w).ravel()
    c = np.bincount(Kflat, weights=W, minlength=L).astype(np.float64, copy=False)
    if c.size == 0:
        return c, np.inf
    cmax = float(c.max())
    s = float(w.sum())
    if not np.isfinite(s) or s < 1e-300:
        return c, np.inf
    return c, float(2.0 * N * cmax / (s * s))


def mirror_descent_weights(Kflat, m, L, N, w0, rng, time_limit):
    Kflat = np.asarray(Kflat, dtype=np.int64).ravel()
    w = safe_normalize(w0)
    m = int(m)
    L = int(L)
    N = int(N)
    time_limit = max(0.0, float(time_limit))
    c, score = sparse_c_score(Kflat, L, N, w)
    if not np.isfinite(score):
        return w, np.inf
    best_w = w.copy()
    best_score = float(score)
    if time_limit <= 1e-6 or L <= 0 or Kflat.size == 0:
        return best_w, best_score

    start = time.monotonic()
    deadline = start + time_limit
    eta = 0.08
    beta_factor = 0.2
    pow_mode = 1
    top_prob = 0.3
    no_improve = 0
    need_c = False
    step = 0

    while time.monotonic() < deadline:
        if need_c:
            c, score = sparse_c_score(Kflat, L, N, w)
            need_c = False
            if not np.isfinite(score):
                w = best_w.copy()
                c, score = sparse_c_score(Kflat, L, N, w)
        if c.size == 0:
            break

        if score < best_score:
            best_score = float(score)
            best_w = w.copy()
            no_improve = 0
        else:
            no_improve += 1

        cmax = float(c.max())
        if cmax <= 1e-300:
            break

        if no_improve > 120:
            w = best_w.copy()
            w = w * np.exp(rng.normal(0.0, 0.25, m))
            w = safe_normalize(w)
            need_c = True
            eta = 0.08
            beta_factor = 0.25
            pow_mode = 1
            top_prob = 0.4
            no_improve = 0
            continue

        if top_prob < 1.0 and rng.random() < top_prob:
            r = rng.random()
            if r < 0.25:
                q = np.zeros(L, dtype=np.float64)
                q[int(np.argmax(c))] = 1.0
            else:
                k = int(min(L, max(10, m // 5)))
                if k >= L:
                    idx = np.arange(L)
                else:
                    idx = np.argpartition(-c, k)[:k]
                vals_top = c[idx]
                temp = max(cmax * 0.1, 1e-300)
                z = (vals_top - float(vals_top.max())) / temp
                z = np.clip(z, -50.0, 0.0)
                ez = np.exp(z)
                q = np.zeros(L, dtype=np.float64)
                q[idx] = ez / ez.sum()
        else:
            if pow_mode == 1:
                vals = c
            else:
                vals = c * c
            vmax = float(vals.max())
            if vmax <= 1e-300:
                break
            beta = max(beta_factor * vmax, 1e-300)
            z = (vals - vmax) / beta
            z = np.clip(z, -50.0, 0.0)
            ez = np.exp(z)
            q = ez / ez.sum()

        qflat = q[Kflat]
        grad = 2.0 * (qflat.reshape(m, m).dot(w))
        gmax = float(grad.max())
        if not np.isfinite(gmax) or gmax <= 1e-300:
            break
        grad = grad / gmax

        improved = False
        for ety in (eta, eta * 0.5, eta * 1.5):
            if ety <= 0:
                continue
            decay = np.exp(-ety * grad)
            wn = w * decay
            s = float(wn.sum())
            if not np.isfinite(s) or s <= 1e-300:
                continue
            wn = wn / s
            maxw = float(wn.max())
            if m > 20 and maxw > 0.10:
                uniform = 1.0 / m
                wn = 0.8 * wn + 0.2 * uniform
                wn = wn / wn.sum()
            if wn.min() < 1e-22:
                wn = np.maximum(wn, 1e-24)
                wn = wn / wn.sum()
            c2, score2 = sparse_c_score(Kflat, L, N, wn)
            if np.isfinite(score2) and score2 < score:
                w = wn
                c = c2
                score = float(score2)
                improved = True
                no_improve = 0
                if score < best_score:
                    best_score = score
                    best_w = w.copy()
                eta = min(0.35, eta * 1.08)
                beta_factor = max(0.02, beta_factor * 0.985)
                break

        if not improved:
            no_improve += 1
            eta = max(0.008, eta * 0.88)
            if no_improve > 40:
                beta_factor = min(0.6, beta_factor * 1.15)
                top_prob = min(0.8, top_prob * 1.05)
                pow_mode = 2
            elif no_improve > 80:
                beta_factor = 0.35
                top_prob = 0.6
                pow_mode = 1

        step += 1
        if step > 50000:
            break

    c, score = sparse_c_score(Kflat, L, N, best_w)
    if np.isfinite(score):
        best_score = float(score)
    return best_w, best_score


def optimize_weights_for_support(S, N, time_limit, seed, init_w=None):
    S = np.asarray(S, dtype=np.int64).ravel()
    N = int(N)
    m = int(S.size)
    if m <= 0 or m > 800:
        return None, np.inf
    time_limit = max(0.0, float(time_limit))
    rng = np.random.default_rng(int(seed))

    try:
        Kflat, L = precompute_support_kflat(S, N)
    except Exception:
        return None, np.inf
    if Kflat.size == 0 or L <= 0:
        return None, np.inf

    inits = []
    if init_w is not None:
        w = safe_normalize(init_w)
        if w.size == m:
            inits.append(w)
    inits.append(np.full(m, 1.0 / m, dtype=np.float64))

    if time_limit > 15.0:
        inits.append(rng.dirichlet(np.ones(m, dtype=np.float64)))
    if time_limit > 30.0:
        p = np.linspace(0.4, 2.5, m, dtype=np.float64)
        inits.append(safe_normalize(p))
        inits.append(safe_normalize(p[::-1]))
    if time_limit > 70.0:
        inits.append(rng.dirichlet(np.ones(m, dtype=np.float64) * 0.25))

    start = time.monotonic()
    deadline = start + time_limit
    best_w = None
    best_score = np.inf
    n_inits = max(1, len(inits))

    for i, wi in enumerate(inits):
        rem = max(0.0, deadline - time.monotonic())
        if rem < 3.0:
            break
        # More time for earlier initializations.
        t = min(rem, max(3.0, time_limit / n_inits * max(1.0, 2.0 - i / max(1, n_inits))))
        try:
            w, sc = mirror_descent_weights(Kflat, m, L, N, wi, rng, t)
        except Exception:
            continue
        if w is not None and np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_w = w

    rem = max(0.0, deadline - time.monotonic())
    if best_w is not None and rem > 20.0:
        try:
            w, sc = mirror_descent_weights(Kflat, m, L, N, best_w, rng, rem * 0.65)
            if w is not None and np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_w = w
        except Exception:
            pass

    if best_w is not None:
        _, best_score = sparse_c_score(Kflat, L, N, best_w)
    return best_w, best_score


def full_local_search(a0, N, rng, time_limit):
    N = int(N)
    a = prepare_full_sequence(a0)
    if a.size != N:
        a = np.ones(N, dtype=np.float64)
    time_limit = max(0.0, float(time_limit))
    if time_limit < 5.0:
        return a, fast_full_score(a)

    best = a.copy()
    best_score = fast_full_score(a)
    rng = np.random.default_rng(int(rng))
    start = time.monotonic()
    deadline = start + time_limit
    idx_base = np.arange(N, dtype=np.int64)
    eta = 0.02
    no_improve = 0

    while time.monotonic() < deadline:
        conv, score = full_conv_score(a)
        if conv.size == 0:
            break
        if np.isfinite(score) and score < best_score:
            best_score = float(score)
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        kmax = int(np.argmax(conv))
        idx = kmax - idx_base
        valid = (idx >= 0) & (idx < N)
        grad = np.zeros(N, dtype=np.float64)
        grad[valid] = a[idx[valid]]
        gmax = float(grad.max())

        if gmax > 1e-300:
            decay = np.exp(-eta * (grad / gmax))
            a_new = a * decay

            if rng.random() < 0.18:
                cnt = min(80, max(1, N // 100))
                idxs = rng.integers(0, N, cnt)
                amp = max(1e-12, float(a.max()))
                a_new[idxs] = amp * rng.uniform(0.001, 0.02, cnt)

            a_new = a_new * np.exp(rng.normal(0.0, 0.004, N))
            a_new = np.clip(a_new, 0.0, 1000.0)
            s = float(a_new.sum())
            if s > 0.01:
                a_new = a_new / s
                score_new = fast_full_score(a_new)
                if np.isfinite(score_new) and score_new < score:
                    a = a_new
                else:
                    if no_improve > 80:
                        a = best.copy()
                        no_improve = 0
                        eta = 0.01
            else:
                if no_improve > 50:
                    a = best.copy()
                    no_improve = 0

        if no_improve > 300:
            a = best.copy()
            no_improve = 0
            eta = min(0.08, eta * 1.3)
        elif no_improve < 20:
            eta = min(0.08, eta * 1.02)

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Sparse support+weight optimizer: build low-collision supports, then balance
    pair-sum loads with mirror descent. Final polishing uses FFT-guided full search."""
    seed = int(seed)
    budget_s = max(60.0, float(budget_s))
    N = get_base_length()
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()

    support_time = min(700.0, max(25.0, 0.25 * budget_s))
    weight_time = min(1500.0, max(60.0, 0.55 * budget_s))
    final_time = max(15.0, budget_s - support_time - weight_time - 15.0)
    scale = max(0.1, (budget_s - 10.0) / max(1e-6, support_time + weight_time + final_time))
    support_time *= scale
    weight_time *= scale
    final_time = max(10.0, budget_s - 10.0 - support_time - weight_time)

    parent = globals().get("height_sequence_1", None)
    parent_full = None
    if parent is not None:
        try:
            parent_full = prepare_full_sequence(parent)
            if parent_full.size != N:
                parent_full = resize_or_fallback(parent_full, N)
        except Exception:
            parent_full = None
    if parent_full is None or parent_full.size != N:
        parent_full = np.ones(N, dtype=np.float64)

    parent_score = fast_full_score(parent_full)
    best_full = parent_full.copy()
    best_score = parent_score
    print(f"parent fast score: {best_score:.8f}, N={N}")

    support_candidates = []

    # If the provided parent is already sparse, exploit its support.
    try:
        nz = np.flatnonzero(parent_full > 1e-12)
        if 5 < nz.size <= 800:
            w = parent_full[nz]
            w = safe_normalize(w)
            Kflat, L = precompute_support_kflat(nz, N)
            _, sc = sparse_c_score(Kflat, L, N, w)
            add_support_candidate(support_candidates, nz, w, sc, N)
            print(f"parent sparse support: m={nz.size}, estimated score={sc:.8f}")
    except Exception:
        pass

    max_m = int(min(800, max(120, int(8.0 * math.sqrt(max(2, N))))))
    q0 = largest_prime_below(max(3, int(math.sqrt(max(4, N))) - 1))

    support_deadline = total_start + support_time
    configs = []
    for v in range(4):
        configs.append(("ff", v, 0))
    for m0 in (40, 80, 130, 180, 250):
        configs.append(("rand", None, m0))

    for idx_cfg, cfg in enumerate(configs):
        if time.monotonic() >= support_deadline:
            break
        rem = max(0.0, support_deadline - time.monotonic())
        t = min(rem, max(5.0, support_time / max(1, len(configs) - idx_cfg) * 0.75))
        if t < 2.0:
            continue
        try:
            kind = cfg[0]
            if kind == "ff":
                S0 = finite_field_support(N, q0, cfg[1], rng)
            else:
                S0 = random_support(rng, N, cfg[2])
            snaps = greedy_add_support(S0, N, rng, max_m, t)
            for S, sc in snaps:
                add_support_candidate(support_candidates, S, None, sc, N)
        except Exception as e:
            print("support config failed:", repr(e))

    # Add fixed-size random supports for diversity, especially larger m.
    if time.monotonic() < support_deadline - 5.0:
        for size in (120, 200, 300, 420, 550, max_m):
            if size > N:
                continue
            for rep in range(2):
                try:
                    S = random_support(rng, N, size)
                    _, _, sc, _ = support_uniform_info(S, N)
                    add_support_candidate(support_candidates, S, None, sc, N)
                except Exception:
                    pass
                if time.monotonic() >= support_deadline:
                    break
            if time.monotonic() >= support_deadline:
                break

    selected = select_support_candidates(support_candidates, max_sel=28)
    print(f"selected {len(selected)} support candidates for weight optimization")

    weight_deadline = total_start + support_time + weight_time
    for i, (S, w0, est) in enumerate(selected):
        rem = max(0.0, weight_deadline - time.monotonic())
        if rem < 8.0:
            break
        # Early candidates get more time; later ones get a lower bound.
        t = min(rem, max(12.0, 130.0 * math.exp(-0.07 * i)))
        try:
            w, sc = optimize_weights_for_support(S, N, t, seed + 1000 + i, w0)
        except Exception as e:
            print("weight opt failed:", repr(e))
            continue
        if w is None or not np.isfinite(sc):
            continue
        sc = float(sc)
        if sc < best_score:
            a = np.zeros(N, dtype=np.float64)
            a[S] = w
            a = prepare_full_sequence(a)
            fs = fast_full_score(a)
            if np.isfinite(fs):
                a = a  # keep
            best_score = min(best_score, fs if np.isfinite(fs) else sc)
            best_full = a.copy()
            print(f"support {i} m={S.size} sparse_score={sc:.8f} full_fast={fast_full_score(a):.8f}")

    final_deadline = total_start + budget_s - 8.0
    rem_final = max(10.0, final_deadline - time.monotonic())
    if rem_final > 20.0 and best_full is not None:
        try:
            a_fin, s_fin = full_local_search(best_full, N, rng, rem_final * 0.75)
            if np.isfinite(s_fin) and s_fin < best_score:
                best_score = float(s_fin)
                best_full = a_fin
                print(f"full local improved: {best_score:.8f}")
        except Exception as e:
            print("full local failed:", repr(e))

        rel = final_deadline - time.monotonic()
        if rel > 20.0:
            try:
                a_par, s_par = full_local_search(parent_full, N, rng, rel * 0.5)
                if np.isfinite(s_par) and s_par < best_score:
                    best_score = float(s_par)
                    best_full = a_par
                    print(f"parent full local improved: {best_score:.8f}")
            except Exception:
                pass

    final_deadline = total_start + budget_s - 4.0
    # Verify the final answer with the official evaluator.
    candidates_to_verify = [("parent", parent_full), ("best", best_full)]
    dedup = []
    seen_keys = set()
    for name, a in candidates_to_verify:
        try:
            a = prepare_full_sequence(a)
            if a.size != N:
                a = resize_or_fallback(a, N)
            key = tuple(np.round(a, 12).tolist())
            if key in seen_keys:
                continue
            seen_keys.add(key)
            dedup.append((name, a))
        except Exception:
            continue

    final_a = parent_full.copy()
    final_score = np.inf
    for name, a in dedup:
        try:
            s = float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            s = np.inf
        if np.isfinite(s) and s < final_score:
            final_score = s
            final_a = a
        print(f"verified {name}: {s:.8f}")

    if not np.isfinite(final_score) or final_a is None or final_a.size != N:
        final_a = prepare_full_sequence(parent_full)
        try:
            s = float(evaluate_sequence([float(x) for x in final_a]))
        except Exception:
            final_a = np.ones(N, dtype=np.float64)

    return [float(x) for x in final_a]


def resize_or_fallback(a, N):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == N:
        return prepare_full_sequence(a)
    if a.size == 0:
        return np.ones(N, dtype=np.float64)
    # Simple linear resample, then normalize.
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, N)
    out = np.interp(x_new, x_old, a)
    return prepare_full_sequence(out)
