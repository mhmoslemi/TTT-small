def project_simplex(v):
    """Project v onto the standard simplex {x >= 0, sum(x) = 1}. O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]

def project_simplex(v):
    """Project v onto the standard simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute score for a normalized sequence (sum=1)."""
    n = len(a)
    c = np.convolve(a, a, mode='full')

"""
Minimize the evaluation score using two-phase projected gradient descent:
Phase 1: Minimize L2 norm of autocorrelation (smooth convex problem)
Phase 2: Minimize peak of autocorrelation (subgradient method)
Uses simplex projection to enforce non-negativity and normalization.
Explores multiple sequence lengths and starting points.
"""

import numpy as np
import time


def project_simplex(v):
    """Project v onto the standard simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_score_normalized(a):
    """Compute score for a normalized sequence (sum=1). O(n log n) via FFT."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * np.max(c)


def optimize_sequence(a, deadline, verbose=True):
    """Optimize a normalized sequence to minimize the evaluation score."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = np.sum(a)
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = eval_score_normalized(a)
    a_curr = a.copy()

    # Phase 1: Minimize sum_k c[k]^2 (smooth convex)
    step = 0.5
    n_l2_iters = min(3000, max(100, int((deadline - time.monotonic()) * 200)))

    for it in range(n_l2_iters):
        if time.monotonic() >= deadline - 60:
            break
        c = np.convolve(a_curr, a_curr, mode='full')
        a_rev = a_curr[::-1]
        conv_ca = np.convolve(c, a_rev, mode='full')
        grad = 2.0 * conv_ca[n - 1:n - 1 + n]

        a_new = a_curr - step * grad
        a_new = project_simplex(a_new)

        sc = eval_score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a_curr = a_new
        step *= 0.998
        if verbose and it % 500 == 0:
            print(f"    L2 phase iter {it}, best score: {best_score:.6f}")

    # Phase 2: Minimize max c[k] using subgradient
    step = 0.1
    k_max_cached = int(np.argmax(np.convolve(a_curr, a_curr, mode='full')))
    eval_interval = 10

    while time.monotonic() < deadline:
        if eval_interval > 0 and (
            (int(time.monotonic() * 100) % eval_interval) < 1
        ):
            c = np.convolve(a_curr, a_curr, mode='full')
            k_max_cached = int(np.argmax(c))

        # Subgradient from cached max lag
        grad = np.zeros(n)
        j_lo = max(0, k_max_cached - n + 1)
        j_hi = min(n - 1, k_max_cached)
        if j_hi >= j_lo:
            source = k_max_cached - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] = 2.0 * a_curr[source]

        a_new = a_curr - step * grad
        a_new = project_simplex(a_new)
        a_curr = a_new
        step = max(step * 0.9998, 0.01)

        # Periodic evaluation
        if int(time.monotonic() * 10) % 50 == 0:
            sc = eval_score_normalized(a_curr)
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                if verbose:
                    print(f"    Max phase improvement: score = {best_score:.6f}")

    return best_a, best_score
