```python
"""
LP-based and subgradient optimization for flat autocorrelation sequences.
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 by iteratively finding directions
that flatten the autocorrelation, using linear programming and projected
subgradient methods.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_convolution_matrix(a):
    """Build sparse matrix A where A[k,i] = a[k-i] if valid, else 0.
    Shape: (2n-1) x n
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for i in range(n):
        # Column i: A[k, i] = a[k-i] for k in [i, i+n-1] (clipped to [0, 2n-2])
        k_start = i  # k-i = 0 => k = i
        k_end = min(i + n - 1, 2*n - 2)
        for k in range(k_start, k_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n)), len(vals)

def solve_lp_direction(a, M, verbose=False):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 3000:
        # Too large for efficient LP, return None
        return None
    
    A, nnz = build_convolution_matrix(a)
    if nnz > 10_000_000:
        return None
    
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, method='highs', 
                     options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def subgradient_step(a, eta, rng):
    """One step of projected subgradient descent on max(a*a)."""
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    # Find all active constraints
    active = np.where(np.abs(c - M) < 1e-10 * M)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    # Subgradient: average of gradients at active points
    # Gradient of (a*a)[k] w.r.t. a[i] = 2*a[k-i] if 0 <= k-i < n
    grad = np.zeros(n)
    for k in active:
        # grad[i] += 2*a[k-i] for valid i
        # k-i in [0, n-1] => i in [k-(n-1), k]
        i_start = max(0, k - n + 1)
        i_end = min(n-1, k)
        for i in range(i_start, i_end + 1):
            grad[i] += 2 * a[k - i]
    
    # Average over active set
    grad /= len(active)
    
    # Step and project
    a_new = a - eta * grad
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(n) / n
    else:
        a_new /= s
    
    return a_new, M

def line_search(a, g, t_max=0.5):
    """Find best t in [0, t_max] to minimize max(((1-t)a + tg)*((1-t)a + tg))."""
    n = len(a)
    best_t = 0
    best_M = np.inf
    
    # Golden section search
    phi = (np.sqrt(5) - 1) / 2
    a_lo = 0.0
    a_hi = t_max
    c_pt = a_hi - phi * (a_hi - a_lo)
    d_pt = a_lo + phi * (a_hi - a_lo)
    f_c = max(np.convolve((1-c_pt)*a + c_pt*g, (1-c_pt)*a + c_pt*g))
    f_d = max(np.convolve((1-d_pt)*a + d_pt*g, (1-d_pt)*a + d_pt*g))
    
    for _ in range(20):

            a_hi = d_pt
            d_pt = c_pt
            f_d = f_c
            c_pt = a_hi - phi * (a_hi - a_lo)
            f_c = max(np.convolve((1-c_pt)*a + c_pt*g, (1-c_pt)*a + c_pt*g))
        else:
            a_lo = c_pt
            c_pt = d_pt
            f_c = f_d
            d_pt = a_lo + phi * (a_hi - a_lo)
            f_d = max(np.convolve((1-d_pt)*a + d_pt*g, (1-d_pt)*a + d_pt*g))
    
    best_t = (a_lo + a_hi) / 2
    return best_t

    best_score = 2 * n * best_M  # since sum = 1
    
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() < deadline - 10:
        c = np.convolve(a, a)
        M = c.max()
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
            best_score = 2 * n * M
            no_improve = 0
            if verbose and iter_count % 50 == 0:

        else:
            no_improve += 1
        
        # Decide method
        if use_lp and n <= 2000 and iter_count % 3 == 0:
            # LP-based step
            g = solve_lp_direction(a, M)
            if g is not None and g.sum() > 1e-10:
                g = g / g.sum()  # Normalize to same sum as a
                t = line_search(a, g, t_max=0.3)
                a = (1 - t) * a + t * g

                if s > 1e-15:

            else:
```

```python
"""
LP-based and subgradient optimization for flat autocorrelation sequences.
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using:
1. LP direction finding (maximize sum(g) s.t. a*g <= M, g>=0)
2. Projected subgradient descent as fast alternative
3. Line search for effective step sizes
4. Multiple diverse starting points for robustness
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix(a):
    """Build sparse matrix A where A[k,i] = a[k-i] if 0<=k-i<n else 0. Shape (2n-1, n)."""
    n = len(a)
    cols = np.repeat(np.arange(n), n)
    rows = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    data = np.tile(a, n)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 2500:
        return None
    try:
        A = build_conv_matrix(a)
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        res = linprog(c, A_ub=A, b_ub=b, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = np.clip(res.x, 0, None)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None


def subgradient_step(a, step_size):
    """One step of projected subgradient descent on max(a*a). Returns (new_a, M)."""
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    if M <= 0:
        return a, M
    tol = 1e-8 * M
    active = np.where(c >= M - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    grad = np.zeros(n)
    for k in active:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            grad[i_start:i_end + 1] += 2.0 * a[k - i_start:k - i_start + len(range(i_start, i_end + 1))]
    grad /= len(active)
    a_new = a - step_size * grad
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(n) / n
    else:
        a_new /= s
    return a_new, M


def line_search(a, g, t_max=0.4):
    """Find best t in [0, t_max] to minimize max(((1-t)a+tg)*((1-t)a+tg))."""
    n = len(a)
    M_at_0 = np.convolve(a, a).max()
    best_t = 0.0
    best_M = M_at_0
    # Try several t values (faster than golden section for our purpose)
    t_vals = np.array([0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0])
    t_vals = t_vals[t_vals <= t_max]
    if len(t_vals) == 0:
        return 0.0
    for t in t_vals:
        a_cand = (1 - t) * a + t * g
        a_cand = np.clip(a_cand, 0, None)
        s = a_cand.sum()
        if s < 1e-15:
            continue
        a_cand /= s
        M_cand = np.convolve(a_cand, a_cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_t = t
    return best_t


def optimize_with_lp(a, deadline, verbose=True):
    """Iteratively optimize using LP direction finding + line search."""
    a = np.clip(a.astype(np.float64), 0, None).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    best_a = a.copy()
    best_M = np.convolve(a, a).max()
    best_score = 2.0 * n * best_M
    iter_count = 0
    stuck = 0
    lp_count = 0
    while time.monotonic() < deadline - 15:
        c = np.convolve(a, a)
        M = c.max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
            best_score = 2.0 * n * M
            stuck = 0
            if verbose and iter_count % 20 == 0:
                print(f"    [LP] iter={iter_count}, M={M:.8f}, score={best_score:.6f}")
        else:
            stuck += 1
        # Determine step method
        use_lp = (n <= 2500 and (iter_count % 2 == 0 or stuck > 50))
        if use_lp:
            g = solve_lp_direction(a, M)
            if g is not None:
                lp_count += 1
                g = g / g.sum()
                t = line_search(a, g, t_max=0.3)
                if t > 0:
                    a_new = (1 - t) * a + t * g
                    a_new = np.clip(a_new, 0, None)
                    s2 = a_new.sum()
                    if s2 > 1e-15:
                        a = a_new / s2
                    else:
                        a = best_a.copy()
                        stuck += 10
                else:
                    # No improvement from LP direction, use subgradient
                    eta = 0.5 / (M * n)
                    a, _ = subgradient_step(a, eta)
            else:
                eta = 0.5 / (M * n)
                a, _ = subgradient_step(a, eta)
        else:
            eta = 0.5 / (M * n)
            a, _ = subgradient_step(a, eta)
        iter_count += 1
        if stuck > 200:
            # Random restart within simplex
            rng = np.random.default_rng(iter_count + n)
            perturbation = rng.standard_normal(n) * 0.05
            a = a + perturbation
            a = np.clip(a, 0, None)
            s3 = a.sum()
            if s3 > 1e-15:
                a /= s3
            else:
                a = best_a.copy()
            stuck = 0
    return best_a, best_score


def optimize_with_subgradient(a, deadline, verbose=True):
    """Optimize using projected subgradient descent with decreasing step size."""
    a = np.clip(a.astype(np.float64), 0, None).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    best_a = a.copy()
    best_M = np.convolve(a, a).max()
    best_score = 2.0 * n * best_M
    a_curr = a.copy()
    base_step = 1.0 / (n ** 0.5)
    iter_count = 0
    while time.monotonic() < deadline - 15:
        c = np.convolve(a_curr, a)
        M = c.max()
        if M < best_M:
            best_M = M
            best_a = a_curr.copy()
            best_score = 2.0 * n * M
            if verbose and iter_count % 100 == 0:
                print(f"    [SG] iter={iter_count}, M={M:.8f}, score={best_score:.6f}")
        # Polyak step size
        if best_M > 0 and M > best_M:
            step_size = (M - best_M) / (M * M * max(1, M))
            step_size = min(step_size, base_step)
        else:
            step_size = base_step / (1 + iter_count * 0.01)
        a_curr, _ = subgradient_step(a_curr, step_size)
        # Periodically add small noise to escape local minima
        if iter_count % 500 == 499:
            rng = np.random.default_rng(iter_count)
            a_curr = a_curr + 0.02 * rng.standard_normal(n)
            a_curr = np.clip(a_curr, 0, None)
            s2 = a_curr.sum()
            if s2 > 1e-15:
                a_curr /= s2
        iter_count += 1
    return best_a, best_score


def local_search(a, deadline, verbose=True):
    """Fine local search with small random perturbations."""
    a = np.clip(a.astype(np.float64), 0, None).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * n
    a /= s
    best_a = a.copy()
    best_M = np.convolve(a, a).max()
    best_score = 2.0 * n * best_M
    rng = np.random.default_rng(12345)
    scale = 0.01
    iter_count = 0
    while time.monotonic() < deadline - 5:
        # Block perturbation
        block_size = max(1, n // 50)
        start = int(rng.integers(0, n - block_size + 1))
        factor = np.exp(rng.normal(0, 0.05))
        a_cand = a.copy()
        a_cand[start:start + block_size] *= factor
        a_cand = np.clip(a_cand, 0, None)
        s2 = a_cand.sum()
        if s2 > 1e-15:
            a_cand /= s2
        M_cand = np.convolve(a_cand, a_cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = a_cand.copy()
            a = a_cand
            best_score = 2.0 * n * M_cand
            if verbose and iter_count % 100 == 0:
                print(f"    [LS] iter={iter_count}, M={M_cand:.8f}, score={best_score:.6f}")
        else:
            # Decrease scale occasionally
            if iter_count % 200 == 199:
                scale *= 0.9
        iter_count += 1
    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(convolve(a,a))/sum(a)^2.
    
    Uses LP-based direction finding combined with subgradient descent and local search.
    """
    deadline = time.monotonic() + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(seed)
    
    best_sequence = None
    best_score = np.inf
    
    # Phase 1: Generate diverse starting points
    starts = []
    
    # From height_sequence_1 if available
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.size > 0 and np.all(np.isfinite(hs)):
            hs_clipped = np.clip(hs, 0, None)
            if hs_clipped.sum() > 1e-15:
                starts.append(hs_clipped)
    except (NameError, Exception):
        pass
    
    # Random starts of various sizes
    for n in [400, 800, 1500, 2500]:
        starts.append(rng.random(n))
    
    # Structured starts
    for n in [600, 1200]:
        t = np.arange(n) / n
        starts.append(np.maximum(0.01, 0.5 + 0.5 * np.cos(2 * np.pi * t)))
        starts.append(np.maximum(0.01, 0.3 + 0.7 * np.abs(np.sin(2 * np.pi * t))))
    
    # Binary starts
    for n in [400, 800]:
        starts.append((rng.random(n) > 0.5).astype(float))
    
    # Trim height_sequence_1 to different sizes
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.size > 1000:
            for trim_n in [1000, 2000, 3000]:
                if trim_n < hs.size:
                    starts.append(np.clip(hs[:trim_n], 0, None))
    except (NameError, Exception):
        pass
    
    print(f"Total starting points: {len(starts)}")
    
    # Phase 2: Quick pass with subgradient to rank starting points (60s)
    quick_deadline = time.monotonic() + 60
    ranked = []
    for i, start in enumerate(starts):
        if time.monotonic() > quick_deadline:
            ranked.append((999.0, start))
            continue
        result, score = optimize_with_subgradient(start, min(deadline, time.monotonic() + 15), verbose=False)
        ranked.append((score, result))
        print(f"  Quick pass {i}: n={len(start)}, score={score:.4f}")
    
    # Sort by score, keep top ones
    ranked.sort(key=lambda x: x[0])
    
    # Phase 3: LP-based refinement on top starting points
    print("\nPhase 3: LP-based refinement")
    top_k = min(5, len(ranked))
    for i in range(top_k):
        if time.monotonic() > deadline - 60:
            break
        score_i, seq_i = ranked[i]
        print(f"  Refining candidate {i} (initial score={score_i:.4f}, n={len(seq_i)})")
        result, score = optimize_with_lp(seq_i, min(deadline - 60, time.monotonic() + 300), verbose=True)
        # Evaluate with the actual function
        actual_score = evaluate_sequence(result.tolist())
        if np.isfinite(actual_score) and actual_score < best_score:
            best_score = actual_score
            best_sequence = result
            print(f"    *** New best: {best_score:.6f} ***")
    
    # Also try the best quick-pass result with LP
    if time.monotonic() < deadline - 120:
        print("\n  Additional LP refinement on best quick-pass result:")
        result, score = optimize_with_lp(ranked[0][1], min(deadline - 60, time.monotonic() + 300), verbose=True)
        actual_score = evaluate_sequence(result.tolist())
        if np.isfinite(actual_score) and actual_score < best_score:
            best_score = actual_score
            best_sequence = result
            print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 4: Local search on the best solution
    print("\nPhase 4: Local search")
    if best_sequence is not None and time.monotonic() < deadline - 20:
        result, score = local_search(best_sequence, deadline, verbose=True)
        actual_score = evaluate_sequence(result.tolist())
        if np.isfinite(actual_score) and actual_score < best_score:
            best_score = actual_score
            best_sequence = result
            print(f"    *** New best after local search: {best_score:.6f} ***")
    
    # Fallback
    if best_sequence is None:
        print("Using fallback all-ones sequence")
        best_sequence = np.ones(100)
        best_score = 2.0
    
    print(f"\nFinal best score: {best_score:.6f}, sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
