from scipy import sparse
from scipy.optimize import linprog

def build_conv_matrix(f):
    """Build the sparse convolution matrix C where (C@g)[k] = (f*g)[k]."""
    n = len(f)
    m = 2*n - 1
    rows = []
    cols = []
    vals = []
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # g index i = k - j, so for j in [j_start, j_end], i = k - j is in [k-j_end, k-j_start]
        # Actually, let me re-derive.
        # (f*g)[k] = sum_j f[j] * g[k-j] where 0<=j<n and 0<=k-j<n
        # Let i = k-j, so j = k-i. Need 0<=k-i<n and 0<=i<n.
        # So i ranges from max(0, k-n+1) to min(n-1, k).
        # And f[k-i] is the coefficient.
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    C = sparse.csr_matrix((vals, (rows, cols)), shape=(m, n))
    return C

    return None

def build_conv_matrix_sparse(f):
    """Build sparse matrix C where (C @ g)[k] = (convolve(f, g))[k]."""
    n = len(f)
    m = 2 * n - 1
    # For output position k, the input index i ranges from max(0, k-n+1) to min(n-1, k)
    # C[k, i] = f[k - i]
    
    row_indices = []
    col_indices = []
    data = []
    
    for k in range(m):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        i_range = range(i_start, i_end + 1)
        row_indices.extend([k] * len(i_range))
        col_indices.extend(list(i_range))
        data.extend(f[k - i] for i in i_range)
    
    C = sparse.csr_matrix((data, (row_indices, col_indices)), shape=(m, n))
    return C
