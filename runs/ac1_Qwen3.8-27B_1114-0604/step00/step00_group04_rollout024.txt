            score = evaluate_sequence(f.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = f.copy()
                elapsed = time.monotonic() - t0
                print(f"  p={p} n={n} score={score:.10f} (elapsed {elapsed:.1f}s)")
        if time.monotonic() > deadline - 60:
            break

    print(f"Phase 1 best: {best_score:.10f}")

    if best_seq is None:
        # Fallback: uniform sequence gives score = 2
        best_seq = np.ones(100, dtype=np.float64)
        best_score = 2.0

    # Phase 2: Local search with block perturbations
    if time.monotonic() < deadline:
        print("Phase 2: Local search with block perturbations")
        cur = best_seq.copy()
        cur_score = best_score
        n = len(cur)
        step = 0
        no_improve = 0
        t_start = time.monotonic()

        while time.monotonic() < deadline:
            step += 1
            if step % 500 == 0:
                elapsed = time.monotonic() - t_start
                print(f"  Step {step}: cur={cur_score:.10f} best={best_score:.10f} "
                      f"({elapsed:.0f}s elapsed)")

            # Annealing: reduce perturbation size over time
            progress = min(1.0, step / 10000.0)
            sigma = 0.08 * (1.0 - 0.7 * progress)

            # Choose block width (wider blocks for larger n)
            min_w = max(1, n // 500)
            max_w = max(min_w + 1, n // 5)
            w = int(rng.integers(min_w, max_w))
            s = int(rng.integers(0, max(1, n - w)))

            cand = cur.copy()
            cand[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
            np.clip(cand, 0.0, 1000.0, out=cand)

            sc = evaluate_sequence(cand.tolist())
            if np.isfinite(sc) and sc < cur_score:
                cur = cand
                cur_score = sc
                no_improve = 0
                if sc < best_score:
                    best_score = sc
                    best_seq = cur.copy()
            else:
                no_improve += 1

            # Escape mechanism: try a larger random perturbation if stuck
            if no_improve > 300:
                no_improve = 0
                cand = cur.copy()
                w2 = int(rng.integers(n // 20, n // 2))
                s2 = int(rng.integers(0, max(1, n - w2)))
                cand[s2:s2 + w2] *= np.exp(rng.normal(0.0, 0.2, size=w2))
                np.clip(cand, 0.0, 1000.0, out=cand)
                sc = evaluate_sequence(cand.tolist())
                if np.isfinite(sc) and sc < cur_score:
                    cur = cand
                    cur_score = sc
                    if sc < best_score:
                        best_score = sc
                        best_seq = cur.copy()

    print(f"Final best score: {best_score:.10f}")
    return [float(x) for x in best_seq]
