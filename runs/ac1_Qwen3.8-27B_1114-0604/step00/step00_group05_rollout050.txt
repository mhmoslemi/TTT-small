    n = len(a)

def proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target} via water-filling."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u * rho) > (cssv - target)
    idx = np.flatnonzero(cond)
    if len(idx) == 0:
        return np.full(n, target / n)
    r = idx[-1] + 1
    theta = (cssv[r - 1] - target) / float(r)
    return np.maximum(v - theta, 0.0)

    """Build (2n-1)x n matrix A where A[k,j]=a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            A[k, js:je + 1] = a[k - js:k - je + 1][::-1]
    return A

    return None

    g /= max(1, len(active))
    return g
