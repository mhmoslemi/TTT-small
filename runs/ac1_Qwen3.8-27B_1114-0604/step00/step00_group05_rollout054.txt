"""
Minimize  2*n*max(convolve(a,a)) / (sum a)^2  over non-negative a.

Key insight: the left-triangle autocorrelation of the binomial sequence
    a[i] = C(2i,i)/4^i  (coefficients of (1-x)^{-1/2})
is exactly flat (equal to a[0]^2) for lags 0..n-1, so its score tends to
pi/2 ~= 1.5708 from above (well below the 2.00 uniform baseline).  Because
the binomial leaves the "right" half of the autocorrelation well below the
peak, the peak sits on the left edge and can be trimmed: lowering the large
leading element while (via the sum=1 renormalisation) lifting the small tail
reduces max R without increasing the peak, lowering the score further.

Algorithm:
  1. Seed with the binomial (and a couple of analytic "lean" variants).
  2. Run exponentiated-gradient (multiplicative) descent on a sharp
     power-mean of the autocorrelation, with an FFT-based closed-form
     gradient and a sum=1 / non-negative constraint maintained by
     construction.  This trims the peak lag and leans the mass.
  3. Polish with evaluator-confirmed multiplicative block local search.
Always returns the best sequence found (never the worse 2.00 baseline).
"""

import time
import numpy as np


def _next_pow2(x):
    L = 1
    while L < x:
        L *= 2
    return L


def _fft_conv_full(a, b):
    n = a.size
    m = b.size
    L = _next_pow2(n + m - 1)
    fa = np.fft.fft(a, L)
    fb = np.fft.fft(b, L)
    return np.fft.ifft(fa * fb)[:n + m - 1].real

    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    R = _fft_conv_full(a, a)[:2 * n - 1]
    mb = float(np.max(R))
    return float(2.0 * n * mb / (s * s))


def _binomial(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    a /= a.sum()
    return a


def _init_lean(n, tail):
    a = _binomial(n)
    half = n // 2
    if tail > 1.0:
        a[half:] *= tail
    else:
        a[:half] *= (1.0 / tail)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _init_expdecay(n, alpha):
    t = np.arange(n, dtype=np.float64) / float(n)
    a = np.exp(-alpha * t)
    a /= a.sum()
    return a


def _gradient_descent(a0, deadline, p0=4.0, p1=128.0, lr0=0.03,
                      lr_decay=0.9998, lr_min=2e-4, plateau_iters=4000):
    a = np.array(a0, dtype=np.float64, copy=True)
    n = a.size
    L = _next_pow2(3 * n - 2)
    fa = np.fft.fft(a, L)
    best = a.copy()
    best_sc = _true_score(a)

    R = (np.fft.ifft(fa * fa)[:2 * n - 1].real)
    best_sc = float(2.0 * n * np.max(R))

    it = 0
    no_improve = 0
    irange = _next_pow2(10)  # plain counter base
    while True:
        if time.monotonic() >= deadline:
            break
        # sharpen the power-mean over time
        frac = (it / irange) if it < irange else 1.0
        p = p0 * (p1 / p0) ** frac
        lr = max(lr_min, lr0 * (lr_decay ** it))

        fa = np.fft.fft(a, L)
        R = np.fft.ifft(fa * fa)[:2 * n - 1].real
        sc = float(2.0 * n * np.max(R))

        Q = np.power(np.clip(R, 1e-300, None), p - 1.0)
        Qr = Q[::-1]
        fQr = np.fft.fft(Qr, L)
        C = np.fft.ifft(fa * fQr)[:3 * n - 2].real
        Csel = C[n - 1:2 * n - 1][::-1]
        g = (4.0 / p) * Csel

        m = float(a @ g)
        step = lr * (g - m)
        step = np.clip(step, -5.0, 5.0)        # stability guard
        a = a * np.exp(-step)
        a = np.clip(a, 1e-300, None)
        a = a / a.sum()

        if sc < best_sc - 1e-9:
            best_sc = sc
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        it += 1
        if no_improve >= plateau_iters:
            break
    return best, best_sc


def _local_search(a0, deadline, seed=0, max_calls=200000):
    rng = np.random.default_rng(seed)
    a = np.array(a0, dtype=np.float64, copy=True)
    n = a.size
    best = a.copy()
    best_sc = _true_score(a)
    sigma = 0.06
    calls = 0
    while time.monotonic() < deadline and calls < max_calls:
        b = a.copy()
        w = int(rng.integers(max(1, n // 200), max(2, n // 8) + 1))
        s = int(rng.integers(0, n - w + 1))
        b[s:s + w] *= np.exp(rng.normal(0.0, sigma))
        if rng.random() < 0.30:
            lo = rng.uniform(0.4, 1.6)
            hi = rng.uniform(0.4, 1.6)
            b *= np.linspace(lo, hi, n)
        b = np.clip(b, 0.0, None)
        sb = b.sum()
        if sb < 1e-12:
            it2 = 0
            continue
        b = b / sb
        calls += 1
        sc = _true_score(b)
        if sc < best_sc - 1e-9:
            best_sc = sc
            best = b
            a = b.copy()
            sigma = max(0.01, sigma * 1.02)
        else:
            # mild cooling + occasional analytic re-injection to escape
            sigma *= 0.9995
            if calls % 400 == 0:
                mix = _binomial(n)
                a = a * (1.0 - 0.10) + mix * 0.10
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
    a = best
    return a, best_sc


def propose_candidate(**kwargs):
    budget = kwargs.get("budget_s", 2400)
    deadline = time.monotonic() + max(30.0, float(budget) - 6.0)
    base_n = int(kwargs.get("n", 4498))

    # ---- guaranteed strong analytic seed ----
    best = _binomial(base_n)
    best_sc = _true_score(best)
    print(f"seed binomial n={base_n} score={best_sc:.6f}", flush=True)

    # ---- gradient refinement from several analytic inits ----
    inits = [
        ("binomial", _binomial(base_n)),
        ("lean2", _init_lean(base_n, 2.0)),
        ("lean5", _init_lean(base_n, 5.0)),
        ("expdec", _init_expdecay(base_n, 3.0)),
    ]
    for name, init in inits:
        if time.monotonic() >= deadline:
            break
        sub_deadline = min(deadline, time.monotonic() + 240.0)
        cand, sc = _gradient_descent(init, sub_deadline)
        sc = _true_score(cand)
        print(f"grad[{name}] score={sc:.6f} best={best_sc:.6f}", flush=True)
        if sc < best_sc:
            best_sc, best = sc, cand

    # ---- polish the current best with local search ----
    if time.monotonic() < deadline:
        sub_deadline = min(deadline, time.monotonic() + max(60.0, (deadline - time.monotonic()) * 0.5))
        cand, sc = _local_search(best, sub_deadline, seed=12345)
        sc = _true_score(cand)
        print(f"local1 score={sc:.6f} best={best_sc:.6f}", flush=True)
        if sc < best_sc:
            best_sc, best = sc, cand

    # ---- try a couple of other lengths if time remains ----
    if time.monotonic() < deadline - 60.0:
        for alt_n in (2400, 8192):
            if time.monotonic() >= deadline - 30.0:
                break
            init = _binomial(alt_n)
            gen_d = min(deadline - 20.0, time.monotonic() + 180.0)
            cand, sc = _gradient_descent(init, gen_d)
            sc = _true_score(cand)
            if sc < best_sc:
                best_sc, best = sc, cand
            print(f"grad alt n={alt_n} score={sc:.6f} best={best_sc:.6f}", flush=True)
            if time.monotonic() < deadline - 30.0:
                ld = min(deadline - 15.0, time.monotonic() + 120.0)
                cand2, sc2 = _local_search(cand, ld, seed=777)
                sc2 = _true_score(cand2)
                if sc2 < best_sc:
                    best_sc, best = sc2, cand2

    # ---- final short local polish with remaining time ----
    if time.monotonic() < deadline - 8.0:
        ld = deadline - 4.0
        cand, sc = _local_search(best, ld, seed=999)
        sc = _true_score(cand)
        if sc < best_sc:
            best_sc, best = sc, cand
        print(f"final local score={sc:.6f} best={best_sc:.6f}", flush=True)

    out = [float(x) for x in np.clip(np.asarray(best), 0.0, 1000.0)]
    print(f"RETURN n={len(out)} internal_score={best_sc:.6f}", flush=True)
    return out
