import numpy as np
import time
import math

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using projected gradient descent
on a power-mean smoothed objective with FFT-based gradient computation.
Initializes from an antisymmetric sine perturbation of uniform sequence,
which already achieves score < 2. Multiple restarts with perturbations
help escape local minima.
"""


def _fft_size(n):
    """Next power of 2 >= 3n."""
    N = 1
    while N < 3 * n:
        N *= 2
    return N


def _score(a, N):
    """Fast score computation via FFT."""
    n = len(a)
    a_fft = np.fft.fft(a, N)
    c = np.real(np.fft.ifft(a_fft * a_fft))[:2 * n - 1]
    max_c = np.max(c)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    return 2.0 * n * max_c / (s * s)


def _compute_grad(a, N, p):
    """
    Compute gradient of M_p(c) where c = a*a (linear conv).
    Returns (grad, M_p).
    grad[m] = 2 * M_p^(1-p) * sum_j c[m+j]^(p-1) * a[j]
    computed via FFT cross-correlation.
    """
    n = len(a)
    a_padded = np.zeros(N)
    a_padded[:n] = a
    a_fft = np.fft.fft(a_padded)

    c = np.real(np.fft.ifft(a_fft * a_fft))[:2 * n - 1]
    c = np.maximum(c, 1e-30)

    c_p = c ** p
    sum_cp = np.sum(c_p)
    M_p = sum_cp ** (1.0 / p)
    scale = 2.0 * M_p ** (1.0 - p)

    w = c ** (p - 1)
    w_padded = np.zeros(N)
    w_padded[:2 * n - 1] = w
    w_fft = np.fft.fft(w_padded)

    # cross_correlation(w, a)[m] = sum_j w[m+j]*a[j] = ifft(fft(w)*conj(fft(a)))
    grad = np.real(np.fft.ifft(w_fft * np.conj(a_fft)))[:n]
    grad *= scale

    return grad, M_p

    it = 0
    stall = 0
    patience = 3000
    p = p_start

    while time.monotonic() < deadline:
        # Gradually increase p
        p = min(p_end, p + 0.001)

        grad, M_p = _compute_grad(a, N, p)
        grad -= np.mean(grad)  # zero-sum constraint

        g_norm = np.max(np.abs(grad))
        if g_norm < 1e-18:
            break

        # Momentum
        momentum_buf = beta * momentum_buf + (1 - beta) * grad
        step_size = lr / (g_norm + 1e-18)
        a = a - step_size * momentum_buf

        # Project
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-15:
            a = best_a.copy()
            s = np.sum(a)
        a = a / s

        sc = _score(a, N)
        it += 1

        if sc < s_best:
            best_a = a.copy()
            s_best = sc
            stall = 0
            if it % 500 == 0 or s_best < 1.55:
                print(f"  [{verbose_tag}] iter={it} score={s_best:.6f} p={p:.1f}")
        else:
            stall += 1

        # Decay lr
        lr *= 0.9997

        # Restart on stall
        if stall >= patience:
            a = best_a.copy()
            noise = rng.normal(0, 0.03 / math.sqrt(n), n)
            a = a + noise
            a = np.maximum(a, 0.0)
            s = np.sum(a)
            a = a / s
            momentum_buf = np.zeros(n)
            lr = 0.04 / n
            stall = 0
            p = p_start
            print(f"  [{verbose_tag}] Restart at iter={it}, best={s_best:.6f}")

        # Periodic small perturbation
        if it % 2000 == 0:
            noise = rng.normal(0, 0.01 / math.sqrt(n), n)
            a = a + noise
            a = np.maximum(a, 0.0)
            s = np.sum(a)
            a = a / s

    print(f"  [{verbose_tag}] Done: iter={it}, best={s_best:.6f}")
    return best_a, s_best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-start projected gradient descent minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses antisymmetric sine initializations that exploit the antisymmetry
    to reduce the center lag of the autocorrelation below the uniform baseline.
    Gradient computed in O(n log n) via FFT cross-correlation.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s) - 10.0

    # Determine n from parent
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad parent")
        n = int(parent.size)
    except Exception:
        n = 6121
        parent = None

    N = _fft_size(n)
    t = np.arange(n, dtype=np.float64)

    # ---- Initialization 1: single antisymmetric sine + 2nd harmonic ----
    a1 = 1.0 + 0.95 * np.sin(2.0 * np.pi * (t + 0.5) / n)
    a1 += 0.08 * np.sin(4.0 * np.pi * (t + 0.5) / n)
    a1 = np.maximum(a1, 0.0)
    a1 = a1 / np.sum(a1)

    # ---- Initialization 2: multi-frequency antisymmetric ----
    a2 = np.ones(n)
    for k in range(1, 8):
        amp = 0.12 / k
        a2 += amp * np.sin(2.0 * np.pi * k * (t + 0.5) / n)
    a2 = np.maximum(a2, 0.0)
    a2 = a2 / np.sum(a2)

    # ---- Initialization 3: parent (if valid) ----
    a3 = None
    if parent is not None:
        a3 = parent.copy()
        s3 = np.sum(a3)
        if s3 > 0:
            a3 = a3 / s3

    # Evaluate inits
    candidates = []
    candidates.append(("sine1", a1.copy()))
    candidates.append(("sine_multi", a2.copy()))
    if a3 is not None:
        candidates.append(("parent", a3.copy()))

    scores = []
    for tag, a in candidates:
        sc = _score(a, N)
        scores.append(sc)
        print(f"Init {tag}: score = {sc:.6f}")

    # Allocate time: give most time to best initializer
    t_main = total_budget * 0.7
    t_sec = total_budget * 0.25
    t_ter = total_budget * 0.05

    # Sort by score
    order = sorted(range(len(candidates)), key=lambda i: scores[i])
    main_tag, main_a = candidates[order[0]]
    sec_tag, sec_a = candidates[order[1]] if len(candidates) > 1 else (main_tag, main_a.copy())
    ter_tag, ter_a = candidates[order[2]] if len(candidates) > 2 else (main_a.copy() + rng.normal(0, 0.01, n))
    ter_a = np.maximum(ter_a, 0)
    ter_a = ter_a / np.sum(ter_a)

    best_a, best_sc = _pgd_loop(main_a, n, N, 10.0, 40.0, t_main, rng, main_tag)

    a2_out, s2 = _pgd_loop(sec_a, n, N, 10.0, 30.0, t_sec, rng, sec_tag)
    if s2 < best_sc:
        best_a, best_sc = a2_out, s2

    if len(candidates) > 2:
        a3_out, s3 = _pgd_loop(ter_a, n, N, 10.0, 25.0, t_ter, rng, ter_tag)
        if s3 < best_sc:
            best_a, best_sc = a3_out, s3

    # Final polish: short high-p run on the best
    remaining = total_budget - t_main - t_sec - (t_ter if len(candidates) > 2 else 0)
    if remaining > 30:
        best_a, best_sc = _pgd_loop(best_a, n, N, 20.0, 60.0, remaining, rng, "polish")

    # Verify with actual evaluator
    final_list = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    print(f"FINAL: score = {final_score:.6f}, n = {n}")

    return final_list
