"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 over non-negative step heights.

Strategy
--------
1. Build a pool of candidate initializations (provided parent, power
   transforms of it, and cosine/U-shaped parametric families) and keep the
   best one.
2. Primary driver: LP-based iterative refinement (the state-of-the-art idea).
   Given current a and M = max(conv(a,a)), solve the LP
       max sum(b)  s.t.  conv(a,b)[k] <= M  (active/top constraints),  b >= 0,
   which yields the maximally-massive non-negative direction whose
   convolution with a stays under the current peak.  Rescale b to unit sum
   and blend a <- (1-t)a + t b.  Only the top-fraction of convolution
   positions are used as constraints to keep the LP fast.
3. Fine-tuning: FFT-accelerated projected subgradient descent on
   max conv position on the remaining time budget.
4. Memory-safe sparse constraint construction (numpy arrays, no giant python
   lists) and robust fallbacks so the function always returns a valid list.
"""

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy import sparse


def _conv_fft(x):
    """Full linear convolution of a real 1-D array via FFT (zero-padded)."""
    nn = int(len(x))
    size = 2 * nn - 1
    fs = 1
    while fs < size:
        fs <<= 1
    X = fft(x, fs)
    return ifft(X * X).real[:size]


def _build_csr(a, active):
    """
    Sparse matrix C of shape (len(active), len(a)) with
    C[i, j] = a[k - j] where k = active[i] and 0 <= k - j < len(a), else 0.
    Built with numpy arrays (no per-element python list appends of indices).
    """
    nn = len(a)
    n_act = len(active)
    if n_act == 0:
        return sparse.csr_matrix((0, nn))

    lengths = np.empty(n_act, dtype=np.int64)
    starts = np.empty(n_act, dtype=np.int64)
    ends = np.empty(n_act, dtype=np.int64)
    for i, k in enumerate(active):
        lo = 0 if k - nn + 1 < 0 else (k - nn + 1)
        hi = nn - 1 if k > nn - 1 else k
        lengths[i] = hi - lo + 1
        starts[i] = lo
        ends[i] = k - hi  # index in `a` of a[k-hi] (the smallest value in row)
    total = int(lengths.sum())
    indptr = np.concatenate(([0], np.cumsum(lengths)).astype(np.int64))
    data = np.empty(total, dtype=np.float64)
    indices = np.empty(total, dtype=np.int64)
    pos = 0
    for i in range(n_act):
        L = int(lengths[i])
        if L <= 0:
            continue
        lo = int(starts[i])
        # row values: a[k - j] for j = lo..hi  =>  a[e_hi]..a[e_lo] with e_lo = k-lo, e_hi=k-hi
        # stored re-sliced then reversed to be contiguous from `ends[i]` upward
        seg = a[ends[i]: ends[i] + L][::-1]
        data[pos:pos + L] = seg
        indices[pos:pos + L] = np.arange(lo, lo + L, dtype=np.int64)
        pos += L
    return sparse.csr_matrix((data, indices, indptr), shape=(n_act, nn))


def _lp_solve(a, M, top_frac, time_limit):
    """Solve reduced LP, return unit-sum non-negative b, or None."""
    nn = len(a)
    c = _conv_fft(a)
    if not np.isfinite(M) or M <= 0.0:
        return None
    threshold = np.quantile(c, 1.0 - float(top_frac))
    active = np.where(c >= threshold)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(c))])
    C = _build_csr(a, active)
    c_obj = -np.ones(nn, dtype=np.float64)
    b_ub = np.full(active.size, M, dtype=np.float64)
    bounds = [(0.0, None)] * nn
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if getattr(res, "status", 1) == 0 and res.x is not None:
            b = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, None)
            s_b = float(b.sum())
            if s_b > 1e-10:
                return b / s_b
    except Exception as e:
        print("LP error:", e, flush=True)
    return None


def _subgradient_step(a, eta, top_frac):
    """One projected subgradient step. Returns (a_new, max_c, new_eta)."""
    nn = len(a)
    c = _conv_fft(a)
    max_c = float(c.max())
    if not np.isfinite(max_c) or max_c <= 1e-15:
        return a, max_c, eta * 0.5
    threshold = np.quantile(c, 1.0 - float(top_frac))
    S = np.where(c >= threshold)[0]
    if S.size == 0:
        return a, max_c, eta
    s_vec = np.zeros(2 * nn - 1, dtype=np.float64)
    s_vec[S] = 1.0
    fs = 1
    while fs < 2 * nn - 1:
        fs <<= 1
    S_f = fft(s_vec, fs)
    a_f = fft(a, fs)
    r = ifft(np.conj(S_f) * a_f).real[:nn]
    g = 2.0 * r / float(S.size)
    a_new = np.clip(a - eta * g, 0.0, None)
    s_new = float(a_new.sum())
    if s_new < 1e-10:
        return a, max_c, eta * 0.5
    a_new = a_new / s_new
    return a_new, max_c, eta

def _maybe_update(a, best, best_score):
    sc = _safe_score(a)
    if sc < best_score:
        return sc, a.copy(), True
    return best_score, best, False
