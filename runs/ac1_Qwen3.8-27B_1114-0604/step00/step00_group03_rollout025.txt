def smooth_topk_subgrad(c, n, topk, tau):
    """Compute subgradient of smooth top-k objective w.r.t. sequence a.
    Returns gradient g of shape (n,) such that d(sum_topk)/da_i = g[i]."""
    m = len(c)
    softweights = np.exp(-((m - 1.0 - np.arange(m)) * tau))
    softweights /= np.sum(softweights)
    cvals = np.asarray(c, dtype=np.float64)
    # Largest topk-1 get weight 1.0, the rest get softweights
    k = min(topk, m)
    # Use partition to find top-k
    if k >= m:
        lam = 0.0
        w = np.ones(m)
    else:
        part = np.argpartition(cvals, m - k)
        top_part = part[m - k:]
        lam = cvals[top_part].min()
        w = np.where(cvals >= lam, 1.0, 0.0).astype(np.float64)
        below_idx = np.where(cvals < lam)[0]
        if len(below_idx) > 0:
            # Bound softweights to below-threshold elements
            sw_below = softweights[below_idx]
            sw_sum = np.sum(sw_below)
            if sw_sum > 1e-30:
                lambda_adj = lam - np.log(sw_sum)
                add = np.exp(-(cvals[below_idx] - lambda_adj))
                w[below_idx] = add
    # For each lag k with weight w[k], subgrad contribution to g[i] is a[k-i]
    # Vectorized: for each valid (i, k) pair
    g = np.zeros(n, dtype=np.float64)
    for k_val in range(m):
        wv = w[k_val]
        if wv <= 0:
            continue
        # g[i] += wv * a[k_val - i] for valid i
        # Valid: 0 <= i < n, 0 <= k_val - i < n
        i_start = max(0, k_val - n + 1)
        i_end = min(k_val, n - 1)
        if i_start > i_end:
            continue
        # indices: i from i_start to i_end, a indices: k_val-i_end to k_val-i_start (reversed)
        a_slice = a_ref[i_end - (k_val - i_end): i_start - (k_val - i_start): -1] if False else None
        # Simpler: a[k_val - i] for i in [i_start, i_end]
        # k_val - i ranges from k_val-i_start down to k_val-i_end
        g[i_start:i_end + 1] += wv * a_ref[k_val - np.arange(i_start, i_end + 1)]
    return g

            best_a = a.copy()

        else:

        # Anneal
        if it % 2000 == 1999:
            topk = max(1, topk // 2)
            tau = max(0.01, tau * 0.5)
            lr = max(0.001, lr * 0.7)

        # Compute subgradient (using current a as a_ref)
        a_ref = a
        g = smooth_topk_subgrad(c, n, topk, tau)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Step
        a_new = a - lr * g / g_norm
        # Project to simplex {x >= 0, sum = 1}
        a_new = np.maximum(0, a_new)
        sn = a_new.sum()
        if sn < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new /= sn
        a = a_new

        # Coordinate descent polish every 500 iterations
        if it % 500 == 499:
            a = coord_descent_polish(a, n, deadline)
            c_check = np.convolve(a, a)
            if float(np.max(c_check)) < best_maxc:
                best_maxc = float(np.max(c_check))
                best_a = a.copy()

        if no_improve > 20000 and it > 1000:
            break

    a_ref = best_a
    return best_a, 2.0 * n * best_maxc
