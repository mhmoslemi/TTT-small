def _lp_direction(a, M, max_time=5.0):
    """Try to find an LP-based improving direction. Returns b or None."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    n = len(a)
    S = a.sum()
    a_n = a / S
    # Build constraint matrix: convolve(a_n, b)[k] <= M/S for all k
    # A[k, i] = a_n[k-i] if 0 <= k-i < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(a_n[j])
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_rhs = np.full(2 * n - 1, M / S)
    bounds = [(0, None)] * n
    t0 = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A_sparse, b_ub=b_rhs, bounds=bounds,
                      method='highs', options={'time_limit': max_time, 'mip_rel_gap': 0.0})
    except Exception:
        return None
    if time.monotonic() - t0 > max_time + 2:
        return None
    if res.status == 0 and res.x is not None:
        b = np.maximum(res.x, 0)
        if b.sum() > S * 1.01:
            return b
    return None

    inits.append((np.arange(1, n + 1, dtype=np.float64) / n) ** 3)
    inits.append(np.exp(np.linspace(0, 2.5, n)))
    inits.append(np.exp(np.linspace(2.5, 0, n)))
    inits.append(np.sin(np.linspace(0.05, np.pi - 0.05, n)))
    # Asymmetric: high-left, low-right
    inits.append(np.concatenate([np.full(n // 2, 2.0 / n), np.full(n - n // 2, 1.0 / n)]))
    # Given sequence
    try:
        gs = np.array(height_sequence_1, dtype=np.float64)
        inits.append(np.clip(gs, 0, 1000))
    except Exception:
        pass

    best = None
    best_score = np.inf
    for init in inits:
        init = np.clip(init, 0, 1000)
        score = evaluate_sequence(init.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = init.copy()

    print(f"Phase 1: best_score={best_score:.6f}")

    if best_score < 1.4:
        print("Already below 1.4, proceeding to refine.")

    # --- Phase 2: Gradient descent with soft-max ---
    a = best.copy()
    S0 = a.sum()
    beta = 15.0
    alpha = 0.5
    patience = 0
    max_patience = 3000
    rng = np.random.default_rng(seed)
    lp_count = 0

    start_time = time.monotonic()
    for it in range(1000000):
        if time.monotonic() > deadline:
            break

        # Check if we should do an LP step (rare)
        if lp_count < 5 and it % 10000 == 5000 and time.monotonic() < deadline - 60:
            c_cur = np.convolve(a, a)
            M_cur = c_cur.max()
            b_dir = _lp_direction(a, M_cur, max_time=8.0)
            lp_count += 1
            if b_dir is not None:
                b_norm = b_dir / b_dir.sum()
                # Mix toward b
                for t in [0.3, 0.5, 0.7]:
                    a_try = (1 - t) * a + t * b_norm * S0
                    a_try = np.clip(a_try, 0, 1000)
                    sc = evaluate_sequence(a_try.tolist())
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best = a_try.copy()
                        a = a_try.copy()
                        alpha *= 0.8
                        patience = 0
                        print(f"  LP step {lp_count} (t={t}): score={best_score:.6f}")
                        break

        # Gradient descent step
        a, cmax = _grad_desc_step(a, beta, alpha, N)

        # Periodic evaluation
        if it % 200 == 0:
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = a.copy()
                patience = 0
                alpha *= 0.998  # gentle decay
                if it % 2000 == 0:
                    print(f"Iter {it}: score={best_score:.6f}, beta={beta:.1f}, alpha={alpha:.4f}")
            else:
                patience += 1
                if patience > max_patience:
                    # Diversify
                    a = a * np.exp(0.3 * rng.normal(0, 1, n))
                    a = np.clip(a, 0, 1000)
                    patience = 0
                    # Increase beta for sharper peak tracking
                    beta = min(beta * 1.3, 200.0)
                    alpha = min(alpha * 1.2, 2.0)
                elif patience % 500 == 0:
                    # Adapt beta
                    beta *= 1.1
                    if beta > 100:
                        beta = 15.0  # reset
                    alpha = min(alpha * 1.1, 1.5)

    # --- Phase 3: Final fine-tuning with smaller steps ---
    if time.monotonic() < deadline - 30:
        a = best.copy()
        alpha = 0.05
        beta = 50.0
        for it in range(50000):
            if time.monotonic() > deadline - 5:
                break
            a, cmax = _grad_desc_step(a, beta, alpha, N)
            if it % 500 == 0:
                score = evaluate_sequence(a.tolist())
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best = a.copy()
                    alpha *= 0.99
                    if it % 5000 == 0:
                        print(f"Phase 3 iter {it}: score={best_score:.6f}")

    final_score = evaluate_sequence(best.tolist())
    print(f"Final score: {final_score:.6f}")
    return [float(x) for x in best]
