"""
Search for a non-negative step-function sequence minimizing
    2*n*max(convolve(a,a))/sum(a)^2.

Strategy:
  * Use FFT-based linear convolution and a smooth log-sum-exp surrogate for the
    maximum convolution coefficient.
  * Optimize full sequences with adaptive multiplicative (natural-gradient)
    updates in log-space, plus a diagonal "sharpening" move that increases
    asymmetry for the currently hottest lag.
  * Warm-start from a low-dimensional antisymmetric cosine basis optimized by
    Adam, and also try several random/structured initializations.
  * Keep a short-sequence fallback (including length 2), and finally return the
    best sequence found (scaled innocuously).
"""

import numpy as np
import time
from numpy.fft import fft, ifft


def next_power_of_two(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(1024, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-12:
        a = np.ones_like(a)
        s = float(a.sum())
    a = a / s
    a = np.maximum(a, 1e-12)
    s2 = float(a.sum())
    if s2 > 0.0:
        a = a / s2
    return a


def scale_for_return(a):
    a = normalize_positive(a)
    n = int(a.size)
    target = float(np.sqrt(2.0 * max(1, n)))
    s = float(a.sum())
    if s > 0.0:
        a = a / s * target
    return a


def safe_evaluate(seq):
    try:
        val = evaluate_sequence(seq)
        val = float(val)
        return val if np.isfinite(val) else float("inf")
    except Exception:
        return float("inf")


def evaluate_array(a):
    if a is None:
        return float("inf")
    out = scale_for_return(a)
    if out.size < 1:
        return float("inf")
    return safe_evaluate([float(x) for x in out])


def get_parent_sequence():
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        arr = np.ones(4096, dtype=np.float64)

def make_uniform(n):
    n = max(2, int(n))
    a = np.ones(n, dtype=np.float64)
    return a / float(n)


def make_cosine_antisym(n, amp):
    n = max(2, int(n))
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = 1.0 + float(amp) * np.cos(np.pi * x)
    return normalize_positive(a)


def make_antisym_random(n, rng, amp):
    n = max(2, int(n))
    z = rng.standard_normal(n)
    z = 0.5 * (z - z[::-1])
    std = float(z.std())
    if std > 1e-12:
        z = z / std
    a = 1.0 + float(amp) * z
    return normalize_positive(a)


def perturb_sequence(a, rng, strength):
    n = int(a.size)
    out = np.asarray(a, dtype=np.float64).copy()
    strength = float(abs(strength))
    mode = int(rng.integers(0, 4))
    if mode == 0:
        out *= np.exp(rng.normal(0.0, strength, n))
    elif mode == 1:
        width = max(1, int(n * rng.uniform(0.02, 0.15)))
        width = min(width, n)
        start = int(rng.integers(0, max(1, n - width + 1)))
        out[start:start + width] *= np.exp(rng.normal(0.0, max(0.05, 1.5 * strength)))
    elif mode == 2:
        z = rng.standard_normal(n)
        z = 0.5 * (z - z[::-1])
        out *= np.exp(0.5 * strength * z)
    else:
        rev = out[::-1]
        m = float(rng.uniform(0.2, 0.8))
        out = m * out + (1.0 - m) * rev
    return normalize_positive(out)


def softmax_x(x, tau):
    M = int(x.size)
    if M <= 0:
        return np.array([], dtype=np.float64)
    p = np.full(M, 1.0 / M, dtype=np.float64)
    t = float(tau) * x
    t = np.where(np.isfinite(t), t, -np.inf)
    if not np.any(np.isfinite(t)):
        return p
    tmax = float(np.max(t))
    if not np.isfinite(tmax):
        return p
    e = np.exp(t - tmax)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return p
    p = e / s
    return p


def estimate_score_with_size(a, N):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n < 1:
        return float("inf")
    A = fft(a, N)
    b = np.real(ifft(A * A))
    b = b[:2 * n - 1]
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    val = 2.0 * n * float(b.max()) / (s * s)
    return float(val) if np.isfinite(val) else float("inf")


def make_fft_context(n):
    N = next_power_of_two(max(4, 3 * int(n) - 2))
    neg_idx = (-np.arange(N, dtype=np.int64)) % N
    phase_neg = np.exp(-2j * np.pi * (int(n) - 1) * np.arange(N) / float(N))
    return N, neg_idx, phase_neg


def correlate_from_A(p, A, N, neg_idx, phase_neg, n):
    Fp = fft(p, N)
    D = phase_neg * A[neg_idx]
    C = ifft(Fp * D).real
    d = C[int(n) - 1:int(n) - 1 + int(n)]
    d = np.where(np.isfinite(d), d, 0.0)
    d = np.maximum(d, 0.0)
    return d


def diag_sharpen(a, k, gamma=4.0):
    n = int(a.size)
    k = int(k)
    out = np.asarray(a, dtype=np.float64).copy()
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo > hi:
        return out
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    j = k - idx
    s = out[idx] + out[j]
    good = s > 1e-15
    if not np.any(good):
        return out
    r = np.zeros(s.shape, dtype=np.float64)
    r[good] = out[idx[good]] / s[good]
    r = np.clip(r, 1e-12, 1.0 - 1e-12)
    g = float(gamma)
    rw = np.power(r[g], g) if isinstance(g, tuple) else np.power(r, g)
    sw = np.power(1.0 - r, g)
    w = rw / (rw + sw)
    out[idx] = s * w
    return out


def param_optimize(n, budget_s, seed):
    budget_s = max(0.0, float(budget_s))
    if budget_s < 5.0:
        return None, float("inf")
    n = max(64, int(n))
    rng = np.random.default_rng(int(seed))
    R = 10
    modes = 2.0 * np.arange(R, dtype=np.float64) + 1.0
    idx = np.arange(n, dtype=np.float64)
    basis = np.cos(np.pi * (modes[None, :] * idx[:, None]) / max(1, n - 1))
    sumbasis = basis.sum(axis=0)
    M = 2 * n - 1
    N, neg_idx, phase_neg = make_fft_context(n)

    best_a = None
    best_score = float("inf")
    n_restarts = 4
    per_time = max(5.0, budget_s / float(n_restarts))

    for r in range(n_restarts):
        start_t = time.monotonic()
        c = rng.normal(0.0, 0.08, R)
        if r == 0:
            c[0] = 0.45
        elif r == 1:
            c[0] = 0.30
            c[1] = 0.20
        elif r == 2:
            c[0] = 0.25
            c[1] = -0.15
            c[2] = 0.12
        else:
            c = rng.normal(0.0, 0.15, R)
        c = np.clip(c, -1.0, 1.0)

        m = np.zeros(R, dtype=np.float64)
        v = np.zeros(R, dtype=np.float64)
        t = 0
        lr = 0.07
        tau = 4.0
        local_best = float("inf")
        local_a = None

        while time.monotonic() - start_t < per_time and t < 1800:
            a_raw = 1.0 + basis.dot(c)
            if not np.all(np.isfinite(a_raw)):
                break
            if float(np.min(a_raw)) < 1e-4:
                an_raw = a_raw.copy()
                an_raw[an_raw < 1e-4] = 1e-4
            else:
                an_raw = a_raw
            S = float(an_raw.sum())
            if S < 1e-4:
                break
            an = an_raw / S

            A = fft(an, N)
            b = np.real(ifft(A * A))[:M]
            b = np.where(np.isfinite(b), b, 0.0)
            b = np.maximum(b, 0.0)
            score = 2.0 * n * float(b.max())
            if not np.isfinite(score):
                break
            if score < local_best:
                local_best = score
                local_a = an.copy()

            x = b * float(M)
            p = softmax_x(x, tau)
            p = 0.95 * p + 0.05 / float(M)

            d = correlate_from_A(p, A, N, neg_idx, phase_neg, n)
            grad_a = 2.0 * float(M) * d

            gB = basis.T.dot(grad_a)
            inner = float(grad_a.dot(an))
            grad_c = gB / S - inner * sumbasis / (S * S)
            grad_c = np.where(np.isfinite(grad_c), grad_c, 0.0)

            t += 1
            b1 = 0.9
            b2 = 0.999
            m = b1 * m + (1.0 - b1) * grad_c
            v = b2 * v + (1.0 - b2) * (grad_c * grad_c)
            mh = m / (1.0 - b1 ** t)
            vh = v / (1.0 - b2 ** t)
            c = c - lr * mh / (np.sqrt(vh) + 1e-8)
            c = np.clip(c, -1.0, 1.0)
            tau = min(tau * 1.003, 25.0)

        if local_a is not None and local_best < best_score:
            best_score = local_best
            best_a = local_a.copy()

    return best_a, best_score


def full_optimize(a0, budget_s, seed):
    a0 = normalize_positive(a0)
    n = int(a0.size)
    n = max(2, n)
    budget_s = max(0.0, float(budget_s))
    if budget_s < 2.0:
        return a0, float("inf")

    rng = np.random.default_rng(int(seed))
    a = normalize_positive(a0)
    M = 2 * n - 1
    N, neg_idx, phase_neg = make_fft_context(n)

    best_a = a.copy()
    best_score = estimate_score_with_size(a, N)
    if not np.isfinite(best_score):
        best_score = 2.0

    prev_score = best_score
    tau = 1.0
    eta = 0.12
    beta = 0.55
    momentum = np.zeros(n, dtype=np.float64)
    last_improve = time.monotonic()
    deadline = time.monotonic() + budget_s
    it = 0
    timeout = max(8.0, 0.12 * budget_s)

    while time.monotonic() < deadline:
        A = fft(a, N)
        b = np.real(ifft(A * A))[:M]
        b = np.where(np.isfinite(b), b, 0.0)
        b = np.maximum(b, 0.0)
        s = float(a.sum())
        if s < 1e-12:
            a = make_uniform(n)
            continue
        score = 2.0 * n * float(b.max()) / (s * s)
        if not np.isfinite(score):
            a = perturb_sequence(a, rng, 0.35)
            momentum = 0.5 * momentum
            continue

        if score < best_score - 1e-9:
            best_score = score
            best_a = a.copy()
            last_improve = time.monotonic()

        if score > prev_score * 1.35:
            eta = max(0.04, eta * 0.65)
        else:
            eta = min(0.9, eta * 1.004)
        prev_score = score

        x = b * float(M)
        p = softmax_x(x, tau)
        p = 0.55 * p
        kmax = int(np.argmax(x))
        use_k = n - 1 if x[n - 1] >= x[kmax] - 1e-6 * max(1.0, x[kmax]) else kmax

        w = max(1, int(M * 0.02))
        lo = max(0, use_k - w)
        hi = min(M, use_k + w)
        p[lo:hi] += 0.30 / float(max(1, hi - lo))
        p += 0.15 / float(M)
        ps = float(p.sum())
        if ps > 0.0 and np.isfinite(ps):
            p = p / ps

        d = correlate_from_A(p, A, N, neg_idx, phase_neg, n)
        dmax = float(d.max()) if d.size else 0.0
        if dmax > 1e-15:
            direction = d / dmax
        else:
            direction = np.ones(n, dtype=np.float64)

        momentum = beta * momentum + (1.0 - beta) * direction
        vmax = float(momentum.max()) if momentum.size else 0.0
        if vmax > 1e-15:
            a = a * np.exp(-eta * (momentum / vmax))
            a = normalize_positive(a)
        else:
            a = perturb_sequence(a, rng, 0.25)

        if it > 50 and (it % 150 == 50):
            for gamma in (2.0, 4.0, 8.0):
                sh = diag_sharpen(a, use_k, gamma)
                for lam in (0.05, 0.15, 0.30):
                    cand = (1.0 - lam) * a + lam * sh
                    cand = normalize_positive(cand)
                    sc = estimate_score_with_size(cand, N)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = cand.copy()
                        a = cand.copy()
                        last_improve = time.monotonic()
                        break

        tau = min(tau * 1.006, 50.0)
        if time.monotonic() - last_improve > timeout:
            strength = 0.12 + 0.30 * min(1.0, (time.monotonic() - last_improve) / max(1.0, 2.0 * timeout))
            a = perturb_sequence(a, rng, strength)
            momentum *= 0.3
            tau = min(tau, 12.0)
            eta = max(eta, 0.25)

        it += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):

    try:
        total_budget = float(budget_s)

        total_budget = 2400.0
    t_start = time.monotonic()
    deadline = t_start + max(0.0, total_budget - 25.0)
    seed = int(seed)
    rng = np.random.default_rng(seed)

    parent = get_parent_sequence()
    if parent.size <= 30000:
        a_best = normalize_positive(parent)
        best_score = evaluate_array(a_best)
    else:
        a_best = make_uniform(8192)
        best_score = evaluate_array(a_best)
    if not np.isfinite(best_score):
        a_best = make_uniform(4096)
        best_score = evaluate_array(a_best)
    if not np.isfinite(best_score):
        best_score = 2.0

    print(f"initial parent score={best_score:.10f} len={a_best.size}", flush=True)

    # Small explicitly good / potentially useful short candidates.
    extras = [
        np.array([2.0, 1.0], dtype=np.float64),
        np.array([1.4, 1.4, 0.6, 0.6], dtype=np.float64),
        np.array([1.4, 1.2, 0.8, 0.6], dtype=np.float64),
        np.array([1.35, 1.25, 1.05, 0.75, 0.45], dtype=np.float64),
        np.array([1.6, 1.2, 0.8, 0.6, 0.5, 0.4], dtype=np.float64),
    ]
    for ex in extras:
        ac = normalize_positive(ex)
        sc = evaluate_array(ac)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            a_best = ac.copy()
            print(f"extra candidate improved to {best_score:.10f} len={ac.size}", flush=True)

    # Choose a large length to invest in.
    if 1500 <= a_best.size <= 12000:
        L_big = int(a_best.size)
    else:
        L_big = 4096
    L_other = 8192 if L_big < 8192 else max(4096, int(L_big // 2))

    inits = []
    inits.append(a_best.copy())

    now = time.monotonic()
    if now < deadline - 120:
        pb = min(120.0, max(20.0, (deadline - now) * 0.15))
        pa, ps = param_optimize(L_big, pb, seed + 101)
        if pa is not None and np.isfinite(ps) and ps < best_score + 0.35:
            pa = normalize_positive(pa)
            inits.append(pa)
            print(f"param L_big candidate estimate {ps:.10f}", flush=True)

        if time.monotonic() < deadline - 260 and L_other != L_big:
            pb2 = min(60.0, max(15.0, (deadline - now) * 0.08))
            pa2, ps2 = param_optimize(L_other, pb2, seed + 102)
            if pa2 is not None and np.isfinite(ps2) and ps2 < best_score + 0.35:
                pa2 = normalize_positive(pa2)
                inits.append(pa2)
                print(f"param other candidate estimate {ps2:.10f}", flush=True)

    inits.append(make_antisym_random(L_big, rng, 0.70))
    inits.append(make_cosine_antisym(L_big, 0.60))
    if L_other != L_big:
        inits.append(make_antisym_random(L_other, rng, 0.70))

    if len(inits) > 6:
        inits = inits[:6]

    # Main large/local optimization attempts.
    while time.monotonic() < deadline - 120 and len(inits) > 0:
        arr = inits.pop(0)
        arr = normalize_positive(arr)
        n_arr = int(arr.size)
        remaining = max(0.0, deadline - time.monotonic())
        if remaining < 60:
            break
        if n_arr <= 4000:
            slice_time = min(350.0, max(60.0, remaining / max(1, len(inits) + 1)))
        elif n_arr <= 10000:
            slice_time = min(300.0, max(55.0, remaining / max(1, len(inits) + 1)))
        else:
            slice_time = min(220.0, max(45.0, remaining / max(1, len(inits) + 1)))

        ac, sc = full_optimize(arr, slice_time, seed + 500 + len(inits))
        if ac is not None and np.isfinite(sc) and sc < best_score + 0.05:
            true_sc = evaluate_array(ac)
            if np.isfinite(true_sc) and true_sc < best_score:
                best_score = true_sc
                a_best = normalize_positive(ac)
                print(f"full large improved to {best_score:.10f} len={a_best.size}", flush=True)

    # Attempt short sequences; they are cheap and provide a robust fallback.
    small_lens = [2, 3, 4, 5, 6, 8, 12, 16]
    for i, n_s in enumerate(small_lens):
        if time.monotonic() < deadline - 30:
            if n_s == 2:
                arr = np.array([2.0, 1.0], dtype=np.float64)
            else:
                arr = make_antisym_random(n_s, rng, 1.0)
            remaining = max(0.0, deadline - time.monotonic())
            slice_time = min(25.0, max(5.0, remaining / max(1, len(small_lens) - i)))
            ac, sc = full_optimize(arr, slice_time, seed + 700 + n_s)
            if ac is not None and np.isfinite(sc) and sc < best_score + 0.05:
                true_sc = evaluate_array(ac)
                if np.isfinite(true_sc) and true_sc < best_score:
                    best_score = true_sc
                    a_best = normalize_positive(ac)
                    print(f"small len {n_s} improved to {best_score:.10f}", flush=True)

    # Final polishing of the current best, if time remains.
    if time.monotonic() < deadline - 45:
        final_budget = max(0.0, deadline - time.monotonic() - 20.0)
        ac, sc = full_optimize(a_best, final_budget, seed + 999)
        if ac is not None and np.isfinite(sc) and sc < best_score + 0.03:
            true_sc = evaluate_array(ac)
            if np.isfinite(true_sc) and true_sc < best_score:
                best_score = true_sc
                a_best = normalize_positive(ac)
                print(f"final polish improved to {best_score:.10f}", flush=True)

    out = scale_for_return(a_best)
    out_list = [float(x) for x in out]
    final_check = safe_evaluate(out_list)
    if not np.isfinite(final_check):
        out_list = [4.0 / 3.0, 2.0 / 3.0]  # length-2 fallback, objective 16/9

    print(f"final score estimate/eval={safe_evaluate(out_list):.10f} len={len(out_list)}", flush=True)
    return out_list
