def self_conv(a):
    """Compute self-convolution of a using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]

def compute_grad(a, weights):
    """
    Compute grad[i] = 2 * sum_{j=0}^{n-1} a[j] * weights[j+i] for i = 0,...,n-1.
    weights has length 2n-1.
    Uses FFT: grad[i] = 2 * (a_rev * weights)[n-1+i]
    """
    n = len(a)
    N = 1 << (3 * n - 3).bit_length()
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(weights, N)
    conv = np.fft.irfft(fa_rev * fw, N)
    return 2.0 * conv[n-1 : n-1+n]

The Lagrangian is: sum_j g[j] - lambda * (sum_j a[k*-j] * g[j] - M)
= sum_j (1 - lambda * a[k*-j]) * g[j] + lambda * M

For the optimal to have g[j] > 0, we need 1 - lambda * a[k*-j] >= 0, i.e., a[k*-j] <= 1/lambda.

At the optimum, the constraint is binding: sum_j a[k*-j] * g[j] = M.

So we want to find lambda such that:
sum_{j: a[k*-j] <= 1/lambda} a[k*-j] * g[j] = M

And g[j] = 0 for a[k*-j] > 1/lambda... wait, that's not right either.

Actually, by the complementary slackness condition, g[j] > 0 only if a[k*-j] = min_j' a[k*-j'] (i.e., the minimum coefficient). So the optimal is:
- Find m = min_j a[k*-j] (where a[k*-j] > 0, or consider only valid j).
- If m = 0: put all mass on g[j] where a[k*-j] = 0.
- If m > 0: put all mass on g[j] where a[k*-j] = m.

Wait, that's the extreme point solution. The LP optimum is at an extreme point.

The extreme points of the polytope {g >= 0, sum_j c_j * g[j] <= M} (where c_j = a[k*-j]) are:
- g = 0 (trivial)
- g[e_j] * (M / c_j) for each j where c_j > 0 (put all mass in one variable)

So the optimal extreme point is: g_j = M / c_j = M / a[k*-j] for the j that minimizes c_j = a[k*-j].

Wait no. The extreme points of the simplex {g >= 0, c^T g <= M} are:
- The origin g = 0
- Points where one variable is at its bound: g_j = M/c_j (all others 0)

The objective is sum(g) = g_j = M/c_j = M/a[k*-j].
To maximize this, we want to minimize c_j = a[k*-j].

So the optimal single-constraint solution is: g[j*] = M / a[k*-j*] where j* = argmin_j a[k*-j].

"""
Optimize non-negative step function coefficients to minimize 2*n*max_conv/(sum_a)^2.
Uses FFT-based self-convolution, soft-max gradient descent with multiplicative updates,
LP-based refinement, and random perturbation local search across multiple lengths and starts.
"""

import numpy as np
import time


def self_conv(a):
    """Compute self-convolution of a using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]


def compute_grad(a, weights):
    """
    Compute grad[i] = 2 * sum_{j=0}^{n-1} a[j] * weights[j+i] for i=0..n-1.
    weights has length 2n-1. Uses FFT via reversed a convolution.
    """
    n = len(a)
    N = 1 << (3 * n - 3).bit_length()
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(weights, N)
    conv = np.fft.irfft(fa_rev * fw, N)
    return 2.0 * conv[n-1:n-1+n]


def optimize_gradient(a0, eta0=1.0, tau_frac=0.1, max_iters=3000, tol=1e-12):
    """
    Multiplicative gradient descent with soft-max smoothing.
    Minimizes 2n * soft_max(self_conv(a)) with sum(a)=1.
    Returns (best_a, best_obj).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n, dtype=np.float64) / n
        S = 1.0
    a = a / S

    c = self_conv(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    eta = eta0
    patience = 0
    max_patience = 800

    for it in range(max_iters):
        c = self_conv(a)
        c_max = c.max()
        if c_max < 1e-15:
            break
        tau = max(tau_frac * c_max, 1e-12)

        exp_c = np.exp((c - c_max) / tau)
        Z = exp_c.sum()
        if Z < 1e-30:
            break
        soft_max = c_max + tau * np.log(Z)
        obj = 2.0 * n * soft_max

        if obj < best_obj - tol:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience > max_patience:
            break

        grad = compute_grad(a, exp_c)
        gmax = grad.max()
        if gmax > 1e-15:
            grad = grad / gmax

        exponent = -eta * grad
        exponent = np.clip(exponent, -60.0, 60.0)
        a_new = a * np.exp(exponent)
        a_new = np.maximum(a_new, 1e-30)
        S_new = a_new.sum()
        if S_new < 1e-15:
            eta *= 0.5
            continue
        a_new = a_new / S_new

        c_new = self_conv(a_new)
        obj_new = 2.0 * n * c_new.max()

        if obj_new < best_obj:
            a = a_new
            eta = min(eta * 1.03, 8.0)
        else:
            eta = max(eta * 0.7, 0.01)

    return best_a, best_obj


def optimize_lp_refine(a, max_iters=30, blend_t=0.4):
    """
    LP-based refinement: at each step solve LP to find best g under convolution constraint.
    Only effective for n <= 2048 due to matrix size.
    """
    from scipy.optimize import linprog
    n = len(a)
    if n > 2048:
        return a, 2.0 * n * self_conv(a).max()

    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-15:
        return a, float('inf')
    a = a / S

    c = self_conv(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    # Build constraint matrix once (it changes with a, so rebuild each iter)
    for it in range(max_iters):
        c = self_conv(a)
        M = c.max()
        if M < 1e-15:
            break

        # A_ub[k, i] = a[i] if 0 <= k-i < n else 0
        A_ub = np.zeros((2 * n - 1, n))
        for i in range(n):
            # a[i] appears in rows k = i, i+1, ..., i+n-1
            A_ub[i:i+n, i] = a
        b_ub = np.full(2 * n - 1, M)
        c_obj = -np.ones(n)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                continue
            g = np.maximum(-res.x, 0.0)
            S_g = g.sum()
            if S_g < 1e-12:
                continue
            g = g / S_g
            a_new = (1.0 - blend_t) * a + blend_t * g
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-12:
                continue
            a_new = a_new / S_new
            c_new = self_conv(a_new)
            obj_new = 2.0 * n * c_new.max()
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
        except Exception:
            break

    return best_a, best_obj


def make_starting_points(n, rng):
    """Generate diverse non-negative starting points of length n."""
    x = np.linspace(0.0, 1.0, n)
    points = []
    # U-shaped
    points.append((2.0 * x - 1.0) ** 2 + 0.01)
    # Inverted dome
    points.append(0.5 + 0.5 * np.cos(2.0 * np.pi * x))
    # Bimodal
    points.append(np.cos(4.0 * np.pi * x) ** 2 + 0.01)
    # Triangular
    points.append(1.0 - np.abs(2.0 * x - 1.0) + 0.01)
    # Uniform
    points.append(np.ones(n))
    # Random
    points.append(rng.random(n) + 0.01)
    # U^4 (sharper U)
    points.append((2.0 * x - 1.0) ** 4 + 0.01)
    # Edge peaks

    # Three peaks
    points.append(np.exp(-30.0 * (x - 0.15) ** 2) + np.exp(-30.0 * (x - 0.5) ** 2) + np.exp(-30.0 * (x - 0.85) ** 2))
    # Log-uniform noise
    points.append(10.0 ** rng.uniform(-1, 1, n))
    return points
