take a step in that direction.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def build_convolution_matrix(a):
    """Build the (2n-1) x n matrix C such that (C @ b)_k = (a*b)_k."""
    n = len(a)
    C = lil_matrix((2*n - 1, n))
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            C[k, j] = a[k - j]
    return csr_matrix(C)


def solve_lp_for_direction(a, M):
    """Solve the LP: max sum(b) s.t. C@b <= M*ones, b >= 0."""
    n = len(a)
    C = build_convolution_matrix(a)
    
    # Minimize -sum(b) => c = -ones
    c = -np.ones(n)
    A_ub = C  # (2n-1) x n
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    return res.x

This is a non-convex problem, but we can use:

- Penalty methods
- Interior point methods (for a smoothed version)

For the subgradient method:
- At each step, compute the subgradient at the current point
- Take a step in the negative subgradient direction
- Project onto the constraint set $\{a: a \ge 0, \sum a = 1\}$

The subgradient of $g(a) = \max_k (a*a)_k$ at $a$ is:
- Let $K^* = \{k: (a*a)_k = M\}$ be the set of maximizing indices.
- The subgradient is the set $\{\sum_{k \in K^*} \lambda_k \nabla_k g : \lambda_k \ge 0, \sum \lambda_k = 1\}$
- where $\nabla_k g$ at position $i$ is: $a_{k-i} + a_{i-k}$ (if both indices valid, else just one term)

- $a_i^{new} = \max(0, a_i - \mu)$ for some $\mu$ chosen so that $\sum a_i^{new} = 1$
- This is the "water-filling" projection.

    autocorr = np.convolve(a, a)
    M = np.max(autocorr)

    active = np.where(autocorr >= M - 1e-10)[0]
    
    # Compute subgradient (random convex combination of active gradients)
    # For simplicity, use uniform combination
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            # (a*a)_k's derivative w.r.t. a_i
            # = a[k-i] + a[i-k] (if valid)
            if 0 <= k - i < n:
                grad[i] += a[k - i]
            if 0 <= i - k < n:
                grad[i] += a[i - k]
    grad /= len(active)
    
    # Take step
    new_a = a - step_size * grad
    new_a = np.clip(new_a, 0, None)
    
    # Project onto sum = 1
    S = np.sum(new_a)
    if S > 0:
        new_a = new_a / S
    else:
        new_a = np.ones(n) / n
    
    return new_a

import time


def compute_subgradient_fast(a, active_mask):
    """Compute subgradient of max(autocorr) at a, given active set mask."""
    n = len(a)
    # v[i] = sum_j a[j] * active_mask[i+j] for i=0,...,n-1
    # This is cross-correlation
    # Use direct computation via numpy correlate
    # np.correlate(active_mask, a, 'valid') gives:
    # result[i] = sum_j active_mask[i+j] * a[j] for i = 0,..., (len(active_mask)-len(a))
    # len(active_mask) = 2n-1, len(a) = n, so i = 0,...,n-1. Perfect!
    v = np.correlate(active_mask, a, mode='valid')
    count = active_mask.sum()
    if count == 0:
        return np.zeros(n)
    return 2.0 * v / count


def project_onto_simplex(a, target_sum):
    """Project a onto {x: x >= 0, sum(x) = target_sum}."""
    n = len(a)
    # Water-filling
    a_pos = np.maximum(a, 0)
    if a_pos.sum() == 0:
        return np.full(n, target_sum / n)
    # Binary search for threshold
    a_sorted = np.sort(a_pos)[::-1]
    cumsum = np.cumsum(a_sorted)
    rho = np.nonzero(a_sorted * np.arange(1, n+1) > (cumsum - target_sum))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cumsum[rho] - target_sum) / (rho + 1)
    return np.maximum(a_pos - theta, 0)


def evaluate_fast(a):
    """Fast evaluation without list conversion."""
    n = len(a)
    if n == 0 or a.sum() < 0.01:
        return np.inf
    a = np.clip(a, 0, 1000)
    autocorr = np.convolve(a, a)
    max_b = np.max(autocorr)
    sum_a = np.sum(a)
    return float(2 * n * max_b / (sum_a ** 2))


def subgradient_descent(a, max_iter, deadline, rng, verbose=False):
    """Run projected subgradient descent to minimize max(autocorr)."""
    n = len(a)
    a = a / a.sum()  # normalize to sum=1
    
    best_a = a.copy()
    best_obj = np.max(np.convolve(a, a))
    
    step_size = 1.0 / np.sqrt(n)
    patience = 500
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute autocorrelation and active set
        autocorr = np.convolve(a, a)
        M = np.max(autocorr)
        
        if M < best_obj:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find active set (indices where autocorr is close to M)
        tolerance = max(1e-12, 1e-8 * M)
        active_mask = (autocorr >= M - tolerance).astype(float)
        
        # Compute subgradient
        grad = compute_subgradient_fast(a, active_mask)
        grad_norm = np.linalg.norm(grad)
        
        if grad_norm < 1e-15:
            break
        
        # Adaptive step size
        # Polyak-type: step = (M - lower_bound) / ||grad||^2
        lower_bound = 1.0 / (2 * n - 1)
        polyak_step = (M - lower_bound) / (grad_norm ** 2 + 1e-15)
        step = min(step_size, polyak_step)
        
        # Take step and project
        a_new = a - step * grad
        a_new = project_onto_simplex(a_new, 1.0)
        
        # Check if step was too large (objective increased too much)
        new_M = np.max(np.convolve(a_new, a_new))
        if new_M > M * 1.5:  # too much increase
            step_size *= 0.5
            a_new = a - step_size * 0.1 * grad / (grad_norm + 1e-15)
            a_new = project_onto_simplex(a_new, 1.0)
        elif new_M < M * 0.9:  # good progress
            step_size = min(step_size * 1.2, 1.0)
        else:
            step_size *= 0.95
        
        a = a_new
        
        if verbose and it % 100 == 0:
            obj = 2 * n * new_M  # since sum=1
            print(f"    SG iter {it}: max_ac={new_M:.8f}, obj≈{obj:.4f}, step={step:.6f}")
        
        if no_improve > patience:
            # Random restart within the method
            a = a * np.exp(rng.normal(0, 0.1, n))
            a = project_onto_simplex(a, 1.0)
            no_improve = 0
            step_size = 1.0 / np.sqrt(n)
    
    return best_a


def lp_direction(a, M, deadline):
    """Solve LP to find best direction. Returns None if too slow."""
    n = len(a)
    t0 = time.monotonic()
    if time.monotonic() - t0 > deadline:
        return None
    
    # Build constraint matrix C: (2n-1) x n, C[k,j] = a[k-j] if valid
    # C @ b = a * b
    # Use sparse construction
    j_indices = np.arange(n)
    # For each column j, the non-zero entries are at rows j, j+1, ..., j+n-1
    # with values a[0], a[1], ..., a[n-1]
    rows = []
    cols = []
    vals = []
    for j in range(n):
        row_range = np.arange(j, j + n)
        rows.append(row_range)
        cols.append(np.full(n, j))
        vals.append(a)
    
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)
    
    C = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    
    # LP: max sum(b) s.t. C @ b <= M * 1, b >= 0
    # => min -sum(b) s.t. C @ b <= M * 1, b >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=[(0, None)]*n,
                      method='highs', options={'time_limit': max(1, int(deadline - time.monotonic()))})
        if res.status != 0:
            return None
        return res.x
    except Exception:
        return None
