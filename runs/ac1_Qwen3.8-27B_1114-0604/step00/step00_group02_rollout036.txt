from scipy.fft import fft, ifft

def convolve_fft(a, b):
    """Linear convolution using FFT."""
    n = len(a) + len(b) - 1
    N = 1
    while N < n:
        N *= 2
    fa = fft(a, N)
    fb = fft(b, N)
    return ifft(fa * fb).real[:n]

def simplex_project(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    v = np.sort(v)[::-1]
    cssv = np.cumsum(v)
    rho = np.where(v * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

import numpy as np
import time
from scipy.fft import fft, ifft

"""
Algorithm: Smoothed gradient descent with adaptive temperature and simplex projection.
Minimizes 2n * max(f*f) / (sum f)^2 subject to f >= 0.
Uses FFT for O(n log n) convolution and cross-correlation.
Includes random perturbations for exploration and anti-symmetrization to break palindromic structure.
"""

def _next_power_of_2(n):
    """Return the next power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def _convolve_self(f):
    """Compute linear self-convolution f*f using FFT. Returns array of length 2n-1."""
    n = len(f)
    N = _next_power_of_2(2 * n - 1)
    F = fft(f, N)
    result = ifft(F * F).real
    return result[:2 * n - 1]

def _cross_corr(p, f):
    """
    Compute cross-correlation (p ⋆ f)[m] = sum_l p[l+m] * f[l] for m = 0, ..., len(f)-1.
    p has length 2*len(f)-1, f has length n.
    """
    n = len(f)
    N = _next_power_of_2(len(p) + n - 1)
    P = fft(p, N)
    F = fft(f, N)
    result = ifft(np.conj(P) * F).real
    return result[:n]

def _project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    if v.sum() <= 0:
        return np.full(n, target_sum / n)
    
    # We want to find x = max(0, v - tau) such that sum(x) = target_sum
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find rho: largest index where u[rho] > (cssv[rho] - target_sum) / (rho+1)
    # i.e., u[rho] * (rho+1) > cssv[rho] - target_sum
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    tau = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - tau, 0)

def _internal_score(f):
    """Compute 2n * max(f*f) / (sum f)^2."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    c = _convolve_self(f)
    return 2.0 * n * c.max() / (s * s)

def _gradient_step(f, beta, lr):
    """One step of smoothed gradient descent with simplex projection."""
    n = len(f)
    c = _convolve_self(f)
    
    # Softmax weights
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    p_sum = exp_bc.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_bc / p_sum
    
    # Gradient: grad[m] = 2 * (p ⋆ f)[m]
    grad = 2.0 * _cross_corr(p, f)
    
    # Update
    f_new = f - lr * grad
    f_new = _project_simplex(f_new, 1.0)
    return f_new, c, p

def _antisymmetric_perturbation(f, rng, strength=0.1):
    """Add a perturbation that breaks palindromic symmetry."""
    n = len(f)
    f_new = f.copy()
    # Randomly swap or adjust pairs (j, n-1-j)
    for _ in range(n // 10):
        j = rng.integers(0, n // 2)
        k = n - 1 - j
        total = f_new[j] + f_new[k]
        if total < 1e-15:
            continue
        # Bias towards making one larger, one smaller
        alpha = rng.beta(2, 1)  # biased towards 1
        f_new[j] = alpha * total
        f_new[k] = (1 - alpha) * total
    return _project_simplex(f_new, 1.0)

    if len(init) == 0 or init.sum() < 1e-10:
        init = np.ones(2048, dtype=np.float64)
    
    n = len(init)
    f = init / init.sum()
    
    best = f.copy()
    best_score = _internal_score(f)
    print(f"n={n}, initial score={best_score:.6f}")
    
    # Hyperparameters
    beta = 100.0
    beta_max = 2000.0
    lr = 0.02
    it_count = 0
    start_time = time.monotonic()
    
    while time.monotonic() < deadline:
        # Anneal temperature
        elapsed = time.monotonic() - start_time
        progress = min(1.0, elapsed / (budget_s * 0.7))
        beta = 50.0 + (beta_max - 50.0) * progress
        lr = 0.03 / (1.0 + it_count / 5000.0)
        
        # Gradient step
        f_new, c, p = _gradient_step(f, beta, lr)
        f = f_new
        it_count += 1

        if it_count % 2000 == 1000:
