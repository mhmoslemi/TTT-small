The constraint a[j]*a[k-j] <= t is not linear, but we can linearize it: for a given upper bound on a[j], the constraint becomes a[k-j] <= t/a[j].

**Iterative water-filling:**
1. Initialize a (e.g., binomial sequence).
2. Compute c = convolve(a, a). M = max(c).
3. For the peak positions (where c[k] ≈ M), the contributing pairs (j, k-j) have large products a[j]*a[k-j].
4. Reduce the larger of each contributing pair.
5. Renormalize.
6. Repeat.

This is O(n^2) per step if done naively, but O(n log n) if we use FFT to compute c and then only modify the peak-contributing elements.

Actually, for the peak position k*, the contributing elements are a[0], a[k*], a[1], a[k*-1], etc. (up to min(k*, n-1) pairs). So we only need to modify O(min(k*, n)) elements per step.

"""
Step-function optimization to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses binomial coefficient sequence as starting point, then applies:
1) Bump/tail perturbations to break the one-sided-flat structure
2) LP-based direction finding
3) Iterative peak-reduction (water-filling) via FFT
4) Multi-start with different lengths and seeds
Target: beat 1.5027

import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
import time
import sys


def binomial_coeffs(n):
    """Generate C(2k,k)/4^k for k=0,...,n-1 (recursively stable)."""
    a = np.ones(n, dtype=np.float64)
    for k in range(1, n):
        a[k] = a[k - 1] * (4.0 * k - 2.0) / (4.0 * k)
    return a


def convolve_self(a):
    """Compute linear a*a via FFT."""
    n = len(a)
    out_len = 2 * n - 1
    L = next_fast_len(out_len)
    fa = fft(a, L)
    result = np.real(ifft(fa * fa, L))[:out_len]
    return result


def compute_score(a):
    """Fast score computation: 2*n*max(c)/S^2."""
    n = len(a)
    c = convolve_self(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * np.max(c) / (S * S)


def normalize_nonneg(a):
    """Clip to [0, inf) and normalize sum to 1."""
    a = np.clip(a, 0.0, None)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / S


def add_bump(a, bump_type=0, eps=0.1):
    """Add a perturbation bump to the binomial sequence."""
    n = len(a)
    a = a.copy()
    if bump_type == 0:
        # sin bump centered at n/2
        x = np.arange(n) / max(1, n - 1)
        a += eps * np.sin(np.pi * x) * a
    elif bump_type == 1:
        # Gaussian bump centered at n/3
        x = np.arange(n) / max(1, n - 1)
        center = 1.0 / 3.0
        sigma = 0.15
        a += eps * np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif bump_type == 2:
        # Tail boost: increase the latter half
        x = np.arange(n) / max(1, n - 1)
        a += eps * np.maximum(0, x - 0.5) * a
    elif bump_type == 3:
        # Anti-binomial shape: favor the tail
        x = np.arange(n) / max(1, n - 1)
        a += eps * x * (1 - x)
    return a


def peak_reduce(a, lr=0.03, top_k=30, n_steps=5000, recalc_every=5):
    """Iteratively reduce the peak of self-convolution.
    
    Each step: find peak, reduce top contributing elements, renormalize.
    Uses FFT to recompute convolution every `recalc_every` steps.

    a = normalize_nonneg(a)
    n = len(a)
    c = convolve_self(a)
    best_local = np.max(c)
    
    for step in range(n_steps):
        if step % recalc_every == 0:
            c = convolve_self(a)
            M = np.max(c)
        else:
            M = np.max(c)
        
        if M < 1e-15:
            break
            
        k_star = int(np.argmax(c))
        j_max = min(k_star, n - 1)
        search_range = min(j_max + 1, top_k * 3)
        j_idx = np.arange(search_range)
        k_j = k_star - j_idx
        valid = (k_j >= 0) & (k_j < n)
        j_idx = j_idx[valid]

        prods = a[j_idx] * a[k_j]

def lp_direction(a, M):
    """Solve LP to find direction g maximizing sum(g) subject to
    g >= 0 and convolve(a, g) <= M.
    Returns the direction (or None if LP fails).
    """
    n = len(a)
    # Build constraint matrix A (shape: (2n-1, n))
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    # This is a banded matrix - build sparsely
    A_rows = []
    A_cols = []
    A_vals = []
    
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A_rows.extend([k] * (j_hi - j_lo + 1))
            A_cols.extend(range(j_lo, j_hi + 1))
            A_vals.extend(a[j_lo:j_hi + 1].tolist())
    
    from scipy.sparse import csc_matrix
    A = csc_matrix((A_vals, (A_rows, A_cols)), shape=(2 * n - 1, n))
    
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.success:
            g = np.clip(res.x, 0, None)
            if np.sum(g) > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_optimize(a, n_lp_iters=10, t=0.3, total_time=600):
    """LP-based optimization loop."""
    a = normalize_nonneg(a)
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = time.monotonic() + total_time
    
    for it in range(n_lp_iters):
        if time.monotonic() > deadline:
            break
        S = np.sum(a)
        c = convolve_self(a)
        M = np.max(c)
        
        g = lp_direction(a, M)
        if g is None:
            break
        
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum  # normalize to sum=1
        
        # Update
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0.0, None)
        a_new = a_new / np.sum(a_new)
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        
        print(f"  LP iter {it}: score={score:.6f}, best={best_score:.6f}", flush=True)
    
    return best_a, best_score
