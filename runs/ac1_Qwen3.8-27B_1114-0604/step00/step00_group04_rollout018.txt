Now, let me think about the FFT-based approach more carefully.

If we zero-pad `a` to length N >= 2n-1, then the linear convolution `a * a` equals the inverse FFT of `|FFT(a)|^2`.

So `conv = ifft(|fft(a, N)|^2)` (taking the real part).

We want to minimize `max(conv)`. In the frequency domain, this means we want `|FFT(a)|^2` to be as "flat" as possible (in some sense).

But we also need `a` to be non-negative. So the problem becomes: find a non-negative sequence `a` of length `n` such that `|FFT(a, N)|^2` is as flat as possible.

This is similar to the problem of designing a window function with a flat top in the frequency domain (or equivalently, a flat frequency response in the time domain for a filter).

**Gerchberg-Saxton style algorithm:**

1. Start with a non-negative sequence `a`.
2. Compute `A = FFT(a, N)`.
3. Modify the amplitude: `|A_new| = |A|^alpha * C^(1-alpha)` for some constant C and mixing parameter alpha < 1 (this pulls the amplitude towards flat).
4. Keep the phase: `A_new = |A_new| * exp(i * angle(A))`.
5. `a_new = real(ifft(A_new)[:n])`.
6. Clip: `a_new = max(a_new, 0)`.
7. Repeat.

The parameter alpha should start close to 1 (small changes) and decrease over time (larger changes towards flat).

import numpy as np
import time

def fast_evaluate(a):
    """Fast evaluation without validation overhead."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def fft_shaping(a, n_iter=2000, alpha_start=0.99, alpha_end=0.5):
    """Iteratively flatten the power spectrum."""
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N *= 2
    a = a.copy().astype(np.float64)
    for i in range(n_iter):
        alpha = alpha_start + (alpha_end - alpha_start) * i / max(1, n_iter - 1)
        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2
        # Geometric mean with flat target
        target = np.full_like(power, power.sum() / len(power))
        new_power = power ** alpha * target ** (1 - alpha)
        new_amplitude = np.sqrt(new_power)
        phase = A / (np.abs(A) + 1e-30)
        A_new = new_amplitude * phase
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.clip(a_new, 0, None)
        if np.sum(a_new) < 1e-10:
            break
        a = a_new
    return a

def subgradient_step(a, step_size, momentum_arr=None, momentum=0.9):
    """One subgradient step to reduce the convolution peak."""
    n = len(a)
    conv = np.convolve(a, a)
    max_val = np.max(conv)
    # Find all active peaks (within 1% of max)
    active = np.where(conv >= 0.99 * max_val)[0]
    
    # Compute subgradient (average over active peaks)
    grad = np.zeros(n)
    for k in active:
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                grad[j] += a[m]
    grad *= 2.0 / len(active)
    
    # Update with momentum
    if momentum_arr is None:
        momentum_arr = -grad * step_size
    else:
        momentum_arr = momentum * momentum_arr - grad * step_size
    
    a_new = a + momentum_arr
    a_new = np.clip(a_new, 0, None)
    s = np.sum(a_new)
    if s < 1e-10:
        return a, momentum_arr
    a_new = a_new / s  # normalize to sum=1
    return a_new, momentum_arr
```

def compute_subgradient_fast(a, conv, active_threshold=0.99):
    n = len(a)
    max_val = np.max(conv)
    active = np.where(conv >= active_threshold * max_val)[0]
    
    grad = np.zeros(n)
    for k in active:
        # We need: for j in [0, n-1], grad[j] += a[k-j] if 0 <= k-j < n
        # Valid j range: max(0, k-n+1) <= j <= min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            # a[k-j] for j in [j_start, j_end] = a[k-j_end : k-j_start+1] reversed
            m_start = k - j_end
            m_end = k - j_start
            grad[j_start:j_end+1] += a[m_start:m_end+1][::-1]
    grad *= 2.0 / len(active)
    return grad
```

"""
Multi-phase optimization for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-based spectral shaping, subgradient descent, and diverse local search.
"""

import numpy as np
import time


def fast_evaluate(a):
    """Compute the objective function value without validation overhead."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a):
    """Compute linear convolution using FFT for speed."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    conv = np.real(np.fft.irfft(A * A, N))[:2 * n - 1]
    return conv


def gs_flatten(a, n_iter=5000, alpha_start=0.99, alpha_end=0.3, n_fft=None):
    """
    Gerchberg-Saxton style iterative flattening of the power spectrum.
    """
    n = len(a)
    if n_fft is None:
        N = 1
        while N < 2 * n - 1:
            N *= 2
    else:
        N = n_fft
    a = a.astype(np.float64).copy()
    
    for i in range(n_iter):
        frac = i / max(1, n_iter - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        
        A = np.fft.rfft(a, N)
        amp = np.abs(A)
        phase = A / (amp + 1e-30)
        
        # Target: flat amplitude
        mean_amp = amp.mean()
        target = np.full_like(amp, mean_amp)
        
        # Geometric mean blending
        new_amp = np.exp(alpha * np.log(amp + 1e-30) + (1 - alpha) * np.log(target + 1e-30))
        
        A_new = new_amp * phase
        a_new = np.real(np.fft.irfft(A_new, N)[:n])
        a_new = np.clip(a_new, 0.0, None)
        
        if np.sum(a_new) < 1e-10:
            return a
        
        a = a_new
    
    return a

    Subgradient descent to reduce the convolution peak.
    Returns the best sequence found during the descent.

    n = len(a)

    best = a.copy()
    best_score = fast_evaluate(best)
    
    momentum = np.zeros(n)
    
    for step_i in range(n_steps):

        # Compute convolution and find peak

    inits.append(np.linspace(0.5, 2, n))
    
    # 6. Cosine modulated
    for freq in [1, 2, 3, 5, 8, 13]:
        inits.append(1.0 + 0.8 * np.cos(2 * np.pi * freq * np.arange(n) / n))
    
    # 7. Gaussian window
    sigma = n / 4
    center = n / 2
    inits.append(np.exp(-(np.arange(n) - center) ** 2 / (2 * sigma ** 2)))
    
    # 8. Rational (1/(i+1))
    inits.append(1.0 / (np.arange(n) + 1))
    
    # 9. Random uniform
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # 10. Random (exponential distribution, shifted)
    inits.append(rng.exponential(1.0, n) + 0.1)
    
    # 11. Two-level: alternating high/low
    base = np.ones(n)
    base[::2] = 1.5
    base[1::2] = 0.5
    inits.append(base)
    
    # 12. Sawtooth
    period = max(2, n // 10)
    inits.append(1.0 + 0.5 * (np.arange(n) % period) / period)
    
    return inits
