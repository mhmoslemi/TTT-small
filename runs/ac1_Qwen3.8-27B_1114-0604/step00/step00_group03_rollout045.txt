"""
LP-based local search for minimizing 2n*max(a*a)/(sum a)^2 over non-negative sequences.

Key reformulation: normalize so that sum(a) = sqrt(2n). Then (sum a)^2 = 2n and the
objective reduces to simply max(a*a) (the peak of the auto-convolution), which is a
clean scalar to minimize.

At each iteration we solve the LP
    max sum(g)  s.t.  g >= 0,  (f*g)[k] <= M   for all k,
where M = max(f*f). The LP solution g is a non-negative direction that is "flat"
against the current peak; convex-combining f with a same-sum rescale of g and line
searching the step typically lowers the peak. Random perturbations and multiple
restarts escape degeneracy / local optima (constant sequences make the LP degenerate).
"""
import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a, b):
    """Linear convolution via FFT."""
    size = len(a) + len(b) - 1
    nfft = 1
    while nfft < size:
        nfft <<= 1
    return np.fft.ifft(np.fft.fft(a, nfft) * np.fft.fft(b, nfft)).real[:size]


def auto_conv_peak(a):
    """Return max(a*a) for a non-negative 1-d array (assumes sum ~ sqrt(2n))."""
    return float(np.max(fast_convolve(a, a)))


def true_score(a):
    """2 n max(a*a) / (sum a)^2, matching the evaluator for in-range values."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return float(2.0 * n * np.max(fast_convolve(a, a)) / (s * s))


def build_conv_matrix(f):
    """(2n-1) x n matrix A with A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[k, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return A


def lp_direction(f, M, rng, time_limit=25.0):
    """Solve the LP for an improving non-negative direction; None on failure."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n) + 1e-8 * rng.standard_normal(n)  # break degeneracy
    b_ub = np.full(2 * n - 1, M + 1e-12)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    g = np.maximum(res.x, 0.0)
    if np.sum(g) < 1e-12:
        return None
    return g


def smooth_random_perturb(f, rng, scale):
    """Add smooth non-negative-ish noise and renormalize to sum sqrt(2n)."""
    n = len(f)
    noise = rng.standard_normal(n)
    noise = 0.5 * (noise + np.roll(noise, 1))  # keep it smooth
    cand = f + scale * np.sqrt(2.0 * n) * noise
    cand = np.clip(cand, 0.0, None)
    s = np.sum(cand)
    if s < 1e-9:
        cand = np.abs(cand) + 1e-6
        s = np.sum(cand)
    return cand * np.sqrt(2.0 * n) / s


def optimize(f, rng, deadline, max_iters, label="", do_lp=True):
    """
    Run local search from starting vector f (length n).  Returns (best_a, best_score)
    found during this run using the peak objective (sum normalized to sqrt(2n)).
    """
    n = len(f)
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.clip(f, 0.0, None)
    s = np.sum(f)
    if s < 1e-9:
        f = np.ones(n)
        s = n
    f = f * np.sqrt(2.0 * n) / s

    best_a = f.copy()
    best_score = auto_conv_peak(f)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = auto_conv_peak(f)
        if M < best_score:
            best_score = M
            best_a = f.copy()
            stall = 0
            if it % 5 == 0 or M < 1.6:
                print(f"  [{label}] it{it} peak={M:.6f}")

        moved = False
        if do_lp:
            g = lp_direction(f, M, rng)
            if g is not None:
                sg = np.sum(g)
                s_f = np.sum(f)
                g = g * (s_f / sg)  # same-sum rescale

                # line search over convex-combination step
                local_best = M
                local_t = 0.0
                ffg = None
                for t in (0.05, 0.10, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9):
                    cand = (1.0 - t) * f + t * g
                    v = auto_conv_peak(cand)
                    if v < local_best:
                        local_best = v
                        local_t = t
                if local_t > 0.0 and local_best < M * (1.0 - 1e-9):
                    f = (1.0 - local_t) * f + local_t * g
                    moved = True
        if not moved:
            # exploratory step to escape
            f = smooth_random_perturb(f, rng, 0.04 + 0.02 * (stall // 10))
            stall += 1
        else:
            stall = 0

    return best_a, best_score


def make_initializations(rng):
    inits = []

    if "height_sequence_1" in globals():

    for n in (256, 512, 1024, 2048):
        # smooth positive noise
        a = np.abs(rng.standard_normal(n))
        for _ in range(3):
            a = 0.5 * (a + np.roll(a, 1))
        inits.append(a + 0.05)
        # cosine taper with jitter
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))
        inits.append(a + 0.04 * np.abs(rng.standard_normal(n)))

    for n in (512, 1024):
        # gap-in-middle (kills the central peak)
        a = np.ones(n)
        a[n // 3:2 * n // 3] = 0.0
        inits.append(a + 0.04 * np.abs(rng.standard_normal(n)))
        # two-hump
        a = np.zeros(n)
        a[:n // 3] = 1.0
        a[2 * n // 3:] = 1.0
        inits.append(a + 0.04 * np.abs(rng.standard_normal(n)))

    # de-dupe by length/content signature
    seen = set()
    out = []
    for a in inits:
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, None)
        if np.sum(a) < 0.01:
            continue
        key = (a.size, int(np.round(np.mean(a) * 100)), int(np.round(np.std(a) * 1000)))
        if key in seen:
            continue
        seen.add(key)
        out.append(a)
    return out


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing 2n*max(a*a)/(sum a)^2."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 12.0)

    best_a = None
    best_score = np.inf

    def consider(a, score, tag=""):
        nonlocal best_a, best_score
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = np.asarray(a, dtype=np.float64).copy()
            print(f"  ** new global best {tag}: peak={score:.6f} (true={true_score(best_a):.6f})")

    inits = make_initializations(rng)
    print(f"Generated {len(inits)} initializations")

    # Phase 1: LP-based optimization with limited iterations per start to stay diverse.
    n_starts = len(inits)
    per_start = 40
    for idx, init in enumerate(inits):
        if time.monotonic() > deadline:
            break
        n = len(init)
        iters = max(15, int(per_start * (768.0 / n)))  # more iters for shorter seqs
        print(f"[start {idx + 1}/{n_starts}] n={n} iters={iters}")
        a, sc = optimize(init, rng, deadline, iters, label=f"s{idx}", do_lp=True)
        consider(a, sc, tag=f"s{idx}/n{n}")

    # Phase 2: plain smooth local search (cheap) to squeeze a bit more, from the best.
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            print("Phase 2: cheap local refinement of best")
            for retry in range(20):
                if time.monotonic() > deadline:
                    break
                a, sc = optimize(smooth_random_perturb(best_a, rng, 0.03),
                                 rng, deadline, 30, label="ref", do_lp=False)
                consider(a, sc, tag=f"ref{retry}")

    # Phase 3: if time remains, run longer LP polish on the global best.
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60.0:
            print(f"Phase 3: LP polish (remaining {remaining:.0f}s)")
            a, sc = optimize(best_a * (1.0 + 0.02 * rng.standard_normal(len(best_a))),
                             rng, deadline, 400, label="polish", do_lp=True)
            consider(a, sc, tag="polish")

    if best_a is None:
        best_a = np.ones(128, dtype=np.float64)

    # Final exact verification / selection.
    final_true = evaluate_sequence(best_a.tolist())
    print(f"\nFINAL: peak-fast={best_score:.6f}  true-evaluator={final_true:.6f}  n={len(best_a)}")
    return [float(x) for x in best_a]
