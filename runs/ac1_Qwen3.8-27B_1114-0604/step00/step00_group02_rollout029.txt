from scipy.sparse import csr_matrix
import time

def build_conv_matrix(f):
    """Build the matrix A such that (f @ g) = A @ g where f@g is convolution."""
    n = len(f)
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # i.e., A[k, j] = f[j] * 1[max(0, k-n+1) <= j <= min(n-1, k)]

    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[js].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_iteration(f, M):
    """Solve the LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)

"""
LP-based iterative optimization for low-peak autocorrelation sequences.
Uses the primal LP to find the "best" direction to move, blends with 
current iterate, and iterates.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def build_conv_matrix(f):
    """Build sparse matrix A such that A @ g = linear_convolution(f, g)."""
    n = len(f)
    # Column j of A: f[j] in rows j to j+n-1
    rows_list = []
    cols_list = []
    vals_list = []
    for j in range(n):
        rows_list.append(np.arange(j, j + n))
        cols_list.append(np.full(n, j, dtype=np.int32))
        vals_list.append(np.full(n, f[j], dtype=np.float64))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def compute_score(f):
    """Compute the evaluation score for sequence f."""
    n = len(f)
    c = np.convolve(f, f)
    max_b = np.max(c)
    sum_a = np.sum(f)
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def lp_step(f, verbose=False):
    """Perform one LP iteration. Returns new g (normalized to sum=1) or None."""
    n = len(f)
    c = np.convolve(f, f)
    M = np.max(c)
    
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not result.success:
        if verbose:
            print(f"  LP failed: {result.message}")
        return None
    
    g = result.x
    g = np.clip(g, 0, None)
    g_sum = np.sum(g)
    if g_sum < 1e-10:
        return None
    
    g = g / g_sum  # normalize to sum=1
    return g

            g = lp_step(f)
            if g is None:

            # Try different step sizes
            best_local = f.copy()
            best_local_score = current_score

"""
LP-based iterative optimization for low-peak autocorrelation sequences.
Minimizes 2n*max(f*f)/(sum(f))^2 using iterative LP subproblems.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def build_conv_matrix(f):
    """Build sparse convolution matrix A where A@g = linear_conv(f, g)."""
    n = len(f)
    # Column j: f[j] in rows j..j+n-1
    total_nz = n * n
    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    vals = np.empty(total_nz, dtype=np.float64)
    
    idx = 0
    for j in range(n):
        rows[idx:idx+n] = np.arange(j, j+n, dtype=np.int32)
        cols[idx:idx+n] = j
        vals[idx:idx+n] = f[j]
        idx += n
    
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def compute_score(f):
    """Compute score = 2n * max(f*f) / (sum(f))^2."""
    n = len(f)
    c = np.convolve(f, f)
    max_b = np.max(c)
    sum_a = np.sum(f)
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def lp_solve(f, M):
    """Solve LP: max sum(g) s.t. A_f @ g <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={"presolve": True, "dual_feasibility_tolerance": 1e-7,
                              "complementarity_tolerance": 1e-7})
    
    if not result.success:
        return None
    
    g = np.clip(result.x, 0, None)
    g_sum = np.sum(g)
    if g_sum < 1e-15:
        return None
    
    return g / g_sum

            break
        
        c = np.convolve(f, f)
        M = np.max(c)

        if g is None:

        improved = False
        for t in [0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            f_try = (1.0 - t) * f + t * g
            f_try = np.clip(f_try, 0, None)
            s = np.sum(f_try)

            f_try = f_try / s

                stall_count = 0
        
        if verbose and (it + 1) % 20 == 0:
            print(f"  Iter {it+1}: score={current:.6f}, M={M:.6e}, elapsed={elapsed:.1f}s")

LP-based iterative search for low-peak autocorrelation non-negative sequences.
Minimizes 2n*max(f*f)/(sum f)^2 using iterative LP subproblems from random starts.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def build_conv_matrix(f):
    """Build sparse matrix A where A@g = linear_conv(f, g). Shape (2n-1, n)."""
    n = len(f)
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    for j in range(n):
        s = slice(j * n, (j + 1) * n)
        rows[s] = np.arange(j, j + n, dtype=np.int32)
        cols[s] = j
        vals[s] = f[j]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def get_score(f):
    """Compute 2n * max(f*f) / (sum f)^2."""
    n = len(f)
    c = np.convolve(f, f)
    s = np.sum(f)
    if s < 0.01:
        return float('inf')
    return 2.0 * n * c.max() / (s * s)


def solve_lp(f):
    """Solve max sum(g) s.t. conv(f,g) <= max(conv(f,f)), g>=0. Return g (sum=1) or None."""
    n = len(f)
    c_ff = np.convolve(f, f)
    M = c_ff.max()
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0.0, None)] * n, method='highs')
    if not res.success:
        return None
    g = np.clip(res.x, 0.0, None)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs
