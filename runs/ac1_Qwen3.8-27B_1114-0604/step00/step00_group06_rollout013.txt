Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using projected gradient descent
with subgradient of the max, combined with LP-based refinement and random restarts.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_subgradient(a, b, tau_frac=1e-3):
    """Compute subgradient of max(b) where b = a*a.
    Uses the active set (indices where b is close to max)."""
    n = len(a)
    max_b = np.max(b)
    threshold = max_b - tau_frac * max_b
    active = np.where(b >= threshold)[0]
    
    grad = np.zeros(n)
    for k in active:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        grad[j_lo:j_hi+1] += a[k-j_hi:k-j_lo+1][::-1]
    
    grad *= 2.0 / len(active)
    return grad, active, max_b


def solve_lp_for_g(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c = -np.ones(n)  # minimize -sum(g)
    
    # Build constraint matrix A (sparse)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 300})
    
    if result.success and result.x is not None:
        return result.x
    return None
