import numpy as np
import time


def _autocorr(f, size):
    """Compute linear autocorrelation of f via FFT. size is pre-allocated FFT size."""
    F = np.fft.rfft(f, n=size)
    return np.fft.irfft(np.abs(F) ** 2, n=size)


def _gradient_from_weights(f, weights, size):
    """Compute gradient: 2 * sum_k weights[k] * f[k-m] for m=0..n-1."""
    n = len(f)
    rev_f = f[::-1]
    W = np.fft.rfft(weights, n=size)
    R = np.fft.rfft(rev_f, n=size)
    conv = np.fft.irfft(W * R, n=size)
    return 2.0 * conv[n - 1: 2 * n - 1]


def _adam_step(f, grad, m_adam, v_adam, t, lr, beta1, beta2, eps):
    """One Adam update step with projection onto probability simplex."""
    m_adam = beta1 * m_adam + (1.0 - beta1) * grad
    v_adam = beta2 * v_adam + (1.0 - beta2) * grad * grad
    m_hat = m_adam / (1.0 - beta1 ** t)
    v_hat = v_adam / (1.0 - beta2 ** t)
    f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
    f_new = np.maximum(f_new, 0.0)
    s = np.sum(f_new)
    if s < 1e-15:
        return f_new / len(f), m_adam, v_adam
    return f_new / s, m_adam, v_adam


def _mirror_descent_step(f, grad, lr):
    """Multiplicative weight update on the simplex."""
    f_new = f * np.exp(-lr * grad)
    s = np.sum(f_new)
    if s < 1e-15:
        return np.ones_like(f) / len(f)
    return f_new / s
