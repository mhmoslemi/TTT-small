"""
LP-based iterative optimization minimizing 2*n*max(autocorr(a))/sum(a)^2.

Core idea (state-of-the-art direction): given current non-negative sequence a
normalized so sum(a) is fixed, let M = max_k (a*a)_k. Solve an LP that finds b>=0
maximizing sum(b) subject to (a*b)_k <= M for all lags k, then rescale b to match
the normalization and take a convex blend a <- (1-t)a + t b, choosing t that most
reduces the new peak. Repeat until the peak stops decreasing. Multiple sizes and
non-uniform initializations are tried to escape the all-ones fixed point and to
search the larger-attack-surface regime. Autocorrelation is computed with FFT and
the LP constraint matrix is built in CSR form. A valid all-ones fallback guarantees
a usable return.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_autocorr(a):
    """Full linear autocorrelation of a, equivalent to np.convolve(a, a)."""
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    r = np.fft.irfft(fa * fa.conjugate(), fft_size)[:size]
    return r


def score_of(a):
    """Write the objective 2*n*max(autocorr)/sum^2 directly (scale invariant)."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    m = fft_autocorr(a).max()
    return 2.0 * n * m / (s * s)


def build_lp_matrix(a):
    """Build (2n-1) x n matrix A with A[k,i] = a[k-i] for 0<=k-i<n.

    Column i holds a[0..n-1] on rows i..i+n-1, so A@b = a*b (full linear
    convolution padded into a 2n-1 vector). Returned in CSR form.
    """
    n = a.size
    cols = np.repeat(np.arange(n), n)              # [0,0,0,1,1,1,...]
    off = np.tile(np.arange(n), n)                 # [0,1,2,0,1,2,...]
    rows = cols + off
    vals = np.tile(a, n)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def best_blend_step(a, b, step_options):
    """Return the blend (1-t)a + t b (clipped >=0) minimizing the new peak."""
    best_a = a
    best_m = np.inf
    for t in step_options:
        cand = (1.0 - t) * a + t * b
        np.maximum(cand, 0.0, out=cand)
        if cand.sum() < 0.01:
            continue
        m = fft_autocorr(cand).max()
        if m < best_m:
            best_m = m
            best_a = cand
    return best_a


def run_lp_loop(a0, deadline, tag):
    """Run LP-based descent from a0 until time limit or stagnation.

    Returns (best_a, best_score) found during the loop.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 1e-10, 1000.0)
    n = a.size
    if a.sum() < 0.01:
        return a, np.inf

    best_a = a.copy()
    best_score = score_of(a)
    step_options = (0.2, 0.35, 0.5, 0.7, 0.9)
    prev_m = np.inf
    stagnant = 0

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        s = a.sum()
        r = fft_autocorr(a)
        m = float(r.max())
        score = 2.0 * n * m / (s * s)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if it % 25 == 0 or best_score < 1.8:
                print("  [%s] it=%d score=%.6f M=%.4f" % (tag, it, best_score, m))

        # convergence / stagnation detection
        if prev_m - m < 1e-9 * max(1.0, m):
            stagnant += 1
        else:
            stagnant = 0
        prev_m = m
        if stagnant > 20:
            break

        A = build_lp_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, m)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,

        if sum_b < 1e-12:

        b = b * (s / sum_b)
        a = best_blend_step(a, b, step_options)
