non-negative sequences.

Strategy:

    (uniform plus edge-weighted "U" and "cap" arches) at the parent length,

  * Drive each shape with a log-sum-exp smoothed gradient descent on the

    annealed smoothing (tau) and step size, and projection onto
    {sum(a)=S, a>=0} after every step (S = sqrt(2n) so the score is

  * Validate with evaluate_sequence and return the best valid list found

import numpy as np
import time


def _next_pow2(n):
    size = 1
    while size < n:
        size <<= 1
    return size


def _conv_fft(a, b):
    n = len(a) + len(b) - 1
    N = _next_pow2(n)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:n]


def _smoothed_obj_grad(a, tau, Nc, Nx):
    """Return (score, gradient) of log-sum-exp smoothed peak convolution."""
    n = len(a)
    S = np.sum(a)
    c = np.fft.irfft(np.fft.rfft(a, Nc) * np.fft.rfft(a, Nc), Nc)[:2 * n - 1]
    max_c = c.max()
    e = np.exp(tau * (c - max_c))
    Z = e.sum()
    p = e / Z
    score = 2.0 * n * max_c / (S * S)
    A = np.fft.rfft(a, Nx)
    P = np.fft.rfft(p, Nx)
    term1 = np.fft.irfft(np.conj(P) * A, Nx).real[:n]   # (p * a)[j]
    term2 = np.fft.irfft(A * P, Nx).real[:n]            # (a * p)[j]
    grad = term1 + term2
    return score, grad


def _score_fast(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-8:
        return np.inf
    c = _conv_fft(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def _project(a, S):
    """Project a onto {x: sum(x)=S, x>=0} via water-level bisection."""
    amax = a.max()
    aq = a.min()
    lo = aq - 1.0
    hi = amax + 1.0
    if hi - lo < 1e-12:
        out = a.copy()
    else:
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            if np.sum(np.maximum(0.0, a - mid)) > S:
                lo = mid
            else:
                hi = mid
        out = np.maximum(0.0, a - 0.5 * (lo + hi))
    s = out.sum()
    if s > 1e-12:
        out = out * (S / s)
    return out

    Nx = _next_pow2(3 * n)

    eta = eta0
