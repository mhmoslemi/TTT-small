"""
Minimize 2*n*max(full convolve(a,a)) / (sum(a))^2 for a non-negative step function.

Algorithm (scale-invariant: we normalize sum(a)=1 and minimize 2*n*peak(a*a)):
  Phase 1: Fast multiplicative (mirror-descent) subgradient steps using FFT
           auto-convolution to locate the active (max) lag and deflate the
           contributing coefficients. Multiple random/restructured starts.
  Phase 2: LP-based Frank-Wolfe (state-of-art idea) with a cutting-plane
           constraint set: solve max sum(g) s.t. (f*g)_k <= M on the active
           set and g >= 0, add the most-violated lag, line-search a convex
           combination update to reduce the peak.
  Phase 3: Local block/multiplicative refinement of the incumbent.
Always returns a valid non-negative list (fallback to the giving sequence).
"""
import time
import numpy as np
from scipy.fft import rfft, irfft


def full_convolve_fft(a, b):
    """Full linear convolution of 1-D arrays via rFFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = b.size
    size = 1
    need = n + m - 1
    while size < need:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    out = irfft(fa * fb)
    return out[:need]


def get_peak_conv(a):
    """Return (peak_value, full_auto_convolution) for a non-negative array."""
    conv = full_convolve_fft(a, a)
    np.maximum(conv, 0.0, out=conv)
    return float(np.max(conv)), conv


def normalize1(a):
    """Clip to >=0 and rescale to sum 1 (fallback if sum too small)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.full(a.shape, 1.0 / a.size)
    return a / s


def subgradient_at(f, k):
    """Subgradient of the single lag (f*f)_k w.r.t. f (length n)."""
    n = f.size
    idx = k - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    g = np.zeros(n, dtype=np.float64)
    g[valid] = 2.0 * f[idx[valid]]
    return g


def multiplicative_phase(f, deadline, max_steps, tag):
    """Multiplicative mirror-descent using the active lag. Returns best f,peak."""
    best = f.copy()
    best_peak, _ = get_peak_conv(best)
    eta = 0.5
    f = f.copy()
    stalled = 0
    for it in range(max_steps):
        if time.monotonic() > deadline:
            break
        _, conv = get_peak_conv(f)
        k = int(np.argmax(conv))
        grad = subgradient_at(f, k)
        f_new = f * np.exp(-eta * grad)
        np.maximum(f_new, 1e-300, out=f_new)
        f_new = normalize1(f_new)
        peak_new, _ = get_peak_conv(f_new)
        if peak_new < best_peak - 1e-13:
            best = f_new
            best_peak = peak_new
            f = f_new
            stalled = 0
        else:
            f = f_new
            stalled += 1
            if stalled % 250 == 0:
                eta *= 0.6
        if it % 400 == 0:
            print("   %s multstep %d peak=%.8f" % (tag, it, best_peak))
    return best, best_peak


def lp_frank_wolfe(f, deadline, max_iters, sub_budget):
    """LP Frank-Wolfe with cutting plane. Returns best f,peak."""
    import cvxpy as cp
    n = f.size
    best = f.copy()
    best_peak, _ = get_peak_conv(best)
    f = f.copy()
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M, _ = get_peak_conv(f)
        # Build cutting-plane LP: active lags start at the center (peak of f*f).
        _, convf = get_peak_conv(f)
        center_k = int(np.argmax(convf))
        gvar = cp.Variable(n)
        constraints = []
        active_ks = set()

        def row_for(k):
            js = np.arange(max(0, k - n + 1), min(k, n - 1) + 1)
            return js, f[k - js]

        def add_k(k):
            js, coef = row_for(int(k))
            if js.size == 0:
                return
            constraints.append(coef @ gvar[js] <= M)
            active_ks.add(int(k))

        add_k(center_k)
        g_opt = None
        for cp_it in range(40):
            if time.monotonic() > deadline:
                break
            prob = cp.Problem(cp.Maximize(cp.sum(gvar)), constraints)
            solved = False
            for solver in (cp.SCS, cp.ECOS, cp.GLPK):
                try:
                    prob.solve(solver=solver, verbose=False)
                    solved = True
                    break
                except Exception:
                    continue
            if not solved or gvar.value is None:
                break
            g_val = np.maximum(np.asarray(gvar.value, dtype=np.float64), 0.0)
            cg = full_convolve_fft(f, g_val)
            np.maximum(cg, 0.0, out=cg)
            viol = cg > M * (1.0 + 1e-6) + 1e-12
            if not np.any(viol):
                g_opt = g_val
                break
            worst = int(np.argmax(np.where(viol, cg, -1.0)))
            if worst not in active_ks:
                add_k(worst)
        if g_opt is None:
            break
        gsum = float(g_opt.sum())
        if gsum < 1e-12:
            break
        g_opt = g_opt / gsum
        # Line search over convex combination.
        chosen = None
        chosen_peak = M
        for t in (0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02):
            cand = normalize1((1.0 - t) * f + t * g_opt)
            pk, _ = get_peak_conv(cand)
            if pk < chosen_peak:
                chosen_peak = pk
                chosen = cand
        if chosen is None:
            pk, _ = get_peak_conv(g_opt)
            if pk < M:
                chosen = g_opt
                chosen_peak = pk
            else:
                break
        f = chosen
        npk = chosen_peak
        if npk < best_peak:
            best = f.copy()
            best_peak = npk
        if it % 10 == 0:
            print("   LP it %d M=%.8f new=%.8f best=%.8f" % (it, M, npk, best_peak))
        if abs(M - npk) < 1e-12:
            break
    return best, best_peak


def local_refine(f, deadline, max_steps):
    """Small multiplicative / block local refinement of incumbent."""
    rng = np.random.default_rng(12345)
    n = f.size
    best = f.copy()
    best_peak, _ = get_peak_conv(best)
    for it in range(max_steps):
        if time.monotonic() > deadline:
            break
        f = f.copy()
        w = int(rng.integers(max(2, n // 200), max(3, n // 12)))
        s = int(rng.integers(0, n - w))
        f[s:s + w] *= float(np.exp(rng.normal(0.0, 0.06)))
        f = normalize1(f)
        pk, _ = get_peak_conv(f)
        if pk < best_peak:
            best = f.copy()
            best_peak = pk
    return best, best_peak
