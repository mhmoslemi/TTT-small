```python
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                idx = k - j
                if 0 <= idx < n:
                    row[j] = best[idx]
            A_rows.append(row)
        A = np.array(A_rows)
        b_vec = np.full(2*n - 1, M)

            g0 = res.x
            g_sum = np.sum(g0)
            if g_sum > 1e-10:
                g = g0 * S_target / g_sum
                # Blend
                t = 0.5
                candidate = (1 - t) * best + t * g
                candidate = np.clip(candidate, 0, 1000)
                score = evaluate_sequence(candidate.tolist())
                if score < best_score:
                    best = candidate
                    best_score = score
                    print(f"Improved to {best_score:.6f} at t={t}")
                else:
                    # Try smaller blend

                    candidate = (1 - t) * best + t * g
                    candidate = np.clip(candidate, 0, 1000)

                    if score < best_score:

        else:

            break
    
    return [float(x) for x in best]
```

If k = 2i: ∂c[k]/∂a[i] = a[i] + a[-i] = a[i] + 0 = a[i] (for i > 0)
If k ≠ 2i: ∂c[k]/∂a[i] = a[k-i] + a[i-k] (both terms are 0 unless the index is in [0,n-1])

OK so in general: ∂c[k]/∂a[i] = a[k-i] + a[i-k] with the convention a[m]=0 for m < 0 or m >= n.

For the subgradient of f(a) = 2n * max_k c[k] / S^2:
Let k* be an argmax of c. Then:
∇f = (2n/S^2) * ∂c[k*]/∂a - (4n*c[k*]/S^3) * 1

where ∂c[k*]/∂a[i] = a[k*-i] + a[i-k*] (with zero extension).

So the descent direction for a[i] is:
d[i] = (a[k*-i] + a[i-k*]) - 2*c[k*]/S

We want to move a in the direction -d (to decrease f).

So: a_new[i] = a[i] - lr * d[i] = a[i] - lr * (a[k*-i] + a[i-k*] - 2*c[k*]/S)

Then project onto [0, 1000] and ensure sum >= 0.01.

```python
    size = 1 << (n - 1).bit_length()  # Next power of 2
    fa = np.fft.rfft(a, size)
```

**Let me reconsider the LP approach.**

The LP approach from the state of the art is:
1. Given current f, compute M = ||f*f||_inf
2. Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0
3. Rescale b to have sum = sum(f)
4. Update: f <- (1-t)*f + t*b

The key insight is that this LP is LINEAR (not quadratic), because f is fixed and we're optimizing over b. The constraint (f*b)[k] = sum_j f[j]*b[k-j] <= M is linear in b.

The LP has:
- n variables (b_0, ..., b_{n-1})
- 2n-1 constraints (the convolution constraints)
- n non-negativity constraints

For n=6121, this is a 6121-variable, 12241-constraint LP. With a sparse solver, this should be solvable in a few seconds.

```python
def solve_active_set_lp(f, M, max_constraints=500):
    """Solve the LP using only active constraints, adding violated ones iteratively."""
    n = len(f)
    
    # Start with the constraint at the peak
    c = np.convolve(f, f, mode='full')
    active = [np.argmax(c)]
    
    for iteration in range(50):
        if len(active) >= max_constraints:
            break
        
        # Build LP with current active constraints
        rows = []
        cols = []
        data = []
        b_vec = []
        
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                rows.append(len(b_vec))  # row index in the constraint matrix
                cols.append(j)
                data.append(f[k - j])
            b_vec.append(M)
        
        A = sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))
        
        c_obj = -np.ones(n)  # minimize -sum(b)
        
        res = linprog(c_obj, A_ub=A, b_ub=np.array(b_vec), 
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            return None
        
        b = res.x
        
        # Check for violated constraints
        fb = np.convolve(f, b, mode='full')
        violated = np.where(fb > M * 1.001)[0]
        
        if len(violated) == 0:
            return b
        
        # Add most violated constraints
        violated_sorted = violated[np.argsort(fb[violated])[::-1]]
        for v in violated_sorted[:10]:
            if v not in active:
                active.append(v)
    
    # If we didn't converge, use what we have
    b = res.x
    return b
```
