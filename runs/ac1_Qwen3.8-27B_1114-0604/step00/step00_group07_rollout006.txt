from scipy.optimize import linprog
import time

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. C_f @ g <= M, g >= 0"""
    n = len(f)
    # C_f is (2n-1) x n
    # C_f[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Actually, (f*g)(k) = sum_j f[k-j] * g[j] for valid j
    # So C_f[k, j] = f[k-j] if max(0, k-n+1) <= j <= min(k, n-1), else 0
    
    # For efficiency, build the constraint matrix
    # But for large n, this matrix is sparse (each row has at most n/2 nonzeros on average)
    
    # Actually, each row k has exactly min(k+1, n, 2n-1-k) nonzeros.
    # So the matrix is Toeplitz-like but not exactly (it's tall).
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(k, n-1)+1):
            A_ub[k, j] = f[k-j]
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x

        return None

        return None
    
    # Rescale to match sum of f
    sum_f = np.sum(f)
    g = g * (sum_f / sum_g)
    
    return g
