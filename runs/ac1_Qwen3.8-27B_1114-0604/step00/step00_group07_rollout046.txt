"""
Minimize eval = 2n * max(f*f) / (sum f)^2 over non-negative step sequences.

Key facts exploited:
  * The score is scale-invariant, so we keep sequences normalized to sum==1.
  * Symmetric non-negative sequences satisfy f*f[n-1] = sum f[i]^2 >= (sum f)^2/n,
    giving score >= 2.  Hence strictly asymmetric sequences are required to beat 2.
  * A "flat" autocorrelation approaches the theoretical lower bound of score~1.

Main driver: the LP direction of the previous state-of-the-art.  Given f with
peak M = max(f*f), solve  max 1'.g  s.t.  g>=0 and (f*g)[k] <= M  for all k.
The optimal g spreads mass away from the old peak; blending f <- (1-t)f + t*g_norm
(and re-normalizing) monotonically lowers the peak for a suitable t found by line
search.  We run this on several asymmetric starting points of several lengths,
insert random escapes to break fixed points, then finish with a fast
peak-directed coordinate polish.  The best sequence found is returned.
"""
import numpy as np
import time
from scipy.optimize import linprog


# ---------------------------------------------------------------------------
# Convolution matrix
# ---------------------------------------------------------------------------
def build_conv_matrix(f):
    """Return A with A @ g == np.convolve(f, g) (length 2n-1 by n)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = f[k - j_min:k - j_max - 1:-1]
    return A


def conv_peak(f):
    return np.max(np.convolve(f, f))


# ---------------------------------------------------------------------------
# LP direction
# ---------------------------------------------------------------------------
def lp_direction(f, M):
    """max 1'.g  s.t.  g>=0, (f*g)[k] <= M."""
    n = len(f)
    c = -np.ones(n)
    A_ub = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None
    if res is not None and res.success:
        return np.asarray(res.x, dtype=np.float64)
    return None


def normalize(f):
    s = np.sum(f)
    if s <= 0 or not np.isfinite(s):
        return None
    return f / s


def safe_eval(f):
    try:
        v = evaluate_sequence([float(x) for x in f])
    except Exception:
        return np.inf
    if not np.isfinite(v):
        return np.inf
    return float(v)


# ---------------------------------------------------------------------------
# LP iteration with line search + escapes
# ---------------------------------------------------------------------------
def lp_iterate(f0, budget, rng, max_iters=2000):
    f = np.asarray(f0, dtype=np.float64).copy()
    f = np.clip(f, 0.0, 1000.0)
    fn = normalize(f)
    if fn is None:
        return np.ones(16), safe_eval(np.ones(16))
    f = fn
    best_f = f.copy()
    best_score = safe_eval(f)
    if not np.isfinite(best_score):
        best_score = np.inf
    deadline = time.monotonic() + budget

    stuck_counter = 0
    ts = [1.0, 0.66, 0.5, 0.33, 0.25, 0.16, 0.08, 0.04, 0.02, 0.01]

    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = conv_peak(f)
        if M <= 0:
            break
        g = lp_direction(f, M)
        if g is None or not np.all(np.isfinite(g)):
            break
        sg = np.sum(g)
        if sg <= 1e-12:
            break
        g = g / sg  # sum == 1

        # if LP already agrees with current point, we are at a fixed point
        if np.linalg.norm(g - f) < 1e-9:
            stuck_counter += 1
        else:
            stuck_counter = 0

        improved = False
        # line search: evaluate several t, keep the best
        cand_best = None
        cand_best_score = best_score
        for t in ts:
            f_new = (1.0 - t) * f + t * g
            f_new = np.clip(f_new, 0.0, None)
            fn2 = normalize(f_new)
            if fn2 is None:
                continue
            sc = safe_eval(fn2)
            if sc < cand_best_score:
                cand_best_score = sc
                cand_best = fn2
        if cand_best is not None:
            f = cand_best
            best_f = f.copy()
            best_score = cand_best_score
            improved = True

        if not improved:
            # escape: random asymmetric perturbation of a random block
            n = len(f)
            width = max(1, n // 16)
            start = int(rng.integers(0, n - width + 1))
            f = f.copy()
            f[start:start + width] *= np.exp(rng.normal(0.0, 0.4, size=width))
            # inject asymmetry
            f = np.clip(f, 0.0, None)
            f = normalize(f)
            if f is None:
                break
            if not (improved or np.linalg.norm(f - best_f) > 0):
                stuck_counter += 1
            if stuck_counter > 30:
                break

    return best_f, best_score

# Fast peak-directed coordinate polish
# ---------------------------------------------------------------------------
def coordinate_polish(f0, budget, rng, max_iters=200000):
    f = np.asarray(f0, dtype=np.float64).copy()
    fn = normalize(f)
    if fn is None:
        return f0, safe_eval(f0)
    f = fn
    best_f = f.copy()
    best_score = safe_eval(f)
    deadline = time.monotonic() + budget
    n = len(f)

    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = np.convolve(f, f)
        m = int(np.argmax(c))
        # contributions f[i]*f[m-i] for valid i
        i0 = max(0, m - n + 1)
        i1 = min(n - 1, m)
        if i0 > i1:
            j = int(rng.integers(0, n))
        else:
            idx = np.arange(i0, i1 + 1)
            pair = m - idx
            contrib = f[idx] * f[pair]
            j = int(idx[int(np.argmax(contrib))])
        # move a bit of mass out of j to a position far from the center-fold
        donor = j
        # choose receiver far from m - j
        far = int(np.argmin(np.abs(np.arange(n) - (m - j))))
        if far == donor:
            far = (donor + n // 2) % n
        frac = float(rng.uniform(0.05, 0.35))
        move = f[donor] * frac
        if not np.isfinite(move) or move <= 0:
            move = max(0.0, f[donor]) * 0.1
        f_new = f.copy()
        f_new[donor] -= move
        f_new[far] += move
        f_new = np.clip(f_new, 0.0, None)
        fn2 = normalize(f_new)
        if fn2 is None:
            continue
        sc = safe_eval(fn2)
        if sc < best_score:
            f = fn2
            best_f = f.copy()
            best_score = sc

    return best_f, best_score

    budget_s = float(budget_s)
    total_deadline = time.monotonic() + max(0.0, budget_s) - 2.0

    best_f, best_score = None, np.inf

    # fallback valid sequence
    def fb():
        return np.ones(64, dtype=np.float64)

    if best_f is None:
        best_f = fb()
        best_score = safe_eval(best_f)

    starts = make_starting_points(rng)
    if len(starts) == 0:
        starts = [fb()]

    # rank starts quickly
    scored = []
    for s in starts:
        sc = safe_eval(s)
        scored.append((sc, s))
    scored.sort(key=lambda x: x[0])

    n_start = len(scored)
    print("collected %d starting seeds, best seed score %.6f" % (n_start, scored[0][0] if n_start else -1), flush=True)

    # Phase 1: LP iteration on the best seeds, with growing time budget
    # leave time for phase 2 polishing
    phase2_reserve = min(600.0, budget_s * 0.25)
    used = 0.0
    t0 = time.monotonic()

    # allocate phase-1 time across top seeds, heavier for better seeds
    top_k = min(6, n_start)
    for rank in range(top_k):
        if time.monotonic() + phase2_reserve > total_deadline:
            break
        sc, s = scored[rank]
        # budget per seed is larger for top seeds

    # random restarts in remaining phase-1 time with fresh asymmetric seeds
    while time.monotonic() + phase2_reserve < total_deadline:

        if s is None:

        budget = min(60.0, total_deadline - time.monotonic() - phase2_reserve)
        if budget <= 0:

        f, sc2 = lp_iterate(s, budget, rng, max_iters=400)
        if sc2 < best_score:
            best_score = sc2
            best_f = f.copy()
        print("[restart] n=%d p=%.2f score=%.6f best=%.6f"

    # Phase 2: coordinate polish of the best
    if best_f is not None and time.monotonic() < total_deadline:

        f, sc2 = coordinate_polish(best_f, rem, rng)
        if sc2 < best_score:
            best_score = sc2
            best_f = f.copy()
        print("[polish] n=%d score=%.6f best=%.6f"
              % (len(f), sc2, best_score), flush=True)

    # final sanity: guarantee a valid, finite sequence
    out = np.asarray(best_f, dtype=np.float64)

        out = fb()
    out = np.clip(out, 0.0, 1000.0)
    final_score = safe_eval(out)
    if not np.isfinite(final_score):
        out = fb()
        final_score = safe_eval(out)
    print("FINAL n=%d score=%.6f" % (len(out), final_score), flush=True)
    return [float(x) for x in out]
