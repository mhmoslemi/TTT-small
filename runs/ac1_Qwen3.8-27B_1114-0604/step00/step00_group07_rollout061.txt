```python
"""
Search for nonnegative step-function heights minimizing
    2 * n * max(conv(a, a)) / (sum(a)**2).

The program uses FFT-based autocorrelation scoring, a mixture of structured
smooth starts and random smooth starts, a low-dimensional differential
evolution step, and full-vector L-BFGS optimization on a log-sum-exp
approximation of the autocorrelation peak.  It finishes with exact active-set
subgradient sharpening.
"""

import time
import numpy as np
from scipy.optimize import minimize as _minimize
try:
    from scipy.optimize import differential_evolution as _differential_evolution
except Exception:
    _differential_evolution = None

PARAM_N = 0
PARAM_T = None


def _fft_size(need):
    need = int(need)
    if need <= 1:
        return 1
    return 1 << (need - 1).bit_length()


def square_conv(a):
    """Full self-convolution of a real 1-D array using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    need = 2 * n - 1
    size = _fft_size(need)
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)[:need]
    b = np.maximum(b, 0.0)
    return b


def pressure_fft(w, a):
    """
    Return c_j = sum_i a_i * w_{i+j}, j=0..n-1,
    where len(w) should be 2n-1.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    need = w.size + n - 1
    size = _fft_size(need)
    fa = np.fft.rfft(a[::-1], size)
    fw = np.fft.rfft(w, size)
    c = np.fft.irfft(fa * fw, size)[:need]
    return c[n - 1:2 * n - 1]


def normalize_exp(x):
    """Interpret x as log-heights and normalize to sum 1, positive entries."""
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return np.asarray([], dtype=np.float64)
    if not np.all(np.isfinite(x)):
        x = np.nan_to_num(x, nan=0.0, posinf=30.0, neginf=-30.0)
    x = np.clip(x, -30.0, 30.0)
    x = x - np.max(x)
    a = np.exp(x)
    a = np.maximum(a, 1e-300)
    s = a.sum()
    if not np.isfinite(s) or s <= 0.0:
        n = x.size
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def quick_score(a):
    """Scale-invariant objective estimate for a nonnegative array."""
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return float("inf")
    b = square_conv(a)
    if b.size == 0:
        return float("inf")
    m = float(b.max())
    if not np.isfinite(m):
        return float("inf")
    # Tiny guard against tiny FFT bias.
    return 2.0 * float(a.size) * (m + 1e-12) / (s * s)


def add_candidate(cands, x):
    x = np.asarray(x, dtype=np.float64)
    if x.ndim != 1 or x.size == 0:
        return
    cands.append(normalize_exp(x))


def generate_random_candidates(n, count, seed):
    """Generate random but smooth positive candidates."""
    rng = np.random.default_rng(seed)
    t = np.linspace(0.0, 1.0, n)
    out = []
    for _ in range(int(count)):
        x = (
            rng.normal(0.0, 1.0) * t
            + rng.normal(0.0, 1.0) * (t - 0.5)
            + rng.normal(0.0, 0.6) * (t - 0.5) ** 2
            + rng.normal(0.0, 0.4) * np.sin(2.0 * np.pi * t + rng.uniform(0.0, 2.0 * np.pi))
            + rng.normal(0.0, 0.4) * np.sin(4.0 * np.pi * t + rng.uniform(0.0, 2.0 * np.pi))
        )
        noise = rng.normal(0.0, 0.2, n)
        if n > 1:
            f = np.fft.rfft(noise)
            freq = np.fft.rfftfreq(n)
            sigma = max(2.0, float(n) * 0.02)
            f *= np.exp(-0.5 * (2.0 * np.pi * freq / sigma) ** 2)
            noise = np.fft.irfft(f, n)
        x += noise
        out.append(normalize_exp(x))
    return out


def generate_candidates(n, seed):
    """Generate a diversified set of structured and random smooth starts."""
    rng = np.random.default_rng(seed)
    t = np.linspace(0.0, 1.0, n)
    eps = 1.0 / float(max(1, n))
    cands = []

    add_candidate(cands, np.zeros(n))

    # Exponentials and shifted exponentials.
    for lam in (0.3, 0.6, 1.0, 1.5, 2.0, 3.0, 5.0):
        add_candidate(cands, -lam * t)
        add_candidate(cands, -lam * (1.0 - t))

    # Power-law endpoint behavior.
    for p in (-0.8, -0.5, 0.0, 0.3, 0.7, 1.2, 2.0):
        add_candidate(cands, p * np.log(t + eps))
        add_candidate(cands, p * np.log(1.0 - t + eps))

    # Beta-like two-endpoint family.
    beta_p = (0.0, 0.2, 0.5, 1.0, 1.5, 2.5)
    for p in beta_p:
        for q in beta_p:
            add_candidate(cands, p * np.log(t + eps) + q * np.log(1.0 - t + eps))

    # Low-order skew polynomial family.
    for _ in range(60):
        A = rng.normal(0.0, 1.5)
        B = rng.normal(0.0, 1.0)
        C = rng.normal(0.0, 1.0)
        D = rng.normal(0.0, 0.5)
        add_candidate(cands, A * t + B * (t - 0.5) ** 2 + C * (t - 0.5) ** 3 + D)

    # Gaussians and ridge shifts.
    for mu in (0.2, 0.35, 0.5, 0.65, 0.8):
        for sig in (0.08, 0.15, 0.25, 0.4):
            add_candidate(cands, -0.5 * ((t - mu) / sig) ** 2)
        for k in (1.0, 3.0, 8.0, 20.0):
            add_candidate(cands, -k * (t - mu))

    # Smooth periodic skews.
    for _ in range(30):
        A = rng.normal(0.0, 1.0)
        B = rng.normal(0.0, 0.8)
        C = rng.normal(0.0, 0.8)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        add_candidate(cands, A * t + B * np.sin(2.0 * np.pi * t + ph) + C * np.cos(2.0 * np.pi * t + ph))

    # More aggressively filtered random smooth starts.
    cands.extend(generate_random_candidates(n, 90, seed + 123457))

    return cands


def resize_sequence(seq, n):
    """Resize a nonnegative sequence to length n by linear interpolation."""
    arr = np.asarray(seq, dtype=np.float64)
    arr = np.clip(np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0), 0.0, 1000.0)
    if arr.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    s = arr.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    arr = arr / s
    if arr.size == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old_t = np.linspace(0.0, 1.0, arr.size)
    new_t = np.linspace(0.0, 1.0, n)
    out = np.interp(new_t, old_t, arr)
    out = np.maximum(out, 0.0)
    s2 = out.sum()
    if s2 <= 0.0 or not np.isfinite(s2):
        return np.ones(n, dtype=np.float64) / float(n)
    return out / s2


def mix_candidates(a, b, weight):
    """Geometric/log-domain mixture of two positive normalized sequences."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    la = np.log(np.maximum(a, 1e-300))
    lb = np.log(np.maximum(b, 1e-300))
    return normalize_exp(weight * la + (1.0 - weight) * lb)


def select_candidates(cands, scores, k):
    """Pick k diversified candidates, preferring low score."""
    if not cands:
        return []
    scores = np.asarray(scores, dtype=np.float64)
    order = np.argsort(scores)
    chosen = []
    ids = []

    for idx in order:
        if len(chosen) >= int(k):
            break
        idx = int(idx)
        if idx in ids:
            continue
        a = cands[idx]
        div = True
        for b in chosen:
            tv = 0.5 * float(np.sum(np.abs(a - b)))
            if tv < 0.02:
                div = False
                break
        if div:
            chosen.append(a)
            ids.append(idx)

    if len(chosen) < int(k):
        for idx in order:
            if len(chosen) >= int(k):
                break
            idx = int(idx)
            if idx not in ids:
                chosen.append(cands[idx])
                ids.append(idx)
    return chosen


def param_x(theta, n):
    """Expand a 6-parameter smooth shape."""
    t = np.linspace(0.0, 1.0, n)
    A, B, C, D, E, F = theta
    return (
        A * t
        + B * (t - 0.5) ** 2
        + C * (t - 0.5) ** 3
        + D * np.sin(2.0 * np.pi * t)
        + E * np.cos(2.0 * np.pi * t)
        + F * np.sin(np.pi * t)
    )


def param_shape_score(theta):
    """Objective for low-dimensional parameter optimization."""
    if PARAM_T is None:
        return float("inf")
    x = param_x(theta, len(PARAM_T))
    a = normalize_exp(x)
    return quick_score(a)


def find_param_shape(seed, deadline):
    """Optional low-dimensional global search for a good smooth seed shape."""
    global PARAM_N, PARAM_T
    if _differential_evolution is None:
        return None
    if time.monotonic() > deadline - 180.0:
        return None
    PARAM_N = 1024
    PARAM_T = np.linspace(0.0, 1.0, PARAM_N)
    bounds = [
        (-4.0, 4.0),
        (-3.0, 3.0),
        (-3.0, 3.0),
        (-2.0, 2.0),
        (-2.0, 2.0),
        (-1.5, 1.5),
    ]
    try:
        res = _differential_evolution(
            param_shape_score,
            bounds,
            seed=seed,
            maxiter=30,
            popsize=15,
            tol=1e-8,
            init="latinhypercube",
            polish=False,
        )
        if res is not None and getattr(res, "x", None) is not None:
            x = np.asarray(res.x, dtype=np.float64)
            if np.all(np.isfinite(x)):
                return x
    except Exception:
        pass
    return None


def obj_jac(x, beta, n):
    """
    Smooth scale-invariant objective and softmax gradient.
    Objective approximates 2*n*max(conv(a,a)) with a log-sum-exp.
    """
    x = np.asarray(x, dtype=np.float64)
    if x.size != n:
        x = np.resize(x, n)
    if not np.all(np.isfinite(x)):
        x = np.nan_to_num(x, nan=0.0, posinf=30.0, neginf=-30.0)

    xm = x - np.max(x)
    ex = np.exp(np.clip(xm, -745.0, 0.0))
    s = ex.sum()
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a = ex / s

    b = square_conv(a)
    if b.size == 0:
        return 1e100, np.zeros(n, dtype=np.float64)
    bmax = float(b.max())
    if not np.isfinite(bmax):
        return 1e100, np.zeros(n, dtype=np.float64)

    bm = b - bmax
    ez = np.exp(np.clip(beta * bm, -745.0, 0.0))
    Z = ez.sum()
    if Z <= 0.0 or not np.isfinite(Z):
        val = 2.0 * float(n) * bmax
        w = np.zeros_like(b, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
    else:
        w = ez / Z
        val = 2.0 * float(n) * (bmax + np.log(Z) / beta)

    c = pressure_fft(w, a)
    pressure = 2.0 * c
    pdot = float(np.dot(a, pressure))
    grad = (2.0 * float(n)) * a * (pressure - pdot)

    if not np.isfinite(val):
        val = 1e100
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    return float(val), grad


def pressure_direct(a, idx):
    """Exact subgradient pressure for a set of active convolution indices."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    idx = np.asarray(idx, dtype=np.int64)
    L = idx.size
    c = np.zeros(n, dtype=np.float64)
    if L == 0:
        return c
    for k in idx:
        kk = int(k)
        lo = kk - n + 1
        if lo < 0:
            lo = 0
        hi = kk
        if hi >= n:
            hi = n - 1
        if hi >= lo:
            c[lo:hi + 1] += a[kk - hi:kk - lo + 1][::-1]
    return (2.0 / float(L)) * c


def top_lag_refine(a, max_iter, deadline, seed=0):
    """
    Exact active-set sharpening:
    repeatedly push down the positions contributing to the highest autocorr
    lags, using multiplicative log-updates and exact score checks.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if not np.isfinite(s) or s <= 0.0:
        return a, float("inf")
    a = a / s

    n = a.size
    best = a.copy()
    best_score = quick_score(a)
    x = np.log(np.maximum(a, 1e-300))
    x = x - np.mean(x)

    rng = np.random.default_rng(seed)
    eta = 0.2
    no_improve = 0
    K_cycle = (5, 10, 25, 60)

    for it in range(int(max_iter)):
        if deadline is not None and time.monotonic() > float(deadline):
            break

        b = square_conv(a)
        M = float(b.max())
        score = 2.0 * float(n) * (M + 1e-12)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()

        K = int(K_cycle[it % len(K_cycle)])
        if K >= b.size:
            idx = np.nonzero(b > 0.0)[0]
        else:
            idx = np.argpartition(b, -K)[-K:]

        pressure = pressure_direct(a, idx)
        p = pressure - float(np.median(pressure))
        scale = float(np.std(p))
        if not np.isfinite(scale) or scale <= 1e-20:
            x = x + rng.normal(0.0, 0.03, n)
            a = normalize_exp(x)
            continue

        x_new = x - (eta * p / scale)
        a_new = normalize_exp(x_new)
        s_new = quick_score(a_new)

        if np.isfinite(s_new) and s_new < best_score * (1.0 - 1e-13):
            x = x_new
            a = a_new
            best = a_new.copy()
            best_score = s_new
            eta = min(1.0, eta * 1.1)
            no_improve = 0
        else:
            eta *= 0.7
            no_improve += 1
            if no_improve > 25:
                x = x + rng.normal(0.0, 0.05, n)
                a = normalize_exp(x)
                no_improve = 0

    return best, best_score


def refine_sequence(a, deadline, seed=0):
    """Deep refiner: log-sum-exp L-BFGS plus exact active-set polish."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if not np.isfinite(s) or s <= 0.0:
        return a, float("inf")
    a = a / s

    n = a.size
    best = a.copy()
    best_score = quick_score(a)
    x = np.log(np.maximum(a, 1e-300))
    x = x - np.mean(x)

    betas = (1.0 * n, 10.0 * n, 100.0 * n, 300.0 * n)
    maxiters = (60, 120, 200, 300)

    for beta, mit in zip(betas, maxiters):
        if time.monotonic() > float(deadline) - 30.0:
            break
        try:
            res = _minimize(
                obj_jac,
                x,
                args=(beta, n),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": int(mit), "ftol": 1e-12, "gtol": 1e-10},
            )
            if res is not None and getattr(res, "x", None) is not None:
                x = np.asarray(res.x, dtype=np.float64)
        except Exception:
            pass

        a = normalize_exp(x)
        sc = quick_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()
            x = np.log(np.maximum(best, 1e-300))
            x = x - np.mean(x)

    best, best_score = top_lag_refine(best, 40, deadline, seed)

    # Final high-temperature polish after the exact active-set move.
    if time.monotonic() < float(deadline) - 30.0:
        x = np.log(np.maximum(best, 1e-300))
        x = x - np.mean(x)
        beta = 300.0 * n
        try:
            res = _minimize(
                obj_jac,
                x,
                args=(beta, n),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 150, "ftol": 1e-12, "gtol": 1e-10},
            )
            if res is not None and getattr(res, "x", None) is not None:
                x = np.asarray(res.x, dtype=np.float64)
                a = normalize_exp(x)
                sc = quick_score(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best = a.copy()
        except Exception:
            pass
        best, best_score = top_lag_refine(best, 30, deadline, seed + 17)

    return best, best_score


def propose_candidate(seed=42, budget_s=2400.0):
    """
    Main search function.  Returns a list of nonnegative floats.
    """
    start = time.monotonic()
    # Keep a safety margin so the process can finish cleanly.
    total_deadline = start + max(10.0, float(budget_s) - 35.0)

    print(f"target budget_s={budget_s}, internal deadline margin left={max(0.0,total_deadline-start):.1f}s")

    best_seq = None
    best_score = float("inf")

    # Start from the provided previous construction if available.
    parent_arr = None
    try:
        raw = height_sequence_1  # noqa: F821
        parent_arr = np.asarray(raw, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0:
            parent_arr = np.nan_to_num(parent_arr, nan=0.0, posinf=0.0, neginf=0.0)
            parent_arr = np.clip(parent_arr, 0.0, 1000.0)
            ps = float(parent_arr.sum())
            if np.isfinite(ps) and ps > 0.0:
                parent_arr = parent_arr / ps
                parent_score = quick_score(parent_arr)
                if np.isfinite(parent_score):
                    best_seq = parent_arr.copy()
                    best_score = parent_score
                    print(f"parent length={parent_arr.size}, internal score={parent_score:.8f}")
            else:
                parent_arr = None
    except Exception:
        parent_arr = None

    if budget_s < 300.0:
        n_main = 2048
    elif budget_s < 1000.0:
        n_main = 4096
    else:
        n_main = 8192

    print(f"main sequence length n={n_main}")

    # Optional low-dimensional shape search.
    theta = find_param_shape(int(seed), total_deadline)
    if theta is not None:
        print(f"parameter shape search best={param_shape_score(theta):.8f}")

    cands = generate_candidates(n_main, int(seed))

    if parent_arr is not None:
        p_resized = resize_sequence(parent_arr, n_main)
        add_candidate(cands, np.log(np.maximum(p_resized, 1e-300)))

    if theta is not None:
        add_candidate(cands, param_x(theta, n_main))

    # Ensure at least one safe candidate exists.
    add_candidate(cands, np.zeros(n_main))

    scores = [quick_score(a) for a in cands]
    finite_count = sum(1 for s in scores if np.isfinite(s))
    finite_min = min([s for s in scores if np.isfinite(s)], default=float("inf"))
    print(f"initial candidates={len(cands)} finite={finite_count} best_initial={finite_min:.8f}")

    # Create log-domain mixtures among good initial candidates.
    if len(cands) >= 2 and time.monotonic() < total_deadline - 120.0:
        order = np.argsort(scores)
        top_idx = [int(i) for i in order[:10]]
        base_cands = cands
        for ii in range(len(top_idx)):
            for jj in range(ii + 1, len(top_idx)):
                a1 = base_cands[top_idx[ii]]
                a2 = base_cands[top_idx[jj]]
                for w in (0.25, 0.5, 0.75):
                    am = mix_candidates(a1, a2, w)
                    cands.append(am)
                    scores.append(quick_score(am))

    chosen = select_candidates(cands, scores, 10)
    chosen_scores = [quick_score(a) for a in chosen]
    print(f"selected {len(chosen)} candidates for deep refine")

    chosen_order = np.argsort(chosen_scores)
    for rank, idx in enumerate(chosen_order):
        if time.monotonic() > total_deadline - 60.0:
            break
        idx = int(idx)
        a0 = chosen[idx]
        print(
            f"refine {rank+1}/{len(chosen)} initial={chosen_scores[idx]:.8f} "
            f"t={time.monotonic()-start:.1f}s left={total_deadtime-margin:.1f}s".replace(
                "total_deadtime", "total_deadline"
            ).replace("margin", "")
        )
        res, res_score = refine_sequence(a0, total_deadline - 35.0, int(seed) + idx)
        if np.isfinite(res_score) and res_score < best_score:
            best_score = res_score
            best_seq = res.copy()
            print(f"new best internal={best_score:.10f} at t={time.monotonic()-start:.1f}s")

    if best_seq is None:
        best_seq = np.ones(n_main, dtype=np.float64) / float(n_main)
        best_score = 2.0

    # Additional final exact sharpening on the global best, if time remains.
    if time.monotonic() < total_deadline - 90.0:
        print("final exact active-set sharpening")
        best_seq, best_score = top_lag_refine(best_seq, 350, total_deadline - 60.0, int(seed) + 99991)
        print(f"after final sharpen internal={best_score:.10f}")

    seq = np.asarray(best_seq, dtype=np.float64)
    seq = np.nan_to_num(seq, nan=0.0, posinf=1000.0, neginf=0.0)
    seq = np.clip(seq, 0.0, 1000.0)

    s = float(seq.sum())
    if not np.isfinite(s) or s <= 0.01:
        seq = np.ones(n_main, dtype=np.float64) / float(n_main)

    # Avoid bool/integer output.
    out = [float(v) for v in seq]
    for i, v in enumerate(out):
        if not np.isfinite(v):
            out[i] = 0.0
        out[i] = max(0.0, min(1000.0, out[i]))

    print(f"final length={len(out)} internal best={best_score:.10f}")
    return out
```
