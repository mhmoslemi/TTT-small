    c_max = c.max()
    shifted = tau * (c - c_max)
    shifted -= shifted.max()
    exp_c = np.exp(shifted)
    weights = exp_c / exp_c.sum()
    
    # g[j] = 2 * sum_k weights[k] * a[k-j] (for valid k-j in [0,n-1])
    # Find significant weights to avoid O(n*(2n-1)) computation
    threshold = 1e-7
    sig = np.where(weights > threshold)[0]
    
    g = np.zeros(n)
    for k in sig:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * weights[k] * a[k - hi:k - lo + 1][::-1]
    return g

def _optimize_single(a0, n, deadline, rng, name, verbose=True):
    """Optimize a single initialization using multi-phase subgradient."""
    a = a0.copy()
    local_best = a.copy()
    
    # Compute initial score (scale-invariant, sum=1)
    c = np.convolve(a, a)
    local_best_score = 2.0 * n * c.max()
    
    if verbose:
        print(f"  [{name}] start: {local_best_score:.6f}")
    
    # Phase 1: Smooth optimization with log-sum-exp (decreasing tau = harder)
    for tau in [3.0, 5.0, 10.0, 20.0, 50.0]:
        if time.monotonic() > deadline:
            break
        
        lr = 2.0 / n
        v = np.zeros(n)
        momentum = 0.8
        stagnant = 0
        max_stagnant = 400
        
        for t in range(50000):
            if time.monotonic() > deadline:
                break
            
            c = np.convolve(a, a)
            g = _compute_lsx_gradient(a, c, tau, n)
            
            v = momentum * v + (1.0 - momentum) * g
            a = a - lr * v
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = a0.copy()
                break
            a = a / s
            lr *= 0.99995
            momentum = min(0.95, 0.8 + 0.00002 * t)
            
            if t % 200 == 0:
                c = np.convolve(a, a)
                score = 2.0 * n * c.max()
                if score < local_best_score - 1e-9:
                    local_best_score = score
                    local_best = a.copy()
                    stagnant = 0
                else:
                    stagnant += 200
                if stagnant >= max_stagnant:
                    break
    
    # Phase 2: Hard max subgradient polishing
    a = local_best.copy()
    c = np.convolve(a, a)
    local_best_score = 2.0 * n * c.max()
    
    npeaks_list = [5, 10, 20, 30]
    for npeaks in npeaks_list:
        if time.monotonic() > deadline:
            break
        
        lr = 1.0 / n
        v = np.zeros(n)
        momentum = 0.6
        stagnant = 0
        
        for t in range(20000):
            if time.monotonic() > deadline:
                break
            
            c = np.convolve(a, a)
            peaks = np.argpartition(c, -npeaks)[-npeaks:]
            g = _compute_subgradient(a, peaks, n)
            
            v = momentum * v + (1.0 - momentum) * g
            a = a - lr * v
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            lr *= 0.9999
            momentum = min(0.9, 0.6 + 0.00003 * t)
            
            if t % 300 == 0:
                c = np.convolve(a, a)
                score = 2.0 * n * c.max()
                if score < local_best_score - 1e-10:
                    local_best_score = score
                    local_best = a.copy()
                    stagnant = 0
                else:
                    stagnant += 300
                if stagnant >= 3000:
                    break
    
    # Phase 3: LP-based refinement (try a few times)
    a = local_best.copy()
    for lp_iter in range(5):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a)
        M = c.max()
        
        # LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0
        # Constraint matrix: row k has a[k-i] at position i (where valid)
        # This is a (2n-1) x n matrix
        # For efficiency, only include binding constraints (c[k] > 0.9*M)
        binding = np.where(c > 0.85 * M)[0]
        
        if len(binding) == 0:
            break
        
        # Build constraint matrix for binding constraints only
        # A_eq structure: for each binding constraint k, A[k_buf, i] = a[k-i] (if valid)
        rows = []
        for k in binding:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            row = np.zeros(n)
            row[lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
            rows.append(row)
        
        A_ub = np.array(rows)
        b_ub = np.full(len(binding), M)
        
        # Maximize sum(b) = minimize -sum(b)
        c_obj = -np.ones(n)
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                          method='highs', options={'presolve': True, 'time_limit': 10})
            if res.success and res.x is not None:
                b = res.x
                b = np.maximum(b, 0)
                b_sum = b.sum()
                if b_sum > 1e-10:
                    b = b / b_sum
                    # Mix: a_new = (1-t)*a + t*b
                    for t_mix in [0.1, 0.2, 0.3, 0.5]:
                        a_try = (1 - t_mix) * a + t_mix * b
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / a_try.sum()
                        c_try = np.convolve(a_try, a_try)
                        score_try = 2.0 * n * c_try.max()
                        if score_try < local_best_score - 1e-10:
                            local_best_score = score_try
                            local_best = a_try.copy()
                            a = a_try.copy()
        except Exception:
            pass
    
    # Final score with actual evaluator
    final_score = float(evaluate_sequence(local_best.tolist()))
    
    if verbose:
        print(f"  [{name}] final: {final_score:.6f}")
    
    return final_score, local_best

def _generate_inits(n, parent, rng):
    """Generate diverse initializations, all normalized to sum=1."""
    inits = []
    
    # Current parent
    if parent.sum() > 0:
        inits.append(("parent", parent / parent.sum()))
    
    # Cosine windows
    for freq in [1, 2, 3, 4, 5, 7, 10, 13, 17, 21, 30, 50]:
        x = np.arange(n)
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.clip(a, 0.01, None)
        inits.append((f"cos{freq}", a / a.sum()))
    
    # Random
    for i in range(15):
        a = rng.random(n) + 0.01
        inits.append((f"rand{i}", a / a.sum()))
    
    # Block patterns (alternating on/off)
    for nb in [4, 6, 8, 12, 16, 24, 32, 48, 64]:
        a = np.zeros(n)
        block = max(1, n // nb)
        for i in range(0, n, 2 * block):
            a[i:min(i + block, n)] = 1.0
        if a.sum() > 0:
            inits.append((f"blk{nb}", a / a.sum()))
    
    # Multi-frequency cosine
    for f1, f2 in [(1, 3), (1, 5), (2, 7), (3, 11), (1, 2), (5, 13)]:
        x = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f1 * x / n) + 0.3 * np.cos(2 * np.pi * f2 * x / n)
        a = np.clip(a, 0.01, None)
        inits.append((f"cos_{f1}_{f2}", a / a.sum()))
    
    # Gaussian bumps
    x = np.linspace(0, 1, n)
    for center in [0.25, 0.5, 0.75]:
        for width in [0.05, 0.1, 0.15, 0.25, 0.4]:
            a = np.exp(-0.5 * ((x - center) / width) ** 2)
            a = np.clip(a, 1e-4, None)
            inits.append((f"gau_{center}_{width}", a / a.sum()))
    
    # Linear ramps
    a = np.linspace(0.01, 1.0, n)
    inits.append(("ramp_up", a / a.sum()))
    a = np.linspace(1.0, 0.01, n)
    inits.append(("ramp_down", a / a.sum()))
    
    # Triangular (tent)
    a = 1.0 - 2.0 * np.abs(np.linspace(-0.5, 0.5, n))
    a = np.clip(a, 0.01, None)
    inits.append(("tent", a / a.sum()))
    
    # Semicircle
    x = np.linspace(-1, 1, n)
    a = np.sqrt(np.clip(1 - x**2, 0, None)) + 0.01
    inits.append(("semi", a / a.sum()))
    
    # Raised cosine (Hann window)
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    a = np.clip(a, 0.01, None)
    inits.append(("hann", a / a.sum()))
    
    # Blackman window
    x = np.arange(n)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1))
    a = np.clip(a, 0.01, None)
    inits.append(("blackman", a / a.sum()))
    
    # Chirp-like (quadratic phase in frequency domain -> but we work in time domain)
    # Use a sequence that approximates a "flat spectrum" in time domain
    # a[i] = 1 + eps * cos(2*pi*i^2/N) (chirp)
    for eps in [0.3, 0.5, 0.8]:
        x = np.arange(n)
        a = 1.0 + eps * np.cos(2 * np.pi * x**2 / (2 * n))
        a = np.clip(a, 0.01, None)
        inits.append((f"chirp{eps}", a / a.sum()))
    
    return inits

def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-start subgradient optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    
    Uses:
    - Diverse initializations (cosine, random, block, window functions)
    - Log-sum-exp smoothing for soft optimization
    - Hard max subgradient for polishing
    - LP-based direction finding for escape from local minima
    """
    deadline = time.monotonic() + budget_s - 10.0
    rng = np.random.default_rng(seed)
    
    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    
    if parent is None or parent.size < 10 or not np.all(np.isfinite(parent)):
        parent = np.ones(256, dtype=np.float64)
    
    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size
    
    # Initial best
    best_score = float(evaluate_sequence(parent.tolist()))
    best_a = parent / (parent.sum() + 1e-30)
    print(f"n={n}, initial score={best_score:.6f}")
    print(f"Time budget: {budget_s-10:.0f}s")
    
    # Generate initializations
    inits = _generate_inits(n, parent, rng)
    print(f"Generated {len(inits)} initializations")
    
    # Sort initializations: put promising ones first
    # Quick evaluation of each init (just compute the score)
    init_scores = []
    for name, a0 in inits:
        c = np.convolve(a0, a0)
        score = 2.0 * n * c.max()
        init_scores.append((score, name, a0))
    
    init_scores.sort(key=lambda x: x[0])
    
    # Optimize in order of initial score (best first), with time management
    # Allocate more time to better initializations
    results = []
    total_time = budget_s - 10 - 30  # reserve time
    n_inits = len(init_scores)
    
    # Give top initializations more time
    time_weights = np.zeros(n_inits)
    for i in range(n_inits):
        time_weights[i] = max(0.1, 1.0 - 0.05 * i)
    time_weights /= time_weights.sum()
    
    cumulative_deadline = time.monotonic()
    for i, (init_score, name, a0) in enumerate(init_scores):
        if time.monotonic() > deadline:
            break
        
        # Skip if initial score is already too bad (waste of time)
        if init_score > best_score * 1.5 and i > 0:
            continue
        
        # Allocate time for this init
        alloc_time = time_weights[i] * total_time * (1.0 - i / n_inits * 0.5)
        alloc_time = max(10, min(alloc_time, deadline - time.monotonic()))
        init_deadline = time.monotonic() + alloc_time
        
        score, result_a = _optimize_single(a0, n, init_deadline, rng, name, verbose=True)
        results.append((score, name, result_a))
        
        if score < best_score:
            best_score = score
            best_a = result_a
            print(f"==> New best: {best_score:.6f} (from {name})")
    
    # If we have time left, do additional optimization on the best
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\nRefining best solution, {remaining:.0f}s remaining...")
        score, result_a = _optimize_single(best_a, n, deadline, rng, "refine", verbose=True)
        if score < best_score:
            best_score = score
            best_a = result_a
            print(f"==> Refined best: {best_score:.6f}")
    
    # One more try with different strategy: perturbation of best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Perturb best and re-optimize
        for attempt in range(3):
            if time.monotonic() > deadline:
                break
            noise = rng.normal(0, 0.05, n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            
            sub_deadline = time.monotonic() + min(remaining / 3, 60)
            score, result_a = _optimize_single(a_pert, n, sub_deadline, rng, f"perturb{attempt}", verbose=False)
            if score < best_score:
                best_score = score
                best_a = result_a
                print(f"==> Perturbed best: {best_score:.6f}")
    
    print(f"\nFinal score: {best_score:.6f}")
    return [float(max(0.0, x)) for x in best_a]
