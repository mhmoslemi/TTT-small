"""
Minimize  2*n*max(a*a) / (sum a)^2  for a non-negative step function a.

The objective is scale invariant, so we optimize over the simplex of shapes
using a softmax parameterization  a = softmax(x)  and projected (momentum)
gradient descent on a subgradient of the max-convolution term.  The gradient
of the active lag k* is  d(2n*(a*a)[k*])/dx_j = 4n * a_j * (a[k*-j] - c*),
which is cheap (one full convolution + an O(n) gather) to evaluate.

We run several async initializations (geometric ramps, power ramps, smooth
asymmetric bells) with momentum + backtracking, periodic jitter to escape
local minima, and keep the global best by the exact evaluator.  Symmetric
shapes are provably >= 2, so all starts are deliberately asymmetric.
"""

import numpy as np
import time


# --------------------------------------------------------------------------
# helpers
# --------------------------------------------------------------------------
def _softmax(x):
    x = np.asarray(x, dtype=np.float64)
    x = x - np.max(x)
    e = np.exp(x)
    s = e.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.ones_like(x) / x.size
    return e / s


def _conv_full(a):
    return np.convolve(a, a, mode="full")


def _local_score(a):
    """Exact scale-invariant score (sum normalized to 1 inside)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = a.sum()
    if s <= 1e-12:
        return np.inf
    c = _conv_full(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_and_grad(x):
    """Return (score, grad_x) for the softmax parameterization (sum a = 1)."""
    a = _softmax(x)
    n = a.size
    c = _conv_full(a)
    k = int(np.argmax(c))
    cstar = c[k]
    score = 2.0 * n * cstar
    # gather a[k - i] for valid i
    idx = k - np.arange(n, dtype=np.int64)
    valid = (idx >= 0) & (idx < n)
    aj_k = np.zeros(n, dtype=np.float64)
    aj_k[valid] = a[idx[valid]]
    grad_x = 4.0 * n * a * (aj_k - cstar)
    return score, grad_x


# --- initializers (all asymmetric so they can beat the symmetric bound 2) ---
def _init_geometric(n, c_param):
    r = 1.0 - c_param / n
    if not (0.0 < r < 1.0):
        r = 0.999
    a = r ** np.arange(n, dtype=np.float64)
    a = a / a.sum()
    return np.log(a + 1e-300)


def _init_power_ramp(n, p):
    a = (np.arange(n, dtype=np.float64) + 1.0) ** p
    a = a / a.sum()
    return np.log(a + 1e-300)


def _init_smooth_bell(n, alpha, beta):
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = np.sin(np.pi * np.clip(t, 0.0, 1.0)) ** alpha * np.exp(-beta * t)
    a = np.clip(a, 1e-12, None)
    a = a / a.sum()
    return np.log(a)


# --------------------------------------------------------------------------
# single start optimization
# --------------------------------------------------------------------------
def _optimize_run(x0, rng, deadline, label):
    x = np.array(x0, dtype=np.float64)
    best_x = x.copy()
    best_score = _local_score(_softmax(x))
    print("%s start score=%.6f" % (label, best_score), flush=True)

    v = np.zeros_like(x)
    beta = 0.90
    eta = 0.5
    eta_min = 1e-4
    it = 0

    while time.monotonic() < deadline:
        score, g = _score_and_grad(x)
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_x = x.copy()

        # momentum
        v = beta * v + (1.0 - beta) * g
        v = v - v.mean()              # softmax is shift-invariant; center v

        # backtracking line search
        accepted = False
        step = eta
        trial_x = x - step * v
        tr_score = _local_score(_softmax(trial_x))
        stalls = 0
        while step > eta_min and (not np.isfinite(tr_score) or tr_score >= score - 1e-12):
            step *= 0.5
            if step <= eta_min:
                break
            trial_x = x - step * v
            tr_score = _local_score(_softmax(trial_x))
            stalls += 1
            if stalls > 12:
                break

        if np.isfinite(tr_score) and tr_score < score - 1e-12:
            x = trial_x
            eta = min(eta * 1.10, 5.0)
            it += 1
        else:
            eta = max(eta * 0.5, eta_min)
            # periodic jitter to escape the current basin
            if it % 80 == 79 and it > 200:
                x = x + rng.normal(0.0, 0.05, x.size)
            it += 1

    # make sure best_x corresponds to a finite score
    s = _local_score(_softmax(best_x))
    if not np.isfinite(s):
        best_x = x
        s = _local_score(_softmax(x))
    print("%s best score=%.6f (iters=%d)" % (label, s, it), flush=True)
    return best_x, s

    t_start = time.monotonic()

            n = int(len(height_sequence_1))
    except Exception:
        n = None
    if n is None or n < 16:
        n = 4498
    n = int(min(n, 20000))   # safety cap

    margin = max(5.0, 0.02 * budget_s)
    search_deadline = t_start + max(1.0, float(budget_s) - margin)

    starts = []

    # 1) state-of-the-art parent, made into an x-parameter (if usable)

                starts.append(np.log(p / p.sum() + 1e-300))
    except Exception:
        pass

    # 2) geometric ramps with several decay constants
    for c_param in (0.6, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0):
        starts.append(_init_geometric(n, c_param))

        starts.append(_init_power_ramp(n, p))

    # de-duplicate-ish by shuffling deterministically

    starts = [starts[i] for i in order]

            if np.isfinite(ss) and ss < best_score:
                best_score = ss
                best_x = sx.copy()
        except Exception as e:
            print("polish error: %s" % e, flush=True)

    a = _softmax(best_x)
    a = np.clip(a, 0.0, 1000.0)
    out = [float(v) for v in a]

    except Exception as e:
        print("final evaluate_sequence error: %s" % e, flush=True)

    return out
