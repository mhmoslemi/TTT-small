import cvxpy as cp

def evaluate_convolution(a):
    """Compute the objective value."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)

def lp_refine(a, M=None, t=0.1):
    """Solve the LP to find a better sequence."""
    n = len(a)
    if M is None:
        b = np.convolve(a, a)
        M = np.max(b)
    
    # LP: maximize sum(b) subject to a*b <= M (element-wise) and b >= 0
    b_var = cp.Variable(n, nonneg=True)

    # (a*b)_k = sum_j a_j * b_{k-j} for valid j
    # We need to set up the constraint matrix

    # (a*b)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_j * b_{k-j}

    # sum_j A[k,j] * b_j <= M where A[k,j] = a_{k-j} if 0<=k-j<n, else 0
    
    # This matrix is banded (each row has at most n non-zeros, total nnz = sum of min(k+1, n, 2n-1-k) for k=0..2n-2)
    
    # For n=3809, this is about 7617 rows and 3809 cols with ~3809*3809/2 ≈ 7.3M non-zeros
    # This might be too large for a dense LP but fine for sparse
    
    # Let me use scipy for the sparse matrix
    from scipy.sparse import csr_matrix, diags

    # Row k: A[k, j] = a[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve()
    except:
        return a.copy(), M
    
    if b_var.value is None:
        return a.copy(), M
    
    b_opt = np.clip(b_var.value, 0, None)
    sum_b = np.sum(b_opt)
    sum_a = np.sum(a)
    
    if sum_b < 0.01:
        return a.copy(), M

    b_scaled = b_opt * (sum_a / sum_b)
