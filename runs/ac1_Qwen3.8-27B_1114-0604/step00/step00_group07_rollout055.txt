    K=_lf_k
    if cmax <=0:

    shifted=K*(c-cmax)
    if shifted.max()>0: shifted=shifted-shifted.max()
    e=np.exp(np.minimum(shifted,0.0))
    s=float(np.sum(e))

    else:

_LB_L = 1
_LB_REF = 0
_LB_DEADLINE = 0.0


def _fft_len_for_n(n):
    m = max(1, 2 * int(n) - 1)
    L = 1
    while L < m:
        L <<= 1
    return L


def _softmax(v):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return np.empty(0, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, -1e300)
    m = float(np.max(v))
    if not np.isfinite(m):
        return np.full(v.size, 1.0 / v.size, dtype=np.float64)
    e = np.exp(v - m)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(v.size, 1.0 / v.size, dtype=np.float64)
    return e / s


def _normalize_nonnegative(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-15 or not np.isfinite(s):
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def _regularize_a(a, eps=1e-12):
    a = _normalize_nonnegative(a)
    if a.size == 0:
        return a
    if eps <= 0.0:
        return a
    a = (1.0 - eps) * a + (eps / float(a.size))
    s = float(np.sum(a))
    if s <= 1e-15 or not np.isfinite(s):
        return a
    return a / s


def _make_central_unnormalized(n):
    r = np.empty(int(n), dtype=np.float64)
    if n <= 0:
        return r
    r[0] = 1.0
    for k in range(int(n) - 1):
        r[k + 1] = r[k] * (2.0 * k + 1.0) / (2.0 * k + 2.0)
    return r


def _make_central(n):
    return _normalize_nonnegative(_make_central_unnormalized(n))


def _variant_gamma(r, gamma):
    r = np.asarray(r, dtype=np.float64)
    r = np.where(r > 0.0, r, 0.0)
    a = np.power(r, float(gamma))
    return _normalize_nonnegative(a)


def _variant_add(r, add):
    r = np.asarray(r, dtype=np.float64)
    return _normalize_nonnegative(r + float(add))


def _variant_tilt(a, tilt):
    a = np.asarray(a, dtype=np.float64)
    if a.size <= 1:
        return _normalize_nonnegative(a)
    w = np.exp(float(tilt) * np.linspace(0.0, 1.0, a.size))
    return _normalize_nonnegative(a * w)


def _fast_score(a, L=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n <= 0:
        return np.inf
    s = float(np.sum(a))
    if s <= 1e-15 or not np.isfinite(s):
        return np.inf
    if L is None:
        L = _fft_len_for_n(n)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa)[: 2 * n - 1].real
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax):
        return np.inf
    return float(2.0 * n * cmax / (s * s))


def _conv_and_grad(a, K, L=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n <= 0:
        return np.empty(0), np.empty(0), np.inf
    if L is None:
        L = _fft_len_for_n(n)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa)[: 2 * n - 1].real
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    cmax = float(np.max(c)) if c.size else 0.0
    if cmax <= 0.0 or not np.isfinite(cmax):
        return c, np.zeros(n, dtype=np.float64), np.inf

    K = float(K)
    if not np.isfinite(K) or K <= 0.0:
        p = np.ones_like(c, dtype=np.float64) / float(c.size)
    else:
        shifted = K * (c - cmax)
        shifted = np.where(np.isfinite(shifted), shifted, -1e300)
        mx = float(np.max(shifted))
        if np.isfinite(mx) and mx > 0.0:
            shifted = shifted - mx
        e = np.exp(np.minimum(shifted, 0.0))
        s = float(np.sum(e))
        if s <= 0.0 or not np.isfinite(s):
            p = np.zeros_like(c, dtype=np.float64)
            p[int(np.argmax(c))] = 1.0
        else:
            p = e / s

    fp = np.fft.rfft(p, L)
    corr = np.fft.irfft(fp * np.conj(fa))[:n].real
    corr = np.where(np.isfinite(corr) & (corr > 0.0), corr, 0.0)
    ga = 2.0 * corr
    return c, ga, float(2.0 * n * cmax)


def _a_from_u_global(x):
    n = int(_LB_N)
    u = np.asarray(x, dtype=np.float64).ravel()
    if u.size != n - 1:
        tmp = np.zeros(n - 1, dtype=np.float64)
        m = min(u.size, n - 1)
        if m > 0:
            tmp[:m] = u[:m]
        u = tmp
    ref = int(_LB_REF)
    if ref < 0:
        ref = 0
    if ref > n - 1:
        ref = n - 1

    z = np.empty(n, dtype=np.float64)
    if ref > 0:
        z[:ref] = u[:ref]
    if ref < n - 1:
        z[ref + 1:] = u[ref:]
    z[ref] = 0.0
    return _softmax(z)


def _u_from_a_reference(a, ref):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n <= 1:
        return np.empty(0, dtype=np.float64)
    ref = int(ref)
    if ref < 0:
        ref = 0
    if ref > n - 1:
        ref = n - 1
    base = float(a[ref])
    if base <= 1e-300 or not np.isfinite(base):
        base = 1e-300
    z = np.maximum(a, 1e-300) / base
    u = np.log(z)
    out = np.empty(n - 1, dtype=np.float64)
    if ref > 0:
        out[:ref] = u[:ref]
    if ref < n - 1:
        out[ref:] = u[ref + 1:]
    return out


def _grad_full_to_u(gfull, ref):
    n = int(gfull.size)
    if n <= 1:
        return np.empty(0, dtype=np.float64)
    ref = int(ref)
    if ref < 0:
        ref = 0
    if ref > n - 1:
        ref = n - 1
    out = np.empty(n - 1, dtype=np.float64)
    if ref > 0:
        out[:ref] = gfull[:ref]
    if ref < n - 1:
        out[ref:] = gfull[ref + 1:]
    return out


def _lbfgs_objfun(x):
    n = int(_LB_N)
    a = _a_from_u_global(x)
    if a.size != n:
        return 1e100, np.zeros(max(0, n - 1), dtype=np.float64)

    L = int(_LB_L)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa)[: 2 * n - 1].real
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    cmax = float(np.max(c)) if c.size else 0.0
    if cmax <= 0.0 or not np.isfinite(cmax):
        return 0.0, np.zeros(max(0, n - 1), dtype=np.float64)

    K = float(_LB_K)
    if not np.isfinite(K) or K <= 0.0:
        K = 1.0
    shifted = K * (c - cmax)
    shifted = np.where(np.isfinite(shifted), shifted, -1e300)
    mx = float(np.max(shifted))
    if np.isfinite(mx) and mx > 0.0:
        shifted = shifted - mx
    e = np.exp(np.minimum(shifted, 0.0))
    s = float(np.sum(e))

    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros_like(c, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
    else:
        p = e / s

    F = cmax + float(np.log(max(s, 1e-300))) / K

    fp = np.fft.rfft(p, L)
    corr = np.fft.irfft(fp * np.conj(fa))[:n].real
    corr = np.where(np.isfinite(corr) & (corr > 0.0), corr, 0.0)
    ga = 2.0 * corr

    dot = float(np.dot(a, ga))
    grad_full = (2.0 * n) * a * (ga - dot)
    grad_u = _grad_full_to_u(grad_full, int(_LB_REF))
    grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)

    val = float(2.0 * n * F)
    if not np.isfinite(val):
        val = 1e100
    return val, grad_u


def _lbfgs_callback(xk):
    return time.monotonic() > _LB_DEADLINE


def _run_lbfgs_phases(a, score, deadline):
    if _scipy_opt is None:
        return a, score

    n = int(a.size)
    L = _fft_len_for_n(n)
    _LB_N = n
    _LB_L = L
    best = _normalize_nonnegative(a)
    best_score = float(score)

    phases = (
        (5.0, 800),
        (15.0, 600),
        (50.0, 500),
        (150.0, 400),
        (400.0, 300),
        (1000.0, 250),
        (2500.0, 150),
        (6000.0, 100),
    )

    for beta, maxiter in phases:
        if time.monotonic() >= deadline:
            break
        a_in = _regularize_a(best, 1e-12)
        ref = int(np.argmax(a_in))
        u = _u_from_a_reference(a_in, ref)
        _LB_REF = ref
        _LB_K = float(beta) * float(n)
        _LB_DEADLINE = float(deadline)

        options = {
            "maxiter": int(maxiter),
            "maxfun": int(max(10, 20 * maxiter)),
            "ftol": 1e-12,
            "gtol": 1e-6,
            "maxls": 50,
        }
        try:
            res = _scipy_opt.minimize(
                _lbfgs_objfun,
                u,
                method="L-BFGS-B",
                jac=True,
                options=options,
                callback=_lbfgs_callback,
            )
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if not np.all(np.isfinite(x)):
                x = u
            a_trial = _a_from_u_global(x)
            a_trial = _normalize_nonnegative(a_trial)
            sc = _fast_score(a_trial, L)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best = a_trial
                best_score = sc
                print(f"LBFGS beta={beta:.1f} improved score={best_score:.10f}")
        except Exception:
            pass

    return best, best_score


def _solve_lp(A, b_ub, n, max_time):
    if _scipy_opt is None:
        return None
    c = -np.ones(n, dtype=np.float64)
    bounds = (0.0, 10.0)
    try:
        opts = {"presolve": True, "time_limit": float(max(1.0, max_time))}
        return _scipy_opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options=opts)
    except Exception:
        pass
    try:
        return _scipy_opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    except Exception:
        return None


def _lp_search(a, score, deadline, max_constraints=500, max_iter=200):
    if _scipy_sparse is None or _scipy_opt is None:
        return a, score

    n = int(a.size)
    L = _fft_len_for_n(n)
    best = _normalize_nonnegative(a)
    best_score = float(score)
    max_constraints = int(max(50, max_constraints))

    for it in range(int(max_iter)):
        if time.monotonic() > deadline:
            break

        a_in = _regularize_a(best, 1e-12)
        fa = np.fft.rfft(a_in, L)
        c = np.fft.irfft(fa * fa)[: 2 * n - 1].real
        c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
        M = float(np.max(c)) if c.size else 0.0
        if M <= 0.0 or not np.isfinite(M):
            break

        cur_score = float(2.0 * n * M)
        if cur_score < best_score - 1e-12:
            best = a_in
            best_score = cur_score
            print(f"LP iteration {it} current score={best_score:.10f}")

        argmax = int(np.argmax(c))
        active = np.where(c >= M - 1e-14 * (M + 1.0))[0]
        if active.size == 0:
            active = np.array([argmax], dtype=int)

        if active.size > max_constraints:
            if max_constraints > 1:
                idxs = np.unique(np.linspace(0, active.size - 1, max_constraints - 1).round().astype(int))
                sampled = active[idxs]
            else:
                sampled = np.empty(0, dtype=int)
            active = np.unique(np.concatenate((sampled, np.array([argmax], dtype=int))))
            if active.size > max_constraints:
                active = active[:max_constraints]
            active = active.astype(int, copy=False)
            if active.size == 0 or active[0] != argmax:
                active[0] = argmax

        m = int(active.size)
        scale = 1.0 / M
        row_parts = []
        col_parts = []
        val_parts = []

        for idx, k in enumerate(active.tolist()):
            k = int(k)
            js = np.arange(max(0, k - (n - 1)), min(n - 1, k) + 1, dtype=np.int64)
            if js.size == 0:
                continue
            vals = (a_in[k - js] * scale).astype(np.float64, copy=False)
            row_parts.append(np.full(js.size, idx, dtype=np.int32))
            col_parts.append(js.astype(np.int32, copy=False))
            val_parts.append(vals)

        if not row_parts:
            continue

        rows = np.concatenate(row_parts)
        cols = np.concatenate(col_parts)
        vals = np.concatenate(val_parts)
        A = _scipy_sparse.csr_matrix((vals, (rows, cols)), shape=(m, n))

        remain = max(1.0, float(deadline - time.monotonic()))
        per_lp = max(5.0, min(30.0, remain / 4.0))
        res = _solve_lp(A, np.ones(m, dtype=np.float64), n, per_lp)
        if res is None or getattr(res, "x", None) is None:
            continue

        b = np.asarray(res.x, dtype=np.float64).ravel()
        b = np.where(np.isfinite(b) & (b > 0.0), b, 0.0)
        B = float(np.sum(b))
        if B <= 1.0 + 1e-8 or not np.isfinite(B):
            continue

        g = b / B
        g = _normalize_nonnegative(g)

        accepted = False
        for t in (0.5, 0.25, 0.12, 0.05, 0.02):
            trial = (1.0 - t) * best + t * g
            trial = _normalize_nonnegative(trial)
            sc = _fast_score(trial, L)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best = trial
                best_score = sc
                print(f"LP iter={it} accepted t={t:.3f} score={best_score:.10f}")
                accepted = True
                break

        if not accepted:
            # If no useful move is found, let the next phase handle refinement.
            continue

    return best, best_score


def _mirror_search(a, score, deadline):
    n = int(a.size)
    if n <= 0:
        return a, score
    L = _fft_len_for_n(n)
    best = _normalize_nonnegative(a)
    best_score = float(score)

    loga = np.log(best + 1e-300)
    loga -= float(np.max(loga))

    beta = 20.0
    K = beta * float(n)
    alpha = 0.10
    stalls = 0
    it = 0

    while time.monotonic() < deadline - 5.0:
        cur = best
        if it % 100 == 0:
            beta = min(50000.0, beta * 1.25)
            K = beta * float(n)

        _c, ga, scur = _conv_and_grad(cur, K, L)
        if not np.isfinite(scur):
            break

        if scur < best_score - 1e-12:
            best = cur
            best_score = scur
            stalls = 0

        gmax = float(np.max(ga)) if ga.size else 0.0
        if gmax <= 0.0 or not np.isfinite(gmax):
            alpha *= 0.8
            it += 1
            continue

        g = ga / gmax
        g = g - float(np.mean(g))

        log_trial = loga - alpha * g
        log_trial -= float(np.max(log_trial))
        cand = _softmax(log_trial)
        sc = _fast_score(cand, L)

        if np.isfinite(sc) and sc < best_score - 1e-12:
            best = cand
            best_score = sc
            cur = cand
            loga = log_trial
            alpha = min(0.50, alpha * 1.05)
            stalls = 0
        elif np.isfinite(sc) and sc < scur - 1e-12:
            cur = cand
            loga = log_trial
            stalls = 0
        else:
            stalls += 1
            alpha = max(0.005, alpha * 0.8)
            if stalls > 25:
                cur = best.copy()
                loga = np.log(cur + 1e-300)
                loga -= float(np.max(loga))
                alpha = 0.05
                stalls = 0

        it += 1

    return best, best_score


def _random_polish(a, score, deadline):
    n = int(a.size)
    if n <= 0:
        return a, score
    L = _fft_len_for_n(n)
    best = _normalize_nonnegative(a)
    best_score = float(score)
    rng = np.random.default_rng(12345)
    fails = 0
    sig = 0.08

    min_w = max(8, n // 1000)
    max_w = max(min_w + 2, n // 60)

    while time.monotonic() < deadline - 5.0:
        cand = best.copy()
        width = int(rng.integers(min_w, max_w + 1))
        if width >= n:
            width = max(1, n // 4)
        start = int(rng.integers(0, max(1, n - width + 1)))
        cand[start:start + width] *= np.exp(rng.normal(0.0, sig, size=width))

        if rng.random() < 0.25:
            tilt = 0.05 * float(rng.normal())
            w = np.exp(tilt * np.linspace(0.0, 1.0, n))
            cand *= w

        cand = _normalize_nonnegative(cand)
        sc = _fast_score(cand, L)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best = cand
            best_score = sc
            fails = 0
        else:
            fails += 1
            if fails % 120 == 0:
                sig = min(0.30, sig * 1.15)
                if fails % 480 == 0:
                    sig = 0.08

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for nonnegative coefficients minimizing 2*n*max(a*a)/sum(a)^2.

    Uses a structured central-binomial start, LP-based conditional-gradient
    moves, smooth log-sum-exp L-BFGS refinement, and light local polishing.
    """
    t0 = time.monotonic()
    total = max(30.0, float(budget_s))
    deadline = t0 + max(5.0, total - 10.0)

    N = 6733
    parent = None
    try:
        hs = height_sequence_1  # noqa: F821
    except Exception:
        hs = None

    if hs is not None:
        arr = np.asarray(hs, dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size > 0:
            N = int(arr.size)
            parent = arr

    if total < 120.0:
        N = min(N, 1200)
    if total < 300.0:
        N = min(N, 4000)
    N = max(32, min(N, 20000))

    if parent is not None and parent.size != N:
        if parent.size > N:
            parent = parent[:N]
        else:
            tmp = np.zeros(N, dtype=np.float64)
            mp = min(parent.size, N)
            tmp[:mp] = parent[:mp]
            parent = tmp

    r = _make_central_unnormalized(N)
    candidates = []
    candidates.append(_normalize_nonnegative(r))
    candidates.append(_normalize_nonnegative(r[::-1]))
    for gamma in (0.95, 1.05, 1.15):
        candidates.append(_variant_gamma(r, gamma))
    for add in (1e-3, 5e-3, 2e-2):
        candidates.append(_variant_add(r, add))
    for tilt in (0.02, -0.02):
        candidates.append(_variant_tilt(_normalize_nonnegative(r), tilt))
    if parent is not None:
        candidates.append(_normalize_nonnegative(parent))

    best = None
    best_score = np.inf
    for cand in candidates:
        a = _normalize_nonnegative(cand)
        if a.size != N:
            continue
        sc = _fast_score(a)
        if np.isfinite(sc) and sc < best_score:
            best = a
            best_score = sc

    if best is None:
        best = _make_central(N)
        best_score = _fast_score(best)

    print(f"initial N={N} fast_score={best_score:.10f} time={time.monotonic()-t0:.2f}")

    if total < 120.0:
        lp1_deadline = t0 + max(5.0, total * 0.15)
        lb_deadline = t0 + max(lp1_deadline + 5.0, total * 0.85)
        lp2_deadline = lb_deadline
    else:
        lp1_deadline = t0 + max(30.0, min(360.0, 0.15 * total))
        lb_deadline = t0 + max(lp1_deadline + 60.0, 0.80 * total)
        lp2_deadline = t0 + max(lb_deadline + 30.0, min(total - 120.0, lb_deadline + 0.15 * total))

    lp1_deadline = min(lp1_deadline, deadline)
    lb_deadline = min(lb_deadline, deadline)
    lp2_deadline = min(lp2_deadline, deadline)
    if lb_deadline < lp1_deadline:
        lb_deadline = lp1_deadline
    if lp2_deadline < lb_deadline:
        lp2_deadline = lb_deadline

    best, best_score = _lp_search(
        best,
        best_score,
        lp1_deadline,
        max_constraints=500,
        max_iter=200,
    )
    print(f"after LP1 score={best_score:.10f} time={time.monotonic()-t0:.2f}")

    if time.monotonic() < lb_deadline:
        best, best_score = _run_lbfgs_phases(best, best_score, lb_deadline)
    print(f"after LBFGS score={best_score:.10f} time={time.monotonic()-t0:.2f}")

    if time.monotonic() < min(lp2_deadline, deadline) - 20.0:
        best, best_score = _lp_search(
            best,
            best_score,
            lp2_deadline,
            max_constraints=700,
            max_iter=120,
        )
    print(f"after LP2 score={best_score:.10f} time={time.monotonic()-t0:.2f}")

    mirror_deadline = time.monotonic() + max(30.0, min(240.0, (deadline - time.monotonic()) * 0.55))
    mirror_deadline = min(mirror_deadline, deadline - 10.0)
    if time.monotonic() < mirror_deadline:
        best, best_score = _mirror_search(best, best_score, mirror_deadline)
    print(f"after mirror score={best_score:.10f} time={time.monotonic()-t0:.2f}")

    if time.monotonic() < deadline - 5.0:
        best, best_score = _random_polish(best, best_score, deadline)
    print(f"after polish score={best_score:.10f} time={time.monotonic()-t0:.2f}")

    best = _normalize_nonnegative(best)

    try:
        real_score = evaluate_sequence([float(x) for x in best])  # noqa: F821
        if not np.isfinite(real_score):
            best = _make_central(N)
            best_score = _fast_score(best)
        print(f"final fast_score={best_score:.10f} eval_score={float(real_score):.10f}")
    except Exception:
        pass

    return [float(x) for x in np.asarray(best, dtype=np.float64)]
