```python
import numpy as np
from scipy.optimize import linprog
import time

def solve_lp(f, M):
    """Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    # c[k] = sum_i f[i]*g[k-i] for k = 0, ..., 2n-2
    # We need c[k] <= M for all k.
    # Build the constraint matrix A where A[k, i] = f[i] if k-i is in [0, n-1], else 0.
    # Actually, c[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[i]*g[k-i]
    # Let j = k-i, so i = k-j. c[k] = sum_j f[k-j]*g[j] for valid j.
    # So A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    
    # A is (2n-1) x n.
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    
    # Objective: maximize sum(g) = -minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: A @ g <= M (vector of M's)
    M_vec = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=M_vec, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x
```

"""
LP-based iterative optimization for minimizing the step function evaluation.
Uses the linear programming approach from the state of the art, with
multiple initial sequences and adaptive step sizes.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def build_conv_matrix(f):
    """Build sparse convolution matrix A such that A @ g = convolve(f, g)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            vals.append(f[i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp(f, M):
    """Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            return None
        return res.x
    except:
        return None

def lp_iterate(f, t, max_iters=100, deadline=None):
    """Run LP iterations starting from f."""
    n = len(f)
    best_f = f.copy()
    best_E = float('inf')
    f = f.copy()
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
            
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        S = float(np.sum(f))
        if S < 1e-10:
            break
        E = 2.0 * n * M / (S * S)
        if E < best_E:
            best_E = E
            best_f = f.copy()
        
        g = solve_lp(f, M)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            break
        
        # Rescale g to match f's sum
        g = g * (S / sum_g)
        
        # Update with step size t
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        
        # Check if progress was made
        new_conv = np.convolve(f_new, f_new)
        new_M = float(np.max(new_conv))
        new_S = float(np.sum(f_new))
        new_E = 2.0 * n * new_M / (new_S * new_S)
        
        if new_E < E:
            f = f_new
        else:
            # Try smaller step
            t = t * 0.5
            if t < 1e-6:
                break
    
    # Final evaluation
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S > 0:
        E = 2.0 * n * M / (S * S)
        if E < best_E:
            best_E = E
            best_f = f.copy()
    
    return best_f, best_E
