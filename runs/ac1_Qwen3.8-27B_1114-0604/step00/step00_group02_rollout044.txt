import numpy as np
import time

"""
Optimizes non-negative sequences to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradient descent with anti-palindromic initialization,
momentum, adaptive step size, and random restarts.
Includes optional LP-based refinement via the cross-correlation constraint.
"""

def _fft_convolve(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _score_val(a):
    """Compute the evaluation score (same formula as evaluate_sequence)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = _fft_convolve(a)
    return 2.0 * n * np.max(b) / (S * S)


def _anti_palindromic_phase(n, k, rng=None):
    """Generate an anti-palindromic sinusoid: anti[n-1-i] = -anti[i]."""
    t = np.arange(n, dtype=np.float64)
    return np.sin(2.0 * np.pi * k * (t + 0.5) / n)


def _compute_gradient(a, b, K, n):
    """Compute approximate gradient of score w.r.t. a.
    
    Uses top |K| peak positions of b to approximate the subgradient of max(b).
    Returns gradient vector of same shape as a.
    """
    S = np.sum(a)
    b_max = np.max(b)
    
    # grad_b[j] = (2/|K|) * sum_{k in K} a[k-j] (for valid k-j in [0,n-1])
    grad_b = np.zeros(n)
    for k in K:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            start_idx = k - hi + 1
            end_idx = k - lo + 1
            grad_b[lo:hi] += a[start_idx:end_idx][::-1]
    grad_b *= 2.0 / len(K)
    
    # d(score)/d(a[j]) = (2n/S^2)*grad_b[j] - 4n*b_max/S^3
    grad = (2.0 * n / (S * S)) * grad_b - (4.0 * n * b_max / (S ** 3))
    return grad


def _gradient_descent_phase(a, n, deadline, rng, K_size=50, patience=300):
    """Run gradient descent with momentum and random restarts."""
    best_seq = a.copy()
    best_score = _score_val(a)
    print(f"  GD start: score={best_score:.6f}")
    
    momentum = np.zeros(n)
    mu = 0.85
    lr = 10.0
    no_improve = 0
    iter_count = 0
    min_lr = 0.1
    
    while time.monotonic() < deadline:
        S = np.sum(a)
        if S < 1e-6:
            break
        b = _fft_convolve(a)
        b_max = np.max(b)
        
        # Find top K peaks
        if K_size < len(b):
            top_slice = np.argpartition(b, -K_size)[-K_size:]
        else:
            top_slice = np.arange(len(b))
        
        grad = _compute_gradient(a, b, top_slice, n)
        
        # Momentum update
        momentum = mu * momentum - lr * grad
        a = a + momentum
        a = np.clip(a, 0.0, 1000.0)
        
        # Renormalize to mean=1
        S_new = np.sum(a)
        if S_new > 1e-10:
            a = a * (n / S_new)
        
        new_score = _score_val(a)
        
        if new_score < best_score:
            best_score = new_score
            best_seq = a.copy()
            no_improve = 0
            if iter_count % 2000 == 0:
                print(f"  GD iter {iter_count}: score={best_score:.6f}")
        else:
            no_improve += 1
        
        # Adaptive: reduce lr when stuck
        if no_improve >= patience:
            lr *= 0.7
            no_improve = 0
            momentum = np.zeros(n)
            
            if lr < min_lr:
                # Random restart
                scale = rng.uniform(0.05, 0.2)
                a = best_seq * (1.0 + scale * rng.standard_normal(n))
                a = np.clip(a, 0.0, 1000.0)
                S_new = np.sum(a)
                if S_new > 1e-10:
                    a = a * (n / S_new)
                lr = 10.0
                mu = 0.85
        
        iter_count += 1
    
    print(f"  GD done after {iter_count} iters: best_score={best_score:.6f}")
    return best_seq, best_score


def _lp_refine(a, n, deadline, rng):
    """Try LP-based refinement: find g with (a*g)[k] <= M, blend toward g."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    best_seq = a.copy()
    best_score = _score_val(a)
    print(f"  LP start: score={best_score:.6f}")
    
    # Build the constraint matrix for (a*g)[k] <= M
    # Row k: sum_{i valid} a[i] * g[k-i] <= M
    # Variables: g[0], ..., g[n-1]
    
    S = np.sum(a)
    b = _fft_convolve(a)
    M = np.max(b)
    
    # Build sparse constraint matrix A where A[k, j] = a[k-j] if 0 <= k-j <= n-1
    # For row k, column j has value a[k-j] if 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k
    # and 0 <= j <= n-1.
    
    row_indices = []
    col_indices = []
    data = []
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j  # index into a
            if 0 <= idx < n:
                row_indices.append(k)
                col_indices.append(j)
                data.append(a[idx])
    
    A = csr_matrix((data, (row_indices, col_indices)), shape=(2 * n - 1, n))
    
    # LP: maximize sum(g) subject to A @ g <= M*ones, g >= 0
    # scipy linprog minimizes, so minimize -sum(g)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, 1000.0)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.status == 0 and result.x is not None:
        g = np.clip(result.x, 0, 1000.0)
        Sg = np.sum(g)
        if Sg > 1e-10:
            # Rescale g to match sum of a
            g = g * (S / Sg)
            
            # Try different blend factors
            for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
                candidate = (1 - t) * best_seq + t * g
                candidate = np.clip(candidate, 0.0, 1000.0)
                Sc = np.sum(candidate)
                if Sc < 1e-6:
                    continue
                candidate = candidate * (n / Sc)
                sc = _score_val(candidate)
                if sc < best_score:
                    best_score = sc
                    best_seq = candidate.copy()
                    print(f"    LP blend t={t}: score={best_score:.6f}")
    
    print(f"  LP done: best_score={best_score:.6f}")
    return best_seq, best_score


def propose_candidate(budget_s=2400, **kwargs):

    Strategy:
    1. Initialize with anti-palindromic perturbation of the given base sequence.

    3. Apply LP-based refinement.
    4. Repeat with different random initializations in remaining time.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(2024)
    
    # Get initial sequence
    if 'height_sequence_1' in globals():

    else:
        base = np.ones(7000, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    base = np.maximum(base, 0.0)
    
    # Normalize base to mean=1
    Sb = np.sum(base)
    if Sb > 1e-10:
        base = base * (n / Sb)
    
    print(f"Sequence length n={n}")
    
    overall_best = base.copy()
    overall_best_score = _score_val(overall_best)
    print(f"Initial score: {overall_best_score:.6f}")
    
    # Try multiple randomizations of anti-palindromic structure
    num_tries = 8
    time_per_try = (deadline - time.monotonic()) / max(1, num_tries - 1)
    
    for trial in range(num_tries):
        if time.monotonic() > deadline - 30:
            break
        
        if trial == 0:
            # First try: keep base as is (already score ~2)
            a = base.copy()
        else:
            # Build anti-palindromic perturbation from random sinusoids
            num_harmonics = rng.integers(1, min(5, n // 10) + 1)
            perturbation = np.zeros(n)
            for _ in range(num_harmonics):
                k = rng.integers(1, n // 2 + 1)
                amp = rng.uniform(0.1, 0.6)
                perturbation += amp * _anti_palindromic_phase(n, k)
            
            a = base + perturbation
            a = np.clip(a, 0.0, 1000.0)
            Sa = np.sum(a)

                a = a * (n / Sa)
        
        # Quick score check
        trial_score = _score_val(a)
        if trial_score >= overall_best_score and trial > 0:
            # Skip trials that don't improve on current best at init
            continue

    print(f"Final score (evaluator): {final_score:.6f}")
    print(f"Final score (internal): {overall_best_score:.6f}")
    
    return [float(x) for x in overall_best]
