```python
"""
Searches for a non-negative step function minimizing
    2*n*max(conv(a,a)) / (sum(a)^2)
by combining:
  * analytic asymmetric initial shapes,
  * mirror-descent on a log-sum-exp surrogate of the convolution peak,
  * adaptive sharpening of the active lags,
  * randomized local perturbations.
"""

import numpy as np
import time


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def _next_pow2(m):
    m = max(1, int(m))
    x = 1
    while x < m:
        x <<= 1
    return x


def _self_conv(a, L):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[: (2 * n - 1)]
    return np.maximum(c, 0.0)


def _score1(a, L):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = _self_conv(a, L)
    return float(2.0 * n * float(c.max()) / (s * s))


def _weights(c, p):
    c = np.asarray(c, dtype=np.float64)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.full(c.size, 1.0 / c.size, dtype=np.float64)

    scaled = float(p) * (c / cm - 1.0)
    exps = np.exp(scaled)
    sw = float(exps.sum())
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = exps / sw

    # Small uniform floor keeps edge lags in the objective and avoids
    # becoming too myopic when p is large.
    flat = 0.01
    w = (1.0 - flat) * w + flat / c.size
    ws = float(w.sum())
    if ws > 0.0:
        w = w / ws
    return w


def _h_from_weights(a, w, Lh):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    a_rev = a[::-1]
    Ar = np.fft.rfft(a_rev, Lh)
    W = np.fft.rfft(w, Lh)
    conv = np.fft.irfft(Ar * W, Lh)
    return np.asarray(conv[n - 1: 2 * n - 1], dtype=np.float64)


def _try_init(f, n):
    f = np.asarray(f, dtype=np.float64)
    if f.ndim != 1 or f.size != n:
        return None
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, 1000.0)
    if float(f.sum()) < 1e-10:
        return None
    return _normalize(f)


def _make_initials(n, parent, rng):
    inits = []

    if parent is not None:
        p = _try_init(parent, n)
        if p is not None:
            inits.append(p)

    u_init = _try_init(np.ones(n, dtype=np.float64), n)
    if u_init is not None:
        inits.append(u_init)

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    u = x - 0.5
    cubic = u ** 3 - 0.6 * u

    # Monotone asymmetric ramps. These break the left-right symmetry
    # that makes the uniform sequence stationary for the peak.
    for t in (0.4, 0.7, 1.0, 1.3, 1.6, 1.9):
        f = 1.0 + t * u
        f = _try_init(np.maximum(f, 0.0), n)
        if f is not None:
            inits.append(f)

    # Monotone plus antisymmetric cubic shaping.
    for t in (0.8, 1.0, 1.3):
        for s in (-0.35, -0.15, 0.0, 0.15, 0.35):
            f = 1.0 + t * u + s * cubic
            f = _try_init(np.maximum(f, 0.0), n)
            if f is not None:
                inits.append(f)

    # Exponential ramps.
    for c in (0.5, 1.0, 2.0, 3.0, 5.0, 8.0):
        f = np.exp(c * u)
        f = _try_init(f, n)
        if f is not None:
            inits.append(f)

    # Power ramps from one end.
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        f = x ** p
        f = _try_init(f, n)
        if f is not None:
            inits.append(f)
        g = (1.0 - x) ** p
        g = _try_init(g, n)
        if g is not None:
            inits.append(g)

    # A few smooth random restarts.
    if rng is not None:
        lowpass = 4.0
        for _ in range(6):
            z = rng.normal(0.0, 1.0, n)
            Z = np.fft.rfft(z)
            k = np.fft.rfftfreq(n)
            Z *= np.exp(-lowpass * (n * k) ** 2)
            z = np.fft.irfft(Z, n)
            f = 1.0 + 0.45 * z
            f = np.maximum(f, 0.02)
            f = _try_init(f, n)
            if f is not None:
                inits.append(f)

    # Keep the list reasonably small.
    cleaned = []
    for f in inits:
        f = _normalize(f)
        cleaned.append(f)
    return cleaned


def _run_mirror(a, deadline, Lc, Lh, inits, rng, tag=""):
    n = len(a)
    if n < 4:
        s = _score1(a, Lc)
        return a, s

    a = _normalize(a)
    best = a.copy()
    best_score = _score1(a, Lc)

    p_list = (4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 128)
    p_idx = 0
    p = p_list[0]
    lr = 0.28
    momentum = 0.86
    state = np.zeros(n, dtype=np.float64)

    it = 0
    no_improve = 0
    last_status = 0.0
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        c = _self_conv(a, Lc)
        cm = float(c.max())
        s = float(a.sum())
        if not (np.isfinite(cm) and np.isfinite(s)):
            a = best.copy()
            state = np.zeros(n, dtype=np.float64)
            continue

        score = float(2.0 * n * cm / (s * s))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            no_improve = 0
            last_improve = time.monotonic()

        if no_improve >= 0:
            w = _weights(c, p)
            h = _h_from_weights(a, w, Lh)
            d = h - float(np.dot(a, h))
            scale = float(np.max(np.abs(d)))
            if scale > 1e-14:
                d = d / scale
                state = momentum * state + (1.0 - momentum) * d
                if no_improve > 250:
                    noise = min(0.45, 0.02 + 0.0004 * float(no_improve))
                    state = state + rng.normal(0.0, noise, n)
                state = np.clip(state, -2.6, 2.6)
                a_new = a * np.exp(-lr * state)
                a_new = np.maximum(a_new, 1e-12)
                a = _normalize(a_new)
            else:
                # Symmetric/stagnant point: inject a mild asymmetry kick.
                perturb = np.clip(rng.normal(0.0, 0.02, n), -0.5, 0.5)
                a = _normalize(a * (1.0 + perturb))
                state = np.zeros(n, dtype=np.float64)

        it += 1
        if not (np.isfinite(score) and score < best_score):
            no_improve += 1

        # Periodically progress the sharpness.
        if it % 900 == 0 and p_idx < len(p_list) - 1:
            p_idx += 1
            p = p_list[p_idx]

        # If stuck, cycle sharpness / reduce step.
        if no_improve == 350:
            p_idx = (p_idx + 1) % len(p_list)
            p = p_list[p_idx]
            lr = max(0.05, lr * 0.75)
            state = np.zeros(n, dtype=np.float64)

        # If very stuck, restart from best mixed with a diverse initial.
        if no_improve == 2600:
            if inits:
                base = inits[int(rng.integers(len(inits)))]
            else:
                base = best
            mix = 0.2 + 0.35 * float(rng.random())
            a = (1.0 - mix) * best + mix * base
            a = _normalize(a)
            p_idx = 0
            p = p_list[0]
            lr = 0.28
            state = np.zeros(n, dtype=np.float64)
            no_improve = 0
            last_improve = time.monotonic()

        now = time.monotonic()
        if now - last_status > 8.0:
            print(f"{tag} t={now:.2f} current={score:.6f} best={best_score:.6f} p={p}")
            last_status = now

    return best, best_score


def _local_search(a, best_score, deadline, Lc, inits, rng):
    n = len(a)
    if n < 8:
        return a, best_score

    a = _normalize(a)
    best = a.copy()
    best_score = float(best_score)
    sigma = 0.16
    stagnant = 0
    last_status = 0.0

    while time.monotonic() < deadline:
        cand = best.copy()
        op = int(rng.integers(6))

        if op == 0:
            nb = int(rng.integers(1, 4))
            maxw = max(2, min(n, max(2, n // 25)))
            for _ in range(nb):
                w = int(rng.integers(1, maxw + 1))
                s = int(rng.integers(0, max(1, n - w + 1)))
                cand[s:s + w] *= float(np.exp(rng.normal(0.0, sigma)))

        elif op == 1:
            cand *= np.exp(rng.normal(0.0, sigma * 0.35, n))

        elif op == 2:
            nb = int(rng.integers(1, 4))
            for _ in range(nb):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j:
                    alpha = float(rng.uniform(0.04, 0.28))
                    ai = cand[i]
                    aj = cand[j]
                    cand[i] = (1.0 - alpha) * ai + alpha * aj
                    cand[j] = (1.0 - alpha) * aj + alpha * ai

        elif op == 3:
            # Gentle edge tapering to control endpoint convolution peaks.
            d = float(rng.uniform(0.02, 0.22))
            half = max(1, min(n // 40, 256))
            cand[:half] *= (1.0 - d)
            cand[-half:] *= (1.0 - d)

        elif op == 4:
            # Increase asymmetry by partially transferring mass from one random
            # position to its reversed partner.
            for _ in range(int(rng.integers(1, 4))):
                i = int(rng.integers(n))
                j = n - 1 - i
                if i != j:
                    alpha = float(rng.uniform(0.05, 0.30))
                    ai = cand[i]
                    aj = cand[j]
                    # Try to make the reversed pair more unequal. This reduces
                    # the contribution to the central convolution term locally.
                    if ai < aj:
                        di = alpha * (aj - ai)
                    else:
                        di = -alpha * (ai - aj)
                    cand[i] = ai + di
                    cand[j] = aj - di

        else:
            if inits:
                base = inits[int(rng.integers(len(inits)))]
            else:
                base = best
            mix = float(rng.uniform(0.02, 0.28))
            cand = (1.0 - mix) * best + mix * base

        cand = _normalize(cand)
        c = _self_conv(cand, Lc)
        s = float(cand.sum())
        sc = float(2.0 * n * float(c.max()) / (s * s)) if s > 0 else np.inf

        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            stagnant = 0
            sigma = max(0.02, sigma * 0.985)
        else:
            stagnant += 1
            if stagnant > 120:
                stagnant = 0
                sigma = max(0.02, sigma * 0.65)

        now = time.monotonic()
        if now - last_status > 10.0:
            print(f"local t={now:.2f} best={best_score:.6f}")
            last_status = now

    return best, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Returns a non-negative coefficient list. Runs for at most budget_s seconds.
    """
    start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0
    deadline = start + max(30.0, budget - 45.0)

    try:
        parent_raw = height_sequence_1
    except Exception:
        parent_raw = None

    parent = None
    n = 0
    if parent_raw is not None:
        p_arr = np.asarray(parent_raw, dtype=np.float64)
        if p_arr.ndim == 1 and p_arr.size > 0:
            p_arr = np.where(np.isfinite(p_arr), p_arr, 0.0)
            p_arr = np.clip(p_arr, 0.0, 1000.0)
            if float(p_arr.sum()) > 1e-12:
                parent = p_arr
                n = int(p_arr.size)

    if n < 4:
        if parent is not None and parent.size > 0:
            return [float(x) for x in _normalize(parent)]
        return [1.0]

    rng = np.random.default_rng(seed)
    Lc = _next_pow2(2 * n - 1)
    Lh = _next_pow2(3 * n - 2)

    inits = _make_initials(n, parent, rng)
    if not inits:
        inits = [np.full(n, 1.0 / n, dtype=np.float64)]

    # Pick the best deterministic/random initial quickly.
    best = inits[0]
    best_score = _score1(best, Lc)
    for arr in inits[1:]:
        sc = _score1(arr, Lc)
        if np.isfinite(sc) and sc < best_score:
            best = arr
            best_score = sc
    print(f"initial best score={best_score:.9f} n={n}")

    total = max(60.0, deadline - start)
    local_reserve = min(420.0, max(80.0, 0.20 * total))
    polish_reserve = min(180.0, max(25.0, 0.07 * total))

    main_deadline = deadline - local_reserve - polish_reserve
    local_deadline = deadline - polish_reserve

    if time.monotonic() < main_deadline:
        best, best_score = _run_mirror(
            best, main_deadline, Lc, Lh, inits, rng, tag="mirror-main"
        )

    if time.monotonic() < local_deadline:
        best, best_score = _local_search(
            best, best_score, local_deadline, Lc, inits, rng
        )

    if time.monotonic() < deadline - 2.0:
        best, best_score = _run_mirror(
            best, deadline - 2.0, Lc, Lh, inits, rng, tag="mirror-polish"
        )

    best = _normalize(best)
    final = [float(x) for x in best]

    # The exact evaluator is authoritative. If it rejects the result for any
    # reason, fall back to a safe normalized initial.
    try:
        exact = float(evaluate_sequence(final))
        if not np.isfinite(exact):
            exact = None
        else:
            print(f"final exact score={exact:.9f}")
    except Exception:
        exact = None

    if exact is None:
        fb = _normalize(inits[0])
        final = [float(x) for x in fb]

    return final
```
