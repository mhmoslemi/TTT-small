"""
Minimize peak autocorrelation of a nonnegative step function.

Score = 2*n*max(convolve(f,f)) / (sum f)^2.  Because the score is
scale-invariant, every candidate is normalized to sum = sqrt(2*n); under
that normalization the score equals exactly max(convolve(f,f)), so the
whole problem reduces to flattening the autocorrelation peak.

Method
------
1. Smooth the non-smooth "max of the convolution" with a softmax
   (log-sum-exp) of temperature tau, giving a smooth objective whose
   gradient is a single weighted cross-correlation (O(n log n)).
2. Optimize with heavy-ball momentum + backtracking line search on the
   smoothed objective, projecting each step onto {x >= 0, sum x = S}.
3. Anneal tau toward 0 so the smoothed objective tracks the true max.
4. Periodically escape local minima with normalized block perturbations.
5. Run from several starts (the provided SOTA seed and a tapered window)
   and keep the best, doing a final deterministic local polish.

Only numpy / math / time are used; evaluate_sequence is assumed imported.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# small helpers (all top level, no closures / lambdas)
# ----------------------------------------------------------------------

def _normalize(f, S):
    """Scale f so sum = S, keeping it nonnegative and finite."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, None)
    s = f.sum()
    if s <= 1e-30 or not math.isfinite(s):
        f = np.ones_like(f) * (S / max(1, f.size))
        return f
    return f * (S / s)


def _project_simplex(x, S):
    """Exact Euclidean projection onto {y >= 0, sum y = S}."""
    n = int(x.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    if S <= 0:
        return np.zeros(n, dtype=np.float64)
    x = np.asarray(x, dtype=np.float64)
    order = np.argsort(x)[::-1]
    xs = x[order]
    cs = np.cumsum(xs)
    rho = np.arange(1, n + 1)
    theta_candidates = (cs - S) / rho
    cond = xs > theta_candidates
    if np.any(cond):
        k = int(np.max(np.where(cond)[0])) + 1
    else:
        k = 1
    theta = theta_candidates[k - 1]
    return np.maximum(x - theta, 0.0)


def _convolve_full(a, b):
    """Full linear convolution via FFT (fast path for 1-D real inputs)."""
    n = a.size + b.size - 1
    m = 1
    while m < n:
        m <<= 1
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b, m)
    out = np.fft.irfft(fa * fb, m)
    return out[:n]


def _smoothed_value_and_grad(f, tau):
    """Return (phi_tau(f), grad) for phi_tau = tau*logsumexp(C/tau).

    grad_j = 2 * sum_i w_{i+j} f_i  = 2 * correlate(w, f, full)[j].
    """
    C = _convolve_full(f, f)
    cmax = C.max()
    z = (C - cmax) / tau
    # stable logsumexp
    zmax = z.max()
    lse = cmax + tau * (zmax + np.log(np.exp(z - zmax).sum()))
    w = np.exp(z - zmax)
    w /= w.sum()
    grad = 2.0 * np.correlate(w, f, mode="full")[: f.size]
    return lse, grad


def _fast_score(f):
    """Score under S = sqrt(2 n) normalization equals max(full conv)."""
    C = _convolve_full(f, f)
    return float(C.max())


def _local_polish(best, S, deadline, n_iter=2000, tag=""):
    """Normalized block perturbations that only keep improvements."""
    rng = np.random.default_rng(20240501)
    f = best.copy()
    n = f.size
    best_score = _fast_score(f)
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        width = int(rng.integers(max(2, n // 400), max(3, n // 8)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        cand = f.copy()
        factor = float(np.exp(rng.normal(0.0, rng.normal(0.0, 0.25) + 0.15)))
        cand[start:stop] *= factor
        cand = np.clip(cand, 0.0, None)
        cand = _normalize(cand, S)
        sc = _fast_score(cand)
        if sc < best_score:
            f = cand
            best_score = sc
    if tag and it % 250 == 0:
        print("local %s score=%.6f" % (tag, best_score))
    return f, best_score


def _optimize_from(start, S, deadline, base_tag):
    """Momentum + backtracking on the smoothed objective with tau annealing."""
    f = _normalize(start.copy(), S)
    n = f.size
    best = f.copy()
    best_score = _fast_score(f)

    tau = max(0.5, 0.05 * math.sqrt(n))   # starts far, anneals to refine
    tau_floor = max(1e-3, 0.002 * math.sqrt(n))
    tau_decay = 0.9990
    beta = 0.6          # heavy-ball momentum
    alpha = 0.5         # starting step (scaled by backtracking)
    v = np.zeros_like(f)
    it = 0
    last_t = input_check = time.monotonic()

    while time.monotonic() < deadline:
        it += 1
        phi, grad = _smoothed_value_and_grad(f, tau)

        # backtracking line search with momentum
        a = alpha
        tried = 0
        accepted = False
        while tried < 25:
            tried += 1
            v = beta * v - a * grad
            f_cand = _project_simplex(f + v, S)
            phi_c, _ = _smoothed_value_and_grad(f_cand, tau)
            if phi_c <= phi + 1e-12:
                # descent on smoothed objective -> move
                f = f_cand
                # keep a projection-safe correction of momentum
                alpha = max(alpha * 1.05, 1e-8)
                accepted = True
                break
            a *= 0.5
        if not accepted:
            # no descent found: damped restart of momentum
            v = np.zeros_like(f)
            alpha = a
            # still attempt a tiny raw projection step for safety
            f = _project_simplex(f - 0.5 * a * grad, S)

        # track true (unsmoothed) best
        sc = _fast_score(f)
        if sc < best_score:
            best_score = sc
            best = f.copy()

        # anneal temperature
        tau = max(tau * tau_decay, tau_floor)
        # slow momentum decay to keep stability at low tau
        beta = max(0.4, 0.9 * beta)

        # periodic local escape from the current best
        if it % 400 == 0:
            f, sc = _local_polish(best, S, deadline, n_iter=200,
                                  tag=base_tag + "#" + str(it))
            if sc < best_score:
                best_score = sc
            f = _normalize(f, S)

        # progress logging
        if it % 500 == 0:
            print("%s it=%d smoothed=%.5f peak=%.6f tau=%.4f"
                  % (base_tag, it, _smoothed_value_and_grad(f, tau)[0],
                     best_score, tau))

    return best, best_score


def _tapered_init(n):
    """Window that tapers to zero at both edges (good autocorr start)."""
    i = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * (i + 0.5) / n) ** 2
