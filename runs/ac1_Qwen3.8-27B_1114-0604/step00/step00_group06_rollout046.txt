def project_simplex(v, S):
    """Project v onto {x: sum(x) = S, x >= 0} using bisection."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Find lambda such that sum(max(0, v - lambda)) = S
    # sum(max(0, v-lambda)) is decreasing in lambda
    # Lower bound: lambda_min = min(v) - S (if all positive)
    # Upper bound: lambda_max = max(v)
    
    # Sort for efficient computation
    v_sorted = np.sort(v)[::-1]  # descending
    prefix = np.concatenate([[0], np.cumsum(v_sorted)])
    
    def sum_above(lam):
        # Count of elements > lam
        idx = np.searchsorted(-v_sorted, -lam, side='left')  
        # v_sorted[idx] >= lam... hmm this is getting complicated
        # Just do it directly for simplicity
        return np.sum(np.maximum(0, v - lam))
    
    lo = np.min(v) - S - 1
    hi = np.max(v) + 1
    
    for _ in range(60):
        mid = (lo + hi) / 2
        s = sum_above(mid)
        if s > S:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2
    return np.maximum(0, v - lam)


def compute_score_fast(a):
    """Compute score without calling evaluate_sequence (for speed)."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_c / (sum_a ** 2))


def subgradient_step(a, alpha):
    """One subgradient step on the hard max of a*a."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = np.argmax(c)
    # grad_i = 2 * a_{k_star - i} for valid indices
    # a_{k_star - i} is valid when 0 <= k_star - i < n
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    # Update
    a_new = a - alpha * grad
    # Project onto simplex (sum = current sum, non-negative)
    S = np.sum(a)
    a_new = project_simplex(a_new, S)
    return a_new


def hard_max_gradient(a, T=0.0):
    """Compute gradient of smoothed max of conv(a,a)."""
    n = len(a)
    c = np.convolve(a, a)
    if T <= 0:
        k_star = np.argmax(c)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        return grad
    else:
        # Softmax
        c_shifted = c - np.max(c)
        w = np.exp(c_shifted / T)
        w /= np.sum(w)
        grad = (2.0 / T) * np.correlate(w, a, mode='valid')
        return grad

def generate_ramp(n, direction=1):
    """Generate an increasing (or decreasing) ramp sequence."""
    a = np.linspace(0.1, 1.0, n)
    if direction == -1:
        a = a[::-1]

def generate_step(n, ratio=0.5, high_val=2.0):
    """Generate a step function: low on left, high on right."""
    split = int(n * ratio)
    a = np.ones(n)
    a[split:] = high_val
    return a


def generate_two_blocks(n, block_ratio=0.4, gap_pos=0.6):
    """Generate a sequence with two blocks and a gap."""
    a = np.ones(n)
    gap_start = int(n * gap_pos)
    gap_end = min(n, gap_start + max(1, int(n * 0.05)))
    a[gap_start:gap_end] = 0
    return a


def generate_geometric(n, ratio=0.99):
    """Generate a geometric sequence."""
    a = np.array([ratio ** (n-1-i) for i in range(n)])
    return a

"""
Minimizes score = 2*n*max(conv(a,a))/(sum a)^2 using non-palindromic initializations
and smoothed subgradient descent. Key insight: palindromic sequences always have peak
at center (score>=2), so we need asymmetric structures to break below 2.
"""

import numpy as np
import time


def project_simplex(v, S):
    """Project v onto {x: sum(x)=S, x>=0} via bisection on threshold."""
    n = v.size
    v = np.asarray(v, dtype=np.float64)
    lo = np.min(v) - S - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.maximum(0.0, v - mid))
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.maximum(0.0, v - lam)


def compute_score_fast(a):
    """Compute score without calling evaluate_sequence."""
    n = a.size
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = np.convolve(a, a)
    max_c = np.max(c)
    return float(2.0 * n * max_c / (sum_a * sum_a))


def smoothed_gradient(a, T):
    """Compute gradient of T*log(sum exp(c_k/T)) w.r.t. a."""
    n = a.size
    c = np.convolve(a, a)
    if T < 1e-10:
        # Hard max: pick the argmax
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        return grad
    # Softmax gradient
    c_shift = c - np.max(c)
    w = np.exp(c_shift / T)
    w /= np.sum(w)
    grad = (2.0 / T) * np.correlate(w, a, mode='valid')
    return grad


def optimize_subgradient(a, time_budget=50.0, seed=0):
    """Smoothed subgradient descent with adaptive parameters."""
    rng = np.random.default_rng(seed)
    n = a.size
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(0.0, a)
    S = np.sum(a)
    if S < 0.01:
        return a
    a = a / S

    best_a = a.copy()
    best_score = compute_score_fast(a)
    start = time.monotonic()

    alpha = 0.05
    T = 5.0 / n
    T_min = 0.1 / n
    T_decay = 0.999
    stall = 0
    stall_limit = 100

    while time.monotonic() - start < time_budget:
        grad = smoothed_gradient(a, T)
        a_new = a - alpha * grad
        a_new = project_simplex(a_new, 1.0)

        score = compute_score_fast(a_new)
        if score < best_score:
            best_a = a_new.copy()
            best_score = score
            stall = 0
            alpha = max(0.001, alpha * 0.999)
        else:
            stall += 1
            alpha = max(0.0001, alpha * 0.97)

        T = max(T_min, T * T_decay)

        if stall >= stall_limit:
            # Escape: add random anti-symmetric perturbation
            eps = 0.05 / n
            r = rng.random(n) - 0.5
            r = r - r[::-1]  # anti-symmetric
            a_new = a + eps * r
            a_new = project_simplex(a_new, 1.0)
            score_new = compute_score_fast(a_new)
            if score_new < best_score * 1.05:
                a = a_new
            else:
                a = project_simplex(a + eps * rng.normal(0, 1, n), 1.0)
            stall = 0
            alpha = 0.03
            T = max(T_min, 2.0 / n)

        a = a_new

    return best_a


def local_search(a, time_budget=30.0, seed=0):
    """Local search with coordinate, block, and swap moves."""
    rng = np.random.default_rng(seed)
    n = a.size
    a = np.asarray(a, dtype=np.float64).copy()

    S = np.sum(a)
    if S < 0.01:

    best_score = compute_score_fast(a)

        elif move < 0.5:

            s = rng.integers(0, max(1, n - w))
            a_new[s:s+w] *= np.exp(rng.normal(0, 0.3, size=w))
            a_new = np.maximum(0.0, a_new)
        elif move < 0.7:

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                eps = min(a_new[i], a_new[j]) * rng.uniform(0.1, 0.5)
                a_new[i] -= eps
                a_new[j] += eps

def make_ramp(n, direction=1):
    a = np.linspace(0.1, 1.0, n, dtype=np.float64)
    if direction == -1:
        a = a[::-1]
    return a


def make_step(n, split_frac, high_val):
    a = np.ones(n, dtype=np.float64)
    split = int(n * split_frac)
    a[split:] = high_val
    return a


def make_geometric(n, ratio):
    idx = np.arange(n, dtype=np.float64)
    return ratio ** (n - 1 - idx)


def make_bias(n, frac_low, low_val, high_val):
    a = np.ones(n, dtype=np.float64)
    split = int(n * frac_low)
    a[:split] = low_val
    a[split:] = high_val
    return a
