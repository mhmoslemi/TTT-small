import numpy as np
import time

def propose_candidate(height_sequence_1=None, budget_s=2400):
    """
    Minimize 2n*max(conv(a,a))/sum(a)^2 using iterative peak reduction.
    
    Strategy:
    1. Start from multiple initializations (uniform, random, known sequences)
    2. Use a "peak shaving" approach: identify the lag with maximum convolution
       value, then adjust the sequence to reduce that peak.
    3. Use simulated annealing to escape local minima.
    4. Use FFT for fast convolution computation.
    """
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(12345)
    
    best_score = np.inf
    best_seq = None
    
    def fast_conv_nonneg(a):
        """Compute linear convolution using FFT for non-negative sequences."""
        n = len(a)
        # Need at least 2n-1 points
        m = 1
        while m < 2*n - 1:
            m *= 2
        fa = np.fft.rfft(a, m)
        c = np.fft.irfft(fa * fa, m)[:2*n-1]
        return c
    
    def compute_score(a):
        """Compute the objective value without calling evaluate_sequence."""
        n = len(a)
        S = np.sum(a)
        if S < 0.01:
            return np.inf
        c = fast_conv_nonneg(a)
        max_c = np.max(c)
        return 2.0 * n * max_c / (S * S)
    
    def peak_shave(a, rng, alpha=0.1):
        """
        Try to reduce the peak of the convolution.
        Identify the lag k* with max c[k*], then reduce contributions.
        """
        n = len(a)
        c = fast_conv_nonneg(a)
        k_star = np.argmax(c)
        M = c[k_star]
        
        # Find which pairs (i, j) with i + j = k_star contribute most to c[k_star]
        # c[k_star] = sum_i a[i] * a[k_star - i] for valid i
        pairs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                pairs.append((i, j, a[i] * a[j]))
        
        if not pairs:
            return a.copy()
        
        # Sort by contribution
        pairs.sort(key=lambda x: -x[2])
        
        # Try reducing the top contributors
        new_a = a.copy()
        for reduction_frac in [0.05, 0.1, 0.2, 0.3]:
            # Reduce top few contributors
            n_reduce = max(1, len(pairs) // 4)
            for i, j, contrib in pairs[:n_reduce]:
                # Reduce a[i] (and a[j] if different)
                factor = 1.0 - reduction_frac * alpha
                new_a[i] = max(0, new_a[i] * factor)
                if j != i:
                    new_a[j] = max(0, new_a[j] * factor)
            
            # Renormalize to keep sum the same
            S = np.sum(new_a)
            if S > 0:
                new_a = new_a * (np.sum(a) / S)
            
            new_score = compute_score(new_a)
            if new_score < compute_score(a):
                return new_a
        
        return a.copy()
    
    def random_perturbation(a, rng, scale=0.1):
        """Add random noise and clip to non-negative."""
        n = len(a)
        noise = rng.normal(0, scale, n)
        new_a = np.maximum(0, a + noise)
        # Renormalize
        S = np.sum(new_a)
        if S > 0:
            new_a = new_a * (np.sum(a) / S)
        return new_a
    
    # Try different sequence lengths
    candidate_lengths = [256, 512, 1024, 2048]
    # If we have a parent, use its length
    if height_sequence_1 is not None and len(height_sequence_1) > 0:
        parent_len = len(height_sequence_1)
        candidate_lengths = [parent_len] + [l for l in candidate_lengths if l != parent_len]
    
    for n in candidate_lengths:
        # Multiple initializations for this length
        for init_type in ['uniform', 'random', 'parent']:
            if init_type == 'uniform':
                a = np.ones(n)
            elif init_type == 'random':
                a = np.maximum(0, rng.normal(1.0, 0.5, n))
            elif init_type == 'parent' and height_sequence_1 is not None:
                parent = np.array(height_sequence_1, dtype=np.float64)
                if len(parent) >= n:
                    a = parent[:n].copy()
                else:
                    a = np.zeros(n)
                    a[:len(parent)] = parent
            else:
                continue
            
            a = np.maximum(0, a)
            S = np.sum(a)
            if S < 0.01:
                continue
            a = a * (n / S)  # Normalize so sum = n
            
            score = compute_score(a)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()
            
            print(f"n={n}, init={init_type}: score={score:.6f}")
            
            # Local search from this starting point
            current = a.copy()
            current_score = score
            temp = 1.0  # Temperature for annealing
            n_iter = 0
            iter_deadline = min(deadline, time.monotonic() + 120)  # Max 120s per (n, init)
            
            while time.monotonic() < iter_deadline and n_iter < 5000:
                n_iter += 1
                strategy = rng.random()
                
                if strategy < 0.5:
                    # Peak shaving
                    candidate = peak_shave(current, rng, alpha=0.1)
                elif strategy < 0.8:
                    # Random perturbation
                    candidate = random_perturbation(current, rng, scale=0.05)
