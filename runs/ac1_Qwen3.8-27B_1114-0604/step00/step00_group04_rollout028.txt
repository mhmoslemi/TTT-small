    """Build sparse LP constraint matrix efficiently."""
    total_rows = 2 * n - 1
    
    # For each row k, the non-zero columns are i in [max(0, k-n+1), min(n-1, k)]
    # and the values are f[k-i]

    data = np.empty(total_nnz, dtype=np.float64)

        i_end = min(n, k + 1)
        length = i_end - i_start
        rows[idx:idx+length] = k
        cols[idx:idx+length] = np.arange(i_start, i_end)
        data[idx:idx+length] = f[k - np.arange(i_start, i_end)]
        idx += length
    
    A = csr_matrix((data, (rows, cols)), shape=(total_rows, n))
    return A

"""
Iterative LP-based minimization of autocorrelation peak.
Combines sparse LP solving with FFT-based peak computation,
multiple initializations, and adaptive line search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _precompute_structure(n):
    """Precompute the row/col/index structure for the LP constraint matrix."""
    total_rows = 2 * n - 1
    rows_list = []
    cols_list = []
    f_idx_list = []
    for k in range(total_rows):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        length = i_end - i_start
        rows_list.extend([k] * length)
        cols_list.extend(range(i_start, i_end))
        f_idx_list.extend(range(k - i_end + 1, k - i_start + 1))
    return (np.array(rows_list, dtype=np.int32),
            np.array(cols_list, dtype=np.int32),
            np.array(f_idx_list, dtype=np.int32))


def _peak_fft(f):
    """Compute max of linear convolution f*f using FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    conv = np.fft.irfft(F * F, fft_size)
    return float(conv[:size].max())


def _lp_direction(f, structure, n, M):
    """Solve LP to find direction g maximizing sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    rows, cols, f_idx = structure
    data = f[f_idx]
    A = csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    c = -np.ones(n)
    b = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _iterate_lp(f, n, structure, deadline, max_stalls=15):
    """Run LP-based iterations to minimize the peak of f*f."""
    f = np.clip(f.astype(np.float64).copy(), 0.0, None)
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / s
    
    M = _peak_fft(f)
    stall = 0
    best_M = M
    
    while time.monotonic() < deadline:
        g = _lp_direction(f, structure, n, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        # Line search
        best_t = 0.0
        best_M_new = M
        for t in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05):
            cand = (1.0 - t) * f + t * g
            cand = np.clip(cand, 0.0, None)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            M_c = _peak_fft(cand)
            if M_c < best_M_new:
                best_M_new = M_c
                best_t = t
        
        if best_t == 0.0:
            break
        
        f = (1.0 - best_t) * f + best_t * g
        f = np.clip(f, 0.0, None)
        f = f / f.sum()
        M = best_M_new
        
        if M < best_M - 1e-12:
            best_M = M
            stall = 0
        else:
            stall += 1
            if stall >= max_stalls:
                break
    
    return f


def _cosine_init(n, rng):
    x = np.linspace(0.0, np.pi, n)
    f = np.sin(x / 2.0) ** 2
    f = f + 1e-6 * rng.random(n)  # small perturbation to break symmetry
    return np.clip(f, 0.0, None)


def _gaussian_init(n, rng):
    x = np.linspace(-1.0, 1.0, n)
    sigma = 0.4 + 0.3 * rng.random()
    f = np.exp(-x**2 / (2 * sigma**2))
    f = f + 1e-6 * rng.random(n)
    return np.clip(f, 0.0, None)


def _random_smooth_init(n, rng):
    f = rng.random(n)
    # Smooth with a small Gaussian kernel
    kernel_size = max(3, n // 20)
    kernel = np.exp(-np.linspace(-3, 3, kernel_size)**2)
    kernel /= kernel.sum()
    f = np.convolve(f, kernel, mode='same')
    f = f + 1e-6 * rng.random(n)
    return np.clip(f, 0.0, None)


def _triangular_init(n, rng):
    x = np.linspace(-1.0, 1.0, n)
    f = np.maximum(0.0, 1.0 - np.abs(x))
    f = f + 1e-6 * rng.random(n)
    return np.clip(f, 0.0, None)
