- If `0 <= k-j < n`: `f[k-j] + f[k-j] = 2*f[k-j]`

    else:
        f0 = f0 / s
    
    deadline = time.monotonic() + budget_s - 20.0
    best_f = f0.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print(f"n={n}, initial score={best_score:.6f}")
    
    # FFT size for convolution
    conv_size = _fft_size(2 * n - 1)
    # FFT size for correlation (gradient)
    corr_size = _fft_size(3 * n - 2)
    
    rng = np.random.default_rng(12345)
    n_restarts = 5
    time_per_restart = (budget_s - 20.0) / n_restarts
    
    for r in range(n_restarts):
        t_end = time.monotonic() + time_per_restart
        
        # Different initializations
        if r == 0:
            f = f0 + 0.02 * rng.standard_normal(n)
        elif r == 1:
            f = f0 * np.linspace(0.5, 1.5, n)
        elif r == 2:
            f = f0 * (1.0 + 0.5 * np.sin(2 * np.pi * np.arange(n) / n))
        elif r == 3:
            f = f0 * (1.0 + 0.3 * np.cos(4 * np.pi * np.arange(n) / n))
        else:
            f = f0 * (1.0 + 0.5 * rng.standard_normal(n))
        
        f = np.clip(f, 0, None)
        f = _project_simplex(f)
        
        # Adam state
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        lr = 0.01
        beta1, beta2, eps = 0.9, 0.999, 1e-10
        T = 1e-4  # softmax temperature
        
        step = 0
        local_best = float('inf')
        eval_every = 10000
        stagnant = 0
        
        while time.monotonic() < t_end:
            # Forward FFT
            F = np.fft.rfft(f, conv_size)
            
            # Convolution
            conv = np.fft.irfft(F * F, conv_size)[:2 * n - 1]
            
            # Softmax weights
            scaled = conv / T
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()
            
            # Gradient via correlation
            W = np.fft.rfft(w, corr_size)
            Fp = np.fft.rfft(f, corr_size)
            corr = np.fft.irfft(W * np.conj(Fp), corr_size)
            grad = 2.0 * corr[:n]
            
            # Adam
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project
            f = _project_simplex(f)
            
            step += 1
            
            if step % eval_every == 0:
                score = float(evaluate_sequence(f.tolist()))
                if score < local_best:
                    local_best = score
                    stagnant = 0
                    if score < best_score:
                        best_score = score
                        best_f = f.copy()
                        print(f"  R{r} s{step}: NEW BEST {best_score:.6f}")
                else:
                    stagnant += 1
                    if stagnant >= 5:
                        # Add noise to escape
                        f = f + 0.005 * rng.standard_normal(n)
                        f = _project_simplex(f)
                        stagnant = 0
                
                if step % (eval_every * 5) == 0:
                    print(f"  R{r} s{step}: local={local_best:.6f} best={best_score:.6f} maxc={conv.max():.7f}")
        
        score = float(evaluate_sequence(f.tolist()))
        if score < best_score:
            best_score = score
            best_f = f.copy()
        print(f"Restart {r} done: local={local_best:.6f} global={best_score:.6f}")
    
    print(f"FINAL: {best_score:.6f}")
    return [float(x) for x in best_f]
