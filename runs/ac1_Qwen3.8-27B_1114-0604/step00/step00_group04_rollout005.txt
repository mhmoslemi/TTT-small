    cols = np.repeat(j_idx, n)
    vals = np.repeat(f, n)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_direction_lp(f, M):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    n = len(f)
    A_ub = build_convolution_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res


def lp_optimize(f, n, deadline, t_values, seed):
    """Run LP-based iterations to minimize the autocorrelation peak."""
    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * n)
    f = f / np.sum(f) * S
    best_f = f.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    stagnant = 0
    
    while time.monotonic() < deadline:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        
        res = solve_direction_lp(f, M)
        if not res.success:
            stagnant += 1
            if stagnant > 10:
                break
            # Perturb and retry
            f = np.clip(f + rng.normal(0, 0.05 * S / np.sqrt(n), n), 0, None)
            f = f / np.sum(f) * S
            continue
        
        g = res.x
        S_g = float(np.sum(g))
        if S_g < 1e-10:
            stagnant += 1
            if stagnant > 10:
                break
            f = np.clip(f + rng.normal(0, 0.05 * S / np.sqrt(n), n), 0, None)
            f = f / np.sum(f) * S
            continue
        
        g = g * S / S_g
        
        # Try different blending parameters
        improved = False
        for t in t_values:
            f_t = (1.0 - t) * f + t * g
            f_t = np.clip(f_t, 0.0, None)
            s_t = float(np.sum(f_t))
            if s_t < 1e-10:
                continue
            f_t = f_t / s_t * S
            score_t = float(evaluate_sequence(f_t.tolist()))
            if score_t < best_score:
                best_score = score_t
                best_f = f_t.copy()
                f = f_t.copy()
                improved = True
                stagnant = 0
                break
        
        if not improved:
            stagnant += 1
            if stagnant > 15:
                # Large restart
                f = 0.5 * best_f + 0.5 * rng.random(n) * S
                f = np.clip(f, 0, None)
                f = f / np.sum(f) * S
                stagnant = 0
    
    return best_f, best_score


def local_search_polish(f, n, deadline, seed):
    """Simple local search with random perturbations."""
    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * n)
    f = f / np.sum(f) * S
    best_f = f.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    
    # Adaptive step size
    step_scale = 0.1
    fail_count = 0
    
    while time.monotonic() < deadline:
        # Random block perturbation
        width = int(rng.integers(max(5, n // 100), max(10, n // 10)))
        start = int(rng.integers(0, max(1, n - width)))
        
        f_try = f.copy()
        f_try[start:start + width] *= np.exp(rng.normal(0, step_scale, width))
        f_try = np.clip(f_try, 0.0, None)
        s = float(np.sum(f_try))
        if s < 1e-10:
            fail_count += 1
            continue
        f_try = f_try / s * S
        
        score = float(evaluate_sequence(f_try.tolist()))
        if score < best_score:
            best_score = score
            best_f = f_try.copy()
            f = f_try.copy()
            fail_count = 0
            step_scale = min(0.5, step_scale * 1.1)
        else:
            fail_count += 1
            if fail_count > 50:
                step_scale = max(0.01, step_scale * 0.7)
                fail_count = 0
    
    return best_f, best_score
