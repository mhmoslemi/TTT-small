    a = a / a.sum()  # normalize

        M = np.max(c)

        c_scaled -= np.max(c_scaled)
        p = np.exp(c_scaled)

        pa = np.convolve(p, a)

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new

    n = len(a)

    rows = []
    cols = []

        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

    n = len(a)
    A = build_constraint_matrix(a)
    # Maximize sum(g) = - minimize -sum(g)
    c_obj = -np.ones(n)
    # Constraints: A @ g <= M
    b_ub = np.full(2*n-1, M)
    # g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
