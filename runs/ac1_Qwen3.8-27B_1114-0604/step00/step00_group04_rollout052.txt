import time
from scipy.optimize import linprog
from scipy import sparse

def build_lp_matrix(a):
    """Build the convolution constraint matrix for the LP."""
    n = len(a)
    # A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n
    # Column j has a[0], ..., a[n-1] at rows j, j+1, ..., j+n-1
    rows = np.concatenate([np.arange(j, j+n) for j in range(n)])

    data = np.tile(a, n)
    A = sparse.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()

def build_conv_matrix(a):
    """Build sparse convolution constraint matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows_parts = []
    cols_parts = []
    data_parts = []
    for j in range(n):
        rows_parts.append(np.arange(j, j + n, dtype=np.int32))
        cols_parts.append(np.full(n, j, dtype=np.int32))
        data_parts.append(a)
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    data = np.concatenate(data_parts)
    return sparse.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def solve_lp(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='revised simplex',
                      options={'maxiter': 5000})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def soft_max_step(a, T, eta):
    """One soft-max gradient descent step to reduce max of convolution."""
    n = len(a)
    conv = np.convolve(a, a)
    Tconv = T * conv
    Tconv -= Tconv.max()
    p = np.exp(Tconv)
    p_sum = p.sum()
    if p_sum < 1e-300:
        return a.copy()
    p /= p_sum
    # Gradient: g_j = 2 * sum_k p[k] * a[k-j] = 2 * correlate(p,a,'full')[n-1+j]
    corr = np.correlate(p, a, 'full')
    g = 2.0 * corr[n - 1:]
    g -= g.mean()
    a_new = a - eta * g
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def hard_max_step(a, eta):

    n = len(a)
    conv = np.convolve(a, a)

    g = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:

    a_new = a - eta * g
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s < 1e-10:

    return a_new / s

    print(f"Search budget: {budget_s}s, deadline: {deadline - time.monotonic():.1f}s remaining")

    # Try lengths from small to large; allocate more time to smaller lengths
    lengths = [4, 5, 6, 7, 8, 10, 12, 14, 16, 20, 24, 32, 40, 48, 64,
               80, 96, 128, 160, 192, 256, 320, 384, 512, 640, 768,
               1024, 1280, 1536, 2048, 2560, 3072, 3809]

    # Allocate time: give more budget to smaller n (where improvements are larger)
    # Use a scheduling approach: try each length with a decreasing time budget
    total_budget = deadline - time.monotonic()
    # Weight smaller lengths more heavily
    weights = [1.0 / (1.0 + i * 0.3) for i in range(len(lengths))]
    w_sum = sum(weights)
    allocated = [total_budget * w / w_sum for w in weights]

    # Ensure minimum time for each
    min_time = 2.0
    for i in range(len(lengths)):
        if allocated[i] < min_time and total_budget > 0:
            allocated[i] = min_time

    for i, n in enumerate(lengths):
        if time.monotonic() > deadline - 5:
            break

        # Don't spend more allocated time than remaining
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        n_deadline = min(deadline - 2, time.monotonic() + allocated[i])

        print(f"  Optimizing n={n} (budget {allocated[i]:.1f}s)...")
        t0 = time.monotonic()

        a_opt, score = optimize_length(n, n_deadline, rng)

        elapsed = time.monotonic() - t0
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"    n={n}: NEW BEST score={score:.6f} (took {elapsed:.1f}s)")
        else:
            print(f"    n={n}: best={score:.6f} (took {elapsed:.1f}s)")

        # If we've already beaten the target, we can stop or do a final polish
        if best_score < 1.5027:
            print(f"  Target 1.5027 surpassed! Current best: {best_score:.6f}")
            # Do a bit more refinement on the best
            if time.monotonic() < deadline - 30 and best_seq is not None:
                best_seq = gradient_refine(best_seq, deadline, max_iters=5000)
                best_seq = local_search(best_seq, deadline, rng, max_evals=500)
                score = evaluate_sequence(best_seq.tolist())
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    print(f"  After polish: {best_score:.6f}")
            break

    # If we still have time, do final refinement on the best solution
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"  Final refinement on best (n={len(best_seq)}, score={best_score:.6f})...")
        best_seq = gradient_refine(best_seq, deadline, max_iters=10000)
        best_seq = local_search(best_seq, deadline, rng, max_evals=1000)
        score = evaluate_sequence(best_seq.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            print(f"  After final refinement: {best_score:.6f}")
        else:
            # Revert if it got worse
            best_score = max(best_score, score)

    print(f"\n=== FINAL: best_score = {best_score:.6f}, length = {len(best_seq) if best_seq is not None else 0} ===")

    if best_seq is not None:
        return [float(x) for x in best_seq]

    # Fallback: small sequence
    return [0.5, 0.25, 0.25]
