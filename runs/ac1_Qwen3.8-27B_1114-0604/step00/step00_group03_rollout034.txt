def multiplicative_update(a, n_iter=50, eta=0.3):
    """Multiplicative weight updates to flatten auto-convolution."""
    a = a.copy().astype(np.float64)
    n = len(a)
    for _ in range(n_iter):
        conv = np.convolve(a, a)
        total = conv.sum()
        if total < 1e-15:
            break
        w = conv / total
        # load[i] = sum_j a[j] * w[i+j] for valid i+j
        # = convolve(a_reversed, w)[n-1 : 2n-1]
        load = np.convolve(a[::-1].copy(), w)
        load = load[n - 1:2 * n - 1]
        if load.max() > 0:
            a = a * np.exp(-eta * load / load.max())
        a = np.clip(a, 0, 1000)
    return a


def spectral_flatten(a, n_iter=20, t=0.3):
    """Spectral magnitude flattening via FFT."""
    a = a.copy().astype(np.float64)
    n = len(a)
    L = 1  # power of 2 >= 2n-1
    while L < 2 * n - 1:
        L *= 2
    for _ in range(n_iter):
        A = np.fft.rfft(a, L)
        mag = np.abs(A)
        phase = np.angle(A)
        target_mag = np.mean(mag)
        new_mag = (1 - t) * mag + t * target_mag

        a_new = np.fft.irfft(A_new)
        a = a_new[:n]
        a = np.clip(a, 0, 1000)
    return a

            a_new = spectral_flatten(best, n_iter=5, t=0.15)
            score = compute_score(a_new)
            if score < best_score - 1e-9:
                real_score = evaluate_sequence(a_new.tolist())
                if real_score < best_score - 1e-9:
                    best_score = real_score
                    best = a_new.copy()
                    no_improve = 0
                    improved = True

        # Random perturbation periodically
        if not improved and iter_count % 30 == 0:
            sigma = 0.05 + 0.02 * min(iter_count // 30, 10)
            a_new = best * np.exp(rng.normal(0, sigma, n))
            a_new = np.clip(a_new, 0, 1000)
            score = compute_score(a_new)
            if score < best_score - 1e-9:
                real_score = evaluate_sequence(a_new.tolist())
                if real_score < best_score - 1e-9:
                    best_score = real_score
                    best = a_new.copy()
                    no_improve = 0
                    improved = True
                    print(f"Perturb #{iter_count} (sigma={sigma:.3f}): score={best_score:.6f}")

        if not improved:
            no_improve += 1

    print(f"Done: score={best_score:.6f}, n={n}, iters={iter_count}")
    return best.tolist()
