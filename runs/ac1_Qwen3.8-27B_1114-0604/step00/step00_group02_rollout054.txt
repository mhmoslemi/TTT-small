```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time

def build_conv_matrix(f, n):
    """Build the sparse convolution matrix A where (A@g)[k] = conv(f,g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        # conv[k] = sum_j f[k-j] * g[j] where 0 <= k-j < n and 0 <= j < n
        j_min = max(0, k - (n-1))
        j_max = min(k, n-1)
        for j in range(j_min, j_max + 1):
            fi = f[k - j]
            if fi != 0:
                rows.append(k)
                cols.append(j)
                vals.append(fi)
    return coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()

def lp_improve(f, M, n):
    """Solve LP to find g maximizing sum(g) s.t. conv(f,g) <= M, g >= 0."""
    c = -np.ones(n)  # minimize -sum(g)
    A = build_conv_matrix(f, n)
    b = np.full(2*n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.status != 0:
        return None
    return res.x

def improve_lp(f, max_iters=50, t_start=0.5, t_decay=0.95):
    """Iteratively improve f using LP-based approach."""
    f = f.copy()
    n = len(f)
    t = t_start
    best_f = f.copy()
    best_score = 2*n*(f*f).max() / f.sum()**2  # quick eval
    
    for it in range(max_iters):
        conv = np.convolve(f, f)
        M = conv.max()
        g = lp_improve(f, M, n)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g < 1e-10:
            break
        # Normalize g to same sum as f
        g = g * f.sum() / sum_g
        # Blend
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        if f_new.sum() < 1e-10:
            break
        f_new = f_new / f_new.sum() * f.sum()
        
        conv_new = np.convolve(f_new, f_new)
        M_new = conv_new.max()
        if M_new < M:
            f = f_new
            best_f = f.copy()
            best_score = 2*n*M_new / f.sum()**2
        t *= t_decay
    
    return best_f, best_score
```

```python
"""
LP-based iterative optimization for minimizing self-convolution peak.
Starts from structured initializations (two-block, three-block) and uses
linear programming to find improvement directions, combined with local search.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def _build_conv_matrix(f, n):
    """Build sparse convolution matrix A where (A @ g)[k] = sum_j f[k-j]*g[j]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            v = f[k - j]
            if v > 0:
                rows.append(k)
                cols.append(j)
                vals.append(v)
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    return A


def _lp_step(f, M, n, rows_cached=None, cols_cached=None):
    """One LP step: find g maximizing sum(g) s.t. conv(f,g) <= M, g >= 0."""
    if rows_cached is not None and cols_cached is not None:

        pass
```
