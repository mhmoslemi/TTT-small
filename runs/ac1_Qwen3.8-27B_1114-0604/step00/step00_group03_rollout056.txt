def convolve(a,b=None):
    a=np.asarray(a,dtype=np.float64)
    if b is None: b=a
    else: b=np.asarray(b,dtype=np.float64)
    n=a.size; m=b.size; out=n+m-1
    if n==0 or m==0: return np.array([])
    if out <= 4096:
        c=np.convolve(a,b)
    else:

        while size < out: size <<=1
        fa=np.fft.rfft(a,size)
        fb=np.fft.rfft(b,size)
        c=np.fft.irfft(fa*fb,size)[:out]
    c = np.where(c < 0.0, 0.0, c)
    return c

"""
Search for low autoconvolution-peak sequences.

The objective is scale invariant:
    F(a) = 2 n max_k (a*a)_k / (sum_i a_i)^2,
with a_i >= 0.  Internally we scale sequences so max(a*a) ~= 1 and maximize
the sum.  The main loop alternates:
  * smooth log-sum-exp / active-set first-order directions,
  * reduced linear-programming directions (packing LP),
  * occasional perturbations/restarts to escape local minima.

Initial points include truncated (1-x)^-1/2 coefficients, power laws,
exponentials, and smoothed random sequences.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

from operator import itemgetter

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")

_EPS = 1e-12


def clip_valid(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    return np.clip(a, 0.0, 1000.0)


def convolve(a, b=None):
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)

    n = a.size
    m = b.size
    out_len = n + m - 1
    if n == 0 or m == 0:
        return np.zeros(0, dtype=np.float64)

    if out_len <= 2048:
        c = np.convolve(a, b)
    else:
        size = 1
        while size < out_len:
            size <<= 1
        fa = np.fft.rfft(a, size)
        if b is a:
            c = np.fft.irfft(fa * fa, size)[:out_len]
        else:
            fb = np.fft.rfft(b, size)
            c = np.fft.irfft(fa * fb, size)[:out_len]

    if c.size:
        np.maximum(c, 0.0, out=c)
    return c


def analyse(a):
    a = clip_valid(a)
    n = a.size
    if n == 0:
        return np.inf, np.zeros(0, dtype=np.float64), 0.0, 0.0

    c = convolve(a, a)
    if c.size == 0:
        return np.inf, c, 0.0, 0.0

    mx = float(np.max(c))
    s = float(np.sum(a))
    if not np.isfinite(mx) or not np.isfinite(s) or s < 0.01 or mx <= 0.0:
        return np.inf, c, mx, s

    score = float(2.0 * n * mx / (s * s))
    if not np.isfinite(score):
        score = np.inf
    return score, c, mx, s


def normalized_sum(a, target_sum):
    a = clip_valid(a)
    s = float(np.sum(a))
    if s <= _EPS:
        return np.full(a.size, float(target_sum) / max(1, a.size), dtype=np.float64)
    return a * (float(target_sum) / s)


def make_sqrt_sequence(n):
    a = np.empty(max(1, int(n)), dtype=np.float64)
    a[0] = 1.0
    for k in range(1, a.size):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)
    return a


def make_power_sequence(n, alpha):
    x = np.arange(1, int(n) + 1, dtype=np.float64)
    return x ** (-float(alpha))


def make_exponential_sequence(n, direction, rng):
    x = np.arange(int(n), dtype=np.float64)
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    x /= float(n - 1)
    lam = float(rng.uniform(1.0, 4.0))
    if direction == 0:
        return np.exp(lam * x)
    return np.exp(lam * (1.0 - x))


def make_smooth_random_sequence(n, mode, rng):
    if mode == 0:
        v = 1.0 + 0.9 * rng.standard_normal(n)
    elif mode == 1:
        v = rng.random(n)
    else:
        v = np.abs(rng.standard_normal(n)) + 0.05

    if n > 5:
        kern = np.ones(5, dtype=np.float64) / 5.0
        v = np.convolve(v, kern, mode="same")

    v = np.maximum(v, 0.0)
    if float(np.sum(v)) <= _EPS:
        v = np.ones(n, dtype=np.float64)
    return v


def make_tail_boost(base, lam, power):
    n = base.size
    if n <= 1:
        return base.copy()
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    return base * (1.0 + float(lam) * (t ** float(power)))


def try_add_candidate(pool, a, kind, max_score=3.5):
    a = clip_valid(np.asarray(a, dtype=np.float64))
    if a.size < 16:
        return

    sc, _, mx, _ = analyse(a)
    if not np.isfinite(sc) or sc > max_score:
        return

    if mx > 0.0:
        a = a / np.sqrt(mx)

    pool.append((float(sc), str(kind), int(a.size), a))


def choose_starts(pool, max_starts=5, rng=None):
    if not pool:
        return []

    pool.sort(key=itemgetter(0))

    starts = []
    seen = set()

    for entry in pool:
        key = (entry[1], entry[2])
        if key in seen:
            continue
        starts.append(entry[3].copy())
        seen.add(key)
        if len(starts) >= int(max_starts):
            break

    # Ensure at least one exploratory random-like start is present.
    has_rand = False
    for entry in pool[:20]:
        if "rand" in str(entry[1]):
            has_rand = True
            break
    if not has_rand and rng is not None:
        n = 2048
        if pool:
            n = min(4096, max(1024, int(pool[0][2])))
        starts.append(make_smooth_random_sequence(n, 0, rng))

    return starts


def compute_h(a, w):
    n = a.size
    # h_i = sum_l a_l w_{i+l}, for i=0..n-1.
    c = convolve(a[::-1], w)
    return c[n - 1:2 * n - 1]


def lp_direction_reduced(a, c, rng, max_rows=800):
    if linprog is None:
        return None

    n = a.size
    total = 2 * n - 1
    if total <= 0:
        return None

    max_rows = int(max(64, min(total, int(max_rows))))

    if total <= 1024:
        ks = np.arange(total, dtype=np.int64)
    else:
        top_k = max(1, int(max_rows * 0.65))
        if top_k >= total:
            part = np.arange(total, dtype=np.int64)
        else:
            part = np.argpartition(c, -top_k)[-top_k:]

        even_k = max(0, max_rows - part.size)
        if even_k > 0:
            even = np.linspace(0, total - 1, even_k).round().astype(np.int64)
            ks = np.unique(np.concatenate((part.astype(np.int64), even)))
            if ks.size > max_rows:
                idx = rng.choice(ks.size, size=max_rows, replace=False)
                ks = ks[idx]
        else:
            ks = part.astype(np.int64)
            if ks.size > max_rows:
                idx = rng.choice(ks.size, size=max_rows, replace=False)
                ks = ks[idx]

    ks = np.ascontiguousarray(ks, dtype=np.int64)
    if ks.size == 0:
        return None

    A = np.zeros((ks.size, n), dtype=np.float64)

    for r in range(ks.size):
        k = int(ks[r])
        if k < n:
            A[r, :k + 1] = a[k::-1]
        else:
            start = k - n + 1
            stop = start - 1
            A[r, start:] = a[n - 1:stop:-1]

    good_rows = np.any(A > 1e-300, axis=1)
    if not np.any(good_rows):
        return None
    A = A[good_rows]

    cvec = -np.ones(n, dtype=np.float64)
    # Tiny random perturbation helps break degeneracy without changing the LP much.
    cvec += 1e-9 * rng.standard_normal(n)

    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            cvec,
            A_ub=A,
            b_ub=np.ones(A.shape[0], dtype=np.float64),
            bounds=bounds,
            method="highs",
            options={"disp": False},
        )
        if getattr(res, "status", None) == 0:
            g = np.asarray(res.x, dtype=np.float64)
            if g.size == n and np.all(np.isfinite(g)):
                s = float(np.sum(g))
                if s > 1e-12:
                    return np.maximum(g, 0.0)
    except Exception:
        pass

    return None


def scaled_candidate_score(b):
    b = clip_valid(b)
    n = b.size
    if n == 0:
        return b, np.inf

    c = convolve(b, b)
    if c.size == 0:
        return b, np.inf

    mx = float(np.max(c))
    s = float(np.sum(b))
    if not np.isfinite(mx) or not np.isfinite(s) or s < 0.01 or mx <= 0.0:
        return b, np.inf

    score = float(2.0 * n * mx / (s * s))
    if not np.isfinite(score):
        return b, np.inf

    return b / np.sqrt(mx), score


def maybe_log(global_best, score, n, origin):
    if score < global_best[2] - 1e-4:
        t = time.monotonic()
        print(f"[t={t:.0f}] {origin} n={n} score={score:.6f}", flush=True)
        global_best[2] = score


def optimize_candidate(a0, global_best, deadline, seed=0, use_lp=True):
    a0 = clip_valid(np.asarray(a0, dtype=np.float64))
    if a0.size < 16:
        sc, _, _, _ = analyse(a0)
        return a0, sc

    rng = np.random.default_rng(int(seed) & ((1 << 64) - 1))

    sc, _, mx, _ = analyse(a0)
    if not np.isfinite(sc) or mx <= 0.0:
        return a0, np.inf

    a = a0 / np.sqrt(mx)
    local_best = a.copy()
    local_best_score = float(sc)
    a_init = a.copy()

    if local_best_score < global_best[1]:
        global_best[0] = a.copy()
        global_best[1] = local_best_score
        maybe_log(global_best, local_best_score, a.size, "init")

    beta = 5.0
    no_improve = 0
    it = 0

    while time.monotonic() < deadline - 3.0:
        it += 1
        n = a.size

        c = convolve(a, a)
        if c.size == 0:
            break

        mx = float(np.max(c))
        s = float(np.sum(a))
        if not np.isfinite(mx) or not np.isfinite(s) or s < 0.01 or mx <= 0.0:
            break

        cur = float(2.0 * n * mx / (s * s))
        a = a / np.sqrt(mx)

        if cur < global_best[1] - 1e-12:
            global_best[0] = a.copy()
            global_best[1] = cur
            maybe_log(global_best, cur, n, "current")

        if cur < local_best_score - 1e-12:
            local_best = a.copy()
            local_best_score = cur

        if it % 400 == 0:
            beta = min(30.0, beta * 1.2)

        # Weights on convolution constraints.  Large beta focuses on active peaks.
        z = beta * c
        zmax = float(np.max(z))
        if not np.isfinite(zmax):
            break
        z -= zmax
        w = np.exp(z)
        wsum = float(np.sum(w))
        if wsum <= _EPS or not np.isfinite(wsum):
            active = (c >= mx * (1.0 - 1e-6)).astype(np.float64)
            if float(np.sum(active)) <= 0.0:
                active[int(np.argmax(c))] = 1.0
            w = active / float(np.sum(active))
        else:
            w = w / wsum

        h = compute_h(a, w)
        force = 2.0 / s - 2.0 * beta * h
        g_grad = force - float(np.min(force))
        g_grad = np.maximum(g_grad, 0.0)
        g_grad = normalized_sum(g_grad, s)

        directions = []

        if use_lp and linprog is not None and time.monotonic() < deadline - 8.0:
            lp_every = 3 if n <= 2500 else 6
            if it % lp_every == 1:
                total = 2 * n - 1
                if n <= 1024:
                    max_rows = total
                elif n <= 1800:
                    max_rows = min(total, 1200)
                elif n <= 3000:
                    max_rows = min(total, 1200)
                else:
                    max_rows = min(total, 700)

                g_lp = lp_direction_reduced(a, c, rng, max_rows=max_rows)
                if g_lp is not None:
                    g_lp = normalized_sum(g_lp, s)
                    g_mix = 0.7 * g_lp + 0.3 * g_grad
                    g_mix = normalized_sum(g_mix, s)
                    directions.append(g_mix)
                    directions.append(g_lp)

        directions.append(g_grad)

        if no_improve > 50:
            noise = 1.0 + 0.15 * rng.standard_normal(n)
            g_rand = np.maximum(a * noise, 0.0)
            g_rand = normalized_sum(g_rand, s)
            directions.append(g_rand)

        accepted = False

        for g in directions[:4]:
            for alpha in (0.10, 0.03):
                b = (1.0 - alpha) * a + alpha * g
                cand, nsc = scaled_candidate_score(b)
                if np.isfinite(nsc) and nsc < local_best_score - 1e-9:
                    a = cand
                    local_best = cand.copy()
                    local_best_score = nsc
                    no_improve = 0
                    accepted = True

                    if nsc < global_best[1] - 1e-12:
                        global_best[0] = cand.copy()
                        global_best[1] = nsc
                        maybe_log(global_best, nsc, n, "step")
                    break
            if accepted:
                break

        if not accepted:
            no_improve += 1

            # Controlled restarts to escape local minima.
            if no_improve == 80 or no_improve == 160 or no_improve == 320:
                strength = 0.35 if no_improve == 80 else (0.70 if no_improve == 160 else 1.20)
                base = a_init
                noise = 1.0 + strength * rng.standard_normal(n)
                b = np.maximum(base * noise, 0.0)
                cand, nsc = scaled_candidate_score(b)
                if np.isfinite(nsc):
                    a = cand
                    local_best = cand.copy()
                    local_best_score = nsc
                    no_improve = 0

            if no_improve > 600:
                if global_best[0] is not None and global_best[0].size == n:
                    base = global_best[0]
                else:
                    base = a_init
                noise = 1.0 + 0.5 * rng.standard_normal(n)
                b = np.maximum(base * noise, 0.0)
                cand, nsc = scaled_candidate_score(b)
                if np.isfinite(nsc):
                    a = cand
                    local_best = cand.copy()
                    local_best_score = nsc
                    no_improve = 0

    return local_best, local_best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    budget = max(10.0, budget)

    search_deadline = start + max(1.0, budget - 30.0)
    reserve_init = min(300.0, max(20.0, budget * 0.25))

    rng = np.random.default_rng(int(seed) & ((1 << 64) - 1))
    pool = []

    lengths = [512, 1024, 2048, 3072, 4096]
    if budget > 2200.0:
        lengths.append(6144)

    gseq = globals().get("height_sequence_1", None)
    if gseq is not None:
        try:
            glen = int(np.size(gseq))
            if 64 <= glen <= 8192 and glen not in lengths:
                lengths.append(glen)
        except Exception:
            pass

    lengths = sorted(set(lengths))

    for n in lengths:
        if time.monotonic() > search_deadline - reserve_init:
            break

        base = make_sqrt_sequence(n)
        try_add_candidate(pool, base, "sqrt")

        for lam in (0.5, 1.2, 2.0):
            try_add_candidate(pool, make_tail_boost(base, lam, 1.0), "tail")
            try_add_candidate(pool, make_tail_boost(base, lam, 2.0), "tail2")

        for alpha in (0.45, 0.5, 0.55):
            try_add_candidate(pool, make_power_sequence(n, alpha), "pow")

        for rmode in (0, 1, 2):
            try_add_candidate(pool, make_smooth_random_sequence(n, rmode, rng), "rand")

        for d in (0, 1):
            try_add_candidate(pool, make_exponential_sequence(n, d, rng), "exp")

    cands = choose_starts(pool, max_starts=5, rng=rng)

    if not cands:
        if budget > 30.0:
            cands = [make_sqrt_sequence(2048), make_tail_boost(make_sqrt_sequence(2048), 1.0, 1.0)]
        else:
            cands = [make_sqrt_sequence(512)]

    # Canonicalize start candidates and initialize global best.
    global_best = [None, np.inf, np.inf]

    cleaned_cands = []
    for a in cands:
        sc, _, mx, _ = analyse(a)
        if np.isfinite(sc) and mx > 0.0:
            a = a / np.sqrt(mx)
            cleaned_cands.append(a)
            if sc < global_best[1]:
                global_best[0] = a.copy()
                global_best[1] = float(sc)
                maybe_log(global_best, float(sc), a.size, "starting")
        else:
            cleaned_cands.append(a.copy())

    cands = cleaned_cands

    if time.monotonic() >= search_deadline - 30.0:
        pass
    else:
        remaining = search_deadline - time.monotonic()
        base_slice = int(max(30, min(150, remaining / (max(1, len(cands)) * 2.0))))

        rounds = 0
        while time.monotonic() < search_deadline - 25.0 and rounds < 12:
            for i, a in enumerate(cands):
                if time.monotonic() > search_deadline - 25.0:
                    break
                slice_s = min(base_slice, int(search_deadline - time.monotonic() - 5.0))
                if slice_s < 25:
                    break

                use_lp = (linprog is not None and a.size <= 4096)
                newa, news = optimize_candidate(
                    a,
                    global_best,
                    time.monotonic() + slice_s,
                    seed=int(seed) + 100 * rounds + i + 1,
                    use_lp=use_lp,
                )
                if newa is not None and newa.size > 0:
                    cands[i] = newa
            rounds += 1
            base_slice = max(25, int(base_slice * 0.9))

        # Extra polish on the global best.
        if global_best[0] is not None and time.monotonic() < search_deadline - 10.0:
            extra = min(240.0, search_deadline - time.monotonic() - 5.0)
            if extra > 30.0:
                use_lp = (linprog is not None and global_best[0].size <= 4096)
                newa, _ = optimize_candidate(
                    global_best[0],
                    global_best,
                    time.monotonic() + extra,
                    seed=int(seed) + 10000,
                    use_lp=use_lp,
                )
                if newa is not None and newa.size > 0:
                    cands[0] = newa

    # Final validation: choose the best by the actual evaluator when possible.
    final_arrays = []
    if global_best[0] is not None:
        final_arrays.append(global_best[0])
    for a in cands:
        final_arrays.append(a)

    # Safe fallbacks.
    for n in (1024, 2048):
        final_arrays.append(make_sqrt_sequence(n))

    best_seq = None
    best_val = np.inf

    for arr in final_arrays:
        seq = [float(x) for x in clip_valid(arr)]
        if not seq:
            continue
        try:
            val = float(evaluate_sequence(seq))
        except Exception:
            val = np.inf

        if np.isfinite(val) and val < best_val:
            best_val = val
            best_seq = seq

    if best_seq is None:
        # Guaranteed valid fallback, not necessarily optimal.
        a = make_sqrt_sequence(512)
        _, _, mx, _ = analyse(a)
        if mx > 0.0:
            a = a / np.sqrt(mx)
        best_seq = [float(x) for x in clip_valid(a)]

    return best_seq
