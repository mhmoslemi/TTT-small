def _get_subgradient(a, m_star, n):
    """Compute the hard subgradient at peak position m_star."""
    g = np.zeros(n)
    if m_star < n:
        g[:m_star + 1] = 2.0 * a[m_star::-1]
    else:
        start = m_star - n + 1
        g[start:] = 2.0 * a[n - 1 - np.arange(n - start)]
    return g

def _soft_subgradient(a, conv, epsilon):
    """Compute soft subgradient using log-sum-exp weights."""
    n = len(a)
    logits = conv / epsilon
    logits -= logits.max()
    p = np.exp(logits)
    p /= p.sum()
    # grad[j] = 2 * sum_m p[m] * a[m-j] for valid m-j
    # = 2 * cross_correlation(a, p)[j]
    # Use FFT: ifft(fft(a) * conj(fft(p))) gives circular correlation
    N = 1
    while N < 2 * n - 1 + n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    corr = np.fft.irfft(A * np.conj(P), N)[:n]
    return 2.0 * corr

    A = np.fft.rfft(a, nfft)
