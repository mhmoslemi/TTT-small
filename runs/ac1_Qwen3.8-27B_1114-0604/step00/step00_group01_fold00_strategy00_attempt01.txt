import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    # --- Security Checks ---
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
        
    try:
        arr = np.array(sequence, dtype=np.float64)
    except Exception:
        return np.inf
        
    valid_types = (int, float, bool)
    if isinstance(sequence, list) and not all(isinstance(x, valid_types) for x in sequence):
        return np.inf
        
    nan_exp_inf = np.logical_or(np.isnan(arr), np.isinf(arr))
    if np.any(nan_exp_inf):
        return np.inf
        
    # defensive clip to keep coefficients nonnegative and within [0, 1000]
    clipped = np.clip(arr, 0.0, 1000.0)
    if np.any(clipped != arr):
        # It passed NaN/inf check, but we must ensure defenseiveness against negative numbers
        if np.any(arr < 0.0):
            return np.inf
            
    nonnegative = np.all(clipped >= 0)
    bounded = np.all(clipped <= 1000)
    
    if not nonnegative or not bounded:
        # Checked via eval function might not care, so do it explicitly.
        return np.inf
        
    n = len(sequence)
    sum_a = np.sum(arr)
    
    # --- Sum Check ---
    if sum_a < 0.01:
        return np.inf
        
    # Convolution using mode='full' produces 2n-1 coefficients
    conv_result = np.convolve(arr, arr)
    max_b = np.max(conv_result)
    
    return float(2 * n * max_b / (sum_a ** 2)) + np.inf
