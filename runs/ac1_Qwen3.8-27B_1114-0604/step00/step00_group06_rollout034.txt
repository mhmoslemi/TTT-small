```python
import numpy as np
from scipy.optimize import linprog
import time

def lp_step(a, M):
    n = len(a)
    # Maximize sum(b) subject to (a*b)_k <= M for all k, b >= 0
    c = -np.ones(n)  # Minimize -sum(b) = maximize sum(b)
    A_ub = []
    b_ub = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            if 0 <= j < n:
                row[j] += a[i]
        A_ub.append(row)
        b_ub.append(M)
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return res.x

def iterative_refine(a, max_iters=1000, t=0.1):
    a = a.copy()
    S = np.sum(a)
    a = a / S  # Normalize to sum = 1
    for it in range(max_iters):
        conv = np.convolve(a, a)
        M = np.max(conv)
        b = lp_step(a, M)
        if b is None:
            break
        B = np.sum(b)
        if B <= S * (1 + 1e-10):
            break  # No improvement possible
        b = b / B * S  # Rescale to same sum
        a_new = (1-t)*a + t*b
        # Check improvement
        conv_new = np.convolve(a_new, a_new)
        if np.max(conv_new) < M:
            a = a_new
    return a
```

```python
import numpy as np

import time

def compute_conv(a):
    return np.convolve(a, a)

def max_conv(a):
    return np.max(np.convolve(a, a))

def lp_solve(a, M):
    """Solve the LP: max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    c = -np.ones(n)
    # Build constraint matrix
    # (a*b)_k = sum_i a_i * b_{k-i} for valid i, k-i
    # For each k, the coefficient of b_j is a_{k-j} (if k-j is valid)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                  bounds=[(0, np.inf)]*n, method='highs')
    if res.status != 0:
        return None
    return res.x
```
