def project_simplex(v, S):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def project_simplex(v, S):
    """Project v onto the simplex {x >= 0, sum(x) = S}."""
    n = len(v)

import numpy as np
import time


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1.0) > (cssv - S)
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_convolve(a):
    """Compute autoconvolution a*a using FFT (returns length 2n-1)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]


def compute_objective(a):
    """Compute 2n * max(a*a) assuming sum(a)=1."""
    n = len(a)
    c = fft_convolve(a)
    return 2.0 * n * c.max()

    time_per_start = (phase1_deadline - time.monotonic()) / num_starts
    mu_schedule = [50, 200, 1000]
    
    fista_best = best_overall.copy()
    fista_best_score = best_score
    
    for i, name in enumerate(start_names):
        if time.monotonic() >= phase1_deadline:
            break
        local_end = min(time.monotonic() + time_per_start, phase1_deadline)
        print(f"[FISTA {i+1}/{num_starts}] {name}, time={time_per_start:.0f}s")
        
        a_curr = starts[name].copy()
        for mu in mu_schedule:
            if time.monotonic() >= local_end:
                break
            seg_end = local_end - (local_end - time.monotonic()) * (mu_schedule.index(mu) + 1) / len(mu_schedule)
            seg_end = min(seg_end, local_end)
            if time.monotonic() >= seg_end:
                continue
            a_curr, sc = fista_smoothed(a_curr, n, mu, seg_end)
        
        # Final evaluation
        final_sc = compute_objective(a_curr)
        if final_sc < fista_best_score:
            fista_best = a_curr.copy()
            fista_best_score = final_sc
            print(f"  FISTA {name}: {final_sc:.6f} (new best)")
    
    # Phase 2: Subgradient polish on best FISTA result (30% of time)
    phase2_deadline = time.monotonic() + (budget_s - 30.0) * 0.3
    print(f"\n[Phase 2] Subgradient polish, time={phase2_deadline - time.monotonic():.0f}s")
    
    a_best, sc = subgradient_polish(fista_best, n, phase2_deadline, verbose=True)
    if sc < fista_best_score:
        fista_best = a_best.copy()
        fista_best_score = sc
    
    if sc < best_score:
        best_overall = fista_best.copy()
        best_score = sc
        print(f"  Phase 2 best: {best_score:.6f}")
    
    # Also try subgradient on the original parent
    if time.monotonic() < phase2_deadline:
        a_best2, sc2 = subgradient_polish(parent_norm, n, phase2_deadline, verbose=False)
        if sc2 < best_score:
            best_overall = a_best2.copy()
            best_score = sc2
            print(f"  Subgrad on parent: {best_score:.6f}")
    
    # Phase 3: Final polish with smaller steps (10% of time)
    phase3_deadline = deadline
    remaining = phase3_deadline - time.monotonic()
    if remaining > 20:
        print(f"\n[Phase 3] Final polish, time={remaining:.0f}s")
        # Try with different step scales
        for ss in [0.15, 0.25, 0.5, 0.8]:
            if time.monotonic() >= phase3_deadline:
                break
            seg = phase3_deadline - (phase3_deadline - time.monotonic()) * 0.3
            a_best3, sc3 = subgradient_polish(best_overall, n, seg, verbose=False)
            if compute_objective(a_best3) < best_score:
                # Verify
                ver = float(evaluate_sequence(a_best3.tolist()))
                if np.isfinite(ver) and ver < best_score:
                    best_overall = a_best3.copy()
                    best_score = ver
                    print(f"  Polish ss={ss}: {best_score:.6f}")
    
    # Final verification
    final_score = float(evaluate_sequence(best_overall.tolist()))
    print(f"\nFinal verified score: {final_score:.6f}")
    return [float(x) for x in best_overall]
