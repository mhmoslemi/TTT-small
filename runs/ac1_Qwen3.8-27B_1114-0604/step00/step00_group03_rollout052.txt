def _fft_conv(a, L):
    """Compute linear convolution a*a using FFT of length L."""
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)

def _compute_gradient(a, w, L):
    """
    Compute gradient of sum_k w_k * (a*a)_k w.r.t. a.
    w is the weight vector (length 2n-1).
    Returns gradient of length n.
    """
    n = len(a)
    # Build triangle vector t of length 2n-1
    # t[m] = a[abs(m - (n-1))] for m in 0..2n-2
    idx = np.abs(np.arange(2*n-1) - (n-1))
    t = a[idx]
    
    # Cross-correlation of w and t:
    # corr[j] = sum_m w[m+j] * t[m] (linear, zero-padded)
    W = np.fft.rfft(w, L)
    T = np.fft.rfft(t, L)
    corr = np.fft.irfft(W * np.conj(T), L)
    
    # grad[i] = 2 * corr[i - (n-1)]
    indices = np.arange(n) - (n-1)
    grad = 2.0 * corr[indices]
    return grad
