def next_pow2(x):

    p=1
    while p < x: p <<=1

def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - float(np.mean(z))
    m = float(np.max(z))
    e = np.exp(z - m)
    s = float(np.sum(e))
    if s <=0 or not np.isfinite(s):
       return np.ones(z.size)/z.size
    return e/s

def score_only_z(z,n,N):
    a=softmax_from_z(z)
    A=np.fft.rfft(a,N)
    b=np.fft.irfft(A*A,N)[:2*n-1]
    if b.size: b=np.where(b>0,b,0.0)
    return float(2.0*n*np.max(b))

def objective_grad_z(z,T,n,N):
    a=softmax_from_z(z)
    A=np.fft.rfft(a,N)
    b=np.fft.irfft(A*A,N)[:2*n-1]
    b = np.where(b > 0.0, b, 0.0)
    c = n*b
    max_c = float(np.max(c))
    score = 2.0*max_c
    mm = max_c
    y = (c-mm)/T
    # To avoid overflow/underflow, y max 0
    ex = np.exp(y)
    se = float(np.sum(ex))
    if se <=0: # fallback
       F = 0.0
       s = np.ones(2*n-1)/ (2*n-1)
    else:
       F = mm/T + math.log(se)
       s = ex/se
    S = np.fft.rfft(s, N)
    corr = np.fft.irfft(S * np.conj(A), N).real[:n]
    G = (2.0*n/T)*corr
    dot = float(np.dot(G,a))
    grad = a * (G - dot)
    grad = grad - float(np.mean(grad))
    # if nonfinite, fallback
    if not np.all(np.isfinite(grad)):
       grad = np.zeros(n)
    return F, score, grad

```python
"""
Local softmax gradient optimizer for non-negative low-convolution-peak sequences.

The returned sequence is a point found by:
  1. Generating many smooth/spectral shape initializations for several lengths.
  2. Optimizing log-softmax variables with a temperature-annealed log-sum-exp
     surrogate for the peak of the full linear self-convolution.
  3. Refining the best points with small log-domain local search.

The objective used internally is scale-invariant:
    score(a) = 2*n*max(convolve(a,a)) / sum(a)^2.
For softmax parameters, sum(a)=1, so minimizing score is equivalent to
minimizing max(n * convolve(a,a)).
"""

import numpy as np
import time
import math


def next_pow2(x):
    """Return the smallest power of two >= x."""
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def stable_softmax(z):
    """Stable softmax with input mean removed."""
    z = np.asarray(z, dtype=np.float64).copy()
    if z.size == 0:
        return z
    z -= float(np.mean(z))
    m = float(np.max(z))
    e = np.exp(z - m)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def score_only_logz(z, n, N):
    """Return internal score for softmax(z), using FFT convolution."""
    a = stable_softmax(z)
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)[: (2 * n - 1)]
    b = np.where(b > 0.0, b, 0.0)
    if b.size == 0:
        return float("inf")
    return float(2.0 * n * float(np.max(b)))


def score_only_a(a):
    """Return internal score for a nonnegative array."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    n = a.size
    N = next_pow2(2 * n)
    A = np.fft.rfft(np.asarray(a, dtype=np.float64) / s, N)
    b = np.fft.irfft(A * A, N)[: (2 * n - 1)]
    b = np.where(b > 0.0, b, 0.0)
    # Since a/s is normalized, score = 2*n*max(b)
    return float(2.0 * n * float(np.max(b)))


def objective_grad_logz(z, T, n, N):
    """
    Return smooth objective, true score, and gradient w.r.t. z for softmax(z).
    Smooth objective is logsumexp(n * conv(a,a) / T).
    """
    a = stable_softmax(z)
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)[: (2 * n - 1)]
    b = np.where(b > 0.0, b, 0.0)
    c = n * b

    max_c = float(np.max(c)) if c.size else 0.0
    score = 2.0 * max_c

    y = (c - max_c) / T
    ex = np.exp(y)
    se = float(np.sum(ex))
    if not np.isfinite(se) or se <= 0.0:
        F = 0.0
        soft = np.ones(2 * n - 1, dtype=np.float64) / float(2 * n - 1)
    else:
        F = max_c / T + math.log(se)
        soft = ex / se

    S = np.fft.rfft(soft, N)
    corr = np.fft.irfft(S * np.conj(A), N).real[:n]

    # d/d a of logsumexp(c/T), where c = n*conv(a,a)
    G_a = (2.0 * float(n) / float(T)) * corr
    dot = float(np.dot(G_a, a))

    # softmax chain rule: grad_z_k = a_k * (G_a_k - sum_i a_i G_a_i)
    grad_z = a * (G_a - dot)
    grad_z -= float(np.mean(grad_z))

    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros(n, dtype=np.float64)

    if not np.isfinite(F):
        F = float(max_c / max(T, 1e-9))
    if not np.isfinite(score):
        score = float("inf")

    return float(F), float(score), grad_z


def random_spectral_z(n, rng, order):
    """Generate a smooth random log-softmax vector using low Fourier modes."""
    z = np.zeros(n, dtype=np.float64)
    if n < 8:
        return z

    K = min(80, n // 2)
    x = np.arange(n, dtype=np.float64) / float(n)
    for k in range(1, K + 1):
        amp = float(rng.standard_normal()) / float((float(k) ** order))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z += amp * np.sin(2.0 * np.pi * float(k) * x + ph)

    z -= float(np.mean(z))
    return np.clip(z, -50.0, 50.0)


def make_z_shape(name, param, n, rng):
    """Create a log-softmax vector from a named parametric shape."""
    x = np.arange(n, dtype=np.float64)
    c = 0.5 * float(n - 1)

    if name == "uniform":
        a = np.ones(n, dtype=np.float64)

    elif name == "exp_left":
        a = np.exp(-float(param) * x / float(max(1, n)))

    elif name == "exp_right":
        a = np.exp(-float(param) * (float(n - 1) - x) / float(max(1, n)))

    elif name == "exp_center":
        a = np.exp(-float(param) * np.abs(x - c) / float(max(1, n)))

    elif name == "exp_edges":
        edge = np.abs(x - c) / float(max(1, c + 1.0))
        a = np.exp(-float(param) * (1.0 - edge))

    elif name == "power_left":
        a = (x + 1.0) ** float(param)

    elif name == "power_right":
        a = (float(n) - x) ** float(param)

    elif name == "sine":
        a = np.sin(np.pi * (x + 0.5) / float(n))
        a = np.maximum(a, 1e-12)

    elif name == "sine2":
        a = np.sin(np.pi * (x + 0.5) / float(n))
        a = a * a + 1e-12

    elif name == "cos":
        a = np.cos(np.pi * x / float(max(1, n)))
        a = a * a + 1e-12

    elif name == "triangle":
        a = 1.0 - np.abs(x - c) / float(max(1, c + 1.0))
        a = np.maximum(a, 1e-12)

    elif name == "quad":
        t = (2.0 * (x + 0.5) / float(n)) - 1.0
        a = 1.0 - t * t
        a = np.maximum(a, 1e-12)

    elif name == "gauss":
        sigma = max(1.0, float(param) * float(n))
        a = np.exp(-0.5 * ((x - c) / sigma) ** 2)

    elif name == "bimodal":
        w = max(1.0, float(param) * float(n))
        a = (
            np.exp(-(((x - 0.25 * float(n - 1)) / w) ** 2)) +
            np.exp(-(((x - 0.75 * float(n - 1)) / w) ** 2)) +
            1e-12
        )

    elif name == "random":
        return random_spectral_z(n, rng, float(param if param > 0 else 1.0))

    else:
        a = np.ones(n, dtype=np.float64)

    a = np.asarray(a, dtype=np.float64)
    if not np.all(np.isfinite(a)) or float(np.sum(a)) <= 0.0:
        a = np.ones(n, dtype=np.float64)

    a = np.clip(a, 1e-300, 1e300)
    z = np.log(a)
    z -= float(np.mean(z))
    return np.clip(z, -50.0, 50.0)


def z_from_base_sequence(base, n):
    """Resample/pad a base sequence to length n and convert to log-softmax."""
    base = np.asarray(base, dtype=np.float64).ravel()
    base = np.clip(base, 0.0, 1000.0)
    if base.size == 0:
        return np.zeros(n, dtype=np.float64)

    if base.size >= n:
        a = base[:n].copy()
    else:
        a = np.empty(n, dtype=np.float64)
        a[: base.size] = base
        a[base.size:] = base[-1]

    a = np.clip(a, 1e-12, 1000.0)
    if float(np.sum(a)) <= 0.0:
        a = np.ones(n, dtype=np.float64)

    z = np.log(a)
    z -= float(np.mean(z))
    return np.clip(z, -50.0, 50.0)


def get_global_initial_sequence():
    """Return height_sequence_1 if present and valid, otherwise None."""
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        a = np.asarray(seq, dtype=np.float64).ravel()
        a = np.clip(a, 0.0, 1000.0)
        if a.size > 0 and float(np.sum(a)) > 0.0:
            return a
    except Exception:
        pass
    return None


def build_shape_defs():
    """Return a list of (name, param) shapes to try."""
    defs = []

    defs.append(("uniform", 0.0))

    for p in (0.5, 1.0, 2.0, 4.0, 8.0):
        defs.append(("exp_left", p))
        defs.append(("exp_right", p))
        defs.append(("exp_center", p))
        defs.append(("exp_edges", p))

    for p in (-2.0, -1.0, -0.5, 0.25, 0.5, 1.0, 2.0, 4.0):
        defs.append(("power_left", p))
        defs.append(("power_right", p))

    defs.append(("sine", 0.0))
    defs.append(("sine2", 0.0))
    defs.append(("cos", 0.0))
    defs.append(("triangle", 0.0))
    defs.append(("quad", 0.0))

    for p in (0.02, 0.05, 0.1, 0.2, 0.4):
        defs.append(("gauss", p))

    for p in (0.05, 0.1, 0.2):
        defs.append(("bimodal", p))

    for order in (0.5, 1.0):
        for _ in range(18):
            defs.append(("random", order))

    return defs


def candidate_key(item):
    """Sort key for candidate tuples (score, z, n)."""
    return item[0]


def improve_by_adam_logz(z0, n, budget_seconds, rng):
    """
    Anneal temperature and run Adam on softmax log-variables.

    Returns (best_z, best_score).
    """
    z0 = np.asarray(z0, dtype=np.float64).copy()
    z0 -= float(np.mean(z0))
    z0 = np.clip(z0, -50.0, 50.0)

    N = next_pow2(2 * n)
    best_score = score_only_logz(z0, n, N)
    best_z = z0.copy()

    if budget_seconds <= 0.0:
        return best_z, best_score

    quick_Ts = (0.35, 0.2, 0.12, 0.07, 0.045, 0.03, 0.022)
    lrs = (0.9, 0.35)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-7
    deadline = time.monotonic() + float(budget_seconds)

    max_iter_phase = int(max(30, min(650, 40 + 8.0 * float(budget_seconds))))

    for T in quick_Ts:
        if time.monotonic() >= deadline:
            break
        for lr in lrs:
            if time.monotonic() >= deadline:
                break

            z = best_z.copy()
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            t = 0
            stall = 0
            phase_stop = time.monotonic() + max(1.0, budget_seconds / 64.0)

            while t < max_iter_phase and time.monotonic() < phase_stop and time.monotonic() < deadline:
                F, score, grad = objective_grad_logz(z, T, n, N)
                if not np.isfinite(score):
                    break

                if score < best_score - 1e-12:
                    best_score = score
                    best_z = z.copy()
                    stall = 0
                else:
                    stall += 1

                t += 1
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                mhat = m / (1.0 - beta1 ** t)
                vhat = v / (1.0 - beta2 ** t)

                z = z - lr * mhat / (np.sqrt(vhat) + eps)

                if t % 120 == 0:
                    z = z + rng.normal(0.0, 2e-4, n)

                z = np.where(np.isfinite(z), z, 0.0)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))

                if stall > 120:
                    break

            # If this lr helped, do not try the smaller lr for same T.
            score_now = score_only_logz(z, n, N)
            if np.isfinite(score_now) and score_now < best_score:
                best_score = score_now
                best_z = z.copy()
                break

    final_score = score_only_logz(best_z, n, N)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    return best_z, best_score


def local_search_logz(z0, n, budget_seconds, rng):
    """Small log-domain block perturbation refinement."""
    z0 = np.asarray(z0, dtype=np.float64).copy()
    z0 -= float(np.mean(z0))
    N = next_pow2(2 * n)

    best_z = z0.copy()
    best_score = score_only_logz(best_z, n, N)
    if budget_seconds <= 0.0:
        return best_z, best_score

    deadline = time.monotonic() + float(budget_seconds)
    sigma = 0.35
    no_improve = 0

    while time.monotonic() < deadline:
        cand = best_z.copy()

        wmin = max(3, int(n * 0.002))
        wmax = max(wmin + 2, int(n * 0.25))
        width = int(rng.integers(wmin, wmax))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        cand[start:stop] += rng.normal(0.0, sigma, stop - start)

        if rng.random() < 0.18:
            cand += rng.normal(0.0, 0.05 * sigma, n)

        if rng.random() < 0.25:
            cand = 0.5 * (cand + cand[::-1].copy())

        cand = np.where(np.isfinite(cand), cand, 0.0)
        cand = np.clip(cand, -50.0, 50.0)
        cand -= float(np.mean(cand))

        score = score_only_logz(cand, n, N)
        if np.isfinite(score) and score < best_score - 1e-12:
            best_z = cand
            best_score = score
            sigma = min(0.8, sigma * 1.08)
            no_improve = 0
        else:
            sigma *= 0.965
            no_improve += 1
            if sigma < 1e-4 or no_improve > 250:
                break

    return best_z, best_score


def recombine_score(z1, z2, w, n):
    """Score a convex blend of two log-variables of same length."""
    cand = (1.0 - float(w)) * np.asarray(z1, dtype=np.float64) + float(w) * np.asarray(z2, dtype=np.float64)
    cand -= float(np.mean(cand))
    N = next_pow2(2 * n)
    return score_only_logz(cand, n, N)


def propose_candidate(budget_s=2400, seed=42, **kwargs):
    """Search for a valid non-negative sequence with low evaluation score."""
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    budget = max(2.0, budget)
    deadline = start_time + budget - 5.0

    rng = np.random.default_rng(seed)
    base = get_global_initial_sequence()

    # Choose lengths. Include the provided construction if useful, and some
    # FFT-friendly lengths.
    n_list = []
    if base is not None and 256 <= base.size <= 12000:
        n_list.append(int(base.size))
    for L in (2048, 4096, 8192, 12288):
        if 256 <= L <= 12000:
            dup = False
            for m in n_list:
                if abs(int(m) - int(L)) <= 0.12 * int(L):
                    dup = True
                    break
            if not dup:
                n_list.append(L)

    n_list = list(dict.fromkeys(n_list))
    if len(n_list) > 4:
        n_list = n_list[:4]
    if not n_list:
        n_list = [4096]

    shape_defs = build_shape_defs()
    candidates = []

    # Generate and score initial candidates.
    for n in n_list:
        N = next_pow2(2 * n)

        # Use base sequence if available.
        if base is not None:
            z = z_from_base_sequence(base, n)
            sc = score_only_logz(z, n, N)
            if np.isfinite(sc):
                candidates.append((sc, z, n))

        # Parametric and random spectral shapes.
        for name, param in shape_defs:
            if time.monotonic() > deadline:
                break
            z = make_z_shape(name, param, n, rng)
            sc = score_only_logz(z, n, N)
            if np.isfinite(sc):
                candidates.append((sc, z, n))

    if not candidates:
        n = 1024
        a = np.ones(n, dtype=np.float64)
        return [float(v) for v in a]

    candidates.sort(key=candidate_key)

    # Ensure diversity: keep top overall and best per length.
    selected = candidates[:12]
    best_by_n = {}
    for item in candidates:
        n = item[2]
        if n not in best_by_n:
            best_by_n[n] = item

    for item in best_by_n.values():
        already = False
        for existing in selected:
            if existing[2] == item[2] and abs(existing[0] - item[0]) < 1e-12:
                already = True
                break
        if not already and len(selected) < 18:
            selected.append(item)

    best_score = float(selected[0][0])
    best_z = np.asarray(selected[0][1], dtype=np.float64).copy()
    best_n = int(selected[0][2])

    # Improve selected candidates in sequence, sharing remaining time.
    selected = selected[: min(16, len(selected))]
    for idx, item in enumerate(selected):
        rem = deadline - time.monotonic()
        if rem <= 1.0:
            break

        remaining_after = max(1, len(selected) - idx - 1)
        share = rem / float(remaining_after + 1)
        share = min(share, max(4.0, 0.9 * rem / 3.0))
        if share < 2.0:
            share = max(1.0, 0.5 * rem)

        n = int(item[2])
        z = np.asarray(item[1], dtype=np.float64).copy()

        # Adam refinement.
        if time.monotonic() < deadline:
            z, sc = improve_by_adam_logz(z, n, max(1.0, 0.78 * share), rng)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_z = z.copy()
                best_n = n

        # Local refinement.
        if time.monotonic() < deadline:
            z, sc = local_search_logz(z, n, max(0.5, 0.22 * share), rng)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_z = z.copy()
                best_n = n

    # Recombination among top solutions of the same length, if time remains.
    if time.monotonic() < deadline:
        # Trim selected to best 10 and group by length.
        groups = {}
        for item in selected[:10]:
            n = int(item[2])
            groups.setdefault(n, []).append(np.asarray(item[1], dtype=np.float64))

        for n, zs in groups.items():
            if len(zs) < 2:
                continue
            N = next_pow2(2 * n)
            for i in range(len(zs)):
                if time.monotonic() > deadline:
                    break
                for j in range(i + 1, len(zs)):
                    if time.monotonic() > deadline:
                        break
                    for w in (0.35, 0.5, 0.65):
                        if time.monotonic() > deadline:
                            break
                        cand = (1.0 - float(w)) * zs[i] + float(w) * zs[j]
                        cand -= float(np.mean(cand))
                        cand = np.clip(cand, -50.0, 50.0)
                        sc = score_only_logz(cand, n, N)
                        if np.isfinite(sc) and sc < best_score:
                            best_score = float(sc)
                            best_z = cand.copy()
                            best_n = n
                            # Continue from this candidate briefly.
                            if time.monotonic() < deadline:
                                cand, sc2 = local_search_logz(cand, n, min(3.0, 0.05 * (deadline - time.monotonic())), rng)
                                if np.isfinite(sc2) and sc2 < best_score:
                                    best_score = float(sc2)
                                    best_z = cand.copy()
                                    best_n = n

    # Final long local search on the current best solution.
    rem = deadline - time.monotonic()
    if rem > 3.0:
        best_z, best_score = local_search_logz(best_z, best_n, rem * 0.95, rng)

    # Convert to a concrete nonnegative sequence.
    a = stable_softmax(best_z)
    a = np.clip(a, 0.0, 1000.0)

    # If the runtime evaluator is available, use it as a final sanity check.
    eval_fn = globals().get("evaluate_sequence", None)
    if eval_fn is not None:
        try:
            eval_score = float(eval_fn([float(x) for x in a]))
            if not np.isfinite(eval_score):
                a = np.ones(1024, dtype=np.float64)
        except Exception:
            pass

    return [float(x) for x in a]
```
