    max_c = conv.max()
    return T * (max_c + np.log(np.sum(np.exp(T * (conv - max_c)))))

def soft_max_conv_grad(a, T):
    """Compute the gradient of T * log-sum-exp(conv(a,a)) with respect to a."""
