def _fft_len(m):

```python
"""
Search for non-negative step-function coefficients minimizing

    2 * n * max(a * a) / (sum(a)**2)

by optimizing the shape of the sequence in log-space.  The main idea is that
non-symmetric sequences can make the largest lag of the linear autoconvolution
smaller than for the symmetric uniform sequence.  The optimizer uses an
annealed p-norm surrogate of the convex-in-a max constraint, estimated by
smooth weights proportional to conv_k ** (p - 1).  The sequence is kept
positive and normalized, and updates are performed multiplicatively in log-space
with Adam-style momentum.  Multiple non-symmetric initializations, resampling,
and perturbation restarts are used to avoid symmetric local optima.
"""

import numpy as np
import time


def _normalize_positive(a):
    """Return a finite non-negative 1D array with sum 1, or a small uniform array."""
    try:
        arr = np.asarray(a, dtype=np.float64)
    except Exception:
        arr = np.ones(64, dtype=np.float64)

    if arr.ndim != 1:
        arr = arr.ravel()

    if arr.size == 0:
        arr = np.ones(64, dtype=np.float64)

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)

    s = float(np.sum(arr))
    if (not np.isfinite(s)) or s <= 1e-300:
        arr = np.full(arr.size if arr.size > 0 else 64, 1.0 / max(1, arr.size if arr.size > 0 else 64), dtype=np.float64)
    else:
        arr = arr / s

    return arr


def _full_conv_len(a, L):
    """Linear auto-convolution computed by FFT with pre-allocated FFT length L."""
    n = len(a)
    A = np.fft.rfft(a, L)
    out = np.fft.irfft(A * A, L)
    return out[:2 * n - 1]


def _corr_len(p, a, L):
    """
    Compute corr[j] = sum_k p[k] * a[k-j] for j = 0..n-1.
    This is the gradient contribution for peak-weighted convolution objectives.
    """
    n = len(a)
    P = np.fft.rfft(p, L)
    Arev = np.fft.rfft(a[::-1], L)
    out = np.fft.irfft(P * Arev, L)
    return out[n - 1:n - 1 + n]


def _add_perturbation(a, sigma, rng):
    """Multiplicative log-normal perturbation, renormalized to sum 1."""
    a = _normalize_positive(a)
    n = len(a)
    x = np.clip(sigma * rng.standard_normal(n), -10.0, 10.0)
    b = a * np.exp(x)
    return _normalize_positive(b)


def _resample_sequence(a, n_new):
    """Linearly resample a normalized sequence to a new length."""
    a = _normalize_positive(a)
    n_old = len(a)
    n_new = int(max(1, n_new))
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    b = np.interp(x_new, x_old, a)
    return _normalize_positive(b)


def _pattern_initial(n, kind, rng):
    """Create mostly non-symmetric positive starting shapes."""
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)

    if kind == "saw_down":
        a = 1.0 + 0.90 * (0.5 - x)
    elif kind == "saw_up":
        a = 1.0 + 0.90 * (x - 0.5)
    elif kind == "exp_down":
        a = np.exp(-2.0 * x)
    elif kind == "exp_up":
        a = np.exp(2.0 * x)
    elif kind == "bump_left":
        a = 0.05 + np.exp(-9.0 * (x - 0.25) ** 2)
    elif kind == "bump_right":
        a = 0.05 + np.exp(-9.0 * (x - 0.75) ** 2)
    elif kind == "quad_left":
        a = 1.0 + 0.80 * (x - 0.5) - 0.90 * (x - 0.5) ** 2
    elif kind == "quad_right":
        a = 1.0 - 0.80 * (x - 0.5) - 0.90 * (x - 0.5) ** 2
    else:
        a = 1.0

    # For diversity, also expose a random low-pass non-symmetric initializer.
    if kind == "random":
        a = 1.0
        r1 = rng.random()
        r2 = rng.random()
        if r1 < 0.45:
            slope = rng.uniform(0.35, 1.65)
            if r2 < 0.5:
                slope = -slope
            a += slope * (x - 0.5)
        elif r1 < 0.75:
            curv = rng.uniform(0.4, 1.5)
            if r2 < 0.5:
                curv = -curv
            a += curv * (x - 0.5) ** 2
        else:
            bump_loc = rng.uniform(0.15, 0.85)
            bump_amp = rng.uniform(0.4, 1.6)
            a += bump_amp * np.exp(-8.0 * (x - bump_loc) ** 2)

        n_modes = int(rng.integers(1, 3))
        for _ in range(n_modes):
            freq = int(rng.integers(1, 4))
            amp = rng.uniform(0.10, 0.55)
            phase = rng.uniform(0.0, 2.0 * np.pi)
            a += amp * np.cos(2.0 * np.pi * freq * x + phase)

    a = np.clip(a, 0.05, None)
    a += 0.025 * rng.standard_normal(n)
    return _normalize_positive(a)


def _random_initial(n, rng):
    """Random non-symmetric positive initial sequence."""
    return _pattern_initial(n, "random", rng)


def _make_initial_states(parent_seq, rng):

    states = []

    if parent_seq is not None:
        parent_seq = _normalize_positive(parent_seq)
        n_parent = len(parent_seq)
        if n_parent <= 4096:
            states.append((parent_seq.copy(), "parent_native"))
        for n in [256, 512, 1024, 2048]:
            if n != n_parent:
                states.append((_resample_sequence(parent_seq, n), "parent_resamp_%d" % n))

    length_plan = [
        (96, 2),
        (192, 2),
        (384, 3),
        (768, 3),

        (3072, 2),
    ]
    patterns = [
        "saw_down",
        "saw_up",
        "exp_down",

        "quad_left",

    for n, count in length_plan:

def _optimize_slice(a_init, slice_s, rng, p0, p1, lr, tag=""):
    """
    Optimize one sequence for a fixed wall-clock slice.

    p is annealed from p0 to p1.  For a given p, weights for convolution lags
    are proportional to conv^(p-1), approximating the gradient of the p-norm
    surrogate of max(conv).  Updates are in log-space via Adam-like momentum.
    """
    a = _normalize_positive(a_init)
    n = len(a)
    if slice_s <= 0.0:
        conv = _full_conv_len(a, 1 << (2 * n - 1).bit_length())
        score = 2.0 * n * float(np.max(conv))
        return a, score

    L_conv = 1 << (2 * n - 1).bit_length()
    L_corr = 1 << (3 * n - 2).bit_length()

    best_a = a.copy()
    conv = _full_conv_len(a, L_conv)
    conv = np.maximum(conv, 0.0)
    best_score = 2.0 * n * float(np.max(conv))
    if not np.isfinite(best_score):
        best_score = 2.0

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    step = 0

    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    min_lr = 0.002
    max_lr = 0.25
    current_lr = float(lr)

    no_improve = 0
    start = time.monotonic()
    deadline = start + float(slice_s)
    tiny = 1e-300

    while time.monotonic() < deadline:
        conv = _full_conv_len(a, L_conv)
        conv = np.maximum(conv, 0.0)
        cmax = float(np.max(conv))
        score = 2.0 * n * cmax
        score = float(score) if np.isfinite(score) else float(best_score)

        if np.isfinite(score) and score < best_score * (1.0 - 1e-8):
            best_score = score
            best_a = a.copy()
            no_improve = 0
            current_lr = min(max_lr, current_lr * 1.015)
        else:
            no_improve += 1
            if no_improve > 300:
                current_lr = max(min_lr, current_lr * 0.96)
            if no_improve > 8000:
                break

        elapsed = max(1e-6, time.monotonic() - start)
        frac = min(1.0, elapsed / max(1e-6, float(slice_s)))
        p = float(p0) + (float(p1) - float(p0)) * (frac ** 1.25)

        if cmax <= tiny:
            w = np.ones(2 * n - 1, dtype=np.float64) / float(2 * n - 1)
        else:
            floor = cmax * 1e-15
            logw = (p - 1.0) * (np.log(np.maximum(conv, floor)) - np.log(cmax))
            logw = np.clip(logw, -745.0, 0.0)
            w = np.exp(logw)
            ws = float(np.sum(w))

            if (not np.isfinite(ws)) or ws <= 0.0:
                w = np.zeros(2 * n - 1, dtype=np.float64)
                w[int(np.argmax(conv))] = 1.0
            else:
                w = w / ws

                # If stuck, mix in a top-lag hard-gradient component.
                if no_improve > 700 and (step % 5 == 0):
                    topk = max(1, (2 * n - 1) // 20)
                    if topk < len(conv):
                        idx = np.argpartition(conv, -topk)[-topk:]
                        topw = np.zeros_like(conv)
                        topw[idx] = 1.0 / float(topk)
                    else:
                        topw = np.ones_like(conv) / float(len(conv))
                    w = 0.5 * w + 0.5 * topw
                    ws2 = float(np.sum(w))
                    if ws2 > 0.0 and np.isfinite(ws2):
                        w = w / ws2

        corr = _corr_len(w, a, L_corr)
        corr = np.maximum(corr, 0.0)
        ga = 2.0 * corr
        tdot = float(np.dot(a, ga))
        gz = a * (ga - tdot * a)

        # Occasionally inject a tiny log-space perturbation if the trajectory is stagnant.
        if no_improve > 500 and (step % 25 == 0):
            nz = rng.standard_normal(n)
            gnorm = float(np.linalg.norm(gz))
            if gnorm > 0.0 and np.isfinite(gnorm):
                scale = min(0.05, 0.001 + 0.000015 * float(no_improve) * current_lr)
                gz = gz + scale * gnorm * nz
            else:
                gz = 1e-6 * nz

        step += 1
        m = beta1 * m + (1.0 - beta1) * gz
        v = beta2 * v + (1.0 - beta2) * gz * gz

        b1_hat = 1.0 - beta1 ** step
        b2_hat = 1.0 - beta2 ** step
        mhat = m / b1_hat
        vhat = v / b2_hat

        delta = current_lr * mhat / (np.sqrt(vhat) + eps)
        delta = np.clip(delta, -25.0, 25.0)

        a = a * np.exp(-delta)

        amax = float(np.max(a))
        amin = float(np.min(a))

        if (not np.isfinite(amax)) or amax <= tiny or (not np.isfinite(amin)):
            a = _normalize_positive(best_a)
            m[:] = 0.0
            v[:] = 0.0
            step = 0
            continue

        if amin <= tiny:
            a = np.where(a > tiny, a, tiny)

        s = float(np.sum(a))
        if (not np.isfinite(s)) or s <= tiny:
            a = _normalize_positive(best_a)
            m[:] = 0.0
            v[:] = 0.0
            step = 0
            continue

        a = a / s

    a = _normalize_positive(best_a)
    out_best = 2.0 * n * float(np.max(np.maximum(_full_conv_len(a, 1 << (2 * n - 1).bit_length()), 0.0)))
    if not np.isfinite(out_best):
        out_best = best_score
    return a, float(out_best)


def _evaluate_authorized(sequence):
    """Safely call the provided evaluate_sequence if available."""
    try:
        vals = [float(x) for x in sequence]
    except Exception:
        return float("inf")
    try:
        val = evaluate_sequence(vals)
    except Exception:
        return float("inf")
    val = float(val)
    return val if np.isfinite(val) else float("inf")


def _sanitize_output(sequence):
    """Return a valid list of finite non-negative floats."""
    a = np.asarray(sequence, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        a = np.ones(256, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if (not np.isfinite(s)) or s <= 0.01:
        a = np.ones(256, dtype=np.float64)
        s = 1.0
    # Keep scale modest and safe; the objective is scale-invariant.
    if s > 1.0:
        a = a / s
    return [float(x) for x in a]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-initialization log-space optimizer for the auto-convolution peak score.
    """
    rng = np.random.default_rng(seed)
    now = time.monotonic()
    total = float(budget_s) if budget_s is not None else 2400.0
    deadline = now + max(5.0, total - 5.0)

    parent = globals().get("height_sequence_1", None)
    parent_seq = _normalize_positive(parent) if parent is not None else None

    if parent_seq is not None:
        best_seq = parent_seq.copy()
        best_actual = _evaluate_authorized(best_seq)
    else:
        best_seq = _normalize_positive(np.ones(256, dtype=np.float64))
        best_actual = _evaluate_authorized(best_seq)

    print("initial candidate length=%d score=%.10f" % (len(best_seq), best_actual), flush=True)

    # Phase 1: broad exploration from diverse non-symmetric starts.
    states = _make_initial_states(best_seq, rng)
    remaining = deadline - time.monotonic()
    exploration_time = min(1100.0, max(20.0, remaining * 0.38))
    if len(states) > 0:
        base_slice = max(2.0, min(55.0, exploration_time / float(len(states))))

    for seq, tag in states:
        rem = deadline - time.monotonic()
        if rem < 4.0:
            break
        seq = _normalize_positive(seq)
        n = len(seq)
        if n < 32 or n > 8192:
            continue
        slice_s = min(base_slice, max(2.0, rem - 1.0))
        p1 = 512.0 if n <= 2048 else 1024.0
        lr = 0.07 if n <= 1024 else 0.05
        seq, _ = _optimize_slice(seq, slice_s, rng, 2.0, p1, lr, tag=tag)
        actual = _evaluate_authorized(seq)
        if actual < best_actual:
            best_actual = actual
            best_seq = _normalize_positive(seq)
            print("explored %s length=%d score=%.10f" % (tag, len(best_seq), best_actual), flush=True)

    # Phase 2: focused refinement around the best found sequence.
    refine_attempt = 0
    refine_no_improve = 0
    while time.monotonic() < deadline - 4.0:
        rem = deadline - time.monotonic()
        if rem < 8.0:
            break

        best_seq = _normalize_positive(best_seq)
        nbest = len(best_seq)

        if refine_attempt % 9 == 0:
            sigma = 0.015 + 0.005 * min(3, refine_no_improve // 8)
            start = _add_perturbation(best_seq, sigma, rng)
            tag = "refine_soft"
            p0 = 16.0
        elif refine_attempt % 9 == 1:
            n2 = int(np.clip(int(round(nbest * 1.5)), 128, 8192))
            start = _resample_sequence(_add_perturbation(best_seq, 0.03, rng), n2)
            tag = "refine_resample_x1_5"
            p0 = 32.0
        elif refine_attempt % 9 == 2:
            sigma = 0.05 + 0.01 * min(4, refine_no_improve // 4)
            start = _add_perturbation(best_seq, sigma, rng)
            tag = "refine_medium"
            p0 = 64.0
        elif refine_attempt % 9 == 3:
            if nbest <= 4096:
                n2 = int(np.clip(int(round(nbest * 2.0)), 128, 8192))
            else:
                n2 = int(np.clip(int(round(nbest * 0.5)), 128, 8192))
            start = _resample_sequence(_add_perturbation(best_seq, 0.05, rng), n2)
            tag = "refine_len_change"
            p0 = 32.0
        elif refine_attempt % 9 == 4:
            start = _random_initial(nbest, rng)
            tag = "refine_random"
            p0 = 2.0
        elif refine_attempt % 9 == 5:
            start = _pattern_initial(nbest, "saw_down", rng)
            tag = "refine_saw"
            p0 = 2.0
        elif refine_attempt % 9 == 6:
            start = _pattern_initial(nbest, "bump_left", rng)
            tag = "refine_bump"
            p0 = 2.0
        elif refine_attempt % 9 == 7:
            start = _add_perturbation(best_seq, 0.12 + 0.02 * min(3, refine_no_improve // 6), rng)
            tag = "refine_strong"
            p0 = 128.0
        else:
            start = _add_perturbation(best_seq, 0.01, rng)
            tag = "refine_tiny"
            p0 = 256.0

        start = _normalize_positive(start)
        nstart = len(start)
        if nstart < 32 or nstart > 8192:
            start = _resample_sequence(start, nbest)
            nstart = len(start)

        p1 = 1024.0 if nstart >= 768 else 512.0
        lr = 0.055 if nstart <= 2048 else 0.045
        slice_s = min(45.0, max(8.0, rem / 3.0))

        seq, _ = _optimize_slice(start, slice_s, rng, p0, p1, lr, tag=tag)
        actual = _evaluate_authorized(seq)

        if actual < best_actual:
            best_actual = actual
            best_seq = _normalize_positive(seq)
            refine_no_improve = 0
            print("refined %s length=%d score=%.10f" % (tag, len(best_seq), best_actual), flush=True)
        else:
            refine_no_improve += 1

        refine_attempt += 1

        # If stuck badly, occasionally try a fresh resampled best sequence.
        if refine_no_improve >= 18 and refine_attempt % 3 == 0:
            n2 = int(np.clip(int(round(len(best_seq) * (1.0 + 0.25 * rng.integers(-1, 3)))), 128, 8192))
            fresh = _resample_sequence(best_seq, n2)
            fresh = _add_perturbation(fresh, 0.07, rng)
            seq, _ = _optimize_slice(fresh, min(25.0, max(6.0, deadline - time.monotonic() - 4.0)), rng, 8.0, 512.0, 0.06, tag="stuck_refresh")
            actual2 = _evaluate_authorized(seq)
            if actual2 < best_actual:
                best_actual = actual2
                best_seq = _normalize_positive(seq)
                refine_no_improve = 0
                print("stuck refresh length=%d score=%.10f" % (len(best_seq), best_actual), flush=True)

    final_seq = _sanitize_output(best_seq)
    final_score = _evaluate_authorized(final_seq)
    if not np.isfinite(final_score):
        final_seq = _sanitize_output(np.ones(256, dtype=np.float64))

    print("final length=%d score=%.10f" % (len(final_seq), _evaluate_authorized(final_seq)), flush=True)
    return final_seq
```
