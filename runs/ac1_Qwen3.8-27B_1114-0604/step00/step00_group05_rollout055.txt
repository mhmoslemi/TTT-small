"""
Search for nonnegative step functions minimizing
    2*n*max(convolve(a,a))/(sum(a)**2).

Strategy:
  * Work with normalized sequences (sum = 1), since the objective is scale-invariant.
  * Use FFT-based fast evaluation and subgradient/smooth-max optimization on log-weights.
  * Start from several families, including a central-binomial profile, smoothed random
    profiles, power-law profiles, and any previously found sequence.
  * Progressive multiresolution search: optimize at 256 ... 8192, always carrying the
    best sequence forward.
  * Use an optional small-size linear-program polishing step, inspired by the LP active-set
    method, when SciPy/HiGHS is available.
  * Finalize by exact evaluation of a few stored candidates if time remains.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
    from scipy.sparse import lil_matrix
    _SCIPY_OK = True
except Exception:
    _SCIPY_OK = False


def _as_1d_float(x):
    """Return a finite 1D float array, replacing bad entries by zero."""
    try:
        y = np.asarray(x, dtype=np.float64).ravel()
    except Exception:
        y = np.zeros(1, dtype=np.float64)
    y = np.where(np.isfinite(y), y, 0.0)
    if y.size == 0:
        y = np.ones(1, dtype=np.float64)
    return y


def _fft_len(n):
    """Smallest power of two >= 2*n-1."""
    target = 2 * n - 1
    L = 1
    while L < target:
        L <<= 1
    return L


def _positive_normalized(a, floor=1e-12):
    """Clip to nonnegative, replace NaN/Inf, and normalize sum to 1."""
    a = _as_1d_float(a)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    a = a / s
    a = np.maximum(a, floor)
    s2 = float(np.sum(a))
    if s2 <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s2


def _conv_fast(a):
    """FFT-based full linear convolution of a normalized 1D array with itself."""
    a = _positive_normalized(a)
    m = a.size
    L = _fft_len(m)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[: 2 * m - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    return np.maximum(c, 0.0)


def _fast_max(a):
    """Fast FFT estimate of max(convolve(a,a))."""
    c = _conv_fast(a)
    return float(np.max(c)) if c.size else 0.0


def _fast_score(a):
    """Fast FFT estimate of the exact objective."""
    a = _as_1d_float(a)
    s = float(np.sum(np.maximum(a, 0.0)))
    if s <= 1e-12:
        return float("inf")
    m = a.size
    if m == 0:
        return float("inf")
    M = _fast_max(a)
    if not np.isfinite(M):
        return float("inf")
    val = 2.0 * m * M / (s * s)
    return float(val) if np.isfinite(val) else float("inf")


def _evaluate_exact(a):
    """Call global evaluate_sequence if available; otherwise use a conservative fallback."""
    a = _positive_normalized(a)
    ev = globals().get("evaluate_sequence", None)
    if ev is not None:
        try:
            val = float(ev(a.tolist()))
            if np.isfinite(val):
                return val
        except Exception:
            pass

    # Fallback: direct convolution for smaller lengths; otherwise fast estimate.
    if a.size <= 2000:
        try:
            conv = np.convolve(a, a, mode="full")
            s = float(np.sum(a))
            if s <= 1e-12:
                return float("inf")
            val = 2.0 * a.size * float(np.max(conv)) / (s * s)
            return float(val) if np.isfinite(val) else float("inf")
        except Exception:
            pass
    return _fast_score(a)


def _resample_to_length(a, m):
    """Resample a sequence linearly to length m and normalize."""
    a = _positive_normalized(a)
    m = int(max(1, m))
    if a.size == m:
        return a.copy()
    if a.size == 1:
        return np.ones(m, dtype=np.float64) * float(a[0] / m)
    x0 = np.linspace(0.0, 1.0, a.size)
    x1 = np.linspace(0.0, 1.0, m)
    y = np.interp(x1, x0, a)
    return _positive_normalized(y)


def _central_sequence(m):
    """
    Central-binomial profile b_i = C(2i,i)/4^i.
    This profile has exactly flat early convolution coefficients and gives
    score about pi/2 for large m.
    """
    m = int(max(1, m))
    a = np.zeros(m, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, m):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return _positive_normalized(a)


def _central_weighted_sequence(m, beta=1.0):
    """Central profile multiplied by a linear ramp to increase back mass."""
    m = int(max(1, m))
    base = _central_sequence(m)
    if m == 1:
        return base
    ramp = 1.0 + float(beta) * np.linspace(0.0, 1.0, m)
    return _positive_normalized(base * ramp)


def _power_sequence(m, p=0.5, delta=0.5):
    """Power-law profile (i + delta)^(-p)."""
    m = int(max(1, m))
    i = np.arange(m, dtype=np.float64)
    if abs(float(p)) < 1e-12:
        a = np.ones(m, dtype=np.float64)
    else:
        a = np.power(i + float(delta), -float(p))
    a = np.where(np.isfinite(a), a, 0.0)
    return _positive_normalized(a)


def _linear_sequence(m, alpha=0.5):
    """Asymmetric linear ramp."""
    m = int(max(1, m))
    if m == 1:
        return np.ones(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, m)
    a = 1.0 + float(alpha) * (1.0 - 2.0 * x)
    a = np.maximum(a, 0.0)
    return _positive_normalized(a)


def _smooth_perturb(a, seed, amp=0.25):
    """Multiplicative smooth random perturbation."""
    a = _positive_normalized(a)
    m = a.size
    rng = np.random.default_rng(seed)
    noise = rng.normal(0.0, 1.0, m)
    # Low-pass by several box filters.
    for w in (1, 3, 7, 15, 31):
        k = w + 1
        kernel = np.ones(k, dtype=np.float64) / float(k)
        noise = np.convolve(noise, kernel, mode="same")
    std = float(np.std(noise))
    if std > 1e-15:
        noise = float(amp) * noise / std
    else:
        noise = 0.0
    q = a * np.exp(noise)
    return _positive_normalized(q)


def _random_smooth_sequence(m, seed=0, amp=0.45):
    """Start from a smooth random multiplicative profile."""
    m = int(max(1, m))
    base = np.ones(m, dtype=np.float64)
    return _smooth_perturb(base, seed, amp=amp)


def _make_starts(m, seed, carry, global_seq, max_starts=6):
    """Return candidate starting arrays of length m."""
    starts = []
    m = int(max(1, m))

    if carry is not None:
        try:
            starts.append(_resample_to_length(carry, m))
        except Exception:
            pass

    if global_seq is not None:
        try:
            starts.append(_resample_to_length(global_seq, m))
        except Exception:
            pass

    starts.append(_central_sequence(m))
    starts.append(_central_weighted_sequence(m, beta=1.0))
    starts.append(_power_sequence(m, p=0.35, delta=1.0))
    starts.append(_power_sequence(m, p=0.55, delta=0.5))
    starts.append(_power_sequence(m, p=0.75, delta=0.5))
    starts.append(_linear_sequence(m, alpha=0.5))
    starts.append(_random_smooth_sequence(m, seed=seed + 7919, amp=0.45))

    # Remove exact obvious duplicates by short fingerprint to save time.
    seen = []
    out = []
    for arr in starts:
        arr = _positive_normalized(arr)
        # Cheap fingerprint: first/last/sum/max and a few hashes.
        fp = (arr.size, round(float(arr[0]), 14), round(float(arr[-1]), 14),
              round(float(arr.max()), 14), round(float(arr[abs(arr.size)//2]), 14))
        if fp in seen:
            continue
        seen.append(fp)
        out.append(arr)
        if len(out) >= int(max_starts):
            break
    return out


def _soft_core(a, kappa):
    """
    For normalized a, compute:
      soft = smoothed max of convolution,
      grad_a = gradient of smoothed max w.r.t. a,
      maxc = exact fast max.
    kappa is sharpness: tau = maxc/kappa.
    """
    a = _positive_normalized(a)
    m = a.size
    L = _fft_len(m)

    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[: 2 * m - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)

    maxc = float(c.max()) if c.size else 0.0
    if maxc <= 1e-300:
        return 0.0, np.zeros(m, dtype=np.float64), 0.0

    tau = maxc / float(max(1.0, kappa))
    u = c / tau
    u = u - float(u.max())
    p = np.exp(u)
    p = np.where(np.isfinite(p), p, 0.0)
    s = float(p.sum())
    if s <= 1e-300:
        # Fallback: uniform over all lags.
        p = np.full(c.size, 1.0 / c.size, dtype=np.float64)
        s = 1.0

    soft = maxc + tau * float(np.log(s))

    # p weights over convolution lags. The derivative of c_k w.r.t. a_j is
    # 2*a_{k-j} when valid, so grad_a = 2 * circular correlation(p, a).
    P = np.fft.rfft(p, L)
    grad_a = 2.0 * np.fft.irfft(P * np.conj(F), L)[:m]
    grad_a = np.where(np.isfinite(grad_a), grad_a, 0.0)
    # Correct tiny negative roundoff.
    grad_a = np.maximum(grad_a, 0.0)

    return float(soft), grad_a, maxc


def _adam_optimize(a, time_budget=30.0, max_iter=200000, lr=0.06,
                   kappa_start=4.0, kappa_end=3000.0, seed=0,
                   score_every=4, reset_limit=1500, noise_amp=0.18):
    """
    Minimize smooth-max objective over simplex via Adam on log-weights.
    Returns best normalized sequence found within time_budget.
    """
    start_t = time.monotonic()
    deadline = start_t + max(0.5, float(time_budget))
    rng = np.random.default_rng(seed)

    a = _positive_normalized(a)
    m = a.size

    best_a = a.copy()
    best_sc = _fast_score(best_a)

    x = np.log(np.maximum(a, 1e-12))
    x = x - float(np.max(x))
    v1 = np.zeros(m, dtype=np.float64)
    v2 = np.zeros(m, dtype=np.float64)
    t = 0
    no_imp = 0

    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8

    for i in range(max(0, int(max_iter))):
        if time.monotonic() >= deadline:
            break

        # Current a from x.
        x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
        x = x - float(np.max(x))
        x = np.clip(x, -70.0, 25.0)
        ax = np.exp(x)
        s = float(np.sum(ax))
        if s <= 1e-15:
            break
        cur = ax / s
        cur = np.maximum(cur, 1e-15)
        cur = cur / float(np.sum(cur))

        # Anneal sharpness: double about every 1000 Adam steps, cap at kappa_end.
        prog = float(i + 1) / 1000.0
        kappa = float(kappa_start) * (2.0 ** prog)
        kappa = min(float(kappa_end), max(1.0, kappa))

        try:
            soft, grad_a, maxc = _soft_core(cur, kappa)
        except Exception:
            x = np.log(np.maximum(best_a, 1e-12))
            x = x - float(np.max(x))
            v1[:] = 0.0
            v2[:] = 0.0
            continue

        if not (np.isfinite(soft) and np.isfinite(maxc)):
            x = np.log(np.maximum(best_a, 1e-12))
            x = x - float(np.max(x))
            v1[:] = 0.0
            v2[:] = 0.0
            continue

        dot = float(np.dot(cur, grad_a))
        grad_x = cur * (grad_a - dot)
        grad_x = np.nan_to_num(grad_x, nan=0.0, posinf=0.0, neginf=0.0)

        t += 1
        v1 = beta1 * v1 + (1.0 - beta1) * grad_x
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_x * grad_x)
        m1 = v1 / (1.0 - beta1 ** t)
        m2 = v2 / (1.0 - beta2 ** t)
        step = lr * m1 / (np.sqrt(m2) + eps)
        step = np.nan_to_num(step, nan=0.0, posinf=0.0, neginf=0.0)

        x = x - step
        x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
        x = x - float(np.max(x))
        x = np.clip(x, -70.0, 25.0)

        if (i + 1) % int(max(1, score_every)) == 0:
            ax = np.exp(x)
            s = float(np.sum(ax))
            if s > 1e-15:
                new_a = ax / s
                new_a = np.maximum(new_a, 1e-15)
                new_a = new_a / float(np.sum(new_a))
                sc = _fast_score(new_a)
                if np.isfinite(sc) and sc < best_sc * (1.0 - 1e-8):
                    best_sc = sc
                    best_a = new_a.copy()
                    no_imp = 0
                else:
                    no_imp += int(max(1, score_every))

        # Escape local plateaus by perturbing the best sequence.
        if no_imp > int(reset_limit):
            pert = _smooth_perturb(best_a, seed=int(rng.integers(0, 2**31 - 1)), amp=noise_amp)
            x = np.log(np.maximum(pert, 1e-12))
            x = x - float(np.max(x))
            v1[:] = 0.0
            v2[:] = 0.0
            no_imp = 0

    return np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)


def _build_conv_matrix(f):
    """Dense-to-sparse matrix A where (A @ g)_k = (f * g)_k."""
    f = _positive_normalized(f)
    m = f.size
    L = 2 * m - 1
    A = lil_matrix((L, m), dtype=np.float64)
    for j in range(m):
        A[j:j + m, j] = f
    return A.tocsr()


def _lp_polish(a, time_budget=30.0, max_iters=2):
    """
    Optional LP direction polishing:
      max sum(g)  s.t.  f*g <= M, g >= 0.
    Used only at moderate lengths because the matrix becomes large.
    """
    a = _positive_normalized(a)
    if not _SCIPY_OK:
        return a
    if a.size > 512:
        return a

    deadline = time.monotonic() + max(1.0, float(time_budget))
    current = a.copy()
    current_sc = _fast_score(current)
    best = current.copy()
    best_sc = current_sc

    c_obj = -np.ones(current.size, dtype=np.float64)

    for _ in range(int(max_iters)):
        rem = deadline - time.monotonic()
        if rem < 15.0:
            break

        conv = _conv_fast(current)
        M = float(conv.max())
        if M <= 1e-300:
            break

        try:
            A = _build_conv_matrix(current)
            b = np.ones(2 * current.size - 1, dtype=np.float64)
            res = linprog(
                c_obj,
                A / M,
                b_ub=b,
                bounds=[(0.0, None)] * current.size,
                method="highs"
            )
        except Exception:
            break

        if res is None or getattr(res, "status", 1) != 0:
            break

        g = np.asarray(res.x, dtype=np.float64)
        g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
        gs = float(np.sum(g))
        if gs <= 1e-12:
            break
        g = g / gs

        improved = False
        for tv in (0.05, 0.10, 0.20, 0.30, 0.50, 1.0):
            h = (1.0 - tv) * current + tv * g
            h = _positive_normalized(h)
            sc = _fast_score(h)
            if np.isfinite(sc) and sc < current_sc * (1.0 - 1e-7):
                current = h
                current_sc = sc
                improved = True
                break

        if not improved:
            for tv in (0.01, 0.02, 0.005):
                h = (1.0 - tv) * current + tv * g
                h = _positive_normalized(h)
                sc = _fast_score(h)
                if np.isfinite(sc) and sc < current_sc * (1.0 - 1e-7):
                    current = h
                    current_sc = sc
                    improved = True
                    break

        if current_sc < best_sc:
            best_sc = current_sc
            best = current.copy()

        if not improved:
            break

    return np.nan_to_num(best, nan=0.0, posinf=0.0, neginf=0.0)


def _is_better(candidate_score, current_score):
    """Strict improvement test with finite handling."""
    return (np.isfinite(candidate_score) and
            (not np.isfinite(current_score) or candidate_score < current_score))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search entry point.
    Returns a list of nonnegative floats minimizing the provided objective.
    """
    start_t = time.monotonic()
    requested = max(60.0, float(budget_s))
    deadline = start_t + max(10.0, requested - 4.0)
    total_budget = max(60.0, requested - 15.0)

    rng = np.random.default_rng(seed)

    # Optional previously found sequence.
    global_seq = None
    prev = globals().get("height_sequence_1", None)
    if prev is not None:
        try:
            prev_arr = _as_1d_float(prev)
            if prev_arr.size > 0:
                global_seq = _positive_normalized(prev_arr)
        except Exception:
            global_seq = None

    best_seq = None
    best_score = float("inf")
    candidate_pool = []

    def store_candidate(arr):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = _positive_normalized(arr)
        sc = _fast_score(arr)
        if _is_better(sc, best_score):
            if best_seq is not None:
                candidate_pool.append(best_seq.copy())
                if len(candidate_pool) > 24:
                    candidate_pool.pop(0)
            best_score = sc
            best_seq = arr.copy()
            candidate_pool.append(best_seq.copy())
            if len(candidate_pool) > 24:
                candidate_pool.pop(0)
            print(f"[store] length={arr.size} fast_score={sc:.10g} t={time.monotonic()-start_t:8.1f}s")

    # Multiresolution stages: length, fraction of total budget, max starts.
    stages = [
        (256, 0.035, 7),
        (512, 0.070, 6),
        (1024, 0.160, 5),
        (2048, 0.280, 4),
        (4096, 0.330, 3),
        (8192, 0.125, 2),
    ]

    for stage_idx, (length, frac, max_starts) in enumerate(stages):
        t_now = time.monotonic()
        if t_now + 5.0 >= deadline:
            break

        stage_budget = frac * total_budget
        stage_budget = min(stage_budget, max(0.0, deadline - t_now - 2.0))
        if stage_budget < 12.0:
            continue

        starts = _make_starts(length, seed + 1000 + length, best_seq, global_seq, max_starts)
        if not starts:
            continue

        print(f"[stage] length={length} stage_budget={stage_budget:7.1f}s starts={len(starts)}")

        # Allocate most time to the carried best sequence if it exists.
        if best_seq is not None and len(starts) > 1:
            carry_budget = 0.58 * stage_budget
            other_budget = stage_budget - carry_budget
        else:
            carry_budget = 0.0
            other_budget = stage_budget

        stage_t0 = time.monotonic()
        for idx, cand in enumerate(starts):
            t_now = time.monotonic()
            rem = deadline - t_now - 1.0
            if rem < 5.0:
                break

            if idx == 0 and best_seq is not None and carry_budget > 0:
                alloc = min(carry_budget, rem)
                carry_budget = 0.0
            else:
                n_other = max(1, len(starts) - (1 if best_seq is not None else 0))
                alloc = min(other_budget / float(n_other), rem)
                other_budget = max(0.0, other_budget - alloc)

            if alloc < 3.0:
                continue

            try:
                opt = _adam_optimize(
                    cand,
                    time_budget=alloc,
                    max_iter=200000,
                    lr=0.06,
                    kappa_start=4.0,
                    kappa_end=3000.0,
                    seed=int(seed + 100000 + length * 31 + idx),
                    score_every=4,
                    reset_limit=1500,
                    noise_amp=0.18,
                )
            except Exception:
                opt = cand

            if opt is not None:
                store_candidate(opt)

            # Stage-level LP polish at small lengths.
            if length <= 512 and _SCIPY_OK and (time.monotonic() + 25.0 < deadline):
                lp_cand = _resample_to_length(best_seq if best_seq is not None else cand, length)
                lp_budget = min(30.0, max(0.0, deadline - time.monotonic() - 8.0))
                if lp_budget > 15.0:
                    try:
                        lpop = _lp_polish(lp_cand, time_budget=lp_budget, max_iters=2)
                        store_candidate(lpop)
                    except Exception:
                        pass

    # If nothing was found, use a safe central sequence.
    if best_seq is None:
        best_seq = _central_sequence(2048)
        best_score = _fast_score(best_seq)
        print(f"[fallback] central length={best_seq.size} score={best_score:.10g}")

    # Final multiresolution polish on the best sequence.
    while time.monotonic() + 20.0 < deadline:
        cur_len = int(best_seq.size)
        next_len = min(8192, max(cur_len * 2, 1024))
        if next_len <= cur_len:
            # Polish at current length.
            alloc = min(70.0, max(0.0, deadline - time.monotonic() - 5.0))
            if alloc < 15.0:
                break
            try:
                opt = _adam_optimize(
                    best_seq,
                    time_budget=alloc,
                    max_iter=200000,
                    lr=0.05,
                    kappa_start=8.0,
                    kappa_end=5000.0,
                    seed=int(seed + 999999),
                    score_every=4,
                    reset_limit=1200,
                    noise_amp=0.12,
                )
                store_candidate(opt)
            except Exception:
                break
            break

        big = _resample_to_length(best_seq, next_len)
        alloc = min(120.0, max(0.0, deadline - time.monotonic() - 5.0))
        if alloc < 15.0:
            break
        try:
            opt = _adam_optimize(
                big,
                time_budget=alloc,
                max_iter=200000,
                lr=0.06,
                kappa_start=6.0,
                kappa_end=4000.0,
                seed=int(seed + 888888 + next_len),
                score_every=4,
                reset_limit=1300,
                noise_amp=0.15,
            )
            store_candidate(opt)
        except Exception:
            break

    # Optional final exact evaluation of a few stored candidates.
    exact_best_seq = best_seq
    exact_best_score = _fast_score(best_seq)

    final_candidates = []
    if best_seq is not None:
        final_candidates.append(best_seq)
    for arr in candidate_pool[-4:]:
        if arr is not None:
            final_candidates.append(arr)

    for arr in final_candidates:
        if arr is None:
            continue
        arr = _positive_normalized(arr)
        need = 35.0 if arr.size > 6000 else 12.0
        if time.monotonic() + need >= deadline:
            continue
        try:
            ev = _evaluate_exact(arr)
        except Exception:
            ev = float("inf")
        if _is_better(ev, exact_best_score):
            exact_best_score = ev
            exact_best_seq = arr.copy()
            print(f"[exact] length={arr.size} score={ev:.10g}")

    final = _positive_normalized(exact_best_seq if exact_best_seq is not None else best_seq)
    final = np.clip(final, 0.0, 1000.0)
    s = float(np.sum(final))
    if s > 0.0:
        final = final / s

    print(f"[final] length={final.size} reported={exact_best_score:.10g} "
          f"final_fast={_fast_score(final):.10g} time={time.monotonic()-start_t:.1f}s")

    return [float(v) for v in final]
