One iterative method:

    """Conjugate gradient for A x = rhs where A is symmetric positive definite."""
    n = len(rhs)
    x = x0.copy()
    r = rhs - A_matvec(x)
    p = r.copy()
    rs_old = np.dot(r, r)
    for i in range(max_iter):
        if np.sqrt(rs_old) < tol * np.linalg.norm(rhs):
            break
        Ap = A_matvec(p)
        alpha = rs_old / np.dot(p, Ap)
        x += alpha * p
        r -= alpha * Ap
        rs_new = np.dot(r, r)
        if rs_new < tol * tol:
            break
        p = r + (rs_new / rs_old) * p
        rs_old = rs_new
    return x

```python
def compute_descent_direction(a, e, max_cg_iter=50):
    """
    Compute the descent direction for reducing ||b - T||^2.
    Solves C_a^T C_a * delta = -C_a^T * e / 2 using CG.
    """
    n = len(a)
    
    def matvec(x):
        # C_a^T C_a x = C_a^T (C_a x) = C_a^T (conv(a, x))
        y = np.convolve(a, x)
        return conv_T_matvec(y, a)
    
    rhs = -conv_T_matvec(e, a) / 2
    
    # CG
    x = np.zeros(n)
    r = rhs - matvec(x)  # = rhs since x=0
    p = r.copy()
    rs_old = np.dot(r, r)
    for i in range(max_cg_iter):
        if np.sqrt(rs_old) < 1e-10:
            break
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if abs(pAp) < 1e-15:
            break
        alpha = rs_old / pAp
        x += alpha * p
        r -= alpha * Ap
        rs_new = np.dot(r, r)
        if rs_new < 1e-20:
            break
        p = r + (rs_new / rs_old) * p
        rs_old = rs_new
    
    return x
```

Each CG iteration requires 2 matvec evaluations:
- `C_a @ x = convolve(a, x)`: O(n log n)
- `C_a^T @ y = convolve(a[::-1], y)[n-1:2n-1]`: O(n log n)

So each CG iteration is O(n log n). With 50 iterations, total is O(n log n * 50). For n=6121, this is very fast (milliseconds).

def conv_adjoint(v, a):
    """Compute C_a^T @ v where C_a is the linear convolution matrix for a."""
    n = len(a)
    result = np.convolve(a[::-1], v)
    return result[n-1:2*n-1]

def cg_solve(a, rhs, max_iter=100, tol=1e-10):
    """Solve C_a^T C_a x = rhs using conjugate gradient."""
    n = len(a)
    
    def matvec(x):
        y = np.convolve(a, x)  # C_a @ x
        return conv_adjoint(y, a)  # C_a^T y
    
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)

            break
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if abs(pAp) < 1e-30:

        x += alpha * p
        r -= alpha * Ap
        rs_new = np.dot(r, r)
        if rs_new < tol * tol:

import numpy as np
import time

def conv_adjoint(v, a):
    """Compute the adjoint of the linear convolution operator."""
    n = len(a)
    return np.convolve(a[::-1], v)[n-1:2*n-1]

def cg_solve(a, rhs, max_iter=100, tol=1e-10):
    """Solve C_a^T C_a x = rhs using conjugate gradient."""
    n = len(a)
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    for _ in range(max_iter):
        if rs < tol:
            break
        y = np.convolve(a, p)
        Ap = conv_adjoint(y, a)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r -= alpha * Ap
        rs_new = np.dot(r, r)
        if rs_new < tol:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x
