Uses soft-max gradient descent with simplex projection, multiple restarts,
and peak-repulsion updates.

import numpy as np
import time

def _softmax_weights(c, T):
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    s = exp_c.sum()
    if s == 0:
        return np.ones(len(c)) / len(c)
    return (c / T) * exp_c / s

def _grad_from_weights(w, a):
    """Compute gradient of soft-max objective w.r.t. a."""
    return 2.0 * np.correlate(w, a, mode='valid')

def _project_simplex(v, s=1.0):
    n = len(v)
    v = v.astype(np.float64).copy()
    if v.max() <= 0:
        return np.full(n, s / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0) * s

    n = len(a)
    deadline = time.monotonic() + time_limit
    
    alpha = 0.1  # repulsion strength
    
    step = 0
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        c_max = c.max()
        if c_max < 1e-10:
            break
        
        # Find all peak positions
        peak_threshold = c_max * 0.95
        peak_positions = np.where(c >= peak_threshold)[0]

        # We want to reduce these products
        # Strategy: multiply a[i] by (1 - alpha * a[k-i]/c_max) for i contributing to peaks
        # This is equivalent to: a[i] *= prod_k (1 - alpha * a[k-i]/c_max) for k in peaks

        # But this might make some entries go to 0.
        
        # Alternative: reduce the entries that are "aligned" with the peak
        # For peak at k: entries a[i] and a[k-i] are "aligned"
        # Reduce the larger of the two
        
        for k in peak_positions:
            # c[k] = sum_i a[i] * a[k-i]
            # For each valid i, the pair (a[i], a[k-i]) contributes

            for i in range(min(n, max(1, k+1))):
                j = k - i
                if 0 <= j < n:
                    # Both a[i] and a[j] contribute to c[k]
                    # Reduce the product

from scipy.optimize import linprog

def _lp_subproblem(a_current, c_current):
    """Solve the LP subproblem at the current point."""
    n = len(a_current)
    # Variables: a_0, ..., a_{n-1}, t
    # Minimize: t
    # Subject to:
    #   2*conv(a_current, a)[k] - c_current[k] <= t  for k = 0, ..., 2n-2
    #   sum(a) = 1
    #   a >= 0
    #   t >= 0
    
    # c[k] = sum_i a[i] * A[k-i] where A = a_current
    # 2*conv(a_current, a)[k] = 2 * sum_i a_current[i] * a[k-i]
    # This is linear in a.
    # The constraint is: 2 * sum_i a_current[i] * a[k-i] - c_current[k] - t <= 0
    
    # Let's build the A_ub matrix.
    # For constraint k: 2 * sum_{i: 0<=k-i<n} a_current[i] * a[k-i] - t <= c_current[k]
    # The coefficient of a[j] in constraint k is: 2 * a_current[k-j] (if 0 <= k-j < n), else 0
    # The coefficient of t is: -1
    
    # So A_ub[k, j] = 2 * a_current[k-j] if 0 <= k-j < n, else 0
    #    A_ub[k, n] = -1
    # b_ub[k] = c_current[k]
    
    # Plus: sum(a) = 1 (equality constraint)
    # And: a >= 0, t >= 0 (bounds)
    
    num_constraints = 2 * n - 1
    A_ub = np.zeros((num_constraints, n + 1))
    b_ub = c_current.copy()
    
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = 2.0 * a_current[idx]
        A_ub[k, n] = -1.0
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: a >= 0, t >= 0
    bounds = [(0, None)] * (n + 1)
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        a_new = result.x[:n]
        t_new = result.x[n]
    else:
        a_new = a_current
        t_new = c_current.max()
    
    return a_new, t_new

I think the key insight is that we need a sequence whose power spectrum is "moderately concentrated" - not too flat (which gives a peaked convolution) and not too concentrated (which gives a wide convolution).

This is a fundamental trade-off, and the optimum is somewhere in between.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
- Sequential LP linearization with random perturbations
- Soft-max gradient descent
- Multiple diverse initializations
- Peak-targeted coordinate updates

import numpy as np
import time
from scipy.optimize import linprog

def _softmax_weights(c, T):
    c_max = c.max()
    if T < 1e-15:
        idx = c.argmax()
        w = np.zeros_like(c)
        w[idx] = 1.0
        return w
    exp_c = np.exp((c - c_max) / T)
    s = exp_c.sum()
    if s < 1e-15:
        return np.ones(len(c)) / len(c)
    return (c / T) * exp_c / s

def _grad_from_weights(w, a):
    return 2.0 * np.correlate(w, a, mode='valid')

def _project_simplex(v, s=1.0):
    n = len(v)
    v = v.astype(np.float64).copy()
    if v.max() <= 0:
        return np.full(n, s / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0) * s

def _lp_iteration(a_cur, c_cur, rng, step_size=1.0):
    n = len(a_cur)
    num_k = 2 * n - 1
    
    # Build constraint matrix
    # A_ub[k, j] = 2 * a_cur[k-j] if 0 <= k-j < n, else 0
    # A_ub[k, n] = -1
    # b_ub[k] = c_cur[k]
    
    # Efficient construction using the fact that a_cur is fixed
    A_ub = np.zeros((num_k, n + 1))
    b_ub = c_cur.copy()
    
    for k in range(num_k):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        # A_ub[k, lo:hi] = 2 * a_cur[k-hi+1 : k-lo+1] (reversed)
        # a_cur[k-j] for j in [lo, hi)
        # = a_cur[k-lo : k-hi+1+1][::-1]  ... let me just do it directly
        for j in range(lo, hi):
            A_ub[k, j] = 2.0 * a_cur[k - j]
        A_ub[k, n] = -1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        a_lp = result.x[:n]
        a_lp = np.clip(a_lp, 0, None)
        a_lp = a_lp / max(a_lp.sum(), 1e-15)
        
        # Add small noise to break symmetry
        noise = rng.normal(0, 0.001, n)
        a_lp = _project_simplex(a_lp + noise, 1.0)
        
        # Take a step
        a_new = (1 - step_size) * a_cur + step_size * a_lp
        return _project_simplex(a_new, 1.0)
    else:
        return a_cur.copy()

    n = len(a_cur)
    current_score = 2 * n * c_cur.max()  # assuming sum(a)=1

        a_new = _lp_iteration(a_cur, c_cur, rng, step_size)
        c_new = np.convolve(a_new, a_new)
        new_score = 2 * n * c_new.max()
        if new_score < current_score:
            return a_new, new_score
        # Otherwise backtrack
    return a_cur, current_score
