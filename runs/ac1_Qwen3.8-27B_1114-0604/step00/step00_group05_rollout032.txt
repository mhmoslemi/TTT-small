def compute_score(x):
    """Compute the score efficiently using FFT."""
    n = len(x)
    S = np.sum(x)
    if S < 0.01:
        return np.inf
    # Linear convolution via FFT
    L = 1
    while L < 2*n - 1:
        L *= 2
    X = fft(x, L)
    c = ifft(X * X).real
    c = c[:2*n-1]  # truncate to valid length
    max_b = np.max(c)
    return 2.0 * n * max_b / (S * S)

def simplex_project(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

from scipy.optimize import linprog

def bcd_step(x, M):
    """Solve LP to find optimal y given x."""
    n = len(x)
    # Maximize sum(y) subject to:
    # (x*y)[k] <= M for k = 0, ..., 2n-2
    # y >= 0
    
    # Variables: y[0], ..., y[n-1]
    # Objective: minimize -sum(y) => c = [-1, -1, ..., -1]
    # Constraints: A_ub @ y <= b_ub
    #   A_ub[k, j] = x[k-j] if 0 <= k-j < n, else 0 (for k = 0, ..., 2n-2)
    #   b_ub = [M, M, ..., M] (2n-1 entries)
    
    # Build A_ub
    # A_ub has shape (2n-1, n)
    # A_ub[k, j] = x[k-j] if 0 <= k-j < n, else 0
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A_ub[k, j] = x[k-j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return -res.fun, res.x  # (maximized sum, optimal y)

    """Project v onto {x: sum(x) = target_sum, x >= 0}."""

def compute_conv(x, L=None):
    """Compute linear convolution x*x via FFT."""
    n = len(x)
    if L is None:

        while L < 2 * n - 1:

    """Compute 2n * max(conv) / (sum(x))^2 efficiently."""
    n = len(x)

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

- Score = 2n * max(c) (since sum(x) = 1)

The gradient of the log-sum-exp objective (1/τ) log(sum exp(τ c_k)):
= sum_k (τ exp(τ c_k) / sum exp(τ c)) * (1/τ) * ∂c_k/∂x
= sum_k w_k * ∂c_k/∂x

where w_k = exp(τ c_k) / sum exp(τ c).

And ∂c_k/∂x_j = 2 * x_{k-j} if 0 <= k-j < n, else 0.

So the gradient is:
∇f_j = sum_k w_k * 2 * x_{k-j} (valid indices)
= 2 * sum_{k=max(j,0)}^{min(j+n-1, 2n-2)} w_k * x_{k-j}

Let m = k-j:
= 2 * sum_{m=0}^{n-1} w_{m+j} * x_m (where w_m = 0 if m >= 2n-1)

This is a "cross-correlation" of w with x, specifically:
∇f_j = 2 * (w ⋆ x)_j = 2 * sum_m x_m * w_{m+j}

Using FFT:
Let X = fft(x, L), W = fft(w, L) where L >= 2n-1.
Then ifft(W * conj(X)) gives the circular cross-correlation.

Specifically: ifft(W * conj(X))_j = sum_m W_m * conj(X_m) * e^{2πimj/L} / L
= sum_m sum_m' w_{m'} e^{-2πim'm/L} * conj(sum_m'' x_{m''} e^{-2πim''m/L}) * e^{2πimj/L} / L
= sum_m' sum_m'' w_{m'} x_{m''} e^{-2πim'm/L} e^{2πim''m/L} e^{2πimj/L} / L
= sum_m' sum_m'' w_{m'} x_{m''} e^{2πim(m'' - m' + j)/L} / L
= sum_m' sum_m'' w_{m'} x_{m''} δ_{m''-m'+j, 0 mod L}
= sum_m' w_{m'} x_{m'-j mod L}

For m' - j in [0, L-1]: if 0 <= m'-j < n, this is w_{m'} * x_{m'-j}. If m'-j >= n, x_{m'-j} = 0.
For m' - j < 0: m' - j + L >= L - (n-1) > 2n-2 (since L >= 2n-1), so w_{m'-j+L} = 0.

So: ifft(W * conj(X))_j = sum_{m'=j}^{L-1} w_{m'} * x_{m'-j} = sum_{m'=j}^{n+j-1} w_{m'} * x_{m'-j} (since x is 0 outside [0,n-1])
= sum_{m=0}^{n-1} w_{m+j} * x_m ✓

"""
Algorithm: Minimize 2n*max(x*x)/(sum(x))^2 for non-negative sequences x.

Uses projected gradient descent with log-sum-exp smoothing of the max-convolution
objective, combined with top-K subgradient focusing, multiple sequence lengths,
and local search refinement. The key insight is that we want the self-convolution
to be as flat as possible (peak-to-average ratio → (2n-1)/n for the ideal).
"""

import numpy as np
from scipy.fft import fft, ifft
import time


def compute_conv_fft(x):
    """Compute linear convolution x*x via FFT. Returns (c, X, L)."""
    n = len(x)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    X = fft(x, L)
    c = np.real(ifft(X * X))[:2 * n - 1]
    return c, X, L


def compute_score_fast(x):
    """Score = 2n * max(conv) / (sum(x))^2."""
    n = len(x)
    S = np.sum(x)
    if S < 1e-10:
        return np.inf
    c, _, _ = compute_conv_fft(x)
    return 2.0 * n * np.max(c) / (S * S)


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if np.max(v) <= 0:
        return np.full(n, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)


def grad_topk(c, x, X, L, topk):
    """Compute gradient using top-K convolution coefficients."""
    n = len(x)
    if topk >= len(c):
        # Full gradient via log-sum-exp equivalent (uniform weights)
        w = np.ones(len(c)) / len(c)
    else:
        # Find top-K indices
        part = np.argpartition(-c, topk)[:topk]
        w = np.zeros(len(c))
        w[part] = 1.0 / topk
    W = fft(w, L)
    grad = 2.0 * np.real(ifft(W * np.conj(X)))[:n]
    return grad


def grad_logsumexp(c, x, X, L, tau):
    """Compute gradient using log-sum-exp weighting."""
    n = len(x)
    c_max = c.max()
    shifted = np.clip((c - c_max) * tau, -400, 400)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    W = fft(w, L)
    grad = 2.0 * np.real(ifft(W * np.conj(X)))[:n]
    return grad


def optimize_gradient(n, seed, max_iters, time_budget, use_topk=False, topk=50):
    """Projected gradient descent for minimizing max(x*x) with sum(x)=1, x>=0."""
    rng = np.random.default_rng(seed)
    
    x = rng.random(n) + 0.01
    x = simplex_project(x)
    
    best_x = x.copy()
    best_score = compute_score_fast(x)
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        
        # Compute convolution
        c, X, L = compute_conv_fft(x)
        
        # Choose gradient method
        if use_topk:
            grad = grad_topk(c, x, X, L, topk)
        else:
            tau = 5.0 + 20.0 * frac
            grad = grad_logsumexp(c, x, X, L, tau)
        
        # Adaptive step size
        alpha = 1.0 / (0.5 + np.linalg.norm(grad) * n * 0.1)
        alpha = min(alpha, 2.0 / n)
        
        x_new = x - alpha * grad
        x_new = simplex_project(x_new)
        
        x = x_new
        
        if it % 30 == 0:
            s = compute_score_fast(x)
            if s < best_score:
                best_score = s
                best_x = x.copy()
    
    # Final check
    s = compute_score_fast(x)
    if s < best_score:
        best_score = s
        best_x = x.copy()
    
    return best_x, best_score


def optimize_structured(n, seed, time_budget):

def local_search(x, seed, time_budget):
    """Block perturbation local search."""
    rng = np.random.default_rng(seed)
    n = len(x)
    best_x = x.copy()
    best_score = compute_score_fast(x)
    
    start = time.monotonic()
    step_count = 0
    
    while time.monotonic() - start < time_budget:
        step_count += 1
        
        # Adaptive block size
        min_block = max(1, n // 500)
        max_block = max(min_block + 1, n // 10)
        block = int(rng.integers(min_block, max_block))
        start_i = int(rng.integers(0, n - block))
        
        candidate = best_x.copy()
        
        # Multiplicative perturbation
        sigma = 0.05 + 0.15 * (time.monotonic() - start) / max(1, time_budget)
        pert = np.exp(rng.normal(0.0, sigma, size=block))
        candidate[start_i:start_i + block] *= pert
        
        # Occasionally flip a segment
        if rng.random() < 0.1:
            flip_len = int(rng.integers(max(2, n // 20), n // 3))
            flip_start = int(rng.integers(0, n - flip_len))
            candidate[flip_start:flip_start + flip_len] = candidate[flip_start:flip_start + flip_len][::-1]
        
        candidate = simplex_project(candidate)
        
        score = compute_score_fast(candidate)
        if score < best_score:
            best_score = score
            best_x = candidate.copy()
        
        if step_count % 500 == 0:
            print(f"    Local search step {step_count}: score={best_score:.6f}")
    
    return best_x, best_score


def lp_bcd_step(x):
    """
    One BCD step: solve LP max sum(y) s.t. (x*y)[k] <= M, y >= 0.
    Returns rescaled y and the improvement ratio.
    Only feasible for smaller n.
    """
    n = len(x)
    if n > 2000:
        return None, 1.0
    
    from scipy.optimize import linprog
    
    c, _, _ = compute_conv_fft(x)
    M = c.max()
    S = x.sum()
    
    # Build constraint matrix A_ub[k, j] = x[k-j] if 0 <= k-j < n
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A_ub[:, j] has x[j], x[j+1], ..., x[n-1] in rows n-1-j to 2n-2-j... 
        # Actually A_ub[k, j] = x[k-j] if 0 <= k-j < n
        # Row k, col j: non-zero when 0 <= k-j < n, i.e., j <= k < j+n
        rows = np.arange(j, min(2 * n - 1, j + n))
        A_ub[rows, j] = x[rows - j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            y = res.x
            ratio = y.sum() / S
            y_rescaled = y * (S / y.sum())
            return y_rescaled, ratio
    except Exception:
        pass
    
    return None, 1.0
