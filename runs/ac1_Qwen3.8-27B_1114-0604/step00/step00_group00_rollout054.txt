def compute_objective_fast(a):
    """Compute 2*n*max(conv)/sum^2 efficiently using FFT."""
    n = len(a)
    # Use FFT for convolution
    size = 2*n - 1
    # Next power of 2 for FFT efficiency
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, n=fft_size)
    fc = np.fft.irfft(fa * fa, n=fft_size)[:size]
    max_b = np.max(fc)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def binomial_init(n):
    """Generate binomial coefficient sequence a[k] = C(2k,k)/4^k."""
    a = np.empty(n)
    a[0] = 1.0
    for k in range(1, n):
        # C(2k,k)/4^k = C(2k-2,k-1)/4^{k-1} * (2k-1)(2k-2)/(4k^2)
        # = prev * (2k-1)/(2k) ... let's use recurrence
        # C(2k,k) = C(2k-2,k-1) * (2k)(2k-1)/(k^2)
        # So C(2k,k)/4^k = C(2k-2,k-1)/4^{k-1} * (2k)(2k-1)/(4*k^2)
        # = prev * (2k-1)/(2k)
        a[k] = a[k-1] * (2*k - 1) / (2*k)
    return a

    deadline = time.monotonic() + time_budget

        improved = False
        # Random order of indices
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-10:
