import numpy as np
import time
from scipy.optimize import linprog

def solve_lp_for_g(f, M):
    """
    Solve: max sum(g) subject to:
      (f * g)[k] <= M for all k = 0, ..., 2n-2
      g[i] >= 0 for all i
    where f * g is the full convolution.
    """
    n = len(f)
    # Objective: minimize -sum(g), i.e., c = -1 for all variables
    c = -np.ones(n)
    
    # Constraints: A_ub @ g <= b_ub
    # (f * g)[k] = sum_i f[max(0, k-n+1)..min(k,n-1)]_i * g[i]
    # For each k, the coefficients of g[i] are f[k-i] (when valid)
    
    # A_ub is (2n-1) x n matrix
    # A_ub[k, i] = f[k-i] if 0 <= k-i < n, else 0
    # i.e., A_ub[k, i] = f[k-i] if max(0, k-n+1) <= i <= min(k, n-1), else 0
    
    # This is a Toeplitz-like matrix
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        A_ub[k, i_min:i_max+1] = f[k-i_max:k-i_min+1][::-1]
    
    b_ub = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return None

"""
Minimize the autocorrelation peak factor of a non-negative sequence.
Uses FFT-based multiplicative weights with periodic LP refinement.
"""
import numpy as np
import time
import math

def fft_convolve(a, b):
    """Full convolution using FFT."""
    n = len(a)
    m = len(b)
    N = 1
    while N < n + m - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:n + m - 1]

def fft_cross_correlate(a, b):
    """
    Compute cross-correlation: (a ⋆ b)[i] = sum_j a[j] * b[i+j]
    """
    n = len(a)
    m = len(b)
    N = 1
    while N < n + m - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(np.conj(A) * B, N)[:n]

    it = 0
    stall_count = 0
    
    while time.monotonic() < deadline:
        # Compute convolution
        conv = fft_convolve(f, f)
        max_conv = conv.max()

        current_score = 2 * n * max_conv  # since sum = 1

        if K_top < len(conv):

        else:
            top_indices = np.arange(len(conv))

        conv_top = conv[top_indices]
        w_top = np.exp(eta * conv_top / max_conv - eta)  # normalized so max is 1

        w = np.zeros(len(conv))

        grad = fft_cross_correlate(f, w)
        grad = np.maximum(grad, 0)

            f_new = f * np.exp(-eta * grad / grad.max())
            f_new = np.clip(f_new, 0, None)
            f_new_sum = f_new.sum()
            if f_new_sum > 1e-10:

            else:

            if score < best_score:

                stall_count = 0
                if it % 100 == 0:
