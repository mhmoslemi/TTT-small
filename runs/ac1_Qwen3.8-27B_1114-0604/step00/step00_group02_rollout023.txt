"""
LP-guided iterative search minimizing the peak of the self-convolution.
Uses the SOTA approach: solve an LP at each step to find the best direction,
then perform a line search update. Falls back to subgradient method if LP is too slow.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def build_convolution_matrix_indices(n):
    """Precompute the row/col structure of the convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
    return np.array(rows), np.array(cols)


def compute_subgradient(f, n):
    """Compute a subgradient of max_k (f*f)[k] with respect to f."""
    conv = np.convolve(f, f)
    M = conv.max()
    # Find indices where conv is close to M
    thresh = M - 0.01 * M
    active = np.where(conv >= thresh)[0]
    
    grad = np.zeros(n)
    for k in active:
        weights = 1.0 / len(active)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += weights * 2.0 * f[idx]
    return grad, M, conv
