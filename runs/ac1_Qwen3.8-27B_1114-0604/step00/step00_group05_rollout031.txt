import numpy as np
import time

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_gradient_max_conv(a, k_star):
    """Compute gradient of c[k_star] = sum_i a[i]*a[k_star-i] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            if idx == j:
                grad[j] = a[idx]  # a[j] * a[j], derivative is a[j]
            else:
                grad[j] = 2 * a[idx]  # a[j]*a[idx] + a[idx]*a[j]
    return grad

def simplex_project(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n <= 1:
        return np.ones(n) * (1.0 / n) if n == 1 else v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (cssv - 1.0) / np.arange(1, n + 1)
    rho = np.nonzero(u > cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_gradient(a, k_star):
    """Compute gradient of c[k_star] = sum_i a[i]*a[k_star-i] w.r.t. a."""
    n = a.size
    indices = k_star - np.arange(n)
    valid = (indices >= 0) & (indices < n)
    grad = np.zeros(n)
    grad[valid] = 2.0 * a[indices[valid]]
    # Fix diagonal: where k_star - j == j, i.e., j == k_star/2
    if k_star % 2 == 0:
        j = k_star // 2
        if 0 <= j < n:
            grad[j] = a[j]  # Only one factor, not two
    return grad

- c[1] = 2a[0]a[1]
- c[2n-2] = a[n-1]^2

For the convolution to be approximately uniform at value V = (sum a)^2/(2n-1):
- a[0] ≈ sqrt(V)
- a[1] ≈ V/(2*sqrt(V)) = sqrt(V)/2
- a[2] ≈ ... (from c[2] = 2a[0]a[2] + a[1]^2 = V)

This gives a[0] > a[1] > a[2] > ... which means the sequence is monotonically decreasing (or increasing). And the values decrease rapidly, so the sum is dominated by the first few terms. This creates a contradiction: the sum can't be all that large if the values are decreasing rapidly, but we need the sum to be large to make V small.
