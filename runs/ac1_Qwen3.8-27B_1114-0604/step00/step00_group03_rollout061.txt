```python
"""
Search for a nonnegative sequence minimizing 2*n*max(auto-convolution)/sum^2.

Method:
  * Parametric initialization over smooth positive families (polynomials, mild exponentials).
  * Optional coarse LP water-filling refinement using scipy, if available.
  * Main refinement by multiplicative mirror-descent / top-peak gradient updates.
  * Final validation with the authoritative evaluate_sequence() routine.
"""

import numpy as np
import time

try:
    import scipy.optimize as _skopt
    import scipy.sparse as _sp
    _HAVE_SCIPY = True
except Exception:
    _skopt = None
    _sp = None
    _HAVE_SCIPY = False


def _sort_key(item):
    return item[0]


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        n = max(1, a.size)
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _finalize(a):
    a = _normalize(a)
    # Remove numerically meaningless dust, then re-normalize.
    a[a < 1e-18] = 0.0
    s = float(a.sum())
    if s <= 1e-300:
        n = max(1, a.size)
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _pad_auto(n):
    return 1 << (2 * n - 1).bit_length()


def _pad_grad(n):
    # convolution of weights of length (2n-1) with sequence of length n
    return 1 << (3 * n - 2).bit_length()


def _autocorr(a, pad=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if pad is None:
        pad = _pad_auto(n)
    F = np.fft.rfft(a, pad)
    c = np.fft.irfft(F * F, pad)[:2 * n - 1]
    c = np.real(c)
    c[c < 0.0] = 0.0
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    c = _autocorr(a)
    m = float(c.max())
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * a.size * m)


def _topk_weights(c, k):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = len(c)
    k = int(min(max(k, 1), L))
    w = np.zeros(L, dtype=np.float64)
    if k == L:
        w.fill(1.0 / float(L))
        return w
    idx = np.argpartition(c, -k)[-k:]
    w[idx] = 1.0 / float(k)
    return w


def _gradient_topk(w, a, pad=None):
    """
    Gradient for F(w,a) = sum_k w_k * conv(a,a)_k = sum_{i,j} w[i+j] a[i] a[j].
    grad[i] = 2 * sum_j w[i+j] a[j].
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = len(a)
    if pad is None:
        pad = _pad_grad(n)
    w_rev = w[::-1]
    Fw = np.fft.rfft(w_rev, pad)
    Fa = np.fft.rfft(a, pad)
    conv = np.fft.irfft(Fw * Fa, pad)[:2 * n - 1]
    conv = np.real(conv)
    # c[i] = conv[(2n-2) - i]
    c = conv[2 * n - 2::-1]
    return 2.0 * c


def _mul_update(a, grad, eta):
    a = np.asarray(a, dtype=np.float64).ravel()
    grad = np.asarray(grad, dtype=np.float64).ravel()
    gm = float(grad.max())
    if not np.isfinite(gm) or gm <= 0.0:
        return None
    scaled = grad / gm
    new = a * np.exp(-float(eta) * scaled)
    new = np.clip(new, 1e-300, None)
    s = float(new.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return new / s


def _random_smooth(a, rng, strength):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    x = np.linspace(0.0, 1.0, n)
    field = np.zeros(n, dtype=np.float64)
    for m in range(1, 9):
        ang = np.pi * float(m) * x
        field = field + rng.normal() * np.sin(ang) + rng.normal() * np.cos(ang)
    loga = np.log(np.maximum(a, 1e-300))
    loga = loga + float(strength) * field
    mx = float(loga.max())
    if not np.isfinite(mx):
        return a.copy()
    loga = loga - mx
    b = np.exp(loga)
    b = np.clip(b, 1e-300, None)
    s = float(b.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return a.copy()
    return b / s


def _make_poly(x, x2, x3, x4, c1, c2, c3, c4):
    a = 1.0 + float(c1) * x + float(c2) * x2 + float(c3) * x3 + float(c4) * x4
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def _make_exp(x, x2, x3, x4, c1, c2, c3, c4):
    loga = float(c1) * x + float(c2) * x2 + float(c3) * x3 + float(c4) * x4
    if not np.all(np.isfinite(loga)):
        return None
    mx = float(loga.max())
    if not np.isfinite(mx):
        return None
    loga = loga - mx
    # Keep in a safe range.
    loga = np.clip(loga, -50.0, 0.0)
    a = np.exp(loga)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def _add_top(top, score, a, top_size=4):
    if not np.isfinite(score):
        return top
    a = np.asarray(a, dtype=np.float64).ravel()
    top.append((float(score), np.array(a, dtype=np.float64)))
    top.sort(key=_sort_key)
    if len(top) > top_size:
        top = top[:top_size]
    return top


def _parametric_search(N, time_budget, seed=42):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, N)
    x2 = x * x
    x3 = x2 * x
    x4 = x2 * x2

    top = []
    a = np.ones(N, dtype=np.float64) / float(N)
    top = _add_top(top, _score(a), a)

    deadline = time.monotonic() + max(1.0, float(time_budget))

    # Strong simple starts, including useful linear tilts.
    for c1 in (0.0, 1.0, 1.5, 2.0, 2.5, 3.0, -1.0, -0.5):
        a = _make_poly(x, x2, x3, x4, c1, 0.0, 0.0, 0.0)
        if a is not None:
            top = _add_top(top, _score(a), a)

    # Linear sweep.
    for c1 in np.linspace(-5.0, 6.0, 65):
        if time.monotonic() > deadline:
            break
        a = _make_poly(x, x2, x3, x4, float(c1), 0.0, 0.0, 0.0)
        if a is not None:
            top = _add_top(top, _score(a), a)

    # Small quadratic sweep.
    vals = np.linspace(-4.0, 4.0, 9)
    for c1 in vals:
        for c2 in vals:
            if time.monotonic() > deadline:
                break
            a = _make_poly(x, x2, x3, x4, float(c1), float(c2), 0.0, 0.0)
            if a is not None:
                top = _add_top(top, _score(a), a)

    # Random smooth positive candidates until time budget is used.
    while time.monotonic() < deadline:
        # Random polynomial part.
        c1 = rng.uniform(-6.0, 6.0)
        c2 = rng.uniform(-6.0, 6.0)
        c3 = rng.uniform(-6.0, 6.0)
        c4 = rng.uniform(-4.0, 4.0)
        a = _make_poly(x, x2, x3, x4, c1, c2, c3, c4)
        if a is not None:
            top = _add_top(top, _score(a), a)

        # Mild exponential part.
        e1 = rng.uniform(-2.0, 2.0)
        e2 = rng.uniform(-3.0, 3.0)
        e3 = rng.uniform(-2.0, 2.0)
        e4 = rng.uniform(-1.5, 1.5)
        a = _make_exp(x, x2, x3, x4, e1, e2, e3, e4)
        if a is not None:
            top = _add_top(top, _score(a), a)

    return top


def _downsample(a, m):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == m:
        return a.copy()
    if m <= 0:
        return a.copy()
    old = np.linspace(0.0, 1.0, n)
    new = np.linspace(0.0, 1.0, m)
    b = np.interp(new, old, a)
    return _normalize(b)


def _upsample(a, big_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if big_n <= 0:
        return a.copy()
    if n == big_n:
        return a.copy()
    old = np.linspace(0.0, 1.0, n)
    new = np.linspace(0.0, 1.0, big_n)
    b = np.interp(new, old, a)
    return _normalize(b)


def _lp_direction(f, M):
    if not _HAVE_SCIPY:
        return None
    f = np.asarray(f, dtype=np.float64).ravel()
    n = len(f)
    L = 2 * n - 1
    if M <= 0.0 or not np.isfinite(M):
        return None

    rows = []
    cols = []
    vals = []

    for k in range(L):
        start = max(0, k - (n - 1))
        end = min(n - 1, k) + 1
        if start >= end:
            continue
        j = np.arange(start, end, dtype=np.int64)
        rows.extend([k] * int(end - start))
        cols.extend(j.tolist())
        vals.extend(np.asarray(f[k - j], dtype=np.float64).tolist())

    if len(vals) == 0:
        return None

    try:
        try:
            A0 = _sp.coo_array((vals, rows, cols), shape=(L, n))
        except AttributeError:
            A0 = _sp.coo_matrix((vals, rows, cols), shape=(L, n))
        A = A0.tocsr() / M
    except Exception:
        return None

    c = -np.ones(n, dtype=np.float64)
    b = np.ones(L, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = _skopt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    except Exception:
        return None

    if res is None or not hasattr(res, "success") or not res.success:
        return None
    if res.x is None:
        return None

    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return None
    return g / s


def _lp_refine(a, deadline, seed=123):
    if not _HAVE_SCIPY:
        a = _normalize(a)
        return a, _score(a)

    a = _normalize(a)
    best = a.copy()
    best_score = _score(a)
    f = a.copy()
    rng = np.random.default_rng(seed)
    it = 0

    while time.monotonic() < deadline:
        conv = _autocorr(f)
        M = float(conv.max())
        if not np.isfinite(M) or M <= 0.0:
            break

        g = _lp_direction(f, M)
        if g is None:
            break

        improved = False
        for t in (0.1, 0.25, 0.5, 0.75, 1.0):
            cand = (1.0 - float(t)) * f + float(t) * g
            cand = _normalize(cand)
            s = _score(cand)
            if np.isfinite(s) and s < best_score:
                best = cand.copy()
                best_score = s
                f = cand.copy()
                improved = True

        if not improved:
            cand = _random_smooth(f, rng, 0.15)
            s = _score(cand)
            if np.isfinite(s) and s < best_score:
                best = cand.copy()
                best_score = s
                f = cand.copy()
                improved = True

        it += 1
        if it % 5 == 0:
            print(f"LP refine it={it} local_best={best_score:.6f}", flush=True)
        if it > 300:
            break

    return best, best_score


def _refine(a, deadline, seed=1):
    a = _normalize(a)
    n = len(a)
    pad_auto = _pad_auto(n)
    pad_grad = _pad_grad(n)

    current = a.copy()
    current_conv = _autocorr(current, pad_auto)
    current_score = float(2.0 * n * current_conv.max())

    best = a.copy()
    best_score = current_score

    rng = np.random.default_rng(seed)
    eta = 0.08
    no_improve = 0
    it = 0
    max_K = max(64, len(current_conv) // 16)

    while time.monotonic() < deadline:
        K = min(max(32, 32 + it // 2), max_K)
        w = _topk_weights(current_conv, K)
        grad = _gradient_topk(w, current, pad_grad)
        gm = float(grad.max())
        if not np.isfinite(gm) or gm <= 0.0:
            break

        improved = False
        tries = (eta, eta * 0.5, eta * 0.2, eta * 0.05)
        for tr in tries:
            cand = _mul_update(current, grad, float(tr))
            if cand is None:
                continue
            cand_conv = _autocorr(cand, pad_auto)
            cand_score = float(2.0 * n * cand_conv.max())
            if not np.isfinite(cand_score):
                continue

            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

            if cand_score < current_score:
                current = cand.copy()
                current_conv = cand_conv
                current_score = cand_score
                eta = max(0.002, float(tr))
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            eta = max(0.002, eta * 0.7)

            if no_improve >= 100:
                strength = 0.15 + 0.05 * min(10.0, float(no_improve) / 100.0)
                cand = _random_smooth(best, rng, strength)
                cand_conv = _autocorr(cand, pad_auto)
                cand_score = float(2.0 * n * cand_conv.max())
                if np.isfinite(cand_score):
                    if cand_score < best_score:
                        best = cand.copy()
                        best_score = cand_score
                        current = cand.copy()
                        current_conv = cand_conv
                        current_score = cand_score
                        eta = 0.08
                        no_improve = 0
                    elif cand_score < best_score * 1.08 + 1e-12:
                        current = cand.copy()
                        current_conv = cand_conv
                        current_score = cand_score
                        no_improve = 0

            if no_improve > 5000:
                break

        it += 1
        if it % 5000 == 0:
            print(
                f"mirror refine it={it} local_best={best_score:.6f} "
                f"current={current_score:.6f}",
                flush=True,
            )

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Return a nonnegative sequence with small evaluation score."""
    start = time.monotonic()
    budget = max(1.0, float(budget_s))
    deadline = start + max(5.0, budget - 20.0)

    # 4096 keeps final O(n^2) evaluation cheap and is large enough for the
    # finite-length asymptotics to be useful.
    N = 4096 if budget > 900.0 else 2048

    remaining = deadline - time.monotonic()
    param_budget = min(240.0, max(5.0, 0.10 * remaining))
    top = _parametric_search(N, param_budget, seed)

    if not top:
        top = [(float("inf"), np.ones(N, dtype=np.float64) / float(N))]

    best = np.array(top[0][1], dtype=np.float64).copy()
    best_score = float(top[0][0])
    print(f"parametric best score (approx) = {best_score:.6f}", flush=True)

    # Optional coarse LP phase.
    remaining = deadline - time.monotonic()
    lp_budget = min(300.0, max(0.0, 0.10 * remaining))
    lp_candidate = None
    if _HAVE_SCIPY and time.monotonic() < deadline - lp_budget - 120.0:
        lp_n = 512
        lp_start = _downsample(best, lp_n)
        lp_deadline = time.monotonic() + lp_budget
        lp_best, lp_score = _lp_refine(lp_start, lp_deadline, seed + 101)
        lp_up = _upsample(lp_best, N)
        lp_up_score = _score(lp_up)
        print(f"LP upsample score (approx) = {lp_up_score:.6f}", flush=True)
        if np.isfinite(lp_up_score) and lp_up_score < best_score:
            best = lp_up.copy()
            best_score = lp_up_score
            lp_candidate = best.copy()
        else:
            lp_candidate = lp_up

    # Main mirror-descent refinement on the current best.
    remaining = deadline - time.monotonic()
    if remaining > 25.0:
        first_deadline = time.monotonic() + 0.55 * remaining
        rb, rs = _refine(best.copy(), first_deadline, seed + 202)
        if np.isfinite(rs) and rs < best_score:
            best = rb.copy()
            best_score = rs
            print(f"main refine score (approx) = {best_score:.6f}", flush=True)

    # Try a second strong parametric start if it has not been ruined.
    remaining = deadline - time.monotonic()
    if remaining > 25.0 and len(top) > 1:
        alt = top[1][1].copy()
        alt_score = _score(alt)
        if np.isfinite(alt_score) and alt_score < best_score * 1.20 + 1e-9:
            alt_deadline = time.monotonic() + 0.45 * remaining
            rb2, rs2 = _refine(alt, alt_deadline, seed + 303)
            if np.isfinite(rs2) and rs2 < best_score:
                best = rb2.copy()
                best_score = rs2
                print(f"alt refine score (approx) = {best_score:.6f}", flush=True)

    # Polish current best with whatever time remains.
    remaining = deadline - time.monotonic()
    if remaining > 20.0:
        final_deadline = time.monotonic() + remaining
        rb3, rs3 = _refine(best.copy(), final_deadline, seed + 404)
        if np.isfinite(rs3) and rs3 < best_score:
            best = rb3.copy()
            best_score = rs3
            print(f"polish score (approx) = {best_score:.6f}", flush=True)

    # Exact final selection among a few valid candidates.
    exact_candidates = []
    exact_candidates.append(best.copy())
    if lp_candidate is not None:
        exact_candidates.append(lp_candidate.copy())
    for _, a in top:
        if len(exact_candidates) >= 6:
            break
        exact_candidates.append(np.array(a, dtype=np.float64).copy())

    final_list = None
    final_exact = float("inf")

    for cand in exact_candidates:
        cand = _finalize(cand)
        lst = [float(v) for v in cand]
        try:
            ex = evaluate_sequence(lst)
        except Exception:
            ex = float("inf")
        if np.isfinite(ex) and ex < final_exact:
            final_exact = ex
            final_list = lst

    # Guaranteed valid fallback.
    if final_list is None:
        fb = [1.0 / float(N)] * N
        try:
            fb_score = evaluate_sequence(fb)
        except Exception:
            fb_score = float("inf")
        if np.isfinite(fb_score):
            final_list = fb
            final_exact = fb_score
        else:
            final_list = fb

    print(f"final exact score = {float(final_exact):.6f}", flush=True)
    return final_list
```
