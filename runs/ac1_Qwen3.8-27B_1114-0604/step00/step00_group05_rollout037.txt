```python
def solve_lp_for_g(a, M):
    """Solve: max sum(g) s.t. convolve(a,g) <= M, g >= 0"""
    n = len(a)
    # Constraint matrix: A @ g <= M * ones(2n-1)
    # A[k, i] = a[i] if 0 <= k-i < n
    # Build as sparse matrix
    
    # Row k has non-zeros at columns i where max(0, k-n+1) <= i <= min(k, n-1)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n-1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(g) = minimize -sum(g)
    c = -np.ones(n)
    b_constraints = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_constraints, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

def build_conv_matrix(a):
    """Build sparse matrix A where A @ g = convolve(a, g)."""
    n = len(a)
    k_range = np.arange(2 * n - 1)
    col_starts = np.maximum(0, k_range - n + 1)
    col_ends = np.minimum(k_range, n - 1)
    nnz_per_row = col_ends - col_starts + 1
    rows = np.repeat(k_range, nnz_per_row)
    cols = np.concatenate([np.arange(s, e+1) for s, e in zip(col_starts, col_ends)])
    vals = a[cols]
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_step(a, M, max_time=30):
    """Solve LP: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    start = time.monotonic()
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, 
                  method='highs', options={'time_limit': max_time, 'presolve': True})
    
    if res.success:
        return res.x
    return None

from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_matrix(a):
    n = len(a)
    k_range = np.arange(2 * n - 1)
    col_starts = np.maximum(0, k_range - n + 1)
    col_ends = np.minimum(k_range, n - 1)
    nnz_per_row = col_ends - col_starts + 1
    rows = np.repeat(k_range, nnz_per_row)
    cols = np.concatenate([np.arange(s, e+1) for s, e in zip(col_starts, col_ends)])
    vals = a[cols]
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
