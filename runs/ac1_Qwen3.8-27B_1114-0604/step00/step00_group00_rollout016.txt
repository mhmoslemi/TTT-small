    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 1e-15:
        a = a * (target_sum / S)
    else:
        a = np.full_like(a, target_sum / len(a))
    return a

def _evaluate_safe(a):
    """Evaluate using our own computation (faster, no overhead)."""
    n = len(a)
    c = _fft_convolve(a)
    max_b = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf, max_b
    return 2.0 * n * max_b / (S * S), max_b

    A = lil_matrix((2*n-1, n))

import numpy as np
import time
from scipy.fft import rfft, irfft

def _fft_convolve(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = rfft(a, size)
    result = irfft(fa * fa, size)[:2*n-1]
    return np.maximum(result, 0.0)

def _subgrad_at_peaks(a, c, n, frac=0.001, max_k=50):
    """Compute subgradient at the peak lags of autocorrelation c."""
    max_b = c.max()
    threshold = max_b * (1.0 - frac)
    K_star = np.where(c >= threshold)[0]
    
    if len(K_star) > max_k:
        # Keep only the top max_k
        vals = c[K_star]
        idx = np.argsort(vals)[-max_k:]
        K_star = K_star[idx]
    
    if len(K_star) == 0:
        K_star = np.array([np.argmax(c)])
    
    g = np.zeros(n)
    for k in K_star:
        # g[i] += 2*a[k-i] for i where 0 <= k-i < n
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_max >= i_min:
            js = k - np.arange(i_min, i_max + 1)
            g[i_min:i_max+1] += 2.0 * a[js]
    
    g /= len(K_star)
    return g

def _project(a, target_sum):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 1e-15:
        a *= target_sum / S
    else:
        a[:] = target_sum / len(a)
    return a

def _score(a):
    """Compute objective value."""
    n = len(a)
    c = _fft_convolve(a)
    max_b = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_b / (S * S)

    """Run subgradient descent to minimize the objective."""

        # Nesterov momentum
        v = momentum * v + g
        a = a - lr * v
        a = _project(a, target_sum)

            lr *= 0.7
            patience = 0
