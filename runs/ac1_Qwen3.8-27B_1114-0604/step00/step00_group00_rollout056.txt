def build_convolution_matrix(a):
    """Build sparse matrix A where (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # a[k-j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        k_start = j
        k_end = j + n  # exclusive
        k_lo = max(0, k_start)
        k_hi = min(2 * n - 1, k_end)
        if k_lo < k_hi:
            ks = np.arange(k_lo, k_hi)
            rows.extend(ks.tolist())
            cols.extend([j] * (k_hi - k_lo))
            vals.extend((a[ks - j]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_iteration(a, M, time_limit=30):

def greedy_lp(a, M, max_sweeps=100):
    """Greedy column-by-column maximization.
    
    For each j, find the maximum b[j] such that conv(a, b) <= M.
    This is O(n) per coordinate, O(n^2) per sweep.
    """
    n = len(a)
    b = np.zeros(n)  # start from 0
    conv_val = np.zeros(2*n-1)  # current conv(a,b)
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            # Find max delta we can add to b[j]
            # Constraint: for all k with a[k-j] > 0 (0 <= k-j < n):
            #   conv_val[k] + delta * a[k-j] <= M
            # So: delta <= (M - conv_val[k]) / a[k-j]
            
            k_lo = max(0, j)
            k_hi = min(2*n-1, j+n)
            if k_lo >= k_hi:
                continue
            
            ks = np.arange(k_lo, k_hi)
            a_vals = a[ks - j]
            positive = a_vals > 1e-12
            
            if not np.any(positive):
                # No constraint, set b[j] to infinity (cap at 1000)
                delta = 1000.0
            else:
                slack = M - conv_val[ks[positive]]
                max_delta = np.min(slack / a_vals[positive])
                if max_delta <= 0:
                    continue
            
            delta = max(0, min(max_delta, 1000.0 - b[j]))
            if delta > 1e-12:
                b[j] += delta
                conv_val[ks] += delta * a_vals
                improved = True
        
        if not improved:
            break
    
    return b

"""

Algorithm:
1. Generate multiple candidate starting shapes (two-level, Fourier, etc.)

import numpy as np

def compute_score(a):
    """Compute the objective: 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


def convolve_and_score(a):
    """Return (score, convolution, sum)."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf, None, s
    conv = np.convolve(a, a)
    score = 2.0 * n * np.max(conv) / (s * s)
    return score, conv, s


def generate_two_level(n, ratio=4.0 / 3.0):

def generate_fourier(n, coeffs):
    """Generate from cosine Fourier coefficients: a[x] = 1 + sum c_k cos(2pi(k+1)x/n)."""
    x = np.arange(n) / n
    a = np.ones(n)
    for k, c in enumerate(coeffs):
        a += c * np.cos(2.0 * np.pi * (k + 1) * x)
    return np.clip(a, 0.0, 1000.0)


def generate_beta_shape(n, p, q):
    """Generate a Beta(p,q) shaped sequence."""
    x = np.linspace(1e-6, 1.0 - 1e-6, n)

    if np.any(np.isnan(a)) or np.any(np.isinf(a)):

    a = a / np.max(a)
    return a

def build_convolution_matrix(a):
    """Build sparse matrix A where (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_lo = max(0, j)
        k_hi = min(2 * n - 1, j + n)
        if k_lo < k_hi:
            ks = np.arange(k_lo, k_hi)
            rows.extend(ks.tolist())
            cols.extend([j] * (k_hi - k_lo))
            vals.extend(np.asarray(a[ks - j]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_iteration(a, M, time_limit=30):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns b or None."""
    n = len(a)
    A = build_convolution_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def local_search_polish(a, score, deadline, print_interval=500):
    """Local search with random perturbations."""
    n = len(a)
    best = a.copy()

    no_improve = 0
    while time.monotonic() < deadline:
        it += 1
        # Adaptive perturbation size

            scale = 0.1
        elif no_improve > 50:
            scale = 0.05
        else:
            scale = 0.02

        if rng.random() < 0.6:
            # Block perturbation
            width = max(2, n // (50 + no_improve // 20))
            start = rng.integers(0, max(1, n - width))
            candidate = best.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, scale, size=width))
        else:
            # Full small perturbation
            candidate = best * np.exp(rng.normal(0, scale * 0.5, size=n))

                print(f"    LS iter {it}: score={best_score:.8f}")

            no_improve += 1

        ca2_deadline = min(deadline - 30, time.monotonic() + (budget_s * 0.15))
        best_seq, best_score = coordinate_ascent(best_seq, 150, ca2_deadline)
        print(f"Phase 4 done in {time.monotonic()-t3:.1f}s, best={best_score:.8f}")

    # Phase 5: Local search polishing
    if time.monotonic() < deadline - 10:
        print("Phase 5: Local search polishing")
        t4 = time.monotonic()
        best_seq, best_score = local_search_polish(best_seq, best_score, deadline - 5)
        print(f"Phase 5 done in {time.monotonic()-t4:.1f}s, best={best_score:.8f}")

    # Final verification
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final verified score: {final_score:.8f}")
    print(f"Sequence length: {len(best_seq)}")
    print(f"Sum: {np.sum(best_seq):.4f}, Max: {np.max(best_seq):.4f}")

    return [float(x) for x in best_seq]
