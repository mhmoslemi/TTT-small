import numpy as np
from scipy.optimize import linprog
import time

"""
Minimizes score = 2*n*max(conv(a,a))/(sum(a))^2 using subgradient descent
and LP-based direction finding. Explores multiple sequence lengths with
parametric initializations.
"""


def compute_score(a):
    n = len(a)
    if n == 0:
        return float('inf')
    b = np.convolve(a, a)
    S = np.sum(a)
    if S < 1e-10:
        return float('inf')
    return 2.0 * n * float(np.max(b)) / (S * S)


def subgrad_step(a, lr):
    n = len(a)
    b = np.convolve(a, a)
    S = np.sum(a)
    M = np.max(b)
    max_idx = np.where(b == M)[0]

    grad = np.zeros(n)
    for k in max_idx:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            grad[i_start:i_end + 1] += 2.0 * a[j_start:j_end + 1]
    grad /= len(max_idx)

    score_grad = 2.0 * n * grad / (S * S) - 4.0 * n * M / (S ** 3)
    a_new = a - lr * score_grad
    return np.clip(a_new, 0.0, 1000.0)


def lp_direction(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            A_ub[k, j] = a[i]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_refine(a, max_iters=50, time_limit=30.0):
    n = len(a)
    a = a.copy()
    start = time.monotonic()
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        b = np.convolve(a, a)
        M = np.max(b)
        g = lp_direction(a, M)
        if g is None or np.sum(g) <= 1e-12:
            break
        g_sum = np.sum(g)
        S = np.sum(a)
        g_scaled = g * S / g_sum
        current_score = compute_score(a)
        best_t = 0.0
        best_s = current_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
            a_trial = (1 - t) * a + t * g_scaled
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            s = compute_score(a_trial)
            if s < best_s - 1e-12:
                best_s = s
                best_t = t
        if best_t == 0.0:
            break
        a = (1 - best_t) * a + best_t * g_scaled
        a = np.clip(a, 0.0, 1000.0)
    return a


def subgrad_descent(a, max_steps=50000, lr0=0.5, decay=0.9999, time_limit=60.0):
    n = len(a)
    a = a.copy()
    lr = lr0 / max(n, 1)
    start = time.monotonic()
    for step in range(max_steps):
        if time.monotonic() - start > time_limit:
            break
        a = subgrad_step(a, lr)
        if step % 200 == 0:
            lr *= decay
    return a


def make_init(n, kind, rng):
    if kind == 0:
        # linear decreasing
        return np.linspace(2.0, 0.5, n)
    elif kind == 1:
        # quadratic decreasing
        x = np.linspace(1.0, 0.1, n)
        return x ** 2
    elif kind == 2:
        # cubic decreasing
        x = np.linspace(1.0, 0.05, n)
        return x ** 3
    elif kind == 3:
        # exponential decay
        return np.exp(-0.3 * np.arange(n))
    elif kind == 4:
        # power law
        x = np.arange(n, 0, -1, dtype=np.float64)
        return x ** 0.8
    elif kind == 5:
        # power law steeper
        x = np.arange(n, 0, -1, dtype=np.float64)
        return x ** 1.5
    elif kind == 6:
        # random smoothed
        raw = rng.random(n) + 0.1
        kernel = np.ones(3) / 3.0
        return np.convolve(raw, kernel, mode='same')
    elif kind == 7:
        # sharp front then tail
        front = np.exp(-np.arange(n) * 0.5)
        return front + 0.1
    elif kind == 8:
        # linear from high to low
        return np.linspace(3.0, 0.1, n)
    else:
        return np.ones(n)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0

    best_seq = np.ones(4)
    best_score = compute_score(best_seq)
    print(f"Initial best score: {best_score:.6f} (n={len(best_seq)})")

    lengths = [4, 5, 6, 7, 8, 9, 10, 12, 14, 16, 18, 20, 24, 28, 32, 40, 48, 56, 64, 80, 100, 128, 160, 200]
    total_inits_per_n = 4

    for n in lengths:
        if time.monotonic() > deadline:
            break

        nits = min(total_inits_per_n, 9)
        for kind in range(nits):
            if time.monotonic() > deadline:
                break
            a = make_init(n, kind, rng)
            a = np.clip(a, 0.01, 100.0)

            # Subgradient descent
            a = subgrad_descent(a, max_steps=20000, lr0=1.0, decay=0.9998,
                                time_limit=20.0)

            # LP refinement for moderate n
            if n <= 120:
                a = lp_refine(a, max_iters=30, time_limit=20.0)

            # More subgradient after LP
            a = subgrad_descent(a, max_steps=10000, lr0=0.5, decay=0.9999,
                                time_limit=15.0)

            score = compute_score(a)
            if score < best_score - 1e-10:
                best_score = score
                best_seq = a.copy()
                print(f"  n={n}, kind={kind}: score={score:.6f}")

    # Final polish on best
    if time.monotonic() < deadline:
        n_best = len(best_seq)
        # Extended subgradient
        remaining = deadline - time.monotonic()
        a = subgrad_descent(best_seq, max_steps=100000, lr0=0.3, decay=0.99999,
                            time_limit=max(10.0, remaining * 0.4))
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_seq = a
            print(f"After extended subgrad: {best_score:.6f}")

        # LP polish if feasible
        if len(best_seq) <= 150 and time.monotonic() < deadline:
            a = lp_refine(best_seq, max_iters=80,
                          time_limit=max(5.0, (deadline - time.monotonic()) * 0.3))
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_seq = a
                print(f"After final LP: {best_score:.6f}")

        # Tiny subgradient steps
        if time.monotonic() < deadline:
            a = subgrad_descent(best_seq, max_steps=50000, lr0=0.1, decay=0.99999,
                                time_limit=max(5.0, deadline - time.monotonic() - 2.0))
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_seq = a
                print(f"After tiny steps: {best_score:.6f}")

    print(f"FINAL: n={len(best_seq)}, score={best_score:.8f}")
    return [float(x) for x in best_seq]
