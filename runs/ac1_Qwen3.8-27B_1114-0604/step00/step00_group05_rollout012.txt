from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_convolution_matrix(f):
    """Build the sparse convolution matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    f_nz = np.where(f > 1e-12)[0]
    
    all_rows = []
    all_cols = []
    all_data = []
    
    for d in f_nz:
        length = n - d
        all_rows.append(np.arange(d, d + length))
        all_cols.append(np.arange(length))
        all_data.append(np.full(length, f[d]))
    
    if not all_rows:
        return csr_matrix((2*n-1, n))
    
    rows = np.concatenate(all_rows)
    cols = np.concatenate(all_cols)
    data = np.concatenate(all_data)
    
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
