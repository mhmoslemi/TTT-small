Minimize 2*n*max(conv(a,a))/sum(a)^2 using exponentiated gradient descent
with combined energy and peak subgradients. Normalizes sum(a)=sqrt(2n) so

import numpy as np

def normalize_to_sum(a, S):
    """Normalize array to have sum S, maintaining non-negativity."""
    s = a.sum()
    if s < 1e-10:
        return np.ones_like(a) * S / a.size
    return a / s * S


def compute_energy_gradient(B, a):
    """Compute gradient of sum(B[k]^2) w.r.t. a.
    g[m] = 4 * sum_j B[m+j] * a[j] = 4 * correlate(B, a, 'valid')[m]
    """
    C = np.correlate(B, a, mode='valid')
    return 4.0 * C


def compute_peak_gradient(a, k_peak):
    """Compute subgradient of max B[k] at a single peak location k_peak.
    dB[k*]/da[m] = 2*a[k*-m] for 0 <= k*-m < n.
    """
    n = len(a)
    g = np.zeros(n)
    m_lo = max(0, k_peak - n + 1)
    m_hi = min(k_peak, n - 1)
    if m_lo <= m_hi:
        m_vals = np.arange(m_lo, m_hi + 1)
        g[m_vals] = 2.0 * a[k_peak - m_vals]
    return g


def exponentiated_update(a, g, eta):
    """Multiplicative (exponentiated gradient) update preserving non-negativity."""
    scale = np.max(np.abs(g))
    if scale < 1e-15:
        return a.copy()
    g_norm = g / scale
    log_a = np.log(np.maximum(a, 1e-300)) - eta * g_norm
    # Shift to prevent overflow/underflow
    log_a = log_a - np.max(log_a)
    a_new = np.exp(log_a)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s


def spectral_init(n, rng):
    """Initialize a non-negative sequence with approximately flat power spectrum."""
    # Generate random phases
    N = 2 * n
    phases = rng.uniform(0, 2 * np.pi, N)
    # Target flat magnitude spectrum
    mag = np.ones(N) * np.sqrt(n / N)
    mag[0] = 0  # DC
    # Build complex spectrum
    Z = mag * np.exp(1j * phases)
    # Enforce Hermitian symmetry for real output
    Z = (Z + np.conj(Z[::-1])) / 2
    Z = np.fft.irfft(Z, n=N)
    # Take first n samples and make non-negative
    a = np.abs(Z[:n])
    return normalize_to_sum(a, np.sqrt(2.0 * n))


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    a = np.abs(rng.normal(0, 1, n))
    return normalize_to_sum(a, np.sqrt(2.0 * n))
