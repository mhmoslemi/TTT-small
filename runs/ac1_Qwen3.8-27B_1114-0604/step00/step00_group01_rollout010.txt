Minimizes 2*n*max(conv(a,a))/sum(a)^2 using a combination of:
1. Spectral flattening (FFT-based magnitude equalization)
2. Subgradient descent on the peak of the self-convolution
3. Adaptive random block perturbations with restart
Multiple initializations are tried to avoid local minima.

def _normalize(a, n):
    """Project onto non-negative and scale so sum = n."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n, dtype=np.float64)
    a = a * (n / s)
    np.minimum(a, 1000.0, out=a)
    return a


def _conv_full_fft(a, L):
    """Full linear self-convolution via zero-padded FFT."""
    n = len(a)
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)
    return C[:2 * n - 1]


def _spectral_flatten(a, n, L, alpha=0.5, n_iters=300):
    """Iteratively flatten the magnitude spectrum to reduce convolution peak."""
    a = a.copy()
    for _ in range(n_iters):
        A = np.fft.rfft(a, L)
        mag = np.abs(A)
        target = mag[0] / np.sqrt(len(mag))
        with np.errstate(divide='ignore', invalid='ignore'):
            scale = (target / (mag + 1e-15)) ** alpha
        scale = np.clip(scale, 1e-4, 1e4)
        A_new = A * scale
        a_new = np.fft.irfft(A_new, L)[:n]
        a = _normalize(a_new, n)
    return a


def _subgrad_step(a, lr, L):
    """One subgradient descent step to reduce max(conv(a,a))."""
    n = len(a)

    idx = c_star - np.arange(n)

    grad[valid] = 2.0 * a_pad[idx[valid]]
