```python
from scipy.optimize import linprog

def linear_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    idx = np.arange(1, n+1)
    rho = np.nonzero(v_sorted * idx > (cumsum - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    lambda_ = (cumsum[rho] - 1) / (rho + 1)
    return np.maximum(v - lambda_, 0.0)
```

```python
import time
from scipy.optimize import linprog
from scipy.sparse import csc_matrix

def linear_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    idx = np.arange(1, n+1)
    rho = np.nonzero(v_sorted * idx > (cumsum - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    lambda_ = (cumsum[rho] - 1) / (rho + 1)
    return np.maximum(v - lambda_, 0.0)

def subgradient_step(a, k_star, eta):
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    if lo <= hi:
        g[lo:hi+1] = 2.0 * a[k_star - hi : k_star - lo + 1][::-1]
    v = a - eta * g
    return project_simplex(v)

def lp_refinement(a, t=0.3):
    """One LP-based refinement step. Returns updated a."""
    n = len(a)
    b = linear_convolve(a)
    M = b.max()
    
    # Build constraint matrix A (2n-1 x n): A[k,j] = a[k-j] if 0 <= k-j <= n-1
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',

    if result.success:
        g = np.maximum(result.x, 0)
        g_sum = g.sum()
        if g_sum > 0:
            g = g * (a.sum() / g_sum)
            new_a = (1 - t) * a + t * g

    return a
```

```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csc_matrix, csr_matrix

def linear_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je+1] = a[k-je : k-js+1][::-1]
    return A.tocsr()

def lp_step(a, t=0.5, time_limit=60):
    """One LP-based refinement step."""
    n = len(a)
    b = linear_convolve(a)
    M = b.max()
    if M <= 0:
        return a
    
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'time_limit': time_limit})
    
    if result.success:
        g = np.maximum(result.x, 0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            g = g * (a.sum() / g_sum)
            new_a = (1 - t) * a + t * g
            return np.maximum(new_a, 0)
    return a

    a = 0.5 * (1 + np.cos(2 * np.pi * t))
```

One key optimization: I don't need to rebuild the constraint matrix from scratch every iteration. But since $a$ changes each iteration, I do need to update it. However, the sparsity pattern doesn't change (only the values do). So I can reuse the sparsity structure.

```python
def build_conv_matrix_structure(n):
    """Build the sparsity pattern of the convolution matrix."""
    from scipy.sparse import lil_matrix
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je+1] = 1  # just mark positions
    return A

def fill_conv_matrix(A_pattern, a):
    """Fill the convolution matrix with values from a."""
    n = len(a)
    A = A_pattern.tolil()
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je+1] = a[k-je : k-js+1][::-1]
    return A.tocsr()
```
