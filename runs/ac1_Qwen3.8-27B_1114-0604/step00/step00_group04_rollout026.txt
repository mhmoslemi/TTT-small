def _compute_gradient(a, w):
    """
    Compute gradient of log-sum-exp(max approximation) w.r.t. a.
    grad[j] = 2 * sum_{l=0}^{n-1} w[j+l] * a[l]
    Using FFT: grad/2 = (a_rev * w)[n-1 : 2n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    L = 1
    while L < n + len(w) - 1:
        L <<= 1
    A = np.fft.rfft(a_rev, L)
    W = np.fft.rfft(w, L)
    h = np.fft.irfft(A * W, L)
    return 2.0 * h[n-1 : 2*n-1]

    """Compute grad of weighted sum of c_k w.r.t. a. grad[j] = 2*sum_l w[j+l]*a[l]."""
