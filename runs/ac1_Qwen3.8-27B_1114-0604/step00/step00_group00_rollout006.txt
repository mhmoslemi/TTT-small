def local_optimize(p, n, rng, deadline):
    """Local optimization starting from p."""
    p = p.copy()
    p = np.clip(p, 0, None)
    if p.sum() < 1e-10:
        p = np.ones(n) / n
    p = p / p.sum()
    
    best_p = p.copy()
    best_score = fast_score(p)
    
    # Phase 1: Mirror descent
    eta = 1.0
    for t in range(5000):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(p, p)
        k_star = np.argmax(conv)
        
        # Subgradient
        g = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                g[i] = 2 * p[j]
        
        p = p * np.exp(-eta * g)
        p = p / p.sum()
        
        score = fast_score(p)
        if score < best_score:
            best_score = score
            best_p = p.copy()
        
        # Decay eta
        eta *= 0.999
    
    # Phase 2: Random perturbations (simulated annealing)
    temperature = 0.01
    for t in range(10000):
        if time.monotonic() > deadline:
            break
        
        # Perturb
        p_new = p * np.exp(rng.normal(0, 0.1, n))
        p_new = np.clip(p_new, 0, None)
        if p_new.sum() < 1e-10:
            continue
        p_new = p_new / p_new.sum()
        
        score_new = fast_score(p_new)
        if score_new < best_score or rng.random() < np.exp((best_score - score_new) / temperature):
            p = p_new
            if score_new < best_score:
                best_score = score_new
                best_p = p_new.copy()
        
        temperature *= 0.999
    
    return best_p, best_score
