```python
"""
Search for non-negative step-function heights minimizing
2 * n * max(full_convolution(a, a)) / sum(a)^2.

The algorithm:
1. Builds a diverse set of positive initial sequences from the supplied base
   sequence and simple parametric families.
2. Runs projected-gradient / smoothed-max local searches on the best starts.
   The smoothing uses a log-sum-exp approximation of the maximum convolution
   coefficient, with annealed temperature and momentum.
3. Periodically uses a multiplicative mirror-descent step and an active-set
   LP "best response" step when SciPy/HiGHS is available.
4. Finally verifies a few candidates with evaluate_sequence and returns the
   best exact sequence.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
    _HAS_LPROG = True
except Exception:
    _HAS_LPROG = False


def next_power_of_two(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def normalize_nonneg(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return np.array([1.0], dtype=np.float64)
    if not np.isfinite(v).all():
        return np.full(n, 1.0 / n, dtype=np.float64)
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    v /= s
    return v


def convolution_full(a, G):
    """Full convolution a*a and its real FFT, cached in fa."""
    n = a.size
    fa = np.fft.rfft(a, G)
    c = np.fft.irfft(fa * fa, G)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c, fa


def score_and_conv(a, G):
    """Objective assuming arbitrary nonnegative sum, plus the convolution."""
    n = a.size
    S = float(a.sum())
    fa = np.fft.rfft(a, G)
    c = np.fft.irfft(fa * fa, G)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    if not np.isfinite(S) or S <= 1e-15:
        return np.inf, c
    return float(2.0 * n * float(np.max(c)) / (S * S)), c


def score_fft(a):
    n = a.size
    G = next_power_of_two(2 * n - 1)
    sc, _ = score_and_conv(a, G)
    return sc


def make_phase(n, G):
    k = np.fft.rfftfreq(G) * (2.0 * np.pi)
    return np.exp(-1j * k * (n - 1))


def correlate_weights_fft(weights, fa, phase, G, n):
    """
    Return corr[i] = sum_j weights[i+j] * a[j] for i=0..n-1.
    This is the center of convolve(weights, reverse(a), mode='full').
    """
    fw = np.fft.rfft(weights, G)
    far = phase * fa.conj()
    out = np.fft.irfft(fw * far, G)
    corr = out[n - 1:n - 1 + n]
    np.maximum(corr, 0.0, out=corr)
    return corr


def lse_weights(c, beta, n):
    cmax = float(c.max())
    scale = float(beta) * float(n)
    e = np.exp(scale * (c - cmax))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        idx = np.where(c >= cmax - 1e-15)[0]
        if idx.size == 0:
            idx = np.array([0], dtype=int)
        w = np.zeros_like(c)
        w[idx] = 1.0 / float(idx.size)
        return w
    return e / s


def top_lags(c, k):
    if c.size <= k:
        return np.arange(c.size, dtype=int)
    idx = np.argpartition(c, -k)[-k:]
    idx = idx[np.argsort(c[idx])[::-1]]
    return idx.astype(int, copy=False)


def lp_active_step(a, lags, M):
    """
    Solve restricted LP:
        maximize sum(b)
        subject to (a * b)_k <= M for selected lags k, b >= 0.
    Returns normalized b and optimum sum, or None if unavailable.
    """
    if not _HAS_LPROG:
        return None, 0.0

    n = a.size
    K = lags.size
    if K == 0:
        return None, 0.0
    if K > 96:
        lags = lags[:96]
        K = 96

    A = np.zeros((K, n), dtype=np.float64)
    for r, kk in enumerate(lags):
        kk = int(kk)
        lo = max(0, kk - (n - 1))
        hi = min(n - 1, kk)
        if lo <= hi:
            A[r, lo:hi + 1] = a[kk - hi:kk - lo][::-1]

    cobj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(K, float(M), dtype=np.float64)

    try:
        try:
            res = linprog(
                cobj,
                A_ub=A,
                b_ub=b_ub,
                bounds=(0.0, None),
                method="highs",
                options={"time_limit": 10.0}
            )
        except Exception:
            res = linprog(
                cobj,
                A_ub=A,
                b_ub=b_ub,
                bounds=(0.0, None),
                method="highs"
            )
    except Exception:
        return None, 0.0

    if hasattr(res, "success") and res.success and res.x is not None:
        b = np.asarray(res.x, dtype=np.float64)
        B = float(b.sum())
        if np.isfinite(B) and B > 1.0000001:
            b = np.maximum(b, 0.0)
            Bs = float(b.sum())
            if Bs > 0.0:
                return b / Bs, B
    return None, 0.0


def add_candidate(starts, label, x):
    x = np.asarray(x, dtype=np.float64)
    if x.ndim != 1 or x.size == 0:
        return
    x = np.clip(x, 0.0, 1e12)
    s = float(x.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return
    a = x / s
    try:
        sc = score_fft(a)
    except Exception:
        sc = np.inf
    if np.isfinite(sc):
        starts.append((str(label), a, float(sc)))


def beta_shape(x, alpha, beta):
    out = np.ones_like(x, dtype=np.float64)
    if alpha > 0.0:
        out *= np.power(np.maximum(x, 1e-300), alpha)
    if beta > 0.0:
        out *= np.power(np.maximum(1.0 - x, 1e-300), beta)
    return out


def make_starts(n, rng, base):
    starts = []
    ar = np.arange(n, dtype=np.float64)
    x = (ar + 0.5) / float(n)
    theta = 2.0 * np.pi * ar / float(max(1, n))

    if base is not None:
        add_candidate(starts, "base", base)

    add_candidate(starts, "uniform", np.ones(n, dtype=np.float64))

    for lam in (0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0):
        add_candidate(starts, "exp-%.2f" % lam, np.exp(-lam * x))

    for p in (0.25, 0.5, 1.0, 2.0, 4.0, 8.0):
        add_candidate(starts, "pow-%.2f" % p, np.power(np.maximum(1.0 - x, 1e-300), p))

    for alpha in (0.0, 0.25, 0.5, 1.0, 2.0):
        for beta in (0.0, 0.25, 0.5, 1.0, 2.0, 4.0):
            if alpha == 0.0 and beta == 0.0:
                continue
            add_candidate(starts, "beta-%.2f,%.2f" % (alpha, beta), beta_shape(x, alpha, beta))

    for amp in (0.25, 0.5, 0.75, 1.0):
        add_candidate(starts, "cos-%.2f" % amp, 1.0 + amp * np.cos(theta))
        add_candidate(starts, "cos-%.2f+pi" % amp, 1.0 + amp * np.cos(theta + np.pi))

    for amp in (0.25, 0.5, 0.75, 1.0, 1.5):
        for ph in (0.0, np.pi / 2.0, np.pi, 1.5 * np.pi):
            add_candidate(starts, "sine-%.2f-%.1f" % (amp, ph), np.exp(amp * np.sin(theta + ph)))

    for lam in (0.5, 1.0, 1.5, 2.0):
        for amp in (0.2, 0.4, 0.7):
            add_candidate(
                starts,
                "exp+sin-%.1f-%.1f" % (lam, amp),
                np.exp(-lam * x) * (1.0 + amp * np.sin(theta))
            )

    for trial in range(4):
        state = 0.0
        z = np.empty(n, dtype=np.float64)
        for i in range(n):
            state = 0.9 * state + float(rng.normal())
            z[i] = state
        z -= z.max()
        add_candidate(starts, "ar1-%d" % trial, np.exp(z))

    for trial in range(4):
        b = np.zeros(n, dtype=np.float64)
        half = n // 2
        for i in range(half):
            if rng.random() < 0.5:
                b[i] = 1.0
            else:
                b[n - 1 - i] = 1.0
        if n % 2 == 1:
            b[half] = 1.0 if rng.random() < 0.5 else 0.0
        a = 0.8 * b + 0.2 * (1.0 / float(n))
        add_candidate(starts, "binary-%d" % trial, a)

    if not starts:
        add_candidate(starts, "fallback-uniform", np.ones(n, dtype=np.float64))
    return starts


def start_score_key(t):
    return float(t[2])


def local_search(initial, deadline, rng, label, use_lp=False):
    """
    Smoothed-max projected-gradient local search with optional mirror
    descent and active-set LP polish.
    """
    n = int(np.asarray(initial).size)
    G = next_power_of_two(3 * n - 2)
    phase = make_phase(n, G)

    best = normalize_nonneg(np.asarray(initial, dtype=np.float64).copy())
    best_c, _ = convolution_full(best, G)
    Sbest = float(best.sum())
    best_score = float(2.0 * n * float(best_c.max()) / (Sbest * Sbest)) if Sbest > 1e-15 else np.inf

    curr = best.copy()
    v = np.zeros(n, dtype=np.float64)

    it = 0
    last_improve = 0
    beta = 5.0
    step_base = 2.0
    mu = 0.9

    lp_next = time.monotonic() + 35.0
    mirror_next = time.monotonic() + 75.0
    last_print = time.monotonic()

    print("local_search start %s n=%d score=%.8f" % (label, n, best_score), flush=True)

    while time.monotonic() < deadline:
        c, fa = convolution_full(curr, G)
        S = float(curr.sum())
        if not np.isfinite(S) or S <= 1e-15:
            curr = best.copy()
            v = np.zeros(n, dtype=np.float64)
            continue

        score = float(2.0 * n * float(c.max()) / (S * S))
        if np.isfinite(score) and score < best_score * (1.0 - 1e-12) - 1e-15:
            best_score = score
            best = curr.copy()
            best_c = c.copy()
            last_improve = it

        stall = it - last_improve

        weights = lse_weights(c, beta, n)
        corr = correlate_weights_fft(weights, fa, phase, G, n)
        g = corr - float(corr.mean())
        if not np.isfinite(g).all():
            g = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)

        v = mu * v + g
        v -= float(v.mean())

        maxv = float(np.max(np.abs(v)))
        if not np.isfinite(maxv) or maxv <= 1e-300:
            curr = best.copy()
            v = np.zeros(n, dtype=np.float64)
        else:
            step = step_base * (1.0 / float(n)) / maxv
            trial = curr - step * v
            if not np.isfinite(trial).all():
                trial = best.copy()
            curr = normalize_nonneg(trial)

        it += 1
        frac = min(1.0, float(it) / 10000.0)
        beta = 5.0 + 200.0 * frac
        step_base = 2.0 + 8.0 * frac
        mu = 0.9 - 0.25 * frac

        if stall > 800:
            r = rng.random()
            if r < 0.6:
                curr = best.copy()
            else:
                sigma = float(rng.uniform(0.2, 1.0))
                curr = normalize_nonneg(best * np.exp(rng.normal(0.0, sigma, n)))
            v = np.zeros(n, dtype=np.float64)
            beta = max(5.0, 0.5 * beta)
            mu = 0.5
            last_improve = it
            stall = 0

        if time.monotonic() > last_print + 30.0:
            print(
                "local_search %s it=%d best=%.8f stall=%d beta=%.1f"
                % (label, it, best_score, stall, beta),
                flush=True
            )
            last_print = time.monotonic()

        if time.monotonic() > mirror_next and deadline - time.monotonic() > 40.0:
            gscale = float(np.max(np.abs(g)))
            if np.isfinite(gscale) and gscale > 1e-30:
                zbase = np.log(np.maximum(curr, 1e-30))
                for mq in (0.4, 0.8, 1.6, 3.2):
                    z = zbase - mq * (g / gscale)
                    z -= float(z.max())
                    trial = np.exp(z)
                    ts = float(trial.sum())
                    if not np.isfinite(ts) or ts <= 0.0:
                        continue
                    trial /= ts
                    sc, ct = score_and_conv(trial, G)
                    if np.isfinite(sc) and sc < best_score * (1.0 - 1e-12) - 1e-15:
                        best_score = sc
                        best = trial
                        best_c = ct
                        curr = best.copy()
                        v = np.zeros(n, dtype=np.float64)
                        last_improve = it
                        break
            mirror_next = time.monotonic() + 130.0

        if use_lp and _HAS_LPROG and time.monotonic() > lp_next and deadline - time.monotonic() > 80.0:
            K = 96
            lags = top_lags(best_c, K)
            if lags.size > 0:
                alp = best.copy()
                alp += 1e-12
                alp = normalize_nonneg(alp)
                M = max(float(best_c.max()) * 1.05 + 1e-15, 1e-12)
                bdir, B = lp_active_step(alp, lags, M)
                if bdir is not None:
                    for t in (0.25, 0.5, 0.75, 1.0):
                        trial = (1.0 - t) * best + t * bdir
                        trial = normalize_nonneg(trial)
                        sc, ct = score_and_conv(trial, G)
                        if np.isfinite(sc) and sc < best_score * (1.0 - 1e-12) - 1e-15:
                            best_score = sc
                            best = trial
                            best_c = ct
                            curr = best.copy()
                            v = np.zeros(n, dtype=np.float64)
                            last_improve = it
                            print("local_search %s LP improvement=%.8f" % (label, best_score), flush=True)
                            break
            lp_next = time.monotonic() + 65.0

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Return best sequence found before the deadline."""
    start_time = time.monotonic()
    total_budget = max(0.0, float(budget_s))
    deadline = start_time + max(5.0, total_budget - 12.0)
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base = None

    n = 6121
    base_arr = None
    if base is not None:
        base = np.clip(base, 0.0, 1e6)
        if 50 <= base.size <= 30000:
            n = int(base.size)
            base_arr = base
        else:
            base_arr = None

    # Generate and rank initial candidates.
    t0 = time.monotonic()
    starts = make_starts(n, rng, base_arr)
    starts.sort(key=start_score_key)
    start_time_used = time.monotonic() - t0
    print("start generation: %d candidates, best_fft=%.8f, time=%.2fs"
          % (len(starts), starts[0][2] if starts else float("inf"), start_time_used), flush=True)

    global_best = starts[0][1].copy() if starts else np.ones(n, dtype=np.float64)
    global_best = normalize_nonneg(global_best)
    global_best_score = starts[0][2] if starts else score_fft(global_best)

    # Time slices for the best independent starts.
    top_count = min(5, len(starts))
    default_slices = (900.0, 500.0, 350.0, 250.0, 150.0)
    main_reserve = 420.0

    used_time = start_time_used
    for i in range(top_count):
        remaining = deadline - time.monotonic() - main_reserve
        if remaining < 20.0:
            break
        slice_s = min(default_slices[i], remaining)
        if slice_s < 10.0:
            break
        end = time.monotonic() + slice_s
        label = ("start%d" % i) + (("/" + starts[i][0]) if i < len(starts) else "")
        try:
            cand, sc = local_search(starts[i][1], end, rng, label, use_lp=(i < 2))
        except Exception:
            cand, sc = starts[i][1], np.inf
        if np.isfinite(sc) and sc < global_best_score:
            global_best = cand
            global_best_score = sc
            print("new global best fft score after %s: %.8f" % (label, global_best_score), flush=True)
        used_time += slice_s

    # Final polish from the current global best.
    remaining = deadline - time.monotonic() - 25.0
    if remaining > 120.0:
        try:
            cand, sc = local_search(global_best, time.monotonic() + remaining, rng, "final", use_lp=True)
        except Exception:
            cand, sc = global_best, np.inf
        if np.isfinite(sc) and sc < global_best_score:
            global_best = cand
            global_best_score = sc
            print("new global best fft score final: %.8f" % global_best_score, flush=True)

    # Evaluate a small set of candidates exactly using the authoritative function.
    candidates = []
    candidates.append(("final_best", global_best))
    for i in range(min(5, len(starts))):
        candidates.append(("start%d" % i, starts[i][1]))
    # Add a few deterministically safe fallbacks.
    if n > 0:
        candidates.append(("uniform", np.ones(n, dtype=np.float64) / float(n)))
        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        candidates.append(("exp1", np.exp(-x)))

    best_exact = None
    best_exact_score = np.inf
    for name, a in candidates:
        a = normalize_nonneg(np.asarray(a, dtype=np.float64).copy())
        if a.size == 0:
            continue
        try:
            s = evaluate_sequence([float(y) for y in a])
        except Exception:
            s = np.inf
        if np.isfinite(s) and s < best_exact_score:
            best_exact_score = float(s)
            best_exact = a
            print("exact candidate %s score=%.8f" % (name, best_exact_score), flush=True)

    if best_exact is None:
        best_exact = normalize_nonneg(np.ones(n, dtype=np.float64) / float(n))

    return [float(y) for y in best_exact]
```
