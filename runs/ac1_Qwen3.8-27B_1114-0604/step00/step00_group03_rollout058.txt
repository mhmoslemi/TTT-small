```
    n = int(_N_OPT)
```

```python
"""
Search for a low-score non-negative step function.

The evaluator minimizes:
    2 * n * max(np.convolve(f, f)) / (sum(f) ** 2)

This program uses:
- cheap deterministic/ random initial profiles,
- FFT-based smoothing of the max-convolution objective,
- L-BFGS in a softmax parameterization,
- multiplicative projected-gradient refinements,
- exact final verification through evaluate_sequence().
"""

import time
import math

import numpy as np

try:
    from scipy.optimize import minimize as _scipy_minimize
except Exception:
    _scipy_minimize = None

_BETA = 1.0


def _normalize(f, floor=1e-12):
    f = np.asarray(f, dtype=np.float64)
    if f.ndim != 1 or f.size == 0:
        return np.ones(1, dtype=np.float64)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, None)
    if floor > 0:
        f = np.maximum(f, floor)
    s = float(np.sum(f))
    if not np.isfinite(s) or s <= 0:
        f = np.ones(f.size, dtype=np.float64)
        s = float(f.size)
    f = f / s
    return f


def _fft_conv(f):
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    F = np.fft.rfft(f, L)
    g = np.fft.irfft(F * F, L)
    g = g[:need]
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    return g


def _fft_corr(h, f):
    h = np.asarray(h, dtype=np.float64)
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    need = h.size + n - 1
    L = 1
    while L < need:
        L <<= 1
    H = np.fft.rfft(h, L)
    Fr = np.fft.rfft(f[::-1], L)
    c = np.fft.irfft(H * Fr, L)
    out = c[n - 1:2 * n - 1]
    if out.size < n:
        out = np.pad(out, (0, n - out.size), mode="constant")
    elif out.size > n:
        out = out[:n]
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    return out


def _fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    s = float(np.sum(f))
    if n <= 0 or not np.isfinite(s) or s <= 0:
        return float("inf")
    g = _fft_conv(f)
    if g.size == 0:
        return float("inf")
    maxb = float(np.max(g))
    if not np.isfinite(maxb):
        return float("inf")
    return float((2.0 * n * maxb) / (s * s))


def _f_from_u(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return np.ones(1, dtype=np.float64)
    u = u - float(np.mean(u))
    z = u - float(np.max(u))
    z = np.clip(z, -700.0, 0.0)
    ez = np.exp(z)
    s = float(np.sum(ez))
    if not np.isfinite(s) or s <= 0:
        return np.ones(u.size, dtype=np.float64) / u.size
    return ez / s


def _u_from_f(f):
    f = _normalize(f)
    u = np.log(np.maximum(f, 1e-12))
    u = u - float(np.mean(u))
    return u


def _smooth_objective_u(x):
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    beta = float(_BETA)
    if n <= 0:
        return (1e300, np.zeros(0, dtype=np.float64))
    if not np.all(np.isfinite(x)):
        return (1e300, np.zeros_like(x, dtype=np.float64))

    f = _f_from_u(x)
    g = _fft_conv(f)
    m = float(np.max(g))
    if not np.isfinite(m):
        return (1e300, np.zeros_like(x, dtype=np.float64))

    if beta <= 1e-12:
        val = float(2.0 * n * m)
        h = np.full(g.shape, 1.0 / g.size, dtype=np.float64)
    else:
        shifted = beta * (g - m)
        shifted = np.clip(shifted, -700.0, 0.0)
        e = np.exp(shifted)
        P = float(np.sum(e))
        if not np.isfinite(P) or P <= 0:
            e = np.full(g.shape, 1.0 / g.size, dtype=np.float64)
            P = 1.0
        else:
            e = e / P
        val = float(2.0 * n * (m + math.log(P) / beta))
        h = e

    corr = _fft_corr(h, f)
    grad_f = 4.0 * float(n) * corr
    du = grad_f * f
    dot = float(np.dot(du, f))
    du = du - f * dot
    du = np.where(np.isfinite(du), du, 0.0)

    if not np.isfinite(val):
        val = 1e300
    return (val, du)


def _lbfgs_anneal(f, deadline, maxiter=200):
    f = _normalize(f)
    n = f.size
    best_f = f.copy()
    best_score = _fast_score(f)
    u = _u_from_f(f)

    if _scipy_minimize is None:
        return best_f, best_score

    mults = (1, 2, 4, 8, 16, 32, 64)
    for mult in mults:
        if time.monotonic() >= deadline:
            break
        _BETA = float(mult * n)
        try:
            res = _scipy_minimize(
                _smooth_objective_u,
                u,
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": int(maxiter),
                    "ftol": 1e-10,
                    "gtol": 1e-7,
                    "maxls": 80,
                    "maxcor": 12,
                },
            )
            if hasattr(res, "x"):
                rx = np.asarray(res.x, dtype=np.float64)
                if rx.size == n and np.all(np.isfinite(rx)):
                    f2 = _f_from_u(rx)
                    sc2 = _fast_score(f2)
                    if np.isfinite(sc2) and sc2 < best_score:
                        best_f = f2
                        best_score = sc2
                    u = rx.copy()
        except Exception:
            pass

    return best_f, best_score


def _multiplicative_anneal(f, rng, deadline, max_iters=20000):
    if time.monotonic() >= deadline:
        return _normalize(f), float("inf")

    f = _normalize(f)
    n = f.size
    best_f = f.copy()
    best_score = _fast_score(f)

    base_beta = 1.0
    factor = 0.08
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        g = _fft_conv(f)
        m = float(np.max(g))
        score = float(2.0 * n * m)
        if not np.isfinite(score):
            break

        if score < best_score - 1e-12:
            best_f = f.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            w = int(max(1, n // int(rng.integers(30, 120))))
            if w >= n:
                w = max(1, n // 10)
            st = int(rng.integers(0, max(1, n - w + 1)))
            f2 = f.copy()
            f2[st:st + w] *= float(np.exp(rng.normal(0.0, 0.7)))
            f2 = _normalize(f2)
            sc2 = _fast_score(f2)
            if np.isfinite(sc2) and sc2 < best_score:
                f = f2
                score = sc2
                best_f = f.copy()
                best_score = sc2
                no_improve = 0
                factor = 0.04
            else:
                no_improve = 0

        beta = float(base_beta * n)
        shifted = beta * (g - m)
        shifted = np.clip(shifted, -700.0, 0.0)
        e = np.exp(shifted)
        P = float(np.sum(e))
        if P <= 0 or not np.isfinite(P):
            e = np.full(g.shape, 1.0 / g.size, dtype=np.float64)
        else:
            e = e / P

        corr = _fft_corr(e, f)
        maxc = float(np.max(corr))
        if maxc <= 0 or not np.isfinite(maxc):
            break

        accepted = False
        for sf in (factor, 0.5 * factor, 0.25 * factor, 2.0 * factor):
            f2 = f * np.exp(-sf * corr / maxc)
            f2 = _normalize(f2)
            sc2 = _fast_score(f2)
            if np.isfinite(sc2) and (sc2 < score - 1e-12 or sc2 < best_score - 1e-12):
                f = f2
                score = sc2
                if sc2 < best_score:
                    best_f = f.copy()
                    best_score = sc2
                factor = min(0.6, factor * 1.08)
                no_improve = 0
                accepted = True
                break

        if not accepted:
            factor = max(0.005, factor * 0.80)
        else:
            base_beta = min(120.0, base_beta * 1.01)

    return best_f, best_score


def _two_block_profile(n, q, r):
    n = int(n)
    m = int(round(float(q) * n))
    m = max(1, min(n - 1, m))
    f = np.full(n, float(r), dtype=np.float64)
    f[:m] = 1.0
    return _normalize(f)


def _three_block_profile(n, l1, l2, h1, h2, h3):
    n = int(n)
    m1 = int(round(float(l1) * n))
    m2 = int(round(float(l2) * n))
    m1 = max(1, min(n - 2, m1))
    m2 = max(1, min(n - m1 - 1, m2))
    m3 = n - m1 - m2
    f = np.empty(n, dtype=np.float64)
    f[:m1] = float(h1)
    f[m1:m1 + m2] = float(h2)
    f[m1 + m2:] = float(h3)
    return _normalize(f)


def _geometric_profile(n, ratio, nblocks):
    n = int(n)
    ratios = []
    r = 1.0
    for _ in range(int(nblocks)):
        ratios.append(r)
        r *= float(ratio)
    s = sum(ratios)
    f = np.zeros(n, dtype=np.float64)
    pos = 0
    for rr in ratios:
        m = int(round((rr / s) * n))
        if pos + m >= n:
            m = n - pos
        if m > 0:
            val = 1.0 / math.sqrt(max(rr, 1e-12))
            f[pos:pos + m] = val
        pos += m
        if pos >= n:
            break
    return _normalize(f)


def _random_profile(rng, n, x_units):
    m = int(rng.integers(6, 20))
    x = np.linspace(0.0, 1.0, m)
    mode = int(rng.integers(0, 4))
    if mode == 0:
        y = np.exp(rng.normal(0.0, 1.0, m))
    elif mode == 1:
        y = 1.0 + rng.normal(0.0, 0.7, m)
        y = np.maximum(y, 1e-3)
    elif mode == 2:
        y = 1.0 + float(rng.normal(0.0, 0.8)) * (2.0 * x - 1.0)
        y = np.maximum(y, 1e-3)
    else:
        y = np.abs(rng.normal(0.0, 1.0, m)) + 1e-3
    f = np.interp(x_units, x, y)
    return _normalize(f)


def _score_sort_profiles(profs):
    arr = []
    for i, p in enumerate(profs):
        sc = _fast_score(p)
        if np.isfinite(sc):
            arr.append((float(sc), int(i), p))
    arr.sort()
    return [p for sc, i, p in arr]


def _base_profiles(n, rng, limit=20, n_random=20):
    n = int(n)
    x_units = np.linspace(0.0, 1.0, n)
    profs = []

    for q in (0.05, 0.10, 0.15, 0.20, 0.25, 0.30, 0.40, 0.50):
        for r in (0.10, 0.25, 0.50, 0.75, 1.00, 1.50, 2.00, 3.00, 5.00):
            profs.append(_two_block_profile(n, q, r))

    heights = (
        (2.0, 1.0, 1.0),
        (1.0, 2.0, 1.0),
        (1.0, 1.0, 2.0),
        (2.0, 0.5, 1.0),
        (0.5, 2.0, 1.0),
        (2.0, 1.0, 0.5),
        (0.5, 1.0, 2.0),
        (1.0, 0.2, 0.5),
    )
    for l1 in (0.10, 0.20, 0.30):
        for l2 in (0.10, 0.20, 0.30):
            for h in heights:
                profs.append(_three_block_profile(n, l1, l2, h[0], h[1], h[2]))

    for ratio in (2.0, 3.0, 4.0):
        for nb in (3, 4, 5):
            profs.append(_geometric_profile(n, ratio, nb))

    for _ in range(max(0, int(n_random))):
        profs.append(_random_profile(rng, n, x_units))

    # A few smooth polynomial-like profiles.
    for _ in range(12):
        x = x_units
        c1 = float(rng.normal(0.0, 1.0))
        c2 = float(rng.normal(0.0, 1.0))
        c3 = float(rng.normal(0.0, 0.5))
        f = (
            1.0
            + c1 * (2.0 * x - 1.0)
            + c2 * (3.0 * x * x - 3.0 * x + 1.0)
            + c3 * (4.0 * x * x * x - 6.0 * x * x + 3.0 * x)
        )
        f = np.maximum(f, 1e-3)
        profs.append(_normalize(f))

    ordered = _score_sort_profiles(profs)
    return ordered[:max(1, int(limit))]


def _upsample_f(f, n_new):
    f = _normalize(f)
    n = f.size
    n_new = int(n_new)
    if n == n_new:
        return f.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x_new, x_old, f))


def _exact_score(seq):
    ev = globals().get("evaluate_sequence", None)
    if ev is None:
        return float("inf")
    try:
        return float(ev(seq))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 30.0)
    end_time = start + total_budget
    rng = np.random.default_rng(seed)

    fallback = [2.0, 1.0, 1.0, 1.0, 1.0]
    best_f = _normalize(fallback)
    best_score = _exact_score(fallback)
    if not np.isfinite(best_score):
        best_score = _fast_score(best_f)

    # Use provided construction if it exists.
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.ndim == 1 and hs.size > 0 and np.all(np.isfinite(hs)):
            hf = _normalize(hs)
            hs_score = _fast_score(hf)
            if np.isfinite(hs_score) and hs_score < best_score:
                best_f = hf
                best_score = hs_score
                print("Initial provided construction score:", best_score)
    except Exception:
        pass

    print("Initial best score (fast):", best_score)

    # Small cheap profile pass.
    if time.monotonic() < end_time:
        d = min(end_time, start + max(10.0, total_budget * 0.02))
        profs = _base_profiles(128, rng, limit=12, n_random=10)
        for p in profs:
            sc = _fast_score(p)
            if np.isfinite(sc) and sc < best_score:
                best_f = p.copy()
                best_score = sc
            if time.monotonic() >= d:
                break
        print("After 128 profile pass:", best_score)

    # Medium optimization on N=512.
    if time.monotonic() < end_time - 120.0:
        n = 512
        inits = _base_profiles(n, rng, limit=8, n_random=10)
        inits.append(_upsample_f(best_f, n))
        inits = _score_sort_profiles(inits)[:5]

        for idx, p in enumerate(inits):
            if time.monotonic() >= end_time - 180.0:
                break
            give = max(45.0, min(180.0, (end_time - 180.0 - time.monotonic()) / max(1, len(inits) - idx)))
            dd = time.monotonic() + give
            f, sc = _lbfgs_anneal(p, dd, maxiter=140)
            if time.monotonic() < end_time - 120.0:
                dd2 = time.monotonic() + max(15.0, min(60.0, (end_time - 120.0 - time.monotonic()) / 2.0))
                f2, sc2 = _multiplicative_anneal(f, rng, dd2, max_iters=3000)
                if np.isfinite(sc2) and sc2 < sc:
                    f = f2
                    sc = sc2
            if np.isfinite(sc) and sc < best_score:
                best_f = f.copy()
                best_score = sc
                print("N=512 improvement:", best_score)

    # Main optimization on N=1024.
    if time.monotonic() < end_time - 120.0:
        n = 1024
        inits = []
        inits.append(_upsample_f(best_f, n))
        inits.extend(_base_profiles(n, rng, limit=8, n_random=10))
        inits = _score_sort_profiles(inits)[:4]

        for idx, p in enumerate(inits):
            remaining = end_time - 150.0 - time.monotonic()
            if remaining <= 0:
                break
            give = max(60.0, min(900.0, remaining / max(1, len(inits) - idx)))
            dd = time.monotonic() + give
            f, sc = _lbfgs_anneal(p, dd, maxiter=220)
            if time.monotonic() < end_time - 90.0:
                give2 = max(20.0, min(240.0, (end_time - 90.0 - time.monotonic()) / 2.0))
                dd2 = time.monotonic() + give2
                f2, sc2 = _multiplicative_anneal(f, rng, dd2, max_iters=12000)
                if np.isfinite(sc2) and sc2 < sc:
                    f = f2
                    sc = sc2
            if np.isfinite(sc) and sc < best_score:
                best_f = f.copy()
                best_score = sc
                print("N=1024 improvement:", best_score)

    # Final refinement at N=2048 using the best shape found.
    if time.monotonic() < end_time - 60.0:
        n = 2048
        f = _upsample_f(best_f, n)
        sc = _fast_score(f)
        if np.isfinite(sc) and sc < best_score:
            best_f = f.copy()
            best_score = sc
        f2, sc2 = _multiplicative_anneal(f, rng, end_time - 20.0, max_iters=12000)
        if np.isfinite(sc2) and sc2 < best_score:
            best_f = f2.copy()
            best_score = sc2
            print("N=2048 final improvement:", best_score)

    final_f = _normalize(best_f)
    final_list = [float(x) for x in final_f]

    # Exact verification; fall back to a known valid sub-2 sequence if needed.
    final_score = _exact_score(final_list)
    fb_score = _exact_score(fallback)
    if not np.isfinite(fb_score):
        fb_score = 2.0

    if (not np.isfinite(final_score)) or final_score > fb_score:
        return fallback

    print("Final selected score:", final_score)
    return final_list
```
