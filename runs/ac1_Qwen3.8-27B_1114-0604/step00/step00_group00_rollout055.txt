```python
import numpy as np
import time

def next_power_of_2(n):
    return 1 << (n - 1).bit_length()

def project_simplex(v):
    """Project vector v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_convolution_fft(a, N_fft):
    """Compute linear convolution of a with itself using FFT."""
    A = np.fft.rfft(a, N_fft)
    B = A * A
    b = np.fft.irfft(B, N_fft)
    return b[:2*len(a)-1]

def compute_gradient(a, p, tau, N_fft):
    """Compute gradient of smooth max of auto-convolution."""
    n = len(a)
```
