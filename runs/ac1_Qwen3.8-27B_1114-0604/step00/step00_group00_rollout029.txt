    n = len(f)
    b = np.convolve(f, f)
    k_star = int(np.argmax(b))
    # Subgradient: for each j where k_star - j is in [0, n-1], grad_j = f[k_star-j]
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n - 1, k_star)
    grad = np.zeros(n)
    grad[j_lo:j_hi+1] = f[k_star - j_hi : k_star - j_lo + 1][::-1]
    # Wait, let me recompute. b[k_star] = sum_j f[j] * f[k_star - j]
    # d(b[k_star])/d(f[j]) = f[k_star - j] (for valid j)
    # So for j in [j_lo, j_hi]: grad[j] = f[k_star - j]
    grad = np.zeros(n)
    for j in range(j_lo, j_hi + 1):
        grad[j] = f[k_star - j]
    
    f_new = f - eta * grad
    return normalize(f_new)

def block_perturb(p, rng, block_frac_range=(0.005, 0.2), sigma_range=(0.05, 0.5)):

"""
Algorithm: Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
- Phase 0: Subgradient warm-up to flatten convolution peak.
- Phase 1: LP-based refinement (alternating between fixing f and optimizing g via LP).
- Phase 2: Simulated annealing with structured moves (block perturbations, symmetric updates).
- Phase 3: Fine-tuning with small perturbations.
Uses numpy FFT for fast convolution and scipy HiGHS for LP solving.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def safe_eval(seq):
    """Evaluate a sequence safely, returning inf on failure."""
    try:
        arr = np.asarray(seq, dtype=np.float64)
        if arr.size == 0:
            return float('inf')
        score = evaluate_sequence([float(x) for x in arr])
        if np.isfinite(score):
            return float(score)
    except Exception:
        pass
    return float('inf')


def norm_unit(p):
    """Normalize to unit sum, ensuring non-negativity."""
    p = np.maximum(np.asarray(p, dtype=np.float64), 0.0)
    s = np.sum(p)
    if s < 1e-300:
        return np.ones(p.shape[0]) / p.shape[0]
    return p / s


def build_conv_mat(f, n):
    """Build sparse matrix A where (A@g)_k = sum_j f_j * g_{k-j} for valid j."""
    n_rows = 2 * n - 1
    total_nnz = n * n
    data = np.empty(total_nnz, dtype=np.float64)
    indices = np.empty(total_nnz, dtype=np.int32)
    indptr = np.empty(n_rows + 1, dtype=np.int64)
    pos = 0
    indptr[0] = 0
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        length = j_hi - j_lo + 1
        data[pos:pos + length] = f[j_lo:j_hi + 1]
        indices[pos:pos + length] = np.arange(j_lo, j_hi + 1, dtype=np.int32)
        pos += length
        indptr[k + 1] = pos
    return csr_matrix((data, indices, indptr), shape=(n_rows, n))


def lp_one_step(f, time_limit=100.0):
    """Solve one LP refinement step. Returns g or None."""
    n = len(f)
    f = norm_unit(f)
    b = np.convolve(f, f)
    M = float(np.max(b))

    t0 = time.monotonic()
    A = build_conv_mat(f, n)
    t_build = time.monotonic() - t0

    if time.monotonic() - t0 > time_limit * 0.4:
        return None

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    t0 = time.monotonic()
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'dual_obj_bound': -1e10,
                               'primal_obj_bound': -1e10, 'time_limit': time_limit - (time.monotonic() - t0)})
    except Exception as e:
        print(f"    LP exception: {e}")
        return None

    t_solve = time.monotonic() - t0
    if not res.success:
        print(f"    LP failed: {res.message}")
        return None
    if np.sum(res.x) < 1e-10:
        return None
    return norm_unit(res.x)


def lp_refine(f, max_time=250.0):
    """Iterative LP-based refinement."""
    f = norm_unit(f)
    best_f = f.copy()
    best_score = safe_eval(f)
    start = time.monotonic()
    t_mix = 0.4
    n_iters = 0
    print(f"  LP start: score={best_score:.6f}, n={len(f)}")

    while time.monotonic() - start < max_time:
        n_iters += 1
        rem = max_time - (time.monotonic() - start)
        if rem < 60:
            break
        g = lp_one_step(f, time_limit=min(120, rem - 10))
        if g is None:
            t_mix = max(t_mix * 0.5, 0.1)
            continue

        f_new = (1 - t_mix) * f + t_mix * g
        f_new = norm_unit(f_new)
        score = safe_eval(f_new)
        print(f"  LP iter {n_iters}: score={score:.6f} (best={best_score:.6f}, t={t_mix:.3f})")

        if score < best_score:
            best_f = f_new.copy()
            best_score = score
            f = f_new
        else:
            t_mix = max(t_mix * 0.5, 0.05)
            if t_mix <= 0.05:
                # Try full replacement
                score_full = safe_eval(g)
                if score_full < best_score:
                    best_f = g.copy()
                    best_score = score_full
                    f = g
                    t_mix = 0.4
                else:
                    break

    print(f"  LP done: {n_iters} iters, best={best_score:.6f}, time={time.monotonic()-start:.0f}s")
    return best_f


def subgrad_warmup(f, n_iters=500, eta_start=0.05, eta_end=0.001):

    t_phase3 = min(60.0, 0.03 * (deadline - time.monotonic()))
    if t_phase3 > 10.0:
        print(f"Phase 3: Final polish (budget {t_phase3:.0f}s)")
        f_pol = final_polish(overall_best, max_time=t_phase3, seed=seed + 2000)
        s = safe_eval(f_pol)
        if s < overall_best_score:
            overall_best_score = s
            overall_best = f_pol.copy()

    # Final check
    final_score = safe_eval(overall_best)
    if not np.isfinite(final_score):
        print("WARNING: final sequence is invalid, returning uniform")
        return [1.0 / n] * n

    print(f"FINAL: score={final_score:.6f}, n={len(overall_best)}")
    return [float(x) for x in overall_best]
