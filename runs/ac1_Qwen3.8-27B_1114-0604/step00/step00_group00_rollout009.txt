def next_pow2(n):
    """Next power of 2 >= n."""
    n = int(n)
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def autocorr(f):
    """Linear autocorrelation f*f via FFT. Returns array of length 2*len(f)-1."""
    n = len(f)
    size = next_pow2(2 * n)
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * np.conj(F), size)[:2 * n - 1]

    """One Gordon step. Returns (f_new, r)."""
    n = len(f)
    r = autocorr(f)
