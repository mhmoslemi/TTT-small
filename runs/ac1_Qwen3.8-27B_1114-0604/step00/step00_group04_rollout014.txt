import numpy as np
import time
from scipy.optimize import linprog

def solve_lp(f, M):
    """Solve the LP: maximize sum(g) subject to f*g <= M, g >= 0."""
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)_k <= M for k = 0, ..., 2n-2
    # (f*g)_k = sum_i f_i * g_{k-i} where g_j = 0 if j not in [0, n-1]
    # For k from 0 to 2n-2:
    #   sum_{i=max(0,k-n+1)}^{min(n-1,k)} f_i * g_{k-i} <= M
    
    # Build constraint matrix A such that A @ g <= b
    # A has 2n-1 rows and n columns
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n-1, k)
        for i in range(i_min, i_max + 1):
            A[k, i] = f[i]
    
    b = np.full(2*n-1, M)

    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')

        return None

"""
LP best-response iteration + local search to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses alternating LP: given f, solve max sum(g) s.t. cross-correlation f*g <= ||f*f||_inf,
then blend. Multiple starting points and sequence lengths explored.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix(f):
    """Build sparse banded matrix A where (A @ g)[k] = sum_i f[i]*g[k-i]."""
    n = len(f)
    row_parts = []
    col_parts = []
    data_parts = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        cnt = hi - lo + 1
        if cnt > 0:
            row_parts.append(np.full(cnt, k, dtype=np.int32))
            col_parts.append(np.arange(lo, hi + 1, dtype=np.int32))
            data_parts.append(f[lo:hi + 1])
    if row_parts:
        rows = np.concatenate(row_parts)
        cols = np.concatenate(col_parts)
        data = np.concatenate(data_parts)
    else:
        rows = np.zeros(0, dtype=np.int32)
        cols = np.zeros(0, dtype=np.int32)
        data = np.zeros(0, dtype=np.float64)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_best_response(f, M, time_limit=20.0):
    """Solve LP: max 1'g s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A = build_conv_matrix(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_iterate(f, S, deadline, verbose=True):
    """Run best-response LP iterations until deadline. Returns (best_f, best_score, iters)."""
    n = len(f)
    best_f = f.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    iters = 0
    stall = 0

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 20:
            break
        iters += 1
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))

        remaining = deadline - time.monotonic()
        t0 = time.monotonic()
        g = solve_best_response(f, M, time_limit=max(3.0, remaining * 0.7))
        lp_time = time.monotonic() - t0

        if g is None or not np.isfinite(g).all():
            if verbose:
                print(f"  iter {iters}: LP failed ({lp_time:.1f}s), stopping")
            break

        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            if verbose:
                print(f"  iter {iters}: degenerate solution, stopping")
            break

        g = g / sum_g * S

        # Adaptive step: start with 0.5, reduce if oscillating
        alpha = 0.5
        new_f = (1.0 - alpha) * f + alpha * g
        new_f = np.maximum(new_f, 0.0)

        new_score = float(evaluate_sequence(new_f.tolist()))
        if np.isfinite(new_score) and new_score < best_score:
            best_f = new_f.copy()
            best_score = new_score
            stall = 0
            if verbose and (iters <= 5 or iters % 10 == 0):
                print(f"  iter {iters}: score={best_score:.6f}, M={M:.4f}, lp={lp_time:.1f}s")
        else:
            stall += 1

        f = new_f

        if best_score < 1.5:
            if verbose:
                print(f"  Target reached! score={best_score:.6f} at iter {iters}")
            break
        if stall > 40:
            if verbose:
                print(f"  Stalled after {stall} iters (best={best_score:.6f}), stopping")
            break

    return best_f, best_score, iters


def local_polish(f, deadline, seed=777):
    """Greedy local search with multiplicative block perturbations."""
    n = len(f)
    best_f = f.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    rng = np.random.default_rng(seed)
    iters = 0
    step_size = 0.15

    while time.monotonic() < deadline:
        iters += 1
        if time.monotonic() > deadline - 2:
            break
        width = max(2, int(rng.integers(n // 200, max(3, n // 4))))
        start = int(rng.integers(0, n - width + 1))
        end = min(n, start + width)

        candidate = best_f.copy()
        candidate[start:end] *= np.exp(rng.normal(0.0, step_size, size=end - start))
        candidate = np.clip(candidate, 0.0, 1000.0)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_f = candidate
            best_score = score
            step_size = max(0.02, step_size * 0.95)
        else:
            if iters % 200 == 0:
                step_size = min(0.3, step_size * 1.1)

        if iters % 200 == 0 and best_score < 1.55:
            print(f"  polish iter {iters}: score={best_score:.6f}")

    return best_f, best_score
