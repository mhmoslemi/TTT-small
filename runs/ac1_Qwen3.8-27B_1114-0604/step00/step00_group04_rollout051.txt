from scipy.optimize import linprog

def _normalize(x):
    """Normalize to sum 1, clip to non-negative."""
    x = np.clip(x, 0.0, None)
    s = x.sum()
    if s < 1e-15:
        return np.ones_like(x) / len(x)
    return x / s

def _score(a):
    """Compute 2n*max(a*a)/S^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(a, a, mode='full')
    return 2.0 * n * conv.max() / (S * S)

def _lp_response(a, max_constraints=200, max_iters=30):
    """Find b minimizing max_k (a*b)_k s.t. b>=0, sum(b)=1.
    Uses constraint generation."""
    n = a.size
    # Start with a few constraints
    # Include the center lag and a few others
    center = n - 1
    initial_lags = list(set([center, max(0, center-1), min(2*n-2, center+1), 0, 2*n-2, n//2]))
    
    for outer in range(max_iters):
        # Solve LP with current constraints
        K = len(initial_lags)
        # Build constraint matrix: for lag k, A[i] = a[k-i] if 0<=k-i<n
        A = np.zeros((K, n))
        for row, k in enumerate(initial_lags):
            # a[k-i] for i=0..n-1, valid when 0<=k-i<n i.e. k-n+1<=i<=k
            lo = max(0, k - n + 1)
            hi = min(n-1, k)
            if hi >= lo:
                for i in range(lo, hi+1):
                    A[row, i] = a[k - i]
        
        # Variables: b_0..b_{n-1}, M
        # Minimize M
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        # Inequality: A@b - M <= 0  =>  [A, -1] @ [b; M] <= 0
        A_ub = np.hstack([A, -np.ones((K, 1))])
        b_ub = np.zeros(K)
        
        # Equality: sum(b) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * (n + 1)
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        b = np.clip(res.x[:n], 0, None)
        M = max(0.0, res.x[n])
        
        # Check all constraints
        conv = np.convolve(a, b, mode='full')
        # Find violated constraints
        violated = np.where(conv > M + 1e-10)[0]
        
        if len(violated) == 0:
            return b, M

        for v in violated[:50]:  # add at most 50 new constraints per iteration
            if v not in initial_lags:
                initial_lags.append(v)
        
        if len(initial_lags) > max_constraints:
            # Keep the most violated ones
            conv_vals = np.convolve(a, b, mode='full')
            top_idx = np.argsort(conv_vals)[::-1][:max_constraints]
            initial_lags = list(set(top_idx.tolist()))
    
    return b, M

def _block_perturb(a, rng, strength=0.05):
    """Apply multiplicative block perturbation."""
    n = len(a)
    x = a.copy()
    n_blocks = rng.integers(1, max(2, n // 500))

        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        x[start:stop] *= float(np.exp(rng.normal(0, strength)))
    return np.clip(x, 0, None)

"""
Minimize 2n*max(a*a)/S^2 using LP-based alternating direction finding with
constraint generation, adaptive blending, local perturbations, and symmetrization.
The core idea: given current sequence a, solve an LP to find b minimizing
max_k (a*b)_k subject to b>=0, sum(b)=1. Blend and iterate.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _normalize(x):
    """Clip to non-negative and normalize to sum 1."""
    x = np.clip(x, 0.0, None)
    s = x.sum()
    if s < 1e-15:
        return np.ones_like(x) / len(x)
    return x / s


def _score(a):
    """Compute 2n*max(a*a)/S^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    conv = np.convolve(a, a, mode='full')
    return 2.0 * n * conv.max() / (S * S)


def _build_constraint_row(a, k):
    """Build one row of the constraint matrix for lag k."""
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if hi < lo:
        return np.zeros(n)
    indices = np.arange(lo, hi + 1)
    row = np.zeros(n)
    row[indices] = a[k - indices]
    return row


def _lp_response(a, max_constraints=120, max_iters=25, time_limit=60.0):
    """
    Find b minimizing max_k (a*b)_k s.t. b>=0, sum(b)=1.
    Uses constraint generation. Returns (b, M) or None on failure.
    """
    n = a.size
    # Initial constraint set: center, edges, and a spread of lags
    center = n - 1
    initial_lags = list(set([
        center, center - 1, center + 1, 0, 2 * n - 2,
        n // 2, n // 4, 3 * n // 4, n - 2, 2, n - 3
    ]))

    t_start = time.monotonic()

    for outer in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            return None

        K = len(initial_lags)
        # Build constraint matrix vectorized
        A = np.array([_build_constraint_row(a, k) for k in initial_lags])

        # Variables: b_0..b_{n-1}, M
        c = np.zeros(n + 1)
        c[n] = 1.0

        A_ub = np.hstack([A, -np.ones((K, 1))])
        b_ub = np.zeros(K)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * (n + 1)

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            return None

        if not res.success:
            break

        b = np.clip(res.x[:n], 0, None)
        M = max(0.0, float(res.x[n]))

        # Check all lags for violations
        conv = np.convolve(a, b, mode='full')
        violated = np.where(conv > M + 1e-12)[0]

        if len(violated) == 0:
            b = _normalize(b)
            return b, M

        # Add violated constraints
        for v in violated[:80]:
            if v not in initial_lags:
                initial_lags.append(v)

        # Cap total constraints
        if len(initial_lags) > max_constraints:
            conv_vals = np.convolve(a, b, mode='full')
            top = np.argsort(conv_vals)[::-1][:max_constraints]
            initial_lags = sorted(set(int(x) for x in top))

    # Fallback: return best b found (or uniform)
    if 'b' in dir() and b is not None:
        return _normalize(b), M
    return None


def _block_perturb(a, rng, strength=0.04):
    """Apply multiplicative block perturbation."""
    n = len(a)
    x = a.copy()
    n_blocks = int(rng.integers(1, max(2, n // 600) + 1))
    for _ in range(n_blocks):
        width = int(rng.integers(max(1, n // 120), max(2, n // 6)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        x[start:stop] *= float(np.exp(rng.normal(0.0, strength)))
    return np.clip(x, 0.0, None)


def _gaussian_perturb(a, rng, strength=0.02):
    """Apply additive Gaussian perturbation."""
    n = len(a)
    x = a + rng.normal(0, strength, n) * a.max()
    return np.clip(x, 0.0, None)


def _try_symmetrize(a):
    """Return symmetrized version."""
    return 0.5 * (a + a[::-1])
