    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    i = np.where(cond)[0][-1]
    theta = cssv[i] / (i + 1)
    return np.maximum(v - theta, 0)

    h = f*f (linear convolution).
    Returns grad of the same shape as f.

def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    i = np.where(cond)[0][-1]
    theta = cssv[i] / (i + 1)
    return np.maximum(v - theta, 0.0)
