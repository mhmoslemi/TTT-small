def compute_gradient(a, w):
    """Compute grad[i] = 2 * sum_j a[j] * w[i+j] for i=0,...,n-1."""
    n = len(a)
    # We need: result[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # This is the cross-correlation of a and w (with w being longer).
    # 
    # Using FFT: 
    # Let A = FFT(a) padded to length L >= 2n-1
    # Let W = FFT(w) padded to length L >= 2n-1

    # is related to the convolution of a_reversed and w.
    #

    # Let p = n-1-j: = sum_p a[p] * w[k-n+1+p]

"""
Gradient-based optimization of non-negative sequences to minimize
2n * max(autocorrelation) / (sum)^2.
Uses FFT for fast autocorrelation/gradient, Nesterov momentum,
adaptive softmax temperature, and multiple restarts with diverse initializations.
"""

import numpy as np
import time
import math


def fft_autocorrelation(a, fft_size):
    """Compute full linear autocorrelation using FFT."""
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * np.conj(fa), fft_size)


def fft_gradient(a, w, fft_size):
    """
    Compute grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j] for i=0..n-1.
    Uses FFT cross-correlation trick.
    """
    n = len(a)
    a_rev = a[::-1]
    fa = np.fft.rfft(a_rev, fft_size)
    fw = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(fa * fw, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def compute_score_fast(a):
    """Compute 2n * max(C) / (sum a)^2 using FFT."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2
    C = fft_autocorrelation(a, fft_size)[:2 * n - 1]
    max_C = C.max()
    return 2.0 * n * max_C / (s * s)


def optimize_with_momentum(a_init, n_iter, deadline, rng, alpha_start=5.0, alpha_end=80.0,
                           lr_start=0.05, lr_end=0.001, beta=0.9, verbose=False, tag=""):
    """
    Optimize a sequence to minimize peak autocorrelation using
    Nesterov momentum with adaptive softmax temperature.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2

    best = a.copy()
    best_maxC = np.inf
    v = np.zeros(n)  # momentum

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        # Parameter schedule
        t = it / max(1, n_iter - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * t
        lr = lr_start * (lr_end / lr_start) ** t

        # Compute autocorrelation
        C = fft_autocorrelation(a, fft_size)[:2 * n - 1]
        max_C = C.max()

        if max_C < best_maxC:
            best_maxC = max_C
            best = a.copy()

        # Softmax weights (temperature-controlled)
        scaled = alpha * C
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= w_sum

        # Compute gradient via FFT
        grad = fft_gradient(a, w, fft_size)

        # Nesterov momentum
        v = beta * v + (1 - beta) * grad
        v_norm = np.linalg.norm(v)
        if v_norm > 1.0:
            v = v / v_norm

        # Step
        a_new = a - lr * v

        # Project to simplex (non-negative, sum=1)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new /= s_new
        else:
            a_new = a.copy()

        a = a_new

        # Occasional random perturbation to escape local minima
        if it > 0 and it % 5000 == 0:
            n_pert = max(1, n // 50)
            idx = rng.integers(0, n, size=n_pert)
            a[idx] = np.maximum(a[idx] * np.exp(rng.normal(0, 0.3, size=n_pert)), 0.0)
            s_p = a.sum()
            if s_p > 1e-15:
                a /= s_p

        if verbose and it % 2000 == 0:
            score_est = 2.0 * n * max_C  # since sum=1
            print(f"  {tag} iter {it}: maxC={max_C:.8f}, est_score={score_est:.6f}, alpha={alpha:.1f}")

    return best


def try_lp_refinement(a, n_lp_calls=5, deadline=None):
    """
    Try LP-based refinement: maximize sum(g) s.t. g>=0, (a*g)[k] <= max(a*a) for all k.
    Then blend: a_new = (1-t)*a + t*g_normalized.
    """
    from scipy.optimize import linprog
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2
    C = fft_autocorrelation(a, fft_size)[:2 * n - 1]
    M = C.max()

    best = a.copy()
    best_score = 2.0 * n * M  # sum=1

    for lp_it in range(n_lp_calls):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Build LP: maximize sum(g) s.t. (a*g)[k] <= M for k=0..2n-2, g >= 0
        # (a*g)[k] = sum_j a[j] * g[k-j] (valid j, k-j)
        # This is a matrix A times g <= M, where A[k,j] = a[j] if 0<=k-j<=n-1 else 0
        # A has shape (2n-1, n)

        # For efficiency, build A as a sparse-ish structure
        # Actually for n~6000, A is 12000x6000 which is ~72M entries. Too large for dense LP.
        # Let's skip LP for large n and only use it for small n.
        if n > 3000:
            break

        # Build constraint matrix A (2n-1 x n)
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            for k in range(j, j + n):
                A[k, j] = a[j]

        c = -np.ones(n)  # minimize -sum(g)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})

        if res.status != 0 or res.x is None:
            break

        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum  # normalize to sum=1

        # Try several blend parameters
        for t in [0.1, 0.2, 0.3, 0.5]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try /= s_try
            fft_size2 = 1
            while fft_size2 < 2 * n - 1:
                fft_size2 *= 2
            C_try = fft_autocorrelation(a_try, fft_size2)[:2 * n - 1]
            score_try = 2.0 * n * C_try.max()
            if score_try < best_score:
                best_score = score_try
                best = a_try.copy()
                a = a_try.copy()
                # Recompute M for next LP
                M = C_try.max()
                break

    return best

    # Allocate time per trial (more for earlier/better ones)
    total_trials = len(trials)
    per_trial_time = (budget_s - 20.0) / total_trials

    for trial_idx, (n, init_type) in enumerate(trials):
        if time.monotonic() > deadline:
            break

        trial_deadline = min(deadline, time.monotonic() + per_trial_time * 1.2)

        elif init_type == 'random':

            a_init[:width // 2] = 0.5
            a_init[-width // 2:] = 0.5
            a_init += rng.random(n) * 0.1
        else:
            a_init = np.ones(n) + rng.random(n) * 0.5

        a_init = np.maximum(a_init, 0.0)
        a_init /= a_init.sum()

        tag = f"trial {trial_idx} (n={n}, {init_type})"

                    best_score = ref_score
                    best_sequence = refined.copy()
                    print(f"  *** LP refined best: {best_score:.6f} ***")

                best_sequence = result.copy()
                print(f"  Polish {restart}: new best = {best_score:.6f}")

    print(f"\nFinal best score: {best_score:.6f}")
    return [float(x) for x in best_sequence]
