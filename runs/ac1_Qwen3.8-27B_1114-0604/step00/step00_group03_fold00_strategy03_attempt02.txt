def apply_defensive_clipping(seq, n):
    """Clamp values and ensure the sum is at least 0.01"""

def shift_effect(seq, shift_amount):
    """General framework for shifting the sequence"""
    # Copy the sequence
    new_seq = copy.deepcopy(seq)
    # Normalize or modify to remove invalid sums
    # Perhaps add zeros or remove parts? But we need to preserve length
    # Do a shift by shift_amount and then clip
    # Since we are adding a fixed offset, we need to clamp to [0,1000] and positive
    shifted_seq = np.roll(new_seq, int(round(shift_amount)))
    # Remove negatives by clipping
    shifted_seq = np.maximum(shifted_seq, 0.0)
    shifted_seq = np.minimum(shifted_seq, 1000.0)
    return shifted_seq.tolist()

    """Generate a block sequence from the parent expression"""
