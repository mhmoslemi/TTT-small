```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def build_A_for_active(a, K, n):
    """Build sparse matrix for active constraints (a_prev * a)_k for k in K."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(K):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(K), n))

def lp_iterate(a, n, active_K, M_bound):
    """Solve LP: min M s.t. (a*a_new)_k <= M for k in active_K, sum(a_new)=1, a_new>=0."""
    c = build_A_for_active(a, active_K, n)
    n_con = len(active_K)
    
    # Variables: [a_0, ..., a_{n-1}, M]
    # Objective: minimize M → c_obj = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: (a*a_new)_k - M <= 0 for k in active_K
    # A_ub * x <= b_ub
    # A_ub has n_con rows, n+1 columns
    # First n columns: c (the cross-convolution matrix)
    # Last column: -1 (for the -M term)
    A_ub = np.hstack([c.toarray(), -np.ones((n_con, 1))])
    b_ub = np.zeros(n_con)
    
    # Equality constraint: sum(a) = 1
    A_eq = np.hstack([np.ones(n), [[0]]])
    b_eq = np.array([1.0])
    
    # Bounds: a_j >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, None
    
    return res.x[:n], res.x[-1]
```

```python
"""
Minimizes 2n*max(autoconv)/(sum)^2 using cutting-plane LP with multiple
initializations and sequence lengths, followed by local search refinement.

Key idea: At each iteration, solve an LP that linearizes the auto-convolution
constraint at the current point. The LP finds the best direction to reduce
the peak cross-convolution. Violated auto-convolution constraints are added
iteratively (cutting plane). Multiple starting points and lengths are explored.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, hstack
import time


def _build_cross_conv_matrix(a, K, n):
    """Build sparse matrix C where C[i,j] = a[K[i]-j] if 0<=K[i]-j<n."""
    rows = []
    cols = []
    vals = []
    nK = len(K)
    for idx in range(nK):
        k = K[idx]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((a[k - js]).tolist())
    if not rows:
        return csr_matrix((nK, n))
    return csr_matrix((vals, (rows, cols)), shape=(nK, n))

    if len(active_K) == 0:
        return None, None
    nK = len(active_K)
    C = _build_cross_conv_matrix(a, active_K, n)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = hstack([C, csr_matrix(-np.ones((nK, 1)))], format='csc')
    b_ub = np.zeros(nK)

    A_eq = hstack([csr_matrix(np.ones((1, n))), csr_matrix(np.zeros((1, 1)))], format='csc')
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None, None

    if not res.success or res.x is None:
        return None, None

    return res.x[:n], res.x[-1]


def _lp_optimize(a_init, n, deadline, max_outer=40):
    """Cutting-plane LP optimization for fixed n."""
    a = np.clip(a_init.copy(), 0, None)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_M = np.inf

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = np.convolve(a, a, mode='full')
        M = c.max()

        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()

        # Active set: top positions
        start_thresh = 0.3
        active_K = list(np.where(c >= start_thresh * M)[0])
        # Always include the peaks
        for k in np.where(c >= 0.99 * M)[0]:
            if k not in active_K:
                active_K.append(int(k))

        if len(active_K) == 0:
            active_K = [int(np.argmax(c))]

        progress = False
        b_last = None

        for inner in range(30):
            if time.monotonic() > deadline:
                break

            b_new, M_lp = _solve_lp(a, n, active_K)
            if b_new is None:
                break

            b_new = np.clip(b_new, 0, None)
            bs = b_new.sum()
            if bs < 1e-10:
                break
            b_new = b_new / bs
            b_last = b_new

            c_new = np.convolve(b_new, b_new, mode='full')
            M_new = c_new.max()

            if M_new <= M_lp + 1e-9 * max(1, M):
                # LP is tight - optimal!
                if M_new < best_M:
                    best_M = M_new
                    best_a = b_new.copy()
                a = b_new
                progress = True
                break

            # Add violated constraints
            violated = np.where(c_new > M_lp + 1e-9 * max(1, M))[0]
            for k in violated:
                if k not in active_K:
                    active_K.append(int(k))

            # If active set too large, accept b_new and restart outer loop
            if len(active_K) > min(800, 2 * n - 1):
                if M_new < best_M:
                    best_M = M_new
                    best_a = b_new.copy()
                a = b_new
                progress = True
                break

        if not progress:
            # Try fraction steps towards b_last
            if b_last is not None:
                accepted = False
                for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
                    a_try = (1.0 - t) * a + t * b_last
                    a_try = np.clip(a_try, 0, None)
                    at = a_try.sum()
                    if at < 1e-10:
                        continue
                    a_try = a_try / at
                    c_try = np.convolve(a_try, a_try, mode='full')
                    if c_try.max() < M - 1e-10:
                        a = a_try
                        progress = True
                        accepted = True
                        break
                if not accepted:
                    break

    return best_a


def _local_search(a, deadline, max_iters=5000):
    """Local search with adaptive step sizes."""
    n = a.size
    rng = np.random.default_rng(12345)
    a = a.copy()
    cur_M = np.convolve(a, a, mode='full').max()
    best_M = cur_M
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Strategy 1: Block perturbation
        if rng.random() < 0.6:
            width = int(rng.integers(max(2, n // 200), max(5, n // 8)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            end = min(n, start + width)
            a_try = a.copy()
            scale = np.exp(rng.normal(0, 0.12, size=end - start))
            a_try[start:end] *= scale
        else:
            # Strategy 2: Targeted peak reduction
            c = np.convolve(a, a, mode='full')
            k_star = int(np.argmax(c))
            # Find j that contributes most to peak
            j_start = max(0, k_star - n + 1)
            j_end = min(n - 1, k_star)
            contribs = a[j_start:j_end + 1] * a[k_star - np.arange(j_start, j_end + 1)]
            top_j = j_start + int(np.argmax(contribs))
            a_try = a.copy()
            a_try[top_j] *= 0.9
            # Boost a low-contribution position
            low_j = j_start + int(np.argmin(contribs))
            a_try[low_j] *= 1.1

        a_try = np.clip(a_try, 0, None)
        if a_try.sum() < 1e-10:
            no_improve += 1
            continue

        c_try = np.convolve(a_try, a_try, mode='full')
        M_try = c_try.max()

        if M_try < best_M - 1e-10:
            a = a_try
            best_M = M_try
            best_a = a_try.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            # Random restart within neighborhood
            a_try = a * np.exp(rng.normal(0, 0.05, n))
            a_try = np.clip(a_try, 0, None)
            if a_try.sum() > 1e-10:
                a = a_try
                no_improve = 0

    return best_a, best_M


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: cutting-plane LP + local search.
    Tries multiple lengths and initializations, keeps the best.
    """
    deadline = time.monotonic() + budget_s - 10.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = np.ones(154) / 154.0

    # Phase 1: LP optimization with various configs
    configs = []
    # Small n: fast LPs, many iterations
    for n in [32, 48, 64, 100, 150, 200]:
        for init in ['cosine', 'hann', 'gaussian_wide', 'gaussian_narrow', 'ramp']:
            configs.append((n, init))
    # Medium n
    for n in [400, 800, 1500]:
        for init in ['cosine', 'hann', 'gaussian_wide']:
            configs.append((n, init))
    # Large n (from provided sequence)
    try:
        h1 = np.asarray(height_sequence_1, dtype=np.float64)
        if h1.ndim == 1 and h1.size > 0 and np.all(np.isfinite(h1)):
            configs.append((h1.size, 'provided'))
    except Exception:
        pass

    for n, init_name in configs:
        if time.monotonic() > deadline - 60:
            break

        if init_name == 'cosine':
            a_init = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / (n + 1))
        elif init_name == 'hann':
            a_init = np.hanning(n)
        elif init_name == 'gaussian_wide':
            a_init = np.exp(-((np.arange(n) - n / 2) ** 2) / (2 * (n / 3) ** 2))
        elif init_name == 'gaussian_narrow':
            a_init = np.exp(-((np.arange(n) - n / 2) ** 2) / (2 * (n / 6) ** 2))
        elif init_name == 'ramp':
            t = np.linspace(0, 1, n)
            a_init = t * (1 - t) * 4  # parabola
        elif init_name == 'provided':
            a_init = np.clip(h1, 0, None)
        else:
            continue

        a_init = np.clip(a_init, 0, None)
        if a_init.sum() < 1e-10:
            continue
        a_init = a_init / a_init.sum()

        # Determine max outer iterations based on n
        max_outer = 60 if n <= 200 else (40 if n <= 1000 else 20)

        t0 = time.monotonic()
        a_opt = _lp_optimize(a_init, n, deadline, max_outer=max_outer)
        elapsed = time.monotonic() - t0

        # Evaluate
        a_eval = a_opt / a_opt.sum() if a_opt.sum() > 0 else a_opt
        score = evaluate_sequence(a_eval.tolist())

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_eval.copy()
            print(f"  n={n}, init={init_name:15s}, score={score:.6f}, "
                  f"LP_time={elapsed:.1f}s, M={np.convolve(a_eval,a_eval).max()*a_eval.sum()**2:.6f}")

    print(f"After LP phase: best_score={best_score:.6f}")

    # Phase 2: Local search refinement
    if time.monotonic() < deadline - 5:
        a_best, M_ls = _local_search(best_seq, deadline)
        a_eval = a_best / a_best.sum() if a_best.sum() > 0 else a_best
        score = evaluate_sequence(a_eval.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_eval.copy()
            print(f"After local search: score={score:.6f}")

    # Phase 3: Try a few more LP runs from perturbed best
    if time.monotonic() < deadline - 30 and best_score < 2.0:
        n_best = best_seq.size
        for trial in range(5):
            if time.monotonic() > deadline - 10:
                break
            a_pert = best_seq * np.exp(rng.normal(0, 0.1, n_best))
            a_pert = np.clip(a_pert, 0, None)
            if a_pert.sum() < 1e-10:
                continue
            a_pert = a_pert / a_pert.sum()
            a_opt = _lp_optimize(a_pert, n_best, deadline, max_outer=30)
            a_eval = a_opt / a_opt.sum() if a_opt.sum() > 0 else a_opt
            score = evaluate_sequence(a_eval.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_eval.copy()
                print(f"  Trial {trial}: score={score:.6f}")

    print(f"FINAL best_score={best_score:.6f}, n={best_seq.size}")
    return [float(x) for x in best_seq]
```
