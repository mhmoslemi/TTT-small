import numpy as np
import time

def next_power_of_2(n):
    return 1 << (n - 1).bit_length()

def compute_convolution(a, fft_size):
    """Compute linear convolution a*a using FFT."""
    a_pad = np.zeros(fft_size)
    a_pad[:len(a)] = a
    A = np.fft.rfft(a_pad)
    c = np.fft.irfft(A * A)
    return c[:2*len(a)-1]

def compute_gradient(a, c, beta, fft_size):
    """Compute gradient of soft-max objective."""
    n = len(a)
    N = 2*n - 1
    # Soft-max weights
    c_shift = c - c.max()
    p = np.exp(beta * c_shift)
    p_sum = p.sum()
    if p_sum == 0:
        p = np.ones(N) / N
    else:
        p /= p_sum
    
    # q = 2*p, zero-padded
    q = np.zeros(fft_size)
    q[:N] = 2 * p
    
    # grad[i] = sum_k q[k] * a[k-i] (cross-correlation)
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    QA = np.fft.rfft(q) * np.conj(np.fft.rfft(a_pad))
