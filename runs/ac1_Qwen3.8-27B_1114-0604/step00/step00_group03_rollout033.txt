import time

def autocorrelation(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_projection(v, target_sum=1.0):
    """Project v onto {x >= 0 : sum(x) = target_sum}."""
    n = len(v)
    # Scale to make the projection standard
    v = v.copy()
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    proj = np.maximum(v - theta, 0)
    return proj

def autocorrelation_fft(a):
    """Compute full autocorrelation of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_projection(v, s=1.0):
    """Project v onto {x >= 0 : sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - s))[0]
    if len(rho) == 0:
        return np.full(n, s/n)
    rho = rho[-1]
    theta = (cv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
