    """Project v onto the simplex {x >= 0, sum(x) = S}."""
    n = len(v)

    u = np.sort(v)[::-1]

    cond = u - (cssv - S) / np.arange(1, n+1)
    rho = np.sum(cond > 0) - 1
    tau = (cssv[rho] - S) / (rho + 1)
    return np.maximum(v - tau, 0)

    conv_ff = np.convolve(f, f)
    M = conv_ff.max()
    
    # Build constraint matrix: A[k, j] = f[k-j] if 0 <= k-j < n
    # This is a banded matrix
    from scipy.sparse import lil_matrix
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k + 1, n)
        if j_start < j_end:
            A[k, j_start:j_end] = f[k - np.arange(j_start, j_end)]

    if res.success:
        g = res.x
        # Rescale
        g = g / g.sum()  # sum = 1
        return g, M
    return None, M

"""
LP-based alternating direction method for minimizing the peak of self-convolution.
Uses constraint generation to solve medium-sized LPs efficiently.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

def project_onto_simplex(v, S=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cv = np.arange(1, n+1)
    cond = u - (cssv - S) / cv
    rho = np.sum(cond > 0) - 1
    if rho < 0:
        return np.full(n, S / n)
    tau = (cssv[rho] - S) / (rho + 1)
    return np.maximum(v - tau, 0)

def solve_lp_with_cg(f, n, M, max_constraints=200, max_cg_iters=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses constraint generation."""
    # Initialize active set with the most "critical" constraints
    conv_ff = np.convolve(f, f)
    # Start with constraints at the top of the convolution
    n_init = min(100, 2*n-1)
    active = np.argsort(conv_ff)[-n_init:]
    
    g = None
    for _ in range(max_cg_iters):
        # Build constraint matrix for active constraints
        A = lil_matrix((len(active), n), dtype=np.float64)
        for idx, k in enumerate(active):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k + 1, n)
            if j_start < j_end:
                cols = np.arange(j_start, j_end)
                A[idx, cols] = f[k - cols]
        A_csr = A.tocsr()
        
        b = np.full(len(active), M)
        c = -np.ones(n)
        
        res = linprog(c, A_ub=A_csr, b_ub=b, bounds=[(0, None)]*n, method='highs')
        
        if not res.success:
            # Try with relaxed constraints
            b = np.full(len(active), M * 1.01)
            res = linprog(c, A_ub=A_csr, b_ub=b, bounds=[(0, None)]*n, method='highs')
            if not res.success:
                return None, active
        
        g = res.x
        if g is None or np.any(g < -1e-10):
            g = np.clip(g, 0, None)
        
        # Check for violated constraints
        conv_fg = np.convolve(f, g)
        violated = np.where(conv_fg > M + 1e-6)[0]
        
        if len(violated) == 0:
            return g, active
        
        # Add violated constraints, keep top max_constraints
        new_active = np.unique(np.concatenate([active, violated]))
        if len(new_active) > max_constraints:
            # Rank by violation amount
            viol_amount = conv_fg[new_active] - M
            order = np.argsort(viol_amount)[::-1]
            new_active = new_active[order[:max_constraints]]
        active = new_active
    
    return g, active

def compute_score(f):
    """Compute the score without calling evaluate_sequence (faster)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_b = conv.max()
    sum_a = f.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a ** 2))
