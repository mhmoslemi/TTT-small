import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    S = 1
    while S < 2*n - 1:
        S *= 2
    A = rfft(a, S)
    return irfft(A * A, S)[:2*n-1]

def compute_grad(a, w):
    n = len(a)
    S = 1
    while S < 3*n - 2:
        S *= 2
    W = rfft(w, S)
    A = rfft(a, S)
    corr = irfft(W.conj() * A, S)
    return 2.0 * corr[:n]

def softmax(x, T):
    x_max = np.max(x)
    exp_x = np.exp((x - x_max) / T)
    return exp_x / np.sum(exp_x)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    # Simple approach: clip and rescale
    v = np.maximum(0, v)
    s = np.sum(v)
    if s == 0:
        return np.ones(n) / n
    return v / s

def optimize_sequence(a, max_time, T_start=1.0, T_end=50.0, lr_start=0.1, seed=42):
    """Optimize a sequence to minimize max(conv(a,a)) subject to sum(a)=1, a>=0."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(0, a)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = 2*n*np.max(compute_conv(a))
    
    start_time = time.monotonic()
    max_time = max_time - 5  # leave 5 sec buffer
    
    num_iters = 0
    lr = lr_start
    
    while time.monotonic() - start_time < max_time:
        it_start = time.monotonic()
        iters_per_sec = num_iters / max(1e-6, time.monotonic() - start_time) if num_iters > 0 else 100
        remaining_frac = max(0, (max_time - (time.monotonic() - start_time)) / max_time)
        
        # Anneal temperature
        progress = 1 - remaining_frac  # 0 to 1
        T = T_start * (T_end / T_start) ** progress  # geometric annealing
        
        # Compute convolution
        c = compute_conv(a)
        max_c = np.max(c)
        
        # Softmax weights
        w = softmax(c, T)
        
        # Compute gradient
        grad = compute_grad(a, w)
        
        # Gradient step
        a_new = a - lr * grad
        
        # Project
        a_new = np.maximum(0, a_new)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        a = a_new
        num_iters += 1
        
        # Evaluate
        c = compute_conv(a)
        obj = 2 * n * np.max(c)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if num_iters % 100 == 0:
            if obj > best_obj * 1.01:
                lr *= 0.9
            else:
                lr = min(lr * 1.05, lr_start)
        
        # Safety: ensure we don't spend too long per iteration
        if iters_per_sec < 10:

    return best_a, best_obj

```python
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def lp_solve(f, M, n):
    """
    Solve: maximize sum(g) subject to conv(f,g) <= M (element-wise), g >= 0.
    f is the current sequence (length n), M is the peak bound.
    """
    # Build constraint matrix A (2n-1 x n) where (A@g)[k] = conv(f,g)[k]
    # A[k,j] = f[k-j] if 0<=k-j<n, else 0
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n-1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
```
