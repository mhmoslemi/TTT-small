import time
from scipy.optimize import linprog

def solve_lp(a, M):
    """Solve max sum(b) s.t. (a*b)_k <= M for all k, b >= 0."""
    n = len(a)
    # c_k = sum_i a_i * b_{k-i} for valid i, k-i
    # This is a linear function of b.
    # c_k = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a_{k-j} * b_j
    
    # Constraint matrix A: A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(k+1, n)):
            A[k, j] = a[k-j]
    
    c = -np.ones(n)  # minimize -sum(b)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    return res

"""
Minimizes F(a) = 2n*max(convolve(a,a))/(sum(a))^2 using LP-based
Frank-Wolfe (conditional gradient) iteration with diverse restarts
across multiple sequence lengths.

Key insight: The uniform sequence gives F=2 (a fixed point of the LP).
By starting from asymmetric sequences and using the LP to find optimal
descent directions, we can escape this fixed point and achieve F < 2.
"""

import numpy as np
import time
from scipy.optimize import linprog


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ b = convolve(a, b)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def normalize_seq(a, n):
    """Clip to [0,1000] and normalize to sum 1."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def run_lp_iteration(a_init, n, time_limit, rng, best_state):
    """
    Run LP-based Frank-Wolfe iteration.
    a_init: starting point (sum=1, non-negative)
    n: length
    time_limit: absolute time to stop
    rng: random number generator
    best_state: dict with keys 'a', 'score' (updated in place)
    """
    a = normalize_seq(a_init.copy(), n)
    local_best = a.copy()
    local_best_score = evaluate_sequence(a.tolist())

    if local_best_score < best_state['score']:
        best_state['score'] = local_best_score
        best_state['a'] = a.copy()

    step_size = 0.5
    stagnant_count = 0
    it = 0

    while time.time() < time_limit:
        it += 1
        conv = np.convolve(a, a)
        M = conv.max()

        A = build_conv_matrix(a)
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')

        if res.status != 0:
            break

        b = res.x
        sum_b = b.sum()

        if sum_b <= 1.0 + 1e-8:
            stagnant_count += 1
            if stagnant_count > 50:
                break
            # Perturb to escape fixed point
            a = normalize_seq(a + 0.05 * (rng.random(n) - 0.5), n)
            step_size = 0.5
            continue

        stagnant_count = 0

        # Adaptive step size (Frank-Wolfe style)
        if it <= 100:
            step = step_size
        else:
            step = min(step_size, 2.0 / (it + 2))

        a_new = (1.0 - step) * a + step * b / sum_b
        a_new = normalize_seq(a_new, n)

        score = evaluate_sequence(a_new.tolist())
        if score < local_best_score:
            local_best_score = score
            local_best = a_new.copy()
            if score < best_state['score']:
                best_state['score'] = score
                best_state['a'] = a_new.copy()
                if it % 100 == 0 or score < best_state.get('print_threshold', 1.9):
                    print(f"    [n={n}, it={it}] score={score:.6f}")
        else:
            step_size = max(0.05, step_size * 0.85)

        a = a_new

    # Polish: coordinate descent
    while time.time() < time_limit and it < 200000:
        it += 1
        improved = False
        for i in range(0, n, max(1, n // 200)):
            if time.time() > time_limit:
                break
            for delta in [0.01, -0.01, 0.05, -0.05]:
                trial = a.copy()
                trial[i] = max(0.0, trial[i] + delta * a.sum() / n)
                trial = normalize_seq(trial, n)
                s = evaluate_sequence(trial.tolist())
                if s < local_best_score - 1e-10:
                    local_best_score = s
                    local_best = trial.copy()
                    a = trial.copy()
                    improved = True
                    break
        if not improved:
            break

    if local_best_score < best_state['score']:
        best_state['score'] = local_best_score
        best_state['a'] = local_best.copy()

    return local_best, local_best_score


def generate_starts(n, rng):

    starts = []
