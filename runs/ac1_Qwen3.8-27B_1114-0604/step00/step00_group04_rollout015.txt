def build_convolution_matrix(f):
    """Build the matrix A such that (A @ g)[k] = (f*g)[k] for full convolution."""
    n = len(f)
    # A is (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n else 0

    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

Define h(f) = max_k (f*f)_k. We want to minimize h(f) subject to sum(f) = S, f >= 0.

The proximal operator of the constraint set {f: sum(f) = S, f >= 0} is the Euclidean projection onto the simplex (well, the intersection of the positive orthant and the hyperplane sum = S).

The forward step (gradient of h) is: at f, compute the subgradient, take a step.
The backward step (proximal) is: project back onto the constraint set.

The subgradient of h(f) = max_k (f*f)_k at f is:
∇_f h(f) = (1/|K|) sum_{k in K} 2 * flip_k(f)
where K = {k : (f*f)_k = max_k (f*f)_k} and flip_k(f)[j] = f[k-j] if 0 <= k-j < n, else 0.

3. Compute subgrad: s[j] = (2/|K|) * sum_{k in K} f[max(0, min(n-1, k-j))] ... (with appropriate bounds)
4. f_temp = f - α * s
5. f_new = project(f_temp) (project onto {f: sum(f) = S, f >= 0})

The projection onto {f: sum(f) = S, f >= 0} is:
- Clip f_temp to [0, ∞)
- Scale so sum = S
- But this doesn't work because clipping changes the sum!

The correct projection onto the "probability simplex" (sum = 1, f >= 0) is:
f_new = max(f_temp - τ, 0) / sum(max(f_temp - τ, 0))
where τ is chosen so that sum(f_new) = 1.

For our case (sum = S, f >= 0):
f_new = max(f_temp - τ, 0) / sum(max(f_temp - τ, 0)) * S

Finding τ requires a 1D search (bisection).

This is a viable approach! And it's much simpler than the LP. Let me implement it.

The key question is: what step size α to use? We can use a backtracking line search or a diminishing step size.

```python
def _subgradient_step(f, alpha):
    """One subgradient step for minimizing max_k (f*f)_k on the simplex."""
    n = len(f)
    S = np.sum(f)
    conv = np.convolve(f, f)
    max_val = np.max(conv)
    # Find all maximizing lags
    K = np.where(conv >= max_val * (1 - 1e-10))[0]
    
    # Compute subgradient: s[j] = (2/|K|) * sum_{k in K} f[k-j] (with bounds)
    s = np.zeros(n)
    for k in K:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                s[j] += f[idx]
    s *= 2.0 / len(K)
    
    # Forward step
    f_temp = f - alpha * s
    
    # Backward step: project onto {f: sum(f) = S, f >= 0}
    # This is the Euclidean projection onto the scaled simplex
    f_temp = np.maximum(f_temp, 0)
    S_new = np.sum(f_temp)
    if S_new < 1e-10:
        return f  # fallback
    f_new = f_temp * S / S_new
    
    return f_new
```

Wait, the projection is not just clipping and rescaling. The correct Euclidean projection onto {f: f >= 0, sum(f) = S} is:
min ||x - f_temp||^2 subject to x >= 0, sum(x) = S

The solution is: x = max(f_temp - τ, 0) where τ is chosen so sum(max(f_temp - τ, 0)) = S.

This requires finding τ, which can be done by sorting f_temp and trying different thresholds.

Let me implement this properly:

```python
def _project_simplex(v, S):
    """Project v onto {x: x >= 0, sum(x) = S} in Euclidean distance."""
    n = len(v)
    # The projection is max(v - tau, 0) where sum(max(v-tau,0)) = S
    # Sort v in descending order
    u = np.sort(v)[::-1]
    # Find the right threshold
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1] + 1
    theta = (cssv[rho-1] - S) / rho
    return np.maximum(v - theta, 0)
```

This is the standard algorithm for projecting onto the simplex, adapted for the case sum = S instead of sum = 1.

```python
"""
Hybrid optimization: LP-based iterations with subgradient polish
to minimize 2*n*max(conv(f,f))/(sum(f))^2 for non-negative sequences.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _precompute_conv_indices(n):
    """Precompute indices for the convolution matrix."""
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    idx = 0
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        count = j_hi - j_lo + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = j_lo + np.arange(count, dtype=np.int32)
        idx += count
    return rows, cols


def _solve_lp(f, M, rows, cols):
    """Solve: max 1^T g s.t. A_f @ g <= M, g >= 0."""
    n = len(f)
    vals = f[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=(0.0, None), method='highs',
                  options={'time_limit': 30})
    if res.success and res.x is not None:
        return res.x
    return None


def _project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        # All projections are zero, distribute evenly
        return np.full(n, S / n)
    rho = rho_idx[-1] + 1
    theta = (cssv[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)
