from scipy.optimize import linprog

"""
Minimizes 2*n*max(convolve(x,x))/(sum(x))^2 for non-negative x using:
1) Multiple structured initializations
2) Subgradient descent with momentum targeting top peaks
3) Multiplicative weight updates (mirror descent)
4) Periodic annealing and multi-start

def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    sv = mu.cumsum()
    rho = np.nonzero(mu * np.arange(1, n + 1) > (sv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (sv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_conv_peak_and_subgrad(x, top_m=10):
    """Compute convolution, peak, and weighted subgradient of max_k (x*x)[k]."""
    n = len(x)
    b = np.convolve(x, x, mode='full')
    peak = b.max()
    
    # Find top M peak positions
    num_peaks = min(top_m, len(b))
    top_indices = np.argsort(b)[::-1][:num_peaks]
    
    # Weighted subgradient: weight by excess over median
    median_b = np.median(b)
    weights = np.maximum(b[top_indices] - median_b, 0.0)
    if weights.sum() < 1e-15:
        weights = np.ones(num_peaks)
    weights = weights / weights.sum()
    
    g = np.zeros(n)
    for w, k in zip(weights, top_indices):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        indices = np.arange(lo, hi + 1)
        vals = x[k - indices] + x[indices]
        if k % 2 == 0:
            mid = k // 2
            if lo <= mid <= hi:
                vals[mid - lo] = 2.0 * x[mid]
        g[lo:hi+1] += w * vals
    
    return b, peak, g


def make_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    
    # Provided parent (symmetrized)
    p = parent.copy()
    p = 0.5 * (p + p[::-1])
    p = np.maximum(p, 0)
    if p.sum() > 0:
        inits.append(p / p.sum())
    
    t = np.linspace(0, 1, n)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine
    rc = (1 + np.cos(2 * np.pi * t)) / 2
    inits.append(rc / rc.sum())
    
    # Inverted raised cosine
    irc = (1 - np.cos(2 * np.pi * t)) / 2 + 0.01
    inits.append(irc / irc.sum())
    
    # Linear ramp
    ramp = np.linspace(0.1, 2.0, n)
    inits.append(ramp / ramp.sum())
    
    # Inverted ramp
    iramp = np.linspace(2.0, 0.1, n)
    inits.append(iramp / iramp.sum())
    
    # Triangle (peak center)
    tri = 1.0 - np.abs(2 * t - 1)
    inits.append(tri / tri.sum())
    
    # Inverted triangle (U-shape)
    itri = np.abs(2 * t - 1) + 0.01
    inits.append(itri / itri.sum())
    
    # Gaussian center
    gauss = np.exp(-((t - 0.5) ** 2) / (2 * 0.2 ** 2))
    inits.append(gauss / gauss.sum())
    
    # Inverted Gaussian (U-shape)
    igauss = np.exp(-((t - 0.5) ** 2) / (2 * 0.4 ** 2))
    igauss = np.maximum(1.0 - igauss, 0.05)
    inits.append(igauss / igauss.sum())
    
    # Double peak
    peak1 = np.exp(-((t - 0.25) ** 2) / (2 * 0.1 ** 2))
    peak2 = np.exp(-((t - 0.75) ** 2) / (2 * 0.1 ** 2))
    two_peaks = peak1 + peak2 + 0.05
    inits.append(two_peaks / two_peaks.sum())
    
    # Cosine^2 window
    cos2 = np.cos(np.pi * (t - 0.5)) ** 2
    inits.append(cos2 / cos2.sum())
    
    # Sinusoidal modulation
    mod = 1.0 + 0.5 * np.sin(2 * np.pi * 3 * t)
    inits.append(mod / mod.sum())
    
    # Random smooth
    rnd = np.abs(rng.normal(1.0, 0.5, size=n))
    kern = np.ones(max(3, n // 200)) / max(3, n // 200)
    rnd = np.convolve(rnd, kern, mode='same')
    inits.append(rnd / rnd.sum())
    
    # Symmetrize all
    sym_inits = []
    for x in inits:
        xs = 0.5 * (x + x[::-1])
        xs = np.maximum(xs, 0)
        if xs.sum() > 0:
            sym_inits.append(xs / xs.sum())
    
    # Also add unsymmetrized versions
    all_inits = sym_inits + inits
    return all_inits

    stagnant = 0
    
    while time.monotonic() - start_time < max_time:
        t_iter += 1
        b, peak, g = compute_conv_peak_and_subgrad(x, top_m)
        
        # Momentum update
        v = momentum * v - alpha * g
        x_new = x + v
        x_new = project_simplex(x_new)
        
        # Cheap objective check

        if t_iter % 5 == 0:

            obj_new = 2.0 * n * b_new.max()
            if obj_new < best_obj - 1e-8:
                best_obj = obj_new
                best_x = x_new.copy()
                stagnant = 0
            else:
                stagnant += 1
            
            # Expensive evaluation every ~0.5s

        if stagnant > 100:

def multiplicative_optimize(x0, max_time, rng, top_m=10, verbose=False):

    n = len(x0)

def lp_refine(f, M, max_time, verbose=False):
    """LP-based refinement: find g maximizing sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    
    # Build constraint matrix: row k has f[i] at column (k-i) mod... 
    # Actually, (f*g)[k] = sum_i f[i] * g[k-i] where 0 <= k-i < n
    # So for constraint k: sum_{i: 0<=k-i<n} f[i] * g[k-i] <= M
    # Let j = k-i, so i = k-j: sum_{j: 0<=j<n, 0<=k-j<n} f[k-j] * g[j] <= M
    # The coefficient of g[j] in constraint k is f[k-j] (if valid) else 0.
    
    # Sparse construction using scipy sparse
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A @ g <= M (array of M)
    b_ub = np.full(2 * n - 1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    # Solve LP
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum  # normalize to sum=1
                return g
    except Exception as e:
        if verbose:
            print(f"  LP failed: {e}")
    
    return None


def lp_iteration(f, max_time, verbose=False):
    """One LP iteration: find optimal g given current f."""
    b = np.convolve(f, f, mode='full')
    M = b.max()
    
    # Solve LP
    g = lp_refine(f, M, max_time, verbose)
    if g is None:
        return f
    return g


def block_perturb(x, rng, strength=0.1):
    """Apply a block perturbation to x."""
    n = len(x)
    x_new = x.copy()
    num_blocks = rng.integers(1, 5)
    for _ in range(num_blocks):
        width = rng.integers(max(2, n // 200), max(3, n // 50))
        start = rng.integers(0, max(1, n - width))
        scale = np.exp(rng.normal(0, strength))
        x_new[start:start+width] *= scale
    
    # Occasionally symmetrize
    if rng.random() < 0.3:
        x_new = 0.5 * (x_new + x_new[::-1])
    
    x_new = np.maximum(x_new, 0)
    s = x_new.sum()
    if s < 1e-10:
        return x
    return x_new / s
