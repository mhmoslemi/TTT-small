import numpy as np
from scipy.signal import correlate
import time

def project_to_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_autocorrelation(f):
    """Compute full linear autocorrelation using FFT."""
    n = len(f)
    size = 1
    while size < 2*n - 1:
        size *= 2
    F = np.fft.rfft(f, size)
    c = np.fft.irfft(F * F, size)[:2*n-1]
    return c

    """Compute gradient of soft-max autocorrelation."""
    # grad_j = 2 * sum_k p_k * f_{k-j}
    # This is 2 * correlate(p, f, 'valid')

One more consideration: the `correlate` function from scipy. Let me make sure I'm using it correctly.

`scipy.signal.correlate(in1, in2, mode='valid')` computes the cross-correlation of in1 and in2, returning only those values where the signals completely overlap. If in1 has length N and in2 has length M (N >= M), the output has length N - M + 1.

The j-th element is: `sum_{k=0}^{n-1} p[k+j] * f[k]`.

This is exactly `sum_{m=0}^{n-1} p[m+j] * f[m]` = the j-th component of the gradient (divided by 2).

So `grad = 2 * correlate(p, f, mode='valid')` is correct.

    size = 1 << (2 * n - 2).bit_length()

def _soft_gradient(c, f, T):
    """Compute soft-max gradient: 2 * correlate(softmax(c/T), f, 'valid')."""
    n = len(f)
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(2 * n - 1)
        p[n - 1] = 1.0
    else:
        p /= s
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad, p


def _mirror_step(f, grad, step):
    """One exponentiated-gradient (mirror descent) step on the simplex."""
    span = grad.max() - grad.min()
    if span < 1e-15:
        return f
    eta = step / span
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    f_new /= s
    return f_new


def _project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _polish(f, n, deadline, rng):
    """Coordinate-descent polish: try small moves in each coordinate."""
    f = f.copy()
    local_best = _eval(f)
    local_best_f = f.copy()
    c = _autocorr_fft(f)
    peak = c.max()

    n_passes = 200
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break

        # Find peak lag
        k_star = int(np.argmax(c))

        # Gradient at the peak lag: g_j = 2 * f_{k_star - j}
        # We want to decrease f_j where f_{k_star - j} is large.
        # Try moving mass from high-contribution positions to low ones.

        contrib = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                contrib[j] = f[j] * f[idx]
            else:
                contrib[j] = 0.0

        # Find the position with highest contribution (to reduce)
        j_donate = int(np.argmax(contrib))
        # Find a position with low contribution (to receive)
        contrib_sorted = np.argsort(contrib)
        j_receive = int(contrib_sorted[0])

        if j_donate == j_receive:
            continue

        # Try a small transfer
        for frac in [0.01, 0.05, 0.1, 0.2]:
            f_try = f.copy()
            amt = f_try[j_donate] * frac
            f_try[j_donate] -= amt
            f_try[j_receive] += amt
            if f_try[j_donate] < 0:
                continue
            f_try /= f_try.sum()
            c_try = _autocorr_fft(f_try)
            if c_try.max() < peak - 1e-12:
                f = f_try
                peak = c_try.max()
                c = c_try
                local_best = _eval(f)
                local_best_f = f.copy()
                break

        # Also try random coordinate swaps
        for _ in range(10):
            i = int(rng.integers(0, n))
            jx = int(rng.integers(0, n))
            if i == jx:
                continue
            f_try = f.copy()
            amt = f_try[i] * 0.05
            f_try[i] -= amt
            f_try[jx] += amt
            if f_try[i] < 0:
                continue
            f_try /= f_try.sum()
            c_try = _autocorr_fft(f_try)
            if c_try.max() < peak - 1e-12:
                f = f_try
                peak = c_try.max()
                c = c_try

    starts['u_narrow'] = 1.0 + 10.0 * ((j - center) / center) ** 2
    starts['bimodal'] = (np.exp(-(j - 0.3 * n) ** 2 / (0.08 * n * n)) +
                         np.exp(-(j - 0.7 * n) ** 2 / (0.08 * n * n)) + 0.01)
    starts['trimodal'] = (np.exp(-(j - 0.2 * n) ** 2 / (0.05 * n * n)) +
                          np.exp(-(j - 0.5 * n) ** 2 / (0.05 * n * n)) +
                          np.exp(-(j - 0.8 * n) ** 2 / (0.05 * n * n)) + 0.01)
    starts['random1'] = rng.exponential(1.0, size=n)
    starts['random2'] = rng.dirichlet(np.ones(n) * 0.05)
    starts['sine'] = 1.0 + 0.5 * np.sin(4.0 * np.pi * j / n)
    starts['sine_abs'] = 1.0 + 0.8 * np.abs(np.sin(2.0 * np.pi * j / n))

    # Evaluate all starts and sort by score
    start_results = []
    for name, f0 in starts.items():
        f = np.maximum(f0, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = np.ones(n)
        else:
            f = f / s
        sc = _eval(f)
        start_results.append((sc, name, f))

    start_results.sort(key=lambda x: x[0])
    print("Starting point scores:")
    for sc, name, _ in start_results:
        print(f"  {name}: {sc:.6f}")

    if start_results[0][0] < best_score:
        best_score = start_results[0][0]
        best_f = start_results[0][2].copy()

    # Allocate time: give more time to better starts
    # Use top 5 starts with decreasing time allocation
    time_per_start = (budget_s - 60) / 5.0 * 0.6  # leave time for polish
    starts_to_optimize = start_results[:5]

    for idx, (sc, name, f0) in enumerate(starts_to_optimize):
        if time.monotonic() > deadline - 120:
            break

        t_alloc = time_per_start * (1.0 - 0.1 * idx)
        start_deadline = time.monotonic() + t_alloc
        print(f"\nOptimizing from '{name}' (time budget: {t_alloc:.0f}s)")

        f_opt, sc_opt = _optimize_from(f0, n, start_deadline, rng, start_label=name)

        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  New best: {best_score:.6f}")

    # Polish phase
    print(f"\nPolishing best solution (score={best_score:.6f})...")
    polish_deadline = time.monotonic() + min(120.0, max(30.0, budget_s - (time.monotonic() - (time.monotonic() - (budget_s - 10.0)))))
    polish_deadline = min(time.monotonic() + 180.0, deadline)

    f_polished, sc_polished = _polish(best_f, n, polish_deadline, rng)
    if sc_polished < best_score:
        best_score = sc_polished
        best_f = f_polished.copy()
        print(f"  Polished to: {best_score:.6f}")

    # One more round of optimization from the polished solution
    if time.monotonic() < deadline - 30:
        print("\nFinal optimization round...")
        deadline2 = min(time.monotonic() + 120.0, deadline)
        f_final, sc_final = _optimize_from(best_f, n, deadline2, rng, start_label="final")
        if sc_final < best_score:
            best_score = sc_final
            best_f = f_final.copy()
            print(f"  Final: {best_score:.6f}")

    # Ensure valid output
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = _eval(best_f)
    if final_score < best_score or not np.isfinite(best_score):
        best_score = final_score

    print(f"\n=== FINAL SCORE: {best_score:.6f} ===")
    return [float(x) for x in best_f]
