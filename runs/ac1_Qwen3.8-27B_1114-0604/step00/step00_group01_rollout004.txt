```python
from scipy.optimize import linprog

def solve_lp(a, M):
    n = len(a)
    # Maximize sum(b) subject to (a*b)_k <= M, b >= 0
    # In standard form: minimize -sum(b) subject to A @ b <= M, b >= 0
    c = -np.ones(n)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = a[i]
    b_ub = np.full(2*n-1, M)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return result.x
```
