import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix

def compute_score_fast(a):
    """Compute the score without calling evaluate_sequence (for speed)."""
    n = len(a)
    c = np.convolve(a, a)
    max_b = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_b / (S * S)

def solve_lp(f, M):
    """Solve the LP to find g that maximizes sum(g) subject to (f*g)[k] <= M, g >= 0."""
    n = len(f)
    # Build the constraint matrix A where A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # We want A @ g <= M for all k = 0, ..., 2n-2
    
    # Using sparse matrix
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A.tocsr(), b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x.copy()
        return g
    return None

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def compute_conv(a):
    """Compute self-convolution efficiently."""
    return np.convolve(a, a)

def compute_score(a):
    """Compute the score manually (faster than calling evaluate_sequence)."""
    n = len(a)
    c = np.convolve(a, a)
    max_b = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_b / (S * S)

def build_toeplitz_matrix(f):
    """Build the (2n-1) x n Toeplitz-like matrix for the LP constraints."""
    n = len(f)
    # Row k, column j: f[k-j] if 0 <= k-j < n, else 0
    # This is a conjugate Toeplitz matrix
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                data.append(f[idx])
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def solve_lp(f, M):
    """Solve: max sum(g) s.t. T(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_toeplitz_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x.copy()
    return None

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """LP-based iterative optimization to minimize peak self-convolution."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0
    
    best_overall = None
    best_score_overall = np.inf

        if time.monotonic() > deadline:
            break
        
        # Try multiple random restarts
        for restart in range(5):
            if time.monotonic() > deadline:
                break
            
            # Random starting point
            a = rng.random(n) * 0.5 + 0.25
            a = a / a.sum()
            
            best = a.copy()
            best_score = compute_score(best)
            
            print(f"n={n}, restart={restart}, initial score={best_score:.6f}")
            
            stagnation = 0
            for it in range(2000):  # max iterations
                if time.monotonic() > deadline:
                    break
                
                c = np.convolve(best, best)
                M = c.max()
                
                g = solve_lp(best, M)
                if g is None:
                    break
                
                g_sum = g.sum()
                if g_sum < 1e-10:
                    break
                
                # Line search
                improved = False
                for t in [0.5, 0.3, 0.2, 0.1, 0.7, 0.8, 1.0]:
                    candidate = (1 - t) * best + t * g

                    score = compute_score(candidate)
                    if score < best_score - 1e-10:

                        best_score = score
                        improved = True

                best_score_overall = best_score
                best_overall = best.copy()
                print(f"  New best: score={best_score:.6f} at n={n}, restart={restart}")
        
        print(f"Best so far after n={n}: {best_score_overall:.6f}")
    
    # Final polish: local search on the best sequence

        best_score = best_score_overall

def greedy_peak_reduction(a, n_steps=10):
    """Reduce the peak of a*a by adjusting entries that contribute to it."""
    n = len(a)
    a = a.copy()
    S = a.sum()
    
    for _ in range(n_steps):
        c = np.convolve(a, a)
        M = c.max()
        peak_lags = np.where(c > 0.9 * M)[0]  # lags close to the peak
        
        # Compute sensitivity for each entry
        # dc[k]/da[m] = 2*a[k-m]
        sensitivity = np.zeros(n)
        for k in peak_lags:
            for m in range(n):
                idx = k - m
                if 0 <= idx < n:
                    sensitivity[m] += 2 * a[idx]
        
        # Reduce entries with high sensitivity, increase those with low
        mean_sens = sensitivity.mean()
        adjustment = sensitivity - mean_sens
        # Small step
        step = 0.01 * M / (S * S)  # scale the adjustment
        a_new = a - step * adjustment
        a_new = np.clip(a_new, 0, 1000)
        a_new = a_new / a_new.sum() * S  # maintain the sum
        
        score_old = compute_score_fast(a)
        score_new = compute_score_fast(a_new)
        if score_new < score_old:
            a = a_new
