the auto-convolution peak of non-negative sequences. Exploits the insight
that non-symmetric sequences can achieve score < 2 by shifting the
convolution peak off-center.

import time

def compute_full_conv(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    L = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:2*n-1]

def softmax_weights(conv, tau):
    """Compute softmax weights."""
    c = np.max(conv)
    w = np.exp(tau * (conv - c))
    s = w.sum()
    if s == 0:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w
