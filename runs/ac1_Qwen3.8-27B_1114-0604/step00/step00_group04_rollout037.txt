```python
def binomial_initial(n):

def solve_lp_for_direction(a, M):
    """Solve LP to find best direction."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    # Constraint: (a * b)[k] <= M for all k
    # Build constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    constraints = [A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    # Solve
    prob.solve(solver=cp.GLOP)  # or SCIP, CBC, etc.
    return b.value
```

```python
"""
LP-based iterative refinement for minimizing peak self-convolution of non-negative sequences.
Uses binomial coefficient sequences as initial points and solves LPs to find improving directions.
"""
import numpy as np
import cvxpy as cp
import time
from math import comb as math_comb

def binomial_initial(n):
    """Generate initial sequence using truncated central binomial coefficients / 4^k."""
    a = np.zeros(n)
    for k in range(n):
        a[k] = math_comb(2*k, k) / (4.0 ** k)
    return a

def compute_score_fast(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv_max = np.max(np.convolve(a, a))
    return 2.0 * n * conv_max / (s * s)

def build_A(a):
    """Build matrix A such that (A @ b)[k] = (a*b)[k] for all k."""
    n = len(a)
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            A[k, j_min:j_max+1] = a[k-j_max:k-j_min+1][::-1]
    return A

def lp_iterate(a, max_time, initial_time=0.0):
    """Run LP-based iterative refinement on sequence a."""
    n = len(a)
    a = a.copy().astype(float)
    best = a.copy()
    best_score = compute_score_fast(a)
    
    A = build_A(a)
    t0 = time.monotonic() - initial_time
    deadline = max_time
    iteration = 0
    stuck_count = 0
    
    while time.monotonic() - t0 < deadline:
        conv = np.convolve(a, a)
        M = np.max(conv)
        
        # Solve LP: max sum(b) s.t. A @ b <= M, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        
        if b_var.value is None:
            break
        
        b_dir = np.maximum(0, b_var.value)
        sum_b = np.sum(b_dir)
        if sum_b < 1e-10:
            break
        
        # Check if we're stuck (b is proportional to a)
        b_scaled = (np.sum(a) / sum_b) * b_dir
        step = b_scaled - a
        if np.linalg.norm(step) < 1e-10 * np.linalg.norm(a):
            stuck_count += 1
            if stuck_count > 10:
                # Add perturbation to escape
                rng = np.random.default_rng(iteration)
                a = a * (1 + 0.01 * rng.standard_normal(n))
                a = np.maximum(a, 0)
                best_score = compute_score_fast(a)
                A = build_A(a)
                stuck_count = 0
                continue
        else:
            stuck_count = 0
        
        # Line search
        best_t = 0.1
        best_s = compute_score_fast(a)
        for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
            a_trial = (1-t)*a + t*b_scaled
            s = compute_score_fast(a_trial)
            if s < best_s:
                best_s = s
                best_t = t
        
        a = (1-best_t)*a + best_t*b_scaled
        score = compute_score_fast(a)
        
        if score < best_score:
            best_score = score
            best = a.copy()
        
        A = build_A(a)
        iteration += 1
        if iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Iter {iteration}: score={best_score:.6f}, M={np.max(np.convolve(a,a)):.6f}, elapsed={elapsed:.1f}s")
    
    return best, best_score

def local_search(a, time_budget, rng):
    """Simple block perturbation local search."""
    a = a.copy().astype(float)
    best = a.copy()
    best_score = compute_score_fast(a)
    n = len(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_budget:
        width = int(rng.integers(max(1, n // 50), max(2, n // 5)))
        start = int(rng.integers(0, n - width))
        candidate = a.copy()
        candidate[start:start+width] *= np.exp(rng.normal(0, 0.05, width))
        candidate = np.maximum(candidate, 0)
        score = compute_score_fast(candidate)
        if score < best_score:
            best_score = score
            best = candidate
            a = candidate
    
    return best, best_score

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0
    
    best_sequence = None
    best_score = np.inf
```

```python
Uses binomial coefficient sequences as initial points, solves LPs to find

import numpy as np
import cvxpy as cp
import time
from math import comb

    a = np.zeros(n)
    for k in range(n):
        a[k] = comb(2*k, k) / (4.0 ** k)
    return a

def score_fast(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    return 2.0 * n * np.max(np.convolve(a, a)) / (s * s)

def make_A(a):
    n = len(a)
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        A[k, lo:hi+1] = a[k-hi:k-lo+1][::-1]
    return A
```
