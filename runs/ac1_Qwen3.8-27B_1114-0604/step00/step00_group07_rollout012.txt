```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def fast_score(a):
    """Compute score efficiently."""
    n = len(a)
    b = np.convolve(a, a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    return 2.0 * n * np.max(b) / (s * s)

def lp_iteration(f, n_vars):
    """One LP iteration: find g maximizing sum(g) s.t. f*g <= max(f*f), g >= 0."""
    n = len(f)
    M = np.max(np.convolve(f, f))
    
    # Build sparse constraint matrix A: (2n-1) x n
    # A[k, j] = f[j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for j in range(n):
        if f[j] == 0:
            continue
        k_start = 0
        k_end = n - 1 + j  # k-j ranges from 0 to n-1, so k ranges from j to j+n-1
        for k in range(k_start, min(k_end, 2*n-1) + 1):
            if 0 <= k - j < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[j])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return f  # fallback
    
def peak_reduction(a, lr, num_peaks=5):
    """Reduce the top peaks of the convolution."""
    n = len(a)
    b = np.convolve(a, a)
    # Find top peaks
    peak_indices = np.argsort(b)[::-1][:num_peaks]
    # Compute average subgradient
    v = np.zeros(n)
    for k in peak_indices:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                v[i] += a[j]
    v /= len(peak_indices)
    a_new = np.maximum(0, a - lr * v)
    return a_new
```

```python
def fast_score(a):
```
