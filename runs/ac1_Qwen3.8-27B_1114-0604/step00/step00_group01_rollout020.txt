The projection onto the simplex {x >= 0, sum(x) = 1} can be done using the algorithm from Duchi et al. (2011):
- Sort x in descending order
- Find the largest rho such that sum of (x_i - theta) for the top rho elements = 1, where theta = (sum of top rho - 1) / rho
- Project: x_i = max(x_i - theta, 0)

```python
def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_hat = rho.max()
    theta = (cssv[rho_hat] - 1) / (rho_hat + 1.0)
    return np.maximum(v - theta, 0)
```

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    s = v.sum()
    if s <= 0:
        return np.ones(n) / n
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = cond.nonzero()[0]
    if len(rho) == 0:
        return np.ones(n) / n
    idx = rho[-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)
