import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_indices(n):
    """Build row/col arrays for the convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            rows_list.append(k)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)

def make_exponential_ramp(n, alpha):
    """Create an exponential ramp sequence."""
    if n == 1:
        return np.ones(1)
    return np.exp(alpha * np.arange(n, dtype=np.float64) / (n - 1))

    rows_idx, cols_idx = build_conv_indices(n)
    nnz = len(rows_idx)
