LP-based iterative optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses direction-finding LP at each iteration, line search blending, multiple
starting points, and adaptive time allocation.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _build_conv_indices(n):
    """Precompute COO indices for the (2n-1) x n linear convolution matrix."""
    counts = np.minimum(
        np.minimum(np.arange(2 * n - 1) + 1, 2 * n - np.arange(2 * n - 1)), n)
    k_arr = np.repeat(np.arange(2 * n - 1, dtype=np.int32), counts)
    j_starts = np.maximum(0, k_arr.astype(np.int64) - n + 1).astype(np.int32)
    total = int(counts.sum())
    offsets = np.empty(total, dtype=np.int32)
    pos = 0
    for c in counts:
        c = int(c)
        offsets[pos:pos + c] = np.arange(c, dtype=np.int32)
        pos += c
    j_arr = (j_starts + offsets).astype(np.int32)
    i_arr = (k_arr - j_arr).astype(np.int32)
    return k_arr, j_arr, i_arr


def _eval_norm(f, n):
    """Evaluate objective 2*n*max(conv(f,f)) for normalized (sum=1) f."""
    return 2.0 * n * float(np.max(np.convolve(f, f)))


def _solve_direction_lp(f, k_arr, j_arr, i_arr, n, M, time_limit):
    """Solve: max sum(b) s.t. A_f @ b <= M*ones, b >= 0. Returns optimal b or None."""
    vals = f[i_arr]
    A = csr_matrix((vals, (k_arr, j_arr)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': time_limit})
    if res.success and res.x is not None:
        return res.x
    return None

def _subgradient_polish(f_init, n, deadline, rng):
    """Subgradient descent polish after LP. Returns (best_f, best_score)."""
    f = f_init.copy()
    best_f = f.copy()
    best_s = _eval_norm(best_f, n)
    lr = 0.5
    it = 0

    while time.monotonic() < deadline:
        it += 1
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            break

        conv = np.convolve(f, f)
        C_max = float(np.max(conv))
        k_star = int(np.argmax(conv))
        S = float(np.sum(f))

        # Subgradient of max(C) w.r.t. a_j = a_{k*-j} + a_{j-k*} (with a_i=0 outside [0,n-1])
        # For linear conv: C_k = sum_i a_i * a_{k-i} (valid indices only)
        # d(C_k)/d(a_j) = a_{k-j} if 0<=k-j<n + a_{j-k} if 0<=j-k<n
        # For k=k_star:
        grad = np.zeros(n)
        # a_{k_star - j} for j in range(max(0, k_star-n+1), min(k_star+1, n))
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star + 1, n)
        for j in range(j_lo, j_hi):
            idx1 = k_star - j
            if 0 <= idx1 < n:
                grad[j] += f[idx1]
        # a_{j - k_star} for j in range(k_star, min(k_star+n, n))
        j_lo2 = k_star
        j_hi2 = min(k_star + n, n)
        for j in range(j_lo2, j_hi2):
            idx2 = j - k_star
            if 0 <= idx2 < n:
                grad[j] += f[idx2]

        # Subtract the (sum) term: 2*C_max/S
        grad = grad - 2.0 * C_max / S

        # Projected descent (non-negative constraint)
        f_try = f - lr * grad
        f_try = np.clip(f_try, 0.0, None)
        s_try = float(np.sum(f_try))
        if s_try < 1e-6:
            lr *= 0.5
            continue
        f_try = f_try / s_try
        obj = _eval_norm(f_try, n)

            best_s = obj
            f = f_try
            lr = min(lr * 1.1, 1.0)

                break

        if it % 20 == 0:
            print(f"  Subgrad iter {it}: score={best_s:.6f} lr={lr:.4f}", flush=True)

    return best_f, best_s
