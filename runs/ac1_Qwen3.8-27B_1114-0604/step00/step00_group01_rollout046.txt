"""
Minimize 2*n*max(a*a)/(sum(a))^2 for non-negative a by:
- diversified starts (uniform, mirror-peak, smooth window, random symmetric),
- mirrored projected-gradient flow on a log-sum-exp surrogate,
- annealed symmetric mixing + occasional random restarts/perturbations,
- final un-mirrored local search to polish the exact score.
"""

import time
import numpy as np
from scipy.signal import fftconvolve

IS_INF_SCORE = float("inf")


def _nonneg(a):
    a = np.asarray(a, dtype=np.float64)
    return np.maximum(a, 0.0)


def _normalize_sum(a):
    s = float(np.sum(a))
    if s > 1e-12:
        return a / s
    return a


def _fast_conv(a, b=None):
    if b is None:
        b = a
    return fftconvolve(np.asarray(a, dtype=np.float64), np.asarray(b, dtype=np.float64))


def _fast_peakconv(a):
    return float(np.max(_fast_conv(a, a)))


def _fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(np.sum(a))
    if S < 0.01:
        return IS_INF_SCORE
    M = _fast_peakconv(a)
    return float(2.0 * n * M / (S * S))


def _mirror(a):
    return np.asarray(a, dtype=np.float64)[::-1].copy()


def _avg_mirror(a):
    return 0.5 * (a + _mirror(a))


def _build_starts(n, rng, base):
    starts = []
    if base is not None:
        b = _nonneg(base)
        if float(b.size) == n and float(b.sum()) > 1e-9:
            starts.append(("base", b.copy()))

    starts.append(("uniform", np.ones(n, dtype=np.float64)))

    # smooth symmetric windows
    j = np.arange(n, dtype=np.float64)
    c = 0.5 * (n - 1.0)
    for wname, w in [
        ("hann", 0.5 - 0.5 * np.cos(2.0 * np.pi * (j + 0.5) / n)),
        ("gauss", np.exp(-0.5 * ((j - c) / (0.25 * max(1.0, n / 6.0))) ** 2)),
        ("cos2", 0.5 + 0.5 * np.cos(np.pi * (j - c) / max(1.0, c + 1.0))),
    ]:
        w = _nonneg(w) + 1e-6
        starts.append((wname, w / w.sum()))

    # random symmetric smooth starts
    for k in range(3):
        base_r = rng.exponential(1.0, size=n)
        sm = _avg_mirror(base_r)
        if float(sm.sum()) < 1e-9:

        starts.append((f"rand_sym_{k}", sm / sm.sum()))

    return starts


def _grad_step(a, beta, lr, use_mirror):
    c = _fast_conv(a, a)
    cmax = float(c.max())
    shifted = beta * (c - cmax)
    e = np.exp(shifted)
    esum = float(e.sum())
    p = e / esum

    n = int(a.size)

    a2 = a - lr * g
    a2 = _nonneg(a2)
    if use_mirror:
        a2 = _avg_mirror(a2)
    return a2

            print(f"[phase2] unmirrored refine score={sc:.6f}")

        # optional second mirrored polish
        if time.monotonic() < deadline - 60:
            a2, sc2 = _alternating_descent(global_best, min(120.0, deadline - time.monotonic() - 60), rng,
                                           mirror_strength=0.20)
            sc2v = min(sc2, _fast_score(a2))
            if sc2v < global_score:
                global_score = sc2v
                global_best = a2.copy()
                print(f"[phase3] second polish score={sc2v:.6f}")

    # final verification
    if global_best is None:
        global_best = np.ones(n, dtype=np.float64)

    global_best = _nonneg(global_best)
    if float(global_best.sum()) < 0.01:
        global_best = np.ones(n, dtype=np.float64)

    # clamp to evaluator bounds
    global_best = np.clip(global_best, 0.0, 1000.0)

    out = [float(x) for x in global_best]
    verified = float(evaluate_sequence(out))
    if np.isfinite(verified) and verified < global_score:
        global_score = verified
        print(f"[final] local_best={global_score:.6f} verified={verified:.6f} len={len(out)}")
    else:
        print(f"[final] local_best={global_score:.6f} verified={verified} len={len(out)}")

    return out
