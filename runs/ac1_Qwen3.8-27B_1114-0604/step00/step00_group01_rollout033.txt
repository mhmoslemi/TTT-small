```python
def compute_smooth_gradient(a, c, K, t):
    """Compute gradient using top-K lags with softmax weights."""
    n = len(a)
    # Get top-K values
    top_idx = np.argpartition(c, -K)[-K:]
    top_vals = c[top_idx]
    # Softmax weights
    top_max = top_vals.max()
    w = np.exp(t * (top_vals - top_max))
    w = w / w.sum()
    
    sg = np.zeros(n)
    for i, k in enumerate(top_idx):
        if w[i] < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        j_vals = np.arange(lo, hi)
        sg[j_vals] += w[i] * 2.0 * a[k - j_vals]
    
    return sg
```

Let me also add a "polishing" step at the end using coordinate descent:
- For each coordinate j, try to find the optimal value (given the rest).
- The contribution of a[j] to c_k is 2*a[j]*a[k-j] (for valid k-j).
- So changing a[j] by delta changes c_k by 2*delta*a[k-j] (approximately, for small delta).
- To minimize the max, we want to reduce the largest c_k that involves a[j].

    """Full convolution using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1

    fft_size = 1
    while fft_size < size:

    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)

import numpy as np

def fft_full_convolve(a, b):
    """Compute full convolution using real FFT."""
    n = len(a)
    m = len(b)
    result_len = n + m - 1
    # Use next power of 2 for FFT efficiency
    fft_len = 1 << (result_len - 1).bit_length()
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    result = np.fft.irfft(fa * fb, fft_len)
    return result[:result_len]

def topk_subgradient(a, c, K, temp):
    """Compute smoothed subgradient using top-K lags with softmax weights."""
    n = len(a)
    c_len = len(c)
    
    if K >= c_len:
        top_idx = np.arange(c_len)
        top_vals = c.copy()
    else:
        # Find top-K indices
        part = np.argpartition(-c, K)[:K]  # K largest values
        top_idx = part[np.argsort(-c[part])]  # sorted descending
        top_vals = c[top_idx]
    
    # Softmax weights (numerically stable)
    vmax = top_vals.max()
    if vmax < 1e-15:
        return np.zeros(n)
    exp_vals = np.exp(temp * (top_vals - vmax))
    w = exp_vals / exp_vals.sum()
    
    # Accumulate gradient
    sg = np.zeros(n)
    for i in range(len(top_idx)):
        wi = w[i]
        if wi < 1e-14:
            continue
        k = top_idx[i]
        # sg[j] += 2 * wi * a[k-j] for j where 0 <= k-j < n
        # j ranges from max(0, k-n+1) to min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:

        # j = lo, lo+1, ..., hi
        # k - j = k - lo, k - lo - 1, ..., k - hi
        # So a[k-j] = a[k-hi : k-lo+1][::-1]
        sg[lo:hi+1] += 2.0 * wi * a[k-hi:k-lo+1][::-1]
    
    return sg

import numpy as np

def fft_full_convolve(a, b):
    """Full linear convolution via real FFT."""
    n = len(a)
    m = len(b)
    result_len = n + m - 1
    fft_len = 1 << (result_len - 1).bit_length()
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    return np.fft.irfft(fa * fb, fft_len)[:result_len]

def topk_subgradient(a, c, K, temp):
    """Smoothed subgradient using top-K lags with softmax weights."""
    n = len(a)
    c_len = len(c)
    K = min(K, c_len)
    
    part = np.argpartition(-c, K - 1)[:K]
    top_idx = part[np.argsort(-c[part])]
    top_vals = c[top_idx]
    
    vmax = top_vals.max()
    if vmax < 1e-15:
        return np.zeros(n)
    
    exps = np.exp(temp * (top_vals - vmax))
    w = exps / exps.sum()
    
    sg = np.zeros(n)
    for i in range(K):
        wi = w[i]
        if wi < 1e-14:
            continue
        k = int(top_idx[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        sg[lo:hi+1] += 2.0 * wi * a[k-hi:k-lo+1][::-1]
    return sg
