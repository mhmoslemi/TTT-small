import numpy as np
from scipy.optimize import linprog
import time

def compute_score(f):
    """Compute the objective value for a sequence."""
    n = len(f)
    conv = np.convolve(f, f)
    max_c = np.max(conv)
    sum_f = np.sum(f)
    if sum_f < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_f ** 2)

def solve_lp(f, M):
    """Solve the LP to find g maximizing sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        A_ub[k, j_min:j_max+1] = f[k - j_max:k - j_min + 1][::-1]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

def lp_iteration(f, max_iters=50, tol=1e-8):

    n = len(f)
    sum_f = np.sum(f)
    f = f.copy()
    
    for it in range(max_iters):
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)
        
        g = solve_lp(f, M)
        if g is None:
            break
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        # Try different step sizes

        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            candidate = (1 - t) * f + t * g
            candidate = np.clip(candidate, 0, 1000)
            score = compute_score(candidate)
            if score < best_score - 1e-10:
                f = candidate
                best_score = score
                improved = True
                break
        
        M_new = np.max(np.convolve(f, f))
        if not improved or abs(M - M_new) / M < tol:
            break
    
    return f, best_score

            print(f"  After grad: score = {score:.8f}")

        # Random polish
        if time.monotonic() < deadline:
            f, score = random_polish(f, deadline, rng)
            print(f"  After random: score = {score:.8f}")

        if score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"  *** New overall best: {best_score:.8f} (n={n}, init={init_type}) ***")

    if best_seq is None:
        best_seq = np.ones(1024, dtype=np.float64)
        best_score = 2.0

    # Final intensive polish on the best
    print(f"\n=== Final polish on best (n={len(best_seq)}, score={best_score:.8f}) ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # Another round of LP
        alloc = remaining * 0.4
        local_deadline = time.monotonic() + alloc
        f, score = lp_optimize(best_seq.copy(), local_deadline, max_stall=15, verbose=True)
        if score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"  Final LP: {best_score:.8f}")

    if time.monotonic() < deadline:
        # Gradient polish
        local_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.4
        f, score = gradient_polish(best_seq.copy(), local_deadline, max_iters=30000)
        if score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"  Final grad: {best_score:.8f}")

    if time.monotonic() < deadline:
        # Random polish
        f, score = random_polish(best_seq.copy(), deadline, rng)
        if score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"  Final random: {best_score:.8f}")

    # Validate with evaluate_sequence
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    eval_score = float(evaluate_sequence(final_list))
    print(f"\nFinal result: n={len(final_list)}, computed_score={best_score:.8f}, "
          f"eval_score={eval_score:.8f}")

    return final_list
