direction b that is "compatible" with the current sequence a (bounded
cross-correlation), then update a in the direction of b.

import numpy as np

from scipy.optimize import linprog


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix where A[k,i] = a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        A[k, i_min:i_max + 1] = a[i_max - k : k - i_min + 1][::-1]
    return A


def score_sequence(a):
    """Compute the evaluation score."""
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * len(a) * conv.max() / (s * s)


def lp_iteration(a, time_limit=20.0):
    """
    One LP iteration: find b >= 0 maximizing sum(b)
    subject to (a*b)[k] <= M for all k.
    Returns b (or None if LP fails).
    """
    n = len(a)
    conv_a = np.convolve(a, a)
    M = conv_a.max()
    
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': time_limit})
    
    if not res.success or res.x is None:
        return None, M
    return res.x, M
