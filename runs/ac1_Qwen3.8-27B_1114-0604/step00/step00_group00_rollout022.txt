The convolution theorem says: `FFT(a * b) = FFT(a) * FFT(b)` where `*` on the left is linear convolution and `*` on the right is pointwise multiplication.

Therefore `conv(a,a) = IFFT(FFT(a)^2)`.

def _fft_self_conv(a, N):
    """Compute conv(a, a) using FFT. N must be >= 2*len(a)-1."""
    F = np.fft.rfft(a, N)
    c = np.fft.irfft(F * F, N)
    return c[:2*len(a)-1]

def _next_pow2(m):
    """Next power of 2 >= m."""
    n = 1
    while n < m:
        n <<= 1
    return n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed projected gradient descent
with FFT-accelerated convolution, multiple initializations, temperature annealing,
and local refinement. Exploits the scale-invariance of the objective to fix sum(a)=1.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_self_conv(a, N):
    """Compute linear conv(a, a) via FFT. N >= 2*len(a)-1."""
    F = np.fft.rfft(a, N)
    c = np.fft.irfft(F * F, N)
    return c[:2 * len(a) - 1]


def _compute_gradient(weights, a, N):
    """Compute grad[j] = 2 * sum_k weights[k] * a[k-j] via FFT cross-correlation."""
    F_w = np.fft.rfft(weights, N)
    F_a = np.fft.rfft(a, N)
    ncorr = np.fft.irfft(F_w * np.conj(F_a), N)
    return 2.0 * ncorr[:len(a)]


def _optimize_sequence(a_init, max_iters=3000, T_start=2.0, T_end=200.0,
                       base_lr=0.005, n_print=500):
    """Optimize non-negative sequence to minimize 2*n*max(conv(a,a)) given sum(a)=1."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    N = _next_pow2(4 * n)  # Enough for cross-correlation (needs >= 3n-2)

    c = _fft_self_conv(a, N)
    best_val = 2.0 * n * np.max(c)
    best_a = a.copy()

    stall_count = 0
    stall_threshold = 500

    for it in range(max_iters):
        progress = it / max_iters
        T = T_start * (T_end / T_start) ** progress

        # Compute convolution
        c = _fft_self_conv(a, N)
        c_max = np.max(c)

        # Softmax weights for smooth max
        exp_c = np.exp(T * (c - c_max))
        sum_exp = np.sum(exp_c)
        weights = exp_c / sum_exp

        # Compute gradient
        grad = _compute_gradient(weights, a, N)

        # Adaptive learning rate
        lr = base_lr / (1.0 + 0.01 * it)

        # Update and project to simplex {a >= 0, sum(a) = 1}
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Evaluate
        c_new = _fft_self_conv(a_new, N)
        val_new = 2.0 * n * np.max(c_new)

        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        # If stalled, add small random perturbation to escape local min
        if stall_count > stall_threshold:
            rng = np.random.default_rng(it)
            noise = rng.normal(0, 0.01, n)
            a = a + noise * (np.max(a) * 0.1)
            A = np.maximum(a, 0)
            sA = np.sum(A)
            if sA > 1e-10:
                a = A / sA
            stall_count = 0

    return best_a, best_val

    rng = np.random.default_rng(seed)

    inits.append(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n) / (2.0 * n)))

    # 8. Random positive
    inits.append(rng.random(n) + 0.01)

    inits.append(np.exp(-np.arange(n) / (n / 3.0)))

    inits.append(np.exp(-np.arange(n - 1, -1, -1) / (n / 3.0)))

    inits.append(np.abs(signal) + 0.1)
