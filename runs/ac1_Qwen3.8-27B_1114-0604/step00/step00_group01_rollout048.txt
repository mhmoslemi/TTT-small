```python
"""
Minimize  2*n*max(a*a) / (sum a)^2  over non-negative sequences a.

Primary driver: reproduce the state-of-the-art LP iteration.
    Given current a (with M = max(a*a)), solve
        max  sum_j b_j   s.t.  b_j >= 0 ,  (a*b)_k <= M  for all k
    as a linear program; the optimal b* gives a descent direction.
    Rescale b* to the normalization and update  a <- (1-t)*a + t*b*.
Alternate several lengths and a cheap FFT-based "push-pull" / smoothed-
subgradient polish to escape local structure.  The exact evaluator
(authoritative) is used for all kept scores.  Several sequence lengths are
tried; the best found is returned.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog


def _next_pow2(m):
    """Smallest power of two >= m (m>=1)."""
    r = 1
    while r < m:
        r <<= 1
    return r


def _conv_sq(a):
    """Linear convolution a * a (length 2n-1) via FFT."""
    n = a.size
    size = 2 * n - 1
    fs = _next_pow2(size)
    A = np.fft.rfft(a, fs)
    out = np.fft.irfft(A * A, fs)
    return out[:size]


def _normalize(a, S):
    """Project to non-negative and rescale to sum S."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.full(a.size, S / a.size)
    return a * (S / s)


def _subgrad_smooth(a, C, temperature):
    """Smoothed subgradient of max(C): g[j] = sum_k W[k] a[k-j]."""
    n = a.size
    size = C.size
    cmax = np.max(C)
    scaled = (C - cmax) / max(temperature, 1e-12)
    scaled = np.maximum(scaled, -60.0)
    W = np.exp(scaled)
    s = W.sum()
    if s < 1e-300:
        W = np.zeros(size)
        W[int(np.argmax(C))] = 1.0
    else:
        W /= s
    a_rev = a[::-1]
    cs = size + n - 1
    fs = _next_pow2(cs)
    G = np.fft.irfft(np.fft.rfft(W, fs) * np.fft.rfft(a_rev, fs), fs)
    return G[n - 1:n - 1 + n].copy()


def _push_pull(a, C, eta):
    """
    Reduce the elements that contribute to the max position and push the
    removed mass onto elements that do NOT contribute.  Returns None when the
    overlap (contributing set) is essentially the whole sequence, in which
    case the caller should fall back to a different move.
    """
    n = a.size
    kstar = int(np.argmax(C))
    lo = max(0, kstar - n + 1)
    hi = min(n - 1, kstar)
    width = hi - lo + 1
    if width >= n - 1:
        return None
    js = np.arange(lo, hi + 1)
    g = 2.0 * a[kstar - js]
    gsum = g.sum()
    if gsum <= 0:
        return None
    a_mean = a.mean()
    mass = g * (eta * a_mean) / gsum
    a_new = a.copy()
    a_new[lo:hi + 1] -= mass
    non = np.ones(n, dtype=bool)
    non[lo:hi + 1] = False
    if non.any():
        total = mass.sum()
        a_non = a_new[non]
        ansum = a_non.sum()
        if ansum > 1e-15:
            a_new[non] += total * a_non / ansum
        else:
            a_new[non] += total / non.sum()
    return a_new


def _hankel(a):
    """Build the (2n-1) x n matrix H with H[k,j] = a[k-j] (0<=k-j<=n-1)."""
    n = a.size
    js = np.tile(np.arange(n, dtype=np.int64), n)
    ds = np.repeat(np.arange(n, dtype=np.int64), n)
    rows = ds + js
    vals = np.repeat(a, n)
    mat = sparse.coo_matrix((vals, (rows, js)), shape=(2 * n - 1, n))
    return mat.tocsr()


def _lp_direction(a, M):
    """Solve max sum(b) s.t. b>=0, a*b <= M.  Returns normalized b* or None."""
    n = a.size
    H = _hankel(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, float(M))
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=H, b_ub=b_ub, bounds=bounds, method="highs")
    except Exception:
        return None
    if res is None or (not res.success) or res.x is None:
        return None
    b = np.maximum(0.0, res.x)
    if b.sum() <= 1e-15:
        return None
    S = np.sum(a)
    return _normalize(b, S)


def _make_init(n, rng):
    """Asymmetric non-negative initializer."""
    t = np.arange(n, dtype=np.float64)
    base = (t + 1.0) * 0.5 + 0.3 * np.exp(-2.0 * t / n) + 0.2 * np.cos(np.pi * t / n)
    base = base + 0.4 * rng.standard_normal(n)
    base = base + 0.3 * (t / max(n - 1, 1))      # break palindromic symmetry
    return _normalize(np.maximum(0.1, base), np.sqrt(2.0 * n))


def _lp_optimize(n, stop_time, rng, best_out, best_score):
    """Run LP-iteration (+ light polish) at length n until stop_time."""
    S = np.sqrt(2.0 * n)
    a = _make_init(n, rng)
    sc = float(evaluate_sequence(a.tolist()))
    if sc < best_score:
        best_out[:] = a
        best_score = sc
        print("  [n=%d] init score=%.6f" % (n, sc))

    it = 0
    while time.monotonic() < stop_time:
        C = _conv_sq(a)
        M = float(np.max(C))
        b = _lp_direction(a, M)
        if b is not None:
            t = 0.55 if it < 30 else 0.25
            a = (1.0 - t) * a + t * b
        else:
            # fallback: push-pull or smoothed subgradient
            a2 = _push_pull(a, C, 0.5)
            if a2 is None:
                g = _subgrad_smooth(a, C, 0.1 * (np.max(C) - np.min(C) + 1e-12))
                a2 = a - (0.3 * a.mean() / (np.max(np.abs(g)) + 1e-12)) * g
                a2 = a2 + 0.02 * rng.standard_normal(n)   # keep symmetry broken
            a = a2
        a = _normalize(a, S)
        it += 1
        if it % 4 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if sc < best_score:
                best_out[:] = a
                best_score = sc
                print("  [n=%d] it=%d score=%.6f t=%.0fs" % (n, it, sc, time.monotonic() - global_t0))
    return best_score


def _polish(a, stop_time, rng, best_out, best_score):
    """Cheap FFT-based polish (push-pull + smoothed subgradient) in place."""
    n = a.size
    S = np.sqrt(2.0 * n)
    a = a.copy()
    it = 0
    while time.monotonic() < stop_time:
        C = _conv_sq(a)
        a2 = _push_pull(a, C, 0.4)
        if a2 is None:
            g = _subgrad_smooth(a, C, 0.05 * (np.max(C) - np.min(C) + 1e-12))
            a2 = a - (0.2 * a.mean() / (np.max(np.abs(g)) + 1e-12)) * g
            a2 = a2 + 0.015 * rng.standard_normal(n)
        a = _normalize(a2, S)
        it += 1
        if it % 30 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if sc < best_score:
                best_out[:] = a
                best_score = sc
                print("  polish it=%d score=%.6f t=%.0fs" % (it, sc, time.monotonic() - global_t0))
    return best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a low-score non-negative sequence; see module docstring."""
    global global_t0
    global_t0 = time.monotonic()
    budget_s = float(budget_s)
    deadline = global_t0 + max(0.0, budget_s - 15.0)
    rng = np.random.default_rng(seed)

    # lengths to try with the (memory-heavy) LP; bounded for memory safety
    lengths = [1536, 2048, 3072]
    try:
        pl = len(np.asarray(height_sequence_1, dtype=np.float64).ravel())
    except Exception:
        pl = 0
    if 512 <= pl <= 4096 and pl not in lengths:
        lengths.append(int(pl))
    lengths = sorted(set(lengths))

    overall = np.ones(1536, dtype=np.float64)
    overall_score = np.inf

    # time slicing: leave room for polish
    remaining = len(lengths)
    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        slice_s = min(450.0, max(60.0, (deadline - time.monotonic() - 60.0) / remaining))
        n_stop = time.monotonic() + slice_s
        remaining -= 1
        try:
            local_best = np.ones(n, dtype=np.float64)
            overall_score = _lp_optimize(n, n_stop, rng, local_best, overall_score)
            if np.all(local_best > 0) or (overall_score > 0 and local_best.shape[0] > 0):
                # adopt this length's best if it improved the global best
                if np.max(np.abs(local_best)) > 0 and np.isfinite(overall_score):
                    overall = local_best.copy()
        except MemoryError:
            print("  MemoryError at n=%d, skipping" % n)
        except Exception as e:
            print("  error at n=%d: %s" % (n, repr(e)))

    # if LP never ran / no improvement, fall back to a polished initializer
    if overall_score == np.inf:
        n = 2048
        overall = _make_init(n, rng)
        overall_score = float(evaluate_sequence(overall.tolist()))

    # final cheap polish on the current best (length = len(overall))
    try:
        overall_score = _polish(overall, deadline - 6.0, rng, overall, overall_score)
    except Exception as e:
        print("  polish error: %s" % repr(e))

    # authoritative final evaluation
    try:
        final_score = float(evaluate_sequence([float(x) for x in overall]))
    except Exception:
        final_score = np.inf
    if not np.isfinite(final_score):
        # guaranteed-valid fallback: a small ramp-derived sequence
        fb = np.arange(256, dtype=np.float64) + 1.0
        final_score = float(evaluate_sequence([float(x) for x in fb]))
        overall = fb

    # sanity clip
    overall = np.clip(np.asarray(overall, dtype=np.float64), 0.0, 1000.0)
    print("FINAL score=%.6f n=%d t=%.0fs" % (final_score, len(overall), time.monotonic() - global_t0))
    return [float(x) for x in overall]
```
