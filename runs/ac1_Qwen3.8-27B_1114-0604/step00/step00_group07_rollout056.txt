import numpy as np
import time
import math

try:
    from scipy.optimize import minimize as _scipy_minimize
except Exception:
    _scipy_minimize = None


_CURRENT_N = 1
_CURRENT_M = 0
_CURRENT_NFFT = 1
_CURRENT_BETA = 1.0
_CURRENT_GRAD_SCALE = 1.0


def _next_pow2(m):
    p = 1
    m = max(1, int(m))
    while p < m:
        p <<= 1
    return p


def _normalize(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.size == 0:
        return np.array([], dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def _score_fft(a, nfft):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n <= 0:
        return float('inf')
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return float('inf')
    arr = a / s
    A = np.fft.rfft(arr, nfft)
    b = np.fft.irfft(A * A, nfft)[:2 * n - 1]
    b = np.maximum(np.asarray(b, dtype=np.float64), 0.0)
    mx = float(np.max(b))
    if not np.isfinite(mx):
        return float('inf')
    return float(2.0 * n * mx)


def _a_to_z(a):
    a = _normalize(a)
    n = int(a.size)
    if n <= 1:
        return np.zeros(0, dtype=np.float64)
    a = np.maximum(a, 1e-12)
    a = a / float(a.sum())
    last = max(float(a[-1]), 1e-12)
    z = np.log(a[:-1] / last)
    z = np.nan_to_num(z, nan=0.0, posinf=50.0, neginf=-50.0)
    return z.astype(np.float64)


def _softmax_z(z):
    n = int(_CURRENT_N)
    m = int(_CURRENT_M)
    z = np.asarray(z, dtype=np.float64).reshape(-1)
    if z.size < m:
        tmp = np.zeros(m, dtype=np.float64)
        tmp[:z.size] = z
        z = tmp
    elif z.size > m:
        z = z[:m]
    z = np.nan_to_num(z, nan=0.0, posinf=50.0, neginf=-50.0)

    full = np.empty(n, dtype=np.float64)
    full[:-1] = z
    full[-1] = 0.0

    mn = float(np.max(full))
    if not np.isfinite(mn):
        return np.ones(n, dtype=np.float64) / float(n)

    full = full - mn
    e = np.exp(full)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def _lbfgs_objective(z):
    n = int(_CURRENT_N)
    if n <= 1:
        return 0.0

    a = _softmax_z(z)
    nfft = int(_CURRENT_NFFT)
    A = np.fft.rfft(a, nfft)
    b = np.fft.irfft(A * A, nfft)[:2 * n - 1]
    b = np.maximum(np.asarray(b, dtype=np.float64), 0.0)
    c = n * b
    mc = float(np.max(c))

    beta = float(_CURRENT_BETA)
    if beta <= 0.0:
        return mc

    ex = np.exp(beta * (c - mc))
    Z = float(np.sum(ex))
    if not np.isfinite(Z) or Z <= 0.0:
        return mc
    return float(mc + math.log(Z))


def _lbfgs_gradient(z):
    n = int(_CURRENT_N)
    m = int(_CURRENT_M)
    if m <= 0:
        return np.zeros(0, dtype=np.float64)

    a = _softmax_z(z)
    nfft = int(_CURRENT_NFFT)
    A = np.fft.rfft(a, nfft)
    b = np.fft.irfft(A * A, nfft)[:2 * n - 1]
    b = np.maximum(np.asarray(b, dtype=np.float64), 0.0)
    c = n * b
    mc = float(np.max(c))

    beta = float(_CURRENT_BETA)
    if beta <= 0.0:
        w = np.zeros_like(c, dtype=np.float64)
        w[np.abs(c - mc) <= 1e-12 + 1e-9 * max(1.0, mc)] = 1.0
        if float(w.sum()) <= 0.0:
            w.fill(1.0 / float(n + 1))
        beta_eff = 1.0
    else:
        ex = np.exp(beta * (c - mc))
        Z = float(np.sum(ex))
        if not np.isfinite(Z) or Z <= 0.0:
            return np.zeros(m, dtype=np.float64)
        w = ex / Z
        beta_eff = beta

    w_rev = w[::-1]
    W = np.fft.rfft(w_rev, nfft)
    convC = np.fft.irfft(A * W, nfft)
    C = convC[2 * n - 2:n - 2:-1]
    C = np.maximum(np.asarray(C, dtype=np.float64), 0.0)

    g_a = 2.0 * beta_eff * C
    S = float(np.dot(g_a, a))
    g_z = a[:-1] * (g_a[:-1] - S)
    g_z = np.nan_to_num(g_z, nan=0.0, posinf=0.0, neginf=0.0)

    g_z *= float(_CURRENT_GRAD_SCALE)

    mx = float(np.max(np.abs(g_z))) if g_z.size else 0.0
    if not np.isfinite(mx):
        return np.zeros(m, dtype=np.float64)
    if mx > 1e8:
        g_z *= (1e8 / mx)
    return g_z.astype(np.float64)


def _run_lbfgs(z0, beta, maxiter):
    global _CURRENT_BETA, _CURRENT_GRAD_SCALE

    z0 = np.asarray(z0, dtype=np.float64).copy()
    z0 = np.nan_to_num(z0, nan=0.0, posinf=50.0, neginf=-50.0)
    if z0.size == 0:
        return z0

    if _scipy_minimize is None:
        return z0

    beta = float(beta)
    n = float(_CURRENT_N)
    scale = max(1.0, (n * n) / max(beta, 1.0))
    if scale > 1e12:
        scale = 1e12
    _CURRENT_BETA = beta
    _CURRENT_GRAD_SCALE = float(scale)

    try:
        res = _scipy_minimize(
            _lbfgs_objective,
            z0,
            jac=_lbfgs_gradient,
            method='L-BFGS-B',
            options={
                'maxiter': int(maxiter),
                'gtol': 1e-12,
                'maxls': 30,
                'disp': False,
            }
        )
    except Exception:
        return z0

    x = np.asarray(res.x, dtype=np.float64)
    x = np.nan_to_num(x, nan=0.0, posinf=50.0, neginf=-50.0)
    return x

def _add_valid(inits, a, n):
    arr = np.asarray(a, dtype=np.float64)
    if arr.size != n:
        return
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 1e-8)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return
    arr = arr / s
    inits.append(arr.astype(np.float64))


def _resample_parent(parent, n):
    try:
        p = np.asarray(parent, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    p = p / s
    if p.size == n:
        return p.astype(np.float64)
    xs = np.linspace(0.0, 1.0, p.size)
    xn = np.linspace(0.0, 1.0, n)
    y = np.interp(xn, xs, p)
    y = np.maximum(np.asarray(y, dtype=np.float64), 1e-8)
    s = float(y.sum())
    if s <= 1e-12:
        return None
    return y / s

            z0 = best_z + rng.normal(0.0, 0.03, _CURRENT_M)
        elif run_count % 6 == 3:
            z0 = best_z + rng.normal(0.0, 0.10, _CURRENT_M)
        else:
            idx = run_count % len(start_pool)
            z0 = start_pool[idx] + rng.normal(0.0, 0.25, start_pool[idx].size)

        beta = beta_cycle[bi % len(beta_cycle)]
        bi += 1

        if rem > 1500.0:
            maxiter = base_iter
        elif rem > 600.0:
            maxiter = int(base_iter * 0.65)
        else:
            maxiter = int(base_iter * 0.4)

        z_new = _run_lbfgs(z0, beta, maxiter)
        a_new = _softmax_z(z_new)
        sc = _score_fft(a_new, nfft)

        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_a = a_new.copy()
            best_z = z_new.copy()
            if best_score < last_print_score - 1e-4:
                print("improved internal score:", best_score, flush=True)
                last_print_score = best_score

        if time.monotonic() - last_exact > 20.0:
            ex = _exact_eval(best_a)
            if np.isfinite(ex):
                print("periodic exact score:", ex, "at internal", best_score, flush=True)
                last_exact = time.monotonic()

    if time.monotonic() < deadline - 40.0:
        z_new = _run_lbfgs(best_z, 20.0, int(base_iter * 0.75))
        a_new = _softmax_z(z_new)
        sc = _score_fft(a_new, nfft)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            best_z = z_new.copy()

    if time.monotonic() < deadline - 15.0:
        z_new = _run_lbfgs(best_z, 50.0, int(base_iter * 0.5))
        a_new = _softmax_z(z_new)
        sc = _score_fft(a_new, nfft)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            best_z = z_new.copy()

    final_a = best_a.copy()
    final_exact = _exact_eval(final_a)

    for i in range(min(3, len(selected))):
        a = selected[i][1]
        ex = _exact_eval(a)
        if np.isfinite(ex) and (not np.isfinite(final_exact) or ex < final_exact):
            final_exact = ex
            final_a = a.copy()

    final_a = _normalize(final_a)
    print("final internal score:", best_score, flush=True)
    print("final exact score:", final_exact, flush=True)

    return [float(x) for x in final_a]
