    # Softmax weights
    b_shifted = b - np.max(b)
    weights = np.exp(tau * b_shifted)
    weights /= np.sum(weights)
    # v = b * weights
    v = b * weights
    # Cross-correlation: cross_corr[i] = sum_j a[j] * v[i+j]
    N = 1 << (n + m - 1).bit_length()
    A = np.fft.rfft(a, N)
    V = np.fft.rfft(v[::-1], N)
    conv = np.fft.irfft(A * V, N)
    cross_corr = conv[n-1:2*n-1][::-1]  # length n
    return 2 * tau * cross_corr

def proposed_gradient_step(a, lr, tau):
    """One gradient step on the smoothed objective."""
    b = np.convolve(a, a)
    g = compute_gradient(a, b, tau)
    a_new = a - lr * g
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
