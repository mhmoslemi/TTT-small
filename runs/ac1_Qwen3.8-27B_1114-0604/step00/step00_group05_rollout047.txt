"""
Minimize  score(a) = 2*n*max(convolve(a,a)) / (sum a)^2  over a>=0.

The score is scale-invariant, so we work on the simplex sum(a)=1 where the
objective reduces to  f(a) = 2*n*max_k (a*a)[k].  We run projected gradient
descent (projection: clip to >=0 then renormalize to sum 1, with the gradient
projected onto the sum-zero tangent of the simplex) from many diverse,
symmetry-breaking starts, followed by random perturb-perturb restarts on the
global best to escape local minima.  Convolutions use an FFT so each step is
O(n log n), which is fast enough for n ~ 4500 within the time budget.
"""

import numpy as np
import time


def _fft_conv(a):
    """Real linear self-convolution via rFFT. Returns array of length 2n-1."""
    n = a.size
    size = 1
    need = 2 * n - 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    C = np.fft.irfft(F * F, size)[:need]
    return C


def _normalize(a):
    """Clip to >=0 and rescale so that the sum is exactly 1."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 0.0:
        return np.full_like(a, 1.0 / a.size, dtype=np.float64)
    return a / s


def _score(a):
    """Return the true objective value for (unnormalized) a, matching the
    provided evaluate_sequence (negative/inf handled by clipping below)."""
    n = a.size
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    m = float(np.max(_fft_conv(a)))
    return float(2.0 * n * m / (s * s))


def _grad_M(a, k):
    """Gradient of M = conv[k] with respect to a, for a with sum 1.
    d(conv[k])/da[i] = 2*a[k-i] when 0 <= k-i < n else 0."""
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def _gd_phase(a, deadline, base_lr, plateau_limit, best_ref):
    """Run projected gradient descent from a until deadline or plateau.

    best_ref is a mutable list [score, array]; updated in place when improved.
    Returns (improved_flag).
    """
    a = _normalize(a.copy())
    n = a.size
    obj = float(2.0 * n * np.max(_fft_conv(a)))  # sum(a)=1 here
    lr = base_lr
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        conv = _fft_conv(a)
        k = int(np.argmax(conv))
        M = float(conv[k])
        cur_obj = 2.0 * n * M
        if cur_obj < best_ref[0] - 1e-12:
            best_ref[0] = cur_obj
            best_ref[1] = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > plateau_limit:
            break

        gM = _grad_M(a, k)
        p = gM - gM.mean()              # tangent to simplex (sum zero)
        accepted = False
        for _ in range(16):
            a_try = a - lr * p
            a_try = np.clip(a_try, 0.0, None)
            st = a_try.sum()
            if st > 0.0:
                a_try = a_try / st
                obj_try = float(2.0 * n * np.max(_fft_conv(a_try)))
                if obj_try < cur_obj:
                    a = a_try
                    cur_obj = obj_try
                    lr *= 1.15           # increase step size when progress
                    accepted = True
                    if cur_obj < best_ref[0] - 1e-12:
                        best_ref[0] = cur_obj
                        best_ref[1] = a.copy()
                        stall = 0
                    break
            lr *= 0.5
        if not accepted:
            lr *= 0.5
            if lr < 1e-9:
                break
        it += 1
    return best_ref[0] < float("inf")


def _make_inits(n, rng, extra=None):
    """Build a diverse list of symmetry-breaking starting points (len n)."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    inits.append(np.ones(n))

    # linear ramps
    inits.append(np.concatenate([np.ones(n // 2), 0.2 * np.ones(n - n // 2)]))
    inits.append(1.0 - x)
    inits.append(x)

    # sigmoids at several centers / steepnesses
    for c, s in [(0.3, 30.0), (0.5, 30.0), (0.7, 30.0),
                 (0.4, 12.0), (0.6, 12.0), (0.5, 70.0)]:
        inits.append(1.0 / (1.0 + np.exp(s * (x - c))))

    # power / exponential shapes (favour one end)
    inits.append((1.0 - x) ** 3.0)
    inits.append((x) ** 3.0)
    inits.append(np.exp(-4.0 * x))
    inits.append(np.exp(-4.0 * (1.0 - x)))

    # gaussian bumps placed off-center
    for mu in [0.3, 0.5, 0.7]:
        inits.append(np.exp(-((x - mu) ** 2) / (2.0 * (0.12) ** 2)))

    # anti-palindromic: strong first half, weak mirrored second half
    ap = np.concatenate([np.ones(n // 2) * 1.0, np.ones(n - n // 2) * 0.15])
    inits.append(ap)
    # "blocky" alternation groups (large-smoothed low autocorrelation idea)
    blocks = np.concatenate([np.full(m, 1.0) if b % 2 == 0 else np.full(m, 0.2)
                             for b, m in enumerate(
                                 [n // 40 + (1 if i < n // 40 % 1 else 0) % 2
                                  for i in range(40)])])[:n]
    if blocks.size == n:
        inits.append(blocks)

    # random starts
    for _ in range(6):
        inits.append(0.5 + rng.random(n))
    for _ in range(4):
        r = rng.random(n)
        # smooth the random signal
        r = np.convolve(r, np.ones(16) / 16.0, mode="same")
        inits.append(r)

    if extra is not None:
        e = np.clip(np.asarray(extra, dtype=np.float64).ravel(), 0.0, None)
        if e.size == n and e.sum() > 0:
            inits.append(e)

    # normalize, dedupe by score cheaply later; return list
    return [_normalize(v) for v in inits]

    rng = np.random.default_rng(seed)
    start_t = time.monotonic()
    deadline = start_t + max(0.0, float(budget_s)) - 3.0

    # Determine length from the provided construction when available.
    n = None
    try:
        n = len(height_sequence_1)
    except Exception:
        n = None
    if n is None or n < 4:

    extra = None
    try:
        extra = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        extra = None
    if extra is not None and extra.size != n:
        extra = None

    best = [float("inf"), np.ones(n)]
    cur_best_val = _score(np.ones(n))
    best[0] = cur_best_val
    best[1] = np.ones(n) / n
    print("n =", n, " initial score =", cur_best_val, flush=True)

    inits = _make_inits(n, rng, extra)

    # Evaluate all starts, keep the promising ones ordered by score.
    scored = []
    for v in inits:
        s = _score(v)
        if np.isfinite(s):
            scored.append((s, v))
    scored.sort(key=lambda t: t[0])

    # Phase 1: gradient descent from each promising start (time-capped per start).
    per_start = min(60.0, max(15.0, (deadline - time.monotonic()) / max(1, len(scored))))
    for s0, v0 in scored:
        if time.monotonic() > deadline:
            break
        local_dead = min(deadline, time.monotonic() + per_start)
        _gd_phase(v0, local_dead, base_lr=1.0, plateau_limit=3000, best_ref=best)
        if best[0] < cur_best_val - 1e-9:
            cur_best_val = best[0]
            print("phase1 best score =", cur_best_val, flush=True)

    # Phase 2: perturb-restart on the global best to escape local minima.
    base_lr = 1.0
    sigma = 0.1
    while time.monotonic() < deadline:
        a0 = best[1] * (1.0 + sigma * rng.normal(0.0, 1.0, n))
        a0 = np.clip(a0, 0.0, None)
        local_dead = min(deadline, time.monotonic() + 25.0)
        before = best[0]
        _gd_phase(a0, local_dead, base_lr=base_lr, plateau_limit=4000, best_ref=best)

            cur_best_val = best[0]
            sigma = min(0.5, sigma * 1.3)
            print("phase2 best score =", cur_best_val, flush=True)
        else:
            sigma = max(0.02, sigma * 0.8)

    final = _normalize(best[1])
    print("FINAL score =", _score(final), " floor(time used)=",
          time.monotonic() - start_t, flush=True)
    return [float(x) for x in final]
