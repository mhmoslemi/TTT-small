"""
Minimize 2*n*max(convolve(p,p))/sum(p)^2 (scale-invariant, so we work with
sum(p)=1 and minimize the autocorrelation peak).

Algorithm:
  * For several sequence lengths n and several smooth non-uniform initializations
    (cosine, Gaussian, triangular, smoothed-random, and the supplied parent),
    run the iterative sparse-LP scheme:
        given f (sum 1, peak M), solve
            maximize sum(g)  s.t.  (f * g)[k] <= M  for all k,  g >= 0
      via a sparse high-S (HiGHS) LP, normalize g to sum 1, then take a
      line-search step  f <- (1-t) f + t g  accepting only when the FFT-computed
      peak decreases.
  * Polish the best sequence with small block multiplicative perturbations.
  * Return the best sequence found (verified against evaluate_sequence).

Peak evaluation uses an FFT (O(n log n)). The LP constraint matrix has a fixed
banded structure per n that is precomputed; only the data vector is refreshed
each iteration by indexing the current f.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _peak_fft(f):
    """Return max of the linear convolution f*f using a real FFT."""
    n = f.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    conv = np.fft.irfft(F * F, fft_size)
    return float(conv[:size].max())


def _build_structure(n):
    """Precompute (rows, cols, f_idx) describing the (f * g) <= M constraint rows.

    Row k corresponds to (f*g)[k] = sum_i f[k-i] * g[i] over valid i.
    """
    rows_all, cols_all, fidx_all = [], [], []
    for k in range(2 * n - 1):
        i0 = 0 if k < n - 1 else k - n + 1
        i1 = k if k < n else n - 1
        rows_all.append(np.full(i1 - i0 + 1, k, dtype=np.int32))
        cols_all.append(np.arange(i0, i1 + 1, dtype=np.int32))
        fidx_all.append(np.arange(k - i1, k - i0 + 1, dtype=np.int32))
    return (np.concatenate(rows_all), np.concatenate(cols_all),
            np.concatenate(fidx_all))


def _lp_direction(f, rows, cols, fidx, n, M):
    """Solve max sum(g) s.t. (f*g)[k] <= M, g>=0. Return normalized g or None."""
    data = f[fidx]
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.empty(2 * n - 1)
    b.fill(M)
    try:
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': 20.0, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def _blend_step(f, g, M):
    """Line-search blend of f toward g; return (new_f, improved)."""
    best_t = 0.0
    best_M = M
    for t in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05):
        cand = (1.0 - t) * f + t * g
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s < 1e-15:
            continue
        cand = cand / s
        Mc = _peak_fft(cand)
        if Mc < best_M - 1e-15:
            best_M = Mc
            best_t = t
    if best_t == 0.0:
        return f, False
    f = (1.0 - best_t) * f + best_t * g
    f = np.clip(f, 0.0, None)
    f = f / f.sum()
    return f, True


def _iterate_lp(f, n, structure, deadline):
    """Drive the LP iterations until no further improvement or time budget hit."""
    f = np.clip(np.asarray(f, dtype=np.float64), 0.0, None)
    if f.sum() < 1e-15:
        return f
    f = f / f.sum()
    rows, cols, fidx = structure
    M = _peak_fft(f)
    stall = 0
    while time.monotonic() < deadline:
        g = _lp_direction(f, rows, cols, fidx, n, M)
        if g is None:
            break
        newf, improved = _blend_step(f, g, M)
        if not improved:
            break
        f = newf
        M = _peak_fft(f)
        stall = 0
        if time.monotonic() > deadline:
            break
    return f


def _polish(f, n, deadline, rng):
    """Small block-multiplicative local search to escape flat regions."""
    f = np.clip(np.asarray(f, dtype=np.float64), 0.0, None)
    f = f / f.sum()
    best = _peak_fft(f)
    while time.monotonic() < deadline:
        width = max(1, n // 40)
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        cand = f.copy()
        cand[start:stop] *= np.exp(rng.normal(0.0, 0.06, size=stop - start))
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s < 1e-15:
            continue
        cand = cand / s
        Mc = _peak_fft(cand)
        if Mc < best - 1e-15:
            f = cand
            best = Mc
    return f


def _init_cosine(n, rng):
    x = np.linspace(0.0, np.pi, n)
    f = np.sin(x * 0.5) ** 2
    f = f + 1e-7 * rng.random(n)
    return f


def _init_gaussian(n, rng):
    x = np.linspace(-1.0, 1.0, n)
    sigma = 0.35 + 0.35 * rng.random()
    f = np.exp(-(x ** 2) / (2.0 * sigma * sigma))
    f = f + 1e-7 * rng.random(n)
    return f


def _init_triangular(n, rng):
    x = np.linspace(-1.0, 1.0, n)
    f = np.maximum(0.0, 1.0 - np.abs(x))
    f = f + 1e-7 * rng.random(n)
    return f


def _init_smooth_random(n, rng):
    f = rng.random(n)
    ks = max(3, n // 25)
    ker = np.exp(-(np.linspace(-3.0, 3.0, ks) ** 2))
    ker /= ker.sum()
    f = np.convolve(f, ker, mode='same')
    f = f + 1e-7 * rng.random(n)
    return f


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for the coefficient sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 6.0)

    best_seq = None
    best_score = np.inf

    n_values = [600, 1200, 2000, 3200, 4500]
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, None)
        if (parent.ndim == 1 and 8 <= parent.size <= 8000
                and np.isfinite(parent).all() and parent.sum() > 0):
            n_values = [int(parent.size)] + n_values
    except Exception:
        pass

    init_fns = (_init_cosine, _init_gaussian, _init_triangular, _init_smooth_random)

    for n in n_values:
        if time.monotonic() > deadline:
            break
        if not (8 <= n <= 8000):
            continue
        try:
            structure = _build_structure(n)
        except Exception:
            continue

        inits = [fn(n, rng) for fn in init_fns]
        try:
            parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, None)
            if parent.ndim == 1 and parent.size == n and parent.sum() > 0:
                inits.append(parent + 1e-7 * rng.random(n))
        except Exception:
            pass

        # split remaining time proportionally to the number of inits for this n
        per_init = max(8.0, (deadline - time.monotonic()) / max(1, len(inits)))
        for init_f in inits:
            if time.monotonic() > deadline:
                break
            sub_dead = min(deadline, time.monotonic() + per_init)
            f_opt = _iterate_lp(init_f, n, structure, sub_dead)
            # brief polish
            pol_dead = min(deadline, time.monotonic() + max(2.0, per_init * 0.25))
            f_opt = _polish(f_opt, n, pol_dead, rng)
            score = 2.0 * n * _peak_fft(f_opt)  # valid since sum(f_opt)=1
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = f_opt.copy()
                print("n=%d score=%.6f peak=%.8f" % (n, score, _peak_fft(f_opt)))

    # final polish with whatever time is left
    if best_seq is not None and time.monotonic() < deadline:
        n = best_seq.size
        best_seq = _polish(best_seq, n, deadline, rng)
        best_score = 2.0 * n * _peak_fft(best_seq)

    if best_seq is not None:
        cand = [float(x) for x in best_seq]
        act = float(evaluate_sequence(cand))
        if np.isfinite(act):
            print("final verified score=%.6f len=%d" % (act, len(cand)))
            return cand

    # safe fallback (uniform -> score 2)
    return [1.0] * 512
