Minimizes 2n*max(f*f)/(sum f)^2 using FFT-based subgradient descent with
adaptive softness, exponential moving average, multiple initializations,
and periodic LP-based direction refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csc_matrix


def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix C where C[k,l]=f[k-l]."""
    n = len(f)
    col_idx = np.repeat(np.arange(n), n)
    row_blocks = np.arange(n * n).reshape(n, n) + np.arange(n)[:, None]
    row_idx = row_blocks.flatten().astype(np.int64)
    vals = np.tile(f, n)
    return csc_matrix((vals, (row_idx, col_idx.astype(np.int64))), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0. Return normalized g."""
    try:
        C = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        result = linprog(c_obj, A_ub=C, b_ub=b, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            g = np.maximum(result.x, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return np.ones(n) / n


def run_subgradient(f, n, rng, max_time, softness_base=30.0, use_momentum=True):
    """FFT-based subgradient descent with momentum and adaptive softness."""
    init_time = time.monotonic()
    N = 1 << (3 * n).bit_length()

    best = f.copy()
    best_val = np.float64(np.convolve(f, f).max())

    lr_base = 0.06
    iteration = 0
    no_improve = 0

    # Momentum buffer
    if use_momentum:
        mom = np.zeros(n)
        beta = 0.9

    while time.monotonic() - init_time < max_time:
        iteration += 1

        # FFT of f (reused for autocorrelation and subgradient)
        f_fft = np.fft.rfft(f, N)

        # Autocorrelation via FFT
        h = np.fft.irfft(f_fft * np.conj(f_fft))[:2 * n - 1]
        M = h.max()

        if M < best_val - 1e-12:
            best_val = M
            best = f.copy()
            no_improve = 0

        no_improve += 1

        # Escape mechanism: if stuck, perturb
        if no_improve > 400:
            mask = rng.random(n) < 0.12
            cnt = int(mask.sum())
            if cnt > 0:
                f[mask] *= np.exp(rng.normal(0, 0.4, cnt))
            f = np.maximum(f, 0)
            s = f.sum()
            if s > 1e-10:
                f /= s
            no_improve = 0
            continue

        # Soft active-set weights
        s_soft = softness_base + 0.015 * iteration
        weights = np.exp(s_soft * (h - M) / max(M, 1e-15))
        wsum = weights.sum()
        if wsum > 1e-30:
            weights /= wsum
        else:
            weights[:] = 0.0
            weights[int(np.argmax(h))] = 1.0

        # Subgradient via cross-correlation FFT
        V_fft = np.fft.rfft(weights, N)
        sg = np.fft.irfft(V_fft * np.conj(f_fft))[:n] * 2.0
        sg -= sg.mean()

        # Momentum
        if use_momentum:
            mom = beta * mom + (1 - beta) * sg
            step_sg = mom
        else:
            step_sg = sg

        # Adaptive learning rate
        lr = lr_base / np.sqrt(1.0 + 0.003 * iteration)
        f -= lr * step_sg
        np.maximum(f, 0.0, out=f)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f[:] = 1.0 / n

    return best, best_val
