   This is a very simple subgradient! It's just a shifted/reversed copy of a.

import numpy as np

from scipy.fft import fft, ifft
from scipy.optimize import linprog


def compute_convolution(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fa = fft(a, n=size)
    c = np.real(ifft(fa * fa))
    return np.maximum(c, 0.0)
