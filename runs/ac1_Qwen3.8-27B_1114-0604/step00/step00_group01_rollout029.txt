    else:

    beta = 0.9
    total_iters = 0

        T = 0.5 * (0.5 ** (phase // 4)) + 0.01

            size = 2 * n - 1
            fft_size = 1 << (size - 1).bit_length()
            A = np.fft.rfft(a, fft_size)
            C = np.fft.irfft(A * A, fft_size)[:size]

            C_max = np.max(C)
            W = np.exp((C - C_max) / T)
            W_sum = np.sum(W)
            if W_sum < 1e-300:

                W[np.argmax(C)] = 1.0

            Wf = np.fft.rfft(W, conv_fft_size)

            G = np.fft.irfft(Wf * ARf, conv_fft_size)
            g = G[n-1:n-1+n].copy()

            momentum = beta * momentum - eta * g
            a = np.maximum(0.0, a + momentum)

            if s < 1e-10:
                a = np.full(n, target_sum / n)

                a = a * target_sum / s
            
            total_iters += 1
        
        # Periodic evaluation
        score = float(evaluate_sequence(a.tolist()))
        if score < best_score:
            best = a.copy()
            best_score = score
