```python
def lp_direction(a, n, M, max_constraints=200):
    """Find the LP direction: max sum(g) s.t. a*g <= M, g >= 0."""
    # Only use the top max_constraints constraints
    c = np.convolve(a, a)
    # Find the indices where c is largest
    top_idx = np.argsort(c)[-max_constraints:]
    
    # Build constraint matrix for these rows
    # A[i, j] = a[k-i-j] where k = top_idx[i], valid if 0 <= k-i-j < n
    # Actually, (a*g)_k = sum_j a_{k-j} * g_j
    # So A[i, j] = a_{top_idx[i] - j} if 0 <= top_idx[i] - j < n else 0
    
    A = np.zeros((len(top_idx), n))
    for i, k in enumerate(top_idx):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[i, j] = a[k-j]
    
    # Solve: max 1^T g s.t. A @ g <= M, g >= 0
    from scipy.optimize import linprog
    c_obj = -np.ones(n)  # minimize -sum(g)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(len(top_idx), M), 
                  bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
"""
Hybrid LP-based local search with constraint generation for minimizing
the peak-to-average ratio of a step function's autocorrelation.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_sparse_constraint_rows(a, k_indices, n):
    """Build sparse constraint matrix rows for given convolution indices."""
    m = len(k_indices)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        ii = k - js
        valid = (ii >= 0) & (ii < n)
        rows.extend((np.full(np.sum(valid), idx)).tolist())
        cols.extend(js[valid].tolist())
        vals.extend(a[ii[valid]].tolist())
    if not rows:
        return csr_matrix((m, n))
    return csr_matrix((vals, (rows, cols)), shape=(m, n))


def solve_lp(a, n, M, n_constraints=80):
    """Solve LP: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    c = np.convolve(a, a)
    # Select top constraints
    k = min(n_constraints, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    # Sort by value descending (largest first)
    top_idx = top_idx[np.argsort(-c[top_idx])]
    
    A = build_sparse_constraint_rows(a, top_idx, n)
    
    c_obj = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
    
    if res.success and res.x is not None:
        return res.x
    return None


def block_perturb(a, rng, n):
    """Apply a random block perturbation."""
    candidate = a.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 10)))
    start = int(rng.integers(0, max(1, n - width)))
    candidate[start:start + width] *= float(np.exp(rng.normal(0.0, 0.15)))
    candidate = np.clip(candidate, 0.0, None)
    s = candidate.sum()
    if s > 1e-10:
        candidate /= s
    return candidate


def random_perturb(a, rng, n, frac=0.05, sigma=0.2):
    """Apply a random perturbation to a fraction of elements."""
    candidate = a.copy()
    mask = rng.random(n) < frac
    if mask.sum() > 0:
        candidate[mask] *= np.exp(rng.normal(0.0, sigma, mask.sum()))
    candidate = np.clip(candidate, 0.0, None)
    s = candidate.sum()
    if s > 1e-10:
        candidate /= s
    return candidate
```

```python
def build_sparse_constraints(a, k_indices, n):
    """Build sparse matrix for LP constraints: row k has a[k-j] at column j."""

        cols_list.append(js)
        vals_list.append(a[ii])
```

```python
                        b[q:q + length].copy(), b[p:p + length].copy()
                a = b
                tot = a.sum()
                if tot > 1e-15:
                    a /= tot
            else:
                # Blend with random
                r = rng.random(n)
                r /= r.sum()
                a = 0.7 * a + 0.3 * r
                a = np.clip(a, 0.0, None)
                tot = a.sum()
                if tot > 1e-15:
                    a /= tot
                else:
                    a = np.ones(n) / n

            no_improve = 0
            patience = min(80, int(patience * 1.3) + 1)

    print(f"Phase 1 done: LPs={total_lp}, evals={total_eval}, "
          f"best={best_score:.6f}, time={time.time()-t0:.0f}s", flush=True)

    # --- Phase 2: Fine-tuning with block perturbations ---
    t2 = time.time()
    pcount = 0
    best_perturb_score = best_score

    while time.time() < deadline:
        # Alternate between different perturbation strengths
        phase = pcount % 3
        if phase == 0:
            a = block_perturb(best, rng, n, sigma=0.08)
        elif phase == 1:
            a = block_perturb(best, rng, n, sigma=0.15)
        else:
            a = multi_block_perturb(best, rng, n, n_blocks=2, sigma=0.1)

        sc = evaluate_sequence(a.tolist())
        pcount += 1
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()
            if pcount % 1000 == 0:
                print(f"  Phase2 #{pcount}: score={sc:.6f} "
                      f"time={time.time()-t2:.0f}s", flush=True)

    print(f"Phase 2 done: perturbations={pcount}, "
          f"best={best_score:.6f}, time={time.time()-t0:.0f}s", flush=True)
    print(f"FINAL SCORE: {best_score:.6f}", flush=True)

    return [float(x) for x in best]
```
