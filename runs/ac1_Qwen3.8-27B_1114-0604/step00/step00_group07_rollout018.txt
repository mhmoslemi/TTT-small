```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def build_constraint_matrix(a):
    """Build the (2n-1) x n constraint matrix for the LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_direction(a, M):
    """Solve the LP to find the best direction b."""
    n = len(a)
    A = build_constraint_matrix(a)
    # Maximize sum(b) s.t. A @ b <= M, b >= 0
    # linprog minimizes, so negate the objective
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.status == 0:
        return res.x
    return None
```

```python
LP-based iterative optimization to minimize max-autocorrelation score.
Solves LPs to find descent directions that reduce the peak of the
autocorrelation function, iterating until convergence.

    """Full autocorrelation via FFT: equivalent to np.convolve(a, a)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

                print(f"  {tag} it={it} score={score:.6f} M={M:.4f}")

        # Convergence
        if prev_M - M < 1e-9 * max(1.0, M):
            stagnant += 1
        else:
            stagnant = 0
        prev_M = M
        if stagnant > 15:
            break

        # Build and solve LP
        A = build_A(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status != 0:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b < 1e-12:
            break
        b = b * (sum_a / sum_b)

        # Best step
        a_new = a
        M_new = M
        for t in step_options:
            cand = (1.0 - t) * a + t * b
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 0.01:
                continue
            M_c = fft_autocorr(cand).max()
            if M_c < M_new:
                M_new = M_c
                a_new = cand

        a = a_new

    return best_a, best_score
```
