def compute_grad(a, weights):
    """Compute grad[i] = 2 * sum_j a[j] * weights[j+i] for i=0,...,n-1."""
    n = len(a)
    # Need FFT size >= n + (2n-1) - 1 = 3n - 2
    N = 1 << (3 * n - 2 - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(weights, N)
    # Cross-correlation: ifft(fft(a) * conj(fft(w))) gives sum_j a[j] * w[j+i]
    # Wait, I need to be careful. The standard formula:
    # np.correlate(a, w, 'full')[k] = sum_j conj(a[j]) * w[j+k-n_a] ... it's confusing.
    # Let me use the property: 
    # ifft(FFT(x) * conj(FFT(y)))[n] = sum_k x[k] * conj(y[k-n]) (circular)
    # For linear: ifft(FFT(x, N) * conj(FFT(y, N)))[n] = sum_k x[k] * y[k-n] (for N large enough)
    # 
    # I want R[i] = sum_j a[j] * w[j+i]
    # = sum_j a[j] * w[(j+i)]
    # Let k = j: R[i] = sum_k a[k] * w[k+i]
    # 
    # Using the property: ifft(FFT(a) * FFT(w))[n] = sum_k a[k] * w[n-k] (circular convolution)
    # That's convolution, not correlation.
    #
    # ifft(FFT(a) * conj(FFT(w)))[n] = sum_k a[k] * conj(w[k-n]) (circular cross-correlation)
    # For real w: = sum_k a[k] * w[k-n]
    #
    # I want sum_k a[k] * w[k+i] = sum_k a[k] * w[(k-n) - (-n-i)]

    #
    # R[i] = sum_k a[k] * w[k+i]
    # Let a_rev[k] = a[n-1-k] (reverse a)
    # Then R[i] = sum_k a[n-1-k'] * w[n-1-k'+i] where k' = n-1-k
    # = sum_{k'} a[n-1-k'] * w[i + n-1-k']
    # Let m = i + n - 1: R[i] = sum_{k'} a[n-1-k'] * w[m-k']
    # = (a_rev * w)[m] = convolution of a_rev and w at position m
    #
    # So R[i] = (a_rev * w)[i + n - 1]

    a_rev = a[::-1]

    # conv[m] = sum_k a_rev[k] * w[m-k] = (a_rev * w)[m]
    # R[i] = conv[i + n - 1]
    
    R = conv[n-1 : n-1+n]
    return 2.0 * R

import numpy as np
import time

def self_conv(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]

def compute_grad_fft(a, weights):
    """Compute grad[i] = 2 * sum_j a[j] * weights[j+i] for i=0,...,n-1 using FFT."""
    n = len(a)
    N = 1 << (3 * n - 3).bit_length()  # Need N >= n + (2n-1) - 1 = 3n-2
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(weights, N)
    conv = np.fft.irfft(fa_rev * fw, N)
    R = conv[n-1 : n-1+n]
    return 2.0 * R

def soft_max_tau(c, tau):
    """Soft max with temperature tau."""
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    return c_max + tau * np.log(exp_c.sum()), exp_c

def optimize_sequence(a0, tau=1.0, eta0=0.1, max_iters=10000, tol=1e-8):
    """Optimize a non-negative sequence to minimize max self-convolution."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    S = a.sum()
    if S <= 0:
        a = np.ones(n) / n
        S = 1.0
    a = a / S  # Normalize to sum = 1
    
    best_a = a.copy()
    best_obj = 2.0 * n * self_conv(a).max()
    eta = eta0
    
    for it in range(max_iters):
        c = self_conv(a)
        c_max_val, exp_c = soft_max_tau(c, tau)
        soft_max_val = c_max_val + tau * np.log(exp_c.sum())
        
        obj = 2.0 * n * soft_max_val
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if obj - best_obj < tol:
            break
        
        # Compute gradient
        grad = compute_grad_fft(a, exp_c)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        S_new = a_new.sum()
        if S_new > 1e-10:
            a_new = a_new / S_new
        
        # Check if improvement
        c_new = self_conv(a_new)
        obj_new = 2.0 * n * c_new.max()
        
        if obj_new < obj:
            a = a_new
            # Increase eta slightly
            eta = min(eta * 1.05, 1.0)
        else:
            # Decrease eta
            eta = max(eta * 0.5, 1e-6)
    
    return best_a, best_obj
