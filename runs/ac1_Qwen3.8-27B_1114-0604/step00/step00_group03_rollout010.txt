import numpy as np
from scipy.optimize import linprog
import time

def solve_lp(a, M):
    """Solve the LP: max sum(b) s.t. convolve(a,b) <= M, b >= 0"""
    n = len(a)
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b), so minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: for each k = 0, ..., 2n-2:
    #   sum_i a[i] * b[k-i] <= M
    #   where b[j] = 0 if j < 0 or j >= n
    #   So the constraint is: sum_{i: 0<=k-i<n} a[i] * b[k-i] <= M
    #   Letting j = k-i: sum_{j: max(0,k-n+1)<=j<=min(n-1,k)} a[k-j] * b[j] <= M
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A_ub[k, j] = a[k - j]
    
    b_ub = np.full(2*n-1, M)
    
    # b >= 0 is handled by bounds
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:

    return None

import time

def solve_lp_step(a, M):
    """Solve: max sum(b) s.t. convolve(a,b) <= M, b >= 0"""
    n = len(a)
    c = -np.ones(n)  # minimize -sum(b)
    
    # Build constraint matrix A where A @ b = convolve(a, b)
    # A has shape (2n-1, n)
    A = np.zeros((2*n - 1, n))
