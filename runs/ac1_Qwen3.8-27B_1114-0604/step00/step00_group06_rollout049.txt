import numpy as np

def clean_input(x):
    if x is None:
        return None
    a = np.asarray(x, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    if not np.all(np.isfinite(a)):
        return None
    if a.sum() <= 0.0:
        return None
    return a


def normalize_to_sum(a, target=None):
    a = np.asarray(a, dtype=np.float64)
    if target is None:
        target = float(a.size)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(a.shape, target / max(1, a.size), dtype=np.float64)
    return a * (target / s)


def convolution_with_fa(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    N = 1 << max(0, (L - 1).bit_length())
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    if c.size:
        mn = float(c.min())
        if mn < 0.0:
            c = np.maximum(c, 0.0)
    return c, fa, N


def fast_score(a):
    a = clean_input(a)
    if a is None:
        return float("inf")
    n = a.size
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    c, _, _ = convolution_with_fa(a)
    M = float(c.max())
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M / (s * s))


def compute_H(a, fa, N, w):
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    wr = w[::-1]
    h = np.fft.irfft(fa * np.fft.rfft(wr, N), N)
    h = h[n - 1:2 * n - 1][::-1]
    if h.size:
        mn = float(h.min())
        if mn < 0.0:
            h = np.maximum(h, 0.0)
    return h


def make_candidate_from_H(a, H, eta):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if H.size != n or n == 0:
        return a.copy()
    hmax = float(H.max())
    if not np.isfinite(hmax) or hmax <= 1e-300:
        return a.copy()
    u = H / hmax
    s = float(a.sum())
    if s <= 0.0:
        return a.copy()
    mean_u = float(np.dot(a, u) / s)
    arg = -float(eta) * (u - mean_u)
    arg = np.clip(arg, -50.0, 50.0)
    cand = a * np.exp(arg)
    maxa = float(a.max())
    if maxa > 0.0:
        cand = np.maximum(cand, 1e-12 * maxa)
    cand = np.minimum(cand, 1000.0)
    s2 = float(cand.sum())
    if s2 <= 0.0:
        return a.copy()
    cand = cand * (s / s2)
    cand = np.clip(cand, 0.0, 1000.0)
    s3 = float(cand.sum())
    if s3 > 0.0:
        cand = cand * (s / s3)
    return cand


def antipal_project(a, target=1.0):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a.copy()
    b = target + 0.5 * (a - a[::-1])
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s <= 0.0:
        b = np.full(n, target, dtype=np.float64)
    else:
        b = b * (float(n) / s)
    b = np.clip(b, 0.0, 1000.0)
    return b


def random_perturb(a, sigma, rng, force_antipal=False):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a.copy()
    b = a * np.exp(rng.normal(0.0, sigma, n))
    b = np.clip(b, 0.0, 1000.0)
    if force_antipal:
        b = antipal_project(b)
    b = normalize_to_sum(b, float(n))
    return b


def make_antipal_noise(n, rng, low_frac=0.0, high_frac=1.0, band=False):
    n = int(max(4, n))
    if band and n > 32:
        spec = np.fft.rfft(rng.normal(size=n))
        nf = spec.size
        keep = np.ones(nf, dtype=bool)
        if nf > 3:
            low = max(1, int(low_frac * nf))
            high = min(nf - 1, int(high_frac * nf))
            idx = np.arange(nf)
            keep = (idx >= low) & (idx <= high)
            keep[0] = False
            if nf > 1:
                keep[-1] = False
        y = np.fft.irfft(spec * keep, n)
    else:
        y = np.fft.irfft(np.fft.rfft(rng.normal(size=n)), n)
    y = y - y[::-1]
    y = np.asarray(y, dtype=np.float64)
    y -= y.mean()
    std = float(y.std())
    if std <= 1e-300:
        y = rng.normal(size=n)
        y -= y.mean()
        std = float(y.std())
        if std <= 1e-300:
            return np.zeros(n, dtype=np.float64)
    return y / std


def make_antipal_binary(n, rng):
    n = int(max(4, n))
    m = n // 2
    z = np.ones(m, dtype=np.float64)
    mask = rng.random(m) < 0.5
    z[mask] = -1.0
    if n % 2 == 0:
        y = np.concatenate((z, -z[::-1]))
    else:
        y = np.concatenate((z, [0.0], -z[::-1]))
    y -= y.mean()
    std = float(y.std())
    if std <= 1e-300:
        return np.zeros(n, dtype=np.float64)
    return y / std


def init_uniform(n):
    return np.ones(int(max(4, n)), dtype=np.float64)


def make_initial(kind, n, rng, provided=None):
    if n is None:
        if provided is not None:
            n = provided.size
        else:
            n = 2048
    n = int(max(4, n))

    if kind == "provided":
        if provided is None:
            a = init_uniform(n)
        else:
            a = np.asarray(provided, dtype=np.float64).copy()
    elif kind == "provided_perturb":
        if provided is None:
            a = init_uniform(n)
        else:
            a = normalize_to_sum(np.asarray(provided, dtype=np.float64), float(provided.size))
        noise = make_antipal_noise(a.size, rng, low_frac=0.05, high_frac=0.95, band=True)
        a = 0.85 * a + 0.15 * (1.0 + 0.5 * noise)
    elif kind == "uniform":
        a = init_uniform(n)
    elif kind == "antipal_binary":
        mod = make_antipal_binary(n, rng)
        a = 1.0 + 0.70 * mod
    elif kind == "antipal_noise":
        mod = make_antipal_noise(n, rng, low_frac=0.05, high_frac=0.95, band=True)
        a = 1.0 + 0.60 * mod
    elif kind == "ramp":
        c = 0.5 * (n - 1)
        z = np.arange(n, dtype=np.float64) - c
        z -= z.mean()
        std = float(z.std())
        if std > 0.0:
            z = z / std
        a = 1.0 + 0.55 * z
    elif kind == "sine":
        i = np.arange(n, dtype=np.float64)
        mod = np.zeros(n, dtype=np.float64)
        lo = max(1, int(0.15 * n // 2))
        hi = max(lo + 1, int(0.85 * n // 2))
        for _ in range(12):
            m = int(rng.integers(lo, hi))
            mod += rng.normal(0.0, 1.0) * np.sin((2.0 * m * np.pi) * (i + 0.5) / float(n))
        mod -= mod.mean()
        std = float(mod.std())
        if std > 0.0:
            mod = mod / std
        a = 1.0 + 0.55 * mod
    else:
        a = init_uniform(n)

    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 0.0:
        a = init_uniform(a.size)
    a = a * (float(a.size) / float(a.sum()))
    a = np.clip(a, 0.0, 1000.0)
    return a


def weights_cap(c, frac):
    peak = float(c.max())
    if not np.isfinite(peak) or peak <= 0.0:
        return None
    thr = peak * max(1e-8, 1.0 - frac)
    w = c - thr
    w = np.maximum(w, 0.0)
    sw = float(w.sum())
    if sw <= 1e-300:
        rel = c / peak
        w = rel * rel
        sw = float(w.sum())
        if sw <= 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            sw = 1.0
    return w / sw


def weights_top(c, frac):
    sz = c.size
    m = max(1, min(sz - 1, int(sz * max(1e-5, frac))))
    idx = sz - 1 - m
    thr = float(np.partition(c, idx)[idx])
    w = (c >= thr).astype(np.float64)
    sw = float(w.sum())
    if sw <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        sw = 1.0
    return w / sw


def weights_soft(c, p):
    peak = float(c.max())
    if not np.isfinite(peak) or peak <= 0.0:
        return None
    rel = c / peak
    w = np.power(rel, float(p))
    sw = float(w.sum())
    if sw <= 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        sw = 1.0
    return w / sw


def local_search(initial, deadline, rng, force_antipal=False,
                 frac_start=0.25, frac_end=0.008,
                 eta_start=0.12, verbose_id=""):

    if a is None:
        a = init_uniform(1024)
    n = a.size
    if n < 4:
        a = init_uniform(1024)
        n = a.size
    a = np.clip(a, 0.0, 1000.0)
    a = normalize_to_sum(a, float(n))
    if force_antipal:
        a = antipal_project(a)
        a = normalize_to_sum(a, float(n))

    c, fa, N = convolution_with_fa(a)
    cur_score = float(2.0 * n * c.max() / (float(a.sum()) ** 2))
    best_a = a.copy()
    best_score = cur_score

    start = time.monotonic()
    last_print = start
    last_print_score = best_score + 1e-3
    it = 0
    stagnant = 0
    eta = float(eta_start)

    while time.monotonic() < deadline:
        it += 1
        peak = float(c.max())
        if not np.isfinite(peak) or peak <= 0.0:
            break

        frac = frac_end + (frac_start - frac_end) * math.exp(-float(it) / 3500.0)
        r = float(rng.random())
        if r < 0.45:
            w = weights_cap(c, frac)
        elif r < 0.80:
            w = weights_top(c, max(0.003, 0.5 * frac))
        else:
            p = min(120.0, 1.0 / max(0.01, frac))
            w = weights_soft(c, p)

        if w is None:
            break

        H = compute_H(a, fa, N, w)
        if H.size != n or float(H.max()) <= 1e-300:
            stagnant += 1
            if stagnant > 200:
                a = random_perturb(a, 0.08, rng, force_antipal)
                c, fa, N = convolution_with_fa(a)
                cur_score = float(2.0 * n * c.max() / (float(a.sum()) ** 2))
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                stagnant = 0
                eta = float(eta_start)
            continue

        etas = [float(eta)]
        if eta > 0.03:
            etas.append(max(0.015, 0.55 * eta))

            etas.append(min(2.0, 1.8 * eta))

        best_cand = None
        best_c = None
        best_fa = None
        best_local = cur_score

        for e in etas:
            cand = make_candidate_from_H(a, H, e)
            if force_antipal:
                cand = antipal_project(cand)
                cand = normalize_to_sum(cand, float(n))
            c2, fa2, _ = convolution_with_fa(cand)
            s2 = float(cand.sum())
            if s2 <= 0.0:
                continue
            sc = float(2.0 * n * c2.max() / (s2 * s2))
            if sc < best_local:
                best_local = sc
                best_cand = cand
                best_c = c2
                best_fa = fa2

        improved = False
        if best_cand is not None:
            if best_local < cur_score * (1.0 - 1e-12):
                a = best_cand
                c = best_c
                fa = best_fa
                cur_score = best_local
                improved = True
                stagnant = 0
                eta = min(1.5, eta * 1.04)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
            elif cur_score > 0.0 and best_local < cur_score * (1.0 + 0.02 * math.exp(-float(it) / 1200.0)):
                if rng.random() < 0.08:
                    a = best_cand
                    c = best_c
                    fa = best_fa
                    cur_score = best_local
                    improved = True
                    stagnant = 0
                    eta = min(1.5, eta * 1.02)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        if not improved:
            stagnant += 1
            eta = max(0.02, eta * 0.92)
            if stagnant > 220:
                sigma = 0.06 + 0.12 * min(1.0, float(stagnant) / 1200.0)
                a = random_perturb(a, sigma, rng, force_antipal)
                c, fa, N = convolution_with_fa(a)
                cur_score = float(2.0 * n * c.max() / (float(a.sum()) ** 2))
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                stagnant = 0
                eta = float(eta_start)

        now = time.monotonic()

            print("local %s t=%6.1f best=%0.6f cur=%0.6f n=%d" %
                  (verbose_id, now - start, best_score, cur_score, n))
            last_print = now
            last_print_score = best_score

    return best_a, best_score


def build_conv_matrix(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_direction(a, M):
    if not _HAS_LINPROG:
        return None
    n = a.size
    if n < 16 or n > 768:
        return None
    if not np.isfinite(M) or M <= 0.0:
        return None
    try:
        A = build_conv_matrix(a)
        cvec = -np.ones(n, dtype=np.float64)
        bvec = np.full(2 * n - 1, float(M), dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds,
                      method="highs", options={"disp": False})
        if res.success and res.x is not None:
            b = np.asarray(res.x, dtype=np.float64)
            b = np.clip(b, 0.0, 1000.0)
            if float(b.sum()) > 0.0:
                return b
    except Exception:
        return None
    return None


def lp_search(initial, deadline, rng, verbose_id=""):
    a = clean_input(initial)
    if a is None:
        a = make_initial("antipal_binary", 512, rng)
    n = a.size
    if n < 16 or n > 768:
        a = make_initial("antipal_binary", 512, rng)
        n = a.size
    a = normalize_to_sum(a, float(n))

    c, _, _ = convolution_with_fa(a)
    cur_score = float(2.0 * n * c.max() / (float(a.sum()) ** 2))
    best_a = a.copy()
    best_score = cur_score

    start = time.monotonic()
    lp_end = min(deadline, start + max(60.0, 0.45 * max(0.0, deadline - start)))
    no_improve = 0
    it = 0

    while time.monotonic() < lp_end:
        it += 1
        c, fa, N = convolution_with_fa(a)
        M = float(c.max())
        b = lp_direction(a, M)
        if b is None:
            break
        old_sum = float(a.sum())
        b = b * (old_sum / float(b.sum()))

        improved = False
        for t in (0.05, 0.10, 0.20, 0.35):
            cand = np.maximum((1.0 - t) * a + t * b, 0.0)
            cand = normalize_to_sum(cand, float(n))
            sc = fast_score(cand)
            if not np.isfinite(sc):
                continue
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
            if sc < cur_score:
                a = cand
                cur_score = sc
                improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                break

    refined, refined_score = local_search(a, deadline, rng, force_antipal=False, verbose_id=verbose_id + "-refine")
    if refined_score < best_score:
        best_a = refined
        best_score = refined_score
    return best_a, best_score


def update_best_with_sequence(best_a, best_score, seq):
    a = clean_input(seq)
    if a is None:
        return best_a, best_score
    s = fast_score(a)
    if np.isfinite(s) and s < best_score:
        return a.copy(), s
    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 35.0)

    provided = None
    try:
        provided = clean_input(height_sequence_1)
    except Exception:
        provided = None

    # Guaranteed sub-2 fallback, then search for much better values.
    xfb = 1.0 + 1.0 / math.sqrt(3.0)
    fallback = np.array([1.0, xfb, 1.0, xfb, 1.0, xfb], dtype=np.float64)
    best_a, best_score = update_best_with_sequence(None, float("inf"), fallback)
    if provided is not None:
        best_a, best_score = update_best_with_sequence(best_a, best_score, provided)

    if deadline <= start + 1.0:
        out = clean_input(best_a)
        if out is None:
            out = fallback.copy()
        return [float(v) for v in out]

    specs = []
    if provided is not None:
        p = provided.size
        if 16 <= p <= 12000:
            specs.append(("provided", p, False, 520.0))
            specs.append(("provided_perturb", p, False, 330.0))
    specs.append(("antipal_binary", 4096, True, 620.0))
    specs.append(("antipal_noise", 8192, False, 620.0))
    specs.append(("ramp", 4096, False, 360.0))
    specs.append(("antipal_binary", 2048, False, 330.0))
    specs.append(("sine", 4096, False, 260.0))
    specs.append(("lp", 512, False, 420.0))

    total_w = sum(s[4] for s in specs)
    for idx, (kind, n_sec, force_antipal, weight) in enumerate(specs):
        if time.monotonic() >= deadline or best_score < 1.02:
            break
        remaining = max(0.0, deadline - time.monotonic())
        alloc = weight / max(1.0, total_w) * remaining
        alloc = max(45.0, min(780.0, alloc))
        if time.monotonic() + alloc > deadline:
            alloc = max(30.0, deadline - time.monotonic())
        if alloc < 30.0:
            break

        if kind == "lp":
            n_use = 512
            init_seq = make_initial("antipal_binary", n_use, rng)
            a, sc = lp_search(init_seq, min(deadline, time.monotonic() + alloc), rng, verbose_id="lp")
        else:
            n_use = n_sec
            init_seq = make_initial(kind, int(n_use), rng, provided)
            a, sc = local_search(
                init_seq,
                min(deadline, time.monotonic() + alloc),
                rng,
                force_antipal=bool(force_antipal),
                verbose_id=kind + str(idx)
            )

        best_a, best_score = update_best_with_sequence(best_a, best_score, a)
        print("after spec %s score=%0.6f len=%d elapsed=%0.1f" %
              (kind, best_score, best_a.size if best_a is not None else 0, time.monotonic() - start))

    out = clean_input(best_a)
    if out is None:
        out = fallback.copy()
    print("final score estimate=%0.6f len=%d elapsed=%0.1f" %
          (fast_score(out), out.size, time.monotonic() - start))
    return [float(v) for v in out]
