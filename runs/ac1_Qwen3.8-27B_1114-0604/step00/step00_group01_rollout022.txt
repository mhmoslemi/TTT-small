2. The sequence should be "spread out" (no single element too large)

"""
Multi-strategy local search to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based fast evaluation with block perturbations, gradient-guided
peak reduction, symmetrization, and simulated annealing.
Tries multiple structured initializations (windows, random, chirp-like).
"""
import numpy as np
import time


def fft_autocorr(a):
    """Compute linear self-convolution a*a using FFT. Result has length 2n-1."""
    n = len(a)
    N = 1
    N <<= (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(result, 0.0)


def fast_score(a):
    """Compute 2*n*max_b/S^2 quickly using FFT."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return float('inf')
    b = fft_autocorr(a)
    return 2.0 * n * b.max() / (S * S)


def make_gaussian(n, sigma_frac):
    """Gaussian window."""
    x = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((x - (n - 1) / 2.0) / max(1.0, n * sigma_frac)) ** 2)


def make_hann(n):
    """Hann window."""
    x = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))


def make_flat_top(n):
    """Flat-top window."""
    x = np.arange(n, dtype=np.float64)
    return (0.2156 + 0.4160 * np.cos(2 * np.pi * x / (n - 1)) +
            0.2780 * np.cos(4 * np.pi * x / (n - 1)) +
            0.0840 * np.cos(6 * np.pi * x / (n - 1)) +
            0.0087 * np.cos(8 * np.pi * x / (n - 1)))


def init_search(n, rng, provided=None):
    """Generate multiple initializations and return the best."""
    candidates = []
    
    # Structured windows
    for sf in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        candidates.append(make_gaussian(n, sf))
    candidates.append(make_hann(n))
    candidates.append(make_flat_top(n))
    candidates.append(np.ones(n))
    
    # Tapered random
    for sf in [0.3, 0.5, 0.8]:
        g = make_gaussian(n, sf)
        candidates.append(g * np.abs(rng.normal(1.0, 0.5, n)))
    
    # Cosine-modified
    x = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3, 5, 8, 13]:
        for amp in [0.2, 0.4, 0.6]:
            mod = 1.0 + amp * np.cos(2 * np.pi * freq * x / n)
            candidates.append(np.maximum(mod, 0.01))
    
    # Pure random
    candidates.append(np.abs(rng.normal(1.0, 0.5, n)))
    candidates.append(rng.uniform(0.1, 1.0, n))
    
    # Chirp-like (absolute value of quadratic phase, offset for non-negativity)
    for c in [1.0, 2.0, 5.0, 10.0]:
        chirp = np.cos(2 * np.pi * c * x * x / (n * n))
        candidates.append(np.abs(chirp) + 0.3)
    
    # Include provided sequence
    if provided is not None:
        candidates.append(provided)
    
    best = None
    best_s = float('inf')
    for c in candidates:
        c = np.clip(c, 0.0, 1000.0)
        if c.sum() < 0.01:
            continue
        s = fast_score(c)
        if s < best_s:
            best_s = s
            best = c.copy()
    
    return best, best_s


def do_block_perturb(base, rng, n):
    """Block multiplicative perturbation."""
    cand = base.copy()
    min_w = max(1, n // 500)
    max_w = max(min_w + 2, n // 4)
    width = int(rng.integers(min_w, max_w))
    start = int(rng.integers(0, n - width))
    sigma = float(rng.uniform(0.05, 0.35))
    cand[start:start + width] *= np.exp(rng.normal(0.0, sigma, width))
    return np.clip(cand, 0.0, 1000.0)


def do_peak_reduction(base, rng, n):
    """Gradient-inspired peak reduction."""
    b = fft_autocorr(base)
    max_b = b.max()
    if max_b <= 0:
        return np.clip(base, 0.0, 1000.0)
    
    thresh = float(rng.uniform(0.8, 0.99))
    active = np.where(b > thresh * max_b)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    if len(active) > 30:
        chosen = rng.choice(len(active), size=30, replace=False)
        active = active[np.sort(chosen)]
    
    cand = base.copy()
    step = float(rng.uniform(0.02, 0.1))
    S = base.sum()
    
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        direction = base[k - hi:k - lo + 1][::-1].copy()
        max_d = direction.max()
        if max_d > 1e-15:
            scale = step * S / max_d
            cand[lo:hi + 1] -= scale * direction
    
    cand *= np.exp(rng.normal(0.0, 0.02, n))
    return np.clip(cand, 0.0, 1000.0)


def do_symmetrize(base, rng, n):
    """Symmetrize with noise."""
    alpha = float(rng.uniform(0.3, 0.7))
    cand = alpha * base + (1 - alpha) * base[::-1]
    cand *= np.exp(rng.normal(0.0, float(rng.uniform(0.05, 0.2)), n))
    return np.clip(cand, 0.0, 1000.0)
