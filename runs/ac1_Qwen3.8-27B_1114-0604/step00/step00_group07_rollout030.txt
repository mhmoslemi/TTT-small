import numpy as np
import time
from scipy.optimize import linprog
from scipy.signal import convolve

def compute_objective(a):
    """Compute 2n * max(convolve(a,a)) / sum(a)^2"""
    n = len(a)
    b = convolve(a, a, mode='full')
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * np.max(b) / (s * s)

def solve_lp(f, M):
    """
    Solve the LP:
    Maximize sum(g)
    Subject to:
      g >= 0
      (f * g)[k] <= M for all k
    """
    n = len(f)
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Inequality constraints: (f * g)[k] <= M for all k
    # (f * g)[k] = sum_j f[j] * g[k-j] (valid j)
    # This is a linear function of g.
    # For each k, the coefficient of g[j] is f[k-j] if 0 <= k-j < n, else 0.
    
    # A_ub matrix: (2n-1) x n
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result
