"""
Gradient-based optimization of non-negative sequences to minimize
the score 2n*max(convolve(f,f))/(sum(f))^2 using log-sum-exp 
smoothing and multiplicative updates.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a)
    m = len(b)
    N = 1
    while N < n + m - 1:
        N *= 2
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    return np.fft.irfft(Fa * Fb, N)[:n + m - 1]


def fft_cross_correlate(p, f):
    """Compute cross-correlation: result[m] = sum_j p[j+m] * f[j] for m=0..len(f)-1."""
    np_ = len(p)
    nf = len(f)
    N = 1
    while N < np_ + nf - 1:
        N *= 2
    Ff = np.fft.rfft(f, N)
    Fp = np.fft.rfft(p, N)
    corr = np.fft.irfft(Fp * np.conj(Ff), N)
    return corr[:nf]


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v_sum = np.sum(v)
    if v_sum <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_candidates = np.nonzero(u * idx > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = np.sum(w)
    if w_sum > 0:
        w /= w_sum  # ensure exact normalization
    else:
        w = np.ones(n) / n
    return w


def get_initial_sequence(n=5582, seed=42):
    """Get initial sequence, trying to use height_sequence_1 if available."""
    rng = np.random.default_rng(seed)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, None)
        if parent.ndim == 1 and parent.size >= 2 and np.isfinite(parent).all() and np.sum(parent) > 0.01:
            return parent / np.sum(parent)
    except Exception:
        pass
    # Fallback: constant sequence
    return np.ones(n) / n


def run_optimization(f, n, rng, time_budget):
    """Run the main optimization loop. Returns (best_f, best_score)."""
    deadline = time.monotonic() + time_budget
    
    b = fft_convolve(f, f)
    max_b = float(np.max(b))
    best_score = 2.0 * n * max_b  # since sum(f) = 1
    best_f = f.copy()
    
    # Parameters
    T = max_b / 3.0
    T_min = max_b / 500.0
    lr = 0.1
    T_decay = 0.99995
    stall_count = 0
    stall_threshold = 500
    
    # Precompute FFT size
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft *= 2
    
    iter_count = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        # Compute convolution and gradient
        b = fft_convolve(f, f)
        max_b = float(np.max(b))
        current_score = 2.0 * n * max_b
        
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_f = f.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Compute softmax probabilities
        b_scaled = b / T
        b_max = float(np.max(b_scaled))
        p = np.exp(b_scaled - b_max)
        p_sum = float(np.sum(p))
        if p_sum > 0:
            p /= p_sum
        else:
            p = np.ones(len(b)) / len(b)
        
        # Compute gradient via cross-correlation
        grad = fft_cross_correlate(p, f)
        grad *= (2.0 / T)
        
        # Multiplicative update (mirror descent)
        step = lr * grad
        # Clip to avoid numerical issues
        step = np.clip(step, -5.0, 5.0)
        f = f * np.exp(-step)
        f = np.clip(f, 1e-15, None)
        f_sum = float(np.sum(f))
        if f_sum > 0:
            f /= f_sum
        else:
            f = best_f.copy()
        
        # Adapt temperature (annealing)
        T = max(T * T_decay, T_min)
        
        # Stall handling: restart from best with perturbation
        if stall_count > stall_threshold:
            noise_scale = 0.01 + 0.05 * rng.random()
            f = best_f + noise_scale * rng.standard_normal(n)
            f = np.clip(f, 1e-10, None)
            f_sum = float(np.sum(f))
            if f_sum > 0:
                f /= f_sum
            else:
                f = best_f.copy()
            T = max(3.0 * max_b, T_min)
            lr = min(lr * 1.5, 0.5)
            stall_count = 0
            stall_threshold = int(stall_threshold * 1.5)  # be more patient
        
        iter_count += 1
        
        # Print progress
        if time.monotonic() - last_print > 10:
            print(f"Iter {iter_count}: score={current_score:.6f}, best={best_score:.6f}, T={T:.2e}, lr={lr:.4f}")
            last_print = time.monotonic()
    
    return best_f, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(convolve(f,f))/(sum(f))^2.
    
    Uses gradient-based optimization with log-sum-exp smoothing of the max

Wait, I have a concern: the `fft_cross_correlate` function. Let me verify the convention.

I want `grad[m] = sum_j p[j+m] * f[j]` for m = 0, ..., n-1, where p has length 2n-1 and f has length n.

The cross-correlation `(p ⋆ f)[m] = sum_j p[j+m] * conj(f[j])`.

In FFT terms: `ifft(FFT(p) * conj(FFT(f)))` gives the circular cross-correlation. With zero-padding to length >= len(p) + len(f) - 1 = (2n-1) + n - 1 = 3n-2, it gives the linear cross-correlation.
