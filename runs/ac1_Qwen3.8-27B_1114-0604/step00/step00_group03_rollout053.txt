def _next_pow2(m):
    n=1
    while n < m: n <<= 1
    return n

def _fft_conv(a):
    n = a.size
    if n == 0: return np.array([], dtype=float)
    need = 2*n -1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim !=1 or arr.size==0: return np.inf
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01: return np.inf
    c = _fft_conv(arr)
    if c.size ==0: return np.inf
    mx = float(np.max(c))
    if mx < 0: mx = 0.0
    return float(2.0 * arr.size * mx / (s*s))

def _make_sine_candidate(n, amp, mode):
    n = int(n)
    if n <=0: return np.array([], dtype=float)
    i=np.arange(n, dtype=float)
    theta=2.0*np.pi*(i+0.5)/float(n)
    delta=np.sin(float(mode)*theta)
    a=(1.0 + float(amp)*delta)/float(n)
    # clip tiny negatives
    a=np.where(a > 0.0, a, 0.0)
    s=a.sum()
    if s <= 0: return np.ones(n)/n
    a /= s
    return a

```python
"""
Low-peak nonnegative step-function search.

The main construction is an anti-palindromic sinusoidal shape:
    a_i = (1 + A sin(2*pi*(i+0.5)/n)) / n,
followed by normalization.  The sinusoidal shape makes the central convolution
lag anti-correlated with itself, while scanning the amplitude A keeps edge and
intermediate lags balanced.  A short active-set mirror refinement is then applied
optionally; only improvements are kept.
"""

import numpy as np
import time


def _next_pow2(m):
    """Smallest power of two >= m."""
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    """
    Fast internal evaluation matching evaluate_sequence, assuming no clipping
    is active.  Used for search; final validation uses evaluate_sequence.
    """
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))


def _make_sine_candidate(n, amp, mode):
    """
    Create a nonnegative normalized sinusoidal anti-palindromic sequence.
    """
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    delta = np.sin(float(mode) * theta)
    a = (1.0 + float(amp) * delta) / float(n)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        a = np.ones(n, dtype=np.float64) / float(n)
        return a
    a = a / s
    return a


def _score_sine_fast(n, mode, amp, delta, base):
    """
    Score a sinusoidal candidate without recomputing the sinusoid.
    """
    a = base * (1.0 + float(amp) * delta)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    return _internal_score(a)


def _optimize_sine_amplitude(n, mode, deadline_abs, guess=0.78):
    """
    One-dimensional amplitude scan/refinement for a fixed length n and mode.
    """
    best_score = np.inf
    best_amp = float(guess)

    n_float = float(n)
    base = 1.0 / n_float
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / n_float
    delta = np.sin(float(mode) * theta)

    # Coarse scan.  Amplitudes > 1 are allowed because clipping produces
    # valid nonnegative shapes, although the optimum is normally below 1.
    amps = np.linspace(0.0, 1.5, 49)
    for amp in amps:
        if time.monotonic() > deadline_abs:
            break
        sc = _score_sine_fast(n, mode, amp, delta, base)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_amp = float(amp)

    # Local grid refinements.
    if np.isfinite(best_score):
        step = 1.5 / 48.0
        for _ in range(5):
            low = max(0.0, best_amp - step)
            high = min(1.5, best_amp + step)
            if high - low <= 1e-7:
                break
            grid = np.linspace(low, high, 21)
            for amp in grid:
                if time.monotonic() > deadline_abs:
                    break
                sc = _score_sine_fast(n, mode, amp, delta, base)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_amp = float(amp)
            step *= 0.5

    return best_amp, best_score


def _scan_family(deadline_abs):
    """
    Search several lengths and amplitudes of the construcing sinusoidal family.
    """
    best_a = None
    best_score = np.inf

    # Start with a length that is already accurate, then test larger lengths.
    lengths = [2048, 4096, 1024, 8192, 512, 256]
    guess_amp = 0.78

    for n in lengths:
        if time.monotonic() > deadline_abs:
            break
        amp, sc = _optimize_sine_amplitude(n, 1, deadline_abs, guess_amp)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = _make_sine_candidate(n, amp, 1)
            guess_amp = amp
            print(f"[scan] n={n} amp={amp:.6f} internal_score={sc:.8f}")

    if best_a is None:
        best_a = _make_sine_candidate(2048, guess_amp, 1)
        best_score = _internal_score(best_a)

    return best_a, best_score


def _local_refine(a, budget_s=60.0):
    """
    Optional light active-set mirror refinement on the simplex.
    Returns the best sequence found internally during the short budget.
    """
    if budget_s <= 1.0:
        arr = np.asarray(a, dtype=np.float64).copy()
        return arr, _internal_score(arr)

    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return arr, np.inf
    arr = arr / s

    best = arr.copy()
    best_score = _internal_score(best)
    if not np.isfinite(best_score):
        return best, best_score

    n = arr.size
    lr = max(1.0, 0.05 * float(n))
    max_lr = 0.25 * float(n)
    period = 10
    max_iters = 6000
    it = 0
    last_improve = 0
    K = 8 if n <= 4096 else 12

    while time.monotonic() - start < budget_s and it < max_iters:
        if time.monotonic() - start > budget_s:
            break
        it += 1

        c = _fft_convolve(arr)
        if c.size == 0:
            break
        maxc = float(np.max(c))
        if not np.isfinite(maxc):
            break

        if c.size <= K:
            active = np.arange(c.size, dtype=np.int64)
        else:
            active = np.argpartition(c, -K)[-K:]

        grad = np.zeros(n, dtype=np.float64)
        cnt = 0
        for kk in active:
            k = int(kk)
            cnt += 1
            i_min = k - (n - 1)
            if i_min < 0:
                i_min = 0
            i_max = k
            if i_max >= n:
                i_max = n - 1
            if i_max >= i_min:
                idx = np.arange(i_min, i_max + 1, dtype=np.int64)
                grad[idx] += 2.0 * arr[k - idx]

        if cnt <= 0:
            break
        grad /= float(cnt)

        # Project to the tangent plane of the simplex by removing the mean.
        g = grad - np.mean(grad)
        if not np.any(g):
            break

        exponent = -lr * g
        exponent = np.clip(exponent, -1.0, 1.0)
        arr = arr * np.exp(exponent)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        ts = float(arr.sum())
        if not np.isfinite(ts) or ts < 0.01:
            arr = best.copy()
            break
        arr = arr / ts

        if it % period == 0:
            sc = _internal_score(arr)
            if np.isfinite(sc) and sc < best_score:
                best = arr.copy()
                best_score = sc
                last_improve = it
                lr = min(max_lr, lr * 1.05)
            else:
                if sc > best_score * 1.02:
                    arr = best.copy()
                    lr = max(1.0, lr * 0.75)
                else:
                    lr = min(max_lr, lr * 1.05)
                if it - last_improve > 300:
                    break

    return best, best_score


def _safe_resolve_sequence(x):
    """Sanitize a sequence before passing it to evaluate_sequence."""
    if x is None:
        return None
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return None
    arr = arr / s
    return arr


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a low evaluation score and return the best valid sequence found.
    """
    t0 = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    hard_deadline = t0 + total_budget - 20.0

    # Allocate enough time for final validation.
    if total_budget > 1200.0:
        scan_deadline = t0 + 520.0
        refine_budget = 120.0
    elif total_budget > 500.0:
        scan_deadline = t0 + max(30.0, total_budget - 250.0)
        refine_budget = 80.0
    else:
        scan_deadline = t0 + max(10.0, total_budget - 80.0)
        refine_budget = 0.0

    scan_a, scan_score = _scan_family(scan_deadline)
    print(f"[search] scan best internal_score={scan_score:.8f} time={time.monotonic()-t0:.2f}")

    refine_a = None
    refine_score = np.inf
    if refine_budget > 5.0 and scan_a is not None:
        remaining_before_refine = hard_deadline - time.monotonic()
        if remaining_before_refine > 90.0:
            refine_budget = min(refine_budget, remaining_before_refine - 70.0)
            refine_a, refine_score = _local_refine(scan_a, refine_budget)
            if np.isfinite(refine_score) and refine_score < scan_score:
                print(f"[refine] improved internal_score={refine_score:.8f} time={time.monotonic()-t0:.2f}")
            else:
                refine_a = None
                print(f"[refine] no improvement time={time.monotonic()-t0:.2f}")

    candidates = []
    if scan_a is not None:
        candidates.append(("scan", scan_a, scan_score))
    if refine_a is not None:
        candidates.append(("refine", refine_a, refine_score))

    # Robust fallbacks.
    fallback1 = _make_sine_candidate(4096, 0.78, 1)
    fallback2 = _make_sine_candidate(2048, 0.78, 1)
    candidates.append(("fallback4096", fallback1, _internal_score(fallback1)))
    candidates.append(("fallback2048", fallback2, _internal_score(fallback2)))

    # Sort by internal score.  Avoid lambdas by using tuple (score, index, arr).
    sortable = []
    for idx, (name, cand, sc) in enumerate(candidates):
        arr = _safe_resolve_sequence(cand)
        if arr is not None:
            sortable.append((float(sc), idx, name, arr))
    sortable.sort()

    best_actual_score = np.inf
    best_out = None

    # Validate at most three candidates.  If the first good one is already
    # well below previous bounds, stop early.
    for sc, idx, name, arr in sortable[:3]:
        if time.monotonic() > hard_deadline:
            break
        try:
            real_score = float(evaluate_sequence(arr.tolist()))
        except Exception:
            real_score = np.inf
        if np.isfinite(real_score) and real_score < best_actual_score:
            best_actual_score = real_score
            best_out = arr
            print(f"[validate] {name} n={arr.size} actual_score={real_score:.8f}")
            if best_actual_score < 1.45:
                break

    if best_out is None:
        # Guaranteed valid fallback: constant sequence has score 2.
        best_out = np.ones(256, dtype=np.float64) / 256.0
        print("[fallback] using constant sequence")

    return [float(x) for x in best_out]
```
