import time

"""
Multi-phase local search for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Key insight: antisymmetric perturbations (a[i]-c = -(a[n-1-i]-c)) reduce the
central convolution peak below the uniform value of 2. Step function at eps=1/3
gives ~1.778. We refine using antisymmetric local search and Fourier smoothing.

def fast_score(a, n):
    """Compute score without calling evaluate_sequence."""
    s = a.sum()
    if s < 0.01:
        return 1e10
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)


def make_step(a_n, eps):
    """Antisymmetric step: lower on left, higher on right."""
    n = a_n
    a = np.ones(n)
    half = n // 2
    a[:half] = 1.0 - eps
    a[half:] = 1.0 + eps
    return np.clip(a, 0.0, 1000.0)


def make_tanh_shape(a_n, eps, k):
    """Antisymmetric tanh shape: smooth transition."""
    n = a_n
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / (n - 1) - 1.0
    a = 1.0 + eps * np.tanh(k * t)
    return np.clip(a, 0.0, 1000.0)


def make_sinusoidal(a_n, eps, freq=1):
    """Antisymmetric sinusoidal perturbation."""
    n = a_n
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.sin(np.pi * freq * i / (n // 2))
    # Ensure antisymmetry: a[i] - 1 = -(a[n-1-i] - 1)
    # sin(pi*freq*i/(n/2)) at position n-1-i gives sin(pi*freq*(n-1-i)/(n/2))
    # For freq=1: sin(pi - pi*i/(n/2)) = sin(pi*i/(n/2)), so it's symmetric not antisymmetric!
    # Need to use a different basis.
    return np.clip(a, 0.0, 1000.0)


def make_antisymmetric_fourier(a_n, coeffs):
    """Build antisymmetric sequence from Fourier sine coefficients.
    Uses basis sin(2*m*pi*i/(n-1)) which is antisymmetric for integer m.
    Actually sin(2*m*pi*(n-1-i)/(n-1)) = sin(2*m*pi - 2*m*pi*i/(n-1))
    = -sin(2*m*pi*i/(n-1)) for integer m. So it IS antisymmetric.
    """
    n = a_n
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for m, c in enumerate(coeffs, start=1):
        a += c * np.sin(2.0 * m * np.pi * i / (n - 1))
    return np.clip(a, 0.0, 1000.0)


def antisymmetric_perturb(a, rng, scale=0.05):
    """Antisymmetric block multiplication perturbation."""
    n = a.size
    half = n // 2
    a_new = a.copy()
    bsize = int(rng.integers(3, max(5, n // 80)))
    start = int(rng.integers(0, half - bsize))
    end = start + bsize
    mstart = n - 1 - (end - 1)
    mend = n - 1 - start
    delta = float(rng.normal(0, scale))
    a_new[start:end] *= np.exp(delta)
    a_new[mstart:mend + 1] *= np.exp(-delta)
    return np.clip(a_new, 0.0, 1000.0)


def regular_perturb(a, rng, scale=0.05):
    """Regular (non-antisymmetric) block perturbation."""
    n = a.size
    a_new = a.copy()
    bsize = int(rng.integers(3, max(5, n // 40)))
    start = int(rng.integers(0, n - bsize))
    end = start + bsize
    delta = float(rng.normal(0, scale))
    a_new[start:end] *= np.exp(delta)
    return np.clip(a_new, 0.0, 1000.0)


def fourier_smooth_perturb(a, rng, n_modes=5, scale=0.02):
    """Perturb in the antisymmetric Fourier basis (low modes)."""
    n = a.size
    a_new = a.copy()
    mode = int(rng.integers(1, n_modes + 1))
    i = np.arange(n, dtype=np.float64)
    basis = np.sin(2.0 * mode * np.pi * i / (n - 1))
    delta = float(rng.normal(0, scale))
    a_new = a_new + delta * basis
    return np.clip(a_new, 0.0, 1000.0)


def targeted_peak_reduction(a, conv, rng, scale=0.03):
    """Targeted move to reduce the convolution peak."""
    n = a.size
    peak_k = int(conv.argmax())
    a_new = a.copy()
    # Find indices where a[i]*a[peak_k - i] contributes significantly
    valid_start = max(0, peak_k - n + 1)
    valid_end = min(peak_k, n - 1)
    idx = np.arange(valid_start, valid_end + 1)
    products = a[idx] * a[peak_k - idx]
    # Pick the index with largest product
    best_i = idx[np.argmax(products)]
    # Antisymmetric move: decrease a[best_i], increase a[n-1-best_i]
    half = n // 2
    if best_i < half:
        a_new[best_i] *= (1.0 - scale)
        a_new[n - 1 - best_i] *= (1.0 + scale)
    else:
        a_new[best_i] *= (1.0 + scale)
        a_new[n - 1 - best_i] *= (1.0 - scale)
    return np.clip(a_new, 0.0, 1000.0)
