import numpy as np
from numpy.fft import fft, ifft, next_fast_len
import time

def compute_convolution_and_grad(v, size, beta):
    """Compute convolution, soft-max weights, and gradient in one pass."""
    n = len(v)
    V = fft(v, n=size)
    b = np.real(ifft(V * V, n=size))[:2*n-1]
    M = float(np.max(b))
    b_shifted = (b - M) * beta
    exp_b = np.exp(np.clip(b_shifted, -80, 0))
    p_sum = float(np.sum(exp_b))
    if p_sum < 1e-300:
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
    else:
        p = exp_b / p_sum
    P = fft(p, n=size)
    grad = np.real(ifft(P * np.conj(V), n=size))[:n]
    return M, grad

def optimize_sequence(v0, deadline, rng, beta_start=20.0, beta_end=300.0,
                      lr_start=0.5, lr_end=0.05, momentum=0.9,
                      noise_every=3000, noise_scale=0.003,
                      symm_every=0, max_iters=10**9):
    """Momentum gradient descent on normalized v to minimize max(v*v)."""
    v = np.clip(v0, 0, None).copy()
    n = len(v)
    S = float(np.sum(v))
    if S < 1e-10:
        return None, np.inf
    v = v / S
    size = next_fast_len(3 * n)
    M, _ = compute_convolution_and_grad(v, size, beta_start)
    best_M = M
    best_v = v.copy()
    velocity = np.zeros(n)
    it = 0
    stagnation = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        frac = min(1.0, it / 15000)
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start + (lr_end - lr_start) * frac

        M, grad = compute_convolution_and_grad(v, size, beta)

        if M < best_M - 1e-12:
            best_M = M
            best_v = v.copy()
            stagnation = 0
        else:
            stagnation += 1

        velocity = momentum * velocity - lr * grad
        v = v + velocity
        v = np.maximum(v, 0)
        S = float(np.sum(v))
        if S < 1e-10:
            v = best_v.copy()
            velocity = np.zeros(n)
            continue
        v = v / S

        if symm_every > 0 and it % symm_every == 0:
            v = 0.5 * (v + v[::-1])
            v = np.maximum(v, 0)
            S = float(np.sum(v))
            if S > 1e-10:
                v = v / S

        if it > 0 and it % noise_every == 0:
            ns = noise_scale * float(np.max(v))
            v = np.maximum(v + rng.normal(0, ns, n), 0)
            S = float(np.sum(v))
            if S > 1e-10:
                v = v / S
            velocity = np.zeros(n)

        if stagnation > noise_every * 3 and it > 1000:
            # Larger restart
            ns = 0.02 * float(np.max(v))
            v = np.maximum(v + rng.normal(0, ns, n), 0)
            S = float(np.sum(v))
            if S > 1e-10:
                v = v / S
            velocity = np.zeros(n)
            stagnation = 0

    return best_v, 2.0 * n * best_M
