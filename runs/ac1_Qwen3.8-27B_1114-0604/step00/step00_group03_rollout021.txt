def compute_convolution(a):
    """Compute self-convolution using FFT for efficiency."""
    n = len(a)
    if n < 64:
        return np.convolve(a, a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.irfft(A * A, fft_size)[:size]
    # Clip tiny negative values from FFT numerical errors
    B = np.maximum(B, 0)
    return B

def simplex_projection(v, target_sum):
    """Project v onto {x : x >= 0, sum(x) = target_sum}. O(n log n)."""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    if np.sum(v) == target_sum and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find rho
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

Actually, for the coordinate step, since $b_k$ is linear in $a_j$ (when all other coordinates are fixed), I can compute the optimal $a_j$ analytically.

Given all coordinates except $a_j$ are fixed, the convolution values are:
$b_k(a_j) = m_k \cdot a_j + c_k$

where $m_k = a_{j-k} + a_{j+k}$ (with zero padding) and $c_k = b_k - m_k \cdot a_j$ (the part not involving $a_j$).

We want to minimize $\max_k (m_k a_j + c_k)$ subject to $a_j \geq 0$ and $\sum_i a_i = 1$.

The sum constraint means: $a_j = 1 - \sum_{i \neq j} a_i$. So $a_j$ is determined by the other coordinates. If we change $a_j$, we need to redistribute the difference to other coordinates.

This is the issue: the simplex constraint couples all coordinates.

A simpler approach for coordinate descent:
- Fix the sum of the "other" coordinates: $S_{-j} = 1 - a_j$.
- If we change $a_j$ to $a_j'$, we need to redistribute $1 - a_j' - S_{-j} = a_j - a_j'$ among the other coordinates.
- The simplest redistribution: add $(a_j - a_j') / (n-1)$ to each other coordinate.

But this is approximate. A cleaner approach:

**Block coordinate descent with 2 coordinates:**
Pick two coordinates $j$ and $k$ (where $k$ is a peak-contributing partner of $j$). Transfer mass between them: $a_j \to a_j - \delta$, $a_k \to a_k + \delta$. This keeps the sum constant.

The objective as a function of $\delta$ is $\max_m (m_m \cdot \delta + c_m)$ which is piecewise linear. We can find the optimal $\delta$ by checking the intersection points.

    result = np.maximum(result, 0)  # Remove tiny negatives from FFT

def simplex_project(v, target):
    """Project onto {x >= 0, sum(x) = target} in O(n log n)."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - target)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rho = idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)

                    a[i] += delta

        s = a.sum()

**Faster approach: Explicit peak reduction**

Instead of subgradient descent, directly reduce the peak:
1. Find the peak lag $k^*$.
2. The peak value is $b_{k^*} = \sum_i a_i a_{i-k^*}$.
3. To reduce $b_{k^*}$, we want to make the product $a_i a_{i-k^*}$ smaller for all $i$.
4. The most effective way: for each pair $(i, i-k^*)$ where both are positive, equalize them (set both to their geometric mean). This minimizes the product given the sum constraint.

But equalizing pairs for one lag $k^*$ might increase other lags. So we need to be careful.

**Alternative: Iterative proportional fitting (IPF) / Sinkhorn-like**

Think of the convolution as a "transport" problem. The value $b_k$ measures how much "mass" is transported across distance $k$. We want this to be as uniform as possible.

A Sinkhorn-like algorithm:
1. Start with $a^{(0)}$.
2. Compute $b^{(0)}_k = \sum_i a^{(0)}_i a^{(0)}_{i-k}$.
3. Define a "correction" based on how $b_k$ deviates from the target (uniform: $1/(2n-1)$).
4. Update $a$ to reduce the peak and increase the troughs.

The update could be:
$a^{(new)}_i \propto a^{(old)}_i \cdot \exp(-\eta \cdot s_i)$

where $s_i = \sum_k w_k (a_{i-k} + a_{i+k})$ is a "stress" measure (how much $a_i$ contributes to the peaks).

This is multiplicative updating, which guarantees non-negativity.

import numpy as np
import time

def compute_convolution(a):
    """Compute self-convolution using FFT for efficiency."""
    n = len(a)
    if n < 128:
        return np.convolve(a, a, mode='full')
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    # Remove tiny negative values from FFT roundoff
    np.maximum(result, 0, out=result)
    return result

def simplex_project(v, target):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rho = idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def trim_zeros(a):
    """Remove leading and trailing zeros from sequence."""
    nz = np.where(a > 1e-15)[0]
    if len(nz) == 0:
        return np.array([1.0])
    return a[nz[0]:nz[-1] + 1]

def stress_vector(a, b, num_peaks=10):
    """Compute the 'stress' on each coordinate from the peak lags."""
    n = len(a)
    s = np.zeros(n)
    top_idx = np.argsort(b)[::-1][:num_peaks]
    top_vals = b[top_idx]
    peak = top_vals[0]
    
    # Soft weights
    spread = max(peak - top_vals[-1], 1e-15) / max(num_peaks - 1, 1)
    w = np.exp(-(top_vals - peak) / spread)
    w /= w.sum()
    
    for wi, k in zip(w, top_idx):
        k = int(k)
        if k >= n:
            k = 2 * n - 2 - k
        if k == 0:
            s += 2.0 * wi * a
        else:
            s[k:] += wi * a[:n - k]
            s[:n - k] += wi * a[k:]
    return s

def multiplicative_optimize(a0, deadline, eta0=0.1, eval_interval=200):
    """
    Multiplicative update to flatten self-convolution.
    a_i <- a_i * exp(-eta * s_i / max(s))
    where s_i is the stress (contribution to peak lags).
    """
    a = a0.copy()
    n = len(a)
    best = a.copy()
    best_score = evaluate_sequence(a.tolist())
    eta = eta0
    no_improve = 0
    iteration = 0
    
    while time.time() < deadline:
        b = compute_convolution(a)
        s = stress_vector(a, b, num_peaks=10)
        
        # Separable amortization update
        a = a * np.exp(-eta * s / (s.max() + 1e-15))
        a = np.clip(a, 0, None)
        total = a.sum()
        if total > 1e-15:
            a = a / total
        else:
            a = np.ones(n) / n
            break
        
        iteration += 1
        
        if iteration % eval_interval == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            # Adaptive eta
            if no_improve >= 5:
                eta *= 0.7
                no_improve = 0
                if
