```python
import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve(a, b):
    """Compute full convolution using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    # Pad to next power of 2 for efficiency
    N = 1
    while N < size:
        N *= 2
    A = fft(a, N)
    B = fft(b, N)
    C = ifft(A * B)
    return C.real[:size]

def project_onto_simplex(a, S):
    """Project a onto {x >= 0, sum(x) = S}."""
    # Binary search for lambda
    a = np.asarray(a, dtype=np.float64)
    lo = a.min()
    hi = a.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.maximum(a - mid, 0)
        s = proj.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.maximum(a - (lo + hi) / 2, 0)

def compute_score_fast(a):
    """Compute the score using FFT convolution."""
    n = len(a)
    c = fft_convolve(a, a)
    max_c = c.max()
    sum_a = a.sum()
    return 2.0 * n * max_c / (sum_a ** 2)
```

```python
"""
Optimizes nonnegative sequences to minimize 2*n*max(convolve(a,a))/sum(a)^2
using gradient descent with simplex projection and FFT-based convolution.
"""

import numpy as np
from scipy.fft import fft, ifft
import time


def next_pow2(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()


def fft_convolve(a, b, N):
    """Full convolution using FFT with precomputed size N."""
    A = fft(a, N)
    B = fft(b, N)
    return ifft(A * B).real[:len(a) + len(b) - 1]


def project_simplex(v, S):
    """Project vector v onto {x >= 0, sum(x) = S} via binary search."""
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.maximum(v - mid, 0.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.maximum(v - lam, 0.0)


def compute_score_fast(a, N):
    """Compute 2*n*max(conv)/sum^2 using FFT."""
    n = len(a)
    c = fft_convolve(a, a, N)
    max_c = c.max()
    S = a.sum()
    return 2.0 * n * max_c / (S * S)
```
