def fft_self_conv(a):
    n = len(a)
    size = 1 << (2*n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2*n-1]

def compute_gradient(a, C, k_peaks=16):
    """Compute the gradient of max(C) with respect to a, using top-k peaks."""
    n = len(a)
    g = np.zeros(n)
    # Top-k peaks
    if k_peaks < len(C):
        top_idx = np.argpartition(C, -k_peaks)[-k_peaks:]
    else:
        top_idx = np.arange(len(C))
    
    for m in top_idx:
        m = int(m)
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            # ∂C[m]/∂a[i] = 2*a[m-i] for valid i
            g[lo:hi+1] += 2.0 * a[m-hi : m-lo+1][::-1]
    return g

def optimize(a0, budget_s, k_peaks=16, lr_init=1.0, lr_decay=0.999):

    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = np.sum(a)
    if S <= 0:
        return a0, np.inf
    
    # Work with softmax parameterization
    a = a / S  # Initialize with sum = 1
    b = np.log(a + 1e-30)  # Inverse softmax (log of a)
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(fft_self_conv(a))
    
    deadline = time.monotonic() + budget_s
    lr = lr_init
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        C = fft_self_conv(a)
        peak = np.max(C)
        score = 2.0 * n * peak  # S=1
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute gradient
        g = compute_gradient(a, C, k_peaks)
        
        # Project gradient to tangent space of simplex (subtract mean weighted by a)
        # ∂f/∂b[i] = a[i] * (g[i] - sum_j a[j]*g[j])
        g_mean = np.dot(a, g)
        grad_b = a * (g - g_mean)
        
        b -= lr * grad_b
        a = np.exp(b - np.max(b))
        a /= np.sum(a)
        
        lr *= lr_decay

            b += 0.1 * np.random.randn(n)
            no_improve = 0
        
        it += 1
        if it % 5000 == 0:
            print(f"  Iter {it}: score={best_score:.6f}, lr={lr:.4f}")
    
    return best_a, best_score
