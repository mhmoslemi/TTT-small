    exp_vals = np.exp(shifted)
    Z = np.sum(exp_vals)
    if Z < 1e-300:
        weights = np.zeros_like(c)
        weights[np.argmax(c)] = 1.0
    else:
        weights = exp_vals / Z
    return M, weights


def lp_kick(a, max_constraints=150, time_limit=30.0):
    """
    Solve LP to find direction d reducing max(convolve(a,a)).
    Minimize t s.t. (a*d)[k] <= t for active k, sum(d)=0, -1<=d<=1.
    Returns direction d or None.
    """
    start = time.monotonic()
    n = len(a)
    c = compute_convolution(a)
    M = np.max(c)
    if M <= 0:
        return None

    # Select active constraints (top portion of c)
    threshold = M - 0.05 * M
    active_mask = c >= threshold
    active_idx = np.where(active_mask)[0]

    # Also add a few more diverse constraints
    if len(active_idx) < max_constraints:
        # Add constraints spread across the range
        extra = np.linspace(0, 2 * n - 2, max_constraints, dtype=int)
        extra = extra[~np.isin(extra, active_idx)]
        active_idx = np.concatenate([active_idx, extra[:max_constraints - len(active_idx)]])

    active_idx = active_idx[:max_constraints]
    num_c = len(active_idx)

    # Build constraint matrix (num_c x (n+1))
    # Row k: sum_j a[j]*d[k-j] - t <= 0
    # coeff of d[m] = a[k-m] if 0 <= k-m < n
    A_ub = np.zeros((num_c, n + 1))
    for i, k in enumerate(active_idx):
        A_ub[i, -1] = -1.0
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        if m_min <= m_max:
            A_ub[i, m_min:m_max + 1] = a[k - m_min:k - m_max + 1][::-1]

    # b_ub = 0
    b_ub = np.zeros(num_c)

    # sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    # Bounds: -1 <= d[i] <= 1, t unbounded below
    bounds = [(-1.0, 1.0)] * n + [(-np.inf, np.inf)]

    try:
        result = linprog(
            np.zeros(n + 1),  # minimize 0 (feasibility) - we just want a direction
            A_ub=A_ub, b_ub=b_ub,
            A_eq=A_eq, b_eq=b_eq,
            bounds=bounds, method='highs'
        )
    except Exception:
        return None

    if time.monotonic() - start > time_limit:
        return None

    if not result.success:
        return None

    d = result.x[:n]
    # Check it's a valid direction (norm > 0)
    if np.linalg.norm(d) < 1e-10:
        return None

    # Verify it actually reduces the max (first-order check on active set)
    # Compute (a*d)[k] for active k
    improvement = 0.0
    for k in active_idx[:50]:
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        val = np.dot(a[k - m_min:k - m_max + 1][::-1], d[m_min:m_max + 1])
        improvement = max(improvement, val)

    if improvement > 0:
        # Direction increases max, reverse it
        d = -d
        improvement = -improvement

    if improvement >= 0:
        return None

    return d


def ramp_init(n):
    """Create a linear ramp initialization."""
    j = np.arange(n, dtype=np.float64)
    a = j / (n - 1) if n > 1 else np.array([1.0])
    return a


def smoothed_ramp_init(n, alpha=0.3):
    """Create a smoothed ramp with some curvature."""
    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1) if n > 1 else np.array([0.5])
    # Ramp with slight S-curve
    a = x + alpha * np.sin(np.pi * x)
    a = np.maximum(a, 0.0)
    return a


def two_hump_init(n):
    """A sequence with two humps on opposite sides."""
    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1) if n > 1 else np.array([0.0])
    a = 0.5 + 0.4 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.01)
    return a


def run_gradient_descent(a, n, S, budget_s, initial_lr=0.5, initial_tau=1.0,
                         tau_decay=0.999, lr_decay=0.9995, print_interval=500):
    """
    Run smooth-max gradient descent to minimize max(convolve(a,a)).
    a is normalized to sum = S = sqrt(2n).
    Returns best a and best M found.
    """
    deadline = time.monotonic() + budget_s
    a = a.copy()
    a = np.maximum(a, 0.0)
    a = a / np.sum(a) * S

    best_a = a.copy()
    c = compute_convolution(a)
    best_M = np.max(c)
    best_score = 2.0 * n * best_M / (S * S)  # = best_M since S^2 = 2n

    size, A = make_fft_setup(a)
    w_rev_base = None
    iteration = 0
    no_improve_count = 0
    stagnation_threshold = 3000

    while time.monotonic() < deadline:
        # Check time frequently for tight loop
        if iteration % 100 == 0 and time.monotonic() > deadline:
            break

        # Compute convolution
        c = compute_convolution(a)
        M = np.max(c)

        # Update best
        if M < best_M:
            best_M = M
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Print progress
        if iteration % print_interval == 0:
            print(f"  iter={iteration}, M={M:.6f}, best_M={best_M:.6f}, "
                  f"tau={initial_tau * tau_decay**iteration:.4f}, "
                  f"lr={initial_lr * lr_decay**iteration:.6f}")

        # Decay tau and lr
        current_tau = initial_tau * (tau_decay ** iteration)
        current_tau = max(current_tau, 0.01)
        current_lr = initial_lr * (lr_decay ** iteration)
        current_lr = max(current_lr, 0.001)

        # If stagnating, increase temperature
        if no_improve_count > stagnation_threshold:
            current_tau = min(current_tau * 2, 5.0)
            current_lr = min(current_lr * 2, 1.0)
            no_improve_count = 0
            print(f"  Stagnation detected, increasing tau to {current_tau:.3f}")

        # Compute smooth max weights
        _, weights = smooth_max_weights(c, current_tau)

        # Compute gradient
        w_rev = weights[::-1]
        Wr = np.fft.rfft(w_rev, size)
        grad = compute_gradient(weights, a, size, A, Wr)

        # Renormalize FFT of a if it changed
        # (a changes each iteration, so we need to update A)
        # Actually A was computed once at the start. We need to recompute.
        # For efficiency, let's recompute A each iteration.
        A = np.fft.rfft(a, size)
        Wr = np.fft.rfft(weights[::-1], size)
        grad = compute_gradient(weights, a, size, A, Wr)

        # Gradient step
        a = a - current_lr * grad

        # Project to non-negative
        a = np.maximum(a, 0.0)

        # Normalize to sum = S
        s = np.sum(a)
        if s > 1e-10:
            a = a / s * S
        else:
            a = ramp_init(n) / np.sum(ramp_init(n)) * S

        iteration += 1

    return best_a, best_M


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for optimal step function coefficients minimizing
    2n * max(convolve(a,a)) / (sum a)^2.
    """
    total_start = time.monotonic()
    total_deadline = total_start + budget_s - 10  # leave 10s buffer
    rng = np.random.default_rng(seed)

    best_global = None
    best_score = np.inf

    # Try multiple strategies
    strategies = [
        ("ramp_5000", 5000, ramp_init),
        ("ramp_8000", 8000, ramp_init),
        ("smooth_ramp_5000", 5000, smoothed_ramp_init),
        ("two_hump_5000", 5000, two_hump_init),
    ]

    # Also try provided sequence if available
    try:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if provided.size > 0 and np.sum(provided) > 0.01:
            strategies.append(("provided", provided.size, None))
    except NameError:
        pass

    # Allocate time: give more time to the first (best) strategy
    # Ramp should be the best starting point
    time_per_strategy = max(60, (budget_s - 30) / len(strategies))

    for name, n, init_func in strategies:
        if time.monotonic() > total_deadline:
            break

        print(f"\n=== Strategy: {name} (n={n}) ===")
        strat_start = time.monotonic()

        if init_func is not None:
            a = init_func(n)
        else:
            a = provided.copy()
            n = a.size

        S = np.sqrt(2.0 * n)
        a = np.maximum(a, 0.0)
        a = a / np.sum(a) * S

        # Phase 1: Gradient descent
        time_remaining = min(time_per_strategy, total_deadline - time.monotonic())
        gd_time = time_remaining * 0.7
        if gd_time < 5:
            continue

        best_a, best_M = run_gradient_descent(
            a, n, S, gd_time,
            initial_lr=0.3, initial_tau=2.0,
            tau_decay=0.9995, lr_decay=0.9997
        )
        score = best_M  # since S^2 = 2n, objective = M
        print(f"  After GD: score={score:.6f}")
        if score < best_score:
            best_score = score
            best_global = best_a.copy()

        # Phase 2: LP kicks
        lp_time = time_remaining * 0.25
        if lp_time > 5 and time.monotonic() < total_deadline:
            a = best_a.copy()
            kick_deadline = time.monotonic() + lp_time
            consecutive_fails = 0

            while time.monotonic() < kick_deadline:
                d = lp_kick(a, max_constraints=100, time_limit=15)
                if d is None:
                    consecutive_fails += 1
                    if consecutive_fails > 5:
                        break
                    continue

                # Apply kick with small step
                d_norm = np.linalg.norm(d)
                if d_norm < 1e-10:
                    continue
                d = d / d_norm

                # Try several step sizes
                best_kick_a = a.copy()
                best_kick_M = np.max(compute_convolution(a))
                for step in [0.1, 0.05, 0.02, 0.01, 0.5]:
                    a_try = a + step * d
                    a_try = np.maximum(a_try, 0.0)
                    s = np.sum(a_try)
                    if s < 1e-10:
                        continue
                    a_try = a_try / s * S
                    M_try = np.max(compute_convolution(a_try))
                    if M_try < best_kick_M:
                        best_kick_M = M_try
                        best_kick_a = a_try.copy()

                if best_kick_M < best_M:
                    best_M = best_kick_M
                    a = best_kick_a.copy()
                    if best_M < best_score:
                        best_score = best_M
                        best_global = a.copy()
                        print(f"  LP kick improved: score={best_M:.6f}")
                    consecutive_fails = 0
                else:
                    consecutive_fails += 1

        # Phase 3: Short GD refinement
        if time.monotonic() < total_deadline:
            gd2_time = min(30, total_deadline - time.monotonic())
            if gd2_time > 5:
                best_a2, best_M2 = run_gradient_descent(
                    best_a, n, S, gd2_time,
                    initial_lr=0.05, initial_tau=0.5,
                    tau_decay=0.9999, lr_decay=0.9999
                )
                if best_M2 < best_M:
                    best_M = best_M2
                    best_a = best_a2
                    if best_M < best_score:
                        best_score = best_M
                        best_global = best_a.copy()

        print(f"  Final for {name}: score={best_score:.6f}")

    # Final: verify the best solution
    if best_global is None:
        # Fallback: use the ramp
        n = 5000
        a = ramp_init(n)
        S = np.sqrt(2.0 * n)
        a = a / np.sum(a) * S
        best_global = a
        best_score = np.max(compute_convolution(a))
        print(f"Fallback ramp: score={best_score:.6f}")

    # Do one final evaluation
    final_score = evaluate_sequence(best_global.tolist())
    print(f"\nFinal score: {final_score:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")

    return [float(x) for x in best_global]
