import numpy as np
import time

def fft_convolve(a, b):
    """Compute full convolution using FFT."""
    n_out = len(a) + len(b) - 1
    size = 1
    while size < n_out:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n_out]

def compute_conv_and_gradient(a, beta):
    """Compute self-convolution and gradient of soft-max smoothed peak."""
    n = len(a)
    conv = fft_convolve(a, a)
    # Numerically stable soft-max
    c_max = np.max(conv)
    exp_bc = np.exp(beta * (conv - c_max))
    sum_exp = np.sum(exp_bc)
    w = exp_bc / sum_exp
    # Gradient: dM/da_i = 2 * sum_j w[i+j] * a[j]
    # = 2 * conv(w, a_flipped)[n-1 : 2n-1]
    a_flipped = a[::-1]
    w_conv_af = fft_convolve(w, a_flipped)
    grad = 2.0 * w_conv_af[n-1:2*n-1]
    return conv, w, grad, c_max

def normalize_sequence(a, target_sum):
    """Normalize sequence to have the given sum, maintaining non-negativity."""
    s = np.sum(a)
    if s < 1e-10:
        return np.full(len(a), target_sum / len(a))
    return a * (target_sum / s)

def make_random_initializer(n, rng, target_sum, style='smooth'):
    """Create a diverse random initial sequence."""
    if style == 'smooth':
        # Sum of random sinusoids
        a = np.ones(n)
        for _ in range(rng.integers(3, 8)):
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2*np.pi)
            amp = rng.uniform(0.1, 0.5)
            a += amp * np.cos(2*np.pi*freq*np.arange(n)/n + phase)
    elif style == 'scrambled':
        # Random permutation of sorted values
        vals = np.linspace(0.5, 1.5, n)
        a = vals[rng.permutation(n)]
    elif style == 'two_peak':
        # Two Gaussian bumps
        c1 = rng.uniform(0.1*n, 0.4*n)
        c2 = rng.uniform(0.6*n, 0.9*n)
        s1 = rng.uniform(0.05*n, 0.15*n)
        s2 = rng.uniform(0.05*n, 0.15*n)
        x = np.arange(n)
        a = np.exp(-0.5*((x-c1)/s1)**2) + np.exp(-0.5*((x-c2)/s2)**2)
    elif style == 'power':
        # Power law
        x = np.arange(n) + 1
        a = x**rng.uniform(-0.5, 0.5)
    else:
        a = np.ones(n)
    
    a = np.clip(a, 0.01, None)
    return normalize_sequence(a, target_sum)
