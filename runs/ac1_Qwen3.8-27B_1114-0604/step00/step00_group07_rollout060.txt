def lp_direction(x, Mx, K, U):
    n=len(x)
    A = np.zeros((len(K), n))
    for row,k in enumerate(K):
        start=max(0,k-n+1); end=min(k,n-1)
        # (x*y)_k = sum_{i=start}^end x[i] * y[k-i]
        # For column j (y_j), k-j=i -> i=k-j, x_{k-j}. So A[row,j]=x[k-j] valid.
        vals = x[k-end:k-start+1][::-1]  # length end-start+1

    bounds = [(0, U)]*n

    if res.status!=0: return None
    y=res.x
    s=y.sum()
    if s<=1e-6: return None
    y = y*(n/s)
    return y, s

def active_lp_direction(x,C,K):
    if linprog is None: return None
    n=x.size; kc=C.size
    if kc<=K: idx=np.arange(kc)
    else: idx=np.argpartition(C, -K)[-K:]
    A=np.zeros((len(idx), n))
    for r,k in enumerate(idx):
        k=int(k)
        start=max(0,k-n+1); end=min(k,n-1)
        if end>=start:
            A[r,start:end+1] = x[k-end:k-start+1][::-1]
    b_ub=np.full(len(idx), float(C.max()))
    cobj=-np.ones(n)
    maxx=float(np.max(x)) if n>0 else 1.0
    U=max(3.0, 1.3*maxx)
    try:
       bounds=[(0.0,U)]*n
       res=linprog(cobj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

       return None
    if res is None or res.status != 0:

    y=np.asarray(res.x,float)
    y=np.where(np.isfinite(y), y, 0.0)
    y=np.maximum(y,0.0)
    s=float(np.sum(y))
    if s <= 1e-9:

    y = y * (n/s)
    # if y not much different, return None
    normx=float(np.linalg.norm(x))
    if normx > 1e-12 and float(np.linalg.norm(y-x)) < 1e-3*normx:

    return y

```python
"""
Heuristic search for low self-convolution nonnegative sequences.

The score is scale invariant.  We optimize x with sum(x)=n, where the returned
sequence is p=x/n.  The local objective is 2*max(conv(x,x))/n.  The main loop
uses smoothed-max subgradient/Adam updates with FFT-based gradients, active-LP
direction steps, and random restarts.  A smaller full-LP stage is used to obtain
a good shape for up-sampling when time permits.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def normalize_x(x):
    """Make x finite, positive, and sum to len(x)."""
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return np.ones(1, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 1e-8)
    s = float(np.sum(x))
    if not np.isfinite(s) or s <= 0.0:
        x = np.ones(x.size, dtype=np.float64)
        s = float(x.size)
    return x * (float(x.size) / s)


def upsample_arr(y, m):
    """Linearly resample y to length m."""
    y = np.asarray(y, dtype=np.float64).ravel()
    y = y[np.isfinite(y)]
    if y.size == 0:
        return np.ones(m, dtype=np.float64)
    if y.size == m:
        return y.copy()
    old = np.arange(y.size, dtype=np.float64)
    new = np.linspace(0.0, y.size - 1.0, m, dtype=np.float64)
    return np.interp(new, old, y)


def get_global_sequence():
    """Return the provided global construction if available and valid."""
    try:
        hs = height_sequence_1
    except Exception:
        return None
    if hs is None:
        return None
    try:
        arr = np.asarray(hs)
        if np.iscomplexobj(arr):
            arr = arr.real
        arr = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = arr[np.isfinite(arr)]
    if arr.size == 0:
        return None
    arr = np.clip(arr, 0.0, 1000.0)
    if float(np.sum(arr)) <= 0.0:
        return None
    return arr


def conv_x(x):
    """Linear convolution x*x, clipped to nonnegative."""
    n = int(x.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n <= 1024:
        c = np.convolve(x, x)
    else:
        need = 2 * n - 1
        L = 1
        while L < need:
            L <<= 1
        fx = np.fft.rfft(x, L)
        c = np.fft.irfft(fx * fx, L)[:need]
    c = np.asarray(c, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    return np.maximum(c, 0.0)


def score_x(x):
    """Local score for x with sum(x)=n: 2*max(conv(x,x))/n."""
    n = int(x.size)
    if n == 0:
        return float("inf")
    c = conv_x(x)
    if c.size == 0:
        return float("inf")
    return 2.0 * float(np.max(c)) / float(n)


def weighted_gradient_fft(x, w):
    """
    Gradient of sum_k w_k * (x*x)_k with respect to x.
    Returns 2 * (cross-correlation-like transform) for each x_i.
    """
    n = int(x.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    need = 3 * n + 1
    L = 1
    while L < need:
        L <<= 1

    Wp = np.zeros(L, dtype=np.float64)
    m = min(int(w.size), L)
    if m > 0:
        Wp[:m] = w[:m]

    Xr = np.zeros(L, dtype=np.float64)
    Xr[0] = x[0]
    if n > 1:
        Xr[L - n + 1:L] = x[n - 1:0:-1]

    g = np.fft.irfft(np.fft.rfft(Wp) * np.fft.rfft(Xr), L)[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    return 2.0 * g


def make_candidate_x(n, rng):
    """Create a diverse set of normalized starts x with sum(x)=n."""
    n = int(max(1, n))
    starts = []
    i = np.arange(n, dtype=np.float64)

    # Basic monotone ramp.
    starts.append(normalize_x(i + 1.0))

    # Padded monotone family.
    pairs = (
        (1.0, 0.0),
        (1.2, 0.0),
        (0.8, 0.0),
        (1.0, 0.25),
        (1.0, 0.5),
        (1.2, 0.5),
        (0.8, 0.5),
        (1.0, 0.75),
        (1.5, 0.25),
        (0.7, 0.25),
    )
    for a, b in pairs:
        x = np.power(i + 1.0, a) * np.power(float(n + 1.0) - i, b)
        starts.append(normalize_x(x))

    # Bounded antisymmetric sinusoid.
    starts.append(normalize_x(1.0 + 0.90 * np.sin(2.0 * np.pi * i / float(n))))

    # Bounded antisymmetric ramp.
    if n > 1:
        starts.append(normalize_x(1.0 + 0.85 * (2.0 * i / float(n - 1) - 1.0)))
    else:
        starts.append(normalize_x(np.ones(1)))

    # Smooth positive random perturbations.
    wl = max(3, n // 100)
    if wl % 2 == 0:
        wl += 1
    win = np.ones(wl, dtype=np.float64)
    win /= float(wl)
    for sigma in (0.30, 0.60):
        r = np.exp(rng.normal(0.0, 0.25, n))
        r = np.convolve(r, win, mode="same")
        r = np.asarray(r, dtype=np.float64)
        mu = float(np.mean(r))
        sd = float(np.std(r))
        if sd > 1e-12:
            r = (r - mu) / sd
        else:
            r = np.zeros_like(r)
        starts.append(normalize_x(1.0 + sigma * r))

    # Resample the previous global construction, if useful.
    g = get_global_sequence()
    if g is not None and g.size > 0:
        starts.append(normalize_x(upsample_arr(g, n)))

    return starts


def build_conv_matrix(v):
    """Dense matrix A such that (A*y)_k = sum_j v_{k-j} y_j."""
    n = int(v.size)
    if n == 0:
        return np.zeros((0, 0), dtype=np.float64)
    k = np.arange(2 * n - 1, dtype=np.int64)[:, None]
    j = np.arange(n, dtype=np.int64)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    dc = np.clip(d, 0, n - 1)
    return np.where(mask, v[dc], 0.0)


def active_lp_direction(x, C, K):
    """
    Heuristic LP direction using the currently most constrained lags.
    Solves max sum(y) subject to (x*y)_k <= Cmax for selected k and y in [0,U].
    """
    if linprog is None:
        return None
    n = int(x.size)
    m = int(C.size)
    if n == 0 or m == 0:
        return None
    K = int(max(1, min(K, m)))
    if m <= K:
        idx = np.arange(m, dtype=np.int64)
    else:
        idx = np.argpartition(C, -K)[-K:].astype(np.int64)

    A = np.zeros((len(idx), n), dtype=np.float64)
    mx = float(np.max(x)) if n else 1.0
    U = max(3.0, 1.3 * mx)
    lim = float(C.max())

    for r, kk in enumerate(idx):
        kk = int(kk)
        start = max(0, kk - n + 1)
        end = min(kk, n - 1)
        if end >= start:
            A[r, start:end + 1] = x[kk - end:kk - start + 1][::-1]

    try:
        bounds = [(0.0, U)] * n
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(len(idx), lim),
                      bounds=bounds, method="highs")
    except Exception:
        return None

    if res is None or res.status != 0:
        return None

    y = np.asarray(res.x, dtype=np.float64)
    y = np.where(np.isfinite(y), y, 0.0)
    y = np.maximum(y, 0.0)
    s = float(np.sum(y))
    if s <= 1e-10:
        return None
    y *= float(n) / s

    nx = float(np.linalg.norm(x))
    if nx > 1e-12 and float(np.linalg.norm(y - x)) < 1e-3 * nx:
        return None
    return y


def optimize_lp_x(x, deadline, seed=0):
    """
    Full-constraint LP iteration for moderate n.
    Works in the x scale, sum(x)=n.
    """
    x = normalize_x(x)
    n = int(x.size)
    if n == 0 or linprog is None:
        return x, score_x(x)

    best_x = x.copy()
    best_score = score_x(x)
    rng = np.random.default_rng(seed)

    extremes = (-np.ones(n),)  # unused, just to avoid accidental lambda-like constructs
    cobj = extremes[0]

    A = build_conv_matrix(x)
    U = max(5.0, 1.5 * float(np.max(x)))
    no_imp = 0
    it = 0

    while time.monotonic() < deadline - 1.0:
        C = conv_x(x)
        lim = float(C.max())
        current = 2.0 * lim / float(n)

        try:
            b_ub = np.full(C.size, lim)
            bounds = [(0.0, U)] * n
            res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
        except Exception:
            break

        if res is None or res.status != 0:
            break

        y = np.asarray(res.x, dtype=np.float64)
        y = np.where(np.isfinite(y), y, 0.0)
        y = np.maximum(y, 0.0)
        s = float(np.sum(y))
        if s <= 1e-10:
            break
        y *= float(n) / s

        accepted = False
        for mix_t in (0.20, 0.10, 0.30, 0.05):
            x_new = (1.0 - mix_t) * x + mix_t * y
            x_new = np.maximum(x_new, 1e-7)
            x_new = normalize_x(x_new)
            s_new = score_x(x_new)
            if s_new < current - 1e-12:
                x = x_new
                current = s_new
                accepted = True
                break

        if accepted:
            if current < best_score:
                best_score = current
                best_x = x.copy()
                no_imp = 0
            else:
                no_imp = 1
            A = build_conv_matrix(x)
            U = max(5.0, 1.5 * float(np.max(x)))
            it += 1
        else:
            no_imp += 1
            if no_imp > 25:
                p = best_x.copy()
                p *= np.exp(rng.normal(0.0, 0.08, n))
                if no_imp > 50:
                    ramp = np.arange(1, n + 1, dtype=np.float64)
                    p = 0.8 * p + 0.2 * ramp
                p = normalize_x(p)
                x = p
                A = build_conv_matrix(x)
                U = max(5.0, 1.5 * float(np.max(x)))
                no_imp = 0
                it = 0

    return best_x, best_score


def optimize_main(x, deadline, seed=0, do_lp=True):
    """
    Main scalable optimizer: smoothed-max Adam on x, with active-LP steps and restarts.
    """
    x = normalize_x(x)
    n = int(x.size)
    if n == 0:
        return x, score_x(x)

    best_x = x.copy()
    best_score = score_x(x)
    rng = np.random.default_rng(seed)

    m1 = np.zeros(n, dtype=np.float64)
    v1 = np.zeros(n, dtype=np.float64)
    adam_t = 0

    lr = 0.10
    temp = 0.08
    no_imp = 0
    it = 0
    perturb_count = 0
    last_lp = -100
    Klp = 40 if n > 4000 else 60

    while time.monotonic() < deadline - 2.0:
        C = conv_x(x)
        c = C / float(n)
        mx = float(C.max())
        current = 2.0 * mx / float(n)

        if current < best_score:
            best_score = current
            best_x = x.copy()

        if no_imp > 70:
            sigmas = (0.10, 0.20, 0.35, 0.60)
            sigma = float(sigmas[perturb_count % 4])
            x = best_x.copy()
            x *= np.exp(rng.normal(0.0, sigma, n))
            if sigma >= 0.35:
                ramp = np.arange(1, n + 1, dtype=np.float64)
                x = 0.75 * x + 0.25 * ramp
            x = normalize_x(x)
            m1 = np.zeros(n, dtype=np.float64)
            v1 = np.zeros(n, dtype=np.float64)
            adam_t = 0
            lr = 0.08
            temp = 0.10
            no_imp = 0
            perturb_count += 1

        # Choose active/smooth weights.
        if it < 60:
            R = 50
            if c.size <= R:
                weights = c.astype(np.float64)
            else:
                idx = np.argpartition(c, -R)[-R:].astype(np.int64)
                weights = np.zeros(c.size, dtype=np.float64)
                weights[idx] = c[idx]
            ws = float(np.sum(weights))
            if not np.isfinite(ws) or ws <= 0.0:
                idx = int(np.argmax(c))
                weights = np.zeros(c.size, dtype=np.float64)
                weights[idx] = 1.0
                ws = 1.0
            weights /= ws
        else:
            shifted = c - c.max()
            weights = np.exp(shifted / temp)
            ws = float(np.sum(weights))
            if not np.isfinite(ws) or ws <= 0.0:
                idx = int(np.argmax(c))
                weights = np.zeros(c.size, dtype=np.float64)
                weights[idx] = 1.0
                ws = 1.0
            weights /= ws

        G = weighted_gradient_fft(x, weights)
        if not np.any(G > 0.0):
            break

        adam_t += 1
        b1 = 0.90
        b2 = 0.99
        m1 = b1 * m1 + (1.0 - b1) * G
        v1 = b2 * v1 + (1.0 - b2) * (G * G)
        mh = m1 / (1.0 - b1 ** adam_t)
        vh = v1 / (1.0 - b2 ** adam_t)
        step = mh / (np.sqrt(vh) + 1e-12)
        step = np.clip(step, -2.0, 2.0)

        accepted = False
        for mult in (1.0, 0.35, 0.12):
            x_new = x - lr * mult * step
            x_new = np.maximum(x_new, 1e-7)
            x_new = normalize_x(x_new)
            s_new = score_x(x_new)
            if s_new < current - 1e-12:
                x = x_new
                current = s_new
                accepted = True
                break

        if accepted:
            if current < best_score:
                best_score = current
                best_x = x.copy()
                no_imp = 0
                if lr < 0.15:
                    lr *= 1.01
                temp = max(0.02, temp * 0.998)
            else:
                no_imp = 1
            it += 1
        else:
            no_imp += 1
            lr = max(0.005, lr * 0.85)
            temp = min(0.35, temp * 1.10)

        # Active LP direction, used as a stronger non-convex move.
        if (do_lp and linprog is not None and n <= 12000 and
                it - last_lp >= 20 and (it % 20 == 0 or no_imp > 20) and
                time.monotonic() < deadline - 35.0):
            y = active_lp_direction(x, C, Klp)
            last_lp = it
            if y is not None:
                lp_accepted = False
                for mix_t in (0.20, 0.10, 0.30, 0.05):
                    x_new = (1.0 - mix_t) * x + mix_t * y
                    x_new = np.maximum(x_new, 1e-7)
                    x_new = normalize_x(x_new)
                    s_new = score_x(x_new)
                    if s_new < current - 1e-12:
                        x = x_new
                        current = s_new
                        lp_accepted = True
                        break
                if lp_accepted:
                    if current < best_score:
                        best_score = current
                        best_x = x.copy()
                        no_imp = 0
                    m1 = np.zeros(n, dtype=np.float64)
                    v1 = np.zeros(n, dtype=np.float64)
                    adam_t = 0
                    it += 1
                else:
                    no_imp += 1

    return best_x, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a nonnegative sequence minimizing
    evaluate_sequence(x / sum(x)).

    Strategy:
      1. Optional small full-LP stage (n=512) to get a useful shape.
      2. Main stage on a larger n using smoothed-max Adam + active LP steps.
      3. Final direct verification with evaluate_sequence.
    """
    start_clock = time.monotonic()
    reserve = 90.0
    total_deadline = start_clock + max(60.0, float(budget_s) - reserve)

    rng = np.random.default_rng(int(seed) if seed is not None else 42)

    best_x = None
    best_score = float("inf")
    small_best_x = None
    small_best_score = float("inf")
    final_candidates = []

    # -------------------------
    # Stage 1: small full-LP
    # -------------------------
    if time.monotonic() + 90.0 < total_deadline:
        n_s = 512
        starts_s = make_candidate_x(n_s, rng)
        scores_s = np.empty(len(starts_s), dtype=np.float64)
        for ii, sx in enumerate(starts_s):
            scores_s[ii] = score_x(sx)
        order_s = np.argsort(scores_s)
        top_s = order_s[:min(2, len(order_s))]

        # Ensure monotone ramp is represented.
        if 0 not in list(top_s):
            top_s[-1] = 0

        for ii, idx in enumerate(list(top_s)):
            if time.monotonic() + 30.0 >= total_deadline:
                break
            chunk = min(80.0, max(20.0, (total_deadline - time.monotonic() - 60.0) / max(1, len(top_s) - ii)))
            x0 = starts_s[idx].copy()
            x_opt, s_opt = optimize_lp_x(x0, time.monotonic() + chunk, seed=int(seed) + 101 + ii)
            x_opt = normalize_x(x_opt)
            s_opt = score_x(x_opt)
            if s_opt < best_score:
                best_score = s_opt
                best_x = x_opt.copy()
            if s_opt < small_best_score:
                small_best_score = s_opt
                small_best_x = x_opt.copy()
            print(f"small LP n={n_s} start={idx} score={s_opt:.8f} elapsed={time.monotonic()-start_clock:.1f}")

    # -------------------------
    # Stage 2: main larger n
    # -------------------------
    remaining = max(0.0, total_deadline - time.monotonic())
    if remaining > 1200.0:
        n_main = 6733
    elif remaining > 600.0:
        n_main = 4096
    else:
        n_main = 2048

    starts_m = make_candidate_x(n_main, rng)

    if small_best_x is not None:
        starts_m.append(normalize_x(upsample_arr(small_best_x, n_main)))

    scores_m = np.empty(len(starts_m), dtype=np.float64)
    for ii, sx in enumerate(starts_m):
        scores_m[ii] = score_x(sx)
    order_m = np.argsort(scores_m)

    selected = list(order_m[:min(4, len(order_m))])
    if 0 not in selected:
        selected[-1] = 0
    selected = selected[:3]

    if time.monotonic() + 30.0 < total_deadline:
        per = max(60.0, (total_deadline - time.monotonic() - 45.0) / max(1, len(selected)))
        for ii, idx in enumerate(selected):
            if time.monotonic() + 45.0 >= total_deadline:
                break
            chunk = min(per, max(20.0, (total_deadline - time.monotonic() - 45.0) / max(1, len(selected) - ii)))
            x0 = starts_m[idx].copy()
            x_opt, s_opt = optimize_main(x0, time.monotonic() + chunk, seed=int(seed) + 201 + ii, do_lp=True)
            x_opt = normalize_x(x_opt)
            s_opt = score_x(x_opt)
            print(f"main n={n_main} start={idx} local_score={s_opt:.8f} elapsed={time.monotonic()-start_clock:.1f}")
            if s_opt < best_score:
                best_score = s_opt
                best_x = x_opt.copy()
            final_candidates.append(x_opt.copy())

    # -------------------------
    # Stage 3: polish global best
    # -------------------------
    if best_x is not None and time.monotonic() + 120.0 < total_deadline:
        x0 = best_x.copy()
        x_opt, s_opt = optimize_main(x0, total_deadline - 60.0, seed=int(seed) + 301, do_lp=True)
        x_opt = normalize_x(x_opt)
        s_opt = score_x(x_opt)
        print(f"polish global local_score={s_opt:.8f} elapsed={time.monotonic()-start_clock:.1f}")
        if s_opt < best_score:
            best_score = s_opt
            best_x = x_opt.copy()
        final_candidates.append(x_opt.copy())

    # -------------------------
    # Final direct verification
    # -------------------------
    if best_x is None:
        best_x = normalize_x(np.arange(1, 1024, dtype=np.float64))

    final_candidates.append(best_x.copy())
    final_candidates.append(normalize_x(np.arange(1, max(128, n_main), dtype=np.float64)))
    if small_best_x is not None:
        final_candidates.append(normalize_x(upsample_arr(small_best_x, n_main)))

    # Keep only a few candidates.
    seen = 0
    trimmed = []
    candidate_limit = 4
    for cand in final_candidates:
        if cand is None:
            continue
        cand = normalize_x(cand)
        # Deduplicate by cheap size/sum/first values.
        dup = False
        for old in trimmed[-5:]:
            if old.size == cand.size and np.allclose(cand[:max(1, old.size // 100)], old[:max(1, old.size // 100)], rtol=1e-12, atol=1e-15):
                dup = True
                break
        if not dup:
            trimmed.append(cand)
        if len(trimmed) >= candidate_limit:
            break

    best_final_seq = None
    best_direct = float("inf")

    for cand in trimmed:
        if cand is None or cand.size == 0:
            continue
        p = cand / float(cand.size)
        seq = [float(v) for v in p]
        try:
            sc = float(evaluate_sequence(seq))
        except Exception:
            sc = float("inf")
        print(f"final candidate n={cand.size} direct_score={sc:.8f}")
        if np.isfinite(sc) and sc < best_direct:
            best_direct = sc
            best_final_seq = seq

    if best_final_seq is None:
        p = best_x / float(best_x.size)
        best_final_seq = [float(v) for v in p]

    return best_final_seq
```
