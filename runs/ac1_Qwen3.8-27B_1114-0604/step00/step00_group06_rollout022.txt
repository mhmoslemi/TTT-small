    n = len(a)
    c = np.convolve(a, a)
    S = np.sum(a)
    
    # Find top-k peaks
    top_indices = np.argsort(c)[-top_k:]
    top_values = c[top_indices]
    # Weights: exponential of (value - max)
    max_val = top_values[-1]
    weights = np.exp((top_values - max_val) * n / 10.0)  # tune the sensitivity
    weights = weights / weights.sum()

    g = np.zeros(n)

def _subgrad_and_step(a, eta, beta=0.7, velocity=None, top_lags=20, rel_tol=0.005):
    """
    Compute subgradient and take a step.
    Returns: (a_new, velocity_new, g_norm)
    """
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    
    # Find near-peak lags (within rel_tol of max)
    threshold = M * (1.0 - rel_tol)
    peak_lags = np.where(c >= threshold)[0]
    
    # Limit to top_lags if too many
    if len(peak_lags) > top_lags:
        # Keep the top ones
        peak_values = c[peak_lags]
        order = np.argsort(peak_values)[-top_lags:]
        peak_lags = peak_lags[order]
    
    # Compute averaged subgradient
    g = np.zeros(n)
    for k in peak_lags:
        partner = k - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        g[mask] += 2.0 * a[partner[mask]]  # factor of 2 from the quadratic
    g /= len(peak_lags)

    g -= g.mean()

        velocity = np.zeros(n)

import numpy as np
import time

"""
Hybrid subgradient optimization with momentum to minimize the normalized
peak autocorrelation of non-negative sequences. Uses near-peak averaging,
adaptive step sizes, periodic restarts, and direct numpy score computation
for speed. The score 2*n*max(conv(a,a))/(sum(a))^2 is scale-invariant, so
we fix sum(a)=n and minimize max(conv(a,a)).
"""


def _fast_score(a):
    """Fast score computation matching evaluate_sequence for valid inputs."""
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * M / (S * S))


def _compute_averaged_subgradient(a, rel_tol=0.005, max_lags=30):
    """
    Compute the averaged subgradient over near-peak autocorrelation lags.
    Returns (g, M, num_active) where g is projected to sum=0.
    """
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()

    threshold = M * (1.0 - rel_tol)
    peak_lags = np.where(c >= threshold)[0]

    if len(peak_lags) > max_lags:
        peak_values = c[peak_lags]
        order = np.argsort(peak_values)[-max_lags:]
        peak_lags = peak_lags[order]

    if len(peak_lags) == 0:
        peak_lags = np.array([c.argmax()])

    # Accumulate subgradients: dq_k/da[j] = 2*a[k-j] (with a=0 outside [0,n))
    g = np.zeros(n)
    for k in peak_lags:
        partner = k - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        g[mask] += a[partner[mask]]

    g /= len(peak_lags)
    # Project onto sum=0 direction (since score is scale-invariant)
    g -= g.mean()

    return g, M, len(peak_lags)


def _try_multiplicative_update(a, k_star, eta):
    """
    Multiplicative (Boltzmann) update to reduce the peak at lag k_star.
    Reduces elements with large 'partners' at the peak lag.
    """
    n = len(a)
    partner = k_star - np.arange(n)
    mask = (partner >= 0) & (partner < n)
    pressure = np.zeros(n)
    pressure[mask] = a[partner[mask]]

    max_p = pressure.max()
    if max_p < 1e-10:
        return a.copy()

    a_new = a * np.exp(-eta * pressure / max_p)
    a_new = np.maximum(a_new, 1e-8)
    a_new = a_new / a_new.sum() * n
    return a_new


def _make_initial_sequence(base, n, rng):
    """Create a non-constant initial sequence by modulating the base."""
    t = np.arange(n, dtype=np.float64)
    # Multiple sinusoidal components to break symmetry richly
    f1 = rng.integers(2, 15)
    f2 = rng.integers(15, 40)
    phi1 = rng.uniform(0, 2 * np.pi)
    phi2 = rng.uniform(0, 2 * np.pi)
    amp1 = rng.uniform(0.1, 0.25)
    amp2 = rng.uniform(0.03, 0.1)
    mod = 1.0 + amp1 * np.sin(2 * np.pi * f1 * t / n + phi1) + amp2 * np.sin(2 * np.pi * f2 * t / n + phi2)
    a = base * mod
    a = np.maximum(a, 0.01)
    a = a / a.sum() * n
    return a

    Returns (best_sequence, best_score).
    """
    best = a.copy()

    print(f"  [{label}] Start: score={best_score:.6f}")

    eta_min = 1e-5

                a = np.maximum(a, 0.01)

                a = np.maximum(a, 0.01)
            else:
                a = _make_initial_sequence(np.ones(n), n, rng)

            eta = 0.5
            no_improve = 0
