```
def _gradient_top(a, c, beta, max_active):
    n=a.size
    if c.size==0: return None
    cmax = float(np.max(c))
    if not np.isfinite(cmax): return None
    # get top max_active indices
    if n == 0: return None
    if c.size > max_active:
       idx = np.argpartition(c, -max_active)[-max_active:]
    else:
       idx = np.arange(c.size)
    # If beta small, maybe include more? ok.
    z = c[idx] - cmax
    # For beta large, exp ok; beta maybe 0? handle
    if beta <= 0:
       w = np.full(idx.size, 1.0/idx.size)
    else:
       w = np.exp(np.clip(beta * z, -50.0, 0.0))
       ws = float(np.sum(w))
       if ws <= 0 or not np.isfinite(ws):
           w = np.full(idx.size, 1.0/idx.size)
       else:
           w = w / ws
    h = np.zeros(n, dtype=np.float64)
    for ii in range(idx.size):
       kk = int(idx[ii]); ww = float(w[ii])
       if ww == 0: continue
       lo = kk - n + 1
       if lo < 0: lo = 0
       hi = kk
       if hi > n - 1: hi = n - 1
       if lo <= hi:
           # a indices from kk-lo down to kk-hi
           seg = a[kk - lo : kk - hi + 1][::-1]
           h[lo:hi+1] += ww * seg
    return h
```

```
def _resample_sequence(seq, n):
    seq = np.asarray(seq,dtype=np.float64).reshape(-1)
    seq = seq[np.isfinite(seq)]
    if seq.size == 0: return None
    seq = np.clip(seq,0,1000)
    if seq.size == n: return seq
    xf = np.linspace(0.0,1.0,seq.size,endpoint=False)
    xn = np.linspace(0.0,1.0,n,endpoint=False)
    out = np.interp(xn, xf, seq)
    return out
```

```
def _random_z_inits(n, rng, count):
```

```
def _sota_direction(a, target_sum, rng, max_time=90.0, maxK=600):
    n = int(a.size)
    if n <=0: return None
    S = float(target_sum)
    floor = max(1e-12, 1e-12 * S / n)
    f = np.asarray(a, dtype=np.float64).copy()
    f[f < floor] = floor
    f = np.where(np.isfinite(f), f, floor)
    s = float(np.sum(f))
    if s <= 0: return None
    f = f * (S / s)
    conv_f = _fft_conv_full(f)
    M = float(np.max(conv_f))
    if not np.isfinite(M) or M <= 0: return None
    Lc = 2*n - 1
    topn = min(200, Lc)
    if Lc > topn:
       top_self = np.argpartition(conv_f, -topn)[-topn:].astype(np.int64)
    else:
       top_self = np.arange(Lc, dtype=np.int64)
    viol = np.array([], dtype=np.int64)
    randn = min(50, Lc)
    rand_init = rng.integers(0, Lc, randn).astype(np.int64) if randn > 0 else np.array([], dtype=np.int64)
    b_upper = max(50.0 * S, 1000.0)
    b = None
    t0 = time.monotonic()
    for inner in range(4):
       if time.monotonic() - t0 > max_time: break
       mself = min(250, maxK // 2)
       mvmax = maxK - mself - 1
       if mvmax < 10: mvmax = 10
       parts = []
       parts.append(top_self[:mself])
       if viol.size > 0:
          parts.append(viol[:mvmax])
       parts.append(np.array([n - 1], dtype=np.int64))
       # add rand if room
       # compute current unique size? simple add rand[:10]
       if rand_init.size > 0:
          parts.append(rand_init[:10])
       K = np.unique(np.concatenate(parts)).astype(np.int64)
       if K.size > maxK:
          K = K[:maxK]
       sol = _solve_lp_for_f(f, M, K, b_upper)
       if sol is None:
          sol = _heuristic_b(f, M, n)
       b = np.asarray(sol, dtype=np.float64)
       if b.size != n: b = np.zeros(n)
       b = np.clip(b, 0.0, b_upper)
       cross = _fft_product_conv(f, b)
       if cross.size == 0: break
       peak = float(np.max(cross))
       if not np.isfinite(peak) or peak <= 0:
          b = np.ones(n, dtype=np.float64) * (M / max(float(np.max(f)), 1e-12))
          continue
       viol_mask = cross > M * (1.0 + 1e-7)
       if np.any(viol_mask):
          viol_idx = np.where(viol_mask)[0].astype(np.int64)
          vtopn = min(300, viol_idx.size)
          if viol_idx.size > vtopn:
             viol = viol_idx[np.argpartition(cross[viol_idx], -vtopn)[-vtopn:]]
          else:
             viol = viol_idx
       else:
          viol = np.array([], dtype=np.int64)
       b = b * (M / peak)
    if b is None:
       b = _heuristic_b(f, M, n)
    cross = _fft_product_conv(f, b)
    if cross.size:
       peak = float(np.max(cross))
       if np.isfinite(peak) and peak > 0:
          b = b * (M / peak)
    sumb = float(np.sum(b))
    if not np.isfinite(sumb) or sumb <= 0:
       return None
    g = b * (S / sumb)
    g = np.clip(g, 0.0, 1000.0)
    sg = float(np.sum(g))
    if not np.isfinite(sg) or sg <= 0:
       return None
    g = g * (S / sg)
    return g
```

```python
"""
Search for nonnegative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2.

The program mainly uses:
  1. log-parameterized local search (a = softmax(z) * scale),
  2. a top-lag subgradient of the max-convolution objective,
  3. inverse-gradient / water-filling directions,
  4. an approximate state-of-the-art LP update:
       maximize sum(b) subject to f*b <= M, b >= 0,
     solved by spaced LP constraint generation with HiGHS when available,
  5. symmetric/inverse restarts and annealing to avoid stationary starts.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _next_pow2(m: int) -> int:
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def _get_initial_sequence():
    obj = globals().get("height_sequence_1")
    if obj is None:
        return None
    try:
        seq = np.asarray(obj, dtype=np.float64)
    except Exception:
        return None
    if seq.ndim != 1:
        seq = seq.reshape(-1)
    if seq.size == 0:
        return None
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    if not np.all(np.isfinite(seq)):
        return None
    return seq


def _resample_sequence(seq, n: int):
    try:
        seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    seq = seq[np.isfinite(seq)]
    if seq.size == 0:
        return None
    seq = np.clip(seq, 0.0, 1000.0)
    if seq.size == int(n):
        return seq.copy()
    old = np.linspace(0.0, 1.0, int(seq.size), endpoint=False)
    new = np.linspace(0.0, 1.0, int(n), endpoint=False)
    return np.interp(new, old, seq)


def _normalize_sum(a, target_sum: float):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return None
    return a * (float(target_sum) / s)


def _fft_conv_full(a):
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        v = float(a[0])
        return np.array([v * v], dtype=np.float64)
    m = 2 * n - 1
    L = _next_pow2(m)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    return np.real(c[:m])


def _fft_product_conv(a, b):
    n = int(a.size)
    m = int(b.size)
    if n == 0 or m == 0:
        return np.array([], dtype=np.float64)
    L = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    c = np.fft.irfft(fa * fb, L)
    return np.real(c[:n + m - 1])


def _score(a) -> float:
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.01:
        return np.inf
    c = _fft_conv_full(a)
    if c.size == 0:
        return np.inf
    mc = float(np.max(c))
    if not np.isfinite(mc):
        return np.inf
    return float(2.0 * a.size * mc / (s * s))


def _softmax_scale(z, target_sum: float):
    z = np.asarray(z, dtype=np.float64).reshape(-1)
    z = np.where(np.isfinite(z), z, 0.0)
    zm = float(np.max(z)) if z.size else 0.0
    z = z - zm
    z = np.clip(z, -50.0, 0.0)
    p = np.exp(z)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 0.0:
        return None
    return p * (float(target_sum) / s)


def _z_from_a(a, limit: float = 40.0):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    tiny = np.finfo(np.float64).tiny
    ap = np.maximum(a, tiny)
    z = np.log(ap)
    z = z - float(np.mean(z))
    return np.clip(z, -float(limit), float(limit))


def _random_z_inits(n: int, rng, count: int):
    out = []
    n = int(n)
    if n <= 0:
        return out

    out.append(np.zeros(n, dtype=np.float64))

    i = np.arange(n, dtype=np.float64)
    L = _next_pow2(2 * n)
    freqs = np.arange(L // 2 + 1, dtype=np.float64)

    for _ in range(int(count)):
        z = np.zeros(n, dtype=np.float64)
        ncomp = int(rng.integers(2, 7))
        for __ in range(ncomp):
            f = int(rng.integers(1, 11))
            amp = float(rng.uniform(0.25, 1.6))
            phase = float(rng.uniform(0.0, 2.0 * math.pi))
            z += amp * np.cos(2.0 * math.pi * f * i / float(n) + phase)

        cutoff = float(rng.uniform(max(1.0, n / 25.0), max(2.0, n / 4.0)))
        mask = 1.0 / (1.0 + (freqs / cutoff) ** 2)
        noise = rng.normal(0.0, 1.0, n)
        nf = np.fft.rfft(noise, L)
        nf *= mask
        z += 0.25 * np.fft.irfft(nf, L)[:n]

        # A few useful one-sided shapes.
        r = rng.random()
        if r < 0.15:
            x = (i - (n - 1) / 2.0) / max(1.0, 0.2 * n)
            z += -0.35 * x * x
        elif r < 0.30:
            x = (i - (n - 1) / 2.0) / max(1.0, 0.25 * n)
            z += 0.25 * np.cos(math.pi * x)
        elif r < 0.45:
            z += 0.3 * i / max(1, n)

        z = z - float(np.mean(z))
        z = np.clip(z, -30.0, 30.0)
        out.append(z)

    return out


def _gradient_top(a, c, beta: float, max_active: int = 96):
    n = int(a.size)
    c = np.asarray(c, dtype=np.float64).reshape(-1)
    if n == 0 or c.size == 0:
        return None

    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return None

    if max_active is None or max_active <= 0:
        max_active = 1
    max_active = int(min(max_active, c.size))

    if c.size > max_active:
        idx = np.argpartition(c, -max_active)[-max_active:]
    else:
        idx = np.arange(c.size)

    if beta <= 0.0:
        w = np.full(idx.size, 1.0 / float(idx.size), dtype=np.float64)
    else:
        z = c[idx] - cmax
        w = np.exp(np.clip(beta * z, -50.0, 0.0))
        ws = float(np.sum(w))
        if not np.isfinite(ws) or ws <= 0.0:
            w = np.full(idx.size, 1.0 / float(idx.size), dtype=np.float64)
        else:
            w = w / ws

    h = np.zeros(n, dtype=np.float64)
    for ii in range(idx.size):
        kk = int(idx[ii])
        ww = float(w[ii])
        if ww == 0.0:
            continue
        lo = kk - n + 1
        if lo < 0:
            lo = 0
        hi = kk
        if hi > n - 1:
            hi = n - 1
        if lo <= hi:
            seg = a[kk - lo:kk - hi + 1][::-1]
            h[lo:hi + 1] += ww * seg
    return h


def _inverse_from_gradient(grad, target_sum: float):
    if grad is None:
        return None
    g = np.asarray(grad, dtype=np.float64).reshape(-1)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    if g.size == 0:
        return None
    umax = float(np.max(g))
    if not np.isfinite(umax) or umax <= 0.0:
        g = np.ones(g.size, dtype=np.float64)
    else:
        g = g / umax
    g = 1.0 / np.maximum(g, 1e-6)
    return _normalize_sum(g, target_sum)


def _heuristic_b(f, M: float, n: int):
    fmax = float(np.max(f)) if f.size else 0.0
    if not np.isfinite(fmax) or fmax <= 0.0:
        fmax = 1e-12
    b = np.full(n, max(M / fmax, 0.0), dtype=np.float64)
    for _ in range(8):
        cross = _fft_product_conv(f, b)
        if cross.size == 0:
            break
        peak = float(np.max(cross))
        if not np.isfinite(peak) or peak <= 0.0:
            break
        if peak <= M * (1.0 + 1e-9):
            break
        b *= (M / peak)
    return b


def _solve_lp_for_f(f, M: float, K, upper: float):
    if linprog is None:
        return None
    n = int(f.size)
    C = int(np.asarray(K).size)
    if n == 0 or C == 0:
        return None

    A = np.empty((C, n), dtype=np.float64)
    K = np.asarray(K, dtype=np.int64).reshape(-1)
    for r in range(C):
        k = int(K[r])
        row = A[r]
        row.fill(0.0)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            row[lo:hi + 1] = f[k - lo:k - hi + 1][::-1]

    cobj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(C, float(M), dtype=np.float64)
    bounds = [(0.0, float(upper))] * n

    for meth in ("highs-ds", "highs"):
        try:
            res = linprog(
                cobj,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method=meth,
                options={"presolve": True},
            )
            if res is not None and getattr(res, "success", False) and res.x is not None:
                x = np.asarray(res.x, dtype=np.float64)
                if x.size == n and np.all(np.isfinite(x)):
                    return np.clip(x, 0.0, float(upper))
        except Exception:
            pass
        try:
            res = linprog(
                cobj,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method=meth,
            )
            if res is not None and getattr(res, "success", False) and res.x is not None:
                x = np.asarray(res.x, dtype=np.float64)
                if x.size == n and np.all(np.isfinite(x)):
                    return np.clip(x, 0.0, float(upper))
        except Exception:
            pass
    return None


def _sota_direction(a, target_sum: float, rng, max_time: float = 60.0, maxK: int = 300):
    n = int(a.size)
    if n <= 0:
        return None
    S = float(target_sum)

    floor = max(1e-12, 1e-12 * S / float(n))
    f = np.asarray(a, dtype=np.float64).reshape(-1).copy()
    f = np.where(np.isfinite(f), f, floor)
    f = np.maximum(f, floor)
    fs = float(np.sum(f))
    if not np.isfinite(fs) or fs <= 0.0:
        return None
    f = f * (S / fs)

    conv_f = _fft_conv_full(f)
    if conv_f.size == 0:
        return None
    M = float(np.max(conv_f))
    if not np.isfinite(M) or M <= 0.0:
        return None

    Lc = 2 * n - 1
    topn = min(200, Lc)
    if Lc > topn:
        top_self = np.argpartition(conv_f, -topn)[-topn:].astype(np.int64)
    else:
        top_self = np.arange(Lc, dtype=np.int64)

    viol = np.array([], dtype=np.int64)
    randn = min(50, Lc)
    if randn > 0:
        rand_init = rng.integers(0, Lc, randn).astype(np.int64)
    else:
        rand_init = np.array([], dtype=np.int64)

    b_upper = max(50.0 * S, 1000.0)
    b = None
    t0 = time.monotonic()

    for _inner in range(3):
        if time.monotonic() - t0 > float(max_time):
            break

        mself = int(max(1, maxK // 2))
        mviol = int(max(0, maxK - mself - 1))

        parts = []
        parts.append(top_self[:mself])
        if viol.size:
            parts.append(viol[:mviol])
        parts.append(np.array([n - 1], dtype=np.int64))
        if rand_init.size:
            parts.append(rand_init[:8])

        K = np.unique(np.concatenate(parts)).astype(np.int64)
        if K.size > maxK:
            K = np.concatenate(
                [
                    top_self[:mself],
                    viol[:mviol],
                    np.array([n - 1], dtype=np.int64),
                ]
            )
            K = np.unique(K).astype(np.int64)

        sol = _solve_lp_for_f(f, M, K, b_upper)
        if sol is None:
            sol = _heuristic_b(f, M, n)
        b = np.asarray(sol, dtype=np.float64).reshape(-1)
        if b.size != n:
            b = np.zeros(n, dtype=np.float64)
        b = np.clip(b, 0.0, b_upper)

        cross = _fft_product_conv(f, b)
        if cross.size == 0:
            break
        peak = float(np.max(cross))
        if not np.isfinite(peak) or peak <= 0.0:
            b = np.ones(n, dtype=np.float64) * (M / max(float(np.max(f)), 1e-12))
            continue

        viol_mask = cross > M * (1.0 + 1e-7)
        if np.any(viol_mask):
            viol_idx = np.where(viol_mask)[0].astype(np.int64)
            vtopn = min(300, viol_idx.size)
            if viol_idx.size > vtopn:
                viol = viol_idx[np.argpartition(cross[viol_idx], -vtopn)[-vtopn:]]
            else:
                viol = viol_idx
        else:
            viol = np.array([], dtype=np.int64)

        b = b * (M / peak)

    if b is None:
        b = _heuristic_b(f, M, n)

    cross = _fft_product_conv(f, b)
    if cross.size:
        peak = float(np.max(cross))
        if np.isfinite(peak) and peak > 0.0:
            b = b * (M / peak)

    sumb = float(np.sum(b))
    if not np.isfinite(sumb) or sumb <= 0.0:
        return None

    g = b * (S / sumb)
    g = np.clip(g, 0.0, 1000.0)
    sg = float(np.sum(g))
    if not np.isfinite(sg) or sg <= 0.0:
        return None
    return g * (S / sg)


def _blend_best(a, g, target_sum: float, current_score: float):
    tv = (0.05, 0.10, 0.20, 0.35, 0.50, 0.75, 1.0)
    best_cand = None
    best_sc = current_score
    for t in tv:
        cand = (1.0 - t) * a + t * g
        cand = _normalize_sum(cand, target_sum)
        if cand is None:
            continue
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_cand = cand
    return best_cand, best_sc


def _full_search(best_a, best_z, best_score: float, target_sum: float, rng, deadline: float):
    n = int(best_a.size)
    work_a = np.asarray(best_a, dtype=np.float64).copy()
    work_z = np.asarray(best_z, dtype=np.float64).copy()
    work_score = float(best_score)

    if not np.isfinite(work_score):
        work_score = np.inf

    best_a = np.asarray(best_a, dtype=np.float64).copy()
    best_z = np.asarray(best_z, dtype=np.float64).copy()
    best_score = float(best_score)
    if not np.isfinite(best_score):
        best_score = np.inf

    beta = 30.0
    T = 0.02
    no_improve = 0
    it = 0
    last_lp = -1.0e30
    alphas = (0.08, 0.16, 0.32, 0.64, 1.28)

    while time.monotonic() < float(deadline):
        it += 1

        if it % 60 == 0:
            beta = min(500.0, beta * 1.15)

        conv = _fft_conv_full(work_a)
        h_raw = _gradient_top(work_a, conv, beta, 96)

        accepted = False
        if h_raw is not None:
            h = h_raw.copy()
            dot = float(np.dot(h, work_a))
            n2 = float(np.dot(work_a, work_a))
            if n2 > 0.0 and np.isfinite(dot):
                h = h - (dot / n2) * work_a

            dz = work_a * h
            dz = dz - float(np.mean(dz))
            rms = float(np.sqrt(np.mean(dz * dz)))
            if np.isfinite(rms) and rms > 1e-14:
                dz = dz / rms
                for alpha in alphas:
                    v = work_z - alpha * dz
                    v = v - float(np.mean(v))
                    v = np.clip(v, -40.0, 40.0)
                    cand = _softmax_scale(v, target_sum)
                    if cand is None:
                        continue
                    sc = _score(cand)
                    if not np.isfinite(sc):
                        continue
                    if sc < work_score - 1e-12:
                        work_z = v
                        work_a = cand
                        work_score = sc
                        accepted = True
                        break
                    if sc < work_score + T and rng.random() < 0.12:
                        work_z = v
                        work_a = cand
                        work_score = sc
                        accepted = True
                        break

        if accepted:
            if work_score < best_score - 1e-12:
                best_a = work_a.copy()
                best_score = work_score
                best_z = work_z.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Cheap inverse-gradient/water-filling move.
        if it % 25 == 0 and h_raw is not None:
            g = _inverse_from_gradient(h_raw, target_sum)
            if g is not None:
                cand, sc = _blend_best(work_a, g, target_sum, work_score)
                if cand is not None and sc < work_score + max(0.02, 0.01 * max(1.0, work_score)):
                    work_a = cand
                    work_score = sc
                    work_z = _z_from_a(work_a)
                    if work_score < best_score - 1e-12:
                        best_a = work_a.copy()
                        best_score = work_score
                        best_z = work_z.copy()
                        no_improve = 0
                    else:
                        no_improve += 1

        # Symmetrization can help, though asymmetry may be optimal; accept only if better.
        if it % 120 == 0 or (no_improve > 25 and it % 12 == 0):
            sym = 0.5 * (work_a + work_a[::-1])
            sym = _normalize_sum(sym, target_sum)
            if sym is not None:
                sc = _score(sym)
                if np.isfinite(sc) and sc < work_score + 0.01:
                    work_a = sym
                    work_score = sc
                    work_z = _z_from_a(work_a)
                    if work_score < best_score - 1e-12:
                        best_a = work_a.copy()
                        best_score = work_score
                        best_z = work_z.copy()
                        no_improve = 0

        # State-of-the-art LP/cross-convolution update, if there is enough time left.
        now = time.monotonic()
        if (
            it % 35 == 0
            and now < float(deadline) - 300.0
            and now - last_lp > 100.0
            and n <= 10000
        ):
            last_lp = now
            g = _sota_direction(work_a, target_sum, rng, max_time=70.0, maxK=300)
            if g is not None:
                cand, sc = _blend_best(work_a, g, target_sum, work_score)
                if cand is not None and sc < work_score + max(0.03, 0.01 * max(1.0, work_score)):
                    work_a = cand
                    work_score = sc
                    work_z = _z_from_a(work_a)
                    if work_score < best_score - 1e-12:
                        best_a = work_a.copy()
                        best_score = work_score
                        best_z = work_z.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
            print("iter", it, "best %.6f work %.6f" % (best_score, work_score))

        # Random restart/perturbation if stuck.
        if no_improve > 55:
            v = work_z + rng.normal(0.0, 0.85, n)
            sym_v = 0.5 * (v + v[::-1])
            use_v = sym_v if rng.random() < 0.35 else v
            use_v = use_v - float(np.mean(use_v))
            use_v = np.clip(use_v, -40.0, 40.0)
            cand = _softmax_scale(use_v, target_sum)
            if cand is not None:
                sc = _score(cand)
                if np.isfinite(sc) and sc < work_score + max(0.08, 0.05 * max(1.0, work_score)):
                    work_a = cand
                    work_score = sc
                    work_z = use_v
                    if work_score < best_score - 1e-12:
                        best_a = work_a.copy()
                        best_score = work_score
                        best_z = work_z.copy()
            no_improve = 0
            beta = 30.0
            T = 0.03

        # If the work sequence drifted badly, reset to the best known point.
        if (
            not np.isfinite(work_score)
            or (np.isfinite(best_score) and work_score > best_score * 1.6 + 0.7)
        ):
            work_a = best_a.copy()
            work_z = best_z.copy()
            work_score = best_score
            no_improve = 0
            beta = 30.0

        if it % 800 == 0:
            print("iter", it, "best %.6f work %.6f" % (best_score, work_score))

    return best_a, best_score, best_z


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start = time.monotonic()
    if budget_s is None:
        total_budget = 2400.0
    else:
        total_budget = float(budget_s)
    deadline = start + max(0.0, total_budget - 12.0)

    rng = np.random.default_rng(seed)

    seq1 = _get_initial_sequence()
    if seq1 is not None and seq1.size >= 256:
        n = int(seq1.size)
    else:
        n = 4096

    # Keep the problem in a range where FFT local search and small LPs are fast.
    n = int(min(max(n, 2048), 8192))
    target = math.sqrt(2.0 * float(n))

    z_list = []

    if seq1 is not None:
        res = _resample_sequence(seq1, n)
        a0 = _normalize_sum(res, target)
        if a0 is not None:
            z_list.append(_z_from_a(a0))

    z_list.extend(_random_z_inits(n, rng, 16))

    best_a = None
    best_z = None
    best_score = np.inf

    for z in z_list:
        a = _softmax_scale(z, target)
        if a is None:
            continue
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_z = z.copy()

    if best_a is None:
        best_a = np.full(n, target / float(n), dtype=np.float64)
        best_z = np.zeros(n, dtype=np.float64)
        best_score = _score(best_a)

    print("initial bound %.6f n=%d" % (best_score, n))

    search_deadline = deadline - 15.0
    if time.monotonic() < search_deadline:
        best_a, best_score, best_z = _full_search(
            best_a, best_z, best_score, target, rng, search_deadline
        )

    print("pre-final bound %.6f n=%d" % (best_score, n))

    final = _normalize_sum(best_a, target)
    if final is None:
        final = np.full(n, target / float(n), dtype=np.float64)

    out = [float(x) for x in final]

    # Final sanity check against the official evaluator if available.
    try:
        official = evaluate_sequence(out)
        if not np.isfinite(official):
            final = np.full(n, target / float(n), dtype=np.float64)
            out = [float(x) for x in final]
    except Exception:
        pass

    return out
```
