        if mb < best_mb:
            best_mb = mb
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > kick_every:
                eta_cur *= 0.6
                if eta_cur < 1e-3:
                    break
                a = proj_simplex(a + kick_size * (np.random.default_rng(it).random(n) - 0.5))
                stall = 0
            elif stall > 100:
                eta_cur = max(0.1, eta_cur * 0.8)

        g = subgrad_at(a, k)
        a = proj_simplex(a - eta_cur * g)
        it += 1
        if verbose and it % 200 == 0:
            print(f"    it={it} mb={mb:.3e} score~{2*n*mb:.4f} eta={eta_cur:.3f}")

    return best_a, 2.0 * n * best_mb


def init_cosine(n, f):
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * f * i / n))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else np.ones(n) / n


def init_bimodal(n, sigma_frac=0.12):
    i = np.arange(n, dtype=np.float64)
    sig = max(1.0, sigma_frac * n)
    a = np.exp(-0.5 * ((i - n / 4.0) / sig) ** 2) + np.exp(-0.5 * ((i - 3.0 * n / 4.0) / sig) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.ones(n) / n


def init_random(n, rng):
    a = rng.random(n)
    s = a.sum()
    return a / s if s > 0 else np.ones(n) / n


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: tries multiple initializations and optimises via projected GD."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 4.0)

    # Collect the provided base sequence
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.clip(base, 0.0, 1000.0)
    except Exception:
        base = np.ones(512, dtype=np.float64)

    best_seq = base.copy().astype(np.float64)
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] base score={best_score:.6f} n={len(base)}")

    # Build list of (name, init_vector) pairs
    inits = []

    # Provide base normalised
    if len(base) >= 16:
        b = base.copy()
        s = b.sum()
        if s > 0:
            inits.append(("base", b / s))

    # Cosine inits at several lengths
    for n_c in [2048, 4096, 8192]:
        for f in [1, 2, 3]:
            inits.append((f"cos_f{f}_n{n_c}", init_cosine(n_c, f)))

    # Bimodal
    for n_b in [2048, 4096]:
        inits.append((f"bim_n{n_b}", init_bimodal(n_b)))

    # Random
    for n_r in [2048, 4096]:
        inits.append((f"rnd_n{n_r}", init_random(n_r, rng)))

    # Give each init a time slice; total budget minus 200s reserve for polishing
    n_strats = len(inits)
    reserve = 200.0
    per_strat = max(30.0, (float(budget_s) - 10.0 - reserve) / max(1, n_strat))

    for name, a0 in inits:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        alloc = min(per_strat, remaining)
        if alloc < 5.0:
            break
        n = len(a0)
        if n < 16 or n > 30000:
            continue
        print(f"\n[{name}] n={n} alloc={alloc:.0f}s")
        t0 = time.monotonic()
        opt, opt_sc = optimize_simplex(a0, alloc, eta=1.0, kick_every=400, kick_size=0.04, verbose=False)
        # Evaluate with the real evaluator
        cand = np.clip(opt, 0.0, 1000.0).tolist()
        sc = evaluate_sequence(cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.array(cand, dtype=np.float64)
            print(f"  NEW BEST {best_score:.6f} from {name}")
        else:
            print(f"  {name} best internal={opt_sc:.4f} eval={sc:.4f}")

    # Polish the overall best
    if time.monotonic() < deadline:
        rem = max(10.0, deadline - time.monotonic())
        print(f"\n[polish] remaining={rem:.0f}s")
        b0 = best_seq.copy()
        s = b0.sum()
        if s > 0:
            b0 = b0 / s
        opt, _ = optimize_simplex(b0, rem * 0.6, eta=0.5, kick_every=600, kick_size=0.02, verbose=True)
        cand = np.clip(opt, 0.0, 1000.0)
        sc = evaluate_sequence(cand.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"  NEW BEST after polish {best_score:.6f}")
        # Second polish round with even smaller steps
        if time.monotonic() < deadline:
            rem2 = max(5.0, deadline - time.monotonic())
            s2 = best_seq.sum()
            if s2 > 0:
                b2 = best_seq / s2
            else:
                b2 = best_seq
            opt2, _ = optimize_simplex(b2, rem2, eta=0.2, kick_every=800, kick_size=0.01, verbose=False)
            cand2 = np.clip(opt2, 0.0, 1000.0)
            sc2 = evaluate_sequence(cand2.tolist())
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = cand2.copy()
                print(f"  NEW BEST after 2nd polish {best_score:.6f}")

    result = np.clip(best_seq.astype(np.float64), 0.0, 1000.0).tolist()
    final_sc = evaluate_sequence(result)
    print(f"\n[done] length={len(result)} score={final_sc:.6f}")
    return result
