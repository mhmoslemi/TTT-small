```python
import numpy as np
import time
import operator

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Main ingredients:
1. A strong analytic seed family based on the truncated generating function
   (1-z)^(-1/2), whose infinite convolution is flat.  Envelopes, endpoint
   boosts, and random multiplicative variations are also generated.
2. A damped spectral-Newton polish that drives conv(a,a) toward a flat vector.
3. A robust exponentiated-gradient line search on the original nonnegative
   simplex.  The gradient of a log-sum-exp surrogate for the maximum
   convolution coefficient is computed with FFT cross-correlation.
4. Periodic random restarting from the current best when the line search gets
   stuck.
"""

def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _normalize_nonnegative(v):
    if v is None:
        return None
    x = np.asarray(v, dtype=np.float64)
    if x.ndim != 1 or x.size == 0:
        return None
    x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    x = np.clip(x, 0.0, 1000.0)
    s = float(x.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return x / s


def _central_base(n):
    n = int(n)
    if n <= 0:
        return None
    c = np.empty(n, dtype=np.float64)
    c[0] = 1.0
    if n > 1:
        i = np.arange(1, n, dtype=np.float64)
        ratios = (2.0 * i - 1.0) / (2.0 * i)
        c[1:] = np.cumprod(ratios)
    return c


def _fast_self_conv(a):
    n = int(a.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    target = 2 * n - 1
    if target == 1:
        return np.array([float(a[0] * a[0])], dtype=np.float64)
    N = _next_pow2(target)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    c = c[:target]
    np.maximum(c, 0.0, out=c)
    return c


def _fast_score(a):
    if a is None:
        return np.inf
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.inf
    c = _fast_self_conv(a)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m / (s * s))


def _cross_weights(w, a):
    """
    Return length-n vector g where
        g[j] = sum_k w[k] * a[k-j]
    with a[m] = 0 outside [0, n-1].
    This is 1/2 of the gradient factor for coefficients of a*a.
    """
    n = int(a.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([float(w[0] * a[0])], dtype=np.float64)

    target = len(w) + n - 1
    N = _next_pow2(target)
    W = np.fft.rfft(w, N)
    AR = np.fft.rfft(a[::-1], N)
    full = np.fft.irfft(W * AR, N)
    start = n - 1
    return full[start:start + n]


def _softmax_weights(c, beta):
    L = int(c.size)
    if L <= 0:
        return np.array([], dtype=np.float64)
    cmax = float(c.max())
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.full(L, 1.0 / L, dtype=np.float64)

    T = beta * cmax
    if T <= 0.0 or not np.isfinite(T):
        T = 1e-15

    x = (c - cmax) / T
    np.maximum(x, -50.0, out=x)
    w = np.exp(x)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = (c >= cmax - 1e-12 * cmax).astype(np.float64)
        s = float(w.sum())
        if s <= 0.0:
            w = np.full(L, 1.0 / L, dtype=np.float64)
        else:
            w /= s
    else:
        w /= s
    return w


def _try_add_seed(seeds, arr):
    y = _normalize_nonnegative(arr)
    if y is not None:
        seeds.append(y)


def _generate_seeds(n, rng):
    seeds = []
    base = _central_base(n)
    if base is None:
        return seeds

    _try_add_seed(seeds, base)
    _try_add_seed(seeds, base[::-1])

    seq = globals().get("height_sequence_1", None)
    if seq is not None:
        arr = _normalize_nonnegative(seq)
        if arr is not None and arr.size == n:
            _try_add_seed(seeds, arr)

    if n > 1:
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        xm = x - 0.5
    else:
        x = np.array([0.0], dtype=np.float64)
        xm = np.array([0.0], dtype=np.float64)

    # Linear envelopes.
    for p in (-1.0, -0.75, -0.5, -0.25, 0.0, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0):
        env = 1.0 + p * x
        if np.min(env) < 0.0:
            env = np.maximum(env, 0.0)
        _try_add_seed(seeds, base * env)

    # Centered linear envelopes.
    for p in (-1.0, -0.5, 0.0, 0.5, 1.0, 2.0):
        env = 1.0 + p * xm
        if np.min(env) < 0.0:
            env = np.maximum(env, 0.0)
        _try_add_seed(seeds, base * env)

    # Quadratic envelopes.
    for p in (-1.0, 0.0, 1.0, 2.0):
        for q in (-1.0, -0.5, 0.0, 0.5, 1.0):
            env = 1.0 + p * x + q * x * (1.0 - x)
            if np.min(env) < 0.0:
                env = np.maximum(env, 0.0)
            _try_add_seed(seeds, base * env)

    # Exponential envelopes.
    for p in (-2.0, -1.0, 0.5, 1.0, 2.0, 3.0):
        env = np.exp(p * x)
        _try_add_seed(seeds, base * env)

    # Endpoint masses: the finite-nefast way to use the last index without
    # immediately disturbing the left half of the convolution.
    for gamma in (0.5, 1.0, 2.0, 5.0, 10.0, 50.0):
        arr = base.astype(np.float64)
        arr[-1] *= (1.0 + gamma)
        _try_add_seed(seeds, arr)

        if n > 20:
            arr = base.astype(np.float64)
            arr[-10:] *= (1.0 + 0.1 * gamma)
            _try_add_seed(seeds, arr)

        if n > 2:
            arr = base.astype(np.float64)
            arr[0] *= (1.0 + 0.1 * gamma)
            _try_add_seed(seeds, arr)

    # Additive brute-tail variations.
    for eps in (0.1, 0.5, 2.0, 10.0):
        _try_add_seed(seeds, base + eps * x * x)
        if n > 10:
            _try_add_seed(seeds, base.astype(np.float64) / (1.0 + eps * x * x))

    # Small reverse mixtures, useful as a differentiable bridge to symmetric
    # structures but kept small to avoid the huge central cross-peak.
    for eps in (0.001, 0.003, 0.01, 0.03, 0.10):
        arr = (1.0 - eps) * base + eps * base[::-1]
        _try_add_seed(seeds, arr)

    # Random multiplicative noises.
    for sigma in (0.05, 0.10, 0.20, 0.40):
        for _ in range(3):
            noise = np.exp(np.clip(rng.normal(0.0, sigma, size=n), -10.0, 10.0))
            _try_add_seed(seeds, base * noise)

        tail_bias = np.exp(np.clip(rng.normal(0.0, 0.25, size=n), -5.0, 5.0) * x)
        _try_add_seed(seeds, base * tail_bias)

    return seeds


def _block_perturb(a, rng):
    n = int(a.size)
    if n <= 1:
        return a.copy()
    width = int(rng.integers(max(1, n // 500), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    scale = float(rng.uniform(0.05, 0.25))
    cand = a.copy()
    cand[start:stop] *= np.exp(np.clip(rng.normal(0.0, scale, size=stop - start), -10.0, 10.0))
    return _normalize_nonnegative(cand)


def _polish_spectral(a, deadline, rng, verbose=True):
    """
    Damped Newton-style polish for A(z)^2 close to a flat vector.
    Works well as a short initial booster from the truncated (1-z)^(-1/2) seed.
    """
    a = _normalize_nonnegative(a)
    if a is None:
        return None, np.inf
    n = int(a.size)
    if n == 1:
        return a, 2.0

    c = _fast_self_conv(a)
    score = float(2.0 * n * c.max())
    best = a.copy()
    best_score = score

    L = 2 * n - 1
    N = _next_pow2(L)
    target = np.full(L, 1.0 / L, dtype=np.float64)

    step = 0.5
    last_improve = time.monotonic()
    it = 0

    while time.monotonic() < deadline:
        it += 1

        if score < best_score - 1e-13:
            best = a.copy()
            best_score = score
            last_improve = time.monotonic()

        if verbose and it % 20 == 0:
            print(f"    spectral it={it} score={score:.8f} best={best_score:.8f} step={step:.4f}", flush=True)

        if time.monotonic() - last_improve > 45.0:
            break

        r = target - c
        r = r - float(np.mean(r))
        rnorm = float(np.linalg.norm(r))
        if not np.isfinite(rnorm) or rnorm < 1e-14:
            break

        fa = np.fft.rfft(a, N)
        fr = np.fft.rfft(r, N)
        denom = 2.0 * fa
        absd = np.abs(denom)
        thr = 1e-8 * float(absd.max()) if absd.size else 1e-12

        spec = np.zeros_like(fa, dtype=np.complex128)
        mask = absd > thr
        if np.any(mask):
            spec[mask] = fr[mask] / denom[mask]

        upd = np.fft.irfft(spec, N)[:n]
        unorm = float(np.linalg.norm(upd))
        anorm = float(np.linalg.norm(a))
        if not np.isfinite(unorm) or unorm < 1e-14:
            break
        scale = anorm / unorm

        accepted = False
        for factor in (1.0, 0.5, 0.25, 0.1, 0.05):
            alpha = step * factor
            cand = a + alpha * scale * upd
            cand = np.where(cand > 0.0, cand, 0.0)
            ss = float(cand.sum())
            if not np.isfinite(ss) or ss <= 0.0:
                continue
            cand /= ss
            cand_c = _fast_self_conv(cand)
            cand_score = float(2.0 * n * cand_c.max())
            if np.isfinite(cand_score) and cand_score < score - 1e-13:
                a = cand
                c = cand_c
                score = cand_score
                accepted = True
                if factor >= 0.99:
                    step = min(1.0, step * 1.1)
                else:
                    step = max(0.02, step * 0.7)
                break

        if not accepted:
            step *= 0.5
            if step < 0.01:
                # One gentle random re-start from the best, then stop if still stuck.
                cand = best.copy()
                cand *= np.exp(np.clip(rng.normal(0.0, 0.08, size=n), -10.0, 10.0))
                cand = _normalize_nonnegative(cand)
                if cand is None:
                    break
                c = _fast_self_conv(cand)
                score = float(2.0 * n * c.max())
                if score < best_score - 1e-13:
                    a = cand
                    best = cand
                    best_score = score
                    last_improve = time.monotonic()
                    step = 0.3
                else:
                    a = best.copy()
                    break

    if score < best_score:
        best = a.copy()
        best_score = score
    return best, best_score


def _polish_gradient(a, deadline, rng, verbose=True):
    """
    Exponentiated-gradient line search on the simplex a >= 0, sum(a)=1.
    Minimizes a log-sum-exp surrogate for max(conv(a,a)).
    """
    a = _normalize_nonnegative(a)
    if a is None:
        return None, np.inf
    n = int(a.size)
    if n == 1:
        return a, 2.0

    c = _fast_self_conv(a)
    score = float(2.0 * n * c.max())
    best = a.copy()
    best_score = score

    eta = 0.30
    beta = 0.15
    eta_min = 0.02
    last_improve = time.monotonic()
    it = 0

    while time.monotonic() < deadline:
        it += 1

        if score < best_score - 1e-13:
            best = a.copy()
            best_score = score
            last_improve = time.monotonic()

        if verbose and (it % 300 == 0 or (time.monotonic() - last_improve > 90.0 and it % 25 == 0)):
            print(
                f"    gradient it={it} score={score:.8f} best={best_score:.8f} "
                f"eta={eta:.4f} beta={beta:.6f}",
                flush=True
            )

        # Gentle annealing: widen the active set early, sharpen it later.
        if it % 200 == 0:
            beta = max(0.005, beta * 0.90)
            eta = min(1.5, eta * 1.05)

        # If no best-like progress, jump from the best point.
        if time.monotonic() - last_improve > 120.0:
            cand = best.copy()
            scale = float(rng.uniform(0.08, 0.20))
            cand *= np.exp(np.clip(rng.normal(0.0, scale, size=n), -10.0, 10.0))
            cand = _normalize_nonnegative(cand)
            if cand is None:
                break
            # Occasionally try a block move too.
            if rng.random() < 0.35:
                cand2 = cand.copy()
                cand2 = _block_perturb(cand2, rng)
                if cand2 is not None:
                    cand_score2 = _fast_score(cand2)
                    cand_score1 = _fast_score(cand)
                    if cand_score2 < cand_score1:
                        cand = cand2

            c = _fast_self_conv(cand)
            score = float(2.0 * n * c.max())
            a = cand
            eta = 0.30
            beta = 0.15
            last_improve = time.monotonic()
            continue

        w = _softmax_weights(c, beta)
        g = 2.0 * _cross_weights(w, a)
        if not np.all(np.isfinite(g)):
            a = best.copy()
            c = _fast_self_conv(a)
            score = float(2.0 * n * c.max())
            eta = 0.1
            continue

        mean_g = float(np.dot(a, g))
        dz = g - mean_g
        maxabs = float(np.max(np.abs(dz)))

        if not np.isfinite(maxabs) or maxabs < 1e-16:
            # No usable direction; small localized restart.
            cand = _block_perturb(a, rng)
            if cand is None:
                cand = _block_perturb(best, rng)
            if cand is None:
                break
            c = _fast_self_conv(cand)
            score = float(2.0 * n * c.max())
            a = cand
            eta = 0.30
            beta = 0.12
            last_improve = time.monotonic()
            continue

        accepted = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            step = eta * factor
            z = -step * dz

            # Removing the maximum makes the update a pure relative perturbation
            # before renormalization; clipping prevents oscillatory blowups.
            zmax = float(np.max(z))
            z = z - zmax
            z = np.clip(z, -10.0, 0.0)

            cand = a * np.exp(z)
            cand = np.where(cand > 0.0, cand, 0.0)
            ss = float(cand.sum())
            if not np.isfinite(ss) or ss <= 0.0:
                continue
            cand /= ss

            cand_c = _fast_self_conv(cand)
            cand_score = float(2.0 * n * cand_c.max())

            if np.isfinite(cand_score) and cand_score < score - 1e-13:
                a = cand
                c = cand_c
                score = cand_score
                accepted = True

                if factor >= 0.99:
                    eta = min(1.5, eta * 1.15)
                elif factor <= 0.125:
                    eta = max(eta_min * 0.5, eta * 0.65)
                break

        if not accepted:
            eta = max(eta_min * 0.5, eta * 0.5)
            if eta < eta_min:
                # A modest random kick, then reset step size.
                cand = a.copy()
                cand *= np.exp(np.clip(rng.normal(0.0, 0.04, size=n), -10.0, 10.0))
                cand = _normalize_nonnegative(cand)
                if cand is None:
                    cand = best.copy()
                c = _fast_self_conv(cand)
                score = float(2.0 * n * c.max())
                a = cand
                eta = 0.30
                beta = 0.12
                last_improve = time.monotonic()

    if score < best_score:
        best = a.copy()
        best_score = score

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a low-score nonnegative sequence and return it as a Python list.
    """
    budget_s = max(120.0, float(budget_s))
    start = time.monotonic()
    main_deadline = start + max(60.0, budget_s - 60.0)
    rng = np.random.default_rng(seed)

    # Length selection: respect the provided length when sensible, otherwise
    # use the known sensible 6121.
    n = 6121
    seqobj = globals().get("height_sequence_1", None)
    if seqobj is not None:
        try:
            arr = np.asarray(seqobj)
            if arr.ndim == 1 and 50 <= arr.size <= 30000:
                n = int(arr.size)
        except Exception:
            pass

    if n <= 1:
        return [1.0]

    print(f"Starting search: n={n}, budget={budget_s:.1f}s", flush=True)

    seeds = _generate_seeds(n, rng)
    if not seeds:
        base = _central_base(n)
        base = _normalize_nonnegative(base)
        if base is None:
            return [1.0] * n
        return [float(x) for x in base]

    scored = []
    for s in seeds:
        if time.monotonic() > main_deadline - 30.0:
            break
        sc = _fast_score(s)
        if np.isfinite(sc):
            scored.append((sc, s))

    if not scored:
        base = _central_base(n)
        base = _normalize_nonnegative(base)
        return [float(x) for x in base]

    scored.sort(key=operator.itemgetter(0))

    best = scored[0][1].copy()
    best_score = float(scored[0][0])
    print(f"Best seed score (FFT): {best_score:.8f}", flush=True)

    total_time = max(120.0, budget_s - 120.0)
    fractions = (0.45, 0.30, 0.15, 0.10)
    max_starts = min(len(scored), 4)

    for idx in range(max_starts):
        now = time.monotonic()
        if now > main_deadline - 45.0:
            break

        frac = fractions[idx] if idx < len(fractions) else 0.1
        slice_budget = min(total_time * frac, main_deadline - now - 30.0)
        if slice_budget < 45.0:
            break

        cur_deadline = now + slice_budget
        seed_arr = scored[idx][1].copy()

        # For the first one or two seeds, give a short spectral-Newton boost.
        if idx < 2 and slice_budget > 150.0:
            sp_budget = min(90.0, slice_budget * 0.18)
            sp_deadline = time.monotonic() + sp_budget
            sp_arr, sp_score = _polish_spectral(seed_arr, sp_deadline, rng, verbose=True)
            if sp_arr is not None and np.isfinite(sp_score) and sp_score <= scored[idx][0] + 1e-12:
                if sp_score < best_score:
                    best = sp_arr
                    best_score = float(sp_score)
                    print(f"Updated global best after spectral: {best_score:.8f}", flush=True)
                seed_arr = sp_arr

        grad_arr, grad_score = _polish_gradient(seed_arr, cur_deadline, rng, verbose=True)
        if grad_arr is not None and np.isfinite(grad_score) and grad_score < best_score:
            best = grad_arr
            best_score = float(grad_score)
            print(f"Updated global best after gradient start {idx}: {best_score:.8f}", flush=True)

    # Use any remaining time on the global best.
    now = time.monotonic()
    if now < main_deadline - 30.0:
        grad_arr, grad_score = _polish_gradient(best, main_deadline, rng, verbose=True)
        if grad_arr is not None and np.isfinite(grad_score) and grad_score < best_score:
            best = grad_arr
            best_score = float(grad_score)
            print(f"Final gradient improved score: {best_score:.8f}", flush=True)

    best = _normalize_nonnegative(best)
    if best is None:
        best = _normalize_nonnegative(_central_base(n))
    if best is None:
        best = np.ones(n, dtype=np.float64) / n

    print(f"Final returned FFT score: {best_score:.8f} (n={best.size})", flush=True)
    return [float(x) for x in best]
```
