"""
Search for low-score nonnegative convolution sequences.

Key idea:
  The score 2*n*max(a*a)/(sum(a)**2) is scale-invariant.  For a continuous
  density f on [0,1] of integral 1, the discrete sequence a_i ~ f(i/n)/n
  gives score approx 2*max_x (f*f)(x).  A strongly non-symmetric periodic
  density such as 1 + cos(2*pi*x + phase) with unit amplitude has very small
  convolution maximum, sometimes close to the absolute lower bound.  This code
  constructs such discrete wave candidates, scans length, phase, and amplitude,
  refines the phase, and returns the best exact-evaluated sequence found.
"""

import numpy as np
import time
import math

_TWOPI = 2.0 * math.pi


def _fft_linear_conv(a):
    """Linear convolution of a 1-D nonnegative real array with itself."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)

    # Direct convolution is fine for tiny vectors and avoids FFT roundoff there.
    if n <= 64:
        c = np.convolve(a, a)
    else:
        m = 1 << (2 * n - 1).bit_length()
        fa = np.fft.rfft(a, m)
        c = np.fft.irfft(fa * fa, m)[:2 * n - 1]

    if c.size:
        c = np.ascontiguousarray(c)
        # FFT roundoff can make tiny negative values.
        c = np.maximum(c, 0.0)
    return c


def _fast_score(a):
    """Fast FFT-based approximation of evaluate_sequence for normalized candidates."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")

    s = float(np.sum(a))
    if not math.isfinite(s) or s < 0.01:
        return float("inf")

    c = _fft_linear_conv(a)
    if c.size == 0:
        return float("inf")

    mb = float(np.max(c))
    if not math.isfinite(mb):
        return float("inf")

    return float(2.0 * n * mb / (s * s))

    Return a normalized nonnegative sequence from a shifted cosine wave.

    a_i = normalize( max(0, 1 + amp * cos(2*pi*(i+0.5)/n + phase)) )
