import numpy as np
import time

def generate_truncated_binomial(n):
    """Generate the truncated central binomial sequence of length n, normalized to sum 1."""
    # u[k] = binom(2k,k) / 4^k for k = 0, ..., n-1
    # Use recurrence: u[0] = 1, u[k] = u[k-1] * (2k-1)/(2k) * ... 
    # Actually: u[k] = binom(2k,k)/4^k
    # u[k]/u[k-1] = [binom(2k,k)/4^k] / [binom(2k-2,k-1)/4^{k-1}]
    # = [binom(2k,k)/binom(2k-2,k-1)] / 4
    # binom(2k,k)/binom(2k-2,k-1) = (2k)!/(k!k!) * ((k-1)!(k-1)!)/(2k-2)!
    # = (2k)(2k-1)/(k^2)
    # So u[k]/u[k-1] = (2k)(2k-1)/(k^2 * 4) = (2k-1)/(2k)
    
    u = np.empty(n, dtype=np.float64)
    u[0] = 1.0
    for k in range(1, n):
        u[k] = u[k-1] * (2*k - 1) / (2*k)
    
    S = np.sum(u)

    return a

def project_to_simplex(v):
    """Project vector v onto the standard simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n+1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0]) + 1  # 1-indexed
    theta = (cssv[rho-1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def convolve_fft(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    # Need output length 2n-1
    L = 1
    while L < 2*n - 1:
        L *= 2
    fa = np.fft.rfft(a, L)
    fa2 = fa * fa
    result = np.fft.irfft(fa2, L)
    return result[:2*n-1]

def compute_objective(a):

def compute_gradient_at_lag(a, k):
    """Compute gradient of b[k] = (a*a)[k] with respect to a.
    grad[j] = 2*a[k-j] if 0 <= k-j < len(a), else 0.
    """
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    # For j in valid range: k-j in [0, n-1], so j in [k-n+1, k]
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    for j in range(j_start, j_end + 1):
        grad[j] = 2.0 * a[k - j]
    return grad

def compute_gradient_at_lag_fast(a, k):
    """Vectorized version."""
    n = len(a)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start > j_end:
        return np.zeros(n, dtype=np.float64)
    indices = np.arange(j_start, j_end + 1)
    grad_vals = 2.0 * a[k - indices]
    grad = np.zeros(n, dtype=np.float64)
    grad[indices] = grad_vals
    return grad

    grad = compute_gradient_at_lag_fast(a, k)
    a_new = a - eta * grad
    a_new = project_to_simplex(a_new)
    return a_new

But wait, this is the step size for the "near" operation in FISTA, which includes both the gradient step and the projection. The standard FISTA step is:
x_{k+1} = prox_{(1/L)*g}(y_k) where g is the gradient and prox is the proximal operator.

For our case, the "prox" is: min_x (1/(2*eta))||x - (y - eta*grad)||^2 s.t. x in simplex.
With eta = 1/L = 0.5: min_x ||x - (y - 0.5*grad)||^2 s.t. x in simplex.
This is just the projection of (y - 0.5*grad) onto the simplex.

So the step is: x_{k+1} = project(y - 0.5 * grad).

import numpy as np
import time

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Truncated central binomial coefficient initialization (baseline ~pi/2 ≈ 1.5708)
2. Nesterov-accelerated projected subgradient descent with FFT convolution
3. Adaptive step size and periodic restarts
4. Best-n scanning across multiple sequence lengths
"""


def generate_truncated_binomial(n):
    """Generate truncated central binomial sequence u[k]=C(2k,k)/4^k normalized to sum 1."""
    u = np.empty(n, dtype=np.float64)
    u[0] = 1.0
    for k in range(1, n):
        u[k] = u[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    S = np.sum(u)
    return u / S


def project_to_simplex(v):
    """Project vector onto standard simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(np.where(cond)[0])) + 1
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def convolve_fft(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    fa = np.fft.rfft(a, L)
    result = np.fft.irfft(fa * fa, L)
    return result[:2 * n - 1]


def compute_gradient_at_lag(y, k):
    """Gradient of (y*y)[k] w.r.t. y: grad[j] = 2*y[k-j] if valid."""
    n = len(y)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    grad = np.zeros(n, dtype=np.float64)
    if j_start <= j_end:
        js = np.arange(j_start, j_end + 1)
        grad[js] = 2.0 * y[k - js]
    return grad


def compute_avg_gradient_active(y, active_lags):
    """Average gradient over active (tied max) lags."""
    n = len(y)
    grad_avg = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            grad_avg[js] += 2.0 * y[k - js]
    grad_avg /= len(active_lags)
    return grad_avg


def objective_value(a):
    """Compute 2*n*max_b/sum_a^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    b = convolve_fft(a)
    return 2.0 * n * np.max(b) / (s * s)


def nesterov_optimize(a0, n, deadline, rng, verbose=False):
    """Nesterov-accelerated projected subgradient descent."""
    a = a0.copy()
    y = a0.copy()
    t_mom = 1.0
    eta = 0.08
    best_local = a0.copy()
    best_score = objective_value(a0)
    stall = 0
    max_stall = 300
    iter_count = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        b = convolve_fft(y)
        max_b = np.max(b)
        s = np.sum(y)
        score = 2.0 * n * max_b / (s * s)

        if score < best_score:
            best_score = score
            best_local = y.copy() if np.all(y >= -1e-12) else a.copy()
            stall = 0
        else:
            stall += 1

        peak_val = max_b
        # Find active lags (within 1e-10 of max)
        active_mask = b >= peak_val - 1e-10 * peak_val
        active_idx = np.where(active_mask)[0]
        active_idx = active_idx[active_idx < n]  # Only lags in [0, n-1] matter for grad

        if len(active_idx) > 50:
            # Use average gradient for many tied lags (compute efficiently)
            # grad_avg[j] = (2/m) * sum_{k in active} y[k-j] for valid
            # For all k in [0, n-1]: grad_avg[j] = (2/n) * cumsum_y[n-1-j]
            if len(active_idx) >= n - 1:
                cumsum_y = np.cumsum(y)
                grad = (2.0 / n) * cumsum_y[::-1].copy()
                # Actually: grad_avg[j] = (2/m) * sum_{k in active, k>=j} y[k-j]
                # For all k in [0,n-1] active: sum_{k=j}^{n-1} y[k-j] = cumsum_y[n-1] - cumsum_y[j-1] (if j>0)
                # = sum_{m=0}^{n-1-j} y[m] = cumsum_y[n-1-j] (0-indexed)
                grad = (2.0 / float(len(active_idx))) * np.zeros(n)
                for j in range(n):
                    grad[j] = (2.0 / float(len(active_idx))) * cumsum_y[min(n - 1, n - 1 - j)]
            else:
                grad = compute_gradient_at_lag(y, int(active_idx[0]))
        elif len(active_idx) > 1:
            # Use a random active lag
            k = int(active_idx[rng.integers(0, len(active_idx))])
            grad = compute_gradient_at_lag(y, k)
        else:
            grad = compute_gradient_at_lag(y, int(peak_idx) if (peak_idx := int(np.argmax(b))) is not None else 0)

        # Gradient step + projection
        x_new = project_to_simplex(y - eta * grad)

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        y = x_new + beta * (x_new - a)

        a = x_new
        t_mom = t_new
        iter_count += 1

        # Adaptive step size
        if stall >= max_stall:
            eta *= 0.5
            stall = 0
            if eta < 1e-10:
                break
            # Restart with perturbation
            noise = rng.normal(0, 0.005, n)
            a = project_to_simplex(a + noise)
            y = a.copy()
            t_mom = 1.0
            eta = max(eta, 0.02)

        if verbose and time.monotonic() - last_print > 20:
            last_print = time.monotonic()
            print(f"    Nesterov iter {iter_count}, score={score:.8f}, best={best_score:.8f}, eta={eta:.6f}")

    return best_local, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Find sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses truncated binomial init + Nesterov projected subgradient descent.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Phase 1: Scan n values for best truncated binomial
    phase1_deadline = time.monotonic() + min(45.0, float(budget_s) * 0.03)
    n_candidates = sorted(set(
        list(range(40, 200, 5)) +
        list(range(200, 600, 10)) +
        list(range(600, 1500, 25)) +

    ))

    for n_try in n_candidates:
        if time.monotonic() > phase1_deadline:
            break
        a_try = generate_truncated_binomial(n_try)
        b_try = convolve_fft(a_try)
        score_try = 2.0 * n_try * np.max(b_try) / (np.sum(a_try) ** 2)
        if score_try < best_score:
            best_score = score_try
            best_seq = a_try.copy()

    n_best = len(best_seq)
    print(f"Phase 1: Best n={n_best}, binomial score={best_score:.8f}")

    # Verify with evaluate_sequence

    print(f"  Verified score: {verified:.8f}")

    a_opt, score_opt = nesterov_optimize(best_seq, n_best, phase2_deadline, rng, verbose=True)
    if score_opt < best_score:

        best_seq = a_opt.copy()
    print(f"Phase 2: Optimized score={best_score:.8f}")

    # Verify
    verified = float(evaluate_sequence([float(x) for x in best_seq]))
    print(f"  Verified: {verified:.8f}")
    if verified < best_score:
        best_score = verified

    # Phase 3: Try a few different n values with shorter optimization
    if time.monotonic() < total_deadline - 60.0:
        # Try neighboring n values
        n_neighbors = sorted(set([
            max(20, n_best // 2), max(20, int(n_best * 0.6)),
            int(n_best * 1.3), min(8000, int(n_best * 1.7)),
        ]))
        per_n_time = max(5.0, (total_deadline - time.monotonic() - 10.0) / max(1, len(n_neighbors)))
