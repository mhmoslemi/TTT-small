        A_ub = np.zeros((K, n))
        for i, k in enumerate(top_idx):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):

            break

        g_sum = g.sum()
        if g_sum < 1e-10:
            break

        g = g / g_sum * target_sum
        
        # Move toward g
        t_mix = 0.3
        a_new = (1 - t_mix) * a + t_mix * g
        a_new = _normalize(a_new, target_sum)
        
        # Check
        c_new = _full_conv(a_new, N_conv)
        new_score = 2.0 * n * c_new.max() / (target_sum * target_sum)
        
        if new_score <= score * 1.001:
            a = a_new
        else:

            t_mix = 0.1
            a_new = (1 - t_mix) * a + t_mix * g
            a_new = _normalize(a_new, target_sum)
            c_new = _full_conv(a_new, N_conv)
            new_score = 2.0 * n * c_new.max() / (target_sum * target_sum)
            if new_score <= score * 1.001:
                a = a_new
            else:
                stall += 1
    
    return best_a, best_score
