```python
"""
Optimizer for nonnegative step functions minimizing
    2*n*max(convolve(a,a)) / sum(a)^2.

The code combines several generic strategies:
- a decaying central-binomial-family prior,
- smoothed-gradient optimization in softmax coordinates using FFT-based
  convolutions and dense active-set gradients,
- simple block/local random search,
- a linear-programming "next best response" step for modest n,
- several tilted and right-bumped starts.
"""

import time
import numpy as np

try:
    from scipy.optimize import minimize, linprog
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False

# Globals used by cached/scipy objective functions.
SC_N = 0
SC_TEMP_REL = 0.1


def _softmax(x):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        return x
    m = float(np.max(x))
    if not np.isfinite(m):
        return np.full(x.size, 1.0 / x.size, dtype=float)
    ex = np.exp(x - m)
    s = float(np.sum(ex))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(x.size, 1.0 / x.size, dtype=float)
    return ex / s


def _fast_conv(a):
    a = np.asarray(a, dtype=float)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=float)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=float)
    if n < 128:
        return np.convolve(a, a)
    r = 2 * n - 1
    N = 1 << max(1, (r - 1).bit_length())
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, N)[:r]
    out = np.asarray(out, dtype=float)
    # FFT can produce tiny negative values; they are never meaningful here.
    out[out < 0.0] = 0.0
    return out


def _fast_score(a):
    a = np.asarray(a, dtype=float)
    if a.size == 0:
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    S = float(np.sum(a))
    if S <= 1e-300 or not np.isfinite(S):
        return float("inf")
    a = a / S
    b = _fast_conv(a)
    if b.size == 0:
        return float("inf")
    M = float(np.max(b))
    if not np.isfinite(M) or M <= 0.0:
        return float("inf")
    return float(2.0 * a.size * M)


def _central(n):
    if n <= 0:
        return np.empty(0, dtype=float)
    a = np.empty(n, dtype=float)
    c = 1.0
    a[0] = 1.0
    for k in range(1, n):
        c *= (2.0 * k - 1.0) / (2.0 * k)
        a[k] = c
    return a


def _weighted_corr(a, w):
    """
    Return h_i = sum_j a_j * w_{i+j} for i=0..n-1.
    This gives 1/2 of the gradient of a weighted sum of convolution coefficients.
    """
    n = a.size
    if n == 0:
        return np.empty(0, dtype=float)
    if n == 1:
        return np.array([a[0] * float(np.sum(w))], dtype=float)
    r = 2 * n - 1
    N = 1 << max(1, (r - 1).bit_length())
    A = np.fft.fft(a, N)
    W = np.fft.fft(w, N)
    h = np.fft.ifft(np.conj(A) * W).real[:n]
    h = np.asarray(h, dtype=float)
    h[h < 0.0] = 0.0
    return h


def _sc_compute(x):
    n = SC_N
    temp_rel = SC_TEMP_REL
    a = _softmax(x)
    b = _fast_conv(a)
    if b.size == 0 or not np.all(np.isfinite(b)):
        return 1.0e6, np.zeros(n, dtype=float)
    M = float(np.max(b))
    if not np.isfinite(M) or M <= 0.0:
        return 0.0, np.zeros(n, dtype=float)

    if temp_rel <= 1e-8:
        tol = max(1e-12, 1e-8 * M)
        active = b >= (M - tol)
        if not np.any(active):
            idx = int(np.argmax(b))
            w = np.zeros(b.size, dtype=float)
            w[idx] = 1.0
            sw = 1.0
        else:
            w = active.astype(float)
            sw = float(np.sum(w))
            if sw <= 0.0 or not np.isfinite(sw):
                idx = int(np.argmax(b))
                w = np.zeros(b.size, dtype=float)
                w[idx] = 1.0
                sw = 1.0
            w /= sw
        B = M
    else:
        temp = max(temp_rel * M, 1e-30)
        z = (b - M) / temp
        w = np.zeros(b.size, dtype=float)
        mask = z > -50.0
        if np.any(mask):
            zm = z[mask]
            zm = np.minimum(zm, 0.0)
            w[mask] = np.exp(zm)
            sw = float(np.sum(w))
            if sw > 0.0 and np.isfinite(sw):
                w /= sw
                B = float(np.dot(w, b))
            else:
                idx = int(np.argmax(b))
                w = np.zeros(b.size, dtype=float)
                w[idx] = 1.0
                B = M
        else:
            idx = int(np.argmax(b))
            w = np.zeros(b.size, dtype=float)
            w[idx] = 1.0
            B = M

    h = _weighted_corr(a, w)
    if not np.all(np.isfinite(h)):
        return 1.0e6, np.zeros(n, dtype=float)

    # f = 2*n*B, and d(B)/da = 2*h.
    g_a = 4.0 * float(n) * h
    da = float(np.dot(a, g_a))
    g_x = g_a - a * da
    if not np.all(np.isfinite(g_x)):
        return 1.0e6, np.zeros(n, dtype=float)
    return float(2.0 * n * B), g_x


def _sc_fun_jac(x):
    return _sc_compute(x)


def _sc_fun_only(x):
    f, g = _sc_compute(x)
    return f


def _sc_grad_only(x):
    f, g = _sc_compute(x)
    return g


def _add_top(seq, sc, top_seqs, top_scores, max_top=12):
    if seq is None:
        return
    seq = np.asarray(seq, dtype=float)
    if seq.size == 0 or not np.all(np.isfinite(seq)):
        return
    seq = np.clip(seq, 0.0, 1000.0)
    S = float(np.sum(seq))
    if S <= 1e-300 or not np.isfinite(S):
        return
    seq = seq / S
    top_seqs.append(seq.copy())
    top_scores.append(float(sc))
    if len(top_seqs) > max_top:
        order = np.argsort(top_scores)
        new_seqs = []
        new_scores = []
        for i in order:
            if len(new_seqs) >= max_top:
                break
            ii = int(i)
            new_seqs.append(top_seqs[ii].copy())
            new_scores.append(top_scores[ii])
        for k in range(len(new_seqs)):
            top_seqs[k] = new_seqs[k]
            top_scores[k] = new_scores[k]
        del top_seqs[len(new_seqs):]
        del top_scores[len(new_scores):]


def _register_candidate(seq, best_pair, top_seqs, top_scores):
    if seq is None:
        return best_pair
    seq = np.asarray(seq, dtype=float)
    if seq.size == 0:
        return best_pair
    seq = np.clip(seq, 0.0, 1000.0)
    S = float(np.sum(seq))
    if S <= 1e-300 or not np.isfinite(S):
        return best_pair
    seq = seq / S
    sc = _fast_score(seq)
    if np.isfinite(sc):
        if sc < best_pair[0]:
            best_pair[0] = sc
            best_pair[1] = seq.copy()
        _add_top(seq, sc, top_seqs, top_scores)
    return best_pair

                    SC_N = n

                    mi = int(max(15, min(80, max_iter // 2)))
                    opt = {
                        "maxiter": mi,
                        "maxfun": int(max(150, mi * 20)),
                        "ftol": 1e-12,

                    res = None
                    try:

                    except Exception:
                        try:
                            res = minimize(_sc_fun_only, x_try, method="L-BFGS-B", jac=_sc_grad_only, options=opt)
                        except Exception:
                            res = None
                    if res is not None and res.x is not None:

                            a2 = _softmax(x2)
                            sc2 = _fast_score(a2)
                            if np.isfinite(sc2) and sc2 < best_sc:
                                best_sc = float(sc2)
                                best_a = a2
                                x_try = x2 - float(np.mean(x2))
    return best_a


def _local_search(a0, rng, max_iter, deadline):
    a = np.asarray(a0, dtype=float)
    n = a.size
    if n == 0:
        return a
    S = float(np.sum(a))
    if S <= 1e-300 or not np.isfinite(S):
        return a
    a = a / S

    x = np.log(np.clip(a, 1e-300, None))
    x = x - float(np.mean(x))
    best_x = x.copy()
    best_a = a.copy()
    best_sc = _fast_score(best_a)

    sigma = 0.06
    minL = max(1, n // 1000)
    maxL = max(minL, n // 25)
    maxL = min(maxL, n)
    no_improve = 0

    for _ in range(int(max_iter)):
        if time.monotonic() > deadline:
            break
        if maxL <= minL:
            L = minL
        else:
            L = int(rng.integers(minL, maxL + 1))
        if rng.random() < 0.12:

        if L >= n:

            std = 0.25 * sigma

        old = x[start:start + L].copy()
        x[start:start + L] = old + dx
        an = _softmax(x)
        sc = _fast_score(an)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_x = x.copy()
            best_a = an.copy()
            no_improve = 0
        else:
            x[start:start + L] = old
            no_improve += 1
        if no_improve > 35:
            sigma *= 0.6
            no_improve = 0
            if sigma < 1e-4:
                break
    return best_a


def _linprog_step(f, M):
    if not HAVE_SCIPY:
        return None
    f = np.asarray(f, dtype=float)
    n = f.size
    if n <= 0 or n > 768:
        return None
    if f[0] <= 1e-12:
        return None
    r = 2 * n - 1
    C = np.zeros((r, n), dtype=float)
    for j in range(n):
        C[j:j + n, j] = f
    c = -np.ones(n, dtype=float)
    b_ub = np.full(r, float(M), dtype=float)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"presolve": True})
    except Exception:
        try:
            res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method="interior-point")
        except Exception:
            return None
    if res is not None and res.success and res.x is not None:
        g = np.asarray(res.x, dtype=float)
        g = np.clip(g, 0.0, None)
        if np.isfinite(g).all() and np.sum(g) > 0.0:
            return g
    return None


def _lp_improve(f, deadline, max_rounds=5):
    f = np.asarray(f, dtype=float)
    if f.size == 0:
        return f
    S = float(np.sum(f))
    if S <= 0.0 or not np.isfinite(S):
        return f

    best = f.copy()
    best_sc = _fast_score(f)
    cur = f.copy()

    for _ in range(int(max_rounds)):
        if time.monotonic() > deadline:
            break
        b = _fast_conv(cur)
        if b.size == 0:
            break
        M = float(np.max(b))
        if not np.isfinite(M) or M <= 0.0:
            break
        g = _linprog_step(cur, M)
        if g is None:
            break
        Sg = float(np.sum(g))
        Sc = float(np.sum(cur))
        if Sg <= 1e-12 or Sc <= 1e-12:
            break
        g = g * (Sc / Sg)

        cur_sc = _fast_score(cur)
        best_cur = cur.copy()
        for t in (0.1, 0.25, 0.5, 0.75, 1.0):
            cand = (1.0 - t) * cur + t * g
            sc = _fast_score(cand)
            if np.isfinite(sc) and sc < cur_sc:
                cur_sc = float(sc)
                best_cur = cand.copy()

        if cur_sc < best_sc:
            best = best_cur.copy()
            best_sc = float(cur_sc)
            cur = best_cur.copy()
        else:
            cur = best_cur.copy()
            break
    return best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start_time = time.monotonic()
    total_budget = max(10.0, float(budget_s))
    hard_end = start_time + total_budget
    deadline = hard_end - 20.0
    rng = np.random.default_rng(int(seed))

    top_seqs = []
    top_scores = []
    best_pair = [float("inf"), None]

    # Use any provided previous construction if available.
    try:
        prev = height_sequence_1  # noqa: F821
        if prev is not None:
            p = np.asarray(prev, dtype=float)
            if p.size > 0:
                p = np.clip(p, 0.0, 1000.0)
                S = float(np.sum(p))
                if S > 1e-12 and np.isfinite(S):
                    p = p / S
                    best_pair = _register_candidate(p, best_pair, top_seqs, top_scores)
    except Exception:
        pass

    # Main lengths.  Keep the largest lengths below four/thousand to keep the
    # final reference evaluator practical.
    lengths = [32, 64, 128, 256, 512, 1024, 2048, 4096]
    central_cache = {}
    for n in lengths:
        if time.monotonic() > deadline - 10.0:
            break
        c = _central(n)
        cs = _fast_score(c)
        if np.isfinite(cs):
            best_pair = _register_candidate(c, best_pair, top_seqs, top_scores)
            central_cache[n] = c.copy()

    # Optimization order: prioritize the ranges where the best scores usually lie.
    optimization_order = [512, 1024, 256, 2048, 128, 4096, 64, 32]
    for n in optimization_order:
        if n not in central_cache:
            continue
        if time.monotonic() > deadline - 30.0:
            break
        c = central_cache[n]

        if n <= 128:
            temps = [0.18, 0.08, 0.03, 0.0]
            maxit = 150
            local_iter = 350
        elif n <= 256:
            temps = [0.18, 0.08, 0.03, 0.0]
            maxit = 120
            local_iter = 300
        elif n <= 512:
            temps = [0.18, 0.08, 0.03, 0.0]
            maxit = 100
            local_iter = 250
        elif n <= 1024:
            temps = [0.14, 0.06, 0.025, 0.0]
            maxit = 90
            local_iter = 220
        elif n <= 2048:
            temps = [0.14, 0.06, 0.025]
            maxit = 70
            local_iter = 160
        else:
            temps = [0.12, 0.05]
            maxit = 45
            local_iter = 90

        aopt = _optimize_gradient(c, temps, maxit, min(deadline, start_time + total_budget - 60), rng, 1)
        best_pair = _register_candidate(aopt, best_pair, top_seqs, top_scores)

        if local_iter > 0 and time.monotonic() < deadline - 25.0:
            aloc = _local_search(aopt, rng, local_iter, deadline)
            best_pair = _register_candidate(aloc, best_pair, top_seqs, top_scores)

        if n <= 512 and time.monotonic() < deadline - 80.0:
            alp = _lp_improve(c, deadline, 4)
            best_pair = _register_candidate(alp, best_pair, top_seqs, top_scores)
            alp2 = _optimize_gradient(alp, [0.08, 0.03, 0.0], 50, deadline, rng, 0)
            best_pair = _register_candidate(alp2, best_pair, top_seqs, top_scores)

        # Additional tilted / right-bumped starts, only where it is cheap enough.
        if n in (256, 512, 1024) and time.monotonic() < deadline - 90.0:
            idx = np.linspace(0.0, 1.0, n)
            for tilt in (-1.8, -0.7, -0.3, 0.3, 0.8, 1.7):
                if time.monotonic() > deadline - 60.0:
                    break
                aw = c * np.exp(float(tilt) * idx)
                scw = _fast_score(aw)
                if np.isfinite(scw) and scw < 1.75:
                    awe = _optimize_gradient(aw, [0.10, 0.04, 0.0], 45, min(deadline, start_time + total_budget - 80), rng, 0)
                    best_pair = _register_candidate(awe, best_pair, top_seqs, top_scores)

            for wf in (0.05, 0.10, 0.18):
                if time.monotonic() > deadline - 60.0:
                    break
                dist = 1.0 - idx
                for bump in (0.35, 1.2, 3.5):
                    if time.monotonic() > deadline - 45.0:
                        break
                    weight = 1.0 + float(bump) * np.exp(-(dist * dist) / (2.0 * wf * wf))
                    aw = c * weight
                    scw = _fast_score(aw)
                    if np.isfinite(scw) and scw < 1.75:
                        awe = _optimize_gradient(aw, [0.10, 0.04, 0.0], 40, min(deadline, start_time + total_budget - 90), rng, 0)
                        best_pair = _register_candidate(awe, best_pair, top_seqs, top_scores)

    # Final polish on the current best, if it has not already cleared the target.
    if best_pair[1] is not None and time.monotonic() < hard_end - 45.0:
        a = np.asarray(best_pair[1], dtype=float)
        best_pair = _register_candidate(
            _optimize_gradient(a, [0.05, 0.025, 0.01, 0.0], 160, hard_end - 25.0, rng, 2),
            best_pair, top_seqs, top_scores
        )
        best_pair = _register_candidate(
            _local_search(a, rng, 900, hard_end - 25.0),
            best_pair, top_seqs, top_scores
        )

        # Try nearby lengths of the currently best sequence.
        base_n = a.size
        nearby = []
        for nn in (base_n // 2, base_n * 2, base_n + 128, base_n - 256):
            if 128 <= nn <= 4096 and nn not in nearby:
                nearby.append(nn)
        for nn in nearby[:2]:
            if time.monotonic() > hard_end - 50.0:
                break
            c = _central(nn)
            c = _optimize_gradient(c, [0.10, 0.04, 0.01, 0.0], 80, hard_end - 25.0, rng, 1)
            best_pair = _register_candidate(c, best_pair, top_seqs, top_scores)

    # Validate a small set of best candidates against the authoritative evaluator.
    true_best = float("inf")
    true_seq = best_pair[1]
    for seq in top_seqs[:10]:
        if time.monotonic() > hard_end - 3.0:
            break
        try:
            val = evaluate_sequence(seq.tolist())  # evaluated externally, already imported in environment.
        except Exception:
            try:
                # Fallback if evaluator is not directly available in this namespace.
                import numpy as _np
                s = np.asarray(seq, dtype=float)
                sr = 2 * s.size * float(np.max(np.convolve(s, s))) / (float(np.sum(s)) ** 2)
                val = sr
            except Exception:
                val = float("inf")
        if np.isfinite(val) and val < true_best:
            true_best = float(val)
            true_seq = seq.copy()

    if true_seq is None:
        if best_pair[1] is not None:
            true_seq = best_pair[1].copy()
        else:
            true_seq = _central(64)

    true_seq = np.asarray(true_seq, dtype=float)
    S = float(np.sum(true_seq))
    if S <= 1e-300 or not np.isfinite(S):
        true_seq = _central(64)
    else:
        true_seq = true_seq / S
        true_seq = np.clip(true_seq, 0.0, 1000.0)

    return [float(x) for x in true_seq]
```
