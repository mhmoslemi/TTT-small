def compute_objective(f):
    """Compute the objective value for a normalized sequence (sum=1)."""
    h = np.convolve(f, f)
    n = len(f)
    return 2 * n * h.max()  # Since sum(f)=1, (sum f)² = 1

def subgradient_step(f, lr, softness=100):
    """One subgradient step to minimize max(f*f)."""
    n = len(f)
    h = np.convolve(f, f)
    M = h.max()
    
    if M == 0:
        return f
    
    # Soft weights: how close each lag is to the max
    # w_k = exp(softness * (h[k] - M) / M) / sum(...)
    # This gives more weight to lags near the max
    weights = np.exp(softness * (h - M) / M)

    sg = np.zeros(n)

        hi = min(n - 1, k)
        indices = np.arange(lo, hi + 1)

    # Step

    s = f_new.sum()
    if s < 1e-10:
        f_new = np.ones(n) / n
    else:
        f_new = f_new / s
    return f_new

    """Solve the LP to find direction g."""
    n = len(f)
    
    # Build constraint matrix C where C[k,l] = f[k-l] for valid k-l
    # C has shape (2n-1, n)
    # c = -ones(n) (to maximize sum(g), we minimize -sum(g))
    c = -np.ones(n)

            C[k, l] = f[k - l]

        s = g.sum()
        if s > 0:
            g = g / s
        else:
            g = np.ones(n) / n
        return g
    else:
        return np.ones(n) / n

    init_time = time.monotonic()

    # Precompute FFT of f (will update each iteration)
    f_pad = np.zeros(N)
    f_pad[:n] = f
    
    # Parameters
    lr = 0.05
    softness = 20.0
    lp_every = 50  # Do LP every 50 subgradient steps
    
    iteration = 0
    best_local = f.copy()
    best_local_score = float(evaluate_sequence(f.tolist()))
    
    while time.monotonic() < deadline and (time.monotonic() - init_time) < budget_per_init:
        iteration += 1
        
        # Compute autocorrelation
        h = np.convolve(f, f)
        M = h.max()
        
        if M < 1e-15:
            break
        
        # Soft weights (higher softness = sharper focus on peak)
        # Increase softness over time
        current_softness = softness + iteration * 0.1
        scaled = (h - M) / M  # in [-1, 0]
        weights = np.exp(current_softness * scaled)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.zeros_like(h)
            weights[int(np.argmax(h))] = 1.0
        else:
            weights /= wsum
        
        # Compute subgradient using FFT
        V_pad = np.zeros(N)
        V_pad[:2*n-1] = weights
        f_pad[:n] = f
        fft_f = np.fft.rfft(f_pad)
        fft_V = np.fft.rfft(V_pad)
        sg = np.fft.irfft(fft_V * np.conj(fft_f))[:n] * 2
        
        # Center
        sg -= sg.mean()
        
        # Adaptive learning rate
        # Scale by the norm of the subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 0:
            step = lr / np.sqrt(iteration + 1)
        else:
            step = 0
        
        # Update
        f = f - step * sg
        f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-10:
            f = np.ones(n) / n
        else:
            f = f / s
        
        # Periodic LP refinement

            M = h.max()
            g = solve_lp(f, M, n)

1. Compute h = f*f, find the peak at lag k*.
2. The peak value is h[k*] = sum_j f[j]*f[k*-j].
3. To reduce this, we want to make the product f[j]*f[k*-j] smaller for the pairs that contribute most.
4. A simple approach: for the pair (j, k*-j) with the largest product, transfer a small amount of mass from the larger to the smaller.
5. But we need to check that this doesn't increase other lags too much.

This is like a "coordinate descent" approach. Let me implement it:

```python
def peak_shave_step(f, n, num_pairs=10):
    """Reduce the peak of f*f by adjusting pairs."""
    h = np.convolve(f, f)
    k_max = int(np.argmax(h))
    M = h[k_max]
    
    # Find the pairs (j, k_max-j) that contribute to h[k_max]
    # Valid j: 0 <= j < n and 0 <= k_max-j < n
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    j_range = np.arange(lo, hi + 1)
    products = f[j_range] * f[k_max - j_range]
    
    # Sort by product (largest first)
    order = np.argsort(products)[::-1]
    
    for idx in order[:num_pairs]:
        j = j_range[idx]
        k = k_max - j
        if j == k:
            continue  # Skip self-pairs for simplicity
        # Transfer mass from larger to smaller
        if f[j] > f[k]:
            source, target = j, k
        else:
            source, target = k, j
        # Small transfer
        delta = 0.01 * f[source]
        f[source] -= delta
        f[target] += delta
    
    f = np.maximum(f, 0)
    f = f / f.sum()
    return f
```

def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix for f."""
    n = len(f)
    col_idx = np.repeat(np.arange(n), n)
    row_blocks = np.arange(n*n).reshape(n, n) + np.arange(n)[:, None]
