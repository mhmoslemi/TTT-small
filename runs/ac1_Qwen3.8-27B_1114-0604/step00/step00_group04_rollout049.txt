"""
Minimize 2n*max(convolve(a,a))/(sum(a))^2 using gradient descent on the
max-convolution subgradient with asymmetric restarts, soft active sets,
and adaptive step sizes. FFT-based for O(n log n) per step.
"""

import numpy as np
import time


def _fft_convolution(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.rfft(a, N)
    c = np.fft.irfft(F * F, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _compute_subgradient(a, w, N_conv):
    """
    Compute g[j] = 2 * sum_{m=0}^{n-1} a[m] * w[j+m] for j=0..n-1.
    w is the (soft) active mask of length 2n-1.
    Uses FFT cross-correlation trick.
    """
    n = len(a)
    w_rev = w[::-1]
    A_f = np.fft.rfft(a, N_conv)
    W_f = np.fft.rfft(w_rev, N_conv)
    conv = np.fft.irfft(A_f * W_f, N_conv)[:3 * n - 2].real
    # g[j] = 2 * conv[2n-2-j], so g = 2 * conv[n-1 : 2n-1][::-1]
    g = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return g


def _soft_active_mask(c, M, frac=0.05):
    """Soft active mask: 1 where c >= M*(1-frac), linear ramp below."""
    threshold = M * (1.0 - frac)
    w = np.zeros_like(c)
    active = c >= threshold
    w[active] = 1.0
    # Smooth transition in band [M*(1-2*frac), M*(1-frac)]
    lower = M * (1.0 - 2.0 * frac)
    band = (c >= lower) & (c < threshold)
    if np.any(band):
        w[band] = (c[band] - lower) / (threshold - lower)
    return w

def _local_search(a, n, deadline, rng, best_score):
