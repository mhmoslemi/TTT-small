def compute_subgradient(a, K_star):
    """Compute the subgradient of max B[k] at the given peak set K_star."""
    n = len(a)
    g = np.zeros(n)
    for k in K_star:
        m_lo = max(0, k - n + 1)
        m_hi = min(k, n - 1)
        if m_lo > m_hi:
            continue
        m_range = np.arange(m_lo, m_hi + 1)
        km_range = k - m_range
        contrib = 2.0 * a[km_range]
        # Fix diagonal
        if k % 2 == 0:
            diag_m = k // 2
            if m_lo <= diag_m <= m_hi:
                contrib[diag_m - m_lo] = a[diag_m]
        g[m_range] += contrib
    g /= len(K_star)
    return g

import numpy as np
import time

def propose_candidate(seed=42, budget_s=2400):
    """
    Minimize 2n*max(conv(a,a))/sum(a)^2 using exponentiated gradient descent
    with energy-based smooth gradient. Normalizes sum(a)=sqrt(2n) so the
    objective equals max(conv(a,a)).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 5.0
    
    # Get initial sequence
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(parent)
    S = np.sqrt(2.0 * n)
    
    s = parent.sum()
    a = parent / s * S if s > 1e-10 else np.ones(n) * S / n
    
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial: n={n}, score={best_score:.6f}")
    
    eta = 0.1  # step size
    patience = 0
    
    iters = 0
    while time.monotonic() < deadline:
        B = np.convolve(a, a)
        max_b = B.max()
        
        # Energy gradient: g[m] = 4 * sum_j B[m+j] * a[j]
        C = np.correlate(B, a, mode='valid')
        g = 4.0 * C
        
        # Also add peak gradient for direct peak reduction
        k_peak = np.argmax(B)
        g_peak = np.zeros(n)
        m_lo = max(0, k_peak - n + 1)
        m_hi = min(k_peak, n - 1)
        g_peak[m_lo:m_hi+1] = 2.0 * a[k_peak - np.arange(m_lo, m_hi + 1)]
        
        # Combine
        g_total = g + g_peak
        scale = np.max(np.abs(g_total)) + 1e-30
        g_norm = g_total / scale
        
        # Exponentiated gradient update
        a_new = a * np.exp(-eta * g_norm)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new / a_new.sum() * S
        
        score = evaluate_sequence(a_new.tolist())
        iters += 1
        
        if score < best_score:
            best_a = a_new.copy()
            best_score = score
            a = a_new
            patience = 0
            # Adaptively increase eta
            eta = min(eta * 1.01, 2.0)
            if iters <= 5 or best_score < 1.51 or iters % 200 == 0:
                print(f"iter={iters}, score={best_score:.6f}, max_b={max_b:.4f}, eta={eta:.4f}, t={time.monotonic()-start_time:.1f}s")
        else:
            patience += 1
            if patience == 10:
                eta *= 0.7
            elif patience == 50:
                # Random perturbation
                a = best_a * np.exp(rng.normal(0, 0.3, n))
                a = np.maximum(a, 0)
                a = a / a.sum() * S
                eta = 0.1
                patience = 0
    
    print(f"Final: score={best_score:.6f}, iters={iters}, time={time.monotonic()-start_time:.1f}s")
    return [float(x) for x in best_a]
