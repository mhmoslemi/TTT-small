def _project_simplex(v):
    n=v.size
    u=np.sort(v)[::-1]
    css=np.cumsum(u)
    idx=np.arange(1,n+1,dtype=float)
    cond = u > (css - 1.0) / idx
    # find last True
    # if no True? theta = max(v)-? but sum constraint; there is at least one.
    rho = np.flatnonzero(cond)[-1]
    theta = (css[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # If due roundoff sum not 1
    s = w.sum()
    if s <= 0:
        return np.full(n, 1.0/n)
    return w / s

           no_improve += 1

           a = _norm_array(best * np.exp(kick))
           bt = min(bt * 1.1, beta_max)
           lr = max(lr, 0.5)

           it = 0

       bc = beta * c
       bcmax = np.max(bc)

def _local_search(a, deadline, seed=0, beta_target=4.0, beta_max=80.0):
    a = _norm_array(a)
    n = a.size
    best = a.copy()
    best_m = _conv_max(best)
    if n <= 1 or deadline <= time.monotonic():
        return best, best_m
    rng = np.random.default_rng(seed)
    lr = 1.0
    bt = float(beta_target)
    prev_mx = None
    no_improve = 0
    it = 0
    while time.monotonic() < deadline:
       c = _fft_conv(a)
       mx = float(c.max())
       if mx < best_m:
           best = a.copy(); best_m = mx; no_improve = 0
       else:
           no_improve += 1
       if no_improve > 2500:
           cand = _fresh_random_simplex(n, rng)
           cand_m = _conv_max(cand)
           if cand_m < best_m:
               a = cand; best = a.copy(); best_m = cand_m; mx = cand_m
           else:
               a = _norm_array(best * np.exp(0.06 * rng.standard_normal(n)))
               mx = _conv_max(a)
           bt = min(bt * 1.1, beta_max)
           lr = max(lr, 0.5)
           no_improve = 0
           it = 0
           prev_mx = mx
           continue
       if mx <= 0.0: break
       beta = bt / mx
       bc = beta * c
       bcm = float(np.max(bc))
       ez = np.exp(bc - bcm)
       se = float(ez.sum())
       if not np.isfinite(se) or se <= 0.0:
           p = np.ones_like(c) / c.size
       else:
           p = ez / se
       g = 2.0 * _cross_corr(p, a)
       gc = g - float(np.mean(g))
       if not np.all(np.isfinite(gc)) or float(np.max(np.abs(gc))) < 1e-300:
           a = _norm_array(a * np.exp(0.03 * rng.standard_normal(n)))
           continue
       accepted = False
       for f in (lr, lr*0.5, lr*0.25):
           a_try = _project_simplex(a - f * gc)
           m_try = _conv_max(a_try)
           if m_try < mx:
               a = a_try; mx = m_try; accepted = True; break
       if not accepted:
           for f in (lr*0.1, lr*0.05):
               a_try = a * np.exp(-f * gc)
               a_try = np.clip(a_try, 0.0, None)

               if m_try < mx:

"""
Hybrid search for low-peak nonnegative convolution sequences.

Strategy:
  1. Generate many nonnegative simplex-normalized starts:
       - structured oscillatory starts (sine/cosine, chirps, Thue-Morse,
         Legendre sequences when length is prime),
       - monotone / V-shaped starts,
       - low-frequency random Fourier / block starts.
  2. Polish candidates with:
       - a projected-gradient / mirror-descent local search minimizing
         max_k convolve(a, a)[k],
       - optional L-BFGS-B polish on a smoothed log-sum-exp surrogate.
  3. Continue polishing the current best until the deadline, then return
     the best sequence.

The objective used internally is scale invariant:
  score(a) = 2 * n * max(convolve(a,a)) / sum(a)^2.
We keep sum(a)=1, so minimizing max(convolve(a,a)) is equivalent.
"""

import numpy as np
import time

    from scipy.optimize import minimize as _sp_minimize
    _HAVE_SCIPY = True
except Exception:
    _HAVE_SCIPY = False


def _score_key(item):
    return item[0]


def _is_prime(n):
    n = int(n)
    if n < 2:
        return False
    if n % 2 == 0:
        return n == 2
    d = 3
    while d * d <= n:
        if n % d == 0:
            return False
        d += 2
    return True


def _thue_values(n):
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    out = np.empty(n, dtype=np.float64)
    for i in range(n):
        out[i] = 1.0 if (bin(i).count("1") % 2 == 0) else -1.0
    return out


def _legendre_values(p):
    p = int(p)
    if p < 2:
        return np.empty(0, dtype=np.float64)
    if not _is_prime(p):
        p = max(2, p - 1)
        while p < 2 or not _is_prime(p):
            p -= 2
    out = np.empty(p, dtype=np.float64)
    e = (p - 1) // 2
    for i in range(p):
        if i == 0:
            out[i] = -1.0
        else:
            w = pow(i, e, p)
            out[i] = 1.0 if w == 1 else -1.0
    return out


def _norm_array(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.empty(0, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.full(arr.shape, 1.0 / arr.size, dtype=np.float64)
    return arr / s


def _next_pow2(x):
    p = 1
    while p < int(x):
        p <<= 1
    return p


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    fs = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fs)
    c = np.fft.irfft(fa * fa, fs)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    return c


def _conv_max(a):
    c = _fft_conv(a)
    if c.size == 0:
        return 0.0
    return float(c.max())


def _cross_corr(p, a):
    p = np.asarray(p, dtype=np.float64).ravel()
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = p.size
    fs = _next_pow2(L + n - 1)
    fp = np.fft.rfft(p, fs)
    fa = np.fft.rfft(a, fs)
    out = np.fft.irfft(fp * np.conj(fa), fs).real[:n]
    return out


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([1.0], dtype=np.float64)

    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (css - 1.0) / idx
    idxs = np.flatnonzero(cond)
    if idxs.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)

    rho = int(idxs[-1])
    theta = (css[rho] - 1.0) / float(rho + 1)
    w = np.maximum(v - theta, 0.0)
    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / sw


def _fresh_random_simplex(n, rng):
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([1.0], dtype=np.float64)

    i = np.arange(n, dtype=np.float64)
    r = float(rng.random())

    if r < 0.35:
        alpha = float(rng.uniform(0.02, 1.00))
        amp = float(rng.uniform(0.5, 1.0))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        v = 1.0 + amp * np.cos(2.0 * np.pi * alpha * i * i + ph)
    elif r < 0.60:
        kmax = max(2, min(n // 3, 16))
        k = int(rng.integers(1, kmax + 1))
        amp = float(rng.uniform(0.5, 1.0))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        v = 1.0 + amp * np.sin(2.0 * np.pi * k * i / n + ph)
    elif r < 0.80:
        q = int(rng.integers(1, 6))
        total_amp = float(rng.uniform(0.35, 0.95))
        coeffs = rng.dirichlet(np.ones(q, dtype=np.float64)) * total_amp
        v = np.ones(n, dtype=np.float64)
        fmax = max(2, min(n, max(2, n // 2)))
        for qq in range(q):
            f = int(rng.integers(1, fmax + 1))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            aq = float(coeffs[qq])
            if rng.random() < 0.5:
                v += aq * np.sin(2.0 * np.pi * f * i / n + ph)
            else:
                v += aq * np.cos(2.0 * np.pi * f * i / n + ph)
    else:
        v = np.ones(n, dtype=np.float64)
        nb = int(rng.integers(2, max(2, min(8, n)) + 1))
        for _bb in range(nb):
            s0 = int(rng.integers(0, max(1, n)))
            w = int(rng.integers(1, max(2, n // 20 + 1)))
            e = min(n, s0 + w)
            v[s0:e] *= float(rng.uniform(0.2, 3.0))

    return _norm_array(v)


def _chirp_sequence(n, alpha, amp, phase, use_sin=False):
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    arg = 2.0 * np.pi * float(alpha) * i * i + float(phase)
    if use_sin:
        v = 1.0 + float(amp) * np.sin(arg)
    else:
        v = 1.0 + float(amp) * np.cos(arg)
    return _norm_array(v)

def _select_candidates(pool, max_total=40):
    if not pool:
        return []

    pool.sort(key=_score_key)

    seen_n = set()
    per_n = []
    for item in pool:
        n = int(item[2])
        if n not in seen_n:
            seen_n.add(n)
            per_n.append(item)

    candidates = []
    seen_ids = set()

    for item in per_n:
        aid = id(item[1])
        if aid not in seen_ids:
            seen_ids.add(aid)
            candidates.append(item)

    for item in pool:
        if len(candidates) >= max_total:
            break
        aid = id(item[1])
        if aid not in seen_ids:
            seen_ids.add(aid)
            candidates.append(item)

    candidates.sort(key=_score_key)
    return candidates


def _lse_objective_grad(z, beta):
    z = np.asarray(z, dtype=np.float64).ravel()
    n = z.size
    if n == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return float("inf"), np.zeros(n, dtype=np.float64)

    # Remove the flat additive direction induced by normalization.
    z = z - float(np.mean(z))

    mz = float(np.max(z))
    v = np.exp(z - mz)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 0.0:
        return float("inf"), np.zeros(n, dtype=np.float64)

    a = v / s
    c = _fft_conv(a)
    if c.size == 0:
        return 0.0, np.zeros(n, dtype=np.float64)

    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1.0

    xr = beta * c
    xmr = float(np.max(xr))
    ez = np.exp(xr - xmr)
    se = float(ez.sum())
    if not np.isfinite(se) or se <= 0.0:
        val = float(np.max(c))
        grad = np.zeros(n, dtype=np.float64)
        return val, grad

    val = float(xmr / beta + np.log(se) / beta)
    p = ez / se
    f = 2.0 * _cross_corr(p, a)
    dot = float(np.dot(a, f))
    gy = a * (f - dot)
    gz = gy - float(np.mean(gy))
    if not np.all(np.isfinite(gz)):
        gz = np.zeros(n, dtype=np.float64)
    return val, gz


def _local_search(a, deadline, seed=0, beta_target=4.0, beta_max=80.0):
    a = _norm_array(a)
    n = a.size
    if n <= 1 or deadline <= time.monotonic():
        best = a.copy()
        best_m = _conv_max(best)
        return best, best_m

    best = a.copy()
    best_m = _conv_max(best)

    rng = np.random.default_rng(int(seed))
    lr = 1.0
    bt = float(beta_target)
    prev_mx = None
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        c = _fft_conv(a)
        mx = float(c.max()) if c.size else 0.0

        if mx < best_m:
            best = a.copy()
            best_m = mx
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2500:
            cand = _fresh_random_simplex(n, rng)
            cand_m = _conv_max(cand)
            if cand_m < best_m:
                a = cand
                best = a.copy()
                best_m = cand_m
                mx = cand_m
            else:
                a = _norm_array(best * np.exp(0.06 * rng.standard_normal(n)))
                mx = _conv_max(a)
            bt = min(bt * 1.1, beta_max)
            lr = max(lr, 0.5)
            no_improve = 0
            it = 0
            prev_mx = mx
            continue

        if mx <= 0.0:
            break

        beta = bt / mx
        bc = beta * c
        bcm = float(np.max(bc))
        ez = np.exp(bc - bcm)
        se = float(ez.sum())
        if not np.isfinite(se) or se <= 0.0:
            p = np.ones_like(c) / float(c.size)
        else:
            p = ez / se

        g = 2.0 * _cross_corr(p, a)
        gc = g - float(np.mean(g))

        if not np.all(np.isfinite(gc)) or float(np.max(np.abs(gc))) < 1e-300:
            a = _norm_array(a * np.exp(0.03 * rng.standard_normal(n)))
            continue

        accepted = False

        # Euclidean projected-step attempts.
        for f in (lr, lr * 0.5, lr * 0.25):
            a_try = _project_simplex(a - f * gc)
            m_try = _conv_max(a_try)
            if m_try < mx:
                a = a_try
                mx = m_try
                accepted = True
                break

        # Multiplicative / mirror-descent attempts.
        if not accepted:
            for f in (lr * 0.1, lr * 0.05):
                a_try = a * np.exp(-f * gc)
                a_try = np.clip(a_try, 0.0, None)
                s = float(a_try.sum())
                if s > 1e-300:
                    a_try = a_try / s
                    m_try = _conv_max(a_try)
                    if m_try < mx:
                        a = a_try
                        mx = m_try
                        accepted = True
                        break

        # Block multiplier perturbations.
        if not accepted:
            for scale in (0.08, 0.03, 0.012):
                width = int(rng.integers(max(1, n // 100), max(2, n // 8 + 1)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                stop = min(n, start + width)
                a_try = a.copy()
                a_try[start:stop] *= np.exp(rng.normal(0.0, scale, size=stop - start))
                a_try = _norm_array(a_try)
                m_try = _conv_max(a_try)
                if m_try < mx:
                    a = a_try
                    mx = m_try
                    accepted = True
                    break

        if not accepted:
            if no_improve > 200 or rng.random() < 0.25:
                a = _norm_array(best * np.exp(0.05 * rng.standard_normal(n)))
            else:
                a = _project_simplex(a + 0.01 * (a + 1.0 / n) * rng.standard_normal(n))
            mx = _conv_max(a)
            if mx < best_m:
                best = a.copy()
                best_m = mx
                no_improve = 0
        else:
            if prev_mx is not None and mx < prev_mx * (1.0 - 1e-7):
                lr = min(lr * 1.04, 10.0)
            else:
                lr = max(lr * 0.95, 1e-3)
            if no_improve > 100:
                bt = min(bt * 1.03, beta_max)

        if not accepted:
            lr = max(lr * 0.85, 1e-3)
            bt = min(bt * 1.05, beta_max)

        prev_mx = mx
        it += 1

    final_m = _conv_max(a)
    if final_m < best_m:
        best = a.copy()
        best_m = final_m

    return best, best_m


def _scipy_polish(a, deadline, seed=0):
    a = _norm_array(a)
    best = a.copy()
    best_m = _conv_max(best)

    if not _HAVE_SCIPY or a.size <= 1 or deadline <= time.monotonic():
        return best, best_m

    n = a.size
    rng = np.random.default_rng(int(seed))

    if n <= 64:
        base_iter = 180
    elif n <= 512:
        base_iter = 90
    elif n <= 2048:
        base_iter = 45
    else:
        base_iter = 18

    for ti, tgt in enumerate((10.0, 40.0)):
        for trial in range(2):
            if time.monotonic() >= deadline:
                break

            mx = _conv_max(a)
            if mx <= 0.0:
                break

            beta = float(tgt / mx)
            z0 = np.log(a + 1e-300)
            if trial:
                z0 = z0 + 0.02 * rng.standard_normal(n)
            z0 = z0 - float(np.mean(z0))

            maxiter = max(5, base_iter - 5 * trial)
            try:
                res = _sp_minimize(
                    _lse_objective_grad,
                    z0,
                    jac=True,
                    args=(beta,),
                    method="L-BFGS-B",
                    options={
                        "maxiter": int(maxiter),
                        "ftol": 1e-12,
                        "gtol": 1e-7,
                        "maxls": 20,
                    },
                )
            except Exception:
                res = None

            if res is not None and hasattr(res, "x"):
                z2 = np.asarray(res.x, dtype=np.float64).ravel()
                if z2.size == n:
                    z2 = z2 - float(np.mean(z2))
                    mz2 = float(np.max(z2))
                    v2 = np.exp(z2 - mz2)
                    s2 = float(v2.sum())
                    if np.isfinite(s2) and s2 > 0.0:
                        a2 = v2 / s2
                        m2 = _conv_max(a2)
                        if m2 < best_m:
                            best = a2.copy()
                            best_m = m2
                            a = a2.copy()

            # A short exact local polish after each smoothed optimization attempt.
            if time.monotonic() < deadline - 2.0:
                local_dl = min(deadline - 1.0, time.monotonic() + 20.0)
                try:
                    a2, m2 = _local_search(a, local_dl, seed=int(seed) + 31 + 7 * int(ti) + trial)
                    if m2 < best_m:
                        best = a2.copy()
                        best_m = m2
                        a = a2.copy()
                except Exception:
                    pass

    return best, best_m

    best_score = float(2.0 * candidates[0][1].size * _conv_max(candidates[0][1]))
    best_seq = _norm_array(candidates[0][1])
    print(f"initial best: {best_score:.6f} n={best_seq.size}")

    if total > 600.0:
        work_deadline = min(final_deadline - 240.0, t0 + total - 240.0)
    else:
        work_deadline = min(final_deadline - 30.0, t0 + total - 30.0)

    if work_deadline < t0 + 1.0:
        work_deadline = final_deadline

    allocs = [800.0, 500.0, 350.0, 250.0, 200.0, 150.0, 120.0, 100.0, 80.0, 70.0]

    for idx, item in enumerate(candidates):
        now = time.monotonic()
        if now >= work_deadline:
            break

        remaining = work_deadline - now
        alloc = allocs[idx] if idx < len(allocs) else 40.0
        alloc = min(alloc, remaining)
        if alloc < 5.0:
            break

        cand = _norm_array(item[1])
        n = int(cand.size)
        cand_deadline = now + alloc

        # Optional smoothed optimizer for candidates that are not excessively large.
        if _HAVE_SCIPY and n <= 16384 and remaining > 60.0:
            sc_dl = min(cand_deadline - 5.0,
                        now + min(90.0, (cand_deadline - now) * 0.25))
            if sc_dl > now + 2.0:
                try:
                    sp_seq, sp_m = _scipy_polish(cand, sc_dl, seed=int(seed) + 101 + 17 * idx)
                except Exception:
                    sp_seq, sp_m = cand, _conv_max(cand)
                sc = float(2.0 * n * sp_m)
                if sc < best_score:
                    best_score = sc
                    best_seq = _norm_array(sp_seq)
                    print(f"scipy improved: n={n} score={best_score:.6f} time={time.monotonic()-t0:.1f}")
                cand = _norm_array(sp_seq)

        # Exact local search for the rest of this candidate's allocation.
        try:
            ls_seq, ls_m = _local_search(cand, cand_deadline, seed=int(seed) + 211 + 29 * idx)
        except Exception:
            ls_seq, ls_m = cand, _conv_max(cand)

        sc = float(2.0 * ls_seq.size * ls_m)
        if sc < best_score:
            best_score = sc
            best_seq = _norm_array(ls_seq)
            print(f"local improved: n={best_seq.size} score={best_score:.6f} time={time.monotonic()-t0:.1f}")

    # Optional final smoothed refinement on the current best.
    final_polish_deadline = final_deadline - 60.0
    if _HAVE_SCIPY and time.monotonic() < final_polish_deadline and best_seq.size <= 16384:
        rem = max(0.0, final_polish_deadline - time.monotonic())
        dl = min(final_polish_deadline, time.monotonic() + max(20.0, min(150.0, 0.5 * rem)))
        try:
            sc_seq, sc_m = _scipy_polish(best_seq, dl, seed=int(seed) + 555)
            sc = float(2.0 * sc_seq.size * sc_m)
            if sc < best_score:
                best_score = sc
                best_seq = _norm_array(sc_seq)
                print(f"final scipy improved: score={best_score:.6f}")
        except Exception:
            pass

    # Final exact local polish until just before the deadline.
    if time.monotonic() < final_deadline:
        try:
            ls_seq, ls_m = _local_search(best_seq, final_deadline, seed=int(seed) + 777)
            sc = float(2.0 * ls_seq.size * ls_m)
            if sc < best_score:
                best_score = sc
                best_seq = _norm_array(ls_seq)
        except Exception:
            pass

    final_list = [float(x) for x in best_seq]

    try:
        ev = evaluate_sequence(final_list)
        if not np.isfinite(ev):
            raise ValueError
    except Exception:
        final_list = [1.0 / 4096.0 for _ in range(4096)]
        try:
            evaluate_sequence(final_list)
        except Exception:
            pass

    return final_list
