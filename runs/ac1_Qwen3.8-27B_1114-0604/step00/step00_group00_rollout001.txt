    print(f"Starting search with budget {budget_s}s")
    
    # Try a variety of uniform sequences of different lengths
    best_seq = None
    best_score = np.inf
    
    # Test several uniform sequences
    for n in [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 4806, 8192, 16384]:
        seq, score = try_uniform(n, 1.0)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    
    print(f"Best uniform score: {best_score:.10f} (length {len(best_seq)})")
    
    # Verify with the actual evaluator
    if best_seq is not None:
        verified_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Verified score with evaluator: {verified_score:.10f}")
        if verified_score < best_score:
            best_score = verified_score
    
    # Try the parent sequence if available
    parent_available = False
    try:
        if 'height_sequence_1' in dir() or 'height_sequence_1' in globals():
            parent = np.asarray(height_sequence_1, dtype=np.float64)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 0:
                parent_available = True
                parent_score = compute_score_fast(parent)
                print(f"Parent sequence: length={parent.size}, fast_score={parent_score:.10f}")
                
                # Verify parent with actual evaluator
                verified_parent = float(evaluate_sequence(parent.tolist()))
                print(f"Parent verified score: {verified_parent:.10f}")
                
                if verified_parent < best_score:
                    best_score = verified_parent
                    best_seq = parent.copy()
    except Exception as e:
        print(f"Could not access parent sequence: {e}")
    
    # Now do some exploration to see if we can beat 2.0
    # Try various structured sequences
    print("\nExploring structured sequences...")
    
    exploration_start = time.monotonic()
    
    # Try cosine-modulated sequences
    for n in [64, 128, 256, 512, 1024]:
        for freq in [1, 2, 3, 5, 8]:
            t = np.arange(n)
            seq = 1.0 + 0.5 * np.cos(2 * np.pi * freq * t / n)
            seq = np.clip(seq, 0.0, 1000.0)
            score = compute_score_fast(seq)
            if score < best_score - 1e-12:
                print(f"  Found better! n={n}, freq={freq}, score={score:.10f}")
                best_score = score
                best_seq = seq.copy()
    
    # Try random sequences
    for n in [16, 32, 64, 128, 256]:
        for _ in range(1000):
            seq = rng.random(n) + 0.1
            score = compute_score_fast(seq)
            if score < best_score - 1e-12:
                print(f"  Found better with random! n={n}, score={score:.10f}")
                best_score = score
                best_seq = seq.copy()
    
    # Try the LP-based approach on the parent
    if parent_available and time.monotonic() < deadline:
        print("\nAttempting LP-based improvement...")
        improved, score = try_lp_based_improvement(parent, max_iters=50)
        if score < best_score - 1e-12:
            print(f"  LP improved! score={score:.10f}")
            best_score = score
            best_seq = improved.copy()
        else:
            print(f"  LP did not improve (score={score:.10f} vs best={best_score:.10f})")
    
    # Try block-based perturbation (similar to previous program but with verification)
    if time.monotonic() < deadline and best_seq is not None:
        print("\nBlock perturbation search...")
        base = best_seq.copy()
        n_base = len(base)
        iterations = 0
        perturbation_start = time.monotonic()
        
        while time.monotonic() < deadline:
            candidate = base.copy()
            width = max(2, n_base // 50)
            start = int(rng.integers(0, max(1, n_base - width)))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.05))
            candidate = np.clip(candidate, 0.0, 1000.0)
            
            score = compute_score_fast(candidate)
            iterations += 1
            
            if score < best_score - 1e-12:
                best_score = score
                best_seq = candidate.copy()
                base = candidate.copy()
                print(f"  iter {iterations}: improved to {score:.10f}")
            
            if iterations % 10000 == 0:
                elapsed = time.monotonic() - perturbation_start
                print(f"  iter {iterations}: best={best_score:.10f}, elapsed={elapsed:.1f}s")
    
    elapsed = time.monotonic() - start_time
    print(f"\nFinal result: score={best_score:.10f}, length={len(best_seq)}, time={elapsed:.1f}s")
    
    # Final verification with the actual evaluator
    final_seq = best_seq
    if final_seq is None:
        # Fallback: uniform sequence
        final_seq = np.ones(256, dtype=np.float64)
    
    # Ensure the sequence is valid
    final_list = [float(max(0.0, min(1000.0, x))) for x in final_seq]
    final_score = float(evaluate_sequence(final_list))
    print(f"Final verified score: {final_score:.10f}")
    
    return final_list
