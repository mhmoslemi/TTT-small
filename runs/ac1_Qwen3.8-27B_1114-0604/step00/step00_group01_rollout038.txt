def _fft_conv(a, b):
    """Linear convolution of two real arrays via FFT."""
    n = len(a) + len(b) - 1
    return np.fft.irfft(np.fft.rfft(a, n) * np.fft.rfft(b, n), n)

def _conv_self(a):
    """Compute a * a (linear self-convolution)."""
    n = len(a)
    size = 2 * n - 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)

def _build_conv_matrix(f, n):
    """Build sparse matrix A where (A @ x) = f * x (linear convolution)."""
    total_nz = sum(min(k+1, n, 2*n-1-k) for k in range(2*n-1))
    rows = np.empty(total_nz, dtype=np.int64)
    cols = np.empty(total_nz, dtype=np.int64)
    vals = np.empty(total_nz, dtype=np.float64)
    offset = 0
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        length = je - js + 1
        rows[offset:offset+length] = k
        cols[offset:offset+length] = np.arange(js, je + 1)
        vals[offset:offset+length] = f[k - np.arange(js, je + 1)]
        offset += length
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def _lp_solve(f, M):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0."""
    n = len(f)

                      method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-7, 'primal_feasibility_tolerance': 1e-7})

def _logsumexp_grad(a, beta):
    """
    Compute gradient of logsumexp(beta * (a*a)) w.r.t. log-param x.
    Returns (grad_x, c_max).

    n = len(a)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    # For numerical stability
    p = np.exp(shifted - shifted.max())
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones_like(c) / len(c)

    pc = float(np.dot(p, c))
    grad_x = 2.0 * beta * a * (g - pc)
    return grad_x, c_max
