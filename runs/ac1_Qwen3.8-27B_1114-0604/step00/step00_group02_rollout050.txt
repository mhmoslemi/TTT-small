"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative a.

Strategy
--------
Scale invariance lets us fix sum(a)=1.  The main engine is the LP
alternating-projection method: with a fixed and M = max(a*a), solve
    max sum(g)  s.t.  g >= 0,  (conv(a,g))[m] <= M  for all m,
then rescale g to the same sum and take a line-searched convex step
a <- (1-t)a + t g, which provably (for small t) never increases the
max self-convolution.  The LP is a structured (convolution/Toeplitz)
program solved by HiGHS on a sparse matrix.

For sequence lengths where the LP matrix is too large, or as a fast
polisher, we use a mirror-descent / multiplicative-weights step on a
log-sum-exp smoothed objective, and a exact-subgradient step on the
true max.  Several lengths and initialisations are tried within the
wall-clock budget; the best exact-evaluated sequence is returned.
All helpers are top-level (no closures, no lambdas, no IO).
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


# ----------------------------------------------------------------------
# helpers
# ----------------------------------------------------------------------
def _normalize_one(a):
    s = float(np.sum(a))
    if s <= 0.0:
        n = a.size
        return np.full(n, 1.0 / max(1, n), dtype=np.float64)
    return a / s


def _conv(a):
    """Exact linear convolution of a with itself."""
    return np.convolve(a, a, mode="full")


def _maxb_sm_t(a):
    b = _conv(a)
    return float(np.max(b)), b


def _exact_score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0:
        return np.inf
    b = _conv(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return float(2 * a.size * float(np.max(b)) / (s * s))


def _build_conv_csr(a):
    """Sparse (2n-1) x n matrix T with (T@g)[m] = (conv(a,g))[m]."""
    n = a.size
    rows = 2 * n - 1
    indptr = np.zeros(rows + 1, dtype=np.int64)
    cols_all = []
    data_all = []
    for m in range(rows):
        jlo = max(0, m - n + 1)
        jhi = min(m, n - 1)
        if jhi < jlo:
            continue
        j = np.arange(jlo, jhi + 1, dtype=np.int64)
        aidx = m - j
        cols_all.append(j)
        data_all.append(a[aidx])
        indptr[m + 1] = indptr[m] + j.size
    if data_all:
        cols = np.concatenate(cols_all)
        data = np.concatenate(data_all)
    else:
        cols = np.zeros(0, dtype=np.int64)
        data = np.zeros(0, dtype=np.float64)
    # fix indptr to be the cumulative offsets
    offsets = np.zeros(rows, dtype=np.int64)
    pos = 0
    for m in range(rows):
        jlo = max(0, m - n + 1)
        jhi = min(m, n - 1)
        ln = max(0, jhi - jlo + 1)
        offsets[m] = pos
        pos += ln
    indptr = np.concatenate([offsets, [pos]])
    return csr_matrix((data, cols, indptr), shape=(rows, n))


def _lp_best_response(a, M):
    """Solve max sum(g) s.t. conv(a,g) <= M, g>=0; return normalized g."""
    n = a.size
    T = _build_conv_csr(a)
    c = -np.ones(n)
    A_ub = T
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return None
    g = np.clip(res.x, 0.0, None)
    gs = float(np.sum(g))
    if gs <= 0.0:
        return None
    return g / gs


def _smooth_grad_step(a, b, eta, eps):
    """Multiplicative-weights step on log-sum-exp smoothed objective."""
    w = np.exp((b - np.max(b)) / eps)
    w = w / w.sum()
    # grad_a[j] = 2 * sum_m w[m] * a[m-j]  (drop constant 2)
    wa = np.convolve(w, a, mode="full")  # length 2n-1, [j] = sum_m w[m] a[m-j]? check index
    # np.convolve(w,a)[k] = sum_i w[i] a[k-i]; we want g[j]=sum_m w[m] a[m-j]
    # that's cross-correlation: use correlate
    ga = np.correlate(a, w, mode="full")
    # np.correlate(a,w,'full') length 2n-1; g[j] aligns: take first n
    ga = ga[:n] if ga.size >= n else np.pad(ga, (0, n - ga.size))
    step = np.clip(ga, 0.0, None)
    new = a * np.exp(-eta * step)
    new = np.clip(new, 0.0, None)
    return _normalize_one(new)


def _subgrad_step(a, b, eta):
    """Mirror step against the exact max subgradient."""
    m = int(np.argmax(b))
    n = a.size
    jlo = max(0, m - n + 1)
    jhi = min(m, n - 1)
    step = np.zeros(n)
    if jhi >= jlo:
        j = np.arange(jlo, jhi + 1)
        step[j] = a[m - j]
    new = a * np.exp(-eta * step)
    new = np.clip(new, 0.0, None)
    return _normalize_one(new)


def _line_search(a, g, cur):
    """Find blend t with best exact score; return (t, newa)."""
    for t in (1.0, 0.66, 0.4, 0.25, 0.15, 0.1, 0.05):
        cand = (1.0 - t) * a + t * g
        cand = np.clip(cand, 0.0, None)
        sc = _exact_score(cand)
        if sc < cur:
            return t, _normalize_one(cand), sc
    return 0.0, a, cur


def _optimize_one(a0, deadline, use_lp, lp_iters, quiet=False):
    a = _normalize_one(np.clip(np.asarray(a0, dtype=np.float64), 0.0, None))
    n = a.size
    M, b = _maxb_sm_t(a)
    best = a.copy()
    best_sc = _exact_score(best)
    if not quiet:
        print("  start n=%d score=%.6f" % (n, best_sc))
    it = 0
    while time.monotonic() < deadline:
        it += 1
        M, b = _maxb_sm_t(a)
        step_done = False
        if use_lp and lp_iters > 0:
            try:
                g = _lp_best_response(a, M)
                if g is not None:
                    t, newa, newsc = _line_search(a, g, best_sc)
                    if newsc < best_sc:
                        a = newa
                        best = a.copy()
                        best_sc = newsc
                        step_done = True
            except Exception:
                use_lp = False
        if not step_done:
            # smoothed / subgradient refinement
            eps = max(0.03 * float(np.max(b)), 1e-12)
            eta = 0.25 / max(1.0, float(np.max(b)) + 1e-12)
            c1 = _smooth_grad_step(a, b, eta, eps)
            c2 = _subgrad_step(a, b, eta)
            s1 = _exact_score(c1)
            s2 = _exact_score(c2)
            if s1 <= s2 and s1 < best_sc:
                a, best, best_sc = c1, c1.copy(), s1
            elif s2 < best_sc:
                a, best, best_sc = c2, c2.copy(), s2
            else:
                # anneal: nudge perturb to escape
                rng = np.random.default_rng(1000 + it)
                a2 = a * np.exp(rng.normal(0.0, 0.05, n))
                a2 = _normalize_one(a2)
                s3 = _exact_score(a2)
                if s3 < best_sc:
                    best = a2.copy()
                    best_sc = s3
                a = best
        if not quiet and (it % 20 == 0 or it < 5):
            print("  it=%d best=%.6f" % (it, best_sc))
    return best, best_sc


def _random_starts(n, rng):
    out = []
    t = np.arange(n) / max(1, n - 1)
    hann = 0.5 * (1.0 + np.cos(2 * np.pi * t)) + 1e-9
    out.append(hann)
    out.append(hann[::-1].copy() if False else hann)  # keep one
    raised = 0.5 * (1.0 - np.cos(2 * np.pi * t)) + 1e-9
    out.append(raised)
    # super-gaussians
    ctr = (n - 1) / 2.0
    for p in (2.0, 4.0, 1.0):
        g = np.exp(-np.abs(t - 0.5) ** p * 4.0) + 1e-9
        out.append(g)
    # ring / bimodal
    ring = np.exp(-((np.abs(t - 0.5) - 0.25) ** 2) / 0.01 + 1e-9)
    out.append(ring)
    # random positive
    for _ in range(2):
        out.append(np.abs(rng.standard_normal(n)) + 1e-6)
    return out


def propose_candidate(seed=42, budget_s=2400, **kwargs):

    Returns a python list of floats.  Runs until ~budget_s seconds, then
    returns the best exact-evaluated sequence found (always a valid list).
    """
    rng = np.random.default_rng(seed)
    total_budget = max(30.0, float(budget_s))
    t0 = time.monotonic()
    deadline = t0 + total_budget - 8.0  # leave margin before hard stop

    best = None
    best_score = np.inf
    best_len = 0

    def _consider(seq):
        nonlocal best, best_score, best_len
        if seq is None:
            return
        seq = np.clip(np.asarray(seq, dtype=np.float64), 0.0, 1000.0)
        if seq.size == 0:
            return
        sc = _exact_score(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = seq.copy()
            best_len = seq.size
            print("  ** new best  n=%d  score=%.6f  (t=%.0fs)" %
                  (seq.size, sc, time.monotonic() - t0), flush=True)

    # ---- initial parent ----------------------------------------------------------------
    parent = None
    try:
        p = globals().get("height_sequence_1", None)
        if p is not None:
            parent = np.clip(np.asarray(p, dtype=np.float64), 0.0, 1000.0)
            if parent.ndim != 1 or parent.size == 0:
                parent = None
    except Exception:
        parent = None
    if parent is None:
        parent = np.ones(2048, dtype=np.float64)
    _consider(parent)

    # ---- choose lengths to explore -----------------------------------------------------
    lengths = []
    pl = int(parent.size)
    for L in [512, 1024, 2048, 3000, 4096, 6144, pl]:
        if L > 0 and L not in lengths:
            lengths.append(L)
    lengths = [L for L in lengths if L <= 8192]
    lengths.sort()

    # time budget split: more time to the most promising (longer) runs,
    # but give every length a real chance.
    n_len = len(lengths)
    # weights favour mid/long lengths slightly
    w = [0.6 + 0.4 * (i / max(1, n_len - 1)) for i in range(n_len)]
    ws = sum(w)
    avail = max(20.0, deadline - time.monotonic())
    alloc = [avail * wv / ws for wv in w]

    for idx, L in enumerate(lengths):
        if time.monotonic() > deadline:

        sub_deadline = min(deadline, time.monotonic() + alloc[idx])
        use_lp = L <= 3000
        print("length n=%d  alloc=%.0fs  lp=%s" % (L, alloc[idx], use_lp),
              flush=True)

        # build candidate starts for this length
        if L == pl and parent is not None:

            starts += _random_starts(L, rng)
        else:
            if use_lp:
                base = parent[:L].copy() if parent.size >= L else parent

                    pad = np.zeros(L - base.size)
                    base = np.concatenate([base, pad]) if base.size else pad
                    base = base + 1e-6
                starts = [base]
            else:
                starts = []
            starts += _random_starts(L, rng)

        local_best = None
        local_sc = np.inf
        for si, st in enumerate(starts):
            if time.monotonic() > sub_deadline:
                break
            try:
                seq, sc = _optimize_one(st, sub_deadline, use_lp,
                                        lp_iters=10 ** 9, quiet=True)
                if sc < local_sc:
                    local_sc = sc
                    local_best = seq
            except Exception:
                pass
        if local_best is not None:
            _consider(local_best)

    # ---- final local polish on the best ---------------------------------------------
    if best is not None and time.monotonic() < deadline:
        print("polishing best (n=%d score=%.6f)" % (best.size, best_score),
              flush=True)
        a = np.clip(best, 0.0, None).copy()
        a = _normalize_one(a)
        n = a.size
        # a few LP steps if feasible

            for _ in range(20):
                if time.monotonic() > deadline:
                    break
                try:
                    M, b = _maxb_sm_t(a)
                    g = _lp_best_response(a, M)
                    if g is not None:
                        t, na, ns = _line_search(a, g, _exact_score(a))
                        if ns < _exact_score(a):
                            a = na
                            _consider(a)
                except Exception:
                    break
        # small multiplicative local search
        rng2 = np.random.default_rng(99)
        for _ in range(2000):
            if time.monotonic() > deadline:
                break
            width = max(2, int(n // 64))
            st = int(rng2.integers(0, max(1, n - width + 1)))
            c = a.copy()
            c[st:st + width] *= float(np.exp(rng2.normal(0.0, 0.05)))
            c = _normalize_one(c)
            _consider(c)

    # ---- guaranteed valid fallback ---------------------------------------------------
    if best is None or not np.all(np.isfinite(best)):
        best = np.ones(2048, dtype=np.float64)

    out = [float(x) for x in np.clip(best, 0.0, 1000.0)]
    print("FINAL n=%d score=%.6f" % (len(out), _exact_score(np.array(out))),
          flush=True)
    return out
