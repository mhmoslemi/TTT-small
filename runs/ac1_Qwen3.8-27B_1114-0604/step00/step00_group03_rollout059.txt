def project_simplex(v):
    n=len(v)
    if n==0: return v
    if not np.all(np.isfinite(v)): return np.ones(n)/n
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho=np.nonzero(u*np.arange(1,n+1) > (cssv-1.0))[0]
    if len(rho)==0:
        theta=0.0
    else:
        theta=(cssv[rho[-1]]-1.0)/(rho[-1]+1.0)
    proj=np.maximum(v-theta,0.0)
    s=proj.sum()
    if s<=0 or not finite: return np.ones(n)/n
    return proj/s
```

def linear_convolution_fft(x):
    x=np.asarray(x, dtype=np.float64)
    n=x.size
    if n==0: return np.array([])
    size=next_power_of_two(2*n-1)
    X=np.fft.rfft(x, size)
    y=np.fft.irfft(X*X, size)

def project_simplex(v):
    v=np.asarray(v,dtype=np.float64)
    n=v.size
    if n==0: return v
    if not np.all(np.isfinite(v)): return np.ones(n,dtype=np.float64)/n
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    idx=np.nonzero(u*np.arange(1,n+1,dtype=np.float64) > (cssv-1.0))[0]
    if idx.size == 0:
        theta=0.0
    else:
        k=int(idx[-1])
        theta=(cssv[k]-1.0)/(k+1.0)
    proj=v-theta
    np.maximum(proj,0.0,out=proj)
    s=proj.sum()
    if s <= 0 or not math.isfinite(s): return np.ones(n)/n
    proj/=s
    return proj
```

def make_bump_sequence(n, p, r, alpha):
    j=np.arange(n,dtype=np.float64)+0.5
    base=np.exp(-p*np.log(j))
    t=j/float(n)
    if r == 1: bump=t
    else: bump=np.power(t, r)
    a=base*(1.0+alpha*bump)
    a=np.where(np.isfinite(a),a,0.0)
    a=np.maximum(a,0.0)
    s=a.sum()
    if s<=0: a=np.ones(n)
    else: a/=s
    return a

def add_smooth_noise(a, rng, amp=0.1):
    a=np.asarray(a)
    n=a.size
    if n <= 1 or amp <=0: return a.copy()
    z=rng.normal(size=n)
    size=next_power_of_two(n)
    Z=np.fft.rfft(z,size)
    freq=np.arange(size//2+1, dtype=np.float64)
    sigma=max(1.0, n/20.0)
    Z *= np.exp(-(freq*freq)/(2.0*sigma*sigma))
    z=np.fft.irfft(Z,size)[:n]
    sd=float(np.std(z))
    if sd > 1e-12:
       z=(z-z.mean())/sd
    else:
       z=np.zeros(n)
    out=a*np.exp(amp*z)
    return np.maximum(out,0.0)

```python
"""
Search for non-negative step-function heights that minimize

    2 * n * max(conv(a, a)) / sum(a)^2.

Strategy:
- Use fast FFT-based scoring for most of the search.
- Start from a broad set of analytic seeds, especially one-sided power-law
  shapes, plus the provided parent sequence.
- Refine using:
  1) an LP-based tangent-direction iteration inspired by the previous
     state-of-the-art construction,
  2) Adam-style projected optimization in log-softmax coordinates with a
     smoothed maximum objective,
  3) local stochastic perturbations / hill climbing.
- Return the best sequence found before the deadline.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog


def next_power_of_two(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def linear_convolution_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = next_power_of_two(2 * n - 1)
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)
    b = b[:2 * n - 1]
    if b.size:
        b = np.maximum(b, 0.0)
    return b


def fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    n = a.size
    s = float(np.sum(a))
    if not math.isfinite(s) or s <= 1e-12:
        return np.inf
    b = linear_convolution_fft(a)
    if b.size == 0:
        return np.inf
    m = float(np.max(b))
    if not math.isfinite(m):
        return np.inf
    if m < 0.0:
        m = 0.0
    return float(2.0 * n * m / (s * s))


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.ones(n, dtype=np.float64) / float(n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.nonzero(u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0))[0]
    if idx.size == 0:
        theta = 0.0
    else:
        k = int(idx[-1])
        theta = (cssv[k] - 1.0) / float(k + 1)

    proj = np.maximum(v - theta, 0.0)
    s = float(np.sum(proj))
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return proj / s


def softmax(x):
    x = np.asarray(x, dtype=np.float64)
    x = x - float(np.max(x))
    e = np.exp(x)
    s = float(np.sum(e))
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(x.size, dtype=np.float64) / float(x.size)
    return e / s


def make_power_sequence(n, p, shift=0.5):
    j = np.arange(n, dtype=np.float64) + float(shift)
    if p == 0.0:
        a = np.ones(n, dtype=np.float64)
    else:
        a = np.exp(-float(p) * np.log(j))
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 0.0 or not math.isfinite(s):
        a = np.ones(n, dtype=np.float64)
    else:
        a /= s
    return a


def make_mix_sequence(n, p_left, p_right, alpha, shift=0.5):
    j = np.arange(n, dtype=np.float64) + float(shift)
    if p_left == 0.0:
        left = np.ones(n, dtype=np.float64)
    else:
        left = np.exp(-float(p_left) * np.log(j))

    right_pos = float(n) - j
    if p_right == 0.0:
        right = np.ones(n, dtype=np.float64)
    else:
        right = np.exp(-float(p_right) * np.log(right_pos))

    a = left + float(alpha) * right
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 0.0 or not math.isfinite(s):
        a = np.ones(n, dtype=np.float64)
    else:
        a /= s
    return a


def make_bump_sequence(n, p, r, alpha):
    j = np.arange(n, dtype=np.float64) + 0.5
    base = np.exp(-float(p) * np.log(j))
    t = j / float(n)
    if r == 1.0:
        bump = t
    else:
        bump = np.power(t, float(r))
    a = base * (1.0 + float(alpha) * bump)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 0.0 or not math.isfinite(s):
        a = np.ones(n, dtype=np.float64)
    else:
        a /= s
    return a


def add_smooth_noise(a, rng, amp=0.1):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1 or amp <= 0.0:
        return a.copy()
    z = rng.normal(size=n)
    size = next_power_of_two(n)
    zhat = np.fft.rfft(z, size)
    f = np.arange(size // 2 + 1, dtype=np.float64)
    sigma = max(1.0, n / 20.0)
    zhat *= np.exp(-(f * f) / (2.0 * sigma * sigma))
    z = np.fft.irfft(zhat, size)[:n]
    sd = float(np.std(z))
    if sd > 1e-12:
        z = (z - float(np.mean(z))) / sd
    else:
        z = np.zeros(n, dtype=np.float64)
    out = a * np.exp(float(amp) * z)
    return np.maximum(out, 0.0)


def correlate_weights(w, a):
    w = np.asarray(w, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = w.size
    if n == 0 or L == 0:
        return np.zeros(n, dtype=np.float64)
    size = next_power_of_two(n + L - 1)
    W = np.fft.fft(w, size)
    A = np.fft.fft(a, size)
    y = np.fft.ifft(W * np.conjugate(A)).real[:n]
    return y


def soft_grad_x(x, tau):
    a = softmax(x)
    n = a.size
    b = linear_convolution_fft(a)
    if b.size == 0:
        return 0.0, np.zeros(n, dtype=np.float64)

    maxb = float(np.max(b))
    s = 2.0 * float(n)
    z = s * float(tau) * (b - maxb)
    ez = np.exp(z)
    ez_sum = float(np.sum(ez))

    if not math.isfinite(ez_sum) or ez_sum <= 0.0:
        w = np.zeros_like(b)
        tol = 1e-12 * max(1.0, abs(maxb))
        idx = np.flatnonzero(b >= maxb - tol)
        if idx.size == 0:
            idx = np.array([int(np.argmax(b))], dtype=int)
        w[idx] = 1.0 / float(idx.size)
        loss = s * maxb
    else:
        w = ez / ez_sum
        loss = s * maxb + (1.0 / float(tau)) * math.log(ez_sum)

    grad_a = 2.0 * s * correlate_weights(w, a)
    grad_x = a * (grad_a - float(np.dot(grad_a, a)))
    return float(loss), grad_x


def adam_update(x, g, m, v, step, beta1, beta2, eps, lr):
    mt = beta1 * m + (1.0 - beta1) * g
    vt = beta2 * v + (1.0 - beta2) * g * g
    b1 = 1.0 - beta1 ** (step + 1)
    b2 = 1.0 - beta2 ** (step + 1)
    mh = mt / b1
    vh = vt / b2
    xn = x - lr * mh / (np.sqrt(vh) + eps)
    return xn, mt, vt


def build_conv_matrix_dense(f):
    n = f.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def run_lp_optimize(f0, deadline, max_iter=400, rng=None):
    if rng is None:
        rng = np.random.default_rng(101)

    f = project_simplex(np.asarray(f0, dtype=np.float64).copy())
    best = f.copy()
    best_score = fast_score(f)

    tilt_scale = 1e-7
    t_values = (0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 0.99)
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        b = linear_convolution_fft(f)
        M = float(np.max(b)) if b.size else 0.0
        if not math.isfinite(M) or M <= 0.0:
            break

        A = build_conv_matrix_dense(f)
        A /= M

        c = -np.ones(f.size, dtype=np.float64)
        r = rng.normal(size=f.size)
        rn = float(np.linalg.norm(r))
        if rn > 1e-30:
            c = c - (tilt_scale / rn) * r

        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=np.ones(2 * f.size - 1, dtype=np.float64),
                bounds=(0.0, None),
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            break
        finally:
            del A

        if res is None or res.x is None:
            break
        if int(res.status) != 0:
            break

        g = np.asarray(res.x, dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        gs = float(np.sum(g))
        if gs <= 1e-15:
            break
        g /= gs

        improved = False
        for t in t_values:
            cand = f + float(t) * (g - f)
            cand = project_simplex(cand)
            sc = fast_score(cand)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best = cand.copy()
                f = cand.copy()
                improved = True

        if not improved:
            cand = g.copy()
            cand = project_simplex(cand)
            sc = fast_score(cand)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best = cand.copy()
                f = cand.copy()
                improved = True

        if not improved:
            break

        it += 1

    return best, best_score


def run_soft_optimize(a0, deadline, max_iter=200000, tau_start=2.0, tau_end=1000.0, lr_start=0.07):
    a = project_simplex(np.asarray(a0, dtype=np.float64).copy())
    x = np.log(np.maximum(a, 1e-300))
    x = x - float(np.mean(x))
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best = a.copy()
    best_score = fast_score(a)

    step = 0
    tau = float(tau_start)
    no_improve = 0

    while time.monotonic() < deadline and step < max_iter:
        if step % 25 == 0:
            tau = min(float(tau_end), tau * 1.06)

        loss, g = soft_grad_x(x, tau)
        if not np.all(np.isfinite(g)):
            break

        if step < 1000:
            lr = lr_start
        elif step < 10000:
            lr = lr_start * 0.35
        else:
            lr = lr_start * 0.12

        x, m, v = adam_update(x, g, m, v, step, 0.9, 0.999, 1e-8, lr)
        x = x - float(np.mean(x))
        step += 1

        if step % 100 == 0:
            a = softmax(x)
            sc = fast_score(a)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 600:
                    x = np.log(np.maximum(best, 1e-300))
                    x = x - float(np.mean(x))
                    m = np.zeros_like(x)
                    v = np.zeros_like(x)
                    tau = max(float(tau_start), tau * 0.5)
                    no_improve = 0

    a = softmax(x)
    sc = fast_score(a)
    if math.isfinite(sc) and sc < best_score - 1e-15:
        best_score = sc
        best = a.copy()

    return best, best_score


def perturb_search(a0, deadline, rng, max_iter=500000, sigma0=0.20):
    f = project_simplex(np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy())
    best = f.copy()
    best_score = fast_score(f)
    n = f.size
    if n == 0:
        return best, best_score

    sigma = float(sigma0)
    no_improve = 0
    it = 0

    min_width = max(1, n // 800)
    max_width = max(min_width, n // 20)

    while time.monotonic() < deadline and it < max_iter:
        width = int(rng.integers(min_width, max_width + 1))
        if width >= n:
            width = n
            start = 0
        else:
            start = int(rng.integers(0, n - width + 1))

        cand = f.copy()
        scale = np.exp(rng.normal(0.0, sigma, size=width))
        cand[start:start + width] *= scale

        if rng.random() < 0.25:
            cand = 0.5 * (cand + cand[::-1])

        cand = project_simplex(np.maximum(cand, 0.0))
        sc = fast_score(cand)

        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc
            best = cand.copy()
            f = cand.copy()
            no_improve = 0
            if sigma < 0.55:
                sigma = min(0.65, sigma * 1.05)
        else:
            no_improve += 1
            if no_improve > 3000:
                f = best.copy()
                no_improve = 0
                sigma = max(0.03, sigma * 0.6)
            elif no_improve > 800 and rng.random() < 0.1:
                f = best.copy()
                no_improve = 0

        if it % 2000 == 0 and it > 0 and sigma < 0.02:
            sigma = 0.06

        it += 1

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start = time.monotonic()
    time_limit = max(1.0, float(budget_s))
    deadline = start + time_limit - 5.0
    if deadline < start + 1.0:
        deadline = start + 1.0

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Parent sequence, if available and reasonably sized.
    parent = globals().get("height_sequence_1", None)
    if parent is not None:
        try:
            parr = np.asarray(parent, dtype=np.float64)
            if parr.ndim == 1 and parr.size > 0:
                parr = np.where(np.isfinite(parr), parr, 0.0)
                parr = np.maximum(parr, 0.0)
                s = float(np.sum(parr))
                if s > 0.0 and math.isfinite(s) and parr.size <= 8192:
                    parr /= s
                    sc = fast_score(parr)
                    if math.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = parr.copy()
                    print(f"parent seq len={parr.size} fast_score={sc:.10f}")
        except Exception:
            pass

    # Build a broad pool of initial candidates.
    inits = []

    lengths = (256, 512, 1024, 2048, 4096)
    powers = (-0.50, -0.48, -0.45, -0.40, -0.35, -0.30, -0.25, -0.20, -0.15, -0.10, 0.0)
    mix_alphas = (0.1, 0.25, 0.5, 1.0, 2.0)
    bump_alphas = (0.1, 0.3, 1.0)

    for n in lengths:
        for p in powers:
            a = make_power_sequence(n, p, shift=0.5)
            sc = fast_score(a)
            inits.append((sc, a, n, "power"))
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a.copy()

        for alpha in mix_alphas:
            a = make_mix_sequence(n, 0.5, 0.5, alpha, shift=0.5)
            sc = fast_score(a)
            inits.append((sc, a, n, "mix"))
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a.copy()

        for alpha in bump_alphas:
            a = make_bump_sequence(n, 0.5, 2.0, alpha)
            sc = fast_score(a)
            inits.append((sc, a, n, "bump"))
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a.copy()

    # A few noisy one-sided power-law starts.
    for n in (512, 1024, 2048):
        for amp in (0.05, 0.15, 0.35):
            base = make_power_sequence(n, -0.5, shift=0.5)
            a = add_smooth_noise(base, rng, amp)
            s = float(np.sum(a))
            if s > 0.0 and math.isfinite(s):
                a /= s
                sc = fast_score(a)
                inits.append((sc, a, n, "noise-power"))
                if math.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a.copy()

    print(f"initial best fast_score={best_score:.10f}, len={0 if best_seq is None else best_seq.size}")

    # Keep a small set of the best initial candidates for augmentation.
    top = []
    keep = 10
    for item in inits:
        sc = item[0]
        if not math.isfinite(sc):
            continue
        if len(top) < keep:
            top.append(item)
        else:
            worst_i = 0
            worst_sc = top[0][0]
            for i in range(1, keep):
                if top[i][0] > worst_sc:
                    worst_sc = top[i][0]
                    worst_i = i
            if sc < worst_sc:
                top[worst_i] = item

    # Stage 1: LP-based tangent direction iteration.
    now = time.monotonic()
    remaining_total = max(0.0, deadline - now)
    lp_end = min(deadline, now + min(600.0, max(60.0, 0.25 * remaining_total)))

    lp_bases = []
    if best_seq is not None and 256 <= best_seq.size <= 1024:
        lp_bases.append(best_seq.copy())

    for n_target in (512, 1024):
        base = make_power_sequence(n_target, -0.5, shift=0.5)
        base = add_smooth_noise(base, rng, 0.05)
        base = project_simplex(base)
        lp_bases.append(base)

    added_top = 0
    for item in top:
        if added_top >= 3:
            break
        if 256 <= item[2] <= 1024:
            lp_bases.append(item[1].copy())
            added_top += 1

    for base in lp_bases:
        now = time.monotonic()
        if now >= lp_end:
            break
        if base.size < 256 or base.size > 1024:
            continue
        budget = min(200.0, lp_end - now)
        if budget < 30.0:
            continue
        sub_deadline = min(lp_end, now + budget)
        try:
            a2, sc2 = run_lp_optimize(base, sub_deadline, max_iter=500, rng=rng)
        except Exception:
            a2, sc2 = base, best_score
        if a2 is not None and sc2 < best_score and math.isfinite(sc2):
            best_score = sc2
            best_seq = a2.copy()
            print(f"LP improved fast_score={best_score:.10f}, len={best_seq.size}")

    # After LP, polish the current best if it is within a practical size.
    now = time.monotonic()
    if best_seq is not None and 128 <= best_seq.size <= 4096:
        sub_deadline = min(deadline, now + min(180.0, max(30.0, 0.10 * max(0.0, deadline - now))))
        if sub_deadline > now + 20.0:
            try:
                a2, sc2 = run_soft_optimize(
                    best_seq,
                    sub_deadline,
                    max_iter=100000,
                    tau_start=5.0,
                    tau_end=1000.0,
                    lr_start=0.06,
                )
            except Exception:
                a2, sc2 = best_seq, best_score
            if a2 is not None and sc2 < best_score and math.isfinite(sc2):
                best_score = sc2
                best_seq = a2.copy()
                print(f"post-LP soft improved fast_score={best_score:.10f}, len={best_seq.size}")

    # Stage 2: soft-max / Adam optimization on multiple bases.
    now = time.monotonic()
    remaining_total = max(0.0, deadline - now)
    soft_end = min(deadline, now + min(1200.0, max(120.0, 0.45 * remaining_total)))

    soft_bases = []
    if best_seq is not None and 128 <= best_seq.size <= 4096:
        soft_bases.append((best_seq.copy(), "best"))

    for n_target in (512, 1024, 2048, 4096):
        base = make_power_sequence(n_target, -0.5, shift=0.5)
        base = add_smooth_noise(base, rng, 0.03)
        base = project_simplex(base)
        soft_bases.append((base, f"power-{n_target}"))

    added_top_soft = 0
    for item in top:
        if added_top_soft >= 4:
            break
        if 256 <= item[2] <= 2048:
            soft_bases.append((item[1].copy(), f"top-{item[3]}"))
            added_top_soft += 1

    for base, desc in soft_bases:
        now = time.monotonic()
        if now >= soft_end:
            break
        if base.size < 128 or base.size > 4096:
            continue
        budget = min(420.0, soft_end - now)
        if budget < 60.0:
            continue
        sub_deadline = min(soft_end, now + budget)
        try:
            a2, sc2 = run_soft_optimize(
                base,
                sub_deadline,
                max_iter=300000,
                tau_start=2.0,
                tau_end=1000.0,
                lr_start=0.07,
            )
        except Exception:
            a2, sc2 = base, fast_score(base)
        if a2 is not None and sc2 < best_score and math.isfinite(sc2):
            best_score = sc2
            best_seq = a2.copy()
            print(f"soft improved via {desc}: fast_score={best_score:.10f}, len={best_seq.size}")

    # Stage 3: local stochastic perturbations on the best sequence.
    now = time.monotonic()
    perturb_end = deadline - 12.0
    if best_seq is not None and perturb_end > now + 30.0:
        try:
            a2, sc2 = perturb_search(
                best_seq,
                perturb_end,
                rng,
                max_iter=1000000,
                sigma0=0.18,
            )
        except Exception:
            a2, sc2 = best_seq, best_score
        if a2 is not None and sc2 < best_score and math.isfinite(sc2):
            best_score = sc2
            best_seq = a2.copy()
            print(f"perturb improved fast_score={best_score:.10f}, len={best_seq.size}")

    # Fallback.
    if best_seq is None:
        best_seq = make_power_sequence(512, -0.5, shift=0.5)
        best_score = fast_score(best_seq)

    # Final cleanup: non-negative, bounded, normalized.
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(np.sum(best_seq))
    if s <= 0.0 or not math.isfinite(s):
        best_seq = make_power_sequence(512, -0.5, shift=0.5)
        s = float(np.sum(best_seq))
    if s > 0.0 and math.isfinite(s):
        best_seq = best_seq / s

    print(f"final fast_score={fast_score(best_seq):.10f}, len={best_seq.size}")

    # If the official evaluator is available, sanity-check and log it.
    eval_func = globals().get("evaluate_sequence", None)
    if eval_func is not None:
        try:
            final_score = eval_func([float(x) for x in best_seq])
            print(f"final evaluate_sequence={final_score}")
        except Exception as exc:
            print(f"final evaluate_sequence failed: {exc}")

    return [float(x) for x in best_seq]
