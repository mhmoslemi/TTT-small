Minimize the autocorrelation peak of a non-negative step function using
exponential-tilt gradient descent with soft-max objective and adaptive
temperature. Multiple random initializations are tried.
"""

import numpy as np
import time


def compute_conv_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

    """Compute the soft-max gradient: g[m] = 2*sum_k w[k]*a[k-m] (valid)."""
    return 2.0 * np.correlate(w, a, mode='valid')

    rng = np.random.default_rng(seed)
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:

    else:
        a = a / s

    best_score = compute_conv_score(a)
    
    T = 1.0  # temperature
    T_min = 0.005
    T_decay = 0.9999
    eta_scale = 1.0

    last_improve_iter = 0

        c_max = c.max()

            a_new_sum = a_new.sum()
            if a_new_sum > 1e-300:
                a = a_new / a_new_sum

        if cur_score < best_score:

            best_a = a.copy()

        T = max(T * T_decay, T_min)

            a = a + kick
            a = np.maximum(a, 0)
            a = a / a.sum()
            last_improve_iter = iter_count

        if iter_count % 10000 == 0:

(1) smooth mean-square phase for fast initial convergence,
(2) soft-max phase for fine-tuning the peak.

import numpy as np

def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def optimize_phase1(a, deadline, seed):
    """Phase 1: minimize sum(c^2) using exponential tilt."""
