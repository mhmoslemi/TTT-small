Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using projected
smoothed-subgradient descent with multiple lengths, initializations,
and random restarts. Uses FFT convolution for O(n log n) iterations.

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    mask = u * idx > (cssv - 1)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Score assuming sum(a)=1: 2*n*max(a*a)."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def smooth_grad(a, b, tau):
    """Smoothed subgradient using softmax weights on b."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0

        w /= s
    # g_j = 2 * sum_k w_k * a_{k-j} = 2 * correlate(w, a, valid)
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using projected
smoothed-subgradient descent with momentum, multi-scale gradients,
multiple sequence lengths, diverse initializations, and local polish.
Uses FFT-based convolution for O(n log n) per iteration.
"""

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def smooth_grad_single(a, b, tau):
    """Smoothed subgradient using softmax weights at one temperature."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g


def multi_scale_grad(a, b, tau_base, n_scales):
    """Average subgradients at multiple temperature scales."""
    g_total = np.zeros(a.size)
    for s in range(n_scales):
        tau = tau_base * (2.0 ** s)
        g_total += smooth_grad_single(a, b, tau)
    return g_total / n_scales


def subgrad_optimize(a, time_budget, rng, use_momentum):
    """
    Run projected smoothed-subgradient descent with optional momentum.
    a is normalized to sum to 1. Returns (best_a, best_score).
    """
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    eta = 0.3
    tau_frac = 0.04
    n_scales = 3
    beta = 0.5  # momentum
    vel = np.zeros(n)

    iters = 0
    last_improve = 0
    stall = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))

        tau = tau_frac * float(np.max(b))
        if tau < 1e-18:
            tau = 1e-18

        if n_scales > 1:
            g = multi_scale_grad(a, b, tau, n_scales)
        else:
            g = smooth_grad_single(a, b, tau)

        if use_momentum:
            vel = beta * vel + (1.0 - beta) * g
            direction = vel
        else:
            direction = g

        a_new = project_simplex(a - eta * direction)

        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))

        # Accept if better or with small probability (simulated annealing)
        accept = (new_sc <= cur_sc) or (rng.random() < 0.03)
        if accept:
            a = a_new
        else:
            a = project_simplex(a - 0.3 * eta * direction)

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters
            stall = 0
        elif iters - last_improve > 2000:
            stall += 1

        iters += 1

        # Adaptive: reduce step if stalling
        if stall > 0 and stall % 3 == 0:
            eta *= 0.7

        # Periodic restart with perturbation
        if iters - last_improve > 8000:
            a = project_simplex(best_a + 0.05 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.2 + 0.3 * rng.random()
            tau_frac = 0.03 + 0.03 * rng.random()
            last_improve = iters
            stall = 0

        # Reheat
        if iters % 15000 == 0:
            eta = 0.2 + 0.4 * rng.random()
            tau_frac = 0.02 + 0.05 * rng.random()
            beta = 0.3 + 0.4 * rng.random()

    return best_a, best_sc


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_two_cluster(n, rng):

    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.02 * np.abs(rng.standard_normal(n))

def init_u_shape(n):
    """High at both edges, low in middle: a_i proportional to 1/sqrt(min(i, n-1-i)+1)."""

    i = np.arange(n, dtype=np.float64)

def polish(a, time_budget, rng):
    """Local polish via random perturbations with decreasing scale."""

    t0 = time.monotonic()
    scale = 0.02
    stuck = 0
    while time.monotonic() - t0 < time_budget and scale > 1e-6:
        improved = False
        for _ in range(10):
            cand = a + scale * rng.standard_normal(n)

            sc = normed_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True
        if not improved:
            stuck += 1
            if stuck > 5:
                break
        else:
            stuck = 0
        scale *= 0.9

    return best_a, best_sc
