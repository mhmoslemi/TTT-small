    size = 2 * n - 1

1. Start with `a - eta * g`.
2. Set negative components to 0.
3. Scale so the sum is 1.

This is not exactly the Euclidean projection (which requires a waterfilling algorithm), but it's a good approximation and much faster.

    a = np.maximum(a, 0.0)
    s = a.sum()

    return a / s


def compute_soft_grad(a, c, temp):
    """Compute gradient of soft-max(c) w.r.t. a using top significant lags."""
    n = len(a)
    cm = c.max()

    w = np.exp(logits)

        scored.append((score, i, s))
    scored.sort(key=lambda x: x[0])
    for score, idx, s in scored[:5]:
        print(f"  Start {idx}: score = {score:.6f}")
    best_score = scored[0][0]
    best_seq = scored[0][2].copy()

    if best_score < 1.51:
        print(f"Already at target! Score = {best_score:.6f}")

    # Phase 2: Optimize from best starting points
    print("\nPhase 2: Gradient optimization...")
    time_per_run = (budget_s - 60.0) / min(5, len(scored))
    # Optimize from top starts, allocating more time to better starts
    top_k = min(5, len(scored))
    time_alloc = [time_per_run * 1.5] + [time_per_run * 0.9] * (top_k - 1)

    for run_idx in range(top_k):
        if time.monotonic() >= deadline:
            break
        _, idx, start = scored[run_idx]
        run_deadline = time.monotonic() + time_alloc[run_idx]
        print(f"\n  Run {run_idx+1}: starting score = {scored[run_idx][0]:.6f}")

        a_opt, score_opt, iters = optimize_run(
            start, run_deadline,
            initial_temp=0.3 + 0.2 * run_idx,
            min_temp=0.003
        )
        print(f"  Run {run_idx+1}: final score = {score_opt:.6f} ({iters} iters)")

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # Phase 3: Fine-tuning from best with low temperature
    print(f"\nPhase 3: Fine-tuning from best (score = {best_score:.6f})...")
    fine_deadline = time.monotonic() + min(300, budget_s - 60.0 - (time.monotonic() - (time.monotonic() - budget_s + 30)))
    fine_deadline = min(fine_deadline, time.monotonic() + 200)

    a_fine = best_seq.copy()
    a_opt, score_fine, iters = optimize_run(
        a_fine, fine_deadline,
        initial_temp=0.02,
        min_temp=0.002
    )
    if score_fine < best_score:
        best_score = score_fine
        best_seq = a_opt.copy()
        print(f"  Fine-tune improved: {best_score:.6f}")

    # Phase 4: Random perturbations and short optimizations
    print(f"\nPhase 4: Random perturbations (best = {best_score:.6f})...")
    pert_deadline = time.monotonic() + 100.0
    for restart in range(20):
        if time.monotonic() >= pert_deadline:
            break
        a_pert = best_seq + rng.normal(0, 0.02, n)
        a_pert = project(a_pert)
        a_opt, score_pert, _ = optimize_run(a_pert, time.monotonic() + 5.0,
                                             initial_temp=0.05, min_temp=0.01)
        if score_pert < best_score:
            best_score = score_pert
            best_seq = a_opt.copy()
            print(f"  Restart {restart}: new best {best_score:.6f}")

    # Final: verify with exact evaluator
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"\nFinal verified score: {final_score:.6f}")
    print(f"  (fast estimate was: {best_score:.6f})")

    # If the exact score is worse (shouldn't happen), retry
    if final_score > best_score + 0.01:
        print("  Warning: exact score differs from estimate, re-checking...")
        # Try parent as fallback
        if parent is not None:
            parent_result = [float(max(0.0, min(1000.0, x))) for x in parent]
            parent_score = evaluate_sequence(parent_result)
            if parent_score < final_score:
                print(f"  Using parent: {parent_score:.6f}")
                return parent_result

    return result
