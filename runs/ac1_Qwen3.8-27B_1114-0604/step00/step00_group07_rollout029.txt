def _cross_corr(x, y):
    """Cross-correlation: result[i] = sum_j x[i+j] * y[j]"""

    size = len(x) + ny - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    X = np.fft.rfft(x, fft_size)
    Y = np.fft.rfft(y, fft_size)
    return np.fft.irfft(X * np.conj(Y), fft_size)[:ny]

    """Compute peaked surrogate of max(conv(a,a)) and its gradient."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = np.max(c)

    else:
        w /= w_sum

    a_new = np.exp(log_a)

    """Cross-correlation: result[i] = sum_j x[i+j] * y[j] (zero-padded)."""

    return np.fft.irfft(X * np.conj(Y), fft_size)[:ny]

def _compute_grad(a, power):
    """Compute gradient of peaked surrogate for max(conv(a,a))."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = np.max(c)
    if max_c <= 0:
        return c, max_c, np.zeros(n)
    # Peaked weights
    ratio = c / max_c
    w = ratio ** power
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(len(c)) / len(c)
    else:
        w /= w_sum
    grad = 2.0 * _cross_corr(w, a)
    return c, max_c, grad
