"""
Minimize  2*n*max(conv(a,a)) / (sum(a))^2  for a non-negative sequence a.

Algorithm
---------
1) Build a set of diverse initialisations (random, quadratic-phase chirp,
   raised-cosine, parent seed) at several lengths and keep the best by exact
   evaluation.
2) Run gradient descent on the smooth surrogate
       F(a) = (2n / beta) * log( sum_k exp( beta * (a*a)_k ) )   (sum a = n)
   whose minimiser approaches the true peak objective as beta grows.
   The gradient
       (grad a)_i = 4n * sum_j w[j+i] * a[j],   w = softmax(beta*(a*a))
   is evaluated in O(n log n) with an FFT cross-correlation (with a one-time
   analytic self-test against a direct O(n^2) reference so the FFT path is
   guaranteed correct).  An Adam scheduler with an annealing beta annealing
   beta and step size is used.
3) Periodic restarts with multiplicative local perturbations of the best
   sequence escape local minima.
4) A final multiplicative-restart polish pass searches around the incumbent.

All helper functions are top level and closure/lambda free.
"""

import numpy as np
from numpy.fft import fft, ifft
import time


# --------------------------------------------------------------------------- #
# Core numerics
# --------------------------------------------------------------------------- #
def _lin_conv(a):
    """Linear convolution a * a via FFT (real output, length 2n-1)."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    return ifft(A * A).real[:size]


def _direct_grad_core(a, w):
    """Reference cross-correlation: r[i] = sum_j w[i+j] * a[j]."""
    n = len(a)
    r = np.empty(n, dtype=np.float64)
    for i in range(n):
        r[i] = np.dot(w[i:i + n], a)
    return r


def _fft_grad_core(a, w):
    """FFT cross-correlation: r[i] = sum_j a[j] * w[i+j]  (i = 0..n-1)."""
    n = len(a)
    N = 1 << (n + len(w) - 2).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    r = ifft(np.conj(A) * W).real[:n]
    return r


def _grad(a, beta, use_fft):
    """Full gradient  (grad a)_i = 4n * sum_j w[j+i] a[j]."""
    n = len(a)
    c = _lin_conv(a)
    cmax = c.max()
    w = np.exp(beta * (c - cmax))
    w /= w.sum()
    core = _fft_grad_core(a, w) if use_fft else _direct_grad_core(a, w)
    return (4.0 * n) * core


def _eval(a):
    """Exact objective via the authoritative evaluator (inf if invalid)."""
    return float(evaluate_sequence([float(x) for x in a]))


# --------------------------------------------------------------------------- #
# Initialisation builders (all top-level)
# --------------------------------------------------------------------------- #
def _init_random(rng, n):
    return rng.uniform(0.1, 1.0, size=n)


def _init_chirp(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 + 0.5 * np.cos(2.0 * np.pi * t * t / (2.0 * n)) + 1e-6


def _init_raised_cos(n):
    t = np.arange(n, dtype=np.float64)
    base = 0.5 + 0.5 * np.cos(2.0 * np.pi * t / n)
    return base * base + 0.01


def _init_inverted_gauss(n):
    t = np.arange(n, dtype=np.float64)
    g = np.exp(-((t - n / 2.0) ** 2) / (2.0 * (n / 4.0) ** 2))
    return g.max() - g + 0.01


def _init_parent():
    if 'height_sequence_1' in globals():
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            return p
    return None


def _normalise(a, target_sum):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-12:
        return None
    a = a * (target_sum / s)
    a = np.clip(a, 0.0, 1000.0)
    return a


# --------------------------------------------------------------------------- #
# Optimiser over a single sequence length
# --------------------------------------------------------------------------- #
def _run_descent(a0, budget_end, rng, use_fft, verbose):
    """Adam soft-max descent on the simplex-sum-n from a0; returns (best, score)."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalise(a, float(n))
    if a is None:
        best, bscore = a0.copy(), _eval(a0)
        return best, bscore

    best = a.copy()
    bscore = _eval(best)

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.80, 0.999, 1e-8

    beta_lo, beta_hi = 5.0, 60.0
    lr0, lr1 = 2.5, 0.3
    no_improve = 0
    eval_every = max(4, n // 800)

    while time.monotonic() < budget_end:
        progress = min(1.0, max(0.0, (time.monotonic() - (budget_end - 120.0)) / 120.0))
        progress = min(1.0, progress + 0.0)  # keep bounded
        # overall schedule uses wall-clock fraction handled by caller via budget_end;
        # here we derive a local anneal from the remaining budget implicitly
        beta = beta_lo + (beta_hi - beta_lo) * (1.0 - np.exp(-3.0 * t_adam / 200.0))
        lr = lr0 * (lr1 / lr0) ** (t_adam / 300.0)

        g = _grad(a, beta, use_fft)

        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * (g * g)
        m_hat = m / (1.0 - b1 ** (t_adam + 1))
        v_hat = v / (1.0 - b2 ** (t_adam + 1))
        a = np.maximum(0.0, a - lr * m_hat / (np.sqrt(v_hat) + eps))
        a = _normalise(a, float(n))
        if a is None:
            a = best.copy()

        t_adam += 1

        if t_adam % eval_every == 0:
            score = _eval(a)
            if np.isfinite(score) and score < bscore:
                bscore = score
                best = a.copy()
                no_improve = 0
                if verbose:
                    print("    descent n=%d  score=%.6f  iter=%d" % (n, bscore, t_adam), flush=True)
            else:
                no_improve += 1

        if no_improve > 160 and time.monotonic() < budget_end - 30.0:
            # multiplicative local restart from the incumbent
            a = best.copy()
            mask = rng.random(n) < 0.25
            a[mask] *= np.exp(rng.normal(0.0, 1.0, int(mask.sum())))
            a = _normalise(a, float(n))
            if a is None:
                a = best.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            no_improve = 0

    return best, bscore


def _polish(a, budget_end, rng, use_fft, verbose):
    """Cheap multiplicative hill-climb around the incumbent."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalise(a, float(n))
    if a is None:
        return a, _eval(a)
    best = a.copy()
    bscore = _eval(best)
    widths = [0.05, 0.10, 0.20, 0.40]
    while time.monotonic() < budget_end:
        for w in widths:
            a2 = best.copy()
            mask = rng.random(n) < 0.2
            a2[mask] *= np.exp(rng.normal(0.0, w, int(mask.sum())))
            a2 = _normalise(a2, float(n))
            if a2 is None:
                continue
            score = _eval(a2)
            if np.isfinite(score) and score < bscore:
                bscore = score
                best = a2.copy()
                if verbose:
                    print("    polish score=%.6f" % bscore, flush=True)
    return best, bscore


# --------------------------------------------------------------------------- #
# Entry point
# --------------------------------------------------------------------------- #
def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for the best non-negative step-function height sequence."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    total = float(budget_s)
    poll_end = t0 + total - 60.0          # wall-budgeted polish vs descent
    polish_start = t0 + total - 60.0

    # --- validate FFT gradient path against the direct reference once -------
    use_fft = True
    try:
        _sa = rng.normal(0.0, 1.0, size=16)
        _sa = np.abs(_sa) + 0.3
        _sc = _lin_conv(_sa)
        _w = np.exp(8.0 * (_sc - _sc.max()))
        _w /= _w.sum()
        _ref = _direct_grad_core(_sa, _w)
        _ft = _fft_grad_core(_sa, _w)
        if not np.allclose(_ref, _ft, rtol=1e-8, atol=1e-10):
            use_fft = False
    except Exception:
        use_fft = False
    print("FFT gradient verified:", use_fft, flush=True)

    # --- Phase 1: seed pool --------------------------------------------------
    best = None
    bscore = np.inf

    def consider(a):
        nonlocal best, bscore
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01 or not np.isfinite(a).all():
            return
        s = _eval(a)
        if np.isfinite(s) and s < bscore:
            bscore = s
            best = a.copy()
            print("  seed n=%d score=%.6f t=%.1fs" % (a.size, s, time.monotonic() - t0), flush=True)

    seed_pool_end = t0 + 25.0
    p = _init_parent()
    if p is not None:
        consider(p)
    for n in (512, 1024, 2048):
        if time.monotonic() > seed_pool_end:
            break
        consider(_init_random(rng, n))
        consider(_init_chirp(n))
        consider(_init_raised_cos(n))
        consider(_init_inverted_gauss(n))

    if best is None:
        best = np.ones(1024)
        bscore = _eval(best)

    print("Seed best: %.6f (n=%d)" % (bscore, best.size), flush=True)

    # --- Phase 2: descent from the best seed, then from a couple of runners ---
    # run descent starting from the current best
    cand = best
    cs, sc = _run_descent(cand, poll_end, rng, use_fft, verbose=True)
    if np.isfinite(sc) and sc < bscore:
        bscore = sc
        best = cs.copy()

    # optionally refine from best parent-derived sequence of any other length
    if p is not None and p.size != best.size and time.monotonic() < poll_end - 2.0:
        cand2 = p / (p.sum() + 1e-30)
        cs2, sc2 = _run_descent(cand2, poll_end, rng, use_fft, verbose=True)
        if np.isfinite(sc2) and sc2 < bscore:
            bscore = sc2
            best = cs2.copy()

    # --- Phase 3: multiplicative polish ---
    if time.monotonic() < polish_start + 40.0:
        ps, pscore = _polish(best, t0 + total - 4.0, rng, use_fft, verbose=True)
        if np.isfinite(pscore) and pscore < bscore:
            bscore = pscore
            best = ps.copy()

    print("FINAL %.6f (n=%d) t=%.1fs" % (bscore, best.size, time.monotonic() - t0), flush=True)
    return [float(x) for x in best]
