import numpy as np
import time

def compute_score(a):
    """Compute 2n * max(a*a) / (sum(a))^2."""
    n = len(a)
    b = np.convolve(a, a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * np.max(b) / (s * s)

def multiplicative_update(a, eta=0.1, n_iters=10000, top_k=1):
    """Run multiplicative update to minimize peak of auto-convolution."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.clip(a, 1e-15, None)
    a = a / a.sum()
    
    best_a = a.copy()
    best_peak = np.max(np.convolve(a, a))
    
    for it in range(n_iters):
        b = np.convolve(a, a)
        peak = b.max()
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()
        
        # Find top-k peak positions
        if top_k == 1:
            k_star = int(np.argmax(b))
            idx = k_star - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            grad = np.zeros(n)
            grad[valid] = a[idx[valid]]
        else:
            top_k_indices = np.argsort(b)[-top_k:]
            grad = np.zeros(n)
            for k in top_k_indices:
                idx = k - np.arange(n)
                valid = (idx >= 0) & (idx < n)
                grad[valid] += a[idx[valid]]
        
        # Multiplicative update
        max_grad = grad.max()
        if max_grad > 0:
            a = a * np.exp(-eta * grad / max_grad)
        
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
    
    return best_a, best_peak
