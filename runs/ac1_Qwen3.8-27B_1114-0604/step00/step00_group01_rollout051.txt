def _project_simplex(v):
    n = v.size
    s = v.sum()
    if s <= 1e-12:
        return np.ones(n)/n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero((u*np.arange(1,n+1)) > (cssv - 1.0))[0]
    if rho.size == 0:
        return np.ones(n)/n
    r = rho[-1]
    theta = (cssv[r]-1.0)/(r+1.0)
    return np.maximum(v-theta, 0.0)

def _peak(a):
    b = np.convolve(a, a, 'full')
    return float(b.max())

def _smooth_grad(a, K, tau):
    n = a.size
    b = np.convolve(a, a, 'full')
    order = np.argsort(b)[::-1]
    topk = order[:K]
    mk = b[topk]
    num = np.exp(mk/tau - mk.max())
    w = num/num.sum()
    grad = np.zeros(n)
    for kk, wt in zip(topk, w):
        ist = max(0, kk - n + 1)
        ien = min(n - 1, kk)
        idx = np.arange(ist, ien + 1)
        grad[idx] += wt * (a[idx] + a[kk - idx])
    return grad, float(mk.max()), int(topk[0])

"""
Minimize 2*n*max(convolve(a,a,'full'))/(sum(a)^2) for non-negative 'a'.

The objective is scale-invariant, so we optimize under sum(a)=1, a>=0 and
maximize progress by minimizing the peak of the full self-convolution.

Approach (different from the previous block-perturbation local search):
  * Smoothed (log-sum-exp over the top few active lags) gradient descent on the
    peak of the self-convolution, projected onto the simplex {a>=0, sum=1}.
  * Symmetry-breaking seeds (the uniform/symmetric point is a fixed point with
    score exactly 2; asymmetry is required to go below 2).
  * Polyak running-average tracking, adaptive step size, and asymmetric restarts
    to escape plateaus.
  * Multiple sequence lengths and random seeds in independent timed rounds.
  * Optional LP polish (maximize sum(g) s.t. (a*g)_k<=M, g>=0, blend) on the
    best moderate-length iterate found, guarded by strict wall-clock checks.
"""
import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} (standard exact projection)."""
    n = int(v.size)
    s = float(v.sum())
    if s <= 1e-12:
        return np.full(n, 1.0 / n, dtype=np.float64)
    u = np.sort(v.astype(np.float64, copy=False))[::-1]
    cssv = np.cumsum(u)
    ar = np.arange(1, n + 1, dtype=np.float64)
    rho = np.nonzero(u * ar > (cssv - 1.0))[0]
    if rho.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    r = int(rho[-1])
    theta = (cssv[r] - 1.0) / float(r + 1)
    return np.maximum(v - theta, 0.0)


def _peak(a):
    """max of the full self-convolution of a (sum=1 => score = 2*n*peak)."""
    return float(np.convolve(a, a, 'full').max())


def _smooth_grad(a, K, tau):
    """Smoothed subgradient of max_k (a*a)_k using top-K active lags."""
    n = int(a.size)
    b = np.convolve(a, a, 'full')
    order = np.argsort(b)[::-1]
    topk = order[:K]
    mk = b[topk].astype(np.float64)
    num = np.exp(mk / tau - mk.max())
    w = num / num.sum()
    grad = np.zeros(n, dtype=np.float64)
    for kk, wt in zip(topk, w):
        ist = max(0, int(kk) - n + 1)
        ien = min(n - 1, int(kk))
        idx = np.arange(ist, ien + 1)
        grad[idx] += wt * (a[idx] + a[int(kk) - idx])
    return grad, float(mk.max()), int(topk[0])


def _make_seed(n, rng, kind):

    else:
        base = np.exp(rng.standard_normal(n)) * (0.5 + 0.5 * np.linspace(0.0, 1.0, n))
    base = np.maximum(base, 0.0)

def _lp_polish(a0, budget, rng, max_solves=60):
