```python
def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

- The gradient computation can also be done via FFT.

Now, the key question is: what n to use? The score is 2*n*max_conv (with sum=1 normalization). For larger n, we have more "freedom" to make the convolution flat, so the max_conv should be smaller. But the factor of 2*n grows linearly. So there's a trade-off.

For uniform: score = 2 regardless of n.
For optimized: we need 2*n*max_conv < 2, i.e., max_conv < 1/n.

The theoretical minimum of max_conv for non-negative p with sum 1 is... well, the average of conv is 1/(2n-1), so max_conv >= 1/(2n-1). If we could achieve max_conv ≈ 1/(2n-1), score ≈ 2n/(2n-1) ≈ 1.

But in practice, for non-negative sequences, it's harder to achieve this. The best known results for "minimum peak-to-mean autocorrelation" for non-negative sequences...

Actually, I recall that for the "minimum peak-to-average power ratio" problem, the optimal sequences are related to Zadoff-Chu or other low-PAPR sequences. But those typically allow complex-valued or ±1 sequences.

For non-negative real sequences, the problem is different. Let me just implement the gradient approach and see what we get.

```python
import numpy as np
import time

def _simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = v.size
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def _smooth_max_gradient(conv_vals, temperature):
    """Compute softmax weights for smooth max approximation."""
    # Shift for numerical stability
    shifted = conv_vals - np.max(conv_vals)
    exp_vals = np.exp(temperature * shifted)
    total = np.sum(exp_vals)
    if total == 0 or not np.isfinite(total):
        # Fallback: uniform weights on max elements
        w = np.zeros_like(conv_vals)
        max_val = np.max(conv_vals)
        w[conv_vals == max_val] = 1.0
        w /= w.sum()
        return w
    w = exp_vals / total
    return w

def _compute_gradient(w, p, n, use_fft=True):
    """Compute gradient of smooth max w.r.t. p."""
    if use_fft and n >= 64:
        # FFT-based computation
        N = 1
        while N < 3 * n - 2:
            N *= 2
        W = np.fft.fft(w, N)
        P_rev = np.fft.fft(p[::-1], N)
        full_conv = np.fft.ifft(W * P_rev).real[:n][::-1]
    else:
        full_conv = np.convolve(w, p[::-1], mode='full')[:n][::-1]
    return 2.0 * full_conv

def _optimize_sequence(p_init, n, rng, time_budget):
    """Optimize the sequence using projected gradient descent."""
    p = p_init.copy()
    p = p / p.sum()
    p = np.maximum(p, 0)
    p = p / p.sum()
    
    best_p = p.copy()
```

```python
import time

"""
Optimizes a non-negative sequence to minimize 2*n*max(autocorrelation)/(sum)^2.
Uses projected gradient descent with log-sum-exp smooth max approximation,
FFT-accelerated gradient computation, and multiple asymmetric starting points.
"""

def _simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x)=1} in O(n log n)."""
    n = v.size
    if n <= 1:
        return np.ones(max(n,1)) / max(n,1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def _next_pow2(m):
    """Next power of 2 >= m."""
    return 1 << (m - 1).bit_length()

def _compute_conv_and_gradient(p, N_fft):
    """Compute convolution and gradient efficiently using FFT."""
    n = p.size
    # Convolution
    F = np.fft.rfft(p, N_fft)
    conv = np.fft.irfft(F * F, N_fft)[:2*n-1]

    return conv
```

```python
def _simplex_project(v):
    """Project v onto the probability simplex {x>=0, sum(x)=1} in O(n log n)."""
    n = v.size
    if n <= 1:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n, dtype=np.float64) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def _next_pow2(m):
    """Smallest power of 2 >= m."""
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

    t_rem = deadline_global - time.monotonic()
    if t_rem > 30:
        n_best = len(best_seq)
        p_start = best_seq / best_seq.sum()
        print(f"\n--- Extended: n={n_best}, time_budget={t_rem:.0f}s ---")
        rng2 = np.random.default_rng(seed + 12345)
        p_opt = _optimize_p(p_start, n_best, t_rem * 0.8, rng2)
        score = evaluate_sequence(p_opt.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = p_opt.copy()
            print(f"  Extended: NEW BEST score={score:.6f}")
        else:
            print(f"  Extended: score={score:.6f} (no improvement)")

    # Final verification
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(result)
    if not np.isfinite(final_score):
        result = [1.0] * 256
        final_score = evaluate_sequence(result)

    print(f"\n{'='*50}")
    print(f"FINAL: score={final_score:.6f}, n={len(result)}")
    print(f"{'='*50}")
    return result
```
