import numpy as np
import time
from scipy.signal import fftconvolve

def project_to_simplex(v, s=1.0):
    """Project v onto simplex {x: sum(x)=s, x>=0}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - s)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_gradient(x, t):
    """Compute gradient of log-sum-exp of convolution."""
    n = len(x)
    conv = np.convolve(x, x)  # length 2n-1
    # Softmax
    shifted = t * conv - t * np.max(conv)  # numerical stability
    p = np.exp(shifted)
    p_sum = np.sum(p)
    p = p / p_sum
    # Gradient
    c1 = fftconvolve(p, x[::-1])[n-1:2*n-1]  # This gives sum_j p[j+i] * x[j]
    c2 = fftconvolve(p, x)[0:n]  # This gives (p*x)[i]
    return c1 + c2

import time
from scipy.signal import fftconvolve

def project_to_simplex(v, s=1.0):
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v

```python
"""
Minimize the peak-to-average ratio of the auto-convolution of a non-negative sequence.
Uses LSE-approximated gradient descent with Adam optimizer, simplex projection,
and periodic restarts to escape local minima.
"""

import numpy as np
import time
from scipy.signal import fftconvolve


def project_to_simplex(v, s=1.0):
    """Project vector v onto the probability simplex {x: sum(x)=s, x>=0}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - s) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Ensure exact sum
    w = w * (s / w.sum()) if w.sum() > 0 else np.full(n, s / n)
    return w


def lse_gradient(x, t):
    """Compute gradient of (1/t)log(sum(exp(t*conv(x,x)))) w.r.t. x."""
    n = len(x)
    conv = np.convolve(x, x)
    shifted = t * (conv - conv.max())
    p = np.exp(shifted)
    s = p.sum()
    if s < 1e-300:
        return np.zeros(n)
    p = p / s
    # c2[i] = sum_j p[j]*x[i-j] = (p*x)[i]
    conv_px = fftconvolve(p, x, 'full')
    c2 = conv_px[0:n]
    # c1[i] = sum_j p[i+j]*x[j]
    conv_prev_x = fftconvolve(p[::-1], x, 'full')
    c1 = conv_prev_x[n - 1:2 * n - 1][::-1]
    return c1 + c2


def compute_score(x):
    """Compute the evaluation score (same formula as evaluate_sequence)."""
    n = len(x)
    conv = np.convolve(x, x)
    max_b = conv.max()
    sum_a = x.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def adam_update(x, grad, m, v, step, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step with simplex projection."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** step)
    v_hat = v / (1 - beta2 ** step)
    x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    x_new = project_to_simplex(x_new, 1.0)
    return x_new, m, v


def run_lse_optimization(x, n, deadline, lr_init=0.05, t_init=20.0, t_max=200.0):
    """Run LSE gradient optimization. Returns best sequence and score."""
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    best = x.copy()
    best_score = compute_score(x)
    t = t_init
    
    last_print = time.monotonic()
    best_score_last = best_score
    
    while time.monotonic() < deadline:
        step += 1
        grad = lse_gradient(x, t)
        lr = lr_init / (1.0 + 0.0005 * step)
        x, m, v = adam_update(x, grad, m, v, step, lr)
        t = min(t_init + step * t_init / 5000.0, t_max)
        
        if step % 5000 == 0:
            s = compute_score(x)
            if s < best_score:
                best = x.copy()
                best_score = s
            now = time.monotonic()
            if now - last_print > 30:
                print(f"  Step {step}: score={best_score:.6f}, t={t:.1f}, lr={lr:.6f}")
                last_print = now
                best_score_last = best_score
    
    # Final check
    s = compute_score(x)
    if s < best_score:
        best = x.copy()
        best_score = s
    
    return best, best_score

            best_score = score2
