"""
Minimize 2*n*max(conv(a,a)) / (sum a)^2 for a non-negative sequence a.

The objective is scale-invariant, so we fix sum(a)=1 and minimize
max_k (a*a)_k.  Search = many independent fresh random initializers at
several lengths, each polished by (a) the convex-LP refinement from the
prior state of the art (max发生率 sum(g) s.t. x*g <= M, g>=0, blend) and
(b) sub-gradient / random-kick local moves.  Best sequence seen is kept.
"""

import time
import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import linprog

# ---------------------------------------------------------------------------
# module level shared state (no closures / no lambdas)
# ---------------------------------------------------------------------------
_BEST_SEQ = None
_BEST_SCORE = np.inf
_RNG = np.random.default_rng(12345)


def _autocorr(a):
    return fftconvolve(a, a, mode="full")


def _project(x):
    n = x.size
    x = np.maximum(x, 0.0)
    s = x.sum()
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / n)
    return x / s


def _int_score(a):
    aac = _autocorr(a)
    s = a.sum()
    if s <= 0:
        return np.inf
    return 2.0 * a.size * aac.max() / (s * s)


def _maybe_update_global(seq):
    global _BEST_SEQ, _BEST_SCORE
    val = _int_score(seq)
    if np.isfinite(val) and val < _BEST_SCORE:
        _BEST_SCORE = val
        _BEST_SEQ = seq.copy()
        print("  new best %.6f  n=%d" % (val, seq.size), flush=True)

# initializers (all return a project()ed length-n, sum-1 array)
# ---------------------------------------------------------------------------
def _init_uniform(n, s):
    return _project(np.ones(n))


def _init_cosine(n, s):
    i = np.arange(n)
    x = 0.5 * (1.0 + np.cos(np.pi * i / max(1, n - 1)))
    return _project(np.maximum(x, 1e-6))


def _init_random(n, s):
    rng = np.random.default_rng(s)
    return _project(rng.exponential(1.0, size=n))


def _init_triangular(n, s):
    i = np.arange(n)
    x = 1.0 - np.abs(i - (n - 1) / 2.0) / ((n - 1) / 2.0)
    return _project(np.maximum(x, 1e-6))


def _init_twosided_exp(n, s):
    rng = np.random.default_rng(s)
    c = rng.uniform(1.0, 6.0)
    i = np.arange(n)
    m = (n - 1) / 2.0
    d = np.abs(i - m) / max(1.0, m)
    x = np.exp(-c * d) + 0.3 * np.exp(-c * (1.0 - d))
    return _project(np.maximum(x, 1e-6))


def _init_noisy_smooth(n, s):
    rng = np.random.default_rng(s)
    k = max(2, n // rng.integers(4, 12) if n > 24 else 2)
    base = np.abs(rng.standard_normal(n))
    # low-pass via cumulative smoothing
    for _ in range(2):
        base = 0.5 * (base + np.roll(base, 1))
    x = base + 0.2 * np.abs(rng.standard_normal(n))
    return _project(np.maximum(x, 1e-6))


_INITS = [_init_uniform, _init_cosine, _init_random,
          _init_triangular, _init_twosided_exp, _init_noisy_smooth]


# ---------------------------------------------------------------------------
# LP refinement primitive (the SoA linear program)
# ---------------------------------------------------------------------------
def _lp_prob(x):
    n = x.size
    aac = _autocorr(x)
    M = aac.max()
    if not np.isfinite(M) or M <= 0.0:
        return None
    if n <= 1600:
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            j = np.arange(lo, hi + 1)
            idx = k - j
            A[k, j] = x[idx]
        cvec = -np.ones(n)
        bvec = np.full(2 * n - 1, M)
        res = linprog(cvec, A_ub=A, b_ub=bvec,
                      bounds=[(0.0, None)] * n, method="highs")
    else:
        return None
    if res is None or not res.success or res.x is None:
        return None
    g = np.maximum(res.x, 0.0)
    if g.sum() <= 1e-12:
        return None
    return g


def _subgrad_dir(x, fmax):
    n = x.size
    aac = _autocorr(x)
    active = np.where(aac >= fmax - max(1e-7 * fmax, 1e-14))[0]
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        grad[lo:hi + 1] += x[k - hi:k - lo + 1][::-1]
        lo2 = k
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] += x[0:hi2 - lo2 + 1]
    return grad


def _line_search_grad(x, fmax):
    d = _subgrad_dir(x, fmax)
    nrm = np.linalg.norm(d)
    if nrm <= 1e-18:
        return x, fmax
    d = d / nrm
    bestx, bestm = x, fmax
    lr = 0.05
    for _ in range(8):
        c = _project(x - lr * d)
        m = _autocorr(c).max()
        if m < bestm - 1e-12:
            bestx, bestm = c, m
        lr *= 0.5
    return bestx, bestm


def _refine(x, max_iter, deadline):
    cur = _project(x)
    Mc = _autocorr(cur).max()
    best = cur.copy()
    bestM = Mc
    _maybe_update_global(best)
    stall = 0
    t = 0.5
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        improved = False
        g = _lp_prob(cur)
        if g is not None:
            gs = g.sum()
            if gs > 1e-12:
                g = g / gs
                for tt in (t, 0.3, 0.2, 0.1, 0.05):
                    cand = _project((1.0 - tt) * cur + tt * g)
                    m = _autocorr(cand).max()
                    if m < Mc - 1e-11:
                        cur, Mc = cand, m
                        improved = True
                        break
        if improved:
            stall = 0
            if Mc < bestM:
                bestM = Mc
                best = cur.copy()
                _maybe_update_global(best)
            t = max(0.1, t * 0.85)
            continue
        # gradient polish
        for _ in range(3):
            cand, m = _line_search_grad(cur, Mc)
            if m < Mc - 1e-11 and time.monotonic() < deadline:
                cur = _project(cand)
                Mc = m
                if Mc < bestM:
                    bestM = Mc
                    best = cur.copy()
                    _maybe_update_global(best)
            else:
                break
        # random kick to escape shallow minima
        if stall >= 3 or it % 5 == 0:
            amp = 0.03 + 0.05 * min(1.0, stall / 8.0)
            kick = cur * np.exp(_RNG.normal(0.0, amp, size=cur.size))
            cur = _project(kick)
            Mc = _autocorr(cur).max()
        stall += 1
        if stall > 14:
            break
    return best, bestM

    if _BEST_SEQ is None:
        fallback = _project(np.ones(256))
        return [float(v) for v in fallback]
    return [float(v) for v in _BEST_SEQ]
