def fft_convolve(a, a2):
    """Full linear convolution using FFT."""
    n1 = len(a)
    n2 = len(a2)
    N = 1 << (n1 + n2 - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(a2, N), N)[:n1+n2-1]
