Minimize evaluate_sequence(a) = 2*n*max(conv(a,a))/sum(a)^2 for a non-negative
sequence a of fixed length. Strategy:
  * Smart initialization: two-level steps, tanh-smoothed steps, ramps, quadratics

      - smooth "tilt"/"curvature" multipliers,
      - block multiplicative perturbations,
      - pairwise mass transfers.
  * Periodic large-noise restarts from the best point to escape local minima.
Fast local scoring via FFT-based np.convolve mirrors evaluate_sequence exactly
(values are kept in [0,1000] with sum well above 0.01).
"""

import numpy as np

def _score(a):
    """Exact mirror of the evaluation objective (fast, vectorised)."""
    n = a.size
    conv = np.convolve(a, a, mode='full')
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _renorm(a, target):
    """Clip to non-negative and rescale so the sum equals target."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        return a
    return a * (target / s)


def _build_candidates(n, parent, rng):
    """Return a list of diverse starting vectors of length n."""
    out = []
    x = np.arange(n, dtype=np.float64)
    t = x / max(n - 1, 1)
    # two-level steps for several alpha
    half = n // 2
    for alpha in (0.3, 0.4, 0.5, 0.6, 0.7):
        c = np.empty(n)
        c[:half] = 1.0
        c[half:] = alpha
        out.append(c)
    # tanh-smoothed steps
    for beta in (0.5, 1.0, 2.0, 3.0, 5.0, 8.0, 12.0):
        out.append(0.5 * (1.0 + np.tanh(beta * (2.0 * t - 1.0))))
    # linear ramps
    out.append(0.2 + 0.8 * t)
    out.append(0.2 + 0.8 * (1.0 - t))
    # quadratic shapes
    out.append(0.2 + 0.8 * t ** 2)
    out.append(0.2 + 0.8 * (1.0 - t) ** 2)
    out.append(0.2 + 0.8 * np.sin(np.pi * t))
    # random smooth shape (low-pass random spectrum)
    spec = rng.normal(0, 1, n // 2 + 1)
    rnd = np.fft.irfft(np.abs(spec) ** 0.3, n=n)
    rnd = rnd - rnd.min()
    if rnd.max() > 0:
        rnd = rnd / rnd.max()
    out.append(0.1 + 0.9 * rnd)
    out.append(rng.uniform(0.1, 1.0, n))
    if parent is not None and parent.size == n:
        out.append(parent.copy())
    return out


def _grad_direction(best, n_scale):
    """Subgradient of the auto-convolution peak, projected to sum-zero."""
    conv = np.convolve(best, best, mode='full')
    peak = conv.max()
    active = np.nonzero(conv >= peak - 1e-9 * peak)[0]
    grad = np.zeros(best.size)
    for k in active:
        # d(conv[k])/da_i = a[k-i] + a[k-i] with the i=k-i term counted once
        ilo = max(0, k - (best.size - 1))
        ihi = min(best.size - 1, k)
        for i in range(ilo, ihi + 1):
            j = k - i
            grad[i] += best[j]
            if j != i:
                pass  # the sum above already adds both i contributions
    # The loop above adds best[j] once per i; the correct subgradient adds
    # best[k-i] for each contributing i, which equals counting a[j] for the
    # mirrored index. Rebuild explicitly for correctness:
    grad = np.zeros(best.size)
    for k in active:
        mgrad = np.zeros(best.size)
        ilo = max(0, k - (best.size - 1))
        ihi = min(best.size - 1, k)
        for i in range(ilo, ihi + 1):
            j = k - i
            mgrad[i] += best[j]
        grad += mgrad
    grad -= grad.mean()  # project onto sum-preserving directions
    return grad
