def compute_score_fast(a):
    """Compute the score directly using numpy for speed (avoids list conversion)."""
    n = len(a)
    if n == 0:
        return float('inf')
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (s * s))

        amp = rng.uniform(0.05, 0.4) * (1.0 + step * 0.005)
        phase = rng.uniform(0, 2.0 * np.pi)
        use_sin = rng.random() < 0.5
        
        perturbation = amp * (np.sin(2.0 * np.pi * harmonic * i / n + phase) if use_sin
                             else np.cos(2.0 * np.pi * harmonic * i / n + phase))
        
        candidate = np.clip(best + perturbation, 0.0, 1000.0)
        cand_score = compute_score_fast(candidate)
        
        if cand_score < best_score:
            best = candidate
            best_score = cand_score
    
    return best, best_score


def block_perturbation(base, rng):
    """Apply a random block perturbation to the sequence."""
    candidate = base.copy()
    n = len(candidate)
    width = int(rng.integers(max(2, n // 100), max(3, n // 10)))
    start = int(rng.integers(0, max(1, n - width)))
    stop = min(n, start + width)
    
    pert_type = rng.random()
    if pert_type < 0.3:
        # Multiplicative change
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.15))
    elif pert_type < 0.5:
        # Additive change
        candidate[start:stop] += rng.normal(0.0, 0.1 * np.mean(base))
    elif pert_type < 0.7:
        # Smooth sine perturbation on block
        idx = np.arange(stop - start)
        block_len = stop - start
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        candidate[start:stop] += 0.1 * np.sin(2 * np.pi * freq * idx / block_len + phase)
    else:
        # Blend with a smoothed version
        window = max(3, width // 5)
        kernel = np.ones(window) / window
        smoothed = np.convolve(candidate[start:stop], kernel, mode='same')
        candidate[start:stop] *= (1 - 0.3)
        candidate[start:stop] += 0.3 * smoothed
    
    return np.clip(candidate, 0.0, 1000.0)


def spectral_perturbation(base, rng, strength=0.1):
    """Perturb in the frequency domain for structured changes."""
    n = len(base)
    spectrum = np.fft.rfft(base).astype(complex)
    
    # Add random noise to a subset of frequency bins
    n_freq = len(spectrum)
    n_perturb = max(1, int(n_freq * 0.1))
    bins = rng.choice(n_freq, size=n_perturb, replace=False)
    
    for b in bins:
        magnitude = rng.uniform(-strength, strength) * np.abs(spectrum[b])
        angle = rng.uniform(0, 2 * np.pi)
        spectrum[b] += magnitude * np.exp(1j * angle)
    
    candidate = np.fft.irfft(spectrum, n=n)
    return np.clip(candidate, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize the convolution peak ratio using parametric initialization
    and multi-strategy local search with simulated annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    # Determine sequence length
    n = 4498
    if 'height_sequence_1' in dir() or 'height_sequence_1' in globals():
        try:
            n = len(globals()['height_sequence_1'])
        except (NameError, TypeError):
            n = 4498
    
    # Phase 1: Generate and evaluate parametric candidates
    print(f"Phase 1: Generating parametric candidates (n={n})...")
    candidates = generate_parametric_candidates(n)
    
    # Also try the provided sequence if available
    try:
        provided = np.array(globals()['height_sequence_1'], dtype=np.float64)
        if len(provided) == n and np.isfinite(provided).all():
            candidates.append(np.clip(provided, 0.0, 1000.0))
    except (NameError, TypeError, ValueError):
        pass
    
    best = None
    best_score = float('inf')
    
    for idx, cand in enumerate(candidates):
        score = compute_score_fast(cand)
        if score < best_score:
            best_score = score
            best = cand.copy()
    
    print(f"  Best parametric: score={best_score:.6f} (after {len(candidates)} candidates)")
    
    # Verify with the actual evaluator
    actual_best_score = float(evaluate_sequence(best.tolist()))
    print(f"  Verified best: score={actual_best_score:.6f}")
    
    # Phase 2: Local search with multiple strategies
    print("Phase 2: Local search...")
    current = best.copy()
    current_score = actual_best_score
    overall_best = best.copy()
    overall_best_score = actual_best_score
    
    temperature = 0.001
    no_improve_count = 0
    iteration = 0
    
    while time.monotonic() < deadline:
        iteration += 1
        
        # Choose perturbation strategy
        strategy = rng.random()
        if strategy < 0.35:
            # Block perturbation
            candidate = block_perturbation(current, rng)
        elif strategy < 0.6:
            # Spectral perturbation
            strength = 0.05 + 0.1 * (1.0 - iteration / max(1, iteration + 100))
            candidate = spectral_perturbation(current, rng, strength=strength)
        elif strategy < 0.8:
            # Single sine perturbation (targeted)
            harmonic = rng.integers(1, 15)
            amp = rng.uniform(0.02, 0.15)
            phase = rng.uniform(0, 2.0 * np.pi)
            i = np.arange(n, dtype=np.float64)
            if rng.random() < 0.5:
                pert = amp * np.sin(2.0 * np.pi * harmonic * i / n + phase)
            else:
                pert = amp * np.cos(2.0 * np.pi * harmonic * i / n + phase)
            candidate = np.clip(current + pert, 0.0, 1000.0)
        else:
            # Multi-harmonic perturbation
            candidate = current.copy()
            i = np.arange(n, dtype=np.float64)
            for _ in range(rng.integers(1, 4)):
                h = rng.integers(1, 20)
                a = rng.uniform(0.01, 0.08)
                p = rng.uniform(0, 2.0 * np.pi)
                candidate += a * np.sin(2.0 * np.pi * h * i / n + p)
            candidate = np.clip(candidate, 0.0, 1000.0)
        
        cand_score = compute_score_fast(candidate)
        
        # Simulated annealing acceptance
        if cand_score < current_score:
            current = candidate
            current_score = cand_score
        else:
            delta = cand_score - current_score
            if temperature > 1e-8 and rng.random() < math.exp(-delta / temperature):
                current = candidate
                current_score = cand_score
        
        # Cooldown
        temperature *= 0.9995
        
        # Track overall best
        if current_score < overall_best_score:
            overall_best = current.copy()
            overall_best_score = current_score
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Periodic reset to best if stuck
        if no_improve_count > 500:
            current = overall_best.copy() + rng.normal(0, 0.01, n)
            current = np.clip(current, 0.0, 1000.0)
            current_score = compute_score_fast(current)
            no_improve_count = 0
            temperature = max(temperature, 0.0005)
        
        # Progress reporting
        if iteration % 2000 == 0:
            elapsed = time.monotonic() - (deadline - float(budget_s) + 5.0)
            remaining = max(0, deadline - time.monotonic())
            print(f"  Iter {iteration}: best={overall_best_score:.6f}, curr={current_score:.6f}, "
                  f"T={temperature:.6f}, remaining={remaining:.0f}s")
    
    # Final verification with actual evaluator
    final_score = float(evaluate_sequence(overall_best.tolist()))
    print(f"Final result: score={final_score:.6f}")
    
    # Safety: verify with actual evaluator that we have a valid result
    if not np.isfinite(final_score) or final_score >= 2.0:
        # Fallback to constant sequence (guaranteed score = 2)
        fallback = np.ones(n, dtype=np.float64)
        fb_score = float(evaluate_sequence(fallback.tolist()))
        print(f"Fallback to constant: score={fb_score:.6f}")
        return [float(x) for x in fallback]
    
    return [float(x) for x in overall_best]
