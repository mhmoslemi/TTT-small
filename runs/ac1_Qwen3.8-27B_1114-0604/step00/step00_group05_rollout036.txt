import numpy as np
import time

"""
Minimizes 2n * max(f*f) / (sum(f))^2 for non-negative sequences f.
Uses projected gradient descent with log-sum-exp smoothing of the max,
FFT-based auto-convolution, and multiple initialization strategies.
Starting from a ramp (which gives ~1.44 for large n), the optimizer
refines the shape to reduce the peak auto-convolution further.
"""


def autoconv_fft(f):
    """Compute full auto-convolution of f using FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    result = np.fft.irfft(F * F, fft_size)
    return result[:size]


def gradient_smooth_max(f, c, t):
    """
    Compute gradient of (1/t)*log(sum_k exp(t*c_k)) w.r.t. f.
    c = auto-convolution of f (length 2n-1).
    Returns gradient vector of length n.
    """
    n = len(f)
    shifted = t * (c - c.max())
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()
    # g[i] = 2 * sum_k p[k] * f[k-i]
    # = 2 * (convolve(p, f_rev))[n-1+n]
    f_rev = f[::-1]
    conv = np.convolve(p, f_rev, mode='full')
    return 2.0 * conv[n - 1:n - 1 + n]


def optimize_from(f0, max_time=300.0, verbose=False):

    Projected gradient descent to minimize max(f*f) subject to f>=0, sum(f)=1.

    """

    iterations = 0

        current_score = 2.0 * n * np.max(c)

        if current_score < best_score:

            last_improve = time.monotonic()

            f_trial = np.maximum(f_trial, 0.0)

def lp_polish(f0, max_time=120.0, n_outer=30):
    """
    Polishing step using the LP-based approach:
    Given f with peak M, find g>=0 maximizing sum(g) s.t. (f*g)[k] <= M for all k.
    Then blend: f <- (1-t)*f + t*g.
    """
    from scipy.optimize import linprog
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0.copy()
    f = f / s

    best_f = f.copy()
    best_score = 2.0 * n * np.max(autoconv_fft(f))
    start_time = time.monotonic()

    for outer in range(n_outer):
        if time.monotonic() - start_time > max_time:
            break

        c = autoconv_fft(f)
        M = np.max(c)
        current_score = 2.0 * n * M

        # Build LP: max sum(g) s.t. A@g <= M, g >= 0
        # A[k, i] = f[i] if 0 <= k-i < n, else 0  (for k=0..2n-2)
        # This is a banded matrix, build it as sparse
        from scipy.sparse import csr_matrix
        rows = []
        cols = []
        vals = []
        for i in range(n):
            if f[i] < 1e-15:
                continue
            # f[i] contributes to (f*g)[k] for k = i, i+1, ..., i+n-1
            for k in range(i, i + n):
                rows.append(k)
                cols.append(i)
                vals.append(f[i])

        A_sparse = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        # Solve: min -sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A_sparse, b_ub=M * np.ones(2 * n - 1),
                      bounds=[(0, None)] * n, method='highs')

        if res.status != 0:
            break

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break

        g = g / g_sum  # Normalize g to sum 1

        # Blend
        alpha = 0.3
        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new

        c_new = autoconv_fft(f_new)
        new_score = 2.0 * n * np.max(c_new)

        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
        f = f_new

    return best_f
