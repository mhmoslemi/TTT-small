import time

def compute_convolution_fft(a, N):
    """Compute full linear convolution of a with itself using FFT."""
    A = np.fft.fft(a, N)
    c = np.fft.ifft(A * A).real
    return c[:2*len(a)-1]

def compute_gradient_surrogate(a, N, alpha):
    """Compute gradient of sum(exp(alpha * c[j])) w.r.t. a."""
    c = compute_convolution_fft(a, N)
    max_c = np.max(c)
    w = np.exp(alpha * (c - max_c))
    # g[i] = 2 * alpha * sum_j w[j] * a[j-i]
    # This is 2*alpha * cross_correlation(w, a)[i]
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    g = np.fft.ifft(W * np.conj(A)).real[:len(a)]
    return g * 2 * alpha
