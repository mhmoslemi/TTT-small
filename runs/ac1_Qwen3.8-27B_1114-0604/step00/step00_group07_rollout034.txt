def smoothed_gradient_step(a, epsilon, learning_rate):
    """One step of smoothed projected gradient descent."""
    n = len(a)
    N = 4 * n  # FFT size
    
    # Compute convolution
    f_a = np.fft.rfft(a, N)
    c = np.fft.irfft(f_a * f_a, N)[:2*n-1]
    
    # Compute softmax weights
    c_max = c.max()
    exp_c = np.exp((c - c_max) / epsilon)
    p = exp_c / exp_c.sum()
    
    # Compute gradient
    f_p = np.fft.rfft(p, N)
    R = np.fft.irfft(np.conj(f_a) * f_p, N)[:n]
    grad = 2.0 * R
    
    # Gradient step
    a_new = a - learning_rate * grad
    
    # Project onto {x >= 0, sum(x) = sum(a)}
    a_new = water_filling_project(a_new, a.sum())
    
    return a_new, c.max()

def water_filling_project(x, target_sum):
    """Project x onto {y >= 0, sum(y) = target_sum}."""
    # Find t such that sum(max(x_i - t, 0)) = target_sum
    # Use bisection
    x_sorted = np.sort(x)[::-1]  # descending
    
    # At the optimum, there exists a threshold t such that:
    # y_i = max(x_i - t, 0)
    # sum(y) = sum_{x_i > t} (x_i - t) = target_sum
    
    # Let k be the number of elements > t.
    # sum_{i=1}^{k} x_sorted[i] - k*t = target_sum
    # t = (sum_{i=1}^{k} x_sorted[i] - target_sum) / k
    
    # k is such that x_sorted[k-1] > t >= x_sorted[k]
    
    # Binary search for k
    n = len(x)
    cumulative = np.cumsum(x_sorted)
    
    # For each possible k, t_k = (cumulative[k-1] - target_sum) / k
    # We need x_sorted[k-1] > t_k >= x_sorted[k] (with x_sorted[n] = -inf)
    
    k = np.searchsorted(-cumulative, -target_sum, side='right')

    # We want the largest k such that t_k <= x_sorted[k-1]
    # i.e., (cumulative[k-1] - target_sum) / k <= x_sorted[k-1]
    # i.e., cumulative[k-1] - target_sum <= k * x_sorted[k-1]
    # i.e., cumulative[k-1] - k * x_sorted[k-1] <= target_sum

    lo = x.min() - 1

Minimize the evaluation function 2*n*max(conv(a,a))/(sum(a))^2 using
smoothed projected gradient descent with FFT-based computations.
Uses log-sum-exp smoothing, water-filling projection, and epsilon annealing.
"""

import numpy as np
import time


def fft_convolve(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N *= 2
    f = np.fft.rfft(a, N)
    return np.fft.irfft(f * f, N)[:2*n-1]


def fft_gradient(a, p, N=None):
    """
    Compute gradient of f_ε(a) = ε*logsumexp(conv(a,a)/ε).
    grad[j] = 2 * sum_k p[k] * a[k-j] (for valid indices)
            = 2 * cross_correlation(a, p)[j]
    """
    n = len(a)
    if N is None:
        N = 1
        while N < 3*n:
            N *= 2
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    corr = np.fft.irfft(np.conj(fa) * fp, N)[:n]
    return 2.0 * corr


def water_filling_project(x, target_sum):
    """Project x onto {y >= 0, sum(y) = target_sum}."""
    # Bisection on threshold t
    # y = max(x - t, 0), find t such that sum(y) = target_sum
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(x - mid, 0.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    t = (lo + hi) / 2.0
    result = np.maximum(x - t, 0.0)
    # Renormalize to fix numerical errors
    if result.sum() > 0:
        result = result * target_sum / result.sum()
    return result


def compute_score_normalized(a):
    """Compute score assuming sum(a) = 1. Score = 2*n*max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a)
    return 2.0 * n * c.max()

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using smoothed projected gradient
descent with FFT-based computations, water-filling projection, and adaptive
epsilon annealing. Tries multiple diverse initializations.
"""

import numpy as np
import time


def _fft_size(n):
    """Next power of 2 >= n."""
    N = 1
    while N < n:
        N *= 2
    return N


def _convolve_fft(a, N=None):
    """Full auto-convolution via FFT."""
    n = len(a)
    if N is None:
        N = _fft_size(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def _gradient_smoothed(a, c, epsilon, N):
    """
    Gradient of epsilon * logsumexp(c/epsilon) w.r.t. a.
    c = conv(a, a), length 2n-1.
    Returns gradient of length n.
    """
    n = len(a)
    # Softmax weights over c
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / epsilon)
    p = exp_c / exp_c.sum()
    
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] (valid indices)
    # = 2 * cross_correlation(a, p)[j]
    # Using FFT: ifft(conj(fft(a)) * fft(p))
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    corr = np.fft.irfft(np.conj(fa) * fp, N)[:n]
    return 2.0 * corr


def _water_filling(x, target):
    """Project x onto {y >= 0, sum(y) = target} via water-filling."""
    if target <= 0:
        return np.zeros_like(x)
    # Bisection for threshold
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.maximum(x - mid, 0.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    y = np.maximum(x - t, 0.0)
    s = y.sum()
    if s > 0:
        y *= target / s
    return y


def _score_normalized(a):
    """Score = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _convolve_fft(a)
    return 2.0 * n * c.max()


def _optimize(a_init, deadline, rng, verbose=False):
    """
    Smoothed projected gradient descent to minimize score.
    a_init: 1D array, will be normalized to sum=1.
    Returns (best_a, best_score_normalized).
    """
    a = np.clip(a_init.copy(), 1e-15, None)
    a /= a.sum()
    
    if not np.isfinite(a).all() or a.sum() <= 0:
        a = np.ones_like(a_init)
        a /= a.sum()
    
    n = len(a)
    N = _fft_size(3 * n)  # FFT size for gradient computation
    
    # Initial score
    best = a.copy()
    best_score = _score_normalized(a)
    
    # Hyperparameters
    epsilon = 3.0 / n
    min_epsilon = 0.02 / n
    lr = 1.0
    min_lr = 0.01
    max_lr = 10.0
    eps_decay = 0.98
    no_improve = 0
    stagnation_limit = 2000
    
    iteration = 0
    while time.monotonic() < deadline:
        # Compute convolution
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        max_c = c.max()
        
        # Smoothed max and its gradient
        exp_c = np.exp((c - max_c) / epsilon)
        p = exp_c / exp_c.sum()
        fp = np.fft.rfft(p, N)
        corr = np.fft.irfft(np.conj(fa) * fp, N)[:n]
        grad = 2.0 * corr
        
        # Gradient step + projection
        a_new = a - lr * grad
        a_new = _water_filling(a_new, 1.0)
        
        # Evaluate
        fa_new = np.fft.rfft(a_new, N)
        c_new = np.fft.irfft(fa_new * fa_new, N)[:2 * n - 1]
        new_score = 2.0 * n * c_new.max()
        
        if new_score < best_score:
            best = a_new.copy()
            best_score = new_score
            no_improve = 0
            # Increase lr slightly if making progress
            lr = min(lr * 1.02, max_lr)
        else:
            no_improve += 1
            # Decrease lr
            lr = max(lr * 0.95, min_lr)
        
        a = a_new
        
        # Decay epsilon
        if iteration % 50 == 0:
            epsilon = max(min_epsilon, epsilon * eps_decay)
        
        # Random perturbation to escape local minima
        if no_improve > stagnation_limit and iteration % 100 == 0:
            a *= np.exp(rng.normal(0, 0.2, n))
            a = np.clip(a, 1e-15, None)
            a /= a.sum()
            no_improve = 0
            lr = 1.0
        
        iteration += 1
        if verbose and iteration % 500 == 0:
            print(f"    iter={iteration}, score={best_score:.10f}, eps={epsilon:.6f}, lr={lr:.4f}")
    
    return best, best_score
