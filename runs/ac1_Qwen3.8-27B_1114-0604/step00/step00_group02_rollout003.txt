import time

def compute_score_fast(a):
    """Fast computation of the score (assuming sum > 0.01)."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a ** 2)

def normalize(a, target_sum=1.0):
    """Normalize a to have the given sum, maintaining non-negativity."""
    s = np.sum(a)
    if s <= 0:
        return np.ones(len(a)) / len(a)
    return a / s * target_sum

def subgradient_step(a, lr, top_k=1):
    """One subgradient descent step to minimize max of full convolution."""
    n = len(a)
    c = np.convolve(a, a)
    # Find top_k largest entries
    if top_k >= len(c):
        active = np.arange(len(c))
    else:
        top_indices = np.argsort(c)[-top_k:]
        # Include all that are within 1% of max
        threshold = c[top_indices[0]] * 0.99
        active = np.where(c >= threshold)[0]

    g = np.zeros(n)

        j_end = min(n - 1, k)

    # Wait, this indexing is tricky. Let me redo.
    # For a given k, dc_k/da[j] = 2*a[k-j] if 0 <= k-j <= n-1
    # So for j in [max(0, k-n+1), min(n-1, k)]: g[j] += 2*a[k-j]
    # a[k-j] for j in [j_start, j_end] is a[k-j_end : k-j_start+1] (reversed)
    
    # Actually: as j goes from j_start to j_end, k-j goes from k-j_start to k-j_end (decreasing)
    # So a[k-j] is a[k-j_start], a[k-j_start-1], ..., a[k-j_end]
    # Which is a[k-j_start : k-j_end-1 : -1] or a[k-j_start : k-j_end : -1]
    
    # Let me just use a direct approach:
    # Create the reversed-and-flipped version
    for k in active:
        valid_j = np.arange(max(0, k-n+1), min(n, k+1))
        valid_k_minus_j = k - valid_j
        g[valid_j] += 2.0 * a[valid_k_minus_j]
