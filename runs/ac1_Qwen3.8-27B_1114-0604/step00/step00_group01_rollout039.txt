```python
"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using smooth gradient descent
with FFT acceleration, exponential parameterization, and peak-directed search.
"""

import numpy as np
import time

def compute_convolution(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    N = 2 * n - 1
    # Next power of 2 for FFT efficiency
    N_fft = 1 << (N - 1).bit_length()
    fa = np.fft.rfft(a, N_fft)
    conv = np.fft.irfft(fa * fa, N_fft)[:N]
    return conv

def smooth_max(c, tau):
    """Compute smooth approximation of max using log-sum-exp."""
    c_shifted = c - c.max()  # For numerical stability
    return c.max() + tau * np.log(np.sum(np.exp(c_shifted / tau)))

def compute_gradient(a, tau):
    """Compute gradient of smooth objective w.r.t. a."""
    n = len(a)
    S = np.sum(a)
    c = compute_convolution(a)
    M = smooth_max(c, tau)
    
    # Weights for log-sum-exp
    c_shifted = c - c.max()
    w = np.exp(c_shifted / tau)
    w /= w.sum()
    
    # corr[j] = sum_{m=0}^{n-1} w[j+m] * a[m]
    # This is a cross-correlation, computable via FFT
    N = 2 * n - 1
    N_fft = 1 << (N - 1).bit_length()
    
    # h[j] = sum_{m=0}^{n-1} w[j+m] * a[m]
    # = full_conv(w, a_rev)[n-1+j] where a_rev = a[::-1]
    a_rev = a[::-1]
    N_fft2 = 1 << (N - 1).bit_length()
    fw = np.fft.rfft(w, N_fft2)
    fa_rev = np.fft.rfft(a_rev, N_fft2)
    full_conv = np.fft.irfft(fw * fa_rev, N_fft2)[:N]
    corr = full_conv[n-1:n-1+n]  # h[0], h[1], ..., h[n-1]
    
    # Gradient of smooth objective
    # f = 2n * M / S^2
    # df/da[j] = (2n/S^2) * (2*corr[j] - 2*M/S)
    grad = (2.0 * n / (S * S)) * (2.0 * (corr - M / S))
    
    return grad, M, S, c, w

def gradient_descent_optimize(a_init, n, tau, lr_init, deadline, max_iter=100000):
    """Run gradient descent with exponential parameterization."""
    # Parameterize a = exp(b) to ensure positivity
    b = np.log(a_init + 1e-10)
    best_a = a_init.copy()
    best_score = np.inf
    lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a = np.exp(b)
        # Renormalize to keep sum reasonable (optional)
        # a = a / a.sum() * n  # Keep mean = 1
        
        grad, M, S, c, w = compute_gradient(a, tau)
        
        # Gradient w.r.t. b: da/db[j] = a[j], so db grad = a[j] * grad[j]
        grad_b = a * grad
        
        # Adam-like update (simple momentum)
        b -= lr * grad_b
        
        # Clip b to keep a in reasonable range [1e-6, 1000]
        b = np.clip(b, np.log(1e-6), np.log(1000))
        
        a = np.exp(b)
        
        # Periodic evaluation
        if it % 100 == 0 or it == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        # Decay learning rate
        lr = lr_init * 0.999 ** (it / 1000)
        
    # Final evaluation
    score = evaluate_sequence(a.tolist())
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

def peak_directed_mutate(a, rng, n_mutations=10):
    """Mutate sequence by targeting the peak of its convolution."""
    a = a.copy()
    n = len(a)
    c = compute_convolution(a)
    
    for _ in range(n_mutations):
        # Find top peaks
        top_k = np.argsort(c)[-5:]
        
        for k_star in top_k:
            # c[k_star] = sum_m a[m] * a[k_star - m]
            # Valid m: max(0, k_star - (n-1)) <= m <= min(n-1, k_star)
            m_min = max(0, k_star - (n - 1))
            m_max = min(n - 1, k_star)
            
            if m_min > m_max:
                continue
            
            # Get contributing pairs
            pairs = [(m, k_star - m) for m in range(m_min, m_max + 1)]
            # Sort by contribution
            contributions = [(a[m] * a[k_star - m], m, k_star - m) for m in range(m_min, m_max + 1)]
            contributions.sort(reverse=True)
            
            # Reduce top contributors
            for contrib, m1, m2 in contributions[:3]:
                if m1 != m2 and a[m1] > 0.01 and a[m2] > 0.01:
                    reduce = 0.1 * min(a[m1], a[m2])
                    a[m1] -= reduce
                    a[m2] -= reduce
                    # Add to random other positions
                    idx1 = rng.integers(0, n)
                    idx2 = rng.integers(0, n)
                    a[idx1] += reduce * 0.5
                    a[idx2] += reduce * 0.5
                elif m1 == m2 and a[m1] > 0.01:
                    reduce = 0.2 * a[m1]
                    a[m1] -= reduce
                    idx = rng.integers(0, n)
                    a[idx] += reduce
            
    a = np.maximum(a, 0)
    return a

def local_search(a, deadline, rng, max_iters=50000):
    """Local search with various perturbation strategies."""
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    n = len(best)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        candidate = best.copy()
        strategy = rng.integers(0, 5)
        
        if strategy == 0:
            # Block perturbation
            width = rng.integers(max(1, n // 100), max(2, n // 10))
            start = rng.integers(0, n - width)
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.3))
        elif strategy == 1:
            # Gaussian bump addition/removal
            center = rng.integers(0, n)
            sigma = rng.integers(max(1, n//100), max(2, n//20))
            x = np.arange(n)
            bump = rng.normal(0, 0.5) * np.exp(-(x-center)**2 / (2*sigma**2))
            candidate += bump
        elif strategy == 2:
            # Peak-directed mutation
            candidate = peak_directed_mutate(best, rng)
        elif strategy == 3:
            # Swap two blocks
            w = rng.integers(max(1, n//50), max(2, n//10))
            i = rng.integers(0, n-w)
            j = rng.integers(0, n-w)
            candidate[i:i+w], candidate[j:j+w] = candidate[j:j+w].copy(), candidate[i:i+w].copy()
        elif strategy == 4:
            # Sinusoidal modulation
            f = rng.integers(1, n//4 + 1)
            phi = rng.uniform(0, 2*np.pi)
            mod = 1 + 0.3 * np.cos(2*np.pi*f*np.arange(n)/n + phi)
            candidate *= mod
        
        candidate = np.maximum(candidate, 0)
        
        score = evaluate_sequence(candidate.tolist())
        if score < best_score:
            best = candidate
            best_score = score
            # Adaptive: reduce perturbation as we improve
            # (implicitly, by keeping best as the new base)
    
    return best, best_score
```

```python
"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using smooth gradient descent
with FFT acceleration, exponential parameterization, and peak-directed search.

Key insight: The objective is scale-invariant. For non-palindromic sequences,
the center convolution value can be reduced below the Cauchy-Schwarz bound,
potentially achieving scores below 2. We use a log-sum-exp smoothed objective
with FFT-based gradient computation for efficiency.
"""

import numpy as np
import time


def compute_convolution_fft(a):
    """Compute linear convolution of a with itself using FFT (O(n log n))."""
    n = len(a)
    N = 2 * n - 1
    N_fft = 1 << (N - 1).bit_length()
    fa = np.fft.rfft(a, N_fft)
    conv = np.fft.irfft(fa * fa, N_fft)[:N]
    return conv


def compute_gradient_and_info(a, tau):
    """
    Compute gradient of smooth objective f = 2n * soft_max(conv(a,a), tau) / S^2.
    Returns (gradient, M_smooth, S, convolution, weights).
    """
    n = len(a)
    S = np.sum(a)
    c = compute_convolution_fft(a)
    
    # Log-sum-exp for numerical stability
    c_max = c.max()
    c_shifted = (c - c_max) / tau
    log_sum = c_max / tau + np.log(np.sum(np.exp(c_shifted)))
    M_smooth = tau * log_sum
    
    # Softmax weights
    w = np.exp(c_shifted)
    w /= w.sum()
    
    # Compute corr[j] = sum_{m=0}^{n-1} w[j+m] * a[m] via FFT
    N = 2 * n - 1
    a_flipped = a[::-1]
    N_full = (2 * n - 1) + n - 1  # full conv length = 3n-2
    N_fft2 = 1 << (N_full - 1).bit_length()
    fw = np.fft.rfft(w, N_fft2)
    faf = np.fft.rfft(a_flipped, N_fft2)
    full_conv = np.fft.irfft(fw * faf, N_fft2)
    corr = full_conv[n - 1:n - 1 + n]  # h[0],...,h[n-1]
    
    # Gradient: df/da[j] = (4n/S^2) * (corr[j] - M_smooth/S)
    grad = (4.0 * n / (S * S)) * (corr - M_smooth / S)
    
    return grad, M_smooth, S, c, w

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    tau = tau_init
    
    no_improve_count = 0
    best_local_for_anneal = best_local
    
    for it in range(max_iter):
        if time.monotonic() > deadline:

        t += 1
        grad, M_s, S, c, w = compute_gradient_and_info(a, tau)

        v_buf = beta2 * v_buf + (1 - beta2) * grad * grad
        m_hat = m_buf / (1 - beta1 ** t)

                    candidate[m2] -= red
                    total_reduce += red
            
            # Distribute to random positions not contributing to peak

                idx = rng.integers(0, n)
                if idx < m_min or idx > m_max:

            candidate[top_idx] *= 0.7

    if parent.ndim != 1 or len(parent) == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(2000, dtype=np.float64)
    
    parent = np.clip(parent, 0, 1000)
    n_parent = len(parent)
    if parent.sum() < 0.01:
        parent = np.ones(n_parent)
    
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial: score={best_score:.6f}, n={n_parent}")
    
    # === Phase 1: Gradient descent from structured non-palindromic initializations ===
    # Choose lengths to try
    lengths_to_try = [n_parent]
    if n_parent != 1000:
        lengths_to_try.append(1000)
    if n_parent != 3000:
        lengths_to_try.append(3000)
    
    total_time = deadline - time.monotonic()
    
    for n_try in lengths_to_try:
        if time.monotonic() > deadline:
            break
        
        time_budget = (deadline - time.monotonic()) * 0.6  # 60% for gradient descent
        per_init = time_budget / 8  # 8 different initializations
        
        init_list = []
        
        # Non-palindromic initializations
        # 1. Ramp (asymmetric)
        a_ramp = np.arange(1, n_try + 1, dtype=np.float64)
        init_list.append(("ramp", a_ramp.copy()))
        
        # 2. Asymmetric sine
        f1 = max(1, n_try // 100)
        a_sin1 = 1.0 + 0.7 * np.sin(2 * np.pi * f1 * np.arange(n_try) / n_try + 0.5)
        a_sin1 = np.maximum(a_sin1, 0.05)
        init_list.append(("asym_sine1", a_sin1.copy()))
        
        # 3. Different frequency
        f2 = max(1, n_try // 37)
        a_sin2 = 1.0 + 0.8 * np.sin(2 * np.pi * f2 * np.arange(n_try) / n_try + 1.2)
        a_sin2 = np.maximum(a_sin2, 0.05)
        init_list.append(("asym_sine2", a_sin2.copy()))
        
        # 4. Random
        a_rand = np.abs(rng.standard_normal(n_try)) + 0.1
        init_list.append(("random", a_rand.copy()))
        
        # 5. Two asymmetric bumps
        a_bump = np.zeros(n_try)
        c1 = n_try // 4
        c2 = 3 * n_try // 5
        sigma1 = n_try // 20
        sigma2 = n_try // 15
        x = np.arange(n_try)
        a_bump += 1.0 * np.exp(-(x - c1) ** 2 / (2 * sigma1 ** 2))
        a_bump += 0.7 * np.exp(-(x - c2) ** 2 / (2 * sigma2 ** 2))
        a_bump += 0.05
        init_list.append

("two_bumps", a_bump.copy()))
        
        # 6. Exponential decay (very asymmetric)
        a_exp = np.exp(-np.arange(n_try) / (n_try / 3.0))
        a_exp += 0.05
        init_list.append(("exp_decay", a_exp.copy()))
        
        # 7. Parent (if same length)
        if n_try == n_parent:
            init_list.append(("parent", parent.copy()))
        
        # 8. Quadratic (parabola shifted)
        x_norm = np.arange(n_try) / n_try
        a_quad = 1.0 + 1.5 * (x_norm - 0.3) ** 2
        a_quad = np.maximum(a_quad, 0.05)
        init_list.append(("quadratic", a_quad.copy()))
        
        for init_name, a_init in init_list:
            if time.monotonic() > deadline:
                break
            
            init_deadline = time.monotonic() + min(per_init, deadline - time.monotonic())
            if init_deadline <= time.monotonic():
                break
            
            tau = n_try / 10.0
            print(f"  GD [{init_name}] n={n_try} tau={tau:.1f}")
            
            a_opt, score = gradient_descent_optimize(a_init, n_try, tau, init_deadline)
            
            if score < best_score:
                best = a_opt
                best_score = score
                print(f"    New best: {best_score:.6f}")
            else:
                print(f"    Score: {score:.6f}")
    
    # === Phase 2: Peak-directed refinement ===
    print(f"\nBefore peak refine: {best_score:.6f}")
    if time.monotonic() < deadline - 30:
        best, best_score = peak_directed_refine(
            best, min(time.monotonic() + (deadline - time.monotonic()) * 0.25, deadline - 30),
            rng, max_evals=5000
        )
        print(f"After peak refine: {best_score:.6f}")
    
    # === Phase 3: Block local search ===
    if time.monotonic() < deadline - 10:
        best, best_score = block_local_search(
            best, deadline, rng, max_evals=20000
        )
        print(f"After block search: {best_score:.6f}")
    
    # === Phase 4: Final fine-tuning with smaller perturbations ===
    if time.monotonic() < deadline - 5:
        n = len(best)
        base = best.copy()
        base_score = best_score
        for it in range(50000):
            if time.monotonic() > deadline - 5:
                break
            candidate = base * (1.0 + 0.005 * rng.standard_normal(n))
            candidate = np.maximum(candidate, 0)
            cs = candidate.sum()
            if cs < 1e-10:
                continue
            candidate = candidate / cs * base.sum()
            score = evaluate_sequence(candidate.tolist())
            if score < base_score:
                base = candidate
                base_score = score
    
    if base_score < best_score:
        best = base
        best_score = base_score
    
    print(f"\nFinal score: {best_score:.6f}, length={len(best)}")
    return [float(x) for x in best]
```
