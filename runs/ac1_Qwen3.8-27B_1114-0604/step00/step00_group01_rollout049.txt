Minimizes 2*n*max(convolve(a,a))/sum(a)^2 for non-negative sequences using
FFT-accelerated subgradient descent in log-space (softmax parameterization).
Uses log-sum-exp as smooth surrogate for the max, with adaptive temperature.
Multiple restarts from diverse initializations for robustness.

import numpy as np
import time
import math
from scipy.fft import fft, ifft


def stable_softmax(v):
    """Numerically stable softmax."""
    v_shifted = v - np.max(v)
    e = np.exp(v_shifted)
    return e / np.sum(e)


def compute_objective_fast(a, fft_size):
    """Compute the objective value without calling evaluate_sequence."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    A = fft(a, fft_size)
    b = ifft(A * A).real[:2*n-1]
    max_b = np.max(b)
    return float(2 * n * max_b / (s * s))


def run_gradient_descent(theta_init, n, fft_size, duration_s, rng, label=""):
    """Run gradient descent in theta-space with softmax parameterization."""
    deadline = time.monotonic() + duration_s
    theta = theta_init.copy().astype(np.float64)
    
    best_a = stable_softmax(theta)
    best_obj = compute_objective_fast(best_a, fft_size)
    best_theta = theta.copy()
    
    # Parameters
    T = 0.1  # Temperature for log-sum-exp
    T_final = 0.01
    eta = 1.0  # Step size
    momentum = np.zeros(n)
    beta = 0.9  # Momentum coefficient
    
    no_improve_count = 0
    improve_threshold = 1e-6
    
    it = 0
    while time.monotonic() < deadline:
        a = stable_softmax(theta)
        
        # Compute b
        A = fft(a, fft_size)
        b = ifft(A * A).real[:2*n-1]
        
        # Compute soft weights (approximate indicator of max)
        b_range = b.max() - b.min()
        if b_range < 1e-12:
            b_range = 1e-12
        # Adaptive temperature: decrease over iterations
        T_adapt = T * b_range
        w = stable_softmax(b / T_adapt)
        
        # Compute gradient: g_a[i] = 2 * sum_k w[k] * a[k-i]
        # = 2 * ifft(fft(a) * conj(fft(w))).real[:n]
        W = fft(w, fft_size)
        g_a = 2.0 * ifft(A * np.conj(W)).real[:n]

        center = np.dot(a, g_a)
        g_theta = a * (g_a - center)

        theta = theta - eta * momentum
        
        # Prevent theta from growing too large (normalize)
        theta_max = np.max(np.abs(theta))
        if theta_max > 100:
            theta = theta * (50.0 / theta_max)
            momentum *= (50.0 / theta_max)

            a_new = stable_softmax(theta)
            obj = compute_objective_fast(a_new, fft_size)
            if obj < best_obj - improve_threshold:
                best_obj = obj
                best_a = a_new
                best_theta = theta.copy()
                no_improve_count = 0

                no_improve_count += 1

"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 for non-negative sequences using
FFT-accelerated gradient descent in log-space (softmax parameterization).
Uses log-sum-exp as smooth surrogate for max with adaptive temperature,
momentum, and multiple diverse restarts.

import numpy as np
import time
from scipy.fft import fft, ifft


def stable_softmax(v):
    v_shifted = v - np.max(v)
    e = np.exp(v_shifted)
    return e / np.sum(e)


def compute_objective_fast(a, fft_size):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    A = fft(a, fft_size)
    b = ifft(A * A).real[:2 * n - 1]
    max_b = np.max(b)
    return float(2 * n * max_b / (s * s))


def run_gradient_descent(theta_init, n, fft_size, duration_s, rng, label=""):

    best_a = a.copy()
