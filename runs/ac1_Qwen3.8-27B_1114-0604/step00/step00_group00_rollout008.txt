"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using LP-based iterative refinement
with subgradient fine-tuning. The LP approach (from the state of the art) finds
the best direction to move by solving a linear program that maximizes the total
mass of a new sequence while keeping its convolution with the current sequence
bounded by the current peak.
"""

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy import sparse


def _conv_fft(x):
    """Compute full linear convolution using FFT."""
    nn = len(x)
    size = 2 * nn - 1
    fs = 1
    while fs < size:
        fs <<= 1
    X = fft(x, fs)
    return ifft(X * X).real[:size]


def _build_csr(a, active_indices):
    """Build sparse matrix C where C[i, j] = a[k-j] for k = active_indices[i]."""
    nn = len(a)
    n_act = len(active_indices)
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(active_indices):
        j_lo = max(0, k - nn + 1)
        j_hi = min(nn - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[k - j])
    return sparse.csr_matrix((vals, (rows, cols)), shape=(n_act, nn))


def _lp_solve(a, M, top_frac=0.35, time_limit=60.0):
    """Solve reduced LP: max sum(b) s.t. (a*b)[k] <= M for top constraints, b>=0."""
    nn = len(a)
    c = _conv_fft(a)
    threshold = np.quantile(c, 1.0 - top_frac)
    active = np.where(c >= threshold)[0]
    
    if len(active) == 0:
        return None
    
    C = _build_csr(a, active)
    c_obj = -np.ones(nn)
    b_ub = np.full(len(active), M)
    
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                      bounds=[(0, None)] * nn, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            b = np.clip(res.x, 0.0, None)
            s_b = b.sum()
            if s_b > 1e-10:
                return b / s_b
    except Exception:
        pass
    return None


def _subgradient_step(a, eta, top_frac=0.05):
    """One projected subgradient step for minimizing max(conv(a,a))."""
    nn = len(a)
    c = _conv_fft(a)
    max_c = c.max()
    
    if max_c <= 1e-15:
        return a, max_c, 0.0
    
    threshold = np.quantile(c, 1.0 - top_frac)
    S = np.where(c >= threshold)[0]
    
    if len(S) == 0:
        return a, max_c, eta
    
    # Compute gradient: g[i] = 2/|S| * sum_{k in S} a[k-i]
    s_vec = np.zeros(2 * nn - 1)
    s_vec[S] = 1.0
    fs = 1
    while fs < 2 * nn - 1:
        fs <<= 1
    # Cross-correlation r[i] = sum_k s_vec[k] * conj(a[k-i])... 
    # Actually for real signals: r[i] = sum_k s_vec[k] * a[k-i]
    # Using FFT: r = ifft(conj(FFT(s_vec)) * FFT(a)) ... let me verify
    # sum_k s[k] * a[k-i]: let m = k-i, then k = m+i, sum_m s[m+i] * a[m]
    # This is the cross-correlation of s and a at lag i.
    # Cross-correlation via FFT: ifft(conj(FFT(s)) * FFT(a)) gives
    # r[i] = sum_k s[k+i] * a[k] (circular), which for linear is what we want
    # when i >= 0 and we take the first nn elements.
    S_f = fft(s_vec, fs)
    a_f = fft(a, fs)
    r = ifft(np.conj(S_f) * a_f).real[:nn]
    g = 2.0 * r / len(S)
    
    a_new = a - eta * g
    a_new = np.clip(a_new, 0.0, None)
    s_new = a_new.sum()
    if s_new < 1e-10:
        return a, max_c, eta * 0.5
    a_new = a_new / s_new
    return a_new, max_c, eta

    return best_score, best
