"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.

Approach:
  * Work with the sum-normalized vector p (sum(p)=1). The score is then
    homogeneous in the scale, so it is proportional to max_k (p*p)[k];
    we minimize that peak and finally rescale p to a valid output.
  * Fast full convolution via real FFT (O(n log n) per evaluation).
  * A smoothed subgradient of the 'max of convolution lags' using an
    adaptive-temperature softmax over the top K lags, giving a dense,
    O(K*n)-cheap direction.
  * Nesterov-accelerated projected subgradient descent onto the simple
    {a>=0, sum=1}, with adaptive step size, top-K and temperature
    annealing, and geometric random restarts to escape plateaus.
  * A small set of diverse analytic/random initializations, time-budgeted,
    followed by a final polishing pass on the global best.
  * The exact evaluator (evaluate_sequence) is invoked periodically and on
    the final candidate; the FFT-computed peak is used as a cheap proxy
    between exact calls.

All helpers are top-level, no closures / lambdas are used.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Fast full (linear) convolution via real FFT
# ----------------------------------------------------------------------
def fft_full_convolve(a, b):
    """Full linear convolution of 1-D real arrays a, b using real FFT."""
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.zeros(0, dtype=np.float64)
    result_len = n + m - 1
    fft_len = 1 << (result_len - 1).bit_length()
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    return np.fft.irfft(fa * fb, fft_len)[:result_len]


# ----------------------------------------------------------------------
# Smoothed subgradient of max_k c_k using top-K softmax weights
# ----------------------------------------------------------------------
def topk_subgradient(a, c, K, temp):
    """
    Return sum_k w_k * dc_k/da where
      dc_k/da[j] = 2*a[k-j]   (0 <= k-j < n, else 0)
      w           = softmax(temp * c) over the K largest lags.
    This is a smoothed subgradient direction for the peak of c.
    """
    n = a.size
    len_c = c.size
    K = min(int(K), len_c)
    if K <= 0:
        return np.zeros(n, dtype=np.float64)

    # K largest lags
    part = np.argpartition(-c, K - 1)[:K]
    order = np.argsort(-c[part])
    top_idx = part[order]

    top_vals = c[top_idx]
    vmax = top_vals.max()
    if vmax < 1e-15:
        return np.zeros(n, dtype=np.float64)

    exps = np.exp(np.clip(temp * (top_vals - vmax), -50.0, 50.0))
    wsum = exps.sum()
    if wsum < 1e-300:
        return np.zeros(n, dtype=np.float64)
    w = exps / wsum

    sg = np.zeros(n, dtype=np.float64)
    for idx in range(K):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = int(top_idx[idx])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        # a[k-j] for j = lo..hi  ==  reverse of a[k-hi : k-lo+1]
        sg[lo:hi + 1] += 2.0 * wi * a[k - hi:k - lo + 1][::-1]
    return sg


# ----------------------------------------------------------------------
# Project onto {a >= 0, sum = 1}
# ----------------------------------------------------------------------
def project_simplex(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


# ----------------------------------------------------------------------
# Core accelerator for one starting point
# ----------------------------------------------------------------------
def run_optimization(a_init, n, deadline, rng):
    """Nesterov projected subgradient descent from a_init (sum-normalized)."""
    a = np.ascontiguousarray(np.maximum(a_init, 0.0), dtype=np.float64)
    a = project_simplex(a)

    best_a = a.copy()
    c = fft_full_convolve(a, a)
    best_peak = float(c.max())

    v = np.zeros(n, dtype=np.float64)      # momentum
    beta = 0.85                            # momentum coeff
    eta = 0.08                             # base step size
    K = 40                                 # top-K lags
    temp = 4.0                             # softmax temperature
    v_cap = 0.05                           # max per-coordinate velocity

    no_improve = 0
    patience = 600
    restarts = 0

    it = 0
    while time.monotonic() < deadline:
        # ---- Nesterov look-ahead ------------------------------------------------
        y = a + beta * v
        if not np.all(np.isfinite(y)):
            y = a.copy()
        cy = fft_full_convolve(y, y)
        cy_peak = float(cy.max())

        if cy_peak < best_peak * (1.0 - 1e-9):
            best_peak = cy_peak
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---- geometric random restart if stuck ---------------------------------
        if no_improve > patience:
            restarts += 1
            scale = 0.04 * (1.0 + 0.4 * min(restarts, 12))
            a = best_a + scale * rng.standard_normal(n)
            a = project_simplex(a)
            v = np.zeros(n, dtype=np.float64)
            eta = min(0.12, eta * 0.9)          # shrink step
            K = min(80, K + 8)                  # broaden view
            temp = max(2.0, temp * 0.6)         # smooth more
            patience = max(250, int(patience * 0.85))
            no_improve = 0
            it += 1
            continue

        # ---- smoothed subgradient at look-ahead point --------------------------
        sg = topk_subgradient(y, cy, K, temp)
        if not np.all(np.isfinite(sg)):
            sg = np.zeros(n, dtype=np.float64)

        # ---- Nesterov momentum update ------------------------------------------
        v = beta * v - eta * sg
        # cap velocity to avoid overshoot beyond the simplicity of projection
        v = np.clip(v, -v_cap, v_cap)
        a = a + v
        a = project_simplex(a)

        # ---- monitor with the actual iterate -----------------------------------
        ca = fft_full_convolve(a, a)
        ca_peak = float(ca.max())
        if ca_peak < best_peak * (1.0 - 1e-9):
            best_peak = ca_peak
            best_a = a.copy()
            no_improve = 0

        it += 1

        # ---- parameter annealing ------------------------------------------------
        if it % 700 == 0:
            temp = min(150.0, temp * 1.18)      # sharpen toward true max
            K = max(4, int(K * 0.94))
            eta = max(0.002, eta * 0.95)
            beta = min(0.95, beta * 1.003)

    return best_a

# Optional cheap local block search on the current best (exact-score driven)
# ----------------------------------------------------------------------
def local_block_search(seq, n, deadline, rng, best_score, parent_sum):
    """Short, exact-evaluated block-perturbation polish on a list seq."""
    best = np.array(seq, dtype=np.float64).copy()
    best = np.clip(best, 0.0, 1000.0)
    local_best = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(local_best):
        return best.tolist(), best_score

    it = 0
    while time.monotonic() < deadline:
        cand = best.copy()
        width = int(rng.integers(max(1, n // 80), max(2, n // 8)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.08)))
        if rng.random() < 0.30:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            it += 1
            continue
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < local_best:
            local_best = sc
            best = cand
            if local_best < best_score:
                best_score = local_best
        it += 1
    return best.tolist(), best_score
