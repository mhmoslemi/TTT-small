def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv/ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.argmax(cond) - 1
    theta = cssv[i] / (i + 1)
    return np.maximum(v - theta, 0)
