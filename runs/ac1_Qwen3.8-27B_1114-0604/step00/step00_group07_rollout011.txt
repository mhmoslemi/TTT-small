def solve_lp_direction(f, M):
    """Find g that maximizes sum(g) subject to g >= 0 and f*g <= M."""
    n = len(f)
    c = -np.ones(n)  # minimize -sum(g)

    # (f*g)[k] = sum_i f[i] * g[k-i]

from scipy.optimize import linprog

def build_conv_matrix(f):
    """Build the matrix A such that A @ g = f * g (convolution)."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        A[k, j_min:j_max+1] = f[k-j_min : k-j_max-1 : -1]

    """Solve LP: max sum(g) s.t. g >= 0, f*g <= M."""
    n = len(f)
    c = -np.ones(n)
    A_ub = build_conv_matrix(f)
    b_ub = np.full(2*n-1, M)
