```python
import numpy as np
import time
from scipy.fft import fft, ifft

def compute_autocorr(f):
    """Compute linear autocorrelation (convolution with self) using FFT."""
    n = len(f)
    N = 2 * n - 1
    F = fft(f, N)
    return np.real(ifft(F * F))

def score_from_autocorr(f, h):
    """Compute the objective given f and its autocorrelation h."""
    n = len(f)
    s = np.sum(f)
    if s < 1e-10:
        return np.inf
    return 2.0 * n * np.max(h) / (s * s)

def subgradient_descent(f, n_steps=10000, eta0=0.1):
    """Run subgradient descent to minimize max autocorrelation."""
    f = f.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = np.sum(f)
    if s < 1e-10:
        return f
    f = f / s  # normalize to sum 1
    
    n = len(f)
    best_f = f.copy()
    best_score = score_from_autocorr(f, compute_autocorr(f))
    
    for t in range(n_steps):
        h = compute_autocorr(f)
        k_star = np.argmax(h)
        
        # Subgradient of (f*f)[k_star] with respect to f
        s_grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                if k_star == 2 * j:
                    s_grad[j] = f[idx]
                else:
                    s_grad[j] = 2.0 * f[idx]
        
        # Adaptive step size
        eta = eta0 / (1.0 + t * 0.01)
        
        f_new = f - eta * s_grad
        f_new = np.maximum(f_new, 0)
        s_new = np.sum(f_new)
        if s_new < 1e-10:
            break
        f_new = f_new / s_new
        
        h_new = compute_autocorr(f_new)
        score_new = score_from_autocorr(f_new, h_new)
        
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
        
        f = f_new
    
    return best_f
```

```python
import numpy as np
```

def _compute_autocorr(f):
    """Compute linear autocorrelation (f convolved with f) using FFT."""
    n = len(f)
    N = 2 * n - 1
    F = fft(f, N)
    return np.real(ifft(F * F))

def _soft_subgradient(f, h, tau):
    """Compute soft subgradient of max_k h_k using softmax weighting."""
    n = len(f)
    N = 2 * n - 1
    # Softmax weights
    h_shifted = h / max(tau, 1e-10) - np.max(h) / max(tau, 1e-10)
    w = np.exp(h_shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(N) / N
    else:
        w = w / w_sum
    
    # s_j = 2 * sum_i f_i * w_{i+j} for j = 0, ..., n-1
    # This is the cross-correlation of f and w (with w on [0, 2n-2])
    L = 4 * n  # enough to avoid circular wraparound
    F_fft = fft(f, L)
    W_fft = fft(w, L)
    cross = np.real(ifft(np.conj(F_fft) * W_fft))
    s_grad = 2.0 * cross[:n]
    
    # Handle the diagonal case (where k = 2j, the derivative is f_j not 2f_j)
    # For simplicity, ignore this O(n) correction (it's a small effect)
    
    return s_grad
