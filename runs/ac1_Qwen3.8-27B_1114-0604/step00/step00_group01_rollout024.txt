        w = np.exp(c_norm)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum

        # Gradient of log-sum-exp: (1/k) * cross_corr(w, a)
        grad = _fft_cross_corr(w, a) / k
        grad = np.maximum(grad, 0.0)  # ensure non-negativity for stability

        # Gradient step
        a = a - lr * grad

        # Project onto {a >= 0, sum(a) = 1}
        a = _project_nonneg_unit(a)

        it += 1

        # Anneal temperature and learning rate
        if it % 30 == 0:
            k = max(k * 0.97, k_min)
            lr = max(lr * 0.998, 1e-5)

        # Track improvements
        if it % 30 == 0:
            score = 2.0 * n * c_max
            if score < best_score:
                best_a = a.copy()
                best_score = score
                last_improve = it
                stall_count = 0
            else:
                stall_count += 1

        # Escape stagnation: temporarily increase k
        if stall_count > 200:
            k = min(k * 1.3, k_init)
            lr = min(lr * 1.2, base_lr)
            stall_count = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize the peak auto-convolution ratio using:
    1. Projected gradient descent with log-sum-exp smoothing (annealed)
    2. FFT-based O(n log n) convolution and cross-correlation
    3. Multiple random restarts with diverse initializations
    4. Final fine-tuning with small temperature
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 12.0)

    # Obtain initial sequence
    try:
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if a0.ndim != 1 or a0.size == 0:
            raise ValueError
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-10:
            a0 = np.ones(6121)
    except Exception:
        a0 = np.ones(6121)

    n = a0.size
    a0 = a0 / a0.sum()

    best_global = a0.copy()
    best_score = 2.0 * n * _fft_convolve_full(a0, a0).max()
    print(f"n={n}, initial score={best_score:.6f}")

    # --- Phase 1: Descent from given sequence (50% budget) ---
    p1_deadline = time.monotonic() + 0.50 * (float(budget_s) - 12.0)
    p1_deadline = min(p1_deadline, total_deadline)
    a1, s1 = _run_descent(best_global, n, p1_deadline, base_lr=0.06,
                          k_init=n / 8.0, k_min=n * 1e-6, label="phase1")
    if s1 < best_score:
        best_global = a1
        best_score = s1
    print(f"Phase 1 done: score={best_score:.6f}, time used={time.monotonic()- (total_deadline - float(budget_s) + 12.0):.1f}s")

    # --- Phase 2: Random restarts (35% budget) ---
    p2_budget = 0.35 * (float(budget_s) - 12.0)
    p2_start = time.monotonic()
    p2_deadline = p2_start + p2_budget
    p2_deadline = min(p2_deadline, total_deadline)

    n_restarts = 10
    per_restart = p2_budget / n_restarts

    for ri in range(n_restarts):
        if time.monotonic() >= p2_deadline - 10:
            break

        # Diverse initializations
        r = rng.random()
        if ri % 5 == 0:
            # Gaussian bump (centered)
            x = np.arange(n, dtype=np.float64)
            a_init = np.exp(-0.5 * ((x - n / 2.0) / (n / 4.0)) ** 2)
        elif ri % 5 == 1:
            # Perturbed best
            a_init = best_global + 0.3 * rng.standard_normal(n) * best_global
        elif ri % 5 == 2:
            # Exponential (front-loaded)
            a_init = np.exp(-np.arange(n, dtype=np.float64) * 4.0 / n)
        elif ri % 5 == 3:
            # Uniform random
            a_init = rng.random(n)
        else:
            # Heavy-tailed
            a_init = 1.0 / (1.0 + 0.1 * np.abs(rng.standard_normal(n)))

        a_init = np.maximum(a_init, 0.0)
        if a_init.sum() < 1e-10:
            a_init = np.ones(n)

        rd = p2_start + (ri + 1) * per_restart
        rd = min(rd, p2_deadline - 5)

        a_res, s_res = _run_descent(a_init, n, rd, base_lr=0.04,
                                    k_init=n / 6.0, k_min=n * 1e-6,
                                    label=f"restart{ri}")
        if s_res < best_score:
            best_global = a_res
            best_score = s_res
            print(f"  Restart {ri}: improved to {best_score:.6f}")

    print(f"Phase 2 done: best score={best_score:.6f}")

    # --- Phase 3: Fine-tuning (15% budget) ---
    p3_deadline = total_deadline
    a3, s3 = _run_descent(best_global, n, p3_deadline, base_lr=0.015,
                          k_init=n / 40.0, k_min=n * 1e-8, label="finetune")
    if s3 < best_score:
        best_global = a3
        best_score = s3

    print(f"Phase 3 done: final score={best_score:.6f}")

    # Scale result: individual values ~O(1), sum ~ n
    result = best_global * n

    # Final verification
    exact = evaluate_sequence(result.tolist())
    print(f"Exact evaluator score: {exact:.6f}")

    if not np.isfinite(exact):
        print("WARNING: non-finite score, using fallback")
        result = np.ones(n)
        exact = evaluate_sequence(result.tolist())
        print(f"Fallback score: {exact:.6f}")

    return [float(x) for x in result]
