import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    N = 1
    while N < n:
        N *= 2
    fa = np.fft.fft(a, N)
    fb = np.fft.fft(b, N)
    return np.fft.ifft(fa * fb).real[:n]

def compute_gradient(f, p):
    """
    Compute dM_smooth/df where M_smooth = (1/T)*log(sum exp(T*c))
    and p[k] = softmax(T*c[k]).
    
    dM_smooth/df[i] = 2 * sum_j p[i+j] * f[j] for j=0..n-1
    """
    n = len(f)
    # p has length 2n-1
    # We want G[i] = sum_{j=0}^{n-1} p[i+j] * f[j] for i = 0, ..., n-1
    # This is the cross-correlation of p and f (p "leads" f)

    p_rev = p[::-1]  # length 2n-1
    # conv(p_rev, f) has length (2n-1) + n - 1 = 3n-2
    # G[i] = conv(p_rev, f)[(2n-2) - i]

    """Compute the exact objective value."""
    n = len(f)
    c = np.convolve(f, f)
    max_c = c.max()
    sum_f = f.sum()
    if sum_f < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_f ** 2)

import numpy as np
import time


"""
Minimizes 2*n*max(conv(f,f))/(sum(f))^2 using FFT-accelerated gradient descent
with soft-max smoothing, Adam optimizer, and multiple restarts.
Uses exp-parameterization to naturally maintain non-negativity.
"""


def _compute_score(f):
    """Compute the exact objective value."""
    n = len(f)
    c = np.convolve(f, f)
    max_c = c.max()
    s = f.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


def _run_gradient_descent(f, time_budget, seed=0):
    """Run gradient descent to minimize the objective using FFT and Adam."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = np.clip(f, 1e-8, 1000.0)

    # Normalize to convenient scale (problem is scale-invariant)
    target_sum = np.sqrt(2.0 * n)
    f = f / f.sum() * target_sum

    # FFT size for convolution (need at least 2n-1)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    # FFT size for gradient computation (need at least 3n-2)
    N2 = 1
    while N2 < 3 * n - 2:
        N2 <<= 1

    deadline = time.monotonic() + time_budget
    best_score = _compute_score(f)
    best_f = f.copy()

    T = 5.0  # Soft-max temperature
    lr = 0.3  # Learning rate
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_step = 0

    # Adam state
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)

    start_time = time.monotonic()
    it = 0
    stall_count = 0
    stall_threshold_score = best_score

    while time.monotonic() < deadline:
        it += 1
        t_step += 1

        # Compute convolution c = conv(f, f) via FFT
        F = np.fft.rfft(f, N)
        c = np.fft.irfft(F * F, N)[:2 * n - 1]

        max_c = c.max()
        sum_f = f.sum()

        if sum_f < 1e-6 or not np.isfinite(max_c):
            break

        score = 2.0 * n * max_c / (sum_f * sum_f)
        if score < best_score - 1e-9:
            best_score = score
            best_f = f.copy()
            stall_count = 0
            stall_threshold_score = best_score
        else:
            stall_count += 1

        # Soft-max weights: p[k] proportional to exp(T * c[k] / max_c)
        scaled = T * c / max_c
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        p /= p_sum

        # Gradient: dM_smooth/df[i] = 2 * sum_j p[i+j] * f[j]
        # Compute G[i] = sum_j p[i+j] * f[j] via cross-correlation (FFT)
        # G[i] = conv(p_reversed, f)[2n-2-i]
        p_rev = p[::-1]
        Pr = np.fft.rfft(p_rev, N2)
        Fr = np.fft.rfft(f, N2)
        conv_val = np.fft.irfft(Pr * Fr, N2)[:3 * n - 2]
        G = conv_val[2 * n - 2:n - 1:-1]  # length n, indices 2n-2 down to n-1
        dM_df = 2.0 * G

        # Full gradient of J = 2n*M/S^2:
        # dJ/df[i] = (2n/S^2) * (dM/df[i] - 2*M/S)
        # Use max_c as M approximation
        grad = dM_df - 2.0 * max_c / sum_f

        # Adam update
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * grad * grad
        m_hat = m_buf / (1.0 - beta1 ** t_step)
        v_hat = v_buf / (1.0 - beta2 ** t_step)

        f -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project to non-negative and renormalize
        f = np.clip(f, 1e-8, 1000.0)
        f = f / f.sum() * target_sum

        # Adapt parameters periodically
        if it % 300 == 0:
            lr *= 0.97
            T = min(T * 1.03, 80.0)

        # If stuck for too long, perturb
        if stall_count > 2000:
            rng_local = np.random.default_rng(seed + it)
            perturb_idx = rng_local.integers(0, n, size=max(1, n // 50))
            f[perturb_idx] *= np.exp(rng_local.normal(0, 0.3, size=len(perturb_idx)))
            f = np.clip(f, 1e-8, 1000.0)
            f = f / f.sum() * target_sum
            stall_count = 0
            lr = max(lr, 0.1)

        if it % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"      iter={it}, score={score:.6f}, best={best_score:.6f}, "
                  f"T={T:.1f}, lr={lr:.4f}, elapsed={elapsed:.0f}s")

    return best_f, best_score


def _local_search(f, time_budget, seed=0):
    """Local search: try random block perturbations and keep improvements."""
    n = len(f)
    f = f.astype(np.float64).copy()
    best_score = _compute_score(f)
    best_f = f.copy()
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + time_budget

    while time.monotonic() < deadline:
        # Random block perturbation
        width = int(rng.integers(max(1, n // 300), max(2, n // 10)))
        start = int(rng.integers(0, max(1, n - width)))
        candidate = best_f.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0, 0.15, size=width))
        candidate = np.clip(candidate, 0.0, 1000.0)

        if rng.random() < 0.3:
            # Also try a second block
            w2 = int(rng.integers(max(1, n // 300), max(2, n // 10)))
            s2 = int(rng.integers(0, max(1, n - w2)))
            candidate[s2:s2 + w2] *= np.exp(rng.normal(0, 0.15, size=w2))
            candidate = np.clip(candidate, 0.0, 1000.0)

        score = _compute_score(candidate)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = candidate

    return best_f, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Minimize 2*n*max(conv(f,f))/(sum(f))^2 using FFT-accelerated gradient descent
    with soft-max smoothing and Adam optimizer, followed by local search polish.
    Tries multiple sequence lengths and random initializations.
    """
    deadline = time.monotonic() + budget_s - 15
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_sequence = None

    # Sequence lengths to try (mixture of Mersenne and round numbers)
    lengths = [3809, 5000, 4095]

    # Reserve last 10% for local search
    main_time = budget_s * 0.85
    local_time = budget_s * 0.15

    # Time allocation per length
    time_per_length = main_time / len(lengths)

    for n in lengths:
        if time.monotonic() > deadline - local_time - 30:
            break

        print(f"\n=== Trying n={n}, budget={time_per_length:.0f}s ===")
        length_start = time.monotonic()

        # 3 restarts per length
        n_restarts = 3
        time_per_restart = time_per_length / n_restarts

        for r in range(n_restarts):
            remaining_total = min(deadline - local_time - 30 - time.monotonic(),
                                  length_start + time_per_length - time.monotonic())
            if remaining_total < 20:
                break

            # Different initializations
            if r == 0:
                # Start near uniform with noise
                f = 1.0 + 0.3 * rng.standard_normal(n)
            elif r == 1:
                # Sum of random cosines (creates structured patterns)
                f = np.ones(n)
                t = np.arange(n)
                for _ in range(4):
                    freq = rng.uniform(0.5, 10.0)
                    phase = rng.uniform(0, 2 * np.pi)
                    f += 0.4 * np.cos(2 * np.pi * freq * t / n + phase)
                f = np.clip(f, 0.05, None)
            else:
                # Exponential decay pattern + noise
                decay = np.exp(-np.arange(n) / (n / 5.0))
                f = decay * rng.uniform(0.5, 1.5, n) + 0.05
                # Combine forward and reversed for symmetry breaking
                f = 0.5 * f + 0.5 * f[::-1] * rng.uniform(0.8, 1.2, n)

            f = np.clip(f, 1e-4, 1000.0)
            f = f / f.sum() * np.sqrt(2.0 * n)

            f_opt, score = _run_gradient_descent(f, remaining_total, seed=seed + r * 100)

            if score < best_score:
                best_score = score
                best_sequence = f_opt.copy()
                print(f"    New best: {best_score:.6f} (n={n}, restart={r})")

        print(f"  Best after gradient descent (n={n}): {best_score:.6f}")

    # Local search polish
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Local search polish, budget={local_time:.0f}s ===")
        local_budget = min(local_time, deadline - 10 - time.monotonic())
        if local_budget > 10:
            best_sequence, best_score = _local_search(best_sequence, local_budget, seed=seed + 9999)
            print(f"  After local search: {best_score:.6f}")

    # Final verification
    if best_sequence is not None:
        final_score = _compute_score(best_sequence)
        if final_score < best_score:
            best_score = final_score
        print(f"\nFinal score: {best_score:.6f}, length={len(best_sequence)}")
    else:
        best_sequence = np.ones(4)
        best_score = _compute_score(best_sequence)
        print(f"Fallback score: {best_score:.6f}")

    return [float(x) for x in best_sequence]
