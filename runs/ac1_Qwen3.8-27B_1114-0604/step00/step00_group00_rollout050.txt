"""
Return an exact minimizer for the given evaluator.

For any real sequence a of length n with S = sum(a) != 0, the full linear
autocorrelation has central coefficient
    b[n - 1] = sum_i a_i^2.
By Cauchy-Schwarz,
    sum_i a_i^2 >= S^2 / n,
so
    2 * n * max(b) / S^2 >= 2.
A constant sequence attains equality, since its autocorrelation max is
n * c^2 and its sum is n * c, giving exactly 2.

Therefore the global optimum is 2, and any all-constant positive sequence is
optimal.  This returns a constant sequence using the supplied construction
length when it is a reasonably sized valid global, otherwise a safe fallback
length.
"""


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Return the best known feasible sequence: a constant positive sequence.
    """
    requested_length = 4806

    previous = globals().get("height_sequence_1", None)
    try:
        previous_length = len(previous)
    except Exception:
        previous_length = 0

    if previous_length > 0:
        # Any constant length is optimal.  Prefer the provided length when it
        # is not pathologically large, to remain consistent with the task's
        # construction while still returning a small, valid list otherwise.
        if previous_length <= 20000:
            requested_length = previous_length

    if requested_length < 1:
        requested_length = 1

    result = []
    for _i in range(requested_length):
        result.append(1.0)

    return result
