from scipy import optimize

def compute_score(a):
    """Compute the score for a non-negative sequence with sum=1."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = np.max(c)
    return 2 * n * max_c  # since sum(a) = 1

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def log_sum_exp(c, beta):

    beta_c = beta * c

    exp_bc = np.exp(beta_c - max_bc)
    sum_exp = np.sum(exp_bc)
    lse = (max_bc + np.log(sum_exp)) / beta
    p = exp_bc / sum_exp
    return lse, p

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1.0) > (cssv - 1.0))[0]

        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
