```
import numpy as np
import time

def _score_np(a):
    a=np.asarray(a,dtype=np.float64)
    n=a.size
    if n==0: return np.inf
    S=a.sum()
    if S < 1e-8: return np.inf
    b=np.convolve(a,a)
    return float(2.0*n*b.max()/(S*S))

def _greedy_sidon(max_start, gap, max_k, allow_fail=True):
    if max_start < 0: return []
    centers=[0]
    sums={0}
    while len(centers)<max_k:
        start=centers[-1]+gap
        if start > max_start: break
        found=False
        # To avoid scanning whole range repeatedly from start if many, just for x in range(start, max_start+1):
        for x in range(start, max_start+1):
            ok=True
            for y in centers:
                if x+y in sums:
                    ok=False
                    break
            if ok:
                for y in centers:
                    sums.add(x+y)
                centers.append(x)
                found=True
                break
        if not found:
            break
    return centers

def _make_cluster_seq(N, L, gap, max_k=200):
    centers=_greedy_sidon(N-L, gap, max_k)
    k=len(centers)
    if k<2 or L<1: return None
    a=np.zeros(N,dtype=np.float64)
    val=1.0/(float(k)*float(L))
    for c in centers:
        a[c:c+L] += val
    S=a.sum()
    if S <= 0: return None
    a /= S
    return a
```

```
def _lp_refresh(a):
    n=a.size
    M=float(np.max(np.convolve(a,a)))
    if M <=0: return None
    A=np.zeros((2*n-1,n), dtype=np.float64)
    for k in range(2*n-1):
        lo=max(0,k-(n-1)); hi=min(n-1,k)
        if lo<=hi:
            A[k, lo:hi+1] = a[lo:hi+1][::-1]
    c=-np.ones(n)
    b_ub=np.full(2*n-1, M)
    bounds=[(0.0, None)] * n
    try:
        res=linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None
    if res.success:
        x=res.x
        x[x<0]=0
        return x
    return None
```

```python
"""
Search for a short nonnegative step function with small
    2*n*max(conv(a,a))/sum(a)^2.

The main idea is to place several uniform-length clusters on a strictly
Sidon set of starting positions.  A Sidon condition makes the peaks of distinct
cluster-pair convolutions avoid one another, while the long uniform clusters
spread each pair contribution over many lags.  The code searches several
lengths, cluster lengths, and minimum gaps, evaluates the exact objective,
and keeps the best sequence found.  A small LP-based polish is included as a
fallback for short sequences.
"""

import time
import math
import numpy as np

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _score_np(a):
    """Exact objective for a numpy array, matching evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    b = np.convolve(a, a)
    mb = float(np.max(b))
    return float(2.0 * n * mb / (s * s))


def _greedy_sidon(max_start, gap, max_k):
    """
    Greedy strictly-Sidon set in [0, max_start] with adjacent gap at least
    `gap`.  All pair sums x_i + x_j, i <= j, are distinct.
    """
    if max_start < 0 or max_k < 1:
        return []

    centers = [0]
    sums = set()
    sums.add(0)

    while len(centers) < max_k:
        start = centers[-1] + gap
        if start > max_start:
            break

        found = False
        for x in range(start, max_start + 1):
            ok = (2 * x) not in sums
            if ok:
                for y in centers:
                    if (x + y) in sums:
                        ok = False
                        break

            if ok:
                for y in centers:
                    sums.add(x + y)
                sums.add(2 * x)
                centers.append(x)
                found = True
                break

        if not found:
            break

    return centers


def _sequence_from_centers(centers, cluster_len, max_len):
    """Build a normalized nonnegative sequence from cluster starts."""
    if not centers or cluster_len <= 0:
        return None

    centers = list(centers)
    max_c = max(centers)
    n = max_c + cluster_len
    if n < 2 or n > max_len:
        return None

    a = np.zeros(n, dtype=np.float64)
    val = 1.0 / (float(len(centers)) * float(cluster_len))

    for c in centers:
        a[c:c + cluster_len] += val

    s = float(np.sum(a))
    if s < 1e-12:
        return None

    a /= s
    return a


def _construction_search(deadline, best_seq, best_score, n_targets, l_targets):
    """Evaluate many Sidon-cluster constructions and update the best result."""
    for N in n_targets:
        if time.monotonic() > deadline - 10:
            break

        for L in l_targets:
            if time.monotonic() > deadline - 10:
                break
            if L >= N or L < 4:
                continue

            sqrt_l = max(1, int(math.sqrt(L)))
            gap_set = {1, 2, 4, 8, sqrt_l, 2 * sqrt_l, max(1, L // 4)}
            gap_set = {g for g in gap_set if 1 <= g < N}

            for gap in sorted(gap_set):
                if time.monotonic() > deadline - 10:
                    break

                max_k = int(8.0 * math.sqrt(float(N))) + 60
                centers = _greedy_sidon(N - L, gap, max_k)
                if len(centers) < 2:
                    continue

                seq = _sequence_from_centers(centers, L, N)
                if seq is None:
                    continue

                sc = _score_np(seq)
                if np.isfinite(sc) and sc < best_score:
                    best_seq = seq.copy()
                    best_score = float(sc)
                    print(
                        f"construction N_target={N} L={L} gap={gap} "
                        f"len={seq.size} centers={len(centers)} score={best_score:.9f}"
                    )

    return best_seq, best_score


def _lp_refresh(a):
    """
    Solve:
        maximize sum(g)
        subject to (a * g)_k <= max(a * a), g >= 0.
    """
    if linprog is None:
        return None

    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n < 2 or n > 350:
        return None

    b = np.convolve(a, a)
    M = float(np.max(b))
    if not np.isfinite(M) or M <= 0:
        return None

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = a[lo:hi + 1][::-1]

    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
        )
    except Exception:
        return None

    if res is None or not res.success or res.x is None:
        return None

    x = np.asarray(res.x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    return x


def _lp_local_polish(a, deadline, max_outer=120):
    """
    A short local search using the LP direction from the previous state of the
    art philosophy.  It is only invoked for short sequences and only accepts
    exact improvements.
    """
    if a is None or a.size > 350:
        return a

    best = np.asarray(a, dtype=np.float64).copy()
    best_score = _score_np(best)
    work = best.copy()
    misses = 0

    for _ in range(max_outer):
        if time.monotonic() > deadline:
            break

        g = _lp_refresh(work)
        if g is None:
            break

        sg = float(np.sum(g))
        sa = float(np.sum(work))
        if not np.isfinite(sg) or not np.isfinite(sa) or sg <= 0 or sa <= 0:
            break

        g = g * (sa / sg)
        improved = False

        for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9):
            cand = (1.0 - t) * work + t * g
            cand = np.maximum(cand, 0.0)
            sc = _score_np(cand)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = float(sc)
                best = cand.copy()
                work = cand.copy()
                improved = True

        if improved:
            misses = 0
        else:
            misses += 1
            if misses >= 3:
                break

    return best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Return the best sequence found.  The function is self-contained and will
    return a valid sequence before the allocated deadline.
    """
    start_time = time.monotonic()
    budget = max(0.0, float(budget_s))
    deadline = start_time + max(0.0, budget - 5.0)

    best_seq = None
    best_score = np.inf

    # Use the provided parent sequence as a fallback candidate.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.clip(parent, 0.0, 1000.0)
            ps = _score_np(parent)
            if np.isfinite(ps):
                best_seq = parent.copy()
                best_score = float(ps)
                print(f"parent length={parent.size} score={best_score:.9f}")
    except Exception:
        pass

    # Primary deterministic construction search.
    n_targets = [256, 384, 512, 768, 1024, 1536, 2048]
    l_targets = [16, 24, 32, 48, 64, 96, 128]

    best_seq, best_score = _construction_search(
        deadline, best_seq, best_score, n_targets, l_targets
    )

    # If the primary search did not reach the target, try longer / wider
    # constructions.  These are only evaluated if time remains.
    if best_score > 1.5027 and time.monotonic() < deadline - 60:
        n_targets2 = [3072, 4096]
        l_targets2 = [64, 96, 128, 192, 256]
        best_seq, best_score = _construction_search(
            deadline, best_seq, best_score, n_targets2, l_targets2
        )

    # Optional LP polish for short best candidates.
    if (
        best_seq is not None
        and best_seq.size <= 300
        and best_score > 1.02
        and time.monotonic() < deadline - 45
    ):
        polished = _lp_local_polish(best_seq, deadline)
        if polished is not None:
            ps = _score_np(polished)
            if np.isfinite(ps) and ps < best_score:
                best_seq = polished
                best_score = float(ps)
                print(f"LP polish score={best_score:.9f}")

    # Final safety: if everything failed, return a valid constant sequence.
    if best_seq is None or not np.all(np.isfinite(best_seq)):
        best_seq = np.ones(128, dtype=np.float64)
        best_score = _score_np(best_seq)

    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Tiny positive floor.  This is numerically negligible but ensures all
    # entries are strictly positive if desired.
    best_seq = best_seq + 1e-12
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_list = [float(x) for x in best_seq]

    try:
        final_score = float(evaluate_sequence(final_list))
    except Exception:
        final_score = float(best_score)

    if (not np.isfinite(final_score)) or final_score > 2.0:
        fallback = np.ones(128, dtype=np.float64) + 1e-12
        final_list = [float(x) for x in fallback]
        try:
            final_score = float(evaluate_sequence(final_list))
        except Exception:
            final_score = 2.0

    elapsed = time.monotonic() - start_time
    print(
        f"final length={len(final_list)} score={final_score:.12f} "
        f"elapsed={elapsed:.2f}s"
    )

    return final_list
```
