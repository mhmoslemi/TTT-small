Minimizes the step function evaluation score 2*n*max(convolve(a,a))/sum(a)^2
using log-sum-exp smoothed gradient descent with adaptive parameters,
multiple diverse initializations, and periodic LP-based refinement.

from scipy.sparse import csr_matrix


def compute_soft_gradient(a, alpha):
    """Compute gradient of log-sum-exp of convolve(a,a)."""
    b = np.convolve(a, a)
    b_shift = b - b.max()
    exp_b = np.exp(alpha * b_shift)
    p_sum = exp_b.sum()
    if p_sum < 1e-300 or not np.isfinite(p_sum):
        p = np.ones_like(b) / b.size
    else:
        p = exp_b / p_sum
    g = 2.0 * np.correlate(p, a, mode='valid')
    return g

def lp_refine(f, M, time_budget=30.0):
    """Solve LP to find g maximizing sum(g) s.t. g>=0, convolve(f,g)<=M. Returns g or None."""
    n = len(f)
    M = float(M)
    deadline = time.monotonic() + time_budget

    # Build constraint matrix for convolve(f, g) <= M
    # Row k: sum_j f[j]*g[k-j] <= M for valid j
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)

    # Bounds: 0 <= g[j] <= 1000
    bounds = [(0, 1000)] * n

    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=bounds, method='highs')

    if res.status == 0 and res.fun is not None:
        g = np.maximum(res.x, 0)
        gs = g.sum()
        if gs > 0.01:
            return g / gs
    return None


def lp_blend_step(f, time_budget=30.0, t=0.3):
    """One LP-based blend step: compute M, solve LP for g, blend f and g."""
    n = len(f)
    b = np.convolve(f, f)
    M = float(b.max())
    g = lp_refine(f, M, time_budget=time_budget)
    if g is None:
        return f
    f_new = (1 - t) * f + t * g
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 0:
        f_new = f_new / s
    return f_new
