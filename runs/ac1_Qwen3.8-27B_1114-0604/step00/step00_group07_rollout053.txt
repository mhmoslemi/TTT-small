def downsample_sequence(arr,n):
    arr=np.asarray(arr,float).ravel()
    arr=np.clip(arr,0,1000)
    if arr.size==0: return np.ones(n)
    if arr.size==n: return arr
    if arr.size < n:
       # repeat/pad
       reps=(n+arr.size-1)//arr.size
       arr=np.tile(arr,reps)[:n]
    else:
       win=(arr.size+n-1)//n
       pad=win*n - arr.size
       if pad>0: arr=np.pad(arr,(0,pad),mode='edge')
       arr=arr.reshape(n,win).mean(axis=1)
    return arr

"""
Search for low peak self-convolution nonnegative step functions.

The optimizer combines:
  * small-n full linear-programming iterations (the active-set style method
    from the state of the art, but with non-uniform restarts),
  * larger-n active-constraint LP cutting planes,
  * mirror-descent smoothing of the largest convolution lags,
  * a final local multiplicative/smooth search.

The objective is scale invariant, so candidates are kept at sum = sqrt(2n),
where the score equals max(convolve(a, a)) up to numerical error.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

_MATRIX_CACHE = {}


def _fft_length(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_target(arr, target):
    """Normalize arr to be nonnegative, bounded by 1000, and sum target."""
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if arr.size == 0:
        return arr

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())

    if s <= 1e-12:
        val = target / float(arr.size)
        if val > 1000.0:
            val = 1000.0
        return np.full(arr.size, val, dtype=np.float64)

    arr = arr * (target / s)
    arr = np.clip(arr, 0.0, 1000.0)

    s2 = float(arr.sum())
    if s2 <= 1e-12:
        val = target / float(arr.size)
        if val > 1000.0:
            val = 1000.0
        return np.full(arr.size, val, dtype=np.float64)

    arr = arr * (target / s2)
    return np.clip(arr, 0.0, 1000.0)


def convolution_ab(a, b):
    """Full linear convolution, using direct multiply-add for small sizes."""
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    if a.size == 0 or b.size == 0:
        return np.empty(0, dtype=np.float64)

    m = a.size + b.size - 1

    # Direct convolution is cheaper and exact for modest sizes.
    if m <= 4096:
        return np.convolve(a, b)

    L = _fft_length(m)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    c = np.fft.irfft(fa * fb, L)[:m]
    c = np.where(np.isfinite(c), c, 0.0)
    return c


def convolution_full(a):
    """Full self-convolution of a."""
    return convolution_ab(a, a)


def conv_max_sequence(arr):
    """Maximum coefficient of arr * arr."""
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.inf
    c = convolution_full(arr)
    if c.size == 0:
        return np.inf
    return float(np.max(c))


def objective_sequence(arr):
    """Internal objective mirroring evaluate_sequence, except using FFT for large n."""
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    m = conv_max_sequence(arr)
    n = arr.size
    return float(2.0 * n * m / (s * s))


def build_full_conv_matrix(f):
    """
    Build dense matrix A for fixed f such that (A @ g)_k = sum_j f[k-j] g[j].
    Only for modest n; cached index data are reused.
    """
    n = int(f.shape[0])
    if n not in _MATRIX_CACHE:
        rows = np.arange(2 * n - 1, dtype=np.int64)[:, None]
        cols = np.arange(n, dtype=np.int64)[None, :]
        d = rows - cols
        mask = (d >= 0) & (d < n)
        flat = np.nonzero(mask)[0]
        dvals = d[mask].astype(np.int64, copy=False)
        _MATRIX_CACHE[n] = (flat, dvals, 2 * n - 1, n)

    flat, dvals, mrows, ncols = _MATRIX_CACHE[n]
    A = np.zeros((mrows, ncols), dtype=np.float64)
    A[flat] = f[dvals]
    return A


def build_active_conv_matrix(f, rows):
    """Dense matrix for a subset of convolution-lag constraints."""
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    m = int(len(np.atleast_1d(rows)))
    A = np.zeros((m, n), dtype=np.float64)

    for ii in range(m):
        k = int(rows[ii])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

    return A


def solve_lp_probabilities(A, M, n):
    """
    Solve:
        maximize sum(g)
        subject to A @ g <= M, g >= 0.
    """
    if linprog is None:
        return None
    if A is None or A.size == 0 or n <= 0:
        return None

    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(A.shape[0], float(M), dtype=np.float64)
    bounds = np.empty((n, 2), dtype=np.float64)
    bounds[:, 0] = 0.0
    bounds[:, 1] = np.inf

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
        )
    except Exception:
        return None

    if res is None:
        return None
    if getattr(res, "status", 1) != 0:
        return None

    g = getattr(res, "x", None)
    if g is None:
        return None

    g = np.asarray(g, dtype=np.float64).ravel()
    if g.size != n or not np.all(np.isfinite(g)):
        return None

    g = np.maximum(g, 0.0)
    if float(g.sum()) <= 1e-12:
        return None

    return g


def lp_line_search(f, g, target, current_m):
    """Try several LP-directions steps and accept the first descent."""
    g = np.asarray(g, dtype=np.float64).ravel()
    sg = float(g.sum())
    if sg <= 1e-12:
        return None

    g = target * g / sg
    g = np.clip(g, 0.0, 1000.0)

    sg2 = float(g.sum())
    if sg2 <= 1e-12:
        return None
    g = target * g / sg2

    best_cand = None
    best_m = None

    alphas = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125]
    for alpha in alphas:
        cand = (1.0 - alpha) * f + alpha * g
        cand = normalize_target(cand, target)
        m = conv_max_sequence(cand)
        if m < current_m - 1e-11:
            return cand, m

    return None


def optimize_lp_full(f0, deadline, rng=None):
    """Full-constraint LP iteration for relatively small n."""
    f = np.asarray(np.asarray(f0, dtype=np.float64).ravel(), dtype=np.float64)
    n = f.size
    if n == 0:
        return f, np.inf

    if linprog is None:
        f = normalize_target(f, np.sqrt(2.0 * f.size))
        return f, objective_sequence(f)

    if n > 1400:
        f = normalize_target(f, np.sqrt(2.0 * f.size))
        return f, objective_sequence(f)

    target = np.sqrt(2.0 * n)
    f = normalize_target(f, target)

    best = f.copy()
    best_m = conv_max_sequence(f)

    while time.monotonic() < deadline:
        cur_m = conv_max_sequence(f)
        if cur_m < best_m - 1e-11:
            best_m = cur_m
            best = f.copy()

        A = build_full_conv_matrix(f)
        g = solve_lp_probabilities(A, cur_m, n)
        if g is None:
            break

        upd = lp_line_search(f, g, target, cur_m)
        if upd is None:
            break

        f, cur_m = upd
        if cur_m < best_m - 1e-11:
            best_m = cur_m
            best = f.copy()

    score = objective_sequence(best)
    return best, score


def optimize_lp_active(f0, deadline, rng=None, max_rows=256):
    """
    Active-constraint LP/cutting-plane iteration for larger n.
    This solves LPs with only the most important constraints and adds
    violated constraints when they appear.
    """
    f = np.asarray(f0, dtype=np.float64).ravel()
    n = f.size
    if n == 0:
        return f, np.inf

    target = np.sqrt(2.0 * n)
    f = normalize_target(f, target)

    best = f.copy()
    best_m = conv_max_sequence(f)
    no_improve = 0

    if linprog is None:
        return best, objective_sequence(best)

    max_rows = int(min(max_rows, max(32, n // 16)))

    while time.monotonic() < deadline:
        cur_m = conv_max_sequence(f)
        if cur_m < best_m - 1e-11:
            best_m = cur_m
            best = f.copy()

        c = convolution_full(f)
        if c.size == 0:
            break

        full_m = float(np.max(c))
        if full_m <= 0.0:
            break

        kth_rows = min(max_rows, max(32, c.size // 16))
        if kth_rows >= c.size:
            rows = np.arange(c.size, dtype=np.int64)
        else:
            kth = c.size - kth_rows
            rows = np.argpartition(c, kth)[kth:].astype(np.int64, copy=False)

        in_rows = np.zeros(c.size, dtype=bool)
        in_rows[rows] = True
        row_list = [int(x) for x in rows]

        g = None

        for _inner in range(4):
            if len(row_list) == 0:
                break

            A = build_active_conv_matrix(f, row_list)
            g = solve_lp_probabilities(A, full_m, n)
            if g is None:
                break

            fg = convolution_ab(f, g)
            if fg.size < c.size:
                fg = np.pad(fg, (0, max(0, c.size - fg.size)))

            viol_mask = fg > full_m * (1.0 + 1e-6) + 1e-9
            viol_idx = np.nonzero(viol_mask)[0]

            if viol_idx.size == 0:
                break

            if len(row_list) >= max_rows:
                break

            add_budget = min(24, max_rows - len(row_list))
            if add_budget <= 0:
                break

            viol_vals = fg[viol_idx]
            if viol_vals.size > add_budget:
                part = np.argpartition(viol_vals, -add_budget)[-add_budget:]
                add_idx = viol_idx[part]
            else:
                add_idx = viol_idx

            added = 0
            for idx in add_idx:
                idx = int(idx)
                if not in_rows[idx]:
                    row_list.append(idx)
                    in_rows[idx] = True
                    added += 1
                    if added >= add_budget:
                        break

        if g is None:
            break

        upd = lp_line_search(f, g, target, full_m)
        if upd is None:
            no_improve += 1
            if no_improve > 4:
                break
            continue

        f, cur_m = upd
        no_improve = 0

        if cur_m < best_m - 1e-11:
            best_m = cur_m
            best = f.copy()

    score = objective_sequence(best)
    return best, score


def optimize_mirror(f0, deadline, rng=None):
    """
    Mirror-descent peak smoothing:
      p <- normalize(p * exp(-eta * grad)),
    where grad is the weighted subgradient from the largest lags.
    """
    f = np.asarray(f0, dtype=np.float64).ravel()
    n = f.size
    if n == 0:
        return f, np.inf

    target = np.sqrt(2.0 * n)
    f = normalize_target(f, target)

    best = f.copy()
    best_m = conv_max_sequence(f)
    topk = int(min(128, max(16, n // 32)))

    etas = [0.35, 0.18, 0.09, 0.045, 0.0225, 0.01125]

    while time.monotonic() < deadline:
        c = convolution_full(f)
        if c.size == 0:
            break

        cur_m = float(np.max(c))
        if cur_m < best_m - 1e-11:
            best_m = cur_m
            best = f.copy()

        if cur_m <= 0.0:
            break

        if topk >= c.size:
            idx = np.arange(c.size, dtype=np.int64)
        else:
            kth = c.size - topk
            idx = np.argpartition(c, kth)[kth:].astype(np.int64, copy=False)

        w = c[idx].astype(np.float64)
        w = np.where(np.isfinite(w), w, 0.0)
        w = np.maximum(w, 0.0)
        sw = float(w.sum())
        if sw <= 1e-12:
            break
        w = w / sw

        grad = np.zeros(n, dtype=np.float64)

        for ii in range(idx.size):
            k = int(idx[ii])
            weight = float(w[ii])
            if weight <= 0.0:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                r_start = k - hi
                r_end = k - lo
                grad[lo:hi + 1] += (2.0 * weight) * f[r_start:r_end + 1][::-1]

        accepted = None
        accepted_m = None

        for eta in etas:
            logp = -eta * grad
            mn = float(np.min(logp))
            if not np.isfinite(mn):
                continue
            scaled = np.exp(logp - mn)
            cand_raw = f * scaled
            cand = normalize_target(cand_raw, target)
            m = conv_max_sequence(cand)
            if m < cur_m - 1e-11:
                accepted = cand
                accepted_m = m
                break

        if accepted is None:
            # If rng is available, try a tiny multiplicative jog to escape plateaus.
            if rng is not None and rng.random() < 0.5:
                jog = np.exp(rng.normal(0.0, 0.05, n))
                cand = normalize_target(f * jog, target)
                m = conv_max_sequence(cand)
                if m < cur_m - 1e-11:
                    f, cur_m = cand, m
                    continue
            break

        f, cur_m = accepted, accepted_m
        if cur_m < best_m - 1e-11:
            best_m = cur_m
            best = f.copy()

    score = objective_sequence(best)
    return best, score


def downsample_sequence(arr, n):
    """Best-effort resampling of a previous sequence to length n."""
    arr = np.asarray(arr, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)

    if n <= 0:
        return np.empty(0, dtype=np.float64)

    if arr.size == 0:
        return np.full(n, 1.0, dtype=np.float64)

    if arr.size == n:
        return arr.copy()

    if arr.size < n:
        reps = (n + arr.size - 1) // arr.size
        arr = np.tile(arr, reps)[:n]
        return arr

    win = (arr.size + n - 1) // n
    pad = win * n - arr.size
    if pad > 0:
        arr = np.pad(arr, (0, pad), mode="edge")

    return arr.reshape(n, win).mean(axis=1)


def make_initials(n, rng, count):
    """Produce several diverse non-uniform nonnegative starting sequences."""
    if n <= 0:
        return []

    target = np.sqrt(2.0 * n)
    out = []

    theta = np.linspace(0.0, 2.0 * np.pi, n, endpoint=False)
    u = np.linspace(0.0, 1.0, n)

    patterns = []

    # Halves of a cosine: break left-right symmetry while staying smooth.
    for amp in (0.3, 0.6, 0.9, 1.2):
        patterns.append(1.0 + amp * np.cos(np.pi * u))
    for amp in (0.3, 0.6, 0.9, 1.2):
        patterns.append(1.0 + amp * np.sin(np.pi * u))

    # Sinusoidal perturbations at several frequencies/phases.
    for amp in (0.35, 0.7, 1.0, 1.3):
        for freq in (1, 2, 3, 5, 8):
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            patterns.append(1.0 + amp * np.cos(freq * theta + phase))

    # A few short sliding-window smoothed random fields.
    for _ in range(5):
        noise = rng.normal(size=n)
        k = min(11 if n >= 256 else 5, max(1, n))
        kernel = np.ones(k, dtype=np.float64) / float(k)
        x = np.convolve(noise, kernel, mode="same")
        mx = float(np.mean(x))
        sd = float(np.std(x))
        if sd > 1e-12:
            x = (x - mx) / sd
        else:
            x = rng.normal(size=n)
        amp = 0.5 + 0.8 * float(rng.random())
        patterns.append(1.0 + amp * x)

    idxs = rng.choice(len(patterns), size=min(max(1, count), len(patterns)), replace=False)
    for idx in idxs:
        x = patterns[idx]
        x = np.maximum(x, 0.02)
        out.append(normalize_target(x, target))

    while len(out) < max(1, count):
        x = 1.0 + 0.8 * rng.normal(size=n)
        x = np.maximum(x, 0.02)
        out.append(normalize_target(x, target))

    # The previous incumbent can be useful if available.
    try:
        hs = globals().get("height_sequence_1", None)
        if hs is not None:
            x = downsample_sequence(hs, n)
            if x.size == n and float(x.sum()) > 0.0:
                out.append(normalize_target(x, target))
    except Exception:
        pass

    return out


def local_search(arr, deadline, rng):
    """Final small-improvement local search: smooth noise, block, and Fourier-mode moves."""
    f = np.asarray(arr, dtype=np.float64).ravel()
    n = f.size
    if n == 0:
        return f, np.inf

    target = np.sqrt(2.0 * n)
    f = normalize_target(f, target)

    best = f.copy()
    best_m = conv_max_sequence(f)
    no_improve = 0

    while time.monotonic() < deadline:
        r = rng.random()

        if r < 0.40:
            sigma = 0.04 + 0.12 * rng.random()
            cand = f * np.exp(rng.normal(0.0, sigma, n))

        elif r < 0.70:
            cand = f.copy()
            lo_w = max(1, n // 64)
            hi_w = max(lo_w + 1, n // 8 + 1)
            width = int(rng.integers(lo_w, hi_w + 1))
            width = min(width, n)
            start = int(rng.integers(0, max(1, n - width + 1)))
            scale = np.exp(rng.normal(0.0, 0.18))
            cand[start:start + width] *= scale

        else:
            maxf = max(1, n // 2)
            freq = int(rng.integers(1, maxf + 1))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            amp = 0.02 + 0.10 * rng.random()
            cand = f + amp * np.cos(2.0 * np.pi * freq * np.arange(n, dtype=np.float64) / float(n) + phase)

        cand = normalize_target(cand, target)
        m = conv_max_sequence(cand)

        if m < best_m - 1e-12:
            best_m = m
            best = cand.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                break

    score = objective_sequence(best)
    return best, score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Entry point. Searches within budget_s seconds and returns the best sequence
    found as a Python list of non-negative floats.
    """
    seed = int(seed) if seed is not None else 42
    try:
        time_budget = float(budget_s) if budget_s is not None else 2400.0
    except Exception:
        time_budget = 2400.0

    if not np.isfinite(time_budget) or time_budget <= 0.0:
        time_budget = 2400.0

    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    hard_deadline = t_start + max(1.0, time_budget - 2.0)

    B = max(10.0, time_budget - 5.0)
    final_reserve = min(0.22 * B, max(5.0, 0.06 * B))
    search_deadline = t_start + max(1.0, B - final_reserve - 5.0)

    # Safe fallback: uniform sequence has score 2.
    n_fallback = 512
    target_fb = np.sqrt(2.0 * n_fallback)
    best_arr = np.full(n_fallback, target_fb / float(n_fallback), dtype=np.float64)
    best_score = 2.0
    try:
        fb_score = objective_sequence(best_arr)
        if np.isfinite(fb_score):
            best_score = float(fb_score)
    except Exception:
        best_score = 2.0

    # Task list: (n, method, weight, initial_count).
    # Small LP runs first to secure a reliable improvement, then larger/smooth
    # optimizations are given the rest of the budget.
    tasks = [
        (256, "lp_full", 9, 4),
        (512, "lp_full", 22, 6),
        (1024, "lp_full", 16, 4),
        (2048, "lp_active", 13, 2),
        (1024, "mirror", 8, 3),
        (4096, "mirror", 18, 4),
        (8192, "mirror", 14, 3),
    ]

    total_w = float(sum(t[2] for t in tasks))

    for n, method, w, icount in tasks:
        now = time.monotonic()
        if now >= search_deadline - 5.0:
            break

        remaining = max(1.0, float(search_deadline - now))
        phase_time = max(5.0, (float(w) / total_w) * remaining)
        if phase_time > remaining - 2.0:
            phase_time = max(1.0, remaining - 2.0)

        phase_end = now + phase_time

        inits = make_initials(int(n), rng, int(icount))
        if not inits:
            continue

        for ii, f0 in enumerate(inits):
            now = time.monotonic()
            if now >= phase_end - 1.0:
                break

            rem = max(1, len(inits) - ii)
            sub_time = max(2.0, (float(phase_end) - now) / float(rem))
            attempt_deadline = min(phase_end - 0.5, now + sub_time)

            if attempt_deadline <= now:
                attempt_deadline = now + max(0.5, phase_end - now)

            if method == "lp_full":
                cand, score = optimize_lp_full(f0, attempt_deadline, rng)
            elif method == "lp_active":
                cand, score = optimize_lp_active(f0, attempt_deadline, rng, max_rows=256)
            elif method == "mirror":
                cand, score = optimize_mirror(f0, attempt_deadline, rng)
            else:
                cand, score = optimize_mirror(f0, attempt_deadline, rng)

            if cand is not None and score < best_score:
                best_arr = np.asarray(cand, dtype=np.float64).copy()
                best_score = float(score)
                print(f"improved: n={int(cand.size)}, method={method}, score={best_score:.8f}")

    # Final polishing.
    now = time.monotonic()
    if now < hard_deadline - 5.0:
        final_deadline = min(hard_deadline - 1.0, now + max(10.0, final_reserve))
        n_best = int(best_arr.size)

        if n_best <= 1024 and linprog is not None:
            cand, score = optimize_lp_full(best_arr, final_deadline, rng)
        elif n_best <= 4096 and linprog is not None:
            cand, score = optimize_lp_active(best_arr, final_deadline, rng, max_rows=256)
        else:
            cand, score = optimize_mirror(best_arr, final_deadline, rng)

        if cand is not None and score < best_score:
            best_arr = np.asarray(cand, dtype=np.float64).copy()
            best_score = float(score)
            print(f"final polish improved: n={int(cand.size)}, score={best_score:.8f}")

        now = time.monotonic()
        if now < hard_deadline - 2.0:
            local_deadline = min(hard_deadline - 1.0, now + max(2.0, final_reserve * 0.35))
            cand, score = local_search(best_arr, local_deadline, rng)
            if cand is not None and score < best_score:
                best_arr = np.asarray(cand, dtype=np.float64).copy()
                best_score = float(score)
                print(f"local search improved: n={int(cand.size)}, score={best_score:.8f}")

    # Final sanitization and exact evaluation.
    best_arr = np.asarray(best_arr, dtype=np.float64).ravel()
    best_arr = np.where(np.isfinite(best_arr), best_arr, 0.0)
    best_arr = np.clip(best_arr, 0.0, 1000.0)

    out = [float(x) for x in best_arr]

    try:
        exact = float(evaluate_sequence(out))
    except Exception:
        exact = best_score

    if not np.isfinite(exact):
        out = [1.0] * 256
        try:
            exact = float(evaluate_sequence(out))
        except Exception:
            exact = 2.0

    print(f"returned: length={len(out)}, exact_score={exact:.8f}")
    return out
