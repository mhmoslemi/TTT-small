def water_filling(a, M, max_sweeps=100):
    """Find b that maximizes sum(b) subject to conv(a,b) <= M, b >= 0."""
    n = len(a)
    b = a.copy()  # start from a (feasible)
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            # Find the maximum we can increase b[j]
            # Constraint: for all k, sum_i a[i]*b[k-i] <= M
            # After increasing b[j] by delta: for all k with a[k-j] > 0,
            #   current_conv[k] + delta * a[k-j] <= M
            # So delta <= (M - current_conv[k]) / a[k-j] for all k with a[k-j] > 0
            
            current_conv = np.convolve(a, b)
            # For position j, the affected conv positions are k = j, j+1, ..., j+n-1
            # (where a[k-j] > 0, i.e., 0 <= k-j < n, i.e., j <= k < j+n)
            
            max_delta = np.inf

                if 0 <= i < n and a[i] > 1e-10:

                b[j] += max_delta
                improved = True
        
        if not improved:
            break

def build_conv_matrix(a):

```python
def build_conv_matrix_fast(a):
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n
    # For each j, the non-zero entries are at positions k = j, j+1, ..., j+n-1
    # with values a[0], a[1], ..., a[n-1]
    
    # This is a banded matrix with lower bandwidth n-1 and upper bandwidth n-1
    # (actually it's a "convolution matrix" or "Toeplitz-like" matrix)
    
    # Use scipy.sparse.diags or build from COO
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        k_start = j
        k_end = j + n  # exclusive
        # A[k, j] = a[k-j] for k in [j, j+n-1]
        # But k must be in [0, 2n-2]
        k_lo = max(0, k_start)
        k_hi = min(2*n-1, k_end)  # exclusive
        if k_lo < k_hi:
            ks = np.arange(k_lo, k_hi)
            rows.extend(ks)
            cols.extend([j] * (k_hi - k_lo))
            vals.extend(a[ks - j])
    
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
```

from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def compute_score_fast(a):
    """Compute the objective function quickly without calling evaluate_sequence."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)

def generate_shape_polynomial(coeffs, n):
    """Generate a sequence from polynomial coefficients on [0,1]."""
    x = np.linspace(0, 1, n, endpoint=False) + 0.5/n  # midpoint sampling
    a = np.zeros(n)
    for k, c in enumerate(coeffs):
        a += c * x**k
    return np.clip(a, 0, 1000)

def shape_score(coeffs, n):
    """Score of the polynomial shape."""
    a = generate_shape_polynomial(coeffs, n)
    s = np.sum(a)
    if s < 0.01:
        return 1e10
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)

def optimize_polynomial(K, n, max_iter=1000):
    """Optimize K+1 polynomial coefficients to minimize the score."""
    def objective(coeffs):
        return shape_score(coeffs, n)
    
    best = None
    best_val = np.inf
    
    # Try multiple random starting points
    rng = np.random.default_rng(42)
    for trial in range(20):
        if trial == 0:
            x0 = [1.0] + [0.0] * K
        else:

                      options={'maxiter': max_iter, 'xatol': 1e-8, 'fatol': 1e-10})
        if res.fun < best_val:
            best_val = res.fun
            best = res.x
    
    return best, best_val

def lp_iteration(a, max_time=30):
    """One LP iteration: find b maximizing sum(b) s.t. conv(a,b) <= max(conv(a,a)), b>=0."""
    n = len(a)
    M = np.max(np.convolve(a, a))
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_lo = max(0, j)
        k_hi = min(2*n-1, j+n)
        if k_lo < k_hi:
            ks = np.arange(k_lo, k_hi)
            rows.append(ks)
            cols.append(np.full(k_hi-k_lo, j))
            vals.append(a[ks-j])
    
    if not rows:
        return None
    
    A = csr_matrix((np.concatenate(vals), (np.concatenate(rows), np.concatenate(cols))),
                   shape=(2*n-1, n))
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                     method='highs', options={'time_limit': max_time})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    3. Polishing with local random perturbations.

        nonlocal best_score, best_seq
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New best: score={score:.6f}, n={len(seq)}")

                a = np.clip(a, 0, 1000)
                score = evaluate_sequence(a.tolist())
                update_best(a, score)
    
    # Phase 2: LP-based refinement

            if b is None:

            sb = np.sum(b)

            a_new = (1-t)*a + t*(b/sb*np.sum(a))
            a_new = np.clip(a_new, 0, 1000)
            score_new = evaluate_sequence(a_new.tolist())
            if np.isfinite(score_new) and score_new < best_score:
                update_best(a_new, score_new)
                a = a_new
            else:
                # Try without relaxation
                a_new2 = np.clip(b/sb*np.sum(a), 0, 1000)
                score_new2 = evaluate_sequence(a_new2.tolist())
                if np.isfinite(score_new2) and score_new2 < best_score:
                    update_best(a_new2, score_new2)
                    a = a_new2
                else:
                    break
    
    # Phase 3: Local search polishing
    print("Phase 3: Local search")
    if best_seq is not None:
        a = best_seq.copy()
        current_score = best_score
        rng = np.random.default_rng(99)
        n = len(a)
        while time.monotonic() < deadline - 5:

        return [float(x) for x in best_seq]
    
    # Absolute fallback
    print("Using fallback uniform sequence")
    return [1.0] * 256

Actually, let me try a simpler approach: **iterative proportional fitting (IPF)** or **alternating projection**.

Algorithm:
1. Start with a (e.g., two-level).
2. Compute M = max(conv(a,a)).
3. For each k where conv(a,a)[k] = M (peak positions), reduce the "contribution" to that peak by adjusting a.
4. Repeat.

This is like a "subgradient" method for the max.

For a specific peak at position k: conv(a,a)[k] = sum_i a[i]*a[k-i]. To reduce this, I can:
- Decrease a[i] for i where a[i]*a[k-i] is large.
- Or increase a[j] for j not involved in the peak (to increase the sum without increasing the peak).

A simple heuristic:
- Find the peak position k*.
- For each i, the "contribution" to the peak is a[i]*a[k*-i].
- Reduce a[i] proportionally to its contribution: a[i] *= (1 - eta * a[i]*a[k*-i] / conv(a,a)[k*])
- This reduces the peak but might increase other convolution values.

This is similar to the "power method" or "mirror descent."
