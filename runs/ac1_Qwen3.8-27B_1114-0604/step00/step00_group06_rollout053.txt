def generate_seed(n, style, rng):
    x = np.zeros(n)
    t = np.linspace(0.0,1.0,n)
    if style==0: x[:] = 1.0
    elif style==1: lam = rng.uniform(0.5,3.0); x = np.exp(-lam*t)
    elif style==2: # best geometric
       lam = 1.2 + rng.normal(0,0.3); x=np.exp(-lam*t)
    elif style==3: # truncated gaussian off-center
       center=rng.uniform(0.05,0.8); sigma=rng.uniform(0.05,0.35); x=np.exp(-((t-center)**2)/(2*sigma*sigma))
    elif style==4: # sine modulated exp
       lam=rng.uniform(0.5,2.0); A=rng.uniform(0.2,1.0); f=rng.integers(1,4); phi=rng.uniform(0,2*np.pi); x=np.exp(-lam*t)*(1+A*np.sin(2*np.pi*f*t+phi))
    elif style==5: # two level
       s=rng.uniform(0.2,0.8); a1=rng.uniform(0.2,3.0); a2=rng.uniform(0.1,2.0); x[t<s]=a1; x[t>=s]=a2
    else: # random smooth
       x = 1.0
       for m in range(1,5):
           amp=rng.uniform(0.0,0.9)/(m+1)
           phase=rng.uniform(0,2*np.pi)
           x += amp*np.sin(2*np.pi*m*t+phase)
       x *= np.exp(-rng.uniform(0.0,2.0)*t)
    x=np.clip(x,0,None)
    if x.sum()<=0: x=np.ones(n)
    return x/x.sum()

def _top_weights(c, P):
   L=c.size
   if L <=0: return np.zeros(0)
   P = int(max(1, min(P, L)))
   if P == L:
      return np.full(L, 1.0/L, dtype=np.float64)
   idx = np.argpartition(c, -P)[-P:]
   q = np.zeros(L, dtype=np.float64)
   q[idx] = 1.0 / float(P)
   return q
```

```python
"""
Search for non-negative step-function heights minimizing:
    2 * n * max(conv(a, a)) / sum(a)^2

The algorithm:
1. Generates diverse smooth one-dimensional seeds:
   - exponential decays,
   - Gaussian-like bumps,
   - sinusoid-modulated envelopes,
   - two-level blocks,
   - random smooth/power/exponential models.
2. Scores them with an FFT-linear-convolution evaluator.
3. Refines promising seeds using a peak-driven multiplicative update:
   the convolution coefficients are computed by FFT, the largest coefficients
   are used as a penalty gradient, and weights are multiplicatively reduced
   where they contribute most to the current peaks.
4. Re-starts / mutates to escape local minima, upsamples good links to larger
   lengths, and returns the best sequence according to the exact evaluator.
"""

import time
import numpy as np

# ---------------------------------------------------------------------------
# Small generic helpers
# ---------------------------------------------------------------------------

def _candidate_key(item):
    return item[0]


def _fft_len(length):
    m = max(1, int(length))
    return 1 << max(0, m - 1).bit_length()


def _normalize_array(x):
    x = np.atleast_1d(np.asarray(x, dtype=np.float64)).copy()
    if x.size == 0:
        return np.ones(1, dtype=np.float64)
    x[~np.isfinite(x)] = 0.0
    np.maximum(x, 0.0, out=x)
    s = float(np.sum(x))
    if not np.isfinite(s) or s <= 0.0:
        u = np.ones_like(x, dtype=np.float64)
        return u / float(u.size)
    x /= s
    return x


def _conv_stats(x, nfft):
    """
    For normalized non-negative x, return (max coefficient, coefficients, FFT).
    The FFT length nfft must be large enough for the requested convolution.
    """
    x = np.asarray(x, dtype=np.float64)
    n = int(x.size)
    if n <= 0:
        return float("inf"), None, None
    L = 2 * n - 1
    X = np.fft.rfft(x, int(nfft))
    c = np.fft.irfft(X * X, int(nfft))[:L]
    c = np.where(c > 0.0, c, 0.0)
    M = float(np.max(c)) if c.size else 0.0
    return M, c, X


def _fast_score(x):
    """
    Fast FFT-based score for normalized / non-normalized candidates.
    This is intended for local search only; the exact evaluator is used
    for final selection.
    """
    xn = _normalize_array(x)
    n = int(xn.size)
    if n <= 0:
        return float("inf")
    L = 2 * n - 1
    nfft = _fft_len(L)
    M, _, _ = _conv_stats(xn, nfft)
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M)


def _penalty_fft(x, q, nfft, X):
    """
    Given convolution-coefficient weights q_k (summing to 1), compute
    d_i = sum_k q_k x_{k-i}
    which is the mixed derivative direction used by the multiplicative update.
    """
    n = int(x.size)
    L = 2 * n - 1
    q = np.asarray(q, dtype=np.float64)
    if q.size != L or not np.any(q > 0):
        return np.zeros(n, dtype=np.float64)
    qrev = q[::-1]
    Q = np.fft.rfft(qrev, int(nfft))
    conv = np.fft.irfft(X * Q, int(nfft))
    idx = L - 1 - np.arange(n, dtype=np.int64)
    d = conv[idx]
    d = np.where(d > 0.0, d, 0.0)
    return d


def _top_weights(c, P):
    L = int(c.size)
    if L <= 0:
        return np.zeros(0, dtype=np.float64)
    P = int(max(1, min(P, L)))
    if P == L:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    idx = np.argpartition(c, -P)[-P:]
    q = np.zeros(L, dtype=np.float64)
    q[idx] = 1.0 / float(P)
    return q

        p2 = float(rng.uniform(0.0, 2.0 * np.pi))

        x = (

        a2 = float(rng.uniform(0.2, 2.5))
        a3 = float(rng.uniform(0.2, 2.5))
        x[t < s1] = a1
        x[(t >= s1) & (t < s2)] = a2
        x[t >= s2] = a3

    x = np.asarray(x, dtype=np.float64)
    x[~np.isfinite(x)] = 0.0
    np.maximum(x, 0.0, out=x)

    if float(np.sum(x)) <= 1e-300 or not np.all(x < 1e300):

def _resize_sequence(x, n2, rng=None):
    n2 = int(max(1, n2))
    x = np.asarray(x, dtype=np.float64)
    x = np.atleast_1d(x)
    x[~np.isfinite(x)] = 0.0
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if x.size == 0 or s <= 1e-300 or not np.isfinite(s):
        return np.ones(n2, dtype=np.float64) / float(n2)
    x = x / s

    if int(x.size) == n2:
        out = x.astype(np.float64, copy=True)
    else:
        xs = np.linspace(0.0, 1.0, x.size, dtype=np.float64)
        xt = np.linspace(0.0, 1.0, n2, dtype=np.float64)
        out = np.interp(xt, xs, x.astype(np.float64, copy=False))

    if rng is not None:
        out = out + rng.normal(0.0, 1e-6, n2)

    return _normalize_array(out)


def _mutate_sequence(x, rng, blend=0.25):
    x = np.asarray(x, dtype=np.float64)
    x = np.atleast_1d(x)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    n = int(x.size)
    if n == 0:
        return np.ones(1, dtype=np.float64)
    x = _normalize_array(x)

    r = float(rng.random())
    if r < 0.35:
        y = _generate_seed(n, rng, int(rng.integers(0, 15)))
        z = (1.0 - blend) * x + blend * y
    elif r < 0.65:
        zlog = np.log(np.maximum(x, 1e-300))
        sigma = float(rng.uniform(0.15, 0.55))
        zlog = zlog + rng.normal(0.0, sigma, n)
        zlog = zlog - float(np.mean(zlog))
        zlog = zlog - float(np.max(zlog))
        z = np.exp(zlog)
    elif r < 0.85:
        z = x.copy()
        for _ in range(int(rng.integers(1, 5))):
            if n < 3:
                break
            start = int(rng.integers(0, n - 1))
            end = int(rng.integers(start + 1, n + 1))
            z[start:end] *= float(np.exp(rng.normal(0.0, 0.25)))
    else:
        # Gentle sharpening / spreading, often useful near convergence.
        gamma = float(rng.uniform(-0.25, 0.25))
        z = x ** (1.0 + gamma)
        s = float(np.sum(z))
        if s <= 0.0 or not np.isfinite(s):
            z = x.copy()
    return _normalize_array(z)


# ---------------------------------------------------------------------------
# Local refinement
# ---------------------------------------------------------------------------

def _refine_sequence(x, budget_s, rng=None):
    """
    Peak-driven multiplicative local search.
    Returns (best_sequence, best_max_convolution_coeff_for_sum_1).
    """
    start = time.monotonic()
    budget_s = max(0.0, float(budget_s))
    x = _normalize_array(x)
    n = int(x.size)
    if n <= 0 or budget_s <= 0.0:
        score = _fast_score(x)
        M = score / (2.0 * max(1, int(_normalize_array(x).size))) if np.isfinite(score) else float("inf")
        return x, M

    L = 2 * n - 1
    # For the penalty convolution x * reversed(q), we need length at least 3n-2.
    nfft = _fft_len(max(L, 3 * n - 2))

    M, c, X = _conv_stats(x, nfft)
    best_x = x.copy()
    best_M = M
    if not np.isfinite(best_M):
        return best_x, best_M

    P0 = max(8, min(128, L // 64))
    P = P0
    lr = 0.12
    floor_rel = 1e-9
    fails = 0
    it = 0
    deadline = start + budget_s

    while time.monotonic() < deadline:
        it += 1
        if not np.isfinite(M) or M <= 0.0:
            break

        q = _top_weights(c, P)
        d = _penalty_fft(x, q, nfft, X)
        dmax = float(np.max(d)) if d.size else 0.0
        if not np.isfinite(dmax) or dmax <= 1e-300:
            break

        pen = d / dmax

        # Main proposed update.
        x2 = x * np.exp(-lr * pen)
        x2 = (1.0 - floor_rel) * x2 + (floor_rel / float(n))
        x2 = _normalize_array(x2)
        M2, c2, X2 = _conv_stats(x2, nfft)

        if np.isfinite(M2) and M2 < M:
            x, c, X, M = x2, c2, X2, M2
            if M < best_M:
                best_M = M
                best_x = x.copy()
            lr = min(0.35, lr * 1.04)
            fails = 0
        else:
            accepted = False
            for fac in (0.5, 0.25):
                xh = x * np.exp(-lr * fac * pen)
                xh = (1.0 - floor_rel) * xh + (floor_rel / float(n))
                xh = _normalize_array(xh)
                Mh, ch, Xh = _conv_stats(xh, nfft)
                if np.isfinite(Mh) and Mh < M:
                    x, c, X, M = xh, ch, Xh, Mh
                    lr = min(0.35, lr * fac)
                    if M < best_M:
                        best_M = M
                        best_x = x.copy()
                    accepted = True
                    fails = 0
                    break

            if not accepted:
                lr = max(0.01, lr * 0.85)
                fails += 1
                if fails > 20:
                    P = min(L, P * 2)
                if fails > 70 and rng is not None:
                    base = best_x if best_x.size == n else x
                    x = _mutate_sequence(base, rng, 0.25)
                    M, c, X = _conv_stats(x, nfft)
                    if np.isfinite(M) and M < best_M:
                        best_M = M
                        best_x = x.copy()
                    lr = 0.12
                    P = P0
                    fails = 0

        # Periodic gentle escape move.
        if it % 2500 == 0 and rng is not None:
            xm = _mutate_sequence(x, rng, 0.1)
            Mm, cm, Xm = _conv_stats(xm, nfft)
            if np.isfinite(Mm):
                if Mm < M * 1.25 or Mm < best_M * 1.1:
                    x, c, X, M = xm, cm, Xm, Mm
                    if M < best_M:
                        best_M = M
                        best_x = x.copy()

    return best_x, best_M


# ---------------------------------------------------------------------------
# Candidate management / finalization
# ---------------------------------------------------------------------------

def _add_candidate(cands, seq, limit=50):
    if seq is None:
        return
    x = _normalize_array(seq)
    if x.size == 0:
        return
    score = _fast_score(x)
    if not np.isfinite(score):
        return
    cands.append((score, x.copy()))
    if len(cands) > limit:
        cands.sort(key=_candidate_key)
        del cands[limit:]


def _get_top(cands, k):
    if not cands:
        return []
    tmp = list(cands)
    tmp.sort(key=_candidate_key)
    k = max(1, int(k))
    return [x.copy() for _, x in tmp[:k]]


def _finalize_sequence(x):
    x = _normalize_array(x)
    if x.size == 0:
        return [1.0]
    maxv = float(np.max(x))
    if maxv <= 0.0:
        return [1.0]
    x = np.where(x > 1e-12 * maxv, x, 0.0)
    nz = np.nonzero(x > 0.0)[0]
    if nz.size == 0:
        return [1.0]
    a = int(nz[0])
    b = int(nz[-1]) + 1
    x = x[a:b]
    return [float(v) for v in x]


def _exact_score(seq_list):
    try:
        v = float(evaluate_sequence(seq_list))
        return v if np.isfinite(v) else float("inf")
    except Exception:
        return float("inf")


def _fallback_sequence():
    n = 256
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    x = np.exp(-1.25 * t)
    x = _normalize_array(x)
    return _finalize_sequence(x)


# ---------------------------------------------------------------------------
# Main entry point
# ---------------------------------------------------------------------------

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Runs the search and returns a list of non-negative floats.
    """
    start = time.monotonic()
    total_budget = max(0.0, float(budget_s))
    safe_deadline = start + max(0.0, total_budget - 30.0)
    main_end = safe_deadline - 90.0
    if main_end < start:
        main_end = safe_deadline

    rng = np.random.default_rng(seed)
    cands = []

    # If an external starting construction is available, use it as one seed.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.atleast_1d(parent)
        parent = np.where(np.isfinite(parent), parent, 0.0)
        parent = np.maximum(parent, 0.0)
        if parent.size > 0:
            _add_candidate(cands, parent, 50)
    except Exception:
        pass

    # Phase 1: broad seed generation and short refinements.
    phase1_budget = min(160.0, max(20.0, 0.07 * total_budget))
    phase1_end = min(main_end, start + phase1_budget)

    if not cands:
        for n in (128, 256, 512):
            if time.monotonic() > phase1_end:
                break
            local_deadline = min(phase1_end, time.monotonic() + max(5.0, phase1_budget / 3.0))
            while time.monotonic() < local_deadline:
                nuse = int(rng.choice(np.array([96, 128, 192, 256, 384, 512, 768])))
                style = int(rng.integers(0, 15))
                x = _generate_seed(nuse, rng, style)
                _add_candidate(cands, x, 50)

    top_phase1 = _get_top(cands, 5)
    if top_phase1 and time.monotonic() < phase1_end:
        frac = (0.5, 0.25, 0.15, 0.07, 0.03)
        for i, seq in enumerate(top_phase1):
            if time.monotonic() >= phase1_end:
                break
            share = max(3.0, phase1_budget * float(frac[i] if i < len(frac) else 0.0))
            seq = _resize_sequence(seq, 512 if seq.size > 1024 or seq.size < 256 else seq.size, rng)
            res_x, _ = _refine_sequence(seq, min(share, phase1_end - time.monotonic()), rng)
            _add_candidate(cands, res_x, 60)
        top_score = _get_top(cands, 1)
        if top_score:
            print("phase1 best fast score: %.8f" % top_score[0][0], flush=True)

    # Phase 2: repeated medium-size refinement, with occasional resampling.
    last_best = float("inf")
    if cands:
        last_best = _get_top(cands, 1)[0][0]

    no_improve = 0
    chunk_count = 0

    while time.monotonic() < main_end:
        if not cands:
            x = _generate_seed(512, rng, int(rng.integers(0, 15)))
            _add_candidate(cands, x, 60)
            if not cands:
                break

        top = _get_top(cands, 5)
        if not top:
            break

        # Choose best normally, but diversify after stagnation.
        rank = 1
        if no_improve >= 2:
            rank = 2
        if no_improve >= 5:
            rank = 3
        if rank > len(top):
            rank = 1
        seq = top[rank - 1]

        # Target length: moderate sizes are a good trade-off for discrete search.
        cur_n = int(seq.size)
        best_fast = top[0][0]
        remaining = max(0.0, safe_deadline - time.monotonic())

        if cur_n > 2048:
            target_n = 1024
        elif cur_n < 1024:
            target_n = 1024
        elif cur_n < 2048 and best_fast < 1.60 and remaining > 500.0:
            target_n = 2048
        else:
            target_n = cur_n

        if target_n != cur_n:
            seq = _resize_sequence(seq, target_n, rng)

        # Mild mutation after stagnation to avoid re-polishing same local basin.
        if no_improve >= 1 and float(rng.random()) < 0.55:
            seq = _mutate_sequence(seq, rng, 0.12)

        chunk_budget = min(150.0, max(0.0, main_end - time.monotonic()))
        if chunk_budget < 8.0:
            break

        res_x, res_M = _refine_sequence(seq, chunk_budget, rng)
        if res_x is not None and res_x.size > 0 and np.isfinite(res_M):
            _add_candidate(cands, res_x, 70)

        # Occasionally inject fresh parameterized seeds to maintain diversity.
        chunk_count += 1
        if chunk_count % 4 == 0 and time.monotonic() < main_end - 120.0:
            style = int(rng.integers(0, 15))
            fresh = _generate_seed(target_n if target_n > 0 else 512, rng, style)
            _add_candidate(cands, fresh, 80)

        cur_best = _get_top(cands, 1)
        if not cur_best:
            break
        cb = cur_best[0][0]
        if np.isfinite(cb) and cb < last_best - 1e-9:
            last_best = cb
            no_improve = 0
        else:
            no_improve += 1

        if chunk_count % 3 == 0:
            best_seq = _get_top(cands, 1)
            if best_seq and time.monotonic() < main_end - 600.0:
                b0 = best_seq[0]
                for target in (1024, 2048):
                    if int(b0.size) < target:
                        up = _resize_sequence(b0, target, rng)
                        _add_candidate(cands, up, 80)
                        break

    # Phase 3: exact final evaluation of a small set of strong candidates.
    final_candidates = _get_top(cands, 20)
    # Include fallback to guarantee a valid return.
    fallback = _fallback_sequence()

    # Also explicitly consider the parent one more time if present.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.atleast_1d(parent)
        parent = np.where(np.isfinite(parent), parent, 0.0)
        parent = np.maximum(parent, 0.0)
        if parent.size > 0:
            finalized_parent = _finalize_sequence(parent)
            final_candidates.append(finalized_parent)
    except Exception:
        pass

    best_out = None
    best_exact = float("inf")
    evaluated_large = 0

    for item in final_candidates:
        if time.monotonic() > safe_deadline - 3.0:
            break
        try:
            out = _finalize_sequence(item)
            nout = len(out)
            if nout > 7000 and evaluated_large >= 1:
                continue
            if nout > 7000:
                evaluated_large += 1
            ex = _exact_score(out)
            if np.isfinite(ex) and ex < best_exact:
                best_exact = ex
                best_out = out
        except Exception:
            pass

    if best_out is None:
        best_out = fallback

    # Final safety: ensure array/list type and finite non-negative floats.
    try:
        final = [float(v) for v in best_out]
        final = [v if np.isfinite(v) and v >= 0.0 else 0.0 for v in final]
        if not final or float(np.sum(final)) <= 0.0:
            final = _fallback_sequence()
        return final
    except Exception:
        return _fallback_sequence()
