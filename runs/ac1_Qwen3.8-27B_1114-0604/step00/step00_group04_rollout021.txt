def compute_conv_max(a):
    """Compute max of conv(a, a) using FFT."""
    n = len(a)

    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    C = ifft(A * A)
    return np.real(C[:size])

def fast_conv_cross(a, b):
    """Compute linear convolution of a and b using FFT."""
    n1, n2 = len(a), len(b)

    A = fft(a, fft_size)
