def sliding_window_max(arr, window_size):
    """Compute sliding window maximum in O(n)."""
    n = len(arr)
    result = np.empty(n - window_size + 1)
    dq = deque()
    for i in range(n):
        while dq and arr[dq[-1]] <= arr[i]:
            dq.pop()
        dq.append(i)
        if dq[0] <= i - window_size:
            dq.popleft()
        if i >= window_size - 1:
            result[i - window_size + 1] = arr[dq[0]]
    return result

    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)

def sliding_window_max(arr, w):
    """O(n) sliding window maximum using deque."""
    n = len(arr)
    out = np.empty(n - w + 1, dtype=np.float64)

The LP is: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.

The dual variables y are non-zero only at constraints that are tight, i.e., where (f*g)[k] = M.

- g >= 0, sum(g) is maximized
- (f*g)[k] <= M for all k
- (f*g)[k] = M for k in supp(y)
- sum_k y[k] * f[k-j] = 1 for j in supp(g)
