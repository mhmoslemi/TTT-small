def fft_convolve(a):
    """Compute linear self-convolution of a using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

def compute_score(a, b=None):
    """Compute the evaluation score. If b is None, compute convolution."""
    n = len(a)
    if b is None:
        b = fft_convolve(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    M = np.max(b)
    return 2.0 * n * M / (S * S)

def subgradient_direction(a, b):
    """
    Compute the descent direction for minimizing max(b) subject to sum(a)=const.
    Returns the direction d such that a + eta*d decreases the objective.
    """
    n = len(a)
    M = np.max(b)
    S = np.sum(a)
    if S < 1e-10:
        return None
    
    # Find active set: indices where b is close to max
    tol = max(1e-12, 1e-10 * M)
    active = np.where(b >= M - tol)[0]
    
    # Compute grad_max: (1/|K*|) * sum_{k in K*} a_{k-i} * 1[0 <= k-i < n]
    grad_max = np.zeros(n)
    for k in active:
        # a_{k-i} is valid for i in [max(0, k-n+1), min(n-1, k)]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # grad_max[i] += a[k-i] for i in [lo, hi]
            # a[k-i] for i=lo is a[k-lo], for i=hi is a[k-hi]
            # So the slice is a[k-hi : k-lo+1] reversed
            grad_max[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    
    grad_max /= len(active)
    
    # Descent direction (negative subgradient of F, up to positive scale):
    # d = M/S - grad_max
    d = M / S - grad_max
    return d

    Returns g (rescaled to have same sum as a) or None if LP fails.
    """
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    n = len(a)
    
    # Build constraint matrix A (2n-1) x n where A[k,j] = a[k-j] if 0 <= k-j < n
    # This is a banded matrix
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # A[k, j] = a[k-j] for j in [lo, hi]
            js = np.arange(lo, hi + 1)
            vs = a[k - js]  # a[k-lo], a[k-lo+1], ..., a[k-hi]
            # Wait: for j in [lo, hi], A[k,j] = a[k-j]
            # So for j=lo: a[k-lo], j=lo+1: a[k-lo-1], ..., j=hi: a[k-hi]
            # Actually: A[k,j] = a[k-j]. For j=lo: a[k-lo]. For j=hi: a[k-hi].
            # The values are a[k-lo], a[k-lo-1], ..., a[k-hi]
            # Which is a[k-hi : k-lo+1] reversed? Let me check.

            # So the values are a[k-hi], a[k-hi+1], ..., a[k-lo] = a[k-hi:k-lo+1]
            rows.extend([k] * (hi - lo + 1))
            cols.extend(range(lo, hi + 1))

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)

    bounds = [(0, None)] * n

        Sg = np.sum(g)

    return None
