        if g is None:
            break

        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            break

        # Try the update
        new_f = (1 - t_step) * f + t_step * g
        np.clip(new_f, 0, None, out=new_f)
        ns = float(new_f.sum())
        if ns < 1e-10:
            break
        new_f /= ns

        score = float(evaluate_sequence(new_f.tolist()))
        if score < best_local_score:
            best_local_score = score
            f = new_f.copy()
            if score < best_score:
                best_score = score
                best_seq = new_f.copy()
                print(f"  [n={n}] LP iter {lp_count}: score={best_score:.6f}")
            stall = 0
            t_step = min(0.5, t_step * 1.02)
        else:
            stall += 1
            t_step *= 0.7
            if t_step < 0.02:
                # Perturb to escape
                f = block_perturb(f, rng, n, sigma=0.3)
                t_step = 0.3
                if stall > 30:
                    break

        lp_count += 1
        if lp_count % 10 == 0 and lp_count > 0:
            print(f"  [n={n}] LP iter {lp_count}: best={best_local_score:.6f}, M={M:.8f}")

    # Local polish
    remaining = deadline - time.monotonic()
    if remaining > 15 and lp_count > 0:
        cur_score = best_local_score
        no_imp = 0
        while time.monotonic() < deadline - 5 and no_imp < 200:
            width = int(rng.integers(max(1, n // 80), max(2, n // 8)))
            start = int(rng.integers(0, n - width + 1))
            found = False
            for sigma in [0.05, 0.1, 0.02]:
                factor = float(np.exp(rng.normal(0.0, sigma)))
                cand = f.copy()
                cand[start:start + width] *= factor
                np.clip(cand, 0, None, out=cand)
                sc = cand.sum()
                if sc < 1e-10:
                    continue
                cand /= sc
                s_val = float(evaluate_sequence(cand.tolist()))
                if s_val < cur_score:
                    f = cand
                    cur_score = s_val
                    if s_val < best_score:
                        best_score = s_val
                        best_seq = cand.copy()
                        print(f"  [n={n}] Polish: score={best_score:.6f}")
                    no_imp = 0
                    found = True
                    break
            if not found:
                no_imp += 1

    return best_score, best_seq


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for optimal step function coefficients using LP-based refinement
    with multi-restart and multi-length strategy.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0, budget_s)

    # Start from the given sequence
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000.0)
    n0 = parent.size
    parent = parent / parent.sum()

    best_score = float(evaluate_sequence(parent.tolist()))
    best_seq = parent.copy()
    print(f"Start: n={n0}, score={best_score:.6f}")

    # Check time budget
    remaining = deadline - time.monotonic()
    if remaining < 60:
        return [float(x) for x in best_seq]

    # Phase 1: Try the original length with LP refinement
    phase1_deadline = time.monotonic() + remaining * 0.5
    best_score, best_seq = try_length(n0, parent, phase1_deadline, rng, best_score, best_seq)
    print(f"After phase 1 (n={n0}): score={best_score:.6f}")

    # Phase 2: Try a shorter length (faster LP, more iterations)
    if time.monotonic() < deadline - 60:
        n2 = min(512, n0)
        phase2_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.35
        best_score, best_seq = try_length(n2, best_seq, phase2_deadline, rng, best_score, best_seq)
        print(f"After phase 2 (n={n2}): score={best_score:.6f}")

    # Phase 3: Try medium length
    if time.monotonic() < deadline - 40:
        n3 = min(1024, n0)
        if n3 != n0 and n3 != n2:
            phase3_deadline = deadline - 10
            best_score, best_seq = try_length(n3, best_seq, phase3_deadline, rng, best_score, best_seq)
            print(f"After phase 3 (n={n3}): score={best_score:.6f}")

    # Final local polish on the best sequence
    remaining = deadline - time.monotonic()
    if remaining > 10:
        f = best_seq.copy()
        n = len(f)
        cur = best_score
        no_imp = 0
        while time.monotonic() < deadline - 3 and no_imp < 500:
            width = int(rng.integers(max(1, n // 100), max(2, n // 10)))
            start = int(rng.integers(0, n - width + 1))
            found = False
            for sigma in [0.03, 0.07, 0.15, 0.01]:
                factor = float(np.exp(rng.normal(0.0, sigma)))
                cand = f.copy()
                cand[start:start + width] *= factor
                np.clip(cand, 0, None, out=cand)
                sc = cand.sum()
                if sc < 1e-10:
                    continue
                cand /= sc
                s_val = float(evaluate_sequence(cand.tolist()))
                if s_val < cur:
                    f = cand
                    cur = s_val
                    if s_val < best_score:
                        best_score = s_val
                        best_seq = cand.copy()
                        print(f"  Final polish: score={best_score:.6f}")
                    no_imp = 0
                    found = True
                    break
            if not found:
                no_imp += 1

    print(f"FINAL: score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
