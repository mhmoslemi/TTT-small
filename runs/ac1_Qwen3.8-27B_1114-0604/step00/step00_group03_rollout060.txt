"""
Block-soft-max optimizer for minimizing 2n * max(a*a) / (sum(a))^2.

The sequence is parameterized as block-constant values:
    a_i = c_{block(i)} / block_len
with sum(c) = 1, implemented through a softmax.  A smoothed peak objective
is minimized with L-BFGS-B and then polished by full-length multiplicative
gradient steps and a short random local search.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None

BLOCK_COUNT = 256
BLOCK_LEN = 10
SEQ_LEN = BLOCK_COUNT * BLOCK_LEN

RECT_CONV = np.convolve(np.ones(BLOCK_LEN, dtype=np.float64),
                        np.ones(BLOCK_LEN, dtype=np.float64),
                        mode="full")

N_FFT = 1
while N_FFT < max(3 * SEQ_LEN, 2 * SEQ_LEN):
    N_FFT <<= 1

FFT_IDX = np.empty(SEQ_LEN, dtype=np.intp)
FFT_IDX[0] = 0
FFT_IDX[1:] = np.arange(N_FFT - 1, N_FFT - SEQ_LEN, -1, dtype=np.intp)

BETA_STEPS = [
    5.0 * SEQ_LEN,
    20.0 * SEQ_LEN,
    100.0 * SEQ_LEN,
    400.0 * SEQ_LEN,
]

FULL_BETA_STEPS = [
    10.0 * SEQ_LEN,

    200.0 * SEQ_LEN,
    800.0 * SEQ_LEN,
]

FULL_ETAS = [0.03, 0.06, 0.12, 0.25, 0.5]


def stable_softmax(v):
    """Stable softmax returning a non-negative vector of length v."""
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size == 0:
        return np.array([1.0])
    v = np.where(np.isfinite(v), v, 0.0)
    z = v - float(np.max(v))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(v) / float(v.size)
    return e / s


def build_block_b(c):
    """Exact convolution of the block-constant sequence c / BLOCK_LEN."""
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size != BLOCK_COUNT:
        c = np.resize(c, BLOCK_COUNT)
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    s = float(c.sum())
    if s <= 0.0:
        c = np.ones(BLOCK_COUNT, dtype=np.float64) / float(BLOCK_COUNT)
    else:
        c = c / s

    q = np.convolve(c, c, mode="full")
    S = np.zeros((2 * (BLOCK_COUNT - 1) * BLOCK_LEN) + 1, dtype=np.float64)
    S[:q.size:BLOCK_LEN] = q
    b = np.convolve(S, RECT_CONV, mode="full")
    b /= float(BLOCK_LEN * BLOCK_LEN)
    return np.maximum(b, 0.0)


def block_b_and_soft(c, beta):
    """Block convolution and soft-max weights / objective value."""
    b = build_block_b(c)
    bmax = float(np.max(b))
    if not np.isfinite(bmax) or bmax <= 0.0:
        w = np.ones_like(b) / float(b.size)
        return b, bmax, w, bmax

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1.0

    z = np.exp(beta * (b - bmax))
    s = float(z.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
        J = bmax
    else:
        w = z / s
        J = bmax + float(np.log(s)) / beta

    return b, bmax, w, J


def block_objective(u, beta):
    """Smoothed peak objective for L-BFGS."""
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size != BLOCK_COUNT or not np.all(np.isfinite(u)):
        return float("inf")

    c = stable_softmax(u)
    _, _, _, J = block_b_and_soft(c, beta)
    if not np.isfinite(J):
        return float("inf")
    return float(J * SEQ_LEN * SEQ_LEN)


def correlate_weights_a(w, a):
    """
    Cross-correlation slice needed for gradients:
        corr[i] = sum_k w[k] * a[k - i], i = 0..len(a)-1
    with out-of-range a-index treated as zero.
    """
    w = np.asarray(w, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n != SEQ_LEN:
        return None
    x = np.fft.ifft(np.fft.fft(w, N_FFT) * np.conj(np.fft.fft(a, N_FFT))).real
    return x[FFT_IDX]


def block_grad(u, beta):
    """Gradient of smooth block objective with respect to softmax logits u."""
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size != BLOCK_COUNT or not np.all(np.isfinite(u)):
        return np.zeros_like(u)

    c = stable_softmax(u)
    _, _, w, _ = block_b_and_soft(c, beta)

    a = np.repeat(c / float(BLOCK_LEN), BLOCK_LEN)
    corr = correlate_weights_a(w, a)
    if corr is None:
        return np.zeros_like(u)

    grad_a = 2.0 * corr
    G = grad_a.reshape(BLOCK_COUNT, BLOCK_LEN).sum(axis=1)
    meanG = float(np.dot(c, G))
    grad_u = c * (G - meanG)

    scale = float(SEQ_LEN * SEQ_LEN / BLOCK_LEN)
    grad_u *= scale
    gen = np.where(np.isfinite(grad_u), grad_u, 0.0)
    return gen


def run_lbfgs_phase(u, beta):
    """Run one L-BFGS-B phase if SciPy is available."""
    u = np.asarray(u, dtype=np.float64).ravel()
    if minimize is None:
        return u

    try:
        res = minimize(
            block_objective,
            u,
            method="L-BFGS-B",
            jac=block_grad,
            args=(float(beta),),
            options={
                "maxiter": 120,
                "ftol": 1e-14,
                "gtol": 1e-10,
                "maxls": 40,
            },
        )
        xu = np.asarray(res.x, dtype=np.float64).ravel()
        if xu.size == BLOCK_COUNT and np.all(np.isfinite(xu)):
            return np.clip(xu, -50.0, 50.0)
    except Exception:
        pass
    return u


def full_from_blocks(c):
    """Expand normalized block masses c to a full sequence summing to 1."""
    c = np.asarray(c, dtype=np.float64).ravel()
    c = stable_softmax(c)
    return np.repeat(c / float(BLOCK_LEN), BLOCK_LEN)


def block_score(c):
    """Exact score of a block-constant sequence normalized to sum 1."""
    b = build_block_b(c)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return float("inf")
    return float(2.0 * SEQ_LEN * mb)


def fft_score_full(a):
    """Fast FFT score for a full-length nonnegative sequence."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return float("inf")

    fa = np.fft.fft(a, N_FFT)
    b = np.fft.ifft(fa * fa).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return float("inf")
    return float(2.0 * n * mb / (s * s))

def full_b_weights_grad(a, beta):
    """Full-length soft-peak objective gradient components."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n != SEQ_LEN:
        return None, None, 0.0

    fa = np.fft.fft(a, N_FFT)
    b = np.fft.ifft(fa * fa).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    bmax = float(np.max(b))
    if not np.isfinite(bmax) or bmax <= 0.0:
        return None, None, 0.0

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1.0

    z = np.exp(beta * (b - bmax))
    s = float(z.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
    else:
        w = z / s

    corr = correlate_weights_a(w, a)
    if corr is None:
        return None, None, 0.0

    grad = 2.0 * corr
    maxg = float(np.max(grad))
    if not np.isfinite(maxg) or maxg <= 0.0:
        maxg = 0.0
    return b, grad, maxg

    if maxg <= 0.0 or grad.size != a.size:
        return a.copy()

    a2 = a * np.exp(-float(eta) * grad / float(maxg))
    a2 = np.where(np.isfinite(a2), a2, 0.0)
    a2 = np.clip(a2, 1e-300, 1000.0)
    s = float(a2.sum())
    if not np.isfinite(s) or s <= 0.0:
        return a.copy()
    return a2 / s


def full_gradient_polish(a, current_score, deadline):
    """Polish a full sequence with multiplicative soft-peak gradient steps."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    best_score = float(current_score)
    if not np.isfinite(best_score):
        best_score = fft_score_full(a)

    for beta in FULL_BETA_STEPS:
        if time.monotonic() >= deadline:
            break
        for _ in range(300):
            if time.monotonic() >= deadline:
                break
            _, grad, maxg = full_b_weights_grad(a, beta)
            if grad is None or maxg <= 0.0:
                break

            best_cand = a
            best_cand_score = best_score
            for eta in FULL_ETAS:
                if time.monotonic() >= deadline:
                    break
                cand = full_eta_candidate(a, grad, maxg, eta)
                sc = fft_score_full(cand)
                if np.isfinite(sc) and sc < best_cand_score - 1e-15:
                    best_cand = cand
                    best_cand_score = sc

            if best_cand is a:
                break
            a = best_cand
            best_score = best_cand_score

    return a, best_score


def random_local_polish(a, current_score, deadline, seed):
    """Short random multiplicative hill-climb in full length."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    best_score = float(current_score)
    if not np.isfinite(best_score):
        best_score = fft_score_full(a)

    rng = np.random.default_rng(seed)
    n = a.size
    sigma = 0.02
    stagnant = 0
    max_no_improve = 600

    while time.monotonic() < deadline:
        if stagnant > max_no_improve:
            sigma *= 0.5
            stagnant = 0
            if sigma < 1e-3:
                sigma = 0.05
                max_no_improve = 250

        cand = a.copy()
        r = float(rng.random())

        if r < 0.55:
            width = int(rng.integers(1, max(2, n // 50) + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            cand[start:start + width] *= np.exp(rng.normal(0.0, sigma, size=width))
        elif r < 0.80:
            width = int(rng.integers(max(1, n // 40), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            cand[start:start + width] *= np.exp(rng.normal(0.0, 0.5 * sigma, size=width))
        else:
            k = int(max(1, n // 100))
            idx = rng.integers(0, n, size=k)
            cand[idx] *= np.exp(rng.normal(0.0, sigma, size=k))

        cand = np.clip(cand, 0.0, 1000.0)
        s = float(cand.sum())
        if not np.isfinite(s) or s < 0.01:
            stagnant += 1
            continue

        sc = fft_score_full(cand)
        if np.isfinite(sc) and sc < best_score - 1e-15:
            a = cand
            best_score = sc
            stagnant = 0
        else:
            stagnant += 1

    return a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluator.
    Returns the best sequence found within the time budget.
    """
    t_start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0

    total_deadline = t_start + max(1.0, budget) - 1.0
    bfgs_deadline = max(t_start + 1.0, total_deadline - 900.0)
    full_deadline = max(t_start + 1.0, total_deadline - 120.0)
    local_deadline = total_deadline - 0.2

    seed_int = int(seed) if np.isfinite(float(seed)) else 42
    profiles = make_seed_profiles(seed_int)
    infos = []
    for c in profiles:
        sc = block_score(c)
        if not np.isfinite(sc):
            sc = float("inf")
        infos.append((sc, c))

    nprof = len(infos)
    if nprof == 0:
        fallback = np.ones(BLOCK_COUNT, dtype=np.float64) / float(BLOCK_COUNT)
        best_a = full_from_blocks(fallback)
        best_score = exact_score_from_a(best_a)
        return [float(x) for x in np.clip(best_a, 0.0, 1000.0)]

    scores = np.empty(nprof, dtype=np.float64)
    for j in range(nprof):
        scores[j] = infos[j][0]
    order = np.argsort(scores)

    best_a = None
    best_score = float("inf")

    # Evaluate a few promising block profiles exactly.
    for jj in range(nprof):
        idx = int(order[jj])
        if jj >= min(nprof, 12):
            break
        c = infos[idx][1]
        a = full_from_blocks(c)
        sc = exact_score_from_a(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print("seed profile score:", best_score)

    if best_a is None:
        idx0 = int(order[0])
        best_a = full_from_blocks(infos[idx0][1])
        best_score = exact_score_from_a(best_a)

    print("initial exact score:", best_score)

    # L-BFGS phase-zoo on the best seed profiles.
    top_count = min(nprof, 6)
    for jj in range(top_count):
        if time.monotonic() >= bfgs_deadline:
            break
        idx = int(order[jj])
        c = infos[idx][1]
        u = np.log(np.maximum(c, 1e-300))
        u = u - float(np.max(u))
        u = np.clip(u, -50.0, 50.0)
        saw_improvement = False

        for beta in BETA_STEPS:
            if time.monotonic() >= bfgs_deadline:
                break
            u = run_lbfgs_phase(u, beta)
            c2 = stable_softmax(u)
            a2 = full_from_blocks(c2)
            sc2 = exact_score_from_a(a2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_a = a2.copy()
                saw_improvement = True
                print("BFGS improved score:", best_score, "beta:", beta)

        if not saw_improvement:
            # Still continue with next seed briefly; stop early if too slow.
            pass

    print("after BFGS score:", best_score)

    # Full-length gradient polish.
    if time.monotonic() < full_deadline:
        polished_a, _ = full_gradient_polish(best_a, best_score, full_deadline)
        sc_pol = exact_score_from_a(polished_a)
        if np.isfinite(sc_pol) and sc_pol < best_score:
            best_score = sc_pol
            best_a = polished_a.copy()
            print("full gradient polish score:", best_score)

    # Short random local polish.
    if time.monotonic() < local_deadline:
        local_a, _ = random_local_polish(best_a, best_score, local_deadline, seed_int + 1)
        sc_loc = exact_score_from_a(local_a)
        if np.isfinite(sc_loc) and sc_loc < best_score:
            best_score = sc_loc
            best_a = local_a.copy()
            print("random local polish score:", best_score)

    # Final safety checks and authoritative evaluation (used only for logging).
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.where(np.isfinite(best_a), best_a, 0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if not np.isfinite(ss) or ss < 0.01:
        best_a = np.ones(SEQ_LEN, dtype=np.float64) / float(SEQ_LEN)

    out = [float(x) for x in best_a]
    try:
        final_sc = evaluate_sequence(out)
        print("final evaluator score:", final_sc)
    except Exception:
        final_sc = fft_score_full(best_a)
        print("final approximate score:", final_sc)

    return out
