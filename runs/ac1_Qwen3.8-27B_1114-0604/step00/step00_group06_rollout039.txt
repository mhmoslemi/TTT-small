    n = len(a)
    a = a / sum(a)  # normalize
    for step in range(num_steps):
        b = np.convolve(a, a)
        k_star = np.argmax(b)
        # Compute gradient of b[k_star] with respect to a
        grad = np.zeros(n)
        for m in range(n):
            idx = k_star - m
            if 0 <= idx < n:
                if idx == m:
                    grad[m] = a[idx]
                else:
                    grad[m] = 2 * a[idx]
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            a -= step_size * grad / grad_norm
        # Project onto sum(a) = 1, a >= 0
        a = np.maximum(a, 0)
        s = sum(a)
        if s > 0:
            a = a / s
    return a

"""
Minimizes 2*n*max(auto_conv) / sum(a)^2 for non-negative sequences using
smooth-max (log-sum-exp) gradient descent with simplex projection.
Tries multiple sequence lengths and initializations within the time budget,
then refines with a local search.
"""

import numpy as np
import time


def _fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def _grad_smooth_max(a, weights):
    """Gradient of smooth max of auto-convolution w.r.t. a using FFT cross-correlation."""
    n = len(a)
    size = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(weights, size)
    cc = np.fft.irfft(W * np.conj(A), size)
    return 2.0 * cc[:n]


def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    d = 1.0 + u * np.arange(1, n + 1, dtype=np.float64) - cssv
    valid = np.where(d > 0)[0]
    if len(valid) == 0:
        return np.ones(n, dtype=np.float64) / n
    rho = valid[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _objective_fft(a):
    """Compute 2*n*max(b) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * float(np.max(_fft_convolve(a)))


def _optimize_inner(a0, time_budget, rng):

    n = len(a0)

    t = 3.0 / n

    step = 0
    stall = 0

        try:
            gs = np.asarray(height_sequence_1, dtype=np.float64)
            if len(gs) != n:
                old_i = np.linspace(0, 1.0, len(gs))
                new_i = np.linspace(0, 1.0, n)
                gs = np.interp(new_i, old_i, gs)
            gs = np.maximum(gs, 0.0)
            if gs.sum() > 0:
                inits.append(('given', gs / gs.sum()))
        except Exception:
            pass

        per_init = remaining_for_n / (len(inits) + 1)

        for name, a0 in inits:
            if time.monotonic() >= deadline:
                break
            t_budget = min(per_init * 1.2, deadline - time.monotonic())
            if t_budget < 3:
                break
            a_opt, obj = _optimize_inner(a0, t_budget, rng)
            sc = float(evaluate_sequence(a_opt.tolist()))
            tag = " *** NEW BEST ***" if (np.isfinite(sc) and sc < best_score_global) else ""
            print(f"  {name}: obj={obj:.6f} eval={sc:.6f}{tag}")
            if np.isfinite(sc) and sc < best_score_global:
                best_score_global = sc
                best_global = a_opt.copy()

    # Final local search phase
    local_time = max(5.0, (deadline - time.monotonic()) * 0.9)
    print(f"\n--- Local search, budget={local_time:.0f}s ---")
    best_local, sc_local = _local_search(best_global, local_time, rng)
    if sc_local < best_score_global:
        best_score_global = sc_local
        best_global = best_local.copy()

    print(f"\nFinal: score={best_score_global:.6f}, n={len(best_global)}")
    return [float(x) for x in best_global]
