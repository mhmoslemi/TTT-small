    c = fast_convolve(a)
    S = np.sum(a)

    active = np.where(c >= P * 0.95)[0]

    d = np.zeros(n)

        # ∂c[k]/∂a[j] = 2*a[k-j] for valid k-j
        for j in range(n):

Minimize 2n * max(convolve(a,a)) / sum(a)^2 for non-negative sequence a.
Uses FFT-based gradient descent with adaptive step size, multiple initializations,
and simulated annealing to escape local minima.
"""

import numpy as np
import time


def _fft_convolve_square(a):
    """Compute full convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)[:size]
    return C


def _objective_from_conv(a, c):
    """Compute objective given a and its self-convolution c."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    P = np.max(c)
    return 2.0 * n * P / (S * S)


def _make_two_level_step(n, h, m):
    """Create a two-level step sequence: h for first m elements, 1 for rest."""
    a = np.ones(n)
    a[:m] = h
    return a


def _gradient_direction(a, c, k_peaks=5):
    """Compute gradient descent direction for the peak positions of c."""
    n = len(a)
    S = np.sum(a)
    P = np.max(c)
    
    # Find top-k peak positions
    if k_peaks >= n:
        top_k = np.arange(n)
    else:
        # Use partial sort for efficiency when n is large
        if n > 100:
            top_k = np.argpartition(c, -k_peaks)[-k_peaks:]
        else:
            top_k = np.argsort(c)[-k_peaks:]
    top_k = top_k[np.argsort(c[top_k])[::-1]]
    
    # Compute b[j] = (1/k) * sum_{k in top_k} a[k-j] for valid indices
    b = np.zeros(n)
    for k_star in top_k:
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            b[lo:hi+1] += a[k_star - hi : k_star - lo + 1][::-1]
    b /= len(top_k)
    
    # Descent direction: want to reduce c[k] for peak k
    # d[j] proportional to (P/S - 2*b[j])
    d = P / S - 2.0 * b
    # Center to maintain sum constraint
    d = d - np.mean(d)
    return d

    a_cubic = x**3
    a_cubic = np.maximum(a_cubic, 0.01)
    initializations.append(a_cubic)
    
    # Two-level with additional structure (staircase)
    a_stair = np.zeros(n)
    for seg in range(5):
        seg_start = seg * n // 5
        seg_end = (seg + 1) * n // 5
        a_stair[seg_start:seg_end] = 0.2 + 0.2 * seg
    initializations.append(a_stair)
    
    # Random initialization
    a_rand = rng.random(n) * 0.8 + 0.1
    initializations.append(a_rand)
    
    # Evaluate all initializations quickly
    for i, a in enumerate(initializations):
        a = np.maximum(a, 0)
        S = np.sum(a)
        if S < 1e-10:
            continue
        a = a / S
        c = _fft_convolve_square(a)
        obj = _objective_from_conv(a, c)
        if obj < best_score:
            best_score = obj
            best_a = a.copy()
    
    print(f"Phase 1 best: {best_score:.6f}")
    
    # Phase 2: Gradient descent from the best initialization
    # and from a few other good initializations
    phase2_deadline = time.monotonic() + (float(budget_s) - 100.0)
    
    # Sort initializations by score and take top few for gradient descent
    scored_inits = []
    for a in initializations:
        a = np.maximum(a, 0)
        S = np.sum(a)
        if S < 1e-10:
            continue
        a = a / S
        c = _fft_convolve_square(a)
        obj = _objective_from_conv(a, c)
        scored_inits.append((obj, a))
    scored_inits.sort(key=lambda x: x[0])
    
    # Run gradient descent from top initializations
    for init_idx in range(min(5, len(scored_inits))):
        if time.monotonic() > phase2_deadline:
            break
        
        obj_init, a_init = scored_inits[init_idx]
        print(f"  Gradient descent from init {init_obj_id} (score={obj_init:.6f})")
        
        max_iters = 30000
        a_result, obj_result = _gradient_descent(
            a_init, max_iters=max_iters, seed=seed + init_idx)
        
        if obj_result < best_score:
            best_score = obj_result
            best_a = a_result.copy()
            print(f"    Improved to: {best_score:.6f}")
        
        # Also try gradient descent from the current best (to build on it)
        if best_a is not None and time.monotonic() < phase2_deadline:
            max_iters2 = min(max_iters, max(5000, int((phase2_deadline - time.monotonic()) * 2000)))
            a_result2, obj_result2 = _gradient_descent(
                best_a + rng.normal(0, 0.005, n),
                max_iters=max(1000, max_iters2),
                seed=seed + 100 + init_idx)
            if obj_result2 < best_score:
                best_score = obj_result2
                best_a = a_result2.copy()
                print(f"    Improved to: {best_score:.6f}")
    
    print(f"Phase 2 best: {best_score:.6f}")
    
    # Phase 3: Local search from the best solution
    if time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        n_iters = int(remaining * 1500)  # ~1500 iterations per second
        n_iters = max(1000, min(n_iters, 200000))
        print(f"  Local search with {n_iters} iterations...")
        
        a_result, obj_result = _local_search(
            best_a, n_iterations=n_iters, seed=seed + 200)
        
        if obj_result < best_score:
            best_score = obj_result
            best_a = a_result.copy()
            print(f"    Improved to: {best_score:.6f}")
    
    # Phase 4: Final gradient descent polish
    if time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            max_iters = int(remaining * 2000)
            max_iters = max(500, min(max_iters, 50000))
            print(f"  Final gradient polish with {max_iters} iterations...")
            a_result, obj_result = _gradient_descent(
                best_a, max_iters=max_iters, seed=seed + 500)
            if obj_result < best_score:
                best_score = obj_result
                best_a = a_result.copy()
                print(f"    Improved to: {best_score:.6f}")
    
    # Convert to list of floats for the final answer
    result = [float(max(0.0, x)) for x in best_a]
    
    # Verify with the actual evaluator
    final_score = float(evaluate_sequence(result))
    if not np.isinf(final_score):
        print(f"Final verified score: {final_score:.6f}")
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * n
    
    return result
