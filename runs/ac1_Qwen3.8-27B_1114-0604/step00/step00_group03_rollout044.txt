"""
Minimizes F(a) = 2n*max(conv(a,a))/(sum(a))^2 for a non-negative step function.
Strategy: soft-max subgradient descent on the autocorrelation peak (scale-invariant,
so we fix sum(a)=sqrt(2n) and minimize max(a*a)), with adaptive temperature,
patience-based random restarts, periodic blending toward uniform to avoid sparsity,
and occasional LP-based direction finding (maximize sum(g) s.t. a*g <= peak, g>=0)
to escape plateaus. Multiple sequence lengths and starting points are explored under
a strict time budget; the best valid sequence found is always returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

_DEADLINE = [0.0]


def _autocorr(a):
    return np.convolve(a, a)


def _objective(a):
    n = len(a)
    if n == 0:
        return np.inf
    r = _autocorr(a)
    max_r = float(np.max(r))
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    return 2.0 * n * max_r / (S * S)


def _soft_max_grad(a, beta, top_m=120):
    """Gradient of soft_max(conv(a,a), beta) w.r.t. a, using top-m lags.
    Returns (gradient, current peak)."""
    n = a.size
    r = np.convolve(a, a)
    m = min(top_m, r.size)
    top_idx = np.argpartition(r, -m)[-m:]
    r_top = r[top_idx]
    r_max = float(np.max(r_top))
    r_shifted = r_top - r_max
    weights = np.exp(beta * r_shifted)
    wsum = weights.sum()
    if wsum <= 1e-300:
        weights = np.full(m, 1.0 / m)
    else:
        weights = weights / wsum
    g = np.zeros(n)
    for i in range(m):
        w = weights[i]
        if w < 1e-12:
            continue
        k = int(top_idx[i])
        j_min = k - n + 1
        if j_min < 0:
            j_min = 0
        j_max = k
        if j_max >= n:
            j_max = n - 1
        if j_min > j_max:
            continue
        j_idx = np.arange(j_min, j_max + 1)
        a_idx = k - j_idx
        g[j_idx] += 2.0 * w * a[a_idx]
    return g, float(np.max(r))


def _lp_direction(a, M):
    """Find g>=0 maximizing sum(g) subject to (a*g)_k <= M for all k.
    Returns rescaled g (sum = sum(a)) or None."""
    n = a.size
    if n < 2:
        return None
    K = 2 * n - 1
    # Build constraint matrix A of shape (K, n): A[k,j] = a[k-j] if 0<=k-j<n
    A = np.zeros((K, n))
    for k in range(K):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            j_idx = np.arange(j_min, j_max + 1)
            A[k, j_idx] = a[k - j_idx]
    c = -np.ones(n)  # minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=np.full(K, M), bounds=[(0, None)] * n,
                  method='highs', options={"presolve": True})
    if not res.success:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = g.sum()
    if s < 1e-12:
        return None
    return g / s


def _blend(a, target_sum):
    b = a / a.sum()
    s = b.sum()
    if s < 1e-12:
        return None
    b = b / s
    t = target_sum / b.sum()
    return b * t


def _try_update(best_a, best_F, cand, target_sum):
    s = cand.sum()
    if s < 1e-12:
        return best_a, best_F, False
    cand = cand * (target_sum / s)
    F = _objective(cand)
    if F < best_F:
        return cand.copy(), F, True
    return best_a, best_F, False
