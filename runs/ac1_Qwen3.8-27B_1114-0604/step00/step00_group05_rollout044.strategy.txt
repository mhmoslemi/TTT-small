       # Initialization logic

           arr = np.clip(arr, 0.1, 1000) / (sum(arr)+1e-12)
           return arr

       current = generate_candidate(best_solution, base_sequence)  # Or generate long
       low_high_ratio = 0  # Placeholder for future risks
       progress_step = max(1, round(parent.size / 10)) if any in the fallback else 1
