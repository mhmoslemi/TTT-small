def _next_pow2(n):
    """Return smallest power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _convolve_fft(a, fft_size):
    """Compute linear convolution a*a via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    buf = np.zeros(fft_size)
    buf[:n] = a
    A = np.fft.rfft(buf)
    c = np.fft.irfft(A * A)
    return c[:2 * n - 1]


def _grad_softmax(a, c, beta, fft_size):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*c[k])) w.r.t. a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    N = 2 * n - 1
    cmax = c.max()
    e = np.exp(beta * (c - cmax))
    Z = e.sum()
    if Z < 1e-300:
        p = np.full(N, 1.0 / N)
    else:
        p = e / Z
    # q[k] = 2*p[k], zero-padded
    q = np.zeros(fft_size)
    q[:N] = 2.0 * p
    ap = np.zeros(fft_size)
    ap[:n] = a
    # Cross-correlation: grad[i] = sum_k q[k]*a[k-i]
    g = np.fft.irfft(np.fft.rfft(q) * np.conj(np.fft.rfft(ap)))[:n]
    return g
