"""
Minimize  2*n*max(convolve(a,a)) / (sum(a))**2  for a non-negative sequence a.

Approach:
  - The score is scale-invariant, so we optimize on the simplex sum(a)=1, a>=0.
  - The hard max is replaced by a smooth log-sum-exp S(c)=log(sum_k exp(c_k/tau)),
    whose gradient w.r.t. a is  (2/tau) * sum_k p_k * a[k-t]  with p=softmax(c/tau).
  - Temperature tau is annealed from "soft" (broad) to "hard" (peak-focused).
  - Autocorrelation (full self-convolution) is computed with a zero-padded FFT.
  - The gradient step is a direction on the simplex; a log-spaced line search
    (backtracking + a few increasing steps) picks the best feasible move.
  - When stuck, the iterate is reset to the run best and a multiplicative block
    perturbation is applied to escape local minima. Several random/structured
    starts are tried over the time budget, then a final polish from the global best.

Authoritative scoring is always done with evaluate_sequence().
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n *= 2
    return n


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a)
        s = a.size
    return a / s


def _autocorr(a, N):
    """Full self-convolution (same order as np.convolve(a,a,'full')) via FFT."""
    fa = np.fft.fft(a, N)
    c = np.real(np.fft.ifft(fa * fa, N))[: 2 * a.size - 1]
    return np.maximum(c, 0.0)


def _softmax_weights(c, tau):
    cs = (c - c.max()) / tau
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros_like(c)
        cm = int(np.argmax(c))
        p[cm] = 1.0
    else:
        p = p / s
    return p


def _smooth_grad(a, p, tau):
    """(2/tau) * sum_k p[k] * a[k-t]  for each t = 0..n-1 (gradient of LSE)."""
    n = a.size
    g = np.correlate(p, a, mode="full")[n - 1: 2 * n - 1]
    return (2.0 / tau) * g


def _block_perturb(a, rng):
    n = a.size
    b = a.copy()
    nblocks = int(rng.integers(1, 4))
    for _ in range(nblocks):
        w = int(rng.integers(max(1, n // 120), max(2, n // 8)))
        st = int(rng.integers(0, max(1, n - w)))
        st = min(st, n - w)
        b[st:st + w] *= float(np.exp(rng.normal(0.0, 0.45, size=w)))
    return _normalize(b)


def _optimize_run(start, deadline, rng, n, N, global_best, global_score):
    """Run smooth-gradient line-search descent from start until deadline.
    Returns (best_a, best_score, improved_global_best, improved_global_score)."""
    a = _normalize(np.asarray(start, dtype=np.float64))
    cur_score = float(evaluate_sequence(a.tolist()))
    run_best_a = a.copy()
    run_best_score = cur_score

    tau_ratio = 0.10
    stagnation = 0
    it = 0

    while time.monotonic() < deadline:
        c = _autocorr(a, N)
        max_c = float(c.max())
        if max_c < 1e-15:
            break
        tau = max(tau_ratio * max_c, 1e-12)
        p = _softmax_weights(c, tau)
        g = _smooth_grad(a, p, tau)
        gl2 = float(np.sqrt(np.dot(g, g)))
        if gl2 < 1e-15:
            break
        al2 = float(np.sqrt(np.dot(a, a)))
        t0 = 0.12 * al2 / gl2

        best_cand_score = cur_score
        best_cand = a
        best_cand_is_new = False
        for f in (4.0, 2.0, 1.0, 0.5, 0.25, 0.125, 0.0625):
            t = t0 * f
            a_try = a - t * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-12:
                continue
            a_try = a_try / s
            sc = float(evaluate_sequence(a_try.tolist()))
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_cand = a_try
                best_cand_is_new = True

        if best_cand_is_new and best_cand_score < cur_score - 1e-12:
            a = best_cand
            cur_score = best_cand_score
            stagnation = 0
            if cur_score < run_best_score:
                run_best_score = cur_score
                run_best_a = a.copy()
        else:
            stagnation += 1
            if stagnation >= 8:
                a = _block_perturb(run_best_a, rng)
                cur_score = float(evaluate_sequence(a.tolist()))
                if cur_score < run_best_score:
                    run_best_score = cur_score
                    run_best_a = a.copy()
                stagnation = 0

        tau_ratio = max(0.004, tau_ratio * 0.9985)
        it += 1

    # propagate to global
    improved_a = global_best
    improved_score = global_score
    if run_best_score < improved_score:
        improved_a = run_best_a
        improved_score = run_best_score

    return run_best_a, run_best_score, improved_a, improved_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for the best non-negative coefficient sequence within budget_s."""
    global_best = None
    global_score = np.inf

    try:
        global_best = _normalize(np.asarray(height_sequence_1, dtype=np.float64))
        global_score = float(evaluate_sequence(global_best.tolist()))
    except Exception:
        global_best = None
        global_score = np.inf

    if global_best is None or global_best.size == 0:
        n = 6121
        global_best = np.ones(n, dtype=np.float64)
        global_score = float(evaluate_sequence(global_best.tolist()))
    else:
        n = int(global_best.size)

    N = _next_pow2(2 * n - 1)
    rng = np.random.default_rng(seed)

    t0 = time.monotonic()
    hard_deadline = t0 + max(10.0, float(budget_s)) - 8.0

    print(f"[init] n={n} fftN={N} global_score={global_score:.6f}", flush=True)

    # Build a few starts: current best, uniform, and random positive sequences.
    starts = [global_best.copy()]
    starts.append(np.ones(n, dtype=np.float64))
    for k in range(3):
        base = 1.0 + 0.6 * rng.standard_normal(n)
        starts.append(_normalize(base))
    # one "blocky random" start to add structure
    bstart = np.ones(n, dtype=np.float64)
    seg = max(1, n // 200)
    for i in range(0, n, seg):
        bstart[i:min(n, i + seg)] = float(rng.uniform(0.3, 1.7))
    starts.append(_normalize(bstart))

    # Give each start a time slice, then reserve the tail for a global polish.
    n_starts = len(starts)
    slice_frac = 0.65 / n_starts
    for idx, sp in enumerate(starts):
        slice_deadline = t0 + (slice_frac * (budget_s - 8.0)) * (idx + 1)
        if time.monotonic() > slice_deadline:
            continue
        print(f"[run {idx}] start ...", flush=True)
        _, _, global_best, global_score = _optimize_run(
            sp, slice_deadline, rng, n, N, global_best, global_score
        )
        print(f"[run {idx}] done, global_score={global_score:.6f}", flush=True)

    # Final polish from the global best with all remaining time.
    if time.monotonic() < hard_deadline:
        _, _, global_best, global_score = _optimize_run(
            global_best, hard_deadline, rng, n, N, global_best, global_score
        )

    # Validate and return.
    out = _normalize(global_best)
    out = np.clip(out, 0.0, 1000.0)
    final_score = float(evaluate_sequence(out.tolist()))
    print(f"[final] n={n} score={final_score:.6f}", flush=True)
    return [float(x) for x in out]
