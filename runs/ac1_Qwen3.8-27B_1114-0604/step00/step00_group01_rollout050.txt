peak auto-convolution of non-negative sequences. Uses FFT-based O(n log n)

import numpy as np
import time

def _next_pow2(n):
    """Smallest power of 2 >= n."""
    if n <= 1:
        return 1
    p = 1
    while p < n:
        p <<= 1
    return p

def _auto_conv(a):
    """Auto-convolution via FFT."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2*n-1]

def _grad_from_weights(a, w):
    """
    Compute g[m] = 2*sum_{j=0}^{n-1} a[j]*w[m+j] via FFT.
    a has length n, w has length 2n-1.
    Returns array of length n.
    """
    n = len(a)
    w_rev = w[::-1]
    N = _next_pow2(3 * n - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(A * W, N)

"""
Iterative soft-max gradient descent with multiplicative updates to minimize
2n*max(conv(a,a))/(sum a)^2 over non-negative sequences.
Uses FFT-based O(n log n) gradient computation. Multiple restarts with
diverse initializations; deep-optimizes the best candidate found.
"""

import numpy as np
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    p = 1
    while p < n:
        p <<= 1
    return p

def _auto_conv(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]

def _grad_from_weights(a, w):
    n = len(a)
    w_rev = w[::-1]
    N = _next_pow2(3 * n - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(A * W, N)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]

def _softmax_weights(C, beta):
    Cmax = C.max()
    e = np.exp(beta * (C - Cmax))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(C)
        w[C.argmax()] = 1.0
    else:
        w = e / s
    return w

def _optimize(a0, deadline, beta_start, alpha_start):
    n = len(a0)
    a = np.clip(a0.copy(), 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return np.inf, a
    a /= s

    best_val = np.inf
    best_a = a.copy()

    C = _auto_conv(a)
    obj = 2.0 * n * C.max()
    best_val = obj
    best_a = a.copy()

    beta = beta_start
    alpha = alpha_start
    it = 0
    stalled = 0
    EVAL_EVERY = 200

    while time.monotonic() < deadline:
        it += 1
        C = _auto_conv(a)
        w = _softmax_weights(C, beta)
        g = _grad_from_weights(a, w)

        gc = g - g.mean()
        cap = 5.0 / alpha
        gc = np.clip(gc, -cap, cap)
        a = a * np.exp(-alpha * gc)
        a = np.maximum(a, 0.0)
        sn = a.sum()
        if sn < 1e-15:
            alpha *= 0.5
            a = best_a.copy()
            a /= a.sum()
            continue
        a /= sn

        if it % EVAL_EVERY == 0:
            Cnow = _auto_conv(a)
            obj = 2.0 * n * Cnow.max()
            if obj < best_val - 1e-12:

                stalled = 0

    return best_val, best_a

def _init_hann(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 + 0.5 * np.cos(2.0 * np.pi * t / max(n - 1, 1))

def _init_gauss(n, sf=0.25):
    t = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    return np.exp(-0.5 * ((t - c) / (sf * n)) ** 2)

def _init_taper(n, pw=1.5):
    t = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    x = np.abs(t - c) / max(c, 1.0)
    return np.maximum(1.0 - x ** pw, 0.0)

def _init_bell(n, pw=3.0):
    t = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    x = (t - c) / (0.3 * n)
    return np.exp(-np.abs(x) ** pw)

def _init_cosine_taper(n, pw=2.0):

    c = (n - 1) / 2.0
    x = (t - c) / max(c, 1.0)
    return np.cos(0.5 * np.pi * np.abs(x)) ** pw
