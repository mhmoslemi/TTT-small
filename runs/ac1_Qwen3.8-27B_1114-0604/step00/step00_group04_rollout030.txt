```python
from scipy import sparse
from scipy.optimize import linprog

def solve_lp(f, M, n):
    # Build constraint matrix A (2n-1 by n)
    # A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    b_rhs = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_rhs, bounds=[(0, None)]*n, method='highs')
    return res.x
```

```python
import numpy as np
```

def fft_convolution(a, b):
    """Fast linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def normalize_positive(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def mirror_descent(a0, n, rng, time_limit=60.0, initial_eta=0.05):
    """
    Mirror descent (multiplicative weights) to minimize max(conv(a,a))
    subject to a >= 0, sum(a) = 1.
    Returns (best_a, best_score) where best_score = 2*n*peak (with sum=1).
    """
    a = normalize_positive(a0.astype(np.float64).copy())
    best_a = a.copy()
    
    conv = fft_convolution(a, a)
    best_peak = np.max(conv)
    
    start = time.monotonic()
    eta = initial_eta
    time_at_last_improve = start
    
    while time.monotonic() - start < time_limit:
        k_star = int(np.argmax(conv))
        
        # Subgradient: grad[j] = 2*a[k_star-j] if 0<=k_star-j<n
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            grad[lo:hi+1] = 2.0 * a[k_star - hi : k_star - lo + 1][::-1]
        
        # Multiplicative update
        a = a * np.exp(-eta * grad)
        a = np.clip(a, 1e-30, None)
        total = a.sum()
        if total < 1e-30:
            a = np.ones(n) / n
        else:
            a = a / total
        
        conv = fft_convolution(a, a)
        peak = np.max(conv)

            best_peak = peak
