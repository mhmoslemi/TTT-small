    a = np.clip(a.astype(np.float64), 1e-6, 1000.0)
    a = a / a.sum()  # Normalize
    rng = np.random.default_rng(seed)
    
    deadline = time.monotonic() + time_budget
    tau = initial_tau
    alpha = initial_alpha
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    while time.monotonic() < deadline:

        b = np.convolve(a, a)

        b_max = b.max()
        exp_b = np.exp((b - b_max) / tau)
        weights = exp_b / exp_b.sum()
        
        # Gradient
        grad = 2.0 * np.correlate(weights, a, mode='valid')

                best = a_new.copy()
        
        a = a_new
        iteration += 1
        
        # Anneal
        tau *= 0.999
        alpha *= 0.9995
    
    return best

def project_to_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * (np.arange(1, n+1)) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def projected_subgradient(a_init, n, time_budget, seed=0, tau0=0.05, alpha0=0.5):

    rng = np.random.default_rng(seed)
