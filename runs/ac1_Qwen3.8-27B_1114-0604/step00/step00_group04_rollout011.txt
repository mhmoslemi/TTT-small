import time
from scipy.fft import fft, ifft

def fast_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    # Pad to next power of 2 for efficiency
    size = 1
    while size < n:
        size *= 2
    Fa = fft(a, size)
    Fb = fft(b, size)
    result = np.real(ifft(Fa * Fb))
    return result[:n]

def compute_objective(a):
    """Compute the objective value for sequence a."""
    n = len(a)
    b = fast_convolve(a, a)
    max_b = np.max(b)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_b / (S * S)

        S = 1.0

    eta = 0.1  # learning rate

        b = fast_convolve(a, a)
        max_b = np.max(b)
        obj = 2.0 * n * max_b  # S=1
        if obj < best_obj:

            best_a = a.copy()

    fa = fft(a, size)
    result = np.real(ifft(fa * fa))

    g = np.zeros(n)
    j_lo = max(0, k - (n-1))
    j_hi = min(k, n-1)
    if j_lo <= j_hi:
        idx = np.arange(j_lo, j_hi+1)
        g[idx] = a[k-idx] + a[idx]
    return g, k, b

def optimize(a0, budget_s, rng):
    """Local optimization from initial point a0."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 1e-10, None)
    
    best_a = a.copy()
    best_obj = np.inf
    
    # Initial evaluation
    b = fast_convolve(a)
    S = np.sum(a)
    obj = 2.0 * n * np.max(b) / (S*S)
    best_obj = obj
    best_a = a.copy()
    
    eta = 0.1
    deadline = time.monotonic() + budget_s
    iterations = 0
    
    while time.monotonic() < deadline:
        g, k, b = subgradient_at_max(a)
        
        if np.max(g) > 0:
            a *= np.exp(-eta * g / np.max(g))
            a = np.clip(a, 1e-15, None)
            S = np.sum(a)
            if S < 1e-10:
                a = np.ones(n)
                S = n
            a = a / S
        
        # Evaluate
        b = fast_convolve(a)
        S = np.sum(a)
        obj = 2.0 * n * np.max(b) / (S*S)
        iterations += 1
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Adaptive eta
        if iterations % 100 == 0:
            eta *= 0.95
    
    return best_a, best_obj

def make_initialized_sequence(n, mode, rng):
    """Create an initial sequence of length n using the specified mode."""
    if mode == 'uniform':
        return np.ones(n)
    elif mode == 'cosine':
        alpha = 1.0
        i = np.arange(n)
        return 1.0 + alpha * np.cos(np.pi * i / (n-1))
    elif mode == 'random_antisym':
        f = rng.standard_normal(n)
        f = (f - f[::-1]) / 2.0  # antisymmetrize
        a = 1.0 + 0.5 * f
        return np.clip(a, 0.01, None)
    elif mode == 'cosine_multi':
        i = np.arange(n)
        a = np.ones(n)
        for m in [1, 3, 5, 7]:
            alpha_m = 0.3 / m
            a += alpha_m * np.cos(np.pi * m * i / (n-1))
        return np.clip(a, 0.01, None)
    else:
        return np.ones(n)
