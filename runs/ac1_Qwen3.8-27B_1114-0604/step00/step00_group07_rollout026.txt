Minimizes the evaluation function 2*n*max(convolve(a,a))/sum(a)^2 
using projected smoothed-subgradient descent with multiple initializations,
adaptive temperature, and random restarts. Exploits scale invariance 
(normalizing sum to 1) and FFT-based convolution for efficiency.

import numpy as np
import time


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def full_convolution(a):
    """Compute the full linear convolution a * a using FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    result = np.fft.irfft(fa * fa, fsize)
    return result[:size]


def compute_score_normalized(a):
    """Compute 2*n*max(b) assuming sum(a)=1."""
    n = a.size
    b = full_convolution(a)
    return 2.0 * n * np.max(b)


def smoothed_subgradient(a, b, tau):
    """
    Compute smoothed subgradient of max(b) where b = a*a.
    Uses softmax weighting over the convolution values.
    """
    n = a.size
    size_b = b.size  # 2n-1
    # Softmax weights
    b_shifted = b / tau - np.max(b) / tau
    w = np.exp(b_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(size_b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    # Smoothed gradient: g_j = 2 * sum_k w_k * a_{k-j}
    # = 2 * (cross-correlation of w and a at lag j)
    # np.correlate(w, a, 'valid') gives c[k] = sum_n a[n]*w[n+k]
    g = 2.0 * np.correlate(w, a, mode='valid')
    # Ensure length is n
    if g.size != n:
        # This shouldn't happen, but just in case
        g_new = np.zeros(n)
        m = min(g.size, n)
        g_new[:m] = g[:m]
        g = g_new
    return g

    Run projected smoothed-subgradient descent.
    a is normalized to sum to 1.
    Returns best a and best score found.
    """
    n = a.size
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    
    tau = tau_init
    eta = eta_init
    n_iter = 0
    last_improve_iter = 0
    stall_count = 0
    
    start_time = time.monotonic()
    
    while time.monotonic() - start_time < max_time:
        b = full_convolution(a)
        current_score = 2.0 * n * np.max(b)
        
        # Compute smoothed subgradient
        g = smoothed_subgradient(a, b, tau)
        
        # Projected gradient step
        a_new = project_simplex(a - eta * g)
        
        # Evaluate new point
        b_new = full_convolution(a_new)
        new_score = 2.0 * n * np.max(b_new)
        
        # Accept/reject

            a = a_new
        else:

        if new_score < best_score:

            best_a = a_new.copy()
            last_improve_iter = n_iter
