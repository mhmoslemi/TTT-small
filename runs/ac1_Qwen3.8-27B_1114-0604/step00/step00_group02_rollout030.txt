import time

def _next_pow2(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length() if n > 1 else 1

def _conv_fft(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    N = _next_pow2(n)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n]

def _xcorr_fft(p, a):
    """Cross-correlation (p ⋆ a)[j] = sum_i p[i+j]*a[i], computed via FFT."""
    n_out = len(p) + len(a) - 1
    N = _next_pow2(n_out)
    return np.fft.irfft(np.conj(np.fft.rfft(p, N)) * np.fft.rfft(a, N), N).real[:n_out]

def _project_sum_nonneg(x, S):
    """Project x onto {v: sum(v)=S, v>=0} via bisection on the water level."""
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.maximum(0.0, x - mid)) > S:
            lo = mid
        else:
            hi = mid
    return np.maximum(0.0, x - 0.5 * (lo + hi))

def _fast_score(a):
    """Fast score computation using FFT convolution."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-6:
        return np.inf
    c = _conv_fft(a, a)
    return 2.0 * n * float(np.max(c)) / (S * S)

    Nx = _next_pow2(len(a) + n - 1 + n)  # for xcorr: len(p)=2n-1, len(a)=n
    # Actually xcorr output length is (2n-1) + n - 1 = 3n-3
    Nx = _next_pow2(3 * n)

    tau = tau0
    eta = eta0

    beta = 0.9  # momentum coefficient

    while time.time() < deadline:
        # Convolution
        A = np.fft.rfft(a, Nc)
        c = np.fft.irfft(A * A, Nc)[:2 * n - 1]

            best = a.copy()

        p = np.exp(tau * (c - max_c))

        a = _project_sum_nonneg(a, S)

def _lp_iteration(f, M, n):
    """
    Solve the LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (unnormalized).
    Uses scipy.optimize.linprog with sparse matrix.
    """
    from scipy.optimize import linprog
    from scipy.sparse import lil_matrix
    
    # Build constraint matrix: A[k, j] = f[k-j] if 0 <= k-j < n
    # This is a lower "convolution" matrix
    A = lil_matrix((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        for k in range(j, j + n):
            A[k, j] = f[k - j]
    A = A.tocsr()
    
    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M), 
                  bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x
    return None
