"""
Search for non-negative step-function heights minimizing
    2*n*max(convolve(a,a))/(sum(a)**2).

Strategy:
- Use smooth, zero-mean anti-symmetric/sinusoidal basis functions to generate
  strong deterministic and random starts.
- Run a small coefficient-space parameter search (random + Nelder-Mead if scipy
  is available).
- Refine full-length probabilities with an exponentiated-gradient descent method
  using smoothed max-convolution subgradients computed by FFT.
- Keep the best sequence found and use the exact evaluator only for final sanity.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as _scipy_minimize
    _HAVE_SCIPY = True
except Exception:
    _HAVE_SCIPY = False


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _positive_normalize(a):
    x = np.asarray(a, dtype=np.float64).ravel()
    if x.size == 0:
        x = np.ones(1, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 0.0, 1000.0)
    if not np.any(x > 0.0):
        return np.ones(x.size, dtype=np.float64) / float(x.size)
    x = np.maximum(x, 1e-12)
    s = float(np.sum(x))
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(x.size, dtype=np.float64) / float(x.size)
    x = x / s
    return x


def _fft_convolve_full(a, b=None):
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    need = a.size + b.size - 1
    if need <= 0:
        return np.empty(0, dtype=np.float64)
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    if b is a:
        out = np.fft.irfft(fa * fa, N)
    else:
        fb = np.fft.rfft(b, N)
        out = np.fft.irfft(fa * fb, N)
    return out[:need]


def _prob_score(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.inf, np.empty(0, dtype=np.float64)
    s = float(np.sum(p))
    if s <= 1e-300 or not np.isfinite(s):
        return np.inf, np.empty(0, dtype=np.float64)
    p = p / s
    if n == 1:
        return 2.0, np.array([1.0], dtype=np.float64)

    q = _fft_convolve_full(p, None)
    q = np.maximum(q, 0.0)
    if q.size == 0:
        return np.inf, q
    maxq = float(np.max(q))
    score = float(2.0 * n * maxq)
    return score, q


def _cross_corr(p, w):
    """
    Returns r_i = sum_j w_{j+i} p_j, the gradient field for weighted
    convolution terms.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    P = np.fft.rfft(p, N)
    W = np.fft.rfft(w, N)
    c = np.fft.irfft(np.conj(P) * W, N)
    return c[:n]


def _make_basis(n, d):
    if n <= 0:
        return np.empty((0, 0), dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    basis = np.empty((d, n), dtype=np.float64)
    basis[0, :] = 2.0 * x - 1.0
    idx = 1
    modes = [1, 3, 5, 7, 9, 11, 13, 15]
    for m in modes:
        if idx >= d:
            break
        basis[idx, :] = np.cos(m * np.pi * x)
        idx += 1
        if idx >= d:
            break
        basis[idx, :] = np.sin(m * np.pi * x)
        idx += 1

    for j in range(d):
        row = basis[j]
        row -= float(np.mean(row))
        mx = float(np.max(np.abs(row)))
        if mx > 1e-15:
            row /= mx
    return basis


def _projected_basis_p(basis, coeff):
    n = basis.shape[1]
    if n == 0:
        return np.empty(0, dtype=np.float64)
    c = np.asarray(coeff, dtype=np.float64).ravel()
    h = basis.T.dot(c)
    raw = 1.0 + h
    mn = float(np.min(raw))
    if mn < -1e-9:
        raw *= (-1.0 / mn)
    raw = np.maximum(raw, 1e-10)
    s = float(np.sum(raw))
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return raw / s


def _coeff_objective(coeff, basis, n):
    c = np.asarray(coeff, dtype=np.float64).ravel()
    if c.size == 0:
        return 3.0
    if not np.all(np.isfinite(c)):
        return 1e6
    if float(np.max(np.abs(c))) > 4.0:
        return 1000.0 + 0.1 * float(np.sum(c * c))
    p = _projected_basis_p(basis, c)
    score, _ = _prob_score(p)
    if not np.isfinite(score):
        return 1e6
    return float(score)


def _insert_top(score, coeff, top, max_top):
    if not np.isfinite(score):
        return
    c = np.asarray(coeff, dtype=np.float64).ravel().copy()
    pos = 0
    for i in range(len(top)):
        if score < top[i][0]:
            pos = i
            break
        pos = i + 1
    top.insert(pos, (float(score), c))
    if len(top) > max_top:
        top.pop()


def _parameter_search(basis, n, deadline, rng):
    d = basis.shape[0]
    top = []
    max_top = 6

    manuals = []

    manuals.append(np.zeros(d, dtype=np.float64))

    c = np.zeros(d, dtype=np.float64)
    if d > 1:
        c[1] = 0.9
    if d > 3:
        c[3] = -0.2
    if d > 5:
        c[5] = 0.1
    if d > 7:
        c[7] = 0.04
    manuals.append(c)

    c = np.zeros(d, dtype=np.float64)
    if d > 1:
        c[1] = 0.8
    if d > 2:
        c[2] = 0.2
    if d > 3:
        c[3] = -0.1
    if d > 4:
        c[4] = -0.15
    if d > 5:
        c[5] = 0.18
    if d > 7:
        c[7] = -0.05
    manuals.append(c)

    c = np.zeros(d, dtype=np.float64)
    if d > 1:
        c[1] = 0.95
    if d > 2:
        c[2] = -0.2
    if d > 3:
        c[3] = -0.12
    if d > 4:
        c[4] = 0.1
    if d > 5:
        c[5] = 0.05
    if d > 6:
        c[6] = 0.08
    manuals.append(c)

    c = np.zeros(d, dtype=np.float64)
    if d > 0:
        c[0] = 0.25
    if d > 1:
        c[1] = 0.9
    if d > 5:
        c[5] = -0.06
    manuals.append(c)

    c = np.zeros(d, dtype=np.float64)
    if d > 0:
        c[0] = -0.2
    if d > 1:
        c[1] = 0.85
    if d > 3:
        c[3] = 0.1
    manuals.append(c)

    for c in manuals:
        s = _coeff_objective(c, basis, n)
        _insert_top(s, c, top, max_top)

    max_random = 1200 if n >= 4096 else 2500
    for it in range(max_random):
        if time.monotonic() > deadline - 15.0:
            break

        coeff = rng.normal(0.0, 0.38, size=d)
        if rng.random() < 0.35:
            coeff *= 0.65
        if rng.random() < 0.30:
            cutoff = max(1, d // 2)
            coeff[cutoff:] *= 0.25

        s = _coeff_objective(coeff, basis, n)
        _insert_top(s, coeff, top, max_top)

        if (it + 1) % 400 == 0:
            best = top[0][0] if top else np.inf
            print("param search", it + 1, "best", best, flush=True)

    if _HAVE_SCIPY and len(top) > 0 and time.monotonic() < deadline - 10.0:
        maxiter = 90 if d >= 8 else 140
        for _, c0 in list(top[:3]):
            if time.monotonic() > deadline - 8.0:
                break
            try:
                res = _scipy_minimize(
                    _coeff_objective,
                    c0,
                    args=(basis, n),
                    method="Nelder-Mead",
                    options={"maxiter": maxiter, "xatol": 1e-5, "fatol": 1e-6},
                )
                if hasattr(res, "x") and res.x is not None:
                    s = _coeff_objective(res.x, basis, n)
                    _insert_top(s, res.x, top, max_top)
            except Exception:
                pass

    return top


def _random_basis_p(basis, rng, amp=0.42):
    d = basis.shape[0]
    coeff = rng.normal(0.0, amp, size=d)
    if d > 5:
        coeff[5:] *= 0.7
    return _projected_basis_p(basis, coeff)


def _make_weights(q, kind, val):
    q = np.asarray(q, dtype=np.float64)
    if q.size == 0:
        return None
    q = np.maximum(q, 0.0)
    qmax = float(np.max(q))
    if qmax <= 0.0 or not np.isfinite(qmax):
        return None

    if kind == "top":
        K = int(max(1, val))
        if K >= q.size:
            w = np.ones_like(q)
        elif K == 1:
            w = np.zeros_like(q)
            w[int(np.argmax(q))] = 1.0
        else:
            kth = max(0, q.size - K)
            idx = np.argpartition(q, kth)[-K:]
            w = np.zeros_like(q)
            w[idx] = 1.0
    elif kind == "frac":
        frac = float(val)
        if frac >= 1.0:
            w = np.ones_like(q)
        else:
            thresh = qmax * frac
            w = (q >= thresh).astype(np.float64)
    else:
        med = float(np.median(q))
        scale = max(qmax - med, qmax * 0.05, 1e-15)
        beta = float(val) / scale
        w = np.exp(beta * (q - qmax))
        s = float(np.sum(w))
        if s <= 0.0 or not np.isfinite(s):
            return None
        w = w / s

    s = float(np.sum(w))
    if s <= 0.0 or not np.isfinite(s):
        return None
    return w / s


def _local_optimize(p0, deadline, rng):
    p = _positive_normalize(p0)
    n = p.size
    if n < 2:
        s, _ = _prob_score(p)
        return p, float(s if np.isfinite(s) else 2.0)

    score, q = _prob_score(p)
    if not np.isfinite(score):
        p = np.ones(n, dtype=np.float64) / float(n)
        score, q = _prob_score(p)

    best_p = p.copy()
    best_score = float(score)
    uniform = np.ones(n, dtype=np.float64) / float(n)

    bigK = max(3, n // 500)
    schemes = [
        ("top", 1),
        ("frac", 0.85),
        ("exp", 2.0),
        ("exp", 12.0),
        ("frac", 0.65),
        ("top", bigK),
    ]

    stag = 0
    it = 0
    max_iter = 12000

    while time.monotonic() < deadline and it < max_iter:
        if time.monotonic() > deadline - 0.75:
            break

        score, q = _prob_score(p)
        if not np.isfinite(score):
            p = best_p.copy()
            score, q = _prob_score(p)

        if score < best_score - 1e-13:
            best_p = p.copy()
            best_score = float(score)

        improved = False

        for kind, val in schemes:
            w = _make_weights(q, kind, val)
            if w is None:
                continue
            field = _cross_corr(p, w)
            dfield = field - float(np.mean(field))
            scale = float(np.max(np.abs(dfield)))
            if scale <= 1e-15:
                continue

            t_list = (0.05, 0.18, 0.55, 1.6, 3.5)
            best_cand = None
            best_cand_score = float(score)

            for t in t_list:
                cand = p * np.exp(-t * dfield / scale)
                cand = np.maximum(cand, 1e-12)
                ssum = float(np.sum(cand))
                if ssum <= 0.0 or not np.isfinite(ssum):
                    continue
                cand = cand / ssum
                cs, _ = _prob_score(cand)
                if np.isfinite(cs) and cs < best_cand_score - 1e-13:
                    best_cand_score = float(cs)
                    best_cand = cand

            if best_cand is not None:
                p = best_cand
                score = best_cand_score
                improved = True
                stag = 0
                break

        if not improved:
            stag += 1

            if stag % 3 == 0:
                if rng.random() < 0.55:
                    mix = 0.82 * best_p + 0.18 * uniform
                    p = _positive_normalize(mix)
                else:
                    noise = rng.normal(0.0, 0.18, size=n)
                    p = _positive_normalize(p * np.exp(noise))
                score, q = _prob_score(p)
                if np.isfinite(score) and score < best_score:
                    best_p = p.copy()
                    best_score = float(score)

            if stag > 60:
                if best_score < 1.47:
                    break
                p = _positive_normalize(best_p * np.exp(rng.normal(0.0, 0.45, size=n)))
                score, q = _prob_score(p)
                stag = 0

        it += 1
        if it % 200 == 0:
            print("local iter", it, "score", score, "best", best_score, flush=True)

    # Final rapid polish around best_p.
    for kind, val in schemes:
        if time.monotonic() > deadline - 0.25:
            break
        score, q = _prob_score(best_p)
        if not np.isfinite(score):
            continue
        w = _make_weights(q, kind, val)
        if w is None:
            continue
        field = _cross_corr(best_p, w)
        dfield = field - float(np.mean(field))
        scale = float(np.max(np.abs(dfield)))
        if scale <= 1e-15:
            continue

        for t in (0.02, 0.08, 0.3, 1.0, 3.0, 10.0):
            if time.monotonic() > deadline - 0.1:
                break
            cand = best_p * np.exp(-t * dfield / scale)
            cand = np.maximum(cand, 1e-12)
            ssum = float(np.sum(cand))
            if ssum <= 0.0 or not np.isfinite(ssum):
                continue
            cand = cand / ssum
            cs, _ = _prob_score(cand)
            if np.isfinite(cs) and cs < score - 1e-13:
                best_p = cand
                best_score = float(cs)

    return best_p, float(best_score)


def _resample_prob(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n_old = p.size
    if n_old == 0 or n_new <= 0:
        return np.ones(max(1, int(n_new)), dtype=np.float64)
    if n_old == n_new:
        return _positive_normalize(p)

    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    density = float(n_old) * p
    new_density = np.interp(x_new, x_old, density)
    new_density = np.maximum(new_density, 0.0)
    return _positive_normalize(new_density)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Returns the best non-negative sequence found within the time budget.
    """
    start = time.monotonic()
    budget = float(max(0.0, budget_s))
    if budget < 5.0:
        return [1.0] * 128

    deadline = start + max(1.0, budget - 2.0)
    rng = np.random.default_rng(int(seed))

    # Use a length that gives an FFT-friendly full convolution size.
    n = 4096
    d = 9
    basis = _make_basis(n, d)

    best_p = np.ones(n, dtype=np.float64) / float(n)
    best_score, _ = _prob_score(best_p)

    # Parameter-space search for good smooth starts.
    param_deadline = start + min(180.0, max(20.0, 0.15 * budget))
    top = _parameter_search(basis, n, param_deadline, rng)

    starts = []
    if len(top) == 0:
        starts.append(_random_basis_p(basis, rng))
    else:
        for _, coeff in top[:3]:
            p = _projected_basis_p(basis, coeff)
            starts.append(p)

    # Add a couple of random smooth starts for diversity.
    starts.append(_random_basis_p(basis, rng))
    starts.append(_random_basis_p(basis, rng))

    # Remove obvious duplicates by score sequence is not needed; cap to 4.
    starts = starts[:4]

    # Run local optimization on each start, spending most time on the first.
    fractions = (0.66, 0.24, 0.08, 0.04)
    for i, p0 in enumerate(starts):
        if time.monotonic() > deadline - 5.0:
            break
        remaining = max(1.0, deadline - time.monotonic())
        if i < len(fractions):
            alloc = remaining * fractions[i]
        else:
            alloc = remaining * 0.05
        local_deadline = time.monotonic() + max(2.0, alloc)
        local_deadline = min(deadline - 2.0, local_deadline)

        p2, s2 = _local_optimize(p0, local_deadline, rng)
        if np.isfinite(s2) and s2 < best_score:
            best_p = p2
            best_score = float(s2)
        print("candidate", i, "score", s2, "best", best_score, flush=True)

    # If not already comfortably under the target, try a longer sequence once.
    if best_score > 1.5027 and time.monotonic() < deadline - 180.0:
        n2 = 8192
        p2 = _resample_prob(best_p, n2)
        s2, _ = _prob_score(p2)
        rescue_deadline = time.monotonic() + min(900.0, max(120.0, 0.35 * budget))
        rescue_deadline = min(deadline - 2.0, rescue_deadline)
        p3, s3 = _local_optimize(p2, rescue_deadline, rng)
        if np.isfinite(s3) and s3 < best_score:
            best_p = p3
            best_score = float(s3)
        print("rescue longer", s3, "best", best_score, flush=True)

    # Final exact/evaluator sanity check, but keep the sequence that our fast
    # objective selected.
    try:
        exact = float(evaluate_sequence([float(x) for x in best_p]))
        if np.isfinite(exact):
            print("final exact evaluator score", exact, flush=True)
    except Exception:
        pass

    return [float(x) for x in best_p]
