def frank_wolfe_step(p, n_active_max=20):
    """One Frank-Wolfe step to minimize max(p*p)."""
    n = len(p)
    conv = np.convolve(p, p)
    max_val = np.max(conv)
    # Find active set (indices where conv is close to max)
    active = np.where(conv >= max_val * 0.999 + max_val * 1e-10)[0]
    if len(active) == 0:
        active = np.where(conv == max_val)[0]
    
    # Compute subgradient
    s = np.zeros(n)
    for j in active:
        for k in range(n):
            if 0 <= j - k < n:
                s[k] += 2 * p[j - k]
    s /= len(active)
    
    # Find linear minimizer (vertex of simplex)
    k_star = np.argmin(s)
    return k_star, s, max_val
