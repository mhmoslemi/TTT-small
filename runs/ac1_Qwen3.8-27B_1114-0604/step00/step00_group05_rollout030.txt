from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def build_convolution_matrix(a):
    """Build the matrix C such that C @ g = convolve(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def project_simplex(v):
    """Project v onto the standard simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Bisection on lambda
    lo = min(v)
    hi = max(v)
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > 1:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.maximum(v - lam, 0)

    """Compute score assuming sum(a)=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * float(np.max(b))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Compute gradient of tau * logsumexp(convolve(a,a)/tau)."""

    a_rev = a[::-1]

    grad = 2.0 * conv[n-1 : 2*n-1]
