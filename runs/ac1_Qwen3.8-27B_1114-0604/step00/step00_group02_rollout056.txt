"""
Optimization for a step-function sequence minimizing
    2*n*max(convolve(a,a)) / (sum(a)**2).

Approach:
1. Build a low-dimensional smooth positive family using orthonormal
   polynomial/trig features, and optimize a lenient-logit (LSE) surrogate
   with L-BFGS-B if available.  This escapes the trivial uniform minimum.
2. Use the best smooth candidate as a seed for full-dimensional projected
   mirror-style optimization in log-space.  The surrogate minimizes a
   log-sum-exp of the convolution peaks and is driven by the softmax of
   the current convolution lags.
3. Run several discrete seeds with different time slices, then optionally
   upscale the best sequence to a larger length and continue optimizing.
"""

import math
import time
import numpy as np

try:
    from scipy.optimize import minimize as scipy_minimize
except Exception:
    scipy_minimize = None


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_seq(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if (not math.isfinite(s)) or s <= 0.0:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def softmax_stable(v):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v.copy()
    m = float(np.max(v))
    w = np.exp(v - m)
    s = float(np.sum(w))
    if s <= 0.0 or not math.isfinite(s):
        return np.ones_like(v, dtype=np.float64) / float(v.size)
    return w / s


def fft_full_conv(a):
    n = len(a)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    need = 2 * n - 1
    M = next_pow2(need)
    A = np.fft.rfft(a, M)
    b = np.fft.irfft(A * A, M)[:need]
    b = np.maximum(b, 0.0)
    return b


def fast_score(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return float("inf")
    s = float(np.sum(a))
    if (not math.isfinite(s)) or s < 0.01:
        return float("inf")
    b = fft_full_conv(a)
    mb = float(np.max(b)) if b.size else 0.0
    if not math.isfinite(mb) or mb < 0.0:
        return float("inf")
    return float(2 * len(a) * mb / (s * s))


def corr_shifts(a, e):
    """
    Return q[i] = sum_m a[m] e[i+m], where e has length 2*n-1.
    This is the needed derivative kernel for the convolution objective.
    """
    n = len(a)
    L = 2 * n - 1
    need = 3 * n - 2
    M = next_pow2(need)
    A = np.fft.rfft(a, M)
    E = np.fft.rfft(e[::-1], M)
    conv = np.fft.irfft(A * E, M)[:3 * n - 2]
    return conv[n - 1:2 * n - 1][::-1]


def resample_seq(a, n):
    a = normalize_seq(a)
    if len(a) == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, len(a))
    x_new = np.linspace(0.0, 1.0, int(n))
    out = np.interp(x_new, x_old, a)
    return normalize_seq(out)


def make_orth_features(n, K):
    x = np.linspace(0.0, 1.0, int(n))
    t = x - 0.5
    cols = [t, t * t, t * t * t]
    for m in range(1, int(K) + 1):
        w = 2.0 * math.pi * m * x
        cols.append(np.cos(w))
        cols.append(np.sin(t * 2.0 * math.pi * m))
    F = np.column_stack(cols)
    F -= np.mean(F, axis=0)
    try:
        Q, R = np.linalg.qr(F)
    except Exception:
        # Last-resort: just normalize columns.
        norms = np.linalg.norm(F, axis=0)
        norms[norms < 1e-12] = 1.0
        return F / norms
    diag = np.abs(np.diagonal(R))
    if diag.size:
        mask = diag > (1e-9 * float(max(1.0, np.max(diag))))
    else:
        mask = np.ones(F.shape[1], dtype=bool)
    if mask.sum() > 0:
        return Q[:, mask]
    return np.ones((len(F), 1), dtype=np.float64)


def coef_fun(c, Q, beta):
    c = np.asarray(c, dtype=np.float64).reshape(-1)
    if not np.all(np.isfinite(c)):
        return 1e100
    a = softmax_stable(Q.dot(c))
    b = fft_full_conv(a)
    if b.size == 0 or not np.all(np.isfinite(b)):
        return 1e100
    m = float(np.max(b))
    w = np.exp(beta * b - beta * m)
    s = float(np.sum(w))
    if s <= 0.0 or not math.isfinite(s):
        return 1e100
    return float(beta * m + math.log(s))


def coef_jac(c, Q, beta):
    c = np.asarray(c, dtype=np.float64).reshape(-1)
    d = Q.shape[1]
    if not np.all(np.isfinite(c)):
        return np.zeros(d, dtype=np.float64)
    a = softmax_stable(Q.dot(c))
    b = fft_full_conv(a)
    e = softmax_stable(beta * b)
    q = 2.0 * beta * corr_shifts(a, e)
    w = a * q
    mu = float(np.sum(w))
    s = Q.T.dot(a)
    fw = Q.T.dot(w)
    grad = fw - mu * s
    if not np.all(np.isfinite(grad)):
        return np.zeros(d, dtype=np.float64)
    return grad


def optimize_discrete(seed, deadline, seed_int=0,
                      beta_start=30.0, beta_end=120.0,
                      base_lr0=0.45, noise0=0.60):
    """
    Full-dimensional mirror/LSE optimizer.  Works directly on log(a)
    and uses a line search over a few step sizes with mild annealing.
    """
    start_t = time.monotonic()
    a = normalize_seq(seed)
    n = len(a)
    if n == 0:
        return np.ones(1, dtype=np.float64), float("inf")

    loga = np.log(np.maximum(a, 1e-12))
    loga -= float(np.mean(loga))

    best_a = a.copy()
    best_sc = fast_score(a)
    cur_sc = best_sc
    rng = np.random.default_rng(int(seed_int))

    total_s = max(1.0, float(deadline) - start_t)
    lr_scale = 1.0
    reject_count = 0
    last_best_t = start_t
    factors = (0.30, 1.00, 2.50)

    while time.monotonic() < float(deadline):
        elapsed = time.monotonic() - start_t
        frac = min(1.0, elapsed / total_s)

        beta = beta_start + (beta_end - beta_start) * frac
        lr = base_lr0 * lr_scale * (1.0 - 0.45 * frac)
        if lr < 0.025:
            lr = 0.025
        noise = noise0 * max(0.05, (1.0 - frac) ** 2)

        a_now = np.exp(loga - float(np.max(loga)))
        s_now = float(np.sum(a_now))
        if s_now <= 0.0 or not math.isfinite(s_now):
            loga = np.log(np.maximum(best_a, 1e-12))
            loga -= float(np.mean(loga))
            cur_sc = best_sc
            continue
        a_now /= s_now

        b_now = fft_full_conv(a_now)
        e_now = softmax_stable(beta * b_now)
        q = corr_shifts(a_now, e_now)

        q = q - float(np.mean(q))
        qstd = float(np.std(q))
        if qstd > 1e-12:
            q = q / qstd
        else:
            q = rng.standard_normal(n)

        noise_vec = rng.standard_normal(n) if noise > 1e-5 else None

        best_cand_sc = float("inf")
        best_cand_loga = None
        best_cand_a = None

        for f in factors:
            cand_loga = loga - lr * f * q
            if noise_vec is not None:
                cand_loga = cand_loga + (noise * math.sqrt(f)) * noise_vec
            cand_loga -= float(np.mean(cand_loga))
            cand_loga = np.clip(cand_loga, -50.0, 50.0)

            cand = np.exp(cand_loga)
            cs = float(np.sum(cand))
            if cs <= 0.0 or not math.isfinite(cs):
                continue
            cand /= cs
            sc = fast_score(cand)
            if sc < best_cand_sc:
                best_cand_sc = sc
                best_cand_loga = cand_loga
                best_cand_a = cand

        if best_cand_loga is None:
            reject_count += 1
            continue

        T = 0.02 * (1.0 - frac) + 0.0005
        accept = False
        if best_cand_sc <= cur_sc + 1e-12:
            accept = True
        else:
            delta = float(best_cand_sc - cur_sc)
            if delta < 700.0 * T:
                if math.exp(-delta / T) > float(rng.random()):
                    accept = True

        if accept:
            loga = best_cand_loga
            a_now = best_cand_a
            cur_sc = best_cand_sc
            reject_count = 0
            if cur_sc < best_sc - 1e-12:
                best_sc = cur_sc
                best_a = a_now.copy()
                last_best_t = time.monotonic()
        else:
            reject_count += 1

        if reject_count > 140:
            loga = np.log(np.maximum(best_a, 1e-12))
            loga -= float(np.mean(loga))
            cur_sc = best_sc
            reject_count = 0
            lr_scale = max(0.55, lr_scale * 0.75)
        elif time.monotonic() - last_best_t > 120.0:
            kick = 0.32 + 0.08 * min(3.0, (time.monotonic() - last_best_t) / 120.0)
            loga = np.log(np.maximum(best_a, 1e-12))
            loga = loga + kick * rng.standard_normal(n)
            loga -= float(np.mean(loga))
            loga = np.clip(loga, -50.0, 50.0)
            a_try = np.exp(loga)
            st = float(np.sum(a_try))
            if st > 0.0 and math.isfinite(st):
                a_try /= st
                cur_sc = fast_score(a_try)
                reject_count = 0
            last_best_t = time.monotonic()

    return best_a, best_sc
