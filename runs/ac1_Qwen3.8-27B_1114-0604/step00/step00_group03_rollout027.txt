def self_convolution(f):
    n = len(f)
    N = 1
    while N < 2*n - 1:
        N *= 2
    F = np.fft.fft(f, N)
    c = np.fft.ifft(F * F).real

def sa_refine(f, deadline, rng, verbose=False):
    """Simulated annealing refinement."""
    n = len(f)
    f = f.copy().astype(np.float64)
    c = self_convolution(f)
    M = float(np.max(c))
    best_f = f.copy()
    best_M = M
    
    # Temperature schedule
    T_current = M * 0.05
    
    iteration = 0
    no_improve = 0
    while time.monotonic() < deadline:
        batch_size = 1000
        for _ in range(batch_size):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            
            # Perturbation: transfer mass from i to j
            # Scale perturbation to be relative to f[i]
            delta = f[i] * 0.02 * (1.0 + float(rng.random()))
            if delta > f[i]:
                delta = f[i] * 0.5
            if delta < 1e-18:
                continue
            
            # Compute update
            k_start = min(i, j)
            k_end = min(max(i + n - 1, j + n - 1), 2 * n - 2)
            k_range = np.arange(k_start, k_end + 1)
            lk = len(k_range)
            
            dc = np.zeros(lk)
            mi = (k_range >= i) & (k_range < i + n)
            mj = (k_range >= j) & (k_range < j + n)
            dc[mi] -= 2.0 * delta * f[k_range[mi] - i]
            dc[mj] += 2.0 * delta * f[k_range[mj] - j]
            
            old_vals = c[k_range].copy()
            c[k_range] += dc
            new_M = float(np.max(c))

                    best_f = f.copy()
