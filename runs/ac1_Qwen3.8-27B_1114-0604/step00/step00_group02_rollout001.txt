import numpy as np
import time

def project_nonneg_simplex(a, target_sum=1.0):
    """Project a onto the non-negative simplex {x: x >= 0, sum(x) = target_sum}."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    # Use the water-filling algorithm
    if a.sum() <= 0:
        return np.full(n, target_sum / n)
    # Sort in descending order
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target_sum)
    rho_hat = rho[cond][-1]
    theta = (cssv[cond][-1] - target_sum) / rho_hat
    return np.maximum(a - theta, 0)
