    # g = convolve(w, u_flipped)[n-1 : 2n-1]
    u_flipped = u[::-1]
    conv_len = (2*n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Wf = np.fft.rfft(w, fft_size)
    Uf = np.fft.rfft(u_flipped, fft_size)
    conv = np.fft.irfft(Wf * Uf, fft_size)
    return conv[n-1 : 2*n-1]
