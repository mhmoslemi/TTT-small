def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        A[k, i_min:i_max + 1] = a[i_max - k : k - i_min + 1][::-1]
    return A


def score_of(a):
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * len(a) * conv.max() / (s * s)


def one_lp_step(a, time_limit=20.0):
    n = len(a)
    conv_a = np.convolve(a, a)

        return None
    return np.asarray(res.x, dtype=np.float64)

"""
LP-iterated peak autocorrelation minimization.
Solves at each step an LP to find the most spread-out sequence b
compatible with current a (bounded cross-correlation), then updates a
toward b. Multiple non-uniform starts, line search, perturbation on stall,
and a final gradient polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A[k,i] = a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        A[k, i_min:i_max + 1] = a[i_max - k: k - i_min + 1][::-1]
    return A


def score_of(a):
    """Compute the evaluation score without calling evaluate_sequence."""
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * len(a) * conv.max() / (s * s)


def one_lp_step(a, time_limit=25.0):
    """Solve max sum(b) s.t. (a*b)[k]<=M, b>=0. Returns b or None."""
    n = len(a)
    conv_a = np.convolve(a, a)
    M = float(conv_a.max())
    A = build_conv_matrix(a)
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': time_limit, 'presolve': True})
    if not res.success or res.x is None:
        return None
    return np.asarray(res.x, dtype=np.float64)


def gradient_polish(a, budget_s=120.0, init_eta=0.05):
    """Simple gradient-based polish using the subgradient of max autocorrelation."""
    n = len(a)
    a = a / a.sum()
    eta = init_eta
    t0 = time.monotonic()
    while time.monotonic() - t0 < budget_s:
        conv = np.convolve(a, a)
        M = conv.max()
        if M < 1e-30:
            break
        # Soft-max weights over the convolution
        temp = 15.0
        w = np.exp(temp * (conv - M) / M)
        w /= w.sum()
        # grad[i] = 4 * sum_{j=0}^{n-1} w[i+j] * a[j]
        # = 4 * np.convolve(w, a[::-1], 'full')[i + n - 1]
        c = np.convolve(w, a[::-1], mode='full')
        grad = 4.0 * c[n - 1: 2 * n - 1]
        # Normalized step
        gn = np.linalg.norm(grad)
        if gn < 1e-30:
            break
        step = eta * grad / gn
        a_new = a - step
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s
        sc_old = score_of(a)
        sc_new = score_of(a_new)
        if sc_new < sc_old:
            a = a_new
            eta = min(eta * 1.05, 0.5)
        else:
            eta *= 0.7
            if eta < 1e-6:
                break
    return a


def make_starts(rng):
    """Generate diverse non-uniform starting sequences."""
    starts = []
    # Given sequence
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        init = np.clip(init, 0.0, None)
        if init.sum() > 0:
            starts.append(init / init.sum())
    except Exception:
        pass
    for n in [2048, 3072, 4096]:
        # Cosine bump
        t = np.linspace(0, 1, n)
        r = 0.5 + 0.5 * np.cos(np.pi * t)
        r = np.clip(r, 0.001, None)
        starts.append(r / r.sum())
        # Two bumps
        r = 0.5 + 0.4 * np.cos(2 * np.pi * t)
        r = np.clip(r, 0.001, None)
        starts.append(r / r.sum())
        # Triangle
        r = 1.0 - np.abs(2.0 * t - 1.0)
        r = np.clip(r, 0.001, None)
        starts.append(r / r.sum())
        # Random smooth via IFFT of low-freq noise
        spec = np.zeros(n // 2 + 1, dtype=np.complex128)
        freqs = np.arange(n // 2 + 1)
        mask = freqs <= 5
        spec[mask] = rng.standard_normal(mask.sum()) + 1j * rng.standard_normal(mask.sum())
        r = np.abs(np.fft.irfft(spec, n=n))
        r = np.clip(r, 0.001, None)
        starts.append(r / r.sum())
    return starts


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: LP-iterated optimization with multiple starts."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 10.0

    best_a = None
    best_score = np.inf

    starts = make_starts(rng)
    # Sort by initial score
    starts.sort(key=lambda x: score_of(np.clip(x, 0, None)))

    print(f"Initialized {len(starts)} starting points, best initial={score_of(np.clip(starts[0],0,None)):.6f}", flush=True)

    max_n = 4096
    time_per_start = max(120.0, (deadline - t0) / max(1, len(starts)) * 0.8)

    for si, a_start in enumerate(starts):
        if time.monotonic() > deadline - 120.0:
            break

        a = np.clip(a_start, 0.0, None).copy()
        n = len(a)
        if n > max_n:
            a = np.interp(np.linspace(0, 1, max_n), np.linspace(0, 1, n), a)
            n = max_n
        a = a / a.sum()

        sc = score_of(a)
        start_clk = time.monotonic()
        print(f"[{si}] n={n} init_score={sc:.6f} budget={time_per_start:.0f}s", flush=True)

        stall = 0
        it = 0
        while time.monotonic() - start_clk < time_per_start and time.monotonic() < deadline - 120.0:
            it += 1
            b = one_lp_step(a, time_limit=25.0)
            if b is None:
                break

            sum_b = b.sum()
            sum_a = a.sum()

            if sum_b <= sum_a * (1.0 + 1e-9):
                stall += 1
                if stall >= 5:
                    break
                a = a + 0.03 * rng.standard_normal(n)
                a = np.clip(a, 0.0, None)
                if a.sum() < 1e-10:
                    break
                a = a / a.sum()
                continue

            b = b * (sum_a / sum_b)
            b = np.clip(b, 0.0, None)
            sb = b.sum()
            if sb < 1e-10:
                break
            b = b / sb

            # Line search
            best_t = 0.0
            best_sc_t = sc
            for tv in [1.0, 0.7, 0.5, 0.33, 0.2, 0.1]:
                a_try = (1.0 - tv) * a + tv * b
                a_try = np.clip(a_try, 0.0, None)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = score_of(a_try)
                if sc_try < best_sc_t:
                    best_sc_t = sc_try
                    best_t = tv

            if best_t > 0.0:
                a = (1.0 - best_t) * a + best_t * b
                a = np.clip(a, 0.0, None)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                sc = best_sc_t
                stall = 0
            else:
                stall += 1
                if stall >= 8:
                    a = a + 0.08 * rng.standard_normal(n)
                    a = np.clip(a, 0.0, None)
                    if a.sum() < 1e-10:
                        break
                    a = a / a.sum()
                    sc = score_of(a)
                    stall = 0
                    continue
                break

            # Update best
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"  [{si}] it={it} NEW BEST={sc:.8f} t={time.monotonic()-t0:.0f}s", flush=True)
            elif it % 30 == 0:
                print(f"  [{si}] it={it} score={sc:.6f} t={time.monotonic()-t0:.0f}s", flush=True)

    # Gradient polish on best
    if best_a is not None and time.monotonic() < deadline - 30.0:
        remain = min(180.0, deadline - 30.0 - time.monotonic())
        print(f"Polishing with {remain:.0f}s remaining, current best={best_score:.8f}", flush=True)
        a_pol = gradient_polish(best_a.copy(), budget_s=remain, init_eta=0.05)
        sc_pol = score_of(a_pol)
        if sc_pol < best_score:
            best_score = sc_pol
            best_a = a_pol
            print(f"  POLISH IMPROVED: {sc_pol:.8f}", flush=True)

    if best_a is None:
        best_a = np.ones(256, dtype=np.float64)

    print(f"FINAL best_score={best_score:.8f} n={len(best_a)} time={time.monotonic()-t0:.1f}s", flush=True)
    return [float(x) for x in best_a]
