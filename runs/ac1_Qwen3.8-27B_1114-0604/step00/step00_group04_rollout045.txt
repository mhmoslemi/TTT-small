"""
Multi-phase optimization for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-based spectral flattening (Gerchberg-Saxton), subgradient descent,
and diverse local search with simulated annealing.
"""

import numpy as np
import time


def _fast_evaluate(a):
    """Compute objective without validation overhead. Assumes a is valid."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _fft_convolve(a):
    """Compute linear convolution using FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    conv = np.real(np.fft.irfft(A * A, N))[:2 * n - 1]
    return conv


def _fft_size(n):
    """Next power of 2 >= 2n-1."""
    N = 1
    while N < 2 * n - 1:
        N *= 2
    return N


def _gs_flatten(a, n_iter, alpha_start, alpha_end, rng=None):
    """
    Gerchberg-Saxton style iterative flattening of the power spectrum.
    Alternates between flattening the frequency amplitude and clipping to non-negative.
    """
    n = len(a)
    N = _fft_size(n)
    a = a.astype(np.float64).copy()

    for i in range(n_iter):
        frac = i / max(1, n_iter - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        A = np.fft.rfft(a, N)
        amp = np.abs(A)
        phase = A / (amp + 1e-30)

        mean_amp = amp.mean()
        target = np.full_like(amp, mean_amp)
        new_amp = np.exp(alpha * np.log(amp + 1e-30) + (1.0 - alpha) * np.log(target + 1e-30))

        A_new = new_amp * phase
        a_new = np.real(np.fft.irfft(A_new, N)[:n])
        a_new = np.clip(a_new, 0.0, None)

        if np.sum(a_new) < 1e-10:
            return a
        a = a_new

    return a


def _subgradient_optimize(a, n_steps, step_init, momentum=0.9):
    """
    Subgradient descent targeting the current convolution peak.
    Returns best sequence and score found during the run.
    """
    n = len(a)
    N = _fft_size(n)
    a = a.astype(np.float64).copy()
    best = a.copy()
    best_score = _fast_evaluate(best)

    momentum_arr = np.zeros(n)

    for i in range(n_steps):
        s = step_init * (0.97 ** i)

        A = np.fft.rfft(a, N)
        conv = np.real(np.fft.irfft(A * A, N))[:2 * n - 1]
        max_val = np.max(conv)

        active = np.where(conv >= 0.99 * max_val)[0]

        grad = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_start <= j_end:
                m_start = k - j_end
                m_end = k - j_start
                grad[j_start:j_end + 1] += a[m_start:m_end + 1][::-1]
        grad *= 2.0 / len(active)

        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            grad /= grad_norm

        momentum_arr = momentum * momentum_arr - s * grad
        a_new = a + momentum_arr
        a_new = np.clip(a_new, 0.0, None)

        if np.sum(a_new) < 1e-10:
            break

        a = a_new
        score = _fast_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()

    return best, best_score

def _make_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    i = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n))

    # Geometric decreasing
    inits.append(0.999 ** i)

    # Linear decreasing
    inits.append(np.linspace(2.0, 0.5, n))

    # Linear increasing
    inits.append(np.linspace(0.5, 2.0, n))

    # Cosine modulated (various frequencies)

    # Gaussian window
    sigma = n / 4.0
    inits.append(np.exp(-(i - n / 2.0) ** 2 / (2.0 * sigma ** 2)))

    # Reciprocal
    inits.append(1.0 / (i + 1.0))

    # Random uniform
    inits.append(rng.uniform(0.2, 1.0, n))

    # Random exponential + offset
    inits.append(rng.exponential(1.0, n) + 0.1)

    # Two-level alternating
    base = np.ones(n)
    base[::2] = 1.5
    base[1::2] = 0.5
    inits.append(base)

    # Sawtooth
    period = max(2, n // 10)
    inits.append(1.0 + 0.5 * (i % period) / period)

    # Quadratic peak (asymmetric)
    inits.append(np.exp(-(i / n) ** 2 * 4.0))

    return inits
