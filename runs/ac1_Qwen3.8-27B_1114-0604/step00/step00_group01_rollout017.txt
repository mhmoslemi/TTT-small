LP-based iterative optimization for minimizing the auto-correlation peak.
Uses the approach: given current sequence a, solve LP to find best direction g,
then blend a towards g. Repeat with decreasing step size.

import numpy as np
import time
import scipy.optimize as opt
from scipy.sparse import csr_matrix


def build_conv_matrix(a):
    """Build the sparse matrix A where (A @ g)[k] = (a * g)[k]."""
    n = len(a)
    # Column j has non-zeros at rows k from j to j+n-1, with values a[0], a[1], ..., a[n-1]
    cols = []
    rows = []
    vals = []
    for j in range(n):
        # rows k from j to min(2n-2, j+n-1)
        hi = min(2*n - 2, j + n - 1)
        length = hi - j + 1
        rows.extend(range(j, hi + 1))
        cols.extend([j] * length)
        vals.extend(a[:length])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n - 1, n))
    return A


def solve_lp_direction(a, M):
    """Solve LP: max 1^T g s.t. A@g <= M*1, g >= 0. Returns g."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = M * np.ones(2*n - 1)
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

    n = len(a)
    conv = np.convolve(a, a)
    # Start with the top few constraints
    top_k = np.argsort(conv)[-min(10, 2*n-1):]
    active = set(top_k.tolist())

"""
LP-based iterative optimization for minimizing auto-correlation peak of non-negative sequences.
Uses active-set LP to find descent directions, with multiple restarts for robustness.
"""

import numpy as np
import time
import scipy.optimize as opt
from scipy.sparse import csr_matrix


def build_conv_submatrix(a, K):
    n = len(a)
    m = len(K)
    if m == 0:
        return csr_matrix((0, n))
    row_idx = []
    col_idx = []
    data = []
    for i, k in enumerate(K):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi < lo:
            continue
        for j in range(lo, hi + 1):
            row_idx.append(i)
            col_idx.append(j)
            data.append(a[k - j])
    if not row_idx:
        return csr_matrix((m, n))
    return csr_matrix((data, (row_idx, col_idx)), shape=(m, n))


def lp_find_direction(a, M, max_active=50, time_budget=30):
    """Find direction g using active-set LP."""
    n = len(a)
    start_time = time.monotonic()
    
    # Initial active set: top constraints
    conv = np.convolve(a, a)
    top_indices = np.argsort(conv)[-min(20, 2*n-1):]
    active = list(set(top_indices.tolist()))
    
    g = None
    for iteration in range(max_active):
        if time.monotonic() - start_time > time_budget:
            break
        
        A = build_conv_submatrix(a, active)
        c = -np.ones(n)
        b = M * np.ones(len(active))
        
        try:
            res = opt.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        except Exception:
            break
        
        if not res.success:
            break
        
        g = res.x
        
        # Check which constraints are violated
        new_conv = np.convolve(a, g)
        violated = np.where(new_conv > M * (1 + 1e-6))[0]
        if len(violated) == 0:
            break
        # Add up to 5 most violated
        worst = violated[np.argsort(new_conv[violated])[-5:]]
        new_active = set(active) | set(worst.tolist())
        if new_active == set(active):
            break
        active = list(new_active)
    
    return g
