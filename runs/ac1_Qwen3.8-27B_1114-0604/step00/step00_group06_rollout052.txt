```
def make_mass_transfer(p, src_start, src_len, dst_start, dst_len, alpha, proportional=True):

    sm = src.sum()

    # subtract proportional
```

```python
import numpy as np
import time


"""
Search for a non-negative sequence minimizing

    2 * n * max(conv(a, a)) / sum(a)**2

by starting from strong structured initializers (especially the finite-binomial
sequence whose first-half convolution is perfectly flat), then applying:
  1. a dense grid of mass-transfer moves,
  2. a block-weight refinement search,
  3. a long randomized local search with mass-transfer/ratio/tilt/sine moves.

The evaluator is call-only; all main scoring uses a fast FFT convolution.
"""


def next_power_of_two(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    if a is None:
        return None
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return None
    a = np.nan_to_num(a, nan=0.0, posinf=1.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 0.0:
        return None
    a = a / s
    return a


def resample_to_length(arr, n):
    arr = np.asarray(arr, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=1.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.size == n:
        return arr.copy()
    if arr.size == 1:
        return np.full(n, float(arr[0]))
    xold = np.linspace(0.0, 1.0, arr.size)
    xnew = np.linspace(0.0, 1.0, n)
    return np.interp(xnew, xold, arr)


def fast_score(a, n, N, m):
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    if abs(s - 1.0) > 1e-12:
        a = a / s
    F = np.fft.rfft(a, N)
    F = F * F
    c = np.fft.irfft(F, N)
    mx = float(np.max(c[:m]))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * n * mx)


def binomial_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * float(2 * i - 1) / float(2 * i)
    return a


def generate_initial_candidates(n, N, m, prev):
    out = []

    b = normalize_positive(binomial_sequence(n))
    if b is not None:
        out.append(b)
    br = normalize_positive(b[::-1].copy())
    if br is not None:
        out.append(br)

    i = np.arange(n, dtype=np.float64)
    if n > 1:
        x = i / float(n - 1)
    else:
        x = np.zeros(1, dtype=np.float64)

    for alpha in (0.30, 0.40, 0.45, 0.50, 0.55, 0.65, 0.75):
        a = np.power(i + 1.0, -alpha)
        out.append(normalize_positive(a))

    for lam in (0.5, 1.0, 2.0, 5.0, 10.0, 20.0):
        out.append(normalize_positive(np.exp(-lam * x)))

    for mode in (1, 2, 3):
        for amp in (0.2, 0.4, 0.6, 0.8):
            a1 = 1.0 + amp * np.sin(2.0 * np.pi * mode * x)
            a2 = 1.0 + amp * np.cos(2.0 * np.pi * mode * x)
            out.append(normalize_positive(np.clip(a1, 1e-8, None)))
            out.append(normalize_positive(np.clip(a2, 1e-8, None)))

    if prev is not None:
        rp = resample_to_length(prev, n)
        if rp is not None:
            out.append(normalize_positive(rp))

    return out


def apply_mass_transfer(cand, sr, sl, dr, dl, alpha, uniform_dest=False):
    if sl <= 0 or dl <= 0 or alpha <= 0.0:
        return False
    src = cand[sr:sr + sl]
    sm = float(src.sum())
    if not np.isfinite(sm) or sm <= 0.0:
        return False
    transfer = float(alpha) * sm
    cand[sr:sr + sl] = src * (1.0 - alpha)
    dst = cand[dr:dr + dl]
    dm = float(dst.sum())
    if uniform_dest or dm <= 1e-300:
        cand[dr:dr + dl] = dst + transfer / float(dl)
    else:
        cand[dr:dr + dl] = dst * (1.0 + transfer / dm)
    return True


def finish_candidate(cand):
    if cand is None:
        return None
    cand = np.asarray(cand, dtype=np.float64)
    np.maximum(cand, 0.0, out=cand)
    s = float(cand.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    cand = cand / s
    return cand


def move_early_late(p, rng, n, alpha_max):
    cand = p.copy()
    low = max(1, n // 2000)
    src_hi = max(low + 1, n // 3)
    dst_hi = max(low + 1, n // 2)
    ls = int(rng.integers(low, src_hi))
    ld = int(rng.integers(low, dst_hi))
    if ls + ld > n:
        ld = n - ls
    if ls <= 0 or ld <= 0:
        return None
    alpha = float(rng.uniform(0.01, alpha_max))
    uniform_dest = bool(rng.random() < 0.35)
    apply_mass_transfer(cand, 0, ls, n - ld, ld, alpha, uniform_dest)
    return finish_candidate(cand)


def move_random_transfer(p, rng, n):
    cand = p.copy()
    half = n // 2
    low = max(1, n // 5000)
    high = max(low + 1, n // 15)
    sl = int(rng.integers(low, high))
    if half <= sl:
        return None
    sr = int(rng.integers(0, half - sl + 1))
    dl = int(rng.integers(low, high))
    min_dr = max(half, sr + sl)
    max_dr = n - dl + 1
    if min_dr >= max_dr:
        return None
    dr = int(rng.integers(min_dr, max_dr))
    alpha = float(rng.uniform(0.01, 0.55))
    uniform_dest = bool(rng.random() < 0.25)
    apply_mass_transfer(cand, sr, sl, dr, dl, alpha, uniform_dest)
    return finish_candidate(cand)


def move_ratio(p, rng, n, sigma):
    cand = p.copy()
    low = max(1, n // 5000)
    high = max(low + 1, n // 10)
    l1 = int(rng.integers(low, high))
    l2 = int(rng.integers(low, high))
    s1 = int(rng.integers(0, max(1, n - l1 + 1)))
    s2 = int(rng.integers(0, max(1, n - l2 + 1)))
    if s2 < s1 + l1 and s1 < s2 + l2:
        return None
    delta = float(np.clip(rng.normal(0.0, sigma), -2.0, 2.0))
    f = float(np.exp(delta))
    cand[s1:s1 + l1] *= f
    cand[s2:s2 + l2] *= 1.0 / f
    return finish_candidate(cand)


def move_single(p, rng, n, sigma):
    cand = p.copy()
    low = max(1, n // 5000)
    high = max(low + 1, n // 10)
    l = int(rng.integers(low, high))
    s = int(rng.integers(0, max(1, n - l + 1)))
    f = float(np.exp(np.clip(rng.normal(0.0, sigma), -2.0, 2.0)))
    cand[s:s + l] *= f
    return finish_candidate(cand)


def move_tilt(p, rng, x, sigma):
    cand = p.copy()
    t = float(np.clip(rng.normal(0.0, sigma * 0.6), -2.0, 2.0))
    cand *= np.exp(t * (x - 0.5))
    return finish_candidate(cand)


def make_sine_cache(n):
    M = 4
    cache = np.empty((M, 2, n), dtype=np.float64)
    js = np.arange(n, dtype=np.float64) / float(n)
    for k in range(M):
        mode = k + 1
        angle = 2.0 * np.pi * mode * js
        cache[k, 0, :] = np.sin(angle)
        cache[k, 1, :] = np.cos(angle)
    return cache


def move_sine(p, rng, sine_cache, sigma):
    if sine_cache is None:
        return None
    cand = p.copy()
    k = int(rng.integers(0, sine_cache.shape[0]))
    amp = float(np.clip(rng.uniform(0.05, 0.80), 0.0, 0.95))
    phase = float(rng.uniform(0.0, 2.0 * np.pi))
    cp = float(np.cos(phase))
    sp = float(np.sin(phase))
    mod = 1.0 + amp * (sine_cache[k, 0] * cp + sine_cache[k, 1] * sp)
    cand *= mod
    return finish_candidate(cand)


def grid_mass_transfers(p, best_score, n, N, m, deadline, rng):
    src_fracs = (0.01, 0.02, 0.05, 0.10, 0.20, 0.30)
    dst_starts = (0.30, 0.40, 0.50, 0.60, 0.70, 0.80)
    dst_lens = (0.10, 0.20, 0.30, 0.50)
    alphas = (0.05, 0.10, 0.15, 0.20, 0.25, 0.35)

    for sf in src_fracs:
        ls = max(1, int(sf * n))
        for ds in dst_starts:
            dr = int(ds * n)
            for dl in dst_lens:
                dlen = max(1, int(dl * n))
                dend = min(n, dr + dlen)
                dlen = dend - dr
                if dr < ls or dlen <= 0:
                    continue
                for un ifalse in (0, 1):
                    uniform_dest = bool(un ifalse)
                    for alpha in alphas:
                        if time.monotonic() >= deadline:
                            return p, best_score
                        cand = p.copy()
                        apply_mass_transfer(cand, 0, ls, dr, dlen, alpha, uniform_dest)
                        cand = finish_candidate(cand)
                        if cand is None:
                            continue
                        sc = fast_score(cand, n, N, m)
                        if np.isfinite(sc) and sc < best_score - 1e-12:
                            p = cand
                            best_score = sc
                            print("grid mass transfer: %.8f" % best_score)
    return p, best_score


def make_block_idx(n, B):
    if B <= 1:
        return np.zeros(n, dtype=np.int64)
    return (np.arange(n, dtype=np.int64) * np.int64(B) // np.int64(n)).astype(np.int64)


def theta_candidate(base, block_idx, theta):
    th = np.clip(theta, -5.0, 5.0)
    a = base * np.exp(th[block_idx])
    s = float(a.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-300:
        return base.copy()
    return a / s


def theta_starts(B):
    starts = []
    starts.append(np.zeros(B, dtype=np.float64))
    if B > 1:
        js = np.linspace(0.0, 1.0, B, dtype=np.float64)
    else:
        js = np.array([0.0], dtype=np.float64)

    for c in (0.20, 0.50, 1.00, 2.00):
        starts.append(c * (js - 0.5))
        starts.append(-c * (js - 0.5))

    idx = np.arange(B, dtype=np.float64)
    for c in (0.30, 0.80, 1.50):
        th = np.zeros(B, dtype=np.float64)
        q1 = max(1, B // 4)
        th[:q1] = -0.70 * c
        th[-q1:] = c
        starts.append(th)

        th = np.zeros(B, dtype=np.float64)
        q2 = max(1, B // 8)
        th[:q2] = -c
        th[-q2:] = c
        starts.append(th)

    for c in (0.30, 0.80, 1.50):
        starts.append(c * np.sin(2.0 * np.pi * idx / float(B)))
        starts.append(c * np.sin(4.0 * np.pi * idx / float(B) + 0.5))

    return starts


def optimize_block_weights(base, B, n, N, m, deadline, seed):
    rng = np.random.default_rng(seed)
    block_idx = make_block_idx(n, B)

    base_cand = base.copy()
    base_score = fast_score(base_cand, n, N, m)

    best_cand = base_cand.copy()
    best_score = base_score
    best_theta = np.zeros(B, dtype=np.float64)

    start_theta = np.zeros(B, dtype=np.float64)
    start_cand = base_cand.copy()
    start_score = base_score

    for th in theta_starts(B):
        cand = theta_candidate(base, block_idx, th)
        sc = fast_score(cand, n, N, m)
        if np.isfinite(sc):
            if sc < start_score - 1e-12:
                start_score = sc
                start_theta = th.copy()
                start_cand = cand.copy()
            if sc < best_score - 1e-12:
                best_score = sc
                best_theta = th.copy()
                best_cand = cand.copy()

    cur_theta = start_theta.copy()
    cur_cand = start_cand.copy()
    cur_score = start_score
    steps = np.full(B, 0.35, dtype=np.float64)
    no_improve = 0

    while time.monotonic() < deadline:
        j = int(rng.integers(B))
        improved = False

        trial = cur_theta.copy()
        trial[j] += steps[j]
        trial_cand = theta_candidate(base, block_idx, trial)
        sc = fast_score(trial_cand, n, N, m)
        if np.isfinite(sc) and sc < cur_score - 1e-12:
            cur_theta = trial
            cur_cand = trial_cand
            cur_score = sc
            improved = True
        else:
            trial = cur_theta.copy()
            trial[j] -= steps[j]
            trial_cand = theta_candidate(base, block_idx, trial)
            sc = fast_score(trial_cand, n, N, m)
            if np.isfinite(sc) and sc < cur_score - 1e-12:
                cur_theta = trial
                cur_cand = trial_cand
                cur_score = sc
                improved = True
            else:
                steps[j] = max(0.02, steps[j] * 0.85)

        if improved:
            no_improve = 0
            steps[j] = min(3.0, steps[j] * 1.15)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_theta = cur_theta.copy()
                best_cand = cur_cand.copy()
                print("block optimize: %.8f" % best_score)
        else:
            no_improve += 1

        if no_improve > 700:
            jump = best_theta + 0.50 * rng.normal(size=B)
            jump = np.clip(jump, -4.0, 4.0)
            jcand = theta_candidate(base, block_idx, jump)
            jscore = fast_score(jcand, n, N, m)
            if np.isfinite(jscore) and jscore < best_score + 0.05:
                cur_theta = jump
                cur_cand = jcand
                cur_score = jscore
            else:
                cur_theta = best_theta.copy()
                cur_cand = best_cand.copy()
                cur_score = best_score
            no_improve = 0

    return best_cand, best_score


def local_search(p, n, N, m, deadline, seed):
    rng = np.random.default_rng(seed)
    best = p.copy()
    best_score = fast_score(best, n, N, m)

    cur = best.copy()
    cur_score = best_score

    if n > 1:
        x = np.arange(n, dtype=np.float64) / float(n - 1)
    else:
        x = np.zeros(1, dtype=np.float64)
    sine_cache = make_sine_cache(n)

    sigma = 0.28
    no_improve = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        batch_limit = 64
        for _ in range(batch_limit):
            if time.monotonic() >= deadline:
                break
            mode = int(rng.integers(0, 6))

            if mode == 0:
                cand = move_early_late(cur, rng, n, 0.55)
            elif mode == 1:
                cand = move_random_transfer(cur, rng, n)
            elif mode == 2:
                cand = move_ratio(cur, rng, n, sigma)
            elif mode == 3:
                cand = move_single(cur, rng, n, sigma)
            elif mode == 4:
                cand = move_tilt(cur, rng, x, sigma)
            else:
                cand = move_sine(cur, rng, sine_cache, sigma)

            if cand is None:
                continue

            sc = fast_score(cand, n, N, m)
            if not np.isfinite(sc):
                continue

            if sc < cur_score - 1e-12:
                cur = cand
                cur_score = sc
                no_improve = 0
                if sc < best_score - 1e-12:
                    best = cand
                    best_score = sc
                    print("local search: %.8f at t=%.1f" % (best_score, time.monotonic() - (deadline - max(0, 2400))))
            else:
                no_improve += 1

        if no_improve > 2500:
            jc = cur.copy()
            ls = max(1, int(0.07 * n))
            dr = int(0.60 * n)
            ld = min(n - dr, max(1, int(0.35 * n)))
            alpha = float(rng.uniform(0.15, 0.55))
            uniform_dest = bool(rng.random() < 0.4)
            apply_mass_transfer(jc, 0, ls, dr, ld, alpha, uniform_dest)
            jc = finish_candidate(jc)
            js = fast_score(jc, n, N, m) if jc is not None else np.inf
            if np.isfinite(js) and js < best_score + 0.05:
                cur = jc
                cur_score = js
            else:
                cur = best.copy()
                cur_score = best_score
            no_improve = 0
            sigma = min(1.2, sigma * 1.25)

        if no_improve == 0:
            sigma = max(0.05, sigma * 0.985)

        now = time.monotonic()
        if now - last_print > 60.0:
            last_print = now
            print("local search active best: %.8f" % best_score)

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start_time = time.monotonic()
    total_budget = max(0.0, float(budget_s))
    deadline = start_time + max(0.0, total_budget - 5.0)

    prev = None
    try:
        prev_arr = height_sequence_1
        if prev_arr is not None and len(prev_arr) > 0:
            prev = np.asarray(prev_arr, dtype=np.float64)
    except NameError:
        prev = None

    if prev is not None and 2048 <= prev.size <= 10000:
        n = int(prev.size)
    else:
        n = 4096

    n = max(1024, min(12000, int(n)))
    m = 2 * n - 1
    N = next_power_of_two(m)

    inits = generate_initial_candidates(n, N, m, prev)
    best = None
    best_score = np.inf
    fallback_base = None

    base_norm = normalize_positive(binomial_sequence(n))
    if base_norm is not None:
        fallback_base = base_norm.copy()

    for cand in inits:
        if cand is None:
            continue
        sc = fast_score(cand, n, N, m)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
    if best is None and fallback_base is not None:
        best = fallback_base.copy()
        best_score = fast_score(best, n, N, m)

    print("initial best: %.8f length=%d" % (best_score, n))

    grid_deadline = start_time + max(45.0, 0.08 * total_budget)
    block_deadline = start_time + max(180.0, 0.25 * total_budget)
    grid_deadline = min(grid_deadline, block_deadline)
    block_deadline = min(block_deadline, deadline - 60.0)

    if best is not None and time.monotonic() < grid_deadline:
        best, best_score = grid_mass_transfers(
            best, best_score, n, N, m, grid_deadline,
            np.random.default_rng(seed + 101)
        )
        print("after grid: %.8f" % best_score)

    if best is not None and time.monotonic() < block_deadline:
        cand, sc = optimize_block_weights(
            best, 32, n, N, m, block_deadline, seed + 202
        )
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best = cand
            best_score = sc
            print("after block optimization: %.8f" % best_score)

    if best is not None and time.monotonic() < deadline:
        best, best_score = local_search(best, n, N, m, deadline, seed + 303)

    print("final fast score: %.8f" % best_score)

    if fallback_base is not None:
        candidates_exact = [
            [float(x) for x in best],
            [float(x) for x in fallback_base],
        ]
    else:
        candidates_exact = [[float(x) for x in best]]

    best_seq = [float(x) for x in best]
    best_exact = np.inf

    for seq in candidates_exact:
        try:
            sc = float(evaluate_sequence(seq))
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < best_exact:
            best_exact = sc
            best_seq = seq

    if not all(np.isfinite(v) for v in best_seq) or len(best_seq) == 0:
        best_seq = [float(x) for x in fallback_base]

    s = float(sum(max(0.0, v) for v in best_seq))
    if s <= 0.0:
        best_seq = [1.0]
    else:
        best_seq = [max(0.0, min(1000.0, float(v) / s)) for v in best_seq]
        if sum(best_seq) <= 0.0:
            best_seq = [1.0]

    return best_seq
```
