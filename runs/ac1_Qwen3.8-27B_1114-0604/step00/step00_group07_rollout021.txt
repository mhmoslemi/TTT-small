def solve_lp(p, M):
    """Solve the LP: maximize sum(b) subject to b >= 0 and (p*b)[k] <= M for all k."""
    n = len(p)
    # Variables: b[0], ..., b[n-1]
    # Objective: minimize -sum(b)
    c = -np.ones(n)
    
    # Inequality constraints: (p*b)[k] <= M for k = 0, ..., 2n-2

    return [float(x) for x in best_p]

import numpy as np
import time
from scipy.optimize import linprog

def build_conv_matrix(n, p):
    """Build the (2n-1) x n convolution constraint matrix."""
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    diffs = k_idx - j_idx
    mask = (diffs >= 0) & (diffs < n)
    A = np.where(mask, p[diffs.clip(0, n-1)], 0.0)
    return A

def solve_lp_for_b(p, M):
    """Solve LP: max sum(b) s.t. (p*b)[k] <= M for all k, b >= 0."""
    n = len(p)
    A = build_conv_matrix(n, p)
    b_ub = np.full(2*n-1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.status != 0:
        return None
    return result.x
