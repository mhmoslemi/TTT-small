```python
"""Search for non-negative step heights minimizing 2*n*max(conv(a,a))/sum(a)^2.

Strategy:
1. Start from several smooth skew/cosine-log low-dimensional shapes and refine
   them with coordinate search + global Nelder-Mead polish.
2. Use the best sequence as an incumbent.
3. Improve it with a linear-programming best-response iteration:
   maximize sum(g) subject to a*g <= max(a*a), g >= 0, then line-search convex
   combinations a <- (1-t)a + t*g and keep only exact improvements.
4. If time remains, do cheap randomized local searches.
All candidates are normalized to sum sqrt(2*n), making the objective numerically
equal to max(a*a) up to roundoff.
"""

import numpy as np
import time

import scipy.sparse as sp
from scipy.optimize import linprog, minimize


def fast_score(arr):
    """FFT-based objective for internal ranking; evaluate_sequence is final."""
    a = np.asarray(arr, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf

    need = 2 * n - 1
    m = 1 << ((need - 1).bit_length())
    fa = np.fft.rfft(a, m)
    cv = np.fft.irfft(fa * fa, m)[:need]
    maxc = float(cv.max())
    return float(2.0 * n * maxc / (s * s))


def normalize_to_sum(a, target_sum):
    """Return a finite non-negative array with exact target sum, clipped."""
    a = np.asarray(a, dtype=np.float64).reshape(-1).copy()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.full(a.size, float(target_sum) / float(a.size), dtype=np.float64)
    a *= float(target_sum) / s
    a = np.minimum(a, 1000.0)
    s2 = float(a.sum())
    if s2 > 0.0:
        a *= float(target_sum) / s2
    return a


def logcos_sequence(p, cos_terms, target_sum):
    """Construct positive sequence exp(sum p_j cos(j*pi*x)) normalized."""
    z = cos_terms.dot(p)
    np.clip(z, -50.0, 50.0, out=z)
    f = np.exp(z)
    s = float(f.sum())
    if not np.isfinite(s) or s <= 1e-300:
        f = np.ones_like(f)
        s = float(f.size)
    a = f * (float(target_sum) / s)
    return normalize_to_sum(a, target_sum)


def logcos_score_value(p, cos_terms, target_sum):
    """Value function for scipy.optimize.minimize."""
    a = logcos_sequence(p, cos_terms, target_sum)
    return fast_score(a)


def score_logcos(p, cos_terms, target_sum):
    """Return (score, sequence) for a log-cosine parameter vector."""
    a = logcos_sequence(p, cos_terms, target_sum)
    return fast_score(a), a


def coordinate_logcos(p0, cos_terms, target_sum, deadline, max_passes=35):
    """Coordinate hill-climbing in low-dimensional log-cosine space."""
    p = np.asarray(p0, dtype=np.float64).copy()
    val, seq = score_logcos(p, cos_terms, target_sum)
    if not np.isfinite(val):
        p = np.zeros_like(p)
        val, seq = score_logcos(p, cos_terms, target_sum)

    best_p = p.copy()
    best_val = val
    best_seq = seq.copy()

    delta = 0.45
    passes = 0
    while passes < max_passes and time.monotonic() < deadline - 1.5:
        passes += 1
        improved = False

        for j in range(p.size):
            current = float(p[j])
            best_current_val = val
            best_seq_trial = seq
            best_p_trial = p

            for sign in (-1.0, 1.0):
                for frac in (1.0, 0.5, 0.25):
                    q = p.copy()
                    q[j] = current + sign * frac * delta
                    if q[j] < -4.0 or q[j] > 4.0:
                        continue
                    v, aa = score_logcos(q, cos_terms, target_sum)
                    if np.isfinite(v) and v < best_current_val - 1e-12:
                        best_current_val = v
                        best_seq_trial = aa
                        best_p_trial = q.copy()

            if best_current_val < val - 1e-12:
                p = best_p_trial.copy()
                val = best_current_val
                seq = best_seq_trial.copy()
                improved = True
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                    best_p = p.copy()

        if not improved:
            delta *= 0.72
            if delta < 0.02:
                break
        else:
            if delta < 0.85:
                delta = min(1.25, delta * 1.06)

    return best_p, best_seq, best_val


def lowdim_search(K, n, target_sum, deadline, initial_seq=None):
    """Optimize a smooth low-dimensional family as a starting point."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    cos_terms = np.empty((K, n), dtype=np.float64)
    for j in range(1, K + 1):
        cos_terms[j - 1] = np.cos(j * np.pi * x)

    if initial_seq is None:
        best_seq = np.full(n, target_sum / float(n), dtype=np.float64)
    else:
        best_seq = normalize_to_sum(initial_seq, target_sum)
    best_val = fast_score(best_seq)
    best_p = np.zeros(K, dtype=np.float64)
    if not np.isfinite(best_val):
        best_val = 1e100

    rng = np.random.default_rng(20240525)

    starts = []
    starts.append(np.zeros(K, dtype=np.float64))

    p = np.zeros(K, dtype=np.float64)
    p[0] = 1.0
    starts.append(p.copy())

    p = np.zeros(K, dtype=np.float64)
    p[0] = 0.9
    if K > 1:
        p[1] = -0.6
    if K > 2:
        p[2] = 0.4
    starts.append(p.copy())

    p = np.zeros(K, dtype=np.float64)
    p[0] = 1.25
    if K > 1:
        p[1] = 0.55
    if K > 2:
        p[2] = -0.75
    if K > 3:
        p[3] = 0.35
    starts.append(p.copy())

    for _ in range(4):
        starts.append(rng.uniform(-0.9, 0.9, size=K).astype(np.float64))

    # Coordinate search from a few reasonable starts.
    for p0 in starts:
        if time.monotonic() > deadline - 8.0:
            break
        p, seq, val = coordinate_logcos(p0, cos_terms, target_sum, deadline - 2.0, max_passes=28)
        if np.isfinite(val) and val < best_val - 1e-10:
            best_val = float(val)
            best_seq = normalize_to_sum(seq, target_sum)
            best_p = np.asarray(p, dtype=np.float64).copy()

    # A short Nelder-Mead polish on the best low-dimensional point.
    if time.monotonic() < deadline - 40.0:
        try:
            res = minimize(
                logcos_score_value,
                best_p.copy(),
                args=(cos_terms, target_sum),
                method="Nelder-Mead",
                options={"maxiter": 250, "xatol": 1e-5, "fatol": 1e-7},
            )
            v = float(res.fun)
            if np.isfinite(v) and v < best_val - 1e-10:
                best_val = v
                best_seq = logcos_sequence(np.asarray(res.x, dtype=np.float64), cos_terms, target_sum)
        except Exception:
            pass

    return normalize_to_sum(best_seq, target_sum), float(best_val)


def build_A(a):
    """Sparse matrix A with (A*g)[k] = (a*g)[k] for linear convolution."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    m = 2 * n - 1
    nz = np.nonzero(a > 1e-15)[0].astype(int)
    if nz.size == 0:
        return sp.csr_matrix((m, n), dtype=np.float64)

    one = np.ones(n, dtype=np.float64)
    diags = []
    for d in nz:
        diags.append(float(a[d]) * one)
    return sp.diags(diags, offsets=nz, shape=(m, n), format="csr", dtype=np.float64)


def solve_max_sum_lp(A, a, M, target_sum, time_limit=30.0):
    """Maximize sum(g) subject to a*g <= M, g >= 0."""
    n = A.shape[1]
    m = A.shape[0]

    c = -np.ones(n, dtype=np.float64)
    amax = float(np.max(a)) if a.size else 0.0
    if amax > 1e-15:
        w = a / amax
        c = c + 0.08 * (w - float(w.mean()))

    b_ub = float(M) * np.ones(m, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": float(max(1.0, time_limit))},
        )
    except Exception:
        return None, False

    if not res.success:
        return None, False

    g = np.asarray(res.x, dtype=np.float64).reshape(-1)
    if g.size != n or not np.all(np.isfinite(g)):
        return None, False

    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-15:
        return None, False

    g *= float(target_sum) / s
    g = normalize_to_sum(g, target_sum)
    return g, True


def solve_min_cross_lp(A, a, target_sum, time_limit=30.0):
    """Minimize t subject to a*g <= t, sum(g)=target_sum, g>=0."""
    n = A.shape[1]
    m = A.shape[0]

    I = sp.identity(m, format="csr", dtype=np.float64)
    A_aug = sp.hstack([A, -I], format="csr", dtype=np.float64)

    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([float(target_sum)], dtype=np.float64)

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(
            c,
            A_ub=A_aug,
            b_ub=np.zeros(m, dtype=np.float64),
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"time_limit": float(max(1.0, time_limit))},
        )
    except Exception:
        return None

    if not res.success:
        return None

    g = np.asarray(res.x[:n], dtype=np.float64).reshape(-1)
    if not np.all(np.isfinite(g)):
        return None

    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-15:
        return None

    g *= float(target_sum) / s
    return normalize_to_sum(g, target_sum)


def lp_improve(a_best, best_score, target_sum, deadline):
    """LP-direction local search with exact acceptance."""
    a = normalize_to_sum(a_best, target_sum)
    best = a.copy()
    best_value = float(best_score)

    n = a.size
    no_improve = 0
    it = 0

    while time.monotonic() < deadline - 30.0:
        it += 1
        rem = max(10.0, deadline - time.monotonic())
        tlimit = max(5.0, min(30.0, rem / 8.0))

        conv = np.convolve(a, a)
        M = float(np.max(conv)) if conv.size else 0.0
        if not np.isfinite(M) or M <= 0.0:
            break

        A = build_A(a)

        g, ok = solve_max_sum_lp(A, a, M, target_sum, time_limit=tlimit)
        if (not ok) or g is None or float(np.linalg.norm(g - a)) < 1e-8 * target_sum:
            g2 = solve_min_cross_lp(A, a, target_sum, time_limit=tlimit)
            if g2 is not None:
                g = g2

        if g is None:
            break

        g = normalize_to_sum(g, target_sum)

        best_trial = None
        best_trial_val = float(best_value) + 1e-30
        for t in (0.0, 0.02, 0.05, 0.10, 0.18, 0.30, 0.50, 0.75):
            cand = (1.0 - t) * a + t * g
            cand = normalize_to_sum(cand, target_sum)
            val = fast_score(cand)
            if np.isfinite(val) and val < best_trial_val - 1e-12:
                best_trial_val = float(val)
                best_trial = cand.copy()

        improved = False
        if best_trial is not None and best_trial_val < best_value - 1e-9:
            exact = float(evaluate_sequence(best_trial.tolist()))
            if np.isfinite(exact) and exact < best_value - 1e-12:
                best_value = exact
                best = best_trial.copy()
                a = best_trial.copy()
                improved = True
                no_improve = 0
                print(f"LP iter {it}: score={best_value:.8f}", flush=True)

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                rng = np.random.default_rng(9173 + it)
                pert = a * np.exp(rng.normal(0.0, 0.12, n))
                pert = normalize_to_sum(pert, target_sum)
                if fast_score(pert) < best_value + 0.25:
                    a = pert
                else:
                    a = best.copy()
                no_improve = 0

    return normalize_to_sum(best, target_sum), float(best_value)


def random_improve(a_best, best_score, target_sum, deadline):
    """Cheap random/block local search as a fallback and finisher."""
    a = normalize_to_sum(a_best, target_sum)
    best = a.copy()
    best_value = float(best_score)
    n = a.size
    rng = np.random.default_rng(55555)

    while time.monotonic() < deadline - 10.0:
        mode = float(rng.random())
        cand = a.copy()

        if mode < 0.45:
            sigma = float(rng.choice((0.03, 0.07, 0.15, 0.30, 0.55)))
            cand *= np.exp(rng.normal(0.0, sigma, n))
        elif mode < 0.80:
            width = int(rng.integers(max(1, n // 500), max(2, n // 4)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.25)))
            if rng.random() < 0.25:
                cand = 0.5 * (cand + cand[::-1])
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            move = float(a[i]) * float(rng.choice((-0.75, -0.25, 0.25, 0.75)))
            cand[i] -= move
            cand[j] += move

        cand = normalize_to_sum(cand, target_sum)
        val = fast_score(cand)
        if np.isfinite(val) and val < best_value - 1e-8:
            exact = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact) and exact < best_value - 1e-12:
                best_value = exact
                best = cand.copy()
                a = cand.copy()
                print(f"Random local: score={best_value:.8f}", flush=True)

    return normalize_to_sum(best, target_sum), float(best_value)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Return best non-negative coefficient sequence found within time budget."""
    try:
        budget_s = max(10.0, float(budget_s))
    except Exception:
        budget_s = 2400.0

    deadline = time.monotonic() + max(1.0, budget_s - 5.0)

    # Choose a working length. Prefer the provided construction length when sane.
    n = 2535
    initial_seq = None
    initial_score = np.inf

    try:
        h = height_sequence_1
        if h is not None:
            h = np.asarray(h, dtype=np.float64).reshape(-1)
            if 1000 <= h.size <= 3000 and np.all(np.isfinite(h)):
                n = int(h.size)
                S = float(np.sqrt(2.0 * n))
                init = normalize_to_sum(h, S)
                sc = float(evaluate_sequence(init.tolist()))
                if np.isfinite(sc):
                    initial_seq = init
                    initial_score = sc
    except Exception:
        pass

    S = float(np.sqrt(2.0 * n))

    if initial_seq is None or not np.isfinite(initial_score):
        initial_seq = np.full(n, S / float(n), dtype=np.float64)
        sc = float(evaluate_sequence(initial_seq.tolist()))
        if np.isfinite(sc):
            initial_score = sc
        else:
            initial_score = 2.0

    best_seq = normalize_to_sum(initial_seq, S)
    best_score = float(initial_score)
    print(f"Initial: n={n}, score={best_score:.8f}", flush=True)

    # Low-dimensional smooth-shape exploration.
    low_budget = max(30.0, min(210.0, 0.22 * max(0.0, budget_s - 20.0)))
    low_deadline = time.monotonic() + low_budget
    try:
        cand_seq, cand_val = lowdim_search(10, n, S, low_deadline, initial_seq)
        exact = float(evaluate_sequence(cand_seq.tolist()))
        if np.isfinite(exact) and exact < best_score:
            best_seq = normalize_to_sum(cand_seq, S)
            best_score = exact
        print(f"Low-dim: best={best_score:.8f}", flush=True)
    except Exception as e:
        print("Low-dim failed:", repr(e), flush=True)

    # LP local search if needed and time remains.
    if best_score > 1.5027 + 1e-10 and time.monotonic() < deadline - 180.0:
        lp_deadline = min(deadline - 90.0, time.monotonic() + max(
            120.0, 0.58 * max(0.0, deadline - time.monotonic())
        ))
        try:
            best_seq, best_score = lp_improve(best_seq, best_score, S, lp_deadline)
        except Exception as e:
            print("LP failed:", repr(e), flush=True)

    # Random finisher if time remains.
    if time.monotonic() < deadline - 30.0:
        try:
            best_seq, best_score = random_improve(best_seq, best_score, S, deadline - 5.0)
        except Exception as e:
            print("Random failed:", repr(e), flush=True)

    # Exact final verification and safe normalization.
    final = normalize_to_sum(best_seq, S)
    final_score = float(evaluate_sequence(final.tolist()))

    if not np.isfinite(final_score):
        final = np.full(n, S / float(n), dtype=np.float64)
        final_score = float(evaluate_sequence(final.tolist()))
        if not np.isfinite(final_score):
            final_score = 2.0

    print(f"FINAL: n={n}, score={final_score:.10f}", flush=True)
    return [float(x) for x in final]
```
