**Smooth approximation:**
f_smooth(a) = (1/T) * log(sum_k exp(T * (a*a)[k]))

As T → ∞, this approaches max_k (a*a)[k]. For T = 10-50, it's a good approximation and is smooth.

The gradient of f_smooth is:
∇f_smooth = (1/T) * sum_k exp(T*(a*a)[k]) / sum_j exp(T*(a*a)[j]) * ∇(a*a)[k]
= sum_k softmax_k * 2 * P_k * a (component-wise)

This is a weighted average of the individual gradients, weighted by the softmax.

```python
def optimize_subgradient(init_a, n, n_iters=10000, T=20.0):
    a = np.clip(init_a, 0, None).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2 * n * np.max(np.convolve(a, a))
    
    velocity = np.zeros(n)
    beta = 0.85
    lr = 0.01  # Initial learning rate
    
    for it in range(n_iters):
        conv = np.convolve(a, a)
        max_conv = conv.max()
        score = 2 * n * max_conv
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Softmax weights
        shifted = T * (conv - max_conv)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Gradient via cross-correlation
        grad = 2.0 * np.correlate(weights, a, mode='valid')
        grad = grad - grad.mean()  # Project onto sum=0
        
        # Momentum update
        velocity = beta * velocity - lr * grad
        a_new = a + velocity
        
        # Projection
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a_new / s
        
        # Adaptive learning rate (decrease slowly)
        lr = 0.01 / (1 + it * 0.001)
    
    return best_a, best_score
```

```python
def compute_smooth_gradient(a, conv, T):
    """Compute the gradient of the smooth approximation to max(conv)."""
    max_conv = np.max(conv)
    shifted = T * (conv - max_conv)
    # Avoid overflow
    shifted -= np.max(shifted)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient: grad[j] = 2 * sum_k weights[k] * a[k-j] (for valid k-j)
    # = 2 * correlate(weights, a, 'valid')[j]
    n = len(a)
    grad = 2.0 * np.correlate(weights, a, mode='valid')
    return grad, weights
```

```python
def project_simplex(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    if np.sum(x) <= 0:
        return np.ones(n) / n
    # Sort in decreasing order
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)
```
