def make_weights(c, rng):
    m = c.size
    c = np.asarray(c, dtype=np.float64)
    # remove nonfinite
    if not np.all(np.isfinite(c)):
       c = np.where(np.isfinite(c), c, 0.0)
    cmax = float(np.max(c))
    if cmax <= 0 or not np.isfinite(cmax):
       w = np.full(m,1.0/m); return w
    r = float(rng.random())
    if r < 0.45:
       sigma = float(rng.choice(np.array([0.02,0.05,0.1,0.2,0.5], dtype=float)))
       temp = max(1e-300, cmax*sigma)
       val = (c-cmax)/temp
       # clipping val < -700 to avoid underflow? exp handles underflow to 0; no overflow since val<=0
       w = np.exp(val)

def resize_sequence(seq,n):
  arr=np.asarray(seq,float)
  arr=np.where(np.isfinite(arr),arr,0)
  arr=np.clip(arr,0,None)
  s=arr.sum()
  if arr.size==0 or not finite(s) or s<=0:
     return np.ones(n)/n
  arr=arr/s
  if arr.size==n: return arr
  x_old=np.linspace(0.0,1.0,arr.size)

  out=np.interp(x_new,x_old,arr)

```python
"""
Search for a non-negative step function minimizing
    2*n*max(conv(a,a)) / sum(a)^2

The optimizer uses FFT-based smoothed-max mirror/multiplicative updates,
multiple smooth random starts, local refinement, and optional LP polishing
for shorter sequences.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    _HAVE_SCIPY_LP = True
except Exception:
    _HAVE_SCIPY_LP = False

_LP_CACHE = {}


def _candidate_key(item):
    return item[0]


def next_pow2(m):
    s = 1
    m = int(m)
    while s < m:
        s <<= 1
    return s


def normalize_probability(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return np.ones(1, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.clip(v, 0.0, None)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return v / s


def objective_from_conv(c, n):
    c = np.real(np.asarray(c, dtype=np.float64))
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    if m < 0.0:
        if m > -1e-14:
            m = 0.0
        else:
            return np.inf
    return float(2.0 * int(n) * m)


def fft_convolve_square(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    m = 2 * n - 1
    S = next_pow2(m)
    r = np.fft.rfft(p, n=S)
    c = np.fft.irfft(r * r, n=S)[:m]
    return np.real(c)


def cross_corr_weights_p(w, p):
    w = np.asarray(w, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    m = int(w.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    S = next_pow2(m + n - 1)
    W = np.fft.rfft(w, n=S)
    P = np.fft.rfft(p, n=S)
    corr = np.fft.irfft(W.conj() * P, n=S)
    return np.real(corr[:n])


def make_weights(c, rng):
    c = np.asarray(c, dtype=np.float64)
    m = int(c.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.full(m, 1.0 / float(m), dtype=np.float64)

    r = float(rng.random())
    if r < 0.45:
        sigma_values = (0.02, 0.05, 0.10, 0.20, 0.50)
        sigma = sigma_values[int(rng.integers(0, len(sigma_values)))]
        temp = max(1e-300, cmax * sigma)
        val = (c - cmax) / temp
        w = np.exp(val)
        s = float(np.sum(w))
        if np.isfinite(s) and s > 0.0:
            return w / s

    k_values = (4, 8, 16, 32, 64, 128)
    K = k_values[int(rng.integers(0, len(k_values)))]
    if K > m:
        K = m

    if K == 1:
        idx = np.array([int(np.argmax(c))], dtype=np.int64)
        widx = np.array([1.0], dtype=np.float64)
    else:
        if K == m:
            idx = np.arange(m, dtype=np.int64)
        else:
            idx = np.argpartition(c, -K)[-K:]
        if int(rng.integers(0, 10)) < 3:
            widx = np.full(len(idx), 1.0 / float(len(idx)), dtype=np.float64)
        else:
            vals = c[idx]
            s = float(np.sum(vals))
            if np.isfinite(s) and s > 0.0:
                widx = vals / s
            else:
                widx = np.full(len(idx), 1.0 / float(len(idx)), dtype=np.float64)

    w = np.zeros(m, dtype=np.float64)
    w[idx] += widx
    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    return w / s


def generate_start(n, start_id, rng, height_seq):
    n = int(n)
    sid = int(start_id) % 9

    if sid == 8 and height_seq is not None:
        return resize_sequence(height_seq, n)

    t = np.linspace(0.0, 1.0, n)

    if sid == 0:
        # nearly uniform, but broken symmetry
        p = np.exp(rng.normal(0.0, 0.05, n))
    elif sid == 1:
        alpha_values = (1.0, 0.5, 2.0)
        alpha = alpha_values[int(rng.integers(0, 3))]
        p = rng.dirichlet(np.full(n, alpha, dtype=np.float64))
    elif sid == 2:
        a = float(rng.uniform(-2.0, 2.0))
        b = float(rng.uniform(-2.0, 2.0))
        p = np.exp(a * t + b * t * t)
    elif sid == 3:
        f = np.ones(n, dtype=np.float64)
        for k in (1, 2, 3):
            amp = float(rng.uniform(0.0, 0.7))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            f = f + amp * np.cos(2.0 * np.pi * k * t + phase)
        p = np.clip(f, 1e-8, None)
    elif sid == 4:
        a = float(rng.uniform(1.0, 8.0))
        p = np.exp(a * t)
    elif sid == 5:
        a = float(rng.uniform(1.0, 8.0))
        p = np.exp(-a * t)
    elif sid == 6:
        p = np.abs(rng.normal(0.0, 1.0, n))
    elif sid == 7:
        mu = float(rng.uniform(0.2, 0.8))
        sig = float(rng.uniform(0.05, 0.25))
        p = np.exp(-0.5 * ((t - mu) / sig) ** 2)
        p = p + 0.1 * np.ones(n, dtype=np.float64)
    else:
        p = np.ones(n, dtype=np.float64)

    return normalize_probability(p)


def local_search(n, p_init, seconds, rng):
    n = int(n)
    p = normalize_probability(p_init)
    c = fft_convolve_square(p)
    obj = objective_from_conv(c, n)

    best_p = p.copy()
    best_c = c.copy()
    best_obj = obj

    start = time.monotonic()
    deadline = start + max(1.0, float(seconds))

    eta = 1.0
    no_improve = 0
    iter_no = 0

    while time.monotonic() < deadline:
        iter_no += 1

        if iter_no % 25 == 0:
            eta = float(np.clip(eta * float(rng.normal(1.0, 0.3)), 0.05, 20.0))

        w = make_weights(c, rng)
        h = cross_corr_weights_p(w, p)
        hmax = float(np.max(h)) if h.size else 0.0

        floor = 1e-6 / float(n)

        if not np.isfinite(hmax) or hmax <= 0.0:
            p = normalize_probability(p * np.exp(rng.normal(0.0, 0.2, n)) + floor)
            c = fft_convolve_square(p)
            obj = objective_from_conv(c, n)
            no_improve = 0
            eta = float(rng.uniform(0.2, 3.0))
            if not np.isfinite(obj) or obj > best_obj * 1.05:
                p = normalize_probability(best_p.copy() + floor)
                c = fft_convolve_square(p)
                obj = objective_from_conv(c, n)
            continue

        x = h / hmax
        use_exp = (iter_no % 7 == 0)

        etas = [eta]
        if no_improve > 20:
            etas.append(max(0.05, eta * 0.5))
            etas.append(min(20.0, eta * 2.0))

        best_trial = None
        best_trial_c = None
        best_trial_obj = obj
        worst_trial = None
        worst_trial_c = None
        worst_trial_obj = np.inf

        for et_raw in etas:
            et = min(5.0 if use_exp else 20.0, float(et_raw))
            if use_exp:
                q = p * np.exp(-et * x)
            else:
                q = p / (1.0 + et * x)
            q = normalize_probability(q)
            cq = fft_convolve_square(q)
            oq = objective_from_conv(cq, n)
            if np.isfinite(oq):
                if oq < best_trial_obj:
                    best_trial_obj = oq
                    best_trial = q
                    best_trial_c = cq
                if oq < worst_trial_obj:
                    worst_trial_obj = oq
                    worst_trial = q
                    worst_trial_c = cq

        if best_trial is not None and best_trial_obj < obj - 1e-12:
            p = best_trial
            c = best_trial_c
            obj = best_trial_obj
            eta = min(20.0, eta * 1.15)
            no_improve = 0
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_p = p.copy()
                best_c = c.copy()
        elif no_improve > 10 and worst_trial is not None and worst_trial_obj < obj * 1.02:
            # simulated-annealing style escape: accept a mild worsening
            p = worst_trial
            c = worst_trial_c
            obj = worst_trial_obj
            no_improve = 0
            eta = max(0.05, eta * 0.7)
        else:
            no_improve += 1
            if no_improve > 30:
                sigma = 0.05 + 0.15 * min(1.0, no_improve / 200.0)
                p = normalize_probability(p * np.exp(rng.normal(0.0, sigma, n)) + floor)
                c = fft_convolve_square(p)
                obj = objective_from_conv(c, n)

                if np.random.RandomState is None:
                    pass  # placeholder to keep style simple

                if (not np.isfinite(obj)) or obj > best_obj * 1.05:
                    p = normalize_probability(best_p.copy() + floor)
                    c = fft_convolve_square(p)
                    obj = objective_from_conv(c, n)

                no_improve = 0
                eta = float(rng.uniform(0.2, 3.0))

                if float(rng.random()) < 0.3:
                    init = normalize_probability(np.exp(rng.normal(0.0, 0.5, n)) + floor)
                    p = normalize_probability(0.7 * p + 0.3 * init)
                    c = fft_convolve_square(p)
                    obj = objective_from_conv(c, n)
                    if (not np.isfinite(obj)) or obj > best_obj * 1.05:
                        p = normalize_probability(best_p.copy() + floor)
                        c = fft_convolve_square(p)
                        obj = objective_from_conv(c, n)
            else:
                eta = max(0.05, eta * 0.75)

    return best_p, best_obj


def resize_sequence(seq, n):
    arr = np.asarray(seq, dtype=np.float64)
    if arr.size == 0:
        return np.ones(int(n), dtype=np.float64) / float(n)

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, None)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(int(n), dtype=np.float64) / float(n)

    arr = arr / s
    n = int(n)
    if arr.size == n:
        return arr

    x_old = np.linspace(0.0, 1.0, arr.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, arr)
    return normalize_probability(out)


def get_lp_indices(n):
    cached = _LP_CACHE.get(n)
    if cached is not None:
        return cached

    n = int(n)
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    pos = 0
    for j in range(n):
        rows[pos:pos + n] = np.arange(j, j + n, dtype=np.int32)
        cols[pos:pos + n] = j
        pos += n

    _LP_CACHE[n] = (rows, cols, total)
    return rows, cols, total


def lp_polish(p_init, seconds, rng):
    p = normalize_probability(p_init)
    n = int(p.size)
    c = fft_convolve_square(p)
    obj = objective_from_conv(c, n)

    best_p = p.copy()
    best_c = c.copy()
    best_obj = obj

    if (not _HAVE_SCIPY_LP) or n > 1024 or float(seconds) < 5.0:
        return best_p, best_obj

    start = time.monotonic()
    deadline = start + float(seconds)
    rows, cols, total = get_lp_indices(n)

    no_improve = 0

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 8.0:
            break

        M = float(np.max(c))
        if not np.isfinite(M) or M <= 0.0:
            break

        vals = np.tile(p * float(n), n)
        try:
            A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        except Exception:
            break

        b = (float(n) * M) * np.ones(2 * n - 1, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        tl = max(1.0, min(10.0, rem - 3.0))

        try:
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=b,
                bounds=bounds,
                method='highs',
                options={'time_limit': tl}
            )
        except Exception:
            break

        if res is None or res.x is None:
            break

        q = np.asarray(res.x, dtype=np.float64)
        q = np.where(np.isfinite(q), q, 0.0)
        q = np.clip(q, 0.0, None)
        qsum = float(q.sum())
        if not np.isfinite(qsum) or qsum <= 1e-12:
            break
        q = q / qsum

        best_trial = None
        best_trial_c = None
        best_trial_obj = obj
        ts = (0.10, 0.25, 0.50, 0.75, 1.00)

        for tt in ts:
            cand = (1.0 - tt) * p + tt * q
            cand = normalize_probability(cand)
            cc = fft_convolve_square(cand)
            oo = objective_from_conv(cc, n)
            if np.isfinite(oo) and oo < best_trial_obj:
                best_trial_obj = oo
                best_trial = cand
                best_trial_c = cc

        if best_trial is not None and best_trial_obj < obj - 1e-12:
            p = best_trial
            c = best_trial_c
            obj = best_trial_obj
            no_improve = 0
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_p = p.copy()
                best_c = c.copy()
        else:
            no_improve += 1
            if no_improve > 2:
                break

    return best_p, best_obj


def consider_candidate(candidates, best_score, best_seq_arr, seq, score):
    seq = np.asarray(seq, dtype=np.float64)
    if seq.size == 0:
        return best_score, best_seq_arr, candidates

    n = int(seq.size)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, None)
    ss = float(seq.sum())
    if not np.isfinite(ss) or ss <= 1e-300:
        return best_score, best_seq_arr, candidates
    seq = seq / ss

    if not np.isfinite(score):
        c = fft_convolve_square(seq)
        score = objective_from_conv(c, n)

    if np.isfinite(score):
        candidates.append((float(score), n, seq.copy()))
        if score < best_score:
            best_score = float(score)
            best_seq_arr = seq.copy()

    return best_score, best_seq_arr, candidates


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    start_time = time.monotonic()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 2400.0
    if total_budget < 60.0:
        total_budget = 60.0

    deadline = start_time + max(60.0, total_budget - 20.0)
    rng = np.random.default_rng(seed)

    height_seq = None
    try:
        height_seq = height_sequence_1
    except NameError:
        height_seq = None

    if height_seq is not None:
        try:
            arr = np.asarray(height_seq, dtype=np.float64)
            if arr.size > 0:
                height_seq = arr
            else:
                height_seq = None
        except Exception:
            height_seq = None

    n_fallback = 1024
    best_seq_arr = np.ones(n_fallback, dtype=np.float64) / float(n_fallback)
    best_score = 2.0
    candidates = []
    candidates.append((2.0, n_fallback, best_seq_arr.copy()))
    print("initial best: %.6f" % best_score)

    # Quick diversified exploration.
    sid = 0
    quick_plan = (
        (1024, 3, 45.0),
        (2048, 3, 45.0),
        (4096, 2, 45.0),
        (8192, 2, 45.0),
    )

    for n, num_starts, sec in quick_plan:
        for i in range(num_starts):
            if time.monotonic() >= deadline - 75.0:
                break
            p = generate_start(n, sid + i, rng, height_seq)
            seq, score = local_search(n, p, sec, rng)
            best_score, best_seq_arr, candidates = consider_candidate(
                candidates, best_score, best_seq_arr, seq, score
            )
            print("quick n=%d start=%d score=%.6f best=%.6f" % (n, sid + i, score, best_score))
        sid += num_starts
        if time.monotonic() >= deadline - 75.0:
            break

    # Focus on the best candidates found.
    if candidates:
        candidates.sort(key=_candidate_key)
        top = candidates[:4]
        for i in range(len(top)):
            rem = deadline - time.monotonic()
            if rem < 90.0:
                break
            _, n_i, seq_i = top[i]
            focus_sec = min(300.0, max(20.0, rem / max(1, len(top) - i)))
            focus_sec = min(focus_sec, deadline - time.monotonic() - 30.0)
            if focus_sec < 15.0:
                break
            seq, score = local_search(n_i, seq_i, focus_sec, rng)
            best_score, best_seq_arr, candidates = consider_candidate(
                candidates, best_score, best_seq_arr, seq, score
            )
            print("focus n=%d score=%.6f best=%.6f" % (n_i, score, best_score))

    # Create a short version of the global best if it is long, for LP polishing.
    if best_seq_arr.size > 1024:
        small = resize_sequence(best_seq_arr, 1024)
        small_score = objective_from_conv(fft_convolve_square(small), 1024)
        best_score, best_seq_arr, candidates = consider_candidate(
            candidates, best_score, best_seq_arr, small, small_score
        )
        print("shortened best n=1024 score=%.6f global_best=%.6f" % (small_score, best_score))

    # LP polish on the best short candidates.
    if _HAVE_SCIPY_LP and candidates:
        candidates.sort(key=_candidate_key)
        lp_runs = 0
        for item in candidates:
            if lp_runs >= 2:
                break
            rem = deadline - time.monotonic()
            if rem < 120.0:
                break
            score_i, n_i, seq_i = item
            if n_i > 1024:
                continue
            if score_i > best_score + 0.15:
                continue
            lp_sec = min(300.0, rem - 30.0)
            if lp_sec < 30.0:
                break
            seq, score = lp_polish(seq_i, lp_sec, rng)
            best_score, best_seq_arr, candidates = consider_candidate(
                candidates, best_score, best_seq_arr, seq, score
            )
            print("lp n=%d score=%.6f best=%.6f" % (n_i, score, best_score))
            lp_runs += 1

    # Final long refinement of the current global best.
    final_sec = deadline - time.monotonic() - 15.0
    if final_sec > 20.0:
        n_final = int(best_seq_arr.size)
        seq, score = local_search(n_final, best_seq_arr, final_sec, rng)
        best_score, best_seq_arr, candidates = consider_candidate(
            candidates, best_score, best_seq_arr, seq, score
        )
        print("final n=%d score=%.6f best=%.6f" % (n_final, score, best_score))

    best_seq_arr = normalize_probability(best_seq_arr)
    final_list = [float(x) for x in best_seq_arr]

    try:
        exact = evaluate_sequence(final_list)
        print("exact score check: %.6f" % float(exact))
    except Exception:
        pass

    return final_list
```
