"""
Minimize  2*n*max_b / sum_a**2   (b = convolve(a,a),  a >= 0).

Key mathematical insight used here:
  The uniform sequence gives exactly 2.0 (the previous SOTA).  Small
  perturbations around uniform *increase* the score, so uniform is a
  local minimum for the naive search.  However, a MONOTONICALLY
  INCREASING sequence makes the central inner product <a, reverse(a)>
  small relative to (sum a)^2 (a and its reversal are "misaligned").

  For the polynomial family  a_i = (i+1)**p  (an increasing, convex
  sequence) the asymptotic score (independent of n in the large-n limit)
  is  2 * (p+1)^2 * Gamma(p+1)**2 / Gamma(2*p+2):
      p=1  -> 4/3  ~ 1.333
      p=2  -> 0.6
      p=3  -> ~0.23
  which is far below the 1.5027 target.  The evaluator clips every value
  to [0, 1000], so after building each shape we rescale it so its maximum
  is exactly 1000 (clipping then becomes a no-op).  Because the objective
  is scale-invariant, the scale choice does not affect the score.

Strategy
  1.  Build a pool of polynomial shapes (varying exponent p and length n),
      rescale each to max=1000, and evaluate with the real evaluator.
  2.  Keep the best shape found.
  3.  Run a black-box local search on the best shape (random multiplicative
      block noise + occasional rebuild at a new exponent) until the time
      budget expires, always tracked against the true evaluator so the best
      sequence found so far is preserved.
  4.  Return the best sequence (validated, non-negative, finite).
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Top-level helpers (no closures, no lambdas)
# ----------------------------------------------------------------------

def rescale_to_max(seq, cap=1000.0):
    """Scale a non-negative array so its maximum equals ``cap``.

    Keeps every value <= cap so the evaluator's own clipping is a no-op.
    The objective is scale-invariant, so this never changes the score.
    """
    arr = np.asarray(seq, dtype=np.float64)
    m = float(arr.max())
    if not np.isfinite(m) or m <= 1e-12:
        # Degenerate / zero sequence: return a safe uniform-ish fallback.
        out = np.full(arr.shape, 1.0)
        return out
    out = arr * (cap / m)
    # Guard against tiny float overshoots above cap.
    out = np.clip(out, 0.0, cap)
    return out


def poly_shape(n, p):
    """Build the polynomial / pseudo-polynomial shape  a_i = (i+1)**p  and
    rescale so max = 1000.  This is an increasing (anti-palindromic)
    non-negative sequence."""
    base = np.arange(1, n + 1, dtype=np.float64)
    if p == 1.0:
        raw = base.copy()
    else:
        # (i+1)**p computed stably.
        raw = base ** p
    return rescale_to_max(raw)


def safe_score(seq_list):
    """Wrap the evaluator so any failure returns +inf."""
    try:
        s = float(evaluate_sequence(seq_list))
    except Exception:
        return float(np.inf)
    return s


def block_perturb(arr, rng, sigma_rel=0.03):
    """Return a non-negative candidate by multiplying a random contiguous
    block (or the whole array) by log-normal noise."""
    cand = arr.copy()
    n = cand.size
    if rng.random() < 0.55:
        # random block
        width = int(rng.integers(max(1, n // 50), max(2, n // 3)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        cand[start:stop] *= np.exp(rng.normal(0.0, sigma_rel, size=stop - start))
    else:
        # gentle global tilt to explore the exponent direction
        tilt = np.exp(rng.normal(0.0, sigma_rel, size=n))
        tilt *= np.linspace(0.5, 1.5, n) ** rng.normal(0.0, 1.5)
        cand *= tilt
    cand = np.clip(cand, 0.0, None)
    return rescale_to_max(cand)


# ----------------------------------------------------------------------
# Main search
# ----------------------------------------------------------------------

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a low-cost non-negative step-function height sequence.

    See module docstring for the math.  Returns the best sequence found
    under ``budget_s``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(30.0, float(budget_s)) - 3.0

    best_seq = None
    best_score = float(np.inf)

    def consider(arr, tag=""):
        """Evaluate ``arr`` (a numpy array) and update the global best."""
        nonlocal best_seq, best_score
        arr = rescale_to_max(arr)
        if not (arr.size > 0 and np.isfinite(arr).all() and arr.max() > 1e-9):
            return
        score = safe_score(arr.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = arr.copy()
            if tag:
                now = time.monotonic()
                print(f"[{now - t0:7.1f}s] {tag}: score={score:.6f} (n={arr.size})")

    # ---- Stage 1: parametric pool of increasing polynomial shapes ----
    print("Stage 1: parametric polynomial pool")
    exponents = [1.0, 1.2, 1.4, 1.6, 1.8, 2.0, 2.2, 2.5, 2.8, 3.0, 3.5]
    lengths = [200, 400, 800, 1500, 2500, 4000]
    for n in lengths:
        # keep convolution cost reasonable in the pool
        if n > 4000:
            continue
        for p in exponents:
            if time.monotonic() > deadline:
                break
            consider(poly_shape(n, p), tag=f"poly n={n}, p={p}")
    print(f"  pool best score = {best_score:.6f}")

    # ---- Stage 2: also consider the provided starting construction ----
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0.0, None)
        if hs.ndim == 1 and hs.size > 0 and np.isfinite(hs).all():
            consider(hs, tag="height_sequence_1")
    except Exception as e:  # noqa: BLE001
        print("  (skipped provided sequence:", e, ")")

    # Safety fallback if nothing worked.
    if best_seq is None:
        best_seq = poly_shape(1000, 1.5)
        best_score = safe_score(best_seq.tolist())
        print(f"  fallback score = {best_score:.6f}")

    # ---- Stage 3: black-box local search on the best shape ----
    # Use a copy sized so that per-evaluation cost stays small enough to fit
    # many iterations before the deadline.
    if best_seq.size > 3000:
        # Down-sample the shape (resample the curve at a smaller length) so
        # the convolution in the evaluator is cheap during the search.
        m = 1500
        ii = np.linspace(0.0, 1.0, m)
        jj = (ii * (best_seq.size - 1)).astype(np.int64)
        work = best_seq[jj].copy()
    else:
        work = best_seq.copy()

    work = rescale_to_max(work)
    search_score = safe_score(work.tolist())
    print(f"Stage 3: local search, n={work.size}, start score={search_score:.6f}")

    it = 0
    no_improve = 0
    last_print = time.monotonic()
    while time.monotonic() < deadline:
        # occasionally rebuild the whole shape at a fresh exponent to escape
        # basin locking, otherwise do a local multiplicative perturbation.
        if rng.random() < 0.06:
            p = float(rng.uniform(1.2, 3.2))
            cand = poly_shape(work.size, p)
        else:
            sigma = float(rng.uniform(0.01, 0.06))
            cand = block_perturb(work, rng, sigma_rel=sigma)

        cand = rescale_to_max(cand)
        score = safe_score(cand.tolist())
        it += 1
        if np.isfinite(score) and score < search_score:
            search_score = score
            work = cand
            no_improve = 0
            # keep the global best in sync (search never returns worse than
            # the global best because we start from it, but sync anyway).
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
        else:
            no_improve += 1

        if it % 500 == 0:
            now = time.monotonic()
            print(f"[{now - t0:7.1f}s] local search it={it} "
                  f"search_score={search_score:.6f} best={best_score:.6f}")

        # adaptive early-stop is NOT used: keep polishing or exploring until
        # the hard deadline, since every iteration is cheap.

    # ---- Final validation and return ----
    out = rescale_to_max(best_seq)
    # ensure strictly valid: positive max, finite, non-negative
    out = np.clip(out, 0.0, 1000.0)
    finite = all(np.isfinite(x) for x in out)
    if not finite or out.max() <= 1e-9:
        out = poly_shape(1000, 1.5)

    final_score = safe_score(out.tolist())
    print(f"FINAL score = {final_score:.6f}  (n={out.size})")

    return [float(x) for x in out]
