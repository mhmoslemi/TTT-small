"""
Step-function coefficient search minimizing  2*n*max(conv(a,a))/(sum a)^2  .

The objective is scale-invariant, so the search normalizes every candidate to
sum(a)=1 and minimizes  2*n*max(conv(a,a)).  A pure uniform vector has
max(conv)=1/n (score 2.0).  To beat that we run projected gradient descent on a
smoothed-max (soft-max) surrogate of max(conv), with an annealed temperature and
a per-step backtracking line-search on the surrogate.  The gradient of the
soft-max surrogate is a single FFT-free cross-correlation with the softmax
weights.  The global best is tracked by the authoritative evaluate_sequence and
returned.  Several diverse seeds (ramp, parabola, bump, smooth-noise) are tried
and stagnation triggers a re-seed from the global best plus noise.
"""

import time
import numpy as np


# --------------------------------------------------------------------------- #
# Top-level helper functions (no closures, no lambdas)
# --------------------------------------------------------------------------- #
def _project_simplex(a):
    """Clip to non-negative and renormalize to sum == 1."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return np.full(a.shape, 1.0 / a.size, dtype=np.float64)
    return a / s


def _softmax_conv(conv, alpha):
    """Return (w, s) where w = softmax(alpha*conv) (sum 1) and s = smoothed max."""
    m = float(conv.max())
    x = alpha * (conv - m)
    e = np.exp(x)
    se = e.sum()
    if se <= 0.0 or not np.isfinite(se):
        # fall back to one-hot at the argmax
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
        s = m
        return w, s
    w = e / se
    s = m + float(np.log(se) / alpha)
    return w, s


def _grad_surrogate(a, w):
    """Gradient (up to a positive constant) of the soft-max surrogate w.r.t. a.

    d/da_i [sum_j w_j (a*a)_j]  ==  2 * sum_j w_j a_{j-i}
    which is a single cross-correlation, computed with one linear convolution
    against the reversed weight vector.  The constant factor 2 is irrelevant to
    the line-search and is omitted.
    """
    lenw = w.size
    full = np.convolve(a, w[::-1])
    idx = lenw - 1 - np.arange(a.size)
    return full[idx]


def _surrogate_value(a, alpha):
    conv = np.convolve(a, a)
    _, s = _softmax_conv(conv, alpha)
    return s


def _make_seeds(n, rng):
    """A handful of diverse, simplex-normalized starting vectors."""
    seeds = []
    # 1. uniform + tiny noise
    seeds.append(_project_simplex(np.ones(n) + 1e-3 * rng.standard_normal(n)))
    # 2. ramp
    seeds.append(_project_simplex(np.arange(n, dtype=np.float64) + 1.0))
    # 3. reverse ramp (same multiset-convex class but different orientation)
    seeds.append(_project_simplex(np.arange(n, dtype=np.float64) + 1.0)[::-1])
    # 4. parabola (concave bump)
    t = np.linspace(-1.0, 1.0, n)
    seeds.append(_project_simplex(np.maximum(1.0 - t * t, 0.0) + 1e-6))
    # 5. gaussian bump
    t = np.linspace(0.0, 1.0, n)
    seeds.append(_project_simplex(np.exp(-((t - 0.5) / 0.18) ** 2) + 1e-6))
    # 6. smooth random (cumsum of log-normal)
    smooth = np.cumsum(np.exp(0.6 * rng.standard_normal(n)))
    smooth = np.maximum(smooth, 0.0)
    if smooth.sum() > 0:
        seeds.append(_project_simplex(smooth))
    # 7. two-bump (breaks left-right symmetry strongly)
    t1 = np.linspace(0.0, 1.0, n)
    seeds.append(_project_simplex(
        np.exp(-((t1 - 0.25) / 0.12) ** 2) +
        0.7 * np.exp(-((t1 - 0.75) / 0.15) ** 2) + 1e-6))
    return seeds


def _gradient_descend(start, budget_s, alpha0, alpha1, seed):
    """Projected soft-max gradient descent with annealing and re-seeding.

    Returns the best sequence (as a normalized list) and its best score found
    by evaluate_sequence during the run.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s)

    working = np.asarray(start, dtype=np.float64).copy()
    working = _project_simplex(working)

    best = working.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    lr_list = (1.0 / np.sqrt(working.size)) * np.array(
        [0.05, 0.15, 0.4, 1.0, 2.5, 6.0], dtype=np.float64)

    iters = 0
    no_improve = 0
    eval_every = 12

    while time.monotonic() < deadline:
        iters += 1
        frac = min(1.0, iters / 2500.0)
        alpha = alpha0 + (alpha1 - alpha0) * frac

        conv = np.convolve(working, working)
        w, s_cur = _softmax_conv(conv, alpha)
        if not np.isfinite(s_cur):
            break
        grad = _grad_surrogate(working, w)
        gnorm = float(np.linalg.norm(grad))
        if gnorm <= 1e-300:
            break

        # backtracking line-search on the surrogate value
        best_cand = None
        best_s = np.inf
        for lr in lr_list:
            cand = _project_simplex(working - lr * grad)
            cs = _surrogate_value(cand, alpha)
            if cs < best_s:
                best_s = cs
                best_cand = cand
        if best_cand is None:
            break

        # accept if we made any (tiny) decrease, otherwise add a nudge so we
        # are not perfectly stuck
        if best_s <= s_cur * (1.0 + 1e-12):
            working = best_cand
        else:
            working = _project_simplex(best_cand + 1e-4 * rng.standard_normal(working.size))

        if (iters % eval_every) == 0:
            score = float(evaluate_sequence(working.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = _project_simplex(working)
                no_improve = 0
            else:
                no_improve += eval_every

        # stagnation re-seed from the global best, with a fresh temperature
        if no_improve >= 480:
            noise = np.exp(0.05 * rng.standard_normal(best.size))
            working = _project_simplex(best * noise)
            no_improve = 0

    final_score = float(evaluate_sequence(best.tolist()))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)
        best = _project_simplex(best)
    return best, best_score


def _pick_best_seed(seeds):
    rng = np.random.default_rng(0)
    scored = []
    for s in seeds:
        sc = float(evaluate_sequence(_project_simplex(s).tolist()))
        scored.append((sc if np.isfinite(sc) else np.inf, s))
    scored.sort(key=lambda p: p[0])
    return scored[0][1]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search entry point.  Returns a list of non-negative floats."""
    rng = np.random.default_rng(seed)

    # Choose a working length: a few thousand terms gives plenty of degrees of
    # freedom to flatten the convolution peak while keeping each O(n^2) convolve
    # (both ours and the evaluator's) fast enough for thousands of iterations.
    n = 3809
    seeds = _make_seeds(n, rng)
    start = _pick_best_seed(seeds)

    remainder = max(0.0, float(budget_s) - 8.0)
    if remainder <= 0.0:
        rem = _project_simplex(start)
        score = float(evaluate_sequence(rem.tolist()))
        if not np.isfinite(score):
            rem = np.ones(n) / n
        return [float(x) for x in rem]

    # Run several independent gradient-descent trajectories, each starting from
    # current global best plus a different perturbation / seed, splitting the
    # remaining time.  The best over all trajectories is returned.
    best = start.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    tries = 4
    per_try = remainder / float(tries)
    for k in range(tries):
        start = _project_simplex(best)
        if k == 0:
            init = start
        else:
            scale = 0.03 + 0.04 * k
            init = _project_simplex(best * np.exp(scale * rng.standard_normal(best.size)))
        cand, sc = _gradient_descend(init, per_try, alpha0=8.0, alpha1=180.0,
                                     seed=1000 + k)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = cand
        print("try", k, "score", best_score, flush=True)

    # final guaranteed-valid fallback
    out = _project_simplex(best)
    ssum = float(out.sum())
    if ssum < 0.01 or not np.isfinite(ssum) or not np.isfinite(out).all():
        out = np.ones(n) / n
    score = float(evaluate_sequence([float(x) for x in out]))
    print("final score", score, "n", n, flush=True)
    return [float(x) for x in out]
