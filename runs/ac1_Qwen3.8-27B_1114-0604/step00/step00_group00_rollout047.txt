"""
Gradient descent with log-sum-exp smoothing to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses softmax parameterization for automatic simplex constraint satisfaction.
Multiple restarts with diverse initializations, adaptive temperature and step size.
Final phase: local random perturbations around the best found solution.

import numpy as np
import time


def fft_self_conv(a, L):
    """Compute linear self-convolution of a via FFT. Returns array of length L (first 2*len(a)-1 are valid)."""
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)


def grad_lse_wrt_a(a, p, L2):
    """
    Compute g[k] = 2 * sum_j a[j] * p[j+k] for k=0,...,n-1.
    This is the gradient of LSE(T*conv(a,a)) w.r.t. a.
    Uses FFT: g = 2 * (a_rev * p)[n-1 : 2n-1] where a_rev = a[::-1].
    """
    n = len(a)
    a_rev = a[::-1]
    Ap = np.fft.rfft(p, L2)
    Ar = np.fft.rfft(a_rev, L2)
    conv_full = np.fft.irfft(Ar * Ap, L2)
    return 2.0 * conv_full[n-1:2*n-1]
