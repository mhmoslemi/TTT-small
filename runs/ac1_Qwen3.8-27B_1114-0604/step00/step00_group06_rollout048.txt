def resample_sequence(parent, n):
   if parent is None or len<=1: return None
   p=np.clip(parent,0,1000)
   if p.size == n: return p
   if p.size > n:
       # average blocks
       if p.size % n ==0: return p.reshape(n,-1).mean(axis=1)
       # take nearest
       idx=np.linspace(0,p.size-1,n).astype(int); return p[idx]
   else:
       # interpolate
       x_old=np.linspace(0,1,p.size); x_new=np.linspace(0,1,n); return np.interp(x_new,x_old,p)

def _next_fast_size(x):
    x = int(max(1, x))
    if x == 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_sequence(x, target=1.0):
    x = np.asarray(x, dtype=np.float64).copy()
    if x.size == 0:
        return x
    bad = ~np.isfinite(x)
    if np.any(bad):
        x[bad] = 0.0
    x = np.clip(x, 0.0, 1000.0)
    s = float(x.sum())
    if s <= 1e-300:
        x.fill(float(target) / float(x.size))
    else:
        x *= float(target) / s
    return x


def full_convolution_fft(a, L=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    if L is None or int(L) < m:
        L = _next_fast_size(m)
    else:
        L = int(L)
    fa = np.fft.rfft(a, L)
    out = np.fft.irfft(fa * fa, L)[:m]
    out = np.real(out)
    out = np.maximum(out, 0.0)
    return out


def score_normalized(c, n):
    if c.size == 0 or n <= 0:
        return np.inf
    cm = float(c.max())
    if not np.isfinite(cm) or cm < 0.0:
        return np.inf
    return float(2.0 * n * cm)


def make_reverse_phase(n, L):
    freq = np.fft.rfftfreq(int(L))
    return np.exp(-2j * np.pi * freq * (int(n) - 1))


def direction_from_c(a, c, alpha, L, phase):
    n = int(a.size)
    if n <= 1:
        return np.zeros(n, dtype=np.float64)
    m = 2 * n - 1
    cmax = float(c.max())
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.zeros(n, dtype=np.float64)

    x = c / cmax
    np.clip(x, 0.0, 1.0, out=x)

    if float(alpha) <= 0.0:
        w = np.full(m, 1.0 / float(m), dtype=np.float64)
    else:

    fw = np.fft.rfft(w, L)

    if n <= 0:

        return np.ones(1, dtype=np.float64)
    cur = np.array([1.0, -1.0], dtype=np.float64)

def balanced_sign_sequence(n, rng):

        return None

        return p.copy()
    if p.size > n:

    old_x = np.linspace(0.0, 1.0, p.size)

        high = max(low + 1, n // 4)

            last_print = now

        it += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start = time.monotonic()
    total_budget = float(max(0.0, budget_s))
    if total_budget < 1.0:
        return [1.0]

    rng = np.random.default_rng(seed)
    deadline = start + max(1.0, total_budget - 6.0)
    parent = get_parent_sequence()

    if total_budget < 700.0:
        lengths = [128, 256, 512]
    elif total_budget < 1600.0:
        lengths = [128, 256, 512, 1024, 2048]
    else:
        lengths = [128, 256, 512, 1024, 2048, 4096]

    results = []
    for n in lengths:
        for kind in BASE_KINDS:
            if kind == 'parent' and parent is None:
                continue
            reps = 3 if kind in FIRST_RANDOM_KINDS else 1
            for rep in range(reps):
                init = make_initial(n, kind, rng, parent)
                if init is None:
                    continue
                init = normalize_sequence(init, 1.0)
                init = np.maximum(init, 1e-12)
                init = normalize_sequence(init, 1.0)
                c = full_convolution_fft(init, None)
                sc = score_normalized(c, n)
                if np.isfinite(sc):
                    results.append((sc, n, kind, rep, init))

    if not results:
        fallback = np.ones(256, dtype=np.float64)
        try:
            auth = evaluate_sequence(fallback.tolist())
        except Exception:
            auth = np.inf
        if not np.isfinite(auth):
            fallback = np.ones(128, dtype=np.float64)
        return [float(x) for x in fallback]

    results_sorted = sorted(results, key=first_key)
    selected = []
    used = set()

    # Add a few overall best initial seeds.
    for r in results_sorted:
        if len(selected) >= 4:
            break
        key = (r[1], r[2], r[3])
        if key not in used:
            used.add(key)
            selected.append(r)

    # Ensure each length is represented at least once.
    for nn in lengths:
        for r in results_sorted:
            if r[1] != nn:
                continue
            key = (r[1], r[2], r[3])
            if key not in used:
                used.add(key)
                selected.append(r)
                break

    # Add diverse kinds.
    for kind in BASE_KINDS:
        if len(selected) >= 14:
            break
        for r in results_sorted:
            if r[2] != kind:
                continue
            key = (r[1], r[2], r[3])
            if key not in used:
                used.add(key)
                selected.append(r)
                break

    selected = sorted(selected, key=first_key)

    remaining = max(0.0, deadline - time.monotonic())
    refine_reserve = max(90.0, 0.22 * remaining)
    candidate_time = max(30.0, remaining - refine_reserve)

    M = min(10, len(selected))
    while M > 1 and candidate_time / float(M) < 18.0:
        M -= 1
    selected = selected[:M]

    alloc_fracs = []
    rem = 1.0
    if len(alloc_fracs) < M:
        val = 1.0 if M == 1 else 0.42
        val = min(rem, val)
        alloc_fracs.append(val)
        rem -= val
    if len(alloc_fracs) < M:
        val = 0.26
        val = min(rem, val)
        alloc_fracs.append(val)
        rem -= val
    if len(alloc_fracs) < M:
        val = 0.16
        val = min(rem, val)
        alloc_fracs.append(val)
        rem -= val
    while len(alloc_fracs) < M:
        cnt = max(1, M - len(alloc_fracs))
        val = max(0.0, rem / float(cnt))
        alloc_fracs.append(val)
        rem -= val

    best_seq = None
    best_score = np.inf

    for i, r in enumerate(selected):
        now = time.monotonic()
        alloc = candidate_time * float(alloc_fracs[i])
        alloc = min(alloc, max(0.0, deadline - refine_reserve - now))
        if alloc < 6.0:
            break
        sub_end = now + alloc
        init = r[4]
        tag = "init"
        try:
            local_best, local_score = optimize_candidate(init, sub_end, rng, tag=tag)
        except Exception:
            local_best, local_score = init, r[0]

        local_best = np.asarray(local_best, dtype=np.float64)
        local_best = np.clip(local_best, 0.0, 1000.0)
        sb = float(local_best.sum())
        if local_best.size == 0 or sb < 1e-12 or not np.isfinite(sb):
            continue
        local_best = normalize_sequence(local_best, 1.0)

        auth = np.inf
        try:
            auth = evaluate_sequence(local_best.tolist())
        except Exception:
            auth = np.inf

        use_score = auth if np.isfinite(auth) else float(local_score)
        if best_seq is None or use_score < best_score:
            best_seq = local_best.copy()
            best_score = use_score
            print(f"[propose] new best score={best_score:.6f} n={local_best.size} alloc={alloc:.1f}")

    now = time.monotonic()
    if best_seq is None:
        best_seq = normalize_sequence(results_sorted[0][4], 1.0)
        best_score = results_sorted[0][0]

    remaining = max(0.0, deadline - now)
    if remaining > 45.0:
        polish_time = max(12.0, 0.55 * remaining - 5.0)
        end = time.monotonic() + polish_time
        try:
            local_best, local_score = optimize_candidate(best_seq, end, rng, tag="polish")
        except Exception:
            local_best, local_score = best_seq, best_score
        local_best = np.asarray(local_best, dtype=np.float64)
        local_best = np.clip(local_best, 0.0, 1000.0)
        if local_best.size > 0 and float(local_best.sum()) > 1e-12:
            local_best = normalize_sequence(local_best, 1.0)
            auth = np.inf
            try:
                auth = evaluate_sequence(local_best.tolist())
            except Exception:
                auth = np.inf
            use_score = auth if np.isfinite(auth) else float(local_score)
            if use_score < best_score:
                best_seq = local_best.copy()
                best_score = use_score
                print(f"[propose] polished best score={best_score:.6f} n={local_best.size}")

    now = time.monotonic()
    remaining = max(0.0, deadline - now)
    if remaining > 45.0:
        num = 3 if remaining > 220.0 else 1
        sub = (remaining - 20.0) / float(max(1, num))
        for k in range(num):
            now = time.monotonic()
            take = min(sub, max(0.0, deadline - 15.0 - now))
            if take < 12.0:
                break
            sub_end = now + take
            pert = stochastic_perturb(best_seq, rng, strength=0.18 + 0.12 * float(k))
            try:
                local_best, local_score = optimize_candidate(pert, sub_end, rng, tag=f"pert{k}")
            except Exception:
                local_best, local_score = pert, best_score
            local_best = np.asarray(local_best, dtype=np.float64)
            local_best = np.clip(local_best, 0.0, 1000.0)
            if local_best.size > 0 and float(local_best.sum()) > 1e-12:
                local_best = normalize_sequence(local_best, 1.0)
                auth = np.inf
                try:
                    auth = evaluate_sequence(local_best.tolist())
                except Exception:
                    auth = np.inf
                use_score = auth if np.isfinite(auth) else float(local_score)
                if use_score < best_score:
                    best_seq = local_best.copy()
                    best_score = use_score
                    print(f"[propose] perturbed best score={best_score:.6f} n={local_best.size}")

    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.size == 0 or float(best_seq.sum()) < 1e-12:
        best_seq = np.ones(256, dtype=np.float64)
    else:
        best_seq = normalize_sequence(best_seq, 1.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        if not np.isfinite(final_score):
            fb = np.ones(256, dtype=np.float64)
            fb_score = evaluate_sequence(fb.tolist())
            if np.isfinite(fb_score):
                best_seq = fb
    except Exception:
        pass

    return [float(x) for x in best_seq]
