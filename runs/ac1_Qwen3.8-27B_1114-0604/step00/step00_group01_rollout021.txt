from scipy.optimize import linprog

def solve_lp(f, M, n):
    # Maximize sum(g) subject to g >= 0 and (f*g)[k] <= M for all k
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
    # A matrix with shape (2n-1, n)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(k, n-1)+1):
            idx = k - j  # index in g
            A[k, idx] = f[j]
    
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    return res.x
```

    conv = np.convolve(a, a)
    S = a.sum()
    M = conv.max()
    
    # Find top peaks
    top_indices = np.argsort(conv)[-n_peaks:][::-1]
    
    # Compute pressure
    pressure = np.zeros_like(a)
    for k in top_indices:
        # (a*a)[k] = sum_j a[j]*a[k-j]
        # contribution to pressure on a[m] is a[k-m]

            if 0 <= m < n:
                pressure[m] += a[j]
    
    # Create new direction: reduce where pressure is high
    # g[m] proportional to a[m] / (1 + pressure[m]/M * some_scale)
    scale = pressure.max() if pressure.max() > 0 else 1
    g = a / (1 + pressure / scale)
    g = g / g.sum() * S
    
    # Update
    new_a = (1 - t) * a + t * g
    return np.clip(new_a, 0, 1000)

def gradient_step(a, lr, eps=1e-6):
    n = len(a)
    conv = np.convolve(a, a)
    M = conv.max()
    
    # Active set: positions where conv is close to max
    active = np.where(conv >= M - eps * M)[0]
    
    # Subgradient of max(conv)
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            m = k - j
            grad[m] += a[j]
    grad *= 2.0 / len(active)
    
    # Center the gradient (to maintain sum constraint)
    grad -= grad.mean()
    
    # Update
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    a_new = a_new * n / a_new.sum()  # rescale to sum = n
    return a_new

At the optimum, we'd expect several positions in the convolution to be at the maximum (active constraints). The KKT conditions would say:
- For each active k: d/d a[m] [(a*a)[k]] = lambda + mu[m] for all m
  where lambda is the Lagrange multiplier for the sum constraint and mu[m] >= 0 is for a[m] >= 0 (with mu[m]*a[m] = 0).

So `a[k-m]` should be constant (summed over active k) for all m where a[m] > 0. This is a system of equations that determines the optimal a.

def aggressive_step(a, p, lr):
    """Directly adjust a based on the peak distribution p."""
    # p[k] is the "weight" of each convolution position
    # We want to reduce a where it contributes to peaks
    # Pressure on a[m] = sum_k p[k] * a[k-m]
    # New a should be: a[m] * (1 - lr * pressure[m] / mean_pressure)
    
    # Compute pressure via FFT (same as soft_grad but without the factor of 2)
    n = len(a)
    a_rev = a[::-1]
    L = next_pow2(3 * n - 1)
    A = np.zeros(L); A[:n] = a_rev
    P = np.zeros(L); P[:2*n-1] = p
    corr = np.fft.irfft(np.fft.rfft(A) * np.fft.rfft(P), L)
    pressure = corr[n-1:2*n-1]
    
    mean_p = pressure.mean()
    ratio = pressure / max(mean_p, 1e-30)
    a_new = a * (1 - lr * (ratio - 1))
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new
```
