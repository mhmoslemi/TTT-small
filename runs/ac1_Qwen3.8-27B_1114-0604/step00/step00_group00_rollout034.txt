"""
LP-constraint-generation optimizer for minimizing 2n*max(a*a)/(sum a)^2.
Uses asymmetric initializations (score < 2 requires asymmetry), reduced LP
with constraint generation for the update direction, and FFT-based fast
evaluation for the line search.

import numpy as np
import scipy.optimize
import time


def convolve_fast(a, b):
    """Full linear convolution via FFT, O(n log n)."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    N = 1 << (size - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:size]


def fast_score(a):
    """Compute 2n*max(a*a)/(sum a)^2 using FFT."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    M = convolve_fast(a, a).max()
    return 2.0 * n * M / (S * S)


def solve_lp(a, M, lags):
    """Solve max 1'g s.t. (a*g)[k]<=M for k in lags, g>=0. Returns g or None."""
    n = len(a)
    K = len(lags)
    if K == 0:
        return np.ones(n)
    
    A_ub = np.zeros((K, n))
    for idx, k in enumerate(lags):
        j0 = max(0, k - n + 1)
        j1 = min(n, k + 1)
        if j0 < j1:
            A_ub[idx, j0:j1] = a[k - np.arange(j0, j1)]
    
    try:
        res = scipy.optimize.linprog(
            -np.ones(n), A_ub=A_ub, b_ub=np.full(K, M),
            bounds=[(0, None)] * n, method='highs',
            options={'presolve': True}
        )
        if res.success:
            return res.x
    except Exception as e:
        print(f"  LP error: {e}")
    return None


def lp_direction(a, max_lags=250, max_inner=10):
    """Find LP-based update direction g."""
    n = len(a)
    b_aa = convolve_fast(a, a)
    M = b_aa.max()
    
    active = np.argsort(b_aa)[-max_lags:]
    g = None
    
    for _ in range(max_inner):
        g = solve_lp(a, M, active)
        if g is None:
            return None
        ag = convolve_fast(a, g)
        viol = np.where(ag > M * (1 + 1e-9) + 1e-12)[0]
        if len(viol) == 0:
            break
        active = np.unique(np.concatenate([active, viol]))
        if len(active) > 600:
            v = ag[active]
            active = active[np.argsort(v)[-400:]]
    
    return g

    x = np.arange(n, dtype=np.float64)

        c = rng.uniform(0.2, 0.8) * n
        s = rng.uniform(0.1, 0.4) * n
        return np.exp(-0.5 * ((x - c) / s) ** 2) + 1e-6
    elif mode == 1:
        return np.linspace(0.05, 1.0, n) * np.exp(rng.normal(0, 0.3, n)) + 1e-6
    elif mode == 2:

    elif mode == 3:
        c1, c2 = sorted(rng.uniform(0.15, 0.85, 2) * n)
        s = rng.uniform(0.05, 0.15) * n
        return (np.exp(-0.5*((x-c1)/s)**2) + 0.3*np.exp(-0.5*((x-c2)/s)**2)) + 1e-6
    elif mode == 4:
        return np.maximum(0.01, np.cos(np.pi*(x/n - 0.1) - 0.2)**2)
    else:
        return np.exp(rng.normal(0, 0.7, n)) + 1e-6
