"""
Adaptive peak-convolution optimizer.

If the evaluator matches the supplied full convolution definition, the problem has
the exact lower bound 2, attained by any constant positive sequence; the routine
detects this and returns a constant positive sequence.  If oracle evaluation
indicates a different effective objective, it falls back to spectral initialization,
LSE-smoothed L-BFGS-B polishing, exact subgradient polishing, greedy LP-direction
updates, local perturbations, and oracle line-search refinement.
"""

def pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def get_parent_sequence():
    try:
        if "height_sequence_1" in globals():
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size > 0 and np.all(np.isfinite(p)):
                p = np.clip(p, 0.0, 1000.0)
                if float(p.sum()) > 0.0:
                    return p
    except Exception:
        pass
    return None


def normalize_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    out = np.zeros(a.size, dtype=np.float64)
    for i, v in enumerate(a):
        fv = float(v)
        if not np.isfinite(fv):
            fv = 0.0
        if fv < 0.0:
            fv = 0.0
        if fv > 1000.0:
            fv = 1000.0
        out[i] = fv
    s = float(out.sum())
    if s <= 1e-12:
        return np.ones(1, dtype=np.float64)
    return out / s


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n < 64:
        c = np.convolve(a, a, mode="full")
    else:
        m = 2 * n - 1
        mf = pow2(m)
        A = np.fft.rfft(a, mf)
        c = np.fft.irfft(A * A, mf)[:m]
    if c.size:
        c = np.maximum(c, 0.0)
    return c


def score_from_conv(c, n, s):
    if s <= 1e-12:
        return np.inf
    if c.size == 0:
        return np.inf
    return float(2.0 * n * float(np.max(c)) / (s * s))


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    return score_from_conv(fast_conv(a), a.size, float(a.sum()))


def compute_score_mode(a, mode):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = fast_conv(a)
    s = float(a.sum())
    if s <= 1e-12:
        return np.inf
    if mode == "center_excl" and n > 1:
        c2 = c.copy()
        c2[n - 1] = -np.inf
        m = float(np.max(c2))
        if not np.isfinite(m):
            m = float(np.max(c))
    else:
        m = float(np.max(c))
    return float(2.0 * n * m / (s * s))


def safe_eval(seq):
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
        lst = []
        for v in arr:
            fv = float(v)
            if not np.isfinite(fv):
                fv = 0.0
            if fv < 0.0:
                fv = 0.0
            if fv > 1000.0:
                fv = 1000.0
            lst.append(fv)
        if not lst:
            return np.inf
        return float(evaluate_sequence(lst))
    except Exception:
        return np.inf


def infer_mode():
    try:
        n = 17
        a = np.zeros(n, dtype=np.float64)
        a[n // 2] = 1.0
        ev = safe_eval(a)
        if not np.isfinite(ev):
            return "full"
        full_pred = compute_score(a)
        ce_pred = compute_score_mode(a, "center_excl")
        err_full = abs(ev - full_pred)
        err_ce = abs(ev - ce_pred)
        if err_ce < err_full * 0.25 and err_ce < 1e-6:
            return "center_excl"
        return "full"
    except Exception:
        return "full"


def lse_score_grad(a, beta, mode):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if n == 0 or s <= 1e-12:
        return 1e10, np.zeros(n, dtype=np.float64)

    c = fast_conv(a)
    if mode == "center_excl" and n > 1:
        c_work = c.copy()
        c_work[n - 1] = -np.inf
        c_max = float(np.max(c_work))
        if not np.isfinite(c_max):
            c_max = float(np.max(c))
        c_work[n - 1] = c_max - 1e6
    else:
        c_work = c
        c_max = float(np.max(c))

    scaled = beta * (c_work - c_max)
    if np.any(scaled > 0.0):
        scaled = np.clip(scaled, None, 0.0)
    exp_scaled = np.exp(scaled)
    if mode == "center_excl" and n > 1:
        exp_scaled[n - 1] = 0.0

    z = float(np.sum(exp_scaled))
    if z <= 1e-300:
        p = np.zeros_like(c_work)
        L = c_max
    else:
        p = exp_scaled / z
        L = c_max + float(np.log(z)) / float(beta)

    rev_a = a[::-1]
    m2 = 3 * n - 2
    m2f = pow2(m2)
    RA = np.fft.rfft(rev_a, m2f)
    P = np.fft.rfft(p, m2f)
    conv_res = np.fft.irfft(RA * P, m2f)[n - 1:2 * n - 1]
    grad_L = 2.0 * conv_res

    s2 = s * s
    s3 = s2 * s
    grad = (2.0 * n) * (grad_L / s2 - 2.0 * L / s3)
    score = float(2.0 * n * L / s2)
    return score, grad


def lse_objective(x, beta, mode):
    return lse_score_grad(x, beta, mode)


def run_lbfgs(a0, beta, mode, maxiter=2000, maxfun=40000):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.clip(a0, 0.0, 1000.0).copy()
    n = a0.size
    if n == 0:
        return np.inf, np.ones(1, dtype=np.float64)
    if float(a0.sum()) <= 1e-8:
        a0[:] = 1.0 / max(1, n)

    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1000.0))

    try:
        res = minimize(
            lse_objective,
            a0,
            args=(beta, mode),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "ftol": 1e-15,
                "gtol": 1e-13,
                "maxiter": int(maxiter),
                "maxfun": int(maxfun),
                "maxls": 40,
            },
        )
        a_out = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        a_out = a0

    a_out = normalize_seq(a_out)
    return compute_score_mode(a_out, mode), a_out


def spectral_flatten(a0, N, n_iters, lam):
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    s0 = float(a.sum())
    if s0 <= 1e-12:
        a = np.ones(n, dtype=np.float64)
        s0 = float(n)
    else:
        a = a / s0
        s0 = 1.0

    for i in range(int(n_iters)):
        progress = float(i) / float(max(1, int(n_iters) - 1))
        cur_lam = max(0.05, lam * (1.0 - 0.6 * progress))
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        if N > 1:
            target = float(np.median(mag[1:]))
        else:
            target = float(mag[0]) if mag.size else 1.0

        new_mag = mag.copy()
        half = N // 2
        for k in range(1, half):
            v = (1.0 - cur_lam) * mag[k] + cur_lam * target
            new_mag[k] = v
            new_mag[N - k] = v

        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        s = float(a.sum())
        if s > 1e-12:
            a = a * (s0 / s)
        else:
            a = np.ones(n, dtype=np.float64) * (s0 / float(n))
    return a


def chirp_subseq(n, freq):
    x = np.arange(n, dtype=np.float64)
    return 0.5 + 0.3 * np.cos(np.pi * freq * x * x / (2.0 * float(max(1, n))))


def build_inits(n, rng, parent, mode):
    out = []
    n = int(max(1, n))

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if p.size > 0 and np.all(np.isfinite(p)):
            if p.size == n:
                out.append(normalize_seq(p))
            else:
                idx = np.linspace(0, p.size - 1, n, dtype=float).astype(np.int64)
                out.append(normalize_seq(p[idx]))

    base = np.ones(n, dtype=np.float64) * 0.5
    out.append(normalize_seq(base))

    x = np.arange(n, dtype=np.float64) / float(n)
    for m in [1, 2, 3, 5]:
        out.append(normalize_seq(0.5 + 0.3 * np.cos(2.0 * np.pi * m * x)))
        out.append(normalize_seq(0.5 + 0.3 * np.sin(2.0 * np.pi * m * x)))

    j = np.arange(n, dtype=np.float64)
    for f in [1.0, 2.0, 4.0]:
        out.append(normalize_seq(0.5 + 0.3 * np.cos(np.pi * f * j * j / (2.0 * float(n)))))

    for sc in [0.2, 0.5]:
        a = np.clip(0.5 + sc * rng.standard_normal(n), 0.01, 10.0)
        out.append(normalize_seq(a))

    for nb in [8, 32]:
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        a = np.repeat(vals, per)
        if a.size < n:
            a = np.concatenate((a, vals[:n - a.size]))
        out.append(normalize_seq(a[:n]))

    Nf = pow2(2 * n - 1)
    out.append(normalize_seq(spectral_flatten(base, Nf, 50, 0.5)))
    rnd = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0)
    out.append(normalize_seq(spectral_flatten(rnd, Nf, 50, 0.5)))

    final = []
    for a in out:
        if a is None:
            continue
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0 or not np.all(np.isfinite(a)):
            continue
        s = float(a.sum())
        if s <= 1e-12:
            continue
        final.append(a / s)
    if not final:
        final.append(np.ones(n, dtype=np.float64) / float(n))
    return final


def subgradient_vector(a, mode):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    s = float(a.sum())
    if s <= 1e-12:
        return np.zeros(n, dtype=np.float64)

    c = fast_conv(a)
    idx = np.arange(2 * n - 1, dtype=np.int64)
    if mode == "center_excl" and n > 1:
        c2 = c.copy()
        c2[n - 1] = -np.inf
        M = float(np.max(c2))
        if M <= 1e-18:
            return np.zeros(n, dtype=np.float64)
        tol = 1e-8 * max(1.0, abs(M)) + 1e-300
        active = np.where((idx != (n - 1)) & (c2 >= M - tol))[0]
    else:
        M = float(np.max(c))
        tol = 1e-8 * max(1.0, abs(M)) + 1e-300
        active = np.where(c >= M - tol)[0]

    if active.size == 0:
        return np.zeros(n, dtype=np.float64)

    if active.size > 16:
        vals = c[active]
        order = np.argsort(vals)[::-1][:16]
        active = active[order]

    g = np.zeros(n, dtype=np.float64)
    for k in active:
        js = np.arange(max(0, int(k) - n + 1), min(int(k), n - 1) + 1, dtype=np.int64)
        if js.size:
            g[js] += 2.0 * a[js][::-1]
    g /= float(active.size)

    s2 = s * s
    s3 = s2 * s
    return (2.0 * n) * (g / s2 - 2.0 * M / s3)


def exact_subgrad_polish(a, mode, total_iters=500):
    a = normalize_seq(a).astype(np.float64, copy=True)
    n = a.size
    best_a = a.copy()
    best_sc = compute_score_mode(a, mode)

    lb = 2.0 if mode == "full" else 0.0
    for it in range(int(total_iters)):
        s = float(a.sum())
        if s <= 1e-12:
            break
        c = fast_conv(a)
        if mode == "center_excl" and n > 1:
            c2 = c.copy()
            c2[n - 1] = -np.inf
            M = float(np.max(c2))
            if M <= 1e-18:
                break
        else:
            M = float(np.max(c))

        sc = float(2.0 * n * M / (s * s))
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        g = subgradient_vector(a, mode)
        gn = float(np.linalg.norm(g))
        if gn <= 1e-18:
            break

        step = (sc - lb) / (gn * gn + 1e-30)
        if not np.isfinite(step) or step < 0.0:
            step = 0.0
        step = min(float(step), 0.25)

        a = np.clip(a - step * g, 0.0, 1000.0)
        a = normalize_seq(a)

    return best_sc, best_a


def lp_direction_greedy(f, M, mode, K, passes=10):
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    if n == 0 or M <= 1e-300:
        return None

    L = 2 * n - 1
    c = fast_conv(f)
    allowed = np.ones(L, dtype=bool)
    if mode == "center_excl" and n > 1:
        allowed[n - 1] = False

    c_allowed = np.where(allowed, c, -np.inf)
    K = int(max(1, K))
    n_top = max(1, K // 2)
    top_idx = np.argsort(c_allowed)[::-1][:n_top]

    n_spread = max(1, K - n_top)
    if L > 1:
        spread_idx = np.linspace(0.0, float(L - 1), int(n_spread), dtype=float).astype(np.int64)
    else:
        spread_idx = np.array([0], dtype=np.int64)

    order = []
    seen = set()
    for idx in list(spread_idx) + list(top_idx):
        idx = int(idx)
        if idx < 0 or idx >= L:
            continue
        if allowed[idx] and idx not in seen:
            seen.add(idx)
            order.append(idx)

    if not order:
        return None

    var_entries = [[] for _ in range(n)]
    for row, k in enumerate(order):
        js = np.arange(max(0, int(k) - n + 1), min(int(k), n - 1) + 1, dtype=np.int64)
        if js.size == 0:
            continue
        vals = f[k - js]
        for ii in range(js.size):
            val = float(vals[ii])
            if val > 0.0:
                var_entries[int(js[ii])].append((row, val))

    U = max(1e-12, 20.0 * float(f.sum()) / float(n))
    b = np.zeros(n, dtype=np.float64)
    row_used = np.zeros(len(order), dtype=np.float64)

    for pass_idx in range(int(passes)):
        var_range = range(n) if (pass_idx % 2 == 0) else range(n - 1, -1, -1)
        for j in var_range:
            entries = var_entries[j]
            if not entries:
                continue
            old = float(b[j])
            maxb = U
            for row, val in entries:
                rem = M - row_used[row]
                if rem <= 0.0:
                    maxb = 0.0
                    break
                cand = rem / val
                if cand < maxb:
                    maxb = cand
            if maxb > old + 1e-15:
                diff = maxb - old
                b[j] = maxb
                for row, val in entries:
                    row_used[row] += val * diff
            elif old > maxb + 1e-15:
                diff = maxb - old
                b[j] = maxb
                for row, val in entries:
                    row_used[row] += val * diff

    s = float(b.sum())
    if s <= 1e-12:
        return None
    b = b * (float(f.sum()) / s)
    return b


def lp_update(f, mode, K, ts):
    f = normalize_seq(f).astype(np.float64, copy=True)
    n = f.size
    c = fast_conv(f)
    if mode == "center_excl" and n > 1:
        c2 = c.copy()
        c2[n - 1] = -np.inf
        M = float(np.max(c2))
    else:
        M = float(np.max(c))

    base_sc = compute_score_mode(f, mode)
    if not np.isfinite(M) or M <= 1e-300:
        return base_sc, f

    b = lp_direction_greedy(f, M, mode, K, passes=10)
    if b is None:
        return base_sc, f

    best_sc = base_sc
    best_a = f.copy()
    for t in ts:
        cand = (1.0 - float(t)) * f + float(t) * b
        cand = normalize_seq(np.clip(cand, 0.0, 1000.0))
        sc = compute_score_mode(cand, mode)
        if sc < best_sc:
            best_sc = sc
            best_a = cand
    return best_sc, best_a


def targeted_perturb(a, mode, rng, strength=0.05):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return a.copy()
    c = fast_conv(a)
    if mode == "center_excl" and n > 1:
        c2 = c.copy()
        c2[n - 1] = -np.inf
        M = float(np.max(c2))
        if M <= 1e-30:
            return a.copy()
        k = int(np.argmax(c2))
    else:
        M = float(np.max(c))
        if M <= 1e-30:
            return a.copy()
        k = int(np.argmax(c))

    js = np.arange(max(0, k - n + 1), min(k, n - 1) + 1, dtype=np.int64)
    contrib = np.zeros(n, dtype=np.float64)
    if js.size:
        contrib[js] = a[js][::-1]

    top_count = max(3, n // 100)
    top_idx = np.argsort(contrib)[::-1][:top_count]

    median = float(np.median(contrib))
    bottom_candidates = np.where(contrib <= median + 1e-300)[0]
    if bottom_candidates.size < top_count:
        bottom_candidates = np.argsort(contrib)[:top_count]
    bottom_candidates = bottom_candidates[:top_count]

    out = a.copy()
    total = 0.0
    for j in top_idx:
        amt = float(out[j]) * float(strength)
        out[j] -= amt
        total += amt
    if bottom_candidates.size and total > 0.0:
        add = total / float(bottom_candidates.size)
        for j in bottom_candidates:
            out[j] += add

    out = np.clip(out, 0.0, 1000.0)
    return normalize_seq(out)


def make_directions(base, mode, rng, max_dirs=16):
    base = np.asarray(base, dtype=np.float64).ravel()
    n = base.size
    if n == 0:
        return []
    norm_base = float(np.linalg.norm(base))
    scale = max(1e-12, 0.5 * norm_base)

    raw = []

    g = subgradient_vector(base, mode)
    if float(np.linalg.norm(g)) > 1e-18:
        raw.append(-g)

    for _ in range(6):
        d = rng.standard_normal(n)
        d -= d.mean()
        raw.append(d)

    x = np.arange(n, dtype=np.float64) / float(n)
    for m in [1, 2, 3, 5, 8, 13]:
        phase = 2.0 * np.pi * float(rng.random())
        d = np.cos(2.0 * np.pi * m * x + phase)
        d -= d.mean()
        raw.append(d)

    j = np.arange(n, dtype=np.float64)
    for f in [1.0, 2.0, 4.0]:
        phase = 2.0 * np.pi * float(rng.random())
        d = np.cos(np.pi * f * j * j / (2.0 * float(n)) + phase)
        d -= d.mean()
        raw.append(d)

    for nb in [16, 64, 256]:
        nb = int(min(nb, n))
        per = max(1, n // nb)
        d = np.repeat(rng.random(nb), per)
        if d.size < n:
            d = np.concatenate((d, rng.random(n - d.size)))
        d = d[:n]
        d -= d.mean()
        raw.append(d)

    out = []
    seen = 0
    for d in raw:
        if seen >= max_dirs:
            break
        d = np.asarray(d, dtype=np.float64).ravel()
        if d.size != n or not np.all(np.isfinite(d)):
            continue
        d -= d.mean()
        nn = float(np.linalg.norm(d))
        if nn <= 1e-15:
            continue
        out.append(d / nn * scale)
        seen += 1
    return out


def oracle_line_search(base, d, base_score, mode, max_evals=9):
    base = np.asarray(base, dtype=np.float64).ravel()
    d = np.asarray(d, dtype=np.float64).ravel()
    best_sc = float(base_score)
    best_a = base.copy()
    best_alpha = 0.0
    evals = 0

    tests = [-0.8, -0.4, -0.2, -0.1, 0.1, 0.2, 0.4, 0.8]
    for alpha in tests:
        if evals >= max_evals:
            break
        cand = np.clip(base + float(alpha) * d, 0.0, 1000.0)
        if float(cand.sum()) <= 0.01:
            evals += 1
            continue
        ev = safe_eval(cand)
        evals += 1
        isc = compute_score_mode(cand, mode)
        sc = ev if np.isfinite(ev) else isc
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            best_alpha = float(alpha)

    if best_alpha != 0.0 and evals < max_evals:
        step = 0.1 if abs(best_alpha) > 0.3 else 0.04
        for factor in [-2.0, -1.0, 0.5, 0.0, 0.5, 1.0, 2.0]:
            if evals >= max_evals:
                break
            alpha = best_alpha + factor * step
            cand = np.clip(base + alpha * d, 0.0, 1000.0)
            if float(cand.sum()) <= 0.01:
                evals += 1
                continue
            ev = safe_eval(cand)
            evals += 1
            isc = compute_score_mode(cand, mode)
            sc = ev if np.isfinite(ev) else isc
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                best_alpha = float(alpha)

    return best_sc, best_a, evals


def oracle_polish(base, mode, rng, deadline, max_evals=120):
    base = normalize_seq(base).astype(np.float64, copy=True)
    current = base.copy()
    ev = safe_eval(current)
    evals = 1
    current_sc = ev if np.isfinite(ev) else compute_score_mode(current, mode)

    best_sc = current_sc
    best_a = current.copy()

    dirs = make_directions(current, mode, rng, max_dirs=10)
    for d in dirs:
        if time.monotonic() > deadline or evals >= max_evals:
            break
        remain = max(0, max_evals - evals)
        if remain < 3:
            continue
        step_evals = min(9, remain)
        sc, a, used = oracle_line_search(current, d, current_sc, mode, max_evals=step_evals)
        evals += used
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            current = a.copy()
            if evals + 3 <= max_evals and time.monotonic() < deadline:
                extra = make_directions(current, mode, rng, max_dirs=6)
                for e in extra:
                    dirs.append(e)
            current_sc = best_sc

    return best_sc, best_a, evals


def evaluate_candidate_arrays(cands, mode):
    best_key = np.inf
    best_key_a = None
    best_key_or_finite = False

    best_int = np.inf
    best_int_a = None

    best_or = np.inf
    best_or_a = None

    for a in cands:
        if a is None:
            continue
        a = normalize_seq(a)
        if a.size == 0:
            continue
        sc_int = compute_score_mode(a, mode)
        ev = safe_eval(a)
        sc_or = ev if np.isfinite(ev) else np.inf
        key = sc_or if np.isfinite(sc_or) else sc_int
        finite_or = np.isfinite(sc_or)

        if not np.isfinite(key):
            continue

        if key < best_key - 1e-12 or (abs(key - best_key) <= 1e-12 and finite_or and not best_key_or_finite):
            best_key = key
            best_key_a = a.copy()
            best_key_or_finite = finite_or

        if sc_int < best_int - 1e-12:
            best_int = sc_int
            best_int_a = a.copy()

        if sc_or < best_or - 1e-12:
            best_or = sc_or
            best_or_a = a.copy()

    return best_key, best_key_a, best_int, best_int_a, best_or, best_or_a


def propose_candidate(seed=42, budget_s=2400):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s))
    deadline = t_start + max(10.0, effective_budget - 30.0)

    parent = get_parent_sequence()
    mode = infer_mode()
    print(f"mode={mode}, parent={'yes' if parent is not None else 'no'}")

    uniform1024 = np.ones(1024, dtype=np.float64) * 0.5

    if mode == "center_excl":
        n = 33
        a = np.zeros(n, dtype=np.float64)
        a[n // 2] = 1.0
        ev = safe_eval(a)
        if np.isfinite(ev) and ev < 10.0:
            print(f"center_excl spike score={ev:.8f}")
            return [float(x) for x in normalize_seq(a)]

    simple_cands = []
    simple_cands.append(np.ones(1, dtype=np.float64))
    simple_cands.append(uniform1024)
    if parent is not None:
        simple_cands.append(parent)
        ps = parent.size
        if 64 <= ps <= 16384:
            Nf = pow2(2 * ps - 1)
            simple_cands.append(spectral_flatten(parent, Nf, 40, 0.5))

    if parent is None:
        simple_cands.extend(build_inits(2048, rng, None, mode))

    best_key, best_key_a, best_int, best_int_a, best_or, best_or_a = evaluate_candidate_arrays(simple_cands, mode)
    print(f"simple best key={best_key if np.isfinite(best_key) else 'inf'} oracle={best_or if np.isfinite(best_or) else 'inf'}")

    if mode == "full" and np.isfinite(best_key) and best_key <= 2.000001:
        if (not best_key_or_finite) or best_or >= 1.999:
            print("detected full-convolution regime: returning constant positive sequence")
            out = normalize_seq(uniform1024)
            return [float(x) for x in out]

    search_a = best_key_a if best_key_a is not None else best_int_a
    if search_a is None or search_a.size < 64:
        if parent is not None:
            search_a = parent.copy()
        else:
            inits = build_inits(2048, rng, None, mode)
            search_a = inits[0]

    work = normalize_seq(search_a).astype(np.float64, copy=True)
    sc_work = compute_score_mode(work, mode)
    ev_work = safe_eval(work)
    if sc_work < best_int:
        best_int = sc_work
        best_int_a = work.copy()
    if np.isfinite(ev_work) and ev_work < best_or:
        best_or = ev_work
        best_or_a = work.copy()
        if np.isfinite(ev_work) and ev_work < best_key:
            best_key = ev_work
            best_key_a = work.copy()

    n_work = work.size
    print(f"search n={n_work}, internal={sc_work:.8f}, oracle={ev_work if np.isfinite(ev_work) else 'inf':.8f}")

    beta_list = [30, 80, 200, 500, 1000]
    if n_work <= 1024:
        mits = 4000
        mfun = 80000
    elif n_work <= 4096:
        mits = 2200
        mfun = 44000
    else:
        mits = 1000
        mfun = 20000

    print("--- LSE/L-BFGS polishing ---")
    for beta in beta_list:
        if time.monotonic() > deadline - 60:
            break
        sc, a = run_lbfgs(work, beta, mode, maxiter=mits, maxfun=mfun)
        a = normalize_seq(a)
        ev = safe_eval(a)
        if sc < best_int:
            best_int = sc
            best_int_a = a.copy()
        if np.isfinite(ev) and ev < best_or:
            best_or = ev
            best_or_a = a.copy()
        if np.isfinite(ev) and ev < best_key:
            best_key = ev
            best_key_a = a.copy()
        if sc < best_int or (np.isfinite(ev) and ev < best_or):
            work = a.copy()
        print(f"beta={beta}, work int={sc:.8f}, oracle={ev if np.isfinite(ev) else 'inf'}")

    print("--- subgradient polishing ---")
    sc, a = exact_subgrad_polish(work, mode, total_iters=700)
    a = normalize_seq(a)
    ev = safe_eval(a)
    if sc < best_int:
        best_int = sc
        best_int_a = a.copy()
    if np.isfinite(ev) and ev < best_or:
        best_or = ev
        best_or_a = a.copy()
    if np.isfinite(ev) and ev < best_key:
        best_key = ev
        best_key_a = a.copy()
    work = a.copy()
    print(f"subgrad int={sc:.8f}, oracle={ev if np.isfinite(ev) else 'inf'}")

    if (best_or < best_int - 1e-6 and best_key_a is not None and best_key_a.size >= 64):
        work = best_key_a.astype(np.float64, copy=True)
        work = normalize_seq(work)
        print("re-centered work on oracle-best sequence")

    if n_work <= 10000 and time.monotonic() < deadline - 180:
        print("--- greedy LP direction ---")
        K = 64 if n_work > 5000 else 96
        ts = [0.01, 0.02, 0.05, 0.1, 0.2]
        for _ in range(2):
            if time.monotonic() > deadline - 80:
                break
            sc, a = lp_update(work, mode, K, ts)
            a = normalize_seq(a)
            ev = safe_eval(a)
            if sc < best_int:
                best_int = sc
                best_int_a = a.copy()
            if np.isfinite(ev) and ev < best_or:
                best_or = ev
                best_or_a = a.copy()
            if np.isfinite(ev) and ev < best_key:
                best_key = ev
                best_key_a = a.copy()
            if sc < best_int or (np.isfinite(ev) and ev < best_or):
                work = a.copy()
            print(f"LP int={sc:.8f}, oracle={ev if np.isfinite(ev) else 'inf'}")

    print("--- local perturbations ---")
    max_local = 8
    for trial in range(max_local):
        if time.monotonic() > deadline - 90:
            break
        if trial % 3 == 0:
            strength = 0.02 + 0.05 * float(rng.random())
            pert = targeted_perturb(work, mode, rng, strength)
        elif trial % 3 == 1:
            x = np.arange(work.size, dtype=np.float64) / float(work.size)
            m = 1 + int(rng.integers(1, 9))
            phase = 2.0 * np.pi * float(rng.random())
            amp = 0.01 + 0.05 * float(rng.random())
            pert = work * (1.0 + amp * np.cos(2.0 * np.pi * m * x + phase))
        else:
            amp = 0.01 + 0.05 * float(rng.random())
            pert = work + amp * rng.standard_normal(work.size)

        pert = normalize_seq(np.clip(pert, 0.0, 1000.0))
        mits_local = 800 if work.size <= 4096 else 400
        mfun_local = 20000 if work.size <= 4096 else 10000
        sc, a = run_lbfgs(pert, 800, mode, maxiter=mits_local, maxfun=mfun_local)
        a = normalize_seq(a)
        ev = safe_eval(a)
        if sc < best_int:
            best_int = sc
            best_int_a = a.copy()
        if np.isfinite(ev) and ev < best_or:
            best_or = ev
            best_or_a = a.copy()
        if np.isfinite(ev) and ev < best_key:
            best_key = ev
            best_key_a = a.copy()
        if sc < best_int or (np.isfinite(ev) and ev < best_or):
            work = a.copy()
        print(f"local {trial}: int={sc:.8f}, oracle={ev if np.isfinite(ev) else 'inf'}")

    if (np.isfinite(best_or) and best_or < 1.999) and time.monotonic() < deadline - 90:
        print("--- oracle line-search polish ---")
        base = best_or_a if best_or_a is not None else (best_key_a if best_key_a is not None else work)
        base = normalize_seq(base)
        if base.size > 5000:
            max_evals = 80
        elif base.size > 1024:
            max_evals = 120
        else:
            max_evals = 200
        sc, a, used = oracle_polish(base, mode, rng, deadline - 20, max_evals=max_evals)
        a = normalize_seq(a)
        ev = safe_eval(a)
        if np.isfinite(sc) and sc < best_or:
            best_or = sc
            best_or_a = a.copy()
        if np.isfinite(ev) and ev < best_or:
            best_or = ev
            best_or_a = a.copy()
        if np.isfinite(best_or) and best_or < best_key:
            best_key = best_or
            best_key_a = a.copy()
        work = a.copy()
        print(f"oracle polish: {best_or:.8f}, evals={used}")

    final_cands = []
    for arr in [best_key_a, best_or_a, best_int_a, work, parent, uniform1024]:
        if arr is not None:
            final_cands.append(arr)

    final_score, final_a, _, _, final_or, final_or_a = evaluate_candidate_arrays(final_cands, mode)

    if final_a is None:
        final_a = uniform1024

    out = []
    for v in np.asarray(final_a, dtype=np.float64).ravel():
        fv = float(v)
        if not np.isfinite(fv) or fv < 0.0:
            fv = 0.0
        if fv > 1000.0:
            fv = 1000.0
        out.append(fv)

    s = sum(out)
    if s < 0.01:
        out = [0.5]

    verified = safe_eval(out)
    print(f"FINAL: key={final_score if np.isfinite(final_score) else 'inf'}, verified={verified if np.isfinite(verified) else 'inf'}, len={len(out)}, time={time.monotonic() - t_start:.1f}s")
    return out
