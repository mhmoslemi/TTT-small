def fast_conv(a):

    n = len(a)

    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for use with L-BFGS-B."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.maximum(scaled, -50)  # Prevent underflow
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient of L w.r.t. a:
    # dL/da[i] = sum_k p[k] * 2*a[k-i] (for valid k-i)
    # = 2 * cross_corr(a, p)[i] where cross_corr[i] = sum_j a[j]*p[j+i]
    M_fft = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M_fft)
    P = np.fft.rfft(p, M_fft)
    cross = np.fft.irfft(A * np.conj(P), M_fft)[:n]
    dL_da = 2.0 * cross
    
    # Full gradient of score = 2n * L / S^2
    # dscore/da[i] = 2n * (dL_da[i] / S - 2*L/S^2)
    S2 = S * S
    grad = 2.0 * n * (dL_da / S - 2.0 * L / S2)
    
    return score, grad

def lse_objective_wrapper(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)

def targeted_reduction(a, n_iters=200, fraction=0.01):
    """
    Targeted reduction: at each step, reduce top contributors to the peak
    and add the mass to the bottom contributors (or to non-peak elements).
    """

    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        sc = 2.0 * n * float(c[k_star]) / (S * S)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        length = j_end - j_start + 1
        
        # Partners: a[k_star - j] for j in [j_start, j_end]
        # = a[k_star - j_start], a[k_star - j_start - 1], ..., a[k_star - j_end]
        idx = k_star - np.arange(j_start, j_end + 1)
        partners = a[idx]
        
        # Top contributors (reduce these)
        top_k = max(1, int(length * fraction))
        top_local = np.argsort(partners)[::-1][:top_k]
        top_global = np.arange(j_start, j_end + 1)[top_local]
        
        # Bottom contributors (increase these)
        bottom_local = np.argsort(partners)[:top_k]
        bottom_global = np.arange(j_start, j_end + 1)[bottom_local]
        
        # Reduce top, increase bottom
        total_reduce = 0.0
        for j in top_global:
            reduce_amount = a[j] * 0.05
            a[j] -= reduce_amount
            a[j] = max(a[j], 0.0)
            total_reduce += reduce_amount
        
        if total_reduce > 0:
            per_bottom = total_reduce / len(bottom_global)
            for j in bottom_global:
                a[j] += per_bottom
        
        # Also try adding to non-peak elements
        # Renormalize
        S = a.sum()
        if S < 1e-12:
            break
        # Don't renormalize (scale invariant)
    
    return best_a

    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def cross_corr_ap(a, p, M_fft):
    """
    Compute cross-correlation: result[i] = sum_j a[j] * p[i+j] for i=0..n-1.
    Uses FFT of size M_fft (must be >= len(a) + len(p) - 1).
    """
    n = len(a)
    A = np.fft.rfft(a, M_fft)
    P = np.fft.rfft(p, M_fft)
    result = np.fft.irfft(A * np.conj(P), M_fft)[:n]
    return result


def multiplicative_update(a, n_iters, beta_init, beta_final, eta_scale, t_start, deadline):
    """
    Mirror descent / multiplicative weight update with soft-peak weighting.
    a[i] <- a[i] * exp(-eta * grad[i])
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12).copy()
    S = float(a.sum())
    a = a / S

    M_conv_fft = 1 << (2 * n - 2).bit_length()
    M_cross_fft = 1 << (3 * n - 3).bit_length()

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline - 30:
            break

        t_frac = it / max(1, n_iters - 1)
        beta = beta_init * (beta_final / beta_init) ** t_frac

        # Convolution
        A = np.fft.rfft(a, M_conv_fft)
        conv = np.fft.irfft(A * A, M_conv_fft)[:2 * n - 1]

        S = float(a.sum())
        conv_max = float(conv.max())
        score = 2.0 * n * conv_max / (S * S)

        if score < best_sc:
            best_sc = score
            best_a = a.copy()
            if best_sc < _best_score_global:
                _best_score_global = best_sc
                _best_a_global = best_a.copy()

        # Soft peak weights
        scaled = beta * (conv - conv_max)
        scaled = np.clip(scaled, -40.0, 0.0)
        exp_scaled = np.exp(scaled)
        p = exp_scaled / exp_scaled.sum()

        # Gradient of LSE: dL/da[i] = 2 * sum_j a[j] * p[i+j] = 2 * cross_corr[i]
        cross = cross_corr_ap(a, p, M_cross_fft)
        dL_da = 2.0 * cross

        # Full gradient (drop constant factor 2n for step size purposes)
        L = conv_max + float(np.log(exp_scaled.sum())) / beta
        S2 = S * S
        grad = dL_da / S - 2.0 * L / S2

        # Adaptive step size
        eta = eta_scale / (it + 10.0) ** 0.7

        # Multiplicative update
        exponent = -eta * grad
        exponent = np.clip(exponent, -8.0, 8.0)
        a *= np.exp(exponent)
        a = np.maximum(a, 1e-15)

        S_new = a.sum()
        if S_new < 1e-12:
            break
        a = a / S_new

    return best_a


def lse_score_grad_fn(a, beta):
    """LSE-smoothed score and gradient for L-BFGS-B."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -40.0, 0.0)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    # Gradient
    M_cross_fft = 1 << (3 * n - 3).bit_length()
    cross = cross_corr_ap(a, p, M_cross_fft)
    dL_da = 2.0 * cross
    S2 = S * S
    grad = 2.0 * n * (dL_da / S - 2.0 * L / S2)

    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize (returns f, grad)."""
    return lse_score_grad_fn(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lp_direction(f, M, n_acts=80, n_vars=300):
    """
    Solve a reduced LP to find a direction g that maximizes sum(g)
    subject to (f*g)[k] <= M for the top n_acts lags of f*f.
    g is restricted to the top n_vars coordinates of f.
    """
    n = len(f)
    conv = fast_conv(f)

    # Active constraints: lags where conv is high
    k_idx = np.argsort(conv)[::-1][:n_acts]

    # Active variables: coordinates where f is large
    j_idx = np.argsort(f)[::-1][:n_vars]

    m = len(k_idx)
    p = len(j_idx)

    # Build constraint matrix A[k_i, j_j] = f[k_i - j_j] if valid
    A = np.zeros((m, p))
    for i, k in enumerate(k_idx):
        for j_idx_val in range(p):
            j = j_idx[j_idx_val]
            diff = k - j
            if 0 <= diff < n:
                A[i, j_idx_val] = f[diff]

    c_obj = -np.ones(p)
    b_ub = np.full(m, M)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * p,
                      method='highs', options={'time_limit': 30})
        if res.success:
            g = np.zeros(n)
            g[j_idx] = res.x
            return g
    except Exception:
        pass
    return None
