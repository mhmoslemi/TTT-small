"""
Multi-start equalization optimizer for minimum peak autocorrelation sequences.

Algorithm:
  1. Generate diverse starting points (parent, antisymmetric perturbation, random)
  2. For each start: fast exact coordinate descent with incremental convolution
  3. Apply "convolution flattening" via second-order targeted perturbation
  4. Interleave CD sweeps with flattening steps
  5. Final polish phase
  Key improvements over prior: multi-start diversity, second-order flattening,
  block-aware coordinate updates, adaptive perturbation strength.
"""

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def cd_sweep(a):
    """
    One full sweep of exact coordinate descent with incremental conv updates.
    Returns (score, updated_a).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    for i in range(n):
        a_i = a[i]
        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        # Outside max
        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()

        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 4.0)
        t_upper = min(t_upper, 1000.0)

        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-8}
        )
        t_opt = float(res.x)
        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a


def cd_sweep_random(a, rng, seed_offset=0):
    """One sweep of coordinate descent in random order."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    order = rng.permutation(n)

    for i in order:
        i = int(i)
        a_i = a[i]
        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))

            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()

        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 4.0)
        t_upper = min(t_upper, 1000.0)

            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-8}
        )
        t_opt = float(res.x)
        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    return final_score, a


def flatten_step(a, strength=0.1, top_frac=0.1):
    """
    Second-order flavening: identify peak-contributing pairs and redistribute.
    Uses quadratic model of the peak to determine optimal reduction.
    """
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    peak_val = c[k_star]

    # Contributions to peak: c[k*] = sum_i a[i]*a[k*-i]
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end < j_start:
        return a

    contrib_indices = np.arange(j_start, j_end + 1)
    contrib_vals = a[contrib_indices] * a[k_star - contrib_indices]
    total_contrib = contrib_vals.sum()

    if total_contrib < 1e-12:
        return a

    # Target: reduce peak by 'strength' fraction
    target_reduction = peak_val * strength

    # Find top contributors (fraction of total)
    n_top = max(3, int(len(contrib_vals) * top_frac))
    top_idx_in = np.argsort(contrib_vals)[::-1][:n_top]
    top_indices = contrib_indices[top_idx_in]
    top_contrib = contrib_vals[top_idx_in].sum()

    # Distribute reduction among top contributors proportional to their value
    rest_indices = np.setdiff1d(np.arange(n), top_indices)
    if len(rest_indices) == 0:
        return a

    a_new = a.copy()
    # Reduce top contributors
    for idx in top_indices:
        idx = int(idx)
        # Reduce a[idx] by amount that reduces c[k*] by approximately target_reduction * (a[idx]*a[k*-idx])/total_contrib
        pair_val = a[idx] * a[k_star - idx] if (0 <= k_star - idx < n) else 0
        if pair_val > 1e-15:
            reduction = a[idx] * min(0.5, target_reduction / max(peak_val, 1e-15))
            a_new[idx] = max(0.0, a[idx] - reduction)

    # Redistribute saved mass to non-contributing positions
    saved = a.sum() - a_new.sum()
    if saved > 1e-12:
        # Distribute to positions that are far from the peak lag
        dist = np.abs(np.arange(n) - k_star) + np.abs(np.arange(n) - (k_star - (n - 1) * 0))
        # Simpler: distribute uniformly to non-top positions
        a_new[rest_indices] += saved / len(rest_indices)

    return np.clip(a_new, 0.0, 1000.0)


def multi_peak_flatten(a, strength=0.05, n_peaks=5):
    """Flatten targeting multiple near-peak lags simultaneously."""
    n = len(a)
    c = fast_conv(a)
    peak_val = c.max()

    # Find lags where c[k] > 0.8 * peak
    threshold = 0.8 * peak_val
    hot_lags = np.where(c >= threshold)[0]
    hot_lags = hot_lags[:n_peaks]  # Limit to n_peaks

    a_new = a.copy()
    for k in hot_lags:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end < j_start:
            continue
        contrib_indices = np.arange(j_start, j_end + 1)
        contrib_vals = a_new[contrib_indices] * a_new[k - contrib_indices]
        total_c = c[k]
        if total_c < 1e-12:
            continue
        # Reduce top 20% of contributors
        n_top = max(2, len(contrib_vals) // 5)
        top_pos = np.argsort(contrib_vals)[::-1][:n_top]
        for p in top_pos:
            idx = int(contrib_indices[p])
            reduction = a_new[idx] * min(0.3, strength * total_c / max(peak_val, 1e-15))
            a_new[idx] = max(0.0, a_new[idx] - reduction)

    # Redistribute
    saved = a.sum() - a_new.sum()
    if saved > 1e-12:
        # Add to positions with lowest contribution to any hot lag
        min_contrib = np.ones(n) * 1e10
        for k in hot_lags:
            k = int(k)
            for i in range(n):
                if 0 <= k - i < n:
                    min_contrib[i] = min(min_contrib[i], a_new[i] * a_new[k - i])
        # Distribute to positions with minimum total contribution
        n_add = max(5, n // 20)
        add_indices = np.argsort(min_contrib)[:n_add]
        a_new[add_indices] += saved / n_add

    return np.clip(a_new, 0.0, 1000.0)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient for the objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    # Gradient via FFT
    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):

                 'maxls': 50}

  2. For each start: fast exact coordinate descent with incremental convolution

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)
