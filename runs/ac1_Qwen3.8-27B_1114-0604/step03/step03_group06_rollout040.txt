"""
Search for a non-negative height sequence minimizing
    2*n*max(np.convolve(a,a)) / (sum(a))**2
by combining:
  - population-style screening over several lengths and diverse inits,
  - log-softmax reparameterization (robust positivity, no scale issue),
  - LSE-smoothed self-convolution optimization with L-BFGS-B,
  - active-peak weighting plus damped exact-score line search,
  - mirror-descent fallback steps.

The returned sequence is scale-invariant; it is returned normalized to sum 1,
which satisfies sum > 0 and is well within the evaluator's [0, 1000] caps.
"""

import math
import time

import numpy as np
from scipy.optimize import minimize

LOG_BOUND = 70.0
MAX_SEARCH_LEN = 12288
_GLOBAL_DEADLINE = 1.0e300
_BOUNDS_CACHE = {}


def make_bounds(nvar):
    """Cached L-BFGS-B bounds for the log-softmax coordinate shift."""
    nvar = int(nvar)
    if nvar <= 0:
        return []
    key = nvar
    if key not in _BOUNDS_CACHE:
        _BOUNDS_CACHE[key] = [(-LOG_BOUND, LOG_BOUND)] * nvar
    return _BOUNDS_CACHE[key]


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def conv_fft(a):
    """Full self-convolution of a real nonnegative array via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    L = _next_pow2(need)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:need]
    c[c < 0.0] = 0.0
    return c


def matvec_weight_p(w, p):
    """
    For w of length 2n-1 and p of length n, compute G_j =
    sum_k w_k * p_{k-j} where the index k-j must be valid.
    This is one half of the gradient of the weighted sum of convolution
    coefficients with respect to p.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 3 * n - 2
    L = _next_pow2(need)
    Wh = np.fft.rfft(w, L)
    Vh = np.fft.rfft(p[::-1], L)
    out = np.fft.irfft(Wh * Vh, L)[n - 1:2 * n - 1]
    out[out < 0.0] = 0.0
    return out


def normalize_seq(a):
    """Normalize to nonnegative sum-1 sequence."""
    x = np.asarray(a, dtype=np.float64).ravel()
    if x.size == 0:
        return np.array([1.0], dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1.0e-300:
        return np.ones_like(x, dtype=np.float64) / float(x.size)
    return x / s


def add_floor_normalized(p, floor=1.0e-8):
    """Add a tiny positive floor while preserving sum = 1."""
    p = normalize_seq(p)
    n = int(p.size)
    if n <= 0:
        return p
    if floor <= 0.0:
        return p
    floor = min(floor, 0.9)
    return (1.0 - floor) * p + (floor / float(n)) * np.ones(n, dtype=np.float64)


def exact_score(a):
    """Exact internal score using the same formula as evaluate_sequence."""
    x = normalize_seq(a)
    n = int(x.size)
    if n <= 0:
        return np.inf
    c = conv_fft(x)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not math.isfinite(m) or m <= 0.0:
        return np.inf
    return float(2.0 * n * m)


def log_p_from_r(r):
    """
    Convert shifted log-softmax coordinates to a positive sum-1 vector.
    r has length n-1; the last coordinate is anchored at log-prob 0.
    """
    r = np.asarray(r, dtype=np.float64).ravel()
    n = int(r.size) + 1
    p = np.empty(n, dtype=np.float64)
    if n == 1:
        p[0] = 1.0
        return p
    mr = float(np.max(r))
    mx = mr if mr > 0.0 else 0.0
    er = np.exp(r - mx)
    last = math.exp(-mx)
    s = float(er.sum()) + last
    if s <= 1.0e-300 or not math.isfinite(s):
        p[:] = 1.0 / float(n)
        return p
    p[:-1] = er / s
    p[-1] = last / s
    np.maximum(p, 0.0, out=p)
    p /= p.sum()
    return p


def p_to_r(p):
    """Convert positive sum-1 p to shifted log coordinates (last fixed to 0)."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n = int(p.size)
    if n <= 1:
        return np.zeros(0, dtype=np.float64)
    p = np.maximum(p, 1.0e-300)
    pl = float(p[-1])
    if pl <= 1.0e-300:
        pl = 1.0e-300
    r = np.log(p[:-1] / pl)
    np.clip(r, -LOG_BOUND, LOG_BOUND, out=r)
    return r


def lse_stats(c, beta):
    """
    Normalized LSE smooth-max statistics.
    Uses L = cmax + (logsum - logK)/beta, equivalent to standard LSE shifted
    by a constant, so gradients are unchanged and values stay near the max.
    """
    c = np.asarray(c, dtype=np.float64)
    K = int(c.size)
    if K == 0:
        return 0.0, np.array([], dtype=np.float64)
    cmax = float(np.max(c))
    if beta <= 0.0 or not math.isfinite(beta):
        # Broad average-like smoothing fallback.
        w = np.ones(K, dtype=np.float64) / float(K)
        L = float(np.mean(c))
        return L, w
    scaled = beta * (c - cmax)
    # scaled <= 0 up to numerical noise; clamp to avoid overflow.
    scaled = np.minimum(scaled, 0.0)
    e = np.exp(scaled)
    s = float(e.sum())
    if s <= 1.0e-300 or not math.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros(K, dtype=np.float64)
        w[idx] = 1.0
        return cmax, w
    w = e / s
    logsum = math.log(s)
    L = cmax + (logsum - math.log(float(K))) / beta
    if not math.isfinite(L):
        L = cmax
    return L, w


def objective_log_lse(x, beta):
    """Smooth objective and gradient in log-softmax coordinates."""
    if time.monotonic() > _GLOBAL_DEADLINE:
        return 1.0e100, np.zeros_like(x, dtype=np.float64)

    p = log_p_from_r(x)
    n = int(p.size)
    if n <= 0:
        return 1.0e100, np.zeros_like(x, dtype=np.float64)

    c = conv_fft(p)
    if c.size == 0:
        return 1.0e100, np.zeros_like(x, dtype=np.float64)

    L, w = lse_stats(c, beta)
    if not math.isfinite(L):
        L = float(np.max(c))

    val = 2.0 * float(n) * L

    g = matvec_weight_p(w, p)
    grad_p = 4.0 * float(n) * g
    mean_grad = float(np.dot(p, grad_p))
    grad = p * (grad_p - mean_grad)

    if x.size > 0:
        return float(val), grad[:-1]
    return float(val), np.zeros(0, dtype=np.float64)


def weights_top_active(c, K, top_beta=20.0, uniform_blend=0.10):
    """Weights concentrated on the currently largest convolution lags."""
    c = np.asarray(c, dtype=np.float64).ravel()
    m = int(c.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    K = int(K)
    if K <= 0:
        K = 1
    if K >= m:
        return np.ones(m, dtype=np.float64) / float(m)

    idx = np.argpartition(c, m - K)[-K:]
    vals = c[idx]
    vmax = float(np.max(vals))
    ww = np.exp(top_beta * (vals - vmax))
    sw = float(ww.sum())
    if sw <= 1.0e-300 or not math.isfinite(sw):
        ww = np.ones_like(vals, dtype=np.float64) / float(vals.size)
    else:
        ww = ww / sw

    w = np.zeros(m, dtype=np.float64)
    w[idx] = ww

    if uniform_blend > 0.0:
        w = (1.0 - uniform_blend) * w + uniform_blend * (1.0 / float(m))
    s = float(w.sum())
    if s <= 1.0e-300:
        w = np.ones(m, dtype=np.float64) / float(m)
    else:
        w = w / s
    return w


def objective_log_fixed_w(x, w):
    """Fixed-weighted sum objective and gradient in log-softmax coordinates."""
    if time.monotonic() > _GLOBAL_DEADLINE:
        return 1.0e100, np.zeros_like(x, dtype=np.float64)

    p = log_p_from_r(x)
    n = int(p.size)
    if n <= 0:
        return 1.0e100, np.zeros_like(x, dtype=np.float64)

    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size != 2 * n - 1:
        # Fall back to exact max weight at center if mismatch occurs.
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[n - 1] = 1.0
    s = float(w.sum())
    if s <= 1.0e-300:
        w = np.ones(w.size, dtype=np.float64) / float(w.size)
    else:
        w = w / s

    c = conv_fft(p)
    if c.size != w.size:
        w = np.zeros(c.size, dtype=np.float64)
        w[n - 1] = 1.0

    L = float(np.dot(w, c))
    if not math.isfinite(L):
        L = float(np.max(c))

    val = 2.0 * float(n) * L
    g = matvec_weight_p(w, p)
    grad_p = 4.0 * float(n) * g
    mean_grad = float(np.dot(p, grad_p))
    grad = p * (grad_p - mean_grad)

    if x.size > 0:
        return float(val), grad[:-1]
    return float(val), np.zeros(0, dtype=np.float64)


def mirror_update(p, g, scale=1.0):
    """Exponentiated-gradient step on the probability simplex."""
    p = normalize_seq(p)
    g = np.asarray(g, dtype=np.float64)
    if p.size != g.size or p.size == 0:
        return p.copy()
    gm = g - float(np.mean(g))
    mx = float(np.max(np.abs(gm)))
    if mx <= 1.0e-300:
        return p.copy()
    z = scale * gm / mx
    np.clip(z, -50.0, 50.0, out=z)
    q = p * np.exp(-z)
    s = float(q.sum())
    if s <= 1.0e-300 or not math.isfinite(s):
        return p.copy()
    return q / s


def do_minimize_lse(r, beta, bounds, maxfun, run_end):
    """L-BFGS-B for LSE objective with deadline guard."""
    global _GLOBAL_DEADLINE
    maxfun = int(max(100, maxfun))
    maxiter = max(100, maxfun)
    old_deadline = _GLOBAL_DEADLINE
    _GLOBAL_DEADLINE = min(run_end, old_deadline)
    try:
        res = minimize(
            objective_log_lse,
            r,
            args=(beta,),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "maxiter": maxiter,
                "maxfun": maxfun,
                "ftol": 1.0e-18,
                "gtol": 1.0e-11,
                "maxls": 30,
            },
        )
        return res
    except Exception:
        return None
    finally:
        _GLOBAL_DEADLINE = old_deadline


def do_minimize_fixed_w(r, w, bounds, maxfun, run_end):

def run_lbfgs_lse(p0, max_s, beta_factors=None, maxfun=5000):
    """
    LSE continuation in log-softmax coordinates.
    Returns (best_p, best_score, best_p_current_placeholder).
    """
    p = normalize_seq(p0)
    best_p = p.copy()
    best_sc = exact_score(p)
    cur_p = p.copy()
    cur_sc = best_sc

    n = int(p.size)
    if n <= 1:
        return best_p, best_sc, best_p

    t0 = time.monotonic()
    phase_end = min(t0 + max_s, _GLOBAL_DEADLINE)
    if phase_end <= t0:
        return best_p, best_sc, best_p

    if beta_factors is None:
        beta_factors = (0.1, 1.0, 10.0, 100.0, 1000.0)

    bounds = make_bounds(n - 1)
    r = p_to_r(p)

    for bf in beta_factors:
        now = time.monotonic()
        if now >= phase_end:
            break
        beta = max(10.0, float(bf) * float(n))
        mf = int(min(maxfun, max(300, 5000)))
        res = do_minimize_lse(r, beta, bounds, mf, phase_end)
        now = time.monotonic()
        if now >= phase_end:
            break
        if res is None or getattr(res, "x", None) is None:
            continue
        r_new = np.asarray(res.x, dtype=np.float64)
        if r_new.size != r.size:
            continue
        np.clip(r_new, -LOG_BOUND, LOG_BOUND, out=r_new)
        p_new = log_p_from_r(r_new)
        sc_new = exact_score(p_new)
        if not math.isfinite(sc_new):
            continue

        # Accept monotone improvements or very small worsening during smoothing.
        if sc_new <= cur_sc + 2.0e-3:
            r = r_new
            cur_p = p_new
            cur_sc = sc_new
        if sc_new < best_sc:
            best_sc = sc_new
            best_p = p_new.copy()

    return best_p, best_sc, best_p


def run_active_polish(p0, max_s, max_iters=40, top_ks=(16, 64, 256), lbfgs_maxfun=1200):
    """
    Active-peak polishing:
      - choose weighted top convolution lags,
      - run L-BFGS-B on fixed weighted-sum objective,
      - accept only damped updates that improve exact score,
      - fallback mirror steps if no improvement.
    """
    p = normalize_seq(p0)
    best_p = p.copy()
    best_sc = exact_score(p)
    cur_p = p.copy()
    cur_sc = best_sc

    n = int(p.size)
    if n <= 1:
        return best_p, best_sc, best_p

    t0 = time.monotonic()
    phase_end = min(t0 + max_s, _GLOBAL_DEADLINE)
    if phase_end <= t0:
        return best_p, best_sc, best_p

    bounds = make_bounds(n - 1)
    top_ks = tuple(max(1, int(k)) for k in top_ks)
    if not top_ks:
        top_ks = (16,)

    for it in range(int(max_iters)):
        now = time.monotonic()
        if now >= phase_end:
            break
        rem = phase_end - now
        if rem < 1.0:
            break

        c = conv_fft(cur_sc and cur_p or p)
        K = top_ks[it % len(top_ks)]
        w = weights_top_active(c, K)

        r = p_to_r(cur_p)
        mf = int(min(lbfgs_maxfun, max(120, int(rem * 2000))))
        res = do_minimize_fixed_w(r, w, bounds, mf, phase_end)

        now = time.monotonic()
        if now >= phase_end:
            break
        if res is None or getattr(res, "x", None) is None:
            p_cand = cur_p.copy()
        else:
            r_new = np.asarray(res.x, dtype=np.float64)
            if r_new.size != r.size:
                p_cand = cur_p.copy()
            else:
                np.clip(r_new, -LOG_BOUND, LOG_BOUND, out=r_new)
                p_cand = log_p_from_r(r_new)

        best_cand = cur_p.copy()
        best_cand_sc = cur_sc
        for alpha in (1.0, 0.7, 0.4, 0.2, 0.1):
            pc = (1.0 - alpha) * cur_p + alpha * p_cand
            pc = normalize_seq(pc)
            sc = exact_score(pc)
            if math.isfinite(sc) and sc < best_cand_sc:
                best_cand_sc = sc
                best_cand = pc

        if best_cand_sc < cur_sc:
            cur_p = best_cand
            cur_sc = best_cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_p = cur_p.copy()
        else:
            # Fallback: a smooth mirror step against the selected peaks.
            g = matvec_weight_p(w, cur_p)
            pm = mirror_update(cur_p, g, scale=1.0)
            scm = exact_score(pm)
            if math.isfinite(scm) and scm < cur_sc:
                cur_p = pm
                cur_sc = scm
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_p = cur_p.copy()
            elif cur_sc > best_sc * 1.005:
                cur_p = best_p.copy()
                cur_sc = best_sc

    return best_p, best_sc, best_p


def central_binomial_seq(n):
    """
    One-sided square-root-of-flat spectrum init:
    coefficients of (1 - z)^(-1/2), m_k = C(2k,k) / 4^k.
    """
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    a = np.maximum(a, 1.0e-12)
    return normalize_seq(a)


def make_inits(n, rng, parent):
    """Produce diverse normalized initializations of length n."""
    n = int(n)
    if n <= 0:
        return []
    inits = []

    if parent is not None and parent.size > 0:
        pr = np.asarray(parent, dtype=np.float64).ravel()
        pr = np.where(np.isfinite(pr), pr, 0.0)
        pr = np.maximum(pr, 0.0)
        s = float(pr.sum())
        if s > 0.0:
            pr = pr / s
            if pr.size != n:
                idx = np.linspace(0, pr.size - 1, n).astype(int)
                pr = pr[idx]
            pr = add_floor_normalized(pr, 1.0e-7)
            inits.append(pr.copy())
            inits.append(np.ascontiguousarray(pr[::-1]).copy())
            if n >= 4:
                inits.append(np.ascontiguousarray(np.sort(pr)[::-1]).copy())
                inits.append(np.ascontiguousarray(np.sort(pr)).copy())

    # Uniform.
    inits.append(np.ones(n, dtype=np.float64) / float(n))

    # One-sided sqrt family.
    sqrt_p = central_binomial_seq(n)
    inits.append(sqrt_p.copy())

    # Mix with reversed one-sided sqrt.
    for lam in (0.2, 0.5, 0.8):
        xx = (1.0 - lam) * sqrt_p + lam * sqrt_p[::-1]
        inits.append(add_floor_normalized(xx, 1.0e-9))

    # Power-law decays.
    i = np.arange(n, dtype=np.float64) + 0.5
    for expo in (0.25, 0.4, 0.5, 0.75):
        xx = i ** (-expo)
        inits.append(add_floor_normalized(xx, 1.0e-9))

    # Mild symmetric endpoint family.
    j = np.arange(n, dtype=np.float64)
    left = (j + 0.5) ** (-0.25)
    right = (float(n) - 0.5 - j) ** (-0.25)
    inits.append(add_floor_normalized(left * right, 1.0e-9))

    # Modulated sqrt family.
    x = np.arange(n, dtype=np.float64) / float(max(1, n))
    for freq, amp in ((1.0, 0.15), (2.0, 0.20), (3.0, 0.25)):
        mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * x + 0.7)
        xx = sqrt_p * np.maximum(mod, 0.05)
        inits.append(add_floor_normalized(xx, 1.0e-9))

    # Smooth low-pass random positive field.
    z = rng.standard_normal(n)
    F = np.fft.rfft(z)
    if F.size > 1:
        kk = np.arange(1, F.size, dtype=np.float64)
        F[1:] *= np.exp(-(kk * kk) / 20.0)
    z = np.fft.irfft(F, n)
    z = (z - float(np.mean(z)))
    span = float(np.max(np.abs(z)))
    if span > 0.0:
        z = z / span
    xx = np.exp(0.8 * z)
    inits.append(add_floor_normalized(xx, 1.0e-9))

    # Lognormal random positive init.
    xx = np.exp(rng.standard_normal(n))
    inits.append(add_floor_normalized(xx, 1.0e-9))

    # Block random smoothed.
    nb = max(4, min(32, n // 64 if n >= 256 else 4))
    per = max(1, n // nb)
    vals = 0.35 + 0.65 * rng.random(nb)
    block = np.repeat(vals, per)
    if block.size < n:
        block = np.concatenate([block, vals[: n - block.size]])
    xx = block[:n]
    # Light smoothing.
    if n >= 8:
        kernel = np.ones(3, dtype=np.float64) / 3.0
        xx = np.convolve(xx, kernel, mode="same")
    inits.append(add_floor_normalized(xx, 1.0e-9))

    # Limit list size.
    if len(inits) > 16:
        inits = inits[:16]
    return inits


def quick_optimize(p, max_s):
    """Short screening optimization for one initial sequence."""
    p = normalize_seq(p)
    best_p = p.copy()
    best_sc = exact_score(p)

    t0 = time.monotonic()
    phase_end = min(t0 + max_s, _GLOBAL_DEADLINE)
    if phase_end <= t0:
        return best_p, best_sc, best_p

    rem = phase_end - t0
    if rem > 5.0:
        p1, sc1, _ = run_active_polish(
            p,
            min(rem * 0.70, max_s * 0.70),
            max_iters=12,
            top_ks=(16, 64, 192),
            lbfgs_maxfun=500,
        )
        if sc1 < best_sc:
            best_sc = sc1
            best_p = p1.copy()

    rem = phase_end - time.monotonic()
    if rem > 4.0:
        p2, sc2, _ = run_lbfgs_lse(
            best_p,
            min(rem * 0.85, max_s * 0.25),
            beta_factors=(1.0, 10.0, 100.0),
            maxfun=2500,
        )
        if sc2 < best_sc:
            best_sc = sc2
            best_p = p2.copy()

    rem = phase_end - time.monotonic()
    if rem > 3.0:
        p3, sc3, _ = run_active_polish(
            best_p,
            rem,
            max_iters=6,
            top_ks=(8, 32, 128),
            lbfgs_maxfun=350,
        )
        if sc3 < best_sc:
            best_sc = sc3
            best_p = p3.copy()

    return best_p, best_sc, best_p


def deep_optimize(p, max_s):
    """Longer local/global refinement from a promising sequence."""
    p = normalize_seq(p)
    best_p = p.copy()
    best_sc = exact_score(p)

    t0 = time.monotonic()
    phase_end = min(t0 + max_s, _GLOBAL_DEADLINE)
    if phase_end <= t0:
        return best_p, best_sc, best_p

    # 1) Active polishing to level main peaks.
    rem = phase_end - t0
    if rem > 10.0:
        p1, sc1, _ = run_active_polish(
            best_p,
            min(rem * 0.35, 800.0),
            max_iters=70,
            top_ks=(8, 32, 128, 400),
            lbfgs_maxfun=1600,
        )
        if sc1 < best_sc:
            best_sc = sc1
            best_p = p1.copy()

    # 2) Smoothed continuation.
    rem = phase_end - time.monotonic()
    if rem > 10.0:
        p2, sc2, _ = run_lbfgs_lse(
            best_p,
            min(rem * 0.30, 650.0),
            beta_factors=(0.1, 1.0, 10.0, 100.0, 1000.0),
            maxfun=6000,
        )
        if sc2 < best_sc:
            best_sc = sc2
            best_p = p2.copy()

    # 3) More active polishing with finer peak control.
    rem = phase_end - time.monotonic()
    if rem > 10.0:
        p3, sc3, _ = run_active_polish(
            best_p,
            min(rem * 0.35, 900.0),
            max_iters=90,
            top_ks=(5, 16, 64, 256),
            lbfgs_maxfun=2000,
        )
        if sc3 < best_sc:
            best_sc = sc3
            best_p = p3.copy()

    # 4) Final LSE high-beta push.
    rem = phase_end - time.monotonic()
    if rem > 8.0:
        p4, sc4, _ = run_lbfgs_lse(
            best_p,
            min(rem * 0.60, 450.0),
            beta_factors=(10.0, 100.0, 1000.0, 10000.0),
            maxfun=5000,
        )
        if sc4 < best_sc:
            best_sc = sc4
            best_p = p4.copy()

    return best_p, best_sc, best_p


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Searches for a non-negative sequence minimizing the
    given evaluation function within budget_s seconds.
    """
    global _GLOBAL_DEADLINE

    try:
        t_start = time.monotonic()
    except Exception:
        t_start = time.time()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not math.isfinite(budget) or budget < 30.0:
        budget = 2400.0

    reserve = 45.0
    deadline = t_start + budget - reserve
    _GLOBAL_DEADLINE = deadline

    try:
        seed_i = int(seed) if seed is not None else 42
    except Exception:
        seed_i = 42
    rng = np.random.default_rng(seed_i)

    # Optional parent solution.
    parent = None
    try:
        pg = globals().get("height_sequence_1", None)
        if pg is not None:
            arr = np.asarray(pg, dtype=np.float64).ravel()
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.maximum(arr, 0.0)
            s = float(arr.sum())
            if arr.size >= 32 and s > 0.0:
                parent = arr / s
    except Exception:
        parent = None

    # Length selection: parent length plus nearby powers of two / practical sizes.
    base_lengths = [2048, 4096, 6144, 8192, 10240, 12288]
    if parent is not None:
        psize = int(parent.size)
        target_len = psize if psize <= MAX_SEARCH_LEN else MAX_SEARCH_LEN
        if target_len < 2048:
            target_len = 2048
        lengths_set = set()
        lengths_set.add(target_len)
        for b in base_lengths:
            lengths_set.add(b)
        dist = []
        anchor = psize if psize <= MAX_SEARCH_LEN else target_len
        for nn in lengths_set:
            nn = int(nn)
            if 2048 <= nn <= MAX_SEARCH_LEN:
                dist.append((abs(nn - anchor), nn))
        dist.sort()
        lengths = [nn for _, nn in dist][:4]
        if not lengths:
            lengths = [4096, 8192]
    else:
        lengths = [4096, 8192, 12288]

    best_p = None
    best_sc = float("inf")

    # -------------------------
    # Phase 1: population screen
    # -------------------------
    screen_budget = min(600.0, max(120.0, (budget - reserve) * 0.22))
    screen_end = t_start + screen_budget
    candidates = []
    uid = 0

    if screen_end > t_start + 30.0:
        per_len = screen_budget / float(max(1, len(lengths)))
        max_inits = 3
        for n in lengths:
            if time.monotonic() > screen_end - 6.0:
                break
            n = int(n)
            inits = make_inits(n, rng, parent)
            inits = inits[:max_inits]
            if not inits:
                inits = [np.ones(n, dtype=np.float64) / float(n)]
            qtime_base = per_len / float(max(1, len(inits)))
            for idx, a0 in enumerate(inits):
                if time.monotonic() > screen_end - 4.0:
                    break
                qtime = max(8.0, min(180.0, qtime_base))
                if time.monotonic() + qtime > screen_end:
                    qtime = max(4.0, screen_end - time.monotonic() - 2.0)
                if qtime < 3.0:
                    continue
                p_best, sc_best, _ = quick_optimize(a0, qtime)
                if math.isfinite(sc_best):
                    label = f"len_{n}_init_{idx}"
                    candidates.append((sc_best, uid, p_best, int(n), label))
                    if sc_best < best_sc:
                        best_sc = sc_best
                        best_p = p_best.copy()
                    uid += 1

    if not candidates:
        n = int(parent.size) if parent is not None else 4096
        n = int(min(max(n, 2048), MAX_SEARCH_LEN))
        inits = make_inits(n, rng, parent)
        if not inits:
            inits = [np.ones(n, dtype=np.float64) / float(n)]
        p0 = normalize_seq(inits[0])
        s0 = exact_score(p0)
        candidates.append((s0, 0, p0, n, "default"))
        if s0 < best_sc:
            best_sc = s0
            best_p = p0.copy()

    # Select a small diverse set of candidates for deep optimization.
    candidates.sort()
    selected = []
    len_count = {}
    for sc, uid_val, p_val, n_val, label in candidates:
        if len(selected) >= 3:
            break
        if not math.isfinite(sc):
            continue
        c = len_count.get(n_val, 0)
        if c >= 2 and len(selected) >= 1:
            continue
        selected.append((sc, p_val, int(n_val), label))
        len_count[n_val] = c + 1

    # If not enough distinct candidates, fill from sorted list.
    if len(selected) < 3:
        seen_uids = set()
        for sc, uid_val, p_val, n_val, label in candidates:
            if len(selected) >= 3:
                break
            if not math.isfinite(sc):
                continue
            if uid_val in seen_uids:
                continue
            seen_uids.add(uid_val)
            selected.append((sc, p_val, int(n_val), label))

    # -------------------------
    # Phase 2: deep refine best candidates
    # -------------------------
    deep_end = deadline - 20.0
    for i, (sc_sel, p_sel, n_sel, label) in enumerate(selected):
        remaining = deep_end - time.monotonic()
        if remaining < 70.0:
            break
        share = remaining / float(max(1, len(selected) - i))
        max_t = min(share - 2.0, 1800.0)
        if max_t < 30.0:
            max_t = remaining - 8.0
        if max_t <= 5.0:
            continue
        p_deep, sc_deep, _ = deep_optimize(p_sel, max_t)
        if math.isfinite(sc_deep) and sc_deep < best_sc:
            best_sc = sc_deep
            best_p = p_deep.copy()
            print(f"[deep {label}] NEW BEST: {best_sc:.10f}  n={int(n_sel)}  t={time.monotonic() - t_start:.1f}s")

    # -------------------------
    # Phase 3: final absolute polish on global best
    # -------------------------
    final_end = deadline - 12.0
    if best_p is not None and time.monotonic() < final_end:
        rem = final_end - time.monotonic()
        if rem > 80.0:
            p_f, sc_f, _ = run_active_polish(
                best_p,
                rem * 0.55,
                max_iters=110,
                top_ks=(5, 16, 64, 256),
                lbfgs_maxfun=2500,
            )
            if sc_f < best_sc:
                best_sc = sc_f
                best_p = p_f.copy()
                print(f"[final active] NEW BEST: {best_sc:.10f}  n={int(best_p.size)}  t={time.monotonic() - t_start:.1f}s")

        rem = final_end - time.monotonic()
        if rem > 50.0:
            p_f, sc_f, _ = run_lbfgs_lse(
                best_p,
                rem * 0.55,
                beta_factors=(10.0, 100.0, 1000.0, 10000.0, 100000.0),
                maxfun=7000,
            )
            if sc_f < best_sc:
                best_sc = sc_f
                best_p = p_f.copy()
                print(f"[final LSE] NEW BEST: {best_sc:.10f}  n={int(best_p.size)}  t={time.monotonic() - t_start:.1f}s")

        rem = final_end - time.monotonic()
        if rem > 30.0:
            p_f, sc_f, _ = run_active_polish(
                best_p,
                rem,
                max_iters=60,
                top_ks=(4, 10, 40, 160),
                lbfgs_maxfun=2000,
            )
            if sc_f < best_sc:
                best_sc = sc_f
                best_p = p_f.copy()
                print(f"[final active2] NEW BEST: {best_sc:.10f}  n={int(best_p.size)}  t={time.monotonic() - t_start:.1f}s")

    # Final safe output.
    if best_p is None:
        out = [0.5] * 1024
    else:
        out_arr = np.asarray(best_p, dtype=np.float64)
        out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
        out_arr = np.maximum(out_arr, 0.0)
        out_arr = np.minimum(out_arr, 1000.0)
        s = float(out_arr.sum())
        if s < 0.01:
            out = [0.5] * 1024
        else:
            # Keep sum at 1; scale invariance makes this safe.
            out_arr = out_arr / s
            out = [float(v) for v in out_arr]

    verified = float("inf")
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc if math.isfinite(best_sc) else float("inf")

    elapsed = time.monotonic() - t_start
    print(f"FINAL internal={best_sc:.12f} verified={verified:.12f} len={len(out)} time={elapsed:.1f}s budget={budget:.1f}s")
    return out
