Actually, let me reconsider the LSE approach. The LSE (Log-Sum-Exp) smoothing is:
`LSE(c, beta) = (1/beta) * log(sum(exp(beta * c)))`

This approximates `max(c)` when beta is large. The gradient is:
`grad LSE = softmax(beta * c)` (weighted by the exponential)

So the gradient of the LSE-smoothed score is:
`grad score = (2n/S^2) * (2 * conv(p, a) - 2 * LSE/S)`

where `p = softmax(beta * c / beta) = softmax(c - max(c))`... wait, let me be more careful.

`L = max(c) + (1/beta) * log(sum(exp(beta*(c - max(c)))))`

`p_k = exp(beta*(c_k - max(c))) / sum_j exp(beta*(c_j - max(c)))`

`grad_a L = 2 * (p * a)_reversed` (the convolution of p with a, appropriately indexed)

`grad_a score = (2n/S^2) * (grad_a L - 2*L/S * sum_a_grad)`

where `sum_a_grad = ones(n)`.

So `grad_a score = (2n/S^2) * (2*(p*a)_conv - 2*L/S * 1)`

This is what the previous code computes. The issue is that for large beta, the gradient is very peaked (concentrated on the argmax), which makes L-BFGS slow.

**Alternative: Use a fixed beta that's not too large**, so the LSE is a good smooth approximation but the gradient is well-distributed.

```python
            # unless it reduces the peak. For now, skip zero coordinates.
            continue
```

```python
def peak_surgery(a, rng, strength=0.05):
    """Targeted perturbation that directly attacks the peak."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    
    # Find contributors to the peak
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib_idx = np.arange(j_start, j_end + 1)
    contrib_vals = a[contrib_idx] * a[k_star - contrib_idx]
    
    # Top contributors
    n_top = max(3, n // 30)
    top_local = np.argsort(contrib_vals)[::-1][:n_top]
    top_global = contrib_idx[top_local]
    
    # Bottom contributors (don't contribute to peak)
    bottom_local = np.argsort(contrib_vals)[:n - n_top]
    bottom_global = contrib_idx[bottom_local]
    
    # Also find coordinates that don't contribute to the peak at all
    all_idx = np.arange(n)
    peak_contrib_mask = np.zeros(n, dtype=bool)
    peak_contrib_mask[j_start:j_end+1] = True
    non_peak_idx = all_idx[~peak_contrib_mask]
    
    a_pert = a.copy()
    total_moved = 0.0
    
    # Move weight from top contributors to non-peak positions
    for j in top_global:
        move_amt = a_pert[j] * strength * rng.random()
        a_pert[j] -= move_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_moved += move_amt
    
    # Distribute to random non-peak positions
    if total_moved > 0 and len(non_peak_idx) > 0:
        targets = non_peak_idx[rng.integers(0, len(non_peak_idx), size=min(10, len(non_peak_idx)))]
        per = total_moved / len(targets)
        a_pert[targets] += per
    
    return np.clip(a_pert, 0.0, 1000.0)
```

```python
"""
Optimized coordinate descent with peak surgery and LSE refinement
for minimizing 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key improvements over previous approach:
- Much faster CD via vectorized grid evaluation (no scipy minimize_scalar)
- Peak surgery perturbation that directly targets the convolution peak
- Tighter search bounds using analytical estimates
- Length variation to escape local minima
- LSE refinement with multiple betas

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fast_cd_sweep(a, order):
    """
    Fast coordinate descent sweep with vectorized 1D solver.
    Uses coarse+fine grid instead of scipy minimize_scalar.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    for idx in range(len(order)):
        i = int(order[idx])

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        # Outside max (lags not affected by a[i])
        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        base = conv[k_lo:k_hi + 1].copy()

        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i if False else 2.0 * a[i] * a_vals
        old_contrib[is_2i] = a[i] * a[i]
        base -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        if is_2i.any():
            beta_coef[np.where(is_2i)[0][0]] = 1.0

        S_rest = S - a[i]
        a_i = a[i]

        # Determine search range
        # Upper bound: where inside catches up to outside, or heuristic
        if outside_max > 0 and np.any(alpha > 0):
            # Tighter bound: smallest t where any linear term reaches outside_max
            valid = alpha > 1e-15
            if np.any(valid):
                t_catch = (outside_max - base[valid]) / alpha[valid]
                t_catch = t_catch[t_catch > 0]
                if len(t_catch) > 0:
                    t_upper = min(1.2 * float(np.min(t_catch)) + 0.01, 1000.0)
                else:
                    t_upper = max(2.0 * a_i + 0.01, S_rest / (4.0 * n))
                    t_upper = min(t_upper, 1000.0)
            else:
                t_upper = max(2.0 * a_i + 0.01, S_rest / (4.0 * n))
                t_upper = min(t_upper, 1000.0)
        else:
            t_upper = max(2.0 * a_i + 0.01, S_rest / (4.0 * n))
            t_upper = min(t_upper, 1000.0)

        if t_upper < 1e-10:
            t_upper = 1e-3

        # Coarse grid (vectorized)
        n_coarse = 20
        t_grid = np.linspace(0.0, t_upper, n_coarse)
        # Shape: (n_coarse, inside_len)
        conv_grid = base[np.newaxis, :] + t_grid[:, np.newaxis] * alpha[np.newaxis, :] + (t_grid[np.newaxis, :] ** 2) * beta_coef[np.newaxis, :]
        max_grid = np.maximum(np.max(conv_grid, axis=1), np.full(n_coarse, outside_max))
        S_new = S_rest + t_grid
        safe_S = np.maximum(S_new, 1e-12)
        scores = 2.0 * n * max_grid / (safe_S * safe_S)

        min_idx = int(np.argmin(scores))

        # Fine grid around the minimum
        if min_idx == 0:
            lo, hi = 0.0, t_grid[1]
        elif min_idx == n_coarse - 1:
            lo, hi = t_grid[n_coarse - 2], t_upper
        else:
            lo, hi = t_grid[min_idx - 1], t_grid[min_idx + 1]

        n_fine = 12
        t_fine = np.linspace(lo, hi, n_fine)
        conv_fine = base[np.newaxis, :] + t_fine[:, np.newaxis] * alpha[np.newaxis, :] + (t_fine[np.newaxis, :] ** 2) * beta_coef[np.newaxis, :]
        max_fine = np.maximum(np.max(conv_fine, axis=1), np.full(n_fine, outside_max))
        S_fine = S_rest + t_fine
        safe_Sf = np.maximum(S_fine, 1e-12)
        scores_fine = 2.0 * n * max_fine / (safe_Sf * safe_Sf)

        t_opt = float(t_fine[int(np.argmin(scores_fine))])

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base + t_opt * alpha + (t_opt * t_opt) * beta_coef

    return compute_score(a), a


def hot_order_cd_sweep(a, order=None):
    """CD sweep that prioritizes coordinates contributing to the peak."""
    n = len(a)
    a_work = a.astype(np.float64).copy()
    conv = fast_conv(a_work)
    k_star = int(np.argmax(conv))

    # Compute contribution to peak
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a_work[j_start:j_end + 1] * a_work[k_star - j_start:k_star - j_start][::-1]

    # Hottest first
    order = np.argsort(contrib)[::-1]
    return fast_cd_sweep(a_work, order)


def peak_surgery(a, rng, strength=0.04):
    """Directly attack the convolution peak by redistributing mass."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))

    # Find all lags near the peak (top few)
    top_lags = np.argsort(c)[::-1][:5]

    a_pert = a.copy()
    total_moved = 0.0
    moved_mask = np.zeros(n, dtype=bool)

    for k_star2 in top_lags:
        k_star2 = int(k_star2)
        j_start = max(0, k_star2 - n + 1)
        j_end = min(k_star2, n - 1)
        if j_end < j_start:
            continue
        idx_range = np.arange(j_start, j_end + 1)
        contrib = a_pert[idx_range] * a_pert[k_star2 - idx_range]
        n_top = max(2, len(idx_range) // 20)
        top_local = np.argsort(contrib)[::-1][:n_top]
        for tl in top_local:
            j = int(idx_range[tl])
            if moved_mask[j]:
                continue
            move_amt = a_pert[j] * strength * (0.5 + 0.5 * rng.random())
            a_pert[j] = max(a_pert[j] - move_amt, 0.0)
            total_moved += move_amt
            moved_mask[j] = True

    # Distribute to random positions
    if total_moved > 0:
        n_targets = min(max(5, n // 50), n)
        targets = rng.integers(0, n, size=n_targets)
        per = total_moved / len(targets)
        a_pert[targets] += per

    return np.clip(a_pert, 0.0, 1000.0)


def gaussian_perturb(a, rng, scale=0.02):
    """Gaussian multiplicative perturbation."""
    n = len(a)
    a_pert = a * (1.0 + scale * rng.standard_normal(n))
    return np.clip(a_pert, 0.0, 1000.0)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -100, 100))
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_wrapper(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def resample_sequence(a, new_len):
    """Resample a sequence to a new length using linear interpolation."""
    n_old = len(a)
    if new_len == n_old:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, new_len)
    a_new = np.interp(x_new, x_old, a)
    return a_new

        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}", flush=True)
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s", flush=True)
    print(f"{'=' * 60}", flush=True)
    return out
```
