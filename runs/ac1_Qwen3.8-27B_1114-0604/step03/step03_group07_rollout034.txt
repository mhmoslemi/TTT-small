"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses anti-palindromic structure, LSE-smoothed optimization, and multi-lag subgradient methods.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta, k_top=1):
    """LSE-smoothed score for top-k lags."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    if k_top == 1:
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        exp_s = np.exp(scaled)
        L = c_max + np.log(exp_s.sum()) / beta
        p = exp_s / exp_s.sum()
    else:
        # Penalize top-k lags
        idx = np.argsort(c)[::-1][:k_top]
        c_max = float(c[idx[0]])
        # Weight each of the top lags
        weights = np.ones(2*n-1) * 1e-10
        for i, ix in enumerate(idx):
            weights[ix] = 1.0 / (i + 1)
        scaled = beta * weights * (c - c_max)
        exp_s = np.exp(scaled)
        L = c_max + np.log(exp_s.sum()) / beta
        p = exp_s / exp_s.sum()
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, k_top=1, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj(a):
        return lse_score_grad(a, beta, k_top)
    
    try:
        res = minimize(
            obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0, False
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out, res.success

def multi_lag_subgrad(a, max_iters=5000, k_lags=5):
    """Subgradient descent targeting top-k lags simultaneously."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top-k lags
        top_k = np.argsort(c)[::-1][:k_lags]
        
        # Compute subgradient contribution from each top lag
        grad = np.zeros(n)
        for k_star in top_k:
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                grad[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad / S2 - 2.0 * M / S3)
        
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        
        # Polyak step with adaptive lower bound
        sc_lb = 2.0 * n / (2.0 * n - 1.0)
        step = (sc - sc_lb) / gn2
        step = min(step, 0.5 / (1.0 + 0.01 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a

def coordinate_descent(a, max_rounds=5, top_k=10):

    a = a.astype(np.float64).copy()

    best_a = a.copy()
    
    for rd in range(max_rounds):
        c = fast_conv(a)
        S = float(a.sum())
        top_lags = np.argsort(c)[::-1][:top_k]
        
        for i in range(n):

            for frac in [0.05, 0.1, 0.15, 0.2]:

                if amt > 0:
                    a_try[i] -= amt
                    a_try[n - 1 - i] += amt

def spectral_init(n, rng, n_modes=20):
    """Generate sequences via spectral parameterization."""
    x = np.arange(n, dtype=np.float64) / n
    inits = []
    
    for _ in range(5):
        # Random spectral coefficients
        coeffs_c = rng.standard_normal(n_modes) * 0.3
        coeffs_s = rng.standard_normal(n_modes) * 0.3
        # Add a deterministic anti-palindromic base
        base = 0.3 + 0.7 * (x - 0.5) * rng.uniform(0.5, 2.0)
        
        f = base.copy()
        for k in range(1, n_modes + 1):
            f += coeffs_c[k-1] * np.cos(2 * np.pi * k * x)
            f += coeffs_s[k-1] * np.sin(2 * np.pi * k * x)
        
        a = np.exp(np.clip(f, -5, 5))
        inits.append(np.clip(a, 0.01, 1000.0))
    
    return inits

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines anti-palindromic initializations, LSE-continuation L-BFGS, log-space optimization,
multi-lag subgradient descent, and adaptive coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta, k_top=1):
    """LSE-smoothed objective and gradient for top-k lag penalty."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    if k_top <= 1:
        c_max = float(c.max())
        shifted = beta * (c - c_max)
        exp_s = np.exp(shifted)
        L = c_max + np.log(exp_s.sum()) / beta
        p = exp_s / exp_s.sum()
    else:
        idx = np.argsort(c)[::-1][:k_top]
        c_max = float(c[idx[0]])
        weights = np.zeros(2 * n - 1)
        for rank, ix in enumerate(idx):
            weights[ix] = 1.0 / (rank + 1)
        shifted = beta * weights * (c - c_max)
        exp_s = np.exp(shifted)
        L = c_max + np.log(exp_s.sum()) / beta
        p = exp_s / exp_s.sum()

    score = 2.0 * n * L / (S * S)

    # Gradient: dL/da[i] = 2 * sum_m p[i+m] * a[m] (cross-correlation)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs_bounded(a0, beta, k_top=1, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective with non-negativity bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def objective(a):
        return lse_obj_grad(a, beta, k_top)

    try:
        res = minimize(
            objective, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0, False

    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out, res.success


def run_lbfgs_logspace(z0, beta, k_top=1, maxiter=5000, maxfun=100000):
    """Run L-BFGS in log-space: a = exp(z), optimizing over z (unconstrained)."""
    n = len(z0)
    z0 = z0.astype(np.float64).copy()
    a0 = np.exp(np.clip(z0, -20, 20))

    def objective(z):
        a = np.exp(np.clip(z, -20, 20))
        sc, grad_a = lse_obj_grad(a, beta, k_top)
        # Chain rule: d/dz = d/da * da/dz = d/da * a
        grad_z = grad_a * a
        return sc, grad_z

    try:
        res = minimize(
            objective, z0, method='L-BFGS-B', jac=True,
            options={'ftol': 1e-16, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0, False

    z_out = res.x
    a_out = np.exp(np.clip(z_out, -20, 20))
    a_out = np.clip(a_out, 0.0, 1000.0)
    return compute_score(a_out), a_out, res.success


def build_inits(n, rng, parent):
    """Generate diverse initializations emphasizing anti-palindromic structure."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else x

    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())

    # Uniform baseline
    inits.append(np.ones(n) * 0.5)

    # Anti-palindromic tilts (various strengths)
    for tilt in [0.3, 0.5, 0.7, 0.9, 1.0, 1.3, 1.5]:
        inits.append(np.clip(1.0 + tilt * (2 * xn - 1), 0.01, 1000.0))
        inits.append(np.clip(1.0 - tilt * (2 * xn - 1), 0.01, 1000.0))

    # Exponential tilts
    for rate in [1.0, 2.0, 3.0, 5.0]:
        inits.append(np.clip(np.exp(rate * (xn - 0.5)), 0.01, 1000.0))
        inits.append(np.clip(np.exp(-rate * (xn - 0.5)), 0.01, 1000.0))

    # Power tilts
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.clip(xn ** p + 0.05, 0.01, 1000.0))
        inits.append(np.clip((1 - xn) ** p + 0.05, 0.01, 1000.0))

    # Tilt + sinusoidal texture
    for tilt in [0.5, 0.8, 1.0]:
        for freq in [3, 7, 13, 23]:
            inits.append(np.clip(
                1.0 + tilt * (2 * xn - 1) + 0.15 * np.sin(2 * np.pi * freq * xn),
                0.01, 1000.0
            ))
            inits.append(np.clip(
                1.0 + tilt * (2 * xn - 1) + 0.15 * np.cos(2 * np.pi * freq * xn + 0.5),
                0.01, 1000.0
            ))

    # Sine-based (odd harmonics give anti-palindromic structure)
    for freq in [1, 3, 5, 7]:
        inits.append(np.clip(0.3 + 0.4 * np.sin(np.pi * freq * xn) + 0.3 * xn, 0.01, 1000.0))

    # Smoothed random
    a_rand = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0)
    ksize = max(1, n // 200)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(a_rand, kernel, mode='same'))

    # Anti-palindromic + random texture
    for _ in range(3):
        tilt = rng.uniform(0.5, 1.2)
        noise = rng.standard_normal(n)
        ksize2 = max(1, n // 100)
        kernel2 = np.ones(ksize2) / ksize2
        noise = np.convolve(noise, kernel2, mode='same')
        inits.append(np.clip(1.0 + tilt * (2 * xn - 1) + 0.1 * noise, 0.01, 1000.0))

    # Two-block
    half = n // 2
    inits.append(np.concatenate([np.ones(half) * 0.9, np.ones(n - half) * 0.2]))
    inits.append(np.concatenate([np.ones(half) * 0.2, np.ones(n - half) * 0.9]))

    return inits


def multi_lag_subgrad(a, max_iters=5000, k_lags=5):
    """Subgradient descent targeting top-k lags simultaneously with Polyak step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-k lags
        top_k = np.argsort(c)[::-1][:k_lags]

        # Accumulate subgradient from each top lag
        grad = np.zeros(n)
        for k_star in top_k:
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                grad[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad / S2 - 2.0 * M / S3)

        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break

        step = (sc - sc_lb) / gn2
        # Adaptive damping
        damp = 1.0 / (1.0 + 0.005 * it)
        step = min(step * damp, 0.3)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def targeted_coord_descent(a, max_rounds=3, top_k=8, step_fracs=None):
    """Coordinate descent: transfer mass from peak-contributing elements to anti-palindromic partners."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_sc = compute_score(a)
    best_a = a.copy()
    if step_fracs is None:
        step_fracs = [0.03, 0.06, 0.10, 0.15]

    for rd in range(max_rounds):
        c = fast_conv(a)
        S = float(a.sum())
        top_lags = np.argsort(c)[::-1][:top_k]

        # Identify elements that contribute most to top lags
        contrib = np.zeros(n)
        for k_star in top_lags:
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                for j in range(j_start, j_end + 1):
                    contrib[j] += a[j] * a[k_star - j]

        # Try transferring from high-contributing to their anti-palindromic partners
        sorted_idx = np.argsort(contrib)[::-1]
        n_try = min(n // 5, 200)

        for i in range(n_try):
            j = sorted_idx[i]
            partner = n - 1 - j
            if partner == j:
                continue
            for frac in step_fracs:
                amt = a[j] * frac
                if amt < 1e-10:
                    break
                a_try = a.copy()
                a_try[j] -= amt
                a_try[partner] += amt
                if a_try[j] < 0:
                    a_try[j] = 0
                a_try = np.clip(a_try, 0.0, 1000.0)
                sc_try = compute_score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

    return best_sc, best_a


def adaptive_perturb(a, rng, n_best):
    """Generate a perturbed candidate based on autocorrelation structure."""
    c = fast_conv(a)
    n = len(a)
    strat = rng.integers(0, 5)

    if strat == 0:
        # Gaussian multiplicative perturbation
        scale = 0.02 + 0.03 * rng.random()
        a_pert = a * (1.0 + scale * rng.standard_normal(n))
    elif strat == 1:
        # Anti-palindromic drift: shift mass from earlier to later (or vice versa)
        a_pert = a.copy()
        direction = rng.choice([-1, 1])
        shift_frac = 0.02 + 0.03 * rng.random()
        for j in range(0, n // 2, max(1, n // 100)):
            amt = a_pert[j] * shift_frac
            a_pert[j] -= amt
            a_pert[n - 1 - j] += direction * amt
    elif strat == 2:
        # Frequency-domain smoothing
        M_fft = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(a, M_fft)
        freqs = np.arange(len(A))
        att = 1.0 - 0.08 * (freqs / len(A)) ** 2
        A *= att
        a_pert = np.real(np.fft.irfft(A, M_fft))[:n]
    elif strat == 3:
        # Random block noise
        a_pert = a.copy()
        n_blocks = rng.integers(5, 20)
        for _ in range(n_blocks):
            start = rng.integers(0, n)
            length = rng.integers(1, max(2, n // 100))
            length = min(length, n - start)
            a_pert[start:start + length] *= (1.0 + 0.05 * rng.standard_normal())
    else:
        # Peak-targeting: reduce elements at the peak lag
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_pert = a.copy()
        if j_end >= j_start:
            # Identify top contributors at peak lag
            contrib = np.zeros(n)
            for j in range(j_start, j_end + 1):
                contrib[j] = a[j] * a[k_star - j]
            top_count = max(5, n // 200)
            top_idx = np.argsort(contrib)[::-1][:top_count]
            for j in top_idx:
                a_pert[j] *= (1.0 - 0.05 * rng.random())
            # Compensate at low contributors
            low_idx = np.argsort(contrib)[:top_count]
            for j in low_idx:
                a_pert[j] *= (1.0 + 0.03 * rng.random())

    a_pert = np.clip(a_pert, 0.0, 1000.0)
    if a_pert.sum() < 1e-6:
        a_pert = a * 0.9
    return a_pert

        report(sc, a_opt, f"P4 subgrad k={k_lags}")

    # Coordinate descent
    if time.monotonic() < phase4_end:
        sc, a_opt = targeted_coord_descent(state['best_a'], max_rounds=3, top_k=10)
        report(sc, a_opt, "P4 coord")

    # Another round of subgradient after coord descent
    if time.monotonic() < phase4_end:
        remaining = phase4_end - time.monotonic()
        n_iters = max(300, int(remaining * 30))
        sc, a_opt = multi_lag_subgrad(state['best_a'], max_iters=n_iters, k_lags=5)
        report(sc, a_opt, "P4 subgrad2")

    # ============================================================
    # Phase 5: Adaptive local search (until deadline)
    # ============================================================
    print(f"\n=== Phase 5: Adaptive Local Search ===")

    n_best = len(state['best_a'])
    trial = 0
    while time.monotonic() < deadline and trial < 30:
        trial += 1
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_pert = adaptive_perturb(state['best_a'], rng, n_best)

        # Choose optimization method based on time
        per_iter = 0.3 + 0.0001 * n_best
        if remaining > 60:
            maxit = max(500, min(5000, int(remaining / per_iter * 0.3)))
            sc, a_opt, _ = run_lbfgs_bounded(a_pert, 5000, maxiter=maxit, maxfun=maxit * 15)
        else:
            # Quick subgradient polish
            n_iters = max(200, int(remaining * 25))
            sc, a_opt = multi_lag_subgrad(a_pert, max_iters=n_iters, k_lags=5)

        report(sc, a_opt, f"P5 @{trial}")

        # Occasional coordinate descent
        if trial % 4 == 0 and time.monotonic() < deadline - 30:
            sc, a_opt = targeted_coord_descent(state['best_a'], max_rounds=2, top_k=8)
            report(sc, a_opt, f"P5 coord @{trial}")

    # ============================================================
    # Final polish
    # ============================================================
    if time.monotonic() < deadline - 3:
        sc, a_opt = multi_lag_subgrad(state['best_a'], max_iters=2000, k_lags=5)
        report(sc, a_opt, "final subgrad")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in state['best_a']]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={state['best_score']:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
