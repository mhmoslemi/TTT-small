"""
Optimizer for minimizing 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
Uses a combination of: (1) spectral error reduction, (2) L-BFGS with log-sum-exp smoothing,
(3) iterative flattening, and (4) multi-scale exploration.
Baseline: uniform sequences achieve score = 2.0 (theoretical minimum by Cauchy-Schwarz).
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed objective and gradient for the max-conv problem."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def error_reduction(a, n_iters=200, alpha=0.7):
    """Newton-like error reduction in frequency domain to flatten convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0
        damp = 0.5 + 0.5 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    return best_a


def spectral_flatten(a, n_iters=300, gamma=0.3):
    """
    Alternative spectral method: directly modify the power spectrum to
    flatten the autocorrelation. Uses a soft version of the optimum spectrum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n)
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M = 1 << (2 * n - 1).bit_length()
    M_conv = 2 * n - 1

    for it in range(n_iters):
        # Compute current autocorrelation
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        P = A * np.conj(A)  # Power spectrum
        conv = np.fft.ifft(P).real[:M_conv]

        # Target: make all autocorrelation values equal to their mean
        target_val = float(conv.sum()) / M_conv
        # Soft target: blend between current and flat
        conv_target = (1.0 - gamma) * conv + gamma * target_val
        conv_target = np.maximum(conv_target, 0.0)

        # Compute desired power spectrum via FFT of target autocorrelation
        conv_target_full = np.zeros(M, dtype=np.float64)
        conv_target_full[:M_conv] = conv_target
        P_target = np.fft.fft(conv_target_full)
        P_target = np.maximum(P_target.real, 0.0) + 1e-10

        # Interpolate between current and target spectrum (in log domain)
        log_P = np.log(P + 1e-20)
        log_P_t = np.log(P_target)
        blend = 0.1 + 0.3 * (it / n_iters)
        log_P_new = (1.0 - blend) * log_P + blend * log_P_t
        P_new = np.exp(log_P_new)

        # Recover time domain: A_new = sqrt(P_new) * exp(i*phase)
        phase = np.angle(A)
        mag = np.sqrt(P_new)
        A_new = mag * np.exp(1j * phase)

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def iterative_peak_reduction(a, n_iters=500):
    """
    Iteratively reduce the peak of the autocorrelation by adjusting
    the sequence elements that contribute most to the peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n)
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        # Find contributing pairs
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        # Contribution vector: a[j] contributes a[j]*a[k_star-j] to c[k_star]
        # Gradient of c[k_star] w.r.t. a[j] is a[k_star-j] (and symmetric)
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            partner = k_star - j
            if 0 <= partner < n:
                contrib[j] += a[partner]
                contrib[partner] += a[j]
        # But we only want the direct gradient: dc/d a[j] = sum of partners
        # Actually for c[k_star] = sum_j a[j]*a[k_star-j], dc/da[j] = a[k_star-j]
        # (counting each pair once in the sum)
        # For the score, we need gradient of 2n*M/S^2
        # Let's use a simpler approach: reduce elements with high contribution

        # Top contributors to the peak
        local_contrib = contrib[j_start:j_end + 1]
        top_k = max(10, n // 50)
        top_local = np.argsort(local_contrib)[::-1][:top_k]

        # Reduce top contributors, increase bottom ones
        reduction = 0.01 * a[j_start + top_local[0]] if local_contrib.max() > 0 else 0.01
        bottom_local = np.argsort(local_contrib)[:top_k]

        a_new = a.copy()
        for li in top_local:
            gi = j_start + li
            a_new[gi] = max(0.0, a_new[gi] - reduction)
        total_red = float(a[j_start + top_local].sum() - a_new[j_start + top_local].sum())
        if total_red > 0 and len(bottom_local) > 0:
            per = total_red / len(bottom_local)
            for li in bottom_local:
                gi = j_start + li
                a_new[gi] = min(1000.0, a_new[gi] + per)

        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        else:
            # If no improvement, try smaller step
            a_try = a.copy()
            for li in top_local[:max(1, top_k // 2)]:
                gi = j_start + li
                a_try[gi] = max(0.0, a_try[gi] * 0.98)
            S_try = a_try.sum()
            if S_try > 1e-12:
                a_try = a_try / S_try
                sc_try = compute_score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    break
            else:
                break
        a = a_new

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 1.0)  # Uniform (score = 2)
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(1.0 + 0.2 * np.sin(4.0 * np.pi * x))
    inits.append(0.5 + 0.5 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(1.0 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(1.0 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.5 + 0.5 * (2.0 * x - 1.0) ** 2)
    # Windowed sequences
    inits.append(np.sinc(x - 0.5) ** 2 + 0.01)
    inits.append(1.0 / (1.0 + 50.0 * (x - 0.5) ** 2))
    return inits
