Hybrid optimizer for minimizing 2n*max(self_conv)/(sum)^2 over non-negative sequences.
Uses improved Adam with lookahead, LP-guided subgradients, diverse restarts,
and multi-phase L-BFGS polishing. Key innovations over previous version:

- Lookahead line search for better step sizes

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution a * b via real FFT."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + np.log(exp_scaled.sum()) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_score_a(a):
    """Exact score and subgradient w.r.t. a at the argmax lag."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def weighted_subgrad_a(a, top_k=10, temperature=0.5):
    """Score and weighted subgradient over top-k lags with softmax weights."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c_max = float(c.max())
    score = 2.0 * n * c_max / (S * S)
    
    # Get top-k lags
    top_idx = np.argsort(c)[::-1][:top_k]
    # Softmax weights over the top-k (sharper = more focused on the max)
    top_vals = c[top_idx]
    shifted = (top_vals - top_vals.max()) / temperature
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    sg = np.zeros(n)
    for idx_i, k in enumerate(top_idx):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            idx = k - np.arange(js, je + 1)
            sg[js:je + 1] += weights[idx_i] * 2.0 * a[idx]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
    return score, grad


def lp_weighted_subgrad(a, top_k=20):
    """
    Use LP dual to find optimal weights on top-k lags for subgradient.
    Solves: maximize -sum_k y_k * c[k] subject to sum_k y_k * g[k] = 0, y >= 0, sum(y) = 1
    which gives the 'most balanced' subgradient direction.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c_max = float(c.max())
    score = 2.0 * n * c_max / (S * S)
    
    # Get top-k lags
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    
    # Build gradient matrix: G[i, k] = dc[k]/da[i] for i in 0..n-1, k in top_idx
    # G[i, k] = 2*a[k-i] (if 0 <= k-i < n) + 2*a[i-k] (if 0 <= i-k < n and i != k-i)
    # Actually for self-conv: dc[k]/da[i] = a[k-i] + a[i-k] (each term if index valid)
    # But since we're computing the subgradient of the score, we need:
    # grad_score[i] = 2n * (sg[i]/S^2 - 2*c_max/S^3)
    # where sg[i] = sum_k y_k * dc[k]/da[i]
    
    # For efficiency, we'll compute the gradient contributions
    # dc[k]/da[i] = a[k-i] if 0<=k-i<n, plus a[i-k] if 0<=i-k<n (and i != k-i, else just a[i])
    
    # Actually, let's just use the softmax-weighted approach as approximation
    # The LP would be expensive for large n. Use temperature-based softmax instead.
    temperature = np.std(top_vals) + 1e-10
    shifted = (top_vals - top_vals.max()) / temperature
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    sg = np.zeros(n)
    for idx_i, k in enumerate(top_idx):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            idx = k - np.arange(js, je + 1)
            sg[js:je + 1] += weights[idx_i] * 2.0 * a[idx]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
    return score, grad


def adam_with_lookahead(a_init, lr=0.005, n_steps=100000, beta_lse=100,
                        time_limit_s=None, t_start=None, top_k=10,
                        temperature=0.5, normalize=True, verbose_interval=5000):
    """
    Adam optimizer with lookahead line search and weighted subgradient.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        lr_t = lr / (1.0 + 1e-5 * t)
        
        # Use weighted subgradient for smoother gradient
        if t % 3 == 0:
            score, grad = weighted_subgrad_a(a, top_k=top_k, temperature=temperature)
        else:
            score, grad = lse_score_grad_a(a, beta_lse)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        
        # Base update direction
        direction = m_hat / (np.sqrt(v_hat) + eps)
        
        # Lookahead: try a few step sizes
        candidates = [0.5 * lr_t, lr_t, 2.0 * lr_t]
        best_sc = score
        best_alpha = lr_t
        
        for alpha in candidates:
            a_trial = a - alpha * direction
            a_trial = np.clip(a_trial, 0.0, 1e10)
            if normalize:
                s = a_trial.sum()
                if s > 1e-12:
                    a_trial = a_trial / s
            sc_trial = compute_score(a_trial)
            if sc_trial < best_sc:
                best_sc = sc_trial
                best_alpha = alpha
        
        a = a - best_alpha * direction
        a = np.clip(a, 0.0, 1e10)
        
        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a / s
        
        if t % verbose_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


def lbfgs_refine(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B refinement with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        x_clip = np.clip(x, 0.0, 1e10)
        s = x_clip.sum()
        if s > 1e-12:
            x_norm = x_clip / s
        else:
            x_norm = x_clip
        sc, gr = lse_score_grad_a(x_norm, beta)
        return sc, gr
    
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def selective_coord_descent(a, top_k=50, n_sweeps=20, time_limit_s=None, t_start=None):
    """Coordinate descent on top-K coordinates per sweep."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        score, grad = subgrad_score_a(a)
        if score < best_score:
            best_score = score
        
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            
            a_old = a[j]
            score, _ = compute_score(a)
            
            if grad[j] > 1e-10:
                ratios = np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.0, 1.02])
                candidates = a_old * ratios
            elif grad[j] < -1e-10:
                candidates = np.array([0.0, a_old * 0.5, a_old, a_old * 1.2, a_old * 1.5, a_old * 2.0])
            else:
                continue
            
            candidates = np.clip(candidates, 0.0, 1000.0)
            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    
    return best_score, a


def frequency_perturb(a, rng, strength=0.1):
    """Perturb in frequency domain: modify the magnitude spectrum."""
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Multiply magnitude by a random smooth function
    freq = np.arange(len(mag)) / len(mag)
    modulator = 1.0 + strength * np.cos(2 * np.pi * rng.uniform(1, 10) * freq + rng.uniform(0, 2 * np.pi))
    modulator = 1.0 + strength * np.cos(2 * np.pi * rng.uniform(1, 10) * freq + rng.uniform(0, 2 * np.pi))
    mag_new = mag * modulator
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    
    if a_new.sum() < 1e-10:
        a_new = a * (1.0 + 0.05 * rng.standard_normal(n))
        a_new = np.clip(a_new, 0.0, 1000.0)
    
    return a_new


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer with lookahead Adam, diverse restarts, and L-BFGS polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    effective_budget = total_budget - 20.0  # Leave time for final checks
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    if parent is not None:
        parent_score = compute_score(parent)
        print(f"Parent loaded: n={len(parent)}, score={parent_score:.8f}")
    else:
        parent = np.ones(4096)
        parent_score = compute_score(parent)
        print(f"No parent, using uniform n=4096, score={parent_score:.8f}")
    
    # Normalize parent to sum=1 for optimization
    best_a = parent / parent.sum()
    best_score = compute_score(best_a)
    n_best = len(best_a)
    
    def try_update(sc, a_arr, label):
        nonlocal best_score, best_a, n_best
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_arr.copy(), 0.0, 1000.0)
            n_best = len(a_arr)
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={n_best} | t={elapsed:.0f}s/{total_budget:.0f}s")
            return True
        return False
    
    # ==================================================================
    # Phase 1: Adam with continuation (35% of time)
    # ==================================================================
    p1 = effective_budget * 0.35
    print(f"\n{'='*50}")
    print(f"Phase 1: Adam continuation (budget={p1:.0f}s)")
    print(f"{'='*50}")
    
    a_cur = best_a.copy()
    
    # Continuation: low beta -> high beta
    beta_schedule = [(50, 0.01, 0.3), (150, 0.005, 0.25), (500, 0.003, 0.25), (1500, 0.001, 0.2)]
    for beta, lr, frac in beta_schedule:
        remaining = p1 - (time.monotonic() - t_start)
        if remaining < 10:
            break
        time_alloc = remaining * frac
        top_k = max(5, min(20, int(2000 / max(1, beta ** 0.3))))
        sc, a_opt = adam_with_lookahead(a_cur, lr=lr, n_steps=500000, beta_lse=beta,
                                         time_limit_s=time_alloc, t_start=t_start,
                                         top_k=top_k, temperature=max(0.1, 50.0/beta))
        try_update(sc, a_opt, f"Adam beta={beta}")
        a_cur = a_opt if sc < compute_score(a_cur) else a_cur
    
    # ==================================================================
    # Phase 2: L-BFGS polish (15% of time)
    # ==================================================================
    p2 = effective_budget * 0.15
    print(f"\n{'='*50}")
    print(f"Phase 2: L-BFGS Polish (budget={p2:.0f}s)")
    print(f"{'='*50}")
    
    for beta in [500, 1500, 4000]:
        remaining = p2 - (time.monotonic() - t_start)
        if remaining < 10:
            break
        t0 = time.monotonic()
        sc, a_opt = lbfgs_refine(best_a, beta, maxiter=8000, maxfun=150000)
        try_update(sc, a_opt, f"L-BFGS beta={beta}")
        print(f"    L-BFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # ==================================================================
    # Phase 3: Diverse restarts (35% of time)
    # ==================================================================
    p3 = effective_budget * 0.35
    print(f"\n{'='*50}")
    print(f"Phase 3: Diverse Restarts (budget={p3:.0f}s)")
    print(f"{'='*50}")
    
    restart_deadline = t_start + p1 + p2 + p3 - 5.0
    n_restarts = 16
    per_restart_time = (p3 - (time.monotonic() - t_start)) / max(1, n_restarts)
    
    for r in range(n_restarts):
        if time.monotonic() > restart_deadline:
            break
        
        strategy = r % 8
        a_pert = best_a.copy()
        
        if strategy == 0:
            # Gaussian perturbation (small)
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strategy == 1:
            # Gaussian perturbation (medium)
            scale = 0.05 + 0.05 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strategy == 2:
            # Peak reduction: reduce elements contributing to top lag
            c = fast_conv(best_a)
            top_lags = np.argsort(c)[::-1][:3]
            for k in top_lags:
                k = int(k)
                js = max(0, k - n_best + 1)
                je = min(k, n_best - 1)
                a_pert[js:je+1] *= 0.7
            # Add back mass to random positions
            removed = float(best_a.sum() - a_pert.sum())
            if removed > 0:
                add_positions = rng.choice(n_best, size=min(n_best//2, 500), replace=False)
                a_pert[add_positions] += removed / len(add_positions)
        elif strategy == 3:
            # Frequency domain perturbation
            a_pert = frequency_perturb(best_a, rng, strength=0.1 + 0.1 * rng.random())
        elif strategy == 4:
            # Block modification
            n_blocks = rng.integers(2, 6)
            block_size = n_best // n_blocks
            for _ in range(n_blocks):
                start = int(rng.integers(0, n_best - block_size))
                a_pert[start:start+block_size] *= rng.uniform(0.3, 1.5)
        elif strategy == 5:
            # Sinusoidal modulation
            freq = rng.uniform(0.5, 5.0)
            phase = 2.0 * np.pi * rng.random()
            x = np.arange(n_best) / n_best
            mod = 1.0 + 0.05 * np.cos(2*np.pi*freq*x + phase)
            a_pert = best_a * mod
        elif strategy == 6:
            # "Anti-peak": boost positions that don't contribute to peak
            c = fast_conv(best_a)
            k_star = int(c.argmax())
            js = max(0, k_star - n_best + 1)
            je = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            peak_mask = np.zeros(n_best, dtype=bool)
            peak_mask[js:je+1] = True
            non_peak_mask = ~peak_mask
            if non_peak_mask.sum() > 0:
                boost = rng.uniform(1.1, 1.5)
                a_pert[non_peak_mask] *= boost
                a_pert[js:je+1] *= 0.8
        else:
            # Mirror + perturb: reverse halves and mix
            half = n_best // 2
            a_pert = best_a.copy()
            a_pert[:half] = best_a[::-1][:half] * 0.5 + best_a[:half] * 0.5
            a_pert[half:] = best_a[n_best-1-half:n_best][::-1] * 0.5 + best_a[half:] * 0.5
        
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s
        
        # Quick optimization: Adam + L-BFGS
        time_left = restart_deadline - time.monotonic()
        if time_left < 5:
            break
        adam_time = min(15.0, time_left * 0.5)
        sc, a_r = adam_with_lookahead(a_pert, lr=0.003, n_steps=100000, beta_lse=300,
                                       time_limit_s=adam_time, t_start=t_start, top_k=10)
        if time.monotonic() < restart_deadline - 3:
            sc2, a_r2 = lbfgs_refine(a_r, 1000, maxiter=3000, maxfun=50000)
            if sc2 < sc:
                sc, a_r = sc2, a_r2
        try_update(sc, a_r, f"restart#{r} s{strategy}")
    
    # ==================================================================
    # Phase 4: Coordinate descent + final polish (15% of time)
    # ==================================================================
    p4 = effective_budget * 0.15
    print(f"\n{'='*50}")
    print(f"Phase 4: Coord Descent + Final Polish (budget={p4:.0f}s)")
    print(f"{'='*50}")
    
    final_deadline = t_start + effective_budget - 10.0
    
    # Coordinate descent
    t0 = time.monotonic()
    cd_time = min(30.0, p4 * 0.4)
    sc, a_opt = selective_coord_descent(best_a, top_k=50, n_sweeps=50,
                                         time_limit_s=cd_time, t_start=t_start)
    try_update(sc, a_opt, "Coord descent")
    print(f"    Coord descent: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # High-beta L-BFGS
    for beta in [3000, 8000]:
        if time.monotonic() > final_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt = lbfgs_refine(best_a, beta, maxiter=10000, maxfun=200000)
        try_update(sc, a_opt, f"Final L-BFGS beta={beta}")
        print(f"    Final L-BFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # Final Adam with very small lr
    if time.monotonic() < final_deadline - 5:
        t0 = time.monotonic()
        sc, a_opt = adam_with_lookahead(best_a, lr=0.0005, n_steps=200000, beta_lse=3000,
                                         time_limit_s=final_deadline - time.monotonic() - 5,
                                         t_start=t_start, top_k=5, temperature=0.1)
        try_update(sc, a_opt, "Final Adam")
        print(f"    Final Adam: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # One more coordinate descent
    if time.monotonic() < final_deadline:
        sc, a_opt = selective_coord_descent(best_a, top_k=60, n_sweeps=10,
                                             time_limit_s=min(15.0, final_deadline - time.monotonic()),
                                             t_start=t_start)
        try_update(sc, a_opt, "Final coord")
    
    # ==================================================================
    # Phase 5: Try slightly different lengths
    # ==================================================================
    if time.monotonic() < final_deadline - 20:
        print(f"\n{'='*50}")
        print("Phase 5: Length Variation")
        print(f"{'='*50}")
        
        n_parent = len(best_a)
        for factor in [0.9, 1.1]:
            new_n = int(n_parent * factor)
            if new_n < 256 or new_n > 50000:
                continue
            if time.monotonic() > final_deadline - 10:
                break
            x_old = np.arange(n_parent)
            x_new = np.linspace(0, n_parent - 1, new_n)
            a_res = np.interp(x_new, x_old, best_a)
            a_res = np.clip(a_res, 0.0, 1000.0)
            if a_res.sum() > 1e-10:
                a_res = a_res / a_res.sum()
                sc, a_r = adam_with_lookahead(a_res, lr=0.003, n_steps=50000, beta_lse=500,
                                               time_limit_s=10.0, t_start=t_start, top_k=10)
                if time.monotonic() < final_deadline - 5:
                    sc2, a_r2 = lbfgs_refine(a_r, 2000, maxiter=3000, maxfun=50000)
                    if sc2 < sc:
                        sc, a_r = sc2, a_r2
                try_update(sc, a_r, f"n={new_n} (x{factor})")
    
    # ==================================================================
    # Final: verify and return
    # ==================================================================
    # Scale back to ensure sum >= 0.01 (safety)
    out = best_a.copy()
    s = out.sum()
    if s < 0.01:
        out = out * (1.0 / s) if s > 0 else np.ones(64) * 0.5
    out = np.clip(out, 0.0, 1000.0)
    
    out_list = [float(x) for x in out]
    verified = float(evaluate_sequence(out_list))
    
    # If verified is worse than internal (shouldn't happen), use parent
    if verified > best_score + 0.01 and parent is not None:
        print(f"WARNING: verified={verified:.6f} > internal={best_score:.6f}, using parent")
        out_list = [float(max(0.0, min(1000.0, x))) for x in parent]
        verified = float(evaluate_sequence(out_list))
    
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out_list)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out_list
