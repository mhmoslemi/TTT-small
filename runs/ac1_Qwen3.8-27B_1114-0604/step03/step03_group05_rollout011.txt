    x = a0.astype(np.float64).copy()
    y = x.copy()
    best_x = x.copy()
    best_f = lse_score_grad(x, beta)[0]
    step = step_init / (n ** 0.5)

    for it in range(maxiter):
        f_y, g_y = lse_score_grad(y, beta)
        # Nesterov step
        x_new = np.clip(y - step * g_y, 0.0, 1000.0)
        # Update momentum
        y = np.clip(x_new + mu * (x_new - x), 0.0, 1000.0)
        x = x_new

        f_x = lse_score_grad(x, beta)[0]
        if f_x < best_f:
            best_f = f_x
            best_x = x.copy()

        # Backtracking line search every 10 iterations
        if it % 10 == 0 and it > 0:
            f_check, g_check = lse_score_grad(x, beta)
            gn = float(np.linalg.norm(g_check))
            if gn > 1e-10:
                # Adaptive step: target reduction
                target_reduction = 0.5 * (f_y - f_check)
                if target_reduction > 0:
                    step = min(step * 1.5, target_reduction / (gn * gn + 1e-20))
                else:
                    step = max(step * 0.7, 1e-8 / (gn + 1e-20))
                step = min(step, 10.0 / (float(x.sum()) + 1e-10))

    return best_x, best_f


def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def peak_shave_linesearch(a, iters=5000):
    """Subgradient descent with backtracking line search on max-conv."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find all near-max peaks (within 1% of max)
        thresh = M * 0.99
        peak_indices = np.where(c >= thresh)[0]

        # Compute combined subgradient
        sg = np.zeros(n)
        # Weight peaks by their values (soft)
        weights = np.exp(50.0 * (c[peak_indices] - M))
        weights /= weights.sum()

        for w, k_star in zip(weights, peak_indices):
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            sg[j_start:j_end + 1] += w * 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Backtracking line search
        step = min(1.0 / (gn * S + 1e-20), 1.0 / (S + 1e-10))
        f_curr = sc
        for _ls in range(20):
            a_trial = np.clip(a - step * grad, 0.0, 1000.0)
            c_trial = fast_conv(a_trial)
            S_trial = float(a_trial.sum())
            if S_trial < 1e-12:
                break
            f_trial = 2.0 * n * float(c_trial.max()) / (S_trial * S_trial)
            if f_trial < f_curr * 0.999:
                a = a_trial
                break
            step *= 0.5
        else:
            # Accept anyway if no improvement found
            a = np.clip(a - step * grad, 0.0, 1000.0)

    return best_sc, best_a


def freq_init(n, rng):
    """Initialize with approximately flat power spectrum, projected to non-negative."""
    # Start with random phases in frequency domain
    L = 1 << (2 * n - 1).bit_length()
    A = np.zeros(L, dtype=np.complex128)
    # Set DC component
    A[0] = np.sqrt(L)
    # Set other frequencies with random phases and roughly constant magnitude
    mags = np.ones(L // 2) * (1.0 + 0.3 * rng.standard_normal(L // 2))
    mags = np.clip(mags, 0.1, None)
    phases = 2.0 * np.pi * rng.random(L // 2)
    A[1:L // 2 + 1] = mags * np.exp(1j * phases)
    # Make conjugate symmetric for real output
    A[L // 2 + 1:] = np.conj(A[1:L // 2 + 1][::-1])
    a = np.fft.ifft(A).real[:n]
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-6:
        a = np.full(n, 0.5)
    return a


def structured_init(n, rng):
    """Structured initialization using harmonic series."""
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    # Add several sinusoidal components with random phases
    for freq in range(1, min(20, n // 4)):
        amp = 0.1 / np.sqrt(freq)
        phase = 2.0 * np.pi * rng.random()
        a += amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-6:
        a = np.full(n, 0.5)
    return a


def cosine_taper_init(n, rng):
    """Cosine-tapered initialization with random perturbation."""
    x = np.arange(n, dtype=np.float64)
    a = 0.5 + 0.5 * np.cos(np.pi * x / (n - 1)) if n > 1 else np.ones(n) * 0.5
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.0, None)
    return a


def optimize_sequence(a0, time_budget_s, verbose=True):
    """Full optimization pipeline for a single sequence length."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.astype(np.float64).copy()
    sc = compute_score(a)
    if verbose:
        print(f"    Initial (n={n}): {sc:.8f}")

    # Phase A: Variance flattening (quick)
    if time.monotonic() - t0 < time_budget_s * 0.1:
        # Simple: run a few LSE iterations at low beta
        for beta in [10, 20]:
            if time.monotonic() - t0 > time_budget_s * 0.1:
                break
            a = run_lbfgs(a, lse_obj_func, (beta,), maxiter=300, maxfun=6000)
        sc = compute_score(a)
        if verbose:
            print(f"    After flatten: {sc:.8f}")

    # Phase B: LSE continuation
    beta_schedule = [20, 50, 100, 200, 400, 800, 1600, 3200, 6400, 12800]
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_budget_s * 0.6:
            break
        # L-BFGS-B round
        a = run_lbfgs(a, lse_obj_func, (beta,), maxiter=1500, maxfun=30000)
        # Nesterov round (helps escape flat regions)
        a_new, f_new = nesterov_lse(a, beta, maxiter=min(1000, int((time_budget_s * 0.6 - (time.monotonic() - t0)) * 50)), mu=0.7)
        sc_new = compute_score(a_new)
        sc_cur = compute_score(a)
        if sc_new < sc_cur:
            a = a_new
            sc = sc_new
        else:
            sc = sc_cur
        if verbose and beta % 400 == 0:
            remaining = time_budget_s * 0.6 - (time.monotonic() - t0)
            print(f"    beta={beta}: {sc:.8f} (remaining phase B: {remaining:.0f}s)")

    # Phase C: High-beta refinement
    for beta in [12800, 25600, 51200]:
        if time.monotonic() - t0 > time_budget_s * 0.8:
            break
        a = run_lbfgs(a, lse_obj_func, (beta,), maxiter=2000, maxfun=40000)
        sc = compute_score(a)
        if verbose:
            print(f"    Refine beta={beta}: {sc:.8f}")

    # Phase D: Peak shaving with line search
    iters = min(5000, max(500, int((time.monotonic() - t0) * 100)))
    if time.monotonic() - t0 < time_budget_s * 0.95:
        sc, a = peak_shave_linesearch(a, iters=min(iters, 3000))
        if verbose:
            print(f"    After peak shave: {sc:.8f}")

    return sc, a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation function.
    Multi-phase: multi-length search, LSE continuation with Nesterov, peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine which lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
        sc_parent = compute_score(parent)
        report(sc_parent, parent, "parent")

    # Add other lengths to try (spread across the budget)
    candidate_lengths = [2048, 4096, 6962, 8192, 12288]
    for cl in candidate_lengths:
        if cl not in lengths:
            lengths.append(cl)

    # Sort by priority: parent length first, then others
    if parent is not None:
        lengths.sort(key=lambda x: (x != len(parent), abs(x - (len(parent) if parent is not None else 4096))))
    else:
        lengths.sort(key=lambda x: x)

    # Budget allocation: give more time to the best-known length
    # First pass: quick evaluation of different lengths
    # Second pass: deep optimization of the best

    # Quick pass: try each length briefly to identify the best
    quick_budget_per = min(120.0, (total_budget - 60.0) / (2 * len(lengths)))
    quick_best_score = np.inf
    quick_best_a = None

    print(f"\n=== Quick Pass: Evaluating {len(lengths)} lengths ===")
    for idx, n_try in enumerate(lengths):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        cur_budget = min(quick_budget_per, remaining / 2)

        # Initialize
        if parent is not None and n_try == len(parent):
            a_init = parent.copy()
        else:
            # Try a couple of initializations
            a1 = freq_init(n_try, rng)
            a2 = structured_init(n_try, rng)
            sc1 = compute_score(a1)
            sc2 = compute_score(a2)
            a_init = a1 if sc1 < sc2 else a2

        sc_init = compute_score(a_init)
        if sc_init < quick_best_score:
            quick_best_score = sc_init
            quick_best_a = a_init.copy()
        print(f"  Length {n_try}: init score = {sc_init:.8f} (budget {cur_budget:.0f}s)")

        # Brief optimization
        if cur_budget > 30:
            sc_opt, a_opt = optimize_sequence(a_init, cur_budget, verbose=False)
            if sc_opt < quick_best_score:
                quick_best_score = sc_opt
                quick_best_a = a_opt.copy()
            report(sc_opt, a_opt, f"quick n={n_try}")

    # Deep optimization of the best found so far
    print(f"\n=== Deep Optimization Phase ===")
    deep_a = quick_best_a if quick_best_a is not None else (parent.copy() if parent is not None else np.full(4096, 0.5))
    remaining = max(60.0, deadline - time.monotonic())
    print(f"  Deep optimizing n={len(deep_a)} with {remaining:.0f}s")

    sc_deep, a_deep = optimize_sequence(deep_a, remaining, verbose=True)
    report(sc_deep, a_deep, "deep")

    # Try a few more restarts if time permits
    print(f"\n=== Restart Phase ===")
    restart_count = 0
    while time.monotonic() < deadline - 30 and restart_count < 10:
        remaining = deadline - time.monotonic() - 30
        if remaining < 60:
            break

        n_restart = len(best_a) if best_a is not None else 4096
        strategy = restart_count % 4

        if strategy == 0:
            # Gaussian perturbation
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_restart))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted: reduce peak contributors
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_restart + 1)
            j_end = min(k_star, n_restart - 1)
            a_pert = best_a.copy()
            contrib = a_pert[j_start:j_end + 1].copy()
            top_k = max(10, n_restart // 50)
            top_idx = np.argsort(contrib)[::-1][:top_k]
            reduce_total = 0.0
            for j in top_idx:
                amt = a_pert[j_start + j] * 0.03
                a_pert[j_start + j] -= amt
                a_pert[j_start + j] = max(a_pert[j_start + j], 0.0)
                reduce_total += amt
            other_idx = set(range(n_restart)) - set(range(j_start, j_end + 1))
            other_idx = list(other_idx)
            if other_idx:
                per = reduce_total / len(other_idx)
                for j in other_idx:
                    a_pert[j] += per
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Frequency domain perturbation
            L = 1 << (2 * n_restart - 1).bit_length()
            A = np.fft.fft(best_a, L)
            A_mag = np.abs(A) * (1.0 + 0.03 * rng.standard_normal(L))
            A_mag = np.clip(A_mag, 0, None)
            A_new = A_mag * np.exp(1j * np.angle(A))
            a_pert = np.fft.ifft(A_new).real[:n_restart]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Different length attempt
            n_alt = int(n_restart * (0.8 + 0.4 * rng.random()))
            n_alt = max(512, min(16384, n_alt))
            if n_alt == n_restart:
                n_alt += 256
            a_pert = freq_init(n_alt, rng)

        cur_budget = min(remaining / 2, 300.0)
        sc_r, a_r = optimize_sequence(a_pert, cur_budget, verbose=False)
        report(sc_r, a_r, f"restart #{restart_count}")
        restart_count += 1

    # Final polish
    if time.monotonic() < deadline - 5 and best_a is not None:
        print(f"\n=== Final Polish ===")
        # One more high-beta LSE
        a_final = run_lbfgs(best_a, lse_obj_func, (50000,), maxiter=1000, maxfun=20000)
        sc_final = compute_score(a_final)
        report(sc_final, a_final, "final LSE")

        # Peak shave
        if time.monotonic() < deadline - 5:
            sc_ps, a_ps = peak_shave_linesearch(best_a, iters=2000)
            report(sc_ps, a_ps, "final peak shave")

    # Output
    if best_a is None:
        best_a = np.full(64, 0.5)

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
