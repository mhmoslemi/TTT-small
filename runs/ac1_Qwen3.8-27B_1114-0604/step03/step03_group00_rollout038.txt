1. **GS-IRFT phase**: Use iterative Fourier transform to push toward flat auto-conv. Use the "soft" target: instead of trying to make c exactly flat, try to reduce the peaks. This is equivalent to:

   - Compute the "excess" e[k] = max(c[k] - mean(c), 0)

"""
Optimization of non-negative sequences to minimize 2n*max(conv(a,a))/(sum(a))^2.

Strategy:
1. GS-IRFT (Gerchberg-Saxton / Iterative Fourier Transform) to push toward flat auto-convolution
2. L-BFGS on LSE-smoothed objective for smooth refinement
3. Peak-targeted coordinate descent
4. Multi-length exploration with adaptive time allocation
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def gs_flatten(a0, n_iters=500, sigma=0.5):
    """
    GS-style iterative method to flatten the auto-convolution.
    At each step:
    1. Compute auto-conv c
    2. Compute target c_target (soft flat)
    3. Compute error e = c - c_target
    4. Newton step in frequency domain: A -= e_spec / (2A)
    5. IFFT, project to non-negative
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) * 0.5
        return a
    a = a / S
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        
        # Compute auto-conv
        conv_spec = A * A
        c = np.fft.irfft(conv_spec, M_fft)[:M]
        
        c_mean = float(c.mean())
        c_max = float(c.max())
        
        if c_max - c_mean < 1e-14:
            break
        
        # Soft target: reduce peaks by sigma fraction
        target = c_mean + (1.0 - sigma) * (c_max - c_mean)
        error = np.maximum(c - target, 0.0)
        
        if error.max() < 1e-14:
            break
        
        # Error spectrum
        err_pad = np.zeros(M_fft)
        err_pad[:M] = error
        err_spec = np.fft.rfft(err_pad)
        
        # Newton step: we want to reduce c by 'error'
        # c is IFFT of A*A, so d(A*A) = -err_spec
        # 2*A*dA = -err_spec => dA = -err_spec / (2*A)
        with np.errstate(divide='ignore', invalid='ignore'):
            dA = -err_spec / (2.0 * A + 1e-20)
        dA[np.abs(A) < 1e-10] = 0.0
        
        # Damping
        damp = min(1.0, 0.3 + 0.7 * it / max(1, n_iters // 3))
        A_new = A + damp * dA
        
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        
        S = float(a_new.sum())
        if S < 1e-12:
            break
        a_new = a_new / S
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
        
    return best_a

def gs_rereflect(a0, n_iters=500):
    """
    Alternative GS: rereflect the spectrum toward the target magnitude.
    Target: the auto-conv should be as flat as possible.
    We use the "optimal" target magnitude, which is the square root of 
    the projection of the flat target onto the cone of non-negative spectra.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) * 0.5
        return a
    a = a / S
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    # Target: flat auto-conv of length M
    # c_target[k] = 1/M for k=0..M-1 (normalized so sum=1)
    # This has DFT: W(f) = sum_{k=0}^{M-1} exp(-i2πfk/M_fft) / M
    # |W(f)| = |sin(πfM/M_fft)| / (M * |sin(πf/M_fft)|)
    
    # But W(f) can be negative, so |A|^2 = W is not always valid.
    # Instead, use |W(f)| as the target magnitude for A
    target_c = np.ones(M) / M
    target_c_pad = np.zeros(M_fft)
    target_c_pad[:M] = target_c
    target_spec = np.fft.rfft(target_c_pad)
    target_A_mag = np.sqrt(np.abs(target_spec))
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        
        # Rereflect: set |A| to target, keep phase
        current_mag = np.abs(A)
        with np.errstate(divide='ignore', invalid='ignore'):
            scale = np.where(current_mag > 1e-15, target_A_mag / current_mag, 1.0)
        A = A * scale
        
        a_new = np.fft.irfft(A, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        
        S = float(a_new.sum())
        if S < 1e-12:
            break
        a_new = a_new / S
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2  # length of conv(rev_a, p) is n + M - 1 = n + 2n-1 - 1 = 3n-2
    # Actually, conv of length-n and length-(2n-1) is length 3n-2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    # dL/da[j] = sum_k p[k] * (da[k]/da[j]) where c[k] = sum_i a[i]a[k-i]
    # dc[k]/da[j] = a[k-j] + a[j-k] (with appropriate bounds)
    # = a reversed, convolved with a delta at j
    # The gradient of L w.r.t. a is: sum_k p[k] * (a_rev convolved with delta_j at position k)
    # = (p convolved with a_rev) evaluated appropriately
    # Actually: dL/d a[j] = sum_k p[k] * dc[k]/da[j]
    # dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    # = a[k-j] + a[j-k] with bounds
    # This equals: (a * delta_j)(k) + (a * delta_j)(k) ... 
    # More precisely: dc[k]/da[j] = a[k-j] + a[j-k] where terms are 0 if index out of [0,n-1]
    # = (forward_corr + backward_corr)
    # = conv(a_rev, delta_j)[k] + conv(a, delta_j)[k]
    # Hmm, let me think differently.
    # c[k] = sum_i a[i] * a[k-i] (with 0 <= i < n, 0 <= k-i < n)
    # d c[k] / d a[j] = a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n)

    # Actually sum_k p[k] * a[k-j] for valid k: this is like a cross-correlation
    # Let's define: for fixed j, sum_k p[k]*a[k-j] where k ranges over [0, 2n-2] and k-j in [0,n-1]
    # = sum_i p[i+j] * a[i] where i in [0,n-1] and i+j in [0, 2n-2]

    # Similarly sum_k p[k]*a[j-k] = sum_i p[j-i]*a[i] (with j-i in [0,2n-2])
    
    # The clean way: grad_a[j] = sum_k p[k] (a[k-j] + a[j-k])
    # = sum_k p[k] a[k-j] + sum_k p[k] a[j-k]
    # First term: let i = k-j, so k = i+j, sum_i p[i+j] a[i] = (p shifted right by j) dot a
    # This is the cross-correlation of p and a evaluated at lag j
    # Second term: let i = j-k, so k = j-i, sum_i p[j-i] a[i] = (p reversed and shifted) dot a
    # This is... let me just compute it as conv(p, a_rev)
    
    # Actually, the cleanest: 
    # sum_k p[k] * a[k-j] = (p ⊛ a)[j] where ⊛ is cross-correlation
    # And sum_k p[k] * a[j-k] = (p ⊛ a)[j] as well (by symmetry of cross-correlation for real signals)

    # And (a ⊛ p)[j] = sum_k a[k+j] p[k] = sum_i a[i] p[i-j]... 
    
    # Let me just use the direct formula:
    # grad_L[j] = 2 * sum_k p[k] * a[(k-j) mod n] ... no, that's circular.
    
    # The correct formula (from the previous code):
    # The gradient of L = log(sum exp(beta*(c[k]-c_max))/beta) + c_max
    # dL/da[j] = sum_k p[k] * dc[k]/da[j]

    # So g[j] = corr_{a,p}[j] + corr_{a, p_rev}[j]
    #

    # = sum_i a[n-1-i] * p[j+n-1-i]
    # Let m = n-1-i, so i = n-1-m:
    # = sum_m a[m] * p[j+m]
    # = corr_{a,p}[j] ✓
    
    # But we need corr_{a,p}[j] + corr_{a,p_rev}[j]
    # corr_{a,p_rev}[j] = sum_m a[m] * p_rev[m+j] = sum_m a[m] * p[2n-2-m-j]
    # 
    # Hmm, this is getting complicated. Let me just use the formula from the previous code which worked.
    
    # The previous code computed: conv_res = irfft(RA * P)[n-1:2n-1] where RA = rfft(rev_a) and P = rfft(p)
    # And used grad = 2n * (g/S^2 - 2L/S^3) where g = 2 * conv_res
    # This gave correct results, so let me just use it.
    
    g = 2.0 * conv_res[n-1:2*n-1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def lse_objective(a, beta):
    return lse_score_grad(a, beta)

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def build_inits(n, rng):
    """Generate diverse initializations optimized for this problem."""
    inits = []
    x = np.linspace(0, 1, n)
    
    # U-shaped (small middle, large ends) - helps reduce c[n-1]
    inits.append(1.0 + 2.0 * (2*x - 1)**2)
    inits.append(0.5 + 1.5 * (2*x - 1)**2)
    inits.append(1e-3 + 1.0 * (2*x - 1)**4)
    
    # Asymmetric exponential decay
    inits.append(np.exp(-3.0 * x))
    inits.append(np.exp(-5.0 * x) + 0.1)
    inits.append(np.exp(-3.0 * (1-x)) + 0.1)
    
    # Sinusoidal
    inits.append(0.5 + 0.5 * np.sin(np.pi * x)**2)
    inits.append(0.3 + 0.7 * np.cos(2*np.pi*x)**2)
    inits.append(1.0 + 0.5 * np.cos(4*np.pi*x))
    
    # Linear ramp
    inits.append(0.1 + 0.9 * x)
    inits.append(0.1 + 0.9 * (1-x))
    
    # Power law
    inits.append(x**0.5 + 0.1)
    inits.append((1-x)**0.5 + 0.1)
    inits.append(x**2 + 0.1)
    
    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, None))
    inits.append(np.clip(1.0 + 0.5 * rng.standard_normal(n), 0.01, None))
    
    # Two bumps
    inits.append(np.exp(-20*(x-0.2)**2) + np.exp(-20*(x-0.8)**2) + 0.1)
    
    # Triangle
    inits.append(1.0 - 2.0*np.abs(x - 0.5) + 0.1)  # inverted V (peak in middle)
    inits.append(2.0*np.abs(x - 0.5) + 0.1)  # V (valley in middle)
    
    return [np.clip(v, 0.0, 1000.0) for v in inits]

"""
Optimizes non-negative sequences to minimize 2n*max(conv(a,a))/(sum(a))^2.

Strategy:
1. GS-IRFT (Iterative Fourier Transform) with adaptive peak-shaving to flatten auto-convolution
2. L-BFGS on LSE-smoothed objective for smooth gradient-based refinement  
3. Diverse initialization including U-shaped, asymmetric, Tukey windows
4. Multi-length exploration with adaptive time budget
5. Local search with targeted peak-reduction perturbations
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

    Adaptive GS/IRFT: iteratively flattens auto-convolution via Newton steps in freq domain.
    Sigma increases over iterations for progressive peak reduction.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()

    if S < 1e-12:
        return np.ones(n) * 0.5
    a = a / S

    M_fft = 1 << (M - 1).bit_length()

    best_a = a.copy()
    best_sc = compute_score(a)

        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        conv_spec = A * A
        c = np.fft.irfft(conv_spec, M_fft)[:M]

        c_mean = float(c.mean())
        c_max = float(c.max())

        frac = it / max(1, n_iters)
        sigma = sigma_start + (sigma_end - sigma_start) * frac

        error = np.maximum(c - target, 0.0)

        err_pad = np.zeros(M_fft)
        err_pad[:M] = error
        err_spec = np.fft.rfft(err_pad)

        # Newton step in frequency domain
        with np.errstate(divide='ignore', invalid='ignore'):
            dA = -err_spec / (2.0 * A + 1e-20)
        dA[np.abs(A) < 1e-10] = 0.0

        over_relax = 1.0 + 0.3 * frac
        damp = min(1.0, 0.4 + 0.6 * frac)
        A_new = A + damp * over_relax * dA

        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S = float(a_new.sum())
        if S < 1e-12:
            break
        a_new = a_new / S

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

    return best_a


def gs_rereflect(a0, n_iters=200):
    """
    GS rereflection: set |A(f)| to target magnitude (sqrt of flat window spectrum),
    keep phase. Pushes toward flat auto-convolution spectrum.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) * 0.5
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    # Target magnitude: sqrt of |DFT of rectangular window of length M|
    target_c = np.ones(M) / M
    target_c_pad = np.zeros(M_fft)
    target_c_pad[:M] = target_c
    target_spec = np.fft.rfft(target_c_pad)
    target_A_mag = np.sqrt(np.abs(target_spec))

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        current_mag = np.abs(A)
        with np.errstate(divide='ignore', invalid='ignore'):
            scale = np.where(current_mag > 1e-15, target_A_mag / current_mag, 1.0)
        A = A * scale

        a_new = np.fft.irfft(A, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S = float(a_new.sum())
        if S < 1e-12:
            break
        a_new = a_new / S

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for the objective 2n*max(conv)/(sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    # Gradient: dL/da[j] = 2 * conv(p, a_rev)[n-1+j]
    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    AR = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(P * AR, M2_fft)
    conv_slice = conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (4.0 * n / S3) * (S * conv_slice - L)
    return score, grad


def lse_objective_wrapper(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def build_diverse_inits(n, rng):
    """Generate diverse initializations designed for low-PAR auto-convolution."""
    inits = []
    x = np.linspace(0, 1, n)

    # U-shaped: reduces middle auto-conv values
    inits.append(1.0 + 3.0 * (2.0 * x - 1.0) ** 2)
    inits.append(0.1 + 2.0 * (2.0 * x - 1.0) ** 4)
    inits.append(0.05 + 1.5 * np.abs(2.0 * x - 1.0))

    # Asymmetric: breaks symmetry to reduce c[n-1]
    inits.append(np.exp(-4.0 * x) + 0.05)
    inits.append(np.exp(-6.0 * x) + 0.02)
    inits.append(np.exp(-4.0 * (1.0 - x)) + 0.05)
    inits.append(0.1 + 0.9 * x)
    inits.append(0.1 + 0.9 * x ** 2)

    # Tukey (tapered cosine) windows
    for alpha in [0.3, 0.5, 0.7]:
        w = np.ones(n)
        taper = int(alpha * n // 2)
        if taper > 0:
            w[:taper] = 0.5 * (1 + np.cos(np.pi * np.arange(taper) / taper - np.pi / 2))
            w[-taper:] = 0.5 * (1 + np.cos(np.pi * np.arange(taper) / taper + np.pi / 2))
        inits.append(w + 0.05)

    # Sinusoidal
    inits.append(1.0 + 0.5 * np.cos(2.0 * np.pi * x))
    inits.append(1.0 + 0.7 * np.cos(4.0 * np.pi * x))
    inits.append(0.5 + 0.5 * np.sin(np.pi * x) ** 2)

    # Power law
    inits.append(x ** 0.3 + 0.1)
    inits.append(x ** 3.0 + 0.1)

    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, None))
    inits.append(np.clip(1.0 + 0.5 * rng.standard_normal(n), 0.01, None))
    inits.append(np.clip(0.2 + 1.5 * rng.random(n), 0.01, None))

    # Two-bump
    inits.append(np.exp(-30 * (x - 0.25) ** 2) + np.exp(-30 * (x - 0.75) ** 2) + 0.05)

    return [np.clip(v, 0.0, 1000.0) for v in inits]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses adaptive GS-IRFT + L-BFGS + local search with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = float(budget_s) - 25.0

    # Load parent if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(2048) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Choose lengths
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n in [8192, 16384, 32768]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))

    print(f"Parent: {'yes' if parent is not None else 'no'} (len={0 if parent is None else len(parent)})")
    print(f"Lengths: {lengths}")
    print(f"Effective budget: {effective_budget:.0f}s")

    # ================================================================
    # Phase 1: GS from diverse initializations (55% of time)
    # ================================================================
    phase1_end = t_start + 0.55 * effective_budget
    print(f"\n=== Phase 1: Adaptive GS-IRFT ===")

    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break
        print(f"\n  --- n={n} ---")
        inits = build_diverse_inits(n, rng)

        # Add parent if applicable
        if parent is not None:
            if len(parent) == n:
                inits.insert(0, parent.copy())
            else:
                x_old = np.linspace(0, 1, len(parent))
                x_new = np.linspace(0, 1, n)
                inits.insert(0, np.interp(x_new, x_old, parent))

        # Adapt number of iterations per init based on remaining time
        remaining_init_time = phase1_end - time.monotonic() - 30
        time_per_init = max(3.0, remaining_init_time / max(1, len(inits)))
        # Estimate: each GS iteration takes roughly n*log(n)/1e6 seconds
        est_time_per_iter = n * np.log2(n) / 5e7
        max_gs_iters = max(50, int(time_per_init / est_time_per_iter))

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end - 20:
                break

            # Run adaptive GS
            a_gs = gs_adaptive(a0, n_iters=max_gs_iters, sigma_start=0.3, sigma_end=0.85)
            sc = compute_score(a_gs)
            report(sc, a_gs, f"GS n={n} i={i}")

            # Also try rereflect (different dynamics)
            if time.monotonic() < phase1_end - 15:
                n_rr = max(30, max_gs_iters // 3)
                a_rr = gs_rereflect(a0, n_iters=n_rr)
                sc2 = compute_score(a_rr)
                report(sc2, a_rr, f"RR n={n} i={i}")

    print(f"\nAfter Phase 1: best = {best_score:.8f}")

    # ================================================================
    # Phase 2: L-BFGS refinement (25% of time)
    # ================================================================
    phase2_end = t_start + 0.80 * effective_budget
    print(f"\n=== Phase 2: L-BFGS Refinement ===")

    # Progressive increase in beta (sharper approximation of max)
    beta_schedule = [30, 80, 200, 500, 1000]
    for beta in beta_schedule:
        if time.monotonic() > phase2_end - 20:
            break
        remaining = phase2_end - time.monotonic() - 20
        if remaining < 20:
            break
        maxiter = min(4000, int(remaining * 15))
        if maxiter < 200:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 20)
        report(sc, a_opt, f"LBFGS b={beta}")

    # Try from alternative starting points (interpolated to other lengths)
    for n_alt in lengths[:2]:
        if time.monotonic() > phase2_end - 40:
            break
        if len(best_a) != n_alt:
            x_old = np.linspace(0, 1, len(best_a))
            x_new = np.linspace(0, 1, n_alt)
            a_start = np.interp(x_new, x_old, best_a)
        else:
            a_start = best_a.copy()

        # Quick GS to refine before L-BFGS
        a_start = gs_adaptive(a_start, n_iters=80, sigma_start=0.4, sigma_end=0.8)
        for beta in [100, 400]:
            if time.monotonic() > phase2_end - 15:
                break
            sc, a_opt, _ = run_lbfgs(a_start, beta, maxiter=2000, maxfun=40000)
            report(sc, a_opt, f"LBFGS alt n={n_alt} b={beta}")

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: Local search with targeted perturbations (12% of time)
    # ================================================================
    phase3_end = t_start + 0.92 * effective_budget
    print(f"\n=== Phase 3: Local Search ===")

    n_best = len(best_a)
    max_trials = 25

    for trial in range(max_trials):
        if time.monotonic() > phase3_end - 25:
            break
        strategy = trial % 7

        if strategy == 0:
            # Small random perturbation
            a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Peak reduction: reduce top contributors to the max lag
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            seg = a_pert[j_start:j_end + 1]
            contrib = seg[::-1]
            top_n = max(20, n_best // 40)
            top_idx = np.argsort(contrib)[::-1][:top_n]
            for li in top_idx:
                gi = j_start + li
                a_pert[gi] *= 0.85
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Sinusoidal modulation
            freq = 0.5 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # GS restart from perturbed point
            a_pert = best_a * (1.0 + 0.05 * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            a_pert = gs_adaptive(a_pert, n_iters=80, sigma_start=0.5, sigma_end=0.9)
        elif strategy == 4:
            # Block rescaling (exploits structure)
            a_pert = best_a.copy()
            n_blocks = 8 + int(rng.integers(0, 8))
            block_size = max(1, n_best // n_blocks)
            for b in range(min(n_blocks, n_best // block_size)):
                s = b * block_size
                e = min(s + block_size, n_best)
                a_pert[s:e] *= 0.6 + 0.8 * rng.random()
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 5:
            # Asymmetric scaling (push toward asymmetry)
            a_pert = best_a.copy()
            half = n_best // 2
            factor = 0.7 + 0.5 * rng.random()
            a_pert[half:] *= factor
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Multi-frequency perturbation
            a_pert = best_a.copy().astype(np.float64)
            x = np.arange(n_best, dtype=np.float64) / n_best
            for _ in range(3):
                f = 1.0 + 10.0 * rng.random()
                ph = 2.0 * np.pi * rng.random()
                am = 0.01 + 0.03 * rng.random()
                a_pert *= (1.0 + am * np.cos(2.0 * np.pi * f * x + ph))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Refine with L-BFGS
        remaining = phase3_end - time.monotonic() - 25
        if remaining > 10:
            maxiter = min(2000, int(remaining * 10))
            sc, a_opt, _ = run_lbfgs(a_pert, 150, maxiter=maxiter, maxfun=maxiter * 15)
            report(sc, a_opt, f"local #{trial} s={strategy}")
        else:
            sc = compute_score(a_pert)
            report(sc, a_pert, f"local #{trial} s={strategy}")

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # ================================================================
    # Phase 4: Final polish (8% of time)
    # ================================================================
    print(f"\n=== Phase 4: Final Polish ===")

    # High-beta L-BFGS (near-exact max optimization)
    if time.monotonic() < t_start + effective_budget - 30:
        sc, a_opt, _ = run_lbfgs(best_a, 2000, maxiter=3000, maxfun=50000)
        report(sc, a_opt, "final LBFGS b=2000")

    # Final GS pass with high sigma
    if time.monotonic() < t_start + effective_budget - 20:
        a_gs = gs_adaptive(best_a, n_iters=200, sigma_start=0.6, sigma_end=0.95)
        sc = compute_score(a_gs)
        report(sc, a_gs, "final GS")

    # One more L-BFGS from GS result
    if time.monotonic() < t_start + effective_budget - 15:
        sc, a_opt, _ = run_lbfgs(best_a, 500, maxiter=2000, maxfun=30000)
        report(sc, a_opt, "polish LBFGS")

    # ================================================================
    # Final verification and return
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
