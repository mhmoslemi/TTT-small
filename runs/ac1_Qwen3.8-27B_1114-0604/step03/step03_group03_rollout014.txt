"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines fast coordinate descent, subgradient-guided perturbation, and differential evolution
on block-parameterized sequences to escape local minima and find better solutions.
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def fast_coord_min(base_inside, alpha, beta_coef, outside_max, S_rest, n, a_i, t_upper):
    """Fast 1D minimization using closed-form candidates + bounded fallback."""
    candidates = [0.0, a_i, t_upper]
    
    # Closed-form optimal t for each affected lag
    ks = np.arange(len(base_inside))
    # t_k = (2*base[k] - alpha[k]*S_rest) / (2*beta[k]*S_rest - alpha[k])
    denom = 2.0 * beta_coef * S_rest - alpha
    denom_safe = np.where(np.abs(denom) > 1e-15, denom, np.where(denom >= 0, 1e-15, -1e-15))
    t_closed = (2.0 * base_inside - alpha * S_rest) / denom_safe
    # Filter valid candidates
    valid = (t_closed > 0) & (t_closed < t_upper)
    if np.any(valid):
        # Only add a few closest to current value to limit candidates
        valid_t = t_closed[valid]
        if len(valid_t) > 10:
            idx = np.argsort(np.abs(valid_t - a_i))[:10]
            candidates.extend(valid_t[idx].tolist())
        else:
            candidates.extend(valid_t.tolist())
    
    # Quick evaluation of all candidates
    best_t = a_i
    best_obj = coord_1d_obj(a_i, base_inside, alpha, beta_coef, outside_max, S_rest, n)
    for t_c in candidates:
        t_c = float(np.clip(t_c, 0.0, t_upper))
        obj = coord_1d_obj(t_c, base_inside, alpha, beta_coef, outside_max, S_rest, n)
        if obj < best_obj:
            best_obj = obj
            best_t = t_c
    
    # If the best candidate is near a boundary or we have few candidates, refine with scipy
    if best_obj > 1e-10:
        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(max(0, best_t * 0.8), min(t_upper, best_t * 1.2 + 0.01)),
            method='bounded',
            options={'xatol': 1e-8, 'maxiter': 50}
        )
        if res.fun < best_obj:
            best_t = float(res.x)
            best_obj = res.fun
    
    return best_t


def coordinate_descent_sweep(a, rng=None, n_coords=None, use_fast=True):
    """One full sweep of coordinate descent with incremental conv updates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    if n_coords is None:
        n_coords = n

    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)

    for idx in range(min(n_coords, n)):
        i = int(order[idx])
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()
        
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)

        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)

        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

def subgradient_perturb(a, rng, step_scale=0.01):
    """Perturb using the subgradient of the max convolution."""
    n = len(a)
    conv = fast_conv(a)
    k_star = int(np.argmax(conv))
    S = float(a.sum())
    
    # Subgradient direction (negative gradient = descent direction)
    # d(conv_k*)/d(a[i]) = 2*a[k*-i] if k*-i in [0,n-1], else 0
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    
    # Normalize
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-12:
        return a.copy()
    grad /= gnorm
    
    # Move in negative gradient direction, clip to non-negative
    step = step_scale * S / n
    a_new = a - step * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    
    # Add some random component for exploration
    noise = step_scale * 0.5 * rng.standard_normal(n) * S / n
    a_new += noise
    a_new = np.clip(a_new, 0.0, 1000.0)
    
    return a_new


def block_perturb(a, rng, n_blocks=50):
    """Perturb by adding correlated noise in blocks."""
    n = len(a)
    block_size = max(1, n // n_blocks)
    n_actual_blocks = (n + block_size - 1) // block_size
    a_new = a.copy()
    for b in range(n_actual_blocks):
        start = b * block_size
        end = min(n, start + block_size)
        delta = rng.normal(0, 0.02 * a[start:end].mean() + 0.001)
        a_new[start:end] += delta
    return np.clip(a_new, 0.0, 1000.0)


def peak_deflation_perturb(a, rng, strength=0.05):
    """Reduces elements that contribute most to the peak convolution."""
    n = len(a)
    conv = fast_conv(a)
    k_star = int(np.argmax(conv))
    
    # Find top contributing pairs
    contrib_products = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            contrib_products[i] = a[i] * a[j]
    
    # Reduce top contributors, increase non-contributors
    top_k = max(5, n // 50)
    top_idx = np.argsort(contrib_products)[::-1][:top_k]
    bottom_idx = np.argsort(contrib_products)[:n - top_k]
    
    a_new = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_new[j] * strength
        a_new[j] = max(a_new[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    
    if len(bottom_idx) > 0 and total_reduced > 0:
        # Distribute to bottom contributors with some randomness
        weights = 1.0 + 0.5 * rng.random(len(bottom_idx))
        weights /= weights.sum()
        a_new[bottom_idx] += total_reduced * weights
    
    return np.clip(a_new, 0.0, 1000.0)


def symmetric_init(n, rng, mode='random'):
    """Generate a symmetric random initialization."""
    half = n // 2
    if mode == 'random':
        a_half = rng.random(half + 1) * 0.5 + 0.1
    elif mode == 'smooth':
        x = np.linspace(0, 1, half + 1)
        a_half = 0.3 + 0.4 * np.sin(np.pi * x) + 0.1 * rng.standard_normal(half + 1)
        a_half = np.abs(a_half)
    elif mode == 'u_shape':
        x = np.linspace(0, 1, half + 1)
        a_half = 0.2 + 0.6 * (x**2) + 0.05 * rng.standard_normal(half + 1)
        a_half = np.abs(a_half)
    else:
        a_half = np.ones(half + 1) * 0.5
    
    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[-2::-1]])
    else:
        a = np.concatenate([a_half, a_half[-1::-1]])
    return np.clip(a, 0.01, 1000.0)


def block_param_eval(params, n_total, M):
    """Evaluate score from block parameters (piecewise constant)."""
    a = np.repeat(params, n_total // M + 1)[:n_total]
    if a.size < n_total:
        a = np.concatenate([a, np.zeros(n_total - a.size)])
    return compute_score(a)


def run_de_on_blocks(n_total, M, best_a, budget_s, rng):
    """Run differential evolution on block parameterization."""
    from scipy.optimize import differential_evolution
    
    # Initialize from current best
    block_vals = np.array([best_a[i*M:(i+1)*M].mean() for i in range(M)])
    block_vals = np.clip(block_vals, 0.01, 100.0)
    
    def objective(params):
        a = np.repeat(params, n_total // M + 1)[:n_total]
        if a.size < n_total:
            pad = n_total - a.size
            a = np.concatenate([a, np.full(pad, params[-1])])
        S = a.sum()
        if S < 0.01:
            return 1e10
        c = fast_conv(a)
        return 2.0 * n_total * float(c.max()) / (S * S)
    
    bounds = [(0.01, 100.0)] * M
    t0 = time.monotonic()
    
    # Use a smaller population for speed
    popsize = max(15, min(40, 4 + 3 * int(np.log10(M))))
    
    try:
        res = differential_evolution(
            objective, bounds,
            seed=rng.integers(2**31),
            maxiter=50,
            popsize=popsize,
            tol=1e-6,
            mutation=(0.3, 1.0),
            recombination=0.7,
            polish=False,
            init='sohal',
            x0=block_vals,
            updating='immediate',
            callback=None
        )
    except Exception as e:
        print(f"  DE failed: {e}")
        return None, None
    
    if res.fun < 1e10:
        a_de = np.repeat(res.x, n_total // M + 1)[:n_total]
        if a_de.size < n_total:
            pad = n_total - a_de.size
            a_de = np.concatenate([a_de, np.full(pad, res.x[-1])])
        a_de = np.clip(a_de, 0.0, 1000.0)
        score = compute_score(a_de)
        elapsed = time.monotonic() - t0
        print(f"  DE (M={M}): score={score:.8f}, time={elapsed:.1f}s")
        return score, a_de
    return None, None


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses coordinate descent, subgradient perturbation, and DE on blocks.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 20.0

    # Load parent
    best_a = None
    best_score = np.inf
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                p = np.clip(p, 0.0, 1000.0)
                best_a = p
                best_score = compute_score(p)
                print(f"Parent score: {best_score:.8f}, length: {len(best_a)}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_a is None:
        n_init = 4096
        best_a = np.ones(n_init) * 0.5
        best_score = compute_score(best_a)
        print(f"Random init score: {best_score:.8f}, length: {len(best_a)}")

    n_work = len(best_a)
    print(f"Working length: {n_work}")

    # ============================================================
    # Phase 1: CD refinement from parent (25% of time)
    # ============================================================
    phase1_deadline = t_start + 0.25 * total_budget
    print(f"\n=== Phase 1: CD refinement ({phase1_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    no_improve = 0
    sweep_count = 0

    while time.monotonic() < phase1_deadline:
        sweep_count += 1
        if sweep_count % 5 == 0:
            sc, a_cur = hot_coordinate_sweep(a_cur, n_sweeps=1)
        else:
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng, use_fast=True)
        
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_cur.copy()
            no_improve = 0
            if sweep_count <= 5 or sweep_count % 20 == 0:
                print(f"  P1 sweep {sweep_count}: {sc:.8f} (NEW BEST)")
        else:
            no_improve += 1

        if sweep_count % 25 == 0:
            print(f"  P1 sweep {sweep_count}: {sc:.8f} (no_improve={no_improve})")

        if no_improve >= 30:
            # Perturb to escape
            strategy = rng.integers(4)
            if strategy == 0:
                a_cur = subgradient_perturb(a_cur, rng, 0.005)
            elif strategy == 1:
                a_cur = peak_deflation_perturb(a_cur, rng, 0.03)
            elif strategy == 2:
                a_cur = block_perturb(a_cur, rng, 30)
            else:
                a_cur = a_cur * (1.0 + 0.01 * rng.standard_normal(n_work))
            a_cur = np.clip(a_cur, 0.0, 1000.0)
            no_improve = 0

    print(f"Phase 1 done: {sweep_count} sweeps, best={best_score:.8f}")

    # ============================================================
    # Phase 2: Diversified perturbation + CD (30% of time)
    # ============================================================
    phase2_deadline = t_start + 0.55 * total_budget
    print(f"\n=== Phase 2: Diversified search ({phase2_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    trial = 0
    while time.monotonic() < phase2_deadline:
        trial += 1
        strategy = trial % 7
        
        if strategy == 0:
            # Subgradient descent
            a_pert = subgradient_perturb(best_a, rng, 0.003 + 0.003 * rng.random())
        elif strategy == 1:
            # Peak deflation
            a_pert = peak_deflation_perturb(best_a, rng, 0.02 + 0.03 * rng.random())
        elif strategy == 2:
            # Block perturbation
            a_pert = block_perturb(best_a, rng, 20 + 30 * rng.integers(3))
        elif strategy == 3:
            # Segment swap
            seg_len = max(20, n_work // 40)
            s1 = int(rng.integers(0, n_work - seg_len))
            s2 = int(rng.integers(0, n_work - seg_len))
            a_pert = best_a.copy()
            tmp = a_pert[s1:s1+seg_len].copy()
            a_pert[s1:s1+seg_len] = a_pert[s2:s2+seg_len]
            a_pert[s2:s2+seg_len] = tmp
        elif strategy == 4:
            # Mirror/antisymmetric perturbation
            n_half = n_work // 2
            a_pert = best_a.copy()
            delta = 0.01 * rng.standard_normal(n_half)
            a_pert[:n_half] += delta
            a_pert[n_half:] -= delta[np.flip(np.arange(n_half) < n_work - n_half)]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 5:
            # Sinusoidal modulation
            freq = 1.0 + 5.0 * rng.random()
            amp = 0.005 + 0.01 * rng.random()
            x = np.arange(n_work) / n_work
            a_pert = best_a * (1.0 + amp * np.cos(2*np.pi*freq*x + 2*np.pi*rng.random()))
        else:
            # Gaussian perturbation
            scale = 0.005 + 0.01 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_work))
        
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        # Quick CD from perturbed point
        for sw in range(4):
            if time.monotonic() > phase2_deadline - 5:
                break
            if sw == 0:
                sc_p, a_pert = hot_coordinate_sweep(a_pert, n_sweeps=1)
            else:
                sc_p, a_pert = coordinate_descent_sweep(a_pert, rng=rng, use_fast=True)
        
        sc_final = compute_score(a_pert)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_pert.copy()
            print(f"  P2 trial {trial} (s={strategy}): {sc_final:.8f} (NEW BEST)")
        elif trial % 10 == 0:
            print(f"  P2 trial {trial} (s={strategy}): {sc_final:.8f}")

    print(f"Phase 2 done: {trial} trials, best={best_score:.8f}")

    # ============================================================
    # Phase 3: DE on blocks (20% of time)
    # ============================================================
    phase3_deadline = t_start + 0.75 * total_budget
    print(f"\n=== Phase 3: DE on Blocks ({phase3_deadline - t_start:.0f}s) ===")

    for M in [100, 200, 400]:
        if time.monotonic() > phase3_deadline - 30:
            break
        sc_de, a_de = run_de_on_blocks(n_work, M, best_a, 60, rng)
        if a_de is not None and sc_de < 1e10:
            # Refine with CD
            a_de_cur = a_de.copy()
            for sw in range(6):
                if time.monotonic() > phase3_deadline - 10:
                    break
                if sw == 0:
                    sc_de, a_de_cur = hot_coordinate_sweep(a_de_cur, n_sweeps=1)
                else:
                    sc_de, a_de_cur = coordinate_descent_sweep(a_de_cur, rng=rng, use_fast=True)
            if sc_de < best_score:
                best_score = sc_de
                best_a = a_de_cur.copy()
                print(f"  Phase3 M={M}: {sc_de:.8f} (NEW BEST)")

    # Also try different sequence lengths
    for n_try in [n_work - 200, n_work + 200, n_work + 500]:
        if time.monotonic() > phase3_deadline - 20:
            break
        if n_try < 100:
            continue
        # Resample best_a to new length
        from numpy interp
        x_old = np.linspace(0, 1, n_work)
        x_new = np.linspace(0, 1, n_try)
        a_resampled = np.interp(x_new, x_old, best_a)
        a_resampled = np.clip(a_resampled, 0.01, 1000.0)
        
        sc_rs, a_resampled = coordinate_descent_sweep(a_resampled, rng=rng, use_fast=True)
        sc_rs, a_resampled = coordinate_descent_sweep(a_resampled, rng=rng, use_fast=True)
        if sc_rs < best_score:
            best_score = sc_rs
            best_a = a_resampled.copy()
            n_work = n_try
            print(f"  Length {n_try}: {sc_rs:.8f} (NEW BEST)")

    print(f"Phase 3 done: best={best_score:.8f}, length={len(best_a)}")

    # ============================================================
    # Phase 4: LSE refinement (10% of time)
    # ============================================================
    phase4_deadline = t_start + 0.88 * total_budget
    print(f"\n=== Phase 4: LSE Refinement ({phase4_deadline - t_start:.0f}s) ===")

    for beta in [500, 2000, 5000]:
        if time.monotonic() > phase4_deadline - 10:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=3000, maxfun=50000)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  LSE beta={beta}: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            print(f"  LSE beta={beta}: {sc:.8f} ({elapsed:.1f}s)")

    # ============================================================
    # Phase 5: Final CD polish (5% of time)
    # ============================================================
    phase5_deadline = deadline
    print(f"\n=== Phase 5: Final Polish ({phase5_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    for sw in range(50):
        if time.monotonic() > phase5_deadline - 5:
            break
        if sw % 3 == 0:
            sc, a_cur = hot_coordinate_sweep(a_cur, n_sweeps=1)
        else:
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng, use_fast=True)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

    # One final LSE if time permits
    if time.monotonic() < deadline - 30:
        sc, a_opt, _ = run_lbfgs(best_a, 3000, maxiter=3000, maxfun=40000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            # Final CD
            sc, a_cur = coordinate_descent_sweep(a_opt.copy(), rng=rng, use_fast=True)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
            print(f"  Final LSE+CD: {best_score:.8f} (NEW BEST)")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
