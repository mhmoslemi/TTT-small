    scaled = np.clip(scaled, -500, 0)
    exps = np.exp(scaled)
    p = exps / exps.sum()
    L = c_max + np.log(exps.sum()) / beta
    
    # dL/da[j] = sum_k p[k] * (a[j+k] + a[j-k]) (with a[x]=0 outside [0,n-1])
    # = sum_k p[k]*a[j+k] + sum_k p[k]*a[j-k]
    # First term: cross-correlation of p and a
    # Second term: cross-correlation of a and p (or convolution)
    
    # Use FFT with sufficient zero-padding
    M = 1 << (3*n - 1).bit_length()
    P = np.fft.rfft(p, M)
    A = np.fft.rfft(a, M)
    
    # sum_k p[k] * a[j+k] = IFFT(conj(P) * A)[j] for j=0..n-1
    term1 = np.fft.irfft(np.conj(P) * A, M)[:n]
    # sum_k p[k] * a[j-k] = IFFT(A * P)[j] for j=0..n-1
    term2 = np.fft.irfft(A * P, M)[:n]
    
    dL = term1 + term2
    S2 = S*S
    S3 = S2*S
    grad = 2*n * (dL/S2 - 2*L/S3)
    return grad

"""
Step function optimization: LSE continuation + soft equalization + Nesterov subgradient.
"""
import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_func_grad(a, beta):
    """Returns (lse_score, gradient) for LSE-smoothed objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    scaled = np.minimum(scaled, 0.0)  # clip upper
    scaled = np.maximum(scaled, -500.0)  # clip lower for stability
    exps = np.exp(scaled)
    exp_sum = float(np.sum(exps))
    if exp_sum < 1e-300:
        exps = np.zeros(len(c))
        exps[int(np.argmax(c))] = 1.0
        exp_sum = 1.0
    p = exps / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    M2 = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, M2)
    A = np.fft.rfft(a, M2)
    term1 = np.fft.irfft(np.conj(P) * A, M2)[:n]
    term2 = np.fft.irfft(A * P, M2)[:n]
    dL = term1 + term2
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, deadline, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    remaining = deadline - time.monotonic()
    if remaining < 3.0:
        return compute_score(a0), a0
    
    options = {
        'ftol': 1e-16,
        'gtol': 1e-14,
        'maxiter': min(10000, int(remaining * 10)),
        'maxfun': min(maxfun, int(remaining * 50)),
        'maxls': 50
    }
    
    def obj(x):
        s, g = lse_func_grad(x, beta)
        return s, g
    
    try:
        res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out

def soft_equalize(a, alpha=0.3, n_iters=50, deadline=None):
    """Soft multiplicative equalization using softmax weights."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M2 = 1 << (3 * n - 1).bit_length()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break
        
        # Softmax weights (temperature-based)
        # Use a "soft" version: weight each lag by exp(-gamma * (c_max - c[k]))
        # Higher gamma = more focus on peak lags
        gamma = 50.0
        weights = np.exp(-gamma * (c_max - c) / max(c_max, 1e-15))
        weights = weights / weights.sum()
        
        # Compute "pressure" at each position: h[j] = sum_k weights[k] * a[j+k]
        # (similar to the LSE gradient but with fixed weights)
        P = np.fft.rfft(weights, M2)
        A = np.fft.rfft(a, M2)
        h = np.fft.irfft(np.conj(P) * A, M2)[:n]
        
        # Add the "other" term for completeness
        h += np.fft.irfft(A * P, M2)[:n]
        
        h = np.maximum(h, 1e-15)
        h_mean = float(h.mean())
        
        # Multiplicative update
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)
        
        # Renormalize
        S_new = float(a.sum())
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a = a * (S / S_new)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

def nesterov_subgrad(a, n_iters=1000, deadline=None):
    """Nesterov-accelerated subgradient with Polyak step and averaging."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    # Normalize to sum = sqrt(2n) for consistency
    if S > 1e-12:
        a = a * (np.sqrt(2.0 * n) / S)
        S = np.sqrt(2.0 * n)
    
    sc_lb = 2.0 * n / (2 * n - 1)  # theoretical lower bound
    best_a = a.copy()
    best_sc = compute_score(a)
    
    y = a.copy()  # Nesterov momentum variable
    v = np.zeros(n)  # velocity
    a_avg = a.copy()  # running average
    beta_mom = 0.9  # momentum
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute exact subgradient at y
        c = fast_conv(y)
        S_y = float(y.sum())
        if S_y < 1e-12:
            break
        M = float(c.max())
        sc_y = 2.0 * n * M / (S_y * S_y)
        
        # Subgradient: use all lags within 1e-10 of max
        threshold = M * (1 - 1e-10)
        active = c >= threshold
        if not active.any():
            active[int(np.argmax(c))] = True
        
        # Average subgradient over active lags
        sg = np.zeros(n)
        for k in range(2 * n - 1):
            if active[k]:
                # d(a*a)[k]/da[j] = a[j+k] + a[j-k] (with a[x]=0 outside)
                j_lo = max(0, k - n + 1)
                j_hi = min(n - 1, k)
                # a[j+k] + a[j-k] for j in [0, n-1]
                # a[j+k] is nonzero when j+k in [0, n-1], i.e., j in [max(0, -k), min(n-1, n-1-k)]
                # a[j-k] is nonzero when j-k in [0, n-1], i.e., j in [max(0, k), min(n-1, k+n-1)]
                # For the subgradient of the sum, we need:
                # d(sum_k active_k * c[k])/da[j] = sum_{k in active} (a[j+k] + a[j-k])
                # This is expensive to compute for many active lags.
                # Let's use FFT: define l = active.astype(float), then
                # sg[j] = sum_k l[k]*(a[j+k]+a[j-k]) = 2 * cross_correl(l, a)[j] (if l is symmetric)
                pass
        
        # This loop approach is too slow. Let me use FFT instead.
        l = active.astype(np.float64)
        l = l / l.sum()  # normalize (average subgradient)
        
        M2 = 1 << (3 * n - 1).bit_length()
        L = np.fft.rfft(l, M2)
        Y = np.fft.rfft(y, M2)
        sg = np.fft.irfft(np.conj(L) * Y, M2)[:n] + np.fft.irfft(Y * L, M2)[:n]
        
        # Full gradient of score = 2n * (M/S^2 - 2*M*1/S^3 * S) ... 
        # score = 2n*M/S^2, d(score)/da = 2n*(dM/da / S^2 - 2*M/S^3)
        # where dM/da is the subgradient of M = max(c)
        S2 = S_y * S_y
        S3 = S2 * S_y
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        
        # Polyak step size
        gap = sc_y - sc_lb
        if gap < 1e-12:
            break
        gn = float(np.linalg.norm(grad))
        if gn < 1e-30:
            break
        step = gap / (gn * gn + 1e-30)
        step = min(step, 0.1)  # cap step size
        
        # Nesterov update
        v = beta_mom * v - step * grad
        x_new = np.clip(y + v, 0.0, 1000.0)
        
        # Renormalize
        S_new = float(x_new.sum())
        if S_new < 1e-12:
            x_new = y  # revert
            S_new = S_y
        else:
            x_new = x_new * (S_y / S_new)
        
        a_avg = a_avg + (x_new - a_avg) / (it + 1)
        y = x_new
        
        # Check improvement
        sc_check = compute_score(y)
        if sc_check < best_sc:
            best_sc = sc_check
            best_a = y.copy()
        
        # Also check the average
        if it % 10 == 0:
            sc_avg = compute_score(a_avg)
            if sc_avg < best_sc:
                best_sc = sc_avg
                best_a = a_avg.copy()
    
    return best_sc, best_a
