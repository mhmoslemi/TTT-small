def lp_update(f, M, n):
    """Find b >= 0 maximizing sum(b) subject to (f*b)_k <= M for all k."""
    # This is an LP with n variables and 2n-1 constraints.
    # (f*b)_k = sum_j f[j] * b[k-j] <= M for k = 0, ..., 2n-2
    # b >= 0
    # Maximize sum(b)
    
    # For scipy, we need: c^T x subject to A x <= b, x >= 0
    # c = -ones(n) (minimize -sum(b))
    # A: (2n-1) x n matrix where A[k, j] = f[j] if 0 <= k-j < n, else 0
    # b: M * ones(2n-1)
    
    from scipy.optimize import linprog
    
    c = -np.ones(n)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            if 0 <= k - j < n:
                A[k, j] = f[j]
    b = M * np.ones(2*n-1)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    return res.x

"""
Step function optimization: LP-based peak equalization with LSE continuation,
subgradient polishing, and structured restarts.
Key innovation: uses linear programming to find the best direction that respects
the convolution constraint, combined with LSE for smooth optimization.
Target: beat 1.5027 (previous best: 1.5093).

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for max-conv objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    # Gradient via FFT
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(a, beta):
    return lse_score_grad(a, beta)

    top_idx = np.argsort(c_ff)[::-1][:K]
    top_idx = np.sort(top_idx)  # keep sorted for structure
    
    # Build sparse A matrix: A[i, j] = f[k_i - j] if 0 <= k_i - j < n
    A = lil_matrix((K, n))
    for i, k in enumerate(top_idx):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            # Coefficient of b[j] is f[k-j] for j in [j_start, j_end]
            # k-j ranges from k-j_end to k-j_start
            coeffs = f[k - j_end:k - j_start + 1][::-1].copy()
            A[i, j_start:j_end + 1] = coeffs
    A = A.tocsr()
    
    c_obj = -np.ones(n)
    b_ub = M * np.ones(K)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.clip(res.x, 0.0, None)
    except Exception:
        pass
    return None


def subgradient_step(a, step_type='polyak'):
    """One subgradient step on the max-conv objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    M = float(c.max())
    k_star = int(np.argmax(c))
    sc = 2.0 * n * M / (S * S)
    
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    sg = np.zeros(n)
    if j_end >= j_start:
        sg[j_start:j_end + 1] = 2.0 * a[j_end:j_start - 1:-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
    gn = float(np.linalg.norm(grad))
    if gn < 1e-15:
        return a
    
    if step_type == 'polyak':
        sc_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
        step = (sc - sc_lb) / (gn * gn + 1e-20)
        step = min(step, 0.5 / S)
        step = max(step, 1e-8 / (gn * S + 1e-20))
    else:
        step = 0.1 / (gn * S + 1e-20)
    
    a_new = a - step * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new


def peak_shave(a, iters=2000, verbose=False):
    """Subgradient descent with adaptive step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    score_history = []
    
    for it in range(iters):
        a = subgradient_step(a, 'polyak')
        sc = 2.0 * n * float(fast_conv(a).max()) / (float(a.sum()) ** 2)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if it % 500 == 0 and verbose:
            print(f"    subgrad iter {it}: score={sc:.8f}")
    
    return best_sc, best_a


def ideal_sequence(n):
    """Generate the 'ideal' sequence: p_k = C(2k,k)/4^k normalized to sum=sqrt(2n)."""
    k = np.arange(n, dtype=np.float64)
    # C(2k,k) / 4^k computed via log-gamma for numerical stability
    from math import lgamma
    log_c = np.array([lgamma(2 * int(ki) + 1) - 2 * lgamma(int(ki) + 1) - int(ki) * np.log(4)
                      for ki in range(n)])
    c = np.exp(log_c)
    c = c / c.sum() * np.sqrt(2.0 * n)
    return c


def symmetric_ideal_sequence(n):
    """Mirror the ideal sequence around the center."""
    c = ideal_sequence(n)
    # Symmetrize: take the "envelope"
    c_sym = (c + c[::-1]) / 2.0
    c_sym = c_sym / c_sym.sum() * np.sqrt(2.0 * n)
    return c_sym


def cosine_window_sequence(n, alpha=1.0):
    """Cosine window: a_k = cos^alpha(pi*k/(2n))."""
    k = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * k / (2.0 * n)) ** alpha
    a = np.clip(a, 0, None)
    a = a / a.sum() * np.sqrt(2.0 * n)
    return a
