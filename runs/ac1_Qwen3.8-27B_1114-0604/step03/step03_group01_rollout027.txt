"""
Spectral-informed optimization for minimum peak self-convolution sequences.
Uses top-K soft-max penalty, spectral initialization, and projected gradient
with adaptive step sizes to minimize 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def topk_penalty(a, K, beta):
    """
    Smooth approximation to top-K sum of convolution values.
    Uses soft-top-K via temperature-scaled softmax.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    # sum_k c_k * (softmax(beta * c))_k truncated

    # This approximates the top-K sum when beta is large

    # soft_topK_sum ≈ sum_k c_k * min(1, K * p_k / sum(p_k[:K]))
    # For simplicity, use: K * weighted_mean where weight = softmax(beta*c)^2

    # K * max_k (c_k - tau) + K*tau = K*max_k c_k (which is just K*max)

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    Returns (score, grad, conv, p_weights, L_val, S).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n), c, None, 0.0, S
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    # Compute dL/da: sum_k p_k * 2 * a_{k-j} for valid k-j
    # = 2 * cross_correlation(p, a)[j]
    L2 = 1 << (3 * n - 2 - 1).bit_length()
    a_flip = a[::-1]
    AF = np.fft.rfft(a_flip, L2)
    PF = np.fft.rfft(p, L2)
    conv_result = np.fft.irfft(AF * PF, L2)
    r = conv_result[n - 1:n - 1 + n]

    dL_da = 2.0 * r
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad, c, p, L, S


def topk_lse_grad(a, K, beta):
    """
    Top-K soft penalty: smooth approximation to mean of top-K convolution values.
    Penalizes a cluster of high peaks, encouraging flattening.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    # Soft top-K: use softmax with temperature 1/beta, then take weighted sum
    # weighted_sum = sum_k c_k * softmax(beta * (c_k - c_max))_k
    # This is a smooth interpolation: beta->0 gives mean(c), beta->inf gives max(c)
    # To get "top-K mean" behavior, use a two-level approach:
    # Level 1: weight = softmax((c_k - c_K) * beta) where c_K is K-th largest
    # Level 2: topK_mean = sum(c_k * weight_k) / sum(weight_k)
    
    # Find approximate K-th largest
    sort_idx = np.argsort(c)[::-1]
    c_K = float(c[sort_idx[min(K - 1, len(c) - 1)]])
    
    scaled = beta * (c - c_K)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    
    # Smooth top-K mean
    L = float(np.dot(c, p))
    score = 2.0 * n * L / (S * S)
    
    # Gradient: dL/da_j = sum_k p_k * 2 * a_{k-j} (same as before)
    L2 = 1 << (3 * n - 2 - 1).bit_length()
    a_flip = a[::-1]
    AF = np.fft.rfft(a_flip, L2)
    PF = np.fft.rfft(p, L2)
    conv_result = np.fft.irfft(AF * PF, L2)
    r = conv_result[n - 1:n - 1 + n]
    
    dL_da = 2.0 * r
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def project_onto_simplex_nonneg(a):
    """Project onto {a >= 0, sum(a) = sum(a_original)} preserving sum."""
    S = float(a.sum())
    if S < 1e-12:
        return a.copy()
    a = np.clip(a, 0.0, None)
    new_S = float(a.sum())
    if new_S > 1e-12:
        a = a * (S / new_S)
    return a


def projected_gradient_descent(a_init, betas, K_schedule, maxiter_per_phase=5000,
                                time_limit_s=None, t_start=None, verbose=True):
    """
    Continuation projected gradient with Armijo backtracking line search.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S_target = float(a.sum())
    a = a / S_target  # Normalize to sum=1
    S_target = 1.0
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for phase, (beta, K) in enumerate(zip(betas, K_schedule)):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        lr = 0.01 / (1.0 + 0.1 * phase)
        c_alpha = 1e-4  # Armijo condition
        rho = 0.5  # Step size reduction
        
        for it in range(maxiter_per_phase):
            if time_limit_s is not None and time.monotonic() > t_start + time_limit_s:
                break
            
            # Get objective and gradient
            if K > 1:
                f, g = topk_lse_grad(a, K, beta)
            else:
                f, g, _, _, _, _ = lse_grad(a, beta)
            
            if not np.isfinite(f):
                break
            
            # Armijo backtracking line search
            step = lr
            improved = False
            for _ in range(20):
                a_new = a - step * g
                a_new = project_onto_simplex_nonneg(a_new)
                
                if K > 1:
                    f_new, _ = topk_lse_grad(a_new, K, beta)
                else:
                    f_new, _, _, _, _, _ = lse_grad(a_new, beta)
                
                if f_new <= f - c_alpha * step * np.dot(g, a - a_new):
                    a = a_new
                    improved = True
                    break
                step *= rho
            
            if not improved:
                break
            
            # Track best with exact score
            if it % 50 == 0:
                exact_sc = compute_score(a)
                if exact_sc < best_score:
                    best_score = exact_sc
                    best_a = a.copy()
                    if verbose:
                        print(f"      PGD phase {phase} (beta={beta},K={K}) it={it}: {exact_sc:.8f}")
        
        # Final check for this phase
        exact_sc = compute_score(a)
        if exact_sc < best_score:
            best_score = exact_sc
            best_a = a.copy()
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    
    return best_score, best_a


def subgrad_exact(a):
    """Exact score and subgradient at the peak lag."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c_max = float(c.max())
    k_star = int(np.argmax(c))
    score = 2.0 * n * c_max / (S * S)
    
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B with bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1000.0)
        sc, g, _, _, _, _ = lse_grad(x, beta)
        if not np.isfinite(sc):
            return 1e10, np.zeros(n)
        return sc, g
    
    res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter,
                            'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coord_descent_smart(a, n_sweeps=10, top_k=50, time_limit_s=None, t_start=None):
    """
    Coordinate descent selecting coordinates by convolution sensitivity.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = np.inf
    if time_limit_s is not None and t_start is not None:
        deadline = t_start + time_limit_s
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        # Compute sensitivity: how much each a_j contributes to the peak
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        sensitivity = np.zeros(n)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sensitivity[js:je + 1] = 2.0 * a[idx]
        
        # Also consider second peak
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) > 1 and c_sorted[1] > 0.8 * c_sorted[0]:
            # Find second peak lag
            c_masked = c.copy()
            c_masked[k_star] = -np.inf
            k_star2 = int(np.argmax(c_masked))
            js2 = max(0, k_star2 - n + 1)
            je2 = min(k_star2, n - 1)
            if je2 >= js2:
                idx2 = k_star2 - np.arange(js2, je2 + 1)
                sensitivity[js2:je2 + 1] += 2.0 * a[idx2]
        
        # Select top-K coordinates by sensitivity * |a_j|
        importance = sensitivity * np.abs(a) + 1e-10 * np.abs(a)
        order = np.argsort(importance)[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            
            a_old = a[j]
            if a_old < 1e-10:
                continue
            
            score = compute_score(a)
            
            # Try different scales
            if sensitivity[j] > 0:
                # Reducing a[j] helps
                ratios = np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.0, 1.1])
            else:
                ratios = np.array([0.5, 0.8, 1.0, 1.2, 1.5, 2.0])
            
            candidates = a_old * ratios
            candidates = np.clip(candidates, 0.0, 1000.0)
            
            best_x = a_old
            best_sc = score
            for x in candidates:
                if x == a_old:
                    continue
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
        
        final_sc = compute_score(a)
        if final_sc < best_score:
            best_score = final_sc
    
    return best_score, a


def spectral_restart(a, rng, perturbation_scale=0.1):
    """
    Restart by modifying the spectrum of the sequence.
    Add random phase perturbations to the DFT, then reconstruct.
    """
    n = len(a)
    A = np.fft.rfft(a)
    
    # Add random phase perturbation (preserves magnitudes approximately)
    phase_noise = rng.standard_normal(len(A)) * perturbation_scale
    # Only perturb the phase, keep magnitude roughly the same
    A_magnitude = np.abs(A)
    A_phase = np.angle(A) + phase_noise
    A_new = A_magnitude * np.exp(1j * A_phase)
    
    a_new = np.fft.irfft(A_new, n)
    a_new = np.clip(a_new, 0.0, None)
    
    # Ensure non-negativity and normalize
    if a_new.sum() < 1e-12:
        a_new = a.copy()
    else:
        a_new = a_new * (float(a.sum()) / float(a_new.sum()))
    
    return a_new


def temporal_perturb_restart(a, rng, scale=0.05):
    """Add structured temporal perturbation."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    freq1 = 1.0 + 3.0 * rng.random()
    freq2 = 4.0 + 8.0 * rng.random()
    phase1 = 2.0 * np.pi * rng.random()
    phase2 = 2.0 * np.pi * rng.random()
    mod = 1.0 + scale * np.cos(2 * np.pi * freq1 * x + phase1) \
              + 0.5 * scale * np.cos(2 * np.pi * freq2 * x + phase2)
    a_new = a * mod
    a_new = np.clip(a_new, 0.0, None)
    if a_new.sum() > 1e-12:
        a_new = a_new * (float(a.sum()) / float(a_new.sum()))
    return a_new


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search: PGD continuation + L-BFGS + coord descent + spectral restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 20.0
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    if parent is not None:
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
    else:
        n_init = 4096
        parent = np.ones(n_init)
        parent_score = compute_score(parent)
        print(f"Generated uniform parent: n={n_init}, score={parent_score:.8f}")
    
    best_score = parent_score
    best_a = parent.copy()
    
    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | t={elapsed:.0f}s")
            return True
        return False
    
    # ==================================================================
    # Phase 1: Projected Gradient with Continuation (40% budget)
    # ==================================================================
    p1_time = effective_budget * 0.40
    print(f"\n{'='*50}")
    print(f"Phase 1: PGD Continuation (budget={p1_time:.0f}s)")
    print(f"{'='*50}")
    
    a_work = best_a / float(best_a.sum())  # Normalize to sum=1
    n = len(a_work)
    
    # Continuation schedule: (beta, K) pairs
    # Start smooth and broad, end sharp and focused
    n_phases = 8
    betas = np.geomspace(5, 5000, n_phases)
    K_schedule = np.linspace(max(2, int(n * 0.5)), 1, n_phases, dtype=int)
    K_schedule = np.clip(K_schedule, 1, n)
    
    t1_deadline = t_start + p1_time
    time_per_phase = p1_time / n_phases
    
    for i in range(n_phases):
        phase_budget = min(time_per_phase, t1_deadline - time.monotonic())
        if phase_budget < 2:
            break
        sc, a_opt = projected_gradient_descent(
            a_work, 
            betas=[betas[i]], 
            K_schedule=[K_schedule[i]],
            maxiter_per_phase=100000,
            time_limit_s=phase_budget,
            t_start=time.monotonic(),
            verbose=False
        )
        try_update(sc, a_opt, f"PGD b={betas[i]:.0f} K={K_schedule[i]}")
        a_work = a_opt
    
    # ==================================================================
    # Phase 2: L-BFGS Polish (25% budget)
    # ==================================================================
    p2_time = effective_budget * 0.25
    print(f"\n{'='*50}")
    print(f"Phase 2: L-BFGS Polish (budget={p2_time:.0f}s)")
    print(f"{'='*50}")
    
    p2_deadline = t_start + p1_time + p2_time
    for beta in [200, 1000, 5000, 20000]:
        if time.monotonic() > p2_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=8000, maxfun=150000)
        try_update(sc, a_opt, f"LBFGS beta={beta}")
        print(f"    LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
        if time.monotonic() > p2_deadline:
            break
    
    # ==================================================================
    # Phase 3: Smart Coordinate Descent (15% budget)
    # ==================================================================
    p3_time = effective_budget * 0.15
    print(f"\n{'='*50}")
    print(f"Phase 3: Coordinate Descent (budget={p3_time:.0f}s)")
    print(f"{'='*50}")
    
    t0 = time.monotonic()
    sc, a_opt = coord_descent_smart(best_a, n_sweeps=50, top_k=60,
                                     time_limit_s=p3_time, t_start=t_start)
    try_update(sc, a_opt, "CoordDescent")
    print(f"    CoordDescent: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # ==================================================================
    # Phase 4: Spectral & Temporal Restarts (15% budget)
    # ==================================================================
    p4_time = effective_budget * 0.15
    print(f"\n{'='*50}")
    print(f"Phase 4: Restarts (budget={p4_time:.0f}s)")
    print(f"{'='*50}")
    
    p4_deadline = t_start + effective_budget * 0.80
    n_restarts = 16
    for r in range(n_restarts):
        if time.monotonic() > p4_deadline:
            break
        
        time_left = p4_deadline - time.monotonic()
        if time_left < 15:
            break
        
        strategy = r % 4
        if strategy == 0:
            a_pert = spectral_restart(best_a, rng, perturbation_scale=0.05 + 0.1 * rng.random())
        elif strategy == 1:
            a_pert = temporal_perturb_restart(best_a, rng, scale=0.03 + 0.05 * rng.random())
        elif strategy == 2:
            # Peak-targeted reduction
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            js = max(0, k_star - n + 1)
            je = min(k_star, n - 1)
            a_pert = best_a.copy()
            for j in range(js, je + 1):
                a_pert[j] *= 0.7
            removed = float(best_a[js:je+1].sum() - a_pert[js:je+1].sum())
            mask = (np.arange(n) < js) | (np.arange(n) > je)
            if mask.sum() > 0:
                a_pert[mask] += removed / mask.sum()
            a_pert = np.clip(a_pert, 0.0, None)
        else:
            # Gaussian noise
            scale = 0.02 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n))
            a_pert = np.clip(a_pert, 0.0, None)
        
        if a_pert.sum() < 1e-12:
            continue
        a_pert = a_pert / float(a_pert.sum())  # re-normalize
        
        # Quick refinement: short PGD + L-BFGS
        adam_time = min(15.0, time_left * 0.4)
        sc_r, a_r = projected_gradient_descent(
            a_pert, betas=[500], K_schedule=[1],
            maxiter_per_phase=10000,
            time_limit_s=adam_time, t_start=time.monotonic(), verbose=False
        )
        if time.monotonic() < p4_deadline - 5:
            sc2, a_r2 = run_lbfgs(a_r, 2000, maxiter=3000, maxfun=50000)
            if sc2 < sc_r:
                sc_r, a_r = sc2, a_r2
        try_update(sc_r, a_r, f"restart#{r} s{strategy}")
    
    # ==================================================================
    # Phase 5: Try different lengths (10% budget)
    # ==================================================================
    p5_time = effective_budget * 0.10
    print(f"\n{'='*50}")
    print(f"Phase 5: Length variations (budget={p5_time:.0f}s)")
    print(f"{'='*50}")
    
    p5_deadline = t_start + effective_budget - 10.0
    n_best = len(best_a)
    # Try a few different lengths
    target_lengths = []
    for factor in [0.7, 0.85, 1.15, 1.3, 1.5]:
        tl = int(n_best * factor)
        if 512 <= tl <= 30000 and tl != n_best:
            target_lengths.append(tl)
    
    for tl in target_lengths[:3]:
        if time.monotonic() > p5_deadline:
            break
        # Resample
        x_old = np.arange(n_best, dtype=np.float64)
        x_new = np.linspace(0, n_best - 1, tl)
        a_res = np.interp(x_new, x_old, best_a)
        a_res = np.clip(a_res, 0.0, None)
        if a_res.sum() < 1e-12:
            continue
        a_res = a_res / float(a_res.sum())
        
        # Optimize
        time_left = p5_deadline - time.monotonic()
        if time_left < 10:
            continue
        sc_r, a_r = projected_gradient_descent(
            a_res, betas=[200, 2000], K_schedule=[1, 1],
            maxiter_per_phase=10000,
            time_limit_s=time_left * 0.5, t_start=time.monotonic(), verbose=False
        )
        if time.monotonic() < p5_deadline - 3:
            sc2, a_r2 = run_lbfgs(a_r, 3000, maxiter=3000, maxfun=50000)
            if sc2 < sc_r:
                sc_r, a_r = sc2, a_r2
        try_update(sc_r, a_r, f"n={tl}")
    
    # ==================================================================
    # Phase 6: Final intensive polish
    # ==================================================================
    print(f"\n{'='*50}")
    print("Phase 6: Final Polish")
    print(f"{'='*50}")
    
    final_deadline = t_start + effective_budget - 3.0
    
    # High-beta L-BFGS
    for beta in [10000, 50000]:
        if time.monotonic() > final_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=10000, maxfun=200000)
        try_update(sc, a_opt, f"final LBFGS beta={beta}")
        print(f"    Final LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # Final coordinate descent
    if time.monotonic() < final_deadline:
        t0 = time.monotonic()
        sc, a_opt = coord_descent_smart(best_a, n_sweeps=20, top_k=80,
                                         time_limit_s=min(30.0, final_deadline - time.monotonic()),
                                         t_start=t_start)
        try_update(sc, a_opt, "final coord")
        print(f"    Final coord: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
