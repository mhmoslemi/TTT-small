    """Solve: max sum(b) s.t. b >= 0, conv(a,b)_k <= M for all k."""
    n = len(a)
    # Build sparse constraint matrix
    # A[k, j] = a[j] if 0 <= k-j <= n-1
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: minimize -sum(b)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    
    if res.status == 0:
        return res.x
    return None

    M_fft = 1 << (M - 1).bit_length()

        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def active_set_grad(a, tol_frac=0.001):
    """
    Compute score and subgradient using active set.
    Returns (score, grad, M, active_mask).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n), 0.0, np.zeros(2*n-1, dtype=bool)
    M = float(c.max())
    score = 2.0 * n * M / (S * S)
    
    # Active set: coefficients within tol_frac of max
    threshold = M * (1.0 - tol_frac)
    active = c >= threshold
    
    # Compute averaged subgradient using FFT
    b = active.astype(np.float64)
    a_flipped = a[::-1]
    L = 1 << (3 * n - 2 - 1).bit_length()
    A_fft = np.fft.rfft(a_flipped, L)
    B_fft = np.fft.rfft(b, L)
    conv_result = np.fft.irfft(A_fft * B_fft, L)
    g = conv_result[n-1:2*n-1].copy()
    
    n_active = max(1, int(active.sum()))
    g /= n_active
    
    # Full gradient of the score
    S2 = S * S
    S3 = S2 * S
    full_grad = 2.0 * n * (g / S2 - 2.0 * M / S3)
    
    return score, full_grad, M, active


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient via FFT
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, func, args=(), maxiter=2000, maxfun=50000):
    """Run L-BFGS-B."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def lp_direction(a, M):
    """
    Solve LP: max sum(b) s.t. b >= 0, conv(a,b)_k <= M.
    Returns b or None if LP fails.
    """
    n = len(a)
    # Build sparse constraint matrix
    # A[k, j] = a[j] if 0 <= k-j <= n-1, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            rows.extend(range(k, k + 1) * (j_end - j_start + 1))
            cols.extend(range(j_start, j_end + 1))
            vals.extend(a[j_start:j_end+1].tolist())
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.status == 0:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def nesterov_subgrad(a, n_iters=5000, tol_frac=0.001, momentum=0.9):
    """
    Nesterov-accelerated subgradient descent on max-conv objective.
    Uses active set for subgradient computation.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    v = np.zeros(n)  # momentum velocity
    gamma = 1.0  # Polyak-like step size
    
    for it in range(n_iters):
        # Nesterov lookahead
        y = a + momentum * v
        
        sc, grad, M, active = active_set_grad(y, tol_frac)
        if np.isinf(sc):
            break
            
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        
        # Polyak step size
        sc_lb = 1.0  # optimistic lower bound
        step = (sc - sc_lb) / (gn * gn + 1e-20)
        step = min(step, 1.0 / float(a.sum()) + 1e-10)
        step = max(step, 1e-8 / (gn + 1e-20))
        
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        a = a_new
        
        if it % 500 == 0:
            print(f"    nesterov it={it}, score={sc:.8f}, best={best_sc:.8f}")
    
    return best_sc, best_a


def plain_subgrad(a, n_iters=5000, tol_frac=0.001):
    """Plain subgradient descent with Polyak step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        sc, grad, M, active = active_set_grad(a, tol_frac)
        if np.isinf(sc):
            break
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        sc_lb = 1.0
        step = (sc - sc_lb) / (gn * gn + 1e-20)
        step = min(step, 1.0 / (float(a.sum()) + 1e-10))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        
        sc_new = compute_score(a)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a.copy()
        
        if it % 1000 == 0:
            print(f"    subgrad it={it}, score={sc:.8f}, best={best_sc:.8f}")
    
    return best_sc, best_a

Step function optimization using LP direction finding, Nesterov-accelerated

    else:
        p = exp_scaled / exp_sum
    L_val = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L_val / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L_val / S3)
    return score, grad

    return lse_score_grad(a, beta)


def run_lbfgs(a0, func, args=(), maxiter=2000, maxfun=50000):

        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def active_set_subgrad_step(a, tol_frac=0.001):
    """
    Compute score and active-set subgradient.
    Returns (score, grad, M, active_count).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n), 0.0, 0
    M = float(c.max())
    score = 2.0 * n * M / (S * S)
    
    threshold = M * (1.0 - tol_frac)
    active = c >= threshold
    n_active = max(1, int(active.sum()))
    
    # Compute averaged subgradient via FFT
    b = active.astype(np.float64)
    a_flipped = a[::-1]
    L = 1 << (3 * n - 2 - 1).bit_length()
    A_fft = np.fft.rfft(a_flipped, L)
    B_fft = np.fft.rfft(b, L)
    conv_result = np.fft.irfft(A_fft * B_fft, L)
    g = conv_result[n-1:2*n-1].astype(np.float64)
    g /= n_active
    
    S2 = S * S
    S3 = S2 * S
    full_grad = 2.0 * n * (g / S2 - 2.0 * M / S3)
    
    return score, full_grad, M, n_active


def nesterov_subgrad(a, n_iters, tol_frac=0.001, momentum=0.9, verbose=False):
    """Nesterov-accelerated subgradient with active set."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
    
    v = np.zeros(n)
    for it in range(n_iters):
        y = a + momentum * v
        y = np.clip(y, 0.0, 1000.0)
        
        sc, grad, M, n_act = active_set_subgrad_step(y, tol_frac)
        if np.isinf(sc):
            break
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        
        # Polyak step
        step = (sc - sc_lb) / (gn * gn + 1e-20)
        step = min(step, 2.0 / (float(y.sum()) + 1e-10))
        step = max(step, 1e-10)
        
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 200 == 0:
            print(f"    nesterov it={it}, score={sc:.8f}, best={best_sc:.8f}, n_act={n_act}, step={step:.2e}")
    
    return best_sc, best_a


def plain_subgrad(a, n_iters, tol_frac=0.001, verbose=False):

def build_lp_matrix(n):
    """Build sparse structure for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            rows.extend([k] * (j_end - j_start + 1))
            cols.extend(range(j_start, j_end + 1))
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def lp_step(a, M, row_idx, col_idx):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)_k <= M."""
    n = len(a)
    # Extract values in the sparse pattern
    # For each entry (k, j) in the pattern, the value is a[j]
    vals = a[col_idx].astype(np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                      options={'presolve': True, 'time_limit': 120})
        if res.status == 0:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None

active-set subgradient descent, and LSE (log-sum-exp) continuation refinement.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for the max-conv objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L_val = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L_val / (S * S)

    vals = a[col_idx].astype(np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))

    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'presolve': True,
                      'time_limit': time_limit})
        if res.status == 0:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def targeted_perturb(a, rng, strength=0.03):
    """Reduce elements contributing most to the peak, redistribute to low-contribution ones."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end+1] = a[j_end:j_start-1:-1] if j_start > 0 else a[j_end::-1]

    top_count = max(10, n // 30)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:max(10, n - top_count)]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: LP direction finding -> Nesterov subgradient -> LSE refinement -> polish.
    Returns best non-negative sequence found within time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    # Load parent construction
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        sc = compute_score(parent)
        report(sc, parent, "parent")
        print(f"Parent: score={sc:.8f}, n={len(parent)}")

    n = len(best_a)
    print(f"\nOptimizing with n = {n}, initial score = {best_score:.8f}")

    # Pre-build LP matrix structure
    lp_rows = lp_cols = None
    if n <= 15000:
        print("Building LP matrix structure...")
        t0 = time.monotonic()
        lp_rows, lp_cols = build_lp_structure(n)
        print(f"  LP structure: {len(lp_rows)} nonzeros, built in {time.monotonic()-t0:.1f}s")

    # ===================================================================
    # Phase 1: LP-based direction finding (18% time)
    # ===================================================================
    phase1_end = t_start + 0.18 * (total_budget - 30.0)
    print("\n=== Phase 1: LP Direction Finding ===")

    if lp_rows is not None and time.monotonic() < phase1_end:
        a_cur = best_a.copy()
        S_cur = float(a_cur.sum())
        M_cur = float(fast_conv(a_cur).max())

        lp_iter = 0
        while time.monotonic() < phase1_end and lp_iter < 20:
            t0 = time.monotonic()
            remaining = phase1_end - time.monotonic()
            tlimit = min(90.0, max(10.0, remaining * 0.4))
            b = lp_step(a_cur, M_cur, lp_rows, lp_cols, time_limit=tlimit)
            if b is None or b.sum() < 1e-10:
                print(f"  LP failed/stopped at iter {lp_iter}")
                break

            S_b = float(b.sum())
            if S_b < 1e-10:
                break
            b_scaled = b * S_cur / S_b

            improved = False
            for t_blend in [0.5, 0.3, 0.2, 0.1, 0.05, 0.8]:
                a_try = (1.0 - t_blend) * a_cur + t_blend * b_scaled
                a_try = np.clip(a_try, 0.0, 1000.0)
                sc_try = compute_score(a_try)
                sc_cur = compute_score(a_cur)
                if sc_try < sc_cur:
                    a_cur = a_try
                    report(sc_try, a_cur, f"LP iter {lp_iter} blend={t_blend}")
                    S_cur = float(a_cur.sum())
                    M_cur = float(fast_conv(a_cur).max())
                    improved = True
                    break

            if not improved:
                print(f"  LP iter {lp_iter}: no improvement, stopping")
                break

            lp_iter += 1
            print(f"  LP iter {lp_iter}: score={compute_score(a_cur):.8f}, "
                  f"time={time.monotonic()-t0:.1f}s")

    # ===================================================================
    # Phase 2: Nesterov active-set subgradient (30% time)
    # ===================================================================
    phase2_end = t_start + 0.48 * (total_budget - 30.0)
    print("\n=== Phase 2: Nesterov Subgradient (tol=0.001) ===")

    a_cur = best_a.copy()
    remaining = max(1.0, phase2_end - time.monotonic())
    est_iters = int(remaining * 250)
    est_iters = max(2000, min(est_iters, 30000))
    sc, a_nest = nesterov_subgrad(a_cur, n_iters=est_iters,
                                   tol_frac=0.001, momentum=0.9, verbose=True)
    report(sc, a_nest, "nesterov A")

    # Second round with tighter tolerance
    if time.monotonic() < phase2_end:
        a_cur = best_a.copy()
        remaining = max(1.0, phase2_end - time.monotonic())
        est_iters = max(1000, int(remaining * 250))
        sc, a_nest2 = nesterov_subgrad(a_cur, n_iters=est_iters,
                                        tol_frac=0.0005, momentum=0.85, verbose=True)
        report(sc, a_nest2, "nesterov B")

    # ===================================================================
    # Phase 3: LSE continuation (30% time)
    # ===================================================================
    phase3_end = t_start + 0.78 * (total_budget - 30.0)
    print("\n=== Phase 3: LSE Continuation ===")

    a_cur = best_a.copy()
    beta_schedule = [30, 60, 120, 300, 600, 1200, 3000, 6000, 12000, 25000]

    for beta in beta_schedule:
        if time.monotonic() > phase3_end:
            break
        t0 = time.monotonic()
        remaining = max(5.0, phase3_end - time.monotonic())
        maxit = max(300, min(3000, int(remaining * 40)))
        a_cur = run_lbfgs(a_cur, lse_obj, (beta,), maxiter=maxit, maxfun=maxit * 20)
        sc = compute_score(a_cur)
        report(sc, a_cur, f"LSE beta={beta}")
        print(f"  beta={beta}: score={sc:.8f}, time={time.monotonic()-t0:.1f}s")

    # ===================================================================
    # Phase 4: Final polish - subgradient + restarts (14% time)
    # ===================================================================
    phase4_end = deadline
    print("\n=== Phase 4: Final Polish ===")

    # Subgradient with very tight tolerance
    a_cur = best_a.copy()
    remaining = max(1.0, (phase4_end - time.monotonic()) * 0.4)
    est_iters = max(1000, int(remaining * 250))
    sc, a_pol = plain_subgrad(a_cur, n_iters=est_iters,
                               tol_frac=0.0003, verbose=True)
    report(sc, a_pol, "final subgrad tight")

    # Nesterov with different momentum
    if time.monotonic() < phase4_end - 30:
        a_cur = best_a.copy()
        remaining = max(1.0, (phase4_end - time.monotonic()) * 0.3)
        est_iters = max(500, int(remaining * 250))
        sc, a_pol2 = nesterov_subgrad(a_cur, n_iters=est_iters,
                                       tol_frac=0.0005, momentum=0.7, verbose=False)
        report(sc, a_pol2, "final nesterov low-mom")

    # Targeted perturbation + quick LSE
    if time.monotonic() < phase4_end - 20:
        a_pert = targeted_perturb(best_a, rng, strength=0.02)
        a_opt = run_lbfgs(a_pert, lse_obj, (10000,), maxiter=500, maxfun=10000)
        sc = compute_score(a_opt)
        report(sc, a_opt, "perturb+LSE")

    # Final high-beta LSE
    if time.monotonic() < deadline - 10:
        a_fin = run_lbfgs(best_a, lse_obj, (50000,), maxiter=500, maxfun=10000)
        sc = compute_score(a_fin)
        report(sc, a_fin, "final LSE high")

    # ===================================================================
    # Output
    # ===================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
