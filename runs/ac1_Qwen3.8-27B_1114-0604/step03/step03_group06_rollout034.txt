Let me also think about the **Riemann method** more carefully.

The Riemann method (from Riemann 1896, used in Barker sequence construction) works as follows:
1. Start with a sequence p
2. Compute the "excess" at each lag: e_k = conv(p,p)[k] - T where T is the target
3. For lags where e_k > 0, "push down" the sequence
4. Update: p_new = p - alpha * sum_{k: e_k > 0} (contribution of lag k to p)

This is essentially a subgradient method on the "excess" part of the objective.

    FFT_p_rev = np.fft.rfft(p_rev, M2_fft)
    FFT_w = np.fft.rfft(p_weights, M2_fft)
    conv_res = np.fft.irfft(FFT_p_rev * FFT_w, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

**Alternative: Use the fact that the problem is scale-invariant**

Since the score depends only on the ratios of the elements (not their absolute values), we can fix sum(a) = 1 and minimize `2n * max(conv(a,a))`.

This simplifies the problem: minimize `max(conv(a, a))` subject to `a >= 0`, `sum(a) = 1`.

The feasible set is the simplex, which is compact. The objective is the max of linear functions (wait, no, conv is quadratic). So it's: minimize `max_k q_k(a)` where `q_k(a) = sum_i a_i a_{k-i}` is a quadratic form.

This is a minimax quadratic problem. There's a Dinkelbach-type approach:
- Binary search on T
- For each T, check feasibility: exists a >= 0, sum(a) = 1, s.t. q_k(a) <= T for all k

The feasibility problem is a system of quadratic constraints, which is generally NP-hard. But for our specific structure, we might be able to solve it approximately.

**Another idea: S-lemma / SDP relaxation**

The problem `minimize max_k a^T Q_k a` subject to `a >= 0`, `1^T a = 1` where `Q_k` are the matrices such that `a^T Q_k a = conv(a,a)[k]`.

An SDP relaxation would be: minimize T subject to `Tr(Q_k X) <= T` for all k, `X >= 0` (as a matrix), `X 1 = 1` (wait, that's not right for the rank-1 constraint).

The standard SDP relaxation for `minimize max_k a^T Q_k a s.t. 1^T a = 1, a >= 0` would use the lift `X = aa^T` and relax to `X >= 0`, `Tr(X Q_k) <= T`, plus constraints from `1^T a = 1` and `a_i >= 0`.

But the `a_i >= 0` constraints are tricky to lift. And solving large SDPs might be too slow.

"""
Optimized minimization of 2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses LSE-smoothed L-BFGS-B, exact subgradient with Polyak step, peak-directed
perturbations, and simulated annealing. Key improvements: adaptive LSE beta,
multi-peak subgradient, backtracking line search, and refined local search.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for the normalized objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    scaled -= scaled.max()  # numerical stability
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient computation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_exact(a, top_m=1):
    """
    Compute subgradient of the exact max-conv objective.
    If top_m > 1, uses average of subgradients at top-m peaks.
    Returns (score, subgradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(np.max(c))
    score = 2.0 * n * M / (S * S)
    
    # Find top-m peaks
    sorted_idx = np.argsort(c)[::-1]
    # Only include peaks that are close to the max
    threshold = M * 0.999
    active_peaks = sorted_idx[c[sorted_idx] >= threshold][:top_m]
    
    # Subgradient: average of subgradients at active peaks
    sg = np.zeros(n)
    for k in active_peaks:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
    sg /= len(active_peaks)
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def project_simplex(a, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * (target_sum / s)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        return lse_score_grad(x, beta)
    
    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def polyak_subgrad_descent(a, n_iters=2000, top_m=5, sc_lb=1.0):
    """
    Subgradient descent with Polyak step size on exact max-conv objective.
    Uses multi-peak subgradient for smoother descent.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    # Normalize to sum = 1
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        score, grad = subgrad_exact(a, top_m=top_m)
        if not np.isfinite(score):
            break
        if score < best_sc:
            best_sc = score
            best_a = a.copy()
        
        # Polyak step size
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (score - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.5)  # cap the step
        
        a_new = a - step * grad
        # Project back to simplex (sum = 1, non-negative)
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / a_new.sum()
        
        a = a_new
    
    return best_sc, best_a


def backtracking_subgrad(a, n_iters=1000, top_m=3):
    """Subgradient descent with backtracking line search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    cur_sc = best_sc
    
    alpha = 0.01
    for it in range(n_iters):
        score, grad = subgrad_exact(a, top_m=top_m)
        if score < best_sc:
            best_sc = score
            best_a = a.copy()
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Backtracking line search
        improved = False
        for _ in range(20):
            a_new = a - alpha * grad
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-10:
                alpha *= 0.5
                continue
            a_new = a_new / a_new.sum()
            sc_new = compute_score(a_new)
            if sc_new < cur_sc:
                a = a_new
                cur_sc = sc_new
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            alpha *= 0.5
            if alpha < 1e-10:
                break
            # Take a small step anyway
            a_new = a - alpha * grad
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() > 1e-10:
                a = a_new / a_new.sum()
    
    return best_sc, best_a


def peak_shaving(a, n_iters=500, top_m=10):
    """
    Targeted peak shaving: at each iteration, identify the top-m convolution
    peaks and take a gradient step specifically to reduce them.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    cur_sc = best_sc
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(np.max(c))
        S = float(a.sum())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top-m peaks
        sorted_idx = np.argsort(c)[::-1][:top_m]
        # Weights: proportional to how much each peak exceeds the "average"
        avg_c = 1.0 / (2 * n - 1)  # since sum(c) = 1
        excess = c[sorted_idx] - avg_c
        w = excess / excess.sum() if excess.sum() > 0 else np.ones(top_m) / top_m
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for m_idx, k in enumerate(sorted_idx):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += w[m_idx] * 2.0 * a[j_start:j_end + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        # Adaptive step size
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            break
        a = a_new / a_new.sum()
    
    return best_sc, best_a


def simulated_annealing_fine(a, n_iters, max_wall_s, rng, eps=0.05):
    """Fine-grained simulated annealing."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    cur_sc = best_sc
    t0 = time.monotonic()
    
    T_start = 0.01
    T_end = 1e-6
    for it in range(n_iters):
        if time.monotonic() - t0 > max_wall_s:
            break
        T = T_start * (T_end / T_start) ** (it / max(n_iters - 1, 1))
        
        # Perturb a small number of coordinates
        n_pert = max(3, n // 200)
        idx = rng.integers(0, n, size=n_pert)
        a_new = a.copy()
        a_new[idx] *= (1.0 + eps * rng.standard_normal(n_pert))
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            a_new = a.copy()
            continue
        a_new = a_new / a_new.sum()
        
        sc_new = compute_score(a_new)
        if sc_new < cur_sc or rng.random() < np.exp(-(sc_new - cur_sc) / max(T, 1e-12)):
            a = a_new
            cur_sc = sc_new
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
    
    return best_sc, best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy() / parent.sum())
        else:
            # Interpolate parent to length n
            idx = np.linspace(0, len(parent) - 1, n)
            idx_floor = idx.astype(int)
            idx_ceil = np.minimum(idx_floor + 1, len(parent) - 1)
            frac = idx - idx_floor
            interpolated = parent[idx_floor] * (1 - frac) + parent[idx_ceil] * frac
            inits.append(interpolated / interpolated.sum())
    
    # Structured
    inits.append(np.ones(n) / n)
    inits.append((0.5 + 0.3 * np.cos(2.0 * np.pi * x)) / n)
    inits.append((0.5 + 0.3 * np.sin(2.0 * np.pi * x)) / n)
    inits.append((0.5 + 0.4 * np.cos(4.0 * np.pi * x)) / n)
    inits.append((0.3 + 0.7 * np.cos(np.pi * x) ** 2) / n)
    inits.append((0.1 + 0.9 * (1.0 - (2.0 * x - 1.0) ** 4)) / n)
    inits.append(np.clip(0.5 + 0.3 * np.sinc(3.0 * (x - 0.5)), 0.01, None) / n)
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, None) / n)
    inits.append(np.clip(0.3 + 0.4 * rng.random(n), 0.01, None) / n)
    
    return inits

"""
Optimized minimization of 2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Key improvements over previous: (1) KKT-aware subgradient with multi-peak averaging,
(2) adaptive LSE beta based on peak spread, (3) Dinkelbach-style binary search on target,
(4) frequency-domain informed perturbations, (5) tighter local search loop.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient (scale-invariant form)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def multi_peak_subgrad(a, top_m=5, tol=0.999):
    """
    Compute multi-peak subgradient of exact max-conv objective.
    Averages subgradients at all lags where conv >= tol * max.
    Returns (score, subgradient, n_active).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n), 0
    M = float(np.max(c))
    score = 2.0 * n * M / (S * S)
    threshold = tol * M
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    # Cap at top_m
    if len(active) > top_m:
        active = np.argsort(c[active])[::-1][:top_m] + np.argsort(c)[::-1][:top_m]
        active = np.argsort(c)[::-1][:top_m]
    # Compute weighted subgradient (equal weights for simplicity)
    sg = np.zeros(n)
    for k in active:
        j_start = max(0, int(k) - n + 1)
        j_end = min(int(k), n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
    sg /= len(active)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad, len(active)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        return lse_score_grad(x, beta)

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def polyak_subgrad(a, n_iters=2000, top_m=10, sc_lb=1.0, tol=0.998):
    """Subgradient descent with Polyak step on exact max-conv."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        score, grad, n_act = multi_peak_subgrad(a, top_m=top_m, tol=tol)
        if not np.isfinite(score):
            break
        if score < best_sc:
            best_sc = score
            best_a = a.copy()
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (score - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / a_new.sum()
        a = a_new
    return best_sc, best_a


def kkt_polish(a, n_iters=2000, top_m=10):
    """
    KKT-inspired polish: at optimum, for all i with a_i > 0, grad_i should be constant.
    We perturb to equalize the gradient on the support.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        score, grad, n_act = multi_peak_subgrad(a, top_m=top_m, tol=0.999)
        if score < best_sc:
            best_sc = score
            best_a = a.copy()
        support = a > a.max() * 1e-6
        if support.sum() < 2:
            break
        # KKT: on support, grad should be constant. Push towards equalizing.
        grad_support = grad[support]
        grad_mean = grad_support.mean()
        # Correction: reduce grad where it's too high, increase where too low
        correction = np.zeros(n)
        correction[support] = grad_support - grad_mean
        cn = float(np.linalg.norm(correction))
        if cn < 1e-15:
            # KKT approximately satisfied; use subgradient step
            gn = float(np.linalg.norm(grad))
            if gn < 1e-15:
                break
            step = (score - 1.0) / (gn * gn + 1e-30)
            step = min(step, 0.3)
            a_new = a - step * grad
        else:
            step = 0.1 / (cn + 1e-10)
            a_new = a - step * correction
            # Also take a small subgradient step
            gn = float(np.linalg.norm(grad))
            if gn > 1e-15:
                sg_step = (score - 1.0) / (gn * gn + 1e-30) * 0.1
                a_new -= min(sg_step, 0.1) * grad
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / a_new.sum()
        a = a_new
    return best_sc, best_a


def dinkelbach_refine(a, n_iters=500, top_m=10):
    """
    Dinkelbach-style: maintain a running target T and minimize max(0, conv - T) type objective.
    This focuses optimization effort on reducing only the peaks above the target.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    # Initial target: current max
    c = fast_conv(a)
    T = float(np.max(c))
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(np.max(c))
        score = 2.0 * n * M / (S * S)
        if score < best_sc:
            best_sc = score
            best_a = a.copy()
        # Update target slightly below current max (aim to push peaks down)
        T = T * 0.999 + M * 0.001  # slow convergence of target
        # Find peaks above target
        above = np.where(c > T)[0]
        if len(above) == 0:
            above = np.array([int(np.argmax(c))])
        if len(above) > top_m:
            above = np.argsort(c[above])[::-1][:top_m] + np.argsort(c)[::-1][len(above)-top_m:len(above)]
            # Just take top_m from all
            above = np.argsort(c)[::-1][:top_m]
        # Subgradient at peaks above target
        sg = np.zeros(n)
        for k in above:
            j_start = max(0, int(k) - n + 1)
            j_end = min(int(k), n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
        sg /= len(above)
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (score - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / a_new.sum()
        a = a_new
    return best_sc, best_a


def freq_informed_perturb(a, rng, n_modes=5):
    """
    Perturb in directions that are eigenfunctions of the convolution operator.
    These are the natural "modes" that most affect the peak.
    """
    n = len(a)
    a_pert = a.copy()
    for _ in range(n_modes):
        freq = rng.random()
        phase = 2.0 * np.pi * rng.random()
        # Use a windowed sinusoid as the perturbation direction
        x = np.arange(n, dtype=np.float64) / n
        direction = np.cos(2.0 * np.pi * freq * x + phase) * np.sin(np.pi * x)
        amp = 0.005 + 0.02 * rng.random()
        a_pert = a + amp * direction
    a_pert = np.clip(a_pert, 0.0, None)
    if a_pert.sum() > 1e-10:
        a_pert = a_pert / a_pert.sum()
    return a_pert


def build_inits(n, rng, parent):
    """Generate diverse initializations (normalized to sum=1)."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            p = parent.astype(np.float64).copy()
            if p.sum() > 0:

        else:
            idx = np.linspace(0, len(parent) - 1, n)
            idx_f = idx.astype(int)
            idx_c = np.minimum(idx_f + 1, len(parent) - 1)
            frac = (idx - idx_f)[:, None]
            interpolated = parent[idx_f] * (1 - frac) + parent[idx_c] * frac
            interpolated = interpolated[:, 0] if interpolated.ndim > 1 else interpolated
            if interpolated.sum() > 0:
                inits.append(interpolated / interpolated.sum())

    inits.append(np.ones(n) / n)
    inits.append((0.5 + 0.4 * np.cos(2.0 * np.pi * x)).clip(0.01, None) / n)
    inits.append((0.5 + 0.4 * np.cos(4.0 * np.pi * x)).clip(0.01, None) / n)

    inits.append((0.1 + 0.9 * np.sin(np.pi * x) ** 2).clip(0.01, None) / n)
    inits.append((0.5 + 0.3 * np.sinc(4.0 * (x - 0.5))).clip(0.01, None) / n)
    inits.append((0.5 + 0.3 * rng.standard_normal(n)).clip(0.01, None) / n)

    # W-shaped

    lengths = sorted(set(lengths))[:5]

    print(f"Parent: {'yes (len=' + str(parent.size) + ')' if parent is not None else 'no'}")
    print(f"Lengths: {lengths}")
    print(f"Budget: {budget_s}s")

    # ===================================================================
    # Phase 1: L-BFGS-B with LSE continuation (45% time)
    # ===================================================================
    p1_dead = t_start + 0.45 * (budget_s - 15.0)
    print(f"\n{'='*55}")
    print(f"Phase 1: L-BFGS-B LSE (deadline {p1_dead - t_start:.0f}s)")
    print(f"{'='*55}")

    beta_schedule = [30, 80, 200, 500, 1000, 2000, 4000]

    for n in lengths:
        if time.monotonic() > p1_dead:
            break
        print(f"\n  n = {n}")
        inits = build_inits(n, rng, parent)
        remaining = p1_dead - time.monotonic()
        max_inits = max(2, min(len(inits), int(remaining / 80.0)))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > p1_dead:
                break
            a_cur = a0.copy()
            for beta in beta_schedule:
                if time.monotonic() > p1_dead:
                    break
                maxiter = min(4000, max(300, int((p1_dead - time.monotonic()) / 8.0)))
                if maxiter < 100:
                    break
                sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=max(maxiter * 3, 2000))
                a_cur = a_opt
                report(sc, a_opt, f"LBFGS n={n} i={i} b={beta}")
    print(f"\nAfter Phase 1: best = {best_score:.8f}")

    # ===================================================================
    # Phase 2: KKT polish + Dinkelbach (25% time)
    # ===================================================================
    p2_dead = t_start + 0.70 * (budget_s - 15.0)
    print(f"\n{'='*55}")
    print(f"Phase 2: KKT + Dinkelbach (deadline {p2_dead - t_start:.0f}s)")
    print(f"{'='*55}")

    a_cur = best_a.copy() / best_a.sum()
    it_count = 0
    while time.monotonic() < p2_dead and it_count < 40:
        phase = it_count % 4
        remaining_iters = max(100, int((p2_dead - time.monotonic()) / 3.0))
        if phase == 0:
            sc, a_new = kkt_polish(a_cur, n_iters=remaining_iters, top_m=10)
        elif phase == 1:
            sc, a_new = dinkelbach_refine(a_cur, n_iters=remaining_iters, top_m=10)
        elif phase == 2:
            sc, a_new = polyak_subgrad(a_cur, n_iters=remaining_iters, top_m=8)
        else:
            sc, a_new = run_lbfgs(a_cur, 2000, maxiter=min(3000, remaining_iters), maxfun=min(9000, remaining_iters * 3))
        a_cur = a_new
        report(sc, a_cur, f"P2 #{it_count} ph={phase}")
        it_count += 1
    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ===================================================================
    # Phase 3: Fine local search with diverse strategies (20% time)
    # ===================================================================
    p3_dead = t_start + 0.90 * (budget_s - 15.0)
    print(f"\n{'='*55}")
    print(f"Phase 3: Fine Local Search (deadline {p3_dead - t_start:.0f}s)")
    print(f"{'='*55}")

    trial = 0
    while time.monotonic() < p3_dead and trial < 40:
        strategy = trial % 5
        n_b = len(best_a)
        a_pert = best_a.copy()

        if strategy == 0:
            # Tiny random perturbation
            scale = 0.005 + 0.015 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_b))
        elif strategy == 1:
            # Peak-targeted sparse perturbation
            c = fast_conv(best_a)
            top_k = np.argsort(c)[::-1][:3]
            for k in top_k:
                j_start = max(0, int(k) - n_b + 1)
                j_end = min(int(k), n_b - 1)
                n_sp = max(10, (j_end - j_start + 1) // 20)
                idx = rng.integers(j_start, j_end + 1, size=n_sp)
                a_pert[idx] *= (1.0 + 0.05 * rng.standard_normal(n_sp))
        elif strategy == 2:
            # Frequency-informed
            a_pert = freq_informed_perturb(best_a, rng, n_modes=3)
        elif strategy == 3:
            # Mirror perturbation: perturb a and its mirror differently
            half = n_b // 2
            scale = 0.01 + 0.03 * rng.random()
            a_pert[:half] *= (1.0 + scale * rng.standard_normal(half))
            a_pert[half:] *= (1.0 - scale * rng.standard_normal(n_b - half))
        else:
            # Block exchange
            bs = max(5, n_b // 40)
            b1 = int(rng.integers(0, n_b - 2 * bs))
            b2 = int(rng.integers(0, n_b - 2 * bs))
            a_pert[b1:b1 + bs], a_pert[b2:b2 + bs] = \
                a_pert[b2:b2 + bs].copy(), a_pert[b1:b1 + bs].copy()

        a_pert = np.clip(a_pert, 0.0, None)
        if a_pert.sum() > 1e-10:
            a_pert = a_pert / a_pert.sum()
            # Quick polish
            sc, a_opt = run_lbfgs(a_pert, 2000, maxiter=2000, maxfun=5000)
            report(sc, a_opt, f"local #{trial} s={strategy}")
        trial += 1
    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # ===================================================================
    # Phase 4: Final high-precision refinement (10% time)
    # ===================================================================
    print(f"\n{'='*55}")
    print(f"Phase 4: Final Refinement")
    print(f"{'='*55}")

    a_fin = best_a.copy() / best_a.sum()

    # Highest beta LSE
    if time.monotonic() < deadline - 40:
        sc, a_new = run_lbfgs(a_fin, 5000, maxiter=8000, maxfun=160000)
        report(sc, a_new, "final LBFGS b=5000")
        a_fin = a_new

    # KKT polish with many iterations
    if time.monotonic() < deadline - 25:
        sc, a_new = kkt_polish(a_fin, n_iters=3000, top_m=15)
        report(sc, a_new, "final KKT")
        a_fin = a_new

    # Dinkelbach
    if time.monotonic() < deadline - 15:
        sc, a_new = dinkelbach_refine(a_fin, n_iters=2000, top_m=15)
        report(sc, a_new, "final Dinkelbach")
        a_fin = a_new

    # Final subgradient
    if time.monotonic() < deadline - 8:
        sc, a_new = polyak_subgrad(a_fin, n_iters=2000, top_m=20, tol=0.995)
        report(sc, a_new, "final subgrad")
        a_fin = a_new

    # One last L-BFGS
    if time.monotonic() < deadline - 4:
        sc, a_new = run_lbfgs(a_fin, 8000, maxiter=5000, maxfun=100000)
        report(sc, a_new, "final FINAL LBFGS")
        a_fin = a_new

    # ===================================================================
    # Output
    # ===================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in a_fin]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*55}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*55}")
    return out
