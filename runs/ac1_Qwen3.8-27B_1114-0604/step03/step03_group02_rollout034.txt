"""
Step function optimization via flat-spectrum (Gerchberg-Saxton) + LSE L-BFGS + improved multiplicative equalization.

Key ideas:
1. Flat-spectrum / GS iterations: alternate time-domain projection (a>=0) with
   frequency-domain magnitude flattening to reduce peak autocorrelation.
2. LSE-smoothed L-BFGS with continuation on beta for local refinement.
3. Improved multiplicative equalization using all lags (not just top-k).
4. Multi-length exploration (8192 primary, 16384 secondary).
5. Subgradient polish for final tightening.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient (verified correct).
    score = 2n * LSE_beta(conv(a,a)) / (sum a)^2
    grad = 2n/S^3 * (S * g - 2*L) where g[j] = 2*sum_m a[m]*p[j+m]
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    m = beta * (c - c_max)
    m_shifted = m - m.max()
    log_sum_exp = m.max() + np.log(np.exp(m_shifted).sum())
    L = c_max + log_sum_exp / beta
    score = 2.0 * n * L / (S * S)
    p = np.exp(m_shifted)
    p = p / p.sum()
    # Gradient of L: g[j] = 2 * sum_m a[m] * p[j+m]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def gs_iterations(a, n_iters=100, alpha_start=0.6, alpha_end=0.15, deadline=None):
    """
    Gerchberg-Saxton style iterations for flat spectrum.
    Alternates between:
    - Time domain: project to a >= 0, normalize
    - Frequency domain: compress |FFT(a)| dynamics
    Returns (best_score, best_a).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Anneal alpha
        frac = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # Frequency domain: compress magnitude
        A = np.fft.rfft(a, M_fft)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        # Power compression
        mag_new = mag ** alpha
        # Blend with flat (geometric mean with mean magnitude)
        mag_mean = mag_new.mean()
        mag_new = np.sqrt(mag_new * mag_mean)
        A_new = phase * mag_new
        # Inverse FFT
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.real(a_new)

        # Time domain: project to non-negative
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new = best_a.copy()
        else:
            a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

    return best_sc, best_a


def gs_with_damping(a, n_iters=80, damping=0.5, alpha_start=0.5, alpha_end=0.1, deadline=None):
    """
    GS iterations with damping for stability.
    a_new = (1-damping) * a + damping * a_gs
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        frac = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        # Increase damping over time for stability
        d = damping * (1.0 - 0.5 * frac)

        A = np.fft.rfft(a, M_fft)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        mag_new = mag ** alpha
        mag_mean = mag_new.mean()
        mag_new = np.sqrt(mag_new * mag_mean)
        A_new = phase * mag_new
        a_gs = np.real(np.fft.irfft(A_new, M_fft)[:n])
        a_gs = np.maximum(a_gs, 0.0)
        S_gs = a_gs.sum()
        if S_gs < 1e-12:
            continue
        a_gs = a_gs / S_gs

        a_new = (1.0 - d) * a + d * a_gs
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            continue
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_sc, best_a


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000, deadline=None):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_wrapper(x, b):
        return lse_score_grad(x, b)

    options = {'ftol': 1e-14, 'gtol': 1e-10, 'maxiter': maxiter,
               'maxfun': maxfun, 'maxls': 30}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 3.0:
            sc, _ = lse_score_grad(a0, beta)
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(50, int(remaining * 150)))
        options['maxiter'] = min(maxiter, max(50, int(remaining * 30)))

    try:
        res = minimize(obj_wrapper, a0, args=(beta,),
                       method='L-BFGS-B', jac=True, bounds=bounds,
                       options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def multiplicative_equalize_v2(a, n_iters=60, alpha=0.2, deadline=None):
    """
    Improved multiplicative equalization using ALL lags with soft weighting.
    Instead of hard threshold on active lags, use softmax weights.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()

    # Temperature for soft active set
    temp = 50.0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break

        # Soft weights: p[k] proportional to exp((c[k] - c_max) / temp)
        scaled = (c - c_max) / temp
        p = np.exp(scaled - scaled.max())
        p = p / p.sum()

        # Compute h[j] = 2 * sum_m a[m] * p[j+m]
        rev_a = a[::-1]
        RA = np.fft.rfft(rev_a, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
        h = 2.0 * conv_res

        h = np.maximum(h, 1e-15)
        h_mean = h.mean()

        # Damped multiplicative update
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)

        S_new = a.sum()
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a = a * (S / S_new)

        # Decrease temperature (sharpen active set)
        temp *= 0.95

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def subgradient_polish(a, total_iters=200, deadline=None):
    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound

    for it in range(total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max(C) w.r.t. a: 2 * a[k_star - j] for valid j
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S2_new = a.sum()
        if S2_new < 1e-12:
            a = best_a.copy()
            break
        a = a * (S / S2_new)

    return best_sc, best_a


def weighted_directional_update(a, n_steps=50, deadline=None):
    """
    Directional update: at each step, identify the peak lag,
    then move mass from elements that contribute to the peak
    toward elements that contribute to low lags.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break

        # Find top few lags
        top_k = np.argsort(c)[::-1][:max(3, n // 100)]
        bottom_k = np.argsort(c)[:max(3, n // 100)]

        # Total contribution-weighted source (from top lags)
        source_weight = np.zeros(n)
        target_weight = np.zeros(n)

        for k in top_k:
            w = c[k] / c_max
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                source_weight[j_start:j_end + 1] += w * a[j_start:j_end + 1][::-1]

        for k in bottom_k:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                target_weight[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]

        # Normalize weights
        sw = source_weight.sum()
        tw = target_weight.sum()
        if sw < 1e-15 or tw < 1e-15:
            break
        source_weight = source_weight / sw
        target_weight = target_weight / tw

        # Transfer small amount of mass
        eps = 0.02 / (1.0 + 0.1 * step)
        a = a - eps * source_weight + eps * target_weight
        a = np.clip(a, 0.0, 1000.0)
        S_new = a.sum()
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a = a * (S / S_new)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        p = parent.astype(np.float64)
        if len(p) != n:
            p = np.interp(np.linspace(0, len(p) - 1, n), np.arange(len(p)), p)
        inits.append(np.clip(p, 0.01, 1000.0).copy())

    x = np.arange(n, dtype=np.float64) / n

    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Cosine windows
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.5 * np.sin(np.pi * x))
    inits.append(0.2 + 0.6 * np.sin(2.0 * np.pi * x) ** 2)
    # Triangular
    tri = 1.0 - 2.0 * np.abs(x - 0.5)
    inits.append(0.2 + 0.8 * tri)
    # Random smooth
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    # Block
    nb = 16
    per = max(1, n // nb)
    vals = 0.2 + 0.6 * rng.random(nb)
    a = np.repeat(vals, per)
    if len(a) < n:
        a = np.concatenate([a, vals[:n - len(a)]])
    inits.append(a[:n].copy())
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                print(f"Parent loaded: length={p.size}, score={compute_score(parent):.8f}")
    except Exception as e:
        print(f"Parent load error: {e}")

    best_score = np.inf
    best_a = np.ones(8192) * 0.5

    def _report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = [8192]
    if parent is not None and parent.size not in lengths and parent.size <= 16384:
        lengths.insert(0, parent.size)
    lengths.append(16384)
    lengths = sorted(set(lengths))[:3]
    print(f"Lengths to try: {lengths}")

    # ==================================================================
    # Phase 1: GS (flat-spectrum) iterations
    # ==================================================================
    phase1_deadline = t_start + 0.30 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 1: GS Flat-Spectrum (deadline at t={phase1_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n  --- GS: n = {n} ---")
        inits = build_inits(n, rng, parent)
        n_trial = min(len(inits), max(2, int((phase1_deadline - time.monotonic()) / 30.0)))

        for i, a0 in enumerate(inits[:n_trial]):
            if time.monotonic() > phase1_deadline:
                break
            t0 = time.monotonic()

            # Standard GS
            sc_gs, a_gs = gs_iterations(a0, n_iters=80, alpha_start=0.5, alpha_end=0.1,
                                         deadline=phase1_deadline)
            _report(sc_gs, a_gs, f"GS n={n} i={i}")

            # Damped GS
            if time.monotonic() < phase1_deadline - 10:
                sc_dgs, a_dgs = gs_with_damping(a0, n_iters=60, damping=0.4,
                                                 alpha_start=0.5, alpha_end=0.1,
                                                 deadline=phase1_deadline)
                _report(sc_dgs, a_dgs, f"DGS n={n} i={i}")

            elapsed = time.monotonic() - t0
            print(f"    init {i}: done in {elapsed:.1f}s")

    print(f"\nAfter Phase 1: best = {best_score:.8f}, n = {len(best_a)}")

    # ==================================================================
    # Phase 2: LSE L-BFGS continuation
    # ==================================================================
    phase2_deadline = t_start + 0.55 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: LSE L-BFGS (deadline at t={phase2_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    # Use best so far as starting point
    n_best = len(best_a)
    a_cur = best_a.copy()
    beta_schedule = [30, 80, 200, 500]

    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 5:
            break
        maxfun = min(50000, max(200, int(remaining * 80)))
        maxiter = min(5000, max(100, int(remaining * 20)))
        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun,
                               deadline=phase2_deadline)
        _report(sc, a_opt, f"LSE b={beta}")
        a_cur = a_opt

    print(f"\nAfter Phase 2: best = {best_score:.8f}, n = {len(best_a)}")

    # ==================================================================
    # Phase 3: Multiplicative equalization + L-BFGS refinement
    # ==================================================================
    phase3_deadline = t_start + 0.75 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 3: Equalization + Refinement (deadline at t={phase3_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    a_cur = best_a.copy()

    for round_i in range(10):
        if time.monotonic() > phase3_deadline:
            break

        # Multiplicative equalization
        sc_eq, a_eq = multiplicative_equalize_v2(a_cur, n_iters=50, alpha=0.2,
                                                  deadline=phase3_deadline)
        _report(sc_eq, a_eq, f"eq r={round_i}")
        a_cur = a_eq

        # L-BFGS polish
        if time.monotonic() < phase3_deadline - 15:
            beta = 200 + 150 * round_i
            maxfun = min(30000, max(100, int((phase3_deadline - time.monotonic()) * 50)))
            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=2000, maxfun=maxfun,
                                   deadline=phase3_deadline)
            _report(sc, a_opt, f"lbfgs r={round_i} b={beta}")
            a_cur = a_opt

    print(f"\nAfter Phase 3: best = {best_score:.8f}, n = {len(best_a)}")

    # ==================================================================
    # Phase 4: Directional updates + local search
    # ==================================================================
    phase4_deadline = t_start + 0.88 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 4: Directional + Local (deadline at t={phase4_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    n_trials = 15

    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        strategy = trial % 4
        a_try = a_cur.copy()

        if strategy == 0:
            # Weighted directional update
            sc_dir, a_try = weighted_directional_update(a_cur, n_steps=40,
                                                         deadline=phase4_deadline)
            _report(sc_dir, a_try, f"dir #{trial}")
        elif strategy == 1:
            # Gaussian perturbation + LSE
            scale = 0.01 + 0.04 * rng.random()
            a_try = a_cur * (1.0 + scale * rng.standard_normal(len(a_cur)))
            a_try = np.clip(a_try, 0.0, 1000.0)
            sc, a_opt = run_lbfgs(a_try, 300, maxiter=1500, maxfun=20000,
                                   deadline=phase4_deadline)
            _report(sc, a_opt, f"gauss #{trial}")
            a_try = a_opt
        elif strategy == 2:
            # GS from current + small perturbation
            a_pert = a_cur * (1.0 + 0.03 * rng.standard_normal(len(a_cur)))
            a_pert = np.clip(a_pert, 0.01, 1000.0)
            sc_gs, a_try = gs_iterations(a_pert, n_iters=30, alpha_start=0.4,
                                         alpha_end=0.15, deadline=phase4_deadline)
            _report(sc_gs, a_try, f"gs_pert #{trial}")
        else:
            # Multiplicative equalization with different params
            sc_eq, a_try = multiplicative_equalize_v2(a_cur, n_iters=30, alpha=0.4,
                                                       deadline=phase4_deadline)
            _report(sc_eq, a_try, f"eq_strong #{trial}")

        if sc < best_score if 'sc' in dir() else False:
            pass
        if (a_try is not a_cur) and time.monotonic() < phase4_deadline - 5:
            cur_sc = compute_score(a_try)
            if cur_sc < best_score:
                a_cur = a_try.copy()

    # ==================================================================
    # Phase 5: Final polynomial-time polishing
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Polish")
    print(f"{'='*60}")

    # Subgradient polish
    if time.monotonic() < deadline - 30:
        sc_sg, a_sg = subgradient_polish(best_a, total_iters=300, deadline=deadline - 30)
        _report(sc_sg, a_sg, "subgrad")

    # High-beta LSE
    for beta in [500, 1000, 2000]:
        if time.monotonic() < deadline - 20:
            maxfun = min(50000, max(100, int((deadline - time.monotonic() - 10) * 60)))
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=3000, maxfun=maxfun,
                                   deadline=deadline - 10)
            _report(sc, a_opt, f"final b={beta}")

    # One more round of equalization + short LSE
    if time.monotonic() < deadline - 15:
        sc_eq, a_eq = multiplicative_equalize_v2(best_a, n_iters=30, alpha=0.15,
                                                  deadline=deadline - 10)
        _report(sc_eq, a_eq, "final eq")
        if time.monotonic() < deadline - 5:
            sc, a_opt = run_lbfgs(a_eq, 1000, maxiter=2000, maxfun=20000,
                                   deadline=deadline - 3)
            _report(sc, a_opt, "final lbfgs2")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
