    # Wait, that's for the convention where (x ⋆ y)[m] = sum_j x[j] * conj(y[j-m])

    # C = ifft(conj(fft(p, N)) * fft(a, N))

    # dL/da_m = 2 * sum_k p_k * a[k-m] (for valid k-m)

    # Then sum_k p_k * a[k-m] = sum_k p_k * r[n-1-k+m]
    # = (p * r)[n-1+m] (convolution of p and r, at index n-1+m)
    # 
    # convolve(p, r)[k] = sum_j p[j] * r[k-j]
    # For k = n-1+m: sum_j p[j] * r[n-1+m-j] = sum_j p[j] * a[j-m]

def exact_subgrad_polish(a, n_iters=1000):
    """Subgradient method on the exact max objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S  # normalize to sum = 1
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        b = fast_conv(a)
        M = float(np.max(b))
        k_star = int(np.argmax(b))
        
        # Subgradient of max(b) at a:
        # d(max b)/da_m = 2 * a[k_star - m] (for valid k_star - m)
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end+1] = 2.0 * a[j_start:j_end+1][::-1]
        
        # Gradient of 2n*M/S^2 w.r.t. a (with S = 1 after normalization):
        # = 2n * sg (since d(S^2)/da = 2S*1 = 2, and we're at S=1)
        # But we should project back to S=1 after each step.
        grad = 2.0 * n * sg  # gradient of 2n*M (S=1 constraint handled by projection)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Polyak step size: t = (f - f*) / ||g||^2, but we don't know f*.
        # Use a heuristic step size.
        step = 1.0 / (gn * n ** 0.5)
        step = min(step, 0.1)
        
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, None)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new  # project back to sum = 1
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    # Scale back (doesn't matter for the score, but keep reasonable values)
    return best_a * 100  # scale to reasonable range

def perturb_sequence(a, rng, strength=0.05):
    """Perturb a sequence for diversity."""
    n = len(a)
    strategy = rng.integers(0, 4)
    if strategy == 0:
        # Gaussian noise

    elif strategy == 1:

        A_noise = A * (1.0 + noise)

def spectral_flatten_init(n, n_iters=200, lam=0.6, seed=0):
    """Generate initial sequence via iterative spectral flattening."""
    rng = np.random.default_rng(seed)
    a = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0)
    N_fft = 1 << (2*n - 1).bit_length()
    
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam_cur = 0.3 + 0.5 * (1.0 - progress)
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        target = np.median(mag[1:n])  # target magnitude for non-DC bins
        new_mag = mag.copy()
        half = N_fft // 2
        for k in range(1, min(n, half + 1)):
            new_mag[k] = (1.0 - lam_cur) * mag[k] + lam_cur * target
            new_mag[N_fft - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (0.5 * n / S)  # normalize to sum ~ n/2
    
    return a

"""
Minimize peak self-convolution of non-negative sequences using L-BFGS with
log-sum-exp smoothing, exact subgradient polishing, spectral flattening
initializations, and diverse perturbation strategies. Includes length
exploration and adaptive beta schedules for progressively tighter peaks.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    b = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(b)) / (S * S)


def lse_objective_and_grad(a, beta):
    """Smoothed objective: 2n*LSE(beta*b)/S^2 and its gradient."""
    n = len(a)
    b = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    b_max = float(np.max(b))
    scaled = beta * (b - b_max)
    exp_scaled = np.exp(scaled)
    sum_exp = float(exp_scaled.sum())
    if sum_exp < 1e-300:
        return 1e10, np.zeros(n)
    L = b_max + np.log(sum_exp) / beta
    p = exp_scaled / sum_exp
    # Gradient of L w.r.t. a: dL/da_m = 2 * convolve(p, reverse(a))[n-1:2n-1]
    r = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    R = np.fft.rfft(r, M2_fft)
    conv_pr = np.fft.irfft(P * R, M2_fft)
    grad_L = 2.0 * conv_pr[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad_f = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)
    obj_val = 2.0 * n * L / S2

    res = minimize(
        lse_objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    return np.clip(res.x, 0.0, 1000.0)


def exact_subgrad_polish(a, n_iters=1000):
    """Subgradient method on exact max-convolution objective (sum=1 normalized)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step_scale = 1.0
    for it in range(n_iters):
        b = fast_conv(a)
        M = float(np.max(b))
        k_star = int(np.argmax(b))
        # Subgradient of 2n*M (at S=1): sg_m = 2*n*2*a[k_star-m] (valid range)
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        grad = 2.0 * n * sg
        # Account for the 1/S^2 term: full grad of 2n*M/S^2 at S=1 is 2n*sg - 4n*M*ones
        ones = np.ones(n)
        grad = grad - 4.0 * n * M * ones
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = step_scale / (gn * np.sqrt(n))
        step = min(step, 0.05)
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, None)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new = a.copy()
            step_scale *= 0.5
            continue
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            step_scale = min(step_scale * 1.1, 5.0)
        else:
            step_scale *= 0.7
        a = a_new
    return best_a


def spectral_flatten_init(n, n_iters, lam, seed):
    """Iterative spectral flattening to generate low-peak initializations."""
    rng = np.random.default_rng(seed)
    a = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0)
    N_fft = 1 << (2 * n - 1).bit_length()
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam_cur = 0.3 + 0.5 * (1.0 - progress)
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        target = float(np.median(mag[1:min(n, N_fft // 2)]))
        new_mag = mag.copy()
        half = N_fft // 2
        lim = min(n, half)
        for k in range(1, lim + 1):
            new_mag[k] = (1.0 - lam_cur) * mag[k] + lam_cur * target
            new_mag[N_fft - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (0.5 * n / S)
    return a


def chirp_init(n, seed):
    """Generate chirp-based initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    inits.append(np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01)
    for f in [1, 3, 7, 13]:
        inits.append(0.3 + 0.3 * np.cos(2.0 * np.pi * f * x / n + rng.random() * 2.0 * np.pi))
    inits.append(0.5 + 0.3 * np.cos(np.pi * x * x / (2.0 * n)) * np.cos(2.0 * np.pi * 3.0 * x / n))
    inits = [np.clip(s, 0.01, 100.0) for s in inits]
    return inits


def perturb_sequence(a, rng, strength):
    """Diverse perturbation of a sequence."""
    n = len(a)
    strategy = int(rng.integers(0, 5))
    if strategy == 0:
        a_pert = a * (1.0 + strength * rng.standard_normal(n))
    elif strategy == 1:
        N_fft = 1 << (2 * n - 1).bit_length()
        A = np.fft.fft(a, N_fft)
        noise = 0.03 * (rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A)))
        A_noise = A * (1.0 + noise)
        A_noise[0] = A[0]
        a_pert = np.real(np.fft.ifft(A_noise))[:n]
    elif strategy == 2:
        x = np.arange(n, dtype=np.float64)
        f = 1.0 + 5.0 * rng.random()
        a_pert = a * (1.0 + 0.05 * np.cos(np.pi * f * x * x / (2.0 * n)))
    elif strategy == 3:
        nb = int(rng.integers(4, 32))
        per = max(1, n // nb)
        block_noise = 0.1 * rng.standard_normal(nb)
        a_pert = a * (1.0 + np.repeat(block_noise, per)[:n])
    else:
        # Targeted: reduce high-contributing elements, boost low-contributing
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        top_k = max(3, n // 100)
        top_idx = np.argsort(contrib)[::-1][:top_k]
        bot_idx = np.argsort(contrib)[:top_k]
        a_pert = a.copy()
        total_red = 0.0
        for j in top_idx:
            red = a_pert[j] * strength * 2.0
            a_pert[j] -= red
            a_pert[j] = max(a_pert[j], 0.0)
            total_red += red
        if len(bot_idx) > 0 and total_red > 0:
            a_pert[bot_idx] += total_red / len(bot_idx)
    return np.clip(a_pert, 0.0, 1000.0)


def try_optimize(a_init, n, beta_schedule, deadline, state):
    """Run L-BFGS beta-sweep from an initialization, updating state."""
    a_cur = a_init.copy()
    S = a_cur.sum()
    if S > 0:
        a_cur = a_cur * (n * 0.5 / S)
    for beta in beta_schedule:
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        est_per_call = 0.003 * max(1.0, n / 4096.0)
        maxfun = min(50000, max(500, int(remaining * 0.15 / est_per_call)))
        maxiter = min(5000, max(100, maxfun // 4))
        a_opt = run_lbfgs(a_cur, beta, maxiter, maxfun)
        a_cur = a_opt
        sc = compute_score(a_cur)
        if sc < state['best_score']:
            state['best_score'] = sc
            state['best_a'] = a_cur.copy()
            print(f"    NEW BEST: {sc:.8f} (n={n}, beta={beta}, t={time.monotonic()-state['t0']:.0f}s)")
    # Subgradient polish
    if time.monotonic() < deadline - 20:
        a_pol = exact_subgrad_polish(a_cur, n_iters=300)
        sc = compute_score(a_pol)
        if sc < state['best_score']:
            state['best_score'] = sc
            state['best_a'] = a_pol.copy()
            print(f"    POLISH BEST: {sc:.8f} (n={n}, t={time.monotonic()-state['t0']:.0f}s)")
    return a_cur


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Multi-phase: spectral init -> L-BFGS beta sweep -> subgrad polish -> perturbations.
    """
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)
    state = {'best_score': np.inf, 'best_a': None, 't0': t0}

    # Get parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.array(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64 and p.max() > 0:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for n_try in [4096, 8192, 12288]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths: {lengths}, Parent: {parent.size if parent is not None else 'none'}")

    # Phase 1: Generate and score initializations
    print("\n--- Phase 1: Initializations ---")
    inits_by_length = {}
    for n in lengths:
        inits = []
        if parent is not None and parent.size == n:
            inits.append(parent.copy())
        for sf_seed in [0, 1, 2, 3]:
            inits.append(spectral_flatten_init(n, n_iters=200, lam=0.5, seed=sf_seed))
        inits.extend(chirp_init(n, seed=n % 200))
        inits.append(np.ones(n) * 0.5)
        inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
        # Golay-like: sum of two random phases
        x = np.arange(n, dtype=np.float64)
        inits.append(0.5 + 0.3 * np.cos(np.pi * x * x / (2.0 * n)) + 0.1 * np.cos(2.0 * np.pi * 5.0 * x / n))

        scored = []
        for a_init in inits:
            a_init = np.clip(a_init, 0.0, 1000.0)
            if a_init.sum() < 0.01:
                continue
            sc = compute_score(a_init)
            scored.append((sc, a_init))
        scored.sort(key=lambda x: x[0])
        inits_by_length[n] = scored[:5]
        print(f"  n={n}: best init = {scored[0][0]:.6f} (from {len(scored)} candidates)")

    # Phase 2: L-BFGS optimization from best inits
    print("\n--- Phase 2: L-BFGS Optimization ---")
    beta_schedules = {
        'short': [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000],
        'long': [10, 30, 60, 120, 250, 500, 1000, 2000, 5000],
    }
    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        print(f"\n  >> n={n}")
        scored_inits = inits_by_length[n]
        n_top = min(3, len(scored_inits))
        schedule = beta_schedules['short'] if n <= 8192 else beta_schedules['long']
        for ci in range(n_top):
            if time.monotonic() > deadline - 30:
                break
            a_init = scored_inits[ci][1].copy()
            remaining = deadline - time.monotonic()
            print(f"    Init {ci} (score={scored_inits[ci][0]:.6f}), remaining={remaining:.0f}s")
            try_optimize(a_init, n, schedule, deadline, state)

    # Phase 3: Diverse perturbations from best
    print(f"\n--- Phase 3: Perturbations (best={state['best_score']:.6f}) ---")
    if state['best_a'] is not None:
        n_best = state['best_a'].size
        for pt in range(20):
            if time.monotonic() > deadline - 30:
                break
            strength = 0.02 + 0.06 * rng.random()
            a_pert = perturb_sequence(state['best_a'], rng, strength)
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            schedule = [200, 500, 1000, 2000, 5000]
            # Scale down schedule if not much time
            n_beta = min(len(schedule), max(2, int(remaining / 15)))
            try_optimize(a_pert, n_best, schedule[:n_beta], deadline, state)

    # Phase 4: Final aggressive polishing
    print(f"\n--- Phase 4: Final Polish (best={state['best_score']:.6f}) ---")
    if state['best_a'] is not None:
        for round_i in range(4):
            if time.monotonic() > deadline - 15:
                break
            n_best = state['best_a'].size
            a_pol = exact_subgrad_polish(state['best_a'], n_iters=800)
            sc = compute_score(a_pol)
            if sc < state['best_score']:
                state['best_score'] = sc
                state['best_a'] = a_pol.copy()
                print(f"    Round {round_i}: {sc:.8f}")
            # High-beta L-BFGS
            if time.monotonic() < deadline - 15:
                a_opt = run_lbfgs(state['best_a'], 10000, 3000, 30000)
                sc = compute_score(a_opt)
                if sc < state['best_score']:
                    state['best_score'] = sc
                    state['best_a'] = a_opt.copy()
                    print(f"    LBFGS hi-beta: {sc:.8f}")

    # Final verification
    if state['best_a'] is None:
        state['best_a'] = np.ones(1024) * 0.5

    out = [float(max(0.0, min(1000.0, x))) for x in state['best_a']]
    if len(out) < 1 or sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t0
    print(f"\n{'='*60}")
    print(f"FINAL: internal={state['best_score']:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
