"""
Frequency-domain Fienup + L-BFGS + Top-M Peak Subgradient optimizer.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovations:
1. Fienup-style iterative method operating on the power spectrum |DFT(a)|^2
2. Soft thresholding with adaptive parameters
3. Top-M peak subgradient with backtracking
4. Multi-length exploration with interpolation-based transfer
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_iteration(a, n_iters=200, mode='soft', threshold_q=0.7, 
                     clip_strength=0.5, normalize=True):
    """
    Fienup-style iteration: flatten the power spectrum |DFT(a)|^2.
    
    Parameters:
    - a: input sequence (non-negative)
    - n_iters: number of iterations
    - mode: 'soft', 'hard', 'quantile'
    - threshold_q: quantile for threshold (if mode='quantile')
    - clip_strength: how aggressively to clip (0 = no clip, 1 = full clip)
    - normalize: whether to normalize to sum=1 after each iteration
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    if normalize:
        a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        # Forward transform
        A = np.fft.rfft(a, M_fft)
        P = (A.real ** 2 + A.imag ** 2)  # Power spectrum
        
        # Determine threshold
        P_mean = float(P.sum()) / len(P)
        if mode == 'hard':
            thresh = P_mean * 1.5
        elif mode == 'soft':
            thresh = P_mean + (1.0 - clip_strength) * (P.max() - P_mean) * 0.5
        else:  # quantile
            thresh = np.quantile(P, threshold_q)
        
        # Reduce power in bins above threshold
        excess = np.maximum(P - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        
        # Modify amplitude: reduce |A| where P is too high
        scale = np.ones_like(P)
        above = P > thresh
        # For soft mode: gradual reduction
        if mode == 'soft':
            scale[above] = np.sqrt(thresh / P[above]) * (1.0 - 0.5 * clip_strength) + 0.5 * clip_strength
            # Actually, let's do: A *= (thresh/P)^(alpha/2) for bins above threshold
            scale[above] = (thresh / P[above]) ** (clip_strength / 2.0)
        elif mode == 'hard':
            scale[above] = np.sqrt(thresh / P[above])
        else:
            scale[above] = (thresh / P[above]) ** (0.5 * clip_strength)
        
        A_modified = A * scale
        
        # Back to time domain
        a_new = np.fft.irfft(A_modified, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        
        if normalize:
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            a_new = a_new / S_new
        
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def top_m_subgradient(a, n_iters=3000, top_m=10, step_init=0.05, decay=0.999):
    """
    Subgradient method using top-M peaks instead of just the max.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top-M peaks
        top_idx = np.argsort(c)[::-1][:top_m]
        
        # Subgradient: sum of contributions from top-M peaks
        sg = np.zeros(n)
        for k in top_idx:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.1)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        step *= decay
    
    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def objective(x):
        return lse_score_grad(x, beta)
    
    res = minimize(
        objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def fienup_adaptive(a, n_iters=300):
    """
    Adaptive Fienup: gradually increase clipping strength.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        # Adaptive parameters
        progress = it / n_iters
        clip_strength = 0.3 + 0.6 * progress  # Start gentle, end aggressive
        n_thresholds = 3 + int(5 * progress)  # More aggressive thresholding over time
        
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        
        P_mean = float(P.mean())
        
        # Multi-level thresholding
        scale = np.ones_like(P)
        for level in range(n_thresholds):
            thresh = P_mean * (1.0 + 0.3 * (level + 1) / n_thresholds)
            above = P > thresh
            if above.any():
                reduction = np.sqrt(np.maximum(thresh / P[above], 0.01))
                scale[above] *= (1.0 - clip_strength) + clip_strength * reduction
        
        A_modified = A * scale
        a_new = np.fft.irfft(A_modified, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0, 1000).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))

    inits.append(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x))

    inits.append(np.clip(0.5 + 0.5 * np.cos(np.pi * (2*x - 1)), 0.01, 1000.0))

"""
Frequency-domain Fienup + L-BFGS + Top-M Subgradient hybrid optimizer.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovations over previous approach:
1. Fienup-style iteration directly on the power spectrum |DFT(a)|^2
2. Adaptive multi-level thresholding schedule
3. Top-M peak subgradient (not just max peak) with decay
4. Cycling: Fienup -> L-BFGS -> Fienup -> L-BFGS for escaping local optima
5. Multi-length exploration with targeted n around known-good region
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_iteration(a, n_iters=300, mode='soft', clip_strength=0.5, threshold_mult=1.3):
    """
    Fienup-style iteration: flatten |DFT(a)|^2 by reducing power in high bins.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Adaptive threshold: start wide, tighten over iterations
        thresh = P_mean * threshold_mult * (1.0 + 0.5 * (1.0 - progress))

        above = P > thresh
        if not above.any():
            break

        if mode == 'soft':
            scale = np.ones_like(P)
            scale[above] = (thresh / P[above]) ** (clip_strength * 0.5)
        elif mode == 'hard':
            scale = np.ones_like(P)
            scale[above] = np.sqrt(thresh / P[above])
        else:  # 'quantile'
            thresh_q = np.quantile(P, 0.8)
            scale = np.ones_like(P)
            scale[P > thresh_q] = (thresh_q / P[P > thresh_q]) ** (clip_strength * 0.5)

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_adaptive(a, n_iters=300):
    """
    Adaptive Fienup with multi-level thresholding that tightens over time.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / n_iters
        clip_str = 0.3 + 0.5 * progress
        n_levels = 2 + int(4 * progress)

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        scale = np.ones_like(P)
        for lv in range(n_levels):
            thresh = P_mean * (1.0 + 0.25 * (lv + 1) / n_levels)
            above = P > thresh
            if above.any():
                red = np.sqrt(np.maximum(thresh / P[above], 0.01))
                scale[above] *= (1.0 - clip_str) + clip_str * red

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

    Aggressive Fienup: strong clipping of top power bins.
    Targets the top 30% of power spectrum bins.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Target top fraction of bins
        top_frac = 0.3 + 0.1 * (it / n_iters)  # Start at 30%, go to 40%
        thresh = np.quantile(P, 1.0 - top_frac)

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            # Strong reduction: bring them close to threshold
            scale[above] = np.sqrt(thresh / P[above]) * 0.95

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def top_m_subgradient(a, n_iters=3000, top_m=10, step_init=0.05, decay=0.9995):
    """
    Subgradient method using top-M peaks. More stable than single-peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.08)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(x, beta):
    """Wrapper for scipy.minimize (top-level, no closure)."""
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=8000, maxfun=150000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coordinate_descent_peak(a, n_sweeps=5, n_top=60):
    """
    Coordinate descent on peak-contributing elements with golden section.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-12:
                continue
            # Try multiplicative perturbations
            candidates = [a_old_j * f for f in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5]]
            best_val = a_old_j
            best_local_sc = best_sc  # use current best as baseline
            # Actually use current score
            cur_sc = compute_score(a)
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-10:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
                a[j] = a_old_j
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(2.0 * np.pi * 2 * x), 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.cos(2.0 * np.pi * 3 * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    return inits


def cycle_fienup_lbfgs(a, n_fienup_iters=100, n_cycles=3):
    """
    Cycling: Fienup -> L-BFGS -> Fienup -> L-BFGS.
    Helps escape local optima by alternating global and local methods.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for cyc in range(n_cycles):
        # Fienup step
        a = fienup_adaptive(a, n_iters=n_fienup_iters)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # L-BFGS step
        beta = 100 + 400 * cyc
        sc2, a2 = run_lbfgs(a, beta, maxiter=3000, maxfun=50000)
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()
            a = a2

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 25.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths - focus on range around previous best (6962)
    lengths = [4096, 6144, 8192]
    if parent is not None:
        lengths.insert(0, len(parent))
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes (n=' + str(len(parent)) + ')' if parent is not None else 'no'}")

    # ==================================================================
    # Phase 1: Fienup from diverse inits (40% of time)
    # ==================================================================
    phase1_end = t_start + 0.40 * (total_budget - 25.0)
    print(f"\n=== Phase 1: Fienup (deadline t={phase1_end - t_start:.0f}s) ===")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  Processing n={n}")
        inits = build_initializations(n, rng, parent)

        # Prioritize parent if available
        inits_sorted = inits  # parent is first if available

        for i, a0 in enumerate(inits_sorted):
            if time.monotonic() > phase1_end:
                break

            # Standard soft Fienup
            a_f = fienup_iteration(a0, n_iters=250, mode='soft', clip_strength=0.6)
            sc = compute_score(a_f)
            update_best(sc, a_f, f"FIENUP-soft n={n} init={i}")

            # Adaptive Fienup on the result
            if time.monotonic() < phase1_end:
                a_af = fienup_adaptive(a_f, n_iters=200)
                sc2 = compute_score(a_af)
                update_best(sc2, a_af, f"FIENUP-adapt n={n} init={i}")

            # Aggressive Fienup
            if time.monotonic() < phase1_end:
                a_ag = fienup_aggressive(a0, n_iters=150)
                sc3 = compute_score(a_ag)
                update_best(sc3, a_ag, f"FIENUP-aggr n={n} init={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Cycling Fienup+L-BFGS (20% of time)
    # ==================================================================
    phase2_end = t_start + 0.60 * (total_budget - 25.0)
    print(f"\n=== Phase 2: Cycling (deadline t={phase2_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    a_cycle = cycle_fienup_lbfgs(best_a, n_fienup_iters=100, n_cycles=3)
    sc = compute_score(a_cycle)
    update_best(sc, a_cycle, "CYCLE")

    # Also try cycling from a perturbed version
    a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(n_best))
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    a_pert = a_pert / a_pert.sum()
    if time.monotonic() < phase2_end:
        a_cycle2 = cycle_fienup_lbfgs(a_pert, n_fienup_iters=80, n_cycles=2)
        sc2 = compute_score(a_cycle2)
        update_best(sc2, a_cycle2, "CYCLE-pert")

    # Try cycling at a different length
    if time.monotonic() < phase2_end and len(lengths) > 1:
        n_alt = lengths[-1] if len(best_a) != lengths[-1] else lengths[0]
        if n_alt != len(best_a):
            a_alt = np.interp(np.linspace(0, 1, n_alt),
                             np.linspace(0, 1, len(best_a)), best_a)
            a_alt = np.maximum(a_alt, 0.01)
            a_cycle3 = cycle_fienup_lbfgs(a_alt, n_fienup_iters=80, n_cycles=2)
            sc3 = compute_score(a_cycle3)
            update_best(sc3, a_cycle3, f"CYCLE n={n_alt}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: L-BFGS with multiple betas (25% of time)
    # ==================================================================
    phase3_end = t_start + 0.85 * (total_budget - 25.0)
    print(f"\n=== Phase 3: L-BFGS (deadline t={phase3_end - t_start:.0f}s) ===")

    for beta in [50, 100, 200, 500, 1000, 2000, 5000]:
        if time.monotonic() > phase3_end:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=10000, maxfun=200000)
        update_best(sc, a_opt, f"LBFGS beta={beta}")

    # L-BFGS from parent at its native length
    if parent is not None and len(parent) != len(best_a):
        if time.monotonic() < phase3_end:
            for beta in [200, 1000]:
                if time.monotonic() > phase3_end:
                    break
                sc, a_opt = run_lbfgs(parent, beta, maxiter=8000, maxfun=120000)
                update_best(sc, a_opt, f"LBFGS-parent beta={beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Top-M subgradient + Coordinate descent (10% of time)
    # ==================================================================
    print(f"\n=== Phase 4: Subgradient + Coord Descent ===")

    for top_m in [5, 10, 20, 40]:
        if time.monotonic() > deadline:
            break
        a_sg = top_m_subgradient(best_a, n_iters=3000, top_m=top_m)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"SG top_m={top_m}")

    if time.monotonic() < deadline - 30:
        a_cd = coordinate_descent_peak(best_a, n_sweeps=4, n_top=80)
        sc = compute_score(a_cd)
        update_best(sc, a_cd, "coord-descent")

        # L-BFGS from coord descent result
        if time.monotonic() < deadline - 20:
            sc2, a_opt = run_lbfgs(a_cd, 500, maxiter=3000, maxfun=50000)
            update_best(sc2, a_opt, "CD+LBFGS")

    # ==================================================================
    # Phase 5: Final Fienup + verification (remaining)
    # ==================================================================
    if time.monotonic() < deadline - 15:
        a_final_f = fienup_adaptive(best_a, n_iters=150)
        sc = compute_score(a_final_f)
        update_best(sc, a_final_f, "FINAL-FIENUP")

    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(best_a, 3000, maxiter=3000, maxfun=40000)
        update_best(sc, a_opt, "FINAL-LBFGS")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
