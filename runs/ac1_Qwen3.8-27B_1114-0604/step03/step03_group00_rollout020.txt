**Approach: Iterative Proportional Fitting (IPF) / Sinkhorn-like algorithm**

The idea:
1. We want to minimize `max_k b[k]` where `b[k] = sum_j a[j]*a[k-j]`
2. This is equivalent to finding `a >= 0` such that `b` is as flat as possible
3. We can use an alternating minimization:
   a. Given `a`, compute `b`
   b. Compute `target = S^2 / (2n-1)` (the flat level)
   c. For each lag k where `b[k] > target`, we want to reduce it
   d. The "direction" to reduce `b[k]` is to decrease `a[j]` for j that contribute to `b[k]`
   e. Specifically, reduce `a[j]` proportionally to `a[k-j]` (the "partner" value)

This gives the multiplicative update:
`a[j] <- a[j] * exp(-eta * sum_{k: b[k]>target} p[k] * a[k-j])`

Where p[k] is some weighting (e.g., proportional to the excess `b[k] - target`).

This is guaranteed to maintain non-negativity and is similar to the "iterative filtering" approach in genomics.

    a = np.maximum(a.astype(np.float64), 1e-10)
    S = a.sum()
    a = a / S  # normalize
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):

        c_max = c.max()
        c_mean = S*S / len(c)  # theoretical flat level
        
        if c_max <= c_mean * 1.01:
            break
        
        # Find top-k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        excess = np.maximum(c[top_idx] - c_mean, 0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            break
        
        # Weight: proportional to excess
        w = excess / total_excess
        
        # Compute the "pressure" on each a[j]:
        # pressure[j] = sum_{k in top_k} w[k_idx] * a[k - j] (for valid k-j)
        pressure = np.zeros(n)

            if w_k < 1e-10:

"""
Improved optimizer for minimizing 2n * max(conv(a,a)) / (sum(a))^2.
Uses FFT-accelerated multiplicative peak reduction, LSE-smoothed L-BFGS,
multi-scale exploration, and annealed restarts.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient computed via FFT."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    # Gradient: grad_LSE[i] = sum_k p[k] * (db[k]/da[i])
    # db[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    # So grad_LSE[i] = sum_k p[k]*a[k-i] (valid) + sum_k p[k]*a[i-k] (valid)
    # = conv_ap[i] + conv_a_pr[2n-2-i] where:
    #   conv_ap = conv(a, p) [linear, length 3n-2]
    #   conv_a_pr = conv(a, p_reversed) [linear, length 3n-2]

    M_conv = 3 * n - 2
    M_conv_fft = 1 << (M_conv - 1).bit_length()
    A_fft = np.fft.rfft(a, M_conv_fft)
    P_fft = np.fft.rfft(p, M_conv_fft)
    PR = p[::-1]
    PR_fft = np.fft.rfft(PR, M_conv_fft)

    conv_ap = np.fft.irfft(A_fft * P_fft, M_conv_fft)[:M_conv]
    conv_a_pr = np.fft.irfft(A_fft * PR_fft, M_conv_fft)[:M_conv]

    # grad_LSE[i] = conv_ap[i] + conv_a_pr[2*n-2-i] for i in [0, n-1]
    # conv_a_pr[2n-2-i] for i=0..n-1: indices 2n-2, 2n-3, ..., n-1
    # = conv_a_pr[n-1 : 2*n-1][::-1]
    grad_LSE = conv_ap[:n] + conv_a_pr[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_LSE / S2 - 2.0 * L * a / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_and_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(1e-10, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 1e-10, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def multiplicative_peak_reduction(a, n_iters=2000, eta=0.05, top_k=20, decay=0.999):
    """
    FFT-accelerated multiplicative peak reduction.
    Maintains non-negativity by construction.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    S = a.sum()
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    M_conv = 3 * n - 2
    M_conv_fft = 1 << (M_conv - 1).bit_length()

    cur_eta = eta
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        c_ideal = S * S / (2 * n - 1)
        c_max = float(c.max())

        if c_max <= c_ideal * 1.005:
            break

        # Weight the peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        excess = np.maximum(c[top_idx] - c_ideal, 0.0)
        total_excess = float(excess.sum())
        if total_excess < 1e-15:
            break
        w = excess / total_excess

        # Build the weight vector W of length 2n-1
        W = np.zeros(2 * n - 1)
        for idx, k in enumerate(top_idx):
            W[int(k)] = w[idx]

        # Compute pressure[j] = sum_k W[k] * a[k-j] (valid k-j in [0,n-1])
        # This is cross-correlation: pressure = IFFT(FFT(W) * conj(FFT(a)))
        W_fft = np.fft.rfft(W, M_conv_fft)
        A_fft = np.fft.rfft(a, M_conv_fft)
        # cross-corr[j] = sum_k W[k] * a[k-j] = IFFT(FFT(W) * conj(FFT(a_padded)))
        # For linear cross-correlation with W (len 2n-1) and a (len n):
        # result has length (2n-1) + n - 1 = 3n-2
        # pressure[j] for j in [0, n-1] is the first n elements
        pressure = np.fft.irfft(W_fft * np.conj(A_fft), M_conv_fft)[:n]
        pressure = np.maximum(pressure, 0.0)

        # Multiplicative update
        a = a * np.exp(-cur_eta * pressure)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        cur_eta *= decay

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def additive_peak_reduction(a, n_iters=1000, step_scale=0.01):
    """
    Additive peak reduction with line search.
    Moves mass from peak contributors to non-peak positions.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    S = a.sum()
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        c_max = float(c[k_star])
        c_ideal = 1.0 / (2 * n - 1)  # since sum(a)=1, sum(c)=1

        if c_max <= c_ideal * 1.01:
            break

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        partners = a[k_star - j_end:k_star - j_start + 1]  # a[k_star - j] for j in [j_start, j_end]

        # Direction: reduce a[j] where partner is large, add to where partner is small
        contrib = seg * partners  # contribution of each a[j] to b[k_star]
        total_contrib = float(contrib.sum())
        if total_contrib < 1e-15:
            break

        # Target: make all contributions approximately equal
        target_contrib = total_contrib / len(contrib)
        # delta[j] proportional to (contrib[j] - target_contrib)
        delta = contrib - target_contrib

        # Apply with small step
        step = step_scale / (1.0 + 0.1 * it / n_iters)
        a[j_start:j_end + 1] -= step * delta

        # Also add to non-peak positions (positions not in [j_start, j_end])
        if j_start > 0:
            left_mass = step * float(np.maximum(delta[:j_start - j_start], 0).sum()) if j_start < len(delta) else 0
            # Simpler: distribute the total reduction to the rest
            total_reduction = float(np.maximum(step * delta, 0).sum())
            if total_reduction > 0:
                mask = np.ones(n, dtype=bool)
                mask[j_start:j_end + 1] = False
                n_other = mask.sum()
                if n_other > 0:
                    a[mask] += total_reduction / n_other

        a = np.maximum(a, 1e-10)
        S = a.sum()
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def frequency_flatten(a, n_iters=200):
    """
    Frequency-domain flattening: modify |A|^2 to reduce autocorrelation peaks,
    then recover a via phase retrieval (keep original phase, adjust magnitude).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()
    best_sc = compute_score(a)

    M = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        mag = np.abs(A)
        phase = np.angle(A)

        # Current autocorrelation (circular)
        ac = np.fft.ifft(mag * mag).real

        ac = np.concatenate([ac[n:], ac[:n]])  # center the autocorrelation
        ac_mean = float(ac.mean())
        ac_max = float(ac.max())

        if ac_max - ac_mean < 1e-14:
            break

        # Reduce peaks in |A|^2 where they contribute to ac peaks
        # The contribution of |A[k]|^2 to ac[m] is |A[k]|^2 * exp(-2*pi*i*k*n/M) ... 
        # Simplified: reduce |A|^2 at indices where the corresponding ac "mode" is peaked
        # Use a softmax-like compression
        beta_flatten = 2.0 + 3.0 * it / n_iters
        ac_scaled = beta_flatten * (ac - ac_max)
        ac_scaled = np.clip(ac_scaled, -50, 0)
        weight = np.exp(ac_scaled)  # lower weight where ac is high
        weight = weight / weight.mean()

        # Apply to |A|^2: new_mag^2 = |A|^2 * weight (broadcast appropriately)
        # The weight has length M, and |A|^2 also has length M
        new_mag2 = (mag * mag) * weight
        new_mag = np.sqrt(np.maximum(new_mag2, 0))

        # Reconstruct with new magnitude, keep phase
        A_new = new_mag * np.exp(1j * phase)

        # Enforce conjugate symmetry for real output
        A_new[-n:] = np.conj(A_new[n:-1][::-1]) if M > 2 * n else A_new[-n:]
        # Actually, for a real signal: A[M-k] = conj(A[k])
        for k in range(1, M // 2 + 1):
            A_new[M - k] = np.conj(A_new[k])
        if M % 2 == 0:
            A_new[M // 2] = float(np.real(A_new[M // 2]))
        A_new[0] = float(np.real(A_new[0]))

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 1e-15)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""

    inits.append(np.ones(n) * 0.5)
    # U-shaped (counters triangular conv)
    inits.append(0.3 + 0.7 * (2.0 * x - 1.0) ** 2)

    # Multi-frequency
    inits.append(0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) * np.cos(2 * np.pi * x))
    inits.append(np.clip(0.5 + 0.2 * np.cos(2 * np.pi * 5 * x) + 0.1 * np.cos(2 * np.pi * 7 * x), 0.01, None))
    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    # Block structure
    for nb in (16, 32, 64):
        per = max(1, n // nb)
        vals = 0.3 + 0.7 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    return inits
