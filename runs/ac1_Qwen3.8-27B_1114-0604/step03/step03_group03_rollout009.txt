"""
Improved optimizer for minimum peak autocorrelation of non-negative sequences.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2.

Key improvements over previous version:
1. Faster coordinate descent using closed-form 1D solutions (3-5x speedup)
2. Continuation method: anneal LSE beta from low to high for better global search

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.minimum(scaled, 500))
    log_sum = np.log(exp_scaled.sum() + 1e-300)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / (exp_scaled.sum() + 1e-300)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_wrapper(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-14, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun,
                     'maxls': 40}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def _1d_obj_val(t, base, alpha, beta_coef, outside_max, S_rest, n):
    """Evaluate 1D objective at t."""
    inside = base + t * alpha + t * t * beta_coef
    im = float(np.max(inside))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def coord_descent_one(a, conv, S, i, t_upper):
    """Optimize single coordinate i with fast closed-form + refinement."""
    n = len(a)
    a_i = a[i]
    M_conv = 2 * n - 1

    k_lo = i
    k_hi = min(M_conv - 1, i + n - 1)

    if k_lo > 0 and k_hi < M_conv - 1:
        outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
    elif k_lo > 0:
        outside_max = float(np.max(conv[:k_lo]))
    elif k_hi < M_conv - 1:
        outside_max = float(np.max(conv[k_hi + 1:]))
    else:
        outside_max = 0.0

    inside_len = k_hi - k_lo + 1
    base = conv[k_lo:k_hi + 1].copy()

    ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
    a_vals = a[ks - i]
    is_2i = (ks == 2 * i)
    old_contrib = 2.0 * a_i * a_vals
    old_contrib[is_2i] = a_i * a_i
    base -= old_contrib

    alpha = 2.0 * a_vals.copy()
    alpha[is_2i] = 0.0
    beta_coef = np.zeros(inside_len)
    beta_coef[is_2i] = 1.0

    S_rest = S - a_i

    # Closed-form attempt
    t = a_i
    t_cf = None
    for _ in range(5):
        inside = base + t * alpha + t * t * beta_coef
        im = float(np.max(inside))
        if outside_max >= im:
            # Outside dominates: objective decreasing in t, try t_upper
            t_cf = t_upper
            break
        k_star = int(np.argmax(inside))
        P = base[k_star]
        Q = alpha[k_star]
        R = beta_coef[k_star]
        denom = 2.0 * R * S_rest - Q
        if abs(denom) < 1e-14:
            t_cf = t
            break
        t_new = (2.0 * P - Q * S_rest) / denom
        t_new = max(0.0, min(t_upper, t_new))
        if abs(t_new - t) < 1e-12:
            t_cf = t_new
            break
        t = t_new
    if t_cf is None:
        t_cf = t

    # Refine with golden section in a bracket around t_cf
    # Evaluate at a few points to find a good bracket
    candidates = [0.0, t_upper, t_cf, max(0.0, t_cf - 0.1 * t_upper), min(t_upper, t_cf + 0.1 * t_upper)]
    best_local = min(candidates, key=lambda t: _1d_obj_val(t, base, alpha, beta_coef, outside_max, S_rest, n))

    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = max(0.0, best_local - 0.3 * t_upper)
    hi = min(t_upper, best_local + 0.3 * t_upper)
    if hi <= lo:
        hi = t_upper
        lo = 0.0

    for _ in range(20):
        if hi - lo < 1e-10 * max(1.0, t_upper):
            break
        m1 = lo + gr * (hi - lo)
        m2 = hi - gr * (hi - lo)
        f1 = _1d_obj_val(m1, base, alpha, beta_coef, outside_max, S_rest, n)
        f2 = _1d_obj_val(m2, base, alpha, beta_coef, outside_max, S_rest, n)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    # Final check against original candidates
    f_opt = _1d_obj_val(t_opt, base, alpha, beta_coef, outside_max, S_rest, n)
    for c in candidates:
        fc = _1d_obj_val(c, base, alpha, beta_coef, outside_max, S_rest, n)
        if fc < f_opt:
            t_opt = c
            f_opt = fc

    # Update
    a[i] = t_opt
    new_S = S_rest + t_opt
    conv[k_lo:k_hi + 1] = base + t_opt * alpha + t_opt * t_opt * beta_coef
    return new_S, t_opt

def block_cd_sweep(a, conv, S, rng, block_size=4, n_blocks=200):

def structured_init(n, kind='gaussian', rng=None):
    """Create structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.arange(n, dtype=np.float64)
    xm = (x - n / 2.0) / (n / 2.0)  # normalize to [-1, 1]

    if kind == 'gaussian':
        sigma = 0.4 + 0.3 * rng.random()
        a = np.exp(-0.5 * (xm / sigma) ** 2)
    elif kind == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(np.pi * xm))
    elif kind == 'polynomial':
        deg = 2 + int(3 * rng.random())
        a = (1.0 - xm ** 2) ** deg
    elif kind == 'uniform':
        a = np.ones(n)
    elif kind == 'u_shaped':
        a = 0.5 + 0.5 * xm ** 2
    elif kind == 'bimodal':
        d = 0.2 + 0.3 * rng.random()
        a = np.exp(-2.0 * (xm - d) ** 2 / 0.1) + np.exp(-2.0 * (xm + d) ** 2 / 0.1)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a / a.sum() * n  # normalize so mean is ~1
    return a


def targeted_perturb_v2(a, rng, strength=0.03):
    """Improved targeted perturbation: reduce peak contributors, add to low contributors."""
    n = len(a)
    c = fast_conv(a)
    # Find top 3 peak lags
    peak_lags = np.argsort(c)[::-1][:3]

    contrib = np.zeros(n)
    for k_star in peak_lags:
        k_star = int(k_star)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]

    # Normalize contribution
    contrib_max = contrib.max()
    if contrib_max > 0:
        contrib_norm = contrib / contrib_max
    else:
        contrib_norm = np.zeros(n)

    # Reduce high contributors, add to low contributors
    threshold_high = 0.7
    threshold_low = 0.2
    high_mask = contrib_norm > threshold_high
    low_mask = contrib_norm < threshold_low

    total_reduced = 0.0
    a_pert = a.copy()
    n_high = high_mask.sum()
    if n_high > 0:
        for j in np.where(high_mask)[0]:
            reduce_amt = a_pert[j] * strength * (0.5 + 0.5 * rng.random())
            a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
            total_reduced += reduce_amt

    n_low = low_mask.sum()
    if n_low > 0 and total_reduced > 0:
        per = total_reduced / n_low
        a_pert[low_mask] += per

    return np.clip(a_pert, 0.0, 1000.0)


def cosine_perturb(a, rng, n_harmonics=3):
    """Perturb using cosine harmonics to explore structured variations."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for h in range(1, n_harmonics + 1):
        freq = h
        amp = (0.01 + 0.02 * rng.random()) / h
        phase = 2.0 * np.pi * rng.random()
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_pert = a * (1.0 + pert)
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses continuation method + fast coordinate descent + structured perturbations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    if parent is not None:
        p_score = compute_score(parent)
        print(f"Parent score: {p_score:.8f}, length: {len(parent)}")
        best_score = p_score
        best_a = parent.copy()
    else:
        n_init = 2048
        best_a = structured_init(n_init, 'gaussian', rng)
        best_score = compute_score(best_a)
        print(f"Random init score: {best_score:.8f}, length: {n_init}")

    n_work = len(best_a)
    print(f"Working length: {n_work}")

    # ============================================================
    # Phase 0: Try structured initializations (quick, 2% of time)
    # ============================================================
    phase0_deadline = t_start + 0.02 * total_budget
    print(f"\n=== Phase 0: Structured Initializations ===")

    init_kinds = ['gaussian', 'raised_cosine', 'polynomial', 'bimodal', 'u_shaped']
    for kind in init_kinds:
        if time.monotonic() > phase0_deadline:
            break
        for _ in range(2):
            a_try = structured_init(n_work, kind, rng)
            # Quick few sweeps
            a_try = np.clip(a_try, 0.0, 1000.0)
            for _ in range(3):
                sc_t, a_try = coordinate_descent_sweep_fast(a_try, rng)
            sc_t = compute_score(a_try)
            if sc_t < best_score:
                best_score = sc_t
                best_a = a_try.copy()
                print(f"  Init '{kind}': {sc_t:.8f} (NEW BEST)")
            else:
                print(f"  Init '{kind}': {sc_t:.8f}")

    # ============================================================
    # Phase 1: Continuation method (40% of time)
    # Start with smooth LSE, gradually sharpen
    # ============================================================
    phase1_deadline = t_start + 0.42 * total_budget
    print(f"\n=== Phase 1: Continuation Method ({phase1_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    betas = [5, 15, 50, 150, 500, 1500, 4000, 10000]

    for bi, beta in enumerate(betas):
        if time.monotonic() > phase1_deadline:
            break
        print(f"  Beta={beta} (step {bi+1}/{len(betas)})")

        # LSE step (fewer iterations for low beta, more for high)
        if beta <= 50:
            maxit = 2000
        elif beta <= 500:
            maxit = 3000
        else:
            maxit = 5000

        t0 = time.monotonic()
        sc_lb, a_opt = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxit * 20)
        elapsed = time.monotonic() - t0
        if sc_lb < best_score:
            best_score = sc_lb
            best_a = a_opt.copy()
            a_cur = a_opt.copy()
            print(f"    LSE: {sc_lb:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            a_cur = a_opt.copy()
            print(f"    LSE: {sc_lb:.8f} ({elapsed:.1f}s)")

        # Coordinate descent sweeps after LSE
        n_sweeps_target = 15 if beta < 100 else 8
        for sw in range(n_sweeps_target):
            if time.monotonic() > phase1_deadline:
                break
            if sw % 3 == 0:
                sc_cd, a_cur, _, _ = coordinate_descent_sweep_hot(a_cur, None, None, rng)
            else:
                sc_cd, a_cur = coordinate_descent_sweep_fast(a_cur, rng)
            if sc_cd < best_score:
                best_score = sc_cd
                best_a = a_cur.copy()
                if sw == 0 or sw % 5 == 0:
                    print(f"    CD sweep {sw}: {sc_cd:.8f} (NEW BEST)")

    print(f"Phase 1 done: best={best_score:.8f}")

    # ============================================================
    # Phase 2: Intensive coordinate descent (30% of time)
    # ============================================================
    phase2_deadline = t_start + 0.72 * total_budget
    print(f"\n=== Phase 2: Intensive CD ({phase2_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    prev_sc = best_score
    no_improve = 0
    sweep_count = 0

    while time.monotonic() < phase2_deadline:
        sweep_count += 1
        # Mix strategies
        strategy = sweep_count % 5
        if strategy == 0:
            sc, a_cur, _, _ = coordinate_descent_sweep_hot(a_cur, None, None, rng)
        elif strategy == 3:
            # Block CD
            a_cur_full = a_cur
            conv_c = fast_conv(a_cur)
            S_c = float(a_cur.sum())
            sc, a_cur, _, _ = block_cd_sweep(a_cur, conv_c, S_c, rng, block_size=4, n_blocks=300)
            sc = compute_score(a_cur)
        else:
            sc, a_cur = coordinate_descent_sweep_fast(a_cur, rng)

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_cur.copy()
            no_improve = 0
            if sweep_count <= 10 or sweep_count % 30 == 0:
                print(f"  Sweep {sweep_count}: {sc:.8f} (NEW BEST)")
        else:
            no_improve += 1

        if sweep_count % 20 == 0:
            print(f"  Sweep {sweep_count}: {sc:.8f} (no_improve={no_improve})")

        # Perturb if stuck
        if no_improve >= 30:
            pert_strength = 0.01 + 0.02 * rng.random()
            if rng.random() < 0.5:
                a_cur = targeted_perturb_v2(a_cur, rng, strength=pert_strength)
            else:
                a_cur = cosine_perturb(a_cur, rng, n_harmonics=2 + int(3 * rng.random()))
            no_improve = 0
            print(f"  Perturbed (sweep {sweep_count})")

    print(f"Phase 2 done: {sweep_count} sweeps, best={best_score:.8f}")

    # ============================================================
    # Phase 3: LSE refinement at high beta (15% of time)
    # ============================================================
    phase3_deadline = t_start + 0.87 * total_budget
    print(f"\n=== Phase 3: High-beta LSE Refinement ===")

    for beta in [8000, 20000, 50000]:
        if time.monotonic() > phase3_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  LSE beta={beta}: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            print(f"  LSE beta={beta}: {sc:.8f} ({elapsed:.1f}s)")
        # Quick CD after LSE
        for _ in range(5):
            if time.monotonic() > phase3_deadline:
                break
            sc2, a_opt = coordinate_descent_sweep_fast(a_opt, rng)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_opt.copy()
                print(f"  CD polish: {sc2:.8f} (NEW BEST)")
                break

    # ============================================================
    # Phase 4: Diversification + final search (7% of time)
    # ============================================================
    phase4_deadline = deadline
    print(f"\n=== Phase 4: Diversification ===")

    for trial in range(40):
        if time.monotonic() > phase4_deadline - 5.0:
            break

        strategy = trial % 6
        n_cur = len(best_a)

        if strategy == 0:
            # Gaussian noise
            scale = 0.01 + 0.02 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_cur))
        elif strategy == 1:
            # Targeted
            a_pert = targeted_perturb_v2(best_a, rng, strength=0.01 + 0.03 * rng.random())
        elif strategy == 2:
            # Cosine perturbation
            a_pert = cosine_perturb(best_a, rng, n_harmonics=4)
        elif strategy == 3:
            # Block zeroing
            block_size = max(20, n_cur // 50)
            start = int(rng.integers(0, max(1, n_cur - block_size)))
            a_pert = best_a.copy()
            saved = a_pert[start:start + block_size].sum()
            a_pert[start:start + block_size] = 0.0
            mask = np.ones(n_cur, dtype=bool)
            mask[start:start + block_size] = False
            a_pert[mask] += saved / max(1, int(mask.sum()))
        elif strategy == 4:
            # Structured re-init blended with best
            a_struct = structured_init(n_cur, rng.choice(['gaussian', 'raised_cosine', 'bimodal']), rng)
            blend = 0.3 + 0.4 * rng.random()
            a_pert = (1 - blend) * best_a + blend * a_struct
        else:
            # Mirror symmetry perturbation
            a_pert = best_a.copy()
            mid = n_cur // 2
            factor = 0.9 + 0.2 * rng.random()
            a_pert[:mid] *= factor
            rest_sum = best_a[mid:].sum()
            if rest_sum > 0:
                a_pert[mid:] *= 1.0 / factor

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # CD from perturbed point
        for sw in range(10):
            if time.monotonic() > phase4_deadline - 5.0:
                break
            if sw % 4 == 0:
                sc_p, a_pert, _, _ = coordinate_descent_sweep_hot(a_pert, None, None, rng)
            else:
                sc_p, a_pert = coordinate_descent_sweep_fast(a_pert, rng)

        sc_p = compute_score(a_pert)
        if sc_p < best_score:
            best_score = sc_p
            best_a = a_pert.copy()
            print(f"  P4 trial {trial} (s={strategy}): {sc_p:.8f} (NEW BEST)")
        elif trial % 8 == 0:
            print(f"  P4 trial {trial} (s={strategy}): {sc_p:.8f}")

    # ============================================================
    # Final polish
    # ============================================================
    print(f"\n=== Final Polish ===")
    a_cur = best_a.copy()
    for sw in range(30):
        if time.monotonic() > deadline - 5.0:
            break
        if sw % 3 == 0:
            sc, a_cur, _, _ = coordinate_descent_sweep_hot(a_cur, None, None, rng)
        else:
            sc, a_cur = coordinate_descent_sweep_fast(a_cur, rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

    print(f"  After CD polish: {best_score:.8f}")

    # Final LSE attempt
    if time.monotonic() < deadline - 10.0:
        sc, a_opt = run_lbfgs(best_a, 30000, maxiter=3000, maxfun=60000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            # One more CD
            sc2, a_opt = coordinate_descent_sweep_fast(a_opt, rng)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_opt.copy()
            print(f"  Final LSE+CD: {best_score:.8f} (NEW BEST)")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
