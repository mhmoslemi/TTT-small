import numpy as np

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())

def weighted_lag_obj_grad(a, K, alpha):
    """
    Objectifive based on weighted sum of top-K convolution lags with soft
    exponential weighting. More focused than LSE, less noisy than hard max.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_sorted = np.sort(c)[::-1]

        a = a_new
    return best_sc, best_a


def adaptive_lag_optimize(a0, n_iters, K, alpha, maxiter_per_round, deadline):
    """
    Iteratively optimize using weighted top-K lag objective,
    updating K and alpha each round.
    """
    a = a0.copy()
    best_sc = compute_score(a)
    best_a = a.copy()
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        sc, a_opt = run_lbfgs_obj(a, lambda x: weighted_lag_obj_grad(x, K, alpha),
                                   maxiter=maxiter_per_round, maxfun=maxiter_per_round * 3)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
        a = a_opt
        # Adapt K: increase as we get closer to convergence
        K = min(K + 2, 50)
        alpha = min(alpha * 1.1, 20.0)
    return best_sc, best_a


def multi_beta_continuation(a0, deadline, betas=None):
    """
    Run L-BFGS-B with increasing beta values for continued refinement.
    """
    if betas is None:
        betas = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000]
    a = a0.copy()
    best_sc = compute_score(a)
    best_a = a.copy()
    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        maxiter = max(500, min(20000, int(remaining / 8.0)))
        sc, a_opt = run_lbfgs_obj(a, lambda x: lse_obj_and_grad(x, beta),
                                   maxiter=maxiter, maxfun=maxiter * 2)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    LSE beta={beta}: {sc:.8f}", flush=True)
        a = a_opt
    return best_sc, best_a


def pnorm_optimize(a0, deadline, p_values=None):
    """Optimize using p-norm objective with increasing p."""

    return best_sc, best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].astype(np.float64).copy())

    # Structured initializations
    inits.append(np.ones(n) * 0.5)
    inits.append(np.clip(0.5 + 0.3 * np.cos(2.0 * np.pi * x), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.3 * np.sin(2.0 * np.pi * x), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * np.cos(4.0 * np.pi * x), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.5 * np.cos(np.pi * x) ** 2, 0.01, 1000.0))
    inits.append(np.clip(0.2 + 0.6 * (1.0 - (2.0 * x - 1.0) ** 2), 0.01, 1000.0))
    # Multi-frequency
    inits.append(np.clip(0.5 + 0.15 * np.cos(2*np.pi*3*x) + 0.15*np.cos(2*np.pi*7*x + 0.5), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.2 * np.sin(2*np.pi*5*x + 1.0), 0.01, 1000.0))
    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.4 + 0.3 * rng.random(n), 0.01, 1000.0))
    # Block structure
    for nb in (16, 32, 64):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a_blk = np.repeat(vals, per)
        if len(a_blk) < n:
            a_blk = np.concatenate([a_blk, vals[:n - len(a_blk)]])
        inits.append(a_blk[:n].copy())
    return inits

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase approach: LSE continuation -> adaptive lag weighting ->
    p-norm refinement -> peak-shaving subgradient -> final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent solution
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)", flush=True)
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        pa = parent.size
        lengths.append(pa)
        for dl in [-512, -256, 256, 512]:
            cand = pa + dl
            if cand >= 2048 and cand not in lengths:
                lengths.append(cand)
    # Always try a couple of standard lengths
    for n_std in [4096, 8192]:
        if n_std not in lengths:
            lengths.append(n_std)
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths to try: {lengths}", flush=True)
    print(f"Parent: {parent is not None}, len={parent.size if parent is not None else 0}", flush=True)

    # =========================================================================
    # Phase 1: Multi-start LSE continuation (50% of time)
    # =========================================================================
    phase1_deadline = t_start + 0.50 * (total_budget - 20.0)
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Multi-start LSE continuation", flush=True)
    print(f"{'='*60}", flush=True)

    # Sort lengths: prioritize parent length first
    lengths_sorted = sorted(lengths, key=lambda n: (abs(n - (parent.size if parent is not None else 6962)), n))

    for n in lengths_sorted:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n--- n = {n} ---", flush=True)
        inits = build_initializations(n, rng, parent)
        remaining = phase1_deadline - time.monotonic()
        max_inits = max(2, min(len(inits), int(remaining / 60.0)))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            sc0 = compute_score(a0)
            print(f"  init {i}: score={sc0:.6f}", flush=True)
            # Run continuation with remaining time budget
            sub_deadline = min(phase1_deadline, time.monotonic() + remaining)
            sc, a_opt = multi_beta_continuation(a0, sub_deadline)
            report(sc, a_opt, f"n={n}, init={i}")
            # Keep the optimized result for this init as potential parent for next
            if i == 0:
                # Use first (parent-based) init result for subsequent inits at this length
                inits[i + 1:] = [x for x in inits[i + 1:]]  # no change, just keeping structure

    print(f"\nAfter Phase 1: best = {best_score:.8f}", flush=True)

    # =========================================================================
    # Phase 2: Adaptive critical-lag optimization (20% of time)
    # =========================================================================
    phase2_deadline = t_start + 0.70 * (total_budget - 20.0)
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Adaptive critical-lag optimization", flush=True)
    print(f"{'='*60}", flush=True)

    a_cur = best_a.copy()
    for round_num in range(6):
        if time.monotonic() > phase2_deadline:
            break
        sc, a_opt = adaptive_lag_optimize(
            a_cur, n_iters=3, K=10 + round_num * 3, alpha=3.0,
            maxiter_per_round=2000, deadline=phase2_deadline
        )
        if sc < best_score:
            report(sc, a_opt, f"adaptive lag round {round_num}")
        a_cur = a_opt
    print(f"\nAfter Phase 2: best = {best_score:.8f}", flush=True)

    # =========================================================================
    # Phase 3: p-norm refinement (15% of time)
    # =========================================================================
    phase3_deadline = t_start + 0.85 * (total_budget - 20.0)
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: p-norm refinement", flush=True)
    print(f"{'='*60}", flush=True)

    a_cur = best_a.copy()
    sc, a_opt = pnorm_optimize(a_cur, phase3_deadline)
    report(sc, a_opt, "p-norm")
    a_cur = a_opt

    # =========================================================================
    # Phase 4: Peak-shaving subgradient + final LSE polish (15% of time)
    # =========================================================================
    phase4_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Peak shaving + final polish", flush=True)
    print(f"{'='*60}", flush=True)

    # Subgradient peak shaving
    if time.monotonic() < phase4_deadline - 30:
        n_iters_sg = max(500, int((phase4_deadline - time.monotonic() - 30) / 0.01))
        n_iters_sg = min(n_iters_sg, 20000)
        sc, a_opt = peak_shave_subgrad(best_a, n_iters_sg)
        report(sc, a_opt, "peak shave subgrad")
        a_cur = a_opt

    # Final high-beta LSE
    if time.monotonic() < phase4_deadline - 10:
        remaining = phase4_deadline - time.monotonic() - 5
        maxiter = max(1000, int(remaining / 5.0))
        sc, a_opt = run_lbfgs_obj(a_cur, lambda x: lse_obj_and_grad(x, 15000),
                                   maxiter=maxiter, maxfun=maxiter * 2)
        report(sc, a_opt, "final LSE beta=15000")

    # One more subgradient round
    if time.monotonic() < phase4_deadline - 5:
        sc, a_opt = peak_shave_subgrad(best_a, 5000)
        report(sc, a_opt, "final subgrad")

    # =========================================================================
    # Final: Try a quick restart from best with different perturbation
    # =========================================================================
    if time.monotonic() < deadline - 5:
        n_best = len(best_a)
        a_pert = best_a * (1.0 + 0.01 * rng.standard_normal(n_best))
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        remaining = deadline - time.monotonic() - 3
        maxiter = max(500, int(remaining / 8.0))
        sc, a_opt = run_lbfgs_obj(a_pert, lambda x: lse_obj_and_grad(x, 10000),
                                   maxiter=maxiter, maxfun=maxiter * 2)
        report(sc, a_opt, "restart perturb")

    # =========================================================================
    # Final verification
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}", flush=True)
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s", flush=True)
    print(f"{'='*60}", flush=True)
    return out
