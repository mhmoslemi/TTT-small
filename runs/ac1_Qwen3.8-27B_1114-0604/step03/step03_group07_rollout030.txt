"""
Refined LSE-continuation optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: finer beta continuation, more iterations at high beta,
multi-peak subgradient handling, spectral pre-conditioning, and adaptive re-heating.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    # Gradient: d(L)/da[j] = sum_k p[k] * d(c[k])/da[j]
    # d(c[k])/da[j] = a[k-j] + a[j-k] (valid indices)
    # This is the cross-correlation of a and p
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapper(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-16,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
        )
    except Exception:
        return compute_score(a0), a0, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out, res.success


def subgradient_polish(a, max_iters=5000, tol=1e-10):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc - tol:
            best_sc = sc
            best_a = a.copy()
        
        # Find active lags (within 0.5% of max)
        threshold = M * 0.995
        active = np.where(c >= threshold)[0]
        
        # Compute subgradient: weighted average of gradients at active lags
        # Use softmax weights for smoothness
        active_c = c[active]
        w = np.exp(100 * (active_c - active_c.max()))
        w /= w.sum()
        
        # Subgradient of max(c) w.r.t. a
        sg = np.zeros(n)
        for m_idx, k in enumerate(active):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w[m_idx] * 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        # Polyak step size with damping
        sc_lb = 1.414  # theoretical lower bound (sqrt(2))
        step = (sc - sc_lb) / gn2
        step = min(step, 1.0 / (1.0 + 0.01 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        
        if it % 1000 == 0 and it > 0:
            print(f"    Subgrad iter {it}: score={sc:.8f}, step={step:.2e}")
    
    return best_sc, best_a


def spectral_reshaping(a, target_flatness=0.3):
    """
    Reshape the power spectrum to be flatter, then project to non-negative.
    """
    n = len(a)
    M_fft = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    P = np.abs(A) ** 2
    # Attenuate peaks in the power spectrum
    P_mean = P.sum() / len(P)
    # Reduce peaks above mean
    P_flat = np.minimum(P, P_mean * (1 + target_flatness))
    # Preserve the DC component
    P[0] = original_dc = P[0]  # DC is (sum a)^2, we want to preserve it
    P_flat[0] = original_dc
    # Reconstruct amplitude (preserve phase)
    A_new = np.sqrt(P_flat) * np.exp(1j * np.angle(A))
    a_new = np.real(np.fft.irfft(A_new, M_fft))[:n]
    # Project to non-negative
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-6:
        return a.copy()
    return a_new

"""
Refined LSE-continuation optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key improvements: finer beta continuation schedule, more iterations at high beta,
multi-peak subgradient handling, spectral pre-conditioning, and adaptive re-heating.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapper(a, beta):
    """Scipy-compatible objective returning (value, gradient)."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective. Returns (score, array, success)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-16,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
        )
    except Exception:
        return compute_score(a0), a0, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out, res.success


def subgradient_polish(a, max_iters=5000):
    """Exact subgradient descent with weighted multi-peak handling."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find active lags with adaptive threshold
        if it < 500:
            threshold = M * 0.99
        else:
            threshold = M * 0.995
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Compute weighted subgradient
        active_c = c[active]
        w = np.exp(200.0 * (active_c - active_c.max()))
        w /= w.sum()

        sg = np.zeros(n)
        for idx, k in enumerate(active):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w[idx] * 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break

        # Adaptive step size
        sc_lb = 1.4142
        step = (sc - sc_lb) / gn2
        damping = 1.0 / (1.0 + 0.005 * it)
        step = min(step, 0.5 * damping)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def spectral_reshaping(a, target_flatness):
    """Reshape power spectrum toward flatness, project to non-negative."""
    n = len(a)
    M_fft = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M_fft)
    P = np.abs(A) ** 2
    P_mean = P.sum() / len(P)
    P_cap = P_mean * (1.0 + target_flatness)
    dc = P[0]
    P_flat = np.minimum(P, P_cap)
    P_flat[0] = dc
    A_new = np.sqrt(P_flat) * np.exp(1j * np.angle(A))
    a_new = np.real(np.fft.irfft(A_new, M_fft))[:n]
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-6:
        return a.copy()
    return a_new


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / max(1, n - 1)

    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())
        inits.append(np.clip(parent + 0.03 * rng.standard_normal(n), 0.01, 1000.0))
        inits.append(np.clip(parent + 0.08 * rng.standard_normal(n), 0.01, 1000.0))

    inits.append(np.ones(n) * 0.5)
    inits.append(np.linspace(0.05, 1.0, n))
    inits.append(np.linspace(1.0, 0.05, n))
    inits.append(0.05 + 0.95 * xn ** 2)
    inits.append(0.05 + 0.95 * (1 - xn) ** 2)

    for freq in [1, 2, 3, 5]:
        inits.append(np.clip(0.5 + 0.3 * np.sin(np.pi * freq * xn), 0.01, 1000.0))

    inits.append(np.clip(0.3 + 0.35 * xn + 0.15 * np.sin(2.0 * np.pi * 11 * xn), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.35 * (1.0 - xn) + 0.15 * np.cos(2.0 * np.pi * 7 * xn), 0.01, 1000.0))

    a_rand = np.clip(0.5 + 0.25 * rng.standard_normal(n), 0.01, 1000.0)
    ksize = max(1, n // 200)
    kernel = np.ones(ksize) / ksize
    a_smooth = np.convolve(a_rand, kernel, mode='same')
    inits.append(np.clip(a_smooth, 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 1000.0))

    return inits


def do_lse_continuation(a_cur, beta_schedule, t_deadline, state, n, label_prefix):
    """Run LSE continuation from a_cur through beta schedule. Updates state."""

            break
        remaining = max(10.0, t_deadline - time.monotonic())
        est_per_iter = 0.002 * n + 0.5
        maxit = max(2000, min(30000, int(remaining / est_per_iter)))
        maxit = min(maxit, 20000)
        sc, a_opt, success = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxit * 25)
        a_cur = a_opt
        elapsed = time.monotonic() - state['t_start']
        if sc < state['best_score']:
            state['best_score'] = sc
            state['best_a'] = a_opt.copy()
            print(f"  [{label_prefix} b={beta}] NEW BEST: {sc:.8f} (n={n}, t={elapsed:.0f}s)")
        else:
            print(f"  [{label_prefix} b={beta}] score={sc:.8f} (no improvement) t={elapsed:.0f}s")
    return a_cur
