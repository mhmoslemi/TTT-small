```python
"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Key insight: By Cauchy-Schwarz, sum(a^2) >= (sum(a))^2/n, so b[0] >= (sum_a)^2/n,
meaning score >= 2*n*(sum_a^2/n)/(sum_a)^2 = 2. Uniform sequences achieve exactly 2.
We verify this and attempt LSE-smoothed optimization to confirm or find edge cases.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_score_a(a):
    """Exact score and subgradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def adam_lse(a_init, lr=0.005, n_steps=50000, beta_lse=100,
             time_limit_s=None, t_start=None):
    """Adam optimizer with LSE-smoothed gradient in a-space."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    a = a / a.sum()  # Normalize to sum=1

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break

        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad_a(a, beta_lse)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        s = a.sum()
        if s > 1e-12:
            a = a / s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def run_lbfgs_a(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B in a-space with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def selective_coord_descent(a, top_k=30, n_sweeps=10, time_limit_s=None, t_start=None):
    """Coordinate descent optimizing top-K coordinates per sweep."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        score, grad = subgrad_score_a(a)
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        for j in order:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if grad[j] > 1e-10:
                candidates = a_old * np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.0])
            elif grad[j] < -1e-10:
                candidates = np.concatenate([[0.0], a_old * np.array([1.5, 2.0, 3.0])])
            else:
                candidates = np.array([a_old])
            candidates = np.clip(candidates, 0.0, 1000.0)
            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a

    n_opt = min(4096, max(1024, n0))
    a_init = np.ones(n_opt) / n_opt

    for beta, lr in [(50, 0.01), (200, 0.005), (500, 0.003)]:
        remaining = effective_budget - (time.monotonic() - t_start)
        if remaining < 10:
            break
        time_alloc = min(remaining * 0.25, 180.0)
        sc, a_opt = adam_lse(a_init, lr=lr, n_steps=100000, beta_lse=beta,
                             time_limit_s=time_alloc, t_start=t_start)
        try_update(sc, a_opt, f"Adam beta={beta}")
        a_init = a_opt

    # Phase 3: L-BFGS polish
    print(f"\n{'='*50}")
    print("Phase 3: L-BFGS polish")
    print(f"{'='*50}")

    for beta in [200, 500, 1000, 2000]:
        remaining = effective_budget - (time.monotonic() - t_start)
        if remaining < 10:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs_a(best_a, beta, maxiter=5000, maxfun=80000)
        try_update(sc, a_opt, f"L-BFGS beta={beta}")
        print(f"    L-BFGS beta={beta}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # Phase 4: Coordinate descent
    print(f"\n{'='*50}")
    print("Phase 4: Coordinate descent")
    print(f"{'='*50}")

    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 10:
        t0 = time.monotonic()
        sc, a_opt = selective_coord_descent(best_a, top_k=30, n_sweeps=10,
                                            time_limit_s=min(remaining * 0.3, 120.0),
                                            t_start=t_start)
        try_update(sc, a_opt, "Coord descent")
        print(f"    Coord descent: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # Phase 5: Try structured sequences (block designs, periodic)
    print(f"\n{'='*50}")
    print("Phase 5: Structured sequences")
    print(f"{'='*50}")

    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 20:
        # Try various structured initializations
        n_struct = 4096
        structures = []

        # Uniform with small perturbation
        structures.append(np.ones(n_struct) / n_struct)

        # Linear ramp (but non-negative)
        x = np.linspace(0.5, 1.5, n_struct)
        structures.append(x / x.sum())

        # Cosine window
        x = np.cos(np.pi * np.arange(n_struct) / (n_struct - 1)) ** 2
        x = np.clip(x, 0.01, None)
        structures.append(x / x.sum())

        # Gaussian-ish
        x = np.exp(-0.5 * ((np.arange(n_struct) - n_struct / 2) / (n_struct / 4)) ** 2)
        structures.append(x / x.sum())

        # Two-lobe
        x = np.exp(-0.5 * ((np.arange(n_struct) - n_struct / 4) / (n_struct / 8)) ** 2) + \
            np.exp(-0.5 * ((np.arange(n_struct) - 3 * n_struct / 4) / (n_struct / 8)) ** 2)
        structures.append(x / x.sum())

        for i, a_s in enumerate(structures):
            sc = compute_score(a_s)
            print(f"  Structure {i}: score={sc:.10f}")
            try_update(sc, a_s, f"structure {i}")
            if time.monotonic() - t_start > effective_budget - 30:
                break

    # Phase 6: Final L-BFGS with highest beta
    print(f"\n{'='*50}")
    print("Phase 6: Final high-beta L-BFGS")
    print(f"{'='*50}")

    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 10:
        for beta in [3000, 5000]:
            if time.monotonic() - t_start > effective_budget - 10:
                break
            sc, a_opt, _ = run_lbfgs_a(best_a, beta, maxiter=8000, maxfun=120000)
            try_update(sc, a_opt, f"final L-BFGS beta={beta}")

    # Final output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [1.0] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
```
