Multi-phase optimizer for minimizing 2n*max(self-conv)/(sum)^2.

def fast_conv(a):

    M_fft = 1 << (M - 1).bit_length()

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())

    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_p_a = np.fft.irfft(RA * P, M2_fft)
    grad_L = 2.0 * conv_p_a[n - 1:2 * n - 1]

    S3 = S2 * S

Multi-phase optimizer for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key techniques: LSE-smoothed Adam with geometric beta annealing, block coordinate L-BFGS
with increasing precision, full L-BFGS polish, and structured perturbations targeting

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_pa = np.fft.irfft(RA * P, M2_fft)
    grad_L = 2.0 * conv_pa[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad

    """Exact score and subgradient w.r.t. a (using all max lags)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    score = 2.0 * n * M / (S * S)
    tol = 1e-10 * M
    max_mask = c >= (M - tol)
    sg = np.zeros(n)
    for k in np.where(max_mask)[0]:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            idx = k - np.arange(js, je + 1)
            sg[js:je + 1] += 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad

    """Perturb by reducing peak-lag contributors and redistributing."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    a_new = a.copy()

    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je > js:
        scale = 1.0 - strength
        a_new[js:je + 1] *= scale
        removed = float((a[js:je + 1] - a_new[js:je + 1]).sum())
        non_contrib = np.concatenate([np.arange(0, js), np.arange(je + 1, n)])
        if len(non_contrib) > 0:
            n_chose = min(100, len(non_contrib))
            chosen = rng.choice(non_contrib, size=n_chose, replace=False)
            a_new[chosen] += removed / n_chose

    return np.clip(a_new, 0.0, 1e6)


def gaussian_perturb(a, rng, scale=0.05):
    """Gaussian perturbation."""
    n = len(a)
    a_new = a * (1.0 + scale * rng.standard_normal(n))
    return np.clip(a_new, 0.0, 1e6)


def block_perturb(a, rng, strength=0.3):
    """Perturb a random block of the sequence."""
    n = len(a)
    a_new = a.copy()
    width = max(10, n // 40)
    center = int(rng.integers(width, n - width))
    a_new[center - width:center + width] *= rng.uniform(1.0 - strength, 1.0 + strength)
    return np.clip(a_new, 0.0, 1e6)


def coord_descent(a, top_k=15, n_sweeps=20):
    """Coordinate descent on top-gradient coordinates."""
    n = len(a)
    a = a.copy()
    best_score = compute_score(a)

    for sweep in range(n_sweeps):
        score, grad = subgrad_score(a)
        if score < best_score:
            best_score = score

        order = np.argsort(np.abs(grad))[::-1][:top_k]
        for j in order:
            j = int(j)
            a_old = a[j]
            if grad[j] > 1e-12:
                candidates = a_old * np.array([0.0, 0.3, 0.5, 0.7, 0.9])
            elif grad[j] < -1e-12:
                candidates = a_old * np.array([1.0, 1.2, 1.5, 2.0, 3.0])
            else:
                continue
            best_val = a_old
            best_sc = score
            for val in candidates:
                val = max(0.0, float(val))
                a[j] = val
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_val = val
            a[j] = best_val

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
    return best_score, a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search: Adam annealing -> Block L-BFGS -> Full L-BFGS
    -> Structured perturbations -> Final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = min(float(budget_s), 2350.0)
    deadline = t_start + total_budget

    # Get parent sequence
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 100:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
    else:
        parent = np.ones(4096) / 4096.0
        parent_score = compute_score(parent)
        print(f"No parent found, using uniform n=4096, score={parent_score:.8f}")

    best_a = parent.copy()
    best_score = parent_score

    def try_update(sc, a, label):
        nonlocal best_a, best_score
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    print(f"Starting optimization. Initial score: {best_score:.8f}")

    # ============================================================
    # Phase 1: LSE-Adam with geometric beta annealing (35% time)
    # ============================================================
    p1_end = t_start + total_budget * 0.35
    print(f"\n{'='*55}")
    print(f"Phase 1: Adam annealing (until t={(p1_end-t_start):.0f}s)")
    print(f"{'='*55}")

    a_norm = best_a / best_a.sum()

    # Beta schedule: geometric, from smooth to sharp
    beta_schedule = [10, 30, 100, 300, 1000, 3000, 10000, 30000]
    # Give more steps to later (sharper) betas
    for beta in beta_schedule:
        if time.monotonic() > p1_end:
            break
        time_left = p1_end - time.monotonic()
        if time_left < 5:
            break
        # Adaptive: more time for higher beta
        frac = 0.12 if beta < 100 else (0.15 if beta < 1000 else 0.18)
        alloc = min(time_left, total_budget * frac)
        sc, a_opt = Adam_anneal(a_norm, [beta], 100000, lr_base=0.01,
                                time_budget=alloc)
        if sc < best_score:
            try_update(sc, a_opt, f"Adam beta={beta}")
            a_norm = a_opt
        else:
            a_norm = a_opt  # Continue from current state even if not best

    print(f"  After Phase 1: best={best_score:.8f}")

    # ============================================================
    # Phase 2: Block L-BFGS with increasing beta (30% time)
    # ============================================================
    p2_end = t_start + total_budget * 0.65
    print(f"\n{'='*55}")
    print(f"Phase 2: Block L-BFGS (until t={(p2_end-t_start):.0f}s)")
    print(f"{'='*55}")

    n_best = len(best_a)

    # Multiple passes with different block sizes and betas
    block_configs = [
        (50, 500),
        (50, 2000),
        (100, 1000),
        (30, 5000),
        (100, 5000),
    ]

    for bs, beta in block_configs:
        if time.monotonic() > p2_end - 10:
            break
        time_left = p2_end - time.monotonic()
        if time_left < 15:
            break

        # Do block L-BFGS pass with time monitoring
        a_work = best_a.copy()
        n_blocks = (n_best + bs - 1) // bs
        # Estimate time per block
        time_per_block = time_left / (n_blocks + 1)

        rng_block = np.random.default_rng(int(time.monotonic() * 10000) % 2**31)
        order = list(range(n_blocks))
        rng_block.shuffle(order)

        bi = 0
        while bi < len(order) and time.monotonic() < p2_end - 5:
            block_idx = order[bi]
            start = block_idx * bs
            end = min(start + bs, n_best)
            if end - start < 2:
                bi += 1
                continue

            a_local = a_work.copy()
            x0 = a_local[start:end].copy()
            try:
                res = minimize(
                    block_obj_grad, x0,
                    args=(a_local, start, end, beta),
                    method='L-BFGS-B', jac=True,
                    bounds=[(0.0, 1e6)] * (end - start),
                    options={'maxiter': 150, 'maxfun': 400,
                             'ftol': 1e-14, 'gtol': 1e-10}
                )
                a_work[start:end] = np.clip(res.x, 0.0, 1e6)
            except Exception:
                pass
            bi += 1

            # Check score periodically
            if bi % 50 == 0:
                sc = compute_score(a_work)
                if sc < best_score:
                    try_update(sc, a_work, f"BlockL({bs},{beta}) blk={bi}")

        sc = compute_score(a_work)
        if sc < best_score:
            try_update(sc, a_work, f"BlockL({bs},{beta})")
        else:
            # Keep the block-optimized version for next pass even if not new best
            best_score_local = compute_score(a_work)
            if best_score_local < best_score + 1e-6:
                best_a = a_work.copy()

    print(f"  After Phase 2: best={best_score:.8f}")

    # ============================================================
    # Phase 3: Full L-BFGS polish (15% time)
    # ============================================================
    p3_end = t_start + total_budget * 0.80
    print(f"\n{'='*55}")
    print(f"Phase 3: Full L-BFGS (until t={(p3_end-t_start):.0f}s)")
    print(f"{'='*55}")

    for beta in [3000, 10000, 30000]:
        if time.monotonic() > p3_end - 10:
            break
        if len(best_a) > 15000:
            # Too large for full L-BFGS to be effective in reasonable time
            break
        t0 = time.monotonic()
        sc, a_opt = full_lbfgs(best_a, beta, maxiter=3000, maxfun=50000)
        elapsed = time.monotonic() - t0
        print(f"    Full L-BFGS beta={beta}: {sc:.8f} ({elapsed:.1f}s)")
        try_update(sc, a_opt, f"FullLBFGS beta={beta}")

    print(f"  After Phase 3: best={best_score:.8f}")

    # ============================================================
    # Phase 4: Structured perturbations + quick opt (10% time)
    # ============================================================
    p4_end = t_start + total_budget * 0.92
    print(f"\n{'='*55}")
    print(f"Phase 4: Perturbations (until t={(p4_end-t_start):.0f}s)")
    print(f"{'='*55}")

    n_restarts = 15
    for r in range(n_restarts):
        if time.monotonic() > p4_end - 20:
            break

        strategy = r % 3
        if strategy == 0:
            a_pert = structured_perturb(best_a, rng, strength=0.1 + 0.05 * rng.random())
        elif strategy == 1:
            a_pert = gaussian_perturb(best_a, rng, scale=0.03 + 0.04 * rng.random())
        else:
            a_pert = block_perturb(best_a, rng, strength=0.2 + 0.2 * rng.random())

        # Quick Adam refinement
        time_left = p4_end - time.monotonic()
        if time_left < 10:
            break
        adam_time = min(15.0, time_left * 0.4)
        a_pert_norm = a_pert / a_pert.sum() if a_pert.sum() > 1e-12 else a_pert
        sc, a_opt = Adam_anneal(a_pert_norm, [500], 5000, lr_base=0.005,
                                time_budget=adam_time)

        # Quick L-BFGS if time allows
        if time.monotonic() < p4_end - 15 and len(a_opt) <= 15000:
            sc2, a_opt2 = full_lbfgs(a_opt, 5000, maxiter=1000, maxfun=20000)
            if sc2 < sc:
                sc, a_opt = sc2, a_opt2

        try_update(sc, a_opt, f"pert#{r} s{strategy}")

    print(f"  After Phase 4: best={best_score:.8f}")

    # ============================================================
    # Phase 5: Final polish (8% time)
    # ============================================================
    p5_end = deadline
    print(f"\n{'='*55}")
    print(f"Phase 5: Final Polish")
    print(f"{'='*55}")

    # High-beta L-BFGS
    if time.monotonic() < p5_end - 30 and len(best_a) <= 15000:
        sc, a_opt = full_lbfgs(best_a, 50000, maxiter=5000, maxfun=100000)
        try_update(sc, a_opt, "Final LBFGS high-beta")

    # Coordinate descent
    if time.monotonic() < p5_end - 15:
        sc, a_opt = coord_descent(best_a, top_k=20, n_sweeps=30)
        try_update(sc, a_opt, "Final coord descent")

    # One more short Adam at very high beta
    if time.monotonic() < p5_end - 10:
        time_left = p5_end - time.monotonic()
        sc, a_opt = Adam_anneal(best_a, [50000], 50000, lr_base=0.001,
                                time_budget=max(5.0, time_left - 5))
        try_update(sc, a_opt, "Final Adam high-beta")

    # ============================================================
    # Try a different length (if time permits)
    # ============================================================
    if time.monotonic() < p5_end - 30:
        n_cur = len(best_a)
        for factor in [1.3, 0.7]:
            n_new = int(n_cur * factor)
            if n_new < 512 or n_new > 20000:
                continue
            if time.monotonic() > p5_end - 15:
                break
            a_res = resample_seq(best_a, n_new)
            sc, a_opt = Adam_anneal(a_res, [1000], 5000, lr_base=0.005,
                                    time_budget=10.0)
            if time.monotonic() < p5_end - 10:
                sc2, a_opt2 = full_lbfgs(a_opt, 5000, maxiter=2000, maxfun=30000)
                if sc2 < sc:
                    sc, a_opt = sc2, a_opt2
            try_update(sc, a_opt, f"resample n={n_new}")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
        best_score = float(evaluate_sequence(out))

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*55}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*55}")
    return out
