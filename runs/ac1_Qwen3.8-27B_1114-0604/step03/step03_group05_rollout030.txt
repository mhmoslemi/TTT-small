"""
Step function optimization using Adam with LSE continuation, multi-scale approach,
and spectral flattening. Target: beat 1.5027.
"""

import numpy as np
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def adam_optimize(a0, beta, max_iters, lr=0.01, verbose=False, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer for LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 1e-6:
        a[:] = 0.5
    
    m = np.zeros(n)
    v = np.zeros(n)
    best_score = compute_score(a)
    best_a = a.copy()
    
    for t in range(1, max_iters + 1):
        score, grad = lse_score_grad(a, beta)
        
        if not np.isfinite(score):
            break
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Learning rate with decay
        lr_t = lr / (1 + 0.001 * t)
        
        a -= lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)
        
        # Check score
        exact_score = compute_score(a)
        if exact_score < best_score:
            best_score = exact_score
            best_a = a.copy()
    
    return best_score, best_a

def spectral_flatten(a, iters=100, verbose=False):
    """Iteratively flatten the power spectrum."""
    n = len(a)
    a = a.astype(np.float64).copy()
    N = 1 << (2 * n - 2).bit_length()
    best_score = compute_score(a)
    best_a = a.copy()
    
    for t in range(iters):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Flatten: reduce high magnitudes, increase low ones
        mag_mean = mag.mean()
        # Soft flattening: mag_new = mag * (mag_mean / mag)^alpha
        alpha = 0.1
        mag_new = mag * (mag_mean / (mag + 1e-10))**alpha
        mag_new = np.maximum(mag_new, 0)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        if a_new.sum() < 1e-12:
            break
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # If it got worse, try smaller alpha
            a = a_new.copy() * 0.5 + best_a * 0.5
            a = np.clip(a, 0.0, 1000.0)
    
    return best_score, best_a

def peak_shave(a, iters=2000):
    """Subgradient descent on max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        sc = 2.0 * n * M / (S * S)
        sc_lb = 1.0
        step = (sc - sc_lb) / (gn * gn + 1e-20)
        step = min(step, 1.0 / S)
        step = max(step, 1e-8 / (gn * S + 1e-20))
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

"""
Step function optimization using Adam + LSE continuation with spectral flattening
and peak shaving. Targets beating 1.5027 by exploiting longer sequences (n=16384),
better adaptive optimization (Adam), and frequency-domain post-processing.
"""

import numpy as np
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    Returns (score, grad) where score ≈ 2n*LSE/c^2 and grad is the gradient.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    # Gradient via convolution
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1: 2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def run_adam(a0, beta, max_iters, lr_init=0.01, decay=0.999, beta1=0.9, beta2=0.999, eps=1e-8, normalize=True):
    """Adam optimizer for LSE-smoothed objective with optional normalization."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 1e-6:
        a[:] = 1.0 / n

    m = np.zeros(n)
    v = np.zeros(n)
    best_score = compute_score(a)
    best_a = a.copy()
    lr = lr_init

    for t in range(1, max_iters + 1):
        score, grad = lse_score_grad(a, beta)
        if not np.isfinite(score):
            break

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)

        if normalize:
            S = a.sum()
            if S > 1e-12:
                a = a / S * a0.sum()

        lr *= decay

        if t % 500 == 0 or t == 1:
            exact = compute_score(a)
            if exact < best_score:
                best_score = exact
                best_a = a.copy()

    # Final check
    exact = compute_score(a)
    if exact < best_score:
        best_score = exact
        best_a = a.copy()

    return best_score, best_a

def spectral_flatten(a, iters=500, alpha_init=0.3, alpha_decay=0.999):
    """
    Iteratively flatten the power spectrum using Gerchberg-Saxton-like approach.
    Reduces peaks in |FFT(a)|^2 to reduce autocorrelation peaks.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    N = 1 << (2 * n - 2).bit_length()
    S_target = a.sum()

    best_score = compute_score(a)
    best_a = a.copy()
    alpha = alpha_init
    no_improve = 0

    for t in range(iters):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        mag_sq = mag * mag

        # Compute target: geometric mean of magnitudes (smooths out peaks)
        log_mag = np.log(mag_sq + 1e-20)
        # Apply a smoothed version: blend with uniform
        target_log = (1.0 - alpha) * log_mag + alpha * np.mean(log_mag)
        target_mag = np.exp(0.5 * target_log)

        A_new = target_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.clip(a_new, 0.0, 1000.0)

        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new * (S_target / S)

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            a = 0.7 * best_a + 0.3 * a_new
            a = np.clip(a, 0.0, 1000.0)
            a = a * (S_target / a.sum())
            no_improve += 1
            if no_improve > 50:
                alpha *= alpha_decay
                if alpha < 0.001:
                    break

    return best_score, best_a

def peak_shave(a, iters=3000):
    """Subgradient descent on max-conv objective with adaptive step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S_target = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        sc = 2.0 * n * M / (S * S)
        # Polyak step with target
        target = best_sc * 1.001
        step = max(0, (sc - target)) / (gn * gn + 1e-20)
        step = min(step, 2.0 / (S * gn + 1e-20))
        step = max(step, 1e-8)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

        # Normalize
        S_new = a.sum()
        if S_new > 1e-12:
            a = a * (S_target / S_new)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def multi_peak_shave(a, iters_per_round=200, num_rounds=5):

            sg = np.zeros(n)
            if j_end >= j_start:
                sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

            S = float(a.sum())
            S2 = S * S
            S3 = S2 * S
            grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
            gn = float(np.linalg.norm(grad))
            if gn < 1e-15:
                continue

            step = 0.01 / (gn * S + 1e-20)
            for _ in range(iters_per_round // num_rounds):
                a -= step * grad
                a = np.clip(a, 0.0, 1000.0)
                S_new = a.sum()
                if S_new > 1e-12:
                    a = a * (S_target / S_new)

                c = fast_conv(a)
                S = float(a.sum())
                M = float(c.max())
                sc = 2.0 * n * M / (S * S)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()

                k_star_actual = int(np.argmax(c))
                j_start = max(0, k_star_actual - n + 1)
                j_end = min(k_star_actual, n - 1)
                sg = np.zeros(n)
                if j_end >= j_start:
                    sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
                S2 = S * S
                S3 = S2 * S
                grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
                gn = float(np.linalg.norm(grad))
                if gn < 1e-15:
                    break
                step = min(0.01 / (gn * S + 1e-20), 1.0 / S)

    return best_sc, best_a


def upsample(a, target_n):
    """Upsample a sequence to target_n using cosine interpolation."""
    n = len(a)
    if n == target_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, target_n)
    # Use spline-like interpolation for smoothness
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Load parent solution, upsample to n=16384
    2. Adam + LSE continuation (main workhorse)
    3. Spectral flattening (frequency domain)
    4. Peak shaving (subgradient)
    5. Final polish and restarts
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    best_score = np.inf
    best_a = None

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # Target length
    n_target = 16384

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={compute_score(parent):.8f}")
        a_work = upsample(parent, n_target)
    else:
        print("No parent available, using random init")
        a_work = 0.5 + 0.1 * rng.standard_normal(n_target)
        a_work = np.clip(a_work, 0.01, None)

    a_work = np.clip(a_work, 0.0, 1000.0)
    sc = compute_score(a_work)
    best_score = sc
    best_a = a_work.copy()
    print(f"Initial: n={n_target}, score={sc:.8f}")

    # ===================================================================
    # Phase 1: Adam + LSE continuation (55% of time)
    # ===================================================================
    phase1_end = t_start + 0.55 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Adam + LSE (budget: {phase1_end - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    beta_schedule = [10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]

    for beta in beta_schedule:
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        remaining = phase1_end - time.monotonic()
        # Estimate iterations: ~0.5ms per iter
        max_iters = int(remaining * 0.8 / 0.0005)
        max_iters = max(500, min(50000, max_iters))

        lr = 0.01 if beta < 100 else 0.005 if beta < 1000 else 0.002

        sc, a_opt = run_adam_nosmooth(a_cur, beta, max_iters=max_iters, lr_init=lr)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  beta={beta}: NEW BEST {sc:.8f} (iters={max_iters}, t={time.monotonic()-t0:.1f}s)")
        else:
            print(f"  beta={beta}: score={sc:.8f}, best={best_score:.8f} (t={time.monotonic()-t0:.1f}s)")
        a_cur = a_opt

    # ===================================================================
    # Phase 2: Spectral flattening (15% of time)
    # ===================================================================
    phase2_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Spectral Flattening (budget: {phase2_end - t_start:.0f}s) ===")

    if time.monotonic() < phase2_end and best_a is not None:
        t0 = time.monotonic()
        max_iters = int((phase2_end - time.monotonic()) * 30)
        max_iters = max(100, min(3000, max_iters))
        sc, a_flat = spectral_flatten(best_a, iters=max_iters, alpha_init=0.2, alpha_decay=0.998)
        if sc < best_score:
            best_score = sc
            best_a = a_flat.copy()
            print(f"  Spectral flatten: NEW BEST {sc:.8f} (t={time.monotonic()-t0:.1f}s)")
        else:
            print(f"  Spectral flatten: score={sc:.8f}, best={best_score:.8f} (t={time.monotonic()-t0:.1f}s)")

        # Try v2
        if time.monotonic() < phase2_end:
            t0 = time.monotonic()
            max_iters2 = int((phase2_end - time.monotonic()) * 30)
            max_iters2 = max(50, min(2000, max_iters2))
            sc, a_flat2 = spectral_flatten_v2(best_a, iters=max_iters2)
            if sc < best_score:
                best_score = sc
                best_a = a_flat2.copy()
                print(f"  Spectral flatten v2: NEW BEST {sc:.8f} (t={time.monotonic()-t0:.1f}s)")
            else:
                print(f"  Spectral flatten v2: score={sc:.8f}, best={best_score:.8f} (t={time.monotonic()-t0:.1f}s)")

    # ===================================================================
    # Phase 3: Peak shaving (15% of time)
    # ===================================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Peak Shaving (budget: {phase3_end - t_start:.0f}s) ===")

    if time.monotonic() < phase3_end and best_a is not None:
        t0 = time.monotonic()
        iters = int((phase3_end - time.monotonic()) * 80)
        iters = max(1000, min(20000, iters))
        sc, a_polished = peak_shave(best_a, iters=iters)
        if sc < best_score:
            best_score = sc
            best_a = a_polished.copy()
            print(f"  Peak shave: NEW BEST {sc:.8f} (iters={iters}, t={time.monotonic()-t0:.1f}s)")
        else:
            print(f"  Peak shave: score={sc:.8f}, best={best_score:.8f} (t={time.monotonic()-t0:.1f}s)")

        # Multi-peak shave
        if time.monotonic() < phase3_end:
            t0 = time.monotonic()
            sc, a_mp = multi_peak_shave(best_a, iters_per_round=200, num_rounds=5)
            if sc < best_score:
                best_score = sc
                best_a = a_mp.copy()
                print(f"  Multi-peak: NEW BEST {sc:.8f} (t={time.monotonic()-t0:.1f}s)")
            else:
                print(f"  Multi-peak: score={sc:.8f}, best={best_score:.8f} (t={time.monotonic()-t0:.1f}s)")

    # ===================================================================
    # Phase 4: Final polish + restarts (15% of time)
    # ===================================================================
    phase4_end = deadline
    print(f"\n=== Phase 4: Final Polish (budget: {phase4_end - t_start:.0f}s) ===")

    if best_a is not None:
        # One more high-beta Adam
        if time.monotonic() < phase4_end:
            t0 = time.monotonic()
            max_iters = int((phase4_end - time.monotonic()) * 0.3 / 0.0005)
            max_iters = max(500, min(5000, max_iters))
            sc, a_final = run_adam_nosmooth(best_a, 20000, max_iters=max_iters, lr_init=0.001)
            if sc < best_score:
                best_score = sc
                best_a = a_final.copy()
                print(f"  Final Adam: NEW BEST {sc:.8f} (t={time.monotonic()-t0:.1f}s)")

        # Final peak shave
        if time.monotonic() < phase4_end:
            t0 = time.monotonic()
            iters = int((phase4_end - time.monotonic()) * 50)
            iters = max(500, min(5000, iters))
            sc, a_pol = peak_shave(best_a, iters=iters)
            if sc < best_score:
                best_score = sc
                best_a = a_pol.copy()
                print(f"  Final shave: NEW BEST {sc:.8f} (t={time.monotonic()-t0:.1f}s)")

        # Quick restarts
        restart_count = 0
        while time.monotonic() < phase4_end and restart_count < 10:
            t0 = time.monotonic()
            strategy = restart_count % 4
            n_best = len(best_a)

            if strategy == 0:
                # Gaussian perturbation
                scale = 0.01 + 0.02 * rng.random()
                a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
                a_pert = np.clip(a_pert, 0.0, 1000.0)
            elif strategy == 1:
                # Targeted: reduce peak contributors
                c = fast_conv(best_a)
                k_star = int(np.argmax(c))
                j_start = max(0, k_star - n_best + 1)
                j_end = min(k_star, n_best - 1)
                a_pert = best_a.copy()
                contrib = np.zeros(n_best)
                if j_end >= j_start:
                    contrib[j_start:j_end + 1] = best_a[j_start:j_end + 1][::-1]
                top_n = max(10, n_best // 200)
                top_idx = np.argsort(contrib)[::-1][:top_n]
                bot_idx = np.argsort(contrib)[top_n:top_n*2]
                red = 0.03 * best_a[top_idx].sum()
                a_pert[top_idx] -= red / len(top_idx)
                a_pert[bot_idx] += red / max(1, len(bot_idx))
                a_pert = np.clip(a_pert, 0.0, 1000.0)
            elif strategy == 2:
                # Sine modulation
                freq = 1.0 + 3.0 * rng.random()
                ph = 2.0 * np.pi * rng.random()
                amp = 0.01 + 0.02 * rng.random()
                x = np.arange(n_best) / n_best
                a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + ph))
                a_pert = np.clip(a_pert, 0.0, 1000.0)
            else:
                # Random block swap
                a_pert = best_a.copy()
                block_size = n_best // 32
                i = rng.integers(0, n_best - 2 * block_size)
                j = rng.integers(0, n_best - 2 * block_size)
                if j > i:
                    i, j = j, i
                i += block_size
                temp = a_pert[i:i+block_size].copy()
                a_pert[i:i+block_size] = a_pert[j:j+block_size]
                a_pert[j:j+block_size] = temp

            # Quick optimize
            max_iters = int(min(2000, (phase4_end - time.monotonic()) * 0.8 / 0.0005))
            max_iters = max(200, max_iters)
            sc, a_opt = run_adam_nosmooth(a_pert, 5000, max_iters=max_iters, lr_init=0.003)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"  Restart {restart_count} (strat={strategy}): NEW BEST {sc:.8f}")
            restart_count += 1

    # ===================================================================
    # Final output
    # ===================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
