"""
Improved optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key innovations:
1. Analytical coordinate updates (dv = S - M/a[mirror]) with line-search verification
2. Multi-peak aware coordinate descent targeting all significant conv peaks
3. Adaptive multi-length search with parent interpolation
4. Momentum-augmented subgradient for final polish
5. Frequency-domain error reduction with adaptive thresholding
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def analytical_coord_descent(a, n_sweeps=20, n_top_peaks=5, patience=5):
    """
    Coordinate descent using the analytical formula dv = S - M/a[mirror].
    Targets the top contributing indices at each active peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    best_a = a.copy()
    best_sc = compute_score(a)
    c = fast_conv(a)
    no_improve = 0

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                break

        # Find top peaks
        c_mean = float(c.mean())
        c_std = float(c.std())
        threshold = c_mean + 0.3 * c_std
        peak_indices = np.where(c > threshold)[0]
        # Sort by value, take top
        if len(peak_indices) > n_top_peaks:
            peak_indices = peak_indices[np.argsort(c[peak_indices])[::-1][:n_top_peaks]]

        # For each peak, find and update top contributors
        updated = set()
        for k in peak_indices:
            M_k = float(c[k])
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg_len = j_end - j_start + 1
            # Contribution: a[j] * a[k-j] for j in [j_start, j_end]
            # Mirror of j is k-j
            mirrors = k - np.arange(j_start, j_end + 1)
            contrib = a[j_start:j_end + 1] * a[mirrors]
            # Top contributors
            top_count = min(20, seg_len)
            top_local = np.argsort(contrib)[::-1][:top_count]

            for li in top_local:
                j = j_start + li
                if j in updated:
                    continue
                mirror = k - j
                a_mirror = float(a[mirror])
                a_j = float(a[j])

                if a_mirror < 1e-15:
                    continue

                # Analytical: dv = S - M/a[mirror]
                dv = S - M_k / a_mirror
                # Clamp to reasonable range
                max_decrease = a_j * 0.5  # Don't decrease by more than 50%
                max_increase = a_j * 0.5 + S / n  # Reasonable increase
                dv = np.clip(dv, -max_decrease, max_increase)
                dv = max(dv, -a_j)  # Non-negativity

                # Try the analytical update
                a[j] = a[j] + dv
                a[j] = max(a[j], 0.0)

                # Verify with line search: try a few scaled versions
                S_new = float(a.sum())
                if S_new < 1e-12:
                    a[j] = a_j
                    continue
                c_new = fast_conv(a)
                sc_new = 2.0 * n * float(c_new.max()) / (S_new * S_new)

                if sc_new < sc:
                    sc = sc_new
                    # Try to improve further with small adjustments
                    for factor in [0.5, 0.75, 1.25, 1.5]:
                        a[j] = a_j + dv * factor
                        a[j] = max(a[j], 0.0)
                        S_test = float(a.sum())
                        if S_test < 1e-12:
                            continue
                        c_test = fast_conv(a)
                        sc_test = 2.0 * n * float(c_test.max()) / (S_test * S_test)
                        if sc_test < sc:
                            sc = sc_test
                        else:
                            a[j] = a[j] - dv * (factor - 1.0)  # revert to best
                            break
                    updated.add(j)

                    # Update c incrementally for remaining peaks in this sweep
                    c = fast_conv(a)
                    S = float(a.sum())
                else:
                    a[j] = a_j  # Revert
        # End of sweep: recompute
        c = fast_conv(a)
        S = float(a.sum())

    return best_a


def multi_peak_subgradient(a, n_iters=3000, momentum=0.9):
    """Subgradient method with momentum, handling multiple active peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)  # Momentum

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find all active peaks (within 5% of max)
        threshold = M * 0.95
        active = np.where(c >= threshold)[0]

        # Subgradient: sum over active peaks
        sg = np.zeros(n)
        for k in active:
            weight = float(c[k]) / M  # Weight by how close to max
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += weight * 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step size
        sc_lb = 1.0
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.05)

        # Momentum
        v = momentum * v - step * grad
        a = a + v
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return best_a


def error_reduction_v2(a, n_iters=200):
    """
    Improved frequency-domain error reduction with adaptive thresholding
    and multiple compression levels.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    M = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        # Adaptive threshold: start aggressive, become gentle
        progress = it / max(1, n_iters)
        alpha = 0.5 + 0.4 * progress  # From 0.5 to 0.9
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break

        # Softer compression
        compression = 0.5 + 0.3 * progress
        conv_mod = conv - compression * excess
        err = conv_mod - conv

        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-14)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.3 + 0.4 * progress
        A_new = A + damp * d
        a = np.fft.ifft(A_new).real[:n]
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            break
        a = a / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_initializations(n, rng, parent=None):

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Analytical coordinate descent (35% of time)
    # ==================================================================
    phase2_deadline = t_start + 0.55 * (float(budget_s) - 10.0)
    print(f"\n--- Phase 2: Analytical Coordinate Descent ---")

    for ci, (sc_init, a_init) in enumerate(candidates):
        if time.monotonic() > phase2_deadline:
            break
        if sc_init > best_score * 1.05:
            continue  # Skip if much worse than current best

        print(f"  Candidate {ci} (init score {sc_init:.8f}), n={len(a_init)}")

        # Run analytical coordinate descent
        n_sweeps = max(10, int((phase2_deadline - time.monotonic()) / len(candidates) / 2.0))
        a_cd = analytical_coord_descent(a_init, n_sweeps=min(n_sweeps, 50),
                                        n_top_peaks=5, patience=8)
        sc = compute_score(a_cd)
        report(sc, a_cd, f"ACD cand{ci}")

    # ==================================================================
    # Phase 3: L-BFGS refinement (25% of time)
    # ==================================================================
    phase3_deadline = t_start + 0.80 * (float(budget_s) - 10.0)
    print(f"\n--- Phase 3: L-BFGS Refinement ---")

    n_best = len(best_a)
    beta_schedule = [50, 100, 200, 500]

    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=5000, maxfun=80000)
        report(sc, a_opt, f"LBFGS beta={beta}")

    # Try from a few perturbed starts
    n_pert = 3
    for pi in range(n_pert):
        if time.monotonic() > phase3_deadline:
            break
        scale = 0.03 + 0.03 * pi
        a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        a_pert = a_pert / a_pert.sum()
        sc, a_opt, _ = run_lbfgs(a_pert, 200, maxiter=3000, maxfun=50000)
        report(sc, a_opt, f"LBFGS pert{pi}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Final polish (20% of time)
    # ==================================================================
    print(f"\n--- Phase 4: Final Polish ---")

    # Multi-peak subgradient with momentum
    a_sg = multi_peak_subgradient(best_a, n_iters=3000, momentum=0.85)
    sc = compute_score(a_sg)
    report(sc, a_sg, "subgrad momentum")

    # More coordinate descent from subgradient result
    a_cd2 = analytical_coord_descent(a_sg, n_sweeps=20, n_top_peaks=3, patience=5)
    sc = compute_score(a_cd2)
    report(sc, a_cd2, "ACD after subgrad")

    # Final L-BFGS with high beta
    if time.monotonic() < deadline - 30:
        sc, a_opt, _ = run_lbfgs(best_a, 1000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "LBFGS beta=1000")

    # One more error reduction
    if time.monotonic() < deadline - 20:
        a_er_final = error_reduction_v2(best_a, n_iters=100)
        sc = compute_score(a_er_final)
        report(sc, a_er_final, "final ER")

    # Final coordinate descent
    if time.monotonic() < deadline - 10:
        a_final_cd = analytical_coord_descent(best_a, n_sweeps=15, n_top_peaks=3, patience=4)
        sc = compute_score(a_final_cd)
        report(sc, a_final_cd, "final ACD")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
