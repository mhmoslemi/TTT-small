"""
Step function optimization using multi-phase approach:
1. LSE continuation for smooth optimization
2. Iterative peak equalization (water-filling on convolution peaks)
3. Coordinate descent local search
4. Final polish with high-beta LSE

Key insights:

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def peak_equalize(a, iters=5000, tol=1e-10):
    """
    Iteratively equalize convolution peaks.
    At each step, find the current max of conv(a,a) and reduce it
    by adjusting the elements that contribute to it.
    Uses a projected subgradient approach with adaptive step size.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Run length for Polyak step
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max(conv) w.r.t. a
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        # The subgradient of (conv(a,a))_{k*} w.r.t. a_j is 2*a_{k*-j} for j in range
        sg = np.zeros(n)
        for j in range(j_start, j_end + 1):
            sg[j] = 2.0 * a[k_star - j]
        
        # Full gradient of score = 2n * M / S^2
        # d(score)/da_j = 2n * (dM/da_j * S^2 - M * 2*S * da_sum/da_j) / S^4
        #              = 2n * (sg_j / S^2 - 2*M/S^3)
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step: decrease step as we get closer to optimum
        # Use Polyak step with target at 1.0 (theoretical lower bound)
        target = 1.0
        if sc > target:
            step = (sc - target) / (gn * gn + 1e-20)
        else:
            step = 1e-6 / (gn * S + 1e-20)
        
        # Cap step to avoid overshooting
        max_step = 0.1 / S
        step = min(step, max_step)
        step = max(step, 1e-10)
        
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        # Check if this is an improvement; if not, reduce step
        sc_new = compute_score(a_new)
        if sc_new > sc:
            # Halve step and try again
            for _ in range(10):
                step *= 0.5
                a_new = a - step * grad
                a_new = np.clip(a_new, 0.0, 1000.0)
                sc_new = compute_score(a_new)
                if sc_new <= sc:
                    break
            if sc_new > sc:
                # Line search failed; try a smaller uniform step
                step = 1e-5 / (gn * S + 1e-20)
                a_new = a - step * grad
                a_new = np.clip(a_new, 0.0, 1000.0)
                sc_new = compute_score(a_new)
        
        a = a_new
    
    return best_sc, best_a


def multi_peak_shave(a, iters=3000):
    """
    Shave multiple peaks simultaneously.
    Uses a weighted subgradient where weights are based on how close
    each convolution value is to the maximum.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Soft-max weights for all peaks (not just the argmax)
        # This helps reduce multiple peaks simultaneously
        beta_soft = 50.0
        c_scaled = beta_soft * (c - M)
        c_scaled = np.clip(c_scaled, -500, 0)
        weights = np.exp(c_scaled) / np.exp(c_scaled).sum()
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for k in range(len(c)):
            if weights[k] < 1e-10:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                sg[j] += weights[k] * 2.0 * a[k - j]
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = (sc - 1.0) / (gn * gn + 1e-20)
        step = min(step, 0.05 / S)
        step = max(step, 1e-8)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a


def coordinate_descent(a, max_rounds=5, step_scale=0.1, rng=None):
    """
    Coordinate descent: for each element, try perturbing it up and down
    to find the local optimum.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = a.astype(np.float64).copy()
    best_sc = compute_score(a)
    
    for rnd in range(max_rounds):
        # Randomize order
        order = rng.permutation(n)
        for idx in order:
            # Try several perturbations of a[idx]
            best_val = a[idx]
            best_local_sc = best_sc
            base = a.copy()
            
            # Try geometric steps up and down
            for factor in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]:
                trial = base.copy()
                trial[idx] = base[idx] * factor
                trial[idx] = max(0.0, min(1000.0, trial[idx]))
                sc = compute_score(trial)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = trial[idx]
            
            a[idx] = best_val
            best_sc = min(best_sc, best_local_sc)
    
    return best_sc, a

"""
Step function optimization: Multi-phase approach combining LSE continuation,
Nesterov-accelerated peak equalization, adaptive multi-peak shaving, and
targeted local search. Key innovations over previous approach:

- Adaptive beta scheduling based on LSE-exact gap

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(a, beta):
    """Objective wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def peak_equalize_nesterov(a, iters=5000, momentum=0.9):
    """
    Nesterov-accelerated subgradient descent on the exact max-conv objective.
    Uses proper backtracking line search and momentum for faster convergence.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Nesterov momentum
    a_prev = a.copy()
    v = np.zeros(n)
    
    step_base = 1.0 / (n ** 0.5)
    
    for it in range(iters):
        # Nesterov lookahead
        a_look = a + momentum * (a - a_prev)
        a_look = np.clip(a_look, 0.0, 1000.0)
        
        c = fast_conv(a_look)
        S = float(a_look.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc_look = 2.0 * n * M / (S * S)
        
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a_look[k_star - j_start:k_star - j_start + (j_end - j_start + 1)][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Polyak step size with Nesterov modification
        target = 1.0
        gap = max(sc_look - target, 0.01)
        step = gap / (gn * gn + 1e-20)
        step = min(step, 0.5 / S)
        step = max(step, 1e-8 / S)
        
        # Update with momentum
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        # Check improvement, backtrack if needed
        sc_new = compute_score(a_new)
        if sc_new > sc_look:
            # Backtrack: reduce step
            for _ in range(5):
                step *= 0.5
                v = momentum * v * 0.5 - step * grad
                a_new = a + v
                a_new = np.clip(a_new, 0.0, 1000.0)
                sc_new = compute_score(a_new)
                if sc_new <= sc_look:
                    break
        
        a_prev = a
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a


def multi_peak_shave_adaptive(a, iters=5000):
    """
    Adaptive multi-peak shaving: uses soft-max weighting of all peaks
    with adaptive temperature. Also includes a "pressure" term that pushes
    up the minimum convolution value to flatten the overall profile.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Momentum
    v = np.zeros(n)
    momentum = 0.7
    
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Adaptive soft-max temperature: start high (uniform), decrease over time
        progress = it / iters
        beta_soft = 10.0 + 100.0 * progress ** 2
        
        c_scaled = beta_soft * (c - M)
        c_scaled = np.clip(c_scaled, -500.0, 0.0)
        weights = np.exp(c_scaled)
        weights_sum = float(weights.sum())
        if weights_sum < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
        else:
            weights /= weights_sum
        
        # Compute weighted subgradient efficiently using FFT
        # sg[j] = sum_k weights[k] * 2 * a[k-j] for valid k
        # This is a cross-correlation of weights with a
        rev_a = a[::-1]
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        W = np.fft.rfft(weights, M2_fft)
        # Cross-correlation: sum_k weights[k] * a[k-j]
        # = ifft(fft(rev_a) * fft(weights))[n-1:2n-1]
        corr = np.fft.irfft(RA * W, M2_fft)[n - 1:2 * n - 1]
        
        sg = 2.0 * corr
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step
        gap = max(sc - 1.0, 0.01)
        step = gap / (gn * gn + 1e-20)
        step = min(step, 0.1 / S)
        step = max(step, 1e-9 / S)
        
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        # Backtrack if needed
        sc_new = compute_score(a_new)
        if sc_new > sc * 1.001:
            for _ in range(3):
                step *= 0.5
                v = momentum * v * 0.5 - step * grad
                a_new = a + v
                a_new = np.clip(a_new, 0.0, 1000.0)
                sc_new = compute_score(a_new)
                if sc_new <= sc * 1.001:
                    break
        
        a = a_new
    
    return best_sc, best_a

def frequency_perturb(a, rng, strength=0.05):
    """Perturb in frequency domain to maintain overall structure."""
    n = len(a)
    L = 1 << (2 * n - 2).bit_length()
    A = np.fft.fft(a, L)
    A_mag = np.abs(A)
    A_phase = np.angle(A)
    
    # Perturb magnitudes with a smooth noise pattern
    noise = rng.standard_normal(L)
    # Low-pass filter the noise to keep perturbations smooth
    noise_freq = np.fft.fft(noise)
    cutoff = L // 4
    noise_freq[cutoff:-cutoff] = 0
    noise = np.fft.ifft(noise_freq).real
    
    A_mag_new = A_mag * (1.0 + strength * noise)
    A_mag_new = np.clip(A_mag_new, 0, None)
    A_new = A_mag_new * np.exp(1j * A_phase)
    a_pert = np.fft.ifft(A_new).real[:n]
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    return a_pert


def variance_flatness_obj(a):
    """
    Objective for flattening: 2n * sum(c^2) / S^4
    This is a smooth proxy that rewards flat autocorrelation.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c2 = float(np.sum(c * c))
    score = 2.0 * n * c2 / (S ** 4)
    
    # Gradient via FFT
    L = 1 << (2 * n - 2).bit_length()
    A = np.fft.fft(a, L)
    E = np.abs(A) ** 2 * np.conj(A)
    E_ifft = np.fft.ifft(E).real
    idx = np.concatenate([[0], np.arange(L - 1, L - n, -1)])
    dc2_da = 4.0 * E_ifft[idx]
    S4 = S ** 4
    S5 = S4 * S
    grad = 2.0 * n * (dc2_da / S4 - 4.0 * c2 / S5)
    return score, grad

                b1 = rng.integers(0, n - block_size)
                b2 = rng.integers(0, n - block_size)
                if abs(b1 - b2) > block_size:
                    a_pert[b1:b1+block_size], a_pert[b2:b2+block_size] = \
                        a_pert[b2:b2+block_size].copy(), a_pert[b1:b1+block_size].copy()
        else:
            # Sinusoidal modulation
            freq = 1.0 + 5.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.03 * rng.random()
            x = np.arange(n, dtype=np.float64) / n
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        # Quick optimization from perturbed start
        time_left = p7_end - time.monotonic()
        if time_left < 5:
            break
        
        if time_left > 30:
            a_opt = run_lbfgs(a_pert, lse_obj, (3000,), maxiter=300, maxfun=8000)
        else:
            iters = int(min(500, time_left * 80))
            sc_tmp, a_opt = peak_equalize_nesterov(a_pert, iters=max(50, iters))
        
        sc = compute_score(a_opt)
        improved = report(sc, a_opt, f"restart {restart_num} (s{strategy})")
        
        # If improved, do a quick polish
        if improved and time.monotonic() < p7_end - 10:
            a_pol = run_lbfgs(a_opt, lse_obj, (5000,), maxiter=200, maxfun=5000)
            sc = compute_score(a_pol)
            report(sc, a_pol, f"restart {restart_num} polish")
        
        restart_num += 1
    
    # ============ Final Output ============
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Total time: {time.monotonic()-t_start:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
