"""
Hybrid MW-LP + Coordinate Descent + LSE optimizer for minimum peak autocorrelation.
Key innovation: Uses multiplicative weights to approximate the LP direction from the
state-of-the-art approach, enabling coordinated multi-coordinate updates that escape
local minima of pure coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def fast_conv(a, b=None):
    """Linear convolution via FFT. If b is None, computes self-convolution."""
    if b is None:
        b = a
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500, 500))
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(conv_in.max())
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)

def coordinate_descent_sweep(a, rng=None, n_coords=None, tol=1e-7):
    """One sweep of coordinate descent with incremental conv updates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    if n_coords is None:
        n_coords = n

    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)

    for idx in range(min(n_coords, n)):
        i = int(order[idx])
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(conv[:k_lo].max()), float(conv[k_hi + 1:].max()))
        elif k_lo > 0:
            outside_max = float(conv[:k_lo].max())
        elif k_hi < M_conv - 1:
            outside_max = float(conv[k_hi + 1:].max())
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()
        
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)

        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': tol}
        )
        t_opt = float(res.x)

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a

def mw_lp_direction(a, M, n_iter=2000, eta=None, verbose=False):
    """
    Use multiplicative weights to approximate the LP:
    maximize sum(b) subject to conv(a,b) <= M, b >= 0.
    Returns b normalized to sum=1.
    """
    n = len(a)
    p = np.ones(n) / n
    q = np.ones(2 * n - 1) / (2 * n - 1)
    
    if eta is None:
        eta = 1.0 / np.sqrt(np.log(2 * n - 1) / n_iter)
    
    # Precompute FFT of a for gradient computation
    N_cor = 1 << (3 * n - 2).bit_length()
    A_f = np.fft.rfft(a, N_cor)
    
    best_p = p.copy()
    best_max_viol = np.inf
    
    for t in range(n_iter):
        # Compute constraint values: z = conv(a, p)
        z = fast_conv(a, p)
        
        # Compute violations
        viol = np.maximum(0.0, z / M - 1.0)
        max_viol = float(viol.max())
        
        if max_viol < best_max_viol:
            best_max_viol = max_viol
            best_p = p.copy()
        
        # Update dual weights
        q *= (1.0 + eta * viol)
        q_sum = q.sum()
        if q_sum < 1e-300:
            break
        q /= q_sum
        
        # Compute primal gradient: (T_a^T q)[i] = sum_j a[j] * q[j+i]
        Q_f = np.fft.rfft(q, N_cor)
        grad = np.fft.irfft(np.conj(A_f) * Q_f, N_cor)[:n]
        grad = np.maximum(grad, 0.0)  # should be non-negative since a, q >= 0
        
        # Update primal weights
        log_p = np.log(p + 1e-300)
        log_p -= eta * grad
        # Normalize in log space for stability
        log_p -= log_p.max()
        p = np.exp(log_p)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(n) / n
        else:
            p /= p_sum
        
        if verbose and t % 500 == 0:
            print(f"    MW iter {t}: max_viol={max_viol:.6f}, eta={eta:.6f}")
    
    return best_p

MW-LP guided coordinate descent + LSE optimizer for minimum peak autocorrelation.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovation: Uses multiplicative weights to approximate the LP direction
(maximize sum(b) s.t. conv(a,b) <= M, b >= 0), enabling coordinated
multi-coordinate updates that escape local minima of pure coordinate descent.
Combined with exact 1D coordinate descent and LSE-based L-BFGS refinement.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a, b=None):
    """Linear convolution via FFT. If b is None, computes self-convolution."""
    if b is None:
        b = a
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500, 500))
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(conv_in.max())
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)

        a_i = a[i]

        k_hi = min(M_conv - 1, i + n - 1)

            outside_max = 0.0

def mw_lp_direction(a, M, n_iter=1500, eta=None):
    """
    Use multiplicative weights to approximate the LP:
    maximize sum(b) subject to conv(a,b) <= M, b >= 0.
    Returns b normalized to sum=1.
    """
    n = len(a)
    p = np.ones(n) / n
    q = np.ones(2 * n - 1) / (2 * n - 1)

    if eta is None:
        eta = 1.0 / np.sqrt(2.0 * np.log(2 * n) / max(1, n_iter))
        eta = min(eta, 5.0)
        eta = max(eta, 0.01)

    N_cor = 1 << (3 * n - 2).bit_length()
    A_f = np.fft.rfft(a, N_cor)

    best_p = p.copy()
    best_max_viol = float((fast_conv(a, p).max()) / M - 1.0)
    if best_max_viol < 0:
        best_max_viol = 0.0

    log_p = np.zeros(n) - np.log(n)

    for t in range(n_iter):
        p = np.exp(log_p - log_p.max())
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(n) / n
        else:
            p /= p_sum

        z = fast_conv(a, p)
        viol = np.maximum(0.0, z / M - 1.0)
        max_viol = float(viol.max())

        if max_viol < best_max_viol:
            best_max_viol = max_viol
            best_p = p.copy()

        if max_viol < 1e-8:
            break

        q *= (1.0 + eta * viol)
        q_sum = q.sum()
        if q_sum < 1e-300:
            break
        q /= q_sum

        Q_f = np.fft.rfft(q, N_cor)
        grad = np.fft.irfft(np.conj(A_f) * Q_f, N_cor)[:n]
        grad = np.maximum(grad, 1e-300)

        grad_norm = grad / grad.max()
        log_p -= eta * grad_norm
        log_p -= log_p.max()

    return best_p


def mw_interp_step(a, n_mw_iter=1500, eta=None, n_cd_sweeps=3):

    b = mw_lp_direction(a, M, n_iter=n_mw_iter, eta=eta)

    best_t = 0.0

    best_a = a.copy()

    for _ in range(n_cd_sweeps):

        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 20.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    if parent is not None:
        p_score = compute_score(parent)
        print(f"Parent score: {p_score:.8f}, length: {len(parent)}")
        best_score = p_score
        best_a = parent.copy()
    else:
        print("No parent available, using random init.")

    n_work = len(best_a)
    print(f"Working length: {n_work}")

    # ============================================================
    # Phase 1: Coordinate descent polish (5 min)
    # ============================================================
    phase1_deadline = t_start + 300.0
    print(f"\n=== Phase 1: CD Polish ({phase1_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    sweep_count = 0
    while time.monotonic() < phase1_deadline:
        sweep_count += 1
        if sweep_count % 3 == 0:
            sc, a_cur = hot_coordinate_sweep(a_cur, n_sweeps=1)
        else:
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            if sweep_count <= 5 or sweep_count % 10 == 0:
                print(f"  P1 sweep {sweep_count}: {sc:.8f} (NEW BEST)")
        if sweep_count % 10 == 0:
            print(f"  P1 sweep {sweep_count}: {sc:.8f}")

    print(f"Phase 1 done: {sweep_count} sweeps, best={best_score:.8f}")

    # ============================================================
    # Phase 2: MW-LP + Interpolation + CD (20 min)
    # ============================================================
    phase2_deadline = t_start + 1500.0
    print(f"\n=== Phase 2: MW-LP Cycles ({phase2_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    mw_cycle = 0
    no_improve_mw = 0
    mw_params_list = [
        (1000, None, 2),
        (1500, None, 2),
        (2000, None, 3),
        (2500, None, 3),
        (2000, 0.5, 2),
        (2000, 1.0, 2),
        (3000, None, 3),
        (1500, 2.0, 2),
    ]

    while time.monotonic() < phase2_deadline:
        mw_cycle += 1
        params = mw_params_list[(mw_cycle - 1) % len(mw_params_list)]
        n_mw, eta_val, n_cd = params

        t0 = time.monotonic()
        sc, a_cur = mw_interp_step(a_cur, n_mw_iter=n_mw, eta=eta_val, n_cd_sweeps=n_cd)
        elapsed = time.monotonic() - t0

        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            no_improve_mw = 0
            print(f"  MW cycle {mw_cycle} ({n_mw}it, eta={eta_val}, {n_cd}cd): {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            no_improve_mw += 1
            if mw_cycle <= 5 or mw_cycle % 5 == 0:
                print(f"  MW cycle {mw_cycle} ({n_mw}it, eta={eta_val}, {n_cd}cd): {sc:.8f} ({elapsed:.1f}s, no_improve={no_improve_mw})")

        if no_improve_mw >= 4:
            if no_improve_mw == 4:
                print(f"  Stuck after {no_improve_mw} MW cycles, perturbing...")
                a_cur = targeted_perturb(best_a, rng, strength=0.02 + 0.02 * rng.random())
                sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cur.copy()
            elif no_improve_mw == 6:
                a_cur = best_a * (1.0 + 0.02 * rng.standard_normal(n_work))
                a_cur = np.clip(a_cur, 0.0, 1000.0)
                sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cur.copy()
            no_improve_mw = 0

    print(f"Phase 2 done: {mw_cycle} MW cycles, best={best_score:.8f}")

    # ============================================================
    # Phase 3: LSE refinement (5 min)
    # ============================================================
    phase3_deadline = t_start + 1800.0
    print(f"\n=== Phase 3: LSE Refinement ({phase3_deadline - t_start:.0f}s) ===")

    for beta in [100, 300, 800, 2000, 5000]:
        if time.monotonic() > phase3_deadline - 30:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=6000, maxfun=100000)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  LSE beta={beta}: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            print(f"  LSE beta={beta}: {sc:.8f} ({elapsed:.1f}s)")

    # ============================================================
    # Phase 4: Perturbation + CD (4 min)
    # ============================================================
    phase4_deadline = t_start + 2040.0
    print(f"\n=== Phase 4: Perturbation Cycles ({phase4_deadline - t_start:.0f}s) ===")

    for trial in range(50):
        if time.monotonic() > phase4_deadline - 5.0:
            break
        strategy = trial % 6
        n_cur = len(best_a)

        if strategy == 0:
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_cur))
        elif strategy == 1:
            a_pert = targeted_perturb(best_a, rng, strength=0.01 + 0.04 * rng.random())
        elif strategy == 2:
            block_size = max(10, n_cur // 80)
            start = int(rng.integers(0, n_cur - block_size))
            a_pert = best_a.copy()
            saved = a_pert[start:start + block_size].sum()
            a_pert[start:start + block_size] = 0.0
            rest_mask = np.ones(n_cur, dtype=bool)
            rest_mask[start:start + block_size] = False
            a_pert[rest_mask] += saved / max(1, int(rest_mask.sum()))
        elif strategy == 3:
            seg_len = max(10, n_cur // 40)
            s1 = int(rng.integers(0, max(1, n_cur - seg_len)))
            s2 = int(rng.integers(0, max(1, n_cur - seg_len)))
            a_pert = best_a.copy()
            tmp = a_pert[s1:s1 + seg_len].copy()
            a_pert[s1:s1 + seg_len] = a_pert[s2:s2 + seg_len]
            a_pert[s2:s2 + seg_len] = tmp
        elif strategy == 4:
            freq = 1.0 + 8.0 * rng.random()
            amp = 0.005 + 0.02 * rng.random()
            x = np.arange(n_cur, dtype=np.float64) / n_cur
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + 2.0 * np.pi * rng.random()))
        else:
            a_pert = best_a.copy()
            idx = rng.integers(0, n_cur, size=max(5, n_cur // 20))
            a_pert[idx] *= (0.5 + rng.random() * 0.5)
            a_pert[~np.isin(np.arange(n_cur), idx)] += a_pert[idx].sum() * 0.1 / max(1, n_cur - len(idx))

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        for sw in range(5):
            if time.monotonic() > phase4_deadline - 5.0:
                break
            if sw % 2 == 0:
                sc_pert, a_pert = hot_coordinate_sweep(a_pert, n_sweeps=1)
            else:
                sc_pert, a_pert = coordinate_descent_sweep(a_pert, rng=rng)

        sc_pert = compute_score(a_pert)
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  P4 trial {trial} (s={strategy}): {sc_pert:.8f} (NEW BEST)")

    # ============================================================
    # Phase 5: Final MW + polish (4 min)
    # ============================================================
    phase5_deadline = deadline
    print(f"\n=== Phase 5: Final Polish ({phase5_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    while time.monotonic() < phase5_deadline - 60:
        sc, a_cur = mw_interp_step(a_cur, n_mw_iter=2000, eta=None, n_cd_sweeps=2)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  P5 MW: {sc:.8f} (NEW BEST)")

    # Final CD sweeps
    a_cur = best_a.copy()
    for _ in range(10):
        if time.monotonic() > deadline - 10:
            break
        sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

    # One final LSE if time permits
    if time.monotonic() < deadline - 30:
        sc, a_opt, _ = run_lbfgs(best_a, 3000, maxiter=4000, maxfun=60000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final LSE: {sc:.8f} (NEW BEST)")
        a_cur = best_a.copy()
        sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  Final CD: {sc:.8f} (NEW BEST)")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
