```python
"""
Advanced hybrid optimizer for minimum peak autocorrelation of non-negative sequences.
Combines fast vectorized coordinate descent, pairwise optimization, peak-targeted
perturbation, and LP-inspired spreading moves.
"""

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize_sum(a, target=None):
    """Normalize sequence to have a specific sum (default: sqrt(2n))."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    if target is None:
        target = np.sqrt(2.0 * n)
    return a * (target / S)

def solve_1d_vectorized(base_inside, alpha, outside_max, S_rest, n, t_upper, has_quad, quad_idx):
    """
    Fast 1D solver.
    base_inside: ndarray of shape (m,) - base convolution values at affected lags
    alpha: ndarray of shape (m,) - linear coefficients
    outside_max: float - max of unaffected lags
    S_rest: float - sum of all coordinates except the one being optimized
    n: int - sequence length
    t_upper: float - upper bound for t
    has_quad: bool - whether there's a quadratic term
    quad_idx: int or None - index of the quadratic term in base_inside/alpha
    """
    m = len(base_inside)
    # Coarse grid
    n_coarse = 120
    t_c = np.linspace(0.0, t_upper, n_coarse)
    # Compute c(t) for all t simultaneously
    c_vals = base_inside[None, :] + t_c[:, None] * alpha[None, :]
    if has_quad:
        c_vals[:, quad_idx] += t_c ** 2  # t_c**2 broadcast to the column
    c_max = np.maximum(c_vals, outside_max).max(axis=1)
    S_new = S_rest + t_c
    f = 2.0 * n * c_max / (S_new * S_new)
    
    # Find coarse minimum
    idx = np.argmin(f)
    t_mid = t_c[idx]
    
    # Fine grid
    delta = t_upper / n_coarse
    lo = max(0.0, t_mid - 3 * delta)
    hi = min(t_upper, t_mid + 3 * delta)
    n_fine = 80
    t_f = np.linspace(lo, hi, n_fine)
    c_vals = base_inside[None, :] + t_f[:, None] * alpha[None, :]
    if has_quad:
        c_vals[:, quad_idx] += t_f ** 2
    c_max = np.maximum(c_vals, outside_max).max(axis=1)
    S_new = S_rest + t_f
    f = 2.0 * n * c_max / (S_new * S_new)
    
    return float(t_f[np.argmin(f)])

def cd_sweep_fast(a, order=None):
    """
    Fast coordinate descent sweep with vectorized 1D solver.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1
    
    if order is None:
        order = np.arange(n)
    
    for i in order:
        i = int(i)
        a_i = a[i]
        
        # Affected lags
        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)
        
        # Outside max
        if k_lo > 0:
            max_left = float(conv[:k_lo].max()) if k_lo > 0 else 0.0
        else:
            max_left = 0.0
        if k_hi < M_conv - 1:
            max_right = float(conv[k_hi+1:].max())
        else:
            max_right = 0.0
        outside_max = max(max_left, max_right)
        
        # Inside values
        inside = conv[k_lo:k_hi+1].copy()
        ks = np.arange(k_lo, k_hi+1)
        offsets = ks - i  # these are in [0, n-1]
        a_vals = a[offsets]
        
        is_2i = (ks == 2 * i)
        
        # Base (remove a_i's contribution)
        old_lin = 2.0 * a_i * a_vals
        old_lin[is_2i] = a_i * a_i  # at k=2i, it's a_i^2 not 2*a_i^2
        base_inside = inside - old_lin
        
        # Alpha (linear coefficient)
        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        
        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, 3.0 * S / n)
        t_upper = min(t_upper, 1000.0)
        
        has_quad = is_2i.any()
        quad_idx = np.where(is_2i)[0]
        quad_idx = int(quad_idx[0]) if has_quad else None
        
        t_opt = solve_1d_vectorized(base_inside, alpha, outside_max, S_rest, n, t_upper, has_quad, quad_idx)
        
        # Update
        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi+1] = base_inside + t_opt * alpha
        if has_quad:
            conv[k_lo + quad_idx] += t_opt * t_opt
    
    return a

def hot_order(a, conv):
    """Get order prioritizing coordinates that contribute to the peak."""
    n = len(a)
    k_star = int(np.argmax(conv))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end+1] = a[j_start:j_end+1][::-1]
    return np.argsort(contrib)[::-1]

def peak_perturb(a, rng, strength=0.03):
    """Perturb by reducing peak contributors and spreading mass."""
    n = len(a)
    conv = fast_conv(a)
    k_star = int(np.argmax(conv))
    
    # Contributors to the peak
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end+1] = a[j_start:j_end+1][::-1]
    
    # Also consider second peak
    conv_copy = conv.copy()
    conv_copy[k_star] = -1
    k_star2 = int(np.argmax(conv_copy))
    j_start2 = max(0, k_star2 - n + 1)
    j_end2 = min(k_star2, n - 1)
    if j_end2 >= j_start2:
        contrib[j_start2:j_end2+1] += 0.5 * a[j_start2:j_end2+1][::-1]
    
    # Top contributors
    top_count = max(5, n // 30)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    rest_idx = np.setdiff1d(np.arange(n), top_idx)
    
    a_new = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        red = a_new[j] * strength * (0.5 + 0.5 * rng.random())
        a_new[j] = max(a_new[j] - red, 0.0)
        total_reduced += red

        # Distribute to non-peak contributors
        weights = 1.0 / (a[rest_idx] + 0.01)  # prefer to add to smaller values
        weights /= weights.sum()
        a_new[rest_idx] += total_reduced * weights
    
    return np.clip(a_new, 0.0, 1000.0)

def random_perturb(a, rng, scale=0.02):

    n = len(a)
    a_new = a * (1.0 + scale * rng.standard_normal(n))
    return np.clip(a_new, 0.0, 1000.0)

def block_zero_perturb(a, rng):
    """Zero out a random block and redistribute."""
    n = len(a)
    block_size = max(5, n // 50)
    start = int(rng.integers(0, n - block_size))

    rest[start:start+block_size] = False
    a_new[rest] += saved / rest.sum()
    return np.clip(a_new, 0.0, 1000.0)
```

Let me also think about a key insight: **the relationship between the score and the sequence structure.**

For a sequence with sum S, the score is 2n * max_conv / S^2. If we write S = n * avg (where avg is the mean), then score = 2 * max_conv / (n * avg^2). So we want to minimize max_conv relative to n * avg^2.

For a "good" sequence, max_conv should be close to the expected value of a random convolution. For a random sequence with values in [0, 2*avg] (so mean is avg), the expected autocorrelation at lag 0 is n * E[a^2] ≈ n * (2/3) * avg^2 (for uniform). The expected autocorrelation at other lags is about (n-k) * avg^2.

So the "random" peak is about n * (2/3) * avg^2, giving score ≈ 2 * (2/3) = 4/3 ≈ 1.33. But this is just expected - the actual max over 2n-1 values will be higher due to concentration. For 2n-1 ≈ 14000 values, the max is about mean + 4*std, so maybe 1.33 * 1.1 ≈ 1.46.

This suggests that the theoretical optimum for large n is around 1.3-1.4, and we're at 1.509, so there's room for improvement.

Actually, I want to also include a key improvement: **analytical 1D solve** instead of grid search. Let me think about this more carefully.

The 1D problem is: minimize f(t) = 2n * C(t) / (S+t)^2 where C(t) = max(max_i (b_i + a_i * t + q_i * t^2), O)

where b_i, a_i are the base and linear coefficients, q_i ∈ {0, 1} (only one q_i = 1), and O is outside_max.

Case 1: The quadratic term q(t) = b_2i + t^2 is the max.
Then C(t) = b_2i + t^2, and f(t) = 2n * (b_2i + t^2) / (S+t)^2.
f'(t) = 2n * [2t(S+t)^2 - 2(S+t)(b_2i + t^2)] / (S+t)^4
      = 2n * 2(S+t)[t(S+t) - (b_2i + t^2)] / (S+t)^4
      = 4n * [tS + t^2 - b_2i - t^2] / (S+t)^3
      = 4n * [tS - b_2i] / (S+t)^3

Setting f'(t) = 0: t = b_2i / S.

But we need to verify that q(t) is indeed the max at this t.

Case 2: A linear term l_j(t) = b_j + a_j * t is the max.
Then C(t) = b_j + a_j * t, and f(t) = 2n * (b_j + a_j * t) / (S+t)^2.
f'(t) = 2n * [a_j(S+t)^2 - 2(S+t)(b_j + a_j*t)] / (S+t)^4
      = 2n * (S+t)[a_j(S+t) - 2(b_j + a_j*t)] / (S+t)^4
      = 2n * [a_j*S + a_j*t - 2*b_j - 2*a_j*t] / (S+t)^3
      = 2n * [a_j*S - a_j*t - 2*b_j] / (S+t)^3

Setting f'(t) = 0: t = (a_j*S - 2*b_j) / a_j = S - 2*b_j/a_j (if a_j > 0).

If a_j = 0, then f(t) = 2n*b_j / (S+t)^2 which is decreasing in t, so minimum at t = t_upper.

Case 3: outside_max O is the max.
Then C(t) = O, and f(t) = 2n*O / (S+t)^2 which is decreasing in t, so minimum at t = t_upper.

So the algorithm is:
1. Compute candidate t values:
   - t = 0
   - t = t_upper
   - t_quad = b_2i / S (from the quadratic case)
   - For each linear term j: t_j = S - 2*b_j/a_j (if a_j > 0 and b_j + a_j*t > 0)
2. For each candidate t, compute C(t) and verify it's the actual max
3. Return the t with minimum f(t)

The issue is that there are O(n) linear terms, so we need O(n) candidate t values. For each, we need to verify the "active set" condition, which is O(n). Total: O(n^2) per coordinate, O(n^3) per sweep. That's too slow for n=6962.

But we can be smarter: we only need to check candidates where the corresponding linear term IS the max. We can do this in two passes:
1. First pass: find which linear term is the max at t=0 and at t=t_upper
2. Second pass: for each "active" term, compute its optimal t and verify

Actually, the number of "active" terms (those that are the max for some t in [0, t_upper]) is typically small (O(1) or O(log n)). So we can:
1. Find the active terms by evaluating C(t) at a few points and tracking which term is max
2. For each active term, solve for the optimal t
3. Also check the breakpoints between active terms

This should be O(n) per coordinate in practice.

```python
def solve_1d_analytical(base, alpha, has_quad, quad_idx, outside_max, S_rest, n, t_upper):
    """
    Analytical 1D solver.
    base: array of base values at affected lags
    alpha: array of linear coefficients
    has_quad: whether there's a quadratic term
    quad_idx: index of quadratic term (if has_quad)
    outside_max: max of unaffected lags
    """
    m = len(base)
    candidates = [0.0, t_upper]
    
    # Quadratic candidate
    if has_quad:
        t_q = base[quad_idx] / S_rest if S_rest > 1e-12 else 0.0
        if 0 <= t_q <= t_upper:
            candidates.append(t_q)
    
    # For each linear term, find its optimal t
    for j in range(m):
        if has_quad and j == quad_idx:
            continue  # skip the quadratic one
        a_j = alpha[j]
        b_j = base[j]
        if a_j > 1e-15:
            t_j = S_rest - 2.0 * b_j / a_j
            if 0 < t_j < t_upper:
                candidates.append(t_j)
        # If a_j == 0, the function is decreasing, so t_upper is the best (already included)
    
    # Also check breakpoints between the top few linear terms and the quadratic
    # (This is important when the max switches)
    # For now, add a few heuristic candidates
    # Evaluate f(t) at all candidates and return the best
    
    best_f = np.inf
    best_t = 0.0
    
    for t in candidates:
        t = max(0.0, min(t, t_upper))
        # Compute C(t)
        c_vals = base + t * alpha
        if has_quad:
            c_vals = c_vals.copy()
            c_vals[quad_idx] += t * t
        C = max(float(np.max(c_vals)), outside_max)
        S_new = S_rest + t
        if S_new < 1e-12:
            continue
        f = 2.0 * n * C / (S_new * S_new)
        if f < best_f:
            best_f = f
            best_t = t
    
    return best_t
```
