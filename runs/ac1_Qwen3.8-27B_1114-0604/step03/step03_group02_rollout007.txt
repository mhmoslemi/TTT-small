We need to minimize: `2 * n * max(conv(a, a)) / (sum(a))^2`

where `conv(a, a)` is the linear convolution of `a` with itself (length 2n-1), and `a` is a non-negative sequence of length n.

Key observations:
1. The objective is scale-invariant: if we scale `a` by `c`, the numerator scales by `c^2` and the denominator scales by `c^2`, so the ratio stays the same.
2. The minimum possible value is 1.0 (achieved when conv(a,a) is as flat as possible, i.e., a "perfect" sequence where all convolution values are equal).
3. For a sequence where all elements are equal (a_i = c), conv(a,a) has values c^2, 2c^2, 3c^2, ..., nc^2, (n-1)c^2, ..., c^2. The max is nc^2, and sum(a) = nc, so the score is 2*n*nc^2 / (nc)^2 = 2. So uniform gives score 2.
4. We want to make the autocorrelation as flat as possible (like a "flat" autocorrelation sequence).

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """Run L-BFGS-B to minimize the LSE-smoothed score."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_func(x, b):
        sc, gr = lse_score_grad(x, b)
        return sc, gr

    options = {'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            sc, _ = lse_score_grad(a0, beta)
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(100, int(remaining * 300)))
        options['maxiter'] = min(maxiter, max(100, int(remaining * 80)))

    try:
        res = minimize(
            obj_func, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options=options
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def soft_equalize(a, n_iters=200, alpha=0.5, beta_range=(1.0, 50.0), deadline=None):
    """
    Soft-weighted multiplicative equalization.
    Instead of hard threshold on active lags, use exponential soft weights.
    Continue in the sharpness parameter.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    # Continuation: start soft, get sharper
    n_continuations = 5
    betas = np.geomspace(beta_range[0], beta_range[1], n_continuations)
    
    iters_per_contin = max(10, n_iters // n_continuations)

    for bi, beta in enumerate(betas):
        for it in range(iters_per_contin):
            if deadline is not None and time.monotonic() > deadline:
                break
            c = fast_conv(a)
            M = float(c.max())
            if M < 1e-15:
                break
            # Soft weights: lambda_k = exp(beta * (c_k - M)) / sum(...)
            # This puts more weight on lags close to the max
            scaled = beta * (c - M)
            lambda_v = np.exp(scaled)
            lambda_sum = lambda_v.sum()
            if lambda_sum < 1e-30:
                lambda_v = np.zeros(len(c))
                lambda_v[int(np.argmax(c))] = 1.0
            else:
                lambda_v = lambda_v / lambda_sum

            # Compute h[j] = sum_k lambda_k * dc_k/da_j
            # Using the same trick as in lse_score_grad
            rev_a = a[::-1]
            M3 = 3 * n - 2
            M3_fft = 1 << (M3 - 1).bit_length()
            RA = np.fft.rfft(rev_a, M3_fft)
            LV = np.fft.rfft(lambda_v, M3_fft)
            conv_res = np.fft.irfft(RA * LV, M3_fft)[:M3]
            h = 2.0 * conv_res[n - 1:2 * n - 1]

            # Multiplicative update to equalize h
            h = np.maximum(h, 1e-15)
            h_mean = h.mean()
            factor = 1.0 - alpha + alpha * (h_mean / h)
            a = a * factor
            a = np.clip(a, 0.0, 1000.0)

            # Renormalize
            S_new = a.sum()
            if S_new < 1e-12:
                a[:] = 0.5
                break
            a = a * (S / S_new)

            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_sc, best_a


def block_coordinate_descent(a, n_blocks=32, iters_per_block=50, deadline=None):
    """
    Block coordinate descent: divide sequence into blocks, optimize each block.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    block_size = n // n_blocks

    for outer in range(10):
        if deadline is not None and time.monotonic() > deadline:
            break
        for b in range(n_blocks):
            if deadline is not None and time.monotonic() > deadline:
                break
            start = b * block_size
            end = start + block_size if b < n_blocks - 1 else n
            if end - start < 2:
                continue

            # Optimize this block using L-BFGS with high beta
            a_block = a[start:end].copy()
            if a_block.sum() < 1e-10:
                a_block[:] = a.mean() / n

            def block_obj(x, b_idx=start, e_idx=end):
                a_trial = a.copy()
                a_trial[b_idx:e_idx] = x
                sc, _ = lse_score_grad(a_trial, 1000.0)
                return sc

            try:
                res = minimize(block_obj, a_block, method='L-BFGS-B',
                              bounds=[(0.0, 1000.0)] * (end - start),
                              options={'maxiter': iters_per_block, 'maxfun': iters_per_block * 5, 'ftol': 1e-12})
                a[start:end] = np.clip(res.x, 0.0, 1000.0)
            except Exception:
                pass

            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_sc, best_a


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to the peak lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses: flat-spectrum init, soft equalization with continuation, 
    LSE L-BFGS, block coordinate descent, and adaptive local search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    N = 8192  # Fixed length

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size > 0:
                if p.size == N:
                    parent = np.clip(p, 0.0, 1000.0)
                else:
                    parent = np.interp(np.linspace(0, p.size - 1, N), np.arange(p.size), p)
                    parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(N) * 0.5

    def _report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 0: Generate diverse initializations
    # ============================================================
    print(f"Phase 0: Generating initializations (N={N})")
    print(f"Parent available: {parent is not None}")

    inits = []

    # Parent (resampled)
    if parent is not None:
        inits.append(('parent', parent.copy()))

    # Flat-spectrum: use IFFT of constant-magnitude spectrum
    # a_j = (1/N) * sum_k exp(i * phi_k) * exp(2*pi*i*j*k/N)
    # For non-negativity, use real sequences: a_j = (2/N) * sum_k cos(2*pi*j*k/N + phi_k) + DC
    # Simple approach: use a chirp
    x = np.arange(N, dtype=np.float64)
    # Chirp: a_j = |cos(pi * j^2 / N)|
    inits.append(('chirp', np.abs(np.cos(np.pi * (x / N) ** 2))))
    # Sum of cosines (low harmonic content)
    for freq in [1, 2, 3, 5, 7]:
        inits.append((f'cos_{freq}', np.abs(np.cos(2 * np.pi * freq * x / N)) + 0.3))
    # Quadratic phase
    inits.append(('quad_phase', np.abs(np.cos(np.pi * x / N * (x / N + 1)))))
    # Random
    inits.append(('rand1', np.clip(0.5 + 0.3 * rng.standard_normal(N), 0.01, None)))
    inits.append(('rand2', np.clip(0.5 + 0.5 * rng.standard_normal(N), 0.01, None)))
    # Block structure
    for nb in [8, 16, 32]:
        per = N // nb
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        inits.append((f'block_{nb}', a[:N].copy()))
    # Uniform
    inits.append(('uniform', np.ones(N) * 0.5))
    # Tapered
    inits.append(('taper', 0.5 + 0.4 * np.cos(np.pi * (x / (N-1) * 2 - 1)) ** 2))

    print(f"  Generated {len(inits)} initializations")

    # ============================================================
    # Phase 1: Quick screening - evaluate all inits, keep top ones
    # ============================================================
    print(f"\nPhase 1: Screening initializations")
    scored_inits = []
    for name, a0 in inits:
        sc = compute_score(a0)
        scored_inits.append((sc, name, a0))
    scored_inits.sort(key=lambda x: x[0])
    for sc, name, _ in scored_inits[:5]:
        print(f"  {name}: {sc:.6f}")

    # Take top 5 for detailed optimization
    top_inits = [(name, a0) for _, name, a0 in scored_inits[:5]]

    # ============================================================
    # Phase 2: Soft equalization + L-BFGS for each top init
    # ============================================================
    phase2_deadline = t_start + 0.5 * total_budget
    print(f"\nPhase 2: Soft equalization + L-BFGS (deadline at t={phase2_deadline-t_start:.0f}s)")

    for name, a0 in top_inits:
        if time.monotonic() > phase2_deadline:
            break
        print(f"\n  --- Init: {name} ---")
        a_cur = a0.copy()

        # Step 1: Soft equalization
        sc, a_eq = soft_equalize(a_cur, n_iters=100, alpha=0.4,
                                beta_range=(2.0, 20.0), deadline=phase2_deadline)
        _report(sc, a_eq, f"{name}/equalize")
        a_cur = a_eq

        # Step 2: L-BFGS with increasing beta
        for beta in [50, 200, 500]:
            if time.monotonic() > phase2_deadline:
                break
            remaining = phase2_deadline - time.monotonic()
            if remaining < 10:
                break
            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=3000, maxfun=50000,
                                  deadline=phase2_deadline)
            _report(sc, a_opt, f"{name}/lbfgs_b{beta}")
