import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)

"""
Improved step function optimization via soft LP-dual equalization and peak-targeted moves.

Key improvements over previous approach:
1. Soft dual weighting (exponential) instead of hard threshold
2. Adaptive peak-reduction moves targeting specific lags
3. Palindromic symmetry exploitation
4. Multi-length resampling with optimization
5. Higher beta LSE continuation for near-exact optimization
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def _lbfgs_obj(x, beta):
    """Wrapper for L-BFGS objective."""
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """Run L-BFGS-B to minimize LSE-smoothed score."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    options = {'ftol': 1e-15, 'gtol': 1e-13, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(100, int(remaining * 150)))
        options['maxiter'] = min(maxiter, max(100, int(remaining * 40)))
    try:
        res = minimize(_lbfgs_obj, a0, args=(beta,), method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def resample_signal(a, n_new):
    """Resample signal to new length using linear interpolation."""
    a = np.asarray(a, dtype=np.float64)
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.arange(n_old, dtype=np.float64)
    x_new = np.linspace(0, n_old - 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def soft_equalize(a, n_iters=100, deadline=None):
    """
    Soft LP-dual inspired equalization.
    Uses exponential weighting for the active lags instead of hard threshold.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (3 * n - 2).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        if M < 1e-15:
            break
        
        # Soft weighting: exponential of (c - M), normalized
        beta_dual = 50.0 + 50.0 * it / n_iters  # Anneal from soft to hard
        scaled = beta_dual * (c - M)
        scaled = np.clip(scaled, -200, 0)
        lambda_v = np.exp(scaled)
        lambda_sum = lambda_v.sum()
        if lambda_sum < 1e-30:
            lambda_v[int(np.argmax(c))] = 1.0
        else:
            lambda_v = lambda_v / lambda_sum

        # Compute h[j] = 2 * sum_k lambda[k] * a[k-j]
        # Cross-correlation of lambda and a at lags 0..n-1
        L = np.fft.rfft(lambda_v, M_fft)
        A = np.fft.rfft(a, M_fft)
        cross = np.fft.irfft(L * np.conj(A), M_fft)
        h = 2.0 * cross[:n]
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()

        # Hybrid update: multiplicative + additive
        alpha = 0.4 + 0.3 * (it / n_iters)  # Increase aggressiveness
        # Multiplicative part
        factor = h_mean / h
        factor = np.clip(factor, 0.1, 10.0)  # Prevent extreme values
        a_new = a * (1.0 - alpha + alpha * factor)
        
        # Ensure non-negativity and renormalize
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new[:] = 0.5
        else:
            a_new = a_new * (S / S_new)
        
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def peak_targeted_reduce(a, n_top=5, strength=0.02, deadline=None, n_rounds=20):
    """
    Targeted reduction of the top convolution peaks.
    For each peak lag, reduce the contributing elements proportionally.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        # Find top lags
        top_idx = np.argsort(c)[::-1][:n_top]
        
        a_new = a.copy()
        for k in top_idx:
            # Find contributing indices
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            # Contribution of each a[j] to c[k] is a[j] * a[k-j]
            contrib = a_new[j_start:j_end+1] * a_new[k - j_end:k - j_start:-1] if j_end > j_start else np.array([a_new[j_start] * a_new[k - j_start]])
            
            # Reduce elements with highest contribution
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            # Proportional reduction
            for i in range(j_start, j_end + 1):
                frac = contrib[i - j_start] / total_contrib
                a_new[i] -= strength * a_new[i] * frac * n_top
                if a_new[i] < 0:
                    a_new[i] = 0.0
        
        # Renormalize
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new * (S / S_new)
        
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def combinatorial_moves(a, n_moves=1000, deadline=None):
    """
    Try pairwise transfers between indices to reduce the peak convolution.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    best_c = fast_conv(a)
    best_M = float(best_c.max())
    
    rng = np.random.default_rng(12345)
    
    for mv in range(n_moves):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Find the peak lag
        k_star = int(np.argmax(best_c))
        
        # Find indices contributing to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        
        # Contribution vector
        contrib = a[j_start:j_end+1] * a[k_star - j_end:k_star - j_start:-1] if j_end > j_start else np.array([a[j_start] * a[k_star - j_start]])
        
        # Pick source: index with highest contribution to peak
        top_local = np.argmax(contrib)
        i_src = j_start + top_local
        
        # Pick destination: random index that contributes least to the peak
        # (or an index far from the peak lag)
        n_candidates = min(50, n)
        candidates = rng.integers(0, n, size=n_candidates)
        
        # Score each candidate by how much it would increase the peak if we added to it
        # Adding to a[j] increases c[k] by a[j] * a[k-j] (for each k)
        # We want to minimize the increase at k_star and other high lags
        best_j = candidates[0]
        best_score = np.inf
        for j in candidates:
            # Increase at peak lag
            inc_peak = a[j] if (0 <= k_star - j <= n - 1) else 0
            # Also check a few other high lags
            other_lags = np.argsort(best_c)[::-1][1:4]
            inc_other = 0.0
            for k in other_lags:
                if 0 <= k - j <= n - 1:
                    inc_other += a[j] * a[k - j]
            total_inc = inc_peak + 0.5 * inc_other
            if total_inc < best_score:
                best_score = total_inc
                best_j = j
        
        i_dst = best_j
        if i_src == i_dst:
            continue
        
        # Try the transfer
        delta = a[i_src] * 0.01  # 1% of source
        if delta < 1e-15:
            continue
        a_trial = a.copy()
        a_trial[i_src] -= delta
        a_trial[i_dst] += delta
        a_trial = np.clip(a_trial, 0.0, 1000.0)
        
        # Check improvement
        c_trial = fast_conv(a_trial)
        M_trial = float(c_trial.max())
        if M_trial < best_M:
            best_M = M_trial
            best_c = c_trial
            best_a = a_trial.copy()
            a = a_trial
            best_sc = compute_score(a)
    
    return best_sc, best_a

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient. Returns (score, grad)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def _lbfgs_obj(x, beta):
    """Top-level wrapper for L-BFGS objective."""
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """Run L-BFGS-B to minimize LSE-smoothed score. Returns (exact_score, array)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    options = {'ftol': 1e-15, 'gtol': 1e-14, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 3.0:
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(100, int(remaining * 120)))
        options['maxiter'] = min(maxiter, max(100, int(remaining * 35)))
    try:
        res = minimize(_lbfgs_obj, a0, args=(beta,), method='L-BFGS-B', jac=True,
                       bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def resample_signal(a, n_new):
    """Resample signal to new length using linear interpolation."""
    a = np.asarray(a, dtype=np.float64)
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.arange(n_old, dtype=np.float64)
    x_new = np.linspace(0, n_old - 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def soft_equalize(a, n_iters=100, deadline=None):

            break
        c = fast_conv(a)
        M = float(c.max())
        if M < 1e-15:
            break

        # Annealed beta: start soft, end hard
        frac = it / max(1, n_iters - 1)
        beta_dual = 20.0 + 200.0 * frac

        scaled = beta_dual * (c - M)
        scaled = np.clip(scaled, -300.0, 0.0)
        lambda_v = np.exp(scaled)
        lambda_sum = lambda_v.sum()
        if lambda_sum < 1e-30:
            lambda_v = np.zeros(len(c))
            lambda_v[int(np.argmax(c))] = 1.0
        else:
            lambda_v = lambda_v / lambda_sum

        # Compute h[j] = 2 * sum_k lambda[k] * a[(k-j) mod valid]
        L_fft = np.fft.rfft(lambda_v, M_fft)
        A_fft = np.fft.rfft(a, M_fft)
        cross = np.fft.irfft(L_fft * np.conj(A_fft), M_fft)
        h = 2.0 * cross[:n]

        h_mean = h.mean()

        factor = np.clip(factor, 0.05, 20.0)

        a = a_new
        sc = compute_score(a)

def palindromize(a):

    return 0.5 * (a + a[::-1])

    total_budget = float(budget_s)
    deadline = t_start + total_budget - 25.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        pv = globals().get('height_sequence_1', None)
        if pv is not None:
            p = np.asarray(pv, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024, dtype=np.float64) * 0.5

    # Helper inline (avoid closure by using nonlocal-free pattern)
    def update_best(sc, aa, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(aa)}, t={time.monotonic()-t_start:.1f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        a_cur = parent.copy()
        sc = compute_score(a_cur)
        update_best(sc, a_cur, "parent")
        print(f"Parent: score={sc:.8f}, n={len(a_cur)}")
    else:
        a_cur = np.ones(4096, dtype=np.float64) * 0.5
        update_best(compute_score(a_cur), a_cur, "uniform")

    # ============================================================
    # Phase 1: Quick structural explorations (120s)
    # ============================================================
    print(f"\n{'='*60}")
    print("Phase 1: Structural Exploration")
    print(f"{'='*60}")

    # Palindromic
    a_pal = palindromize(a_cur)
    update_best(compute_score(a_pal), a_pal, "palindromic")

    # Multi-length
    for n_try in [4096, 8192, 12288, 16384]:
        if time.monotonic() > t_start + 100:
            break
        a_res = resample_signal(best_a, n_try)
        sc, a_opt = run_lbfgs(a_res, 100, maxiter=1500, maxfun=25000, deadline=t_start + 110)
        update_best(sc, a_opt, f"len={n_try}")

    # ============================================================
    # Phase 2: Soft dual equalization + LSE (600s)
    # ============================================================
    phase2_end = t_start + 600.0
    print(f"\n{'='*60}")
    print(f"Phase 2: Soft Equalization (until t={phase2_end - t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    for rnd in range(20):
        if time.monotonic() > phase2_end:
            break
        # Soft equalization
        sc, a_eq = soft_equalize(a_cur, n_iters=50, deadline=phase2_end)
        update_best(sc, a_eq, f"eq r{rnd}")
        a_cur = a_eq

        # LSE refinement with increasing beta
        if time.monotonic() < phase2_end - 20:
            beta = 100 + 80 * rnd
            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=2500, maxfun=40000, deadline=phase2_end - 10)
            update_best(sc, a_opt, f"lbfgs r{rnd} b{beta}")
            a_cur = a_opt

    # ============================================================
    # Phase 3: Peak-targeted reduction (300s)
    # ============================================================
    phase3_end = t_start + 900.0
    print(f"\n{'='*60}")
    print(f"Phase 3: Peak Reduction (until t={phase3_end - t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    for rnd in range(8):
        if time.monotonic() > phase3_end:
            break
        strength = 0.01 + 0.005 * rnd
        sc, a_red = peak_targeted_reduce(a_cur, n_top=5, strength=strength,
                                          deadline=phase3_end, n_rounds=40)
        update_best(sc, a_red, f"peak r{rnd}")

        if time.monotonic() < phase3_end - 15:
            sc, a_opt = run_lbfgs(a_red, 200, maxiter=2000, maxfun=30000, deadline=phase3_end - 10)
            update_best(sc, a_opt, f"polish r{rnd}")
            a_cur = a_opt

    # ============================================================
    # Phase 4: Combinatorial moves (300s)
    # ============================================================
    phase4_end = t_start + 1200.0
    print(f"\n{'='*60}")
    print(f"Phase 4: Combinatorial Moves (until t={phase4_end - t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    sc, a_comb = combinatorial_moves(a_cur, n_moves=5000, deadline=phase4_end)
    update_best(sc, a_comb, "combinatorial")

    # Alternate combinatorial + LSE
    for rnd in range(3):
        if time.monotonic() > phase4_end:
            break
        sc, a_opt = run_lbfgs(best_a, 300, maxiter=2000, maxfun=30000, deadline=phase4_end - 10)
        update_best(sc, a_opt, f"comb-polish r{rnd}")
        sc, a_comb = combinatorial_moves(a_opt, n_moves=2000, deadline=phase4_end)
        update_best(sc, a_comb, f"comb r{rnd}")

    # ============================================================
    # Phase 5: Final high-beta polishing (remaining time)
    # ============================================================
    print(f"\n{'='*60}")
    print("Phase 5: Final Polishing")
    print(f"{'='*60}")

    for beta in [500, 1000, 2000, 4000, 6000]:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - 40 - time.monotonic()
        if remaining < 10:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=4000, maxfun=60000, deadline=deadline - 40)
        update_best(sc, a_opt, f"final b={beta}")

    # Subgradient polish
    if time.monotonic() < deadline - 30:
        sc, a_pol = subgrad_polish(best_a, total_iters=200, deadline=deadline - 25)
        update_best(sc, a_pol, "subgrad")

    # One final equalization + LSE
    if time.monotonic() < deadline - 20:
        sc, a_eq = soft_equalize(best_a, n_iters=80, deadline=deadline - 15)
        update_best(sc, a_eq, "final eq")
        if time.monotonic() < deadline - 10:
            sc, a_opt = run_lbfgs(best_a, 3000, maxiter=3000, maxfun=40000, deadline=deadline - 5)
            update_best(sc, a_opt, "final lbfgs")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic()-t_start:.1f}s / {budget_s}s")
    print(f"{'='*60}")

    if verified < best_score:
        best_score = verified

    return out
