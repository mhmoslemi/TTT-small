import numpy as np
from scipy.optimize import minimize
import time


def fast_self_conv(a):
    """Compute self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    score = 2n * LSE(beta * c) / S^2
    where LSE(x) = max(x) + log(sum(exp(x - max(x))))/beta

    log_sum = float(np.log(exp_scaled.sum())) + scaled.max() / beta
    L = c_max + log_sum / beta

"""
Step function optimization via LSE-smoothed custom Adam optimizer with active-set refinement.

Key improvements over previous approach:
1. Custom Adam optimizer with momentum (faster per-second than L-BFGS for this problem).
2. More gradual LSE continuation with adaptive beta scheduling.
3. Explicit active-set balancing: enforce KKT conditions on top coefficients.
4. Top-k smoothed max for final polishing (convex, better conditioned than exact max).
5. Multi-length exploration with upscaling from coarse solutions.
6. Targeted pairwise exchange local search.

import numpy as np
from scipy.optimize import minimize
import time


def fast_self_conv(a):
    """Compute self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta, M_cache=None):
    """
    LSE-smoothed score and gradient.
    Returns (score, grad).
    """
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled - scaled.max())
    log_sum_val = float(np.log(exp_scaled.sum())) + float(scaled.max()) / beta
    L = c_max + log_sum_val / beta

    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    # g[j] = sum_{l=0}^{n-1} a[l] * p[l+j] = conv(a_rev, p)[n-1+j]
    a_rev = a[::-1]
    N_conv = 1 << (3 * n - 2).bit_length()
    FA_rev = np.fft.rfft(a_rev, N_conv)
    FP = np.fft.rfft(p, N_conv)
    conv_result = np.fft.irfft(FA_rev * FP, N_conv)
    g_coeffs = conv_result[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (4.0 * n) * (g_coeffs / S2 - L * a / S3)

    return score, grad


def topk_score_grad(a, k):
    """
    Top-k smoothed score and subgradient.
    score = 2n * mean(top_k(c)) / S^2
    """
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    topk_indices = np.argpartition(c, -k)[-k:]
    topk_mean = float(c[topk_indices].mean())

    score = 2.0 * n * topk_mean / (S * S)

    # Subgradient: average of subgradients of the top-k c[k]
    # dc[k]/da[j] = 2*a[k-j] if 0<=k-j<n else 0
    # So dL/da[j] = (1/k) * sum_{k in topk} 2*a[k-j]
    p = np.zeros(2 * n - 1)
    p[topk_indices] = 1.0 / k

    a_rev = a[::-1]
    N_conv = 1 << (3 * n - 2).bit_length()
    FA_rev = np.fft.rfft(a_rev, N_conv)
    FP = np.fft.rfft(p, N_conv)
    conv_result = np.fft.irfft(FA_rev * FP, N_conv)
    g_coeffs = conv_result[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (4.0 * n) * (g_coeffs / S2 - topk_mean * a / S3)

    return score, grad


def adam_optimize(a0, beta, max_steps=50000, lr=0.01, deadline=None,
                  beta1=0.9, beta2=0.999, eps=1e-8, decay=0.995,
                  decay_interval=500, use_topk=False, topk_k=50):
    """
    Custom Adam optimizer for the LSE-smoothed objective.
    Much faster per-second than L-BFGS for large n.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 1e-6:
        a[:] = 0.5

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_score = compute_score(a)
    best_a = a.copy()
    score_tol = best_score * 1e-9

    if use_topk:
        topk_k_val = min(topk_k, 2 * n - 1)

    while t < max_steps:
        if deadline is not None and time.monotonic() > deadline:
            break

        if use_topk:
            _, grad = topk_score_grad(a, topk_k_val)
        else:
            _, grad = lse_score_grad(a, beta)

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)

        if t % decay_interval == 0:
            lr *= decay
            # Periodically check exact score
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            if lr < 1e-10:
                break

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_score, best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None, use_topk=False, topk_k=50):
    """Run L-BFGS-B to minimize the smoothed score."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    if use_topk:
        def obj_wrapper(x):
            return topk_score_grad(x, topk_k)
    else:
        def obj_wrapper(x):
            return lse_score_grad(x, beta)

    options = {'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter,
               'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            sc = compute_score(a0)
            return sc, a0
        options['maxfun'] = min(maxfun, max(50, int(remaining * 150)))
        options['maxiter'] = min(maxiter, max(50, int(remaining * 30)))

    try:
        res = minimize(obj_wrapper, a0, method='L-BFGS-B', jac=True,
                       bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

        # Compute g[j] = sum_l a[l] * p[l+j]
        a_rev = a[::-1]
        FA_rev = np.fft.rfft(a_rev, N_conv)
        FP = np.fft.rfft(p, N_conv)
        conv_result = np.fft.irfft(FA_rev * FP, N_conv)
        g_coeffs = conv_result[n - 1:2 * n - 1]

        # KKT condition: g[j] should equal (L/S) * a[j]
        # where L = mean of active c values
        L = float(c[active].mean())
        target = (L / S) * a

        # Residual: how much g deviates from target
        residual = g_coeffs - target
        r_norm = float(np.linalg.norm(residual))
        if r_norm < 1e-12:
            break

        # Update: move a in direction that reduces residual
        # If g[j] > target[j], reduce a[j]; if g[j] < target[j], increase a[j]
        # But we need to be careful about the coupling.
        # Simple approach: multiplicative update
        ratio = np.maximum(target, 1e-15) / np.maximum(g_coeffs, 1e-15)
        factor = 1.0 - alpha + alpha * ratio
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)

        # Renormalize
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a = a * (S / S_new)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def targeted_local_search(a, rng, n_trials=50, deadline=None):
    """
    Targeted local search: perturb positions that contribute most to peak lags.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for trial in range(n_trials):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_self_conv(best_a)
        k_star = int(np.argmax(c))

        # Find positions contributing to the peak lag
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        contrib = np.zeros(n)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = best_a[j_start:j_end + 1][::-1]

        # Pick a few positions to reduce (high contribution) and increase (low contribution)
        top_k = max(5, n // 100)
        top_idx = np.argsort(contrib)[::-1][:top_k]
        bottom_idx = np.argsort(contrib)[:top_k]

        a_pert = best_a.copy()
        total_reduced = 0.0
        scale = 0.02 + 0.06 * rng.random()
        for j in top_idx:
            amt = a_pert[j] * scale * rng.random()
            a_pert[j] -= amt
            a_pert[j] = max(a_pert[j], 0.0)
            total_reduced += amt

        if total_reduced > 0:
            for j in bottom_idx:
                a_pert[j] += total_reduced / len(bottom_idx) * rng.random()

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick Adam refinement
        sc, a_opt = adam_optimize(a_pert, 200, max_steps=2000, lr=0.005,
                                  deadline=deadline, decay=0.99, decay_interval=200)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()

    return best_sc, best_a


def gaussian_perturb_search(a, rng, n_trials=30, deadline=None):
    """Gaussian perturbation + quick re-optimization."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for trial in range(n_trials):
        if deadline is not None and time.monotonic() > deadline:
            break

        scale = 0.01 + 0.05 * rng.random()
        a_pert = a * (1.0 + scale * rng.standard_normal(n))
        a_pert = np.clip(a_pert, 0.0, 1000.0)

        sc, a_opt = adam_optimize(a_pert, 300, max_steps=3000, lr=0.005,
                                  deadline=deadline, decay=0.99, decay_interval=300)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()

    return best_sc, best_a


def build_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        p = parent.astype(np.float64)
        if len(p) != n:
            p = np.interp(np.linspace(0, len(p) - 1, n), np.arange(len(p)), p)
        inits.append(np.clip(p, 0.01, 1000.0).copy())

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Raised cosine (Hann window)
    inits.append(0.3 + 0.7 * np.sin(np.pi * x) ** 2)

    # Various cosine modulations
    inits.append(0.4 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.5 * np.sin(np.pi * x))
    inits.append(0.2 + 0.6 * np.sin(np.pi * x) ** 2)
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x))

    # Gaussian-like
    sigma = 0.3
    inits.append(np.exp(-0.5 * ((x - 0.5) / sigma) ** 2))

    # Quadratic bump
    inits.append(np.maximum(0.1, 1.0 - (2.0 * x - 1.0) ** 2))

    # Random
    inits.append(np.clip(0.5 + 0.1 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))

    # Block structure
    for nb in (4, 8, 16):
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())

    return inits


def upscale_coarse(coarse, n_fine):
    """Upscale a coarse solution to fine resolution using cubic interpolation."""
    n_coarse = len(coarse)
    if n_coarse == n_fine:
        return coarse.copy()
    x_coarse = np.arange(n_coarse, dtype=np.float64)
    x_fine = np.linspace(0, n_coarse - 1, n_fine)
    fine = np.interp(x_fine, x_coarse, coarse, kind='cubic')
    return np.clip(fine, 0.0, 1000.0)


def subgradient_polish(a, n_iters=500, deadline=None):
    """Exact subgradient descent with decreasing step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    base_lower = 2.0 * n / (2 * n - 1)  # Theoretical lower bound

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find all active lags (within 1% of max)
        active = (c >= M * 0.99)
        n_active = int(active.sum())

        # Subgradient: average over active lags
        p = active.astype(np.float64) / n_active
        a_rev = a[::-1]
        N_conv = 1 << (3 * n - 2).bit_length()
        FA_rev = np.fft.rfft(a_rev, N_conv)
        FP = np.fft.rfft(p, N_conv)
        conv_result = np.fft.irfft(FA_rev * FP, N_conv)
        g_coeffs = conv_result[n - 1:2 * n - 1]

        S2 = S * S
        S3 = S2 * S
        grad = (4.0 * n) * (g_coeffs / S2 - M * a / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Step size: (f - f_lower) / ||g||^2 with decay
        gap = max(sc - base_lower, 1e-6)
        step = gap / (gn * gn + 1e-30)
        step = min(step, 0.1 / (1.0 + 0.01 * it))

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

        # Small additional step for progress
        step2 = 0.001 / (1.0 + 0.005 * it)
        a -= step2 * grad / (gn + 1e-15)
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a
