"""
Optimized self-convolution peak minimizer for non-negative sequences.
Uses aggressive LSE-continuation L-BFGS with very high beta values,
Nesterov-accelerated subgradient, and spectral self-convolution flattening.
Key improvements over previous: higher beta schedule, Nesterov acceleration,
spectral phase-aware iterations, and more time devoted to final polishing.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def nesterov_subgrad_polish(a, total_iters=2000):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    gamma = 0.9
    sc_lb = 1.0
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        t = 2.0 / (it + 1 + gamma - 1)
        y = (1 - t) * a + t * v
        v = y - step * grad
        a_new = np.clip(v, 0.0, 1000.0)
        sc_new = compute_score(a_new)
        if sc_new > sc * 1.01:
            a = np.clip(y - 0.5 * step * grad, 0.0, 1000.0)
            v = a.copy()
        else:
            a = a_new
    return best_sc, best_a


def spectral_selfconv_flatten(a0, n_iters=200, lr=0.3):
    """
    Spectral flattening specific to self-convolution peak reduction.
    Iteratively adjusts the spectrum to make IFFT(B^2) flat while maintaining a >= 0.
    """
    a = a0.astype(np.float64).copy()
    n = len(a)
    N_fft = 1 << (2 * n - 1).bit_length()
    for it in range(n_iters):
        B = np.fft.rfft(a, N_fft)
        B2 = B * B
        c = np.fft.irfft(B2, N_fft)[:2 * n - 1]
        c_full = np.fft.irfft(B2, N_fft)
        c_max = float(c.max())
        c_min = float(c.min())
        c_range = c_max - c_min
        if c_range < 1e-10:
            break
        # Target: flatten c by reducing high values and boosting low values
        # In frequency domain: adjust B to reduce peak of IFFT(B^2)
        # Gradient of max(c) wrt real B: if peak at k*, grad_B[m] = 2*Re(B[m]*exp(i*2*pi*m*k*/N))/N
        k_star = int(np.argmax(c))
        phase = np.arange(len(B)) * 2.0 * np.pi * k_star / N_fft
        grad_B = 2.0 * (B * np.exp(1j * phase))
        # Dampen: reduce B where it contributes to the peak
        decay = 1.0 - lr * 0.1
        B_new = B * decay + (1.0 - decay) * B
        # Also add a small flattening of |B|
        mag = np.abs(B)
        if mag[0] > 0:
            non_dc = mag[1:]
            target = np.median(non_dc)
            mag_adjusted = mag.copy()
            half = N_fft // 2
            for k in range(1, min(len(mag) - 1, half + 1)):
                alpha = 0.1 * lr
                mag_adjusted[k] = (1 - alpha) * mag[k] + alpha * target
            if N_fft % 2 == 0 and len(mag) > half:
                mag_adjusted[half] = mag[half]
            # Reconstruct with adjusted magnitude and original phase
            phase_B = np.angle(B)
            B_mag_adj = mag_adjusted * np.exp(1j * phase_B)
            B_new = 0.7 * B_new + 0.3 * B_mag_adj
        a_new = np.real(np.fft.ifft(B_new))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a


def greedy_peak_reduction(a0, n_steps=100):
    """
    Greedy approach: repeatedly find the peak of the convolution and
    redistribute mass from peak-contributing elements to others.
    """
    a = a0.astype(np.float64).copy()
    n = len(a)
    S = a.sum()
    for step in range(n_steps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        # Elements contributing to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        seg = a[j_start:j_end + 1].copy()
        contrib = seg * seg[::-1]  # contribution of each pair
        # Reduce elements that contribute most to the peak
        top_k = max(2, n // 100)
        idx_sorted = np.argsort(contrib)[::-1]
        n_reduce = min(top_k, len(idx_sorted) - 1)
        if n_reduce <= 0:
            break
        reduce_idx = idx_sorted[:n_reduce]
        # Find elements to increase (low contribution, non-peak region)
        all_idx = np.arange(j_start, j_end + 1)
        reduce_global = all_idx[reduce_idx]
        increase_pool = np.setdiff1d(np.arange(n), reduce_global)
        if len(increase_pool) == 0:
            break
        # Pick low-value elements from increase pool
        pool_vals = a[increase_pool]
        n_incr = min(len(reduce_idx), max(2, len(increase_pool) // 10))
        bottom_idx_pool = np.argsort(pool_vals)[:n_incr]
        increase_global = increase_pool[bottom_idx_pool]
        # Redistribute
        total_reduce = 0.0
        for idx in reduce_global:
            amt = a[idx] * 0.05
            a[idx] -= amt
            a[idx] = max(a[idx], 0.0)
            total_reduce += amt
        per_incr = total_reduce / len(increase_global)
        for idx in increase_global:
            a[idx] += per_incr
        a = np.clip(a, 0.0, 1000.0)
        # Renormalize
        S_new = a.sum()
        if S_new > 1e-12:
            a = a * (S / S_new)
    return a


def build_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    # Geometric decay from left
    for r in [0.98, 0.95, 0.9, 0.8]:
        inits.append(r ** x)
    # Geometric decay from right
    for r in [0.98, 0.95, 0.9]:
        inits.append(r ** (n - 1 - x))
    # Asymmetric: large at one end
    inits.append(1.0 + 5.0 * np.exp(-x / (n * 0.1)))
    inits.append(1.0 + 5.0 * np.exp(-(n - 1 - x) / (n * 0.1)))
    # Sinusoidal
    for freq in [1, 2, 3]:
        inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * freq * x / n))
    # Binomial-like
    from math import lgamma
    k = n - 1
    log_binom = np.array([lgamma(k + 1) - lgamma(j + 1) - lgamma(k - j + 1) for j in range(n)])
    binom = np.exp(log_binom - log_binom.max())
    inits.append(binom / binom.sum())
    # Random with smooth structure
    for scale in [0.5, 1.0, 2.0]:
        rough = rng.standard_normal(n)
        # Smooth it
        from scipy.ndimage import uniform_filter1d
        smooth = uniform_filter1d(rough, size=max(2, n // 50))
        inits.append(np.clip(0.5 + scale * smooth, 0.01, 10.0))
    # Chirp-based (for self-convolution)
    for f in [1.0, 2.0, 3.0]:
        inits.append(0.5 + 0.3 * np.cos(np.pi * f * x * x / (2.0 * n)))
    # Two-peak structure
    peak1 = int(n * 0.3)
    peak2 = int(n * 0.7)
    inits.append(np.exp(-((x - peak1) ** 2) / (2 * (n * 0.05) ** 2)) + np.exp(-((x - peak2) ** 2) / (2 * (n * 0.05) ** 2)))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses LSE-continuation L-BFGS with very high beta, Nesterov subgradient,
    and spectral self-convolution flattening.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0
    deadline = t_start + effective_budget

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        n = parent.size
        print(f"Using parent of length n={n}")
    else:
        n = 4096
        print(f"No parent, using n={n}")

    # ============================================================
    # Phase 1: Generate and screen initializations
    # ============================================================
    print("\n--- Phase 1: Initialization ---")
    init_deadline = t_start + 0.08 * effective_budget

    candidates = []
    if parent is not None:
        candidates.append(("parent", parent.copy()))

    # Structured inits
    struct_inits = build_structured_inits(n, rng)
    for i, si in enumerate(struct_inits[:8]):
        candidates.append((f"struct_{i}", si.astype(np.float64)))

    # Random
    for i in range(4):
        candidates.append((f"rand_{i}", np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 10.0)))

    # Quick screen
    print(f"  {len(candidates)} candidates to screen...")
    screened = []
    for name, cand in candidates:
        if time.monotonic() > init_deadline:
            break
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.1:
            continue
        sc = compute_score(cand)
        screened.append((sc, name, cand))
    screened.sort(key=lambda x: x[0])
    print(f"  Top init: {screened[0][1]} with score {screened[0][0]:.6f}")

    # Take top 5 for optimization
    top_inits = [(name, cand) for sc, name, cand in screened[:5]]
    for name, cand in top_inits:
        report(compute_score(cand), cand, f"init:{name}")

    # ============================================================
    # Phase 2: L-BFGS with LSE continuation (main workhorse)
    # ============================================================
    print("\n--- Phase 2: L-BFGS Continuation ---")
    lbfgs_deadline = t_start + 0.65 * effective_budget

    beta_schedule = [30, 50, 100, 200, 500, 1000, 2000, 5000]

    for ci, (init_name, a_init) in enumerate(top_inits):
        if time.monotonic() > lbfgs_deadline:
            break
        print(f"\n  >> Candidate {ci}: {init_name}")
        a_cur = a_init.copy()
        for bi, beta in enumerate(beta_schedule):
            if time.monotonic() > lbfgs_deadline:
                break
            remaining = lbfgs_deadline - time.monotonic()
            if remaining < 10:
                break
            # Scale iterations based on remaining time
            # For n~7000, each LSE eval takes about 0.01-0.03s
            est_per_fcall = 0.02 * (n / 4096)
            maxfun = int(remaining * 0.7 / max(est_per_fcall, 0.005))
            maxfun = min(max(maxfun, 500), 80000)
            maxiter = max(100, maxfun // 3)
            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
            a_cur = a_opt
            report(sc, a_opt, f"lbfgs {init_name} beta={beta}")
            if sc < best_score:
                best_a = a_opt.copy()

        # Final score for this trajectory
        sc_final = compute_score(a_cur)
        report(sc_final, a_cur, f"lbfgs_final {init_name}")

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Very high beta L-BFGS polishing
    # ============================================================
    print("\n--- Phase 3: High-Beta Polish ---")
    polish_deadline = t_start + 0.80 * effective_budget

    for beta in [5000, 10000, 20000]:
        if time.monotonic() > polish_deadline:
            break
        remaining = polish_deadline - time.monotonic()
        if remaining < 10:
            break
        est_per_fcall = 0.02 * (n / 4096)
        maxfun = int(remaining * 0.6 / max(est_per_fcall, 0.005))
        maxfun = min(max(maxfun, 200), 50000)
        maxiter = max(200, maxfun // 3)
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"polish beta={beta}")

    # ============================================================
    # Phase 4: Nesterov subgradient
    # ============================================================
    print("\n--- Phase 4: Nesterov Subgradient ---")
    subgrad_deadline = t_start + 0.92 * effective_budget
    remaining_iters = int((subgrad_deadline - time.monotonic()) / 0.02)
    remaining_iters = max(200, min(remaining_iters, 5000))
    print(f"  Running {remaining_iters} Nesterov iterations...")
    sc, a_opt = nesterov_subgrad_polish(best_a, total_iters=remaining_iters)
    report(sc, a_opt, "nesterov_subgrad")

    # ============================================================
    # Phase 5: Final L-BFGS + spectral + subgradient round
    # ============================================================
    print("\n--- Phase 5: Final Refinement ---")
    final_deadline = deadline - 10

    # One more L-BFGS at very high beta
    if time.monotonic() < final_deadline - 30:
        sc, a_opt = run_lbfgs(best_a, 20000, maxiter=5000, maxfun=30000)
        report(sc, a_opt, "final_lbfgs_20k")

    # Spectral self-convolution flattening
    if time.monotonic() < final_deadline - 30:
        print("  Spectral self-conv flatten...")
        t0 = time.monotonic()
        a_spec = spectral_selfconv_flatten(best_a, n_iters=100, lr=0.5)
        sc_spec = compute_score(a_spec)
        report(sc_spec, a_spec, "spectral")
        print(f"  Spectral done in {time.monotonic()-t0:.1f}s")

        # L-BFGS from spectral result
        if time.monotonic() < final_deadline - 15:
            sc, a_opt = run_lbfgs(a_spec, 5000, maxiter=3000, maxfun=20000)
            report(sc, a_opt, "post_spectral_lbfgs")

    # Final Nesterov
    if time.monotonic() < final_deadline - 10:
        remaining_iters = int((final_deadline - time.monotonic() - 5) / 0.02)
        remaining_iters = max(100, min(remaining_iters, 3000))
        sc, a_opt = nesterov_subgrad_polish(best_a, total_iters=remaining_iters)
        report(sc, a_opt, "final_nesterov")

    # ============================================================
    # Phase 6: Greedy peak reduction (last resort)
    # ============================================================
    if time.monotonic() < final_deadline - 5:
        a_greedy = greedy_peak_reduction(best_a, n_steps=50)
        sc_greedy = compute_score(a_greedy)
        report(sc_greedy, a_greedy, "greedy")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
