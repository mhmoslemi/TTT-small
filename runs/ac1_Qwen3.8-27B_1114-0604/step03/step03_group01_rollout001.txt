from scipy.optimize import linprog

def lp_step(f, M):
    """Solve the LP: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    # Build constraint matrix: (f*g)(k) = sum_j f_j * g_{k-j} <= M
    # For k = 0, ..., 2n-2:
    # (f*g)(k) = sum_{j=max(0,k-n+1)}^{min(k,n-1)} f_{k-j} * g_j
    # So the constraint matrix A has rows indexed by k, columns by j:
    # A[k,j] = f_{k-j} if max(0,k-n+1) <= j <= min(k,n-1), else 0
    
    # This is a banded matrix! Each row has at most n non-zero entries.
    # For efficiency, we can use scipy's sparse matrices.
    
    from scipy.sparse import lil_matrix
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        for j in range(js, je+1):
            A[k, j] = f[k-j]
    
    # Maximize sum(g) = c^T g where c = 1
    # Subject to: A g <= M * 1, g >= 0
    c = np.ones(n)
    b = M * np.ones(2*n-1)
    
    res = linprog(-c, A_ub=A.tocsr(), b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        g = res.x
        return g
    return None

def fast_conv(a):

    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_score(a):
    """Exact score and subgradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

def lp_solve(f, M, lags):

        js = max(0, k - n + 1)
        je = min(k, n - 1)

            S = a.sum()

            candidates = np.clip(candidates, 0.0, 1000.0)
            
            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = x
                if a.sum() < 1e-12:
                    a[j] = a_old
                    continue
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
    
    final_sc = compute_score(a)
    if final_sc < best_score:

def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, np.maximum(a, 0))
    return np.clip(a_new, 0.0, 1000.0)


def softmax_fn(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def adam_lse(a_init, lr=0.005, beta_lse=200, time_limit=60.0, t_start=None):
    """Adam with LSE-smoothed gradient, in a-space (normalized)."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_conv(a).max())
    deadline = (t_start + time_limit) if (t_start is not None) else np.inf
    
    t = 0
    while time.monotonic() < deadline:
        t += 1
        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad(a, beta_lse)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        
        if t % 5000 == 0:
            sc = 2.0 * n * float(fast_conv(a).max())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = 2.0 * n * float(fast_conv(a).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a * S
