from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def topk_lse_score_grad(a, beta, K):
    """
    Top-K LSE smoothed score and gradient.
    LSE is computed only over the K largest convolution values for tighter approximation.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    # Find top-K indices
    if K >= len(c):
        topk_idx = np.arange(len(c))
    else:
        topk_idx = np.argpartition(c, -K)[-K:]
    
    c_topk = c[topk_idx]
    c_max = float(c_topk.max())
    scaled = beta * (c_topk - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    L = c_max + float(np.log(exp_s.sum())) / beta
    p = exp_s / exp_s.sum()  # weights over top-K
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient: dL/da[j] = sum_{k in topK} p_k * a[k-j] (for valid k-j)
    # Use cross-correlation: for each k in topK, the contribution to grad is
    # p_k * a[k-j] reversed
    # grad[j] = sum_{k in topK} p_k * a[k-j] (where 0 <= k-j <= n-1)
    # This is a weighted sum of shifted/reversed versions of a
    # Equivalent to: convolve a with a weighted impulse train at topK positions
    
    # Build the weight vector for the cross-correlation
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    
    # Create array w of length 2n-1 with w[k] = p weighted
    w = np.zeros(2 * n - 1)
    for i, k in enumerate(topk_idx):
        w[k] = p[i]
    
    # grad[j] = sum_k w[k] * a[k-j] for valid k,j
    # = (w * a_rev)[n-1-j] ... let me think more carefully
    # Actually: grad[j] = sum_k w[k] * a[k-j] where 0<=j<n, 0<=k-j<n
    # = sum_{m} w[j+m] * a[m] where m = k-j, 0<=m<n, 0<=j+m<2n-1
    # This is a cross-correlation of w and a
    # grad = cross_corr(w, a) where cross_corr(w,a)[j] = sum_m w[j+m]*a[m]
    
    # Using FFT: cross_corr(w, a) = ifft(conj(fft(w)) * fft(a)) ... 
    # Actually, let me use the direct formula:
    # grad[j] = sum_{m=0}^{n-1} a[m] * w[j+m]
    # This is the (1-lagged) convolution of a with w
    # conv(a, w)[j + n - 1] = sum_m a[m] * w[j + n - 1 - m]... no
    # Let me just use: grad[j] = sum_m a[m] * w[j+m]
    # = (reverse_conv)(a, w)
    # Using FFT: ifft(fft(a) * conj(fft(w))) gives the cross-correlation
    # corr[j] = sum_m a[m] * conj(w[m-j]) ... this is getting confusing
    
    # Simpler approach: use the relationship
    # C(k) = sum_j a[j]*a[k-j], so dC(k)/da[j] = a[k-j]
    # grad_LSE[j] = sum_k p_k * a[k-j] (for valid k-j)
    # Let's define b[j] = sum_k p_k * a[k-j]
    # b[j] for j=0..n-1: b[j] = sum_{k=j}^{j+n-1} p_k * a[k-j] (where p_k = 0 outside topK)
    # b = convolve(p_array, a_rev) ... 
    # b[j] = sum_m a[m] * p[j+m] where m=0..n-1, j+m=0..2n-2
    # This is the "cross-correlation" of p and a: b = corr(p, a)
    # Using FFT: b[j] = ifft(conj(FFT(p)) * FFT(a))[j]... 
    # Actually: sum_m a[m]*p[j+m] = (a * p_rev)[j + n - 1] where p_rev is p reversed
    # Or: let's just compute it as: b = ifft(FFT(p, M2) * FFT(a, M2)) ... no
    
    # The cleanest way: b[j] = sum_m a[m] * p[j+m]
    # Define P[m] = p[m] for m=0..2n-2, 0 elsewhere
    # Then b[j] = sum_m a[m] * P[j+m]
    # Let r[m] = P[2n-2-m] (reverse of P)
    # b[j] = sum_m a[m] * P[j+m] = sum_m a[m] * r[2n-2-j-m]
    # = (a * r)[2n-2-j] (convolution)
    # So b = conv(a, r) reversed and shifted
    
    # Even simpler: use the fact that for each topk index k, the contribution to grad is
    # p_i * [a[k-j] for j=0..n-1 where 0<=k-j<=n-1]
    # = p_i * a[max(0,k-n+1):min(k+1,n)] reversed
    
    # Let me just do it the straightforward way with FFT cross-correlation:
    # b[j] = sum_m a[m] * w[j+m] = cross-correlation of w with a
    # In FFT terms: b = ifft(fft(w, N) * conj(fft(a, N))) ... 
    # Wait no. Let me define:
    # (f ⋆ g)[j] = sum_k f[k] * g[k-j] (cross-correlation)
    # We want b[j] = sum_k w[k] * a[k-j] = (w ⋆ a)[j]
    # In FFT: FFT(w ⋆ a) = conj(FFT(w)) * FFT(a)
    # So b = ifft(conj(FFT(w)) * FFT(a))
    
    WA = np.fft.rfft(w, M2_fft)
    AA = np.fft.rfft(a, M2_fft)
    b = np.fft.irfft(np.conj(WA) * AA, M2_fft)[:n]
    # b[j] should be sum_k w[k]*a[k-j] for 0<=j<n
    # But we need to check the indexing...
    # Actually for linear cross-correlation: (w ⋆ a)[j] = sum_k w[k]*a[k-j]
    # The FFT gives circular cross-correlation, so we need zero-padding
    
    # Let me verify: if w has support [0, 2n-2] and a has support [0, n-1],
    # then (w ⋆ a)[j] for j=0..n-1 involves w[k]*a[k-j] for k=j..j+n-1
    # with w[k] potentially non-zero for k up to 2n-2.
    # The cross-correlation has support [-n+1, 2n-2], so we need padding of at least 3n-2.
    # M2_fft should be enough.
    
    # Actually, I realize there might be an offset issue. Let me just compute directly
    # for small cases to verify, but for now let me use a different approach:
    
    # Direct computation via the "shifted sum" approach:
    # b[j] = sum_{k in topK} p[k] * a[k-j] (where 0 <= k-j <= n-1)
    # For a given j, the valid k range is [j, j+n-1] intersected with topK.
    # This is expensive to compute directly for large K.
    
    # Let me use the FFT approach but be more careful about indexing.
    # The linear cross-correlation (w ⋆ a)[j] = sum_{k} w[k] a[k-j]
    # For the FFT: we zero-pad both to length >= (2n-1)+(n-1) = 3n-2
    # Then ifft(conj(FFT(w)) * FFT(a)) gives the circular cross-correlation
    # which equals the linear one (due to zero-padding).
    # The result has the cross-correlation at indices 0..(3n-3),
    # where (w ⋆ a)[j] is at index j for j >= 0 and at index 3n-2+j for j < 0.
    # We want j = 0..n-1, so we take indices 0..n-1 of the result.
    
    # Wait, I think the standard cross-correlation via FFT is:
    # corr = ifft(conj(FFT(w)) * FFT(a))
    # This gives corr[j] = sum_k w[k] * a[(k-j) mod N] (circular)
    # With zero-padding, for j in [0, len(a)-1], this gives the linear cross-correlation.
    
    # Actually, I realize my formula might have an index offset. Let me just use
    # the "flip and convolve" approach:
    # b[j] = sum_k w[k]*a[k-j] = sum_m w[j+m]*a[m] (substituting m=k-j)
    # = (w_shifted * a)[0] where w_shifted starts at position j
    # This is the same as: b = ifft(FFT(w) * FFT(a)) ... no that's convolution.
    
    # Let me just use: b[j] = sum_m a[m] * w[j+m]
    # Define W of length 2n-1 with W[k] = w[k]. Define A of length n with A[m] = a[m].
    # b[j] = sum_m a[m]*w[j+m] = sum_m A[m] * W[j+m]
    # This is the "anti-convolution" or "cross-correlation" of A and W.
    # In FFT: FFT(A cross W) = conj(FFT(A)) * FFT(W) ... 
    # Hmm, let me just use the direct formula with the flip:
    # b[j] = sum_m A[m] * W[j+m]
    # Let W_rev[m] = W[2n-2-m] (flip W)
    # b[j] = sum_m A[m] * W[j+m] = sum_m A[m] * W_rev[2n-2-j-m]
    # = (A * W_rev)[2n-2-j] (standard convolution)
    
    W_rev = w[::-1]
    # conv = A * W_rev (linear convolution, length n + (2n-1) - 1 = 3n-2)
    conv_full = np.fft.irfft(np.fft.rfft(a, M2_fft) * np.fft.rfft(W_rev, M2_fft), M2_fft)[:3*n-2]
    # b[j] = conv_full[2n-2-j] for j=0..n-1
    b = conv_full[2*n-2 : n-1 : -1]  # indices 2n-2, 2n-3, ..., n-1
    
    g = 2.0 * b
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    
    return score, grad


def topk_lse_wrapper(a, beta, K):
    """Scipy-compatible objective with top-K LSE."""
    return topk_lse_score_grad(a, beta, K)


def run_lbfgs_topk(a0, beta, K, maxiter=3000, maxfun=60000, time_limit_s=None):
    """Run L-BFGS-B on top-K LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    t0 = time.monotonic()
    
    # Custom callback to check time
    def time_callback(xk):
        if time_limit_s is not None and time.monotonic() - t0 > time_limit_s:
            raise TimeoutError()
        return False
    
    try:
        res = minimize(
            topk_lse_wrapper, a0, args=(beta, K),
            method='L-BFGS-B', jac=True, bounds=bounds,
            callback=time_callback,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
        )
    except (TimeoutError, Exception):
        # If timeout, return current best from the optimization
        # We'll recompute from a0 perturbed slightly
        a_out = np.clip(a0, 0.0, 1000.0)
        return compute_score(a_out), a_out, False
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.success


def full_lse_score_grad(a, beta):
    """Full LSE smoothed score and gradient (over all 2n-1 lags)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    L = c_max + float(np.log(exp_s.sum())) / beta
    p = exp_s / exp_s.sum()
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    rev_a = a[::-1]
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    
    return score, grad


def full_lse_wrapper(a, beta):
    return full_lse_score_grad(a, beta)


def run_lbfgs_full(a0, beta, maxiter=3000, maxfun=60000, time_limit_s=None):
    """Run L-BFGS-B on full LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    t0 = time.monotonic()
    
    def time_callback(xk):
        if time_limit_s is not None and time.monotonic() - t0 > time_limit_s:
            raise TimeoutError()
        return False
    
    try:
        res = minimize(
            full_lse_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            callback=time_callback,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
        )
    except (TimeoutError, Exception):
        a_out = np.clip(a0, 0.0, 1000.0)
        return compute_score(a_out), a_out, False
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.success


def peak_shaving(a, max_iters=5000, topM=10, step_decay=0.999):
    """
    Direct peak-shaving: iteratively reduce the M largest autocorrelation peaks
    using subgradient steps with Polyak-style adaptive step size.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
    
    step_scale = 1.0
    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top-M peaks
        if topM >= len(c):
            topk_idx = np.arange(len(c))
        else:
            topk_idx = np.argpartition(c, -topM)[-topM:]
        
        # Compute combined subgradient (average over active peaks)
        grad = np.zeros(n)
        for k in topk_idx:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                grad[j_start:j_end + 1] += 2.0 * seg[::-1]
        grad /= len(topk_idx)
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad / S2 - 2.0 * M / S3)
        
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        
        # Polyak step with decay
        step = (sc - sc_lb) / gn2 * step_scale
        step = min(step, 1.0 / (1.0 + 0.001 * it))
        step = max(step, 1e-10)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        step_scale *= step_decay
        
        if step_scale < 1e-6:
            break
    
    return best_sc, best_a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else x
    
    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())
        # Perturbed parent
        a_p = parent * (1.0 + 0.05 * rng.standard_normal(n))
        inits.append(np.clip(a_p, 0.01, 1000.0))
        a_p = parent * (1.0 + 0.02 * rng.standard_normal(n))
        inits.append(np.clip(a_p, 0.01, 1000.0))
    
    # Linear ramps (anti-palindromic)
    inits.append(np.linspace(0.05, 1.0, n))
    inits.append(np.linspace(1.0, 0.05, n))
    
    # Power ramps
    inits.append(0.05 + 0.95 * xn ** 1.5)
    inits.append(0.05 + 0.95 * (1 - xn) ** 1.5)
    inits.append(0.05 + 0.95 * xn ** 2)
    inits.append(0.05 + 0.95 * (1 - xn) ** 2)
    
    # Anti-palindromic sinusoidal
    for freq in [1, 2, 3, 5, 7]:
        inits.append(np.clip(0.5 + 0.4 * np.sin(np.pi * freq * xn), 0.01, 1000.0))
    
    # Comb-like (high-low alternating blocks)
    block = max(1, n // 20)
    pattern = np.tile([1.0, 0.3], n // block + 1)[:n]
    inits.append(pattern * 0.5 + 0.25)
    
    # Smooth random
    a_rand = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0)
    ksize = max(1, n // 100)
    kernel = np.ones(ksize) / ksize
    a_smooth = np.convolve(a_rand, kernel, mode='same')
    inits.append(np.clip(a_smooth, 0.01, 1000.0))
    
    # Pure random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    
    # Two-block
    half = n // 2
    inits.append(np.concatenate([np.ones(half) * 0.8, np.ones(n - half) * 0.2]))
    inits.append(np.concatenate([np.ones(half) * 0.2, np.ones(n - half) * 0.8]))
    
    # Spectral: try to create flat power spectrum
    # Target: |A(k)|^2 = S^2/(2n) for k != 0
    # A(0) = S, so |A(0)|^2 = S^2
    # We need: sum_j a[j] e^{-i 2pi k j / N} to have magnitude ~ S/sqrt(2n) for k != 0
    # This is hard to achieve directly, but we can try random phase
    N_fft = 1 << (2 * n - 1 - 1).bit_length()
    A_target = np.zeros(N_fft // 2 + 1, dtype=np.complex128)
    A_target[0] = np.sqrt(n)  # |A(0)| = S (with S ~ sqrt(n) for unit-mean)
    for k in range(1, N_fft // 2 + 1):
        A_target[k] = np.sqrt(1.0 / (2 * n)) * np.exp(1j * rng.uniform(0, 2 * np.pi))
    a_spec = np.real(np.fft.irfft(A_target * 2 - 2j * A_target[0], N_fft)[:n])
    # Make non-negative
    a_spec = np.clip(a_spec, 0.01, 1000.0)
    if a_spec.sum() > 0.01:
        inits.append(a_spec)
    
    return inits


def spectral_flat_init(n, rng, target_flatness=1.0):
    """
    Generate a sequence by designing its power spectrum to be as flat as possible,
    then projecting to non-negative.
    """
    N_fft = 1 << (2 * n - 1 - 1).bit_length()
    
    # Target: flat power spectrum (like white noise)
    # A[0] = S (DC component)
    # A[k] for k=1..N/2: random phase, magnitude ~ S/sqrt(2n)
    A = np.zeros(N_fft // 2 + 1, dtype=np.complex128)
    S_target = np.sqrt(n)  # if a ~ O(1)
    A[0] = S_target
    for k in range(1, N_fft // 2 + 1):
        mag = S_target / np.sqrt(2 * n) * (0.8 + 0.4 * rng.random())
        phase = rng.uniform(0, 2 * np.pi)
        A[k] = mag * np.exp(1j * phase)
    
    # Inverse FFT
    a = np.real(np.fft.irfft(A, N_fft)[:n])
    # Project to non-negative
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        a = np.ones(n) * 0.5
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function using Top-K LSE continuation, peak shaving,
    and stochastic refinement to minimize 2n*max(conv(a,a))/sum(a)^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = np.ones(1024) * 0.5
    
    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    # Determine lengths
    if parent is not None:
        n_main = parent.size
    else:
        n_main = 6962
    
    print(f"Parent: {'yes (n=' + str(parent.size) + ')' if parent is not None else 'no'}")
    print(f"Main length: {n_main}")
    
    # ============================================================
    # Phase 1: Quick init evaluation + Top-K LSE continuation
    # ============================================================
    phase1_end = t_start + 0.50 * total_budget
    print(f"\n=== Phase 1: Top-K LSE Continuation (t={phase1_end - t_start:.0f}s) ===")
    
    # Generate inits
    inits = generate_initializations(n_main, rng, parent)
    print(f"  Generated {len(inits)} initializations")
    
    # Quick evaluate all inits
    init_scores = []
    for i, a_init in enumerate(inits):
        sc = compute_score(a_init)
        init_scores.append((sc, i))
        print(f"    Init {i}: score = {sc:.8f}")
    init_scores.sort()
    
    # Sort inits by score, try the best ones first
    # Top-K LSE continuation with adaptive K and beta
    beta_schedule = [5, 10, 30, 100, 300, 1000, 3000]
    K_schedule = [5, 10, 20, 50, 100]
    
    n_inits_to_try = min(len(inits), 8)  # Try up to 8 inits
    
    for init_rank in range(n_inits_to_try):
        if time.monotonic() > phase1_end:
            break
        init_idx = init_scores[init_rank][1]
        a_cur = inits[init_idx].copy()
        
        remaining = phase1_end - time.monotonic()
        per_beta_time = remaining / (len(beta_schedule) * 1.5)
        
        for beta in beta_schedule:
            if time.monotonic() > phase1_end:
                break
            t_beta0 = time.monotonic()
            beta_time_limit = min(per_beta_time, 120.0)
            
            # Use Top-K LSE with K that grows with beta
            K = min(len(K_schedule), max(1, int(beta / 50) + 1))
            K = K_schedule[min(K - 1, len(K_schedule) - 1)]
            
            maxit = max(500, min(5000, int(beta_time_limit * 8)))
            sc, a_opt, _ = run_lbfgs_topk(a_cur, beta, K,
                                           maxiter=maxit, maxfun=maxit * 15,
                                           time_limit_s=beta_time_limit)
            a_cur = a_opt
            report(sc, a_opt, f"P1 init={init_idx} b={beta} K={K}")
        
        # After full continuation for this init, report
        sc_final = compute_score(a_cur)
        report(sc_final, a_cur, f"P1 init={init_idx} final")
    
    print(f"\n  After Phase 1: best = {best_score:.8f}")
    
    # ============================================================
    # Phase 2: Peak shaving (direct subgradient on top-M peaks)
    # ============================================================
    phase2_end = t_start + 0.70 * total_budget
    print(f"\n=== Phase 2: Peak Shaving (t={phase2_end - t_start:.0f}s) ===")
    
    if time.monotonic() < phase2_end:
        n_iters = max(1000, int((phase2_end - time.monotonic()) * 80))
        
        # Try different M values
        for M_val in [5, 10, 20, 50]:
            if time.monotonic() > phase2_end:
                break
            n_it_M = max(500, int((phase2_end - time.monotonic()) * 40))
            sc, a_opt = peak_shaving(best_a, max_iters=n_it_M, topM=M_val)
            report(sc, a_opt, f"P2 M={M_val}")
    
    # ============================================================
    # Phase 3: Full LSE high-beta refinement
    # ============================================================
    phase3_end = t_start + 0.82 * total_budget
    print(f"\n=== Phase 3: Full LSE High-Beta (t={phase3_end - t_start:.0f}s) ===")
    
    if time.monotonic() < phase3_end:
        for beta in [1000, 3000, 10000]:
            if time.monotonic() > phase3_end:
                break
            remaining = phase3_end - time.monotonic()
            tl = min(remaining * 0.8, 300.0)
            maxit = max(500, int(tl * 10))
            sc, a_opt, _ = run_lbfgs_full(best_a, beta, maxiter=maxit,
                                           maxfun=maxit * 15, time_limit_s=tl)
            report(sc, a_opt, f"P3 full b={beta}")
    
    # ============================================================
    # Phase 4: Stochastic restarts
    # ============================================================
    phase4_end = t_start + 0.92 * total_budget
    print(f"\n=== Phase 4: Stochastic Refinement (t={phase4_end - t_start:.0f}s) ===")
    
    trial = 0
    while time.monotonic() < phase4_end - 5 and trial < 20:
        trial += 1
        strat = trial % 5
        n_best = len(best_a)
        
        if strat == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strat == 1:
            # Anti-palindromic bump at random position
            a_pert = best_a.copy()
            center = rng.integers(n_best // 4, 3 * n_best // 4)
            width = max(1, n_best // 20)
            for j in range(n_best):
                d = (j - center) / width
                sign = 1.0 if j < center else -1.0
                a_pert[j] *= (1.0 + 0.1 * sign * np.exp(-d * d))
        elif strat == 2:
            # Frequency-domain smoothing
            M_fft = 1 << (2 * n_best - 1 - 1).bit_length()
            A = np.fft.rfft(a_pert if 'a_pert' in dir() else best_a, M_fft)
            freqs = np.arange(len(A))
            att = 1.0 - 0.15 * (freqs / len(A)) ** 2
            A *= att
            a_pert = np.real(np.fft.irfft(A, M_fft))[:n_best]
        elif strat == 3:
            # Local block swap
            a_pert = best_a.copy()
            blen = max(1, n_best // 30)
            i1 = rng.integers(0, n_best - 2 * blen)
            i2 = rng.integers(0, n_best - 2 * blen)
            if abs(i1 - i2) > blen:
                tmp = a_pert[i1:i1+blen].copy()
                a_pert[i1:i1+blen] = a_pert[i2:i2+blen] * 0.9
                a_pert[i2:i2+blen] = tmp * 0.9
        else:
            # Targeted: reduce elements at peak-contributing positions
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            a_pert = best_a.copy()
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            if j_end >= j_start:
                contrib = np.zeros(n_best)
                for j in range(j_start, j_end + 1):
                    contrib[j] = best_a[j] * best_a[k_star - j]
                top_count = max(5, n_best // 50)
                top_idx = np.argsort(contrib)[::-1][:top_count]
                total_red = 0.0
                for j in top_idx:
                    amt = a_pert[j] * 0.05
                    a_pert[j] -= amt
                    total_red += amt
                bottom_idx = np.argsort(contrib)[:n_best // 5]
                if total_red > 0:
                    a_pert[bottom_idx] += total_red / len(bottom_idx)
        
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 0.01:
            a_pert = best_a * 0.9
        
        # Quick refinement
        remaining = phase4_end - time.monotonic()
        tl = min(remaining * 0.4, 60.0)
        if tl > 5:
            sc, a_opt, _ = run_lbfgs_topk(a_pert, 200, 30,
                                           maxiter=1000, maxfun=20000,
                                           time_limit_s=tl)
            report(sc, a_opt, f"P4 #{trial} s={strat}")
    
    # ============================================================
    # Phase 5: Final peak shaving + polish
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 10
        n_iters = max(500, int(remaining * 60))
        sc, a_opt = peak_shaving(best_a, max_iters=n_iters, topM=10)
        report(sc, a_opt, "P5 final peak_shave")
    
    # One more top-K LSE pass
    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 5
        tl = min(remaining * 0.7, 120.0)
        if tl > 5:
            sc, a_opt, _ = run_lbfgs_topk(best_a, 5000, 100,
                                           maxiter=3000, maxfun=50000,
                                           time_limit_s=tl)
            report(sc, a_opt, "P5 final topK")
    
    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
