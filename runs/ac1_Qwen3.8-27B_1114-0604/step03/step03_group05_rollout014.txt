"""
Step function optimization: Simulated Annealing with LSE-guided local refinement.
Uses structured moves (block operations, frequency-domain mutations) to escape
local minima, combined with LSE optimization for local convergence.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, func, args, maxiter=1000, maxfun=25000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def peak_shave(a, iters=2000):
    """Subgradient descent on max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (best_sc - 1.0) / (gn * gn + 1e-20)
        step = min(step, 1.0 / S)
        step = max(step, 1e-7 / (gn * S))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        sc = 2.0 * n * M / (S * S)  # Use old M for consistency
        # Recompute
        c = fast_conv(a)
        sc = 2.0 * n * float(c.max()) / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_sc, best_a


def sa_block_swap(a, rng):
    """Swap two random blocks of the sequence."""
    n = len(a)
    a_new = a.copy()
    # Choose block sizes
    size1 = rng.integers(max(2, n // 20), max(3, n // 5))
    size2 = rng.integers(max(2, n // 20), max(3, n // 5))
    # Choose positions
    pos1 = rng.integers(0, max(1, n - size1))
    pos2 = rng.integers(0, max(1, n - size2))
    # Avoid overlap
    if pos1 + size1 > pos2 and pos2 + size2 > pos1:
        return a_new
    # Swap
    a_new[pos1:pos1+size1], a_new[pos2:pos2+size2] = a_new[pos2:pos2+size2].copy(), a_new[pos1:pos1+size1].copy()
    return np.clip(a_new, 0.0, 1000.0)


def sa_block_flip(a, rng):
    """Reverse a random block."""
    n = len(a)
    a_new = a.copy()
    size = rng.integers(max(4, n // 10), max(5, n // 3))
    pos = rng.integers(0, max(1, n - size))
    a_new[pos:pos+size] = a_new[pos:pos+size][::-1]
    return a_new


def sa_gaussian_noise(a, rng, scale):
    """Add Gaussian noise."""
    a_new = a + scale * a.mean() * rng.standard_normal(len(a))
    return np.clip(a_new, 0.0, 1000.0)


def sa_freq_dampen(a, rng, damp_level):
    """Dampen high frequencies in the sequence."""
    n = len(a)
    L = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, L)
    freqs = np.arange(L // 2 + 1)
    # Dampen high freqs
    mask = freqs > n * damp_level / 2
    A[mask] *= 0.5
    a_new = np.fft.irfft(A, L)[:n]
    return np.clip(a_new, 0.0, 1000.0)


def sa_freq_boost(a, rng, boost_level):
    """Boost certain frequency bands."""
    n = len(a)
    L = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, L)
    freqs = np.arange(L // 2 + 1)
    # Random band
    f0 = rng.integers(0, L // 4)
    f1 = f0 + rng.integers(1, L // 8)
    band = (freqs >= f0) & (freqs < f1)
    A[band] *= (1.0 + boost_level * rng.random())
    a_new = np.fft.irfft(A, L)[:n]
    return np.clip(a_new, 0.0, 1000.0)


def sa_taper(a, rng, taper_amount):
    """Apply a random taper (multiply by a smooth window)."""
    n = len(a)
    x = np.arange(n) / n
    # Random center and width
    center = rng.random()
    width = 0.1 + 0.4 * rng.random()
    window = 1.0 + taper_amount * np.cos(np.pi * (x - center) / width)
    window = np.clip(window, 0.1, 2.0)
    a_new = a * window
    return np.clip(a_new, 0.0, 1000.0)

"""
Step function optimization: Simulated Annealing with LSE-guided local refinement.
Combines structured exploratory moves (block operations, frequency-domain mutations,
tapering) with LSE optimization for local convergence. Uses temperature-controlled
acceptance to escape local minima, fundamentally different from pure gradient-based
approaches.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(a, beta):
    """Objective wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, func, args, maxiter=1000, maxfun=25000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def peak_shave(a, iters=3000):
    """Subgradient descent on max-conv objective with Polyak step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (best_sc - 1.0) / (gn * gn + 1e-20)
        step = min(step, 1.0 / S)
        step = max(step, 1e-7 / (gn * S))
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_sc, best_a


def sa_gaussian_noise(a, rng, scale):
    """Add proportional Gaussian noise."""
    n = len(a)
    a_new = a + scale * a.mean() * rng.standard_normal(n)
    return np.clip(a_new, 0.0, 1000.0)


def sa_block_swap(a, rng):
    """Swap two random blocks of the sequence."""
    n = len(a)
    a_new = a.copy()
    size1 = int(rng.integers(max(2, n // 25), max(3, n // 4)))
    size2 = int(rng.integers(max(2, n // 25), max(3, n // 4)))
    pos1 = int(rng.integers(0, max(1, n - size1)))
    pos2 = int(rng.integers(0, max(1, n - size2)))
    if pos1 + size1 > pos2 and pos2 + size2 > pos1:
        return a_new
    block1 = a_new[pos1:pos1 + size1].copy()
    block2 = a_new[pos2:pos2 + size2].copy()
    a_new[pos2:pos2 + size2] = block1[:size2]
    a_new[pos1:pos1 + size1] = block2[:size1]
    return np.clip(a_new, 0.0, 1000.0)


def sa_block_flip(a, rng):
    """Reverse a random block."""
    n = len(a)
    a_new = a.copy()
    size = int(rng.integers(max(4, n // 15), max(5, n // 3)))
    pos = int(rng.integers(0, max(1, n - size)))
    a_new[pos:pos + size] = a_new[pos:pos + size][::-1]
    return a_new


def sa_freq_dampen(a, rng, damp_fraction):
    """Dampen high frequencies in the sequence."""
    n = len(a)
    L = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, L)
    freqs = np.arange(L // 2 + 1)
    cutoff = int(n * damp_fraction)
    mask = freqs > cutoff
    A[mask] *= 0.3
    a_new = np.fft.irfft(A, L)[:n]
    return np.clip(a_new, 0.0, 1000.0)


def sa_freq_boost(a, rng):
    """Boost a random frequency band."""
    n = len(a)
    L = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, L)
    freqs = np.arange(L // 2 + 1)
    f0 = int(rng.integers(0, L // 4))
    f1 = f0 + int(rng.integers(2, max(3, L // 10)))
    band = (freqs >= f0) & (freqs < f1)
    A[band] *= 1.0 + 0.5 * rng.random()
    a_new = np.fft.irfft(A, L)[:n]
    return np.clip(a_new, 0.0, 1000.0)


def sa_taper(a, rng, taper_amount):
    """Apply a random smooth window modulation."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    center = rng.random()
    width = 0.15 + 0.5 * rng.random()
    window = 1.0 + taper_amount * np.cos(np.pi * (x - center) / max(width, 0.05))
    window = np.clip(window, 0.05, 2.0)
    a_new = a * window
    return np.clip(a_new, 0.0, 1000.0)


def sa_random_window_mix(a, rng, rng2_unused=None):
    """Mix with a random smooth window (Gaussian or raised cosine)."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    choice = rng.integers(0, 3)
    if choice == 0:
        # Gaussian window
        mu = 0.2 + 0.6 * rng.random()
        sigma = 0.05 + 0.3 * rng.random()
        window = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    elif choice == 1:
        # Raised cosine
        center = 0.2 + 0.6 * rng.random()
        width = 0.2 + 0.6 * rng.random()
        t = (x - center) / width
        window = np.where(np.abs(t) < 1.0, 0.5 * (1 + np.cos(np.pi * t)), 0.0)
    else:
        # Inverted Gaussian (dip)
        mu = 0.2 + 0.6 * rng.random()
        sigma = 0.05 + 0.3 * rng.random()
        window = 1.5 - 1.0 * np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    window = np.clip(window, 0.0, None)
    if window.max() > 0:
        window = window / window.max()
    else:
        window = np.ones(n)
    a_new = a * window
    return np.clip(a_new, 0.0, 1000.0)


def make_move(a, rng, move_type, temp_frac):
    """Generate a perturbation based on move type and temperature fraction."""
    n = len(a)
    if move_type == 0:
        scale = 0.02 + 0.08 * temp_frac
        return sa_gaussian_noise(a, rng, scale)
    elif move_type == 1:
        return sa_block_swap(a, rng)
    elif move_type == 2:
        return sa_block_flip(a, rng)
    elif move_type == 3:
        damp = 0.3 + 0.5 * temp_frac
        return sa_freq_dampen(a, rng, damp)
    elif move_type == 4:
        return sa_freq_boost(a, rng)
    elif move_type == 5:
        taper = 0.2 + 0.5 * temp_frac
        return sa_taper(a, rng, taper)
    elif move_type == 6:
        return sa_random_window_mix(a, rng)
    else:
        # Heavy mutation: large-scale structural change
        scale = 0.1 + 0.3 * temp_frac
        a_noisy = sa_gaussian_noise(a, rng, scale)
        a_noisy = sa_taper(a_noisy, rng, 0.3)
        return a_noisy


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses simulated annealing with structured moves and LSE local refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 25.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    # Initialize
    if parent is not None:
        sc = compute_score(parent)
        if sc < best_score:
            best_score = sc
            best_a = parent.copy()
        print(f"Parent score: {sc:.8f}, length={len(parent)}")
    else:
        for n_try in [2048, 4096]:
            a_rand = 0.5 + 0.5 * rng.random(n_try)
            sc = compute_score(a_rand)
            if sc < best_score:
                best_score = sc
                best_a = a_rand.copy()
        print(f"Random init score: {best_score:.8f}, length={len(best_a)}")

    n = len(best_a)
    print(f"Initial n = {n}, score = {best_score:.8f}")

    # ===================================================================
    # Phase 1: LSE local optimization (10% time)
    # ===================================================================
    print("\n=== Phase 1: LSE Local Optimization ===")
    phase1_end = t_start + 0.10 * total_budget
    a_cur = best_a.copy()
    for beta in [50, 200, 500, 1500]:
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        a_cur = run_lbfgs(a_cur, lse_obj, (beta,), maxiter=600, maxfun=15000)
        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  LSE beta={beta}: NEW BEST {sc:.8f} ({time.monotonic()-t0:.1f}s)")
        else:
            print(f"  LSE beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ===================================================================
    # Phase 2: Simulated Annealing with LSE refinement (65% time)
    # ===================================================================
    print("\n=== Phase 2: Simulated Annealing ===")
    phase2_end = t_start + 0.75 * total_budget

    T_start = 0.3
    T_end = 0.0005
    n_sa_iters = 300
    lse_beta_sa = 500
    lse_maxiter_sa = 250

    T = T_start
    cooling = (T_end / T_start) ** (1.0 / n_sa_iters)

    current_a = best_a.copy()
    current_sc = best_score

    # Maintain a small population of diverse solutions
    population = [best_a.copy() for _ in range(3)]
    pop_scores = [best_score for _ in range(3)]

    sa_round = 0
    consecutive_no_improve = 0

    while time.monotonic() < phase2_end and sa_round < n_sa_iters:
        T = max(T * cooling, T_end)
        sa_round += 1
        temp_frac = T / T_start

        # Choose which population member to perturb
        if sa_round % 3 == 0 and len(population) > 1:
            idx = int(rng.integers(0, len(population)))
            parent_a = population[idx].copy()
        else:
            parent_a = current_a.copy()

        # Generate move
        move_type = int(rng.integers(0, 7))
        candidate_a = make_move(parent_a, rng, move_type, temp_frac)

        # Ensure validity
        if candidate_a.sum() < 0.01:
            candidate_a = parent_a * 0.5 + 0.5 * rng.random(len(parent_a))
        candidate_a = np.clip(candidate_a, 0.0, 1000.0)

        # LSE refinement
        t_move = time.monotonic()
        refined_a = run_lbfgs(candidate_a, lse_obj, (lse_beta_sa,),
                              maxiter=lse_maxiter_sa, maxfun=lse_maxiter_sa * 20)
        refined_sc = compute_score(refined_a)
        move_time = time.monotonic() - t_move

        # Accept/reject based on temperature
        delta = refined_sc - current_sc
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 1e-8:
                ratio = np.exp(-delta / T)
                if rng.random() < ratio:
                    accept = True

        if accept:
            current_a = refined_a.copy()
            current_sc = refined_sc
            if refined_sc < best_score:
                best_score = refined_sc
                best_a = refined_a.copy()
                consecutive_no_improve = 0
                print(f"  SA r={sa_round} T={T:.5f} move={move_type}: "
                      f"NEW BEST {refined_sc:.8f} ({move_time:.1f}s)")
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1
            # Still check if it's the best found
            if refined_sc < best_score:
                best_score = refined_sc
                best_a = refined_a.copy()

        # Update population: keep best, replace worst
        pop_scores.append(refined_sc)
        population.append(refined_a.copy())
        worst_idx = int(np.argmax(pop_scores))
        population.pop(worst_idx)
        pop_scores.pop(worst_idx)

        # Adaptive: if stuck, try a heavier move
        if consecutive_no_improve >= 15:
            heavy_a = make_move(best_a, rng, 7, 1.0)
            if heavy_a.sum() > 0.01:
                heavy_refined = run_lbfgs(heavy_a, lse_obj, (lse_beta_sa,),
                                          maxiter=lse_maxiter_sa,
                                          maxfun=lse_maxiter_sa * 20)
                heavy_sc = compute_score(heavy_refined)
                if heavy_sc < best_score:
                    best_score = heavy_sc
                    best_a = heavy_refined.copy()
                    current_a = heavy_refined.copy()
                    current_sc = heavy_sc
                    consecutive_no_improve = 0
                    print(f"  SA r={sa_round} HEAVY: NEW BEST {heavy_sc:.8f}")

        if sa_round % 20 == 0:
            print(f"  SA r={sa_round}, T={T:.5f}, cur={current_sc:.6f}, "
                  f"best={best_score:.6f}, no_improve={consecutive_no_improve}, "
                  f"t={time.monotonic()-t_start:.0f}s")

    # ===================================================================
    # Phase 3: Final refinement (25% time)
    # ===================================================================
    print("\n=== Phase 3: Final Refinement ===")
    phase3_end = deadline

    # High-beta LSE refinement
    if time.monotonic() < phase3_end:
        a_cur = best_a.copy()
        for beta in [1000, 3000, 8000, 15000]:
            if time.monotonic() > phase3_end:
                break
            t0 = time.monotonic()
            a_cur = run_lbfgs(a_cur, lse_obj, (beta,), maxiter=2000, maxfun=50000)
            sc = compute_score(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                print(f"  Final LSE beta={beta}: NEW BEST {sc:.8f} "
                      f"({time.monotonic()-t0:.1f}s)")
            else:
                print(f"  Final LSE beta={beta}: {sc:.8f} "
                      f"({time.monotonic()-t0:.1f}s)")

    # Peak shaving
    if time.monotonic() < phase3_end:
        t0 = time.monotonic()
        remaining = phase3_end - time.monotonic()
        iters = int(min(5000, max(500, remaining * 80)))
        sc, a_shaved = peak_shave(best_a, iters=iters)
        if sc < best_score:
            best_score = sc
            best_a = a_shaved.copy()
            print(f"  Peak shave: NEW BEST {sc:.8f} ({time.monotonic()-t0:.1f}s)")
        else:
            print(f"  Peak shave: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # One more quick SA burst with the best solution
    if time.monotonic() < deadline - 5:
        print("\n  Final SA burst...")
        T_burst = 0.05
        for burst_i in range(20):
            if time.monotonic() > deadline - 5:
                break
            move_type = int(rng.integers(0, 7))
            burst_a = make_move(best_a, rng, move_type, 0.3)
            if burst_a.sum() > 0.01:
                burst_refined = run_lbfgs(burst_a, lse_obj, (3000,),
                                          maxiter=300, maxfun=6000)
                burst_sc = compute_score(burst_refined)
                if burst_sc < best_score:
                    best_score = burst_sc
                    best_a = burst_refined.copy()
                    print(f"    Burst {burst_i}: NEW BEST {burst_sc:.8f}")

    # ===================================================================
    # Final output
    # ===================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
