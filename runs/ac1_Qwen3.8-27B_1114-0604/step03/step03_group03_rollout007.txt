def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for the objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled - scaled.max())
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    sc = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    # Gradient via FFT
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return sc, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5

    def obj_grad(x):
        return lse_score_grad(x, beta)

    res = minimize(
        obj_grad, a0, jac=True,
        method='L-BFGS-B', bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)

    return exact_sc, a_out

    top_frac = max(0.05, strength)
    a_pert = a.copy()
    # Reduce high-contributing elements
    threshold = np.percentile(contrib, 70)
    high_mask = contrib > threshold
    reduce_amt = a_pert[high_mask] * top_frac
    a_pert[high_mask] -= reduce_amt
    a_pert = np.clip(a_pert, 0.0, 1000.0)

    # Redistribute to low-contributing elements
    low_mask = ~high_mask
    if a_pert.sum() < a.sum() and low_mask.any():
        saved = a.sum() - a_pert.sum()
        weights = np.ones(low_mask.sum()) * (a_pert[low_mask] + 0.01)
        weights /= weights.sum()
        a_pert[low_mask] += saved * weights
    return np.clip(a_pert, 0.0, 1000.0)


def frequency_perturb(a, rng, n_freqs=3):
    """Perturb in frequency domain to create structured variations."""
    n = len(a)
    x = a.copy()
    X = np.fft.rfft(x)
    for _ in range(n_freqs):
        k = int(rng.integers(1, len(X)))
        amp = 0.01 + 0.05 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        X[k] += amp * a.mean() * np.exp(1j * phase)
        if k < len(X) - 1:
            X[-k] = np.conj(X[k])
    x = np.fft.irfft(X, n)
    return np.clip(x, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: fast CD -> LSE refinement -> diversification -> final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 10.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    if parent is not None:
        p_score = compute_score(parent)
        print(f"Parent score: {p_score:.8f}, length: {len(parent)}")
        best_score = p_score
        best_a = parent.copy()
    else:
        print("No parent. Initializing with structured sequence.")
        n_init = 4096
        x = np.arange(n_init) / n_init
        best_a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
        best_a = np.clip(best_a, 0.01, None)
        best_score = compute_score(best_a)
        print(f"Initial score: {best_score:.8f}")

    n_work = len(best_a)
    print(f"Working length: {n_work}")

    # ============================================================
    # Phase 1: Fast coordinate descent (45% of time)
    # ============================================================
    phase1_deadline = t_start + 0.45 * total_budget
    print(f"\n=== Phase 1: Coordinate Descent ({(phase1_deadline - t_start):.0f}s) ===")

    a_cur = best_a.copy()
    no_improve = 0
    sweep = 0

    while time.monotonic() < phase1_deadline:
        sweep += 1
        # Alternate strategies
        if sweep % 5 == 0:
            sc, a_cur = coord_descent(a_cur, hot_order=True)
        elif sweep % 5 == 1:
            sc, a_cur = coord_descent(a_cur, rng=rng)
        else:
            sc, a_cur = coord_descent(a_cur)

        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            no_improve = 0
            if sweep <= 10 or sweep % 25 == 0:
                print(f"  P1 sweep {sweep}: {sc:.8f} (NEW BEST)")
        else:
            no_improve += 1

        if sweep % 25 == 0:
            print(f"  P1 sweep {sweep}: {sc:.8f} (no_improve={no_improve})")

        # Perturb if stuck
        if no_improve >= 30:
            strat = rng.integers(3)
            if strat == 0:
                a_cur = geometric_perturb(a_cur, rng, strength=0.02 + 0.03 * rng.random())
            elif strat == 1:
                a_cur = a_cur * (1.0 + 0.02 * rng.standard_normal(len(a_cur)))
                a_cur = np.clip(a_cur, 0.0, 1000.0)
            else:
                a_cur = frequency_perturb(a_cur, rng, n_freqs=2)
            no_improve = 0
            sc, a_cur = coord_descent(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                print(f"  P1 post-perturb sweep {sweep}: {sc:.8f} (NEW BEST)")

    print(f"Phase 1 done: {sweep} sweeps, best={best_score:.8f}")

    # ============================================================
    # Phase 2: LSE gradient refinement (20% of time)
    # ============================================================
    phase2_deadline = t_start + 0.65 * total_budget
    print(f"\n=== Phase 2: LSE Refinement ===")

    for beta in [50, 150, 500, 1500, 4000, 8000]:
        if time.monotonic() > phase2_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=6000, maxfun=120000)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  LSE beta={beta}: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            print(f"  LSE beta={beta}: {sc:.8f} ({elapsed:.1f}s)")

    print(f"Phase 2 done: best={best_score:.8f}")

    # ============================================================
    # Phase 3: Diversification (20% of time)
    # ============================================================
    phase3_deadline = t_start + 0.85 * total_budget
    print(f"\n=== Phase 3: Diversification ===")

    trial = 0
    while time.monotonic() < phase3_deadline - 5.0:
        trial += 1
        n_cur = len(best_a)
        strat = trial % 6

        if strat == 0:
            # Geometric perturbation
            a_pert = geometric_perturb(best_a, rng, strength=0.03 + 0.04 * rng.random())
        elif strat == 1:
            # Frequency domain perturbation
            a_pert = frequency_perturb(best_a, rng, n_freqs=3)
        elif strat == 2:
            # Block swap
            seg = max(20, n_cur // 25)
            s1 = int(rng.integers(0, n_cur - seg))
            s2 = int(rng.integers(0, n_cur - seg))
            a_pert = best_a.copy()
            tmp = a_pert[s1:s1 + seg].copy()
            a_pert[s1:s1 + seg] = a_pert[s2:s2 + seg]
            a_pert[s2:s2 + seg] = tmp
        elif strat == 3:
            # Mirror + perturbation
            a_pert = best_a.copy()
            half = n_cur // 2
            a_pert[half:] = a_pert[:half][::-1]
            a_pert *= (1.0 + 0.02 * rng.standard_normal(n_cur))
        elif strat == 4:
            # Sinusoidal modulation
            freq = 1.0 + 8.0 * rng.random()
            amp = 0.02 + 0.04 * rng.random()
            x = np.arange(n_cur) / n_cur
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + 2.0 * np.pi * rng.random()))
        else:
            # Coarse random restart blended with best
            a_rand = np.abs(rng.standard_normal(n_cur))
            a_rand /= a_rand.sum() * max(1, best_a.sum() / n_cur)
            mix = 0.1 + 0.2 * rng.random()
            a_pert = (1 - mix) * best_a + mix * a_rand

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick CD from perturbed point
        n_cd = 6 if n_cur > 3000 else 12
        for sw in range(n_cd):
            if time.monotonic() > phase3_deadline - 5.0:
                break
            sc_p, a_pert = coord_descent(a_pert, hot_order=(sw == 0), rng=rng if sw > 0 else None)

        sc_p = compute_score(a_pert)
        if sc_p < best_score:
            best_score = sc_p
            best_a = a_pert.copy()
            print(f"  P3 trial {trial} (s={strat}): {sc_p:.8f} (NEW BEST)")
        elif trial % 5 == 0:
            print(f"  P3 trial {trial} (s={strat}): {sc_p:.8f}")

    print(f"Phase 3 done: {trial} trials, best={best_score:.8f}")

    # ============================================================
    # Phase 4: Final polish (15% of time)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")

    a_cur = best_a.copy()
    polish_sweep = 0
    polish_deadline = deadline - 20.0
    while time.monotonic() < polish_deadline:
        polish_sweep += 1
        if polish_sweep % 4 == 0:
            sc, a_cur = coord_descent(a_cur, hot_order=True)
        else:
            sc, a_cur = coord_descent(a_cur, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            if polish_sweep % 10 == 0:
                print(f"  P4 sweep {polish_sweep}: {sc:.8f} (NEW BEST)")

    print(f"Phase 4 done: {polish_sweep} sweeps, best={best_score:.8f}")

    # Final LSE if time permits
    if time.monotonic() < deadline - 10.0:
        sc, a_opt = run_lbfgs(best_a, 5000, maxiter=3000, maxfun=50000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final LSE: {sc:.8f} (NEW BEST)")
        # One more CD
        a_cur = best_a.copy()
        sc, a_cur = coord_descent(a_cur, hot_order=True)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  Final CD: {sc:.8f} (NEW BEST)")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
