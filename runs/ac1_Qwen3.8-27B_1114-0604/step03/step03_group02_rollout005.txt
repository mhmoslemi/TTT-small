                       options={'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
        a_out = np.clip(res.x, 0.0, 1000.0)
    except:
        a_out = a0
    return compute_score(a_out), a_out

def mul_update(a, n_iters, alpha, temp_frac, deadline):
    """Multiplicative equalization with softmax-weighted peaks."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (3 * n - 2).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        if M < 1e-15:
            break
        temp = M * temp_frac
        if temp < 1e-15:
            temp = c.std() * 0.1
        if temp < 1e-15:
            temp = 1e-10
        weights = np.exp(c / temp - c.max() / temp)
        wsum = weights.sum()
        if wsum < 1e-15:
            break
        weights /= wsum
        
        W = np.fft.rfft(weights, M_fft)
        AR = np.fft.rfft(a[::-1], M_fft)
        cross = np.fft.irfft(W * AR, M_fft)
        h = 2.0 * cross[n-1:2*n-1]
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()
        
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = np.clip(a * factor, 0.0, 1000.0)
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a *= S / S_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

def peak_shave(a, rng, n_trials, deadline):
    """Targeted peak reduction: reduce elements contributing to peak, add to others."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        # Find contributing elements
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            continue
        contrib = a[j_start:j_end+1].copy()
        
        # Pick top contributors to reduce
        top_k = max(5, n // 100)
        top_idx_local = np.argsort(contrib)[::-1][:top_k]
        
        # Pick bottom contributors (or random) to increase
        bottom_idx_local = np.argsort(contrib)[:top_k]
        
        strength = 0.02 + 0.05 * rng.random()
        a_trial = a.copy()
        total_red = 0.0
        for idx in top_idx_local:
            j = j_start + idx
            red = a_trial[j] * strength
            a_trial[j] -= red
            a_trial[j] = max(a_trial[j], 0.0)
            total_red += red
        
        if total_red > 0:
            per = total_red / len(bottom_idx_local)
            for idx in bottom_idx_local:
                j = j_start + idx
                a_trial[j] += per
        
        a_trial = np.clip(a_trial, 0.0, 1000.0)
        sc = compute_score(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
    
    return best_sc, best_a
