"""
Multi-phase optimizer for minimizing 2n*max(self-conv)/(sum)^2.
Combines Adam (LSE-smoothed), LP-based updates, L-BFGS, and coordinate descent.
Uses diverse initializations and adaptive parameters.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time
import math

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def fast_cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    n = max(len(a), len(b))
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(rev_a) + len(p) - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:n - 1 + n]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def subgrad_score_a(a):
    """Exact score and subgradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad

def multiple_subgrad_score_a(a, top_k=5):
    """Score and averaged subgradient over top-k convolution lags."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c_max = float(c.max())
    score = 2.0 * n * c_max / (S * S)
    sorted_idx = np.argsort(c)[::-1][:top_k]
    # Weights: softmax of the top-k values
    top_vals = c[sorted_idx]
    weights = np.exp((top_vals - top_vals.max()) * 2.0)
    weights = weights / weights.sum()
    sg = np.zeros(n)
    for t, k in enumerate(sorted_idx):
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(k_int, n - 1)
        if je >= js:
            idx = k_int - np.arange(js, je + 1)
            sg[js:je + 1] += 2.0 * weights[t] * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
    return score, grad

def run_lbfgs_a(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in a-space with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def adam_adaptive(a_init, lr_init=0.005, n_steps=50000, beta_start=30, beta_end=3000,
                  time_limit_s=None, t_start=None, normalize=True):
    """Adam with adaptive LSE beta (exponential schedule)."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        # Adaptive beta: exponential from beta_start to beta_end
        if t < n_steps:
            frac = t / n_steps
            beta_t = beta_start * (beta_end / beta_start) ** frac
        else:
            beta_t = beta_end
        
        # Learning rate with warmup and decay
        warmup_steps = 100
        if t <= warmup_steps:
            lr_t = lr_init * t / warmup_steps
        else:
            lr_t = lr_init * 0.5 * (1 + math.cos(math.pi * (t - warmup_steps) / (n_steps - warmup_steps)))
        
        score, grad = lse_score_grad_a(a, beta_t)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        
        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a / s
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def lp_based_update(a, K=50, alpha=0.3):
    """
    LP-based update: solve max sum(g) s.t. g>=0, top-K cross-conv constraints.
    Returns updated sequence.
    """
    n = len(a)
    c = fast_conv(a)
    M = float(c.max())
    S = float(a.sum())
    
    # Get top-K lags
    top_k_idx = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix A: A[m, i] = a[k_m - i] if valid
    A = np.zeros((K, n))
    for m, k in enumerate(top_k_idx):
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(k_int, n - 1)
        if je >= js:
            idx = k_int - np.arange(js, je + 1)
            A[m, js:je + 1] = a[idx]
    
    # LP: max 1^T g s.t. A g <= M, g >= 0
    c_obj = -np.ones(n)  # minimize -sum(g)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(K, M), bounds=(0, None), method='highs')
    
    if res.status != 0:
        return a.copy()
    
    g = np.clip(res.x, 0, 1000.0)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return a.copy()
    
    # Rescale g to match sum of a
    g = g * (S / g_sum)
    
    # Convex combination
    a_new = (1.0 - alpha) * a + alpha * g
    a_new = np.clip(a_new, 0.0, 1000.0)
    
    return a_new

def iterative_subgrad_flattening(a, n_iters=100, top_k_peaks=10, reduce_frac=0.05,
                                 time_limit_s=None, t_start=None):
    """
    Iteratively reduce peak contributors and redistribute.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        S = float(a.sum())
        score = 2.0 * n * float(c.max()) / (S * S)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find top-k peaks
        sorted_idx = np.argsort(c)[::-1][:top_k_peaks]
        
        # Compute influence of each a_i on the top peaks
        influence = np.zeros(n)
        for k in sorted_idx:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(k_int, n - 1)
            if je >= js:
                idx = k_int - np.arange(js, je + 1)
                influence[js:je + 1] += a[idx]
        
        # Reduce high-influence elements
        total_mass = 0.0
        n_reduce = max(1, int(n * reduce_frac))
        top_influence_idx = np.argsort(influence)[::-1][:n_reduce]
        
        for j in top_influence_idx:
            reduction = a[j] * 0.3  # reduce by 30%
            a[j] -= reduction
            total_mass += reduction
        
        # Redistribute to low-influence elements
        low_influence_idx = np.argsort(influence)[:n_reduce]
        if total_mass > 0 and len(low_influence_idx) > 0:
            a[low_influence_idx] += total_mass / len(low_influence_idx)
        
        a = np.clip(a, 0.0, 1000.0)
    
    return best_score, best_a

def coordinate_descent(a, top_k=50, n_sweeps=10, time_limit_s=None, t_start=None):
    """Selective coordinate descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        score, grad = subgrad_score_a(a)
        if score < best_score:
            best_score = score
        
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            
            a_old = a[j]
            S = float(a.sum())
            c = fast_conv(a)
            max_b = float(c.max())
            
            if grad[j] > 1e-10:
                ratios = np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.0])
            elif grad[j] < -1e-10:
                ratios = np.array([1.0, 1.2, 1.5, 2.0, 3.0])
            else:
                continue
            
            candidates = a_old * ratios
            candidates = np.clip(candidates, 0.0, 1000.0)
            
            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = x
                sc = 2.0 * n * float(fast_conv(a).max()) / (float(a.sum()) ** 2)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    
    return best_score, a

def diverse_initializations(n, seed=42):
    """Generate diverse initial sequences of length n."""
    rng = np.random.default_rng(seed)
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Cosine taper
    a1 = np.cos(np.pi * x / (2 * n)) ** 2 + 0.1
    inits.append(a1 / a1.sum())
    
    # Sine
    a2 = np.sin(np.pi * (x + 1) / (n + 1)) ** 2 + 0.05
    inits.append(a2 / a2.sum())
    
    # Gaussian
    a3 = np.exp(-0.5 * ((x - n/2) / (n/4)) ** 2) + 0.1
    inits.append(a3 / a3.sum())
    
    # Linear taper (triangle)
    a4 = np.minimum(x, n - 1 - x) + 1.0
    inits.append(a4 / a4.sum())
    
    # Flat
    inits.append(np.ones(n) / n)
    
    # Power spectrum shaping: 1 - cos (raised cosine)
    a5 = 0.5 * (1 - np.cos(2 * np.pi * x / n)) + 0.1
    inits.append(a5 / a5.sum())
    
    # Two Gaussians
    a6 = np.exp(-0.5 * ((x - n/3) / (n/8)) ** 2) + np.exp(-0.5 * ((x - 2*n/3) / (n/8)) ** 2) + 0.05
    inits.append(a6 / a6.sum())
    
    # Random smooth
    noise = rng.standard_normal(n)
    # Smooth with a kernel
    kernel_size = max(3, n // 100)
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size+1) / (kernel_size//3)) ** 2)
    kernel /= kernel.sum()
    a7 = np.convolve(noise, kernel, mode='same')
    a7 = np.abs(a7) + 0.1
    inits.append(a7 / a7.sum())
    
    # Multi-frequency
    freqs = rng.choice([2, 3, 5, 7, 11, 13], size=3, replace=False)
    a8 = np.ones(n)
    for f in freqs:
        a8 += 0.3 * np.cos(2 * np.pi * f * x / n + rng.uniform(0, 2*np.pi))
    a8 = np.abs(a8) + 0.1
    inits.append(a8 / a8.sum())
    
    return inits

"""
Multi-phase optimizer for minimizing 2n*max(self-conv)/(sum)^2.
Combines adaptive-beta Adam (LSE-smoothed), LP-based updates,
iterative subgradient flattening, L-BFGS polishing, and coordinate descent.
Uses diverse perturbations and adaptive parameters to escape local minima.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time
import math


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        return 2.0 * n * c_max / (S * S), np.zeros(n)
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = len(rev_a) + len(p) - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:n - 1 + n]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_score_a(a):
    """Exact score and subgradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def run_lbfgs_a(a0, beta, maxiter=4000, maxfun=80000):
    """L-BFGS-B in a-space with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def adam_adaptive(a_init, lr_init=0.005, n_steps=50000,
                  beta_start=30, beta_end=3000,
                  time_limit_s=None, t_start=None, normalize=True):
    """Adam with adaptive LSE beta (exponential schedule) and cosine LR decay."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    warmup = max(50, n_steps // 100)

    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break

        # Adaptive beta
        if t < n_steps:
            frac = t / max(1, n_steps)
            beta_t = beta_start * (beta_end / beta_start) ** frac
        else:
            beta_t = beta_end

        # Learning rate: linear warmup then cosine decay
        if t <= warmup:
            lr_t = lr_init * t / warmup
        else:
            progress = (t - warmup) / max(1, n_steps - warmup)
            lr_t = lr_init * 0.5 * (1.0 + math.cos(math.pi * progress))

        score, grad = lse_score_grad_a(a, beta_t)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)

        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a / s

        if t % 5000 == 0:
            sc_check = compute_score(a)
            if sc_check < best_score:
                best_score = sc_check
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def lp_based_update(a, K=40, alpha=0.3):
    """
    LP-based update: solve max sum(g) s.t. g>=0, top-K xconv constraints with a.
    Returns updated sequence as convex combination.
    """
    n = len(a)
    c = fast_conv(a)
    M = float(c.max())
    S = float(a.sum())

    # Get top-K lags
    top_k_idx = np.argsort(c)[::-1][:K]

    # Build constraint matrix
    A_mat = np.zeros((K, n))
    for m, k in enumerate(top_k_idx):
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(k_int, n - 1)
        if je >= js:
            idx = k_int - np.arange(js, je + 1)
            A_mat[m, js:je + 1] = a[idx]

    # LP: max 1^T g s.t. A g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'time_limit': 30})
    except Exception:
        return a.copy()

    if res.status != 0 or res.x is None:
        return a.copy()

    g = np.clip(res.x, 0, 1000.0)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return a.copy()

    # Rescale g to match sum of a
    g = g * (S / g_sum)

    # Convex combination
    a_new = (1.0 - alpha) * a + alpha * g
    a_new = np.clip(a_new, 0.0, 1000.0)

    return a_new


def iterative_flattening(a, n_iters=200, top_k_peaks=12, reduce_frac=0.04,
                         time_limit_s=None, t_start=None):
    """
    Iteratively reduce peak contributors and redistribute to low-influence positions.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        score = 2.0 * n * float(c.max()) / (S * S)

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Find top-k peaks
        sorted_idx = np.argsort(c)[::-1][:top_k_peaks]

        # Compute influence of each a_i on the top peaks
        influence = np.zeros(n)
        for k in sorted_idx:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(k_int, n - 1)
            if je >= js:
                idx = k_int - np.arange(js, je + 1)
                influence[js:je + 1] += a[idx]

        # Reduce high-influence elements
        total_mass = 0.0
        n_reduce = max(1, int(n * reduce_frac))
        top_influence_idx = np.argsort(influence)[::-1][:n_reduce]

        for j in top_influence_idx:
            reduction = a[j] * 0.35
            a[j] -= reduction
            total_mass += reduction

        # Redistribute to low-influence elements
        low_mask = np.zeros(n, dtype=bool)
        low_idx = np.argsort(influence)[:n_reduce]
        if total_mass > 0 and len(low_idx) > 0:
            a[low_idx] += total_mass / len(low_idx)

        a = np.clip(a, 0.0, 1000.0)

    return best_score, best_a


def coordinate_descent(a, top_k=50, n_sweeps=10, time_limit_s=None, t_start=None):
    """Selective coordinate descent on highest-gradient coordinates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        score, grad = subgrad_score_a(a)
        if score < best_score:
            best_score = score

        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break

            a_old = a[j]
            cur_score = 2.0 * n * float(fast_conv(a).max()) / (float(a.sum()) ** 2)

            if grad[j] > 1e-10:
                ratios = np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.0])
            elif grad[j] < -1e-10:
                ratios = np.array([1.0, 1.2, 1.5, 2.0])
            else:
                continue

            candidates = a_old * ratios
            candidates = np.clip(candidates, 0.0, 1000.0)

            best_x = a_old
            best_sc = cur_score
            for x in candidates:
                a[j] = x
                S = float(a.sum())
                if S < 1e-12:
                    a[j] = a_old
                    continue
                sc = 2.0 * n * float(fast_conv(a).max()) / (S * S)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc

    return best_score, a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)

    7. Final intensive polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 25.0

    # Get parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
    else:
        n_default = 4096
        parent = np.ones(n_default) / n_default
        parent_score = compute_score(parent)
        print(f"No parent, using uniform n={n_default}, score={parent_score:.8f}")

    best_score = parent_score
    best_a = parent.copy()
    print(f"Initial best: {best_score:.8f}")

    def do_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ==================================================================
    # Phase 1: Adaptive Adam on parent (25% time)
    # ==================================================================
    p1 = effective_budget * 0.25
    print(f"\n{'='*50}")
    print(f"Phase 1: Adaptive Adam on parent (budget={p1:.0f}s)")
    print(f"{'='*50}")

    a_norm = parent / parent.sum()

    adam_configs = [
        (20, 500, 0.01),
        (50, 2000, 0.005),
        (200, 5000, 0.002),
    ]

    for beta_s, beta_e, lr in adam_configs:
        remaining = p1 - (time.monotonic() - t_start)
        if remaining < 10:
            break
        time_alloc = remaining * 0.33
        sc, a_opt = adam_adaptive(a_norm, lr_init=lr, n_steps=200000,
                                   beta_start=beta_s, beta_end=beta_e,
                                   time_limit_s=time_alloc, t_start=t_start)
        if sc < best_score:
            do_update(sc, a_opt, f"Adam beta={beta_s}->{beta_e}")
        a_norm = a_opt

    # ==================================================================
    # Phase 2: LP-based updates (20% time)
    # ==================================================================
    p2 = effective_budget * 0.20
    print(f"\n{'='*50}")
    print(f"Phase 2: LP-based updates (budget={p2:.0f}s)")
    print(f"{'='*50}")

    deadline2 = t_start + p1 + p2
    for it in range(80):
        if time.monotonic() > deadline2:
            break
        if time.monotonic() - t_start > effective_budget - 10:
            break

        alpha = 0.15 + 0.2 * rng.random()
        a_lp = lp_based_update(best_a, K=30, alpha=alpha)
        sc = compute_score(a_lp)

        if sc < best_score:
            # Polish with quick Adam
            t0 = time.monotonic()
            sc2, a_polished = adam_adaptive(a_lp, lr_init=0.003, n_steps=20000,
                                             beta_start=200, beta_end=2000,
                                             time_limit_s=min(8.0, deadline2 - time.monotonic()),
                                             t_start=t0)
            if sc2 < sc:
                sc, a_lp = sc2, a_polished
            do_update(sc, a_lp, f"LP iter {it} a={alpha:.2f}")
        else:
            # Periodically try Adam from LP result even without immediate improvement
            if it % 8 == 0 and it > 0:
                time_left = deadline2 - time.monotonic()
                if time_left > 10:
                    t0 = time.monotonic()
                    sc2, a_polished = adam_adaptive(a_lp, lr_init=0.005, n_steps=30000,
                                                     beta_start=100, beta_end=3000,
                                                     time_limit_s=min(15.0, time_left - 5),
                                                     t_start=t0)
                    do_update(sc2, a_polished, f"LP+Adam iter {it}")

    # ==================================================================
    # Phase 3: Subgradient flattening (15% time)
    # ==================================================================
    p3 = effective_budget * 0.15
    print(f"\n{'='*50}")
    print(f"Phase 3: Subgradient flattening (budget={p3:.0f}s)")
    print(f"{'='*50}")

    deadline3 = deadline2 + p3
    sc, a_opt = iterative_flattening(best_a, n_iters=300, top_k_peaks=15,
                                      reduce_frac=0.03,
                                      time_limit_s=p3 * 0.6, t_start=t_start)
    do_update(sc, a_opt, "Subgrad flatten")

    # Follow up with Adam
    if time.monotonic() < deadline3 - 5:
        t0 = time.monotonic()
        sc2, a_polished = adam_adaptive(a_opt, lr_init=0.003, n_steps=50000,
                                         beta_start=100, beta_end=3000,
                                         time_limit_s=min(20.0, deadline3 - time.monotonic() - 5),
                                         t_start=t0)
        do_update(sc2, a_polished, "Subgrad+Adam")

    # ==================================================================
    # Phase 4: L-BFGS polishing (20% time)
    # ==================================================================
    p4 = effective_budget * 0.20
    print(f"\n{'='*50}")
    print(f"Phase 4: L-BFGS polishing (budget={p4:.0f}s)")
    print(f"{'='*50}")

    deadline4 = deadline3 + p4
    for beta in [500, 1500, 3000, 5000]:
        if time.monotonic() > deadline4:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs_a(best_a, beta, maxiter=5000, maxfun=100000)
        print(f"    L-BFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
        do_update(sc, a_opt, f"L-BFGS beta={beta}")

    # ==================================================================
    # Phase 5: Diverse restarts (10% time)
    # ==================================================================
    p5 = effective_budget * 0.10
    print(f"\n{'='*50}")
    print(f"Phase 5: Diverse restarts (budget={p5:.0f}s)")
    print(f"{'='*50}")

    deadline5 = deadline4 + p5
    n_best = len(best_a)
    restart_count = 0
    max_restarts = 25

    while time.monotonic() < deadline5 and restart_count < max_restarts:
        time_left = deadline5 - time.monotonic()
        if time_left < 12:
            break
        restart_count += 1

        strategy = (restart_count - 1) % 5
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.03 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strategy == 1:
            # Peak cutting - reduce elements contributing to main peak
            c = fast_conv(best_a)
            k_star = int(c.argmax())
            js = max(0, k_star - n_best + 1)
            je = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            a_pert[js:je + 1] *= 0.5
            removed = float(best_a[js:je + 1].sum() - a_pert[js:je + 1].sum())
            mask = (np.arange(n_best) < js) | (np.arange(n_best) > je)
            if mask.sum() > 0:
                a_pert[mask] += removed / mask.sum()
        elif strategy == 2:
            # Multi-frequency modulation
            x = np.arange(n_best, dtype=np.float64) / n_best
            f1 = 1.0 + 4.0 * rng.random()
            p1_val = 2.0 * np.pi * rng.random()
            f2 = 3.0 + 7.0 * rng.random()
            p2_val = 2.0 * np.pi * rng.random()
            a_pert = best_a * (1.0 + 0.03 * np.cos(2*np.pi*f1*x + p1_val)
                               + 0.02 * np.cos(2*np.pi*f2*x + p2_val))
        elif strategy == 3:
            # Block modification
            center = int(rng.integers(n_best // 4, 3 * n_best // 4))
            width = max(10, n_best // 30)
            a_pert = best_a.copy()
            lo = max(0, center - width)
            hi = min(n_best, center + width)
            a_pert[lo:hi] *= rng.uniform(0.3, 1.3)
        else:
            # Invert dynamics: boost low-value regions, reduce high-value
            a_min = float(a_pert_min()) if False else float(best_a.min())
            a_max = float(best_a.max())
            if a_max - a_min > 1e-10:
                a_norm_local = (best_a - a_min) / (a_max - a_min)
                a_pert = best_a * (1.0 + 0.05 * (0.5 - a_norm_local))
            else:
                a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(n_best))

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick optimization
        adam_time = min(15.0, time_left * 0.4)
        t0 = time.monotonic()
        sc, a_r = adam_adaptive(a_pert, lr_init=0.003, n_steps=30000,
                                 beta_start=100, beta_end=2000,
                                 time_limit_s=adam_time, t_start=t0)
        # Quick L-BFGS if time permits
        if time.monotonic() < deadline5 - 8:
            sc2, a_r2, _ = run_lbfgs_a(a_r, 1000, maxiter=2000, maxfun=30000)
            if sc2 < sc:
                sc, a_r = sc2, a_r2
        do_update(sc, a_r, f"restart#{restart_count} s{strategy}")

    # ==================================================================
    # Phase 6: Try different lengths (10% time)
    # ==================================================================
    p6 = effective_budget * 0.10
    print(f"\n{'='*50}")
    print(f"Phase 6: Different lengths (budget={p6:.0f}s)")
    print(f"{'='*50}")

    deadline6 = deadline5 + p6
    n_parent = len(best_a)

    target_lengths = [int(n_parent * 0.7), int(n_parent * 1.3), int(n_parent * 1.5)]
    for tl in target_lengths:
        if time.monotonic() > deadline6:
            break
        if tl < 512 or tl > 60000:
            continue

        a_res = resample_sequence(best_a, tl)

        time_left = deadline6 - time.monotonic()
        if time_left < 20:
            continue
        t0 = time.monotonic()
        sc, a_opt = adam_adaptive(a_res, lr_init=0.003, n_steps=50000,
                                   beta_start=50, beta_end=2000,
                                   time_limit_s=min(15.0, time_left * 0.4),
                                   t_start=t0)
        if time.monotonic() < deadline6 - 5:
            sc2, a_opt2, _ = run_lbfgs_a(a_opt, 1000, maxiter=3000, maxfun=50000)
            if sc2 < sc:
                sc, a_opt = sc2, a_opt2
        do_update(sc, a_opt, f"n={tl}")

    # ==================================================================
    # Phase 7: Final intensive polish (remaining time)
    # ==================================================================
    print(f"\n{'='*50}")
    print(f"Phase 7: Final Polish")
    print(f"{'='*50}")

    final_deadline = t_start + effective_budget - 3.0

    # Multiple L-BFGS with very high beta
    for beta in [5000, 10000, 20000]:
        if time.monotonic() > final_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs_a(best_a, beta, maxiter=6000, maxfun=120000)
        print(f"    Final L-BFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
        do_update(sc, a_opt, f"final LBFGS beta={beta}")

    # Final Adam with high beta and small lr
    if time.monotonic() < final_deadline - 5:
        time_left = final_deadline - time.monotonic() - 3
        t0 = time.monotonic()
        sc, a_opt = adam_adaptive(best_a, lr_init=0.0005, n_steps=100000,
                                   beta_start=5000, beta_end=50000,
                                   time_limit_s=time_left, t_start=t0)
        do_update(sc, a_opt, "final Adam high-beta")

    # Final coordinate descent
    if time.monotonic() < final_deadline:
        time_left = final_deadline - time.monotonic()
        if time_left > 5:
            sc, a_opt = coordinate_descent(best_a, top_k=60, n_sweeps=5,
                                           time_limit_s=min(25.0, time_left - 2),
                                           t_start=time.monotonic())
            do_update(sc, a_opt, "final coord")

    # One more LP round if time permits
    if time.monotonic() < final_deadline - 5:
        a_lp_final = lp_based_update(best_a, K=50, alpha=0.1)
        sc = compute_score(a_lp_final)
        t0 = time.monotonic()
        sc2, a_pol = adam_adaptive(a_lp_final, lr_init=0.002, n_steps=20000,
                                    beta_start=500, beta_end=5000,
                                    time_limit_s=min(10.0, final_deadline - time.monotonic() - 2),
                                    t_start=t0)
        if sc2 < sc:
            sc, a_lp_final = sc2, a_pol
        do_update(sc, a_lp_final, "final LP")

    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
