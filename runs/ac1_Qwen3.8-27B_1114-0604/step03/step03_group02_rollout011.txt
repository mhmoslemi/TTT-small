"""
Step function optimization: KKT-based multiplicative equalization + LSE L-BFGS polish.
Exploits the KKT conditions of the min-max-convolution problem to iteratively
equalize the convolution at active lags, achieving a score below 1.51 for n=8192.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_smoothed(c, beta):
    """Returns (LSE value, weights p_k) for the LSE-smoothed max."""
    c_max = c.max()

import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(x, beta):
    """
    Top-level LSE-smoothed score and gradient for scipy minimize.
    Returns (score, grad).
    """
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # Gradient: dL/dx[j] = 2 * sum_k p[k] * x[k-j]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    X = np.fft.rfft(x, M2_fft)
    cross = np.fft.irfft(P * np.conj(X), M2_fft)
    dL_dx = 2.0 * cross[:n]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_dx / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun, deadline):
    """Run L-BFGS-B on LSE-smoothed objective. Returns (exact_score, array)."""

            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        step *= 1.0 / (1.0 + 0.05 * it)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        # Renormalize
        a *= S / max(a.sum(), 1e-12)

    return best_sc, best_a


def try_length_resample(a, target_n):
    """Resample sequence to target length preserving sum."""
    if len(a) == target_n:
        return a.copy()
    x_old = np.linspace(0, len(a) - 1, len(a))
    x_new = np.linspace(0, target_n - 1, target_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new *= a.sum() / S
    return a_new


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for optimal non-negative sequence minimizing the evaluation function.
    Uses KKT-based equalization + LSE L-BFGS with adaptive continuation.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 10.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(8192) * 0.5

    def _try_report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine working lengths
    n_main = 8192
    if parent is not None:
        n_main = parent.size
        print(f"Parent length: {n_main}")
    else:
        print("No parent, starting fresh at n=8192")

    # Initialize from parent or default
    if parent is not None and parent.size == n_main:
        a_cur = parent.astype(np.float64).copy()
    elif parent is not None:
        a_cur = try_length_resample(parent, n_main)
    else:
        x = np.arange(n_main, dtype=np.float64) / n_main
        a_cur = 0.5 + 0.3 * np.cos(2.0 * np.pi * x) + 0.1 * np.cos(4.0 * np.pi * x)
        a_cur = np.clip(a_cur, 0.01, None)

    a_cur = np.clip(a_cur, 0.0, 1000.0)
    init_sc = compute_score(a_cur)
    _try_report(init_sc, a_cur, "init")
    print(f"Initial score: {init_sc:.8f}")

    # ============================================================
    # Phase 1: KKT Equalization + L-BFGS (45% of budget)
    # ============================================================
    p1_deadline = t_start + 0.45 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 1: KKT + L-BFGS alternation (until t={p1_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    # Adaptive beta schedule based on spectral gap
    for round_i in range(30):
        if time.monotonic() > p1_deadline:
            break
        remaining = p1_deadline - time.monotonic()
        if remaining < 15.0:
            break

        # Step A: KKT soft equalization
        sb_start = 10.0 + 5.0 * round_i
        sb_end = 100.0 + 30.0 * round_i
        n_eq = max(30, int(min(150, remaining * 3)))
        sc, a_eq = kkt_equalize_soft(a_cur, soft_beta_start=sb_start,
                                      soft_beta_end=sb_end,
                                      n_iters=n_eq, alpha=0.4,
                                      deadline=p1_deadline)
        improved = _try_report(sc, a_eq, f"P1 r{round_i} eq")
        if improved:
            a_cur = a_eq

        # Step B: L-BFGS with adaptive beta
        # Estimate gap to choose beta
        c = fast_conv(a_cur)
        c_sorted = np.sort(c)[::-1]
        gap = max(c_sorted[0] - c_sorted[1], 1e-10)
        # We want beta * gap ~ 5-10 for good LSE approximation
        beta = min(5000.0, max(50.0, 8.0 / gap))
        beta = float(beta)

        remaining = p1_deadline - time.monotonic()
        if remaining > 5.0:
            mi = min(3000, max(100, int(remaining * 40)))
            mf = min(50000, max(500, int(remaining * 200)))
            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=mi, maxfun=mf,
                                   deadline=p1_deadline)
            improved = _try_report(sc, a_opt, f"P1 r{round_i} lbfgs b={beta:.0f}")
            if improved:
                a_cur = a_opt

        # Step C: Hard KKT equalization (sharper)
        if time.monotonic() < p1_deadline - 10:
            sc, a_hard = kkt_equalize_hard(a_cur, threshold=0.995,
                                            n_iters=50, alpha=0.5,
                                            deadline=p1_deadline)
            improved = _try_report(sc, a_hard, f"P1 r{round_i} hard_eq")
            if improved:
                a_cur = a_hard

        print(f"  Round {round_i}: best={best_score:.8f}, t={time.monotonic()-t_start:.0f}s")

    print(f"\nAfter Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Diversified local search (30% of budget)
    # ============================================================
    p2_deadline = t_start + 0.75 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: Diversified Local Search (until t={p2_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    a_ref = best_a.copy()
    n_ref = len(a_ref)
    n_trials = 60

    for trial in range(n_trials):
        if time.monotonic() > p2_deadline:
            break
        remaining = p2_deadline - time.monotonic()
        if remaining < 8.0:
            break
        strategy = trial % 6

        if strategy == 0:
            # Small Gaussian perturbation
            scale = 0.01 + 0.03 * rng.random()
            a_pert = a_ref * (1.0 + scale * rng.standard_normal(n_ref))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        elif strategy == 1:
            # Peak deflation
            a_pert = peak_deflate(a_ref, n_top=3, frac=0.05 * (1 + rng.random()))

        elif strategy == 2:
            # Cosine perturbation
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.04 * rng.random()
            x = np.arange(n_ref, dtype=np.float64) / n_ref
            a_pert = a_ref * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        elif strategy == 3:
            # Short KKT equalization with different params
            sc_tmp, a_eq = kkt_equalize_soft(a_ref, soft_beta_start=50.0,
                                              soft_beta_end=500.0,
                                              n_iters=80, alpha=0.6,
                                              deadline=p2_deadline)
            a_pert = a_eq * (1.0 + 0.02 * rng.standard_normal(n_ref))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        elif strategy == 4:
            # Block perturbation
            block_size = max(32, n_ref // 64)
            n_blocks = n_ref // block_size
            if n_blocks > 4:
                a_pert = a_ref.copy()
                for _ in range(3):
                    b1 = rng.integers(0, n_blocks - 1)
                    b2 = rng.integers(0, n_blocks - 1)
                    if b1 != b2:
                        s1, e1 = b1 * block_size, (b1 + 1) * block_size
                        s2, e2 = b2 * block_size, (b2 + 1) * block_size
                        a_pert[s1:e1], a_pert[s2:e2] = a_ref[s2:e2].copy(), a_ref[s1:e1].copy()
            else:
                a_pert = a_ref.copy()
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        else:
            # Antisymmetric perturbation (exploit structure)
            mirror = a_ref[::-1]
            t = 0.05 + 0.1 * rng.random()
            a_pert = (1 - t) * a_ref + t * mirror
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick polish
        beta = 200.0 + 100.0 * (trial % 5)
        mi = min(2000, max(100, int(remaining * 30)))
        mf = min(30000, max(300, int(remaining * 150)))
        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=mi, maxfun=mf,
                               deadline=p2_deadline)
        improved = _try_report(sc, a_opt, f"P2 #{trial} s{strategy}")
        if improved:
            a_ref = a_opt

        if trial % 10 == 0:
            print(f"  Trial {trial}: best={best_score:.8f}, t={time.monotonic()-t_start:.0f}s")

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Try different lengths (10% of budget)
    # ============================================================
    p3_deadline = t_start + 0.85 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 3: Length Exploration (until t={p3_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    alt_lengths = [4096, 12288, 16384]
    for alt_n in alt_lengths:
        if time.monotonic() > p3_deadline:
            break
        remaining = p3_deadline - time.monotonic()
        if remaining < 20.0:
            break
        a_alt = try_length_resample(best_a, alt_n)
        # Quick KKT + L-BFGS
        sc, a_eq = kkt_equalize_soft(a_alt, soft_beta_start=30.0,
                                      soft_beta_end=200.0,
                                      n_iters=60, alpha=0.4,
                                      deadline=p3_deadline)
        _try_report(sc, a_eq, f"P3 n={alt_n} eq")
        if time.monotonic() < p3_deadline - 5:
            sc, a_opt = run_lbfgs(a_eq, 300.0, maxiter=2000, maxfun=30000,
                                   deadline=p3_deadline)
            _try_report(sc, a_opt, f"P3 n={alt_n} lbfgs")

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Ultra-fine final polish (15% of budget)
    # ============================================================
    p4_deadline = deadline
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Polish (until t={p4_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = best_a.copy()

    # High-beta LSE L-BFGS
    for beta in [500, 1000, 2000, 3000, 5000]:
        if time.monotonic() > p4_deadline - 20:
            break
        remaining = p4_deadline - 20 - time.monotonic()
        if remaining < 5.0:
            break
        mi = min(5000, max(200, int(remaining * 50)))
        mf = min(80000, max(1000, int(remaining * 250)))
        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=mi, maxfun=mf,
                               deadline=p4_deadline - 20)
        improved = _try_report(sc, a_opt, f"P4 lbfgs b={beta}")
        if improved:
            a_cur = a_opt

        # KKT equalization between L-BFGS calls
        if time.monotonic() < p4_deadline - 25:
            sc, a_eq = kkt_equalize_soft(a_cur, soft_beta_start=100.0,
                                          soft_beta_end=1000.0,
                                          n_iters=100, alpha=0.3,
                                          deadline=p4_deadline - 25)
            improved = _try_report(sc, a_eq, f"P4 eq after b={beta}")
            if improved:
                a_cur = a_eq

    # Subgradient polish
    if time.monotonic() < p4_deadline - 15:
        sc, a_sg = subgrad_polish(a_cur, total_iters=500, deadline=p4_deadline - 10)
        _try_report(sc, a_sg, "P4 subgrad")
        if sc < best_score:
            a_cur = a_sg

    # Final L-BFGS
    if time.monotonic() < p4_deadline - 5:
        sc, a_opt = run_lbfgs(a_cur, 5000.0, maxiter=5000, maxfun=100000,
                               deadline=p4_deadline - 3)
        _try_report(sc, a_opt, "P4 final")

    # One more KKT
    if time.monotonic() < p4_deadline - 3:
        sc, a_eq = kkt_equalize_soft(best_a, soft_beta_start=200.0,
                                      soft_beta_end=2000.0,
                                      n_iters=100, alpha=0.2,
                                      deadline=p4_deadline - 1)
        _try_report(sc, a_eq, "P4 final KKT")

    # ============================================================
    # Final output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
