    contrib = np.zeros(n)

    if j_end >= j_start:
        contrib[j_start:j_end+1] = a[j_start:j_end+1][::-1]

def cd_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    conv_in = base_inside + t * alpha + t * t * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)

def coordinate_descent(a, order=None, n_sweeps=1, use_hot=False, rng=None, momentum=1.0):
    """
    Coordinate descent with optional hot-spot ordering and momentum.
    momentum < 1 means we don't fully optimize each coordinate.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1
    
    for sweep in range(n_sweeps):
        if use_hot:
            k_star = int(np.argmax(conv))
            contrib = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                contrib[j_start:j_end+1] = a[j_start:j_end+1][::-1]
            order = np.argsort(contrib)[::-1]

"""
Hybrid optimizer for minimum peak autocorrelation of non-negative sequences.
Uses exact coordinate descent with hot-spot reordering, LSE-smoothed L-BFGS,
and diverse targeted perturbation strategies. Key improvements:
- High iteration count CD with adaptive ordering
- Targeted "mass transfer" perturbations (hot->cold coordinates)
- Multi-beta LSE refinement
- Pairwise exchange moves
- Multiple sequence length trials
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def cd_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + t * t * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def coordinate_descent(a, order=None, use_hot=False, rng=None, xatol=1e-7):
    """
    One full sweep of coordinate descent with incremental conv updates.
    Returns (score, updated_a).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    if use_hot:
        k_star = int(np.argmax(conv))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        order = np.argsort(contrib)[::-1]
    elif order is None:
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

    for idx in range(n):
        i = int(order[idx])
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()

        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.05, S_rest / n * 4.0)
        t_upper = min(t_upper, 1000.0)

        res = minimize_scalar(
            cd_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': xatol}
        )
        t_opt = float(res.x)

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -60, 60)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(x, beta):
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def hot_cold_perturb(a, rng, strength=0.03):
    """
    Move mass from coordinates contributing to hot lags to cold coordinates.
    """
    n = len(a)
    c = fast_conv(a)
    peak = float(c.max())
    threshold = 0.90 * peak
    hot_lags = np.where(c >= threshold)[0]

    hot_coords = np.zeros(n)
    for k in hot_lags:
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_end >= j_start:
            hot_coords[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]

    if hot_coords.max() < 1e-15:
        return a * (1.0 + 0.01 * rng.standard_normal(n))

    q75 = np.percentile(hot_coords, 75)
    q25 = np.percentile(hot_coords, 25)
    hot_mask = hot_coords >= q75
    cold_mask = hot_coords <= q25

    n_hot = int(hot_mask.sum())
    n_cold = int(cold_mask.sum())
    if n_hot < 3 or n_cold < 3:
        idx = rng.integers(0, n, size=max(3, n // 50))
        a_pert = a.copy()
        saved = 0.0
        for j in idx:
            amt = a_pert[j] * strength
            a_pert[j] -= amt
            saved += amt
        other = np.setdiff1d(np.arange(n), idx)
        if len(other) > 0:
            a_pert[other] += saved / len(other)
        return np.clip(a_pert, 0.0, 1000.0)

    hot_idx = np.where(hot_mask)[0]
    cold_idx = np.where(cold_mask)[0]
    a_pert = a.copy()
    total_moved = 0.0
    for j in hot_idx:
        amt = a_pert[j] * strength
        a_pert[j] -= amt
        total_moved += amt
    a_pert[cold_idx] += total_moved / n_cold
    return np.clip(a_pert, 0.0, 1000.0)


def pairwise_exchange(a, rng, n_pairs=10):
    """
    Random pairwise mass exchanges: move a small amount from one coordinate to another.
    """
    n = len(a)
    a_pert = a.copy()
    c = fast_conv(a)
    peak = float(c.max())

    for _ in range(n_pairs):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        amt = min(a_pert[i], a_pert[i] * 0.02 * rng.random())
        a_pert[i] -= amt
        a_pert[j] += amt
    return np.clip(a_pert, 0.0, 1000.0)


def gaussian_perturb(a, rng, scale=0.02):
    """Gaussian multiplicative perturbation."""
    n = len(a)
    a_pert = a * (1.0 + scale * rng.standard_normal(n))
    return np.clip(a_pert, 0.0, 1000.0)


def block_perturb(a, rng):
    """Zero a random block and redistribute mass."""
    n = len(a)
    block_size = max(5, n // 200)
    start = int(rng.integers(0, max(1, n - block_size)))
    a_pert = a.copy()
    saved = a_pert[start:start + block_size].sum()
    a_pert[start:start + block_size] = 0.0
    rest = np.ones(n, dtype=bool)
    rest[start:start + block_size] = False
    rest_count = int(rest.sum())
    if rest_count > 0:
        a_pert[rest] += saved / rest_count
    return np.clip(a_pert, 0.0, 1000.0)


def sinusoidal_perturb(a, rng):
    """Apply a sinusoidal modulation."""
    n = len(a)
    freq = 1.0 + 5.0 * rng.random()
    amp = 0.005 + 0.02 * rng.random()

    x = np.arange(n, dtype=np.float64) / n
    mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_pert = a * mod
    return np.clip(a_pert, 0.0, 1000.0)

        pol_count += 1
        if pol_count % 3 == 0:
            sc, a_cur = coordinate_descent(a_cur, use_hot=True, xatol=1e-8)
        else:
            sc, a_cur = coordinate_descent(a_cur, rng=rng, xatol=1e-8)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            if pol_count % 20 == 0:
                print(f"  P3 S{pol_count}: {sc:.8f} (BEST)")

    print(f"Phase 3 done: {pol_count} sweeps, best={best_score:.8f}")

    # ============================================================
    # Phase 4: Diverse perturbation + CD cycles (20% of time)
    # ============================================================
    phase4_deadline = deadline - 30.0
    print(f"\n=== Phase 4: Perturbation Cycles ({phase4_deadline - t_start:.0f}s) ===")

    trial = 0
    while time.monotonic() < phase4_deadline:
        trial += 1
        remaining = phase4_deadline - time.monotonic()
        if remaining < 10.0:
            break

        # Choose perturbation strategy (weighted)
        r = rng.random()
        if r < 0.25:
            a_pert = hot_cold_perturb(best_a, rng, strength=0.01 + 0.03 * rng.random())
        elif r < 0.40:
            a_pert = pairwise_exchange(best_a, rng, n_pairs=5 + int(15 * rng.random()))
        elif r < 0.55:
            a_pert = gaussian_perturb(best_a, rng, scale=0.005 + 0.02 * rng.random())
        elif r < 0.70:
            a_pert = block_perturb(best_a, rng)
        elif r < 0.85:
            a_pert = sinusoidal_perturb(best_a, rng)
        else:
            # Aggressive: combine hot-cold with stronger strength
            a_pert = hot_cold_perturb(best_a, rng, strength=0.04 + 0.04 * rng.random())

        # Quick CD from perturbed point (adapt number of sweeps to remaining time)
        n_cd = min(30, max(3, int(remaining * 0.3)))
        for sw in range(n_cd):
            if time.monotonic() > phase4_deadline - 3.0:
                break
            if sw % 4 == 3:
                sc_p, a_pert = coordinate_descent(a_pert, use_hot=True, xatol=1e-7)
            else:
                sc_p, a_pert = coordinate_descent(a_pert, rng=rng, xatol=1e-7)

        sc_pert = compute_score(a_pert)
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  P4 T{trial}: {sc_pert:.8f} (BEST)")
        elif trial % 10 == 0:
            print(f"  P4 T{trial}: {sc_pert:.8f} (best={best_score:.8f})")

    # ============================================================
    # Phase 5: Try different lengths (if time permits)
    # ============================================================
    phase5_deadline = deadline - 15.0
    if time.monotonic() < phase5_deadline:
        elapsed_so_far = time.monotonic() - t_start
        remaining = phase5_deadline - time.monotonic()
        if remaining > 60:
            print(f"\n=== Phase 5: Length Variants ({remaining:.0f}s remaining) ===")
            n_base = len(best_a)
            # Try a few nearby lengths
            for frac in [0.8, 1.2, 1.5]:
                if time.monotonic() > phase5_deadline - 10:
                    break
                n_new = int(n_base * frac)
                n_new = max(256, min(n_new, 20000))
                # Truncate or pad
                if n_new < n_base:
                    a_new = best_a[:n_new].copy()
                else:
                    a_new = np.zeros(n_new)
                    a_new[:n_base] = best_a
                sc_new = compute_score(a_new)
                print(f"  Length {n_new}: initial score={sc_new:.8f}")
                # Quick CD
                for _ in range(5):
                    if time.monotonic() > phase5_deadline - 10:
                        break
                    sc_new, a_new = coordinate_descent(a_new, rng=rng, xatol=1e-7)
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    print(f"  Length {n_new}: {sc_new:.8f} (BEST!)")
                else:
                    print(f"  Length {n_new}: {sc_new:.8f}")

    # ============================================================
    # Final polish
    # ============================================================
    print(f"\n=== Final Polish ===")
    a_cur = best_a.copy()
    for _ in range(10):
        if time.monotonic() > deadline - 5.0:
            break
        sc, a_cur = coordinate_descent(a_cur, use_hot=True, xatol=1e-8)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

    # One final LSE if time permits
    if time.monotonic() < deadline - 8.0:
        sc, a_opt, _ = run_lbfgs(best_a, 5000, maxiter=3000, maxfun=50000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final LSE: {sc:.8f} (BEST)")
        a_cur = best_a.copy()
        if time.monotonic() < deadline - 3.0:
            sc, a_cur = coordinate_descent(a_cur, rng=rng, xatol=1e-8)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                print(f"  Final CD: {sc:.8f} (BEST)")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 256

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
