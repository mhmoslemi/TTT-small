"""
Multi-phase optimizer for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Uses LSE continuation in normalized space + L-BFGS + peak-targeted coordinate descent + SPSA.
Key improvements over previous: structured initializations, adaptive beta, symmetry exploitation,
and a dedicated peak-reduction phase.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_normalized_obj_grad(a, beta):
    """
    LSE-smoothed objective and gradient for a probability distribution (sum a = 1).
    Returns (score, grad) where score = 2n * LSE approximation of max(a*a),
    and grad is the gradient w.r.t. a (on the normalized simplex).
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    a_norm = a / S
    c = fast_conv(a_norm)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        exp_sum = 1e-300
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L

    p = exp_scaled / exp_sum
    rev_a = a_norm[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad = 2.0 * n * conv_res
    return score, grad

def adam_normalized(a_init, lr=0.01, n_steps=50000, beta=100,
                    time_limit_s=None, t_start=None, verbose_interval=50000):
    """Adam optimizer on the probability simplex using LSE-smoothed gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    deadline = (t_start + time_limit_s) if (time_limit_s is not None and t_start is not None) else np.inf

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break

        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_normalized_obj_grad(a, beta)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        S_new = a.sum()
        if S_new > 1e-12:
            a = a / S_new
        else:
            a[:] = 1.0 / n
            m[:] = 0.0
            v[:] = 0.0

        if t % verbose_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"      AdamNorm t={t}: score={sc:.8f}")

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def run_lbfgs_normalized(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B in log-space (softmax) with LSE smoothing, for probability distributions."""
    n = len(a0)
    a0_safe = np.clip(a0, 1e-12, 1e10)
    x0 = np.log(a0_safe) - np.log(a0_safe).mean()

    def obj_grad(x):
        x_shifted = x - x.max()
        e = np.exp(x_shifted)
        a = e / e.sum()
        sc, grad_a = lse_normalized_obj_grad(a, beta)
        d = float(np.dot(grad_a, a))
        grad_x = a * (grad_a - d)
        return sc, grad_x

    res = minimize(
        obj_grad, x0,
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-16, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    x_out = res.x
    x_shifted = x_out - x_out.max()
    e = np.exp(x_shifted)
    a_out = e / e.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def peak_targeted_coord_descent(a, n_sweeps=20, top_k=50,
                                time_limit_s=None, t_start=None):
    """
    Coordinate descent that specifically targets the peak lag.
    For each coordinate, tries to reduce the value at the max lag while
    maintaining or improving the score.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_limit_s) if (time_limit_s is not None and t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        k_star = int(c.argmax())

        # Compute gradient at peak
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)

        # Score with exact peak
        M = float(c[k_star])
        score = 2.0 * n * M / (S * S)

        if score < best_score:
            best_score = score

        # Gradient w.r.t. a[j] for the peak constraint
        # d(M)/da[j] = a[k_star - j] (if 0 <= k_star - j < n)
        # d(score)/da[j] = 2n * (dM/da[j] / S^2 - 2*M*a[j]/S^3)
        grad = np.zeros(n)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            valid = (idx >= 0) & (idx < n)
            vals = np.zeros(je - js + 1)
            vals[valid] = a[idx[valid]]
            S2 = S * S
            S3 = S2 * S
            grad[js:je + 1] = 2.0 * n * (vals / S2 - 2.0 * M * a[js:je+1] / S3)

        # Also compute full subgradient (in case another lag is also at max)
        top_lags = np.argsort(c)[::-1][:3]
        for k in top_lags[1:]:
            kk = int(k)
            if c[kk] < M * 0.99:
                break
            js2 = max(0, kk - n + 1)
            je2 = min(kk, n - 1)
            if je2 >= js2:
                idx2 = kk - np.arange(js2, je2 + 1)
                valid2 = (idx2 >= 0) & (idx2 < n)
                vals2 = np.zeros(je2 - js2 + 1)
                vals2[valid2] = a[idx2[valid2]]
                grad[js2:je2 + 1] += 2.0 * n * (vals2 / S2 - 2.0 * M * a[js2:je2+1] / S3)

        # Select top-k coordinates by gradient magnitude (prefer those at peak)
        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break

            a_old = a[j]
            g_j = grad[j]

            if g_j > 1e-10:
                # Reducing a[j] helps
                ratios = np.array([0.1, 0.3, 0.5, 0.7, 0.85, 0.95, 1.0])
                candidates = a_old * ratios
            elif g_j < -1e-10:
                # Increasing a[j] helps
                ratios = np.array([1.0, 1.1, 1.3, 1.5, 2.0, 3.0, 5.0])
                candidates = np.concatenate([[0.0], a_old * ratios])
            else:
                continue

            candidates = np.clip(candidates, 0.0, 1000.0)

            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def spsa_optimize(a_init, n_steps=20000, alpha=0.05, gamma=0.01, A=50,
                  delta_scale=0.1, time_limit_s=None, t_start=None):
    """
    SPSA (Simultaneous Perturbation Stochastic Approximation) on the simplex.
    Perturbs the entire vector simultaneously.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S

    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(42)
    deadline = (t_start + time_limit_s) if (time_limit_s is not None and t_start is not None) else np.inf

    for k in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break

        # SPSA step size
        a_k = alpha / ((A + k) ** gamma)

        # Random perturbation direction
        d = rng.choice([-1.0, 1.0], size=n)

        # Evaluate at a +/- delta*d
        delta = delta_scale * a_k * np.maximum(a, 0.01)
        a_plus = np.clip(a + delta * d, 0, 1e10)
        a_plus = a_plus / a_plus.sum()
        a_minus = np.clip(a - delta * d, 0, 1e10)
        a_minus = a_minus / a_minus.sum()

        f_plus = compute_score(a_plus)
        f_minus = compute_score(a_minus)

        # SPSA gradient estimate
        g = (f_plus - f_minus) / (2.0 * np.maximum(delta, 1e-10))

        # Update
        a_new = a - a_k * g
        a_new = np.clip(a_new, 0, 1e10)
        S_new = a_new.sum()
        if S_new > 1e-12:
            a_new = a_new / S_new
        else:
            a_new = a.copy()
        a = a_new

        if k % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"      SPSA k={k}: score={sc:.8f}")

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def make_cosine_init(n, k, phase=0.0, amp=0.3):
    """Create a cosine-modulated initialization."""
    x = np.arange(n, dtype=np.float64) / n
    a = 1.0 + amp * np.cos(2 * np.pi * k * x + phase)
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-10:
        a[:] = 1.0 / n
    else:
        a = a / a.sum()
    return a


def make_parabolic_init(n, curvature=1.0):
    """Parabolic initialization."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 + curvature * (2 * x - 1) ** 2
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-10:
        a[:] = 1.0 / n
    else:
        a = a / a.sum()
    return a


def make_random_structured_init(n, rng, n_freqs=5):
    """Random structured initialization using sum of random sinusoids."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for i in range(n_freqs):
        freq = rng.integers(1, max(2, n // 4))
        phase = 2 * np.pi * rng.random()
        amp = rng.uniform(0.05, 0.3)
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-10:
        a[:] = 1.0 / n
    else:
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def lbfgs_with_callback(a0, beta, maxiter=5000, maxfun=100000,
                        time_limit_s=None, t_start=None):
    """L-BFGS in log-space with time monitoring via callback."""
    n = len(a0)
    a0_safe = np.clip(a0, 1e-12, 1e10)
    x0 = np.log(a0_safe) - np.log(a0_safe).mean()
    best_score = compute_score(a0)
    best_x = x0.copy()
    stop_flag = [False]
    deadline = (t_start + time_limit_s) if (time_limit_s is not None and t_start is not None) else np.inf

    def obj_grad(x):
        if time.monotonic() > deadline:
            stop_flag[0] = True
        x_shifted = x - x.max()
        e = np.exp(x_shifted)
        a = e / e.sum()
        sc, grad_a = lse_normalized_obj_grad(a, beta)
        d = float(np.dot(grad_a, a))
        grad_x = a * (grad_a - d)
        return sc, grad_x

    def callback(xk):
        if time.monotonic() > deadline:
            stop_flag[0] = True
        sc = compute_score(softmax_from_x(xk))
        nonlocal best_score, best_x
        if sc < best_score:
            best_score = sc
            best_x = xk.copy()
        return stop_flag[0]

    res = minimize(
        obj_grad, x0,
        method='L-BFGS-B', jac=True, callback=callback,
        options={'ftol': 1e-16, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    x_out = res.x
    a_out = softmax_from_x(x_out)
    final_sc = compute_score(a_out)
    if final_sc < best_score:
        best_score = final_sc
        best_x = x_out.copy()
    a_best = softmax_from_x(best_x)
    best_sc = compute_score(a_best)
    return best_sc, a_best, res


def softmax_from_x(x):
    """Convert x to probability vector via softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

    # Random structured
    for r in range(5):
        a_init = make_random_structured_init(n_base, rng, n_freqs=5)
        sc = compute_score(a_init)
        candidates.append((f'rand_{r}', a_init, sc))

    # Symmetric cosine (a[j] = a[n-1-j])
    for k in [1, 2, 4]:
        x = np.arange(n_base, dtype=np.float64) / n_base
        a_half = 1.0 + 0.3 * np.cos(2 * np.pi * k * x[:n_base // 2])
        a_half = np.clip(a_half, 0.0, None)
        a_sym = np.concatenate([a_half, a_half[::-1]])
        if len(a_sym) > n_base:
            a_sym = a_sym[:n_base]
        elif len(a_sym) < n_base:
            a_sym = np.concatenate([a_sym, a_sym[-1:]])
        if a_sym.sum() > 1e-10:
            a_sym = a_sym / a_sym.sum()
        sc = compute_score(a_sym)
        candidates.append((f'sym_cos_k{k}', a_sym, sc))

    # Sort by score and take top candidates
    candidates.sort(key=lambda x: x[2])
    print(f"  Generated {len(candidates)} initializations, best: {candidates[0][2]:.8f}")

    # Quick Adam on top 8 candidates
    top_candidates = candidates[:8]
    time_per_init = p1_time / len(top_candidates) / 2

    for label, a_init, sc_init in top_candidates:
        if time.monotonic() > t_start + p1_time:
            break
        remaining = t_start + p1_time - time.monotonic()
        if remaining < 10:
            break
        t_alloc = min(time_per_init, remaining * 0.4)

        sc, a_opt = adam_normalized(a_init, lr=0.01, n_steps=200000,
                                    beta=50, time_limit_s=t_alloc, t_start=t_start)
        try_update(sc, a_opt, f"P1:{label}")
        # Quick L-BFGS
        if time.monotonic() < t_start + p1_time - 5:
            t0 = time.monotonic()
            sc2, a2, _ = run_lbfgs_normalized(a_opt, 200, maxiter=2000, maxfun=30000)
            try_update(sc2, a2, f"P1:{label}+LBFGS")

    # ================================================================
    # Phase 2: LSE Continuation (25% of budget)
    # ================================================================
    p2_time = effective_budget * 0.25
    print(f"\n{'='*60}")
    print(f"Phase 2: LSE Continuation (budget={p2_time:.0f}s)")
    print(f"{'='*60}")

    p2_deadline = t_start + p1_time + p2_time
    betas = [50, 100, 200, 500, 1000, 2000, 5000]
    a_cur = best_a.copy()

    for beta in betas:
        remaining = p2_deadline - time.monotonic()
        if remaining < 20:
            break
        t_alloc = min(remaining * 0.5, 120.0)

        # Adam with this beta
        sc, a_opt = adam_normalized(a_cur, lr=0.005, n_steps=50000,
                                    beta=beta, time_limit_s=t_alloc * 0.6, t_start=t_start)
        try_update(sc, a_opt, f"P2:Adam b={beta}")
        a_cur = a_opt

        # L-BFGS with this beta
        if time.monotonic() < p2_deadline - 10:
            t0 = time.monotonic()
            sc, a_opt, _ = run_lbfgs_normalized(a_cur, beta, maxiter=3000, maxfun=50000)
            try_update(sc, a_opt, f"P2:LBFGS b={beta}")
            a_cur = a_opt
            print(f"    Beta={beta}: score={sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ================================================================
    # Phase 3: High-beta L-BFGS Polish (20% of budget)
    # ================================================================
    p3_time = effective_budget * 0.20
    print(f"\n{'='*60}")
    print(f"Phase 3: High-beta L-BFGS Polish (budget={p3_time:.0f}s)")
    print(f"{'='*60}")

    p3_deadline = t_start + p1_time + p2_time + p3_time
    for beta in [2000, 5000, 10000, 20000]:
        remaining = p3_deadline - time.monotonic()
        if remaining < 15:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs_normalized(best_a, beta, maxiter=8000, maxfun=150000)
        try_update(sc, a_opt, f"P3:LBFGS b={beta}")
        print(f"    LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Also try direct L-BFGS
    if time.monotonic() < p3_deadline - 10:
        a_direct = best_a * best_a.sum()  # Restore to unnormalized
        for beta in [5000, 10000]:
            if time.monotonic() >= p3_deadline - 5:
                break
            t0 = time.monotonic()
            sc, a_opt, _ = run_lbfgs_direct(a_direct, beta, maxiter=5000, maxfun=100000)
            try_update(sc, a_opt, f"P3:LBFGS-direct b={beta}")
            print(f"    LBFGS-direct beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ================================================================
    # Phase 4: Peak-targeted coordinate descent + SPSA (15%)
    # ================================================================
    p4_time = effective_budget * 0.15
    print(f"\n{'='*60}")
    print(f"Phase 4: Peak Coord Descent + SPSA (budget={p4_time:.0f}s)")
    print(f"{'='*60}")

    p4_deadline = t_start + p1_time + p2_time + p3_time + p4_time

    # Coordinate descent
    t0 = time.monotonic()
    remaining = p4_deadline - time.monotonic()
    if remaining > 20:
        sc, a_opt = peak_targeted_coord_descent(
            best_a, n_sweeps=50, top_k=60,
            time_limit_s=remaining * 0.5, t_start=t_start)
        try_update(sc, a_opt, "P4:PeakCoord")
        print(f"    PeakCoord: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # SPSA
    t0 = time.monotonic()
    remaining = p4_deadline - time.monotonic()
    if remaining > 20:
        sc, a_opt = spsa_optimize(best_a, n_steps=10000, alpha=0.1,
                                  gamma=0.6, A=100, delta_scale=0.05,
                                  time_limit_s=remaining * 0.5, t_start=t_start)
        try_update(sc, a_opt, "P4:SPSA")
        print(f"    SPSA: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ================================================================
    # Phase 5: Final intensive polish (10%)
    # ================================================================
    p5_time = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Polish (budget={p5_time:.0f}s)")
    print(f"{'='*60}")

    p5_deadline = t_start + effective_budget - 5.0

    # Very high beta L-BFGS
    for beta in [10000, 30000]:
        if time.monotonic() > p5_deadline - 5:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs_normalized(best_a, beta, maxiter=10000, maxfun=200000)
        try_update(sc, a_opt, f"P5:LBFGS b={beta}")
        print(f"    Final LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Final coordinate descent
    if time.monotonic() < p5_deadline - 5:
        remaining = p5_deadline - time.monotonic()
        t0 = time.monotonic()
        sc, a_opt = peak_targeted_coord_descent(
            best_a, n_sweeps=30, top_k=80,
            time_limit_s=min(remaining * 0.6, 100.0), t_start=t_start)
        try_update(sc, a_opt, "P5:FinalCoord")
        print(f"    FinalCoord: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Final Adam with very high beta
    if time.monotonic() < p5_deadline - 5:
        remaining = p5_deadline - time.monotonic()
        sc, a_opt = adam_normalized(best_a, lr=0.001, n_steps=50000,
                                    beta=10000, time_limit_s=remaining * 0.5,
                                    t_start=t_start)
        try_update(sc, a_opt, "P5:FinalAdam")

    # One more L-BFGS if time
    if time.monotonic() < p5_deadline - 3:
        sc, a_opt, _ = run_lbfgs_normalized(best_a, 20000, maxiter=5000, maxfun=100000)
        try_update(sc, a_opt, "P5:UltimateLBFGS")

    # ================================================================
    # Final verification and return
    # ================================================================
    # Scale best_a to have reasonable magnitude
    out = best_a.copy()
    S = out.sum()
    if S < 0.01:
        out = out * (1.0 / max(S, 1e-12))
    # Ensure all values are in [0, 1000]
    out = np.clip(out, 0.0, 1000.0)

    out_list = [float(x) for x in out]
    verified = float(evaluate_sequence(out_list))
    elapsed = time.monotonic() - t_start

    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out_list)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")

    if verified >= np.inf or not np.isfinite(verified):
        # Fallback
        out_list = [0.5] * 64

    return out_list
