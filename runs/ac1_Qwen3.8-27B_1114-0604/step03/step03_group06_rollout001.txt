"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines Adam with LSE-smoothed objective, multi-resolution warm start,
frequency-domain initialization, LP direction finding, and surgical local search.
"""

import numpy as np
from scipy.optimize import minimize
import time

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except ImportError:
    HAVE_CVXPY = False


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def lse_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled - scaled.max())
    log_sum = np.log(exp_s.sum()) + scaled.max()
    L = c_max + log_sum / beta
    p = exp_s / exp_s.sum()
    sc = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return sc, grad


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        beta = beta_start * (beta_end / beta_start) ** (it / max(n_iters - 1, 1))
        sc, g = lse_grad(a, beta)
        if not np.isfinite(g).all() or np.max(np.abs(g)) > 1e6:
            break
        t = it + 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)
        exact = compute_score(a)
        if exact < best_sc:
            best_sc = exact
            best_a = a.copy()
    return best_sc, best_a


def lbfgs_polish(a0, beta, maxiter=3000, maxfun=50000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lambda x: lse_grad(x, beta), a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-10, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    sc = compute_score(a_out)
    return sc, a_out


def upsample(a, factor):
    n = len(a)
    new_n = n * factor
    idx = np.linspace(0, n - 1, new_n)
    i0 = np.floor(idx).astype(int)
    i1 = np.minimum(i0 + 1, n - 1)
    frac = idx - i0
    return a[i0] * (1 - frac) + a[i1] * frac


def build_inits(n, rng, parent):
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].astype(np.float64).copy())
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    inits.append(np.clip(0.5 + 0.3 * np.sinc(3.0 * (x - 0.5)), 0.01, 1000.0))
    inits.append(np.clip(1.0 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.4 * rng.random(n), 0.01, 1000.0))
    # Chirp
    inits.append(np.clip(0.5 + 0.4 * np.cos(2 * np.pi * 5 * x * x), 0.01, 1000.0))
    # Frequency-domain flat
    N_fft = 1 << (2 * n - 1).bit_length()
    A = np.ones(N_fft // 2 + 1) + 0.1 * rng.standard_normal(N_fft // 2 + 1)
    A_flat = np.zeros(N_fft // 2 + 1)
    A_flat[:] = 1.0 + 0.1 * rng.standard_normal(N_fft // 2 + 1)
    a_fd = np.fft.irfft(A_flat, N_fft)[:n]
    inits.append(np.clip(a_fd / np.max(np.abs(a_fd)) * 0.5 + 0.5, 0.01, 1000.0))
    # Block structure
    for nb in (8, 16, 32):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a_blk = np.repeat(vals, per)
        if len(a_blk) < n:
            a_blk = np.concatenate([a_blk, vals[:n - len(a_blk)]])
        inits.append(a_blk[:n].copy())
    return inits


def solve_lp_direction(a, max_wall_s=20.0):
    if not HAVE_CVXPY:
        return None
    n = len(a)
    t0 = time.monotonic()
    if n > 5000:
        return None
    M = float(np.max(fast_conv(a)))
    if M < 1e-12:
        return a.copy()
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_mat[k_start:k_end, j] = a[:k_end - k_start]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.PDLP, max_time=max_wall_s)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_time=max_wall_s)
        except Exception:
            return None
    if time.monotonic() - t0 > max_wall_s:
        return None
    if b_var.value is None:
        return None
    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None
    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    return b_opt


def lp_step(a, max_wall_s=15.0):
    old_sc = compute_score(a)
    b_opt = solve_lp_direction(a, max_wall_s=max_wall_s)
    if b_opt is None:
        return old_sc, a
    best_sc = old_sc
    best_a = a
    for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
        a_new = (1.0 - t) * a + t * b_opt
        a_new = np.clip(a_new, 0.0, 1000.0)
        if a_new.sum() < 0.01:
            continue
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new
    return best_sc, best_a


def targeted_perturb(a, rng, strength):
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:max(top_count, n // 4)]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def subgrad_polish(a, total_iters=200, deadline=None):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = min(0.1, (sc - 2.0 + 0.01) / (gn * gn + 1e-30))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
        for dl in [-500, 500, -1000, 1000]:
            cand = parent.size + dl
            if cand >= 2048:
                lengths.append(cand)
    for n in [4096, 8192, 12288, 16384]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths: {lengths}, parent: {parent is not None}, CVXPY: {HAVE_CVXPY}")

    # Phase 1: Multi-resolution warm start + Adam (45% of time)
    p1_deadline = t_start + 0.45 * (total_budget - 20.0)
    print(f"\n{'='*60}\nPhase 1: Adam optimization\n{'='*60}")

    for n in lengths:
        if time.monotonic() > p1_deadline:
            break
        print(f"\n--- n={n} ---")
        inits = build_inits(n, rng, parent)
        remaining = p1_deadline - time.monotonic()
        max_inits = max(2, min(len(inits), int(remaining / 60.0)))
        inits = inits[:max_inits]

        # Multi-resolution: optimize small first, then upsample
        if n >= 4096 and remaining > 120:
            small_n = n // 16
            small_inits = build_inits(small_n, rng, None)
            for si, s_init in enumerate(small_inits[:3]):
                if time.monotonic() > p1_deadline:
                    break
                sc_s, a_s = adam_optimize(s_init, 50, 5000, 2000, 0.5, p1_deadline)
                if sc_s < best_score:
                    a_up = upsample(a_s, 16) if len(a_s) * 16 == n else upsample(a_s, n // len(a_s))
                    a_up = np.clip(a_up, 0.01, 1000.0)
                else:
                    a_up = s_init
                    idx = np.linspace(0, len(a_up) - 1, n).astype(int)
                    a_up = a_up[idx]

        for i, a0 in enumerate(inits):
            if time.monotonic() > p1_deadline:
                break
            a_cur = a0.copy()
            # Adam with beta annealing
            n_adam = max(500, int((p1_deadline - time.monotonic()) / 0.05))
            n_adam = min(n_adam, 8000)
            sc, a_opt = adam_optimize(a_cur, 30, 5000, n_adam, 0.5, p1_deadline)
            report(sc, a_opt, f"n={n},i={i},adam")
            a_cur = a_opt
            # L-BFGS polish
            if time.monotonic() < p1_deadline - 10:
                sc2, a_pol = lbfgs_polish(a_cur, 500, maxiter=2000, maxfun=30000)
                report(sc2, a_pol, f"n={n},i={i},lbfgs")
                a_cur = a_pol

    print(f"\nAfter Phase 1: best = {best_score:.8f}")

    # Phase 2: LP direction (20% of time)
    p2_deadline = t_start + 0.65 * (total_budget - 20.0)
    print(f"\n{'='*60}\nPhase 2: LP direction\n{'='*60}")

    if HAVE_CVXPY and len(best_a) <= 5000:
        a_lp = best_a.copy()
        for it_lp in range(20):
            if time.monotonic() > p2_deadline:
                break
            sc, a_new = lp_step(a_lp, max_wall_s=10.0)
            if sc < best_score:
                report(sc, a_new, f"LP {it_lp}")
            a_lp = a_new
            if time.monotonic() < p2_deadline - 5:
                sc2, a_pol = lbfgs_polish(a_lp, 500, maxiter=2000, maxfun=30000)
                report(sc2, a_pol, f"LP+LBFGS {it_lp}")
                a_lp = a_pol
    else:
        print("  Skipping LP")
        a_tk = best_a.copy()
        for it_tk in range(5):
            if time.monotonic() > p2_deadline:
                break
            sc, a_opt = lbfgs_polish(a_tk, 300, maxiter=3000, maxfun=50000)
            report(sc, a_opt, f"lbfgs_iter {it_tk}")
            a_tk = a_opt

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # Phase 3: Local search with diverse strategies (20% of time)
    p3_deadline = t_start + 0.85 * (total_budget - 20.0)
    print(f"\n{'='*60}\nPhase 3: Local Search\n{'='*60}")

    n_best = len(best_a)
    trial = 0
    while time.monotonic() < p3_deadline and trial < 40:
        strategy = trial % 6
        if strategy == 0:
            scale = 0.02 + 0.06 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.05 * rng.random())
        elif strategy == 2:
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            block_sz = max(10, n_best // 20)
            b1 = int(rng.integers(0, n_best - 2 * block_sz))
            b2 = int(rng.integers(block_sz, n_best - block_sz))
            a_pert = best_a.copy()
            a_pert[b1:b1 + block_sz], a_pert[b2:b2 + block_sz] = \
                a_pert[b2:b2 + block_sz].copy(), a_pert[b1:b1 + block_sz].copy()
        elif strategy == 4:
            n_sparse = max(10, n_best // 50)
            a_pert = best_a.copy()
            idx = rng.integers(0, n_best, size=n_sparse)
            a_pert[idx] *= (1.0 + 0.3 * rng.standard_normal(n_sparse))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_struct = 0.5 + 0.3 * np.cos(2.0 * np.pi * rng.random() * 5 * x + rng.random() * 6.28)
            a_pert = 0.7 * best_a + 0.3 * a_struct * (best_a.sum() / max(a_struct.sum(), 1e-10))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        sc, a_opt = lbfgs_polish(a_pert, 800, maxiter=3000, maxfun=50000)
        report(sc, a_opt, f"local#{trial} s={strategy}")
        trial += 1

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # Phase 4: Final refinement (15% of time)
    print(f"\n{'='*60}\nPhase 4: Final Refinement\n{'='*60}")

    if time.monotonic() < deadline - 10:
        sc, a_opt = adam_optimize(best_a, 500, 20000, 3000, 0.1, deadline - 5)
        report(sc, a_opt, "adam_final")

    if time.monotonic() < deadline - 5:
        sc, a_opt = lbfgs_polish(best_a, 2000, maxiter=10000, maxfun=100000)
        report(sc, a_opt, "lbfgs_final")

    if time.monotonic() < deadline - 3:
        sc, a_pol = subgrad_polish(best_a, total_iters=500, deadline=deadline - 1)
        report(sc, a_pol, "subgrad")

    # Final
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
