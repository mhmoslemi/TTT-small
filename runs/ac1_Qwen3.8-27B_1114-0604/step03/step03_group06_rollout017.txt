def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()

    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def build_convol_matrix_sparse(a):
    """Build sparse convolution matrix A where (A@b)[k] = (a*b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        length = k_end - k_start
        rows.extend(range(k_start, k_end))
        cols.extend([j] * length)
        vals.extend(a[:length])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def solve_lp(a, max_wall_s=30.0):

    if M < 1e-12:
        return a.copy()

    constraints = [A @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    
    try:

        except Exception:
            return None

        return None
    if b_var.value is None:
        return None
    b_opt = np.clip(b_var.value, 0.0, 1000.0)

    return b_opt

"""
LP-iterative optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses sparse LP (CVXPY/PDLP) for direction finding with adaptive step sizes,
reduced-constraint LP for speed, L-BFGS-B polishing, and perturbations to escape local minima.
Key improvements: many more LP iterations, reduced LP (top active lags), adaptive line search
from t=1, periodic full replacement, and multi-scale exploration.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy import sparse
import cvxpy as cp


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


class LSEObj:
    """Callable class for L-BFGS-B objective (avoids closures/lambdas)."""
    def __init__(self, beta):
        self.beta = beta
    def __call__(self, x):
        return lse_score_grad(x, self.beta)


def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix A where (A@b)[k] = (a*b)[k]."""
    n = len(a)
    rows_list = []
    cols_list = []
    vals_list = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        length = k_end - k_start
        rows_list.extend(range(k_start, k_end))
        cols_list.extend([j] * length)
        vals_list.extend(a[:length])
    A = sparse.csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n))
    return A


def build_conv_matrix_reduced(a, num_active):
    """Build sparse convolution matrix using only top num_active lags."""
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    # Select lags where c[k] > 0.7 * c_max
    threshold = 0.7 * c_max
    active_lags = np.where(c >= threshold)[0]
    # Ensure we include the maximum lag
    k_star = int(np.argmax(c))
    if k_star not in active_lags:
        active_lags = np.append(active_lags, k_star)
    # Limit to num_active largest
    if len(active_lags) > num_active:
        c_active = c[active_lags]
        top_idx = np.argsort(c_active)[::-1][:num_active]
        active_lags = active_lags[top_idx]
    # Build sparse matrix with only these rows
    rows_list = []
    cols_list = []
    vals_list = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        length = k_end - k_start
        rows_list.extend(range(k_start, k_end))
        cols_list.extend([j] * length)
        vals_list.extend(a[:length])
    A_full = sparse.csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n))
    A_red = A_full[active_lags, :].tocsr()
    return A_red, c[active_lags]


def solve_lp_full(a, time_budget=30.0):
    """Solve full LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k."""
    n = len(a)
    t0 = time.monotonic()
    if n > 12000:
        return None
    c = fast_conv(a)
    M = float(c.max())
    if M < 1e-12:
        return a.copy()
    A = build_conv_matrix_sparse(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    try:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 2:
            return None
        prob.solve(solver=cp.PDLP, max_time=remaining)
    except Exception:
        try:
            remaining = time_budget - (time.monotonic() - t0)
            if remaining > 2:
                prob.solve(solver=cp.ECOS, max_time=remaining)
        except Exception:
            return None
    if b_var.value is None:
        return None
    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None
    return b_opt


def solve_lp_reduced(a, num_active=3000, time_budget=30.0):

def solve_lp_adaptive(a, time_budget=30.0):
    """Solve LP, trying reduced first for speed, falling back to full."""
    n = len(a)
    t0 = time.monotonic()
    # For smaller n, use full LP; for larger, try reduced first

    # Try reduced LP

    # Fall back to full LP
    remaining = time_budget - (time.monotonic() - t0)
    if remaining > 5:
        return solve_lp_full(a, remaining)
    return None


def lp_step_with_linesearch(a, time_budget=30.0):
    """LP step with adaptive line search from t=1 downward."""
    old_sc = compute_score(a)
    b_opt = solve_lp_adaptive(a, time_budget=time_budget)
    if b_opt is None:
        return old_sc, a
    # Normalize to same sum
    S = float(a.sum())
    if b_opt.sum() < 1e-12:
        return old_sc, a
    b_opt = b_opt * (S / b_opt.sum())
    # Line search
    best_sc = old_sc
    best_a = a.copy()
    for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1]:
        a_new = (1.0 - t) * a + t * b_opt
        a_new = np.clip(a_new, 0.0, 1000.0)
        if a_new.sum() < 0.01:
            continue
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new
    return best_sc, best_a


def lbfgs_polish(a, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B polish with LSE objective."""
    n = len(a)
    a0 = np.clip(a.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    obj = LSEObj(beta)
    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out

        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = min(0.2 / (gn + 1e-10), 0.3)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def multi_peak_perturb(a, rng, n_peaks=8, strength=0.05):
    """Add multi-peak perturbation to escape local minima."""
    n = len(a)
    a_pert = a.copy()
    peak_width = max(1, n // (n_peaks * 4))
    for i in range(n_peaks):
        center = int((i + 0.5) * n / n_peaks + rng.integers(-peak_width, peak_width + 1))
        center = max(0, min(n - 1, center))
        lo = max(0, center - peak_width)
        hi = min(n, center + peak_width + 1)
        x = np.arange(lo, hi) / n
        bump = strength * np.exp(-20.0 * (x - center / n) ** 2)
        a_pert[lo:hi] += bump * a[lo:hi].mean()
    return np.clip(a_pert, 0.0, 1000.0)


def coarse_to_fine(parent, n_target, rng):
    """Downsample parent, add structure, upsample to n_target."""
    n_parent = len(parent)
    n_coarse = max(256, n_target // 16)
    # Downsample
    idx = np.linspace(0, n_parent - 1, n_coarse).astype(int)
    a_coarse = parent[idx].copy()
    # Add random structure
    a_coarse = a_coarse * (1.0 + 0.1 * rng.standard_normal(n_coarse))
    a_coarse = np.clip(a_coarse, 0.01, 1000.0)
    # Upsample
    idx2 = np.linspace(0, n_coarse - 1, n_target).astype(int)
    a_fine = a_coarse[idx2].copy()
    return a_fine


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses iterative LP with reduced constraints, L-BFGS-B polishing, and perturbations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    print(f"Parent: {'yes' if parent is not None else 'no'}, len={parent.size if parent is not None else 0}")

    # Initialize
    if parent is not None:
        a = parent.copy()
    else:
        n = 4096
        x = np.arange(n, dtype=np.float64) / n
        a = 0.3 + 0.4 * np.cos(np.pi * x) ** 2 + 0.05 * rng.standard_normal(n)
        a = np.clip(a, 0.01, 1000.0)

    sc = compute_score(a)
    report(sc, a, f"start (n={len(a)})")

    # === Phase 1: L-BFGS-B polishing (20% of time) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: L-BFGS-B polishing")
    print(f"{'='*60}")
    phase1_end = t_start + 0.20 * (budget_s - 25.0)

    for beta in [50, 100, 200, 500, 1000]:
        if time.monotonic() > phase1_end:
            break
        remaining_frac = max(0.1, (phase1_end - time.monotonic()) / (phase1_end - t_start))
        maxiter = int(3000 * remaining_frac) + 500
        sc, a_opt = lbfgs_polish(a, beta, maxiter=maxiter, maxfun=maxiter * 15)
        report(sc, a_opt, f"LBFGS beta={beta}")
        a = a_opt

    print(f"After Phase 1: best = {best_score:.8f}")

    # === Phase 2: Iterative LP (60% of time) ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Iterative LP optimization")
    print(f"{'='*60}")
    phase2_end = t_start + 0.80 * (budget_s - 25.0)

    n_lp = len(a)
    # Time budget per LP based on size
    if n_lp <= 4000:
        lp_time = 15.0
    elif n_lp <= 7000:
        lp_time = 25.0
    elif n_lp <= 10000:
        lp_time = 35.0
    else:
        lp_time = 50.0

    it = 0
    stagnant = 0
    last_improve_it = 0

    while time.monotonic() < phase2_end - 30:
        it += 1
        # Periodic L-BFGS polish every 15 iterations
        if it % 15 == 0 and time.monotonic() < phase2_end - 60:
            sc, a_opt = lbfgs_polish(a, 800, maxiter=2000, maxfun=30000)
            if sc < best_score:
                report(sc, a_opt, f"LP cycle {it//15} polish")
                a = a_opt
                last_improve_it = it
                stagnant = 0
            # Continue with LP

        # LP step
        sc, a_new = lp_step_with_linesearch(a, time_budget=lp_time)
        if sc < best_score:
            report(sc, a_new, f"LP iter {it}")
            a = a_new
            stagnant = 0
            last_improve_it = it
        else:
            stagnant += 1
            # After several stagnant iterations, perturb
            if stagnant >= 4:
                pert_type = stagnant % 3
                if pert_type == 0:
                    scale = 0.03 + 0.04 * rng.random()
                    a = a * (1.0 + scale * rng.standard_normal(n_lp))
                    a = np.clip(a, 0.01, 1000.0)
                elif pert_type == 1:
                    a = multi_peak_perturb(a, rng, n_peaks=6, strength=0.03)
                else:
                    # Targeted: increase elements that contribute least to peak
                    c = fast_conv(a)
                    k_star = int(np.argmax(c))
                    j_start = max(0, k_star - n_lp + 1)
                    j_end = min(k_star, n_lp - 1)
                    contrib = np.zeros(n_lp)
                    if j_end >= j_start:
                        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
                    # Increase lowest contributors
                    bottom_frac = n_lp // 10
                    bottom_idx = np.argsort(contrib)[:bottom_frac]
                    top_idx = np.argsort(contrib)[-bottom_frac:]
                    transfer = a[top_idx].sum() * 0.1
                    a[top_idx] -= transfer / len(top_idx)
                    a[bottom_idx] += transfer / len(bottom_idx)
                    a = np.clip(a, 0.01, 1000.0)
                stagnant = 0
                # Quick check if perturbation helped
                sc_p = compute_score(a)
                if sc_p < best_score:
                    report(sc_p, a, f"perturb iter {it}")

        # Every 40 iterations, try full replacement
        if it % 40 == 0 and n_lp <= 8000 and time.monotonic() < phase2_end - 60:
            b_opt = solve_lp_adaptive(a, time_budget=lp_time)
            if b_opt is not None:
                S = float(a.sum())
                if b_opt.sum() > 1e-10:
                    b_opt = b_opt * (S / b_opt.sum())
                    sc_full = compute_score(b_opt)
                    if sc_full < best_score:
                        report(sc_full, b_opt, f"LP full replace iter {it}")
                        a = b_opt
                        stagnant = 0
                        last_improve_it = it

    print(f"After Phase 2: best = {best_score:.8f}")

    # === Phase 3: Coarse-to-fine exploration (10% of time) ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Coarse-to-fine exploration")
    print(f"{'='*60}")
    phase3_end = t_start + 0.90 * (budget_s - 25.0)

    if parent is not None and time.monotonic() < phase3_end:
        # Try a few different perturbations of the best solution
        for trial in range(5):
            if time.monotonic() > phase3_end:
                break
            a_c2f = coarse_to_fine(best_a, len(best_a), rng)
            sc, a_opt = lbfgs_polish(a_c2f, 500, maxiter=2000, maxfun=30000)
            report(sc, a_opt, f"coarse-to-fine #{trial}")
            if sc < best_score:
                # Continue from this new best with one LP step
                if time.monotonic() < phase3_end - 30:
                    sc2, a2 = lp_step_with_linesearch(a_opt, time_budget=20.0)
                    report(sc2, a2, f"coarse-to-fine #{trial} + LP")
                    a = a2
                else:
                    a = a_opt
            else:
                a = a_opt

    print(f"After Phase 3: best = {best_score:.8f}")

    # === Phase 4: Final refinement (10% of time) ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Final refinement")
    print(f"{'='*60}")

    # High-beta LSE polish
    for beta in [2000, 5000]:
        if time.monotonic() > deadline - 15:
            break
        sc, a_opt = lbfgs_polish(a, beta, maxiter=4000, maxfun=60000)
        report(sc, a_opt, f"final LBFGS beta={beta}")
        a = a_opt

    # Subgradient polish
    if time.monotonic() < deadline - 10:
        sc, a_pol = subgradient_polish(a, n_iters=500)
        report(sc, a_pol, "subgrad polish")
        a = a_pol

    # One more LP
    if time.monotonic() < deadline - 30 and len(a) <= 8000:
        sc, a_new = lp_step_with_linesearch(a, time_budget=25.0)
        report(sc, a_new, "final LP")
        a = a_new

    # Final L-BFGS
    if time.monotonic() < deadline - 5:
        sc, a_opt = lbfgs_polish(a, 3000, maxiter=3000, maxfun=50000)
        report(sc, a_opt, "final final LBFGS")
        a = a_opt

    # === Final verification ===
    out = [float(max(0.0, min(1000.0, x))) for x in a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
