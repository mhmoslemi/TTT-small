"""
Advanced optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Strategy: Aggressive LSE-continuation L-BFGS + efficient coordinate descent +
subgradient polish + multi-restart local search. Key improvements over previous:

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and gradient w.r.t. a.
    
    Key: dL/da[k] = 2 * sum_i a[i] * p[i+k] where p = softmax(beta*(c-max(c)))
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerically stable logsumexp
    max_scaled = scaled.max()
    exp_s = np.exp(scaled - max_scaled)
    log_sum_exp = max_scaled + np.log(exp_s.sum())
    L = c_max + log_sum_exp / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    
    # Gradient: dL/da[k] = 2 * sum_{i=0}^{n-1} a[i] * p[i+k]
    # Compute via FFT: conv(a_rev, p) evaluated at indices [n-1, 2n-2]
    rev_a = a[::-1]
    M2 = len(rev_a) + len(p) - 1  # n + (2n-1) - 1 = 3n-2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]  # length n
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_wrapper(a, beta):
    """Scipy-compatible objective returning (value, gradient)."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective with non-negativity bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        a_out = a0
        exact_sc = compute_score(a_out)
        return exact_sc, a_out, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.success

            else:
                w = a[:min(n, j_end)]

            #   = 0 to min(2n-2-k, n-1)  [since k>=0]
            
            idx_start = max(0, j_start - k)  # = 0 always since j_start >= k
            idx_end = min(n - 1, j_end - k)

def subgrad_polish(a, max_iters=5000):
    """Exact subgradient descent with Polyak step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient: dc[j]/da[k] = 2*a[j-k] for j=k_star
        # So sg[k] = 2*a[k_star - k] if 0 <= k_star-k < n, else 0
        sg = np.zeros(n)
        idx = k_star - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        sg[valid] = 2.0 * a[idx[valid]]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - sc_lb) / gn2
        step = min(step, 0.5 / (1.0 + 0.01 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else x

    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())
        # Perturbed parent
        inits.append(np.clip(parent * (1.0 + 0.05 * rng.standard_normal(n)), 0.0, 1000.0))
        inits.append(np.clip(parent * (1.0 + 0.15 * rng.standard_normal(n)), 0.0, 1000.0))

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Linear ramp (anti-palindromic)
    inits.append(np.linspace(0.05, 1.0, n))
    inits.append(np.linspace(1.0, 0.05, n))

    # Anti-palindromic sinusoidal
    for freq in [1, 2, 3]:
        inits.append(np.clip(0.5 + 0.3 * np.sin(np.pi * freq * xn), 0.01, 1000.0))
        inits.append(np.clip(0.5 + 0.3 * np.cos(np.pi * freq * xn), 0.01, 1000.0))

    # Smoothed random
    a_rand = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0)
    ksize = max(1, n // 100)
    kernel = np.ones(ksize) / ksize
    a_smooth = np.convolve(a_rand, kernel, mode='same')
    inits.append(np.clip(a_smooth, 0.01, 1000.0))

    # Two-block
    half = n // 2
    inits.append(np.concatenate([np.ones(half) * 0.7, np.ones(n - half) * 0.3]))
    inits.append(np.concatenate([np.ones(half) * 0.3, np.ones(n - half) * 0.7]))

    return inits

        other = k_star - j
        if 0 <= other < n:
            contrib[j] = a[j] * a[other]
    
    top_count = max(5, n // 100)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:max(1, n - top_count)]
    total_red = 0.0
    for j in top_idx:
        amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - amt, 0.0)
        total_red += amt
    if len(bottom_idx) > 0 and total_red > 0:
        per = total_red / len(bottom_idx)
        a_pert[bottom_idx] += per
    
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Quick multi-start with LSE-continuation
    Phase 2: Aggressive LSE-continuation with fine beta schedule
    Phase 3: Coordinate descent
    Phase 4: Subgradient + perturbation local search
    Phase 5: Final high-beta polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine working lengths
    lengths = [6962]
    if parent is not None:
        if parent.size not in lengths:
            lengths.insert(0, parent.size)
    lengths = list(dict.fromkeys(lengths))  # deduplicate preserving order
    lengths = lengths[:3]

    main_n = lengths[0]
    print(f"Working lengths: {lengths}, main n={main_n}")
    print(f"Parent available: {parent is not None}")

    # ============================================================
    # Phase 1: Quick multi-start (15% of budget)
    # ============================================================
    phase1_end = t_start + 0.12 * total_budget
    print(f"\n=== Phase 1: Quick Multi-Start (deadline t={phase1_end-t_start:.0f}s) ===")

    beta_schedule_p1 = [50, 200, 800]

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  n={n}")
        inits = build_inits(n, rng, parent if (parent is not None and len(parent) == n) else None)
        time_left = phase1_end - time.monotonic()
        n_inits = min(len(inits), max(3, int(time_left / 25)))

        for i in range(min(n_inits, 8)):
            if time.monotonic() > phase1_end:
                break
            a_cur = inits[i].copy()
            for beta in beta_schedule_p1:
                if time.monotonic() > phase1_end:
                    break
                sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=2000, maxfun=40000)
                report(sc, a_opt, f"P1 n={n} i={i} b={beta}")
                a_cur = a_opt

    if best_a is None:
        best_a = np.ones(main_n) * 0.5
        best_score = compute_score(best_a)

    print(f"  After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Aggressive LSE-continuation (35% of budget)
    # ============================================================
    phase2_end = t_start + 0.47 * total_budget
    print(f"\n=== Phase 2: Aggressive LSE-Continuation ===")

    # Fine beta schedule
    beta_schedule_p2 = [50, 100, 200, 400, 800, 1600, 3200, 6400, 12800, 25600]

    n_work = len(best_a)
    a_cur = best_a.copy()
    
    # Run from best
    for beta in beta_schedule_p2:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        maxit = max(500, min(10000, int(remaining / (0.001 * n_work))))
        sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxit * 20)
        report(sc, a_opt, f"P2 best b={beta}")
        a_cur = a_opt

    # Run from second-best initialization (ramp)
    if time.monotonic() < phase2_end - 60:
        a_ramp = np.linspace(0.05, 1.0, n_work)
        a_cur2 = a_ramp.copy()
        for beta in [200, 800, 3200, 12800]:
            if time.monotonic() > phase2_end:
                break
            sc, a_opt, _ = run_lbfgs(a_cur2, beta, maxiter=3000, maxfun=60000)
            report(sc, a_opt, f"P2 ramp b={beta}")
            a_cur2 = a_opt

    print(f"  After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Coordinate Descent (20% of budget)
    # ============================================================
    phase3_end = t_start + 0.67 * total_budget
    print(f"\n=== Phase 3: Coordinate Descent ===")

    if time.monotonic() < phase3_end:
        n_cd = max(20, int((phase3_end - time.monotonic()) / 2.0))
        print(f"  Running ~{n_cd} CD sweeps")
        sc, a_cd = coord_descent_sweep(best_a, n_sweeps=n_cd, rng=rng)
        report(sc, a_cd, "CD")

        # Another round with LSE refinement
        if time.monotonic() < phase3_end:
            sc, a_ref, _ = run_lbfgs(a_cd, 5000, maxiter=3000, maxfun=60000)
            report(sc, a_ref, "CD+refine")
            if sc < best_score:
                sc2, a_cd2 = coord_descent_sweep(a_ref, n_sweeps=20, rng=rng)
                report(sc2, a_cd2, "CD+refine+CD")

    print(f"  After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Subgradient + Perturbation (20% of budget)
    # ============================================================
    phase4_end = t_start + 0.87 * total_budget
    print(f"\n=== Phase 4: Subgradient + Local Search ===")

    # Subgradient polish
    if time.monotonic() < phase4_end - 60:
        n_iters = max(500, int((phase4_end - 60 - time.monotonic()) * 50))
        sc, a_sg = subgrad_polish(best_a, max_iters=n_iters)
        report(sc, a_sg, "subgrad")

    # Perturbation-based local search
    trial = 0
    while time.monotonic() < phase4_end - 10 and trial < 20:
        trial += 1
        strat = trial % 5
        if strat == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(len(best_a)))
        elif strat == 1:
            # Targeted perturbation
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.05 * rng.random())
        elif strat == 2:
            # Anti-palindromic correction
            a_pert = best_a.copy()
            n_a = len(a_pert)
            for j in range(n_a):
                partner = n_a - 1 - j
                avg = (a_pert[j] + a_pert[partner]) / 2.0
                diff = (a_pert[j] - a_pert[partner]) / 2.0
                a_pert[j] = avg - 0.1 * diff
                a_pert[partner] = avg + 0.1 * diff
        elif strat == 3:
            # Frequency-domain smoothing
            n_a = len(a_pert) if 'a_pert' in dir() else len(best_a)
            a_pert = best_a.copy()
            M_fft = 1 << (2 * len(a_pert) - 1 - 1).bit_length()
            A = np.fft.rfft(a_pert, M_fft)
            freqs = np.arange(len(A))
            att = 1.0 - 0.15 * (freqs / len(A)) ** 2
            A *= att
            a_pert = np.real(np.fft.irfft(A, M_fft))[:len(a_pert)]
        else:
            # Random reset of a small fraction
            a_pert = best_a.copy()
            frac = 0.05
            n_reset = max(1, int(frac * len(a_pert)))
            idx = rng.choice(len(a_pert), n_reset, replace=False)
            a_pert[idx] = rng.uniform(0.01, 2.0, n_reset)

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            a_pert = best_a * 0.9

        sc, a_opt, _ = run_lbfgs(a_pert, 5000, maxiter=3000, maxfun=60000)
        report(sc, a_opt, f"P4 #{trial} s={strat}")

        # Quick CD after each good perturbation
        if sc < best_score + 0.001:
            sc2, a_opt2 = coord_descent_sweep(a_opt, n_sweeps=5, rng=rng)
            report(sc2, a_opt2, f"P4 #{trial} CD")

    print(f"  After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final Polish (remaining time)
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")

    if time.monotonic() < deadline - 5:
        # Very high beta L-BFGS
        for beta in [20000, 50000]:
            if time.monotonic() > deadline - 10:
                break
            sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
            report(sc, a_opt, f"P5 beta={beta}")

        # Final subgradient
        if time.monotonic() < deadline - 10:
            sc, a_opt = subgrad_polish(best_a, max_iters=3000)
            report(sc, a_opt, "final PG")

        # Final CD
        if time.monotonic() < deadline - 5:
            sc, a_opt = coord_descent_sweep(best_a, n_sweeps=10, rng=rng)
            report(sc, a_opt, "final CD")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
