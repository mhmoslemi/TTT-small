Multi-phase optimizer with frequency-domain power iteration and adaptive LSE refinement.
Targets flat autocorrelation via spectrum compression, refined by L-BFGS with
adaptively increasing smoothing parameter.
"""

import time


def fast_conv(a):

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

    c = fast_conv(a)

    c_max = float(c.max())
    scaled = beta * (c - c_max)

    L = c_max + log_sum / beta

    S3 = S2 * S

    Compresses |A(k)| toward its mean, then projects back to time domain

        if target_mag < 1e-10:

        ratio = target_mag / (mags + 1e-15)
        ratio = np.power(ratio, gamma)
        ratio[0] = 1.0  # Don't modify DC
        
        A_new = A * ratio
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        
        # Damped update
        a = (1 - damping) * a + damping * a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def iapf_init(n, n_iters=200, gamma=0.7, rng=None):
    """
    Iterative Amplitude and Phase Filtering to generate initial
    non-negative sequence with flat spectrum.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    
    M = 1 << (2 * n - 1).bit_length()
    # Start with random phase
    phases = rng.uniform(0, 2 * np.pi, M)
    phases[0] = 0
    mags = np.ones(M)
    target_mag = 1.0
    
    a = np.ones(n) * 0.5
    for it in range(n_iters):
        A = mags * np.exp(1j * phases)
        a_new = np.fft.ifft(A).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        a = a_new
        
        # Update spectrum
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        mags_new = np.abs(A)
        mags_new[0] = A[0].real  # Keep DC
        target = np.mean(mags_new[mags_new > 1e-10])
        ratio = np.power(target / (mags_new + 1e-15), gamma)
        ratio[0] = 1.0
        mags = mags_new * ratio
        phases = np.angle(A)
    
    return a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 60}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def error_reduction(a, n_iters=200, alpha=0.7):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess

        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.5 + 0.5 * (it / max(1, n_iters // 4))
        A_new = A + damp * d

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def subgradient_polish(a, total_iters=3000):
    """
    Subgradient descent with adaptive step size and momentum.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    
    # Momentum
    v = np.zeros(n)
    mu = 0.9
    
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        
        # Momentum
        v = mu * v - step * grad
        a = a + v
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def peak_targeted_cd(a, n_sweeps=5, n_top=100):
    """Coordinate descent targeting peak contributors with golden-section search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            # Try reducing this element
            for frac in [0.3, 0.5, 0.7]:
                a[j] = a_old_j * frac
                S = a.sum()
                if S < 1e-12:
                    a[j] = a_old_j
                    continue
                a_norm = a / S
                sc = compute_score(a_norm)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    S = a.sum()
                    # Found improvement, don't restore
                    break
            else:
                # No improvement, restore
                a[j] = a_old_j
                S = a.sum()
                if S > 1e-12:
                    a = a / S

    return best_a
