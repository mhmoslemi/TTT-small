def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)

    # (clipped to valid range [0, 2n-2])
    total_nz = n * n  # each of n columns has exactly n entries
    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    data = np.empty(total_nz, dtype=np.float64)
    
    for j in range(n):
        start = j * n
        rows[start:start+n] = np.arange(j, j+n)
        cols[start:start+n] = j
        data[start:start+n] = f
    
    return sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))


def solve_lp_direction(f, M, n_constraints=None):
    """
    Solve LP: max sum(g) s.t. (f*g) <= M, g >= 0.
    If n_constraints is not None, only enforce top n_constraints tightest constraints.
    Returns (g_optimal, optimal_value) or (None, None) on failure.
    """
    n = len(f)
    
    # Build constraint matrix
    A = build_conv_matrix_sparse(f, n)
    
    # Optionally reduce constraints
    if n_constraints is not None and n_constraints < 2*n-1:
        # Compute current (f*g) for g = ones (as a proxy for which constraints are tight)
        # Actually, we want all constraints (f*g) <= M; the "tightest" are those
        # where the left side is closest to M for the current f.
        # For g=ones: (f*ones)[k] = sum(f) for all valid k (with edge effects)
        # Let's just keep all constraints for now if n is small enough
        if n > 5000:
            # Keep only the middle n_constraints rows (where conv is largest)
            center = n - 1
            half = n_constraints // 2
            k_lo = max(0, center - half)
            k_hi = min(2*n-2, center + half)
            A = A[k_lo:k_hi+1, :]
            M_vec = np.full(k_hi - k_lo + 1, M)
        else:
            M_vec = np.full(2*n-1, M)
    else:
        M_vec = np.full(2*n-1, M)
    
    # Set up LP in cvxpy
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0, A @ g <= M_vec]
    
    prob = cp.Problem(objective, constraints)
    
    try:
        # Try multiple solvers
        for solver in [cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if prob.status == 'optimal' and g.value is not None:
                    g_val = np.asarray(g.value).flatten()
                    g_val = np.clip(g_val, 0, 1000)
                    return g_val, float(prob.value)
            except Exception:
                continue
        
        # Fallback: return uniform
        g_fallback = np.ones(n) / n
        return g_fallback, float(n)
    except Exception:
        return np.ones(n) / n, float(n)


def lp_guided_step(f, M, n_constraints=None):
    """
    One LP-guided step: find direction, line search, update.
    Returns (new_f, new_score, old_score).
    """
    n = len(f)
    S = float(f.sum())
    if S < 1e-12:
        return f, np.inf, np.inf
    
    # Normalize f
    f_norm = f / S
    
    # Solve LP
    g, lp_val = solve_lp_direction(f_norm, M / (S * S), n_constraints)
    if g is None:
        return f, compute_score(f), compute_score(f)
    
    # Normalize g to same sum as f_norm
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return f, compute_score(f), compute_score(f)
    g = g / g_sum
    
    # Line search: minimize score of (1-t)*f_norm + t*g for t in [0, 1]
    best_t = 0.0
    best_sc = compute_score(f_norm)
    
    # Sample t values
    t_candidates = np.concatenate([
        np.linspace(0.01, 0.3, 20),
        np.linspace(0.3, 0.7, 20),
        np.linspace(0.7, 0.99, 10)
    ])
    
    c_ff = fast_conv(f_norm)
    c_gg = fast_conv(g)
    # c_fg = convolve(f_norm, g)
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    F = np.fft.rfft(f_norm, M2_fft)
    G = np.fft.rfft(g, M2_fft)
    c_fg = np.fft.irfft(F * G, M2_fft)[:M2]
    
    for t in t_candidates:
        c_new = (1-t)**2 * c_ff + 2*t*(1-t) * c_fg + t**2 * c_gg
        sc = 2.0 * n * float(np.max(c_new))  # sum is 1 after normalization
        if sc < best_sc:
            best_sc = sc
            best_t = t
    
    if best_t < 0.005:
        # No improvement
        return f, best_sc, compute_score(f)
    
    f_new = (1 - best_t) * f_norm + best_t * g
    f_new = np.maximum(f_new, 0.0)
    S_new = float(f_new.sum())
    if S_new < 1e-12:
        return f, best_sc, compute_score(f)
    f_new = f_new / S_new
    
    return f_new, compute_score(f_new), compute_score(f)


def spectral_flatten(a, n_iters=100, alpha_start=0.3, alpha_end=0.1):
    """
    Iteratively flatten |FFT(a)|^2 to reduce peak of auto-correlation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2*n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max(1, n_iters - 1))
        
        A = np.fft.fft(a, M_fft)
        P = np.abs(A) ** 2
        phase = A / np.maximum(np.abs(A), 1e-30)
        
        # Flatten: compress P[1:] towards its mean
        P_mean = float(P[1:].mean())
        P_new = P.copy()
        excess = np.maximum(P[1:] - P_mean, 0.0)
        P_new[1:] -= alpha * excess
        P_new[1:] += alpha * excess.sum() / (M_fft - 1)  # redistribute
        P_new = np.maximum(P_new, 0.0)
        
        # Reconstruct
        A_new = np.sqrt(P_new) * phase
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    
    # Gradient of L w.r.t. a:
    # d(L)/d(a[j]) = sum_k p[k] * d(c[k])/d(a[j])
    # d(c[k])/d(a[j]) = a[k-j] + a[j] (for the self-convolution, with appropriate bounds)
    # Actually for c[k] = sum_i a[i]*a[k-i], d(c[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j] (if 0<=j-k<n)
    # Wait, for self-conv: c[k] = sum_{i+j=k} a[i]*a[j]
    # d(c[k])/d(a[j]) = a[k-j] (from the a[i] term where i=k-j) + a[k-j] (from the a[j] term where it's the second factor)
    # = 2*a[k-j] if 0 <= k-j < n, else a[k-j] appears once
    # Actually no: c[k] = sum_i a[i] * a[k-i]
    # d(c[k])/d(a[j]) = a[k-j] (differentiating a[i] at i=j) + sum over i where k-i=j, i.e., i=k-j: a[k-(k-j)] = a[j]
    # Wait I'm confusing myself. Let me just use: d(c[k])/d(a[j]) = a[k-j] if 0<=k-j<n else 0
    # No that's wrong too. c[k] = sum_{i} a[i]*a[k-i]. 
    # If I differentiate w.r.t. a[j]: the terms where i=j give a[k-j], and the terms where k-i=j (i.e. i=k-j) give a[j].
    # So d(c[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j] (if 0<=k-j<n, i.e., same condition)
    # Wait: the second term: when k-i = j, i.e., i = k-j. The term is a[i]*a[k-i] = a[k-j]*a[j].
    # Differentiating w.r.t. a[j]: this gives a[k-j].
    # And the first term: when i = j. The term is a[j]*a[k-j].
    # Differentiating w.r.t. a[j]: this gives a[k-j].
    # So total: 2*a[k-j] if 0 <= k-j < n.
    # But wait, if j = k-j (i.e., k = 2j), then the same term is counted twice? No, it's fine because we're summing over all i.
    
    # Hmm, actually I need to be more careful. c[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
    # d(c[k])/d(a[j]) = sum over terms involving a[j]:
    #   - Terms where i = j: a[j]*a[k-j], derivative wrt a[j] = a[k-j] (if 0<=k-j<n)
    #   - Terms where k-i = j, i.e., i = k-j: a[k-j]*a[j], derivative wrt a[j] = a[k-j] (if 0<=k-j<n)
    # If j = k-j (k=2j), these are the same term a[j]*a[j], and derivative is 2*a[j].
    # So in general: d(c[k])/d(a[j]) = a[k-j] * (1 if 0<=k-j<n else 0) + a[k-j] * (1 if 0<=k-j<n else 0)
    # = 2*a[k-j] if 0<=k-j<n, else 0.
    
    # So: d(L)/d(a[j]) = sum_k p[k] * 2*a[k-j] * (indicator 0<=k-j<n)
    # This is 2 * (p * a_rev)[j] where a_rev is the reversed a (padded)
    
    # conv_res[j] = sum_k p[k] * a[k-j] = (p * a)[j] (cross-correlation)
    # = sum_k p[k] * a[k-j] for valid k,j
    # This is the cross-correlation of p and a, evaluated at lag j
    # = (full cross-correlation)[j + n - 1] ... let me just compute it via FFT
    
    rev_a = a[n-1::-1]  # reversed a (length n)
    # (p * a_cv)[j] = sum_k p[k] * a[j-k+n-1] ... 
    # Let me just use: conv(p, rev_a) gives sum_k p[k] * rev_a[j-k] = sum_k p[k] * a[n-1-(j-k)]
    # Hmm, this is getting confusing. Let me use the direct FFT approach.
    
    # d(L)/d(a[j]) = 2 * sum_{k: 0<=k<2n-1, 0<=k-j<n} p[k] * a[k-j]
    # Let m = k-j, so k = m+j: sum_{m: 0<=m<n, 0<=m+j<2n-1} p[m+j] * a[m]
    # = sum_m a[m] * p[j+m] (with p padded to length 2n-1, zeros outside)
    # This is the cross-correlation!
    # = (a * p_rev)[j + n - 1] where p_rev = p[::-1]
    # Or equivalently, it's the (n-1+j)-th element of convolve(a, p_rev)
    
    # Let me just compute via FFT:
    M2 = 3 * n - 1  # enough for conv of length n and (2n-1)
    M2_fft = 1 << (M2 - 1).bit_length()
    FA = np.fft.rfft(a, M2_fft)
    FP = np.fft.rfft(p, M2_fft)
    # We want sum_k p[k]*a[k-j] for each j
    # = sum_k p[k]*a_reversed[n-1-k+j] ... no
    # Let me define: (a ⋆ p)[j] = sum_m a[m]*p[j+m] (cross-correlation, with p indexed from 0)
    # This equals sum_m a[m] * p_rev[n-1-(j+m)] ... no
    # Simplest: (a ⋆ p)[j] = (a * p_rev)[n-1+j] where p_rev = p[::-1] and * is cross-correlation
    # Actually: (a ⋆ p)[j] = sum_m a[m] p[m+j] = (p * a)[j] where * is ... 
    
    # OK let me just compute it directly:
    # For each j, grad_L[j] = 2 * sum_{k=0}^{2n-2} p[k] * a[k-j] (where a[k-j]=0 if k-j outside [0,n-1])
    # This is 2 * cross_correlation(p, a)[j]
    # cross_correlation(p, a)[j] = sum_k p[k] * a[k-j]
    # = sum_k p[k] * a_rev[n-1-k+j] ... no
    # Let a_pad be a padded to length 2n-1 with zeros
    # Then cross_correlation[p, a_pad][j] = sum_k p[k] * a_pad[k-j]
    # = sum_k p[k] * a[k-j] (since a_pad[k-j] = a[k-j] if 0<=k-j<n else 0)
    
    # To compute via FFT: cross_correlation(p, a) = ifft(FFT(p) * conj(FFT(a)))
    # But we need p and a to be the same length. Let's use length 2n-1.
    
    p_pad = np.zeros(2*n-1)
    p_pad[:len(p)] = p
    a_pad = np.zeros(2*n-1)
    a_pad[:n] = a
    
    M3 = 1 << (2*(2*n-1) - 1).bit_length()
    PP = np.fft.rfft(p_pad, M3)
    AP = np.fft.rfft(a_pad, M3)
    cross = np.fft.irfft(PP * np.conj(AP), M3)
    # cross[j] = sum_k p_pad[k] * a_pad[k-j] (circular, but we padded enough)
    # Actually: ifft(FFT(p) * conj(FFT(a)))[n] = sum_k p[k]*a[k+n] ... 
    # Let me just use the definition: cross[k] = sum_j p[j] * a[j-k] (with appropriate indexing)
    # This is getting too complicated. Let me use a simpler approach.
    
    # Simple approach: d(L)/d(a[j]) = 2 * sum_{k: k-j in [0,2n-2]} p[k] * a[k-j] * (1 if k-j < n else 0)
    # = 2 * sum_{m: m in [0,n-1]} p[m+j] * a[m]  (substituting m=k-j, need m+j < 2n-1)
    # So for each j: grad_L[j] = 2 * sum_{m=0}^{n-1} p[m+j] * a[m]
    # This is the cross-correlation of p and a!
    # cross[j] = sum_m a[m] * p[m+j]
    # = (a * p)[j] where * denotes cross-correlation with p "advanced" by j
    
    # Using FFT: cross = ifft(FFT(p_pad) * conj(FFT(a_pad)))
    # where p_pad and a_pad are zero-padded to length >= 3n-2
    # Then cross[n-1+j] = sum_m a[m] * p[m+j] (approximately, with correct indexing)
    
    # Let me just compute it the "dumb" way for correctness:
    # Actually for n up to 10000, I can use the FFT approach:
    # Define: h[j] = sum_m a[m] * p[m+j] for j = 0, 1, ..., n-1
    # This is the "upper triangle" of the cross-correlation.
    # cross = np.correlate(p, a, mode='full')  -- but this is O(n^2) for large n
    # Via FFT: cross = ifft(FFT(p, N) * conj(FFT(a, N)))[:2n-1]
    # where N >= 3n-2
    # Then h[j] = cross[n-1+j] for j = 0, ..., n-1
    
    p_full = np.zeros(2*n-1)
    p_full[:2*n-1] = p  # p already has length 2n-1
    a_full = np.zeros(2*n-1)
    a_full[:n] = a
    
    N_fft = 1 << (3*n - 2).bit_length()
    P_fft = np.fft.rfft(p_full, N_fft)
    A_fft = np.fft.rfft(a_full, N_fft)
    cross_full = np.fft.irfft(P_fft * np.conj(A_fft), N_fft)
    # cross_full[k] = sum_j p_full[j] * a_full[j+k] (circular, but padded enough)
    # Actually: ifft(FFT(p) * conj(FFT(a)))[k] = sum_j p[j] * a[j-k] (with circular indexing)
    # With enough padding, for k in range: = sum_j p[j] * a[j-k] (linear)
    # So cross_full[k] = sum_j p_full[j] * a_full[j-k]
    # We want h[j] = sum_m a[m] * p[m+j] = sum_m a[m] * p_full[m+j]
    # Let l = m+j, so m = l-j: h[j] = sum_l a[l-j] * p_full[l] (where a[l-j] = 0 if l-j outside [0,n-1])
    # = sum_l p_full[l] * a_full[l-j] = cross_full[j] (since cross_full[j] = sum_l p_full[l] * a_full[l-j])
    
    # Wait, let me verify: ifft(FFT(p) * conj(FFT(a)))[k] = (p ⋆ a)[k] = sum_j p[j] * a[j-k]
    # (where ⋆ denotes cross-correlation)
    # So cross_full[j] = sum_l p_full[l] * a_full[l-j]
    # And h[j] = sum_m a[m] * p_full[m+j] = sum_m a[m] * p_full[m+j]
    # Let l = m+j: h[j] = sum_l a[l-j] * p_full[l] = sum_l p_full[l] * a_full[l-j] = cross_full[j] ✓
    
    grad_L = 2.0 * cross_full[:n]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-13,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception as e:
        print(f"  L-BFGS failed: {e}")
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def error_reduction(a, n_iters=100, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err
