Step function optimization via soft-equalization + L-BFGS + coordinate search.
Focuses on escaping local minima through diverse exploration and structured updates.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def soft_equalize(a, n_iters=200, beta_start=10, beta_end=500, alpha=0.5, deadline=None):
    """
    Soft-equalization using softmax-weighted dual variables.
    beta controls the concentration: low beta = smooth, high beta = sharp (near hard threshold).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M_fft = 1 << (3 * n - 2).bit_length()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Anneal beta
        frac = it / max(1, n_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        c = fast_conv(a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break
        
        # Soft lambda: softmax over convolution values
        scaled = beta * c / c_max  # normalize so max is at beta
        # Actually, let's use: lambda[k] = exp(beta * c[k] / c_max) / sum(...)
        # This gives more weight to larger c[k]
        scaled = beta * (c - c_max) / c_max  # so that the max has value 0, others are negative
        exp_scaled = np.exp(scaled)
        lambda_v = exp_scaled / exp_scaled.sum()
        
        # Compute pressure: h[j] = 2 * sum_k lambda[k] * a[j+k] (cross-correlation)
        L = np.fft.rfft(lambda_v, M_fft)
        A = np.fft.rfft(a, M_fft)
        cross = np.fft.irfft(L * np.conj(A), M_fft)
        h = 2.0 * cross[:n]
        
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()
        
        # Damped multiplicative update
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)

        if S_new < 1e-12:

            break

        sc = compute_score(a)
        if sc < best_sc:

            best_a = a.copy()
    
    return best_sc, best_a

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:

    def obj_func(x):

        return sc, g

            options=options
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:

    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def coordinate_descent(a, n_sweeps=10, step_size=0.01, deadline=None):
    """
    Coordinate descent: for each coordinate, try increasing and decreasing.
    Only update if the max convolution decreases.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_sc = compute_score(a)
    best_a = a.copy()

"""
Step function optimization via annealed soft-equalization + L-BFGS + disturbance-reoptimize.

Key innovations over previous approach:
  1. Soft-equalization: uses softmax-weighted dual variables instead of hard threshold,
     providing a smoother, more global equalization signal.
  2. Annealed beta: starts with broad (low-beta) equalization, gradually focusing on peaks.
  3. Disturbance-reoptimize cycle: explicitly perturbs the best solution and re-optimizes
     to escape local minima.
  4. Adaptive alpha schedule in equalization for better convergence.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and its gradient.
    Returns (score, grad).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: conv with p
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def soft_equalize(a, n_iters=300, beta_start=5.0, beta_end=300.0, alpha_start=0.5, alpha_end=0.2, deadline=None):
    """
    Soft-equalization using softmax-weighted dual variables with annealed beta.
    
    At each iteration:
      1. Compute convolution c = conv(a, a)
      2. Compute soft lambda via softmax: lambda[k] = exp(beta*(c[k]-max)/max) / sum(...)
      3. Compute pressure h[j] = 2 * cross_correlation(lambda, a)[j]
      4. Multiplicative update: a[j] <- a[j] * (1 - alpha + alpha * h_mean/h[j])
      5. Renormalize to preserve sum
    
    Beta is annealed from beta_start to beta_end (exponentially).
    Alpha is linearly annealed from alpha_start to alpha_end.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (3 * n - 2).bit_length()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Anneal parameters
        frac = it / max(1, n_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        
        c = fast_conv(a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break
        
        # Soft lambda: softmax over (c - max) / max, so max lag gets weight 1/beta_n_lags
        scaled = beta * (c - c_max) / c_max
        # Clip for numerical stability
        scaled = np.clip(scaled, -500.0, 0.0)
        exp_scaled = np.exp(scaled)
        lambda_sum = exp_scaled.sum()
        if lambda_sum < 1e-300:
            lambda_v = np.zeros(len(c))
            lambda_v[int(np.argmax(c))] = 1.0
        else:
            lambda_v = exp_scaled / lambda_sum
        
        # Compute pressure: h[j] = 2 * sum_k lambda[k] * a[j+k]
        L = np.fft.rfft(lambda_v, M_fft)
        A = np.fft.rfft(a, M_fft)
        cross = np.fft.irfft(L * np.conj(A), M_fft)
        h = 2.0 * cross[:n]
        
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()
        
        # Damped multiplicative update
        factor = 1.0 - alpha + alpha * (h_mean / h)
        factor = np.clip(factor, 0.01, 100.0)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)
        
        # Renormalize to preserve sum
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a = a * (S / S_new)
        
        # Periodic score check
        if it % 20 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                if it % 100 == 0:
                    print(f"    soft_eq iter {it}: score={sc:.8f}, beta={beta:.1f}, alpha={alpha:.3f}")
    
    # Final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_sc, best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000, deadline=None):
    """
    Run L-BFGS-B to minimize the LSE-smoothed score.
    Returns (exact_score, result_array).
    """
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    options = {'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 3.0:
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(50, int(remaining * 120)))
        options['maxiter'] = min(maxiter, max(50, int(remaining * 30)))
    
    def obj_func(x):
        return lse_score_grad(x, beta)
    
    try:
        res = minimize(
            obj_func, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options=options
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    
    return compute_score(a_out), a_out


def build_diverse_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        p = parent.astype(np.float64)
        if len(p) != n:
            p = np.interp(np.linspace(0, len(p) - 1, n), np.arange(len(p)), p)
        inits.append(np.clip(p, 0.01, 1000.0).copy())
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Cosine modulations
    x = np.arange(n, dtype=np.float64) / n
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.5 + 0.2 * np.cos(6.0 * np.pi * x) * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.4 * np.sin(3.0 * np.pi * x) ** 2)
    
    # Quadratic
    inits.append(0.1 + 0.8 * (2.0 * x - 1.0) ** 2)
    
    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    
    # Block structure
    for nb in (4, 8, 16):
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
      Phase 1: Annealed soft-equalization from parent + diverse inits
      Phase 2: L-BFGS refinement with increasing beta
      Phase 3: Disturbance-reoptimize cycle to escape local minima
      Phase 4: Final high-beta polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0  # 25s buffer
    
    # Load parent solution
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                print(f"Loaded parent: length={p.size}")
    except Exception:
        parent = None
    
    # Determine working length
    if parent is not None and 4096 <= parent.size <= 16384:
        n = parent.size
    else:
        n = 8192
    
    print(f"Working with n = {n}")
    
    # Initialize best
    if parent is not None:
        if len(parent) == n:
            best_a = parent.copy()
        else:
            best_a = np.interp(np.linspace(0, len(parent) - 1, n), np.arange(len(parent)), parent)
            best_a = np.clip(best_a, 0.0, 1000.0)
    else:
        best_a = np.ones(n) * 0.5
    
    best_score = compute_score(best_a)
    print(f"Initial score: {best_score:.8f}")
    
    # Phase 1: Global search - soft equalization from multiple inits
    print("\n" + "=" * 60)
    print("Phase 1: Global Search via Soft Equalization")
    print("=" * 60)
    phase1_deadline = t_start + 0.25 * budget_s
    
    inits = build_diverse_inits(n, rng, parent)
    # Sort: parent first, then others
    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        print(f"  Init {i}: running soft equalization...")
        t0 = time.monotonic()
        sc, a_eq = soft_equalize(a0, n_iters=200, beta_start=5.0, beta_end=200.0,
                                  alpha_start=0.5, alpha_end=0.2, deadline=phase1_deadline)
        if sc < best_score:
            best_score = sc
            best_a = a_eq.copy()
            print(f"    NEW BEST: {sc:.8f} (init {i}, t={time.monotonic()-t_start:.0f}s)")
        else:
            print(f"    Score: {sc:.8f} (no improvement, t={time.monotonic()-t_start:.0f}s)")
        # Quick L-BFGS polish from equalized solution
        if time.monotonic() < phase1_deadline - 10:
            sc2, a_lp = run_lbfgs(a_eq, 200, maxiter=3000, maxfun=40000, deadline=phase1_deadline)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_lp.copy()
                print(f"    L-BFGS polish: NEW BEST {sc2:.8f} (t={time.monotonic()-t_start:.0f}s)")
    
    print(f"\nAfter Phase 1: best = {best_score:.8f}")
    
    # Phase 2: L-BFGS refinement with increasing beta
    print("\n" + "=" * 60)
    print("Phase 2: L-BFGS Refinement")
    print("=" * 60)
    phase2_deadline = t_start + 0.50 * budget_s
    
    for beta in [100, 200, 400, 800, 1200]:
        if time.monotonic() > phase2_deadline:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=60000, deadline=phase2_deadline)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  L-BFGS beta={beta}: NEW BEST {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        else:
            print(f"  L-BFGS beta={beta}: {sc:.8f} (t={time.monotonic()-t_start:.0f}s}")
    
    # Mid-run: another soft equalization pass
    if time.monotonic() < phase2_deadline - 20:
        sc, a_eq = soft_equalize(best_a, n_iters=200, beta_start=50.0, beta_end=400.0,
                                  alpha_start=0.4, alpha_end=0.15, deadline=phase2_deadline - 10)
        if sc < best_score:
            best_score = sc
            best_a = a_eq.copy()
            print(f"  Mid soft-eq: NEW BEST {sc:.8f} (t={time.monotonic()-t_start:.0f}s}")
    
    print(f"\nAfter Phase 2: best = {best_score:.8f}")
    
    # Phase 3: Disturbance-reoptimize cycle (key for escaping local minima)
    print("\n" + "=" * 60)
    print("Phase 3: Disturbance-Reoptimize")
    print("=" * 60)
    phase3_deadline = t_start + 0.85 * budget_s
    remaining_p3 = phase3_deadline - time.monotonic()
    n_trials = max(10, int(remaining_p3 / 60.0))  # ~60s per trial
    n_trials = min(n_trials, 30)
    print(f"  {n_trials} trials, ~{remaining_p3/n_trials:.0f}s each")
    
    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline:
            break
        trial_deadline = min(phase3_deadline, time.monotonic() + remaining_p3 / n_trials * 1.8)
        
        strategy = trial % 6
        if strategy == 0:
            # Gaussian perturbation (moderate)
            scale = 0.03 + 0.05 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted: reduce peak contributors
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            a_pert = best_a.copy()
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            contrib = np.zeros(n)
            if j_end >= j_start:
                contrib[j_start:j_end + 1] = best_a[j_start:j_end + 1][::-1]
            top_count = max(10, n // 25)
            top = np.argsort(contrib)[::-1][:top_count]
            rest = np.setdiff1d(np.arange(n), top)
            total_red = a_pert[top].sum() * 0.08
            a_pert[top] -= total_red / len(top)
            a_pert[rest] += total_red / len(rest)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Sinusoidal perturbation
            freq = 1.0 + 10.0 * rng.random()
            ph = 2.0 * np.pi * rng.random()
            amp = 0.03 + 0.07 * rng.random()
            x = np.arange(n, dtype=np.float64) / n
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + ph))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Block swap
            a_pert = best_a.copy()
            bs = max(16, n // 100)
            nb = n // bs
            if nb > 2:
                idx = rng.integers(0, nb, size=2)
                if idx[0] != idx[1]:
                    i, j = int(idx[0]), int(idx[1])
                    tmp = a_pert[i*bs:(i+1)*bs].copy()
                    a_pert[i*bs:(i+1)*bs] = a_pert[j*bs:(j+1)*bs]
                    a_pert[j*bs:(j+1)*bs] = tmp
        elif strategy == 4:
            # Short soft-equalize + perturb
            sc_tmp, a_tmp = soft_equalize(best_a, n_iters=40, beta_start=50.0, beta_end=200.0,
                                           alpha_start=0.5, alpha_end=0.3, deadline=trial_deadline)
            a_pert = a_tmp * (1.0 + 0.03 * rng.standard_normal(n))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Strong random restart (less frequent)
            a_pert = np.clip(0.5 + 0.5 * rng.random(n), 0.01, 1000.0)
            # Blend with best
            blend = 0.3 + 0.4 * rng.random()
            a_pert = blend * best_a + (1.0 - blend) * a_pert
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        # Re-optimize from perturbed starting point
        sc, a_opt = run_lbfgs(a_pert, 300, maxiter=3000, maxfun=40000, deadline=trial_deadline)
        
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Trial {trial} (strat={strategy}): NEW BEST {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
            
            # If improved, do additional refinement
            if time.monotonic() < phase3_deadline - 30:
                sc2, a_eq2 = soft_equalize(a_opt, n_iters=50, beta_start=80.0, beta_end=300.0,
                                            alpha_start=0.4, alpha_end=0.15, deadline=phase3_deadline - 10)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = a_eq2.copy()
                    print(f"    + soft_eq: NEW BEST {sc2:.8f}")
                    if time.monotonic() < phase3_deadline - 20:
                        sc3, a_opt3 = run_lbfgs(a_eq2, 600, maxiter=3000, maxfun=50000,
                                                  deadline=phase3_deadline - 10)
                        if sc3 < best_score:
                            best_score = sc3
                            best_a = a_opt3.copy()
                            print(f"    + lbfgs: NEW BEST {sc3:.8f}")
        else:
            print(f"  Trial {trial} (strat={strategy}): {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
    
    print(f"\nAfter Phase 3: best = {best_score:.8f}")
    
    # Phase 4: Final polish
    print("\n" + "=" * 60)
    print("Phase 4: Final Polish")
    print("=" * 60)
    
    for beta in [1000, 2000, 3000, 5000]:
        if time.monotonic() > deadline - 40:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=80000, deadline=deadline - 30)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final L-BFGS beta={beta}: NEW BEST {sc:.8f}")
        else:
            print(f"  Final L-BFGS beta={beta}: {sc:.8f}")
    
    # Final soft equalization
    if time.monotonic() < deadline - 25:
        sc, a_eq = soft_equalize(best_a, n_iters=150, beta_start=100.0, beta_end=600.0,
                                  alpha_start=0.3, alpha_end=0.1, deadline=deadline - 15)
        if sc < best_score:
            best_score = sc
            best_a = a_eq.copy()
            print(f"  Final soft-eq: NEW BEST {sc:.8f}")
    
    # Very final L-BFGS
    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(best_a, 2000, maxiter=5000, maxfun=60000, deadline=deadline - 5)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Very final L-BFGS: NEW BEST {sc:.8f}")
    
    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic()-t_start:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    
    return out
