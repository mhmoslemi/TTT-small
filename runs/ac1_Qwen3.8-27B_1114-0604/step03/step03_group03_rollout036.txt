    n = len(f)

    M = cp.Variable()
    
    # Build constraint matrix A: A[k,i] = f[k-i] if 0 <= k-i < n
    # (f*b)[k] = sum_i A[k,i] * b[i]
    
    constraints = [b >= 0, cp.sum(b) == 1]

    # For each k in 0..2n-2, (f*b)[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} f[k-i]*b[i] <= M

    prob.solve()

"""
Advanced optimizer for minimum peak self-convolution of non-negative sequences.
Key innovations over previous approach:
1. Analytical 1D solver for coordinate descent (exact, no minimize_scalar)
2. LP-dual Frank-Wolfe direction for global moves (fundamentally different from CD)
3. Multi-scale LSE refinement with adaptive beta
4. Structured perturbation using active-set information from the dual
5. Adaptive length exploration
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    r = np.arange(1, n + 1)
    cond = sv - (cv - 1.0) / r > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(np.where(cond)[0])
    mu = (sv[i] - 1.0) / (i + 1)
    return np.maximum(v - mu, 0.0)


def compute_ATy(f, y):
    """Compute v[i] = sum_{k=0}^{n-1} f[k] * y[k+i] for i=0..n-1.
    f has length n, y has length 2n-1. Uses FFT for O(n log n) computation.
    """
    n = len(f)
    f_rev = f[::-1]
    M = 3 * n - 2
    M_fft = 1 << (M - 1).bit_length()
    F = np.fft.rfft(f_rev, M_fft)
    Y = np.fft.rfft(y, M_fft)
    conv = np.fft.irfft(F * Y, M_fft)[:M]
    return conv[n - 1:2 * n - 1]


def lp_dual_frank_wolfe(f, n_iter=2000, tol=1e-10):
    """
    Solve the dual LP: max min_i (A^T y)[i] s.t. y >= 0, sum(y) = 1.
    Uses Frank-Wolfe (conditional gradient) method.
    Returns (y_opt, lambda_star, i_active_set).
    """
    n = len(f)
    m = 2 * n - 1  # number of lags
    # Initialize y uniformly on the simplex
    y = np.ones(m) / m
    # Precompute argmax of f
    f_max_idx = int(np.argmax(f))
    f_max_val = f[f_max_idx]

    prev_gap = np.inf
    for it in range(n_iter):
        # Compute v = A^T y
        v = compute_ATy(f, y)
        i_star = int(np.argmin(v))
        lambda_current = float(v[i_star])

        # Subgradient: A[:, i_star] has f[k-i_star] for k in [i_star, min(m-1, i_star+n-1)]
        # The FW vertex is e_{k_max} where k_max = argmax_k A[k, i_star]
        # A[k, i_star] = f[k-i_star] for valid k, so k_max = i_star + argmax(f)
        k_max = i_star + f_max_idx
        if k_max >= m:
            k_max = m - 1

        # Step size
        t = 2.0 / (it + 2)

        # Update
        y_new = (1.0 - t) * y
        y_new[k_max] += t
        # Project to ensure non-negativity (should already be on simplex but numerical)
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s > 1e-15:
            y_new /= s
        y = y_new

        # Gap computation (dual objective - primal lower bound)
        # Dual obj = min_i (A^T y)[i] = lambda_current
        # Primal lower bound: for any feasible b, M >= (A^T y)^T b = y^T A b >= ... 
        # Actually gap = max_k (f*b_feasible)[k] - min_i (A^T y)[i] for some feasible b
        # Simple gap: v.max() - v.min() as a proxy for progress
        gap = float(v.max() - v.min())
        if gap < tol and it > 10:
            break
        prev_gap = gap

    v_final = compute_ATy(f, y)
    lambda_star = float(v_final.min())
    # Active set: indices where v[i] is close to lambda_star
    threshold = lambda_star + 0.01 * (float(v_final.max()) - lambda_star + 1e-15)
    i_active = np.where(v_final <= threshold)[0]

    return y, lambda_star, i_active


def analytical_coord_1d(a, i, conv, S, n):
    """
    Solve the 1D subproblem for coordinate i analytically.
    Returns the optimal value t for a[i].
    """
    M_conv = 2 * n - 1
    k_lo = i
    k_hi = min(M_conv - 1, i + n - 1)

    # Outside max
    if k_lo > 0 and k_hi < M_conv - 1:
        outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
    elif k_lo > 0:
        outside_max = float(np.max(conv[:k_lo]))
    elif k_hi < M_conv - 1:
        outside_max = float(np.max(conv[k_hi + 1:]))
    else:
        outside_max = 0.0

    inside_len = k_hi - k_lo + 1
    a_i = a[i]
    S_rest = S - a_i

    # Affected convolution values
    ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
    a_vals = a[ks - i]
    is_2i = (ks == 2 * i)

    # base: conv without a[i] contribution
    old_contrib = 2.0 * a_i * a_vals
    old_contrib[is_2i] = a_i * a_i
    base = conv[k_lo:k_hi + 1] - old_contrib

    # alpha: linear coefficient
    alpha = 2.0 * a_vals.copy()
    alpha[is_2i] = 0.0

    # beta: quadratic coefficient (only at k=2i)
    beta_idx = np.where(is_2i)[0]

    t_upper = max(3.0 * a_i + 0.1, 3.0 * S_rest / max(1, n))
    t_upper = min(t_upper, 1000.0)

    # Candidate t values
    candidates = [0.0, t_upper]

    # For each k, compute the optimal t if k is the active one
    # For c_k = 0 (linear): t_k = S_rest - 2*base[k]/alpha[k] (if alpha[k] != 0)
    # For c_k = 1 (quadratic at k=2i): t_k = (2*base - alpha*S_rest) / (2*S_rest - alpha)

    # Linear candidates
    nonzero_alpha = alpha[alpha > 1e-15]
    if len(nonzero_alpha) > 0:
        idx_nonzero = np.where(alpha > 1e-15)[0]
        t_cands_lin = S_rest - 2.0 * base[idx_nonzero] / alpha[idx_nonzero]
        candidates.extend(t_cands_lin.tolist())

    # Quadratic candidate (at k=2i)
    if len(beta_idx) > 0:
        bi = beta_idx[0]
        denom = 2.0 * S_rest - alpha[bi]
        if abs(denom) > 1e-15:
            t_q = (2.0 * base[bi] - alpha[bi] * S_rest) / denom
            candidates.append(t_q)

    # Also add the point where inside max = outside_max (approximate)
    if outside_max > 0:
        # Find t where the max inside value equals outside_max
        # This is approximate; just add a few points
        max_base = float(np.max(base))
        if max_base < outside_max:
            # Solve max_base + t*max_alpha = outside_max for t
            max_alpha_val = float(np.max(alpha))
            if max_alpha_val > 1e-15:
                t_cross = (outside_max - max_base) / max_alpha_val
                candidates.append(t_cross)

    # Filter and evaluate candidates
    best_t = 0.0
    best_h = np.inf
    for t in candidates:
        if t < 0 or t > t_upper:
            continue
        conv_in = base + t * alpha
        if len(beta_idx) > 0:
            conv_in[beta_idx] += t * t
        im = float(np.max(conv_in))
        tot_max = im if im > outside_max else outside_max
        S_new = S_rest + t
        if S_new < 1e-12:
            continue
        h = 2.0 * n * tot_max / (S_new * S_new)
        if h < best_h:
            best_h = h
            best_t = t

    # Refine with a local search around the best candidate
    if 0 < best_t < t_upper:
        for dt in [0.01, -0.01, 0.001, -0.001, 0.1, -0.1]:
            t_try = best_t + dt * (t_upper + 0.1)
            if 0 <= t_try <= t_upper:
                conv_in = base + t_try * alpha
                if len(beta_idx) > 0:
                    conv_in[beta_idx] += t_try * t_try
                im = float(np.max(conv_in))
                tot_max = im if im > outside_max else outside_max
                S_new = S_rest + t_try
                if S_new < 1e-12:
                    continue
                h = 2.0 * n * tot_max / (S_new * S_new)
                if h < best_h:
                    best_h = h
                    best_t = t_try

    return best_t


def coord_descent_sweep_fast(a, conv, S, n, order=None, analytical=True):
    """One sweep of coordinate descent with incremental conv updates."""
    a = a.astype(np.float64).copy()
    conv = conv.astype(np.float64).copy()
    M_conv = 2 * n - 1

    if order is None:
        order = np.arange(n)

    for idx in range(len(order)):
        i = int(order[idx])
        a_i = a[i]

        t_opt = analytical_coord_1d(a, i, conv, S, n)

        # Update a and conv
        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)

        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base = conv[k_lo:k_hi + 1] - old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        conv[k_lo:k_hi + 1] = base + t_opt * alpha
        if np.any(is_2i):
            conv[k_lo:k_hi + 1][is_2i] += t_opt * t_opt

        a[i] = t_opt
        S = S - a_i + t_opt

    return a, conv, S


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def lse_obj(a_flat):
        return lse_score_grad(a_flat, beta)

    res = minimize(
        lse_obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def lp_direction_update(f, n_fw_iter=3000):
    """
    Use LP-dual Frank-Wolfe to get a direction, then form a new sequence.
    Returns the proposed direction (a new sequence b).
    """
    y, lambda_star, i_active = lp_dual_frank_wolfe(f, n_iter=n_fw_iter)

    # The direction: uniform over active set
    if len(i_active) == 0:
        return f.copy()

    b = np.zeros(len(f))
    b[i_active] = 1.0 / len(i_active)

    # Blend with original for stability
    # Actually, return b as the direction; the caller will mix
    return b


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses coordinate descent + LP-dual direction + LSE refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    if parent is not None:
        p_score = compute_score(parent)
        print(f"Parent score: {p_score:.8f}, length: {len(parent)}")
        best_score = p_score
        best_a = parent.copy()
    else:
        print("No parent available, using random init.")

    n = len(best_a)
    print(f"Working length: {n}")

    # ============================================================
    # Phase 1: Coordinate descent sweeps (35% of time)
    # ============================================================
    phase1_deadline = t_start + 0.35 * total_budget
    print(f"\n=== Phase 1: Fast CD ({phase1_deadline - t_start:.0f}s budget) ===")

    a_cur = best_a.copy()
    conv_cur = fast_conv(a_cur)
    S_cur = float(a_cur.sum())
    no_improve = 0
    sweep_count = 0

    while time.monotonic() < phase1_deadline:
        sweep_count += 1

        if sweep_count % 5 == 0:
            # Hot ordering: prioritize coordinates near the peak
            k_star = int(np.argmax(conv_cur))
            contrib = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                contrib[j_start:j_end + 1] = a_cur[j_start:j_end + 1][::-1]
            order = np.argsort(contrib)[::-1]
        elif sweep_count % 3 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        a_cur, conv_cur, S_cur = coord_descent_sweep_fast(a_cur, conv_cur, S_cur, n, order)
        sc = compute_score(a_cur)

        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            no_improve = 0
            if sweep_count <= 5 or sweep_count % 25 == 0:
                print(f"  CD sweep {sweep_count}: {sc:.8f} (NEW BEST)")
        else:
            no_improve += 1

        if sweep_count % 25 == 0:
            print(f"  CD sweep {sweep_count}: {sc:.8f} (no_improve={no_improve})")

        if no_improve >= 30:
            print(f"  Stuck, perturbing...")
            a_cur = targeted_perturb(a_cur, rng, strength=0.02 + 0.02 * rng.random())
            conv_cur = fast_conv(a_cur)
            S_cur = float(a_cur.sum())
            no_improve = 0

    print(f"Phase 1 done: {sweep_count} sweeps, best={best_score:.8f}")

    # ============================================================
    # Phase 2: LP-dual direction (20% of time)
    # ============================================================
    phase2_deadline = t_start + 0.55 * total_budget
    print(f"\n=== Phase 2: LP-Dual Direction ({phase2_deadline - t_start:.0f}s budget) ===")

    for lp_trial in range(15):
        if time.monotonic() > phase2_deadline - 30.0:
            break

        # Get LP direction
        t0 = time.monotonic()
        b_dir = lp_direction_update(best_a, n_fw_iter=5000)
        lp_time = time.monotonic() - t0

        # Mix and optimize
        for mix_t in [0.3, 0.5, 0.7, 1.0]:
            if time.monotonic() > phase2_deadline - 10.0:
                break
            a_mixed = (1.0 - mix_t) * best_a + mix_t * b_dir
            a_mixed = np.clip(a_mixed, 0.0, 1000.0)

            # Quick CD from mixed point
            conv_m = fast_conv(a_mixed)
            S_m = float(a_mixed.sum())
            for sw in range(3):
                if time.monotonic() > phase2_deadline - 5.0:
                    break
                order = rng.permutation(n)
                a_mixed, conv_m, S_m = coord_descent_sweep_fast(a_mixed, conv_m, S_m, n, order)

            sc_m = compute_score(a_mixed)
            if sc_m < best_score:
                best_score = sc_m
                best_a = a_mixed.copy()
                print(f"  LP trial {lp_trial}, mix={mix_t}: {sc_m:.8f} (NEW BEST, {lp_time:.1f}s)")
            elif lp_trial < 3:
                print(f"  LP trial {lp_trial}, mix={mix_t}: {sc_m:.8f} ({lp_time:.1f}s)")

    print(f"Phase 2 done: best={best_score:.8f}")

    # ============================================================
    # Phase 3: LSE refinement (15% of time)
    # ============================================================
    phase3_deadline = t_start + 0.70 * total_budget
    print(f"\n=== Phase 3: LSE Refinement ({phase3_deadline - t_start:.0f}s budget) ===")

    for beta in [300, 1000, 3000, 8000, 20000]:
        if time.monotonic() > phase3_deadline - 10.0:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=5000, maxfun=80000)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  LSE beta={beta}: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            print(f"  LSE beta={beta}: {sc:.8f} ({elapsed:.1f}s)")

        # Quick CD after each LSE
        if time.monotonic() < phase3_deadline - 15.0:
            a_tmp = best_a.copy()
            conv_tmp = fast_conv(a_tmp)
            S_tmp = float(a_tmp.sum())
            for _ in range(3):
                order = rng.permutation(n)
                a_tmp, conv_tmp, S_tmp = coord_descent_sweep_fast(a_tmp, conv_tmp, S_tmp, n, order)
            sc_tmp = compute_score(a_tmp)
            if sc_tmp < best_score:
                best_score = sc_tmp
                best_a = a_tmp.copy()
                print(f"  CD after LSE beta={beta}: {sc_tmp:.8f} (NEW BEST)")

    # ============================================================
    # Phase 4: Multi-restart perturbation (20% of time)
    # ============================================================
    phase4_deadline = t_start + 0.90 * total_budget
    print(f"\n=== Phase 4: Multi-Restart ({phase4_deadline - t_start:.0f}s budget) ===")

    strategies = ['gaussian', 'targeted', 'block', 'lp_mixed', 'sinusoidal', 'reshape']
    trial = 0
    while time.monotonic() < phase4_deadline - 10.0:
        strategy = strategies[trial % len(strategies)]
        n_cur = len(best_a)

        if strategy == 'gaussian':
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_cur))
        elif strategy == 'targeted':
            a_pert = targeted_perturb(best_a, rng, strength=0.01 + 0.04 * rng.random())
        elif strategy == 'block':
            block_size = max(20, n_cur // 50)
            start = int(rng.integers(0, n_cur - block_size))
            a_pert = best_a.copy()
            saved = a_pert[start:start + block_size].sum()
            a_pert[start:start + block_size] = 0.0
            rest_mask = np.ones(n_cur, dtype=bool)
            rest_mask[start:start + block_size] = False
            a_pert[rest_mask] += saved / max(1, int(rest_mask.sum()))
        elif strategy == 'lp_mixed':
            b_dir = lp_direction_update(best_a, n_fw_iter=2000)
            t_mix = 0.3 + 0.5 * rng.random()
            a_pert = (1.0 - t_mix) * best_a + t_mix * b_dir
        elif strategy == 'sinusoidal':
            freq = 2.0 + 15.0 * rng.random()
            amp = 0.01 + 0.04 * rng.random()
            x = np.arange(n_cur, dtype=np.float64) / n_cur
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + 2.0 * np.pi * rng.random()))
        else:  # reshape
            # Try to reshape towards a different profile
            x = np.arange(n_cur, dtype=np.float64) / n_cur
            shape = 0.5 + 0.5 * np.cos(np.pi * x)  # cosine window
            a_pert = best_a * (1.0 + 0.3 * (shape - shape.mean() / shape.max()))

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # CD from perturbed point
        conv_p = fast_conv(a_pert)
        S_p = float(a_pert.sum())
        max_sweeps = 6
        for sw in range(max_sweeps):
            if time.monotonic() > phase4_deadline - 5.0:
                break
            order = rng.permutation(n)
            a_pert, conv_p, S_p = coord_descent_sweep_fast(a_pert, conv_p, S_p, n, order)

        sc_p = compute_score(a_pert)
        if sc_p < best_score:
            best_score = sc_p
            best_a = a_pert.copy()
            print(f"  P4 trial {trial} ({strategy}): {sc_p:.8f} (NEW BEST)")
        elif trial < 10:
            print(f"  P4 trial {trial} ({strategy}): {sc_p:.8f}")
        trial += 1

    print(f"Phase 4 done: best={best_score:.8f}")

    # ============================================================
    # Phase 5: Final polish (10% of time)
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")
    a_cur = best_a.copy()
    conv_cur = fast_conv(a_cur)
    S_cur = float(a_cur.sum())

    polish_count = 0
    while time.monotonic() < deadline - 5.0:
        polish_count += 1
        if polish_count % 4 == 0:
            k_star = int(np.argmax(conv_cur))
            contrib = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                contrib[j_start:j_end + 1] = a_cur[j_start:j_end + 1][::-1]
            order = np.argsort(contrib)[::-1]
        else:
            order = np.arange(n)

        a_cur, conv_cur, S_cur = coord_descent_sweep_fast(a_cur, conv_cur, S_cur, n, order)
        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

    # One final LSE if time permits
    if time.monotonic() < deadline - 30.0:
        sc, a_opt, _ = run_lbfgs(best_a, 5000, maxiter=3000, maxfun=50000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final LSE: {sc:.8f} (NEW BEST)")
        # Final CD
        a_cur = best_a.copy()
        conv_cur = fast_conv(a_cur)
        S_cur = float(a_cur.sum())
        for _ in range(3):
            if time.monotonic() > deadline - 5.0:
                break
            order = np.arange(n)
            a_cur, conv_cur, S_cur = coord_descent_sweep_fast(a_cur, conv_cur, S_cur, n, order)
        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  Final CD: {sc:.8f} (NEW BEST)")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
