"""
Optimization for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Strategy: Adam with LSE continuation → L-BFGS polish → 1D coordinate descent → diverse restarts.
"""

import numpy as np
from scipy.optimize import minimize
import time

# All helper functions at top level

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and exact gradient (with diagonal correction)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    
    # Compute dL/da using FFT
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_rp = np.fft.irfft(RA * P, M2_fft)
    v = conv_rp[n-1:2*n-1]  # v[m] = sum_j a[j]*p[j+m]
    
    # Diagonal correction: subtract a[m]*p[2m]
    diag_idx = 2 * np.arange(n)
    dL_da = 2.0 * v - a * p[diag_idx]
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L * a / S3)
    
    return score, grad

def adam_optimize(a_init, betas, steps_per_beta, lr_base, time_limit=None, t_start=None, normalize=True):
    """Adam with LSE beta continuation."""
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / len(a)
    if normalize:
        a = a / a.sum()
    
    n = len(a)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for beta in betas:
        for step in range(steps_per_beta):
            t += 1
            if time_limit is not None and t_start is not None:
                if time.monotonic() - t_start > time_limit:
                    return best_score, best_a
            lr = lr_base / (1.0 + 0.01 * t * 1e-4)
            score, grad = lse_score_and_grad(a, beta)
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * grad * grad
            m_hat = m / (1 - b1**t)
            v_hat = v / (1 - b2**t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0.0)
            if normalize:
                s = a.sum()
                if s > 1e-12:
                    a = a / s
        # After each beta phase, check score
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_score, best_a

def lbfgs_optimize(a_init, beta, maxiter=5000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a_init)
    a0 = np.clip(a_init, 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0/n
    bounds = [(0.0, 1000.0)] * n
    
    def fun_grad(x):
        x = np.maximum(x, 0.0)
        sc, g = lse_score_and_grad(x, beta)
        return sc, g
    
    res = minimize(fun_grad, a0, jac=True, method='L-BFGS-B', bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-10, 'maxiter': maxiter, 'maxfun': maxiter*5})
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out

def coord_descent(a, n_sweeps=10, top_k=50, time_limit=None, t_start=None):
    """Selective coordinate descent with 1D line search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_limit) if (time_limit is not None and t_start is not None) else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

                [0.0],
                [a_old * r for r in [0.1, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 2.0]],
                [a_old + grad[j] * s for s in [0.01, 0.05, 0.1, 0.5]]
            ]))
            candidates = np.clip(candidates, 0.0, 1000.0)

                sc = compute_score(a)

                    best_sc = sc
                    best_x = x
            a[j] = best_x
        
        if compute_score(a) < best_score:
            best_score = compute_score(a)
    
    return best_score, a

def resample(a, new_n):
    """Resample sequence to new length."""
    from scipy.interpolate import interp1d
    n = len(a)
    if n == new_n:
        return a.copy()
    f = interp1d(np.arange(n), a, kind='cubic', fill_value='extrapolate')
    x_new = np.linspace(0, n-1, new_n)
    a_new = np.maximum(f(x_new), 0.0)
    return a_new
