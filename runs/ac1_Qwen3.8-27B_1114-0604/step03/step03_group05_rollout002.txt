            # In all cases: coeff[k] = 2*a[k-i] if 0<=k-i<n (including the case k-i==i where it's 2*a[i] but the actual contribution is a[i]^2, which changes by (a_i_new-a[i])*2*a[i] + ... hmm this is getting confusing)
            
            # Let me just do it directly:
            # Remove a[i] from the sequence, compute convolution of the rest
            # No, that's too expensive. Let me use the formula:
            # c_new[k] = c_old[k] + (a_i_new - a_i_old) * (a[k-i] + a[k-i]) 
            # but only for valid k-i, and with the correction for k-i == i.
            
            # Actually, the simplest way:
            # c[k] = sum_{j} a[j] * a[k-j]
            # The terms involving a[i]: j=i gives a[i]*a[k-i], and k-j=i (j=k-i) gives a[k-i]*a[i]
            # If k-i == i, these are the same term: a[i]^2
            # So the total contribution of a[i] to c[k] is:
            #   2*a[i]*a[k-i] if k-i != i and 0<=k-i<n
            #   a[i]^2 if k-i == i (i.e., k = 2i)
            #   0 otherwise
            # When we change a[i] to a_i_new:
            #   new contribution = 2*a_i_new*a[k-i] (if k-i != i) or a_i_new^2 (if k==2i)
            #   change = new - old
            
            # This means the objective is not simply linear in a_i_new (because of the k=2i term).
            # For simplicity, let's handle it numerically.
            
            # For the 1D optimization, I'll just evaluate the full objective for a given a_i.
            # This is O(n) per evaluation (just update the affected c[k] values).
            
            S_other = S - a[i]
            a_i_old = a[i]
            
            def obj(a_i_val):
                # Update the convolution
                # c[k] changes by: (a_i_val - a_i_old) * 2 * a[k-i] for valid k, 
                # except when k=2i, the change is (a_i_val^2 - a_i_old^2) instead of (a_i_val - a_i_old)*2*a[i]
                # Actually (a_i_val^2 - a_i_old^2) = (a_i_val - a_i_old)*(a_i_val + a_i_old), not (a_i_val - a_i_old)*2*a[i]
                # So I need to be careful.
                
                # Let me just compute it from scratch for the affected lags.
                # c[k] for k in [i-(n-1), i+(n-1)] ∩ [0, 2n-2] is affected.
                # Actually all k from max(0, 0) to min(2n-2, 2i + (n-1)) ... 
                # Let me just recompute c for the affected range.
                
                # For efficiency, let's note that c[k] is affected for k where either j=i or k-j=i is in range.
                # j=i: contributes a_i_val * a[k-i] for k-i in [0, n-1], i.e., k in [i, i+n-1]
                # k-j=i: j=k-i, contributes a[k-i] * a_i_val for k-i in [0, n-1], i.e., k in [i, i+n-1]
                # Wait, that's the same range! So k in [i, i+n-1] ∩ [0, 2n-2].
                
                # But when j=i AND k-j=i (i.e., k=2i), the term is a_i_val^2 (counted once).
                # So: c_new[k] = c_old[k] + (a_i_val - a_i_old) * (a[k-i] + a[k-i]) for k in [i, i+n-1], k≠2i
                #            c_new[k] = c_old[k] + (a_i_val^2 - a_i_old^2) for k = 2i (if 2i < 2n-1)
                
                # Hmm wait, I'm confusing myself. Let me think again.
                # c[k] = sum_{j: 0≤j<n, 0≤k-j<n} a[j] * a[k-j]
                # Terms where a[i] appears:
                # 1. j = i: term is a[i] * a[k-i], valid if 0≤k-i<n
                # 2. k-j = i, i.e., j = k-i: term is a[k-i] * a[i], valid if 0≤k-i<n
                # If k-i = i (i.e., k = 2i): both conditions give the same term a[i]*a[i] = a[i]^2
                #   So the total contribution is a[i]^2 (NOT 2*a[i]^2)
                # If k-i ≠ i: contribution is a[i]*a[k-i] + a[k-i]*a[i] = 2*a[i]*a[k-i]
                
                # When we replace a[i] with a_i_val:
                # New contribution at k (k≠2i): 2*a_i_val*a[k-i]
                # Old contribution at k (k≠2i): 2*a_i_old*a[k-i]
                # Change: 2*(a_i_val - a_i_old)*a[k-i]
                
                # New contribution at k=2i: a_i_val^2
                # Old contribution at k=2i: a_i_old^2
                # Change: a_i_val^2 - a_i_old^2
                
                # So the max of c_new is:
                #   max( max_{k≠2i, k in [i, i+n-1]} (c[k] + 2*(a_i_val-a_i_old)*a[k-i]),
                #         (c[2i] + a_i_val^2 - a_i_old^2) if 2i < 2n-1,
                #         max_{k not affected} c[k] )
                
                # For the "not affected" part, the max is just the max of c[k] over the unaffected k.
                # This is a bit expensive to compute each time, but we can precompute it.
                
                # Actually, for the 1D optimization, let me just compute the full max.
                # The affected range is k in [i, min(i+n-1, 2n-2)].
                # Unaffected max: max of c[k] for k not in this range.
                
                # Let me precompute the unaffected max:
                # It's the max of c[0:i] and c[i+n:2n-1] (if i+n ≤ 2n-2)
                
                if a_i_val < 0:
                    a_i_val = 0.0
                
                S_new = S_other + a_i_val
                if S_new < 1e-12:
                    return 1e10
                
                # Compute max of new c
                # Unaffected max
                # Affected range: k from max(0, i) to min(2n-2, i+n-1)
                k_lo = i
                k_hi = min(2*n - 2, i + n - 1)
                
                # Unaffected: k in [0, k_lo) ∪ (k_hi, 2n-2]
                unaff_max = 0.0
                if k_lo > 0:
                    unaff_max = max(unaff_max, float(np.max(c[:k_lo])))
                if k_hi < 2*n - 1:
                    unaff_max = max(unaff_max, float(np.max(c[k_hi+1:])))
                
                # Affected: for each k in [k_lo, k_hi]
                # c_new[k] = c[k] + delta
                # delta = 2*(a_i_val - a_i_old) * a[k-i] for k≠2i
                # delta = a_i_val^2 - a_i_old^2 for k=2i (if in range)
                
                # For efficiency, compute the max over affected k:
                # For k≠2i: c[k] + 2*(a_i_val - a_i_old) * a[k-i]
                # This is linear in a_i_val with slope 2*a[k-i]
                # The max of (c[k] + 2*(a_i_val-a_i_old)*a[k-i]) over k≠2i in range
                # = max_k c[k] + ... no, it's max_k (c[k] + slope_k * (a_i_val - a_i_old))
                
                # This is the max of linear functions, which is convex in a_i_val.
                # I can compute it by evaluating at the argmax points.
                
                # For a given a_i_val, the value at k is: c[k] + 2*(a_i_val-a_i_old)*a[k-i]
                # Let me just compute this for all affected k.
                
                delta_base = a_i_val - a_i_old
                # Vectorized: for k in [k_lo, k_hi], j = k - i, a[j]
                j_lo = 0  # k_lo - i = i - i = 0
                j_hi = k_hi - i  # = min(n-1, 2n-2-i)
                
                affected_vals = c[k_lo:k_hi+1] + 2.0 * delta_base * a[j_lo:j_hi+1]
                
                # Correct for k=2i: the formula gives c[2i] + 2*delta_base*a[i], 
                # but the actual change should be a_i_val^2 - a_i_old^2 = delta_base*(a_i_val + a_i_old)
                # So correction: affected_vals[2i - k_lo] should use delta_base*(a_i_val + a_i_old) instead of 2*delta_base*a[i]
                # Difference: delta_base*(a_i_val + a_i_old) - 2*delta_base*a_i_old = delta_base*(a_i_val - a_i_old) = delta_base^2
                # So: affected_vals[2i-k_lo] += delta_base^2
                
                if 2*i >= k_lo and 2*i <= k_hi:
                    idx_2i = 2*i - k_lo
                    affected_vals = affected_vals.copy()
                    affected_vals[idx_2i] += delta_base * delta_base
                
                aff_max = float(np.max(affected_vals))
                total_max = max(aff_max, unaff_max)
                
                return 2.0 * n * total_max / (S_new * S_new)
            
            # Minimize over a_i_val in [0, 1000]
            res = minimize_scalar(obj, bounds=(0.0, 1000.0), method='bounded',
                                  options={'xatol': 1e-10})
            
            a[i] = max(0.0, min(1000.0, res.x))
        
        # Compute score after sweep
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
        if verbose:
            print(f"    CD sweep {sweep}: score={best_sc:.8f}")
        
        # Convergence check
        if sweep > 0 and abs(prev_sc - best_sc) < 1e-10:
            break
        prev_sc = best_sc
    
    return best_sc, a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lse_obj(a, beta):
    return lse_score_grad(a, beta)

def averaged_subgrad_score_grad(a, top_m=5):
    """Averaged subgradient over top-m peaks."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    
    # Get top-m peaks
    sorted_idx = np.argsort(c)[::-1]
    top_indices = sorted_idx[:top_m]
    
    # Weight by how close to the max (softmax-like)
    weights = np.exp(-0.1 * (c_max - c[top_indices]))
    weights /= weights.sum()
    
    # Average subgradient
    grad = np.zeros(n)
    for w, k_star in zip(weights, top_indices):
        k_star = int(k_star)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            grad[j_start:j_end + 1] += w * 2.0 * a[j_start:j_end + 1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad / S2 - 2.0 * c_max / S3)
    score = 2.0 * n * c_max / (S * S)
    return score, grad

def bb_subgradient_descent(a, iters=2000, top_m=5, verbose=False):
    """Barzilai-Borwein subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    a_prev = a.copy()
    grad_prev = None
    
    for it in range(iters):
        sc, grad = averaged_subgrad_score_grad(a, top_m=top_m)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # BB step size
        if it > 0 and grad_prev is not None:
            s = a - a_prev
            y = grad - grad_prev
            sy = float(np.dot(s, y))
            yy = float(np.dot(y, y))
            if yy > 1e-20:
                if it % 2 == 0:
                    step = sy / yy if sy > 1e-20 else 1.0 / (float(np.linalg.norm(grad)) + 1e-10)
                else:
                    ss = float(np.dot(s, s))
                    step = ss / sy if sy > 1e-20 else 1.0 / (float(np.linalg.norm(grad)) + 1e-10)
            else:
                step = 1.0 / (float(np.linalg.norm(grad)) + 1e-10)
        else:
            step = 1.0 / (float(np.linalg.norm(grad)) + 1e-10)
        
        # Clip step size
        step = max(1e-10, min(step, 10.0 / (float(a.sum()) + 1e-10)))
        
        a_prev = a.copy()
        grad_prev = grad.copy()
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        
        if verbose and it % 200 == 0:
            print(f"    BB iter {it}: score={best_sc:.8f}, step={step:.6e}")
    
    # Final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses: LSE continuation, coordinate descent with exact 1D optimization,
    BB subgradient descent, and multi-start.
    Target: beat 1.5027.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        sc = compute_score(parent)
        report(sc, parent, "parent")
        print(f"Parent: score={sc:.8f}, n={len(parent)}")
    else:
        a0 = np.ones(1024) * 0.5
        sc = compute_score(a0)
        report(sc, a0, "default init")

    n_use = len(best_a)
    print(f"\nWorking with n = {n_use}")
    
    # ===================================================================
    # Phase 1: LSE Continuation (coarse-to-fine)
    # ===================================================================
    phase1_frac = 0.40
    phase1_end = t_start + phase1_frac * (total_budget - 30.0)
    print(f"\n=== Phase 1: LSE Continuation (until t={phase1_end - t_start:.0f}s) ===")
    
    a_cur = best_a.copy()
    beta_schedule = [10, 20, 40, 80, 160, 320, 640, 1280, 2560, 5120, 10240]
    
    for beta in beta_schedule:
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        bounds = [(0.0, 1000.0)] * len(a_cur)
        try:
            res = minimize(lse_obj, a_cur, args=(beta,),
                          method='L-BFGS-B', jac=True, bounds=bounds,
                          options={'ftol': 1e-16, 'gtol': 1e-12,
                                   'maxiter': 2000, 'maxfun': 50000})
            a_cur = np.clip(res.x, 0.0, 1000.0)
        except Exception as e:
            print(f"    L-BFGS-B failed at beta={beta}: {e}")
        
        sc = compute_score(a_cur)
        report(sc, a_cur, f"LSE beta={beta}")
        elapsed = time.monotonic() - t0
        print(f"  beta={beta}: score={sc:.8f}, time={elapsed:.1f}s")
    
    # ===================================================================
    # Phase 2: Coordinate Descent (exact 1D optimization)
    # ===================================================================
    phase2_frac = 0.30
    phase2_end = t_start + (phase1_frac + phase2_frac) * (total_budget - 30.0)
    print(f"\n=== Phase 2: Coordinate Descent (until t={phase2_end - t_start:.0f}s) ===")
    
    a_cur = best_a.copy()
    sweeps_done = 0
    while time.monotonic() < phase2_end:
        # Do one sweep of coordinate descent
        # For efficiency, process coordinates in random order
        order = rng.permutation(n_use)
        sc_before = compute_score(a_cur)
        
        for i in order:
            if time.monotonic() > phase2_end:
                break
            
            c = fast_conv(a_cur)
            S = float(a_cur.sum())
            S_other = S - a_cur[i]
            a_i_old = a_cur[i]
            n_local = n_use
            
            # Precompute unaffected max
            k_lo = i
            k_hi = min(2 * n_local - 2, i + n_local - 1)
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(np.max(c[:k_lo])))
            if k_hi < 2 * n_local - 1:
                unaff_max = max(unaff_max, float(np.max(c[k_hi + 1:])))
            
            # Precompute affected part
            j_hi = k_hi - i
            c_aff = c[k_lo:k_hi + 1].copy()
            a_aff = a_cur[0:j_hi + 1].copy()
            idx_2i = 2 * i - k_lo if (2 * i >= k_lo and 2 * i <= k_hi) else -1
            
            def obj_1d(a_i_val):
                if a_i_val < 0:
                    a_i_val = 0.0
                S_new = S_other + a_i_val
                if S_new < 1e-12:
                    return 1e10
                delta = a_i_val - a_i_old
                aff_vals = c_aff + 2.0 * delta * a_aff
                if idx_2i >= 0:
                    aff_vals = aff_vals.copy()
                    aff_vals[idx_2i] += delta * delta
                aff_max = float(np.max(aff_vals))
                total_max = max(aff_max, unaff_max)
                return 2.0 * n_local * total_max / (S_new * S_new)
            
            res = minimize_scalar(obj_1d, bounds=(0.0, 1000.0), method='bounded',
                                  options={'xatol': 1e-8})
            a_cur[i] = max(0.0, min(1000.0, res.x))
        
        sweeps_done += 1
        sc = compute_score(a_cur)
        report(sc, a_cur, f"CD sweep {sweeps_done}")
        print(f"  CD sweep {sweeps_done}: score={sc:.8f}, improvement={sc_before - sc:.8f}")
        
        # Early stop if no improvement
        if sc_before - sc < 1e-12:
            print("  CD converged, stopping.")
            break

    # ===================================================================
    # Phase 3: BB Subgradient Descent
    # ===================================================================
    phase3_frac = 0.15
    phase3_end = t_start + (phase1_frac + phase2_frac + phase3_frac) * (total_budget - 30.0)
    print(f"\n=== Phase 3: BB Subgradient (until t={phase3_end - t_start:.0f}s) ===")
    
    while time.monotonic() < phase3_end:
        iters = max(200, int((phase3_end - time.monotonic()) * 100))
        iters = min(iters, 5000)
        sc, a_polished = bb_subgradient_descent(best_a, iters=iters, top_m=5)
        report(sc, a_polished, "BB subgrad")
        if sc >= best_score:
            print("  BB not improving, trying with different top_m")
            sc, a_polished = bb_subgradient_descent(best_a, iters=iters//2, top_m=1)
            report(sc, a_polished, "BB subgrad (top1)")
            if sc >= best_score:
                break

    # ===================================================================
    # Phase 4: More Coordinate Descent (finer)
    # ===================================================================
    phase4_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Fine Coordinate Descent (until t={phase4_end - t_start:.0f}s) ===")
    
    a_cur = best_a.copy()
    sweeps_done2 = 0
    while time.monotonic() < phase4_end:
        order = rng.permutation(n_use)
        sc_before = compute_score(a_cur)
        
        for i in order:
            if time.monotonic() > phase4_end:
                break
            c = fast_conv(a_cur)
            S = float(a_cur.sum())
            S_other = S - a_cur[i]
            a_i_old = a_cur[i]
            n_local = n_use
            
            k_lo = i
            k_hi = min(2 * n_local - 2, i + n_local - 1)
            unaff_max = 0.0
