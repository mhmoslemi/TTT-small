This is much harder. The problem is essentially: find a non-negative sequence of length n that minimizes the L-infinity norm of its autocorrelation, relative to the square of its L1 norm.

This is related to the problem of finding sequences with low peak-to-average power ratio (PAPR) in the time domain, or equivalently, flat power spectra.

def spectral_phase_nudge(a, n_iters=20, step_size=0.1):
    """
    Iterative spectral phase rotation to reduce peak autocorrelation.
    At each step: identify peak lag, compute subgradient, apply spectral rotation
    in the direction that reduces the peak, project to non-negative.
    """
    n = len(a)
    a = a.copy()
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of the peak at lag k_star
        # d(conv_k*)/da[j] = 2 * a[k*-j] for valid j
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end+1] = 2.0 * a[j_start:j_end+1][::-1]
        
        # Full gradient (including the 1/S^2 term)
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        # Spectral phase nudge:
        # We want to go in direction -grad, but in the frequency domain
        # to maintain a "global" change rather than local
        A = np.fft.rfft(a, M_fft)
        G = np.fft.rfft(grad, M_fft)
        
        # Rotate the spectrum in the direction that reduces the peak
        # Phase of -G tells us the direction of decrease
        A_new = A - step_size * G
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        # Maintain sum approximately constant
        S_new = float(a_new.sum())
        if S_new > 1e-12:
            a_new = a_new * (S / S_new)
        
        a = a_new
        step_size *= 0.95  # gradual decay
    
    return best_sc, best_a

def subgradient_polish(a, n_iters=500, lower_bound=1.0):

            best_a = a.copy()

        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end+1] = 2.0 * a[j_start:j_end+1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a

"""
Aggressive LSE-continuation + targeted spectral peak reduction + subgradient polish.
Key innovations over previous approach:
1. Fixed gradient computation (proper direct convolution for LSE gradient)
2. Iterative peak-targeted deflation with FFT-based directed updates
3. Peak concentration analysis for alpha continuation
4. Nesterov-accelerated projected gradient as fallback
5. Coarse-to-fine L-BFGS continuation (binary upsampling)

import numpy as np

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_og(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    cmax = float(np.max(c))
    sh = beta * (c - cmax)
    em = np.exp(sh)
    Z = float(em.sum())
    L = cmax + np.log(Z) / beta
    p = em / Z
    # Gradient via direct-to-FFT trick
    M2 = 3 * n - 1
    M2f = 1 << (M2 - 1).bit_length()
    q = np.zeros(M2)
    q[:n] = p[n - 1:]
    dL = np.fft.irfft(np.fft.rfft(a, M2f) * np.fft.rfft(q, M2f), M2f)[:n]
    S2 = S * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n / S2) * dL - (4.0 * n * L / (S2 * S))
    return score, grad

def peak_deflation(a, n_rounds=15, strength=0.06):
    """Reduce peak autocorrelation by deflating contributing elements."""
    n = len(a)
    Mf = 1 << (2 * n - 1).bit_length()
    M2f = 1 << (3 * n - 2).bit_length()
    a = a.copy().astype(np.float64)
    S0 = float(a.sum())
    for _ in range(n_rounds):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(np.max(c))
        ks = int(np.argmax(c))
        js = max(0, ks - n + 1)
        je = min(ks, n - 1)
        w = a[js:je + 1].copy()
        wsum = float(w.sum())
        if wsum < 1e-15:
            break
        # Inverse-difference: peak above fair share
        target = S * S / (2.0 * n - 1.0)
        excess = max(0.0, M - target)
        if excess < 1e-10:
            break
        # Scale w down
        scale = max(0.1, 1.0 - strength * min(excess / (S2_val(S)), 1.0))
        b = a.copy()
        b[js:je + 1] = w * scale
        # Targeted spectral boost to under-represented lags
        A = np.fft.rfft(b, Mf)
        base = np.fft.irfft(A * A, Mf)[:2 * n - 1]
        under = np.maximum(0.0, target - base)
        total_under = float(np.sum(under))
        if total_under > 1e-15:
            m1 = a[js:je + 1].copy()
            ampl = float(np.dot(np.roll(m1, -ks), under)) / total_under
            F = np.fft.rfft(under, Mf)
            d = np.fft.irfft(F * np.conj(A), Mf)[:n]
            b[js:je + 1] = w + strength * ampl * d[js:je + 1]
        b = np.maximum(b, 0.0)
        Sb = float(b.sum())
        if Sb > 1e-12:
            b = b * (S0 / Sb)
        a = b
    return a


def nesterov_projected(a0, fgrad, n_iters=2000):
    n = len(a0)
    a = a0.copy().astype(np.float64)
    y = a.copy()
    best = float('inf')
    best_a = a.copy()
    v = 0.0
    L = 5.0
    for k in range(n_iters):
        val, gr = fgrad(y)
        L = max(L, float(np.abs(gr).max()) + 1.0)
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * v * v)) / 2.0
        v = (1.0 + alpha) / (1.0 + alpha + 1.0)

            if p.size >= 64:
                parent = np.clip(p.astype(np.float64), 0.0, 1000.0)
    except Exception:
        pass

    if parent is None:
        n = 4096
        parent = np.ones(n) * 0.5
        Mf = 1 << (2 * n - 1).bit_length()
        for _ in range(150):
            A = np.fft.fft(parent, Mf)
            mag = np.abs(A)
            ph = np.angle(A)
            med = np.median(mag[1:n])
            nm = mag.copy()
            nm[1:n] = 0.6 * mag[1:n] + 0.4 * med
            nm[n:] = nm[1:n][::-1]
            parent = np.clip(np.real(np.fft.ifft(nm * np.exp(1j * ph)))[:n], 0.0, 1000.0)

    n = len(parent)
    best_score = compute_score(parent)
    best_a = parent.copy()
    t0 + 0

    print(f"Start: n={n}, score={best_score:.8f}")

    def report(sc, arr, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = arr.copy()
            el = time.monotonic() - t0
            print(f"  [{label}] NEW BEST: {sc:.8f} (t={el:.0f}s)")
        return sc < best_score

    # ---- Phase 1: LSE-LBFGS continuation (35%) ----
    d1 = t0 + 0.35 * budget
    print(f"\n--- Phase 1: LSE-LBFGS (to t={d1 - t0:.0f}s) ---")
    betas = [5, 10, 25, 50, 100, 250, 500]
    ac = best_a.copy()
    for b in betas:
        if time.monotonic() > d1:
            break
        rem = d1 - time.monotonic()
        if rem < 5:
            break
        mi = min(5000, max(300, int(rem * 0.8 / max(0.004 * n / 2048 * 15, 0.1))))
        ac = run_lbfgs(ac, b, maxiter=mi, maxfun=min(100000, mi * 25))
        sc = compute_score(ac)
        report(sc, ac, f"P1 b={b}")
    print(f"  P1 done: best={best_score:.8f}")

    # ---- Phase 2: Peak deflation + refine (30%) ----
    d2 = t0 + 0.65 * budget
    print(f"\n--- Phase 2: Peak Deflation (to t={d2 - t0:.0f}s) ---")
    for r in range(8):
        if time.monotonic() > d2:
            break
        rem = d2 - time.monotonic()
        if rem < 5:
            break
        nr = min(20, max(5, int(rem / 3.0)))
        a_def = peak_deflation(best_a, n_rounds=nr, strength=0.05 + 0.02 * rng.random())
        sc = compute_score(a_def)
        report(sc, a_def, f"P2 def r={r}")
        if time.monotonic() < d2 - 3:
            mi = min(3000, max(200, int((d2 - time.monotonic()) * 0.6 / max(0.004 * n / 2048 * 15, 0.1))))
            a_opt = run_lbfgs(a_def, 300, maxiter=mi, maxfun=min(50000, mi * 20))
            sc = compute_score(a_opt)
            report(sc, a_opt, f"P2 ref r={r}")
    print(f"  P2 done: best={best_score:.8f}")

    # ---- Phase 3: High-beta L-BFGS (20%) ----
    d3 = t0 + 0.85 * budget
    print(f"\n--- Phase 3: High-Beta Refine (to t={d3 - t0:.0f}s) ---")
    for b in [500, 1000, 2000, 5000]:
        if time.monotonic() > d3:
            break
        rem = d3 - time.monotonic()
        if rem < 5:
            break
        mi = min(6000, max(500, int(rem * 0.8 / max(0.004 * n / 2048 * 15, 0.1))))
        a_opt = run_lbfgs(best_a, b, maxiter=mi, maxfun=min(120000, mi * 20))
        sc = compute_score(a_opt)
        report(sc, a_opt, f"P3 b={b}")
    print(f"  P3 done: best={best_score:.8f}")

    # ---- Phase 4: Subgradient + Nesterov (10%) ----
    d4 = deadline
    print(f"\n--- Phase 4: Subgrad + Nesterov ---")
    sc, a_opt = subgrad_polish(best_a, n_iters=600, lb=1.0)
    report(sc, a_opt, "P4 subgrad")

    if time.monotonic() < d4 - 15:
        mi = min(4000, max(500, int((d4 - time.monotonic() - 10) * 0.7 / 0.003)))
        sc, a_opt = nesterov_projected(a_opt, lse_og, n_iters=mi)
        a_opt2 = run_lbfgs(a_opt, 2000, maxiter=2000, maxfun=40000)
        sc2 = compute_score(a_opt2)
        report(sc2, a_opt2, "P4 nesterov+lbfgs")

    if time.monotonic() < d4 - 5:
        sc, a_opt = subgrad_polish(best_a, n_iters=400, lb=max(1.0, best_score - 0.005))
        report(sc, a_opt, "P4 subgrad2")

    # ---- Phase 5: Final L-BFGS ----
    if time.monotonic() < deadline - 3:
        a_opt = run_lbfgs(best_a, 3000, maxiter=3000, maxfun=60000)
        sc = compute_score(a_opt)
        report(sc, a_opt, "P5 final")

    # ---- Output ----
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    el = time.monotonic() - t0
    print(f"\n{'='*50}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length={len(out)}, Time={el:.1f}s/{budget_s}s")
    print(f"{'='*50}")
    return out
