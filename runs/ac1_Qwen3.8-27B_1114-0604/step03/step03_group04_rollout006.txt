"""
Improved optimizer for minimum peak autocorrelation of non-negative sequences.
Uses: (1) adaptive spectral flattening, (2) peak-shaving coordinate descent,
(3) L-BFGS with top-k LSE smoothing, (4) structured local search.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def topk_lse_score_grad(a, beta, top_k):
    """Compute smoothed score using top-k log-sum-exp."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    # Get top-k values
    k = min(top_k, len(c))
    topk_idx = np.argpartition(c, -k)[-k:]
    topk_vals = c[topk_idx]
    
    c_max = float(topk_vals.max())
    scaled = beta * (topk_vals - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient: only non-zero at positions that contribute to top-k lags
    p = exp_scaled / exp_scaled.sum()  # weights over top-k lags
    
    # For each top-k lag, the gradient of (a*a)_k w.r.t. a is 2*(shifted a)
    # Weighted sum gives the gradient of the smoothed function
    grad_c = np.zeros(n)
    for idx, pi in zip(topk_idx, p):
        k = int(idx)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            grad_c[j_start:j_end + 1] += 2.0 * pi * seg[::-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * L / S3)
    return score, grad


def spectral_flatten_adaptive(a0, N, n_iters=500, lam_start=0.9, lam_end=0.3, soft_alpha=0.1):
    """Spectral flattening with adaptive lambda and soft non-negativity."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        t = i / max(1, n_iters - 1)
        lam = lam_start + (lam_end - lam_start) * t
        
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: geometric mean of non-DC magnitudes (more aggressive than median)
        non_dc_mag = mag[1:]
        log_mag = np.log(non_dc_mag + 1e-12)
        target = np.exp(np.mean(log_mag))
        
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if (N % 2 == 0 and k == half):
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        
        # Soft non-negativity: use softplus
        if soft_alpha > 0:
            a_new = soft_alpha * np.log1p(np.exp(a_new / soft_alpha))
        else:
            a_new = np.maximum(a_new, 0.0)
        
        a = a_new
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a


def peak_shaving_coordinate_descent(a, n_iters=200, step_decay=0.95):
    """Direct coordinate descent to shave peaks in autocorrelation."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best = a.copy()
    best_sc = compute_score(a)
    
    c = fast_conv(a)
    step = 0.01
    rng = np.random.default_rng(12345)
    
    for it in range(n_iters):
        # Find peak lag
        k_star = int(np.argmax(c))
        M = c[k_star]
        S = float(a.sum())
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
        
        # Find which elements contribute most to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        contrib = np.zeros(n)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        
        # Reduce the element with highest contribution
        # But also check other peaks to avoid creating new ones
        # Use top-3 peaks for robustness
        top3 = np.argpartition(c, -3)[-3:]
        
        total_contrib = np.zeros(n)
        for k in top3:
            gs = max(0, int(k) - n + 1)
            ge = min(int(k), n - 1)
            if ge >= gs:
                total_contrib[gs:ge+1] += a[gs:ge+1][::-1]
        
        # Decrease the top contributor
        i_decrease = int(np.argmax(total_contrib))
        decrement = min(a[i_decrease] * step, S * 0.001)
        a[i_decrease] -= decrement
        
        # Increase a position that doesn't contribute to the top peaks
        # Find positions with low total contribution
        low_positions = np.where(total_contrib < np.median(total_contrib))[0]
        if len(low_positions) > 0:
            i_increase = low_positions[rng.integers(len(low_positions))]
            a[i_increase] += decrement
        
        a = np.clip(a, 0.0, 1000.0)
        
        # Update autocorrelation incrementally (full update for simplicity)
        if it % 5 == 0 or it == n_iters - 1:
            c = fast_conv(a)
        
        step *= step_decay
    
    return best_sc, best


def run_lbfgs_topk(a0, beta, top_k, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj(a):
        return topk_lse_score_grad(a, beta, top_k)
    
    res = minimize(
        obj, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def subgradient_polish(a, total_iters=1000, lb=1.0):
    """Subgradient method with adaptive step size for exact peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    step_size = 0.1
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient at the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        
        # Polyak step size
        step = (sc - lb) / (float(np.dot(grad, grad)) + 1e-30)
        step = min(step, step_size)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        
        # Decay step size
        step_size *= 0.999
        
        # Periodic restart with small perturbation
        if it % 200 == 199:
            a += 0.01 * np.random.default_rng(it).standard_normal(n)
            a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a


def spectral_phase_randomization(a0, N, n_inits=5, rng=None):
    """Generate multiple initializations via random phase in frequency domain."""
    if rng is None:
        rng = np.random.default_rng()
    results = []
    a = a0.astype(np.float64).copy()
    n = len(a)
    S0 = a.sum()
    
    for _ in range(n_inits):
        A = np.fft.fft(a, N)
        # Randomize phase but keep magnitude
        mag = np.abs(A)
        new_phase = rng.uniform(0, 2*np.pi, N)
        # Preserve DC
        new_phase[0] = 0
        # Conjugate symmetry for real output
        new_phase[N//2:] = -new_phase[1:N//2+1][::-1] if N % 2 == 0 else -new_phase[1:N//2][::-1]
        new_A = mag * np.exp(1j * new_phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S > 1e-12:
            a_new = a_new * (S0 / S)
        results.append(a_new)
    
    return results

"""
Improved optimizer combining multi-peak subgradient with momentum,
adaptive spectral flattening, and L-BFGS polishing for minimum
peak autocorrelation of non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def multi_peak_grad(c, a, top_k):
    """Compute gradient as convex combination of top-k peak gradients."""
    n = len(a)
    # Get top-k peaks
    k = min(top_k, len(c))
    topk_idx = np.argpartition(c, -k)[-k:]
    topk_vals = c[topk_idx]
    
    # Weight by softmax of values
    max_val = topk_vals.max()
    weights = np.exp(10 * (topk_vals - max_val))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx, w in zip(topk_idx, weights):
        kk = int(idx)
        j_start = max(0, kk - n + 1)
        j_end = min(kk, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            grad[j_start:j_end + 1] += w * 2.0 * seg[::-1]
    
    return grad

def spectral_flatten_v2(a0, N, n_iters=400, lam_schedule='cosine', soft_alpha=0.05):
    """Improved spectral flattening with cosine annealing and soft non-negativity."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    S0 = a.sum()
    
    for i in range(n_iters):
        t = i / max(1, n_iters - 1)
        if lam_schedule == 'cosine':
            lam = 0.9 * (0.5 + 0.5 * np.cos(np.pi * t)) + 0.1
        elif lam_schedule == 'linear':
            lam = 0.9 - 0.6 * t
        else:
            lam = 0.5
        
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: geometric mean of non-DC magnitudes
        non_dc = mag[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-12)))
        
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if (N % 2 == 0 and k == half):
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        
        # Soft non-negativity
        if soft_alpha > 0:
            # softplus(x) = log(1 + exp(x)) / alpha approximately
            # Use a scaled softplus
            a_new = soft_alpha * np.log1p(np.exp(np.clip(a_new / soft_alpha, -50, 50)))
        else:
            a_new = np.maximum(a_new, 0.0)
        
        a = a_new
        S = a.sum()
        if S > 1e-12:
            a = a * (S0 / S)
    
    return a
