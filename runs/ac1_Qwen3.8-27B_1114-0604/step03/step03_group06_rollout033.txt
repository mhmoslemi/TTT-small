"""
Improved optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LP direction finding with bisection line search, Nesterov-accelerated
subgradient, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time
try:
    import cvxpy as cp
    HAVE_CVXPY = True
except ImportError:
    HAVE_CVXPY = False


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lambda x: lse_score_grad(x, beta), a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

        return None, None

    t0 = time.monotonic()
    if time.monotonic() - t0 > max_wall_s:
        return None, None
    c = fast_conv(a)
    M = float(np.max(c))
    if M < 1e-12:
        return a.copy(), None
    if n > 8000:
        return None, None
    # Build constraint matrix
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_mat[k_start:k_end, j] = a[:k_end - k_start]
    b_var = cp.Variable(n, nonneg=True)
    y_var = cp.Variable(2 * n - 1, nonneg=True)  # dual variable
    constraints = [A_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.PDLP, max_time=max_wall_s)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_time=max_wall_s)
        except Exception:
            return None, None
    if time.monotonic() - t0 > max_wall_s:
        return None, None
    if b_var.value is None:
        return None, None
    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None, None
    # Try to get dual
    dual = None
    try:
        dual = prob.constraints[0].dual_value
        if dual is not None:
            dual = np.abs(dual)
    except Exception:
        pass
    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    return b_opt, dual


def lp_step_with_line_search(a, max_wall_s=30.0):
    """LP step with bisection line search for optimal step size."""
    old_sc = compute_score(a)
    b_opt, dual = solve_lp_direction(a, max_wall_s=max_wall_s)
    if b_opt is None:
        return old_sc, a

    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_sc = old_sc
    
    for _ in range(12):  # 12 bisection steps
        mid = (lo + hi) / 2.0
        a_mid = (1.0 - mid) * a + mid * b_opt
        a_mid = np.clip(a_mid, 0.0, 1000.0)

            continue
        sc_mid = compute_score(a_mid)
        if sc_mid < best_sc:

        a_lo = (1.0 - lo) * a + lo * b_opt
        a_lo = np.clip(a_lo, 0.0, 1000.0)
        sc_lo = compute_score(a_lo) if a_lo.sum() >= 0.01 else np.inf

            lo = mid

            hi = mid
    
    a_best = (1.0 - best_t) * a + best_t * b_opt
    a_best = np.clip(a_best, 0.0, 1000.0)

def nesterov_subgradient(a, n_iters=2000, max_wall_s=60.0):
    """Nesterov-accelerated subgradient for the exact max-conv objective."""
    n = len(a)
    t0 = time.monotonic()
    
    # Normalize to sum = sqrt(2n) for numerical stability
    S0 = float(a.sum())
    if S0 < 1e-12:
        return compute_score(a), a
    scale = np.sqrt(2.0 * n) / S0
    x = a * scale  # now sum(x) = sqrt(2n)
    
    # In this normalization, score = 2n * max(conv(x,x)) / (2n) = max(conv(x,x))
    # So we just minimize max(conv(x,x))
    
    best_x = x.copy()
    best_M = float(np.max(fast_conv(x)))
    
    # Nesterov momentum
    beta = 0.0  # will increase
    v = np.zeros(n)
    
    for it in range(n_iters):
        if time.monotonic() - t0 > max_wall_s:
            break
        
        # Nesterov extrapolation
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        x_extrap = x + (beta / beta_new - 1.0) * v
        
        c = fast_conv(x_extrap)
        M = float(np.max(c))
        k_star = int(np.argmax(c))
        
        # Subgradient of max(conv) at x_extrap
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = x_extrap[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        # Also account for the constraint sum(x) = sqrt(2n)
        # Project gradient onto the hyperplane sum=0
        sg_mean = sg.mean()
        sg_proj = sg - sg_mean
        
        gn = float(np.linalg.norm(sg_proj))
        if gn < 1e-15:
            break
        
        # Polyak step size: (M - M_lower) / ||g||^2
        # Lower bound for M: (sum(x))^2 / (2n-1) = 2n/(2n-1)
        M_lower = 2.0 * n / (2.0 * n - 1.0)
        step = (M - M_lower) / (gn * gn + 1e-30)
        step = min(step, 1.0 / (beta_new * gn))  # cap step size
        
        v = -step * sg_proj
        x = x_extrap + v  # Wait, this isn't quite right for Nesterov
        
        # Actually, let me use the standard Nesterov scheme:
        # x_{k+1} = (beta_k / beta_{k+1}) * x_k + (1 - beta_k/beta_{k+1}) * y_k
        # where y_k = x_k + (beta_{k-1}/beta_k - 1) * (x_k - x_{k-1})
        
        # Simpler: just use standard accelerated gradient
        # x_new = x_extrap - step * grad
        # v = x_new - x
        x_new = x_extrap - step * sg_proj
        x_new = np.clip(x_new, 0.0, 1000.0)
        # Re-project to sum = sqrt(2n)
        S_new = x_new.sum()
        if S_new > 1e-12:
            x_new = x_new * (np.sqrt(2.0 * n) / S_new)
        
        # Update Nesterov state
        v = x_new - x
        x = x_new
        beta = beta_new
        
        # Track best
        M_cur = float(np.max(fast_conv(x)))
        if M_cur < best_M:
            best_M = M_cur
            best_x = x.copy()
    
    # Rescale back
    S_best = float(best_x.sum())
    if S_best > 1e-12:
        best_x = best_x * (S0 / S_best)
    
    best_sc = compute_score(best_x)
    # Also check if the original scaling gives a better score
    return best_sc, best_x

"""
Improved optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key innovations over previous approach:
1. LP dual-weighted subgradient: uses LP dual variables to construct a rich subgradient
   from all active lags (not just the argmax), dramatically improving step quality.
2. Nesterov-accelerated optimization with the weighted subgradient.
3. Bisection line search for LP direction steps.
4. Frequency-domain perturbations for escaping local minima.
5. Adaptive multi-strategy scheduling with stagnation detection.
"""

import numpy as np
from scipy.optimize import minimize
import time

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except ImportError:
    HAVE_CVXPY = False


# Global for L-BFGS-B objective (avoids lambda/closure)
_lse_beta_global = 100.0


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """Linear cross-convolution a*b via FFT."""
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def _lse_obj(x):
    """Top-level objective for L-BFGS-B (uses global beta)."""
    return lse_score_grad(x, _lse_beta_global)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B with LSE-smoothed objective."""
    global _lse_beta_global
    n = len(a0)
    _lse_beta_global = beta
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        _lse_obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def solve_lp_direction(a, max_wall_s=30.0):
    """
    Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k.
    Returns (b_opt, dual_weights) or (None, None).
    """
    global t0_global
    if not HAVE_CVXPY:
        return None, None
    n = len(a)
    t0 = time.monotonic()
    if time.monotonic() - t0 > max_wall_s:
        return None, None
    c = fast_conv(a)
    M = float(np.max(c))
    if M < 1e-12:
        return a.copy(), None
    if n > 9000:
        return None, None

    # Build constraint matrix (sparse-friendly structure)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_mat[k_start:k_end, j] = a[:k_end - k_start]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.PDLP, max_time=max_wall_s)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_time=max_wall_s)
        except Exception:
            return None, None

    if time.monotonic() - t0 > max_wall_s:
        return None, None
    if b_var.value is None:
        return None, None
    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None, None

    # Extract dual for weighted subgradient
    dual = None
    try:
        dual = prob.constraints[0].dual_value
        if dual is not None:
            dual = np.abs(np.asarray(dual, dtype=np.float64))
            # Normalize
            dsum = dual.sum()
            if dsum > 1e-15:
                dual = dual / dsum
    except Exception:
        pass

    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    return b_opt, dual


def lp_step_with_line_search(a, max_wall_s=30.0):
    """LP step with bisection line search for optimal step size."""
    old_sc = compute_score(a)
    b_opt, dual = solve_lp_direction(a, max_wall_s=max_wall_s)
    if b_opt is None:
        return old_sc, a

    # Bisection line search
    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_sc = old_sc

    for _ in range(14):
        mid = (lo + hi) / 2.0
        a_mid = (1.0 - mid) * a + mid * b_opt
        a_mid = np.clip(a_mid, 0.0, 1000.0)
        if a_mid.sum() < 0.01:
            hi = mid
            continue
        sc_mid = compute_score(a_mid)
        if sc_mid < best_sc:
            best_sc = sc_mid
            best_t = mid
        # Determine direction: compare with nearby point
        probe = min(lo + 0.01, 1.0)
        a_probe = (1.0 - probe) * a + probe * b_opt
        a_probe = np.clip(a_probe, 0.0, 1000.0)
        sc_probe = compute_score(a_probe) if a_probe.sum() >= 0.01 else np.inf
        if sc_mid < sc_probe:
            lo = mid
        else:
            hi = mid

    a_best = (1.0 - best_t) * a + best_t * b_opt
    a_best = np.clip(a_best, 0.0, 1000.0)
    return best_sc, a_best


def weighted_subgradient(a, y_weights):
    """
    Compute subgradient using LP dual weights.
    y_weights[k] is the normalized weight for lag k.
    This gives a rich subgradient from all active lags simultaneously.
    """
    n = len(a)
    sg = np.zeros(n)
    active = y_weights > 1e-8
    n_active = int(np.sum(active))
    if n_active == 0:
        # Fallback to argmax
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]
        return sg

    # Efficient computation: for each active lag k, add y_k * 2 * a[j_start:j_end+1][::-1]
    # Group by contribution positions
    for k in np.where(active)[0]:
        w = y_weights[k]
        if w < 1e-10:
            continue
        j_start = max(0, int(k) - n + 1)
        j_end = min(int(k), n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] += w * 2.0 * a[j_start:j_end + 1][::-1]
    return sg


def nesterov_weighted(a, n_iters=2000, max_wall_s=60.0):
    """
    Nesterov-accelerated optimization using LP-dual-weighted subgradient.
    This is the key innovation: uses all active lags (weighted by LP dual)
    instead of just the argmax lag.
    """
    global t0_global
    n = len(a)
    t0 = time.monotonic()
    a = a.astype(np.float64).copy()

    best_a = a.copy()
    best_sc = compute_score(a)

    # Work in normalized space: sum = 1
    S0 = float(a.sum())
    if S0 < 1e-12:
        return best_sc, best_a
    x = a / S0

    best_x = x.copy()
    best_M = float(np.max(fast_conv(x)))

    # Nesterov momentum
    mu = 0.0
    x_prev = x.copy()

    for it in range(n_iters):
        if time.monotonic() - t0 > max_wall_s:
            break

        # Nesterov extrapolation
        t_k = 2.0 / (it + 2.0)
        x_extrap = x + mu * np.clip(x - x_prev, -x, 100.0)
        x_extrap = np.clip(x_extrap, 0, None)

        c = fast_conv(x_extrap)
        M = float(np.max(c))
        k_star = int(np.argmax(c))

        # Get weighted subgradient
        # Use top active lags (those near the max)
        c_sorted = np.sort(c)[::-1]
        threshold = c_star = c_sorted[min(20, len(c_sorted) - 1)]
        # Weights proportional to exp(-(c_star - c)/gap)
        gap = max(c_sorted[0] - c_sorted[min(20, len(c_sorted)-1)], 1e-10)
        w_raw = np.exp(-(c_star - c) / (gap * 5.0))
        w_sum = w_raw.sum()
        if w_sum > 1e-15:
            y_weights = w_raw / w_sum
        else:
            y_weights = np.zeros(len(c))
            y_weights[k_star] = 1.0

        sg = weighted_subgradient(x_extrap, y_weights)

        # Project onto sum=1 hyperplane
        sg = sg - sg.mean()
        gn = float(np.linalg.norm(sg))
        if gn < 1e-15:
            break

        # Adaptive step size
        M_lb = 1.0 / (2.0 * n - 1.0)
        step = (M - M_lb) / (gn * gn + 1e-30)
        step = min(step, 1.0 / (gn * 2.0 + 1e-15))
        step = min(step, 0.5)

        x_new = x_extrap - step * sg
        x_new = np.clip(x_new, 0.0, None)
        S_new = x_new.sum()
        if S_new > 1e-12:
            x_new = x_new / S_new

        mu = t_k
        x_prev = x
        x = x_new

        M_cur = float(np.max(fast_conv(x)))
        if M_cur < best_M:
            best_M = M_cur
            best_x = x.copy()

    best_a_out = best_x * S0
    best_sc = compute_score(best_a_out)
    return best_sc, best_a_out


def nesterov_single_lag(a, n_iters=3000, max_wall_s=60.0):
    """Nesterov with single-lag subgradient (simpler fallback)."""
    n = len(a)
    t0 = time.monotonic()

    best_sc = compute_score(a)

        return best_sc, best_a

    mu = 0.0
    x_prev = x.copy()

    for it in range(n_iters):
        if time.monotonic() - t0 > max_wall_s:
            break

        t_k = 2.0 / (it + 2.0)
        x_extrap = x + mu * (x - x_prev)
        x_extrap = np.clip(x_extrap, 0, None)

        c = fast_conv(x_extrap)
        M = float(np.max(c))
        k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)

        sg = np.zeros(n)
        if j_end >= j_start:

            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        sg = sg - sg.mean()
        gn = float(np.linalg.norm(sg))
        if gn < 1e-15:
            break

        M_lb = 1.0 / (2.0 * n - 1.0)
        step = min((M - M_lb) / (gn * gn + 1e-30), 1.0 / (gn + 1e-15))
        step = min(step, 0.5)

        x_new = x_extrap - step * sg
        x_new = np.clip(x_new, 0.0, None)
        S_new = x_new.sum()
        if S_new > 1e-12:
            x_new = x_new / S_new

        x = x_new

        M_cur = float(np.max(fast_conv(x)))

            best_x = x.copy()

    best_a_out = best_x * S0

    return best_sc, best_a_out


def freq_perturb(a, rng, amp=0.02):
    """Frequency-domain perturbation."""
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = amp * rng.standard_normal(len(A))
    A_pert = A * np.exp(1j * phase_pert)
    a_pert = np.fft.irfft(A_pert, n)
    return np.clip(a_pert, 0, 1000)


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_stars = np.argsort(c)[::-1][:min(20, 2 * n - 1)]
    a_pert = a.copy()
    total_reduce = 0.0
    reduce_idx = []
    for k in k_stars:
        j_start = max(0, int(k) - n + 1)
        j_end = min(int(k), n - 1)
        if j_end >= j_start:
            # Find the peak contributor in this segment
            contrib = a[j_start:j_end + 1][::-1]
            peak_local = int(np.argmax(contrib))
            j = j_start + peak_local
            red = a_pert[j] * strength * rng.random()
            a_pert[j] -= red
            total_reduce += red
            reduce_idx.append(j)
    # Redistribute to low-contribution elements
    c_sorted_asc = np.argsort(c)
    low_idx = c_sorted_asc[:max(len(reduce_idx), n // 10)]
    if len(low_idx) > 0 and total_reduce > 0:
        per = total_reduce / len(low_idx)
        for j in low_idx:
            a_pert[j] += per
    a_pert = np.clip(a_pert, 0, 1000)
    S = a_pert.sum()
    if S > 0.01:
        a_pert = a_pert * (a.sum() / S)
    return a_pert

            sc, a_opt, _ = run_lbfgs(a_cur, 2000, maxiter=3000, maxfun=6000)
            if sc < best_score:
                report(sc, a_opt, f"P2 LP+LBFGS#{it}")
                a_cur = a_opt

        elif mod == 3:
            # Frequency perturbation + Nesterov
            a_pert = freq_perturb(a_cur, rng, amp=0.05)
            if a_pert.sum() > 0.01:
                a_pert = a_pert * (a_cur.sum() / a_pert.sum())
            max_ns = min(30.0, remaining * 0.4)
            sc, a_ns = nesterov_weighted(a_pert, n_iters=1500, max_wall_s=max_ns)
            if sc < best_score:
                report(sc, a_ns, f"P2 freq+NW#{it}")
                a_cur = a_ns

        elif mod == 4:
            # Targeted perturbation + L-BFGS
            a_pert = targeted_perturb(a_cur, rng, strength=0.06)
            if time.monotonic() > phase2_deadline - 30:
                continue
            sc, a_opt, _ = run_lbfgs(a_pert, 1000, maxiter=3000, maxfun=6000)
            if sc < best_score:
                report(sc, a_opt, f"P2 target+LBFGS#{it}")
                a_cur = a_opt
            else:
                a_cur = a_opt if sc < compute_score(a_cur) else a_cur

        else:
            # Nesterov single-lag (different flavor)
            max_ns = min(40.0, remaining * 0.4)
            sc, a_ns = nesterov_single_lag(a_cur, n_iters=2000, max_wall_s=max_ns)
            if sc < best_score:
                report(sc, a_ns, f"P2 NSL#{it}")
                a_cur = a_ns

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # =========================================================================
    # Phase 3: Multi-length exploration (15% of time)
    # =========================================================================
    phase3_deadline = t_start + 0.90 * (total_budget - 10.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-length (deadline at {phase3_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    # Try nearby lengths
    length_factors = [0.85, 0.92, 0.97, 1.03, 1.08, 1.15]
    for factor in length_factors:
        if time.monotonic() > phase3_deadline - 20:
            break
        new_n = int(n_best * factor)
        if new_n < 1024 or new_n > 25000:
            continue
        idx = np.linspace(0, n_best - 1, new_n)
        a_res = np.interp(idx, np.arange(n_best), best_a)
        a_res = np.clip(a_res, 0, 1000)
        sc0 = compute_score(a_res)
        print(f"  n={new_n} (factor={factor}): initial score={sc0:.8f}")
        if time.monotonic() > phase3_deadline - 20:
            continue
        # Polish
        maxit = min(5000, int((phase3_deadline - time.monotonic()) * 60))
        if maxit < 500:
            continue
        sc, a_opt, _ = run_lbfgs(a_res, 500, maxiter=maxit, maxfun=maxit * 2)
        report(sc, a_opt, f"len={new_n}")
        # If promising, do more
        if sc < best_score + 0.005 and time.monotonic() < phase3_deadline - 20:
            sc2, a_opt2, _ = run_lbfgs(a_opt, 1500, maxiter=3000, maxfun=6000)
            report(sc2, a_opt2, f"len={new_n}v2")
            # Try Nesterov on it
            if time.monotonic() < phase3_deadline - 15:
                sc3, a_ns = nesterov_weighted(a_opt2, n_iters=1000, max_wall_s=15.0)
                report(sc3, a_ns, f"len={new_n}v3")

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # =========================================================================
    # Phase 4: Final intensive polish (10% of time)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Polish")
    print(f"{'='*60}")

    a_final = best_a.copy()

    # Progressive beta LSE
    for beta in [500, 1000, 2000, 4000, 8000, 12000]:
        if time.monotonic() > deadline - 5.0:
            break
        remaining = deadline - 5.0 - time.monotonic()
        if remaining < 10:
            break
        maxit = int(min(8000, remaining * 100))
        sc, a_opt, _ = run_lbfgs(a_final, beta, maxiter=maxit, maxfun=maxit * 2)
        if sc < best_score:
            report(sc, a_opt, f"final b={beta}")
            a_final = a_opt

    # Nesterov weighted final
    if time.monotonic() < deadline - 10.0:
        sc, a_ns = nesterov_weighted(a_final, n_iters=3000, max_wall_s=min(25.0, deadline - 10.0 - time.monotonic()))
        if sc < best_score:
            report(sc, a_ns, "final NW")
            a_final = a_ns

    # One more LP
    if time.monotonic() < deadline - 30.0:
        sc, a_lp = lp_step_with_line_search(a_final, max_wall_s=25.0)
        if sc < best_score:
            report(sc, a_lp, "final LP")
            a_final = a_lp
        # Polish after LP
        if time.monotonic() < deadline - 5.0:
            sc, a_opt, _ = run_lbfgs(a_final, 5000, maxiter=3000, maxfun=6000)
            if sc < best_score:
                report(sc, a_opt, "final final")
                a_final = a_opt

    # =========================================================================
    # Final verification
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
