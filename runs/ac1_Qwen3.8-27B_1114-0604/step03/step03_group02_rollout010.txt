from scipy.optimize import minimize
import time

N = 8192
M_CONV = 2 * N - 1  # Length of full convolution: 16383
M_FFT_CONV = 16384  # Next power of 2 for convolution
M_CROSS = 3 * N - 2  # Length needed for cross-correlation: 24574
M_FFT_CROSS = 32768  # Next power of 2 for cross-correlation


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    A = np.fft.rfft(a, M_FFT_CONV)
    return np.fft.irfft(A * A, M_FFT_CONV)[:M_CONV]


def compute_score(a):

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * N * float(np.max(c)) / (S * S)

def freq_equalize(a, n_iters=50, deadline=None):
    """
    Frequency-domain equalization: try to flatten |A(f)|^2.
    Alternates between:
    1. Compute spectrum A(f)
    2. Modify amplitude to be more flat

    """

    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    S = a.sum()

        if deadline is not None and time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a)
        amp = np.abs(A)
        amp_sq = amp ** 2
        
        # Target: flatten amp_sq
        # Use a power transformation: amp_new = amp_sq^gamma where gamma < 1 flattens
        gamma = 0.3 + 0.7 * (it / max(n_iters - 1, 1))  # Start aggressive, end conservative
        amp_target = amp_sq ** gamma
        amp_target = amp_target / max(amp_target.mean(), 1e-10)
        
        # Blend between current and target
        t_blend = 0.3
        amp_new = (1 - t_blend) * amp + t_blend * amp_target * (amp.mean() / max(amp_target.mean(), 1e-10))
        
        # Reconstruct with modified amplitude, preserving phase
        phase = np.angle(A)
        A_new = amp_new * np.exp(1j * phase)
        
        # Make real
        a_new = np.fft.irfft(A_new, n)
        
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        
        # Renormalize to preserve sum
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new * (S / S_new)
        
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

            a_pert[j] = max(a_pert[j], 0.0)
            total_moved += amt
        
        # Add to bottom contributors
        add_idx = contrib_sorted[-top_k:]
        if len(add_idx) > 0 and total_moved > 0:
            # Weighted addition: prefer elements with lower current value
            weights = 1.0 / (a[add_idx] + 0.01)
            weights /= weights.sum()
            for i, j in enumerate(add_idx):
                a_pert[j] += total_moved * weights[i]
        
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        # Renormalize
        S_new = a_pert.sum()
        if S_new > 1e-12:
            a_pert = a_pert * (S / S_new)
        
        sc = compute_score(a_pert)
        if sc < best_sc:
            best_sc = sc
            best_a = a_pert.copy()
            a = a_pert  # Continue from improved solution

"""
Step function optimization for fixed length 8192.
Minimizes: 2*n*max(conv(a,a)) / (sum(a))^2 subject to a >= 0.

Uses LSE-smoothed objective with continuation in beta, frequency-domain
equalization, LP-inspired multiplicative updates, and targeted peak shaving.
"""

import numpy as np
from scipy.optimize import minimize
import time

N = 8192
M_CONV = 2 * N - 1
M_FFT_CONV = 1 << (M_CONV - 1).bit_length()
M_CROSS = 3 * N - 2
M_FFT_CROSS = 1 << (M_CROSS - 1).bit_length()


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    A = np.fft.rfft(a, M_FFT_CONV)
    return np.fft.irfft(A * A, M_FFT_CONV)[:M_CONV]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * N * float(np.max(c)) / (S * S)


def lse_score_grad(x, beta):
    """LSE-smoothed score and gradient as a tuple (f, g) for scipy minimize."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled_shifted = scaled - scaled.max()
    exp_scaled = np.exp(scaled_shifted)
    log_sum = np.log(exp_scaled.sum()) + scaled.max()
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = x[::-1]
    PA = np.fft.rfft(p, M_FFT_CROSS)
    RA = np.fft.rfft(rev_a, M_FFT_CROSS)
    conv_res = np.fft.irfft(PA * RA, M_FFT_CROSS)[n - 1: 2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun, deadline):
    """Run L-BFGS-B on LSE-smoothed score. Returns (exact_score, result_array)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    mi = maxiter
    mf = maxfun
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 3.0:
            return compute_score(a0), a0
        mf = min(maxfun, max(50, int(remaining * 150)))
        mi = min(maxiter, max(50, int(remaining * 40)))
    options = {'ftol': 1e-16, 'gtol': 1e-14, 'maxiter': mi, 'maxfun': mf, 'maxls': 30}
    try:
        res = minimize(lse_score_grad, a0, args=(beta,), method='L-BFGS-B',
                       jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def multiplicative_equalize(a, n_iters, alpha, deadline):
    """LP-dual-inspired multiplicative update to equalize convolution spectrum."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):

            break
        c = fast_conv(a)
        M = float(c.max())
        if M < 1e-15:
            break
        soft_beta = 30.0 + 15.0 * it
        w = np.exp(soft_beta * (c - M))
        w = w / w.sum()
        rev_a = a[::-1]
        W = np.fft.rfft(w, M_FFT_CROSS)
        RA = np.fft.rfft(rev_a, M_FFT_CROSS)
        cross = np.fft.irfft(W * RA, M_FFT_CROSS)[n - 1: 2 * n - 1]
        h = 2.0 * cross
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()
        factor = 1.0 - alpha + alpha * (h_mean / h)

def freq_equalize(a, n_iters, deadline):
    """Frequency-domain equalization to flatten |A(f)|^2."""
    n = len(a)

            break

        amp = np.abs(A)

        amp_sq_mean = amp_sq.mean()
        if amp_sq_mean < 1e-20:
            break
        amp_sq_norm = amp_sq / amp_sq_mean
        gamma = 0.4 + 0.5 * (it / max(n_iters - 1, 1))
        amp_target = amp_sq_norm ** gamma
        amp_target_mean = amp_target.mean()
        amp_current_mean = amp.mean()
        amp_new = (1.0 - 0.3) * amp + 0.3 * amp_target * (amp_current_mean / max(amp_target_mean, 1e-10))
        phase = np.angle(A)
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n)

        if S_new < 1e-12:

        a = a_new
        sc = compute_score(a)

            best_a = a.copy()
    return best_sc, best_a

        contrib = np.zeros(n)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        top_k = max(10, n // 100)
        strength = rng.uniform(0.02, max_strength)
        a_pert = a.copy()
        total_moved = 0.0
        contrib_sorted = np.argsort(contrib)
        reduce_idx = contrib_sorted[-top_k:]
        add_idx = contrib_sorted[:top_k]
        for j in reduce_idx:
            amt = a_pert[j] * strength
            a_pert[j] -= amt
            a_pert[j] = max(a_pert[j], 0.0)
            total_moved += amt
        if len(add_idx) > 0 and total_moved > 0:
            weights = 1.0 / (a[add_idx] + 0.01)
            weights /= weights.sum()
            for i, j in enumerate(add_idx):
                a_pert[j] += total_moved * weights[i]
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S_new = a_pert.sum()
        if S_new > 1e-12:
            a_pert = a_pert * (S / S_new)
        sc = compute_score(a_pert)
        if sc < best_sc:
            best_sc = sc
            best_a = a_pert.copy()
            a = a_pert
    return best_sc, best_a


def random_restart(n, rng):
    """Generate a diverse random initialization."""
    x = np.arange(n, dtype=np.float64) / n
    strategy = rng.integers(0, 7)
    if strategy == 0:
        a = np.maximum(0.3 + 0.4 * rng.standard_normal(n), 0.01)
    elif strategy == 1:
        freq = 1.0 + 5.0 * rng.random()
        a = 0.5 + 0.3 * np.cos(2 * np.pi * freq * x + rng.random() * 2 * np.pi)
        a = np.maximum(a + 0.1 * rng.standard_normal(n), 0.01)
    elif strategy == 2:
        n_blocks = int(rng.integers(8, 64))
        block_size = n // n_blocks
        vals = 0.2 + 0.6 * rng.random(n_blocks)
        a = np.repeat(vals, block_size)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        a = a[:n]
        a = np.maximum(a + 0.05 * rng.standard_normal(n), 0.01)
    elif strategy == 3:
        phase = 2 * np.pi * rng.random(M_FFT_CONV // 2 + 1)
        amp = np.ones(M_FFT_CONV // 2 + 1)
        amp[0] = 0.5
        A_complex = amp * np.exp(1j * phase)
        a = np.fft.irfft(A_complex, M_FFT_CONV)[:n]
        a = np.maximum(a, 0.01)
        a = a / max(a.max(), 1e-10) * 0.5
    elif strategy == 4:
        a = np.cumsum(rng.standard_normal(n))
        a = a - np.mean(a)
        a = np.abs(a)
        a = a / max(a.max(), 1e-10) * 0.8 + 0.1
    elif strategy == 5:
        a = np.where(rng.random(n) > 0.5,
                    0.2 + 0.3 * rng.random(n),
                    0.7 + 0.3 * rng.random(n))
    else:
        a = 0.5 + 0.3 * np.sin(2 * np.pi * rng.random() * n * x) * np.cos(2 * np.pi * rng.random() * n * x)
        a = np.maximum(a, 0.01)
    return np.clip(a, 0.01, 1000.0)


def subgrad_step(a, sc_lb=1.0):
    """One projected subgradient step on the exact max-conv objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    M = float(c.max())
    k_star = int(np.argmax(c))
    sc = 2.0 * n * M / (S * S)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    sg = np.zeros(n)
    if j_end >= j_start:
        sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    gn = float(np.linalg.norm(grad))
    if gn < 1e-15:
        return a
    step = min((sc - sc_lb) / (gn * gn + 1e-30), 0.3)
    a_new = a - step * grad
    return np.clip(a_new, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence of length 8192 minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    n = N

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals():
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size == n:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(n) * 0.5

    # Initial score from parent
    if parent is not None:
        par_sc = compute_score(parent)
        if par_sc < best_score:
            best_score = par_sc
            best_a = parent.copy()
        print(f"Parent score: {par_sc:.8f}")

    # Start score
    u_sc = compute_score(np.ones(n) * 0.5)
    if u_sc < best_score:
        best_score = u_sc
        best_a = np.ones(n) * 0.5
    print(f"Uniform score: {u_sc:.8f}")
    print(f"Initial best: {best_score:.8f}")

    # ==================================================================
    # Phase 1: Multi-start global search with LSE continuation
    # ==================================================================
    phase1_deadline = t_start + 0.38 * budget_s
    beta_schedule = [30, 80, 200, 500]

    print(f"\n{'=' * 60}")
    print(f"Phase 1: Multi-start LSE optimization (t={phase1_deadline - t_start:.0f}s)")
    print(f"{'=' * 60}")

    inits = []
    if parent is not None:
        inits.append(parent.copy())
    inits.append(np.ones(n) * 0.5)
    for i in range(6):
        inits.append(random_restart(n, rng))

    for init_i, a0 in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        a_cur = a0.copy()
        init_start = time.monotonic()
        for beta in beta_schedule:
            if time.monotonic() > phase1_deadline:
                break
            remaining = phase1_deadline - time.monotonic()
            if remaining < 15:
                break
            sc, a_opt = run_lbfgs(a_cur, beta, 2000, 30000, phase1_deadline)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"  [init={init_i}, beta={beta}] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
            a_cur = a_opt
            # Skip remaining betas if no improvement in 200s
            if time.monotonic() - init_start > 200:
                break

    print(f"\nAfter Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Intensive refinement
    # ==================================================================
    phase2_deadline = t_start + 0.70 * budget_s
    print(f"\n{'=' * 60}")
    print(f"Phase 2: Intensive refinement (t={phase2_deadline - t_start:.0f}s)")
    print(f"{'=' * 60}")

    a_cur = best_a.copy()

    for round_i in range(12):
        if time.monotonic() > phase2_deadline:
            break

        # Multiplicative equalization
        sc, a_eq = multiplicative_equalize(a_cur, 50, 0.25, phase2_deadline)
        if sc < best_score:
            best_score = sc
            best_a = a_eq.copy()
            print(f"  [R{round_i+1} eq] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        a_cur = a_eq

        # Frequency equalization
        sc, a_fe = freq_equalize(a_eq, 25, phase2_deadline)
        if sc < best_score:
            best_score = sc
            best_a = a_fe.copy()
            print(f"  [R{round_i+1} freq] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        a_cur = a_fe

        # L-BFGS polish
        if time.monotonic() < phase2_deadline - 15:
            beta = 300 + 200 * round_i
            sc, a_opt = run_lbfgs(a_cur, beta, 2000, 25000, phase2_deadline)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"  [R{round_i+1} lbfgs b={beta}] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
            a_cur = a_opt

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Peak shaving + local search
    # ==================================================================
    phase3_deadline = t_start + 0.87 * budget_s
    print(f"\n{'=' * 60}")
    print(f"Phase 3: Peak shaving and local search (t={phase3_deadline - t_start:.0f}s)")
    print(f"{'=' * 60}")

    a_cur = best_a.copy()
    n_trials = 25

    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline:
            break
        strategy = trial % 5

        if strategy == 0:
            sc, a_pert = peak_shave(a_cur, rng, 25, 0.08, phase3_deadline)
            a_opt = a_pert
        elif strategy == 1:
            scale = 0.01 + 0.03 * rng.random()
            a_pert = a_cur * (1.0 + scale * rng.standard_normal(n))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            sc, a_opt = run_lbfgs(a_pert, 500, 1500, 20000, phase3_deadline)
        elif strategy == 2:
            # Multi-peak shaving
            a_pert = a_cur.copy()
            c = fast_conv(a_pert)
            top_lags = np.argsort(c)[::-1][:3]
            for k_star in top_lags:
                k_star = int(k_star)
                j_start = max(0, k_star - n + 1)
                j_end = min(k_star, n - 1)
                contrib = np.zeros(n)
                if j_end >= j_start:
                    contrib[j_start:j_end + 1] = a_pert[j_start:j_end + 1][::-1]
                top_idx = np.argsort(contrib)[::-1][:15]
                bot_idx = np.argsort(contrib)[:15]
                total = 0.0
                for j in top_idx:
                    amt = a_pert[j] * 0.04
                    a_pert[j] = max(a_pert[j] - amt, 0.0)
                    total += amt
                if total > 0:
                    for j in bot_idx:
                        a_pert[j] += total / len(bot_idx)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            S_pert = a_pert.sum()
            if S_pert > 1e-12:
                a_pert = a_pert * (a_cur.sum() / S_pert)
            sc, a_opt = run_lbfgs(a_pert, 500, 1500, 20000, phase3_deadline)
        elif strategy == 3:
            # Subgradient steps
            a_pert = a_cur.copy()
            for _ in range(20):
                a_pert = subgrad_step(a_pert)
            sc = compute_score(a_pert)
            a_opt = a_pert
        else:
            # Sinusoidal perturbation
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.04 * rng.random()
            x = np.arange(n, dtype=np.float64) / n
            a_pert = a_cur * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            sc, a_opt = run_lbfgs(a_pert, 400, 1500, 20000, phase3_deadline)

        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [trial={trial}, strat={strategy}] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")

        # Quick polish
        if time.monotonic() < phase3_deadline - 8:
            sc2, a_opt2 = run_lbfgs(a_opt, 800, 1000, 12000, phase3_deadline)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_opt2.copy()
                print(f"  [trial={trial} polish] NEW BEST: {sc2:.8f} (t={time.monotonic()-t_start:.0f}s}")
            a_opt = a_opt2

        if compute_score(a_opt) < compute_score(a_cur):
            a_cur = a_opt

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Final high-precision polish
    # ==================================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 4: Final polish")
    print(f"{'=' * 60}")

    for beta in [1000, 2000, 5000]:
        if time.monotonic() > deadline - 40:
            break
        sc, a_opt = run_lbfgs(best_a, beta, 3000, 40000, deadline - 40)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [final b={beta}] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # Final multiplicative equalization
    if time.monotonic() < deadline - 30:
        sc, a_eq = multiplicative_equalize(best_a, 60, 0.15, deadline - 30)
        if sc < best_score:
            best_score = sc
            best_a = a_eq.copy()
            print(f"  [final eq] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # Final L-BFGS
    if time.monotonic() < deadline - 15:
        sc, a_opt = run_lbfgs(best_a, 3000, 2000, 30000, deadline - 15)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [final lbfgs] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
