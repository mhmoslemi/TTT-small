def lse_obj_norm(a,beta):
    n = len(a)
    aw = sanitize_nonneg(a)
    S = float(aw.sum())
    if S < 1e-12:
       return 1e100, np.zeros(n)
    y = aw / S
    b = linear_conv(y)
    cmax = float(np.max(b))
    scaled = beta * (b - cmax)
    scaled = np.minimum(scaled, 0.0)
    expv = np.exp(scaled)
    expsum = float(expv.sum())
    if not np.isfinite(expsum) or expsum < 1e-300:
       lse = cmax
       p = np.zeros(2*n-1, dtype=float)
       p[int(np.argmax(b))] = 1.0
    else:
       lse = cmax + float(np.log(expsum)) / beta
       p = expv / expsum
    gy = weighted_conv_gradient(y,p)
    dot = float(np.dot(y, gy))
    grad = (gy - dot) / S
    grad *= 2.0*n
    if not np.all(np.isfinite(grad)):
       grad = np.zeros(n)
    obj = 2.0*n*lse
    if not np.isfinite(obj): obj = 1e100
    return float(obj), grad

```python
"""
Non-negative sequence search for minimum peak linear convolution.

The objective is
    score(a) = 2*n*max(a * a) / (sum(a))^2
where * denotes linear convolution and a is non-negative.  The score is
scale invariant, so the optimization is done on the simplex sum(a)=1.

The program combines several complementary steps:
1. Use the previously found sequence, if available, as the main seed.
2. Generate many cheap initializations and warm them by a spectral
   square-root flattening step in convolution space.
3. Polish promising points with L-BFGS-B on a normalized LSE approximation
   of the hard max-convolution objective.
4. Finish with deterministic cleanup and a final oracle check via
   evaluate_sequence, if available.

All helper functions are top-level and no lambdas are used.
"""

import numpy as np
import time
from scipy.optimize import minimize


OPT_DEADLINE = 0.0


def lbfgs_callback(x):
    return time.monotonic() > OPT_DEADLINE


def fft_len(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def sanitize_nonneg(a):
    a = np.asarray(a, dtype=float).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    return np.clip(a, 0.0, 1000.0)


def normalize_nonneg(a):
    a = sanitize_nonneg(a)
    if a.size == 0:
        return np.ones(1, dtype=float)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.full(a.size, 1.0 / float(a.size))
    return a / s


def linear_conv(a):
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=float)
    m = 2 * n - 1
    nfft = fft_len(m)
    fac = np.fft.rfft(a, nfft)
    return np.fft.irfft(fac * fac, nfft)[:m]


def compute_score(a):
    a = sanitize_nonneg(a)
    if a.size == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    n = a.size
    b = linear_conv(a)
    if b.size == 0:
        return np.inf
    maxb = float(np.max(b))
    if not np.isfinite(maxb):
        return np.inf
    return float(2.0 * n * maxb / (s * s))


def resize_nonneg(a, n):
    a = sanitize_nonneg(a)
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=float)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=float)
    if m == n:
        return normalize_nonneg(a)
    if m == 1:
        return normalize_nonneg(np.full(n, float(a[0])))

    old = np.arange(m, dtype=float) + 0.5
    new = (np.arange(n, dtype=float) + 0.5) * (float(m) / float(n))
    y = np.interp(new, old, a, left=0.0, right=0.0)
    return normalize_nonneg(y)


def weighted_conv_gradient(a, p):
    """
    Gradient of sum_k p[k] * (a*a)[k] with respect to a, assuming a has length n.
    Returns grad_j = 2 * sum_l p[j+l] * a[l].
    """
    n = int(a.size)
    m = 2 * n - 1
    p = np.asarray(p, dtype=float).ravel()
    if p.size != m:
        tmp = np.zeros(m, dtype=float)
        kk = min(m, p.size)
        if kk > 0:
            tmp[:kk] = p[:kk]
        p = tmp

    nfft = fft_len(m)
    pf = np.fft.fft(p, nfft)
    af = np.fft.fft(a, nfft)
    q = np.fft.ifft(pf * np.conj(af), nfft)[:n].real
    return 2.0 * q


def lse_objective_normalized(a, beta):
    """
    Normalized LSE surrogate for 2*n*max(conv(a,a)), with scale invariance
    handled by evaluating on y = a / sum(a).
    """
    n = int(a.size)
    aw = sanitize_nonneg(a)
    s = float(aw.sum())
    if s < 1e-12:
        return 1e100, np.zeros(n, dtype=float)

    y = aw / s
    b = linear_conv(y)
    cmax = float(np.max(b))

    scaled = beta * (b - cmax)
    np.minimum(scaled, 0.0, out=scaled)
    expv = np.exp(scaled)
    expsum = float(expv.sum())

    if not np.isfinite(expsum) or expsum < 1e-300:
        lse = cmax
        p = np.zeros(2 * n - 1, dtype=float)
        p[int(np.argmax(b))] = 1.0
    else:
        lse = cmax + float(np.log(expsum)) / beta
        p = expv / expsum

    gy = weighted_conv_gradient(y, p)
    dot = float(np.dot(y, gy))
    grad = (gy - dot) / s
    grad *= 2.0 * n

    obj = 2.0 * n * lse
    if not np.isfinite(obj):
        obj = 1e100
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=float)

    return float(obj), grad.copy()


def run_lbfgs_norm(a0, beta, maxiter, maxfun, deadline=None):
    global OPT_DEADLINE

    a0 = normalize_nonneg(a0)
    n = a0.size
    if n == 0:
        return 2.0, np.ones(1, dtype=float)

    bounds = [(0.0, 100.0)] * n
    OPT_DEADLINE = float(deadline) if deadline is not None else (time.monotonic() + 1e6)

    options = {
        "ftol": 1e-15,
        "gtol": 1e-11,
        "maxiter": int(max(1, maxiter)),
        "maxfun": int(max(1, maxfun)),
        "maxls": 25,
        "disp": False,
    }

    try:
        res = minimize(
            lse_objective_normalized,
            a0,
            args=(beta,),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options,
            callback=lbfgs_callback,
        )
        a_out = np.asarray(res.x, dtype=float).copy()
    except Exception:
        a_out = a0.copy()

    a_out = sanitize_nonneg(a_out)
    s = float(a_out.sum())
    if s < 1e-12:
        a_out = np.full(n, 1.0 / float(n))
    else:
        a_out = a_out / s

    sc = compute_score(a_out)
    return sc, a_out


def run_one_optim(a, beta, time_budget):
    n = int(a.size)
    if n <= 4096:
        cap = 20000
    elif n <= 12000:
        cap = 12000
    else:
        cap = 8000

    maxfun = min(cap, max(800, int(time_budget * 1000)))
    maxiter = max(200, int(maxfun // 5))
    deadline = time.monotonic() + max(1.0, float(time_budget) * 0.95)
    return run_lbfgs_norm(a, beta, maxiter, maxfun, deadline)


def lbfgs_continuation(a, sc_init, betas, budget_s):
    a = normalize_nonneg(a)
    best = a.copy()
    bestsc = float(sc_init)

    if budget_s <= 0.0 or len(betas) == 0:
        return best, bestsc

    start = time.monotonic()
    stage_budget = float(budget_s) / float(len(betas))

    for beta in betas:
        elapsed = time.monotonic() - start
        rem = float(budget_s) - elapsed
        if rem < 20.0:
            break
        alloc = min(stage_budget, rem)
        if alloc < 20.0:
            break

        sc, a2 = run_one_optim(a, beta, alloc)
        a2 = normalize_nonneg(a2)
        if np.isfinite(sc) and sc < bestsc:
            bestsc = float(sc)
            best = a2.copy()

        a = a2

    return best, bestsc


def spectral_sqrt_flatten(a, budget_s, max_iter=300):
    """
    Directly shape the convolution spectrum:
    target a flattened a*a, then move a by taking a spectral square root.
    """
    a = normalize_nonneg(a)
    n = int(a.size)
    if n < 2 or budget_s <= 0.0:
        return a

    max_iter = int(max(8, min(max_iter, max(8, int(budget_s / 0.01)))))
    tstart = time.monotonic()
    limit = tstart + float(budget_s)

    best = a.copy()
    b = linear_conv(a)
    bestsc = 2.0 * n * float(np.max(b))
    streak = 0

    for it in range(max_iter):
        if time.monotonic() > limit:
            break

        b = linear_conv(a)
        bmax = float(np.max(b))
        bavg = float(np.mean(b))
        if not (np.isfinite(bmax) and np.isfinite(bavg)) or bavg <= 0.0:
            break

        score = 2.0 * n * bmax
        if score < bestsc:
            bestsc = float(score)
            best = a.copy()
            streak = 0
        else:
            streak += 1

        if streak > 100:
            break
        if bmax <= bavg * (1.0 + 1e-6):
            break

        progress = float(it) / float(max(1, max_iter - 1))

        # Anneal the target peak: first close to current peak, then toward ~1.2*mean.
        t_target = (1.0 - progress) * 0.98 * bmax + progress * 1.20 * bavg
        t_target = max(t_target, bavg * 1.0005)
        if t_target > 0.985 * bmax:
            t_target = 0.985 * bmax
        if t_target >= bmax:
            t_target = bmax * 0.999

        target = b.copy()
        above = target > t_target
        if np.any(above):
            target[above] = t_target
            below = ~(target > t_target)
            nb = int(np.count_nonzero(below))
            if nb > 0:
                excess = float(np.sum(b[above] - t_target))
                if excess > 0.0:
                    target[below] += excess / float(nb)
        target = np.maximum(target, 1e-300)

        nfft = fft_len(2 * n - 1)
        af = np.fft.fft(a, nfft)
        bf = af * af
        tf = np.fft.fft(target, nfft)

        maxbf = float(np.max(np.abs(bf))) if bf.size else 1.0
        tol = 1e-24 * max(1.0, maxbf)
        ratio = np.ones(nfft, dtype=complex)
        mask = np.abs(bf) > tol
        ratio[mask] = tf[mask] / bf[mask]

        eta = 0.25 + (0.05 - 0.25) * progress
        ratio = 1.0 + eta * (ratio - 1.0)

        af_new = af * np.sqrt(ratio + 1e-12j)
        a_new = np.real(np.fft.ifft(af_new))[:n]
        a_new = sanitize_nonneg(a_new)
        snew = float(a_new.sum())
        if snew > 1e-12:
            a_new = a_new / snew
            b_new = linear_conv(a_new)
            sc_new = 2.0 * n * float(np.max(b_new))
            if np.isfinite(sc_new) and sc_new < score - 1e-14:
                a = a_new

    return best


def try_trim_edges(a):
    a = normalize_nonneg(a).copy()
    if a.size < 4:
        return a

    best = a
    bestsc = compute_score(a)
    max_steps = min(200, a.size // 5)

    for _ in range(max_steps):
        if best.size < 4:
            break

        meanv = float(np.mean(best))
        thresh = max(1e-12, 1e-5 * meanv)

        left_bad = float(best[0]) < thresh
        right_bad = float(best[-1]) < thresh

        if not left_bad and not right_bad:
            break

        if left_bad and right_bad:
            cand = best[1:-1]
        elif left_bad:
            cand = best[1:]
        else:
            cand = best[:-1]

        if cand.size < 4:
            break

        sc = compute_score(cand)
        if np.isfinite(sc) and sc < bestsc - 1e-12:
            best = cand
            bestsc = float(sc)
        else:
            break

    return best


def clean_sequence(a):
    a = normalize_nonneg(a)
    if a.size == 0:
        return a

    cands = [a]
    for eps in (1e-12, 1e-10, 1e-8, 1e-6, 1e-4):
        c = np.where(a < eps, 0.0, a)
        s = float(np.sum(c))
        if s > 1e-12:
            c = c / s
            c = try_trim_edges(c)
            cands.append(c)

    best = cands[0]
    bestsc = compute_score(best)
    for c in cands[1:]:
        sc = compute_score(c)
        if np.isfinite(sc) and sc < bestsc:
            bestsc = float(sc)
            best = c
    return best


def build_lengths(parent, budget):
    cand = []
    if parent is not None:
        p = int(parent.size)
        if 32 <= p <= 20000:

    for n in (2048, 4096, 8192, 16384):
        if n not in cand:
            cand.append(n)

    if budget < 1200.0:
        keep = []
        for n in cand:
            if len(keep) < 2:
                keep.append(n)
        cand = keep

    return cand

def warm_key(item):
    return item[0]


def add_to_warm(warm, a, sc, max_keep):
    if not np.isfinite(sc):
        return
    a = normalize_nonneg(a)
    warm.append((float(sc), a.copy()))
    if len(warm) > max_keep:
        warm.sort(key=warm_key)
        warm.pop()


def register_candidate(best, a, label, score=None):
    a = sanitize_nonneg(a)
    if a.size == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf

    if score is None:
        score = compute_score(a)
    if not np.isfinite(score):
        return np.inf

    if float(score) < best[0] - 1e-12:
        best[0] = float(score)
        best[1] = (a / s).copy()
        print("NEW BEST {:.10f} n={} {}".format(best[0], int(a.size), label))

    return float(score)


def generate_inits(n, parent, rng):
    n = int(n)
    inits = []

    if parent is not None:
        inits.append(resize_nonneg(parent, n))

    inits.append(np.ones(n, dtype=float))

    if n > 1:
        x = np.arange(n, dtype=float)
        u = float(n)

        for f in (1, 2, 3, 5):
            a = 1.0 + 0.55 * np.cos(2.0 * np.pi * f * x / u)
            inits.append(np.clip(a, 0.0, None))

        c = np.cos(np.pi * x * x / (2.0 * n))
        s = np.sin(np.pi * x * x / (2.0 * n))
        inits.append(np.clip(1.0 + 0.7 * c, 0.0, None))
        inits.append(np.clip(1.0 + 0.7 * s, 0.0, None))

        a = np.full(n, 0.40, dtype=float)
        for f in range(1, 8):
            amp = 0.30 / float(f)
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            a += amp * np.cos(2.0 * np.pi * f * x / u + ph)
        inits.append(np.clip(a, 0.0, None))

        nb = max(2, n // 32)
        vals = 0.1 + 0.8 * rng.random(nb)
        a = np.repeat(vals, max(1, n // nb))
        if a.size < n:
            a = np.concatenate((a, vals[:n - a.size]))
        inits.append(np.clip(a[:n], 0.0, None))

        h = max(1, n // 4)
        a = np.zeros(n, dtype=float)
        a[:h] += 1.0
        a[-h:] += 1.0
        inits.append(np.clip(a + 0.02, 0.0, None))

        pat = np.tile(np.array([1.0, 1.0, 0.0, 1.0]), n // 4 + 1)[:n] + 0.05
        inits.append(pat)

    out = []
    for arr in inits:
        arr = normalize_nonneg(arr)
        if arr.size == n:
            out.append(arr)

    if len(out) > 8:
        out = out[:8]
    return out

    )

    weights = [0.45, 0.22, 0.15, 0.10, 0.08]
    betas_main = (300, 800, 2000, 5000, 10000, 4000)

    for i, (sc_top, a_top) in enumerate(tops):
        if time.monotonic() > hard_deadline - 60.0:
            break
        if not np.isfinite(sc_top):
            continue
        if sc_top > best[0] + 0.05:
            continue

        rem = hard_deadline - time.monotonic()
        if rem < 80.0:
            break

        alloc = rem * weights[i] if i < len(weights) else rem * 0.05
        if alloc < 60.0:
            alloc = min(rem, 120.0)
        if alloc > 1000.0:
            alloc = 1000.0

        a_fin, sc_fin = lbfgs_continuation(a_top, sc_top, betas_main, alloc)
        register_candidate(best, a_fin, "top_{}_lbfgs".format(i), sc_fin)

        rem2 = hard_deadline - time.monotonic()
        if rem2 > 60.0:
            a_sp = spectral_sqrt_flatten(a_fin, min(80.0, 0.15 * rem2), max_iter=120)
            sc_sp = compute_score(a_sp)
            register_candidate(best, a_sp, "top_{}_spec".format(i), sc_sp)

            if np.isfinite(sc_sp) and sc_sp < sc_fin:
                a_use = a_sp
                sc_use = sc_sp
            else:
                a_use = a_fin
                sc_use = sc_fin
        else:
            a_use = a_fin
            sc_use = sc_fin

        rem3 = hard_deadline - time.monotonic()
        if rem3 > 45.0:
            alloc2 = min(120.0, 0.25 * rem3)
            a_fin2, sc_fin2 = lbfgs_continuation(a_use, sc_use, (8000, 20000, 6000), alloc2)
            register_candidate(best, a_fin2, "top_{}_final".format(i), sc_fin2)

    if best[1] is not None:
        rem = hard_deadline - time.monotonic()
        if rem > 30.0:
            a_clean = clean_sequence(best[1])
            register_candidate(best, a_clean, "best_clean", None)

        rem = hard_deadline - time.monotonic()
        if rem > 60.0:
            alloc = min(200.0, 0.35 * rem)
            a_final, sc_final = lbfgs_continuation(best[1], best[0], (10000, 25000, 8000), alloc)
            register_candidate(best, a_final, "final_lbfgs", sc_final)

        if best[1] is not None:
            a_final = clean_sequence(best[1])
            register_candidate(best, a_final, "final_clean", None)

    # Final selection according to the actual oracle, if available.
    final_candidates = []
    if best[1] is not None:
        final_candidates.append(best[1])
        final_candidates.append(clean_sequence(best[1]))
    if parent is not None:
        final_candidates.append(normalize_nonneg(parent))
    final_candidates.append(np.ones(1024, dtype=float) / 1024.0)

    end = t0 + budget - 5.0
    best_eval = np.inf
    best_out = None

    evaluated_count = 0
    for cand in final_candidates:
        if time.monotonic() > end or evaluated_count >= 4:
            break

        cand = normalize_nonneg(cand)
        if cand.size == 0:
            continue

        out = [float(v) for v in cand]
        try:
            scv = float(evaluate_sequence(out))
        except Exception:
            scv = np.inf

        evaluated_count += 1
        if np.isfinite(scv) and scv < best_eval - 1e-12:
            best_eval = float(scv)
            best_out = out

    if best_out is None:
        out = [float(v) for v in (np.ones(1024, dtype=float) / 1024.0)]
        best_out = out

    total = time.monotonic() - t0
    print(
        "FINAL internal={:.10f} oracle={:.10f} n={} time={:.1f}s".format(
            float(best[0]) if best[1] is not None else float("inf"),
            float(best_eval) if np.isfinite(best_eval) else float("inf"),
            len(best_out),
            total,
        )
    )

    return best_out
```
