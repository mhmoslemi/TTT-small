"""
Step function optimization: Multi-strategy approach combining LSE continuation with Adam,
Nesterov accelerated subgradient, peak-targeted coordinate descent, and structured restarts.
Key improvements over prior approach: Adam-based LSE optimization for better landscape navigation,
Nesterov momentum for faster subgradient convergence, and peak-specific coordinate updates.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(a, beta):
    """Objective wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def variance_score_grad(a):
    """Score and gradient for 2n*sum(c^2)/S^4 (flatness objective)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c2 = float(np.sum(c * c))
    score = 2.0 * n * c2 / (S ** 4)
    L = 1 << (2 * n - 2).bit_length()
    A = np.fft.fft(a, L)
    E = np.abs(A) ** 2 * np.conj(A)
    E_ifft = np.fft.ifft(E).real
    idx = np.concatenate([[0], np.arange(L - 1, L - n, -1)])
    dc2_da = 4.0 * E_ifft[idx]
    S4 = S ** 4
    S5 = S4 * S
    grad = 2.0 * n * (dc2_da / S4 - 4.0 * c2 / S5)
    return score, grad


def variance_obj(a):
    """Objective wrapper for variance minimization."""
    return variance_score_grad(a)


def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    """Run L-BFGS-B optimization. Returns optimized array."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def adam_lse_optimize(a, beta, n_steps, lr_base=0.01, lr_decay=0.92, decay_every=200):
    """Adam optimizer for LSE objective with non-negative projection."""
    n = len(a)
    a = a.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_sc = compute_score(a)
    for t in range(n_steps):

        sc, g = lse_score_grad(a, beta)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        m_hat = m / (1.0 - b1 ** (t + 1))

    sc_final = compute_score(a)

def nesterov_subgrad(a, iters=5000, verbose=False):

    n = len(a)
    a = a.astype(np.float64).copy()

    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    gamma = 2.0 / 3.0
    for it in range(iters):
        a_k = (a + v) / (1.0 + gamma)
        c = fast_conv(a_k)
        S = float(a_k.sum())
        if S < 1e-12:

        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a_k.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a_k[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        t_k = 2.0 / (gamma * (it + 2))
        step = (sc - sc_lb) / (gn * gn + 1e-20)
        step = min(step, 1.0 / (S + 1e-10))
        step = max(step, 1e-8 / (gn * S + 1e-10))
        a_new = a_k - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)

        a = a_new

        best_sc = sc_final
        best_a = a.copy()
    return best_sc, best_a


def peak_coord_descent(a, n_peaks=5, iters_per_peak=200, verbose=False):
    """Coordinate descent targeting the top peaks simultaneously."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(iters_per_peak):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        sc = 2.0 * n * float(c.max()) / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Find top peaks (excluding center which is hard to reduce)
        conv_no_center = c.copy()
        conv_no_center[max(0, n - 5):min(2 * n - 1, n + 5)] = -np.inf
        top_k = np.argsort(conv_no_center)[::-1][:n_peaks]
        improved = False
        for k_star in top_k:
            k_star = int(k_star)
            if c[k_star] < 0:
                continue
            M_peak = float(c[k_star])
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end < j_start:
                continue
            # Elements contributing to this peak
            contrib_idx = list(range(j_start, j_end + 1))
            contrib_vals = a[contrib_idx]
            partner_vals = a[j_end - np.array(contrib_idx)]
            # Reduce the largest contributors proportionally
            order = np.argsort(contrib_vals * np.abs(partner_vals))[::-1]
            n_reduce = max(1, len(contrib_idx) // 10)
            total_reduce = 0.0
            for idx in order[:n_reduce]:
                j = contrib_idx[idx]
                reduce_amt = contrib_vals[idx] * 0.02
                a[j] -= reduce_amt
                a[j] = max(a[j], 0.0)
                total_reduce += reduce_amt
            # Redistribute to random low-contribution positions
            if total_reduce > 0:
                low_mask = a < np.median(a)
                low_idx = np.where(low_mask)[0]
                if len(low_idx) > 0:
                    n_add = min(len(low_idx), max(5, n // 50))
                    add_idx = rng_choice(low_idx, n_add)
                    per_add = total_reduce / n_add
                    for j in add_idx:
                        a[j] += per_add
                    improved = True
        a = np.clip(a, 0.0, 1000.0)
        if not improved:
            break
    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_sc, best_a


def rng_choice(arr, k):
    """Randomly choose k elements from arr without replacement."""
    k = min(k, len(arr))
    idx = np.random.default_rng(int(time.time() * 1000) % 100000).choice(len(arr), k, replace=False)
    return arr[idx]


def targeted_perturb(a, rng, strength=0.05):
    """Reduce elements contributing most to the peak, redistribute to others."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:max(5, n - top_count)]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def multi_peak_shave(a, iters=3000, n_peaks=3):
    """Shave multiple peaks iteratively with adaptive step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Process top peaks
        sorted_idx = np.argsort(c)[::-1][:n_peaks]
        for k_star in sorted_idx:
            k_star = int(k_star)
            M_k = float(c[k_star])
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            sg = np.zeros(n)
            if j_end >= j_start:
                sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
            S2 = S * S
            S3 = S2 * S
            grad = 2.0 * n * (sg / S2 - 2.0 * M_k / S3)
            gn = float(np.linalg.norm(grad))
            if gn < 1e-15:
                continue
            step = 0.02 / (gn * S + 1e-20)
            step = min(step, 0.5 / S)
            a -= step * grad
            a = np.clip(a, 0.0, 1000.0)
    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses Adam-based LSE, Nesterov subgradient, peak coordination, and restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Initialize from parent
    if parent is not None:
        sc = compute_score(parent)
        report(sc, parent, "parent")
        print(f"Parent score: {sc:.8f}, length={len(parent)}")
    else:
        print("No parent found, using random init")
        # Generate a good initial sequence: cosine-tapered
        n_init = 4096
        x = np.arange(n_init) / (n_init - 1)
        best_a = 0.5 + 0.5 * np.cos(np.pi * x)
        best_a = best_a + 0.01 * rng.standard_normal(n_init)
        best_a = np.clip(best_a, 0.01, None)
        sc = compute_score(best_a)
        report(sc, best_a, "init")

    n_use = len(best_a)
    print(f"Working with n = {n_use}, initial score = {best_score:.8f}")

    # ===================================================================
    # Phase 1: Variance flattening - 8% time
    # ===================================================================
    phase1_end = t_start + 0.08 * (total_budget - 30.0)
    print("\n=== Phase 1: Variance Flattening ===")
    a_cur = best_a.copy()
    if time.monotonic() < phase1_end:
        t0 = time.monotonic()
        a_cur = run_lbfgs(a_cur, variance_obj, (), maxiter=300, maxfun=10000)
        sc = compute_score(a_cur)
        report(sc, a_cur, "variance")
        print(f"  Phase 1 done: score={sc:.8f}, time={time.monotonic()-t0:.1f}s")

    # ===================================================================
    # Phase 2: LSE Continuation with L-BFGS-B - 25% time
    # ===================================================================
    phase2_end = t_start + 0.33 * (total_budget - 30.0)
    print("\n=== Phase 2: LSE Continuation (L-BFGS-B) ===")

    a_cur = best_a.copy()
    beta_schedule = [15, 30, 60, 120, 250, 500, 1000, 2000, 4000, 8000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        t0 = time.monotonic()
        prev_sc = compute_score(a_cur)
        a_cur = run_lbfgs(a_cur, lse_obj, (beta,), maxiter=800, maxfun=20000)
        sc = compute_score(a_cur)
        report(sc, a_cur, f"LSE-{beta}")
        improvement = prev_sc - sc
        print(f"  beta={beta}: score={sc:.8f}, imp={improvement:.2e}, time={time.monotonic()-t0:.1f}s")

    # ===================================================================
    # Phase 3: LSE with Adam (different optimizer for landscape navigation) - 25% time
    # ===================================================================
    phase3_end = t_start + 0.58 * (total_budget - 30.0)
    print("\n=== Phase 3: LSE with Adam ===")

    a_cur = best_a.copy()
    adam_betas = [500, 1000, 2000, 4000, 8000, 15000]
    for beta in adam_betas:
        if time.monotonic() > phase3_end:
            break
        t0 = time.monotonic()
        prev_sc = compute_score(a_cur)
        n_steps = int(min(3000, max(500, (phase3_end - time.monotonic()) * 5)))
        sc, a_adam = adam_lse_optimize(a_cur, beta, n_steps, lr_base=0.005, lr_decay=0.90, decay_every=300)
        if sc < compute_score(a_cur):
            a_cur = a_adam
        report(sc, a_cur, f"Adam-{beta}")
        print(f"  Adam beta={beta}: score={sc:.8f}, time={time.monotonic()-t0:.1f}s")

    # ===================================================================
    # Phase 4: Nesterov subgradient on exact objective - 15% time
    # ===================================================================
    phase4_end = t_start + 0.73 * (total_budget - 30.0)
    print("\n=== Phase 4: Nesterov Subgradient ===")

    if time.monotonic() < phase4_end:
        t0 = time.monotonic()
        iters = int(min(8000, max(2000, (phase4_end - time.monotonic()) * 3)))
        sc, a_nest = nesterov_subgrad(best_a, iters=iters)
        report(sc, a_nest, "nesterov")
        a_cur = a_nest if sc < compute_score(a_cur) else a_cur
        print(f"  Nesterov done: score={sc:.8f}, time={time.monotonic()-t0:.1f}s")

    # ===================================================================
    # Phase 5: Multi-peak shaving - 10% time
    # ===================================================================
    phase5_end = t_start + 0.83 * (total_budget - 30.0)
    print("\n=== Phase 5: Multi-Peak Shaving ===")

    if time.monotonic() < phase5_end:
        t0 = time.monotonic()
        iters = int(min(3000, max(500, (phase5_end - time.monotonic()) * 2)))
        sc, a_mp = multi_peak_shave(best_a, iters=iters, n_peaks=3)
        report(sc, a_mp, "multi-peak")
        a_cur = a_mp if sc < compute_score(a_cur) else a_cur
        print(f"  Multi-peak done: score={sc:.8f}, time={time.monotonic()-t0:.1f}s")

    # ===================================================================
    # Phase 6: Peak coordinate descent - 7% time
    # ===================================================================
    phase6_end = t_start + 0.90 * (total_budget - 30.0)
    print("\n=== Phase 6: Peak Coordinate Descent ===")

    if time.monotonic() < phase6_end:
        t0 = time.monotonic()
        sc, a_pcd = peak_coord_descent(best_a, n_peaks=5, iters_per_peak=500)
        report(sc, a_pcd, "coord-desc")
        a_cur = a_pcd if sc < compute_score(a_cur) else a_cur
        print(f"  Coord descent done: score={sc:.8f}, time={time.monotonic()-t0:.1f}s")

    # ===================================================================
    # Phase 7: Structured restarts - 8% time
    # ===================================================================
    phase7_end = deadline
    print("\n=== Phase 7: Structured Restarts ===")

    n_best = len(best_a)
    restart_count = 0
    while time.monotonic() < phase7_end and restart_count < 15:
        t0 = time.monotonic()
        strategy = restart_count % 5
        if strategy == 0:
            # Small Gaussian perturbation
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted perturbation
            a_pert = targeted_perturb(best_a, rng, strength=0.02 + 0.04 * rng.random())
        elif strategy == 2:
            # Low-frequency sinusoidal modulation
            freq = 0.5 + 3.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.02 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # High-frequency modulation
            freq = 5.0 + 20.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.005 + 0.015 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.sin(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Block swap: exchange two random blocks
            a_pert = best_a.copy()
            blk_size = max(32, n_best // 64)
            i = int(rng.integers(0, n_best - 2 * blk_size))
            j = int(rng.integers(0, n_best - 2 * blk_size))
            if abs(i - j) > blk_size:
                temp = a_pert[i:i + blk_size].copy()
                a_pert[i:i + blk_size] = a_pert[j:j + blk_size] * 0.9
                a_pert[j:j + blk_size] = temp * 0.9
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick optimization from perturbed start: try both LSE and Adam
        a_opt1 = run_lbfgs(a_pert, lse_obj, (3000,), maxiter=400, maxfun=8000)
        sc1 = compute_score(a_opt1)
        sc2, a_opt2 = adam_lse_optimize(a_pert, 5000, n_steps=1000, lr_base=0.003, lr_decay=0.90, decay_every=200)
        if sc2 < sc1:
            a_opt = a_opt2
            sc = sc2
        else:
            a_opt = a_opt1
            sc = sc1
        report(sc, a_opt, f"restart#{restart_count}(s{strategy})")
        restart_count += 1
        elapsed_restart = time.monotonic() - t0
        print(f"  Restart {restart_count}: score={sc:.8f}, time={elapsed_restart:.1f}s")

    # ===================================================================
    # Phase 8: Final polish - remaining time
    # ===================================================================
    print("\n=== Phase 8: Final Polish ===")
    if time.monotonic() < deadline:
        # High-beta LSE
        remaining = max(0, deadline - time.monotonic())
        a_final = run_lbfgs(best_a, lse_obj, (20000,), maxiter=int(min(2000, remaining * 5)), maxfun=int(min(50000, remaining * 100)))
        sc = compute_score(a_final)
        report(sc, a_final, "final-LSE")

        # More Nesterov
        if time.monotonic() < deadline - 10:
            iters = int(max(2000, (deadline - time.monotonic() - 5) * 2))
            sc, a_nest2 = nesterov_subgrad(best_a, iters=iters)
            report(sc, a_nest2, "final-nesterov")

        # Final multi-peak
        if time.monotonic() < deadline - 5:
            sc, a_mp2 = multi_peak_shave(best_a, iters=2000, n_peaks=5)
            report(sc, a_mp2, "final-multipick")

    # ===================================================================
    # Final output
    # ===================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    # Verify
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
