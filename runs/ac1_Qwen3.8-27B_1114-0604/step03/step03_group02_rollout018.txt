    exp_s = np.exp(scaled)
    logsum = np.log(exp_s.sum())
    L = M + logsum / beta
    score = 2.0 * n * L / (S*S)
    p = exp_s / exp_s.sum()

    S3 = S2*S
    grad = (2.0 * n) * (g/S2 - 2.0*L/S3)

def compute_h(a, w, M_fft):
    """Compute h_j = 2 * sum_k w_k * a_{k-j} (cross-correlation of w and a, lags 0..n-1)."""
    n = len(a)
    W = np.fft.rfft(w, M_fft)
    A = np.fft.rfft(a, M_fft)
    cross = np.fft.irfft(W * np.conj(A), M_fft)
    return 2.0 * cross[:n]


def softmax_multiplicative_update(a, beta_w=50.0, alpha=0.5, n_iters=500, deadline=None, verbose=False):
    """
    Softmax-weighted multiplicative update with adaptive step size.
    a_j <- a_j * (h_mean / h_j)^alpha, renormalized to preserve sum.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (3 * n - 2).bit_length()
    stagnant = 0
    best_alpha = alpha

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        if M < 1e-15:
            break

        w = softmax_weights(c, beta_w)
        h = compute_h(a, w, M_fft)
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()

        factor = (h_mean / h) ** alpha
        a_new = a * factor
        a_new = np.maximum(a_new, 1e-10)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new * (S / S_new)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            stagnant = 0
            # Increase alpha slightly if improving
            best_alpha = min(1.0, alpha * 1.05)
        else:
            stagnant += 1
            # Decrease alpha if stagnating
            alpha = max(0.02, alpha * 0.9)

        a = a_new
        if verbose and it % 100 == 0:
            print(f"      iter {it}: score={sc:.8f}, best={best_sc:.8f}, alpha={alpha:.4f}")

        if stagnant > 200:
            break

    return best_sc, best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    M = float(c.max())
    scaled = beta * (c - M)
    exp_s = np.exp(scaled)
    logsum = np.log(exp_s.sum())
    L = M + logsum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        sc, g = lse_score_grad(x, beta)
        return sc, g

    options = {'ftol': 1e-16, 'gtol': 1e-14, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}

    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            sc, _ = lse_score_grad(a0, beta)
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(100, int(remaining * 400)))
        options['maxiter'] = min(maxiter, max(100, int(remaining * 100)))

    try:
        res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0

    return compute_score(a_out), a_out


def nesterov_accelerated(a0, beta, n_iters=3000, deadline=None, verbose=False):
    """Nesterov-accelerated proximal gradient on LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    prev_a = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    # Estimate Lipschitz constant
    S = a.sum()
    max_a = float(np.max(a))
    L = (4.0 * n * max_a) / (S * S) + (4.0 * n * 2.0 * (S / n)) / (S * S * S) + 1e-8
    step = 1.0 / L
    best_step = step

    t_val = 1.0
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        theta = (t_val - 1.0) / t_new
        y = a + theta * (a - prev_a)
        y = np.maximum(y, 0.0)

        sc_y, grad = lse_score_grad(y, beta)

        # Gradient step with backtracking
        a_new = y - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = np.minimum(a_new, 1000.0)

        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            best_step = step
        else:
            # Backtrack
            step = step * 0.7
            a_new = y - step * grad
            a_new = np.maximum(a_new, 0.0)

        prev_a = a
        a = a_new
        t_val = t_new

        # Periodically re-estimate Lipschitz
        if it % 500 == 0 and it > 0:
            S = a.sum()
            max_a = float(np.max(a))
            L = (4.0 * n * max_a) / (S * S) + (4.0 * n * 2.0 * (S / n)) / (S * S * S) + 1e-8
            step = max(step, 0.5 / L)

        if verbose and it % 500 == 0:
            print(f"      Nesterov iter {it}: score={sc_new:.8f}, best={best_sc:.8f}, step={step:.6e}")

    return best_sc, best_a


def targeted_rebalance(a, strength=0.03, n_lags=5):
    """Redistribute mass away from positions contributing to top convolution peaks."""
    n = len(a)
    c = fast_conv(a)
    # Find top n_lags peaks
    top_lags = np.argsort(c)[::-1][:n_lags]

    contrib = np.zeros(n)
    for k in top_lags:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]

    n_top = max(20, n // 200)
    top_idx = np.argsort(contrib)[::-1][:n_top]
    bottom_idx = np.argsort(contrib)[:n_top]

    a_new = a.copy()
    total_move = 0.0
    for j in top_idx:
        move = a_new[j] * strength
        a_new[j] -= move
        a_new[j] = max(a_new[j], 0.0)
        total_move += move

    if total_move > 0 and len(bottom_idx) > 0:
        per = total_move / len(bottom_idx)
        for j in bottom_idx:
            a_new[j] += per

    return np.clip(a_new, 1e-10, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 10.0

    # Load parent solution
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(8192) * 0.5
        print("No parent available, starting from uniform.")
    else:
        print(f"Parent loaded: length={len(parent)}")

    n = len(parent)
    best_score = compute_score(parent)
    best_a = parent.copy()
    print(f"Initial score: {best_score:.8f} (n={n})")

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {best_score:.8f} (t={elapsed:.0f}s)")
            return True
        return False

    # ==================================================================
    # Phase 1: Prolonged softmax multiplicative update with beta continuation
    # ==================================================================
    t_p1 = t_start + 0.55 * budget_s
    print(f"\n{'='*60}")
    print(f"Phase 1: Softmax multiplicative update (deadline t={t_p1 - t_start:.0f}s)")
    print(f"{'='*60}")

    # Beta continuation: start high (focus on peak), decrease to smooth
    beta_schedule = [100, 80, 60, 40, 30, 20]
    for bi, beta_w in enumerate(beta_schedule):
        if time.monotonic() > t_p1 - 30:
            break
        remaining_iters = max(50, int((t_p1 - time.monotonic()) / (len(beta_schedule) - bi) / 0.002))
        sc, a_new = softmax_multiplicative_update(
            best_a, beta_w=beta_w, alpha=0.5,
            n_iters=remaining_iters, deadline=t_p1 - 5
        )
        if try_update(sc, a_new, f"phase1 beta_w={beta_w}"):
            pass

    # Fine-tuning passes with small alpha
    for alpha in [0.3, 0.2, 0.1]:
        if time.monotonic() > t_p1 - 10:
            break
        sc, a_new = softmax_multiplicative_update(
            best_a, beta_w=40, alpha=alpha,
            n_iters=300, deadline=t_p1 - 3
        )
        if try_update(sc, a_new, f"phase1 fine alpha={alpha}"):
            pass

    print(f"After Phase 1: score={best_score:.8f}")

    # ==================================================================
    # Phase 2: L-BFGS LSE polishing with continuation
    # ==================================================================
    t_p2 = t_start + 0.75 * budget_s
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS LSE polishing (deadline t={t_p2 - t_start:.0f}s)")
    print(f"{'='*60}")

    for beta in [300, 500, 800, 1500, 3000]:
        if time.monotonic() > t_p2 - 20:
            break
        sc, a_new = run_lbfgs(best_a, beta, maxiter=8000, maxfun=150000, deadline=t_p2 - 5)
        if try_update(sc, a_new, f"phase2 lbfgs beta={beta}"):
            pass

    # ==================================================================
    # Phase 3: Nesterov + alternating multiplicative/L-BFGS
    # ==================================================================
    t_p3 = t_start + 0.88 * budget_s
    print(f"\n{'='*60}")
    print(f"Phase 3: Nesterov + alternating (deadline t={t_p3 - t_start:.0f}s)")
    print(f"{'='*60}")

    # Nesterov pass
    if time.monotonic() < t_p3 - 30:
        sc, a_new = nesterov_accelerated(best_a, beta=800, n_iters=2000, deadline=t_p3 - 30)
        try_update(sc, a_new, "phase3 nesterov")

    # Alternating: multiplicative then L-BFGS, repeated
    for round_i in range(6):
        if time.monotonic() > t_p3 - 10:
            break
        # Multiplicative
        sc, a_new = softmax_multiplicative_update(
            best_a, beta_w=60, alpha=0.3, n_iters=200, deadline=t_p3 - 15
        )
        try_update(sc, a_new, f"phase3 alt-round {round_i} multiplicative")

        # L-BFGS
        if time.monotonic() < t_p3 - 10:
            sc, a_new = run_lbfgs(best_a, beta=1000, maxiter=3000, maxfun=50000, deadline=t_p3 - 5)
            try_update(sc, a_new, f"phase3 alt-round {round_i} lbfgs")

    print(f"After Phase 3: score={best_score:.8f}")

    # ==================================================================
    # Phase 4: Local search with diverse perturbations
    # ==================================================================
    t_p4 = deadline - 20
    print(f"\n{'='*60}")
    print(f"Phase 4: Local search (deadline t={t_p4 - t_start:.0f}s)")
    print(f"{'='*60}")

    n_local = max(10, int((t_p4 - time.monotonic()) / 30))
    for trial in range(n_local):
        if time.monotonic() > t_p4:
            break
        strategy = trial % 5
        if strategy == 0:
            # Targeted rebalance
            a_pert = targeted_rebalance(best_a, strength=0.02 + 0.04 * rng.random(), n_lags=3)
        elif strategy == 1:
            # Gaussian perturbation
            scale = 0.01 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n))
            a_pert = np.clip(a_pert, 1e-10, 1000.0)
        elif strategy == 2:
            # Sinusoidal perturbation
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.04 * rng.random()
            x = np.arange(n, dtype=np.float64) / n
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 1e-10, 1000.0)
        elif strategy == 3:
            # Edge modification
            a_pert = best_a.copy()
            edge = max(32, n // 8)
            a_pert[:edge] = np.clip(a_pert[:edge] * (1.0 + 0.15 * rng.standard_normal(edge)), 1e-10, 1000.0)
            a_pert[-edge:] = np.clip(a_pert[-edge:] * (1.0 + 0.15 * rng.standard_normal(edge)), 1e-10, 1000.0)
        else:
            # Multi-frequency perturbation
            a_pert = best_a.copy()
            for _ in range(3):
                freq = 0.5 + 15.0 * rng.random()
                phase = 2.0 * np.pi * rng.random()
                amp = 0.005 + 0.02 * rng.random()
                x = np.arange(n, dtype=np.float64) / n
                a_pert *= (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 1e-10, 1000.0)

        # Refine with L-BFGS
        sc, a_new = run_lbfgs(a_pert, beta=1000, maxiter=3000, maxfun=50000, deadline=t_p4)
        if sc < best_score:
            try_update(sc, a_new, f"phase4 trial {trial} strat={strategy}")

    # ==================================================================
    # Phase 5: Final intensive polishing
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polishing")
    print(f"{'='*60}")

    # Very high beta L-BFGS
    for beta in [2000, 5000, 10000]:
        if time.monotonic() > deadline - 5:
            break
        sc, a_new = run_lbfgs(best_a, beta, maxiter=5000, maxfun=80000, deadline=deadline - 3)
        if try_update(sc, a_new, f"final lbfgs beta={beta}"):
            pass

    # Final multiplicative micro-adjustments
    if time.monotonic() < deadline - 3:
        for alpha in [0.1, 0.05, 0.02]:
            if time.monotonic() > deadline - 2:
                break
            sc, a_new = softmax_multiplicative_update(
                best_a, beta_w=80, alpha=alpha, n_iters=200, deadline=deadline - 1
            )
            if try_update(sc, a_new, f"final mult alpha={alpha}"):
                pass

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
