"""
Multi-phase optimizer: Mirror Descent + Block Coordinate Descent + Active Set
for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key improvements over previous: entropic mirror descent, block-level optimization,
adaptive LSE sharpening, and active-set-guided perturbations.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_smooth(a, beta):
    """Compute LSE-smoothed max of convolution."""
    c = fast_conv(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    return c_max + np.log(exp_scaled.sum()) / beta, exp_scaled / exp_scaled.sum()


def lse_grad_a(a, beta):
    """Gradient of LSE-smoothed score w.r.t. a (normalized, sum=1)."""
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    L = c_max + np.log(exp_scaled.sum()) / beta
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a[i] = sum_k p[k] * d(c[k])/d(a[i])
    # d(c[k])/d(a[i]) = a[k-i] (with bounds)
    # Using FFT: grad[i] = (p * a_rev)[n-1+i] for i in [0, n-1]
    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    # Score = 2n * L / S^2, with S=1 (normalized), so score = 2n*L
    # grad_score = 2n * conv_res (when S=1)
    grad = 2.0 * n * conv_res
    return 2.0 * n * L, grad


def subgrad_score_a(a, top_k=1):
    """Exact score and (averaged) subgradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    score = 2.0 * n * M / (S * S)
    
    # Get top-k lags
    sorted_idx = np.argsort(c)[::-1][:top_k]
    # Weights: exponential of (c[k] - c_max)
    weights = np.exp(c[sorted_idx] - c[sorted_idx[0]])
    weights /= weights.sum()
    
    sg = np.zeros(n)
    for pos, k in enumerate(sorted_idx):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            idx = k - np.arange(js, je + 1)
            sg[js:je + 1] += weights[pos] * a[idx]
    
    S2 = S * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / (S2 * S))
    return score, grad


def mirror_descent(a_init, n_steps=50000, beta_init=50, beta_final=5000,
                   eta_init=0.1, time_limit_s=None, t_start=None,
                   verbose_interval=5000):
    """
    Mirror descent with LSE smoothing and adaptive beta.
    Operates on the simplex (sum(a)=1, a>=0).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a /= S
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        # Adaptive beta: linearly interpolate (in log space)
        log_beta = np.log(beta_init) + (np.log(beta_final) - np.log(beta_init)) * min(1.0, t / (n_steps * 0.8))
        beta = np.exp(log_beta)
        
        # Adaptive eta: decreasing
        eta = eta_init / (1.0 + 0.01 * t)
        
        # Compute gradient
        _, grad = lse_grad_a(a, beta)
        
        # Mirror descent step: a_new[i] = a[i] * exp(-eta * grad[i]) / Z
        log_a = np.log(a + 1e-300)
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()  # numerical stability
        a_new = np.exp(log_a_new)
        a_new /= a_new.sum()
        
        # Check for improvement
        if t % 100 == 0 or t == 1:
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                if t % verbose_interval == 0:
                    elapsed = time.monotonic() - t_start if t_start else 0
                    print(f"    MD t={t}: score={sc:.8f} beta={beta:.0f} eta={eta:.4f} t={elapsed:.0f}s")
        
        a = a_new
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


def block_coordinate_descent(a, block_size=64, n_sweeps=50, time_limit_s=None, t_start=None,
                             verbose_interval=10):
    """
    Block coordinate descent: optimize one block at a time using L-BFGS.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf
    
    n_blocks = n // block_size
    remainder = n - n_blocks * block_size
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for b in range(n_blocks + (1 if remainder > 0 else 0)):
            if time.monotonic() > deadline:
                break
            
            if b < n_blocks:
                start = b * block_size
                end = start + block_size
            else:
                start = n_blocks * block_size
                end = n
            
            if end - start < 4:
                continue
            
            # Optimize this block
            block = a[start:end].copy()
            block_sum = block.sum()
            
            def block_obj(block_vals, a_rest=a, start=start, end=end, n=n):
                a_full = a_rest.copy()
                a_full[start:end] = np.clip(block_vals, 0, 1000)
                return compute_score(a_full), None
            
            def block_obj_grad(block_vals, a_rest=a, start=start, end=end, n=n):
                a_full = a_rest.copy()
                a_full[start:end] = np.clip(block_vals, 0, 1000)
                sc, grad = lse_grad_a(a_full, 500)
                return sc, grad[start:end]
            
            # Use L-BFGS on the block
            res = minimize(
                block_obj_grad, block,
                method='L-BFGS-B', jac=True,
                bounds=[(0, 1000)] * (end - start),
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8, 'maxfun': 5000}
            )
            
            new_block = np.clip(res.x, 0, 1000)
            a[start:end] = new_block
            improved = True
        
        # Check score at end of sweep

def active_set_perturbation(a, n_perturbations=20, time_limit_s=None, t_start=None):

def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)

def projected_subgrad_descent(a_init, n_steps=100000, top_k=5, 
                               time_limit_s=None, t_start=None):
    """
    Projected subgradient descent on the simplex with multi-active-set.
    Uses backtracking line search.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0/n
    else:
        a /= S
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    # Initial step size
    eta = 1.0 / np.sqrt(n)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        # Compute multi-subgradient
        c = fast_conv(a)
        c_max = c.max()
        S = a.sum()
        score = 2.0 * n * c_max / (S*S)
        
        # Top-k lags
        top_k_actual = min(top_k, 2*n-1)
        sorted_idx = np.argsort(c)[::-1][:top_k_actual]
        # Exponential weights
        weights = np.exp((c[sorted_idx] - c_max) * 10)
        weights /= weights.sum()
        
        # Compute averaged gradient
        grad = np.zeros(n)
        for pos, k in enumerate(sorted_idx):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                idx = k - np.arange(js, je + 1)
                grad[js:je+1] += weights[pos] * a[idx]
        
        # Gradient of score = 2n * (grad/S^2 - 2*c_max*S_grad/S^3)
        # When S=1: grad_score = 2n * grad (approximately)
        # More precisely: d(score)/d(a[i]) = 2n * (gradient_of_max / S^2 - 2*max*gradient_of_S / S^3)
        # gradient_of_S[i] = 1 for all i
        # So: d(score)/d(a[i]) = 2n * (grad[i] - 2*c_max/S) / S^2 ... 
        # Actually: score = 2n * M / S^2 where M = max(c)
        # d(score)/d(a[i]) = 2n * (dM/d(a[i]) * S^2 - M * 2*S * dS/d(a[i])) / S^4
        # = 2n * (dM/d(a[i]) / S^2 - 2*M / S^3)
        # = 2n/S^2 * (dM/d(a[i]) - 2*M/S)
        
        S2 = S * S
        grad_score = 2.0 * n / S2 * (grad - 2.0 * c_max / S)
        
        # Projected gradient step: a_new = project(a - eta * grad_score)
        # Project onto simplex: a >= 0, sum(a) = 1
        a_unproj = a - eta * grad_score
        
        # Projection onto simplex (Duchi et al.)
        a_proj = project_simplex(a_unproj)
        
        # Backtracking line search
        alpha = 1.0
        sc_new = compute_score(a_proj)
        while sc_new > score and alpha > 1e-6:
            alpha *= 0.5
            a_proj = project_simplex(a - alpha * eta * grad_score)
            sc_new = compute_score(a_proj)
        
        a = a_proj
        eta = eta / (1 + 1e-4 * t)  # Decrease step size
        
        if t % 100 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_score, best_a


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu) - 1
    rho = np.nonzero(mu * np.arange(1, n+1) > cv)[0][-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""

block coordinate descent, and active-set-guided perturbations for minimizing
2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key improvements: proper simplex projection, multi-active-set subgradients,
backtracking line search, entropic mirror descent, and block-level L-BFGS.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu) - 1.0
    rho_candidates = np.arange(1, n + 1)
    valid = (mu * rho_candidates) > cv
    if not np.any(valid):
        return np.ones(n) / n
    rho = int(np.max(rho_candidates[valid])) - 1
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_grad_a(a, beta):
    """
    LSE-smoothed score and gradient w.r.t. a.
    Returns (score, grad) where grad = d(score)/d(a).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p_sum = float(exp_scaled.sum())
    if p_sum < 1e-300:
        p = np.zeros_like(c)
        p[int(c.argmax())] = 1.0
        L = c_max
    else:
        L = c_max + np.log(p_sum) / beta
        p = exp_scaled / p_sum
    # conv_res[i] = sum_k p[k] * a[k-i] (circular with proper bounds)
    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = 2.0 * n * (conv_res / S2 - 2.0 * L / S3)
    return score, grad


def multi_subgrad(a, top_k=5):
    """
    Compute score and multi-active-set subgradient.
    Returns (score, grad) where grad is the averaged subgradient over top-k lags.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c_max = float(c.max())
    score = 2.0 * n * c_max / (S * S)

    k_actual = min(top_k, 2 * n - 1)
    sorted_idx = np.argsort(c)[::-1][:k_actual]
    # Exponential weights emphasizing the peak
    weights = np.exp((c[sorted_idx] - c_max) * 20.0)
    w_sum = float(weights.sum())
    if w_sum < 1e-300:
        weights = np.zeros(k_actual)
        weights[0] = 1.0
    else:
        weights /= w_sum

    sg = np.zeros(n)
    for pos in range(k_actual):
        k = int(sorted_idx[pos])
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            idx = k - np.arange(js, je + 1)
            sg[js:je + 1] += weights[pos] * a[idx]

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * c_max / S3)
    return score, grad


def projected_subgrad_descent(a_init, n_steps=100000, top_k=10,
                              eta_init=None, time_limit_s=None, t_start=None,
                              verbose_interval=5000):
    """
    Projected subgradient descent on simplex with multi-active-set and backtracking.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S

    if eta_init is None:
        eta = 2.0 / (n ** 0.55)
    else:
        eta = eta_init

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break

        score, grad = multi_subgrad(a, top_k=top_k)

        # Backtracking line search
        alpha = 1.0
        a_new = project_simplex(a - alpha * eta * grad)
        sc_new = compute_score(a_new)
        backtrack_count = 0
        while sc_new > score * (1.0 - 1e-10) and alpha > 1e-8 and backtrack_count < 15:
            alpha *= 0.5
            a_new = project_simplex(a - alpha * eta * grad)
            sc_new = compute_score(a_new)
            backtrack_count += 1

        a = a_new

        # Adaptive step size
        eta = eta_init / (1.0 + 1e-4 * t) if eta_init is not None else eta / (1.0 + 1e-5 * t)

        if t % 100 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            if t % verbose_interval == 0 and t_start is not None:
                elapsed = time.monotonic() - t_start
                print(f"    PSGD t={t}: score={score:.8f} eta={eta:.6f} bt={backtrack_count} t={elapsed:.0f}s")

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def mirror_descent(a_init, n_steps=50000, beta_init=50, beta_final=10000,
                   eta_init=0.1, time_limit_s=None, t_start=None,
                   verbose_interval=5000):
    """
    Entropic mirror descent with LSE smoothing and adaptive parameters.
    Operates on the simplex (sum(a)=1, a>=0).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S

    best_a = a.copy()
    best_score = compute_score(a)
    log_a = np.log(a + 1e-300)

    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break

        # Adaptive beta: exponential ramp
        frac = min(1.0, t / (0.8 * n_steps))
        log_beta = np.log(beta_init) + (np.log(beta_final) - np.log(beta_init)) * frac
        beta = np.exp(log_beta)

        # Adaptive eta
        eta = eta_init / (1.0 + 0.02 * t)

        # Compute gradient using LSE
        _, grad = lse_grad_a(a, beta)

        # Mirror descent step: a_new[i] = a[i] * exp(-eta * grad[i]) / Z
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new /= s_new
            log_a = np.log(a_new + 1e-300)

        a = a_new

        if t % 200 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            if t % verbose_interval == 0 and t_start is not None:
                elapsed = time.monotonic() - t_start
                print(f"    MD t={t}: score={sc:.8f} beta={beta:.0f} eta={eta:.4f} t={elapsed:.0f}s")

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def block_lse_obj_grad(block_vals, a_full, start, end, beta):
    """Objective and gradient for a block sub-problem."""
    a_trial = a_full.copy()
    a_trial[start:end] = np.clip(block_vals, 0.0, 1000.0)
    sc, grad = lse_grad_a(a_trial, beta)
    return sc, grad[start:end]


def block_coordinate_descent(a, block_size=64, n_sweeps=30, beta_lse=500,
                             time_limit_s=None, t_start=None,
                             verbose_interval=5):
    """
    Block coordinate descent: optimize one block at a time using L-BFGS-B.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf

    n_blocks = (n + block_size - 1) // block_size

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        # Randomize block order
        order = np.random.permutation(n_blocks)

        for bi in order:
            if time.monotonic() > deadline:
                break

            start = bi * block_size
            end = min(start + block_size, n)
            bsize = end - start
            if bsize < 4:
                continue

            block = a[start:end].copy()

            # L-BFGS on the block
            res = minimize(
                block_lse_obj_grad, block,
                args=(a, start, end, beta_lse),
                method='L-BFGS-B', jac=True,
                bounds=[(0.0, 1000.0)] * bsize,
                options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-8, 'maxfun': 3000}
            )

            new_block = np.clip(res.x, 0.0, 1000.0)
            a[start:end] = new_block

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            if (sweep + 1) % verbose_interval == 0 and t_start is not None:
                elapsed = time.monotonic() - t_start
                print(f"    BCD sweep={sweep+1}: score={sc:.8f} t={elapsed:.0f}s")

    return best_score, a


def active_set_perturbation(a, n_perturbations=30, time_limit_s=None, t_start=None,
                            rng=None):
    """
    Generate perturbations focused on the active set (lags where conv is near max).
    """
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf

    for p in range(n_perturbations):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        c_max = c.max()
        threshold = c_max * (0.85 + 0.1 * rng.random())
        active_lags = np.where(c >= threshold)[0]

        # Find contributing coordinates
        contrib_mask = np.zeros(n, dtype=bool)
        for k in active_lags:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            contrib_mask[js:je + 1] = True

        a_pert = a.copy()
        contrib_sum = float(a_pert[contrib_mask].sum())
        non_contrib_sum = float(a_pert[~contrib_mask].sum())

        strategy = p % 4
        if strategy == 0 and contrib_sum > 1e-12 and non_contrib_sum > 1e-12:
            # Move mass from contributing to non-contributing
            frac = 0.03 + 0.08 * rng.random()
            removed = frac * contrib_sum
            a_pert[contrib_mask] *= (1.0 - frac)
            a_pert[~contrib_mask] += (removed / non_contrib_sum) * a_pert[~contrib_mask]

        elif strategy == 1:
            # Gaussian perturbation
            scale = 0.02 + 0.05 * rng.random()
            a_pert = a * (1.0 + scale * rng.standard_normal(n))

        elif strategy == 2:
            # Sinusoidal perturbation
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            x = np.arange(n, dtype=np.float64) / n
            a_pert = a * (1.0 + 0.05 * np.cos(2.0 * np.pi * freq * x + phase))

        else:
            # Block modification
            center = int(rng.integers(n // 4, 3 * n // 4))
            width = max(8, n // 40)
            a_pert[center - width:center + width] *= rng.uniform(0.4, 1.4)

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        s = a_pert.sum()
        if s < 1e-12:
            continue
        a_pert /= s

        # Quick optimization
        time_left = deadline - time.monotonic()
        if time_left < 3:
            break
        time_alloc = min(10.0, time_left * 0.4)
        sc, a_opt = mirror_descent(a_pert, n_steps=5000, beta_init=500, beta_final=5000,
                                   eta_init=0.03, time_limit_s=time_alloc, t_start=t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            if t_start is not None:
                print(f"    ASP pert#{p} s{strategy}: score={sc:.8f} t={time.monotonic()-t_start:.0f}s")

    return best_score, best_a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def final_lbfgs_polish(a, beta=5000, maxiter=5000, time_limit_s=60, t_start=None):
    """Final L-BFGS-B polish on the full sequence."""
    n = len(a)

    def full_obj_grad(a_flat):
        a_clip = np.clip(a_flat, 0.0, 1000.0)
        return lse_grad_a(a_clip, beta)

    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf
    t0 = time.monotonic()

    res = minimize(
        full_obj_grad, a,
        method='L-BFGS-B', jac=True,
        bounds=[(0.0, 1000.0)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-11,
                 'maxfun': maxiter * 10, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)

    print(f"{'='*50}")

    a_norm = best_a / best_a.sum()

    # Multi-stage with different top_k and learning rates
    for top_k, eta_init in [(5, 1.0), (10, 0.5), (20, 0.3)]:
        remaining = p1 - (time.monotonic() - t_start)
        if remaining < 10:
            break
        time_alloc = remaining * 0.4
        sc, a_opt = projected_subgrad_descent(
            a_norm, n_steps=200000, top_k=top_k, eta_init=eta_init,
            time_limit_s=time_alloc, t_start=t_start)
        try_update(sc, a_opt, f"PSGD top_k={top_k}")
        a_norm = a_opt

    # ==================================================================
    # Phase 2: Mirror Descent (20% of time)
    # ==================================================================
    p2 = effective_budget * 0.20
    print(f"\n{'='*50}")
    print(f"Phase 2: Mirror Descent (budget={p2:.0f}s)")
    print(f"{'='*50}")

    for beta_init, beta_final, eta_init in [(100, 2000, 0.08), (500, 10000, 0.04)]:
        remaining = p2 - (time.monotonic() - t_start)
        if remaining < 10:
            break
        time_alloc = remaining * 0.45
        sc, a_opt = mirror_descent(
            best_a / best_a.sum(), n_steps=100000, beta_init=beta_init,
            beta_final=beta_final, eta_init=eta_init,
            time_limit_s=time_alloc, t_start=t_start)
        try_update(sc, a_opt, f"MD beta={beta_init}->{beta_final}")

    # ==================================================================
    # Phase 3: Block Coordinate Descent (25% of time)
    # ==================================================================
    p3 = effective_budget * 0.25
    print(f"\n{'='*50}")
    print(f"Phase 3: Block Coordinate Descent (budget={p3:.0f}s)")
    print(f"{'='*50}")

    for block_size, beta_lse in [(32, 300), (64, 500), (128, 800)]:
        remaining = p3 - (time.monotonic() - t_start)
        if remaining < 30:
            break
        time_alloc = remaining * 0.35
        sc, a_opt = block_coordinate_descent(
            best_a, block_size=block_size, n_sweeps=100, beta_lse=beta_lse,
            time_limit_s=time_alloc, t_start=t_start)
        try_update(sc, a_opt, f"BCD block={block_size}")

    # ==================================================================
    # Phase 4: Active Set Perturbation + Refinement (15% of time)
    # ==================================================================
    p4 = effective_budget * 0.15
    print(f"\n{'='*50}")
    print(f"Phase 4: Active Set Perturbation (budget={p4:.0f}s)")
    print(f"{'='*50}")

    remaining = p4 - (time.monotonic() - t_start)
    if remaining > 10:
        sc, a_opt = active_set_perturbation(
            best_a, n_perturbations=60, time_limit_s=remaining * 0.85,
            t_start=t_start, rng=rng)
        try_update(sc, a_opt, "Active set perturbation")

    # ==================================================================
    # Phase 5: Different lengths (5% of time)
    # ==================================================================
    p5 = effective_budget * 0.05
    print(f"\n{'='*50}")
    print(f"Phase 5: Different Lengths (budget={p5:.0f}s)")
    print(f"{'='*50}")

    n_parent = len(best_a)
    for scale in [0.8, 1.2, 1.5]:
        remaining = p5 - (time.monotonic() - t_start)
        if remaining < 15:
            break
        new_n = int(n_parent * scale)
        if new_n < 512 or new_n > 40000:
            continue
        a_res = resample_sequence(best_a, new_n)
        a_res = a_res / a_res.sum()
        time_alloc = min(20.0, remaining * 0.3)
        sc, a_opt = mirror_descent(
            a_res, n_steps=10000, beta_init=200, beta_final=3000,
            eta_init=0.05, time_limit_s=time_alloc, t_start=t_start)
        try_update(sc, a_opt, f"n={new_n}")

    # ==================================================================
    # Phase 6: Final Polish (5% of time)
    # ==================================================================
    p6 = effective_budget * 0.05
    print(f"\n{'='*50}")
    print(f"Phase 6: Final Polish (budget={p6:.0f}s)")
    print(f"{'='*50}")

    final_deadline = t_start + effective_budget - 5.0

    # L-BFGS polish
    remaining = final_deadline - time.monotonic()
    if remaining > 10:
        sc, a_opt = final_lbfgs_polish(best_a, beta=5000, maxiter=3000,
                                       time_limit_s=min(40.0, remaining * 0.7),
                                       t_start=t_start)
        try_update(sc, a_opt, "Final L-BFGS")

    # One more PSDG pass
    remaining = final_deadline - time.monotonic()
    if remaining > 5:
        a_norm = best_a / best_a.sum()
        sc, a_opt = projected_subgrad_descent(
            a_norm, n_steps=50000, top_k=15, eta_init=0.2,
            time_limit_s=min(20.0, remaining * 0.8), t_start=t_start)
        try_update(sc, a_opt, "Final PSDG")

    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
