            c = d
            fc = fd

the max autocorrelation ratio of non-negative sequences.
Uses golden section search for each coordinate update, with periodic

    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

"""
Coordinate descent with exact 1D subproblem solutions for minimizing
the max autocorrelation ratio of non-negative sequences.
Key insight: for fixed other coordinates, the objective as a function
of a single coordinate is a 1D problem solvable via golden section.
Combined with LSE polishing and structured initializations.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def F_1d(baseline, coeffs, D_other, S_base, x):
    """1D objective for coordinate descent subproblem."""
    vals = baseline + x * coeffs
    m = float(np.max(vals))
    if D_other > m:
        m = D_other
    denom = S_base + x
    return m / (denom * denom)


def golden_section_1d(gs_base, gs_coeffs, gs_D, gs_Sbase, a, b, max_iter=45):
    """Golden section search minimizing F_1d on [a, b]. Returns optimal x."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = F_1d(gs_base, gs_coeffs, gs_D, gs_Sbase, c)
    fd = F_1d(gs_base, gs_coeffs, gs_D, gs_Sbase, d)
    for _ in range(max_iter):
        if b - a < 1e-10 * max(1.0, abs(b)):
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = F_1d(gs_base, gs_coeffs, gs_D, gs_Sbase, c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = F_1d(gs_base, gs_coeffs, gs_D, gs_Sbase, d)
    return (a + b) / 2.0


def coord_desc_sweep(a, c, S, order, gse_iters=40):
    """One sweep of coordinate descent. Modifies a, c in place. Returns new S."""
    n = len(a)
    for i in order:
        old_val = a[i]
        k_lo = max(0, i)
        k_hi = min(2 * n - 2, i + n - 1)
        if k_hi < k_lo:
            continue
        a_idx = np.arange(k_lo, k_hi + 1) - i
        coeffs = a[a_idx]
        baseline = c[k_lo:k_hi + 1].copy() - old_val * coeffs
        if k_lo > 0:
            D_left = float(np.max(c[:k_lo]))
        else:
            D_left = -1e300
        if k_hi + 1 <= 2 * n - 2:
            D_right = float(np.max(c[k_hi + 1:]))
        else:
            D_right = -1e300
        D_other = max(D_left, D_right)
        S_base = S - old_val
        if S_base < 1e-12:
            S_base = 1e-12
        max_coeff = float(np.max(coeffs))
        if max_coeff > 1e-15:
            current_peak = max(float(np.max(baseline)), D_other)
            x_upper = max(5.0 * old_val, 4.0 * current_peak / max_coeff, 1.0)
        else:
            x_upper = max(2.0 * old_val, 1.0)
        x_upper = min(x_upper, 50.0 * old_val + 10.0)
        F0 = F_1d(baseline, coeffs, D_other, S_base, 0.0)
        x_opt = golden_section_1d(baseline, coeffs, D_other, S_base,
                                  0.0, x_upper, max_iter=gse_iters)
        F_opt = F_1d(baseline, coeffs, D_other, S_base, x_opt)
        if F_opt >= F0:
            # Also try a large value (sequence grows = scale invariance)
            x_try = x_upper
            F_try = F_1d(baseline, coeffs, D_other, S_base, x_try)
            if F_try < F0:
                x_opt = x_try
                F_opt = F_try
            else:
                continue
        x_opt = max(0.0, x_opt)
        delta = x_opt - old_val
        if abs(delta) > 1e-14 * max(1.0, old_val):
            c[k_lo:k_hi + 1] += delta * coeffs
            a[i] = x_opt
            S += delta
    return S


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj_wrap(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, func, args, maxiter=500, maxfun=15000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(func, a0, args=args, method='L-BFGS-B', jac=True,
                       bounds=bounds,
                       options={'ftol': 1e-16, 'gtol': 1e-12,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def peak_shave(a, iters=2000):
    """Exact subgradient descent on max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        sc_lb = 1.0
        step = (sc - sc_lb) / (gn * gn + 1e-20)
        step = min(step, 1.0 / S)
        step = max(step, 1e-7 / (gn * S + 1e-30))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def normalize_seq(a, target_sum):
    """Normalize sequence to have given sum."""
    S = float(a.sum())
    if S < 1e-12:
        a[:] = target_sum / len(a)
    else:
        a *= target_sum / S
    return a


def try_init_chirp(n, rng):
    """Create a chirp-based non-negative initial sequence."""
    x = np.arange(n, dtype=np.float64)
    freq = 1.0 + rng.random() * 3.0
    a = 1.0 + 0.4 * np.cos(np.pi * x * x * freq / n)
    a = np.clip(a, 0.01, None)
    return a


def try_init_random(n, rng):
    """Create a random non-negative initial sequence."""
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a


def try_init_box(n, rng):
    """Box + noise initialization."""
    base = 1.0
    noise = 0.3 * rng.standard_normal(n)
    a = np.clip(base + noise, 0.01, None)
    return a

    parent = None
    try:

            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # --- Phase 1: Initialization (5% of time) ---
    init_budget = 0.05 * (total_budget - 30.0)
    init_end = t_start + init_budget
    print("=== Phase 1: Initialization ===")

    candidates = []

    if parent is not None:
        sc_p = compute_score(parent)
        report(sc_p, parent, "parent")
        candidates.append(parent.copy())
        print(f"  Parent: score={sc_p:.8f}, n={len(parent)}")

    # Try structured inits with parent length or 4096
    n_try = len(parent) if parent is not None else 4096
    for init_fn in [try_init_chirp, try_init_random, try_init_box]:
        if time.monotonic() > init_end:
            break
        a_init = init_fn(n_try, rng)

        S = float(a_init.sum())
        c = fast_conv(a_init)
        order = np.arange(n_try)

            target_S = np.sqrt(2.0 * n_try)
            scale = target_S / S
            if abs(scale - 1.0) > 1e-3:
                a_init *= scale
                S = target_S
                c *= scale * scale
        sc = compute_score(a_init)
        report(sc, a_init, f"init:{init_fn.__name__}")
        candidates.append(a_init.copy())

    # Use the best candidate
    a_cur = best_a.copy() if best_a is not None else candidates[0].copy()
    n = len(a_cur)

    # --- Phase 2: Main Coordinate Descent (80% of time) ---
    main_end = t_start + 0.85 * (total_budget - 30.0)
    print("\n=== Phase 2: Coordinate Descent ===")

    c = fast_conv(a_cur)
    target_S = np.sqrt(2.0 * n)

    rng.shuffle(order)
    sweep_count = 0

    while time.monotonic() < main_end:

        # Check for large scale drift and renormalize
        S_cur = float(a_cur.sum())
        if S_cur < target_S * 0.1 or S_cur > target_S * 10.0:

            S = target_S
            c = fast_conv(a_cur)

        # Run one sweep
        S = coord_desc_sweep(a_cur, c, S, order, gse_iters=gse_iters)
        sweep_count += 1

        # Renormalize to prevent drift
        S_cur = float(a_cur.sum())
        if S_cur > 0:

            a_cur *= scale

            c *= scale * scale

        # Evaluate periodically (every 10 sweeps or at the end of a "batch")

            if report(sc, a_cur, f"coord_sweep_{sweep_count}"):

                no_improve_count = 0
                # Refresh order

                no_improve_count += 10
