import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_objective(a, beta):
    return lse_score_grad(a, beta)

def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)

def coordinate_descent_sweep(a, rng=None, n_coords=None):
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    if n_coords is None:
        n_coords = n

    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)

    for idx in range(min(n_coords, n)):
        i = int(order[idx])
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)

        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-7}
        )
        t_opt = float(res.x)

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a

def hot_coordinate_sweep(a, n_sweeps=1):
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    for _ in range(n_sweeps):
        k_star = int(np.argmax(conv))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        order = np.argsort(contrib)[::-1]

        for i in order:
            i = int(i)
            a_i = a[i]

            k_lo = i
            k_hi = min(M_conv - 1, i + n - 1)

            if k_lo > 0 and k_hi < M_conv - 1:
                outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
            elif k_lo > 0:
                outside_max = float(np.max(conv[:k_lo]))
            elif k_hi < M_conv - 1:
                outside_max = float(np.max(conv[k_hi + 1:]))
            else:
                outside_max = 0.0

            inside_len = k_hi - k_lo + 1
            base_inside = conv[k_lo:k_hi + 1].copy()
            ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
            a_vals = a[ks - i]
            is_2i = (ks == 2 * i)
            old_contrib = 2.0 * a_i * a_vals
            old_contrib[is_2i] = a_i * a_i
            base_inside -= old_contrib

            alpha = 2.0 * a_vals.copy()
            alpha[is_2i] = 0.0
            beta_coef = np.zeros(inside_len)
            beta_coef[is_2i] = 1.0

            S_rest = S - a_i
            t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
            t_upper = min(t_upper, 1000.0)

            res = minimize_scalar(
                coord_1d_obj,
                args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
                bounds=(0.0, t_upper),
                method='bounded',
                options={'xatol': 1e-7}
            )
            t_opt = float(res.x)
            a[i] = t_opt
            S = S_rest + t_opt
            conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a

def make_palindromic(a):
    """Make sequence palindromic by averaging with reverse."""
    return (a + a[::-1]) / 2.0

def freq_perturb(a, rng, n_low=3, n_high=None):
    """Perturb in frequency domain: modify random frequency bins."""
    n = len(a)
    A = np.fft.rfft(a)
    if n_high is None:
        n_high = len(A) - 1
    
    n_mod = max(1, n // 10)
    idx = rng.choice(np.arange(n_low, min(n_high, len(A)-1)), size=min(n_mod, len(A)-1-n_low), replace=False)
    A[idx] *= (1.0 + 0.1 * rng.standard_normal(len(idx)))
    a_new = np.fft.irfft(A, n)
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new

def targeted_perturb(a, rng, strength=0.05):
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)

"""
Adaptive LSE-LBFGS + Coordinate Descent optimizer for minimum peak self-convolution.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key improvements over previous approach:
1. Annealed LSE beta schedule (low→high) for smoother global→local transition
2. Palindromic and frequency-domain structured perturbations
3. Adaptive "stuck detection" with LSE escape steps
4. Multi-strategy perturbation with quick refinement pipelines
5. Better time management with progress tracking

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient with numerical safeguards."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -400.0, 400.0)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300 or not np.isfinite(exp_sum):
        return 1e10, np.zeros(n)
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    grad = np.nan_to_num(grad, nan=0.0, posinf=1e6, neginf=-1e6)
    return score, grad


def lse_objective_fn(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=2000, maxfun=40000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective_fn, a0, args=(beta),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-14, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def _cd_single_aug(a, conv, S, i, n, M_conv):
    """Optimize a single coordinate i given current a, conv, S."""
    a_i = a[i]
    k_lo = i
    k_hi = min(M_conv - 1, i + n - 1)

    if k_lo > 0 and k_hi < M_conv - 1:
        outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
    elif k_lo > 0:
        outside_max = float(np.max(conv[:k_lo]))
    elif k_hi < M_conv - 1:
        outside_max = float(np.max(conv[k_hi + 1:]))
    else:
        outside_max = 0.0

    inside_len = k_hi - k_lo + 1
    base_inside = conv[k_lo:k_hi + 1].copy()
    ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
    a_vals = a[ks - i]
    is_2i = (ks == 2 * i)
    old_contrib = 2.0 * a_i * a_vals
    old_contrib[is_2i] = a_i * a_i
    base_inside -= old_contrib

    alpha = 2.0 * a_vals.copy()
    alpha[is_2i] = 0.0
    beta_coef = np.zeros(inside_len)
    beta_coef[is_2i] = 1.0

    S_rest = S - a_i
    t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
    t_upper = min(t_upper, 1000.0)

    res = minimize_scalar(
        coord_1d_obj,
        args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
        bounds=(0.0, t_upper),
        method='bounded',
        options={'xatol': 1e-7}
    )
    t_opt = float(res.x)
    a[i] = t_opt
    S = S_rest + t_opt
    conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef
    return S


def coordinate_descent_sweep(a, rng=None, n_coords=None):
    """One full sweep of coordinate descent with incremental conv updates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    if n_coords is None:
        n_coords = n
    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)

    for idx in range(min(n_coords, n)):
        i = int(order[idx])
        S = _cd_single_aug(a, conv, S, i, n, M_conv)

    final_score = compute_score(a)
    return final_score, a


def hot_coordinate_sweep(a, n_sweeps=1):
    """Coordinate descent prioritizing coordinates contributing to the peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    for _ in range(n_sweeps):
        k_star = int(np.argmax(conv))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        order = np.argsort(contrib)[::-1]

        for i in order:
            i = int(i)
            S = _cd_single_aug(a, conv, S, i, n, M_conv)

    final_score = compute_score(a)
    return final_score, a


def make_palindromic(a):
    """Make sequence palindromic by averaging with its reverse."""
    return (a + a[::-1]) / 2.0


def freq_domain_perturb(a, rng, n_mod=None, amp_scale=0.08):
    """Perturb in frequency domain by modifying random frequency bins."""
    n = len(a)
    A = np.fft.rfft(a)
    if n_mod is None:
        n_mod = max(5, n // 15)
    n_mod = min(n_mod, len(A) - 2)
    idx = rng.integers(1, len(A), size=n_mod)
    A[idx] *= (1.0 + amp_scale * rng.standard_normal(n_mod))
    a_new = np.fft.irfft(A, n)
    return np.clip(a_new, 0.0, 1000.0)


def targeted_perturb(a, rng, strength=0.04):
    """Reduce peak-contributing elements, redistribute mass."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def multi_piece_perturb(a, rng, n_pieces=4):
    """Split sequence into pieces, randomly rescale and shuffle."""
    n = len(a)
    a_pert = a.copy()
    piece_size = n // n_pieces
    scales = 1.0 + 0.15 * rng.standard_normal(n_pieces)
    scales = np.clip(scales, 0.5, 2.0)
    for i in range(n_pieces):
        lo = i * piece_size
        hi = lo + piece_size if i < n_pieces - 1 else n
        a_pert[lo:hi] *= scales[i]
    # Renormalize to original sum
    orig_sum = a.sum()
    new_sum = a_pert.sum()
    if new_sum > 1e-12:
        a_pert *= orig_sum / new_sum
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses annealed LSE + coordinate descent + structured perturbations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 12.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None and parent.sum() > 0.01:
        p_score = compute_score(parent)
        print(f"Parent score: {p_score:.8f}, length: {len(parent)}")
        best_score = p_score
        best_a = parent.copy()
    else:
        n_init = 2048
        best_a = np.ones(n_init) * 0.5
        best_score = compute_score(best_a)
        print(f"No valid parent, using uniform init. Score: {best_score:.8f}")

    n_work = len(best_a)
    print(f"Working length: {n_work}")

    # ============================================================
    # Phase 1: Annealed LSE-LBFGS (22% of time)
    # ============================================================
    phase1_deadline = t_start + 0.22 * total_budget
    print(f"\n=== Phase 1: Annealed LSE ({phase1_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    # Annealing: start smooth (low beta), increase to sharp (high beta)
    beta_schedule = [15, 40, 100, 300, 800, 2000, 5000, 12000, 25000]

    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > phase1_deadline - 3:
            break
        t0 = time.monotonic()
        frac = (phase1_deadline - time.monotonic()) / max(1.0, phase1_deadline - t_start)
        maxfun = int(15000 + 25000 * frac)
        maxiter = max(500, int(1500 * (1.0 + bi * 0.3)))

        sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
        elapsed = time.monotonic() - t0
        a_cur = a_opt.copy()

        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  beta={beta:>6d}: {sc:.8f} *** NEW BEST *** ({elapsed:.1f}s)")
        else:
            print(f"  beta={beta:>6d}: {sc:.8f} (best: {best_score:.8f}) [{elapsed:.1f}s]")

    # Quick CD to finalize
    if time.monotonic() < phase1_deadline - 2:
        sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  Post-LSE CD: {sc:.8f} *** NEW BEST ***")

    print(f"Phase 1 complete. Best: {best_score:.8f}")

    # ============================================================
    # Phase 2: Coordinate descent refinement (33% of time)
    # ============================================================
    phase2_deadline = t_start + 0.55 * total_budget
    print(f"\n=== Phase 2: CD Refinement ({phase2_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    sweep_count = 0
    no_improve = 0
    prev_sc = compute_score(a_cur)

    while time.monotonic() < phase2_deadline - 2:
        sweep_count += 1
        # Cycle through orderings: random, random, hot
        if sweep_count % 3 == 0:
            sc, a_cur = hot_coordinate_sweep(a_cur, n_sweeps=1)
        else:
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_cur.copy()
            no_improve = 0
            if sweep_count <= 10 or sweep_count % 30 == 0:
                print(f"  Sweep {sweep_count:>4d}: {sc:.8f} *** NEW BEST ***")
        else:
            no_improve += 1

        if sweep_count % 30 == 0:
            delta = prev_sc - sc
            prev_sc = sc
            print(f"  Sweep {sweep_count:>4d}: {sc:.8f} (stagnant={no_improve}, delta={delta:+.2e})")

        # Escape mechanism: LSE step when stuck
        if no_improve >= 25 and time.monotonic() < phase2_deadline - 30:
            print(f"  Stuck at sweep {sweep_count}, LSE escape (beta=3000)...")
            t_esc = time.monotonic()
            sc_esc, a_esc, _ = run_lbfgs(a_cur, 3000, maxiter=1500, maxfun=25000)
            a_cur = a_esc.copy()
            sc_esc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
            if sc_esc < best_score:
                best_score = sc_esc
                best_a = a_cur.copy()
                print(f"    Escape: {sc_esc:.8f} *** NEW BEST *** ({time.monotonic()-t_esc:.1f}s)")
            else:
                print(f"    Escape: {sc_esc:.8f} ({time.monotonic()-t_esc:.1f}s)")
            no_improve = 0

    print(f"Phase 2 complete: {sweep_count} sweeps. Best: {best_score:.8f}")

    # ============================================================
    # Phase 3: Structured perturbations (30% of time)
    # ============================================================
    phase3_deadline = t_start + 0.85 * total_budget
    print(f"\n=== Phase 3: Perturbations ({phase3_deadline - t_start:.0f}s) ===")

    trial = 0
    best_in_phase3 = best_score
    stagnant_trials = 0

    while time.monotonic() < phase3_deadline - 5:
        strategy = trial % 8

        if strategy == 0:
            a_pert = make_palindromic(best_a)
        elif strategy == 1:
            a_pert = freq_domain_perturb(best_a, rng, amp_scale=0.05 + 0.05 * rng.random())
        elif strategy == 2:
            a_pert = targeted_perturb(best_a, rng, strength=0.02 + 0.05 * rng.random())
        elif strategy == 3:
            scale = 0.015 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_work))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 4:
            a_pert = multi_piece_perturb(best_a, rng, n_pieces=3 + int(rng.integers(0, 4)))
        elif strategy == 5:
            # Sinusoidal modulation
            freq = 1.0 + 8.0 * rng.random()
            amp = 0.015 + 0.04 * rng.random()
            x = np.arange(n_work, dtype=np.float64) / n_work
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + 2.0 * np.pi * rng.random()))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 6:
            # "Valley" perturbation
            center = int(rng.integers(n_work // 5, 4 * n_work // 5))
            width = max(5, n_work // 25)
            a_pert = best_a.copy()
            lo = max(0, center - width)
            hi = min(n_work, center + width)
            saved = a_pert[lo:hi].sum()
            a_pert[lo:hi] *= 0.2
            removed = saved - a_pert[lo:hi].sum()
            mask = np.ones(n_work, dtype=bool)
            mask[lo:hi] = False
            if mask.sum() > 0:
                a_pert[mask] += removed / mask.sum()
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Combined: targeted + palindromic
            a_pert = targeted_perturb(best_a, rng, strength=0.03)
            a_pert = make_palindromic(a_pert)

        # Quick refinement pipeline: 3 CD + 1 LSE + 2 CD
        for sw in range(3):
            if time.monotonic() > phase3_deadline - 8:
                break
            if sw % 2 == 0:
                sc_p, a_pert = hot_coordinate_sweep(a_pert, n_sweeps=1)
            else:
                sc_p, a_pert = coordinate_descent_sweep(a_pert, rng=rng)

        if time.monotonic() < phase3_deadline - 25:
            sc_p, a_pert, _ = run_lbfgs(a_pert, 1500, maxiter=1000, maxfun=15000)
            sc_p, a_pert = coordinate_descent_sweep(a_pert, rng=rng)
            if time.monotonic() < phase3_deadline - 5:
                sc_p, a_pert = hot_coordinate_sweep(a_pert, n_sweeps=1)

        if sc_p < best_score - 1e-12:
            best_score = sc_p
            best_a = a_pert.copy()
            stagnant_trials = 0
            print(f"  Trial {trial:>3d} (s={strategy}): {sc_p:.8f} *** NEW BEST ***")
        else:
            stagnant_trials += 1
            if trial % 8 == 0:
                print(f"  Trial {trial:>3d} (s={strategy}): {sc_p:.8f}")

        # If very stagnant, try a more aggressive perturbation
        if stagnant_trials >= 16:
            print(f"  Very stagnant ({stagnant_trials}), aggressive restart...")
            # Blend with a fresh random sequence
            a_fresh = np.abs(rng.standard_normal(n_work)) + 0.1
            a_fresh /= a_fresh.max()
            blend = 0.3 + 0.4 * rng.random()
            a_pert = (1 - blend) * best_a + blend * a_fresh * (best_a.sum() / a_fresh.sum())
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            # Quick optimize
            sc_p, a_pert, _ = run_lbfgs(a_pert, 500, maxiter=800, maxfun=10000)
            for sw in range(4):
                if time.monotonic() > phase3_deadline - 5:
                    break
                sc_p, a_pert = coordinate_descent_sweep(a_pert, rng=rng)
            if sc_p < best_score:
                best_score = sc_p
                best_a = a_pert.copy()
                print(f"    Aggressive restart: {sc_p:.8f} *** NEW BEST ***")
            stagnant_trials = 0

        trial += 1

    print(f"Phase 3 complete: {trial} trials. Best: {best_score:.8f}")

    # ============================================================
    # Phase 4: Final polish (10% of time)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")
    a_cur = best_a.copy()
    polish_count = 0
    while time.monotonic() < deadline - 3:
        polish_count += 1
        if polish_count % 4 == 0:
            sc, a_cur = hot_coordinate_sweep(a_cur, n_sweeps=1)
        else:
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_cur.copy()
            if polish_count % 5 == 0:
                print(f"  Polish {polish_count:>3d}: {sc:.8f} *** NEW BEST ***")

    # Final LSE if time permits
    if time.monotonic() < deadline - 25:
        sc, a_opt, _ = run_lbfgs(best_a, 8000, maxiter=2000, maxfun=30000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final LSE: {sc:.8f} *** NEW BEST ***")
        a_cur = best_a.copy()
        if time.monotonic() < deadline - 10:
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                print(f"  Final CD: {sc:.8f} *** NEW BEST ***")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
