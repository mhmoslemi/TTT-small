import numpy as np

from scipy.optimize import minimize


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad_norm(a, beta):
    """LSE-smoothed objective in normalized form. Minimize LSE(beta*c) s.t. sum(a)=1 implicitly.
    Actually we minimize 2n*LSE(beta*c)/(S^2) which handles normalization automatically."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    exp_sum = exp_s.sum()
    if exp_sum < 1e-300:
        L = c_max
        p = np.full(len(c), 1.0 / len(c))
    else:
        L = c_max + np.log(exp_sum) / beta
        p = exp_s / exp_sum
    score = 2.0 * n * L / (S * S)
    # Gradient: dL/d(a_m) = 2 * sum_j a_j * p[m+j] (computed via FFT)
    rev_a = a[::-1]

    """Subgradient using top-K peaks with weighted averaging."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    lb = 2.0 * n / (2.0 * n - 1)

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Use LSE with high beta to approximate max with a smoothed "top-K"
        beta_sub = 500.0
        scaled = beta_sub * (c - c_max)
        scaled = np.clip(scaled, -400, 400)
        exp_s = np.exp(scaled)
        p = exp_s / exp_s.sum()

        # Gradient of sum_k p_k * c_k = LSE-like
        rev_a = a[::-1]
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[:n + 2 * n - 2]
        g = 2.0 * conv_res[n - 1:2 * n - 1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (g / S2 - 2.0 * c_max / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - lb) / gn2
        step = min(step, 0.5 * (0.999 ** (it // 50)))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def build_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else x

    # Anti-palindromic ramp
    inits.append(np.linspace(0.05, 1.0, n))
    inits.append(np.linspace(1.0, 0.05, n))

    # Bell shape
    inits.append(np.exp(-2.0 * (xn - 0.5) ** 2))

    # Inverted bell
    inits.append(1.0 - 0.7 * np.exp(-4.0 * (xn - 0.5) ** 2))

    # Sinusoidal
    inits.append(np.clip(0.5 + 0.4 * np.sin(np.pi * xn), 0.01, 10.0))
    inits.append(np.clip(0.5 + 0.4 * np.sin(3.0 * np.pi * xn), 0.01, 10.0))

    # Random smooth
    a_rand = 0.5 + 0.3 * rng.standard_normal(n)
    ksize = max(3, n // 100)
    kernel = np.ones(ksize) / ksize
    a_smooth = np.convolve(a_rand, kernel, mode='same')
    inits.append(np.clip(a_smooth, 0.01, 10.0))

    # Pure random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 10.0))

    # Two-block
    half = n // 2
    inits.append(np.concatenate([np.ones(half) * 0.7, np.ones(n - half) * 0.3]))
    inits.append(np.concatenate([np.ones(half) * 0.3, np.ones(n - half) * 0.7]))

    # Cosine taper (like a window function)
    inits.append(0.5 + 0.5 * np.cos(np.pi * xn))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * xn))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Multi-phase optimization targeting score < 1.5027.
    Phase 1: Quick LSE refinement of parent (if available)
    Phase 2: Aggressive high-beta LSE continuation
    Phase 3: Multi-argmax subgradient polish
    Phase 4: Peak-shaving local search
    Phase 5: Explore new initializations at n=8192
    Phase 6: Final refinement
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    print(f"Parent: {'yes (n=' + str(parent.size) + ', score=' + str(compute_score(parent)) + ')' if parent is not None else 'no'}")

    # ===== Phase 1: Quick refinement of parent =====
    if parent is not None:
        n_parent = parent.size
        print(f"\n=== Phase 1: Quick LSE on parent (n={n_parent}) ===")
        a_cur = parent.copy()
        for beta in [100, 500, 2000]:
            sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=3000, maxfun=60000)
            a_cur = a_opt
            report(sc, a_opt, f"P1 beta={beta}")
        print(f"  After Phase 1: {best_score:.8f}")

    # ===== Phase 2: Aggressive high-beta continuation on best =====
    print(f"\n=== Phase 2: High-Beta Continuation ===")
    n_work = len(best_a)
    beta_schedule = [5000, 10000, 20000, 50000, 100000, 200000]
    t_phase2_end = t_start + 900.0

    for beta in beta_schedule:
        if time.monotonic() > t_phase2_end:
            break
        remaining = max(60.0, t_phase2_end - time.monotonic())
        maxit = max(2000, min(10000, int(remaining * 3)))
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxit, maxfun=maxit * 20)
        report(sc, a_opt, f"P2 beta={beta}")

    # Subgradient polish between phases
    sc, a_opt = subgrad_polish_multi(best_a, max_iters=3000,
                                      lb=2.0 * len(best_a) / (2.0 * len(best_a) - 1))
    report(sc, a_opt, "P2 subgrad")

    print(f"  After Phase 2: {best_score:.8f}")

    # ===== Phase 3: Multi-argmax subgradient =====
    print(f"\n=== Phase 3: Multi-Argmax Subgradient ===")
    t_phase3_end = t_start + 1200.0
    if time.monotonic() < t_phase3_end:
        n_iters = max(2000, int((t_phase3_end - time.monotonic()) * 50))
        sc, a_opt = multi_argmax_subgrad(best_a, max_iters=n_iters)
        report(sc, a_opt, "P3 multi-subgrad")

    # ===== Phase 4: Peak-shaving local search =====
    print(f"\n=== Phase 4: Peak Shaving ===")
    t_phase4_end = t_start + 1700.0
    if time.monotonic() < t_phase4_end:
        time_left = t_phase4_end - time.monotonic()
        n_trials = max(50, int(time_left / 3))
        sc, a_opt = peak_shave_local(best_a, rng, n_trials=n_trials, time_budget=time_left)
        report(sc, a_opt, "P4 peak-shave")

    # LSE refinement after peak shaving
    if time.monotonic() < t_phase4_end + 100:
        sc, a_opt, _ = run_lbfgs(best_a, 50000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "P4 post-shave LSE")
        sc, a_opt = subgrad_polish_multi(best_a, max_iters=2000,
                                          lb=2.0 * len(best_a) / (2.0 * len(best_a) - 1))
        report(sc, a_opt, "P4 post-shave subgrad")

    print(f"  After Phase 4: {best_score:.8f}")

    # ===== Phase 5: Explore n=8192 with new inits =====
    print(f"\n=== Phase 5: Explore n=8192 ===")
    n_new = 8192
    t_phase5_end = t_start + 2100.0
    if time.monotonic() < t_phase5_end:
        inits = build_diverse_inits(n_new, rng)
        for i, a_init in enumerate(inits[:5]):
            if time.monotonic() > t_phase5_end:
                break
            a_cur = a_init.copy()
            # Quick LSE continuation
            for beta in [100, 1000, 10000, 50000]:
                if time.monotonic() > t_phase5_end:
                    break
                sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=2000, maxfun=40000)
                a_cur = a_opt
                report(sc, a_opt, f"P5 n={n_new} i={i} b={beta}")
            # Quick subgrad
            sc, a_opt = subgrad_polish_multi(a_cur, max_iters=1000,
                                              lb=2.0 * n_new / (2.0 * n_new - 1))
            report(sc, a_opt, f"P5 n={n_new} i={i} subgrad")

    # ===== Phase 6: Final refinement of overall best =====
    print(f"\n=== Phase 6: Final Refinement ===")
    deadline = t_start + budget_s - 30.0
    if time.monotonic() < deadline:
        # One more high-beta LSE
        remaining = deadline - time.monotonic()
        maxit = max(1000, int(remaining * 2))
        sc, a_opt, _ = run_lbfgs(best_a, 100000, maxiter=maxit, maxfun=maxit * 20)
        report(sc, a_opt, "P6 final LSE")

        # Final subgradient
        sc, a_opt = subgrad_polish_multi(best_a, max_iters=3000,
                                          lb=2.0 * len(best_a) / (2.0 * len(best_a) - 1))
        report(sc, a_opt, "P6 final subgrad")

    # ===== Output =====
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"Conv max lag: {int(np.argmax(fast_conv(best_a)))}")
    print(f"{'=' * 60}")
    return out
