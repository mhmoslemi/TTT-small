def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """Linear cross-convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def subgradient_exact(a):
    """Compute exact subgradient of the score function."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(np.argmax(c))
    sc = 2.0 * n * M / (S * S)
    # Subgradient of max(c) w.r.t. a
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    sg_M = np.zeros(n)
    if j_end >= j_start:
        seg = a[j_start:j_end + 1]
        sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        # Fix the diagonal term (where 2j = k_star)
        if k_star % 2 == 0:
            j_diag = k_star // 2
            if 0 <= j_diag < n:
                sg_M[j_diag] = a[j_diag]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
    return sc, grad


def fista_subgradient(a, total_iters=2000, max_wall_s=100.0):
    """FISTA-accelerated subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    t0 = time.monotonic()
    # FISTA state
    y = a.copy()
    tk = 1.0
    # Initial step size
    Sc, gc = subgradient_exact(y)
    gn = float(np.linalg.norm(gc))
    if gn < 1e-15:
        return best_sc, best_a
    step = min((Sc - sc_lb) / (gn * gn + 1e-30), 0.5)
    for it in range(total_iters):
        if time.monotonic() - t0 > max_wall_s:
            break
        c_y = fast_conv(y)
        S_y = float(y.sum())
        if S_y < 1e-12:
            break
        M_y = float(c_y.max())
        k_star = int(np.argmax(c_y))
        Sc_y = 2.0 * n * M_y / (S_y * S_y)
        # Subgradient at y
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = y[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
            if k_star % 2 == 0:
                j_diag = k_star // 2
                if 0 <= j_diag < n:
                    sg_M[j_diag] = y[j_diag]
        S2 = S_y * S_y
        S3 = S2 * S_y
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M_y / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        # Adaptive step size
        step = min((Sc_y - sc_lb) / (gn * gn + 1e-30), 0.3)
        step = max(step, 1e-6)
        # Update y
        y_new = y - step * grad
        y_new = np.clip(y_new, 0.0, 1000.0)
        # FISTA momentum
        tk_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y = y_new + ((tk - 1.0) / tk_new) * (y_new - a)
        y = np.clip(y, 0.0, 1000.0)
        a = y_new
        tk = tk_new
        # Check score
        sc_a = compute_score(a)
        if sc_a < best_sc:
            best_sc = sc_a
            best_a = a.copy()

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B minimization with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5

        lambda x: lse_score_grad(x, beta), a0,

    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_conv_matrix(a):
    """Build the dense convolution matrix A where (A*b)[k] = (a*b)[k]."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_mat[k_start:k_end, j] = a[:k_end - k_start]
    return A_mat


def solve_valley_fill_lp(a, alpha, delta_frac=0.005, max_wall_s=45.0):
    """
    Valley-filling LP: max sum(d) s.t. d>=0, (a*d)_k <= (s_k + delta)/alpha for all k.
    where s_k = M - c_k is the slack and delta = delta_frac * M.
    Returns optimal d or None.
    """
    if not HAVE_CVXPY:
        return None
    n = len(a)
    t0 = time.monotonic()
    if n > 8000:
        return None
    c = fast_conv(a)
    M = float(c.max())
    if M < 1e-12:
        return a.copy()
    s = M - c
    delta = delta_frac * M
    rhs = (s + delta) / alpha
    A_mat = build_conv_matrix(a)
    d_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ d_var <= rhs]
    prob = cp.Problem(cp.Maximize(cp.sum(d_var)), constraints)
    try:
        remaining = max_wall_s - (time.monotonic() - t0)
        if remaining < 5:
            return None
        prob.solve(solver=cp.PDLP, max_time=remaining)
    except Exception:
        try:
            remaining = max_wall_s - (time.monotonic() - t0)
            if remaining < 3:
                return None
            prob.solve(solver=cp.ECOS, max_time=remaining)
        except Exception:
            return None
    if time.monotonic() - t0 > max_wall_s:
        return None
    if d_var.value is None:
        return None
    d_opt = np.clip(d_var.value, 0.0, 1000.0)
    if d_opt.sum() < 1e-10:
        return None
    return d_opt

def targeted_perturb(a, rng, strength):
    """Perturb elements contributing most to peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_count = max(top_count, n // 4)
    bottom_idx = np.argsort(contrib)[:bottom_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)

    # Add back to low-contribution positions
    add_mask = np.zeros(n)
    num_valleys = max(3, n // 30)
    valley_lags = np.argsort(c)[:num_valleys]
    for k in valley_lags:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        add_mask[j_start:j_end + 1] += 1.0
    add_mask = add_mask / (add_mask.max() + 1e-10)
    total_reduced = float((a - a_pert).sum())
    a_pert += total_reduced * add_mask / float(add_mask.sum())
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses valley-filling LP, standard LP, L-BFGS-B with LSE, FISTA subgradient,
    and diverse perturbations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent solution
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        pa = parent.size
        lengths.append(pa)
        for dl in [-100, 100, -500, 500]:
            cand = pa + dl
            if cand >= 512:
                lengths.append(cand)
    for n in [4096, 6144, 8192, 10240]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}, len={parent.size if parent is not None else 0}")
    print(f"CVXPY available: {HAVE_CVXPY}")

    # =========================================================================
    # Phase 1: L-BFGS-B with continuation in beta (25% of time)
    # =========================================================================
    phase1_deadline = t_start + 0.25 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: L-BFGS-B optimization (deadline at {phase1_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    if parent is not None:
        sc_parent = compute_score(parent)
        report(sc_parent, parent, "parent")

    beta_schedule = [20, 50, 100, 200, 400, 800, 1600]

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n--- n = {n} ---")
        inits = build_inits(n, rng, parent)
        remaining = phase1_deadline - time.monotonic()
        max_inits = max(2, min(len(inits), int(remaining / 50.0)))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_cur = a0.copy()
            for bi, beta in enumerate(beta_schedule):
                if time.monotonic() > phase1_deadline:
                    break
                remaining_bi = phase1_deadline - time.monotonic()
                maxiter = min(5000, 300 * min(bi + 2, 6))
                sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=max(maxiter * 2, 1000))
                a_cur = a_opt
                report(sc, a_opt, f"n={n}, init={i}, beta={beta}")

    print(f"\nAfter Phase 1: best = {best_score:.8f}")

    # =========================================================================
    # Phase 2: LP-based refinement (35% of time)
    # =========================================================================
    phase2_deadline = t_start + 0.60 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: LP refinement (deadline at {phase2_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    if len(best_a) <= 8000:
        a_lp = best_a.copy()
        it_lp = 0
        # Alternate between valley-fill and standard LP
        while time.monotonic() < phase2_deadline:
            use_valley = (it_lp % 2 == 0)
            if use_valley:
                sc, a_new = lp_step_valley_fill(a_lp, alpha=0.5, max_wall_s=40.0)
            else:
                sc, a_new = lp_step_standard(a_lp, max_wall_s=40.0)
            if sc < best_score:
                report(sc, a_new, f"{'valley' if use_valley else 'std'} LP {it_lp}")
            a_lp = a_new
            # L-BFGS polish
            if time.monotonic() > phase2_deadline:
                break
            sc, a_pol = run_lbfgs(a_lp, 400, maxiter=2000, maxfun=4000)
            report(sc, a_pol, f"LP+LBFGS {it_lp}")
            a_lp = a_pol
            it_lp += 1
            if it_lp >= 25:
                break
    else:
        print("  Sequence too long for LP, using L-BFGS only")
        a_tk = best_a.copy()
        for it in range(15):
            if time.monotonic() > phase2_deadline:
                break
            sc, a_opt = run_lbfgs(a_tk, 500 + it * 200, maxiter=3000, maxfun=6000)
            report(sc, a_opt, f"LBFGS iter {it}")
            a_tk = a_opt

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # =========================================================================
    # Phase 3: Diversification (25% of time)
    # =========================================================================
    phase3_deadline = t_start + 0.85 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Diversification (deadline at {phase3_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    trial = 0
    while time.monotonic() < phase3_deadline:
        strategy = trial % 8
        if strategy == 0:
            scale = 0.02 + 0.08 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.07 * rng.random())
        elif strategy == 2:
            freq = 1.0 + 10.0 * rng.random()
            phase_val = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.06 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase_val))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            block_sz = max(10, n_best // 20)
            b1 = int(rng.integers(0, max(1, n_best - 2 * block_sz)))
            b2 = int(rng.integers(0, max(1, n_best - 2 * block_sz)))
            a_pert = best_a.copy()
            a_pert[b1:b1 + block_sz], a_pert[b2:b2 + block_sz] = \
                a_pert[b2:b2 + block_sz].copy(), a_pert[b1:b1 + block_sz].copy()
        elif strategy == 4:
            n_sparse = max(10, n_best // 40)
            a_pert = best_a.copy()
            idx = rng.integers(0, n_best, size=n_sparse)
            a_pert[idx] *= (1.0 + 0.3 * rng.standard_normal(n_sparse))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 5:
            a_pert = valley_fill_perturb(best_a, rng, strength=0.03 + 0.05 * rng.random())
        elif strategy == 6:
            # Mix with structured sequence
            x = np.arange(n_best, dtype=np.float64) / n_best
            freq1 = 1.0 + 3.0 * rng.random()
            freq2 = 4.0 + 8.0 * rng.random()
            a_struct = 0.5 + 0.3 * np.cos(2.0 * np.pi * freq1 * x + rng.random() * 6.28) \
                       + 0.2 * np.sin(2.0 * np.pi * freq2 * x + rng.random() * 6.28)
            a_struct = np.clip(a_struct, 0.01, 1000.0)
            mix_frac = 0.2 + 0.3 * rng.random()
            a_pert = (1 - mix_frac) * best_a + mix_frac * a_struct * (best_a.sum() / a_struct.sum())
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # LP on perturbed version
            scale = 0.05 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            if len(a_pert) <= 8000:
                sc_pert, a_pert = lp_step_valley_fill(a_pert, alpha=0.5, max_wall_s=20.0)

        # Polish
        if time.monotonic() < phase3_deadline:
            sc, a_opt = run_lbfgs(a_pert, 600, maxiter=3000, maxfun=5000)
            report(sc, a_opt, f"div #{trial} (strat={strategy})")
        trial += 1
        if trial >= 40:
            break

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # =========================================================================
    # Phase 4: Final refinement (15% of time)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Refinement")
    print(f"{'='*60}")

    # FISTA subgradient
    if time.monotonic() < deadline - 60.0:
        remaining_s = deadline - 60.0 - time.monotonic()
        sc, a_fista = fista_subgradient(best_a, total_iters=3000, max_wall_s=remaining_s)
        report(sc, a_fista, "FISTA subgrad")

    # High-beta LSE
    for beta in [2000, 5000, 10000]:
        if time.monotonic() > deadline - 30.0:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=8000, maxfun=160000)
        report(sc, a_opt, f"final LBFGS beta={beta}")

    # One more LP if available
    if HAVE_CVXPY and len(best_a) <= 8000 and time.monotonic() < deadline - 40.0:
        sc, a_new = lp_step_valley_fill(best_a, alpha=0.3, max_wall_s=30.0)
        report(sc, a_new, "final valley LP")
        if time.monotonic() < deadline - 20.0:
            sc, a_opt = run_lbfgs(best_a, 3000, maxiter=5000, maxfun=10000)
            report(sc, a_opt, "post-LP LBFGS")

    # Final FISTA
    if time.monotonic() < deadline - 5.0:
        remaining_s = max(5.0, deadline - 5.0 - time.monotonic())
        sc, a_fista = fista_subgradient(best_a, total_iters=500, max_wall_s=remaining_s)
        report(sc, a_fista, "final FISTA")

    # =========================================================================
    # Final verification and return
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
