    import cvxpy as cp
    HAVE_CVXPY = True
except ImportError:
    HAVE_CVXPY = False


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """Linear cross-convolution via FFT."""
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_sparse_conv_matrix(a):
    """Build sparse Toeplitz matrix for convolution with a.
    Returns A (2n-1) x n where A @ b = conv(a, b).
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        length = k_end - k_start
        rows.extend(range(k_start, k_end))
        cols.extend([j] * length)
        vals.extend(a[:length])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_for_direction(a, max_time=25.0):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b) <= M."""
    if not HAVE_CVXPY:
        return None, None
    t0 = time.monotonic()
    n = len(a)
    M = float(np.max(fast_conv(a)))
    if M < 1e-12:
        return a.copy(), np.zeros(2 * n - 1)
    
    # Build sparse convolution matrix
    A_mat = build_sparse_conv_matrix(a)
    
    b_var = cp.Variable(n, nonneg=True)
    y_var = cp.Variable(2 * n - 1, nonneg=True)
    constraints = [A_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    
    remaining = max_time - (time.monotonic() - t0)
    if remaining < 2:
        return None, None
    
    try:
        prob.solve(solver=cp.PDLP, max_time=remaining)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_time=remaining)
        except Exception:
            return None, None
    
    if time.monotonic() - t0 > max_time + 5:
        return None, None
    
    if b_var.value is None:
        return None, None
    
    b_opt = np.clip(np.asarray(b_var.value), 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None, None
    
    # Try to get dual values for the constraints
    y_dual = None
    try:
        if constraints[0].dual_value is not None:
            y_dual = np.asarray(constraints[0].dual_value).ravel()
            y_dual = np.clip(y_dual, 0.0, None)
    except Exception:
        pass
    
    return b_opt, y_dual


def topk_score_grad(a, k):
    """Top-k smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_sorted = np.sort(c)[::-1]
    if k < len(c_sorted):
        threshold = c_sorted[k - 1]
        k2 = min(k + 3, len(c_sorted) - 1)
        gap = max(c_sorted[k - 1] - c_sorted[k2], 1e-10)
        width = gap * 0.5 + 1e-10
        w = 1.0 / (1.0 + np.exp(-10.0 * (c - threshold) / width))
    else:
        w = np.ones(len(c))
    w_sum = w.sum()
    if w_sum < 1e-12:
        w = np.ones(len(c))
        w_sum = len(c)
    w = w / w_sum
    L = float(np.dot(w, c))
    score = 2.0 * n * L / (S * S)
    
    # Gradient computation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs_topk(a0, k, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B with top-k objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    res = minimize(
        topk_score_grad, a0, args=(k,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def run_lbfgs_lse(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def multiplicative_update(a, y_dual, damping=0.5):
    """Apply multiplicative update using dual variables.
    a_j <- a_j / ((y*a)_j)^damping, then rescale.
    """
    n = len(a)
    if y_dual is None or len(y_dual) == 0:
        return a
    # Compute (y * a) - this is conv(y, a) but we need the "middle" part
    # (y*a)_j = sum_k y_k * a_{k-j} for valid indices
    # This is actually the cross-correlation, which is conv(y, rev_a) shifted
    rev_a = a[::-1]
    cross = fast_conv_cross(y_dual, rev_a)
    # We need cross[n-1 : 2*n-1] to get (y*a)_j for j=0,...,n-1
    # Actually let me think about this more carefully.
    # (y * a)[k] = sum_j y_j * a_{k-j}
    # We want (y * a)[n-1+j] = sum_i y_i * a_{n-1+j-i} for j=0,...,n-1
    # Hmm, this is getting confusing. Let me just use the direct formula.
    # (y*a)_j in the dual constraint means: sum_k y_k * A[k,j] = sum_k y_k * a_{k-j}
    #   where the sum is over k such that 0 <= k-j < n, i.e., j <= k <= j+n-1
    #   = sum_{k=j}^{j+n-1} y_k * a_{k-j}
    # Let's compute this using convolution:
    # Let z = rev(y), then (z * a)[n-1-j + k] ... this is getting messy.
    # Let me just do it directly with FFT.
    # (y*a)_j = sum_{i=0}^{n-1} y[j+i] * a[i] (with y padded)
    # This is the cross-correlation of y and a at lag j.
    # cross_corr(j) = sum_i y[j+i] * a[i] = (rev_y * a)[n-1-j] ... 
    
    # Actually, let me just use the matrix-vector product approach for clarity.
    # (y*a)_j = sum_k y_k * a_{k-j} for k in [j, j+n-1]
    # = sum_{i=0}^{n-1} y_{j+i} * a_i
    # This is the cross-correlation of y and a.
    
    # Using FFT: cross_corr(j) = sum_i a_i * y_{j+i}
    # = (a * rev(y))_... let me think.
    # conv(a, rev(y))[k] = sum_i a_i * rev(y)[k-i] = sum_i a_i * y[n-1-(k-i)] = sum_i a_i * y[n-1-k+i]
    # We want sum_i a_i * y_{j+i}. Set n-1-k+i = j+i => n-1-k = j => k = n-1-j.
    # So cross_corr(j) = conv(a, rev(y))[n-1-j]
    
    rev_y = y_dual[::-1]
    M = len(a) + len(y_dual) - 1
    M_fft = 1 << (M - 1).bit_length()
    A_fft = np.fft.rfft(a, M_fft)
    Y_fft = np.fft.rfft(rev_y, M_fft)
    conv_result = np.fft.irfft(A_fft * Y_fft, M_fft)[:M]
    # cross_corr(j) = conv_result[n-1-j] for j=0,...,n-1
    ya = conv_result[n-1::-1][:n]  # ya[j] = conv_result[n-1-j]
    
    ya = np.maximum(ya, 1e-10)
    a_new = a / (ya ** damping)
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 1e-10:
        return a
    a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def lp_iteration(a, max_time=25.0, use_multiplicative=True):
    """One LP-based iteration. Returns (score, new_a, improved)."""
    old_sc = compute_score(a)
    b_opt, y_dual = solve_lp_for_direction(a, max_time=max_time)
    
    if b_opt is None:
        return old_sc, a, False
    
    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    
    best_sc = old_sc
    best_a = a.copy()
    
    # Try mixing
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        a_new = (1.0 - t) * a + t * b_opt
        a_new = np.clip(a_new, 0.0, 1000.0)
        if a_new.sum() < 0.01:
            continue
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
    
    # Try multiplicative update
    if use_multiplicative and y_dual is not None:
        for damp in [0.3, 0.5, 0.7]:
            a_mult = multiplicative_update(a, y_dual, damping=damp)
            sc = compute_score(a_mult)
            if sc < best_sc:
                best_sc = sc
                best_a = a_mult.copy()
    
    improved = best_sc < old_sc * (1 - 1e-8)
    return best_sc, best_a, improved


def subgradient_polish(a, total_iters=200):
    """Exact subgradient descent on max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = min((sc - sc_lb) / (gn * gn + 1e-30), 0.2)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a
