```python
def spectral_shaping(p, n_iters=100, alpha=0.5):
    """
    Gerchberg-Saxton style spectral shaping to flatten p*p.
    p is a non-negative sequence with sum 1.
    """
    n = len(p)
    N = 1 << (2*n-1).bit_length()  # FFT length
    target_mag = np.sqrt(N / (2*n-1))  # target flat magnitude
    
    p = p.copy()
    for it in range(n_iters):
        P = np.fft.rfft(p, N)
        # Blend magnitude toward target
        mag = np.abs(P)
        phase = np.angle(P)
        new_mag = alpha * target_mag + (1-alpha) * mag
        P_new = new_mag * np.exp(1j * phase)
        # Recover time domain
        p_new = np.fft.irfft(P_new, N)[:n]
        # Project onto non-negative, sum=1
        p_new = np.maximum(0, p_new)
        s = p_new.sum()
        if s < 1e-12:
            break
        p_new /= s
        p = p_new
    return p
```

Let me try yet another approach: **Optimal control / greedy peak reduction**.

**Greedy peak reduction:**
1. Find the peak position k* in p*p
2. The peak is (p*p)[k*] = sum_i p[i]*p[k*-i]
3. Greedily reduce the pair (p[i], p[k*-i]) with the largest product
4. Redistribute the mass to positions that are far from k* (so they don't contribute to the peak)
5. Repeat

This is similar to the "active set" approach in the previous SOTA, but simpler.

Let me think about the redistribution. If we reduce p[i] by δ and p[k*-i] by δ (or just one of them), we reduce the peak by approximately 2δ*p[other] (or δ*p[other]). We should add the mass δ to positions j where (p*p)[k*] doesn't increase (or increases minimally).

Position j contributes to (p*p)[k*] through p[j]*p[k*-j]. So we want to add mass to j where p[k*-j] is small (ideally 0).

This gives a natural algorithm:
1. Find k* = argmax(p*p)
2. For each pair (i, k*-i) with both in range, compute the "cost" of reducing p[i]:
   - Benefit: reduces (p*p)[k*] by p[k*-i] * δ
   - Cost: might increase other (p*p)[k] values
3. Reduce the p[i] with highest benefit/cost ratio
4. Add the mass to p[j] where p[k*-j] = 0 (or minimal)

This is essentially a coordinate descent with smart variable selection.

For other k: (p*p)[k] = c_k (unchanged)

So f(x) = max(c_k for unchanged k, x*p[k-j]+c_k for changed k ≠ 2j, x^2+c_{2j} for k=2j if valid)

3. The quadratic x^2+c_{2j} has its minimum at x=0
4. The max of all these is minimized where the active constraints balance

"""
Spectral-shaping + exact coordinate descent + continuation L-BFGS
to minimize 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Improves upon previous SOTA of 1.508084 targeting 1.5027.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def subgrad_max_conv(a):
    """
    Return (score, subgradient) of f(a) = 2n*max(a*a)/(sum a)^2.
    Subgradient uses the argmax of the convolution.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    score = 2.0 * n * M / (S * S)
    
    # Find all positions achieving the max (within tolerance)
    tol = 1e-10 * M
    k_indices = np.where(c >= M - tol)[0]
    
    # Subgradient is average of subgradients at each max position
    sg = np.zeros(n)
    for k in k_indices:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        for j in range(js, je + 1):
            # d(a*a)[k]/da[j] = a[k-j] + a[j-k] (but only one is valid typically)
            # Actually: (a*a)[k] = sum_i a[i]*a[k-i]
            # d/da[j] = a[k-j] (if 0<=k-j<=n-1) + a[j-k] (if 0<=j-k<=n-1)
            # Usually only one term is non-zero
            val = 0.0
            if 0 <= k - j <= n - 1:
                val += a[k - j]
            if 0 <= j - k <= n - 1:
                val += a[j - k]
            sg[j] += val
    
    sg /= len(k_indices)  # Average
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / exp_sum
    
    # Compute gradient: grad[j] = 2n * (sum_k p[k]*(a[k-j]+a[j-k])/S^2 - 2L/S^3)
    # sum_k p[k]*a[k-j] is a cross-correlation
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    A = np.fft.rfft(a, M2_fft)
    
    # term1[j] = sum_k p[k]*a[k-j] (cross-correlation at lag -j)
    # Using: IFFT(conj(P)*A)[m] = sum_k p[k]*a[k+m]
    # So term1[j] = IFFT(conj(P)*A)[-j] = IFFT(conj(P)*A)[M2_fft-j]
    corr = np.fft.irfft(np.conj(P) * A, M2_fft)
    term1 = np.zeros(n)
    term1[0] = corr[0]
    if n > 1:
        term1[1:] = corr[M2_fft - np.arange(1, n)]
    
    # term2[j] = sum_k p[k]*a[j-k] = (p*a)[j] (linear convolution)
    conv_pa = np.fft.irfft(P * A, M2_fft)[:n]
    term2 = conv_pa
    
    conv_grad = term1 + term2
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (conv_grad / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out
