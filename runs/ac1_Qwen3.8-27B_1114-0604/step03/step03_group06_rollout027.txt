"""
Optimized search for non-negative sequences minimizing 2n*max(autoconv)/(sum)^2.
Uses frequency-domain shaped initialization, LP direction finding, L-BFGS-B with
continuation, and peak-shaving heuristics to escape local minima.

import numpy as np
from scipy.optimize import minimize
import time
try:
    import cvxpy as cp
    HAVE_CVXPY = True
except ImportError:
    HAVE_CVXPY = False


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """Linear cross-convolution a*b via FFT."""
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: need convolve(rev_a, p)[n-1:2n-1]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def topk_score_grad(a, k):
    """Soft top-k score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_sorted = np.sort(c)[::-1]
    if k < len(c_sorted):
        threshold = c_sorted[min(k - 1, len(c_sorted) - 1)]
        next_val = c_sorted[min(k + 2, len(c_sorted) - 1)]
        gap = max(threshold - next_val, 1e-8)
        width = gap * 0.3 + 1e-8
        w = 1.0 / (1.0 + np.exp(-10.0 * (c - threshold) / width))
    else:
        w = np.ones(len(c))
    w = w / w.sum()
    L = float(np.dot(w, c))
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, use_topk=False, topk_k=5):
    """Run L-BFGS-B with LSE or top-k objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    if use_topk:
        obj_fun = lambda x: topk_score_grad(x, topk_k)
    else:
        obj_fun = lambda x: lse_score_grad(x, beta)
    
    res = minimize(
        obj_fun, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': maxiter * 2, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def solve_lp_direction(a, M, max_wall_s=60.0):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns optimal b or None.
    """
    if not HAVE_CVXPY:
        return None
    n = len(a)
    t0 = time.monotonic()
    if n > 8000:
        return None
    
    # Build the constraint matrix (Toeplitz structure)
    # (a*b)[k] = sum_j a[k-j] * b[j] for valid j
    # We need: for each k = 0,...,2n-2, sum_j T[k,j] * b[j] <= M
    # where T[k,j] = a[k-j] if 0 <= k-j <= n-1, else 0
    
    # Use sparse-like construction
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        length = k_end - k_start
        rows.extend(range(k_start, k_end))
        cols.extend([j] * length)
        vals.extend(a[:length].tolist())
    
    T_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    
    try:
        remaining = max_wall_s - (time.monotonic() - t0)
        if remaining < 5:
            return None
        prob.solve(solver=cp.PDLP, max_time=remaining)
    except Exception:
        try:
            remaining = max_wall_s - (time.monotonic() - t0)
            if remaining < 5:
                return None
            prob.solve(solver=cp.ECOS, max_time=remaining)
        except Exception:
            return None
    
    if time.monotonic() - t0 > max_wall_s:
        return None
    if b_var.value is None:
        return None
    
    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None
    return b_opt


def lp_iterate(a, max_wall_s=60.0):
    """One LP iteration: find direction, line search, return best."""
    n = len(a)
    old_sc = compute_score(a)
    c = fast_conv(a)
    M = float(np.max(c))
    
    b_opt = solve_lp_direction(a, M, max_wall_s)
    if b_opt is None:
        return old_sc, a
    
    target_sum = float(a.sum())
    b_rescaled = b_opt * (target_sum / b_opt.sum())
    b_rescaled = np.clip(b_rescaled, 0.0, 1000.0)
    
    # Line search
    best_sc = old_sc
    best_a = a.copy()
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        a_new = (1.0 - t) * a + t * b_rescaled
        a_new = np.clip(a_new, 0.0, 1000.0)
        if a_new.sum() < 0.01:
            continue
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
    return best_sc, best_a


def freq_shaped_init(n, n_harmonics=3, rng=None):
    """
    Create a sequence whose autocorrelation is approximately flat,
    using frequency-domain shaping.
    """
    # We want (a*a)[k] to be flat. 
    # In the frequency domain (with zero-padding to N >= 2n-1):
    # |A|^2 = DFT(c) where c is the desired autocorrelation.
    # If c[k] = C for k=0,...,2n-2, then |A[m]| = C * |D_{2n-1}(m)| where D is Dirichlet kernel.
    # Choose phase = 0 for simplicity (gives real, even a).
    
    N = 1 << (2 * n - 1).bit_length()  # Power of 2 >= 2n-1
    m = np.arange(N // 2 + 1)
    
    # Dirichlet kernel of length 2n-1: D[m] = sum_{k=0}^{2n-2} exp(-2πimk/N)
    # = exp(-πi m (2n-2)/N) * sin(π m (2n-1)/N) / sin(π m / N)
    # Magnitude: |sin(π m (2n-1)/N) / sin(π m / N)|
    
    num = np.sin(np.pi * m * (2 * n - 1) / N)
    den = np.sin(np.pi * m / N)
    den[0] = 1.0  # Avoid division by zero
    D = np.abs(num / den)
    D[0] = 2 * n - 1  # DC component
    
    # |A[m]|^2 proportional to D[m]
    # A[m] = sqrt(D[m]) (zero phase)
    A = np.zeros(N, dtype=np.complex128)
    A[:N//2+1] = np.sqrt(D)
    for m_idx in range(1, N//2+1):
        A[N - m_idx] = np.sqrt(D[N - m_idx])  # Conjugate symmetry
    
    a = np.fft.ifft(A).real[:n]
    a = np.clip(a, 0.0, 1000.0)
    
    if a.sum() < 1e-6:
        a[:] = 0.5
    return a


def cos_init(n, freq, amp):
    """Cosine-based initialization."""
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    return np.clip(a, 0.0, 1000.0)


def multi_cos_init(n, n_harmonics=4, rng=None):
    """Multi-harmonic cosine initialization."""
    if rng is None:
        rng = np.random.default_rng()
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for h in range(1, n_harmonics + 1):
        freq = h  # h-th harmonic
        amp = rng.uniform(0.1, 0.5) * (1.0 / h)
        phase = 2.0 * np.pi * rng.random()
        a += amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    return np.clip(a, 0.0, 1000.0)


def peak_shave(a, n_shaves=10, strength=0.05):
    """Iteratively reduce peak autocorrelation by adjusting contributing elements."""
    n = len(a)
    a = a.copy()
    for _ in range(n_shaves):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        # Find elements contributing most to c[k_star]
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        contrib = a[j_start:j_end + 1].copy()
        # Reduce the top contributors
        n_reduce = max(3, len(contrib) // 10)
        top_idx = np.argsort(contrib)[::-1][:n_reduce]
        total_reduce = 0.0
        for idx in top_idx:
            reduce_amt = a[j_start + idx] * strength
            a[j_start + idx] -= reduce_amt
            a[j_start + idx] = max(a[j_start + idx], 0.0)
            total_reduce += reduce_amt
        # Distribute to bottom contributors
        bottom_idx = np.argsort(contrib)[:n_reduce]
        per = total_reduce / max(1, len(bottom_idx))
        for idx in bottom_idx:
            a[j_start + idx] += per
        a = np.clip(a, 0.0, 1000.0)
    return a


def subgrad_polish(a, total_iters=200):
    """Subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient
        sg_M = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        # Handle the diagonal term (when 2j = k_star)
        if k_star % 2 == 0:
            j_diag = k_star // 2
            if 0 <= j_diag < n:
                sg_M[j_diag] -= a[j_diag]  # Correct the double counting
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = min(0.1, (sc - 1.0) / (gn * gn + 1e-30))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a
