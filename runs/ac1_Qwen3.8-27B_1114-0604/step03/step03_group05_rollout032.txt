import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(np.sum(exp_scaled))
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    # Gradient: dL/da[j] = 2 * sum_k p[k] * (a[k-j] if k>=j else a[j-k] if ...)
    # Actually: c[k] = sum_i a[i]*a[k-i], so dc[k]/da[j] = a[k-j] (if valid)
    # dL/da[j] = 2 * sum_k p[k] * dc[k]/da[j] = 2 * sum_k p[k] * a[k-j] (where valid)

    # c[k] = sum_i a[i] * a[k-i]

    # So dc[k]/da[j] = a[k-j] + a[j] (if both indices are valid)
    # Wait, that's only if j ≠ k-j, i.e., j ≠ k/2.
    # More carefully: c[k] = sum_i a[i] * a[k-i]
    # dc[k]/da[j] = a[k-j] (from the i=j term) + a[k-j] (from the k-i=j term, i.e., i=k-j)
    # But if j = k-j (i.e., k = 2j), it's just a[j] (not 2*a[j]).
    # Hmm, actually: dc[k]/da[j] = a[k-j] (if 0 ≤ k-j < n) + a[k-j] (if 0 ≤ k-j < n, from the other factor)
    # Wait, I'm confusing myself. Let me be more careful.
    # c[k] = sum_{i} a[i] * a[k-i]
    # dc[k]/da[j] = (d/da[j]) sum_i a[i] * a[k-i]
    # = a[k-j] (from the term where i=j in the first factor) + a[k-j] (from the term where k-i=j, i.e., i=k-j, in the second factor)
    # But if j = k-j, these are the same term, so we get a[j] not 2*a[j].
    # Actually no: when i=j, we get a[j] * a[k-j]. The derivative w.r.t. a[j] of this term is a[k-j] (treating a[k-j] as constant if k-j ≠ j) or 2*a[j] (if k-j = j).
    # And from the term where i=k-j (if k-j ≠ j), we get a[k-j]*a[j], derivative is a[k-j].
    # So: dc[k]/da[j] = a[k-j] if k-j ≠ j (and valid), or 2*a[j] if k-j = j (i.e., k=2j), or just a[k-j] if only one of i=j or i=k-j is valid.
    
    # This is getting complicated. Let me use the FFT approach.
    # sum_k p[k] * dc[k]/da[j] = sum_k p[k] * (a[k-j] + a[k-j] adjusted)
    # = 2 * conv(p, a)[j] (approximately, for interior points)
    # More precisely, it's the "cross-correlation" of p and a, which is conv(p_rev, a) or conv(p, a_rev).
    
    # Let's use: g[j] = sum_k p[k] * dc[k]/da[j]
    # For the "full" convolution sense (ignoring boundary effects for now):
    # g = 2 * conv(p_flipped, a) [padded to length n]
    # where p_flipped = p[::-1]
    
    # Actually, the cleanest way:
    # g[j] = sum_k p[k] * a[k-j] (where k-j is in range [0,n-1])
    #      + sum_k p[k] * a[k-j] (same thing, from the other factor)
    # = 2 * sum_k p[k] * a[k-j] (for interior j where both terms are valid)
    
    # This is 2 * (circular cross-correlation of p and a), which can be computed via FFT.
    # Using: g[j] = 2 * (ifft(fft(p) * conj(fft(a))))[j] ... no, that's not right either.

    # This is (p * a)[j] in the convolution sense (with appropriate padding).
    # h = conv(p, a_rev)[n-1 : 2n-2] ... hmm, let me just use the direct approach.

    # g[j] = sum_{k=0}^{2n-2} p[k] * dc[k]/da[j]
    # where dc[k]/da[j] = a[k-j] (if 0 ≤ k-j ≤ n-1) + (a[j-k+n-1] ... no)

    # = dot(a, conv(p, a_rev))  where a_rev = a[::-1]

    # I'll just use the FFT-based approach that was in the previous code.
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lse_obj(a, beta):
    return lse_score_grad(a, beta)

def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0

def peak_shave(a, iters=3000):
    """Subgradient descent on max-conv with Polyak step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0  # theoretical lower bound
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-20)
        step = min(step, 1.0 / S)
        step = max(step, 1e-6 / (gn * S))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a

def multi_peak_equalize(a, K=5, iters=200):
    """Try to equalize the top K peaks of the convolution."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        
        # Find top K peaks
        sorted_idx = np.argsort(c)[::-1][:K]
        peak_vals = c[sorted_idx]
        target = peak_vals.mean()  # try to make all top K equal
        
        # For each peak, compute the subgradient and try to reduce it
        # while increasing the others
        for k_star in sorted_idx:
            k_star = int(k_star)
            if c[k_star] <= target:
                continue
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            # Elements contributing to this peak
            contrib = a[j_start:j_end+1][::-1].copy()
            # Reduce the top contributors
            top_thresh = np.percentile(contrib[contrib > 0], 70) if (contrib > 0).any() else 0
            top_mask = contrib > top_thresh
            total_reduce = 0.0
            for idx in np.where(top_mask)[0]:
                reduce_amt = 0.01 * contrib[idx]
                a[j_start + idx] -= reduce_amt
                total_reduce += reduce_amt
            # Distribute to bottom elements
            bottom_candidates = np.where(a < 0.3 * a.mean())[0]
            if len(bottom_candidates) > 0:
                per = total_reduce / len(bottom_candidates)
                a[bottom_candidates] += per
            
        a = np.clip(a, 0.0, 1000.0)
        sc_new = compute_score(a)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a.copy()
        elif sc_new > sc * 1.01:
            # Revert if it got too much worse
            a = best_a.copy()
    
    return best_sc, best_a

def simulated_annealing(a0, t_start, t_end, rng):
    """Simulated annealing with convolution-aware moves."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    current_sc = best_sc
    
    # Temperature schedule
    T_start = 0.001
    T_end = 1e-7
    total_iters = int((t_end - t_start) / 0.005)  # estimate
    total_iters = min(total_iters, 50000)
    
    iter_count = 0
    while time.monotonic() < t_end and iter_count < total_iters:
        iter_count += 1
        progress = iter_count / total_iters
        T = T_start * (T_end / T_start) ** progress
        
        # Choose a move type
        move_type = rng.integers(0, 5)
        a_new = a.copy()
        
        if move_type == 0:
            # Pair swap: transfer from one element to another
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                amt = 0.01 * a.mean() * rng.random()
                a_new[i] = max(0, a_new[i] - amt)
                a_new[j] = min(1000, a_new[j] + amt)
        elif move_type == 1:
            # Block perturbation
            block_len = max(1, n // 100)
            start = rng.integers(0, n - block_len)
            amp = 0.02 * a.mean() * rng.random()
            a_new[start:start+block_len] += amp * rng.standard_normal(block_len)
            a_new = np.clip(a_new, 0, 1000)
        elif move_type == 2:
            # Peak-targeted reduction
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            # Reduce top contributors to peak
            contrib = a[j_start:j_end+1][::-1].copy()
            top_count = max(1, len(contrib) // 10)
            top_idx = np.argsort(contrib)[::-1][:top_count]
            total_red = 0
            for idx in top_idx:
                red = 0.005 * contrib[idx]
                a_new[j_start + idx] = max(0, a_new[j_start + idx] - red)
                total_red += red
            # Distribute to random other positions
            other_idx = rng.choice(np.setdiff1d(np.arange(n), j_start + top_idx), size=min(total_red > 0 and 10 or 1, n), replace=False)
            if len(other_idx) > 0:
                a_new[other_idx] += total_red / len(other_idx)
            a_new = np.clip(a_new, 0, 1000)
        elif move_type == 3:
            # Frequency domain nudge
            L = 1 << (2*n - 1).bit_length()
            A = np.fft.fft(a, L)
            k = rng.integers(1, L//2)
            delta = 0.01 * np.abs(A).mean() * complex(rng.standard_normal(), rng.standard_normal())
            A[k] += delta
            A[L-k] += np.conj(delta)
            a_new = np.fft.ifft(A).real[:n]
            a_new = np.clip(a_new, 0, 1000)
        else:
            # Gaussian perturbation with re-normalization
            eps = 0.005 * rng.standard_normal(n)
            a_new = a + eps
            a_new = np.clip(a_new, 0, 1000)
            if a_new.sum() > 0:
                a_new *= a.sum() / a_new.sum()
        
        sc_new = compute_score(a_new)
        
        # Acceptance criterion
        if sc_new < current_sc:
            a = a_new
            current_sc = sc_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()
        else:
            # Boltzmann acceptance
            delta = sc_new - current_sc
            if delta / max(T, 1e-15) < -np.log(rng.random() + 1e-15):
                a = a_new
                current_sc = sc_new
        
        # Report periodically
        if iter_count % 1000 == 0:
            print(f"    SA iter {iter_count}, T={T:.2e}, current={current_sc:.8f}, best={best_sc:.8f}")
    
    return best_sc, best_a

def spectral_flatten(a, iters=1000):
    """Frequency domain spectral flattening."""
    n = len(a)
    L = 1 << (2*n - 1).bit_length()
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    A = np.fft.fft(a, L)
    
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        sc = 2.0 * n * float(c.max()) / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Compute current spectrum
        A = np.fft.fft(a, L)
        C = A * A  # Spectrum of convolution
        
        # Target: flatten C (make it more uniform)
        # Use a soft version: C_target = C^alpha for alpha < 1
        C_abs = np.abs(C)
        C_mean = C_abs.mean()
        # Only modify frequencies where C is above mean
        mask = C_abs > C_mean
        if mask.sum() == 0:
            break
        C_target = C_abs.copy()
        C_target[mask] = C_mean + 0.1 * (C_abs[mask] - C_mean)
        
        # Adjust A to match C_target
        ratio = np.sqrt(C_target / (C_abs + 1e-30))
        A_new = A * ratio
        # Make it conjugate-symmetric (for real output)
        A_new[0] = np.abs(A_new[0])
        for k in range(1, L//2 + 1):
            if k <= L - 1 - k:
                A_new[L-k] = np.conj(A_new[k])
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.clip(a_new, 0, 1000)
        
        # Normalize to maintain sum
        if a_new.sum() > 0:
            a_new *= a.sum() / a_new.sum()
        
        a = a_new
    
    return best_sc, best_a

    conv_res = np.fft.irfft(RA * PC, M2_fft)[n - 1:2 * n - 1]
    dnorm_da = 2.0 * coeff * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dnorm_da / S2 - 2.0 * norm_val / S3)
    return score, grad


def pnorm_obj(a, p_power):
    return pn

orm_score_grad(a, p_power)


def lse_obj(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def run_lbfgs_log(x0, beta, maxiter=2000):
    """L-BFGS-B in log-parametrization: a = exp(x), with mean(x)=0 constraint via projection."""
    n = len(x0)
    x0 = x0.astype(np.float64).copy()
    x0 -= x0.mean()  # enforce mean=0
    bounds = [(-15.0, 15.0)] * n

    def log_obj(x_grad):
        x_grad = x_grad - x_grad.mean()
        a = np.exp(x_grad)
        a = np.clip(a, 1e-15, 1000.0)
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        c_max = float(np.max(c))
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -700.0, 700.0)
        exp_scaled = np.exp(scaled)
        exp_sum = float(exp_scaled.sum())
        if exp_sum < 1e-300:
            p = np.zeros(len(c))
            p[int(np.argmax(c))] = 1.0
        else:
            p = exp_scaled / exp_sum
        L = c_max + np.log(exp_sum) / beta
        score = 2.0 * n * L / (S * S)
        rev_a = a[::-1]
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
        dL_da = 2.0 * conv_res
        S2 = S * S
        S3 = S2 * S
        dL_da_full = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
        # Chain rule: d/dx = d/da * da/dx = d/da * a
        dL_dx = dL_da_full * a
        # Remove mean component (constraint sum(x)=0)
        dL_dx -= dL_dx.mean()
        return score, dL_dx

    try:
        res = minimize(
            log_obj, x0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxiter * 30, 'maxls': 30}
        )
        x_opt = res.x - res.x.mean()
        x_opt = np.clip(x_opt, -15.0, 15.0)
        a_opt = np.exp(x_opt)
        a_opt = np.clip(a_opt, 0.0, 1000.0)
        return a_opt
    except Exception:
        return np.exp(np.clip(x0, -15, 15))


def projected_peak_reduce(a, iters=5000, epsilon=0.02):
    """Reduce peaks using exact projected gradient steps with first-order line search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Gradient of c[k_star] w.r.t. a: g[j] = 2*a[k_star - j] (if valid)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        g = np.zeros(n)
        g[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        gnorm2 = float(g @ g)
        if gnorm2 < 1e-20:
            break

        # Step size: reduce c[k_star] by epsilon * M
        # First-order: c[k*](a - t*g) ≈ M - t * gnorm2
        # So t ≈ epsilon * M / gnorm2
        t_target = epsilon * M / gnorm2
        # Constraint: a - t*g >= 0 => t <= min_{j: g[j]>0} a[j]/g[j]
        pos = g > 1e-15
        if not pos.any():
            break
        t_max = float(np.min(a[pos] / g[pos]))
        t = min(t_target, 0.5 * t_max)
        t = max(t, 1e-10 * t_max)

        a_new = a - t * g
        a_new = np.clip(a_new, 0.0, 1000.0)

        # Check improvement
        sc_new = 2.0 * n * float(fast_conv(a_new).max()) / (float(a_new.sum()) ** 2)
        if sc_new < sc:
            a = a_new
        else:
            # Try smaller step
            a_new = a - t * 0.1 * g
            a_new = np.clip(a_new, 0.0, 1000.0)
            sc_new2 = 2.0 * n * float(fast_conv(a_new).max()) / (float(a_new.sum()) ** 2)
            if sc_new2 < sc:
                a = a_new
            else:
                # Skip this iteration
                epsilon *= 0.9
                epsilon = max(epsilon, 0.001)

    return best_sc, best_a


def simulated_annealing(a0, t_deadline, rng, n_iters_max=100000):
    """Simulated annealing with convolution-aware moves."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    current_sc = best_sc

    T_start = 0.0005
    T_end = 1e-8
    iter_count = 0
    accept_count = 0
    best_count = 0

    while time.monotonic() < t_deadline and iter_count < n_iters_max:
        iter_count += 1
        progress = min(1.0, iter_count / n_iters_max)
        T = T_start * (T_end / T_start) ** progress

        move_type = rng.integers(0, 6)
        a_new = a.copy()
        S = float(a.sum())

        if move_type == 0:
            # Pair transfer (preserves sum exactly)
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                amt = 0.005 * S / n * (0.5 + rng.random())
                a_new[i] = max(0.0, a_new[i] - amt)
                a_new[j] = min(1000.0, a_new[j] + amt)
        elif move_type == 1:
            # Block perturbation
            block_len = max(2, n // 200)
            start = int(rng.integers(0, n - block_len))
            amp = 0.01 * S / n * rng.random()
            pert = amp * rng.standard_normal(block_len)
            # Zero-sum perturbation to preserve total
            pert -= pert.mean()
            a_new[start:start + block_len] += pert
            a_new = np.clip(a_new, 0, 1000)
        elif move_type == 2:
            # Peak-targeted: reduce top contributors to current peak
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            seg = a[j_start:j_end + 1]
            if seg.max() > 0:
                # Reduce top few in the segment
                thresh = seg.max() * 0.7
                top_mask = seg > thresh
                if top_mask.any():
                    total_red = 0.0
                    idx_top = np.where(top_mask)[0]
                    for idx in idx_top:
                        red = 0.01 * seg[idx] * rng.random()
                        a_new[j_start + idx] = max(0, a_new[j_start + idx] - red)
                        total_red += red
                    # Distribute to random other positions
                    other = rng.choice(np.setdiff1d(np.arange(n), j_start + idx_top),
                                       size=min(20, n), replace=False)
                    if len(other) > 0:
                        a_new[other] += total_red / len(other)
            a_new = np.clip(a_new, 0, 1000)
        elif move_type == 3:
            # Frequency domain nudge
            L = 1 << (2 * n - 1).bit_length()
            A = np.fft.rfft(a, L)
            k = int(rng.integers(1, L // 2))
            mag_scale = 0.005 * np.abs(A).mean()
            delta_mag = mag_scale * rng.standard_normal()
            delta_phase = 0.01 * rng.standard_normal()
            A[k] *= (1.0 + delta_mag)
            A[k] *= np.exp(1j * delta_phase)
            a_new = np.fft.irfft(A, L)[:n]
            a_new = np.clip(a_new, 0, 1000)
            if a_new.sum() > 0:
                a_new *= S / a_new.sum()
        elif move_type == 4:
            # Sinusoidal modulation
            freq = 1.0 + 5.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.003 * rng.random()
            x = np.arange(n, dtype=np.float64) / n
            mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase)
            a_new = a * mod
            a_new = np.clip(a_new, 0, 1000)
        else:
            # Pair exchange (swap portions)
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                swap_amt = 0.5 * min(a[i], a[j]) * (0.1 + 0.9 * rng.random())
                a_new[i] -= swap_amt
                a_new[j] += swap_amt
                a_new = np.clip(a_new, 0, 1000)

        sc_new = compute_score(a_new)

        if sc_new < current_sc:
            a = a_new
            current_sc = sc_new
            accept_count += 1
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()
                best_count += 1
        else:
            delta = sc_new - current_sc
            if delta < T:
                if np.log(rng.random() + 1e-15) > -delta / max(T, 1e-15):
                    a = a_new
                    current_sc = sc_new
                    accept_count += 1

        if iter_count % 5000 == 0:
            print(f"    SA iter {iter_count}, T={T:.2e}, cur={current_sc:.8f}, best={best_sc:.8f}, accepts={accept_count}")

    return best_sc, best_a


def spectral_flatten(a, iters=2000):
    """Iteratively flatten the convolution spectrum in frequency domain."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        sc = 2.0 * n * float(c.max()) / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        A = np.fft.rfft(a, L)
        C = A * A  # Frequency content of convolution

        # Target: reduce peak in C (which corresponds to making c flatter)
        C_mag = np.abs(C)
        c_max = float(c.max())
        # Softly reduce peaks in frequency domain
        alpha = 0.05
        mask = C_mag > np.median(C_mag)
        if mask.sum() == 0 or mask.sum() == len(C_mag):
            # Try a different approach: scale toward mean
            C_mag_target = C_mag * (1.0 - alpha) + np.mean(C_mag) * alpha
        else:
            C_mag_target = C_mag.copy()
            C_mag_target[mask] = (1.0 - alpha) * C_mag[mask] + 0.1 * np.mean(C_mag)

        # Update A magnitude
        A_new = A * (C_mag_target / (C_mag + 1e-30)) ** 0.5
        # Ensure conjugate symmetry for real output (rfft handles this)
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.clip(a_new, 0, 1000)
        if a_new.sum() > 0:
            a_new *= S / a_new.sum()
        a = a_new

    return best_sc, best_a


def multi_peak_waterfill(a, K=10, iters=3000):
    """Water-filling: iteratively equalize the top K peaks."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top K peaks
        sorted_idx = np.argsort(c)[::-1][:K]
        top_vals = c[sorted_idx]
        target = float(top_vals.mean())

        # For each peak above target, apply a small reduction step
        for k_star in sorted_idx[:3]:
            k_star = int(k_star)
            if c[k_star] <= target * 1.001:
                continue
            # Subgradient for this peak
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            g = np.zeros(n)
            g[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
            gnorm2 = float(g @ g)
            if gnorm2 < 1e-20:
                continue
            # Step to reduce this peak toward target
            excess = c[k_star] - target
            t = min(excess / gnorm2, 0.1 * float(np.min(a[g > 1e-15] / g[g > 1e-15])) + 1e-10)
            a_new = a - t * g
            a_new = np.clip(a_new, 0, 1000)
            # Accept if it doesn't worsen the global max too much
            sc_check = 2.0 * n * float(fast_conv(a_new).max()) / (float(a_new.sum()) ** 2)
            if sc_check < sc * 1.005:
                a = a_new

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-strategy search: log-param LSE, LSE continuation, p-norm optimization,
    projected peak reduction, water-filling, simulated annealing, spectral flattening.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(2048) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    if parent is not None:
        sc = compute_score(parent)
        report(sc, parent, "parent")
        print(f"Parent: score={sc:.8f}, length={len(parent)}")

    n_use = len(best_a)
    print(f"Working with n = {n_use}")

    # =================================================================
    # Phase 1: Quick LSE to establish baseline (10% time)
    # =================================================================
    phase1_end = t_start + 0.10 * (total_budget - 30.0)
    print("\n=== Phase 1: Initial LSE ===")
    a_cur = best_a.copy()
    for beta in [30, 100, 500]:
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        a_cur = run_lbfgs(a_cur, lse_obj, (beta,), maxiter=500, maxfun=10000)
        sc = compute_score(a_cur)
        report(sc, a_cur, f"LSE beta={beta}")
        print(f"  beta={beta}: score={sc:.8f}, t={time.monotonic()-t0:.1f}s")

    # =================================================================
    # Phase 2: Log-parametrized LSE (different landscape) (15% time)
    # =================================================================
    phase2_end = t_start + 0.25 * (total_budget - 30.0)
    print("\n=== Phase 2: Log-Parametrized LSE ===")
    if time.monotonic() < phase2_end:
        x0 = np.log(np.maximum(best_a, 1e-10))
        x0 -= x0.mean()
        for beta in [50, 200, 1000, 3000]:
            if time.monotonic() > phase2_end:
                break
            t0 = time.monotonic()
            a_log = run_lbfgs_log(x0, beta, maxiter=1000)
            sc = compute_score(a_log)
            report(sc, a_log, f"log-LSE beta={beta}")
            print(f"  log-beta={beta}: score={sc:.8f}, t={time.monotonic()-t0:.1f}s")
            x0 = np.log(np.maximum(a_log, 1e-10))
            x0 -= x0.mean()

    # =================================================================
    # Phase 3: LSE continuation (main, 20% time)
    # =================================================================
    phase3_end = t_start + 0.45 * (total_budget - 30.0)
    print("\n=== Phase 3: LSE Continuation ===")
    a_cur = best_a.copy()
    for beta in [1000, 3000, 10000, 30000]:
        if time.monotonic() > phase3_end:
            break
        t0 = time.monotonic()
        a_cur = run_lbfgs(a_cur, lse_obj, (beta,), maxiter=1500, maxfun=30000)
        sc = compute_score(a_cur)
        report(sc, a_cur, f"cont-LSE beta={beta}")
        print(f"  beta={beta}: score={sc:.8f}, t={time.monotonic()-t0:.1f}s")

    # =================================================================
    # Phase 4: p-norm optimization (10% time)
    # =================================================================
    phase4_end = t_start + 0.55 * (total_budget - 30.0)
    print("\n=== Phase 4: p-Norm Optimization ===")
    if time.monotonic() < phase4_end:
        a_cur = best_a.copy()
        for p_val in [8, 16, 32]:
            if time.monotonic() > phase4_end:
                break
            t0 = time.monotonic()
            a_cur = run_lbfgs(a_cur, pnorm_obj, (p_val,), maxiter=1000, maxfun=20000)
            sc = compute_score(a_cur)
            report(sc, a_cur, f"p-norm p={p_val}")
            print(f"  p={p_val}: score={sc:.8f}, t={time.monotonic()-t0:.1f}s")

    # =================================================================
    # Phase 5: Projected peak reduction (15% time)
    # =================================================================
    phase5_end = t_start + 0.70 * (total_budget - 30.0)
    print("\n=== Phase 5: Projected Peak Reduction ===")
    if time.monotonic() < phase5_end:
        t0 = time.monotonic()
        remaining_s = phase5_end - time.monotonic()
        iters = int(min(20000, remaining_s * 200))
        sc, a_ppr = projected_peak_reduce(best_a, iters=max(1000, iters), epsilon=0.01)
        report(sc, a_ppr, "projected")
        print(f"  Projected: score={sc:.8f}, t={time.monotonic()-t0:.1f}s")
        # Follow up with LSE
        if time.monotonic() < phase5_end:
            a_follow = run_lbfgs(a_ppr, lse_obj, (5000,), maxiter=500, maxfun=10000)
            sc = compute_score(a_follow)
            report(sc, a_follow, "projected+LSE")

    # =================================================================
    # Phase 6: Simulated annealing (15% time)
    # =================================================================
    phase6_end = t_start + 0.85 * (total_budget - 30.0)
    print("\n=== Phase 6: Simulated Annealing ===")
    if time.monotonic() < phase6_end:
        t0 = time.monotonic()
        sc, a_sa = simulated_annealing(best_a, phase6_end, rng)
        report(sc, a_sa, "annealing")
        print(f"  Annealing: score={sc:.8f}, t={time.monotonic()-t0:.1f}s")

    # =================================================================
    # Phase 7: Water-filling + final polish (15% time)
    # =================================================================
    print("\n=== Phase 7: Water-Filling & Final Polish ===")
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        sc, a_wf = multi_peak_waterfill(best_a, K=8, iters=2000)
        report(sc, a_wf, "waterfill")
        print(f"  Waterfill: score={sc:.8f}, t={time.monotonic()-t0:.1f}s")

    if time.monotonic() < deadline:
        # Final high-beta LSE
        t0 = time.monotonic()
        a_final = run_lbfgs(best_a, lse_obj, (50000,), maxiter=1000, maxfun=20000)
        sc = compute_score(a_final)
        report(sc, a_final, "final-LSE")
        print(f"  Final LSE: score={sc:.8f}, t={time.monotonic()-t0:.1f}s")

    if time.monotonic() < deadline:
        # Final subgradient
        t0 = time.monotonic()
        sc, a_sub = projected_peak_reduce(best_a, iters=2000, epsilon=0.005)
        report(sc, a_sub, "final-subgrad")
        print(f"  Final subgrad: score={sc:.8f}, t={time.monotonic()-t0:.1f}s")

    if time.monotonic() < deadline:
        # Spectral flatten
        t0 = time.monotonic()
        sc, a_spec = spectral_flatten(best_a, iters=500)
        report(sc, a_spec, "spectral")
        print(f"  Spectral: score={sc:.8f}, t={time.monotonic()-t0:.1f}s")

    # =================================================================
    # Final output
    # =================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
