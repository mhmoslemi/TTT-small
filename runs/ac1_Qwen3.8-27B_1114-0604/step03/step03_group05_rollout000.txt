"""
Step function optimization using LP-based direction finding, LSE continuation,
and Nesterov-accelerated subgradient descent.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def conv_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a) + len(b) - 1
    M_fft = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:n]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)

    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

        return np.ones(n)

    # = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * d[k-j]
    # Let i = k-j, so j = k-i:
    # = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[k-i] * d[i]

def nesterov_subgradient(a, f_eval, f_grad, max_iter=5000, verbose=False):

    Nesterov accelerated gradient descent for the non-smooth max-conv objective.
    f(a) = 2n * max(c) / S^2

    v = a.copy()  # momentum variable

    best_sc = f_eval(a)
