"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Strategy: LSE-continuation L-BFGS with adaptive beta, subgradient with momentum,
structured perturbations, and multi-length exploration. Improves on previous 1.509482.

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 0)
    exp_s = np.exp(scaled)
    exp_sum = exp_s.sum()
    if exp_sum < 1e-300:
        exp_sum = 1e-300
    L = c_max + np.log(exp_sum) / beta
    p = exp_s / exp_sum
    score = 2.0 * n * L / (S * S)
    # Gradient: need dc/da which involves cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Scipy-compatible objective+gradient."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgrad_polish(a, max_iters=5000, momentum=0.9):
    """Subgradient descent with momentum and Polyak step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    v = np.zeros(n)

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max convolution
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

        # Momentum update
        v = momentum * v - grad
        gn2 = float(v @ v)
        if gn2 < 1e-30:
            break
        # Polyak step with decay
        polyak = (sc - sc_lb) / gn2
        decay = 1.0 / (1.0 + 0.002 * it)
        step = min(polyak, 1.0 * decay)
        a = a - step * v
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else x.copy()

    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())

    # Uniform
    inits.append(np.ones(n) * 1.0)

    # Anti-palindromic ramps
    inits.append(np.linspace(0.05, 1.5, n))
    inits.append(np.linspace(1.5, 0.05, n))

    # Quadratic anti-palindromic
    inits.append(np.clip(0.05 + 1.45 * xn ** 2, 0.01, 1000.0))
    inits.append(np.clip(0.05 + 1.45 * (1.0 - xn) ** 2, 0.01, 1000.0))

    # Anti-palindromic sinusoidal
    for freq in [1, 2, 3]:
        inits.append(np.clip(0.8 + 0.5 * np.sin(np.pi * freq * xn), 0.01, 1000.0))

    # Frequency-modulated
    inits.append(np.clip(0.5 + 0.3 * np.sin(2 * np.pi * 5 * xn) + 0.2 * xn, 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.3 * np.cos(2 * np.pi * 8 * xn) + 0.2 * (1 - xn), 0.01, 1000.0))

    # Smoothed random
    a_rand = np.clip(0.8 + 0.4 * rng.standard_normal(n), 0.01, 1000.0)
    ksize = max(1, n // 100)
    kernel = np.ones(ksize) / ksize
    a_smooth = np.convolve(a_rand, kernel, mode='same')
    inits.append(np.clip(a_smooth, 0.01, 1000.0))

    # Pure random
    inits.append(np.clip(0.8 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))

    # Two-block
    half = n // 2
    inits.append(np.concatenate([np.ones(half) * 1.2, np.ones(n - half) * 0.4]))
    inits.append(np.concatenate([np.ones(half) * 0.4, np.ones(n - half) * 1.2]))

    return inits


def lse_continuation(a0, beta_schedule, maxiter_per_beta=3000):
    """Run LSE continuation: progressively increase beta."""
    a_cur = a0.copy()
    for beta in beta_schedule:
        sc, a_new = run_lbfgs(a_cur, beta, maxiter=maxiter_per_beta, maxfun=maxiter_per_beta * 20)
        a_cur = a_new
    return a_cur


def structured_perturb(a, rng, mode=0):
    """Apply structured perturbations to escape local minima."""
    n = len(a)
    a_pert = a.copy()
    x = np.arange(n, dtype=np.float64) / (n - 1) if n > 1 else np.zeros(n)

    if mode == 0:
        # Gaussian noise
        scale = 0.02 + 0.03 * rng.random()
        a_pert = a * (1.0 + scale * rng.standard_normal(n))
    elif mode == 1:
        # Anti-palindromic bump in the middle
        center = n // 2
        width = max(1, n // 10)
        for j in range(n):
            d = (j - center) / width
            sign = 1.0 if j < center else -1.0
            a_pert[j] *= (1.0 + 0.05 * sign * np.exp(-d * d))
    elif mode == 2:
        # Targeted: reduce top contributors to peak lag
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        contrib = np.zeros(n)
        if j_end >= j_start:
            for j in range(j_start, j_end + 1):
                contrib[j] = a[j] * a[k_star - j]
        top_n = max(5, n // 50)
        top_idx = np.argsort(contrib)[::-1][:top_n]
        bot_idx = np.argsort(contrib)[-top_n * 2:]
        total_red = 0.0
        for j in top_idx:
            amt = a_pert[j] * 0.05
            a_pert[j] -= amt
            a_pert[j] = max(a_pert[j], 0.0)
            total_red += amt
        if len(bot_idx) > 0 and total_red > 0:
            per = total_red / len(bot_idx)
            a_pert[bot_idx] += per
    elif mode == 3:
        # Frequency-domain shaping
        M_fft = 1 << (2 * n - 2).bit_length()
        A = np.fft.rfft(a_pert, M_fft)
        freqs = np.arange(len(A))
        att = 1.0 - 0.05 * (freqs / len(A)) ** 2
        A *= att
        a_pert = np.real(np.fft.irfft(A, M_fft))[:n]
    elif mode == 4:
        # Edge tapering
        taper = 1.0 - 0.3 * np.abs(2.0 * x - 1.0)
        a_pert = a * taper

    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Multi-phase optimization with LSE-continuation,
    subgradient polishing, and structured local search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(512) * 1.0

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for n_try in [6962, 8192, 10240, 12288]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))
    # Keep at most 4 lengths, prioritize parent size
    lengths = lengths[:4]

    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}")

    # ============================================================
    # Phase 1: Quick LSE-continuation from best inits for each n
    # ============================================================
    phase1_end = t_start + 0.35 * total_budget
    print(f"\n=== Phase 1: LSE Continuation (until t={phase1_end - t_start:.0f}s) ===")

    beta_schedule_p1 = [20, 50, 150, 500, 1500]

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"\n  Processing n={n}")
        inits = build_inits(n, rng, parent if (parent is not None and parent.size == n) else None)

        # Prioritize: parent first, then anti-palindromic, then others
        # Limit number of inits based on available time
        time_left = phase1_end - time.monotonic()
        n_inits_max = min(len(inits), max(2, int(time_left / 60)))
        n_inits_max = min(n_inits_max, 8)

        for i in range(n_inits_max):
            if time.monotonic() > phase1_end:
                break
            t0 = time.monotonic()
            a_cur = inits[i].copy()
            # Run LSE continuation
            for beta in beta_schedule_p1:
                if time.monotonic() > phase1_end:
                    break
                remaining = phase1_end - time.monotonic()
                maxit = max(500, min(3000, int(remaining / (0.002 * n + 1))))
                sc, a_new = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxit * 25)
                a_cur = a_new
            sc = compute_score(a_cur)
            report(sc, a_cur, f"P1 n={n} i={i}")
            print(f"    init {i}: score={sc:.8f} (t={time.monotonic()-t0:.1f}s)")

    print(f"\n  After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: High-beta refinement on best solution
    # ============================================================
    phase2_end = t_start + 0.60 * total_budget
    print(f"\n=== Phase 2: High-Beta Refinement (until t={phase2_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    beta_schedule_p2 = [3000, 8000, 15000, 30000]

    for beta in beta_schedule_p2:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        maxit = max(1000, min(8000, int(remaining / (0.003 * n_best + 1))))
        sc, a_new = run_lbfgs(best_a, beta, maxiter=maxit, maxfun=maxit * 20)
        report(sc, a_new, f"P2 beta={beta}")
        print(f"    beta={beta}: score={compute_score(a_new):.8f}")

    # ============================================================
    # Phase 3: Subgradient polish with momentum
    # ============================================================
    phase3_end = t_start + 0.75 * total_budget
    print(f"\n=== Phase 3: Subgradient Polish (until t={phase3_end - t_start:.0f}s) ===")

    if time.monotonic() < phase3_end:
        n_iters = max(1000, int((phase3_end - time.monotonic()) * 80))
        sc, a_opt = subgrad_polish(best_a, max_iters=n_iters, momentum=0.85)
        report(sc, a_opt, "subgrad1")

        # Second round with lower momentum
        if time.monotonic() < phase3_end:
            n_iters2 = max(500, int((phase3_end - time.monotonic()) * 60))
            sc, a_opt = subgrad_polish(best_a, max_iters=n_iters2, momentum=0.7)
            report(sc, a_opt, "subgrad2")

    print(f"\n  After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Local search with structured perturbations + re-optimization
    # ============================================================
    phase4_end = t_start + 0.90 * total_budget
    print(f"\n=== Phase 4: Local Search (until t={phase4_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    trial = 0
    stagnant = 0
    while time.monotonic() < phase4_end - 10 and trial < 40:
        trial += 1
        mode = trial % 5
        a_pert = structured_perturb(best_a, rng, mode=mode)
        if a_pert.sum() < 1e-6:
            a_pert = best_a * 0.9

        # Short L-BFGS refinement
        beta_ls = 5000 if trial < 20 else 10000
        maxit_ls = max(800, min(3000, int((phase4_end - time.monotonic()) / 3)))
        sc, a_new = run_lbfgs(a_pert, beta_ls, maxiter=maxit_ls, maxfun=maxit_ls * 15)
        improved = report(sc, a_new, f"P4 #{trial} m={mode}")
        if improved:
            stagnant = 0
        else:
            stagnant += 1
        if stagnant >= 8:
            # Try stronger perturbation
            a_pert = structured_perturb(best_a, rng, mode=0)
            a_pert = a_pert * (1.0 + 0.1 * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            sc, a_new = run_lbfgs(a_pert, 2000, maxiter=1500, maxfun=30000)
            improved = report(sc, a_new, f"P4 #{trial} restart")
            stagnant = 0

    # ============================================================
    # Phase 5: Final subgradient + L-BFGS polish
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")
    if time.monotonic() < deadline - 30:
        n_iters = max(500, int((deadline - 30 - time.monotonic()) * 50))
        sc, a_opt = subgrad_polish(best_a, max_iters=n_iters, momentum=0.5)
        report(sc, a_opt, "final_subgrad")

    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(best_a, 20000, maxiter=2000, maxfun=40000)
        report(sc, a_opt, "final_lbfgs")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [1.0] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
