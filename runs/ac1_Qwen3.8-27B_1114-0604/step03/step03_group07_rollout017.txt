    try:

    a_out = np.clip(res.x, 0.0, 1.0)
    if a_out.sum() < 1e-12:

    # Rescale back (score is scale-invariant, so we just need the direction)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.success

def nesterov_subgrad(a0, beta_schedule, max_iters=5000, time_limit=None):
    """Nesterov-accelerated LSE subgradient with projection to simplex-like constraint."""
    n = len(a0)
    # Work with sum(a) = 1 normalization
    S = float(a0.sum())
    a = a0 / S if S > 1e-12 else np.ones(n) / n
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Nesterov momentum
    y = a.copy()
    t_k = 1.0
    
    # Initial step size estimate
    c = fast_conv(a)
    M = float(c.max())
    L_est = 2.0 * n * (4.0 * M + 2.0)  # rough Lipschitz estimate
    
    for it in range(max_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        # Interpolate beta
        frac = it / max_iters
        beta = beta_schedule[0] * (beta_schedule[-1] / beta_schedule[0]) ** frac
        
        # Compute LSE at y
        score_y, grad_y = lse_score_and_grad(y, beta)
        
        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        
        # Step size: polyak-like
        # Use fixed decreasing step
        step = 1.0 / (L_est * np.sqrt(t_new))
        
        y_new = y - step * grad_y
        # Project to {x >= 0, sum(x) = 1}
        y_new = np.clip(y_new, 0, None)
        s = y_new.sum()
        if s > 1e-12:
            y_new = y_new / s
        
        # Check at the new point
        sc_new = compute_score(y_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = y_new.copy()
        
        a = y_new
        y = a + momentum * (a - y)
        # Project y
        y = np.clip(y, 0, None)
        s = y.sum()
        if s > 1e-12:
            y = y / s
        else:
            y = a.copy()
        
        t_k = t_new
    
    return best_sc, best_a

"""
Multi-phase convex optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LSE-continuation L-BFGS, Nesterov-accelerated projected subgradient,
and adaptive spectral peak shaving. Focuses optimization depth on a single
length for maximum refinement.

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 0.0)
    exp_s = np.exp(scaled)
    total = exp_s.sum()
    L = c_max + np.log(total) / beta
    p = exp_s / total
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad

def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() < 1e-14:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new

        step = 1.0 / (L_est * np.sqrt(t_new + 1.0))
        # Clamp step to avoid overshooting
        step = min(step, 0.5 / (1.0 + 0.01 * it))

        y_new = y - step * grad_y
        y_new = project_simplex(y_new)

        sc_new = compute_score(y_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = y_new.copy()

        x_prev = a
        a = y_new
        y = a + momentum * (a - x_prev)
        y = project_simplex(y)
        t_k = t_new

    return best_sc, best_a


def adaptive_peak_shave(a, n_iters=100, time_limit=None):
    """Adaptively shave the convolution peak by reducing top contributors."""
    n = len(a)
    a = a.copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    decay = 1.0

    for it in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)

        if j_end < j_start:
            break

        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[j] * a[k_star - j]

        top_k = max(3, n // 80)
        top_idx = np.argsort(contrib)[::-1][:top_k]
        bottom_idx = np.argsort(contrib)[top_k:]

        total_red = 0.0
        strength = 0.015 * decay
        for j in top_idx:
            red = a[j] * strength
            a[j] = max(a[j] - red, 0.0)
            total_red += red

        if len(bottom_idx) > 0 and total_red > 0:
            per = total_red / len(bottom_idx)
            a[bottom_idx] += per

        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-12:
            a = best_a.copy()
            break
        a = a / s

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            decay = max(0.3, decay * 0.97)
        else:
            a = best_a.copy()
            decay *= 0.9
            if decay < 0.01:
                break

    return best_a


def build_inits(n, rng, parent=None):
    """Generate diverse initializations for optimization."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else x

    if parent is not None and len(parent) == n:
        p = parent.astype(np.float64).copy()
        ps = p.sum()
        if ps > 1e-12:
            inits.append(p / ps)

    inits.append(np.ones(n) / n)

    # Anti-palindromic linear
    for amp in [0.3, 0.5, 0.7, 0.9]:
        a = 0.5 + amp * (xn - 0.5)
        a = np.clip(a, 0.001, None)
        inits.append(a / a.sum())

    # Anti-palindromic quadratic
    for amp in [0.3, 0.6]:
        a = 0.5 + amp * (xn - 0.5) * np.abs(xn - 0.5)
        a = np.clip(a, 0.001, None)
        inits.append(a / a.sum())

    # Sinusoidal anti-palindromic
    for freq in [1, 2, 3, 5]:
        a = 0.5 + 0.35 * np.sin(np.pi * freq * (2.0 * xn - 1.0))
        a = np.clip(a, 0.001, None)
        inits.append(a / a.sum())

    # Power law anti-palindromic
    for p in [0.5, 1.5, 2.0]:
        a = 0.3 + 0.7 * np.power(np.abs(xn - 0.5) * 2.0, p)
        a = a * (1.0 - xn) + (1.0 - a) * xn  # make anti-palindromic
        a = np.clip(a, 0.001, None)
        inits.append(a / a.sum())

    # Two-block
    half = n // 2
    for hi in [0.6, 0.8, 0.9]:
        a = np.concatenate([np.ones(half) * hi, np.ones(n - half) * (1.0 - hi) * 0.3])
        inits.append(a / a.sum())

    # Random smoothed
    a = np.abs(rng.standard_normal(n)) + 0.1
    ksize = max(1, n // 200)
    kernel = np.ones(ksize) / ksize
    a = np.convolve(a, kernel, mode='same')
    inits.append(a / a.sum())

    # Pure random
    a = np.abs(rng.standard_normal(n)) + 0.01
    inits.append(a / a.sum())

    return inits


def lse_continuation(a0, n, beta_schedule, phase_end):
    """Run LSE continuation from a0 through the beta schedule."""
    a = a0.astype(np.float64).copy()
    for beta in beta_schedule:
        if time.monotonic() > phase_end:
            break
        remaining = phase_end - time.monotonic()
        if remaining < 3:
            break
        maxit = max(200, min(4000, int(remaining * 1.5)))
        sc, a_opt, _ = run_lbfgs(a, beta, maxiter=maxit, maxfun=maxit * 20)
        a = a_opt
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Quick init evaluation + basis search.
    Phase 2: LSE-continuation L-BFGS from best inits (main workhorse).
    Phase 3: Nesterov-accelerated subgradient refinement.
    Phase 4: Adaptive peak shaving interleaved with L-BFGS.
    Phase 5: Final high-beta polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 25.0

    best_score = np.inf
    best_a = np.ones(6962) / 6962

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64 and p.sum() > 0.01:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # Choose primary length: parent's length or 8192
    if parent is not None:
        primary_n = parent.size
    else:
        primary_n = 8192

    # Also consider 8192 if parent is different
    secondary_n = 8192 if primary_n != 8192 else None
    lengths = [primary_n]
    if secondary_n is not None:
        lengths.append(secondary_n)

    print(f"Primary n={primary_n}, secondary n={secondary_n}")
    print(f"Parent: {'yes' if parent is not None else 'no'}")

    # ============================================================
    # Phase 1: Quick initialization (10% of budget)
    # ============================================================
    phase1_end = t_start + 0.08 * total_budget
    print(f"\n=== Phase 1: Init (until t+{phase1_end - t_start:.0f}s) ===")

    candidate_starts = {}
    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        p = parent if (parent is not None and parent.size == n) else None
        inits = build_inits(n, rng, p)
        best_init_sc = np.inf
        best_init = None
        for i, init in enumerate(inits):
            sc = compute_score(init)
            if sc < best_init_sc:
                best_init_sc = sc
                best_init = init
            report(sc, init, f"init{i} n={n}")
        candidate_starts[n] = best_init
        print(f"  Best init for n={n}: {best_init_sc:.8f}")

    # ============================================================
    # Phase 2: LSE Continuation (45% of budget)
    # ============================================================
    phase2_end = t_start + 0.55 * total_budget
    print(f"\n=== Phase 2: LSE Continuation (until t+{phase2_end - t_start:.0f}s) ===")

    beta_schedule = [20, 50, 100, 200, 400, 800, 1500, 3000, 6000, 12000]

    for n in lengths:
        if time.monotonic() > phase2_end:
            break
        if n not in candidate_starts:
            continue
        print(f"\n  n={n}")

        p = parent if (parent is not None and parent.size == n) else None
        inits = build_inits(n, rng, p)

        # Sort inits by initial score, take top 8
        init_scores = [(compute_score(init), init) for init in inits]
        init_scores.sort(key=lambda x: x[0])
        top_inits = [a for _, a in init_scores[:8]]

        time_left = phase2_end - time.monotonic()
        per_init_time = time_left / len(top_inits)

        for i, init in enumerate(top_inits):
            if time.monotonic() > phase2_end:
                break
            t0 = time.monotonic()
            a_cur = init.copy()
            for beta in beta_schedule:
                if time.monotonic() > phase2_end:
                    break
                remaining = phase2_end - time.monotonic()
                if remaining < 5:
                    break
                maxit = max(200, min(3000, int(remaining * 0.8)))
                sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxit * 15)
                a_cur = a_opt
                report(sc, a_opt, f"P2 n={n} i={i} b={beta}")
            print(f"    Init {i}: final={compute_score(a_cur):.8f}, time={time.monotonic()-t0:.1f}s")
            if compute_score(a_cur) < compute_score(candidate_starts.get(n, a_cur)):
                candidate_starts[n] = a_cur

    print(f"\n  After Phase 2: best={best_score:.8f}")

    # ============================================================
    # Phase 3: Nesterov Subgradient (15% of budget)
    # ============================================================
    phase3_end = t_start + 0.72 * total_budget
    print(f"\n=== Phase 3: Nesterov Subgradient ===")

    n_best = len(best_a)
    a_work = best_a.astype(np.float64).copy()

    # Run Nesterov from current best
    sc, a_opt = nesterov_subgrad(a_work, 50, 8000, max_iters=2000, time_limit=phase3_end)
    report(sc, a_opt, "nesterov")

    # Also try from the best init
    if n_best in candidate_starts:
        a_work2 = candidate_starts[n_best].astype(np.float64).copy()
        sc, a_opt = nesterov_subgrad(a_work2, 50, 8000, max_iters=1500, time_limit=phase3_end)
        report(sc, a_opt, "nesterov-init")

    # ============================================================
    # Phase 4: Peak Shaving + L-BFGS interleaved (15% of budget)
    # ============================================================
    phase4_end = t_start + 0.88 * total_budget
    print(f"\n=== Phase 4: Peak Shaving + Refinement ===")

    a_work = best_a.astype(np.float64).copy()
    for rnd in range(6):
        if time.monotonic() > phase4_end:
            break
        # Peak shave
        a_shaved = adaptive_peak_shave(a_work, n_iters=40, time_limit=phase4_end)
        sc = compute_score(a_shaved)
        report(sc, a_shaved, f"shave r{rnd}")
        a_work = a_shaved

        # L-BFGS refinement
        remaining = phase4_end - time.monotonic()
        if remaining > 15:
            sc, a_opt, _ = run_lbfgs(a_work, 8000, maxiter=2500, maxfun=40000)
            report(sc, a_opt, f"lbfgs r{rnd}")
            a_work = a_opt

    # ============================================================
    # Phase 5: Final high-beta polish (remaining time)
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")
    for beta in [15000, 30000, 60000]:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        maxit = max(500, min(5000, int(remaining * 1.5)))
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxit, maxfun=maxit * 15)
        report(sc, a_opt, f"final b={beta}")

    # Final Nesterov if time permits
    if time.monotonic() < deadline - 15:
        sc, a_opt = nesterov_subgrad(best_a, 200, 20000, max_iters=1000, time_limit=deadline - 5)
        report(sc, a_opt, "final-nesterov")

    # ============================================================
    # Output
    # ============================================================
    out_a = best_a.astype(np.float64).copy()
    S = out_a.sum()
    if S > 0:
        out_a = out_a / max(out_a.max(), 1e-12)
    out_a = np.clip(out_a, 0.0, 1000.0)
    out = [float(x) for x in out_a]

    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
