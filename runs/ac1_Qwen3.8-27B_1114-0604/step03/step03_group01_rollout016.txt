"""
Improved optimizer for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key improvements: active-set coordinate descent targeting specific convolution lags,
adaptive beta continuation to very high values, LP-inspired direction finding,

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300:
        # Degenerate case: all mass on one lag
        idx = int(np.argmax(c))
        L = c_max
        p = np.zeros(len(c))
        p[idx] = 1.0

        p = exp_scaled / exp_sum

def run_lbfgs_a(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B in a-space with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,

    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def adam_direct(a_init, lr=0.005, n_steps=50000, beta_lse=100,
                time_limit_s=None, t_start=None, normalize=True):
    """Adam optimizer in a-space with LSE-smoothed gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad_a(a, beta_lse)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a / s
        if t % 20000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a

                             time_limit_s=None, t_start=None):
    """
    Coordinate descent targeting specific convolution lags.
    Identifies top-K lags near the max and optimizes contributing coordinates.
    """

    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = np.inf
    if time_limit_s is not None and t_start is not None:
        deadline = t_start + time_limit_s

    for sweep in range(200):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score

        # Identify top-K lags
        sorted_idx = np.argsort(c)[::-1][:top_k_lags]

        # and try to reduce the lag by adjusting those coordinates
        for k in sorted_idx:
            if time.monotonic() > deadline:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je < js:

            indices = np.arange(js, je + 1)
            paired = k - indices  # a[indices] * a[paired] contributes to lag k

                continue  # Not close to max, skip

                    if sc_new < best_sc:
                        best_sc = sc_new
                        best_x = a_old * ratio
                a[j] = best_x

        # Periodically re-normalize
        if (sweep + 1) % 10 == 0:
            S = a.sum()
            if S > 1e-12:
                a = a / S

    # Final
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def targeted_coord_descent(a, n_iter=1000, time_limit_s=None, t_start=None):
    """
    More aggressive coordinate descent: at each step, find the coordinate
    whose change most reduces the max convolution lag.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = np.inf
    if time_limit_s is not None and t_start is not None:
        deadline = t_start + time_limit_s

    c = fast_conv(a)
    S = float(a.sum())
    M = float(c.max())

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        # Find the max lag
        k_star = int(c.argmax())
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)

        # Get current gradient (subgradient at the max)
        score = 2.0 * n * M / (S * S)

        # Select a few coordinates to try (those with largest a[paired] value)
        indices = np.arange(js, je + 1)
        if len(indices) == 0:
            continue
        paired = k_star - indices
        contrib = a[indices] * a[paired]
        # Sort by contribution, try top ones
        order = np.argsort(contrib)[::-1][:min(20, len(indices))]

        improved = False
        for idx_pos in order:
            if time.monotonic() > deadline:
                break
            j = int(indices[idx_pos])
            p = int(paired[idx_pos])
            a_old = a[j]
            if a_old < 1e-15:
                continue

            # Try reducing a[j] (since it contributes to the peak lag)
            for ratio in [0.7, 0.85, 0.95]:
                a[j] = a_old * ratio
                # Quick check: what's the new value at lag k_star?
                new_c_k = c[k_star] - a_old * a[p] + a[j] * a[p]
                if new_c_k < M * 0.99:
                    # Check if this actually reduces the overall max
                    c = fast_conv(a)
                    new_M = float(c.max())
                    new_score = 2.0 * n * new_M / (S * S)
                    if new_score < score:
                        M = new_M
                        score = new_score
                        improved = True
                        break
                a[j] = a_old

            if improved:
                if score < best_score:
                    best_score = score
                break

        if not improved:
            # Try increasing non-contributing coordinates
            mask = np.ones(n, dtype=bool)
            mask[indices] = False
            if mask.sum() > 0:
                # Pick a random non-contributing coordinate
                candidates = np.where(mask)[0]
                j = int(candidates[np.random.randint(len(candidates))])
                a_old = a[j]
                # Small increase
                a[j] = a_old * 1.05 if a_old > 1e-15 else 1e-8
                c = fast_conv(a)
                S = float(a.sum())
                new_M = float(c.max())
                new_score = 2.0 * n * new_M / (S * S)
                if new_score >= score:
                    a[j] = a_old  # Revert

    a = a / a.sum()
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def lp_direction_step(a, M, t_interp=0.1):
    """
    LP-inspired direction step.
    Find a direction that reduces the max convolution while maintaining sum.
    Uses a simplified "greedy" approach:
    1. Identify active lags (near max)
    2. For each coordinate, compute how much it contributes to active lags
    3. Move mass from high-contributing to low-contributing coordinates
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

    # Active lags: those within 2% of max
    tol = 0.02 * M
    active = c >= (M - tol)
    active_indices = np.where(active)[0]

    if len(active_indices) == 0:
        return a.copy()

    # For each coordinate j, compute its "contribution score" to active lags
    contrib_score = np.zeros(n)
    for k in active_indices:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            indices = np.arange(js, je + 1)
            paired = k - indices
            contrib_score[indices] += a[paired]

    # Normalize contribution score
    max_contrib = contrib_score.max()
    if max_contrib < 1e-15:
        return a.copy()

    # Direction: reduce high-contributing coords, increase low-contributing
    # But ensure non-negativity
    a_new = a.copy()
    # Reduce: a_new[j] = a[j] * (1 - alpha * contrib_score[j] / max_contrib)
    alpha = t_interp * 0.5
    a_new = a * (1.0 - alpha * contrib_score / max_contrib)
    a_new = np.clip(a_new, 0.0, 1000.0)

    # Renormalize to same sum
    S_new = a_new.sum()
    if S_new > 1e-12:
        a_new = a_new * S / S_new

    # Check if this improves
    c_new = fast_conv(a_new)
    M_new = float(c_new.max())
    S_check = float(a_new.sum())
    old_score = 2.0 * n * M / (S * S)
    new_score = 2.0 * n * M_new / (S_check * S_check)

    if new_score < old_score:
        return a_new
    return a.copy()


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer:
    1. Adam warmup with moderate beta
    2. L-BFGS continuation to high beta
    3. Active-set coordinate descent + L-BFGS alternation
    4. Multi-length exploration
    5. Final ultra-high-beta polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None and parent.sum() > 1e-6:
        print(f"Parent: n={len(parent)}, score={compute_score(parent):.8f}")
    else:
        parent = np.ones(4096)
        print(f"No parent, using uniform n=4096")

    # Normalize to sum=1
    best_a = parent / parent.sum()
    best_score = compute_score(best_a)
    print(f"Initial best: {best_score:.8f}, n={len(best_a)}")

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            # Normalize
            S = best_a.sum()
            if S > 1e-12:
                best_a = best_a / S
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.10f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ==================================================================
    # Phase 1: Adam warmup (20% of time)
    # ==================================================================
    p1_time = effective_budget * 0.20
    print(f"\n{'='*60}")
    print(f"Phase 1: Adam warmup (budget={p1_time:.0f}s)")
    print(f"{'='*60}")

    a_work = best_a.copy()
    for beta, lr in [(50, 0.01), (150, 0.005), (400, 0.003)]:
        remaining = p1_time - (time.monotonic() - t_start)
        if remaining < 3:
            break
        time_alloc = min(remaining, p1_time / 3 - 5)
        if time_alloc < 2:
            break
        sc, a_opt = adam_direct(a_work, lr=lr, n_steps=200000, beta_lse=beta,
                                time_limit_s=time_alloc, t_start=t_start)
        if sc < best_score:
            try_update(sc, a_opt, f"Adam beta={beta}")
        a_work = best_a.copy()

    # ==================================================================
    # Phase 2: L-BFGS continuation (30% of time)
    # ==================================================================
    p2_time = effective_budget * 0.30
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS continuation (budget={p2_time:.0f}s)")
    print(f"{'='*60}")

    betas = [500, 1500, 4000, 10000, 25000]
    for beta in betas:
        remaining = p2_time - (time.monotonic() - t_start)
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, a_opt, res = run_lbfgs_a(best_a, beta, maxiter=3000, maxfun=60000)
        try_update(sc, a_opt, f"LBFGS beta={beta}")
        print(f"    beta={beta}: {sc:.10f} ({time.monotonic()-t0:.1f}s, "
              f"status: {res.message[:40]})")

    # ==================================================================
    # Phase 3: Active-set coordinate descent + L-BFGS (25% of time)
    # ==================================================================
    p3_time = effective_budget * 0.25
    print(f"\n{'='*60}")
    print(f"Phase 3: Active-set CD + L-BFGS (budget={p3_time:.0f}s)")
    print(f"{'='*60}")

    a_work = best_a.copy()
    n_iter = 0
    while time.monotonic() - t_start < t_start + p1_time + p2_time + p3_time - 10:
        remaining = t_start + p1_time + p2_time + p3_time - 10 - time.monotonic()
        if remaining < 5:
            break

        # Active-set coordinate descent
        time_alloc = min(remaining * 0.5, 60.0)
        sc, a_opt = active_set_coord_descent(a_work, top_k_lags=5,
                                             time_limit_s=time_alloc, t_start=t_start)
        if sc < best_score:
            try_update(sc, a_opt, f"ActiveSet CD iter{n_iter}")
        a_work = best_a.copy()

        # LP-inspired direction
        c = fast_conv(a_work)
        M = float(c.max())
        a_lp = lp_direction_step(a_work, M, t_interp=0.3)
        sc_lp = compute_score(a_lp)
        if sc_lp < best_score:
            try_update(sc_lp, a_lp, f"LP direction iter{n_iter}")
            a_work = best_a.copy()

        # Quick L-BFGS polish
        if time.monotonic() - t_start < t_start + p1_time + p2_time + p3_time - 20:
            sc, a_opt, _ = run_lbfgs_a(a_work, 5000, maxiter=2000, maxfun=30000)
            if sc < best_score:
                try_update(sc, a_opt, f"LBFGS-5k iter{n_iter}")
            a_work = best_a.copy()

        n_iter += 1
        if n_iter % 3 == 0:
            print(f"    Iter {n_iter}: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # ==================================================================
    # Phase 4: Multi-length + targeted restarts (15% of time)
    # ==================================================================
    p4_time = effective_budget * 0.15
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-length + restarts (budget={p4_time:.0f}s)")
    print(f"{'='*60}")

    n_current = len(best_a)
    n_restarts = 8
    for r in range(n_restarts):
        remaining = t_start + p1_time + p2_time + p3_time + p4_time - 15 - time.monotonic()
        if remaining < 10:
            break

        strategy = r % 4
        if strategy == 0:
            # Resample to different length
            new_n = int(n_current * (0.8 + 0.15 * rng.random() * 2))
            new_n = max(1024, min(new_n, 20000))
            if new_n != n_current:
                a_pert = resample_sequence(best_a, new_n)
                a_pert = a_pert / a_pert.sum()
            else:
                a_pert = best_a.copy()
        elif strategy == 1:
            # Perturb peak contributors
            c = fast_conv(best_a)
            k_star = int(c.argmax())
            n_best = len(best_a)
            js = max(0, k_star - n_best + 1)
            je = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            peak_mask = np.zeros(n_best, dtype=bool)
            peak_mask[js:je+1] = True
            a_pert[peak_mask] *= (0.7 + 0.2 * rng.random())
            non_peak = ~peak_mask
            if non_peak.sum() > 0:
                removed = float(best_a[peak_mask].sum() - a_pert[peak_mask].sum())
                a_pert[non_peak] += removed / non_peak.sum()
            a_pert = np.clip(a_pert, 0, 1000)
            a_pert = a_pert / a_pert.sum()
        elif strategy == 2:
            # Gaussian perturbation
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(len(best_a)))
            a_pert = np.clip(a_pert, 0, 1000)
            a_pert = a_pert / a_pert.sum()
        else:
            # Sinusoidal modulation
            n_best = len(best_a)
            freq = 2.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            x = np.arange(n_best) / n_best
            mod = 1.0 + 0.02 * np.cos(2 * np.pi * freq * x + phase)
            a_pert = best_a * mod
            a_pert = np.clip(a_pert, 0, 1000)
            a_pert = a_pert / a_pert.sum()

        # Optimize the perturbed candidate
        time_left = t_start + p1_time + p2_time + p3_time + p4_time - 15 - time.monotonic()
        if time_left < 5:
            break
        adam_time = min(15.0, time_left * 0.4)
        sc, a_r = adam_direct(a_pert, lr=0.003, n_steps=30000, beta_lse=500,
                              time_limit_s=adam_time, t_start=t_start)
        if time.monotonic() < t_start + p1_time + p2_time + p3_time + p4_time - 20:
            sc2, a_r2, _ = run_lbfgs_a(a_r, 5000, maxiter=2000, maxfun=30000)
            if sc2 < sc:
                sc, a_r = sc2, a_r2
        try_update(sc, a_r, f"restart#{r} s{strategy}")

    # ==================================================================
    # Phase 5: Final ultra-high-beta polish (10% of time)
    # ==================================================================
    p5_time = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish (budget={p5_time:.0f}s)")
    print(f"{'='*60}")

    final_deadline = t_start + effective_budget - 5.0

    # Ultra high beta L-BFGS
    for beta in [20000, 50000]:
        if time.monotonic() > final_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs_a(best_a, beta, maxiter=5000, maxfun=80000)
        try_update(sc, a_opt, f"Final LBFGS beta={beta}")
        print(f"    beta={beta}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # Final coordinate descent
    if time.monotonic() < final_deadline - 5:
        remaining = final_deadline - time.monotonic()
        sc, a_opt = active_set_coord_descent(best_a, top_k_lags=3,
                                             time_limit_s=min(remaining, 30.0),
                                             t_start=t_start)
        try_update(sc, a_opt, "Final active-set CD")

    # One more targeted pass
    if time.monotonic() < final_deadline - 3:
        c = fast_conv(best_a)
        M = float(c.max())
        a_lp = lp_direction_step(best_a, M, t_interp=0.1)
        sc_lp = compute_score(a_lp)
        try_update(sc_lp, a_lp, "Final LP step")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
