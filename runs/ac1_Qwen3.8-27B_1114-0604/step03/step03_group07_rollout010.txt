"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum a)^2 over non-negative sequences.
Strategy: Aggressive LSE continuation from parent → active-set penalty refinement →
structured local search with multiplicative updates → multi-length exploration → final polish.
Key improvements over previous: richer active-set handling, multiplicative update step,
higher-beta LSE schedule, and more targeted perturbation strategies.

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700, 700)
    exp_s = np.exp(scaled)
    L = c_max + np.log(np.sum(exp_s)) / beta
    p = exp_s / np.sum(exp_s)
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1: 2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
        )
    except Exception:
        return compute_score(a0), a0, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.success


def penalty_objective_grad(a, top_lags, gamma):
    """
    Penalty-based objective: minimize max(c_k) + gamma * sum((c_k - t)_+^2) for top lags.
    Uses smooth approximation of the positive part.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    S2 = S * S
    S3 = S2 * S
    # Smooth max via LSE with high beta
    beta_eff = 50000.0
    scaled = beta_eff * (c - c_max)
    scaled = np.clip(scaled, -700, 700)
    exp_s = np.exp(scaled)
    L = c_max + np.log(np.sum(exp_s)) / beta_eff
    p = exp_s / np.sum(exp_s)
    # Add penalty for top lags exceeding a threshold
    base_score = 2.0 * n * L / S2
    # Gradient base
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1: 2 * n - 1]
    g = 2.0 * conv_res
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    # Penalty: for top lags, add penalty if c_k > 0.98 * c_max
    threshold = 0.98 * c_max
    penalty = 0.0
    pgrad = np.zeros(n)
    for k in top_lags:
        ck = c[k]
        if ck > threshold:
            excess = ck - threshold
            penalty += gamma * excess * excess
            # Gradient of penalty w.r.t. a
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                pgrad[j] += gamma * 2.0 * excess * a[k - j]
    total_score = base_score + penalty / S2
    total_grad = grad + pgrad / S2 - 2.0 * total_score / S * np.ones(n)
    return total_score, total_grad


def penalty_wrapper(a, top_lags, gamma):
    return penalty_objective_grad(a, top_lags, gamma)


def subgrad_polish(a, max_iters=5000, top_k=5):
    """Subgradient descent with Polyak step, targeting top-k lags."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Get top-k lags
        top_lags = np.argsort(c)[::-1][:top_k]
        # Weights: proportional to how close to max
        weights = np.zeros(2 * n - 1)
        for idx, k in enumerate(top_lags):
            weights[k] = max(0.0, 1.0 - 0.1 * idx)
        w_sum = weights.sum()
        weights = weights / w_sum

        # Subgradient as weighted combination
        sg = np.zeros(n)
        for k in top_lags:
            w = weights[k]
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                sg[j] += 2.0 * w * a[k - j]

        S2 = S * S
        S3 = S2 * S
        # Effective M is weighted average of top lags
        M_eff = sum(weights[k] * c[k] for k in top_lags)
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_eff / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - sc_lb) / gn2
        # Adaptive step: decay over time
        decay = 1.0 / (1.0 + 0.01 * it)
        step = min(step, 0.3 * decay)
        step = max(step, 1e-10)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def multiplicative_update(a, n_iter=100, top_k=3):
    """
    Multiplicative update inspired by NMF: naturally maintains non-negativity.
    Update: a[j] <- a[j] * (desirable / undesirable)
    where desirable/undurable come from the autocorrelation structure.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iter):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # For each position j, compute the "pressure" to increase or decrease
        # based on contribution to top lags
        top_lags = np.argsort(c)[::-1][:top_k]
        pressure_num = np.zeros(n)  # numerator for multiplicative ratio
        pressure_den = np.zeros(n) + 1e-10  # denominator

        for k in top_lags:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                # a[j] contributes a[j]*a[k-j] to c_k
                # To reduce c_k, decrease a[j] proportionally to a[k-j]
                pressure_den[j] += a[j] * a[k - j] / max(c[k], 1e-15)

        # Also add a "sum preservation" term
        target_mean = S / n
        pressure_den += 0.01 * a

        # Multiplicative update: a[j] *= (target_mean + 0.5) / pressure_den[j]
        # This is a heuristic; tune the constant
        ratio = (target_mean + 0.5) / pressure_den
        ratio = np.clip(ratio, 0.9, 1.1)
        a = a * ratio
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a

def targeted_perturb(a, rng, strength=0.05, n_top=20):
    """Reduce elements contributing most to peak convolution lags."""
    n = len(a)
    c = fast_conv(a)
    a_pert = a.copy()

    # Target top few lags
    top_lags = np.argsort(c)[::-1][:max(1, c.size // 50)]
    contrib = np.zeros(n)
    for k in top_lags:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            contrib[j] += a[j] * a[k - j]

    top_count = max(5, n // 100)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[top_count:n - top_count]
    total_red = 0.0
    for j in top_idx:
        amt = a_pert[j] * strength
        a_pert[j] -= amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_red += amt
    if len(bottom_idx) > 0 and total_red > 0:
        per = total_red / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def anti_palindromic_perturb(a, rng, strength=0.05):
    """Inject anti-palindromic structure to reduce center autocorrelation."""
    n = len(a)
    a_pert = a.copy()
    center = (n - 1) / 2.0
    width = max(1, n / 20.0)
    for j in range(n):
        d = (j - center) / width
        sign = 1.0 if j < center else -1.0
        a_pert[j] *= (1.0 + strength * sign * np.exp(-d * d * 0.5))
    return np.clip(a_pert, 0.0, 1000.0)


def spectral_flat_perturb(a, rng, strength=0.1):
    """Flatten the spectrum to reduce autocorrelation peak."""
    n = len(a)
    a_pert = a.copy()
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a_pert, M_fft)
    power = np.abs(A) ** 2
    # Target: make power spectrum flatter
    mean_power = power.mean()
    # Attenuate peaks, boost valleys
    correction = 1.0 - strength * (power - mean_power) / (mean_power + 1e-10)
    correction = np.clip(correction, 0.5, 1.5)
    A *= np.sqrt(correction)
    a_pert = np.real(np.fft.irfft(A, M_fft))[:n]
    return np.clip(a_pert, 0.0, 1000.0)

        elif strat == 1:
            a_pert = anti_palindromic_perturb(best_a, rng, strength=0.03 + 0.04 * rng.random())
        elif strat == 2:
            a_pert = spectral_flat_perturb(best_a, rng, strength=0.05 + 0.1 * rng.random())
        elif strat == 3:
            # Gaussian perturbation
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        else:
            # Multiplicative random
            a_pert = best_a * np.exp(0.03 * rng.standard_normal(n_best))

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            a_pert = best_a * 0.9

        # Quick LSE refinement
        for beta in [5000, 30000]:
            if time.monotonic() > phase4_end:
                break
            sc, a_opt, _ = run_lbfgs(a_pert, beta, maxiter=2000, maxfun=40000)
            report(sc, a_opt, f"P4 #{trial} s={strat} b={beta}")
            a_pert = a_opt

        # Subgradient after LSE
        if time.monotonic() < phase4_end:
            sc, a_opt = subgrad_polish(a_pert, max_iters=2000, top_k=3)
            report(sc, a_opt, f"P4 #{trial} subgrad")

    # ============================================================
    # Phase 5: Multi-length exploration (1900 - 2200s)
    # ============================================================
    phase5_end = t_start + 2200.0
    print(f"\n=== Phase 5: Multi-Length Exploration ===")

    lengths_to_try = [8192, 10240, 12288]
    if parent is not None:
        lengths_to_try = [l for l in lengths_to_try if l != parent.size]

    for n_try in lengths_to_try:
        if time.monotonic() > phase5_end:
            break
        print(f"  Trying n={n_try}")
        # Start from a ramp
        a_try = np.linspace(0.05, 1.0, n_try)
        # Quick LSE continuation
        for beta in [100, 1000, 10000]:
            if time.monotonic() > phase5_end:
                break
            sc, a_opt, _ = run_lbfgs(a_try, beta, maxiter=3000, maxfun=60000)
            a_try = a_opt
            report(sc, a_try, f"P5 n={n_try} b={beta}")

        # Subgradient
        if time.monotonic() < phase5_end:
            sc, a_opt = subgrad_polish(a_try, max_iters=2000, top_k=5)
            report(sc, a_opt, f"P5 n={n_try} subgrad")

    # ============================================================
    # Phase 6: Final high-precision polish (2200 - deadline)
    # ============================================================
    print(f"\n=== Phase 6: Final Polish ===")

    if time.monotonic() < deadline - 10:
        # Very high beta LSE
        for beta in [50000, 200000]:
            if time.monotonic() > deadline - 15:
                break
            sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=5000, maxfun=50000)
            report(sc, a_opt, f"P6 b={beta}")

        # Final subgradient
        if time.monotonic() < deadline - 10:
            n_iters = int((deadline - time.monotonic() - 5) * 50)
            n_iters = max(500, n_iters)
            sc, a_opt = subgrad_polish(best_a, max_iters=n_iters, top_k=3)
            report(sc, a_opt, "P6 final subgrad")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
