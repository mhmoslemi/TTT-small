"""
Optimized step function height sequence generator using LSE-continuation L-BFGS
with adaptive beta, Nesterov-accelerated subgradient polish, and multi-start
diverse initializations. Minimizes 2n*max(conv(a,a))/(sum(a))^2 for non-negative a.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    
    # Gradient via cross-correlation with reversed a
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Scipy-compatible objective with gradient."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        a_out = a0
        return compute_score(a_out), a_out
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def build_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / max(n - 1, 1)
    
    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Anti-palindromic ramps
    inits.append(np.linspace(0.05, 1.0, n))
    inits.append(np.linspace(1.0, 0.05, n))
    
    # Quadratic
    inits.append(0.05 + 0.95 * xn ** 2)
    inits.append(0.05 + 0.95 * (1.0 - xn) ** 2)
    
    # Anti-palindromic sinusoidal
    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.clip(0.5 + 0.4 * np.sin(np.pi * freq * xn), 0.01, 1000.0))
    
    # Hann window (smooth bump)
    inits.append(0.5 * (1 - np.cos(2 * np.pi * xn)))
    
    # Raised cosine with offset
    inits.append(0.1 + 0.9 * (0.5 * (1 - np.cos(2 * np.pi * xn))))
    
    # Two-block
    half = n // 2
    inits.append(np.concatenate([np.ones(half) * 0.7, np.ones(n - half) * 0.3]))
    inits.append(np.concatenate([np.ones(half) * 0.3, np.ones(n - half) * 0.7]))
    
    # Smooth random
    a_rand = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1.0)
    ksize = max(1, n // 100)
    kernel = np.ones(ksize) / ksize
    a_smooth = np.convolve(a_rand, kernel, mode='same')
    inits.append(np.clip(a_smooth, 0.01, 1.0))
    
    # Bump at 1/3 and 2/3
    for center_frac in [1/3, 2/3]:
        center = int(center_frac * n)
        width = n // 6
        inits.append(0.1 + 0.9 * np.exp(-((x - center) / width) ** 2))
    
    return inits


def nesterov_subgrad(a, max_iters=10000):
    """Nesterov-accelerated subgradient method."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a_proj = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Lower bound: 2n/(2n-1) (achievable only if autocorrelation is perfectly flat)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(max_iters):
        # Nesterov lookahead
        a_nest = a_proj + v
        a_nest = np.clip(a_nest, 0.0, 1000.0)
        
        c = fast_conv(a_nest)
        S = float(a_nest.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        
        # Project onto simplex-like constraint (a >= 0)
        # Use subgradient
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a_nest[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        # Nesterov step size
        t = 2.0 / (it + 2.0)
        step = t / (np.linalg.norm(grad) + 1e-15)
        step = min(step, 1.0 / (1.0 + 0.1 * it))
        
        # Update
        a_new = a_nest - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        v = (a - a_new) / (1.0 - t) * t
        
        a = a_new
        a_proj = a_new - v
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a


def subgrad_polish(a, max_iters=5000):
    """Plain subgradient with adaptive step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        
        # Polyak-type step size
        step = max(0.0, (sc - sc_lb)) / (gn2 + 1e-15)
        step = min(step, 1.0 / (1.0 + 0.001 * it))
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a


def coordinate_descent(a, n_coords=50, n_rounds=100):
    """Coordinate descent on top contributors to the peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_sc = compute_score(a)
    
    for rd in range(n_rounds):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        # Find top contributors
        if j_end >= j_start:
            contrib = np.zeros(n)
            for j in range(j_start, j_end + 1):
                contrib[j] = a[j] * a[k_star - j]
        
        # Try reducing top contributors and adding to bottom
        top_k = min(n_coords, n // 20)
        top_idx = np.argsort(contrib)[::-1][:top_k]
        bot_idx = np.argsort(contrib)[:top_k]
        
        improved = False
        for j in top_idx:
            for frac in [0.01, 0.02, 0.05, 0.1]:
                a_trial = a.copy()
                amt = a_trial[j] * frac
                a_trial[j] -= amt
                if a_trial[j] < 0:
                    continue
                # Redistribute to bottom contributors
                for b in bot_idx:
                    a_trial[b] += amt / len(bot_idx)
                a_trial = np.clip(a_trial, 0.0, 1000.0)
                sc = compute_score(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    a = a_trial.copy()
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_sc, a

def gerchberg_saxton(a, target_c, max_iters=1000):
    """GS algorithm to find a with autocorrelation close to target_c."""
    n = len(a)
    a = a.astype(np.float64).copy()
    M_fft = 1 << (2 * n - 1 - 1).bit_length()
    
    for it in range(max_iters):
        # Compute current autocorrelation
        A = np.fft.rfft(a, M_fft)
        power = np.abs(A) ** 2
        c_current = np.fft.irfft(power, M_fft)[:2*n-1]
        
        # Compute target power spectrum
        # We want c to be as flat as possible (or match target_c)
        # For now, use target_c directly
        target_power = np.fft.rfft(target_c, M_fft)
        # target_power should be real and non-negative
        target_power = np.maximum(target_power.real, 0)
        target_mag = np.sqrt(target_power)
        
        # Update: keep current phases, use target magnitudes
        phases = np.angle(A)
        A_new = target_mag * np.exp(1j * phases)
        a = np.real(np.fft.irfft(A_new, M_fft))[:n]
        
        # Enforce non-negativity
        a = np.maximum(a, 0)
    
    return a

"""
Optimized step function height sequence generator.
Minimizes 2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences using:
- Multi-start with diverse anti-palindromic and structured initializations
- LSE-continuation L-BFGS with adaptive beta up to very high values

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -700, 700)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Scipy-compatible objective with gradient."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective. Returns (score, array)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def build_inits(n, rng, parent=None):
    """Generate diverse initializations emphasizing anti-palindromic structure."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / max(n - 1, 1)

    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Anti-palindromic ramps (various strengths)
    for lo, hi in [(0.01, 1.0), (0.05, 1.0), (0.1, 1.0), (0.01, 0.5)]:
        inits.append(np.linspace(lo, hi, n))
        inits.append(np.linspace(hi, lo, n))

    # Quadratic anti-palindromic
    inits.append(0.05 + 0.95 * xn ** 2)
    inits.append(0.05 + 0.95 * (1.0 - xn) ** 2)
    inits.append(0.05 + 0.95 * xn ** 3)
    inits.append(0.05 + 0.95 * (1.0 - xn) ** 3)

    # Anti-palindromic sinusoidal (odd harmonics)
    for freq in [1, 2, 3, 4, 5, 7, 11]:
        inits.append(np.clip(0.5 + 0.45 * np.sin(np.pi * freq * xn), 0.01, 1000.0))

    # Mixed: ramp + oscillation
    inits.append(np.clip(0.2 + 0.5 * xn + 0.2 * np.sin(2 * np.pi * 5 * xn), 0.01, 1000.0))
    inits.append(np.clip(0.2 + 0.5 * (1 - xn) + 0.2 * np.cos(2 * np.pi * 3 * xn), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.3 * xn + 0.3 * np.sin(np.pi * 2 * xn + 0.5), 0.01, 1000.0))

    # Hann window
    inits.append(0.05 + 0.95 * 0.5 * (1 - np.cos(2 * np.pi * xn)))

    # Two-block asymmetric
    for ratio in [0.3, 0.4, 0.7]:
        split = int(ratio * n)
        inits.append(np.concatenate([np.ones(split) * 0.8, np.ones(n - split) * 0.2]))
        inits.append(np.concatenate([np.ones(split) * 0.2, np.ones(n - split) * 0.8]))

    # Three-block
    third = n // 3
    inits.append(np.concatenate([np.ones(third) * 0.3, np.ones(n - 2 * third) * 0.8, np.ones(third) * 0.3]))
    inits.append(np.concatenate([np.ones(third) * 0.8, np.ones(n - 2 * third) * 0.3, np.ones(third) * 0.8]))

    # Smooth random
    a_rand = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1.0)
    ksize = max(1, n // 80)
    kernel = np.ones(ksize) / ksize
    a_smooth = np.convolve(a_rand, kernel, mode='same')
    inits.append(np.clip(a_smooth, 0.01, 1.0))

    # Bump profiles
    for center_frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        center = int(center_frac * n)
        width = max(1, n // 8)
        inits.append(0.05 + 0.95 * np.exp(-((x - center) / width) ** 2))

    return inits


def nesterov_subgrad(a, max_iters=10000):
    """Nesterov-accelerated subgradient with momentum."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a_prev = a.copy()
    y = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(max_iters):
        # Compute subgradient at y (lookahead point)
        c = fast_conv(y)
        S = float(y.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = y[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

        # Adaptive step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = (sc - sc_lb) / (gnorm * gnorm + 1e-15)
        step = min(step, 0.5 / (1.0 + 0.005 * it))

        # Nesterov update
        t_k = 2.0 / (it + 2.0)
        a_new = y - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)

        # Update momentum
        y = a_new + (a - a_new) / (1.0 - t_k) * t_k
        y = np.clip(y, 0.0, 1000.0)
        a = a_new

        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

    return best_sc, best_a

    a = np.clip(a, 1e-10, 1000.0).copy()

        eta = 0.5 * (sc - sc_lb) / (a @ grad + 1e-15)

def multi_peak_subgrad(a, max_iters=5000, n_peaks=5):
    """Subgradient using top n_peaks autocorrelation values simultaneously."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        # Top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        top_vals = c[top_idx]
        weights = top_vals / top_vals.sum()
        M_eff = float(weights @ top_vals)

        sc = 2.0 * n * float(c.max()) / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Combined subgradient from top peaks
        sg = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k_star = int(k)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_eff / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break

        step = max(0.0, sc - sc_lb) / (gn2 + 1e-15)
        step = min(step, 0.5 / (1.0 + 0.003 * it))

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def coordinate_descent(a, n_coords=40, n_rounds=200):
    """Coordinate descent on top contributors to the peak autocorrelation."""
    n = len(a)
    a = np.clip(a.astype(np.float64).copy(), 1e-10, 1000.0)
    best_sc = compute_score(a)

    for rd in range(n_rounds):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)

        if j_end < j_start:
            break

        # Compute contributions
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[j] * a[k_star - j]

        top_k = min(n_coords, max(5, n // 30))
        top_idx = np.argsort(contrib)[::-1][:top_k]
        bot_idx = np.argsort(contrib)[:top_k]

        improved_any = False
        for j in top_idx:
            for frac in [0.02, 0.05, 0.1, 0.15]:
                a_trial = a.copy()
                amt = a_trial[j] * frac
                a_trial[j] -= amt
                if a_trial[j] < 0:
                    continue
                # Redistribute to lowest contributors
                per = amt / max(len(bot_idx), 1)
                for b in bot_idx:
                    a_trial[b] += per
                a_trial = np.clip(a_trial, 1e-10, 1000.0)
                sc = compute_score(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    a = a_trial.copy()
                    improved_any = True
                    break

        if not improved_any and rd > 20:
            break

    return best_sc, a


def stochastic_restart(a, rng, scale=0.1):
    """Perturb a solution for restart."""
    n = len(a)
    a_pert = a * (1.0 + scale * rng.standard_normal(n))
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    if a_pert.sum() < 1e-6:
        a_pert = a * 0.8
    return a_pert


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """

    Phase 1: Multi-start LSE (low-mid beta)
    Phase 2: Deep LSE continuation (high beta, adaptive)
    Phase 3: Nesterov + multiplicative subgradient polish
    Phase 4: Coordinate descent + stochastic restarts
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total = float(budget_s)
    deadline = t_start + total - 40.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    n_target = parent.size if parent is not None else 6962
    print(f"Target length: {n_target}, Parent: {'yes' if parent is not None else 'no'}")

    # ============================================================
    # Phase 1: Multi-start (20% budget)
    # ============================================================
    phase1_end = t_start + 0.20 * total
    print(f"\n=== Phase 1: Multi-Start (until {phase1_end - t_start:.0f}s) ===")

    inits = build_inits(n_target, rng, parent)
    # Score and sort initializations
    scored_inits = []
    for i, init in enumerate(inits):
        sc = compute_score(init)
        scored_inits.append((sc, i, init))
    scored_inits.sort(key=lambda x: x[0])

    print(f"  {len(inits)} initializations generated, best base score: {scored_inits[0][0]:.4f}")

    candidates = []
    n_to_try = min(10, len(scored_inits))
    for rank in range(n_to_try):
        if time.monotonic() > phase1_end:
            break
        sc_base, idx, init = scored_inits[rank]
        for beta in [30, 100, 500]:
            if time.monotonic() > phase1_end:
                break
            remaining = phase1_end - time.monotonic()
            maxit = max(1000, min(5000, int(remaining / 5.0)))
            sc, a_opt = run_lbfgs(init, beta, maxiter=maxit, maxfun=maxit * 20)
            report(sc, a_opt, f"P1 r={rank} b={beta}")
            candidates.append((sc, a_opt.copy()))

    # Keep top candidates
    candidates.sort(key=lambda x: x[0])
    candidates = candidates[:6]
    if not candidates:
        candidates = [(best_score, best_a)]
    print(f"  Phase 1 best: {candidates[0][0]:.8f}")

    # ============================================================
    # Phase 2: Deep LSE continuation (40% budget)
    # ============================================================
    phase2_end = t_start + 0.60 * total
    print(f"\n=== Phase 2: Deep LSE Continuation (until {phase2_end - t_start:.0f}s) ===")

    beta_schedule = [1000, 2000, 5000, 10000, 20000, 50000, 100000]

    for c_idx in range(min(4, len(candidates))):
        if time.monotonic() > phase2_end:
            break
        a_cur = candidates[c_idx][1].copy()
        prev_sc = candidates[c_idx][0]
        print(f"\n  Candidate {c_idx}: start score={prev_sc:.8f}")

        for beta in beta_schedule:
            if time.monotonic() > phase2_end:
                break
            remaining = phase2_end - time.monotonic()
            if remaining < 30:
                break
            # Scale iterations by remaining time
            est_time_per_iter = 0.003 * n_target / 1000.0
            maxit = max(2000, min(15000, int(remaining / max(est_time_per_iter, 0.005))))

            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxit * 25)
            a_cur = a_opt
            improved = report(sc, a_opt, f"P2 c={c_idx} b={beta}")
            # If score got worse, revert
            if sc > prev_sc * 1.001:
                a_cur = np.clip(a_cur * 0.99, 0.0, 1000.0)
                continue
            prev_sc = min(prev_sc, sc)

    print(f"\n  Phase 2 best: {best_score:.8f}")

    # ============================================================
    # Phase 3: Subgradient polish (15% budget)
    # ============================================================
    phase3_end = t_start + 0.75 * total
    print(f"\n=== Phase 3: Subgradient Polish (until {phase3_end - t_start:.0f}s) ===")

    if time.monotonic() < phase3_end:
        n_iters = max(2000, int((phase3_end - time.monotonic()) * 40))
        sc, a_opt = nesterov_subgrad(best_a, max_iters=n_iters)
        report(sc, a_opt, "nesterov")

    if time.monotonic() < phase3_end:
        n_iters = max(1000, int((phase3_end - time.monotonic()) * 50))
        sc, a_opt = multiplicative_descend(best_a, max_iters=n_iters)
        report(sc, a_opt, "mult_grad")

    if time.monotonic() < phase3_end:
        n_iters = max(1000, int((phase3_end - time.monotonic()) * 60))
        sc, a_opt = multi_peak_subgrad(best_a, max_iters=n_iters, n_peaks=5)
        report(sc, a_opt, "multi_peak")

    # ============================================================
    # Phase 4: Coordinate descent + restarts (15% budget)
    # ============================================================
    phase4_end = deadline
    print(f"\n=== Phase 4: Coord Descent + Restarts (until {phase4_end - t_start:.0f}s) ===")

    if time.monotonic() < phase4_end - 30:
        sc, a_opt = coordinate_descent(best_a, n_coords=30, n_rounds=100)
        report(sc, a_opt, "coord")

    # Stochastic restarts
    restart_count = 0
    while time.monotonic() < phase4_end - 60 and restart_count < 8:
        restart_count += 1
        scale = 0.05 + 0.1 * rng.random()
        a_pert = stochastic_restart(best_a, rng, scale)
        # Quick LSE optimization from perturbed point
        sc, a_opt = run_lbfgs(a_pert, 5000, maxiter=3000, maxfun=60000)
        improved = report(sc, a_opt, f"restart {restart_count}")
        if not improved:
            # Try multiplicative gradient
            sc2, a_opt2 = multiplicative_descend(a_opt, max_iters=2000)
            report(sc2, a_opt2, f"restart {restart_count} mult")

    # Final subgradient push
    if time.monotonic() < deadline - 10:
        sc, a_opt = nesterov_subgrad(best_a, max_iters=3000)
        report(sc, a_opt, "final_nesterov")

    # Verify
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
