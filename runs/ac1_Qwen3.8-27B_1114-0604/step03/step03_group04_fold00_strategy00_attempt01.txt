def evaluate_sequence(seq):
    if not isinstance(seq, list):
        return np.inf
    if len(seq) == 0:
        return np.inf
    for x in seq:
        # Reject booleans (subclass of int) and other non-numeric types
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        # Reject NaN and infinity values
        if np.isnan(x) or np.isinf(x):
            return np.inf
    # Convert to floats for consistency to overcome potential data type issues
    seq = [float(x) for x in seq]
    # Apply bounds checking (0 to 1000) as a protective measure
    seq = [max(0.0, min(1000.0, x)) for x in seq]
    sum_val = np.sum(seq)
    if sum_val < 0.01:
        return np.inf
    # Perform full convolution to capture autocorrelation properties
    convolve_result = np.convolve(seq, seq, mode='full')
    max_val = np.max(convolve_result)
    return 2.0 * len(seq) * max_val / (sum_val ** 2)

def lse_objective_with_grad(a, beta):
    n = len(a)
    a = a.astype(np.float64).copy()
    sum_a = float(np.sum(a))
    if sum_a < 1e-12:

    return best_a, best_score

def run_lbfgs_sequence(initial_a, n, beta, maxiter=5000, maxfun=100000):
    """Execute L-BFGS optimization on a sequence with the given length."""
    # Clip initial sequence to bounds for immediate safety
    if 'evaluate_sequence' is not None:
        initial_a = safe_clip_scale(initial_a)
    initial_a_array = initial_a.astype(np.float64)
    initial_sum = np.sum(initial_a_array)
    
    # Initialize bounds for optimization (assuming [0, 1000] range if no clipping was explicit)
    bounds = [(0.0, 1000.0)] * n
    
    res = minimize(lse_objective_with_grad, initial_a_array, method='L-BFGS-B', args=(beta),
                    bounds=bounds, options={'maxiter': max(100, int(maxiter/5)), 'maxfun': int(maxfun/10), 'gtol': 1e-9, 'ftol': 1e-10})
    
    # Supervised gradient checking to ensure valid result
    if not res.success or np.isnan(res.fun) or res.x.size != n:
        # Fallback hyperparameter adjustment if results are invalid
        if beta < 100:
            updated_beta = 10 * n * beta / (np.min(res.x) + 10) if np.min(res.x) > 0 else beta
            res, _ = run_lbfgs_sequence(initial_a_array, n, updated_beta)
        if beta < 50:
            # Use higher beta to prevent infinite smoothing issues
            try:
                safe_beta = beta * 2
                res, _ = minimize(lse_objective_with_grad, initial_a_array, method='L-BFGS-B', args=(safe_beta), bounds=bounds, options={'maxiter': maxiter*0.8, 'maxfun': maxfun})
            except Exception as e:
                pass
    
    # Convert result to canonical form for output, apply evaluation bounds if necessary
    convolved_a = np.convolve(res.x, res.x)
    norm_conv_max = np.max(convolved_a[0:len(convolved_a)-1])  # Consider proper indexing per mode='full' output
    score_result = evaluate_sequence(res.x)
    
    if score_result < np.inf and score_result < best_score_hist:
        best_score_hist.append(score_result)
        best_a_hist.append(res.x.copy())
        
    if score_result < 1.5:
        min_reached = True
        
    return score_result, res.x.copy()
