def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = rfft(a, M_fft)
    return irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """Linear cross-convolution a*b via FFT."""
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = rfft(a, M_fft)
    B = rfft(b, M_fft)
    return irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum() + 1e-30)

    S2 = S * S

    # d c[k]/d a_j = 2 * a[k-j] (since c = a*a)
    # So grad_j = 4 * sum_k errors[k] * a[k-j] = 4 * (errors_reversed * a)_j
    # More precisely: sum_k errors[k] * a[k-j] for valid k,j
    # This is a cross-correlation
    er = errors[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    ER = rfft(er, M2_fft)
    A = rfft(a, M2_fft)
    conv_res = irfft(ER * A, M2_fft)[n - 1:2 * n - 1]
    grad_max = 4.0 * conv_res  # gradient of the max-conv part
    # Gradient of S^2 term
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_max / S2 - 2.0 * L / S3)
    return score, grad


def peak_pressure_step(a, n_top=50, strength=0.01):
    """
    Reduce a at positions that contribute most to the top peaks of conv(a,a).
    Redistribute mass to positions with low pressure.
    """
    n = len(a)
    c = fast_conv(a)
    # Find top peaks
    c_sorted = np.sort(c)[::-1]
    threshold = c_sorted[min(n_top, len(c_sorted) - 1)]
    hot = c > threshold
    # Compute pressure at each position
    # pressure[j] = sum over hot k of a[k-j] (if valid)
    # This is: for each hot k, add a[k-j] to pressure[j]
    # Equivalent to: pressure = sum over hot k of shifted_a
    hot_indices = np.where(hot)[0]
    pressure = np.zeros(n)
    # Vectorized: for each hot index k, the contribution to pressure[j] is a[k-j] for valid j
    # pressure[j] = sum_{k in hot} a[k-j] * 1(0 <= k-j < n)
    # Let's compute this as: for j in 0..n-1, pressure[j] = sum_{k in hot, 0<=k-j<n} a[k-j]
    # = sum_{k in hot} a[max(0,k-n+1):k+1] shifted... this is convolution-like
    # Simpler: iterate over hot indices (should be small)
    for k in hot_indices:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            # a[k-j] for j in [j_start, j_end]
            # When j = j_start, k-j = k - j_start; when j=j_end, k-j = k-j_end
            seg = a[j_start:j_end + 1][::-1]
            pressure[j_start:j_end + 1] += seg
    # Weight by excess
    excess = np.maximum(c - threshold, 0.0)
    # Re-weight pressure by excess
    pressure_w = np.zeros(n)
    for k_idx, k in enumerate(hot_indices):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1][::-1]
            pressure_w[j_start:j_end + 1] += seg * excess[k_idx]
    # Normalize
    pw_max = pressure_w.max()
    if pw_max < 1e-30:
        return a.copy()
    pressure_w /= pw_max
    # Update: reduce where pressure is high, increase where low
    S = float(a.sum())
    a_new = a - strength * S * pressure_w
    a_new = np.clip(a_new, 0.0, 1000.0)
    # Renormalize
    if a_new.sum() > 1e-12:
        a_new *= S / a_new.sum()
    return a_new


def solve_lp_direction(a, M, max_wall_s=15.0):
    """
    Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k.
    Returns optimal b or None.
    """
    if not HAVE_CVXPY:
        return None
    n = len(a)
    if n > 8000:
        return None
    t0 = time.monotonic()
    # Build sparse-like constraint: (a*b)[k] = sum_j a[|k-j| offset] * b[j]
    # Actually (a*b)[k] = sum_{j=0}^{n-1} a[k-j] * b[j] where a[x]=0 if x not in [0,n-1]
    # So for each k, the constraint involves b[0..n-1] with coefficients a[k-j]
    # This is a (2n-1) x n matrix with banded structure
    # For efficiency with n~7000, use sparse
    from scipy.sparse import csr_matrix, diags
    
    # Build the convolution matrix (2n-1) x n
    # Entry (k, j) = a[k-j] if 0 <= k-j <= n-1, else 0
    # Row k has nonzeros at j = max(0, k-n+1) .. min(k, n-1)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            rows.extend([k] * (j_end - j_start + 1))
            cols.extend(range(j_start, j_end + 1))
            vals.extend(a[j_end - (j_end - j_start):j_end + 1][::-1][:j_end - j_start + 1])
    
    # Actually let me be more careful
    # (a*b)[k] = sum_j a[k-j] * b[j] for j where 0<=k-j<n
    # For fixed k: j ranges from max(0, k-n+1) to min(k, n-1)
    # Coefficient of b[j] is a[k-j]
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    
    remaining = max_wall_s - (time.monotonic() - t0)
    if remaining < 1.0:
        return None
    try:
        prob.solve(solver=cp.PDLP, max_time=remaining)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_time=remaining)
        except Exception:
            return None
    
    if time.monotonic() - t0 > max_wall_s:
        return None
    if b_var.value is None:
        return None
    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None
    return b_opt


def lp_iteration(a, max_iters=10, max_wall_s=600.0):
    """
    Iterate: solve LP for direction, take step, polish with LSE.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_sc = compute_score(a)
    best_a = a.copy()
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > max_wall_s:
            break
        remaining = max_wall_s - (time.monotonic() - t0)
        if remaining < 60:
            break
        
        c = fast_conv(a)
        M = float(c.max())
        S = float(a.sum())
        
        # Solve LP
        lp_time = min(30.0, remaining * 0.4)
        b_opt = solve_lp_direction(a, M, max_wall_s=lp_time)
        if b_opt is None:
            print(f"    LP failed at iter {it}")
            break
        
        # Rescale b_opt to match sum of a
        b_opt = b_opt * (S / b_opt.sum())
        
        # Line search
        best_step_sc = best_sc
        best_step_a = best_a
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_try = (1.0 - t) * a + t * b_opt
            a_try = np.clip(a_try, 0.0, 1000.0)
            if a_try.sum() < 0.01:
                continue
            sc_try = compute_score(a_try)
            if sc_try < best_step_sc:
                best_step_sc = sc_try
                best_step_a = a_try.copy()
        
        if best_step_sc >= best_sc:
            # LP didn't help, try smaller steps
            for t in [0.05, 0.1, 0.15]:
                a_try = (1.0 - t) * a + t * b_opt
                a_try = np.clip(a_try, 0.0, 1000.0)
                if a_try.sum() < 0.01:
                    continue
                sc_try = compute_score(a_try)
                if sc_try < best_step_sc:
                    best_step_sc = sc_try
                    best_step_a = a_try.copy()
        
        a = best_step_a
        
        # Polish with LSE
        if time.monotonic() - t0 > max_wall_s - 120:
            break
        sc_poly, a_poly, _ = _lbfgs_step(a, 500, maxiter=1500, maxfun=3000)
        if sc_poly < best_sc:
            best_sc = sc_poly
            best_a = a_poly.copy()
        a = a_poly if sc_poly <= compute_score(a) else a
        
        if best_sc < compute_score(best_a) - 1e-10:
            print(f"    LP iter {it}: {best_sc:.8f}")
    
    return best_sc, best_a


def _lbfgs_step(a0, beta, maxiter=2000, maxfun=5000):
    """Single L-BFGS-B step with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        return lse_score_grad(x, beta)
    
    try:
        res = minimize(
            obj_and_grad, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0, None
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out, res


def _err_grad_step(a0, maxiter=2000, maxfun=5000):
    """L-BFGS-B with error-driven gradient."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        return error_driven_grad(np.clip(x, 0, 1000))
    
    try:
        res = minimize(
            obj_and_grad, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-14, 'gtol': 1e-11,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def dual_mwu_step(a, n_iters=50, max_wall_s=60.0):
    """
    Dual multiplicative weight update:
    1. p = current "weight" on convolution positions
    2. Solve inner QP: min a^T T_p a s.t. a>=0, sum(a)=1
    3. Update p multiplicatively based on (a*a)
    """
    n = len(a)
    a = a.astype(np.float64).copy() / a.sum()  # normalize to sum 1
    best_a = a.copy()
    best_sc = compute_score(best_a)
    t0 = time.monotonic()
    
    # Initialize p as uniform
    n_conv = 2 * n - 1
    p = np.ones(n_conv) / n_conv
    
    for it in range(n_iters):
        if time.monotonic() - t0 > max_wall_s:
            break
        
        # Compute (a*a)
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Update p: multiplicative weight
        # p_k proportional to exp(eta * c_k)
        eta = 5.0 + 5.0 * it / n_iters
        p_new = p * np.exp(eta * (c - c_max))
        p = p_new / p_new.sum()
        
        # Solve inner QP: min a^T T_p a s.t. a >= 0, sum(a) = 1
        # Use projected gradient (a few iterations)
        a_qp = a.copy()
        for pg_it in range(20):
            # Compute T_p @ a_qp: (T_p a)_i = sum_j p_{i+j} * a_j
            # This is a Hankel-type product
            tp_a = np.zeros(n)
            for i in range(n):
                # sum_j p[i+j] * a[j] for j where i+j < n_conv
                j_max = min(n, n_conv - i)
                if j_max > 0:
                    tp_a[i] = np.dot(p[i:i + j_max], a_qp[:j_max])
            # Projected gradient step
            grad = 2.0 * tp_a
            a_qp -= 0.01 * grad
            a_qp = np.maximum(a_qp, 0.0)
            if a_qp.sum() > 1e-12:
                a_qp /= a_qp.sum()
        
        sc_qp = compute_score(a_qp)
        if sc_qp < best_sc:
            best_sc = sc_qp
            best_a = a_qp.copy()
        
        # Blend: keep some of the old a
        a = 0.7 * a + 0.3 * a_qp
        a = np.maximum(a, 0.0)
        if a.sum() > 1e-12:
            a /= a.sum() * a.sum()  # keep sum = 1... actually just normalize
            a = a / a.sum()
    
    # Rescale back
    best_a = best_a * (a.sum() / best_a.sum()) if best_a.sum() > 0 else a
    # Actually, since score is scale-invariant, just return best_a
    sb = compute_score(best_a)
    return sb, best_a


def exact_subgrad_polish(a, total_iters=500):
    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    S_orig = float(a.sum())
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max(c) w.r.t. a
        k_star = int(np.argmax(c))
        sg_M = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        
        # If multiple peaks are close, average their subgradients
        c_close = c > M * 0.999
        n_close = int(c_close.sum())
        if n_close > 1:
            sg_M = np.zeros(n)
            for k2 in np.where(c_close)[0]:
                js = max(0, k2 - n + 1)
                je = min(k2, n - 1)
                if je >= js:
                    sg_M[js:je + 1] += 2.0 * a[js:je + 1][::-1]
            sg_M /= n_close
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        step = min(0.1, 0.01 / (gn + 1e-15))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a


def multi_peak_lse_grad(a, beta, n_peaks=5):
    """
    LSE over top n_peaks convolution values (not just global max).
    Smoother than pure LSE for small beta, more targeted than full LSE.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    # Weight: high where c is large
    c_max = float(c.max())
    c_min = float(c.min())
    # Use a soft weighting based on rank
    threshold = np.sort(c)[::-1][min(n_peaks, len(c) - 1)]
    # Smooth weight: sigmoid around threshold
    width = max((c_max - threshold) * 0.5, (c_max - c_min) * 0.01, 1e-10)
    w = 1.0 / (1.0 + np.exp(-5.0 * (c - threshold) / width))
    w_sum = w.sum() + 1e-30
    p = w / w_sum
    L = float(np.dot(p, c))
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = rfft(rev_a, M2_fft)
    P = rfft(p, M2_fft)
    conv_res = irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs_multi_peak(a0, beta, n_peaks, maxiter=2000, maxfun=5000):
    """L-BFGS-B with multi-peak LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        return multi_peak_lse_grad(np.clip(x, 0, 1000), beta, n_peaks)
    
    try:
        res = minimize(
            obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def build_inits(n, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            # Interpolate parent to length n
            idx = np.linspace(0, len(parent) - 1, n)
            lo = idx.astype(int)
            hi = np.minimum(lo + 1, len(parent) - 1)
            frac = idx - lo
            inits.append(parent[lo] * (1 - frac) + parent[hi] * frac)
    
    # Structured
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    # Sinc-like
    inits.append(np.clip(0.5 + 0.3 * np.sinc(3.0 * (x - 0.5)), 0.01, 10.0))
    # Power law
    inits.append(np.clip(0.5 + 0.3 * 4.0 * x ** 0.5 * (1 - x) ** 0.5, 0.01, 10.0))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = np.ones(1024) * 0.5
    
    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False
    
    # Determine primary length
    if parent is not None:
        primary_n = len(parent)
    else:
        primary_n = 6962
    
    print(f"Primary length: {primary_n}, parent: {parent is not None}")
    print(f"CVXPY: {HAVE_CVXPY}")
    
    # =========================================================================
    # Phase 1: LSE continuation with L-BFGS-B (40% of time)
    # =========================================================================
    phase1_end = t_start + 0.40 * (budget_s - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: LSE Continuation (until t={phase1_end - t_start:.0f}s)")
    print(f"{'='*60}")
    
    a_cur = np.clip(parent, 0, 1000).copy() if parent is not None else np.ones(primary_n) * 0.5
    
    # Quick initial polish
    sc, a_cur, _ = run_lbfgs_multi_peak(a_cur, 50, n_peaks=3, maxiter=1000, maxfun=3000)
    report(sc, a_cur, "init polish")
    
    # Beta continuation
    betas = [50, 100, 200, 400, 800, 1500, 3000]
    for beta in betas:
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 30:
            break
        sc, a_cur, _ = run_lbfgs_multi_peak(a_cur, beta, n_peaks=3, maxiter=3000, maxfun=8000)
        report(sc, a_cur, f"LSE beta={beta}")
        
        # Also try standard LSE
        sc2, a2, _ = _lbfgs_step(a_cur, beta, maxiter=1500, maxfun=4000)
        if sc2 < best_score:
            report(sc2, a2, f"std LSE beta={beta}")
            a_cur = a2
    
    print(f"  After Phase 1: {best_score:.8f}")
    
    # =========================================================================
    # Phase 2: LP-based direction iteration (25% of time)
    # =========================================================================
    phase2_end = t_start + 0.65 * (budget_s - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: LP Direction (until t={phase2_end - t_start:.0f}s)")
    print(f"{'='*60")
    
    if HAVE_CVXPY and len(best_a) <= 8000:
        lp_time = phase2_end - time.monotonic()
        sc, a_lp = lp_iteration(best_a.copy(), max_iters=30, max_wall_s=lp_time)
        report(sc, a_lp, "LP done")
    else:
        print("  Skipping LP (no cvxpy or too large)")
        # Use error-driven gradient instead
        a_ed = best_a.copy()
        while time.monotonic() < phase2_end:
            sc_ed, a_ed = _err_grad_step(a_ed, maxiter=2000, maxfun=5000)
            if not report(sc_ed, a_ed, "err-grad"):
                break
    
    print(f"  After Phase 2: {best_score:.8f}")
    
    # =========================================================================
    # Phase 3: Peak pressure + LSE polish (15% of time)
    # =========================================================================
    phase3_end = t_start + 0.80 * (budget_s - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Peak Pressure (until t={phase3_end - t_start:.0f}s)")
    print(f"{'='*60}")
    
    a_pp = best_a.copy()
    n_pp = len(a_pp)
    pp_trial = 0
    while time.monotonic() < phase3_end and pp_trial < 20:
        # Peak pressure step
        strength = 0.005 + 0.01 * rng.random()
        a_new = peak_pressure_step(a_pp, n_top=30, strength=strength)
        # Polish
        sc_new, a_pol = run_lbfgs_multi_peak(a_new, 800, n_peaks=3, maxiter=2000, maxfun=5000)
        if sc_new < best_score:
            report(sc_new, a_pol, f"PP #{pp_trial}")
            a_pp = a_pol
        else:
            # Try from current best with different pressure
            a_new = peak_pressure_step(best_a, n_top=10 + int(20 * rng.random()), strength=strength * 0.5)
            sc_new, a_pol = run_lbfgs_multi_peak(a_new, 800, n_peaks=3, maxiter=2000, maxfun=5000)
            if sc_new < best_score:
                report(sc_new, a_pol, f"PP-best #{pp_trial}")
                a_pp = a_pol
        pp_trial += 1
    
    print(f"  After Phase 3: {best_score:.8f}")
    
    # =========================================================================
    # Phase 4: Error-driven refinement + multi-peak LSE (10% of time)
    # =========================================================================
    phase4_end = t_start + 0.90 * (budget_s - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 4: Error-Driven + Multi-Peak (until t={phase4_end - t_start:.0f}s)")
    print(f"{'='*60")
    
    a_ed = best_a.copy()
    ed_trial = 0
    while time.monotonic() < phase4_end and ed_trial < 15:
        # Error-driven step
        sc_ed, a_ed_new = _err_grad_step(a_ed, maxiter=2000, maxfun=5000)
        if sc_ed < best_score:
            report(sc_ed, a_ed_new, f"ED #{ed_trial}")
            a_ed = a_ed_new
            a_ed = run_lbfgs_multi_peak(a_ed, 1000, n_peaks=5, maxiter=1000, maxfun=3000)[1]
        else:
            # Try multi-peak with different params
            for npk in [5, 10, 20]:
                sc_mp, a_mp = run_lbfgs_multi_peak(a_ed, 2000, n_peaks=npk, maxiter=1500, maxfun=3000)
                if sc_mp < best_score:
                    report(sc_mp, a_mp, f"MP k={npk} #{ed_trial}")
                    a_ed = a_mp
                    break
        ed_trial += 1
    
    print(f"  After Phase 4: {best_score:.8f}")
    
    # =========================================================================
    # Phase 5: Final refinement (10% of time)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Refinement")
    print(f"{'='*60")
    
    # High-beta LSE
    for beta in [3000, 5000, 8000]:
        if time.monotonic() > deadline - 10:
            break
        sc, a_f, _ = _lbfgs_step(best_a, beta, maxiter=5000, maxfun=100000)
        report(sc, a_f, f"final LSE b={beta}")
    
    # Multi-peak final
    for beta in [2000, 5000]:
        if time.monotonic() > deadline - 10:
            break
        sc, a_f = run_lbfgs_multi_peak(best_a, beta, n_peaks=3, maxiter=5000, maxfun=100000)
        report(sc, a_f, f"final MP b={beta}")
    
    # Exact subgradient polish
    if time.monotonic() < deadline - 5:
        sc, a_sg = exact_subgrad_polish(best_a, total_iters=2000)
        report(sc, a_sg, "subgrad")
    
    # One more LSE from subgrad result
    if time.monotonic() < deadline - 3:
        sc, a_f, _ = _lbfgs_step(best_a, 5000, maxiter=5000, maxfun=50000)
        report(sc, a_f, "very final LSE")
    
    # =========================================================================
    # Phase 6: Try different lengths if time remains
    # =========================================================================
    if time.monotonic() < deadline - 120:
        print(f"\n{'='*60}")
        print(f"Phase 6: Length Exploration")
        print(f"{'='*60")
        
        n_best = len(best_a)
        alt_lengths = [n_best - 100, n_best + 100, n_best + 200, n_best - 200]
        alt_lengths = [l for l in alt_lengths if l >= 2000 and l <= 12000]
        
        for n_alt in alt_lengths:
            if time.monotonic() > deadline - 60:
                break
            # Resample best_a to new length
            idx = np.linspace(0, len(best_a) - 1, n_alt)
            lo = idx.astype(int)
            hi = np.minimum(lo + 1, len(best_a) - 1)
            frac = idx - lo
            a_alt = best_a[lo] * (1 - frac) + best_a[hi] * frac
            a_alt = np.clip(a_alt, 0, 1000)
            
            sc, a_opt = run_lbfgs_multi_peak(a_alt, 200, n_peaks=3, maxiter=2000, maxfun=5000)
            report(sc, a_opt, f"len={n_alt}")
            
            sc2, a_opt2, _ = _lbfgs_step(a_opt, 1000, maxiter=2000, maxfun=5000)
            report(sc2, a_opt2, f"len={n_alt} polish")
    
    # =========================================================================
    # Final verification
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
