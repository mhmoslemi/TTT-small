"""
Search for non-negative step-function heights minimizing
    2*n*max(self-convolution) / (sum)^2

Strategy:
- Start from strong analytic/provided candidates:
    * previous best if available,
    * truncations/tilts of the infinite flat-square-root sequence
      coefficients of (1-z)^(-1/2),
    * symmetric variants and a simple random positive initializer.
- Main local optimizer is a dual/multiplicative KKT-style iteration:
    given a probability vector a, choose a dual distribution lambda over convolution lags
    concentrated on near-maximum coefficients, compute sensitivities
        r_i = sum_k lambda_k a_{k-i}
    and update multiplicatively
        a_i <- a_i * (C / r_i)^eta,  C = lambda . (a*a)
    which enforces the KKT condition r_i ~= C on the support.
- The method accepts only moves that do not hurt the exact objective too much,
  occasionally symmetrizes, restarts from perturbed best points, and finally polishes
  with a short LSE-smoothed Adam run.
"""

import os
os.environ.setdefault("OMP_NUM_THREADS", "1")
os.environ.setdefault("OPENBLAS_NUM_THREADS", "1")
os.environ.setdefault("MKL_NUM_THREADS", "1")
os.environ.setdefault("NUMEXPR_NUM_THREADS", "1")

import numpy as np
import time


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.size)
    return a / s


def symmetrize(a):
    a = np.asarray(a, dtype=np.float64)
    a = 0.5 * (a + a[::-1])
    return normalize_sequence(a)


def compute_score(a, n_fft=None, L=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    if L is None:
        L = 2 * n - 1
    if n_fft is None:
        n_fft = next_pow2(L)
    A = np.fft.rfft(a, n_fft)
    c = np.fft.irfft(A * A, n_fft)[:L]
    if c.size == 0:
        return np.inf
    M = float(c.max())
    S = float(a.sum())
    if not np.isfinite(S) or S <= 1e-12 or not np.isfinite(M):
        return np.inf
    return float(2.0 * n * M / (S * S))


def b_init(n, shift=0, tilt=0.0, sym=False):
    """
    Initializers based on coefficients of (1-z)^(-1/2):
        b_i = binom(2i, i) / 4^i,  b_0 = 1.
    The infinite sequence has perfectly flat self-convolution; finite windows/tilts
    are useful starting points.
    """
    n = int(n)
    if n <= 0:
        return np.array([1.0])
    if n == 1:
        return np.array([1.0])
    shift = int(max(0, shift))
    total = shift + n

    idx = np.arange(1, total, dtype=np.float64)
    vals = np.empty(total, dtype=np.float64)
    vals[0] = 1.0
    if total > 1:
        ratios = (2.0 * idx - 1.0) / (2.0 * idx)
        vals[1:] = np.cumprod(ratios)

    a = vals[shift:shift + n].astype(np.float64).copy()
    if tilt != 0.0:
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        a *= np.exp(-tilt * x)
    if sym:
        a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 1e-300)
    s = float(a.sum())
    if s > 1e-300:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def resample_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    new_n = int(new_n)
    if new_n <= 0:
        return np.array([1.0])
    if n == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    if n == new_n:
        return normalize_sequence(a)
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0.0, n - 1.0, new_n, dtype=np.float64)
    out = np.interp(x_new, x_old, a)
    return normalize_sequence(out)


def random_positive_init(n, rng):
    n = int(n)
    a = np.ones(n, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for _ in range(4):
        f = float(rng.uniform(1.0, 8.0))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.05, 0.22))
        a += amp * np.cos(2.0 * np.pi * f * x + ph)
    a = np.maximum(a, 1e-6)
    return normalize_sequence(a)


def perturb_sequence(a, rng, scale=0.02):
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    if n == 0:
        return np.array([1.0])
    x = 1.0 + float(scale) * rng.standard_normal(n)
    x = np.clip(x, 0.2, 5.0)
    a *= x
    a = np.maximum(a, 1e-300)
    s = float(a.sum())
    if s > 1e-300:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def make_lambda(c, M, p, gap_frac, lam_old, momentum, min_active=8):
    """
    Dual weights over convolution lags, concentrated on near-maximum coefficients.
    """
    L = c.size
    if M <= 0.0 or not np.isfinite(M):
        return np.ones(L, dtype=np.float64) / float(L)

    thr = M * (1.0 - float(gap_frac))
    if thr < 0.0:
        thr = 0.0
    active = c >= thr

    if int(active.sum()) < min_active:
        if L > min_active:
            idx = np.argpartition(c, -min_active)[-min_active:]
            active[idx] = True
        else:
            active.fill(True)

    vals = np.zeros(L, dtype=np.float64)
    rel = (c[active] - M) / M
    w = np.exp(np.clip(p * rel, -700.0, 0.0))
    vals[active] = w
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        vals[active] = 1.0
        s = float(active.sum())

    lam_new = vals / s
    if lam_old is not None and momentum > 0.0:
        lam = (1.0 - momentum) * lam_new + momentum * lam_old
        lam = np.maximum(lam, 0.0)
        ls = float(lam.sum())
        if np.isfinite(ls) and ls > 0.0:
            lam /= ls
        else:
            lam = lam_new
    else:
        lam = lam_new
    return lam


def lse_from_c(c, M, p):
    """Unscaled LSE part: max + logsumexp(p*(c-M)/M) / p * M."""
    if M <= 0.0 or not np.isfinite(M):
        return 0.0
    if p <= 0.0:
        return float(M)
    rel = (c - M) / M
    scaled = np.clip(p * rel, -700.0, 0.0)
    s = float(np.exp(scaled).sum())
    if not np.isfinite(s) or s <= 0.0:
        return float(M)
    return float(M * (1.0 + np.log(s) / p))


def dm_optimize(a0, end_time, rng=None, sym=False,
                p0=60.0, p_end=800.0, gap0=0.20, gap_end=0.004,
                eta0=0.70, eta_end=0.15,
                lam_momentum=0.30, log_momentum=0.10,
                max_log=6.0, tag="dm"):
    """Deterministic multiplicative dual-KKT optimizer, time-bounded."""
    a = normalize_sequence(a0)
    t_start = time.monotonic()
    if time.monotonic() >= float(end_time):
        return compute_score(a), a

    n = a.size
    L = 2 * n - 1
    N = next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = compute_score(a, n_fft=N, L=L)

    lam = None
    vel_log = np.zeros(n, dtype=np.float64)
    eta_scale = 1.0
    no_imp = 0
    reject = 0
    last_print = t_start
    iter_count = 0
    budget_alloc = max(1e-6, float(end_time) - t_start)

    while time.monotonic() < float(end_time):
        elapsed = time.monotonic() - t_start
        if elapsed - last_print > 20.0:
            print(f"  [{tag}] it={iter_count:5d} best={best_score:.8f} cur~{best_score:.8f} frac={min(1.0, elapsed/budget_alloc):.2f}")
            last_print = time.monotonic()

        frac = min(1.0, elapsed / budget_alloc)
        p = p0 + (p_end - p0) * frac
        gap = gap0 + (gap_end - gap0) * frac
        eta = max(0.01, (eta0 + (eta_end - eta0) * frac) * eta_scale)

        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A, N)[:L]
        M = float(c.max())
        if not np.isfinite(M) or M <= 0.0:
            break
        score = 2.0 * n * M  # a is normalized

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0

        lam = make_lambda(c, M, p, gap, lam, lam_momentum, min_active=8)

        rev_lam = lam[::-1]
        Lambda = np.fft.rfft(rev_lam, N)
        y = np.fft.irfft(Lambda * A, N)
        r = y[n - 1:2 * n - 1][::-1]

        C = float(np.dot(lam, c))
        if not np.isfinite(C) or C <= 0.0:
            C = M

        min_r = max(C * 1e-12, 1e-300)
        r_safe = np.maximum(r, min_r)

        step = eta * (np.log(C) - np.log(r_safe))
        if log_momentum > 0.0:
            vel_log = log_momentum * vel_log + step
        else:
            vel_log = step
        step = np.clip(vel_log, -max_log, max_log)

        cand = a * np.exp(step)
        cand = np.maximum(cand, 1e-300)
        cs = float(cand.sum())
        accepted = False
        if np.isfinite(cs) and cs > 1e-300:
            cand /= cs
            if sym:
                cand = 0.5 * (cand + cand[::-1])
                cs2 = float(cand.sum())
                if cs2 > 0.0 and np.isfinite(cs2):
                    cand /= cs2

            A2 = np.fft.rfft(cand, N)
            c2 = np.fft.irfft(A2 * A2, N)[:L]
            M2 = float(c2.max())
            score2 = 2.0 * n * M2

            if (np.isfinite(score2) and
                    (score2 <= score * (1.0 + 1e-8) or score2 < best_score)):
                accepted = True
            else:
                # Use a smoothed LSE objective as a secondary acceptance signal.
                lse_cur = lse_from_c(c, M, p)
                lse_cand = lse_from_c(c2, M2, p)
                if (np.isfinite(score2) and np.isfinite(lse_cand) and
                        lse_cand <= lse_cur * (1.0 - 1e-5) and
                        score2 <= best_score * 1.02):
                    accepted = True

            if accepted:
                a = cand
                reject = 0
                eta_scale = min(1.0, eta_scale * 1.02)
                if score2 < best_score - 1e-12:
                    best_score = score2
                    best_a = a.copy()
                    no_imp = 0
                else:
                    no_imp += 1
            else:
                reject += 1
                no_imp += 1
                if reject > 30 and reject % 30 == 0:
                    eta_scale = max(0.05, eta_scale * 0.75)

        # Periodic symmetrization test (objective is reversal invariant).
        if iter_count % 1000 == 0 and not sym:
            sym_a = 0.5 * (a + a[::-1])
            sym_a = normalize_sequence(sym_a)
            sc_sym = compute_score(sym_a, n_fft=N, L=L)
            if np.isfinite(sc_sym) and sc_sym < best_score:
                best_score = sc_sym
                best_a = sym_a.copy()
                a = sym_a
                sym = True
                lam = None
                vel_log.fill(0.0)
                no_imp = 0
                print(f"  [{tag}] symmetrized improvement -> {best_score:.8f}")

        # If stuck, restart from a mild perturbation of the best point.
        if no_imp > 1200 and rng is not None:
            scale = 0.015 + 0.015 * ((no_imp % 3000) // 1000)
            a = perturb_sequence(best_a, rng, scale=scale)
            lam = None
            vel_log.fill(0.0)
            no_imp = 0
            reject = 0
            eta_scale = 1.0
            print(f"  [{tag}] restart from perturbed best at it={iter_count}")

        iter_count += 1

    final_sc = compute_score(best_a, n_fft=N, L=L)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, best_a


def optimize_initial(a_init, budget, rng, sym=False, tag="init",
                     p0=60.0, p_end=800.0, gap0=0.20, gap_end=0.004,
                     eta0=0.70, eta_end=0.15,
                     lam_momentum=0.30, log_momentum=0.10, max_log=6.0):
    if budget <= 3.0:
        a = normalize_sequence(a_init)
        return compute_score(a), a
    a = normalize_sequence(a_init)
    end = time.monotonic() + max(1.0, float(budget))
    sc, a_opt = dm_optimize(
        a, end, rng=rng, sym=sym,
        p0=p0, p_end=p_end, gap0=gap0, gap_end=gap_end,
        eta0=eta0, eta_end=eta_end,
        lam_momentum=lam_momentum, log_momentum=log_momentum,
        max_log=max_log, tag=tag
    )
    if not sym:
        sym_a = symmetrize(a_opt)
        sc_sym = compute_score(sym_a)
        if sc_sym < sc:
            sc = sc_sym
            a_opt = sym_a
    return sc, a_opt


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient in a-space."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 1e30, np.array([0.0])
    L = 2 * n - 1
    A = np.fft.rfft(a, next_pow2(L))
    c = np.fft.irfft(A * A, next_pow2(L))[:L]
    S = float(a.sum())
    if not np.isfinite(S) or S <= 1e-12:
        return 1e30, np.zeros(n, dtype=np.float64)

    c_max = float(c.max())
    if not np.isfinite(c_max) or c_max <= 0.0:
        return 1e30, np.zeros(n, dtype=np.float64)

    scaled = float(beta) * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -700.0, 0.0))
    ez = float(exp_scaled.sum())
    if not np.isfinite(ez) or ez <= 0.0:
        exp_scaled = (c == c_max).astype(np.float64)
        ez = float(exp_scaled.sum())
        if ez <= 0.0:
            return 1e30, np.zeros(n, dtype=np.float64)

    Lse = c_max + np.log(ez) / float(beta)
    score = 2.0 * n * Lse / (S * S)
    p = exp_scaled / ez

    M2 = 3 * n - 2
    M2_fft = next_pow2(M2)
    RA = np.fft.rfft(a[::-1], M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1][::-1]
    g = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * Lse / S3)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    return float(score), grad


def run_adam_lse(a_init, beta, lr, deadline, n_steps=100000, tag="adam_lse"):
    a = normalize_sequence(a_init)
    t0 = time.monotonic()
    if time.monotonic() >= float(deadline):
        return compute_score(a), a

    best_a = a.copy()
    best_score = compute_score(a)

    m = np.zeros_like(a, dtype=np.float64)
    v = np.zeros_like(a, dtype=np.float64)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    while time.monotonic() < float(deadline):
        t += 1
        if t > n_steps:
            break
        sc_lse, grad = lse_score_grad_a(a, beta)
        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        a = a - float(lr) * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 1e-300, 1000.0)
        s = float(a.sum())
        if s > 1e-12:
            a /= s

        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"  [{tag}] t={t} best={best_score:.8f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_score, best_a


def get_parent():
    g = globals()
    if "height_sequence_1" not in g or g["height_sequence_1"] is None:
        return None
    try:
        arr = np.asarray(g["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size < 16:
        return None
    arr = np.clip(arr, 0.0, 1000.0)
    if not np.all(np.isfinite(arr)):
        return None
    if float(arr.sum()) <= 1e-12:
        return None
    return arr


def propose_candidate(seed=42, budget_s=2400):
    """
    Returns the best non-negative height sequence found within the time budget.
    """
    t0 = time.monotonic()
    total = max(120.0, float(budget_s) - 30.0)
    hard_end = t0 + total
    rng = np.random.default_rng(seed)

    best_a = None
    best_score = np.inf

    parent = get_parent()
    if parent is not None:
        parent = normalize_sequence(parent)
        n_parent = int(parent.size)
        base_n = int(np.clip(n_parent, 2048, 12288))
        print(f"  parent length={n_parent}, score={compute_score(parent):.8f}")
    else:
        n_parent = 8192
        base_n = 8192
        print("  no parent available; base_n=8192")

    if base_n > 12288:
        base_n = 12288
    large_n = int(np.clip(int(base_n * 1.35), base_n + 512, 12288))
    if large_n <= base_n:
        large_n = min(12288, base_n + 1024)

    specs = []

    if parent is not None:
        p = normalize_sequence(parent)
        specs.append((p, False, "parent"))
        specs.append((symmetrize(p), True, "parent_sym"))
        if large_n != p.size:
            specs.append((resample_sequence(p, large_n), False, f"parent_large_{large_n}"))
        if base_n != p.size:
            specs.append((resample_sequence(p, base_n), False, f"parent_base_{base_n}"))

    n_list = [base_n]
    if large_n != base_n:
        n_list.append(large_n)

    if parent is None:
        tilts = [-1.0, -0.6, -0.3, -0.15, 0.0, 0.15, 0.3, 0.6]
    else:
        tilts = [-0.6, -0.3, -0.15, 0.0, 0.15, 0.3]

    for n in n_list:
        for t in tilts:
            a = b_init(n, 0, t)
            specs.append((a, False, f"b_n{n}_tilt{t:+.2f}"))
        a = b_init(n, 0, -0.4, sym=True)
        specs.append((a, True, f"b_n{n}_sym0"))
        a = b_init(n, 0, 0.2, sym=True)
        specs.append((a, True, f"b_n{n}_sym1"))
        if n <= 12288:
            a = b_init(n, int(0.25 * n), 0.0)
            specs.append((a, False, f"b_n{n}_shift25"))

    if not specs:
        specs.append((b_init(base_n, 0, 0.0), False, "b_default"))

    pairs = []
    for i, (a, symf, tag) in enumerate(specs):
        sc = compute_score(a)
        pairs.append((sc, i))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    pairs.sort()

    if best_a is None:
        best_a = normalize_sequence(specs[pairs[0][1]][0])
        best_score = compute_score(best_a)

    # Reserve time for final polish.
    reserve_final = 190.0
    elapsed = time.monotonic() - t0
    remaining = max(0.0, total - elapsed - reserve_final)

    # Main experiment on the best quick-scoring initializer.
    if remaining > 80.0 and len(pairs) > 0:
        idx = pairs[0][1]
        a0, symf, tag = specs[idx]
        budgetA = min(max(240.0, remaining * 0.45), 1450.0)
        sc, a_opt = optimize_initial(
            a0, budgetA, rng, sym=symf, tag=f"A_{tag}",
            p0=60.0, p_end=900.0, gap0=0.20, gap_end=0.003,
            eta0=0.70, eta_end=0.15,
            lam_momentum=0.30, log_momentum=0.10, max_log=6.0
        )
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
        print(f"  main experiment {tag}: {sc:.8f}, global best {best_score:.8f}")

    # Additional experiments on next-best initializers.
    for count, (sc_q, idx) in enumerate(pairs[1:6], start=1):
        elapsed = time.monotonic() - t0
        remaining = max(0.0, total - elapsed - reserve_final)
        if remaining < 70.0:
            break
        if count > 2 and best_score < 1.5027:
            break

        a0, symf, tag = specs[idx]
        budgetB = min(remaining * 0.22, 750.0)
        sc, a_opt = optimize_initial(
            a0, budgetB, rng, sym=symf, tag=f"B{count}_{tag}",
            p0=80.0, p_end=800.0, gap0=0.15, gap_end=0.005,
            eta0=0.55, eta_end=0.14,
            lam_momentum=0.30, log_momentum=0.08, max_log=6.0
        )
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
        print(f"  secondary experiment B{count} {tag}: {sc:.8f}, global best {best_score:.8f}")

    # Final polishing.
    final_budget = min(reserve_final, max(0.0, total - (time.monotonic() - t0)))
    if final_budget > 25.0 and best_a is not None:
        # High-conviction DMU polish.
        f1 = min(final_budget * 0.55, final_budget - 30.0)
        if f1 > 20.0:
            is_good_sym = bool(np.allclose(best_a, best_a[::-1], rtol=1e-6, atol=1e-10))
            sc, a_opt = optimize_initial(
                best_a, f1, rng, sym=is_good_sym, tag="final_dm",
                p0=400.0, p_end=1200.0, gap0=0.020, gap_end=0.002,
                eta0=0.22, eta_end=0.05,
                lam_momentum=0.32, log_momentum=0.05, max_log=5.0
            )
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
            print(f"  final DMU: {sc:.8f}, global best {best_score:.8f}")

        # LSE Adam polish.
        elapsed = time.monotonic() - t0
        left = max(0.0, total - elapsed - 10.0)
        if left > 25.0 and best_a is not None:
            A = np.fft.rfft(best_a, next_pow2(2 * best_a.size - 1))
            c = np.fft.irfft(A * A, next_pow2(2 * best_a.size - 1))[:2 * best_a.size - 1]
            M = float(c.max())
            if not np.isfinite(M) or M <= 0.0:
                beta = 10000.0
            else:
                beta = max(1000.0, 50.0 / M)
            beta = min(beta, 2.0e6)
            deadline = min(hard_end - 2.0, time.monotonic() + left * 0.9)
            sc, a_opt = run_adam_lse(best_a, beta, lr=0.006, deadline=deadline,
                                     n_steps=200000, tag="final_adam")
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
            print(f"  final LSE Adam: {sc:.8f}, global best {best_score:.8f}")

    # Last-second symmetrization and validation.
    if best_a is None:
        best_a = b_init(4096, 0, 0.0)
        best_score = compute_score(best_a)

    sym_a = symmetrize(best_a)
    sc_sym = compute_score(sym_a)
    if sc_sym < best_score:
        best_score = sc_sym
        best_a = sym_a

    best_a = np.asarray(best_a, dtype=np.float64)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(best_a.sum())
    if not np.isfinite(s) or s < 0.01:
        out = [0.5] * 64
    else:
        out = [float(x) for x in best_a]

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = float(best_score)

    elapsed = time.monotonic() - t0
    print(f"FINAL internal={best_score:.8f} verify={verified:.8f} len={len(out)} time={elapsed:.1f}s")
    return out
