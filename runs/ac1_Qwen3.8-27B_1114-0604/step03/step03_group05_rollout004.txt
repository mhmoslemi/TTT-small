"""
Step function optimization using LP-based water filling, LSE continuation,
and adaptive peak shaving. Targets sub-1.5027 score.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lse_obj(a, beta):
    return lse_score_grad(a, beta)

def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0

def solve_lp_waterfill(a, M, top_k=None):
    """
    Solve: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0
    Returns the optimal b.
    """
    n = len(a)
    # The convolution (a*b)_k = sum_j a[j] * b[k-j]
    # For k = 0: a[0]*b[0]
    # For k = 1: a[0]*b[1] + a[1]*b[0]
    # ...
    # For k = n-1: a[0]*b[n-1] + a[1]*b[n-2] + ... + a[n-1]*b[0]
    # ...
    # For k = 2n-2: a[n-1]*b[n-1]
    
    # The constraint matrix A_eq has shape (2n-1, n)
    # A_eq[k, j] = a[j] if 0 <= k-j < n, else 0
    # i.e., A_eq[k, j] = a[j] if k-n+1 <= j <= k and 0 <= j < n
    
    # For efficiency, let's build this sparse matrix
    from scipy.sparse import csr_matrix, diags
    
    # Build the constraint matrix
    # Row k, column j: a[j] if max(0, k-n+1) <= j <= min(k, n-1)
    # This is a banded matrix structure
    
    # Actually, for n=7000, building a dense (2n-1) x n matrix is (14000) x 7000 = 98 million entries.
    # That's about 784 MB as float64. Too much for linprog.
    # Let's use the top_k approach or a different formulation.
    
    if top_k is None:
        top_k = min(200, 2 * n - 1)
    
    # Compute convolution to find top peaks
    c = fast_conv(a)
    # Actually, we need (a*b)_k for the LP, which depends on b.
    # The constraint is: for each k, sum_j A[k,j] * b[j] <= M
    # where A[k,j] = a[j] if 0 <= k-j < n else 0
    
    # For the top K peaks of the CURRENT convolution, we add constraints.
    # But we should also add some random constraints to cover the space.
    
    # Let's use the top K peaks plus some evenly spaced ones
    sorted_idx = np.argsort(c)[::-1]
    top_idx = sorted_idx[:min(top_k, 2*n-1)]
    # Add some evenly spaced indices
    if 2*n-1 > top_k:
        extra = np.linspace(0, 2*n-2, min(50, 2*n-1-top_k)).astype(int)
        all_idx = np.unique(np.concatenate([top_idx, extra]))
    else:
        all_idx = top_idx
    
    # Build constraint matrix (sparse)
    rows = []
    cols = []
    vals = []
    for k in all_idx:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j - j_start + (j_start))  # just j
            vals.append(a[j])
    
    # Actually let me just build it properly
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(all_idx):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[j])
    
    from scipy.sparse import csr_matrix
    A_sp = csr_matrix((vals, (rows, cols)), shape=(len(all_idx), n))
    
    # Objective: maximize sum(b) → minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Bounds: b >= 0 (no upper bound, but let's cap at something reasonable)
    # The sum of b will be at most M * (2n-1) / min(a) roughly, but let's just use a generous bound
    b_max = 10.0  # Cap each b_j at 10 (should be more than enough after normalization)
    bounds = [(0, b_max)] * n
    
    try:
        from scipy.optimize import linprog
        res = linprog(c_obj, A_ub=A_sp, b_ub=np.full(len(all_idx), M),
                      bounds=bounds, method='highs')
        if res.success:
            return np.clip(res.x, 0, None)
        else:
            return None
    except Exception:
        return None

def lp_waterfill_iteration(a, top_k=200, line_search_steps=20):
    """
    One iteration of LP-based water filling.
    Returns (new_a, score_old, score_new).
    """
    n = len(a)
    c = fast_conv(a)
    M = float(np.max(c))
    S = float(a.sum())
    score_old = 2.0 * n * M / (S * S)
    
    # Solve LP
    b = solve_lp_waterfill(a, M, top_k=top_k)
    if b is None or b.sum() < 1e-10:
        return a.copy(), score_old, score_old
    
    # Normalize b to have the same sum as a
    b = b * S / b.sum()
    
    # Line search
    best_a = a.copy()
    best_sc = score_old
    for t_candidate in np.linspace(0.05, 0.5, line_search_steps):
        a_trial = (1 - t_candidate) * a + t_candidate * b
        a_trial = np.clip(a_trial, 0, 1000)
        sc = compute_score(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
    
    return best_a, score_old, best_sc

def lse_score_grad_v2(a, beta):

def peak_shave_adaptive(a, iters=2000):
    """Peak shaving with adaptive step size and backtracking."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    step_scale = 1.0
    
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Subgradient step with backtracking
        step = step_scale / (gn * S + 1e-20)
        for _ in range(10):
            a_trial = a - step * grad
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            sc_trial = compute_score(a_trial)
            if sc_trial < sc:
                a = a_trial
                # Increase step scale slightly
                step_scale = min(step_scale * 1.1, 10.0)
                break
            else:
                step *= 0.5
                step_scale = max(step_scale * 0.9, 0.01)
        else:
            # Couldn't improve, try a smaller step or break
            break
    
    return best_sc, best_a

"""
Step function optimization: LP-based water filling with LSE continuation
and adaptive peak shaving. Targets sub-1.5027 score.
Strategy: LP water filling (main) → LSE polish → peak shave → restarts
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=2000, maxfun=50000):
    """Run L-BFGS-B optimization on LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def build_lp_matrix(a, constraint_indices):
    """
    Build the constraint matrix for the LP water-filling problem.
    (a*b)_k = sum_j A[k,j] * b[j] where A[k,j] = a[j] if k-n+1 <= j <= k and 0<=j<n
    """
    n = len(a)
    num_constraints = len(constraint_indices)
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end < j_start:
            continue
        # Add entries for this row
        rows.extend([i] * (j_end - j_start + 1))
        cols.extend(range(j_start, j_end + 1))
        vals.extend(a[j_start:j_end + 1].tolist())
    
    return csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))


def solve_lp_waterfill(a, M, constraint_indices):
    """
    Solve: max sum(b) s.t. (a*b)_k <= M for k in constraint_indices, b >= 0
    Returns optimal b or None if infeasible.
    """
    n = len(a)
    A_mat = build_lp_matrix(a, constraint_indices)
    
    # Objective: maximize sum(b) → minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Bounds
    b_upper = M * n * 2.0  # generous upper bound
    bounds = [(0.0, b_upper)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=np.full(len(constraint_indices), M),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if res.success:
            return np.clip(res.x, 0, None)
        else:
            # Try with relaxed constraints
            res = linprog(c_obj, A_ub=A_mat, b_ub=np.full(len(constraint_indices), M * 1.01),
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 30.0})
            if res.success:
                return np.clip(res.x, 0, None)
            return None
    except Exception:
        return None


def lp_iteration(a, top_k=300, line_search_points=15):
    """
    One LP water-filling iteration.
    Returns (new_a, old_score, new_score).
    """
    n = len(a)
    c = fast_conv(a)
    M = float(np.max(c))
    S = float(a.sum())
    score_old = 2.0 * n * M / (S * S)
    
    # Select constraint indices: top K peaks + some spread
    sorted_idx = np.argsort(c)[::-1]
    top_idx = sorted_idx[:min(top_k, 2 * n - 1)]
    
    # Add some evenly spread indices for coverage
    if 2 * n - 1 > top_k + 50:
        extra = np.linspace(0, 2 * n - 2, 50).astype(int)
        constraint_indices = np.unique(np.concatenate([top_idx, extra]))
    else:
        constraint_indices = top_idx
    
    # Solve LP
    b = solve_lp_waterfill(a, M, constraint_indices)
    if b is None or b.sum() < 1e-10:
        return a.copy(), score_old, score_old
    
    # Normalize b to same sum as a
    b_norm = b * S / float(b.sum())
    
    # Line search
    best_a = a.copy()
    best_sc = score_old
    for t_val in np.linspace(0.1, 0.6, line_search_points):
        a_trial = (1.0 - t_val) * a + t_val * b_norm
        a_trial = np.clip(a_trial, 0.0, 1000.0)
        sc = compute_score(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
    
    return best_a, score_old, best_sc


def peak_shave_adaptive(a, iters=3000):
    """Peak shaving with backtracking line search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    step_scale = 0.5
    
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Backtracking line search
        step = step_scale / (gn * S + 1e-20)
        improved = False
        for _ in range(10):
            a_trial = a - step * grad
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            sc_trial = compute_score(a_trial)
            if sc_trial < sc * (1.0 - 1e-10):
                a = a_trial
                step_scale = min(step_scale * 1.05, 5.0)
                improved = True
                break
            else:
                step *= 0.5
        if not improved:
            step_scale = max(step_scale * 0.8, 0.05)
    
    return best_sc, best_a
