Optimized peak autocorrelation minimizer using projected subgradient + L-BFGS continuation.
Combines exact subgradient descent with smooth LSE polishing for non-negative sequences.

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    # Let me try: ifft(conj(FFT(a)) * FFT(p))[k] = sum_m a[m] * p[m+k] (for appropriate k)
    # Yes! This is what we want (for the non-wrapping case, with N large enough).
    
    M_fft = 1 << (3 * n - 2).bit_length()  # large enough to avoid wrapping
    FA = np.fft.rfft(a, M_fft)
    FP = np.fft.rfft(p, M_fft)
    h_full = np.fft.irfft(np.conj(FA) * FP, M_fft)
    h = h_full[:n]  # h[j] = sum_m a[m] * p[m+j]
    
    # So dC[k]/da[j] = 2 * a[k-j] (if 0 <= k-j < n), 0 otherwise
    # dL/da[j] = sum_k p[k] * dC[k]/da[j] = 2 * sum_k p[k] * a[k-j] (if valid)
    # = 2 * h'[j] where h'[j] = sum_k p[k] * a[k-j]
    # Note: h'[j] = sum_k p[k] * a[k-j] = sum_m p[m+j] * a[m] (letting m = k-j) = h[j]
    # Wait, h[j] = sum_m a[m] * p[m+j] which is the same as sum_k p[k] * a[k-j] (with k = m+j)
    # So dL/da[j] = 2 * h[j]
    
    grad_L = 2.0 * h
    
    # Now, score = 2n * L / S^2
    # d(score)/da[j] = 2n * (dL/da[j] * S^2 - L * 2*S * 1) / S^4
    # = 2n * (2*h[j] * S - 2*L) / S^3
    # = 4n * (h[j] * S - L) / S^3
    
    S2 = S * S
    S3 = S2 * S
    grad = (4.0 * n) * (h * S - L) / S3
    
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """L-BFGS optimization with bound constraints."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        return lse_score_grad(x, beta)
    
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def spectral_flatten(a0, N, n_iters=200, lam=0.5, rng=None):
    """Iterative spectral flattening."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if (N % 2 == 0 and k == half):
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a


def accelerated_subgradient(a0, n_iters=10000, rng=None):
    """Accelerated (FISTA-style) projected gradient on the exact peak objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    y = a.copy()
    S = a.sum()
    if S > 1e-12:
        a = a / S
        y = y / S
    else:
        a = np.ones(n) / n
        y = a.copy()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    t = 1.0
    lr_base = 2.0 / (4.0 * n)  # Lipschitz constant estimate for the gradient on simplex
    
    for it in range(n_iters):
        # Compute subgradient at y
        c = fast_conv(y)
        k_star = int(np.argmax(c))
        g = np.zeros(n)
        idx = k_star - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g[valid] = 2.0 * y[idx[valid]]
        
        # Gradient step
        lr = lr_base / (t + 1.0)
        a_new = y - lr * g
        
        # Project onto simplex (non-negative, sum = 1)
        a_new = project_simplex(a_new)
        
        # Update y for next iteration (FISTA)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        
        a = a_new
        t = t_new
        
        # Track best
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a


def diversity_restart(a_best, n, rng, strength):
    """Create a perturbed version of the best solution for restart."""
    strategies = rng.integers(5)
    if strategies == 0:
        # Gaussian perturbation
        a = a_best * (1.0 + strength * rng.standard_normal(n))
    elif strategies == 1:
        # Multiplicative chirp
        x = np.arange(n, dtype=np.float64)
        f = 1.0 + 5.0 * rng.random()
        a = a_best * (1.0 + strength * np.cos(np.pi * f * x * x / (2.0 * n)))
    elif strategies == 2:
        # Random mask
        mask = (rng.random(n) > 0.1).astype(float)
        a = a_best * mask
        if a.sum() < 1e-6:
            a = a_best.copy()
    elif strategies == 3:
        # Spectral perturbation
        N_fft = 1 << (2 * n - 1).bit_length()
        A = np.fft.fft(a_best, N_fft)
        noise = strength * (rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A)))
        A_noisy = A + noise * np.abs(A)
        a = np.real(np.fft.ifft(A_noisy))[:n]
    else:
        # Block perturbation
        n_blocks = int(2 ** rng.integers(3, 8))
        block_size = max(1, n // n_blocks)
        scales = 1.0 + strength * rng.standard_normal(n_blocks)
        a = np.repeat(scales, block_size * n // n_blocks)[:n] * a_best[:n]
        if len(a) < n:
            a = np.concatenate([a, a_best[len(a):]])
    return np.maximum(a, 0.0)

"""
Hybrid optimizer for minimum peak autocorrelation of non-negative sequences.
Uses accelerated projected subgradient (FISTA) on the exact peak objective,
L-BFGS with adaptive LSE smoothing continuation, spectral flattening inits,
and diversity restarts. Targets score below 1.5027.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full linear autocorrelation via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def peak_subgradient(y, n):
    """Compute subgradient of 2n*max(C) at y (on simplex, so sum=1).
    Returns (gradient, k_star)."""
    c = fast_conv(y)
    k_star = int(np.argmax(c))
    g = np.zeros(n)
    idx = k_star - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    g[valid] = 2.0 * y[idx[valid]]
    return g, k_star


def fista_subgradient(a0, n_iters, lr_init, rng=None):
    """FISTA-accelerated projected subgradient on simplex for peak autocorrelation."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S > 1e-12:
        a = a / S
    else:
        a = np.ones(n) / n
    y = a.copy()

    best_a = a.copy()
    best_sc = compute_score(a)
    t = 1.0
    lr = lr_init
    stagnant = 0

    for it in range(n_iters):
        g, k_star = peak_subgradient(y, n)
        # Gradient step with backtracking
        a_new = y - lr * g
        a_new = project_simplex(a_new)
        sc_new = compute_score(a_new)

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 100:
                lr *= 0.95
                stagnant = 0

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        a = a_new
        t = t_new

        # Periodic re-evaluation of step size
        if it % 500 == 499:
            sc_cur = compute_score(y)
            if sc_cur < best_sc * 1.001:
                lr = min(lr * 1.02, 1.0 / (2.0 * n))

    return best_sc, best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    exp_scaled = np.exp(scaled - scaled.max())
    log_sum = scaled.max() + np.log(exp_scaled.sum())
    L = c_max + (log_sum - scaled.max()) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    # Gradient: dL/da[j] = 2 * sum_k p[k] * a[k-j] (valid indices)
    # h[j] = sum_m a[m] * p[m+j]
    M_fft = 1 << (3 * n - 2).bit_length()
    FA = np.fft.rfft(a, M_fft)
    FP = np.fft.rfft(p, M_fft)
    h = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]

    # d(score)/da[j] = 4n * (h[j]*S - L) / S^3
    S2 = S * S
    S3 = S2 * S
    grad = (4.0 * n) * (h * S - L) / S3

    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun):
    """L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 20}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def spectral_flatten(a0, N, n_iters, lam):
    """Iterative spectral flattening for good initializations."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    original_sum = a.sum()
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam_eff = lam * (1.0 - 0.5 * progress)
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if (N % 2 == 0 and k == half):
                continue
            new_mag[k] = (1.0 - lam_eff) * mag[k] + lam_eff * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (original_sum / S)
    return a
