    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    
    sum_a2 = float(np.sum(a**2))

        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

        # L = beta * n * sum_a2 + 1e-10

def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)

"""
Multi-phase optimizer for flat-autocorrelation non-negative sequences.
Key innovation: First-order FISTA solution of the LP subproblem in the
alternating method (from state-of-the-art), combined with LSE-smoothed
L-BFGS polish and adaptive multi-length exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def self_conv(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    # Gradient of L w.r.t. a: 2 * cross-correlation(a, p)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def lp_solve_fista(a, M, n_fista_iters=3000, beta_start=5.0, beta_end=50.0):
    """
    Solve: max sum(x) s.t. conv(a,x)[k] <= M for all k, x >= 0
    using FISTA on the LSE-smoothed surrogate with adaptive beta.
    Returns x (unnormalized).
    """
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    for it in range(n_fista_iters):
        # Adaptive beta: linear ramp then hold
        frac = min(1.0, it / (n_fista_iters * 0.3))
        beta = beta_start + (beta_end - beta_start) * frac

        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L

        # Compute C @ x = conv(a, x)
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        # C^T @ p = a * sliding_window_sum(p)
        # sliding_window_sum(p)[j] = sum_{k=j}^{j+n-1} p[k]
        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0

        # FISTA update
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def lp_alternating(a, max_outer_iters=200, fista_iters=3000, t_start=0.4, t_min=0.05,
                   time_limit=None, t_check_interval=5):
    """
    LP-based alternating method.
    Given a (normalized to sum=1), iteratively:
    1. Compute M = max(conv(a,a))
    2. Solve LP: max sum(x) s.t. conv(a,x) <= M, x >= 0
    3. Rescale x to sum=1
    4. Update a = (1-t)*a + t*x
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve_count = 0

    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        M = float(self_conv(a).max())
        x = lp_solve_fista(a, M, n_fista_iters=fista_iters)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        # Adaptive mixing parameter
        t = max(t_min, t_start * (1.0 - 0.3 * outer / max(1, max_outer_iters)))

        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= 15:
                # Reduce step and reset counter
                t = t * 0.5
                if t < 0.01:
                    break
                no_improve_count = 0

        a = a_new

    return best_a, best_sc


def error_reduction(a, n_iters=200, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.5 + 0.5 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def subgradient_polish(a, total_iters=3000):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(total_iters):
        c = self_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    # Low-pass filtered noise (smooth)
    noise = rng.standard_normal(n)
    from scipy.ndimage import uniform_filter1d
    smooth = uniform_filter1d(noise, size=max(3, n // 50))
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 1000.0))
    # Multi-frequency
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Primary method: LP-based alternating with FISTA sub-solver.
    Secondary: LSE-smoothed L-BFGS, error reduction, subgradient polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 20.0

    # Try to load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, len={0 if parent is None else len(parent)}")

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [4096, 6962, 8192, 10000]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:4]

    print(f"Lengths to explore: {lengths}")

    # ==================================================================
    # Phase 1: Quick global search with error reduction (15% time)
    # ==================================================================
    phase1_deadline = t_start + 0.15 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 1: Global Search (Error Reduction) ---")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        elapsed = time.monotonic() - t_start
        remaining = phase1_deadline - t_start
        if remaining < 30:
            break
        print(f"  n={n}, remaining={remaining:.0f}s")
        inits = build_initializations(n, rng, parent)
        max_inits = max(2, int(remaining / 10.0))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_er = error_reduction(a0, n_iters=100, alpha=0.7)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} init={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LP-based alternating method (50% time) - MAIN PHASE
    # ==================================================================
    phase2_deadline = t_start + 0.65 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 2: LP-Based Alternating ---")

    # Focus primarily on the best length found so far
    n_main = len(best_a)
    if parent is not None and len(parent) in lengths:
        n_main = len(parent)

    # Run LP alternating on the main length
    a_work = best_a if len(best_a) == n_main else np.interp(
        np.linspace(0, 1, n_main), np.linspace(0, 1, len(best_a)), best_a
    )
    a_work = np.maximum(a_work, 0.0)
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Progressively refine with LP alternating
    fista_budgets = [2000, 3000, 4000, 5000]
    t_schedules = [(0.5, 0.1), (0.4, 0.08), (0.3, 0.05), (0.2, 0.03)]

    for fi, (fista_it, (t_s, t_m)) in enumerate(zip(fista_budgets, t_schedules)):
        if time.monotonic() > phase2_deadline:
            break
        elapsed = time.monotonic() - t_start
        remaining = phase2_deadline - t_start
        max_outer = max(5, int(remaining / max(5, fista_it * n_main / 500000.0)))
        print(f"  LP round {fi}: n={n_main}, fista_iters={fista_it}, t=[{t_s},{t_m}], max_outer~{max_outer}")

        a_out, sc_out = lp_alternating(
            a_work,
            max_outer_iters=max_outer,
            fista_iters=fista_it,
            t_start=t_s,
            t_min=t_m,
            time_limit=phase2_deadline - 5
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{fi}")
        a_work = a_out.copy()

    # Try LP alternating on other lengths if time permits
    for n in lengths:
        if n == n_main:
            continue
        if time.monotonic() > phase2_deadline - 60:
            break
        print(f"  LP alt length: n={n}")
        a_test = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        S = float(a_test.sum())
        if S > 1e-12:
            a_test = a_test / S
        a_out, sc_out = lp_alternating(
            a_test, max_outer_iters=30, fista_iters=2000,
            t_start=0.4, t_min=0.1, time_limit=phase2_deadline - 10
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: L-BFGS refinement (20% time)
    # ==================================================================
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 3: L-BFGS Refinement ---")

    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    beta_schedule = [30, 80, 200, 500, 1000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline:
            break
        sc, a_opt, _ = run_lbfgs(a_norm, beta, maxiter=6000, maxfun=200000)
        report(sc, a_opt, f"LBFGS beta={beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Local search + polish (10% time)
    # ==================================================================
    print(f"\n--- Phase 4: Local Search & Polish ---")

    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    # Try several perturbation strategies
    n_trials = 8
    for trial in range(n_trials):
        if time.monotonic() > deadline - 60:
            break
        strategy = trial % 4
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.01 + 0.03 * rng.random()
            a_pert = a_norm * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted: reduce peak-contributing elements
            c = self_conv(a_norm)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = a_norm.copy()
            seg = a_pert[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = max(5, n_best // 200)
            top_local = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a_pert[gi] * 0.03
                a_pert[gi] = max(a_pert[gi] - red, 0.0)
                total_red += red
            if total_red > 0:
                bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
                per = total_red / len(bottom_local)
                for li in bottom_local:
                    a_pert[j_start + li] += per
        elif strategy == 2:
            # Error reduction restart with perturbation
            a_pert = error_reduction(a_norm * (1 + 0.03 * rng.standard_normal(n_best)),
                                     n_iters=80, alpha=0.8)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Sinusoidal modulation
            freq = 1.0 + 6.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = a_norm * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        S_p = float(a_pert.sum())
        if S_p > 1e-12:
            a_pert = a_pert / S_p
            sc, a_opt, _ = run_lbfgs(a_pert, 200, maxiter=3000, maxfun=80000)
            report(sc, a_opt, f"local #{trial} s={strategy}")

    # Subgradient polish
    if time.monotonic() < deadline - 120:
        print("  Subgradient polish...")
        a_sp = subgradient_polish(best_a, total_iters=3000)
        sc = compute_score(a_sp)
        report(sc, a_sp, "subgrad")

    # Final LP alternating on the very best
    if time.monotonic() < deadline - 80:
        print("  Final LP polish...")
        a_final_in = best_a.copy()
        S = float(a_final_in.sum())
        if S > 1e-12:
            a_final_in = a_final_in / S
        a_final, sc_final = lp_alternating(
            a_final_in, max_outer_iters=20, fista_iters=5000,
            t_start=0.15, t_min=0.02, time_limit=deadline - 30
        )
        sc = compute_score(a_final)
        report(sc, a_final, "final LP")

    # Final L-BFGS with high beta
    if time.monotonic() < deadline - 30:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S
        sc, a_opt, _ = run_lbfgs(a_norm, 2000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final LBFGS beta=2000")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
