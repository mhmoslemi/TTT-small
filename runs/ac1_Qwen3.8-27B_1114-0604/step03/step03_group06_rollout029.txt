"""
Improved optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses power-mean smoothing (adaptive p), FFT-based exact gradients, multi-phase search with
basin hopping, restarts at different lengths, and targeted local refinement.
Key novel ideas: power-mean objective (cheaper than LSE for large p), structured spectral
perturbations for basin hopping, adaptive length exploration, and Pareto frontier tracking.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def power_mean_obj_grad(a, p):
    """
    Power-mean smoothed objective and its gradient.
    f(a) = 2n * PM_p(conv(a,a)) / (sum(a))^2
    where PM_p(c) = (mean(c^p))^(1/p).
    Returns (value, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    m = len(c)  # 2n-1
    c_safe = np.maximum(c, 1e-300)
    c_pow = c_safe ** p
    PM = (c_pow.mean()) ** (1.0 / p)
    if PM < 1e-300:
        return 1e10, np.zeros(n)
    score = 2.0 * n * PM / (S * S)

    # w_k = (c_k / PM)^(p-1) / m
    w = (c_safe / PM) ** (p - 1.0) / m

    # Compute v_j = sum_k w_k * a_{|k-j|} via FFT
    P = 1 << (3 * n - 2 - 1).bit_length()
    W_ft = np.fft.rfft(w, P)
    A_ft = np.fft.rfft(a, P)

    # correlate(w, a, 'full')[j] for j=0..n-1
    corr = np.fft.irfft(W_ft * np.conj(A_ft), P)[:n]
    # convolve(w, a)[j] for j=0..n-1
    conv_val = np.fft.irfft(W_ft * A_ft, P)[:n]
    # v = corr + conv - w[:n] * a[0]
    v = corr + conv_val - w[:n] * a[0]

    S2 = S * S
    S3 = S2 * S
    grad = 4.0 * n * v / S2 - 4.0 * n * PM / S3
    return score, grad


def lse_obj_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500, 500))
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p_weights = exp_scaled / exp_scaled.sum()

    # Gradient via FFT
    P = 1 << (3 * n - 2 - 1).bit_length()
    PW_ft = np.fft.rfft(p_weights, P)
    A_ft = np.fft.rfft(a, P)
    corr = np.fft.irfft(PW_ft * np.conj(A_ft), P)[:n]
    conv_val = np.fft.irfft(PW_ft * A_ft, P)[:n]
    v = corr + conv_val - p_weights[:n] * a[0]

    S2 = S * S
    S3 = S2 * S
    grad = 4.0 * n * v / S2 - 4.0 * n * L / S3
    return score, grad


def run_optimization(a0, objective, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B with given objective function."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def make_pm_obj(p_val):
    """Create a power-mean objective function with fixed p."""
    def obj(a):
        return power_mean_obj_grad(a, p_val)
    return obj


def make_lse_obj(beta_val):
    """Create an LSE objective function with fixed beta."""
    def obj(a):
        return lse_obj_grad(a, beta_val)
    return obj


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].astype(np.float64).copy())

    # Structured initializations
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    # U-shape (high at edges, low in middle)
    inits.append(0.1 + 0.9 * (1.0 - np.abs(2.0 * x - 1.0)) ** 2)
    inits.append(0.1 + 0.9 * (1.0 - np.abs(2.0 * x - 1.0)) ** 4)
    # Inverted U
    inits.append(0.1 + 0.9 * np.abs(2.0 * x - 1.0) ** 2)
    # Multi-frequency
    inits.append(0.5 + 0.2 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x) + 0.1 * np.cos(6 * np.pi * x))
    # Sinc-like
    inits.append(np.clip(0.5 + 0.3 * np.sinc(3.0 * (x - 0.5)), 0.01, 1000.0))
    # Gaussian-like bump
    inits.append(np.clip(0.2 + 0.8 * np.exp(-4.0 * (x - 0.5) ** 2), 0.01, 1000.0))
    # Double bump
    inits.append(np.clip(0.1 + 0.8 * np.exp(-8.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-8.0 * (x - 0.7) ** 2), 0.01, 1000.0))
    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.4 * rng.random(n), 0.01, 1000.0))
    # Low-freq random
    freqs = rng.random(5) * 10 + 1
    phases = rng.random(5) * 2 * np.pi
    amps = rng.random(5) * 0.2
    sig = np.zeros(n)
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2 * np.pi * f * x + ph)
    inits.append(np.clip(0.5 + sig, 0.01, 1000.0))
    # Block structure
    for nb in (8, 16, 32, 64):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    # Alternating
    a_alt = np.ones(n) * 0.5
    a_alt[0::2] = 0.8
    a_alt[1::2] = 0.2
    inits.append(a_alt)
    return inits


def targeted_perturb(a, rng, strength):
    """Perturb elements that contribute most to the peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:max(top_count, n // 4)]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def spectral_perturb(a, rng, n_freqs=5):
    """Perturb in frequency domain for global-ish moves."""
    n = len(a)
    A = np.fft.rfft(a)
    # Perturb low-frequency components more
    n_real = len(A)
    mask = np.ones(n_real)
    cutoff = max(2, n_real // 20)
    mask[:cutoff] = 0.5 + rng.random(cutoff)
    mask[cutoff:] = 0.1 + 0.3 * rng.random(n_real - cutoff)
    A_pert = A * (1.0 + 0.1 * mask * rng.standard_normal(n_real))
    a_pert = np.fft.irfft(A_pert, 2 * n_real - 1)[:n]
    # Re-normalize to keep similar sum
    if a_pert.sum() > 0:
        a_pert = a_pert * (a.sum() / a_pert.sum())
    return np.clip(a_pert, 0.0, 1000.0)


def basin_hopping(a, n_iters, max_wall_s, rng, p_smooth=20):
    """Basin hopping with spectral and targeted perturbations."""
    n = len(a)
    a_cur = a.astype(np.float64).copy()
    best_a = a_cur.copy()
    best_sc = compute_score(a_cur)
    cur_sc = best_sc
    obj = make_pm_obj(p_smooth)
    t0 = time.monotonic()
    for it in range(n_iters):
        if time.monotonic() - t0 > max_wall_s:
            break
        # Choose perturbation strategy
        r = rng.random()
        if r < 0.3:
            a_pert = spectral_perturb(a_cur, rng)
        elif r < 0.6:
            strength = 0.02 + 0.08 * rng.random()
            a_pert = targeted_perturb(a_cur, rng, strength)
        else:
            scale = 0.03 + 0.1 * rng.random()
            a_pert = a_cur * (1.0 + scale * rng.standard_normal(n))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Local optimization from perturbed point
        sc, a_opt = run_optimization(a_pert, obj, maxiter=1000, maxfun=20000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            cur_sc = sc
            a_cur = a_opt
        elif rng.random() < np.exp(-(sc - cur_sc) / (best_sc - 1.0 + 1e-6)):
            a_cur = a_opt
            cur_sc = sc
    return best_sc, best_a


def subgradient_polish(a, total_iters=300):
    """Exact subgradient descent on the true max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    step_scale = 1.0
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max(c) w.r.t. a: 2*a_{|k_star - j|} for |k_star-j|<n
        sg_M = np.zeros(n)
        for j in range(n):
            d = abs(k_star - j)
            if d < n:
                sg_M[j] = 2.0 * a[d]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30) * step_scale
        step = min(step, 0.1)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        # Decay step scale
        if it % 50 == 49:
            step_scale *= 0.7
    return best_sc, best_a

    # Phase 2: Basin hopping (25% of time)
    # =========================================================================
    phase2_end = t_start + 0.75 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: Basin Hopping (ends at {phase2_end - t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    for p_smooth in [16, 32, 64]:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        n_iters = max(10, int(remaining / 8.0))
        sc, a_opt = basin_hopping(best_a, n_iters, max_wall_s=remaining, rng=rng, p_smooth=p_smooth)
        report(sc, a_opt, f"basin p={p_smooth}")
        best_a = a_opt

    # Try basin hopping at different lengths
    for n_alt in lengths:
        if time.monotonic() > phase2_end:
            break
        if n_alt == n_best:
            continue
        # Interpolate parent to new length
        if parent is not None:
            idx = np.linspace(0, len(parent) - 1, n_alt).astype(int)
            a0 = parent[idx].copy()
        else:
            a0 = best_a.copy()
            if len(a0) != n_alt:
                idx = np.linspace(0, len(a0) - 1, n_alt).astype(int)
                a0 = a0[idx]
        remaining = phase2_end - time.monotonic()
        if remaining < 30:
            break
        sc, a_opt = basin_hopping(a0, max(5, int(remaining / 10.0)), max_wall_s=remaining * 0.8, rng=rng, p_smooth=32)
        report(sc, a_opt, f"basin n={n_alt}")

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # =========================================================================
    # Phase 3: Targeted local search (15% of time)
    # =========================================================================
    phase3_end = t_start + 0.90 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Targeted Local Search (ends at {phase3_end - t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    trial = 0
    obj_high_p = make_pm_obj(128)
    while time.monotonic() < phase3_end and trial < 80:
        strategy = trial % 8
        if strategy == 0:
            scale = 0.02 + 0.06 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.07 * rng.random())
        elif strategy == 2:
            a_pert = spectral_perturb(best_a, rng)
        elif strategy == 3:
            # Frequency-domain perturbation at random frequencies
            A = np.fft.rfft(best_a)
            nf = max(3, n_best // 100)
            idx = rng.integers(1, len(A), size=nf)
            A[idx] *= (1.0 + 0.2 * rng.standard_normal(nf))
            a_pert = np.fft.irfft(A, 2 * len(A) - 1)[:n_best]
            if a_pert.sum() > 0:
                a_pert *= best_a.sum() / a_pert.sum()
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 4:
            # Block swap
            block_sz = max(20, n_best // 15)
            b1 = int(rng.integers(0, n_best - 2 * block_sz))
            b2 = int(rng.integers(0, n_best - 2 * block_sz))
            a_pert = best_a.copy()
            a_pert[b1:b1 + block_sz], a_pert[b2:b2 + block_sz] = \
                a_pert[b2:b2 + block_sz].copy(), a_pert[b1:b1 + block_sz].copy()
        elif strategy == 5:
            # Sparse large perturbation
            n_sparse = max(20, n_best // 30)
            a_pert = best_a.copy()
            idx = rng.integers(0, n_best, size=n_sparse)
            a_pert[idx] *= (1.0 + 0.5 * rng.standard_normal(n_sparse))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 6:
            # Mix with structured sequence
            x = np.arange(n_best, dtype=np.float64) / n_best
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            a_struct = 0.5 + 0.3 * np.cos(2.0 * np.pi * freq * x + phase)
            mix = 0.2 + 0.3 * rng.random()
            a_pert = (1 - mix) * best_a + mix * a_struct * (best_a.sum() / a_struct.sum())
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Anti-peak: reduce where convolution is largest
            c = fast_conv(best_a)
            # Find top-K lag positions and reduce contributing elements
            top_lags = np.argsort(c)[::-1][:10]
            a_pert = best_a.copy()
            for k in top_lags:
                k = int(k)
                j_s = max(0, k - n_best + 1)
                j_e = min(k, n_best - 1)
                if j_e >= j_s:
                    for j in range(j_s, j_e + 1):
                        a_pert[j] *= 0.95
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            # Redistribute mass to low-convolution areas
            if a_pert.sum() < best_a.sum():
                deficit = best_a.sum() - a_pert.sum()
                a_pert += deficit / n_best
                a_pert = np.clip(a_pert, 0.0, 1000.0)

        sc, a_opt = run_optimization(a_pert, obj_high_p, maxiter=2000, maxfun=40000)
        report(sc, a_opt, f"local #{trial} (strat={strategy})")
        trial += 1

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # =========================================================================
    # Phase 4: Final refinement (10% of time)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Refinement")
    print(f"{'='*60}")

    # High-p power mean polish
    for p_final in [256, 512]:
        if time.monotonic() > deadline - 10.0:
            break
        obj = make_pm_obj(p_final)
        sc, a_opt = run_optimization(best_a, obj, maxiter=5000, maxfun=100000)
        report(sc, a_opt, f"final p={p_final}")
        best_a = a_opt

    # Subgradient polish
    if time.monotonic() < deadline - 8.0:
        sc, a_pol = subgradient_polish(best_a, total_iters=500)
        report(sc, a_pol, "subgrad polish")
        if sc < best_score:
            best_a = a_pol

    # One more L-BFGS from the polished solution
    if time.monotonic() < deadline - 5.0:
        obj = make_pm_obj(512)
        sc, a_opt = run_optimization(best_a, obj, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final final")

    # =========================================================================
    # Final verification and return
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
