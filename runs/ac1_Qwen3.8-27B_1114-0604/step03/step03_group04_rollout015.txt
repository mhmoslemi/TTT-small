"""

peak self-convolution of non-negative sequences. Combines spectral flattening
for initialization, high-beta LSE continuation via L-BFGS-B, adaptive multi-peak

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta, top_k=1):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_shifted / exp_shifted.sum()
    N_fft = 1 << (3 * n - 2).bit_length()
    rev_a = a[::-1]
    RA = np.fft.rfft(rev_a, N_fft)
    P = np.fft.rfft(p, N_fft)
    conv_res = np.fft.irfft(RA * P, N_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def multi_peak_score_grad(a, beta, top_k=5):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_top = c[sorted_idx]
    c_max_top = float(c_top.max())
    shifted = beta * (c_top - c_max_top)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    L = c_max_top + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p_top = exp_shifted / exp_shifted.sum()
    p = np.zeros(len(c))
    p[sorted_idx] = p_top
    N_fft = 1 << (3 * n - 2).bit_length()
    rev_a = a[::-1]
    RA = np.fft.rfft(rev_a, N_fft)

    conv_res = np.fft.irfft(RA * P, N_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)

def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000, top_k=1, use_multi_peak=False):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        if use_multi_peak:
            return multi_peak_score_grad(x, beta, top_k=top_k)
        else:
            return lse_score_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def spectral_flatten_v2(a0, N_fft, n_iters=300, lam_start=0.8, lam_end=0.3,
                        use_phase_opt=False, rng=None):
    a = a0.astype(np.float64).copy()
    n = len(a)
    a_sum_orig = a.sum()
    for i in range(n_iters):
        t = i / max(1, n_iters - 1)
        lam = lam_start * (1 - t) + lam_end * t
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc = mag[1:]
        half = N_fft // 2
        non_dc_mirror = non_dc[:half] if N_fft % 2 == 0 else non_dc[:half]
        target = np.median(non_dc_mirror)
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        for k in range(1, half + 1):
            if k == half and N_fft % 2 == 0:
                new_mag[k] = mag[k]
            else:
                new_mag[k] = (1.0 - lam) * mag[k] + lam * target
                new_mag[N_fft - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a_sum_orig / S)
    return a


def spectral_flatten_aggressive(a0, N_fft, n_iters=500):

    n = len(a)
    a_sum_orig = a.sum()
    for i in range(n_iters):

        lam = 0.7 * (1 - t) + 0.3 * t
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        half = N_fft // 2
        non_dc = mag[1:half + 1]
        target = np.mean(non_dc)
        spread = np.std(non_dc)
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        for k in range(1, half + 1):
            val = (1.0 - lam) * mag[k] + lam * target
            val = max(val, target - 0.5 * spread * (1 - t))
            new_mag[k] = val
            new_mag[N_fft - k] = val
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a_sum_orig / S)
    return a


def peak_cutting(a, n_peaks=10, strength=0.05, n_iter=50):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iter):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        contrib = np.zeros(n)
        for k in peak_idx:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                contrib[j_start:j_end + 1] += a[j_start:j_end + 1][::-1] * (c[k] / M)
        top_count = max(5, n // 100)
        top_idx = np.argsort(contrib)[::-1][:top_count]
        bottom_count = max(5, n // 100)
        bottom_idx = np.argsort(contrib)[:bottom_count]
        total_reduce = 0.0
        for j in top_idx:
            reduce_amt = a[j] * strength
            a[j] -= reduce_amt
            a[j] = max(a[j], 0.0)
            total_reduce += reduce_amt
        if total_reduce > 0 and len(bottom_idx) > 0:
            per = total_reduce / len(bottom_idx)
            for j in bottom_idx:
                a[j] += per
        a = np.clip(a, 0.0, 1000.0)
        strength *= 0.95
    return best_sc, best_a


def subgrad_polish(a, total_iters=500):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        lr = 0.1 / (1.0 + 0.05 * it)
        step = lr / (gn + 1e-30)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a

    return np.clip(upsampled, 0.0, 1000.0)

    print(f"\n--- Phase 1: Initialization (by t={phase1_deadline - t_start:.0f}s) ---")

    all_cands = []
    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        t0 = time.monotonic()
        inits = build_inits_for_length(n, rng, parent)
        print(f"  n={n}: generated {len(inits)} inits ({time.monotonic()-t0:.1f}s)")
        scored = []
        for a in inits:
            sc = compute_score(a)
            scored.append((sc, a))
        scored.sort(key=lambda x: x[0])
        top_inits = scored[:min(6, len(scored))]
        for sc, a in top_inits:
            all_cands.append((sc, a))
            report(sc, a, f"init n={n}")
        print(f"    best init: {top_inits[0][0]:.6f}")

    all_cands.sort(key=lambda x: x[0])
    print(f"Total candidates: {len(all_cands)}, best={all_cands[0][0]:.6f}")

    # Phase 2: LSE continuation with increasing beta
    phase2_deadline = t_start + 0.65 * effective_budget
    print(f"\n--- Phase 2: LSE L-BFGS (by t={phase2_deadline - t_start:.0f}s) ---")

    beta_schedule = [50, 200, 1000, 5000, 15000]
    top_k_schedule = [1, 1, 3, 3, 5]

    for ci, (sc_init, a_init) in enumerate(all_cands[:8]):
        if time.monotonic() > phase2_deadline:
            break
        n = len(a_init)
        t0 = time.monotonic()
        a_cur = a_init.copy()
        for bi, beta in enumerate(beta_schedule):
            if time.monotonic() > phase2_deadline:
                break
            remaining = phase2_deadline - time.monotonic()
            if remaining < 10:
                break
            tk = top_k_schedule[bi]
            use_mp = bi >= 2 and tk > 1
            est_per_iter = 0.003 * (n / 4096) + 0.002
            maxiter = min(3000, max(200, int(remaining * 0.7 / max(est_per_iter * 15, 0.05))))
            maxfun = min(50000, max(1000, maxiter * 25))
            sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun,
                                       top_k=tk, use_multi_peak=use_mp)
            a_cur = a_opt
            report(sc, a_opt, f"LSE n={n} ci={ci} b={beta} tk={tk}")
        sc_final = compute_score(a_cur)
        report(sc_final, a_cur, f"LSE_done n={n} ci={ci}")
        print(f"  n={n} ci={ci}: {sc_final:.6f} ({time.monotonic()-t0:.1f}s)")

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # Phase 3: Peak cutting + re-optimization cycle
    phase3_deadline = t_start + 0.85 * effective_budget
    print(f"\n--- Phase 3: Peak Cutting (by t={phase3_deadline - t_start:.0f}s) ---")

    if best_a is not None:
        for cycle in range(5):
            if time.monotonic() > phase3_deadline:
                break
            t0 = time.monotonic()
            sc_pc, a_pc = peak_cutting(best_a, n_peaks=10, strength=0.03, n_iter=100)
            report(sc_pc, a_pc, f"peak_cut {cycle}")
            # Re-optimize after peak cutting
            remaining = phase3_deadline - time.monotonic()
            if remaining > 20:
                n = len(a_pc)
                maxiter = min(2000, max(500, int(remaining * 0.5 / 0.02)))
                sc, a_opt, _ = run_lbfgs(a_pc, 10000, maxiter=maxiter, maxfun=50000,
                                          top_k=3, use_multi_peak=True)
                report(sc, a_opt, f"reopt after pc {cycle}")
                best_a_for_pc = a_opt
            print(f"  cycle {cycle}: {best_score:.6f} ({time.monotonic()-t0:.1f}s)")

    # Phase 4: Diverse local search
    phase4_deadline = t_start + 0.95 * effective_budget
    print(f"\n--- Phase 4: Local Search (by t={phase4_deadline - t_start:.0f}s) ---")

    if best_a is not None:
        n_best = len(best_a)
        n_trials = 30
        for trial in range(n_trials):
            if time.monotonic() > phase4_deadline:
                break
            strat = trial % 6
            if strat == 0:
                scale = 0.01 + 0.03 * rng.random()
                a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            elif strat == 1:
                a_pert = peak_cutting(best_a, n_peaks=5, strength=0.02, n_iter=50)[1]
            elif strat == 2:
                freq = 1.0 + 10.0 * rng.random()
                phase = 2.0 * np.pi * rng.random()
                amp = 0.02 + 0.03 * rng.random()
                x = np.arange(n_best) / n_best
                a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            elif strat == 3:
                N_fft = 1 << (2 * n_best - 1).bit_length()
                A = np.fft.fft(best_a, N_fft)
                noise = 0.01 * rng.standard_normal(len(A)) + 0.01j * rng.standard_normal(len(A))
                A_noisy = A * (1.0 + noise)
                A_noisy[0] = A[0]
                a_pert = np.real(np.fft.ifft(A_noisy))[:n_best]
            elif strat == 4:
                a_pert = best_a.copy()
                c = fast_conv(a_pert)
                peak_pos = np.argsort(c)[::-1][:5]
                for k in peak_pos:
                    j_s = max(0, k - n_best + 1)
                    j_e = min(k, n_best - 1)
                    if j_e >= j_s:
                        seg = a_pert[j_s:j_e + 1]
                        a_pert[j_s:j_e + 1] = seg * 0.9
            else:
                x = np.arange(n_best, dtype=np.float64)
                f = 1.0 + 5.0 * rng.random()
                pert = 0.02 * np.cos(np.pi * f * x * x / (2.0 * n_best))
                a_pert = best_a * (1.0 + pert)

            a_pert = np.clip(a_pert, 0.0, 1000.0)
            remaining = phase4_deadline - time.monotonic()
            if remaining < 5:
                break
            maxiter = min(2000, max(300, int(remaining * 0.3 / 0.02)))
            sc, a_opt, _ = run_lbfgs(a_pert, 5000, maxiter=maxiter, maxfun=30000,
                                      top_k=3, use_multi_peak=True)
            report(sc, a_opt, f"local #{trial} s={strat}")

    # Phase 5: Final high-beta polish
    print(f"\n--- Phase 5: Final Polish ---")
    if best_a is not None:
        sc_pg, a_pg = subgrad_polish(best_a, total_iters=300)
        report(sc_pg, a_pg, "subgrad")

        for beta_final in [20000, 50000]:
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            maxiter = min(5000, max(1000, int(remaining * 0.7 / 0.03)))
            sc, a_opt, _ = run_lbfgs(best_a, beta_final, maxiter=maxiter, maxfun=100000,
                                      top_k=3, use_multi_peak=True)
            report(sc, a_opt, f"final_beta={beta_final}")

        sc_pg2, a_pg2 = subgrad_polish(best_a, total_iters=200)
        report(sc_pg2, a_pg2, "final_subgrad2")

    # Final
    if best_a is None:
        out = [0.5] * 128
    else:
        out = [float(max(0.0, min(1000.0, x))) for x in best_a]
        if sum(out) < 0.01:
            out = [0.5] * 128

    verified = evaluate_sequence(out)
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
