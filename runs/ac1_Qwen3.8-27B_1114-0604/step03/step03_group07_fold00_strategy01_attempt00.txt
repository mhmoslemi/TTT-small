def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    if not isinstance(sequence, list):
        return np.inf
    if len(sequence) == 0:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0.0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    conv_result = np.convolve(sequence, sequence, mode="full")
    sum_val = float(sum(sequence))
    if sum_val < 0.01:
        return np.inf
    max_val = float(max(conv_result))
    return float(2.0 * n * max_val / (sum_val ** 2))


def full_convolution(a):
    """Convolution with mode='full' and handles all edge cases."""
    n = len(a)
    if n == 0:
        return np.zeros(2*n - 1) if a == [] else np.array([])
    conv = np.convolve(a, a, mode="full")
    return conv

    """Smoothed LSE score and gradient computation."""
    if not a or len(a) == 0:
        return float('inf')
    n_val = len(a)

    if sum_a < 0.01:
        return float('inf')

        return np.inf, np.array([])
    
    # Smoothing: use linear scaling and log-sum-exp for the max part
