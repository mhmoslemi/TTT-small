For the LP step, the problem is:
maximize sum(b)
subject to:
  b ≥ 0
  (a * b)[k] ≤ M for all k = 0, ..., 2n-2

where M = max(conv(a,a)) and a is the current solution.

This LP has n variables and 2n-1 + n constraints (the convolution constraints plus non-negativity). For n ≈ 7000, that's about 21000 constraints and 7000 variables. This should be solvable in a few seconds with scipy.optimize.linprog (using HiGHS solver if available) or cvxpy.

The constraint (a*b)[k] ≤ M can be written as:
sum_j a[j] * b[k-j] ≤ M for all k

This is a system of linear inequalities. The matrix is a "convolution matrix" (Toeplitz-like). We can use the special structure.

Actually, for scipy.optimize.linprog, we just need to provide the constraint matrix. For n=7000, the matrix is (2n-1) × n ≈ 13999 × 7000 ≈ 100 million entries. That's a lot of memory (about 800 MB as float64). This might be too much.

Alternative: use the LP in a different form. Since the constraint matrix is structured (it's a convolution), we can use a custom solver or exploit the structure.

Actually, a simpler approach: instead of solving the full LP, we can use a "greedy" or "iterative" approach:
1. Start with b = 0.
2. Find the coordinate j that gives the largest increase in sum(b) while maintaining feasibility.
3. Increase b[j] until some constraint becomes tight.
4. Repeat.

This is essentially the "least angle regression" or "simplex-like" approach. It's O(n²) in the worst case but often much faster in practice.

Alternatively, I can use a smaller LP by sacrificing some accuracy. For example, only include the constraints at the top K lags (where K = 100 or so). This gives a much smaller LP that's still a good approximation.

Let me go with the truncated LP approach: include only the top K convolution lags as constraints. This gives a K × n LP which is much more manageable.

"""
Optimized step-function search: FISTA + Truncated LP + Flat-Conv Init + Coordinate Descent.
Builds on parent solution with accelerated proximal gradient methods and LP-based direction finding.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_ab(a, b):
    """Linear cross-convolution a*b via FFT."""
    na, nb = len(a), len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient. Returns (score, grad)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient of L w.r.t. a: convolve(p, a) shifted
    # dL/da[j] = 2 * (a*p)[j + n - 1] for j in [0, n-1]
    rev_p = p[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    AP = np.fft.rfft(a, M2_fft)
    RP = np.fft.rfft(rev_p, M2_fft)
    conv_res = np.fft.irfft(AP * RP, M2_fft)[:M2]
    # conv(a, rev_p)[j + n - 1] for j in 0..n-1
    g_conv = conv_res[n - 1:2 * n - 1]
    g = 2.0 * g_conv
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def estimate_lipschitz(a, beta):
    """Estimate Lipschitz constant via power iteration on the Hessian."""
    n = len(a)
    # Use a few power iterations
    v = np.random.default_rng(0).standard_normal(n)
    v = v / np.linalg.norm(v)
    for _ in range(10):
        # Approximate Hessian-vector product via finite differences of gradient
        eps = 1e-6 * np.linalg.norm(a)
        sc1, g1 = lse_score_grad(a + eps * v, beta)
        sc2, g2 = lse_score_grad(a - eps * v, beta)
        Hv = (g1 - g2) / (2.0 * eps)
        Hv = np.maximum(Hv, 0)  # Hessian should be PSD for convex part
        v_new = Hv / (np.linalg.norm(Hv) + 1e-30)
        ratio = float(v @ Hv) / (float(v @ v) + 1e-30)
        v = v_new
    return max(ratio * 1.5, 1.0)


def fista_optimize(a0, beta, n_iters=5000, step_scale=0.9):
    """FISTA (accelerated proximal gradient) for LSE-smoothed objective with non-negativity."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    L = estimate_lipschitz(a, beta)
    step = step_scale / (L + 1e-10)
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for k in range(n_iters):
        sc, grad = lse_score_grad(y, beta)
        x_new = y - step * grad
        x_new = np.maximum(x_new, 0.0)
        S_new = x_new.sum()
        if S_new < 1e-12:
            break
        x_new = x_new / S_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if k % 200 == 0:
            sc_exact = compute_score(a)
            if sc_exact < best_sc:
                best_sc = sc_exact
                best_a = a.copy()
            # Adapt step size
            L = max(L * 0.95, 1e-6)
            step = step_scale / (L + 1e-10)

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def flat_conv_init(n, S_target=1.0):
    """Generate sequence whose auto-convolution is approximately flat (first n lags)."""
    c = np.zeros(n)
    c[0] = 1.0 / np.sqrt(2.0 * n - 1.0)
    for m in range(1, n):
        cross_sum = np.dot(c[1:m], c[m-1:0:-1])
        c[m] = (1.0 / (2.0 * n - 1.0) - cross_sum) / (2.0 * c[0])
        if c[m] < 0:
            c[m] = 0.0
    S_c = c.sum()
    if S_c < 1e-12:
        return np.ones(n) / n
    a = c / S_c * S_target
    return a


def lp_direction(a, K=50):
    """
    Solve truncated LP to find direction b that maximizes sum(b)
    subject to (a*b)[k] <= M for top K lags, b >= 0.
    Returns normalized direction b.
    """
    n = len(a)
    conv_a = fast_conv(a)
    M = float(conv_a.max())
    S = float(a.sum())

    # Get top K lags by convolution value
    top_k = np.argsort(conv_a)[::-1][:K]

    # Build constraint matrix: A_lp * b <= M for each selected lag
    # (a*b)[k] = sum_j a[j] * b[k-j] for valid j
    A_lp = np.zeros((K, n))
    for i, k in enumerate(top_k):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_lp[i, j] = a[j]

    # Maximize sum(b) subject to A_lp * b <= M, b >= 0
    # scipy linprog minimizes, so we negate
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A_lp, b_ub=np.full(K, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': 30.0})

    if res.status == 0 and res.x is not None:
        b = np.maximum(res.x, 0.0)
        S_b = b.sum()
        if S_b > 1e-12:
            return b / S_b
    return None


def lp_based_update(a, K=50, blend=0.3):
    """LP-based direction finding and update."""
    b_dir = lp_direction(a, K=K)
    if b_dir is None:
        return a
    a_new = (1.0 - blend) * a + blend * b_dir
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new


def coordinate_descent_peaks(a, n_sweeps=5, n_top=100):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            best_local_sc = compute_score(a)
            best_val = a_old_j
            for frac in [0.5, 0.7, 0.8, 0.9, 1.2, 1.5, 0.3]:
                cv = a_old_j * frac
                a[j] = cv
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_polish(a, total_iters=3000):
    """Projected subgradient descent on the exact max-conv objective with averaging."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    # Polyak averaging
    avg_a = np.zeros(n)
    sc_lb = 1.0  # optimistic lower bound

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        avg_a += a
        sc_avg = compute_score(avg_a / (it + 1))
        if sc_avg < best_sc:
            best_sc = sc_avg
            best_a = (avg_a / (it + 1)).copy()

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(flat_conv_init(n))
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    return inits
