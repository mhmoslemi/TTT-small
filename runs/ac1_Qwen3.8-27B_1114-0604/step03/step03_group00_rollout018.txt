"""
Hybrid optimizer combining L2-gradient flattening, LSE peak reduction, and spectral shaping.
Key innovations over previous approach:
1. Novel L2-convolution gradient descent phase to flatten the autocorrelation efficiently
2. Spectral shaping via power spectrum equalization with non-negativity projection
3. Multi-scale LSE refinement with adaptive beta
4. Cross-correlation-based gradient (O(n log n)) for the L2 objective
5. Sequence extension strategy to leverage longer sequences

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    max_c = float(np.max(c))
    return 2.0 * n * max_c / (S * S)


def l2_gradient(a):
    """
    Gradient of sum_k c[k]^2 where c = conv(a,a).
    grad[j] = 4 * sum_{i=0}^{n-1} c[j+i] * a[i]
    Computed via FFT in O(n log n).
    """
    n = len(a)
    c = fast_conv(a)  # length 2n-1
    # We need: grad[j] = 4 * sum_{i=0}^{n-1} a[i] * c[j+i] for j=0..n-1
    # This is cross-correlation of a and c, evaluated at lags 0..n-1
    N_fft = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, N_fft)
    C = np.fft.rfft(c, N_fft)
    cross = np.fft.irfft(A * np.conj(C), N_fft)
    return 4.0 * cross[:n].copy()


def l2_gradient_descent(a, n_iters=2000, verbose=False):
    """
    Gradient descent on sum_k c[k]^2 subject to a>=0, sum(a)=S.
    Uses backtracking line search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S  # normalize to sum=1

    best_a = a.copy()
    best_obj = _l2_obj(a)

    # Initial step size
    grad = l2_gradient(a)
    gnorm = float(np.linalg.norm(grad))
    if gnorm < 1e-15:
        return a
    step = 1.0 / (gnorm + 1e-10)
    step = min(step, 1.0 / n * 10)

    for it in range(n_iters):
        grad = l2_gradient(a)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12:
            break

        # Backtracking line search
        alpha = step
        obj_cur = _l2_obj(a)
        for _ls in range(30):
            a_trial = a - alpha * grad
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-12:
                alpha *= 0.5
                continue
            a_trial = a_trial / s
            obj_trial = _l2_obj(a_trial)
            if obj_trial < obj_cur - 1e-4 * alpha * gnorm * gnorm:

        else:

        a = a - alpha * grad
        a = np.maximum(a, 0.0)

        a = a / s

def _l2_obj(a):
    """sum of squares of convolution."""
    c = fast_conv(a)
    return float(np.sum(c * c))


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    score = 2n * LSE(c, beta) / S^2
    where LSE(c, beta) = (1/beta) * log(sum(exp(beta * c)))
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum() + 1e-300)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    # Gradient: d(score)/da[j]
    # d(L)/da[j] = (1/beta) * sum_k exp(beta*(c[k]-c_max)) * dc[k]/da[j] / sum_k exp(beta*(c[k]-c_max))
    #             = sum_k p[k] * dc[k]/da[j]  where p[k] = exp(...)/sum
    # dc[k]/da[j] = 2*a[k-j] if 0<=k-j<n else 0
    # So d(L)/da[j] = 2 * sum_{i=0}^{n-1} p[j+i] * a[i]  (cross-correlation of a and p)

    p = exp_scaled / (exp_scaled.sum() + 1e-300)
    # p has length 2n-1, a has length n
    N_fft = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, N_fft)
    P = np.fft.rfft(p, N_fft)
    cross = np.fft.irfft(A * np.conj(P), N_fft)[:n]
    dL_da = 2.0 * cross

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_wrap(a, beta):
    """Wrapper for scipy.minimize (returns (f, grad))."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    a0 = a0 / a0.sum()  # normalize
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_wrap, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out[:] = 1.0 / n
    a_out = a_out / a_out.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def spectral_shaping(a, n_iters=100, t_start=0.1, t_end=0.5, verbose=False):
    """
    Spectral shaping: iteratively flatten the power spectrum while maintaining non-negativity.
    In frequency domain: modify |A|^2 to be flatter, then project back to time domain.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    N_fft = 1 << (2 * n - 1 - 1).bit_length()

    for it in range(n_iters):
        t = t_start + (t_end - t_start) * (it / max(1, n_iters - 1))
        A = np.fft.rfft(a, N_fft)
        P = np.abs(A) ** 2  # power spectrum, length N_fft//2 + 1

        # Flatten the power spectrum
        P_mean = P.mean()
        if P_mean < 1e-15:
            break
        # Soft flattening: P_new = P^(1-t) * P_mean^t
        P_new = np.power(np.maximum(P, 1e-30), 1.0 - t) * np.power(P_mean + 1e-30, t)

        # Reconstruct A with new magnitude, same phase
        phase_A = np.angle(A)
        mag_new = np.sqrt(P_new)
        A_new = mag_new * np.exp(1j * phase_A)

        # Go back to time domain
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = a_new / s

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new
        if verbose and it % 20 == 0:
            print(f"    SpectralShaping it={it}, t={t:.3f}, score={sc:.6f}")

    return best_a

    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    M = 1 << (2 * n - 1 - 1).bit_length()

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-14:
            break

        # Multiple peaks: use top-k peaks
        n_peaks = min(5, 2 * n - 1)
        top_k = np.argsort(conv)[::-1][:n_peaks]
        # Weighted excess
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-14:
            break
        conv_mod = conv - 0.7 * excess

        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.3 + 0.4 * (it / max(1, n_iters))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 50 == 0:
            print(f"    ER-v2 it={it}, score={sc:.6f}, max_c={conv_max:.8f}, mean_c={conv_mean:.8f}")

    return best_a


def multi_peak_coordinate_descent(a, n_sweeps=5, n_top_per_peak=30, verbose=False):
    """
    Coordinate descent targeting multiple peaks simultaneously.
    For each peak, identify top contributors and optimize them.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Find top peaks
        n_peaks = min(3, 2 * n - 1)
        peak_indices = np.argsort(c)[::-1][:n_peaks]

        for k_star in peak_indices:
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue

            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            total_contrib = float(np.sum(contrib) * float(seg).sum() / (len(seg) + 1e-10))

            top_count = min(n_top_per_peak, len(contrib))
            top_local = np.argsort(contrib)[::-1][:top_count]
            top_global = j_start + top_local

            for j in top_global:
                a_old = a[j]
                # Try reducing this element
                candidates = [a_old * 0.5, a_old * 0.7, a_old * 0.85, a_old * 0.0, a_old * 1.2, a_old * 1.5]
                best_val = a_old
                best_local_sc = compute_score(a)
                for cv in candidates:
                    cv = max(0.0, cv)
                    if abs(cv - a_old) < 1e-12:
                        continue
                    a[j] = cv
                    sc = compute_score(a)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_val = cv
                a[j] = best_val
                S = a.sum()
                if S > 1e-12:
                    a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if verbose:
            print(f"    MultiPeakCD sweep={sweep}, score={best_sc:.6f}")

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            from numpy import interp
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.maximum(interp(x_new, x_old, parent), 0.0).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.2 * np.sin(2.0 * np.pi * x) + 0.1 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x) * np.cos(4.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.15 * rng.standard_normal(n), 0.01, 100.0))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    # U-shaped
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    # W-shaped
    inits.append(0.3 + 0.5 * np.abs(2.0 * np.cos(2.0 * np.pi * x)))
    # Random blocks
    for nb in (16, 32, 64):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    return [np.maximum(a, 0.0) for a in inits]


def extend_sequence(a, extra_len, rng):
    """Try to extend a sequence by appending optimized tail."""
    n = len(a)
    new_n = n + extra_len
    # Initialize extension with small values
    a_ext = np.zeros(new_n)
    a_ext[:n] = a
    # Small random tail
    a_ext[n:] = 0.01 * rng.random(extra_len)
    a_ext = np.maximum(a_ext, 0.0)
    S = a_ext.sum()
    if S > 1e-12:
        a_ext = a_ext / S
    return a_ext


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses L2-gradient flattening, spectral shaping, LSE+L-BFGS, and local search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 20.0

    # Get parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 32:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n in [8192, 12288, 16384, 20480]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))
    # Limit to at most 4 lengths
    lengths = lengths[:4]

    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes' if parent is not None else 'no'}, "
          f"len={0 if parent is None else len(parent)}")

    # ============================================================
    # Phase 1: Generate candidates and quick evaluation (15% time)
    # ============================================================
    phase1_deadline = t_start + 0.15 * (float(budget_s) - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: Candidate Generation (deadline: {phase1_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    candidates = []  # list of (score, a, label)

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)
        # Limit inits
        remaining_frac = (phase1_deadline - time.monotonic()) / max(1e-6, phase1_deadline - t_start)
        max_inits = max(3, int(len(inits) * min(1.0, remaining_frac * 2)))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.maximum(a0, 0.0)
            S = a0.sum()
            if S < 1e-6:
                continue
            a0 = a0 / S
            sc = compute_score(a0)
            candidates.append((sc, a0, f"init n={n} i={i}"))
            if i == 0 and parent is not None and n == len(parent):
                report(sc, a0, "parent")

    # Sort by score
    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, best={candidates[0][0]:.6f}")

    # Take top few candidates
    top_candidates = candidates[:min(5, len(candidates))]

    # ============================================================
    # Phase 2: L2 gradient descent flattening (25% time)
    # ============================================================
    phase2_deadline = t_start + 0.40 * (float(budget_s) - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: L2 Gradient Flattening (deadline: {phase2_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    l2_results = []
    for sc_init, a_init, label in top_candidates[:3]:
        if time.monotonic() > phase2_deadline:
            break
        print(f"  L2-GradDesc on {label} (init score={sc_init:.6f})")
        a_flat = l2_gradient_descent(a_init, n_iters=1000)
        sc_flat = compute_score(a_flat)
        report(sc_flat, a_flat, f"L2 flatten {label}")
        l2_results.append((sc_flat, a_flat))

    l2_results.sort(key=lambda x: x[0])

    # ============================================================
    # Phase 3: Spectral shaping (10% time)
    # ============================================================
    phase3_deadline = t_start + 0.50 * (float(budget_s) - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Spectral Shaping (deadline: {phase3_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    if l2_results:
        best_l2_sc, best_l2_a = l2_results[0]
        if best_l2_sc < best_score:
            a_spec = spectral_shaping(best_l2_a, n_iters=100, t_start=0.05, t_end=0.4)
            sc_spec = compute_score(a_spec)
            report(sc_spec, a_spec, "spectral shaping")

    # ============================================================
    # Phase 4: Error reduction v2 + LSE/L-BFGS (30% time)
    # ============================================================
    phase4_deadline = t_start + 0.80 * (float(budget_s) - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 4: ER + LSE/L-BFGS (deadline: {phase4_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    # Error reduction from best
    if time.monotonic() < phase4_deadline:
        a_er = error_reduction_v2(best_a, n_iters=200, alpha=0.7)
        sc_er = compute_score(a_er)
        report(sc_er, a_er, "ER-v2")

    # LSE + L-BFGS with increasing beta
    beta_schedule = [20, 50, 100, 300, 800, 2000]
    time_per_beta = (phase4_deadline - time.monotonic()) / len(beta_schedule)

    for beta in beta_schedule:
        if time.monotonic() > phase4_deadline:
            break
        remaining = time.monotonic()
        maxiter = max(500, int(remaining * 200))
        maxfun = max(5000, int(remaining * 2000))
        print(f"  L-BFGS beta={beta} (maxiter={maxiter}, maxfun={maxfun})")
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS beta={beta}")

    # ============================================================
    # Phase 5: Local search with perturbations (10% time)
    # ============================================================
    phase5_deadline = t_start + 0.92 * (float(budget_s) - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 5: Local Search (deadline: {phase5_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    n_trials = 10
    for trial in range(n_trials):
        if time.monotonic() > phase5_deadline:
            break
        strategy = trial % 5
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Peak-targeted: reduce top contributors at the peak
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            seg = a_pert[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = max(5, n_best // 50)
            top_local = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a_pert[gi] * 0.08
                a_pert[gi] = max(a_pert[gi] - red, 0.0)
                total_red += red
            if total_red > 0:
                bottom_idx = np.argsort(a_pert)[:max(1, n_best // 10)]
                per = total_red / len(bottom_idx)
                for bi in bottom_idx:
                    a_pert[bi] += per
        elif strategy == 2:
            # Sinusoidal modulation
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.05 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Short error reduction from perturbed start
            a_pert = error_reduction_v2(
                best_a * (1 + 0.03 * rng.standard_normal(n_best)),
                n_iters=30, alpha=0.8)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Block rescaling
            a_pert = best_a.copy()
            n_blocks = 4 + int(rng.integers(0, 8))
            block_size = max(1, n_best // n_blocks)
            for b in range(min(n_blocks, n_best // block_size)):
                s = b * block_size
                e = min(s + block_size, n_best)
                factor = 0.6 + 0.8 * rng.random()
                a_pert[s:e] *= factor
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S

        # Quick L-BFGS refinement
        remaining = phase5_deadline - time.monotonic()
        if remaining < 10:
            break
        sc, a_opt, _ = run_lbfgs(a_pert, 200, maxiter=2000, maxfun=min(30000, int(remaining * 2000)))
        report(sc, a_opt, f"local #{trial} s={strategy}")

    # ============================================================
    # Phase 6: Final polish (8% time)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final Polish")
    print(f"{'='*60}")

    # High-beta L-BFGS
    if time.monotonic() < deadline - 30:
        sc, a_opt, _ = run_lbfgs(best_a, 5000, maxiter=3000, maxfun=50000)
        report(sc, a_opt, "final LBFGS beta=5000")

    # Multi-peak coordinate descent
    if time.monotonic() < deadline - 20:
        a_cd = multi_peak_coordinate_descent(best_a, n_sweeps=3, n_top_per_peak=40)
        sc_cd = compute_score(a_cd)
        report(sc_cd, a_cd, "multi-peak CD")
        if sc_cd < best_score:
            # One more L-BFGS from CD result
            if time.monotonic() < deadline - 10:
                sc, a_opt, _ = run_lbfgs(a_cd, 1000, maxiter=2000, maxfun=30000)
                report(sc, a_opt, "CD+LBFGS")

    # Final error reduction
    if time.monotonic() < deadline - 10:
        a_er_f = error_reduction_v2(best_a, n_iters=100, alpha=0.85)
        sc_er_f = compute_score(a_er_f)
        report(sc_er_f, a_er_f, "final ER")

    # Final L2 gradient step
    if time.monotonic() < deadline - 5:
        a_l2_f = l2_gradient_descent(best_a, n_iters=200)
        sc_l2_f = compute_score(a_l2_f)
        report(sc_l2_f, a_l2_f, "final L2-grad")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
