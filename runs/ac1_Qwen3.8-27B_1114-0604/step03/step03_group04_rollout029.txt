**The key degrees of freedom**: The shape of a (its "distribution") determines the shape of conv(a,a). 

**Optimal structure**: I believe the optimal sequence has a "bell-shaped" or "U-shaped" profile (not uniform). A U-shaped sequence (large at edges, small in middle) would have:
- Low center convolution (since middle values are small)
- But potentially high edge convolution (a[0]^2 + a[n-1]^2 could be large)

A bell-shaped sequence (large in middle, small at edges) would have:
- High center convolution (middle values are large)
- Low edge convolution

So neither extreme is obviously better. The optimal is somewhere in between.

So the question is: what is `min sum_j a[j] * a[n-1-j]` subject to `sum a[j] = S`, `a[j] >= 0`?

If we let b_j = a[n-1-j], then we're minimizing `sum_j a[j] * b_j` subject to `sum a = sum b = S`, `a, b >= 0`, and `b_j = a[n-1-j]`.

    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1 : 2*n-1]
    
    dLSE_da = 2.0 * conv_res
    
    # Full gradient of score = 2n*L/S^2
    # d(score)/da[i] = 2n/S^2 * dLSE_da[i] - 4n*L/S^3 * 1
    # (since d(S)/da[i] = 1)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n * dLSE_da) / S2 - (4.0 * n * L) / S3
    
    return score, grad

def spectral_flatten(a0, N, n_iters=300, lam_init=0.7, lam_final=0.3, rng=None):
    """Iterative spectral flattening with annealing."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    target_sum = a0.sum()
    
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam = lam_init + (lam_final - lam_init) * progress
        
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude (excluding DC)
        non_dc = mag[1:]
        if N % 2 == 0:
            non_dc = non_dc[:N//2]  # Only consider positive frequencies
        target = np.median(non_dc)
        
        new_mag = mag.copy()
        half = N // 2
        for k in range(1, (N // 2) + 1):
            if N % 2 == 0 and k == half:
                continue  # Nyquist stays
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_mag[0] = mag[0]  # DC unchanged
        if N % 2 == 0:
            new_mag[half] = mag[half]
        
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        
        S = a.sum()
        if S > 1e-12:
            a = a * (target_sum / S)
        
        # occasional random perturbation to escape local minima
        if rng is not None and i % 50 == 25:
            a = np.maximum(a + 0.01 * rng.standard_normal(n), 0.0)
            S = a.sum()
            if S > 1e-12:
                a = a * (target_sum / S)
    
    return a

def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out

def peak_shave(a, rng, n_steps=50, step_scale=0.01):
    """Novel peak shaving: directly reduce the max convolution value."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    S = a.sum()
    
    for step in range(n_steps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        M = float(c[k_star])
        
        # Find contributors to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        contrib = np.zeros(n)
        if j_end >= j_start:
            # contrib[j] = a[j] * a[k_star - j] for valid j
            for j in range(j_start, j_end + 1):
                contrib[j] = a[j] * a[k_star - j]
        
        # Find the top contributor
        top = int(np.argmax(contrib))
        if contrib[top] < 1e-10:
            break
        
        # Find a "safe" position to add mass
        # A position i is "safe" if it doesn't contribute to the peak
        safe_mask = (contrib == 0)
        if not safe_mask.any():
            # All positions contribute; find the one with least contribution
            safe_mask = contrib == contrib.min()
        
        # Choose a random safe position
        safe_indices = np.where(safe_mask)[0]
        if len(safe_indices) == 0:
            break
        safe_idx = rng.choice(safe_indices)
        
        # Move mass
        amt = a[top] * step_scale
        a[top] -= amt
        a[safe_idx] += amt
        a = np.clip(a, 0.0, 1000.0)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

"""
Hybrid spectral-flattening + L-BFGS + peak-shaving optimizer for minimum peak autocorrelation.
Implements: (1) diversified spectral flattening initialization, (2) L-BFGS with LSE-smoothed
objective and adaptive beta schedule, (3) novel iterative spectral re-projection between
L-BFGS phases to escape local minima, (4) direct peak-shaving local search, and (5) subgradient
polish. Works on non-negative real sequences to minimize 2n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(a, beta):
    """Compute LSE-smoothed objective value and its gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: dLSE/da[i] = 2 * sum_m p[i+m] * a[m]
    # Computed via: convolve(a[::-1], p)[n-1 : 2n-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a[::-1], M2_fft)
    P = np.fft.rfft(p.astype(np.float64), M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dLSE_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n * dLSE_da) / S2 - (4.0 * n * L) / S3
    return score, grad


def spectral_flatten(a0, N, n_iters, lam_init, lam_final, rng):

            new_m = (1.0 - lam) * mag[k] + lam * target
            mag[k] = new_m
            mag[N - k] = new_m
        # Keep DC and Nyquist

        a_new = np.real(np.fft.ifft(new_A))[:n]

def peak_shave(a, rng, n_steps, step_scale):
    """Direct peak shaving: move mass from peak contributors to safe positions."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for step in range(n_steps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        # Compute contribution of each a[j] to conv[k_star]
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        contrib = np.zeros(n)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            contrib[js] = a[js] * a[k_star - js]
        top_idx = np.argsort(contrib)[::-1]
        # Identify safe positions (zero contribution to peak)
        safe_mask = contrib < contrib.max() * 0.1
        if not safe_mask.any():
            safe_mask = contrib <= np.sort(contrib)[max(0, n // 4)]
        safe_indices = np.where(safe_mask)[0]
        if len(safe_indices) == 0 or len(top_idx) == 0:
            break
        # Move mass from top contributors to random safe positions
        n_top = max(3, n // 200)
        top_subset = top_idx[:n_top]
        total_move = 0.0
        for j in top_subset:
            amt = a[j] * step_scale
            a[j] -= amt
            if a[j] < 0:
                a[j] = 0.0
            total_move += a[j] if a[j] > 0 else 0
        # Redistribute to safe positions
        if total_move > 0 and len(safe_indices) > 0:
            n_safe_use = min(len(safe_indices), n_top * 2)
            chosen_safe = safe_indices[:n_safe_use]
            per_safe = total_move / n_safe_use
            for j in chosen_safe:
                a[j] += per_safe
        a = np.clip(a, 0.0, 1000.0)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_sc, best_a


def subgradient_polish(a, total_iters, sc_lb):
    """Subgradient method for the non-smooth peak objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient of max(conv) w.r.t. a
        sg_M = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            sg_M[js] = 2.0 * a[k_star - js]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        # Adaptive step size
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.2)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses spectral flattening, L-BFGS with LSE smoothing, spectral re-projection,
    peak shaving, and subgradient polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 30.0
    deadline = t_start + effective_budget

    best_score = np.inf
    best_a = None
    n = 0
    N_fft = 0

    def try_report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.astype(np.float64).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 128:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
    else:
        n = 8192
    N_fft = 1 << (2 * n - 1).bit_length()
    print(f"Using n = {n}, FFT size = {N_fft}")
    print(f"Budget: {effective_budget:.0f}s")

    # ============================================================
    # Phase 1: Generate diverse initializations
    # ============================================================
    print("\n=== Phase 1: Initialization ===")
    inits = []
    if parent is not None:
        inits.append(("parent", parent.astype(np.float64).copy()))

    # Spectral flatten from multiple diverse starts
    chirp = np.cos(np.pi * np.arange(n, dtype=np.float64) ** 2 / (2.0 * n)) ** 2 + 0.01
    starts = [
        ("uniform", np.ones(n) * 0.5),
        ("random", np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 10.0)),
        ("chirp", chirp.copy()),
        ("bimodal", 0.3 + 0.7 * np.exp(-2.0 * ((np.arange(n, dtype=np.float64) / n - 0.5) ** 2) * 20.0)),
    ]
    for name, s in starts:
        sf = spectral_flatten(s, N_fft, n_iters=400, lam_init=0.8, lam_final=0.25, rng=rng)
        inits.append((f"sf_{name}", sf))

    # Score and sort
    scored = []
    for name, a in inits:
        a = np.clip(a, 0.0, 1000.0)
        sc = compute_score(a)
        scored.append((sc, a, name))
        print(f"  {name}: score = {sc:.6f}")
    scored.sort(key=lambda x: x[0])

    # Report best initialization
    for sc, a, name in scored:
        try_report(sc, a, name)

    # ============================================================
    # Phase 2: L-BFGS with spectral re-projection between phases
    # ============================================================
    print("\n=== Phase 2: L-BFGS Optimization ===")
    beta_schedule = [15, 40, 100, 250, 600, 1500, 3000, 6000]

    # Work on top 2 initializations
    n_top = min(2, len(scored))
    for ii in range(n_top):
        sc_init, a_init, name_init = scored[ii]
        a_cur = a_init.copy()
        elapsed_init = time.monotonic() - t_start
        print(f"\n  >> Optimizing init '{name_init}' (score={sc_init:.6f}, elapsed={elapsed_init:.0f}s)")

        for bi, beta in enumerate(beta_schedule):
            if time.monotonic() > deadline - 200:
                break
            remaining = deadline - time.monotonic()
            # Scale iterations based on remaining time
            est_time_per_fun = 0.015 * (n / 8192.0)
            maxiter = int(remaining * 0.6 / (est_time_per_fun * 15))
            maxiter = max(300, min(maxiter, 15000))
            maxfun = maxiter * 25

            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
            a_cur = a_opt
            tag = f"lbfgs_b{beta}_{name_init}"
            try_report(sc, a_cur, tag)
            print(f"    beta={beta}: score={sc:.8f}")

            # Spectral re-projection between L-BFGS phases (every other beta)
            if bi % 2 == 1 and time.monotonic() < deadline - 100:
                a_reproj = spectral_reproject(a_cur, N_fft, n_iters=80, lam=0.3)
                sc_rep = compute_score(a_reproj)
                try_report(sc_rep, a_reproj, f"reproj_b{beta}_{name_init}")
                # Use whichever is better
                if sc_rep < sc:
                    a_cur = a_reproj
                    print(f"    reproj: score={sc_rep:.8f} (improved)")
                # Short L-BFGS from re-projected point
                if time.monotonic() < deadline - 80:
                    sc2, a2 = run_lbfgs(a_cur, beta, maxiter=500, maxfun=10000)
                    try_report(sc2, a2, f"relbfgs_b{beta}_{name_init}")
                    if sc2 < compute_score(a_cur):
                        a_cur = a2

        # Peak shaving after full L-BFGS trajectory
        if time.monotonic() < deadline - 60:
            sc_sh, a_sh = peak_shave(a_cur, rng, n_steps=150, step_scale=0.03)
            try_report(sc_sh, a_sh, f"shave_{name_init}")
            if sc_sh < compute_score(a_cur):
                a_cur = a_sh
            # L-BFGS from shaved
            if time.monotonic() < deadline - 40:
                sc3, a3 = run_lbfgs(a_cur, max(1000, beta_schedule[-1]), maxiter=3000, maxfun=60000)
                try_report(sc3, a3, f"postshave_{name_init}")

    print(f"\nAfter Phase 2: best = {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # ============================================================
    # Phase 3: High-beta refinement with spectral nudging
    # ============================================================
    print("\n=== Phase 3: High-Beta Refinement ===")
    for beta in [5000, 10000, 20000]:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        maxiter = int(remaining * 0.5 / (0.02 * (n / 8192.0) * 10))
        maxiter = max(1000, min(maxiter, 20000))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 20)
        try_report(sc, a_opt, f"refine_b{beta}")
        # Try spectral nudge
        if time.monotonic() < deadline - 40:
            a_nudged = spectral_reproject(a_opt, N_fft, n_iters=40, lam=0.15)
            sc_n = compute_score(a_nudged)
            try_report(sc_n, a_nudged, f"nudge_b{beta}")
            if sc_n < sc:
                sc2, a2 = run_lbfgs(a_nudged, beta, maxiter=2000, maxfun=40000)
                try_report(sc2, a2, f"refine2_b{beta}")

    # ============================================================
    # Phase 4: Peak shaving + local perturbations
    # ============================================================
    print("\n=== Phase 4: Local Search ===")
    n_local_trials = 8
    for trial in range(n_local_trials):
        if time.monotonic() > deadline - 30:
            break
        strategy = trial % 4
        if strategy == 0:
            # Peak shave
            a_pert = best_a.copy()
        elif strategy == 1:
            # Small random perturbation
            a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(n))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Spectral perturbation (phase noise)
            A = np.fft.fft(best_a, N_fft)
            noise = 0.03 * (rng.standard_normal(N_fft) + 1j * rng.standard_normal(N_fft))
            A = A * (1.0 + noise)
            A[0] = np.fft.fft(best_a, N_fft)[0]  # Preserve DC
            a_pert = np.real(np.fft.ifft(A))[:n]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Chirp-modulated perturbation
            x = np.arange(n, dtype=np.float64)
            f = 2.0 + 6.0 * rng.random()
            a_pert = best_a * (1.0 + 0.03 * np.cos(np.pi * f * x / n))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        sc, a_opt = run_lbfgs(a_pert, 3000, maxiter=3000, maxfun=60000)
        try_report(sc, a_opt, f"local_{trial}_s{strategy}")

    # ============================================================
    # Phase 5: Final subgradient polish
    # ============================================================
    print("\n=== Phase 5: Subgradient Polish ===")
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # Theoretical lower bound
    sc_sg, a_sg = subgradient_polish(best_a, total_iters=1000, sc_lb=sc_lb)
    try_report(sc_sg, a_sg, "subgrad")

    # Final L-BFGS from subgradient result
    if time.monotonic() < deadline - 20:
        sc_f, a_f = run_lbfgs(a_sg, 10000, maxiter=5000, maxfun=100000)
        try_report(sc_f, a_f, "final_lbfgs")

    # One more subgradient pass
    if time.monotonic() < deadline - 10:
        sc_sg2, a_sg2 = subgradient_polish(best_a, total_iters=500, sc_lb=sc_lb)
        try_report(sc_sg2, a_sg2, "subgrad2")

    # ============================================================
    # Final output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"Theoretical lower bound: {2.0*n/(2.0*n-1.0):.8f}")
    print(f"Gap from bound: {best_score - 2.0*n/(2.0*n-1.0):.8f}")
    print(f"{'='*60}")
    return out
