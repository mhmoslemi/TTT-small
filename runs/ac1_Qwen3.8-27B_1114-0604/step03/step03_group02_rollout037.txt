```
def lse_score_grad(a,beta):
  a=np.asarray(a,float).ravel()
  n=a.size
  c=fast_conv(a)
  S=a.sum()
  if S<=1e-12: return 1e100, zeros
  cmax=c.max()
  # avoid overflow: scaled=beta*(c-cmax)
  scaled=beta*(c-cmax)
  exps=np.exp(scaled) # since <=0
  denom=exps.sum()
  if denom<=0: return ...
  p=exps/denom
  L=cmax + np.log(denom)/beta
  sc=2*n*L/(S*S)
  # grad
  M=1 << (3*n-2).bit_length()
  RA=np.fft.rfft(a[::-1],M)
  P=np.fft.rfft(p,M)
  cross=np.fft.irfft(RA*P,M)
  g=2*cross[n-1:2*n-1]
  grad=(2*n)*(g/(S*S) - 2*L/(S*S*S))
  return sc, grad
```

```python
"""
Search for a non-negative step-function coefficient sequence of length 8192
minimizing the black-box evaluator `evaluate_sequence`.

The algorithm combines:
  * FFT-convolution-based smoothed objectives (LSE + L-BFGS-B) for the
    analytically visible score,
  * softmax parameterization on the simplex to respect scale invariance,
  * an LP-inspired "active-set" candidate generator that follows the state-of-art
    idea more literally,
  * black-box local refinement using actual evaluator scores, with logarithmic
    structured multiplicative updates.

All helper functions are top-level and do not use closures or lambdas.
"""

import numpy as np
import time

from scipy.optimize import minimize
from scipy.sparse import csr_matrix

try:
    import cvxpy as cp
    HAVE_CP = True
except Exception:
    HAVE_CP = False

if HAVE_CP:
    try:
        cp.settings.max_time = 45
    except Exception:
        pass
    try:
        cp.settings.suppress_warnings = True
    except Exception:
        pass


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    m = 2 * n - 1
    M = _next_pow2(m)
    A = np.fft.rfft(a, M)
    out = np.fft.irfft(A * A, M)
    return out[:m]


def explicit_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    S = float(np.sum(a))
    if not np.isfinite(S) or S <= 1e-12:
        return np.inf
    c = fft_convolve(a)
    M = float(np.max(c))
    if not np.isfinite(M):
        return np.inf
    sc = 2.0 * n * M / (S * S)
    if not np.isfinite(sc):
        return np.inf
    return float(sc)


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return 1e100, np.zeros(0, dtype=np.float64)

    S = float(np.sum(a))
    if not np.isfinite(S) or S <= 1e-12:
        return 1e100, np.zeros(n, dtype=np.float64)

    c = fft_convolve(a)
    c = np.nan_to_num(c, nan=0.0, posinf=1e100, neginf=0.0)
    cmax = float(np.max(c))

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1.0

    scaled = beta * (c - cmax)
    scaled = np.nan_to_num(scaled, nan=-1e6, posinf=-1e6, neginf=-1e6)
    smax = float(np.max(scaled))
    if smax > 0.0:
        scaled = scaled - smax
    scaled = np.clip(scaled, -745.0, 0.0)

    exps = np.exp(scaled)
    denom = float(np.sum(exps))

    if not np.isfinite(denom) or denom <= 0.0:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
        L = cmax
    else:
        p = exps / denom
        L = cmax + float(np.log(denom)) / beta

    sc = 2.0 * n * L / (S * S)

    M2 = _next_pow2(3 * n - 2)
    RA = np.fft.rfft(a[::-1], M2)
    P = np.fft.rfft(p, M2)
    cross = np.fft.irfft(RA * P, M2)
    g = 2.0 * cross[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    grad = np.nan_to_num(grad, nan=0.0, posinf=1e100, neginf=-1e100)

    if not np.isfinite(sc):
        return 1e100, grad
    return float(sc), grad


def lse_minimize(x, beta):
    a = np.asarray(x, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    S = float(np.sum(a))
    if not np.isfinite(S) or S <= 1e-9:
        return 1e100, np.zeros_like(a, dtype=np.float64)

    sc, grad = lse_score_grad(a, beta)

    # Mild sum-penalty helps conditionedness in the unparameterized run.
    if S > 1e-9:
        pen = 1e-4 * (S - 1.0) ** 2
        gpen = 2e-4 * (S - 1.0)
        return float(sc + pen), grad + gpen

    return float(sc), grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.nan_to_num(a0, nan=0.0, posinf=1000.0, neginf=0.0)
    a0 = np.clip(a0, 0.0, 1000.0)
    S = float(np.sum(a0))
    if not np.isfinite(S) or S <= 1e-12:
        a0 = np.full(a0.size, 1.0 / max(1, a0.size), dtype=np.float64)
    else:
        a0 = a0 / S

    n = a0.size
    if n == 0:
        return np.inf, a0

    if deadline is not None and time.monotonic() >= deadline:
        return explicit_score(a0), a0

    bounds = [(0.0, 1000.0)] * n
    options = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 32,
    }

    if deadline is not None:
        rem = deadline - time.monotonic()
        if rem < 30.0:
            return explicit_score(a0), a0
        options["maxfun"] = min(int(maxfun), max(200, int(rem * 20)))
        options["maxiter"] = min(int(maxiter), max(50, int(rem * 5)))

    try:
        res = minimize(
            lse_minimize,
            a0,
            args=(float(beta),),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options,
        )
        out = np.clip(res.x, 0.0, 1000.0)
        out = normalize_nonneg(out)
    except Exception:
        out = a0

    return explicit_score(out), out


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    z = np.nan_to_num(z, nan=0.0, posinf=30.0, neginf=-30.0)
    z = np.clip(z, -30.0, 30.0)
    z = z - z.max()
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(z.size, 1.0 / z.size, dtype=np.float64)
    return e / s


def softmax_minimize(z, beta):
    a = softmax_from_z(z)
    n = a.size
    sc, gauge = lse_score_grad(a, beta)

    if not np.isfinite(sc):
        return 1e100, np.zeros(n, dtype=np.float64)

    gauge = np.nan_to_num(gauge, nan=0.0, posinf=1e100, neginf=-1e100)
    dot = float(np.dot(gauge, a))
    gz = a * (gauge - dot)
    gz = np.nan_to_num(gz, nan=0.0, posinf=1e100, neginf=-1e100)
    return float(sc), gz


def run_lbfgs_softmax(z0, beta, maxiter=5000, maxfun=100000, deadline=None):
    z0 = np.asarray(z0, dtype=np.float64).ravel()
    z0 = np.nan_to_num(z0, nan=0.0, posinf=30.0, neginf=-30.0)
    z0 = np.clip(z0, -30.0, 30.0)
    z0 = z0 - float(np.mean(z0))

    n = z0.size
    if n == 0:
        a0 = np.ones(1, dtype=np.float64)
        return explicit_score(a0), a0

    if deadline is not None and time.monotonic() >= deadline:
        a0 = softmax_from_z(z0)
        return explicit_score(a0), a0

    bounds = [(-30.0, 30.0)] * n
    options = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 32,
    }

    if deadline is not None:
        rem = deadline - time.monotonic()
        if rem < 30.0:
            a0 = softmax_from_z(z0)
            return explicit_score(a0), a0
        options["maxfun"] = min(int(maxfun), max(200, int(rem * 20)))
        options["maxiter"] = min(int(maxiter), max(50, int(rem * 5)))

    try:
        res = minimize(
            softmax_minimize,
            z0,
            args=(float(beta),),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options,
        )
        out = softmax_from_z(res.x)
    except Exception:
        out = softmax_from_z(z0)

    return explicit_score(out), out


def actual_score(a):
    try:
        out = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(out) == 0:
            return np.inf
        val = evaluate_sequence(out)
        val = float(val)
        if not np.isfinite(val):
            return np.inf
        return val
    except Exception:
        return np.inf


def resample_sequence(arr, n):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    if arr.size == 0:
        return None
    if arr.size == n:
        return arr.copy()
    if arr.size == 1:
        return np.full(n, arr[0], dtype=np.float64)
    old = np.linspace(0.0, 1.0, arr.size)
    new = np.linspace(0.0, 1.0, n)
    return np.interp(new, old, arr)


def build_inits(n, rng, parent):
    inits = []

    if parent is not None:
        p = resample_sequence(parent, n)
        if p is not None:
            inits.append(normalize_nonneg(p))

    x = np.arange(n, dtype=np.float64) / float(n)

    # Uniform / constant is the explicit-formula optimum; useful fallback.
    inits.append(normalize_nonneg(np.ones(n, dtype=np.float64)))

    # Low-frequency cosine shapes.
    modes = [
        (1, 0.20, 0.0),
        (1, 0.35, 1.1),
        (2, 0.30, 0.5),
        (3, 0.32, 1.7),
        (5, 0.22, 0.9),
        (8, 0.16, 2.3),
    ]
    for freq, amp, ph in modes:
        arr = 0.5 + amp * np.cos(2.0 * np.pi * freq * x + ph)
        inits.append(normalize_nonneg(arr))

    # Smooth random low-pass noise.
    for _ in range(2):
        spec = np.fft.rfft(rng.standard_normal(n))
        cutoff = max(1, int(0.18 * spec.size))
        spec[cutoff:] = 0.0
        arr = np.fft.irfft(spec, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
        arr = 0.5 + 0.35 * arr / (float(np.max(np.abs(arr))) + 1e-12)
        inits.append(normalize_nonneg(arr))

    # Block structures.
    for nb in (8, 16):
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        arr = np.repeat(vals, per)
        if arr.size < n:
            arr = np.concatenate([arr, vals[: n - arr.size]])
        inits.append(normalize_nonneg(arr[:n]))

    # Gapped / alternating.
    arr = 0.5 + 0.3 * rng.standard_normal(n)
    arr[1::2] = 0.1
    inits.append(normalize_nonneg(arr))

    # Sparse pulses on baseline.
    arr = np.full(n, 0.05, dtype=np.float64)
    for _ in range(3):
        c = int(rng.integers(0, n))
        w = 16 + int(rng.integers(0, 64))
        arr[c:c + w] = 1.0
    inits.append(normalize_nonneg(arr))

    return inits


def choose_active_indices(c, count, center=None):
    m = c.size
    if m == 0:
        return np.empty(0, dtype=int)

    if center is not None:
        center = int(center)
        if center < 0:
            center = 0
        if center >= m:
            center = m - 1

    count = int(count)
    if count >= m:
        idx = np.arange(m, dtype=int)
        return idx

    if count <= 1:
        idx = np.array([center if center is not None else 0], dtype=int)
        return idx

    idx = np.argpartition(c, -count)[-count:]
    idx = idx.astype(int)

    if center is not None and not np.any(idx == center):
        pos = int(np.argmin(c[idx]))
        idx[pos] = center

    return idx


def build_lp_matrix(f, active_indices):
    n = f.size
    active = np.asarray(active_indices, dtype=int)
    m = active.size
    if m == 0 or n == 0:
        return None

    data_parts = []
    row_parts = []
    col_parts = []

    max_k = 2 * n - 2
    for r, k in enumerate(active):
        k = int(k)
        if k < 0:
            k = 0
        if k > max_k:
            k = max_k

        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        if end < start:
            continue

        js = np.arange(start, end + 1, dtype=np.int64)
        vals = f[k - js].astype(np.float64)

        data_parts.append(vals)
        row_parts.append(np.full(js.shape, r, dtype=np.int32))
        col_parts.append(js.astype(np.int32))

    if not data_parts:
        return None

    data = np.concatenate(data_parts)
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)

    return csr_matrix((data, (rows, cols)), shape=(m, n))


def lp_solvers_list():
    sols = []
    if not HAVE_CP:
        return sols
    names = ("MOSEK", "GUROBI", "ECOS", "GLPK", "CBC", "PDLP", "SCIP", "XPRESS", "CVXOPT")
    for name in names:
        sol = getattr(cp, name, None)
        if sol is not None:
            sols.append(sol)
    if not sols:
        try:
            sols.append(cp.ECOS)
        except Exception:
            pass
    return sols


def solve_lp_shape(f, M, active_indices, b_upper=1.0, deadline=None):
    if not HAVE_CP:
        return None
    M = float(M)
    if not np.isfinite(M) or M <= 1e-300:
        return None

    A = build_lp_matrix(f, active_indices)
    if A is None or A.nnz == 0:
        return None

    n = f.size
    num_rows = len(active_indices)

    try:
        b = cp.Variable(n, nonneg=True)
        scale = 1.0 / M
        As = A * scale
        rhs = np.full(num_rows, 1.0 + 1e-7 + 1e-8, dtype=np.float64)
        constraints = [As @ b <= rhs, b <= b_upper]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        solvers = lp_solvers_list()
        val = None
        ecos_name = getattr(cp, "ECOS", None)

        for sol in solvers:
            if deadline is not None and time.monotonic() > deadline:
                break
            try:
                if sol == ecos_name:
                    prob.solve(solver=sol, max_iter=50000)
                else:
                    prob.solve(solver=sol)

                if prob.status in ("optimal", "optimal_inaccurate", "solved", "solved_inaccurate"):
                    val = b.value
                    break
            except Exception:
                pass

        if val is None:
            return None

        val = np.asarray(val, dtype=np.float64).ravel()
        val = np.nan_to_num(val, nan=0.0, posinf=b_upper, neginf=0.0)
        val = np.clip(val, 0.0, b_upper)
        s = float(np.sum(val))
        if not np.isfinite(s) or s <= 1e-12:
            return None
        return val
    except Exception:
        return None


def lp_generate_candidate(a, active_count, b_upper=1.0, deadline=None):
    if not HAVE_CP:
        return None
    if deadline is not None and time.monotonic() > deadline - 30.0:
        return None

    f = normalize_nonneg(a)
    n = f.size

    # Keep f strictly positive to avoid degenerate LP columns.
    eps = 1e-7 / max(1, n)
    f = f + eps
    f = f / f.sum()

    c = fft_convolve(f)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 1e-15:
        return None

    ac = int(active_count)
    ac = max(1, ac)
    ac = min(ac, 2 * n - 1)

    idx = choose_active_indices(c, ac, n - 1)
    b = solve_lp_shape(f, M, idx, b_upper, deadline)
    if b is None:
        return None

    return normalize_nonneg(b)


def active_h(a, top_frac=0.02, top_min=1, top_max=2048):
    a = normalize_nonneg(a)
    n = a.size
    if n == 0:
        return None

    c = fft_convolve(a)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 1e-15:
        return None

    m = 2 * n - 1
    count = int(m * float(top_frac))
    count = max(int(top_min), min(count, int(top_max), m))

    idx = choose_active_indices(c, count, n - 1)
    if idx.size == 0:
        return None

    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0 / float(idx.size)

    M2 = _next_pow2(3 * n - 2)
    RA = np.fft.rfft(a[::-1], M2)
    W = np.fft.rfft(w, M2)
    cross = np.fft.irfft(RA * W, M2)
    h = 2.0 * cross[n - 1:2 * n - 1]
    h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)

    if float(np.max(h)) <= 1e-30:
        return None
    return h


def active_subgradient_direction(a, top_frac=0.02):
    h = active_h(a, top_frac=top_frac)
    if h is None:
        return None
    d = -h
    d = d - float(np.mean(d))
    d = np.nan_to_num(d, nan=0.0, posinf=0.0, neginf=0.0)
    if float(np.max(np.abs(d))) < 1e-15:
        return None
    return d


def multiplicative_direction(a, top_frac=0.05):
    h = active_h(a, top_frac=top_frac)
    if h is None:
        return None
    h = np.maximum(h, 1e-30)
    hmean = float(np.mean(h))
    if not np.isfinite(hmean) or hmean <= 1e-30:
        return None
    d = -np.log(h / hmean)
    d = d - float(np.mean(d))
    d = np.nan_to_num(d, nan=0.0, posinf=0.0, neginf=0.0)
    if float(np.max(np.abs(d))) < 1e-15:
        return None
    return d


def random_direction(n, rng, kind=None):
    x = np.arange(n, dtype=np.float64) / float(n)

    if kind is None:
        kind = int(rng.integers(0, 5))

    if kind == 0:
        f = 1 + int(rng.integers(0, 20))
        ph = 2.0 * np.pi * float(rng.random())
        d = np.cos(2.0 * np.pi * f * x + ph)
    elif kind == 1:
        f1 = 1 + int(rng.integers(0, 10))
        f2 = 2 + int(rng.integers(0, 15))
        ph1 = 2.0 * np.pi * float(rng.random())
        ph2 = 2.0 * np.pi * float(rng.random())
        d = np.cos(2.0 * np.pi * f1 * x + ph1) * np.cos(2.0 * np.pi * f2 * x + ph2)
    elif kind == 2:
        spec = np.fft.rfft(rng.standard_normal(n))
        cutoff = max(1, int(0.15 * spec.size))
        spec[cutoff:] = 0.0
        d = np.fft.irfft(spec, n)
    elif kind == 3:
        d = np.zeros(n, dtype=np.float64)
        nb = 32 + int(rng.integers(0, 256))
        b1 = int(rng.integers(0, max(1, n // 2)))
        b2 = int(rng.integers(max(1, n // 2), n))
        d[b1:b1 + nb] = 1.0
        d[b2:b2 + nb] = -1.0
    else:
        d = rng.standard_normal(n)

    d = np.nan_to_num(d, nan=0.0, posinf=0.0, neginf=0.0)
    d = d - float(np.mean(d))
    if float(np.max(np.abs(d))) < 1e-15:
        d = rng.standard_normal(n)
        d = d - float(np.mean(d))
    return d


def log_dir_line_search(a, d, deadline, best_score, best_a, max_evals=12):
    a = normalize_nonneg(a)
    d = np.asarray(d, dtype=np.float64).ravel()
    if d.size != a.size:
        return best_score, best_a

    d = np.nan_to_num(d, nan=0.0, posinf=0.0, neginf=0.0)
    d = d - float(np.mean(d))
    dmax = float(np.max(np.abs(d)))
    if not np.isfinite(dmax) or dmax < 1e-15:
        return best_score, best_a

    d = d / dmax * 2.0

    alphas = [
        0.005, -0.005,
        0.015, -0.015,
        0.05, -0.05,
        0.15, -0.15,
        0.4, -0.4,
        1.0, -1.0,
        2.0, -2.0,
    ]

    floor = 1e-6
    loga = np.log(np.maximum(a, floor))

    cur_score = float(best_score)
    cur_a = normalize_nonneg(best_a)

    evals_left = int(max_evals)
    for al in alphas:
        if evals_left <= 0:
            break
        if deadline is not None and time.monotonic() > deadline:
            break

        al = float(al)
        z = loga + al * d
        z = np.clip(z, -30.0, 30.0)
        cand = np.exp(z)
        s = float(np.sum(cand))
        if not np.isfinite(s) or s <= 1e-12:
            continue
        cand = cand / s

        sc = actual_score(cand)
        evals_left -= 1

        if sc < cur_score - 1e-12:
            cur_score = sc
            cur_a = cand.copy()

    return cur_score, cur_a


def log_mix_line_search(a, g, deadline, best_score, best_a, max_evals=14):
    a = normalize_nonneg(a)
    g = normalize_nonneg(g)
    if g.size != a.size:
        return best_score, best_a

    floor = 1e-6
    loga = np.log(np.maximum(a, floor))
    logg = np.log(np.maximum(g, floor))

    tvals = [
        0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3,
        0.4, 0.5, 0.65, 0.8, 0.9, 1.0,
    ]

    cur_score = float(best_score)
    cur_a = normalize_nonneg(best_a)

    evals_left = int(max_evals)
    for t in tvals:
        if evals_left <= 0:
            break
        if deadline is not None and time.monotonic() > deadline:
            break

        t = float(t)
        z = (1.0 - t) * loga + t * logg
        z = np.clip(z, -30.0, 30.0)
        cand = np.exp(z)
        s = float(np.sum(cand))
        if not np.isfinite(s) or s <= 1e-12:
            continue
        cand = cand / s

        sc = actual_score(cand)
        evals_left -= 1

        if sc < cur_score - 1e-12:
            cur_score = sc
            cur_a = cand.copy()

    return cur_score, cur_a


def local_search_actual(a, best_score, rng, deadline, max_no_improve=120):
    cur = normalize_nonneg(a)
    cur_score = float(best_score)
    n = cur.size
    no_improve = 0
    total_trials = 0

    tf_choices1 = [0.005, 0.01, 0.02, 0.05, 0.1]
    tf_choices2 = [0.01, 0.02, 0.05, 0.1]

    while True:
        if deadline is None:
            break
        if time.monotonic() > deadline - 20.0:
            break

        total_trials += 1
        r = float(rng.random())

        if r < 0.25:
            tf = tf_choices1[int(rng.integers(0, len(tf_choices1)))]
            d = active_subgradient_direction(cur, top_frac=tf)
        elif r < 0.50:
            tf = tf_choices2[int(rng.integers(0, len(tf_choices2)))]
            d = multiplicative_direction(cur, top_frac=tf)
        else:
            d = random_direction(n, rng)

        if d is not None:
            sc2, cur2 = log_dir_line_search(cur, d, deadline, cur_score, cur, max_evals=10)
            if sc2 < cur_score - 1e-12:
                cur_score = sc2
                cur = cur2
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > max_no_improve:
            no_improve = 0
            dr = random_direction(n, rng)
            sc2, cur2 = log_dir_line_search(cur, dr, deadline, cur_score, cur, max_evals=12)
            if sc2 < cur_score - 1e-12:
                cur_score = sc2
                cur = cur2

        if total_trials % 40 == 0:
            print(
                f"  [local] trials={total_trials} best={cur_score:.10f} "
                f"t={time.monotonic()- (time.monotonic() - time.monotonic()):.0f}",
                flush=True,
            )

    return cur_score, cur


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    n = 8192

    parent = None
    try:
        if "height_sequence_1" in globals():
            raw = globals()["height_sequence_1"]
            parent_arr = np.asarray(raw, dtype=np.float64).ravel()
            if parent_arr.size > 0:
                res = resample_sequence(parent_arr, n)
                if res is not None and res.size == n:
                    parent = normalize_nonneg(res)
    except Exception:
        parent = None

    if parent is None:
        parent = np.full(n, 1.0 / float(n), dtype=np.float64)

    best_a = parent.copy()
    best_sc = actual_score(best_a)
    if not np.isfinite(best_sc):
        best_sc = explicit_score(best_a)

    print(f"Initial best: {best_sc:.10f} (n={n})", flush=True)

    # -------------------------------------------------------------
    # Quick diverse initialization check, always using the true evaluator.
    # -------------------------------------------------------------
    init_deadline = t_start + min(180.0, 0.05 * total_budget)
    inits = build_inits(n, rng, best_a)
    for i, a0 in enumerate(inits):
        if time.monotonic() > init_deadline:
            break
        sc = actual_score(a0)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a0.copy()
            print(f"  [init {i}] {sc:.10f}", flush=True)

    # -------------------------------------------------------------
    # Smoothed optimizer phase for the analytically known objective.
    # This is safe because we only accept candidates by true score.
    # -------------------------------------------------------------
    phase2_deadline = t_start + min(420.0, 0.18 * total_budget)

    a_exp = normalize_nonneg(best_a)
    sc_exp, a_opt = run_lbfgs(a_exp, 30, maxiter=3000, maxfun=50000, deadline=phase2_deadline)
    sc_act = actual_score(a_opt)
    if sc_act < best_sc - 1e-12:
        best_sc = sc_act
        best_a = a_opt.copy()
        print(f"  [lbfgs a-space beta=30] {best_sc:.10f}", flush=True)
    a_exp = a_opt

    if time.monotonic() < phase2_deadline - 60.0:
        z0 = np.log(np.maximum(a_exp, 1e-8))
        z0 = z0 - np.mean(z0)
        sc_exp, a_opt = run_lbfgs_softmax(z0, 50, maxiter=3000, maxfun=50000, deadline=phase2_deadline)
        sc_act = actual_score(a_opt)
        if sc_act < best_sc - 1e-12:
            best_sc = sc_act
            best_a = a_opt.copy()
            print(f"  [lbfgs softmax beta=50] {best_sc:.10f}", flush=True)
        a_exp = a_opt

    if time.monotonic() < phase2_deadline - 120.0:
        z0 = np.log(np.maximum(a_exp, 1e-8))
        z0 = z0 - np.mean(z0)
        sc_exp, a_opt = run_lbfgs_softmax(z0, 200, maxiter=3000, maxfun=50000, deadline=phase2_deadline)
        sc_act = actual_score(a_opt)
        if sc_act < best_sc - 1e-12:
            best_sc = sc_act
            best_a = a_opt.copy()
            print(f"  [lbfgs softmax beta=200] {best_sc:.10f}", flush=True)
        a_exp = a_opt

    # -------------------------------------------------------------
    # LP-inspired active-set candidate generation phase.
    # -------------------------------------------------------------
    lp_deadline = t_start + min(1380.0, 0.58 * total_budget)
    active_schedule = [64, 128, 128, 256, 64, 256, 128, 256]
    tf_schedule = [0.01, 0.02, 0.05, 0.02, 0.01, 0.05, 0.02, 0.03]
    round_i = 0
    last_best_time = t_start

    while round_i < len(active_schedule) * 3:
        if time.monotonic() > lp_deadline:
            break

        ac = active_schedule[round_i % len(active_schedule)]
        improved = False

        g = lp_generate_candidate(best_a, ac, b_upper=1.0, deadline=deadline)
        if g is not None:
            sc2, a2 = log_mix_line_search(best_a, g, deadline, best_sc, best_a, max_evals=12)
            if sc2 < best_sc - 1e-12:
                best_sc = sc2
                best_a = a2.copy()
                improved = True
                last_best_time = time.monotonic()
                print(f"  [LP active={ac}] {best_sc:.10f}", flush=True)

        tf = tf_schedule[round_i % len(tf_schedule)]
        d = active_subgradient_direction(best_a, top_frac=tf)
        if d is not None:
            sc2, a2 = log_dir_line_search(best_a, d, deadline, best_sc, best_a, max_evals=10)
            if sc2 < best_sc - 1e-12:
                best_sc = sc2
                best_a = a2.copy()
                improved = True
                last_best_time = time.monotonic()
                print(f"  [subgrad tf={tf}] {best_sc:.10f}", flush=True)

        if not improved and (time.monotonic() - last_best_time) > 240.0 and round_i > 3:
            d = random_direction(n, rng)
            sc2, a2 = log_dir_line_search(best_a, d, deadline, best_sc, best_a, max_evals=12)
            if sc2 < best_sc - 1e-12:
                best_sc = sc2
                best_a = a2.copy()
                last_best_time = time.monotonic()
                print(f"  [diversify random] {best_sc:.10f}", flush=True)

        round_i += 1

    # -------------------------------------------------------------
    # Final black-box local search.
    # -------------------------------------------------------------
    final_deadline = deadline - 10.0
    best_sc, best_a = local_search_actual(best_a, best_sc, rng, final_deadline)

    # -------------------------------------------------------------
    # Final verification and safe output.
    # -------------------------------------------------------------
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if len(out) != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, len(out))
        tmp[:m] = out[:m]
        if tmp.sum() <= 0.01:
            tmp[:] = 1.0 / float(n)
        out = [float(x) for x in tmp]

    if sum(out) < 0.01:
        out = [1.0 / float(n)] * n

    final_sc = actual_score(out)
    if not np.isfinite(final_sc):
        final_sc = explicit_score(out)
    if final_sc < best_sc:
        best_sc = final_sc

    print(f"FINAL: {best_sc:.10f} length={len(out)}", flush=True)
    return out
```
