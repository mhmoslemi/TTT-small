"""
Improved optimizer for minimum peak autocorrelation sequence.
Uses fast coordinate descent with analytic 1D solver, sensitivity-based
perturbation, and periodic LSE-LBFGS global moves.
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

            # t*(2S - a_j) = 2b_j - a_j*S
            denom = 2.0 * S_rest - a_j
            if abs(denom) < 1e-14:
                t_cand = t_upper
            else:
                t_cand = (2.0 * b_j - a_j * S_rest) / denom
        else:
            if a_j < 1e-14:
                # Constant constraint, f = 2n*b_j/(S+t)^2 decreasing
                # Need to find where another constraint becomes active
                # Just return t_upper (or use fallback)
                break
            t_cand = S_rest - 2.0 * b_j / a_j

        t_cand = max(0.0, min(t_upper, t_cand))

        # Check if this constraint is still the maximum at t_cand
        val = base_inside + t_cand * alpha + (t_cand * t_cand) * beta_coef
        cur_max = float(np.max(val))

        if cur_max > outside_max:
            new_active = int(np.argmax(val))
            if new_active == active:
                # Verify it's a local min
                eps = max(1e-7, 1e-6 * max(1.0, abs(t_cand)))
                S_new = S_rest + t_cand
                f_c = 2.0 * n * cur_max / (S_new * S_new)

                t_plus = min(t_cand + eps, t_upper)
                v_plus = base_inside + t_plus * alpha + (t_plus * t_plus) * beta_coef
                f_plus = 2.0 * n * max(float(np.max(v_plus)), outside_max) / ((S_rest + t_plus) ** 2)

                t_minus = max(t_cand - eps, 0.0)
                v_minus = base_inside + t_minus * alpha + (t_minus * t_minus) * beta_coef
                f_minus = 2.0 * n * max(float(np.max(v_minus)), outside_max) / ((S_rest + t_minus) ** 2)

                if f_c <= f_plus + 1e-12 and f_c <= f_minus + 1e-12:
                    return t_cand
            else:
                active = new_active
                continue
        else:
            # Outside max dominates at t_cand
            # Find largest t where all inside ≤ outside_max
            t_max = t_upper
            for j in range(len(base_inside)):
                if alpha[j] > 1e-14:
                    t_j = (outside_max - base_inside[j]) / alpha[j]
                    if t_j < t_max:
                        t_max = t_j
                if beta_coef[j] > 0.5:
                    # t^2 + a_j*t + b_j - OM ≤ 0
                    c0 = base_inside[j] - outside_max
                    disc = alpha[j] ** 2 - 4.0 * c0
                    if disc >= 0:
                        t_j = (-alpha[j] + np.sqrt(disc)) / 2.0
                    else:
                        t_j = 0.0
                    if t_j < t_max:
                        t_max = t_j
            return max(0.0, min(t_upper, t_max))

    # Fallback: golden section (rare)
    def f_obj(t):
        val = base_inside + t * alpha + (t * t) * beta_coef
        im = float(np.max(val))
        tot = im if im > outside_max else outside_max
        S_new = S_rest + t
        if S_new < 1e-12:
            return 1e10
        return 2.0 * n * tot / (S_new * S_new)

    res = minimize_scalar(f_obj, bounds=(0.0, t_upper), method='bounded',
                          options={'xatol': 1e-8})
    return float(res.x)


def coord_sweep(a, conv, S, mode='random', rng=None):
    """
    One full coordinate descent sweep with incremental conv updates.
    mode: 'random', 'hot', 'cold'
    """
    n = len(a)
    a = a.copy()
    M_conv = 2 * n - 1

    if mode == 'random':
        order = rng.permutation(n) if rng is not None else np.arange(n)
    elif mode == 'hot':
        k_star = int(np.argmax(conv))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        order = np.argsort(contrib)[::-1]
    else:  # cold
        k_star = int(np.argmax(conv))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        order = np.argsort(contrib)

    for i in order:
        i = int(i)
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        # Outside max
        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)

        # Base: conv with a[i] removed
        base_inside = conv[k_lo:k_hi + 1].copy()
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i
        t_upper = max(2.0 * a_i + 0.01, S_rest / n * 4.0)
        t_upper = min(t_upper, 1000.0)

        t_opt = solve_1d_active(base_inside, alpha, beta_coef, outside_max, S_rest, n, t_upper)

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    return a, S


def sensitivity_perturb(a, conv, rng, strength=0.03):
    """
    Perturb based on sensitivity to active (peak) lags.
    Reduce high-sensitivity coordinates, increase low-sensitivity ones.
    """
    n = len(a)
    # Find top active lags
    c_sorted = np.sort(conv)[::-1]
    threshold = c_sorted[0] * 0.95  # lags within 5% of peak
    active_lags = np.where(conv >= threshold)[0]

    # Compute sensitivity of each coordinate to active lags
    sens = np.zeros(n)
    for k in active_lags:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            # sensitivity of conv[k] w.r.t. a[j] is a[k-j] (or 2a[k-j] if j=k-j... simplification)
            sens[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]

    # Normalize sensitivity
    sens_max = sens.max()
    if sens_max < 1e-12:
        return a.copy()

    # Reduce high-sensitivity, increase low-sensitivity
    reduce_mask = sens > 0.7 * sens_max
    increase_mask = sens < 0.3 * sens_max

    total_reduce = 0.0
    a_new = a.copy()
    for i in np.where(reduce_mask)[0]:
        amt = a_new[i] * strength
        a_new[i] -= amt
        total_reduce += amt

    if total_reduce > 0 and increase_mask.any():
        inc_indices = np.where(increase_mask)[0]
        per = total_reduce / len(inc_indices)
        a_new[inc_indices] += per

    return np.clip(a_new, 0.0, 1000.0)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300:
        p = np.ones(len(c)) / len(c)
        L = c_max
    else:
        log_sum = np.log(exp_sum)
        L = c_max + log_sum / beta
        p = exp_scaled / exp_sum

    score = 2.0 * n * L / (S * S)

    # Gradient: d/da[i] of 2n*L/S^2 = 2n*(dL/da[i] * 1/S^2 - 2*L/S^3)
    # dL/da[i] = sum_k p[k] * d(conv[k])/da[i]
    # d(conv[k])/da[i] = a[k-i] (if valid) + a[i-k] (if valid and k-i != i-k)
    # For self-conv: d(conv[k])/da[i] = 2*a[k-i] if i != k-i, a[i] if i == k-i (k=2i)
    #                = a[k-i] if only one side valid

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res  # gradient of L w.r.t. a

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3 * np.ones(n))
    # Actually: d(2n*L/S^2)/da[i] = 2n*(g[i]/S^2 - 2*L/(S^3))
    # The second term is the same for all i, so:
    grad = (2.0 * n / S2) * g - (4.0 * n * L / S3) * np.ones(n)

    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 10.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    # Try multiple initializations
    initializations = []

    if parent is not None:
        initializations.append(('parent', parent.copy()))
        p_score = compute_score(parent)
        print(f"Parent score: {p_score:.8f}, length: {len(parent)}")
        best_score = p_score
        best_a = parent.copy()

    # Additional initialization: raised-cosine window
    if parent is not None:
        n_init = len(parent)
        x = np.linspace(0, 1, n_init)
        rc = 0.5 * (1 + np.cos(2 * np.pi * x))
        initializations.append(('raised_cosine', rc / rc.sum()))

        # Gaussian initialization
        center = n_init / 2
        sigma = n_init / 4
        gauss = np.exp(-0.5 * ((np.arange(n_init) - center) / sigma) ** 2)
        initializations.append(('gaussian', gauss / gauss.sum()))

        # Flat with edge taper
        flat = np.ones(n_init)
        taper = min(50, n_init // 10)
        for i in range(taper):
            flat[i] *= (i + 1) / taper
            flat[n_init - 1 - i] *= (i + 1) / taper
        initializations.append(('edge_taper', flat / flat.sum()))

    if best_a is None:
        n_init = 2048
        best_a = np.ones(n_init) / n_init
        best_score = compute_score(best_a)
        initializations = [('flat', best_a.copy())]
        print(f"No parent. Flat init score: {best_score:.8f}")

    n_work = len(best_a)
    print(f"Working length: {n_work}")

    # ============================================================
    # Phase 1: Coordinate descent from best init (45% of time)
    # ============================================================
    phase1_deadline = t_start + 0.45 * total_budget
    print(f"\n=== Phase 1: Coordinate Descent ({phase1_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    conv_cur = fast_conv(a_cur)
    S_cur = float(a_cur.sum())
    no_improve = 0
    sweep_count = 0
    prev_sc = best_score

    while time.monotonic() < phase1_deadline:
        sweep_count += 1

        # Cycle through modes
        mode = ['random', 'hot', 'random', 'cold'][sweep_count % 4]
        a_cur, S_cur = coord_sweep(a_cur, conv_cur, S_cur, mode=mode, rng=rng)
        conv_cur = fast_conv(a_cur)

        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            no_improve = 0
            if sweep_count <= 5 or sweep_count % 50 == 0:
                print(f"  Sweep {sweep_count}: {sc:.8f} (NEW BEST)")
        else:
            no_improve += 1

        if sweep_count % 25 == 0:
            print(f"  Sweep {sweep_count}: {sc:.8f} (no_improve={no_improve})")

        # Perturb if stuck
        if no_improve >= 30:
            strength = 0.02 + 0.04 * rng.random()
            a_cur = sensitivity_perturb(a_cur, conv_cur, rng, strength)
            conv_cur = fast_conv(a_cur)
            S_cur = float(a_cur.sum())
            no_improve = 0
            print(f"  Stuck at sweep {sweep_count}, perturbing (strength={strength:.3f})")

    sc_final = compute_score(a_cur)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_cur.copy()
    print(f"Phase 1 done: {sweep_count} sweeps, best={best_score:.8f}")

    # ============================================================
    # Phase 2: Try other initializations (10% of time)
    # ============================================================
    phase2_deadline = t_start + 0.55 * total_budget
    print(f"\n=== Phase 2: Alternative Initializations ({phase2_deadline - t_start:.0f}s) ===")

    for init_name, init_a in initializations:
        if init_name == 'parent':
            continue
        if time.monotonic() > phase2_deadline - 5:
            break
        # Quick optimization from this init
        a_try = init_a.copy()
        # Scale to match parent's sum for fair comparison
        if parent is not None:
            a_try = a_try * (parent.sum() / a_try.sum())
        conv_try = fast_conv(a_try)
        S_try = float(a_try.sum())

        for sw in range(15):
            if time.monotonic() > phase2_deadline - 3:
                break
            a_try, S_try = coord_sweep(a_try, conv_try, S_try, mode='hot', rng=rng)
            conv_try = fast_conv(a_try)

        sc_try = compute_score(a_try)
        print(f"  Init '{init_name}': {sc_try:.8f}")
        if sc_try < best_score:
            best_score = sc_try
            best_a = a_try.copy()
            print(f"  → NEW BEST from {init_name}!")

    # ============================================================
    # Phase 3: LSE-LBFGS refinement (20% of time)
    # ============================================================
    phase3_deadline = t_start + 0.75 * total_budget
    print(f"\n=== Phase 3: LSE-LBFGS ({phase3_deadline - t_start:.0f}s) ===")

    betas = [50, 100, 300, 800, 2000, 5000]
    for beta in betas:
        if time.monotonic() > phase3_deadline - 5:
            break
        t0 = time.monotonic()
        sc, a_opt, res = run_lbfgs(best_a, beta, maxiter=4000, maxfun=80000)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  LSE β={beta}: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            print(f"  LSE β={beta}: {sc:.8f} ({elapsed:.1f}s)")

    # ============================================================
    # Phase 4: CD polish after LSE (10% of time)
    # ============================================================
    phase4_deadline = t_start + 0.85 * total_budget
    print(f"\n=== Phase 4: CD Polish ({phase4_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    conv_cur = fast_conv(a_cur)
    S_cur = float(a_cur.sum())
    for sw in range(100):
        if time.monotonic() > phase4_deadline:
            break
        mode = 'hot' if sw % 3 == 0 else 'random'
        a_cur, S_cur = coord_sweep(a_cur, conv_cur, S_cur, mode=mode, rng=rng)
        conv_cur = fast_conv(a_cur)
        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            if sw % 20 == 0:
                print(f"  P4 sweep {sw}: {sc:.8f} (NEW BEST)")

    print(f"Phase 4 done: best={best_score:.8f}")

    # ============================================================
    # Phase 5: Aggressive perturbation + recovery (10% of time)
    # ============================================================
    phase5_deadline = deadline
    print(f"\n=== Phase 5: Perturbation Cycles ({phase5_deadline - t_start:.0f}s) ===")

    n_cur = len(best_a)
    for trial in range(50):
        if time.monotonic() > phase5_deadline - 5:
            break

        strategy = trial % 6
        a_pert = best_a.copy()

        if strategy == 0:
            # Sensitivity-based perturbation
            conv_pert = fast_conv(a_pert)
            a_pert = sensitivity_perturb(a_pert, conv_pert, rng,
                                         strength=0.03 + 0.05 * rng.random())
        elif strategy == 1:
            # Gaussian noise
            scale = 0.02 + 0.04 * rng.random()
            a_pert = a_pert * (1.0 + scale * rng.standard_normal(n_cur))
        elif strategy == 2:
            # Block swap
            seg = max(20, n_cur // 40)
            s1 = int(rng.integers(0, n_cur - seg))
            s2 = int(rng.integers(0, n_cur - seg))
            tmp = a_pert[s1:s1 + seg].copy()
            a_pert[s1:s1 + seg] = a_pert[s2:s2 + seg]
            a_pert[s2:s2 + seg] = tmp
        elif strategy == 3:
            # Edge modification
            edge_len = max(20, n_cur // 20)
            edge_scale = 0.3 + 0.7 * rng.random()
            a_pert[:edge_len] *= edge_scale
            a_pert[-edge_len:] *= edge_scale
        elif strategy == 4:
            # Sinusoidal modulation
            freq = 1.0 + 20.0 * rng.random()
            amp = 0.02 + 0.05 * rng.random()
            x = np.arange(n_cur) / n_cur
            a_pert *= (1.0 + amp * np.cos(2 * np.pi * freq * x + 2 * np.pi * rng.random()))
        else:
            # Targeted: zero out top peak contributors and redistribute
            conv_pert = fast_conv(a_pert)
            k_star = int(np.argmax(conv_pert))
            contrib = np.zeros(n_cur)
            j_start = max(0, k_star - n_cur + 1)
            j_end = min(k_star, n_cur - 1)
            if j_end >= j_start:
                contrib[j_start:j_end + 1] = a_pert[j_start:j_end + 1][::-1]
            top_n = max(5, n_cur // 100)
            top_idx = np.argsort(contrib)[::-1][:top_n]
            saved = a_pert[top_idx].sum() * 0.3
            a_pert[top_idx] *= 0.7
            rest = np.ones(n_cur, dtype=bool)
            rest[top_idx] = False
            a_pert[rest] += saved / rest.sum()

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick CD recovery
        conv_pert = fast_conv(a_pert)
        S_pert = float(a_pert.sum())
        n_recover = 6
        if time.monotonic() > phase5_deadline - 8:
            n_recover = 2
        for sw in range(n_recover):
            a_pert, S_pert = coord_sweep(a_pert, conv_pert, S_pert, mode='hot', rng=rng)
            conv_pert = fast_conv(a_pert)

        sc_pert = compute_score(a_pert)
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  P5 trial {trial} (s={strategy}): {sc_pert:.8f} (NEW BEST)")
        elif trial % 10 == 0:
            print(f"  P5 trial {trial} (s={strategy}): {sc_pert:.8f}")

    # ============================================================
    # Final polish
    # ============================================================
    print(f"\n=== Final Polish ===")
    a_cur = best_a.copy()
    conv_cur = fast_conv(a_cur)
    S_cur = float(a_cur.sum())
    for _ in range(30):
        if time.monotonic() > deadline - 3:
            break
        a_cur, S_cur = coord_sweep(a_cur, conv_cur, S_cur, mode='random', rng=rng)
        conv_cur = fast_conv(a_cur)
        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

    # Final LSE if time permits
    if time.monotonic() < deadline - 20:
        sc, a_opt, _ = run_lbfgs(best_a, 3000, maxiter=3000, maxfun=50000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final LSE: {sc:.8f} (NEW BEST)")
        # Quick CD after
        a_cur = best_a.copy()
        conv_cur = fast_conv(a_cur)
        S_cur = float(a_cur.sum())
        a_cur, S_cur = coord_sweep(a_cur, conv_cur, S_cur, mode='hot', rng=rng)
        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  Final CD: {sc:.8f} (NEW BEST)")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
