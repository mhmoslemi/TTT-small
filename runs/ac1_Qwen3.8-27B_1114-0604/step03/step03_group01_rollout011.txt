"""
Multi-scale LSE-continuation optimizer with adaptive beta, softmax-space Adam,
and multi-phase polishing for minimizing 2n*max(self-conv)/(sum)^2.
Key improvements: progressive resolution, adaptive beta schedule based on LSE-true gap,
and aggressive final polishing with active-set-aware coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a (with sum(a)=1 normalization assumed)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: (1/beta) * sum_k p_k * grad c[k]
    # grad c[k] w.r.t. a[i] = 2*a[k-i] for valid k-i
    # So grad L[i] = (2/beta) * sum_k p[k] * a[k-i] (for valid range)
    # This is (2/beta) * (p * a_reversed)[n-1+i] ... let me compute via FFT
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad_L = (2.0 / beta) * conv_res
    # Full score gradient: d/d a [2n * L / S^2] = 2n * (grad_L / S^2 - 2*L / S^3)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return 2.0 * n * L / S2, grad


def softmax_fn(x):
    """Stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def lse_score_grad_x(x, beta):
    """LSE-smoothed score and gradient w.r.t. x (softmax parametrization)."""
    a = softmax_fn(x)
    score, grad_a = lse_score_grad_a(a, beta)
    d = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - d)
    return score, grad_x


def adam_softmax(a_init, beta_schedule, n_steps_per_beta, lr=0.01,
                 time_limit_s=None, t_start=None, normalize_x=True):
    """
    Adam in softmax (x) space with LSE-smoothed gradient and beta continuation.
    beta_schedule: list of (beta, n_steps) pairs.
    Returns (best_score, best_a, best_x).
    """
    n = len(a_init)
    a_init_safe = np.clip(a_init, 1e-15, 1e15)
    x = np.log(a_init_safe)
    x = x - x.mean()  # Center for numerical stability
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a_init_safe.copy()
    best_score = compute_score(a_init_safe)
    global_t = 0
    
    for beta, n_steps in beta_schedule:
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        # Reset Adam moments at each beta change for stability
        m = np.zeros(n)
        v = np.zeros(n)
        local_t = 0
        
        for step in range(n_steps):
            if time_limit_s is not None and t_start is not None:
                if time.monotonic() - t_start > time_limit_s:
                    break
            
            global_t += 1
            local_t += 1
            lr_t = lr / (1.0 + 1e-4 * global_t)
            
            score, grad_x = lse_score_grad_x(x, beta)
            
            m = b1 * m + (1.0 - b1) * grad_x
            v = b2 * v + (1.0 - b2) * grad_x * grad_x
            m_hat = m / (1.0 - b1 ** local_t)
            v_hat = v / (1.0 - b2 ** local_t)
            x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            
            if step % 2000 == 0:
                a_cur = softmax_fn(x)
                sc = compute_score(a_cur)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cur.copy()
    
    # Final check
    a_final = softmax_fn(x)
    sc_final = compute_score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_score, best_a, x

def run_lbfgs_a(a0, beta, maxiter=5000, maxfun=80000):
    """L-BFGS-B in a-space with bounds [0, 1000] using LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    
    # Normalize to sum=1 for better conditioning
    a0 = a0 / a0.sum()
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-12:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def active_set_coord_descent(a, n_sweeps=20, top_k_lags=10, time_limit_s=None, t_start=None):
    """
    Coordinate descent that focuses on reducing the top-k convolution lags.
    At each sweep, identifies the top-k lags and adjusts coordinates that
    contribute most to those lags.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        c_max = float(c.max())
        score = 2.0 * n * c_max
        if score < best_score:
            best_score = score
        
        # Find top-k lags near the maximum
        threshold = c_max - 0.01 * c_max  # Within 1% of max
        active_lags = np.where(c >= threshold)[0]
        if len(active_lags) == 0:
            active_lags = np.array([int(c.argmax())])
        active_lags = active_lags[:top_k_lags]
        
        # For each active lag, the contributing indices are:
        # i such that 0 <= lag - i < n, i.e., lag-n+1 <= i <= lag
        # Gradient contribution from lag k: 2*a[k-i]
        # We want to reduce a[i] where it contributes to the max lags
        
        # Compute "pressure" on each coordinate
        pressure = np.zeros(n)
        for k in active_lags:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                idx = k - np.arange(js, je + 1)
                pressure[js:je+1] += a[idx]
        
        # Coordinates with high pressure should be reduced
        # Find top-m pressure coordinates
        m = min(50, n // 10)
        top_pressure = np.argsort(pressure)[::-1][:m]
        
        for j in top_pressure:
            if time.monotonic() > deadline:
                break
            
            a_old = a[j]
            if a_old < 1e-15:
                continue
            
            # Try reducing this coordinate
            # Binary search for the best reduction factor
            lo, hi = 0.0, 1.0
            best_ratio = 1.0
            best_sc = score
            
            for _ in range(12):
                mid = (lo + hi) / 2.0
                a[j] = a_old * mid
                S = a.sum()
                if S < 1e-12:
                    a[j] = a_old
                    break
                a_norm = a / S
                sc = compute_score(a_norm)
                if sc < best_sc:
                    best_sc = sc
                    best_ratio = mid
                    lo = mid
                else:
                    hi = mid
                a[j] = a_old
            
            a[j] = a_old * best_ratio
        
        # Renormalize
        S = a.sum()
        if S > 1e-12:
            a = a / S
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    
    return best_score, a

                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            
            a[j] = best_x
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    
    return best_score, a


def resample_sequence(a, new_n):
    """Resample to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n) * a[0] / new_n
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def make_beta_schedule(beta_min, beta_max, n_phases, total_steps):
    """Create a list of (beta, steps_per_phase) pairs with geometric progression."""
    if n_phases <= 1:
        return [(beta_max, total_steps)]
    ratios = (beta_max / beta_min) ** (1.0 / (n_phases - 1))
    schedule = []
    # Allocate more steps to lower betas
    weights = []
    for i in range(n_phases):
        w = 1.0 / (1.0 + 0.1 * i)
        weights.append(w)
    total_w = sum(weights)
    for i in range(n_phases):
        beta = beta_min * (ratios ** i)
        steps = int(total_steps * weights[i] / total_w)
        schedule.append((beta, max(100, steps)))
    return schedule


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer:
    1. Initialize from parent or structured sequence
    2. LSE Adam with beta continuation in softmax space
    3. L-BFGS-B polishing with increasing beta
    4. Active-set coordinate descent
    5. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0
    deadline = t_start + effective_budget
    
    # --- Load parent ---
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    if parent is not None and parent.sum() > 0.01:
        parent = parent / parent.sum()
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.10f}")
    else:
        parent = np.ones(4096) / 4096
        parent_score = 2.0
        print(f"No parent, using uniform n=4096, score={parent_score:.10f}")
    
    best_score = parent_score
    best_a = parent.copy()
    
    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.10f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False
    
    # ==================================================================
    # Phase 1: LSE Adam in softmax space with beta continuation (40% time)
    # ==================================================================
    p1_budget = effective_budget * 0.40
    print(f"\n{'='*60}")
    print(f"Phase 1: LSE Adam softmax-space (budget={p1_budget:.0f}s)")
    print(f"{'='*60}")
    
    n_work = len(best_a)
    # Create beta schedule: from 10 to 20000 over ~8 phases
    beta_schedule = make_beta_schedule(10, 20000, 8, 200000)
    t0 = time.monotonic()
    sc, a_opt, x_opt = adam_softmax(best_a, beta_schedule, 200000, lr=0.01,
                                     time_limit_s=p1_budget, t_start=t_start)
    try_update(sc, a_opt, f"Adam softmax")
    print(f"  Phase 1 done: {sc:.10f} ({time.monotonic()-t0:.1f}s)")
    
    # ==================================================================
    # Phase 2: L-BFGS-B polishing (25% time)  
    # ==================================================================
    p2_budget = effective_budget * 0.25
    p2_deadline = t_start + effective_budget * 0.65
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS-B Polish (budget={p2_budget:.0f}s)")
    print(f"{'='*60}")
    
    a_work = best_a.copy()
    x_work = x_opt.copy()
    
    for beta in [200, 1000, 5000, 20000]:
        if time.monotonic() > p2_deadline:
            break
        t0 = time.monotonic()
        # L-BFGS in a-space
        sc_a, a_opt_a, _ = run_lbfgs_a(a_work, beta, maxiter=5000, maxfun=100000)
        if sc_a < best_score:
            a_work = a_opt_a
            try_update(sc_a, a_opt_a, f"LBFGS-a beta={beta}")
        # L-BFGS in x-space
        x_work_init = np.log(np.clip(a_work, 1e-15, 1e15))
        x_work_init = x_work_init - x_work_init.mean()
        sc_x, a_opt_x, x_work = run_lbfgs_x(x_work_init, beta, maxiter=3000, maxfun=60000)
        if sc_x < best_score:
            a_work = a_opt_x
            try_update(sc_x, a_opt_x, f"LBFGS-x beta={beta}")
        print(f"  LBFGS beta={beta}: best={best_score:.10f} ({time.monotonic()-t0:.1f}s)")
    
    # ==================================================================
    # Phase 3: Active-set coordinate descent (15% time)
    # ==================================================================
    p3_budget = effective_budget * 0.15
    p3_deadline = t_start + effective_budget * 0.80
    print(f"\n{'='*60}")
    print(f"Phase 3: Active-set Coord Descent (budget={p3_budget:.0f}s)")
    print(f"{'='*60}")
    
    t0 = time.monotonic()
    sc, a_opt = active_set_coord_descent(best_a, n_sweeps=30, top_k_lags=15,
                                         time_limit_s=p3_budget, t_start=t_start)
    try_update(sc, a_opt, "Active-set CD")
    print(f"  Active-set CD: {sc:.10f} ({time.monotonic()-t0:.1f}s)")
    
    # ==================================================================
    # Phase 4: Full coordinate descent (10% time)
    # ==================================================================
    p4_budget = effective_budget * 0.10
    p4_deadline = t_start + effective_budget * 0.90
    print(f"\n{'='*60}")
    print(f"Phase 4: Full Coord Descent (budget={p4_budget:.0f}s)")
    print(f"{'='*60}")
    
    t0 = time.monotonic()
    sc, a_opt = full_coord_descent(best_a, n_sweeps=20, n_candidates=8,
                                    time_limit_s=p4_budget, t_start=t_start)
    try_update(sc, a_opt, "Full CD")
    print(f"  Full CD: {sc:.10f} ({time.monotonic()-t0:.1f}s)")
    
    # ==================================================================
    # Phase 5: Multi-length exploration + final polish (10% time)
    # ==================================================================
    p5_deadline = deadline - 5.0
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-length + Final Polish")
    print(f"{'='*60}")
    
    n_best = len(best_a)
    
    # Try slightly different lengths
    for factor in [0.9, 1.1]:
        if time.monotonic() > p5_deadline:
            break
        new_n = int(n_best * factor)
        if new_n < 256 or new_n > 20000:
            continue
        a_res = resample_sequence(best_a, new_n)
        a_res = a_res / a_res.sum()
        
        # Quick Adam refinement
        beta_sched = make_beta_schedule(100, 5000, 3, 50000)
        sc, a_opt = adam_direct_normalized(a_res, beta_sched, 50000, lr=0.003,
                                            time_limit_s=min(30.0, p5_deadline - time.monotonic()),
                                            t_start=t_start)
        # Quick L-BFGS
        if time.monotonic() < p5_deadline - 5:
            sc2, a_opt2, _ = run_lbfgs_a(a_opt, 5000, maxiter=3000, maxfun=50000)
            if sc2 < sc:
                sc, a_opt = sc2, a_opt2
        try_update(sc, a_opt, f"n={new_n} (factor={factor})")
    
    # Final intensive L-BFGS
    if time.monotonic() < p5_deadline - 10:
        for beta in [10000, 50000]:
            if time.monotonic() > p5_deadline - 5:
                break
            t0 = time.monotonic()
            sc, a_opt, _ = run_lbfgs_a(best_a, beta, maxiter=8000, maxfun=120000)
            try_update(sc, a_opt, f"final LBFGS beta={beta}")
            print(f"  Final LBFGS beta={beta}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")
    
    # Final coordinate descent
    if time.monotonic() < p5_deadline - 3:
        sc, a_opt = active_set_coord_descent(best_a, n_sweeps=10, top_k_lags=20,
                                             time_limit_s=min(15.0, p5_deadline - time.monotonic() - 2),
                                             t_start=t_start)
        try_update(sc, a_opt, "final active CD")
    
    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128
    
    # Normalize to a reasonable scale (the evaluator clips to [0, 1000])
    # Keep values in a safe range
    s = sum(out)
    if s > 1000.0 * len(out) * 0.5:
        out = [x * 1000.0 / s * 0.1 for x in out]
        s = sum(out)
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Sum: {s:.4f}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
