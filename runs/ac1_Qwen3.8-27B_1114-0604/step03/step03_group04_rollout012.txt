import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a, L=None):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    M = 2 * n - 1
    if L is None:
        L = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def conv_with_a(a, delta, L=None):
    """Compute conv(a, delta) using FFT."""
    n = len(a)
    if L is None:
        M = 2 * n - 1
        L = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, L)
    D = np.fft.rfft(delta, L)
    return np.fft.irfft(A * D, L)[:2*n-1]


def convolution_newton_step(a, n_peaks=50, tau_quantile=0.99, reg=1e-8):
    """
    Newton step to flatten the auto-convolution.
    Returns (delta, score_improvement_estimate).
    """
    n = len(a)
    M = 2 * n - 1
    L = 1 << (M - 1).bit_length()
    
    c = fast_conv(a, L)
    S = float(a.sum())
    if S < 1e-12:
        return np.zeros(n), 0.0
    
    # Target: uniform distribution of convolution
    c_mean = S * S / M
    # Focus on the peaks: use a threshold
    threshold = np.quantile(c, 1 - n_peaks / M)
    
    # Excess above target at peaks
    excess = np.maximum(c - c_mean, 0)
    # Only target the top n_peaks
    top_mask = c > threshold
    d = np.zeros(M)
    d[top_mask] = excess[top_mask]
    # Normalize so that sum(d) = S^2 * n_peaks / M (roughly)
    d_sum = d.sum()
    if d_sum < 1e-12:
        return np.zeros(n), 0.0
    
    # We want conv(a, delta) ≈ -d/2 (to reduce the peaks)
    # In frequency domain:
    A_freq = np.fft.rfft(a, L)
    d_freq = np.fft.rfft(d, L)
    
    # Solve: A_freq * delta_freq = -d_freq / 2
    # Regularize to avoid division by zero
    denom = A_freq * A_freq.conjug() + reg  # Use |A|^2 for stability
    # Actually, we want A_freq * delta_freq = -d_freq/2
    # So delta_freq = -d_freq / (2 * A_freq)
    # But A_freq is complex, so let's use the pseudo-inverse approach
    # delta_freq = -d_freq * conj(A_freq) / (|A_freq|^2 + reg)
    delta_freq = -d_freq * np.conjugate(A_freq) / (denom)
    delta = np.fft.irfft(delta_freq, L)[:n].real
    
    return delta, float(d.sum())


def conv_newton_phase(a, n_iters=200, max_step=0.5):
    """
    Iterative convolution Newton method.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    S0 = a.sum()
    
    for it in range(n_iters):
        delta, _ = convolution_newton_step(a)
        dn = np.linalg.norm(delta)
        if dn < 1e-15:
            break
        
        # Line search: try a few step sizes
        best_t = 0.0
        best_trial_sc = compute_score(a)
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = np.clip(a + t * delta, 0.0, 1000.0)
            if a_trial.sum() < 0.01:
                continue
            sc_trial = compute_score(a_trial)
            if sc_trial < best_trial_sc:
                best_trial_sc = sc_trial
                best_t = t
        
        if best_t == 0.0:
            # Try negative direction too
            for t in [0.1, 0.05, 0.02]:
                a_trial = np.clip(a - t * delta, 0.0, 1000.0)
                if a_trial.sum() < 0.01:
                    continue
                sc_trial = compute_score(a_trial)
                if sc_trial < best_trial_sc:
                    best_trial_sc = sc_trial
                    best_t = -t
        
        if best_t == 0.0:
            break
        
        a = np.clip(a + best_t * delta, 0.0, 1000.0)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if it % 20 == 0:
            print(f"    conv_newton iter {it}: score={sc:.8f}, |delta|={dn:.4f}, t={best_t:.3f}")
    
    return best_sc, best_a


def peak_pressure_search(a, n_iters=500, rng=None):
    """
    Peak-pressure local search: iteratively reduce the peak of the convolution
    by adjusting the contributing coefficients.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    M = 2 * n - 1
    L = 1 << (M - 1).bit_length()
    
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a, L)
        S = float(a.sum())
        sc = 2.0 * n * c.max() / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        k_star = int(np.argmax(c))
        
        # Find contributing indices
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        if j_end < j_start:
            break
        
        # Contribution of each a[j] to c[k_star]
        # c[k_star] = sum_j a[j] * a[k_star - j]
        # So contribution of a[j] is a[k_star - j] (and a[j] is the "other" factor)
        contrib = np.zeros(n)
        idx_range = range(j_start, j_end + 1)
        contrib[j_start:j_end+1] = a[k_star - np.array(idx_range)]
        
        # Total contribution
        total_contrib = contrib.sum()
        if total_contrib < 1e-12:
            break
        
        # Strategy: reduce high-contributing elements, increase low-contributing ones
        # Use a "pressure" approach
        normalize = contrib / (total_contrib + 1e-12)
        
        # Identify "high pressure" and "low pressure" zones
        high_thresh = np.quantile(contrib[contrib > 0], 0.7)
        low_thresh = np.quantile(contrib[contrib > 0], 0.3)
        
        high_mask = contrib > high_thresh
        low_mask = (contrib < low_thresh) & (contrib >= 0)
        
        # Also consider elements not contributing to the peak at all
        non_contrib_mask = (contrib == 0)
        
        n_high = high_mask.sum()
        n_low = (low_mask | non_contrib_mask).sum()
        
        if n_high == 0 or n_low == 0:
            break
        
        # Amount to transfer (adaptive)
        peak_excess = c[k_star] - c.mean()
        transfer_frac = min(0.1, max(0.01, peak_excess / (S * S) * n))
        
        total_reduce = 0.0
        for j in np.where(high_mask)[0]:
            amt = a[j] * transfer_frac * (normalize[j] / normalize[high_mask].mean() if normalize[high_mask].mean() > 0 else 1.0)
            a[j] -= amt
            a[j] = max(a[j], 0.0)
            total_reduce += amt
        
        # Distribute to low-pressure and non-contributing elements
        target_mask = low_mask | non_contrib_mask
        n_target = target_mask.sum()
        if n_target > 0 and total_reduce > 0:
            per = total_reduce / n_target
            for j in np.where(target_mask)[0]:
                a[j] += per
        
        a = np.clip(a, 0.0, 1000.0)
        
        # Occasionally add random perturbation to escape local minima
        if it % 50 == 49:
            a += 0.01 * rng.standard_normal(n)
            a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a


def lse_objective_and_grad(a, beta, L=None):
    """
    LogSumExp-smoothed objective and its gradient.
    """
    n = len(a)
    M = 2 * n - 1
    if L is None:
        L = 1 << (M - 1).bit_length()
    
    c = fast_conv(a, L)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerically stable logsumexp
    max_scaled = scaled.max()
    log_sum = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
    LSE = c_max + (log_sum - max_scaled) / beta  # Wait, this is wrong
    
    # Correct LSE: c_max + log(sum(exp(beta*(c-c_max))))/beta
    # = c_max + (max_scaled + log(sum(exp(scaled - max_scaled))))/beta
    LSE = c_max + (max_scaled + np.log(np.exp(scaled - max_scaled).sum())) / beta
    
    score = 2.0 * n * LSE / (S * S)
    
    # Gradient of LSE with respect to a[j]
    # d(LSE)/da[j] = sum_k p[k] * dc[dk]/da[j] where p[k] = exp(beta*(c[k]-c_max))/sum(...)
    p = np.exp(scaled - max_scaled) / np.exp(scaled - max_scaled).sum()
    
    # dc[k]/da[j] = a[k-j] + a[j-k] (for valid indices, accounting for the two terms in convolution)
    # Actually: c[k] = sum_i a[i]*a[k-i]
    # dc[k]/da[j] = a[k-j] + a[j-k] (if both indices valid)
    # But in linear convolution: c[k] = sum_i a[i]*a[k-i] where a[i]=0 outside [0,n-1]
    # dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    # Note: the second term appears because da[k-i]/da[j] = 1 when k-i=j, i.e., i=k-j
    # And da[i]/da[j] = 1 when i=j, so the second term is a[k-j] when k-j is valid... 
    # Wait, let me be more careful.
    # c[k] = sum_i a[i] * a[k-i]
    # dc[k]/da[j] = a[k-j] (from the first factor, when i=j) + a[j] * ... 
    # No: dc[k]/da[j] = sum_i [da[i]/da[j] * a[k-i] + a[i] * da[k-i]/da[j]]
    # = a[k-j] (when i=j, first term) + a[j] (when k-i=j, i.e., i=k-j, second term)
    # So dc[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j] (if 0 <= k-j < n)
    # Wait, that's only if k-j is a valid index. If k-j < 0 or k-j >= n, the term is 0.
    # But also, when i = k-j is valid AND i = j (i.e., k = 2j), we get:
    # dc[k]/da[j] = a[k-j] (first term) + a[k-j] (second term) but these are the same when k-j=j
    # So dc[k]/da[j] = 2*a[k-j] when 0 <= k-j < n, else 0.
    # Actually no: when i=j, the first term gives a[k-j]. When i=k-j (and k-j != j), the second term gives a[j].
    # When k-j = j (i.e., k=2j), both terms give a[j], so total is 2*a[j].
    # When k-j != j: first term (i=j) gives a[k-j], second term (i=k-j) gives a[j].
    # So dc[k]/da[j] = a[k-j] + a[j] when 0 <= k-j < n AND k-j != j
    # And dc[k]/da[j] = 2*a[j] when k = 2j (and 0 <= j < n)
    
    # This is getting complicated. Let me use the FFT-based approach from the previous code.
    # Gradient = 2 * (p convolved with reversed a), appropriately indexed.
    
    # From the previous code:
    # rev_a = a[::-1]
    # M2 = 3*n - 2
    # M2_fft = 1 << (M2-1).bit_length()
    # RA = np.fft.rfft(rev_a, M2_fft)
    # P = np.fft.rfft(p, M2_fft)
    # conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    # g = 2.0 * conv_res
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    g = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * LSE / S3)
    
    return score, grad


def run_lbfgs_logparam(a0, beta, maxiter=5000, maxfun=100000, L=None):
    """
    L-BFGS optimization using log-parameterization: a = exp(x)
    This ensures non-negativity.
    """
    n = len(a0)
    
    def obj_and_grad(x):
        a = np.exp(x)
        a = np.clip(a, 1e-10, 1000.0)
        sc, grad_a = lse_objective_and_grad(a, beta, L=L)
        # Chain rule: grad_x = grad_a * a
        grad_x = grad_a * a
        return sc, grad_x
    
    x0 = np.log(np.maximum(a0, 1e-10))
    
    res = minimize(
        obj_and_grad, x0,
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.exp(res.x)
    a_out = np.clip(a_out, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def simulated_annealing(a0, n_iters=2000, T0=0.01, Tend=0.001, rng=None):
    """
    Simulated annealing for the convolution peak minimization problem.
    """
    if rng is None:
        rng = np.random.default_rng(123)
    n = len(a0)
    M = 2 * n - 1
    L = 1 << (M - 1).bit_length()
    
    a = a0.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    S = a.sum()
    
    for it in range(n_iters):
        # Annealing schedule
        progress = it / max(1, n_iters - 1)
        T = T0 * (Tend / T0) ** progress
        
        # Generate perturbation
        strategy = rng.integers(0, 4)
        if strategy == 0:
            # Gaussian perturbation
            scale = T * S / n
            a_new = a + scale * rng.standard_normal(n)
        elif strategy == 1:
            # Swap perturbation: move mass from one position to another
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                amt = min(a[i], T * S / n)
                a_new = a.copy()
                a_new[i] -= amt
                a_new[j] += amt
            else:
                a_new = a.copy()
        elif strategy == 2:
            # Block perturbation
            block_size = max(1, int(n * rng.uniform(0.01, 0.1)))
            start = rng.integers(0, n - block_size)
            a_new = a.copy()
            a_new[start:start+block_size] *= (1 + T * rng.standard_normal())
        else:
            # Spectral perturbation
            A = np.fft.rfft(a, L)
            noise = T * 0.1 * (rng.standard_normal(len(A)) + 1j*rng.standard_normal(len(A)))
            A_noisy = A + noise
            A_noisy[0] = A[0]  # Preserve sum
            a_new = np.fft.irfft(A_noisy, L)[:n]
        
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        if a_new.sum() < 0.01:
            continue
        
        sc_new = compute_score(a_new)
        sc_cur = compute_score(a)
        
        if sc_new < sc_cur:
            a = a_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()
        else:
            # Accept with probability exp(-(sc_new - sc_cur) / T)
            delta = sc_new - sc_cur
            if delta > 0 and delta < 1.0:
                if rng.random() < np.exp(-delta / max(T, 1e-10)):
                    a = a_new
    
    return best_sc, best_a
