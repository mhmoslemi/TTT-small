"""
Multi-stage optimizer for minimum peak autocorrelation of non-negative sequences.
Uses L-BFGS with LSE-smoothed objective at increasing precision, combined with
a lag-focused subgradient method and local search.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_shifted / exp_shifted.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_lse = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_lse / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgrad_lag_focus(a, n_iters=500, top_k=3):
    """Subgradient method focusing on top-k autocorrelation peaks."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Get top K lags
        top_lags = np.argsort(c)[::-1][:top_k]
        
        # Compute subgradient (sum over top K lags)
        sg = np.zeros(n)
        for k in top_lags:
            k = int(k)
            # dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            # Vectorized:
            # First term: for j where 0<=k-j<n, i.e., k-n+1<=j<=k
            lo1 = max(0, k - n + 1)
            hi1 = min(k, n - 1)
            if hi1 >= lo1:
                # a[k-j] for j in [lo1, hi1] => indices k-hi1 to k-lo1 in a
                sg[lo1:hi1+1] += a[k-hi1:k-lo1+1][::-1]
            
            # Second term: for j where 0<=j-k<n, i.e., k<=j<=k+n-1
            lo2 = max(0, k)
            hi2 = min(k + n - 1, n - 1)
            if hi2 >= lo2:
                # a[j-k] for j in [lo2, hi2] => indices j-k in a
                sg[lo2:hi2+1] += a[lo2-k:hi2-k+1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Step size with backtracking
        step = min(1.0 / (gn * gn + 1e-10), 0.1)
        for _ in range(30):
            a_new = np.clip(a - step * grad, 0.0, 1000.0)
            c_new = fast_conv(a_new)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            sc_new = 2.0 * n * float(c_new.max()) / (S_new * S_new)
            if sc_new < sc * (1 - 1e-6):
                a = a_new
                sc = sc_new
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5
    
    return best_sc, best_a


def local_search(a, rng, n_trials=50, beta=2000):
    """Local search with diverse perturbations + L-BFGS polish."""
    n = len(a)

        strategy = trial % 5

            k_star = int(np.argmax(c))
            contrib = np.zeros(n)
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            if hi >= lo:
                contrib[lo:hi+1] = a[k_star - hi:k_star - lo + 1][::-1]
                contrib[lo:hi+1] += a[lo - k_star:hi - k_star + 1] if lo >= k_star else 0
            # Find top contributors and reduce them
            top_count = max(5, n // 50)
            top_idx = np.argsort(contrib)[::-1][:top_count]
            a_pert = a.copy()
            red_amt = a_pert[top_idx] * 0.05
            a_pert[top_idx] -= red_amt
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            # Redistribute to random other positions
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            a_pert[other_idx] += red_amt.sum() / len(other_idx)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:

            M = 2 * n - 1
            M_fft = 1 << (M - 1).bit_length()
            A = np.fft.rfft(a, M_fft)

            A_new[0] = A[0]  # Preserve sum

            a_pert[i:i+block_size], a_pert[j:j+block_size] = a[j:j+block_size].copy(), a[i:i+block_size].copy()
        else:

            best_sc = sc

    res = minimize(
        lse_obj_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgrad_lag_focus(a, n_iters=500, top_k=3):
    """Subgradient method focusing on top-k autocorrelation peaks with line search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_lags = np.argsort(c)[::-1][:top_k]

        sg = np.zeros(n)
        for k in top_lags:
            k = int(k)
            lo1 = max(0, k - n + 1)
            hi1 = min(k, n - 1)
            if hi1 >= lo1:
                sg[lo1:hi1 + 1] += a[k - hi1:k - lo1 + 1][::-1]
            lo2 = max(0, k)
            hi2 = min(k + n - 1, n - 1)
            if hi2 >= lo2:
                sg[lo2:hi2 + 1] += a[lo2 - k:hi2 - k + 1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(1.0 / (gn * gn + 1e-10), 0.1)
        accepted = False
        for _ in range(25):
            a_new = np.clip(a - step * grad, 0.0, 1000.0)
            c_new = fast_conv(a_new)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            sc_new = 2.0 * n * float(c_new.max()) / (S_new * S_new)
            if sc_new < sc * (1 - 1e-8):
                a = a_new
                sc = sc_new
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                accepted = True
                break
            step *= 0.5
        if not accepted:
            a = np.clip(a - 0.005 / (gn + 1e-10) * grad, 0.0, 1000.0)

    return best_sc, best_a


def multiplicative_update(a, n_iters=500):
    """Multiplicative-weight style updates that preserve non-negativity."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = 0.1

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        k_star = int(np.argmax(c))
        sg = np.zeros(n)
        lo1 = max(0, k_star - n + 1)
        hi1 = min(k_star, n - 1)
        if hi1 >= lo1:
            sg[lo1:hi1 + 1] += a[k_star - hi1:k_star - lo1 + 1][::-1]
        lo2 = max(0, k_star)
        hi2 = min(k_star + n - 1, n - 1)
        if hi2 >= lo2:
            sg[lo2:hi2 + 1] += a[lo2 - k_star:hi2 - k_star + 1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        eta_cur = eta * (1.0 / (1.0 + 0.01 * it))
        log_update = -eta_cur * grad / (gn + 1e-10)
        log_update = np.clip(log_update, -2.0, 2.0)
        a = a * np.exp(log_update)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 1e-12:
            a = best_a.copy()

    return best_sc, best_a


def local_search(a, rng, n_trials, beta):
    """Local search with diverse perturbations + L-BFGS polish."""
    n = len(a)
    best_sc = compute_score(a)
    best_a = a.copy()

    for trial in range(n_trials):
        strategy = trial % 5
        if strategy == 0:
            scale = 0.01 + 0.03 * rng.random()
            a_pert = a * (1.0 + scale * rng.standard_normal(n))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            contrib = np.zeros(n)
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            if hi >= lo:
                contrib[lo:hi + 1] = a[k_star - hi:k_star - lo + 1][::-1]
            lo2 = max(0, k_star)
            hi2 = min(k_star + n - 1, n - 1)
            if hi2 >= lo2:
                contrib[lo2:hi2 + 1] += a[lo2 - k_star:hi2 - k_star + 1]
            top_count = max(5, n // 100)
            top_idx = np.argsort(contrib)[::-1][:top_count]
            a_pert = a.copy()
            red_amt = a_pert[top_idx] * 0.05
            a_pert[top_idx] -= red_amt
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            other_mask = np.ones(n, dtype=bool)
            other_mask[top_idx] = False
            other_idx = np.where(other_mask)[0]
            if len(other_idx) > 0 and red_amt.sum() > 0:
                a_pert[other_idx] += red_amt.sum() / len(other_idx)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            M_fft = 1 << (2 * n - 1).bit_length()
            A = np.fft.rfft(a, M_fft)
            phase_noise = 0.01 * rng.standard_normal(len(A))
            A_new = A * np.exp(1j * phase_noise)
            A_new[0] = A[0]
            a_pert = np.real(np.fft.irfft(A_new, M_fft))[:n]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            block_size = max(2, n // 300)
            i = int(rng.integers(0, n - 2 * block_size))
            j = int(rng.integers(0, n - 2 * block_size))
            if abs(i - j) < block_size * 2:
                continue
            a_pert = a.copy()
            tmp = a_pert[i:i + block_size].copy()
            a_pert[i:i + block_size] = a_pert[j:j + block_size]
            a_pert[j:j + block_size] = tmp
        else:
            x = np.arange(n, dtype=np.float64)
            f = 1.0 + 5.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            pert = 0.02 * np.cos(np.pi * f * x * x / (2.0 * n) + phase)
            a_pert = a * (1.0 + pert)
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=400, maxfun=8000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    [LS trial {trial} s={strategy}] {sc:.8f}")

    return best_sc, best_a


def spectral_refine(a, n_iters=200, lam=0.5):
    """Iterative spectral flattening to reduce peak autocorrelation."""
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    a = a.astype(np.float64).copy()
    S0 = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for i in range(n_iters):
        cur_lam = lam * (1.0 - 0.5 * i / n_iters)
        A = np.fft.fft(a, M_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_indices = list(range(1, M_fft // 2 + 1))
        if M_fft % 2 == 0:
            non_dc_indices = [k for k in non_dc_indices if k != M_fft // 2]
        else:
            non_dc_indices = list(range(1, M_fft // 2 + 1))
        non_dc_mag = mag[non_dc_indices]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        for k in non_dc_indices:
            new_mag[k] = (1.0 - cur_lam) * mag[k] + cur_lam * target
            new_mag[M_fft - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (S0 / S)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-stage optimizer: L-BFGS (LSE) at increasing beta, subgradient with lag focus,
    multiplicative updates, spectral refinement, and local search.
    Focuses on refining the parent sequence with careful time management.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 100:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        a = parent.copy()
        n = len(a)
        print(f"Starting from parent: n={n}")
    else:
        n = 4096
        a = np.ones(n) * 0.5
        print(f"No parent found, using uniform n={n}")

    best_sc = compute_score(a)
    best_a = a.copy()
    print(f"Initial score: {best_sc:.8f}")

    # ================================================================
    # Stage 1: L-BFGS with increasing beta (primary refinement)
    # ================================================================
    print("\n--- Stage 1: L-BFGS refinement ---")
    beta_schedule = [30, 100, 300, 1000, 3000, 10000, 30000]
    a_cur = a.copy()

    for beta in beta_schedule:
        if time.monotonic() > deadline - 300:
            break
        t0 = time.monotonic()
        remaining = deadline - time.monotonic() - 300
        if remaining < 10:
            break
        # Allocate equal time budget per beta stage
        num_betas_left = len(beta_schedule) - beta_schedule.index(beta)
        time_per_stage = min(remaining / num_betas_left, 300)
        est_per_iter = max(0.01, n / 200000.0)
        maxiter = min(8000, int(time_per_stage / max(est_per_iter, 0.005)))
        maxfun = min(200000, max(2000, maxiter * 25))
        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=max(100, maxiter), maxfun=maxfun)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"  [L-BFGS beta={beta}] {sc:.8f} ({time.monotonic()-t0:.1f}s) *")
        else:
            print(f"  [L-BFGS beta={beta}] {sc:.8f} ({time.monotonic()-t0:.1f}s)")
        a_cur = a_opt

    print(f"After Stage 1: best = {best_sc:.8f}")

    # ================================================================
    # Stage 2: Subgradient with lag focus
    # ================================================================
    print("\n--- Stage 2: Subgradient (lag-focused) ---")
    if time.monotonic() < deadline - 200:
        for top_k in [1, 2, 3]:
            if time.monotonic() > deadline - 100:
                break
            t0 = time.monotonic()
            a_sg = best_a.copy()
            # Adaptive number of iterations based on time
            time_budget = min(60, deadline - time.monotonic() - 100)
            est_per_iter = 0.02 * (n / 4096)
            n_iters = min(1000, max(100, int(time_budget / max(est_per_iter, 0.01))))
            sc_sg, a_sg_opt = subgrad_lag_focus(a_sg, n_iters=n_iters, top_k=top_k)
            if sc_sg < best_sc:
                best_sc = sc_sg
                best_a = a_sg_opt.copy()
                print(f"  [Subgrad K={top_k}] {sc_sg:.8f} ({time.monotonic()-t0:.1f}s) *")
            else:
                print(f"  [Subgrad K={top_k}] {sc_sg:.8f} ({time.monotonic()-t0:.1f}s)")
                best_a = a_sg_opt  # Keep the subgradient result for next stage

    print(f"After Stage 2: best = {best_sc:.8f}")

    # ================================================================
    # Stage 3: Multiplicative updates
    # ================================================================
    print("\n--- Stage 3: Multiplicative updates ---")
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        time_budget = min(60, deadline - time.monotonic() - 150)
        est_per_iter = 0.02 * (n / 4096)
        n_iters_mu = min(1000, max(200, int(time_budget / max(est_per_iter, 0.01))))
        sc_mu, a_mu = multiplicative_update(best_a, n_iters=n_iters_mu)
        if sc_mu < best_sc:
            best_sc = sc_mu
            best_a = a_mu.copy()
            print(f"  [Mult update] {sc_mu:.8f} ({time.monotonic()-t0:.1f}s) *")
        else:
            print(f"  [Mult update] {sc_mu:.8f} ({time.monotonic()-t0:.1f}s)")

    print(f"After Stage 3: best = {best_sc:.8f}")

    # ================================================================
    # Stage 4: Spectral refinement
    # ================================================================
    print("\n--- Stage 4: Spectral refinement ---")
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        for lam in [0.3, 0.5, 0.7]:
            if time.monotonic() > deadline - 60:
                break
            sc_sp, a_sp = spectral_refine(best_a, n_iters=100, lam=lam)
            if sc_sp < best_sc:
                best_sc = sc_sp
                best_a = a_sp.copy()
                print(f"  [Spectral lam={lam}] {sc_sp:.8f} ({time.monotonic()-t0:.1f}s) *")
            t0 = time.monotonic()

    print(f"After Stage 4: best = {best_sc:.8f}")

    # ================================================================
    # Stage 5: Local search with diverse perturbations
    # ================================================================
    print("\n--- Stage 5: Local search ---")
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic() - 60
        n_trials = max(5, int(remaining / 30))
        n_trials = min(n_trials, 40)
        t0 = time.monotonic()
        sc_ls, a_ls = local_search(best_a, rng, n_trials=n_trials, beta=5000)
        if sc_ls < best_sc:
            best_sc = sc_ls
            best_a = a_ls.copy()
            print(f"  [Local search] {sc_ls:.8f} ({time.monotonic()-t0:.1f}s) *")
        else:
            print(f"  [Local search] {sc_ls:.8f} ({time.monotonic()-t0:.1f}s)")

    print(f"After Stage 5: best = {best_sc:.8f}")

    # ================================================================
    # Stage 6: Final high-beta polish
    # ================================================================
    print("\n--- Stage 6: Final polish ---")
    if time.monotonic() < deadline - 10:
        for beta in [10000, 50000]:
            if time.monotonic() > deadline - 5:
                break
            t0 = time.monotonic()
            remaining = deadline - time.monotonic() - 5
            if remaining < 5:
                break
            maxiter = min(5000, int(remaining * 0.4 / max(0.01, n / 200000.0)))
            maxfun = min(100000, max(1000, maxiter * 20))
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=max(100, maxiter), maxfun=maxfun)
            if sc < best_sc:
                best_sc = sc
                best_a = a_opt.copy()
                print(f"  [Final beta={beta}] {sc:.8f} ({time.monotonic()-t0:.1f}s) *")

    # ================================================================
    # Stage 7: Final subgradient squeeze
    # ================================================================
    print("\n--- Stage 7: Final subgradient ---")
    if time.monotonic() < deadline - 3:
        t0 = time.monotonic()
        n_iters_final = min(500, max(50, int((deadline - time.monotonic() - 3) / 0.05)))
        sc_f, a_f = subgrad_lag_focus(best_a, n_iters=n_iters_final, top_k=1)
        if sc_f < best_sc:
            best_sc = sc_f
            best_a = a_f.copy()
            print(f"  [Final subgrad] {sc_f:.8f} ({time.monotonic()-t0:.1f}s) *")

    # ================================================================
    # Final output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_sc:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
