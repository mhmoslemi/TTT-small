"""
Step function optimization for fixed length 8192 using LSE-smoothed gradient methods,
LSE-weighted multiplicative equalization, and targeted coordinate descent.

Key improvements over previous approach:

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Cross convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient. Returns (score, grad)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient: dL/da_j = 2 * sum_k p[k] * a[k-j] (cross-correlation)
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    g = 2.0 * conv_res[:n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_multiplicative_update(a, beta, alpha=0.5, n_iters=50, deadline=None):
    """
    LSE-weighted multiplicative update.
    Uses continuous softmax weights on convolution peaks (not binary active set).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break
        # Softmax weights
        scaled = beta * (c - c_max)
        exp_scaled = np.exp(scaled)
        p = exp_scaled / exp_scaled.sum()
        # Compute pressure h_j = 2 * sum_k p[k] * a[k-j]
        rev_a = a[::-1]
        RA = np.fft.rfft(rev_a, M_fft)
        P = np.fft.rfft(p, M_fft)
        cross = np.fft.irfft(RA * P, M_fft)
        h = 2.0 * cross[:n]
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()
        # Damped multiplicative update
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)

            break

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Adaptive alpha: reduce if oscillating
        if it > 10 and it % 10 == 0:
            alpha *= 0.85

    return best_sc, best_a


def run_lbfgs_lse(a0, beta, maxiter=3000, maxfun=50000, deadline=None):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x, b):
        return lse_score_grad(x, b)

    options = {'ftol': 1e-14, 'gtol': 1e-12, 'maxiter': maxiter,
               'maxfun': maxfun, 'maxls': 30}
    if deadline is not None:
        remaining = max(1.0, deadline - time.monotonic())
        options['maxfun'] = min(maxfun, max(50, int(remaining * 150)))
        options['maxiter'] = min(maxiter, max(50, int(remaining * 30)))

    try:
        res = minimize(obj_and_grad, a0, args=(beta,),
                       method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out


def coordinate_descent(a, deadline=None, n_full_passes=2):

    Coordinate descent: for each coordinate, try to decrease the score.

    contrib = np.zeros(n)

    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    order = np.argsort(contrib)[::-1]

    delta_base = 0.01
    for path in range(n_full_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

            if deadline is not None and time.monotonic() > deadline:
                break
            a_orig = a[idx]
            if a_orig < 1e-10:

                a[idx] = a_orig * frac
                a[idx] = max(0.0, min(1000.0, a[idx]))
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                else:
                    a[idx] = a_orig
                    break
            # Recompute contribution order every full pass
            if path == 0 and idx == order[0]:
                # Update peak info after significant changes
                pass

    return best_sc, a


def random_coordinate_search(a, rng, n_trials=500, deadline=None):
    """Random coordinate search with geometric step sizes."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_sc = compute_score(a)

    for t in range(n_trials):
        if deadline is not None and time.monotonic() > deadline:
            break
        idx = rng.integers(n)
        a_orig = a[idx]
        if a_orig < 1e-10:
            continue
        frac = 0.5 + rng.random()  # Uniform in [0.5, 1.5]
        a[idx] = a_orig * frac
        a[idx] = max(0.0, min(1000.0, a[idx]))
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
        else:
            a[idx] = a_orig

    return best_sc, a


def block_local_search(a, rng, deadline=None, n_blocks=100):
    """Perturb random blocks of coordinates and keep improvement."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_sc = compute_score(a)
    block_size = max(4, n // n_blocks)

    for _ in range(n_blocks * 5):
        if deadline is not None and time.monotonic() > deadline:
            break
        start = rng.integers(0, n - block_size)
        a_orig = a[start:start + block_size].copy()
        # Random multiplicative perturbation
        pert = 1.0 + 0.1 * rng.standard_normal(block_size)
        a[start:start + block_size] = a_orig * pert
        a[start:start + block_size] = np.clip(a[start:start + block_size], 0.0, 1000.0)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
        else:
            a[start:start + block_size] = a_orig

    return best_sc, a


def subgradient_polish(a, total_iters=200, deadline=None):
    """Projected subgradient descent with Polyak+ step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0  # Theoretical lower bound

    for it in range(total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max conv
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        # Polyak+ step size
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 1.0 / (np.sqrt(it + 1)))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []

    # Parent (resampled if needed)
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if len(p) == n:
            inits.append(np.clip(p, 0.01, 1000.0).copy())
        elif len(p) >= 64:
            p_res = np.interp(np.linspace(0, len(p) - 1, n), np.arange(len(p)), p)
            inits.append(np.clip(p_res, 0.01, 1000.0).copy())

    x = np.arange(n, dtype=np.float64) / n

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Raised cosine window (tapers to ~0 at edges - good for aperiodic autocorr)
    raised = 0.5 * (1 - np.cos(2 * np.pi * x))
    inits.append(0.1 + 0.9 * raised)

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    inits.append(0.05 + 0.95 * hann ** 2)

    # Cosine modulations on top of raised cosine base
    base = 0.3 + 0.7 * raised
    for freq in [2, 4, 6, 10]:
        inits.append(np.clip(base * (1 + 0.3 * np.cos(2 * np.pi * freq * x)), 0.01, 1000.0))

    # Gaussian center
    gc = np.exp(-2.0 * (2 * x - 1) ** 2)
    inits.append(0.1 + 0.9 * gc)

    # Two-peaked
    two_peak = np.exp(-4 * (x - 0.25) ** 2) + np.exp(-4 * (x - 0.75) ** 2)
    inits.append(0.1 + 0.9 * two_peak / two_peak.max())

    # Periodic structure: approximate Barker-like with cosines
    for f1, f2, amp in [(1, 3, 0.4), (2, 5, 0.3), (1, 7, 0.35)]:
        sig = np.cos(2 * np.pi * f1 * x) + amp * np.cos(2 * np.pi * f2 * x)
        sig = 1.0 + sig / (1.0 + amp)  # Make non-negative
        inits.append(np.clip(0.1 + 0.9 * sig / sig.max(), 0.01, 1000.0))

    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.5 * rng.random(n), 0.01, 1000.0))

    # Block structure
    for nb in [8, 16, 32]:
        per = n // nb
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence of length 8192 minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    n = 8192
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = p
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(n) * 0.5
    best_a_sc = compute_score(best_a)
    if best_a_sc < best_score:
        best_score = best_a_sc

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.array(a, dtype=np.float64).copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    print(f"n={n}, parent={'yes' if parent is not None else 'no'}")
    print(f"Initial best: {best_score:.8f}")

    # ==================================================================
    # Phase 1: LSE continuation from diverse inits (~50% time)
    # ==================================================================
    phase1_dl = t_start + 0.50 * budget_s
    print(f"\n{'='*60}\nPhase 1: LSE Continuation (until t={phase1_dl-t_start:.0f}s)\n{'='*60}")

    inits = build_initializations(n, rng, parent)
    # Sort by initial score (best first)
    init_scores = [(compute_score(a), i, a) for i, a in enumerate(inits)]
    init_scores.sort(key=lambda x: x[0])

    beta_schedule = [20, 50, 100, 200, 500, 1000]

    for rank, (isc, idx, a0) in enumerate(init_scores[:6]):  # Top 6 inits
        if time.monotonic() > phase1_dl:
            break
        print(f"\n  Init {rank} (score={isc:.6f}):")
        a_cur = a0.copy()
        cur_best = isc
        for beta in beta_schedule:
            if time.monotonic() > phase1_dl:
                break
            remaining = phase1_dl - time.monotonic()
            if remaining < 5:
                break

            # L-BFGS step
            sc, a_opt = run_lbfgs_lse(a_cur, beta, maxiter=2000, maxfun=40000,
                                       deadline=phase1_dl)
            improved = report(sc, a_opt, f"b={beta}")
            if improved:
                a_cur = a_opt
                cur_best = sc
            else:
                # Try LSE multiplicative update instead
                sc2, a_eq = lse_multiplicative_update(a_cur, beta, alpha=0.4,
                                                       n_iters=30, deadline=phase1_dl)
                if report(sc2, a_eq, f"lse-mul b={beta}"):
                    a_cur = a_eq
                    cur_best = sc2

                if not improved and sc2 >= cur_best and beta >= 100:
                    break  # No progress, move to next init

    print(f"\nAfter Phase 1: best={best_score:.8f}")

    # ==================================================================
    # Phase 2: Alternating LSE-multiplicative + L-BFGS (~25% time)
    # ==================================================================
    phase2_dl = t_start + 0.75 * budget_s
    print(f"\n{'='*60}\nPhase 2: Alternating Refinement (until t={phase2_dl-t_start:.0f}s)\n{'='*60}")

    a_cur = best_a.copy()
    for rnd in range(12):
        if time.monotonic() > phase2_dl:
            break
        # Step A: LSE multiplicative update
        beta_mu = 50 + 50 * rnd
        sc, a_eq = lse_multiplicative_update(a_cur, beta_mu, alpha=0.4,
                                             n_iters=40, deadline=phase2_dl)
        report(sc, a_eq, f"alt-{rnd} mul")
        a_cur = a_eq

        # Step B: L-BFGS polish
        if time.monotonic() < phase2_dl - 10:
            beta_lb = 100 + 200 * rnd
            sc, a_opt = run_lbfgs_lse(a_cur, beta_lb, maxiter=2000, maxfun=40000,
                                       deadline=phase2_dl)
            report(sc, a_opt, f"alt-{rnd} lbfgs")
            a_cur = a_opt

    print(f"\nAfter Phase 2: best={best_score:.8f}")

    # ==================================================================
    # Phase 3: Coordinate descent + local search (~15% time)
    # ==================================================================
    phase3_dl = t_start + 0.90 * budget_s
    print(f"\n{'='*60}\nPhase 3: Local Search (until t={phase3_dl-t_start:.0f}s)\n{'='*60}")

    a_cur = best_a.copy()

    # Coordinate descent (full passes)
    if time.monotonic() < phase3_dl - 30:
        sc, a_cd = coordinate_descent(a_cur, deadline=phase3_dl - 60, n_full_passes=2)
        report(sc, a_cd, "coord-desc")
        a_cur = a_cd

    # Random coordinate search
    if time.monotonic() < phase3_dl - 20:
        sc, a_rcs = random_coordinate_search(a_cur, rng, n_trials=3000,
                                              deadline=phase3_dl - 40)
        report(sc, a_rcs, "rand-coord")
        a_cur = a_rcs

    # Block local search
    if time.monotonic() < phase3_dl - 10:
        sc, a_bl = block_local_search(a_cur, rng, deadline=phase3_dl)
        report(sc, a_bl, "block-search")
        a_cur = a_bl

    # Another coordinate descent pass
    if time.monotonic() < phase3_dl - 5:
        sc, a_cd2 = coordinate_descent(a_cur, deadline=phase3_dl, n_full_passes=1)
        report(sc, a_cd2, "coord-desc-2")
        a_cur = a_cd2

    print(f"\nAfter Phase 3: best={best_score:.8f}")

    # ==================================================================
    # Phase 4: Final polish (~10% time)
    # ==================================================================
    print(f"\n{'='*60}\nPhase 4: Final Polish\n{'='*60}")

    a_cur = best_a.copy()

    # High-beta L-BFGS
    for beta in [2000, 5000]:
        if time.monotonic() > deadline - 15:
            break
        sc, a_opt = run_lbfgs_lse(a_cur, beta, maxiter=3000, maxfun=50000,
                                   deadline=deadline - 15)
        report(sc, a_opt, f"final-b{beta}")
        a_cur = a_opt

    # Subgradient polish
    if time.monotonic() < deadline - 10:
        sc, a_sg = subgradient_polish(a_cur, total_iters=100, deadline=deadline - 10)
        report(sc, a_sg, "subgrad")
        a_cur = a_sg

    # Final L-BFGS from best
    if time.monotonic() < deadline - 5:
        sc, a_opt = run_lbfgs_lse(best_a, 3000, maxiter=2000, maxfun=40000,
                                   deadline=deadline - 3)
        report(sc, a_opt, "final-lbfgs")

    # Final coordinate descent on best
    if time.monotonic() < deadline - 3:
        sc, a_cd3 = coordinate_descent(best_a, deadline=deadline - 2, n_full_passes=1)
        report(sc, a_cd3, "final-coord")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length={len(out)}, Time={time.monotonic()-t_start:.1f}s/{budget_s}s")
    print(f"{'='*60}")

    return out
