    #   Standard: (x ⋆ y)[n] = sum_k x[k]*conj(y[k+n])
    #   In FFT: (x ⋆ y) = ifft(FFT(x) * conj(FFT(y)))
    #   So (a ⋆ p)[j] = sum_k a[k]*p[k+j] = sum_k p[k+j]*a[k] ✓
    #   = term2[j]!
    #   And in FFT: ifft(FFT(a, M2) * conj(FFT(p, M2)), M2)
    
    # Great! So:
    # h[j] = term1[j] + term2[j]
    # term1 = irfft(rfft(p, M2) * rfft(a, M2), M2)  [convolution p*a]
    # term2 = irfft(rfft(a, M2) * conj(rfft(p, M2)), M2)  [cross-correlation a⋆p]
    #   For real sequences, conj(rfft(p)) = rfft(p[::-1] padded)... 
    #   Actually for real p: conj(FFT(p)) = FFT(p) reversed (with conjugate)
    #   Just use: conj(np.fft.rfft(p, M2)) in complex arithmetic.
    
    # But wait: rfft returns complex array. conj works element-wise. Fine.
    
    # Let me just implement this and verify.
    
    M2 = 1 << (3 * n - 1).bit_length()
    Pa = np.fft.rfft(p, M2)
    Aa = np.fft.rfft(a, M2)
    term1 = np.fft.irfft(Pa * Aa, M2)
    term2 = np.fft.irfft(Aa * np.conj(Pa), M2)
    h = (term1 + term2)[:n]
    
    # Final gradient
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n / S3) * (S * h - 2.0 * lse_val * np.ones(n))
    return score, grad


def lse_objective_wrapper(a, beta):
    """Wrapper for scipy minimize."""
    return lse_score_and_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, maxls=50):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': maxls}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def spectral_flatten(a0, N, n_iters=200, lam=0.5):
    """Iterative spectral flattening."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    S_init = a.sum()
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (S_init / S)
    return a


def subgradient_step(a, sc_lb=1.0):
    """One subgradient step on the exact (non-smooth) objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return a, np.inf
    M = float(c.max())
    k_star = int(np.argmax(c))
    sc = 2.0 * n * M / (S * S)
    # Subgradient of M w.r.t. a[j]:
    # M = conv[k_star] = sum_i a[i]*a[k_star - i]
    # dM/da[j] = a[k_star - j] (if valid) + a[j - k_star] (if valid)
    sg_M = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        # For j in [j_start, j_end]: a[k_star - j] is valid
        sg_M[j_start:j_end + 1] = a[k_star - j_end:k_star - j_start + 1][::-1]
    # Also the "other" term: a[j - k_star] valid when 0 <= j - k_star < n
    # i.e., k_star <= j <= k_star + n - 1, but j < n, so j in [max(0,k_star), n-1]
    j_start2 = max(0, k_star)
    j_end2 = min(n - 1, k_star + n - 1)
    if j_end2 >= j_start2:
        # a[j - k_star] for j in [j_start2, j_end2]
        sg_M[j_start2:j_end2 + 1] += a[j_start2 - k_star:j_end2 - k_star + 1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n / S3) * (S * sg_M - 2.0 * M)
    gn = float(np.linalg.norm(grad))
    if gn < 1e-15:
        return a, sc
    step = (sc - sc_lb) / (gn * gn + 1e-30)
    step = min(step, 0.5)
    a_new = a - step * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new, sc


def subgradient_polish(a, total_iters=500):
    """Polish with subgradient method."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    for it in range(total_iters):
        a, sc = subgradient_step(a, sc_lb)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Decay
        if it % 50 == 0 and it > 0:
            pass
    return best_sc, best_a


def mirror_perturb(a, strength=0.03, rng=None):
    """Perturb by anti-correlating a with its version shifted to the peak lag."""
    n = len(a)
    if rng is None:
        rng = np.random.default_rng(0)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    # Create anti-correlation: where a[k*-j] is large, reduce a[j]
    # and vice versa
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    a_pert = a.copy()
    if j_end >= j_start:
        reflected = a[k_star - j_end:k_star - j_start + 1][::-1]
        mean_ref = np.mean(reflected)
        # Reduce where reflected is above mean, increase where below
        diff = reflected - mean_ref
        a_pert[j_start:j_end + 1] -= strength * diff
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    # Preserve sum approximately
    S_orig = a.sum()
    S_new = a_pert.sum()
    if S_new > 1e-12:
        a_pert = a_pert * (S_orig / S_new)
    return a_pert


def random_perturb(a, scale=0.02, rng=None):
    """Additive random perturbation."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a)
    a_pert = a + scale * a.std() * rng.standard_normal(n)
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    # Preserve sum
    S_orig = a.sum()
    S_new = a_pert.sum()
    if S_new > 1e-12:
        a_pert = a_pert * (S_orig / S_new)
    return a_pert


def multiplicative_perturb(a, scale=0.02, rng=None):
    """Multiplicative random perturbation."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a)
    a_pert = a * np.exp(scale * rng.standard_normal(n))
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    return a_pert


def chirp_perturb(a, rng=None):
    """Chirp-based multiplicative perturbation."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a)
    x = np.arange(n, dtype=np.float64)
    f = 1.0 + 8.0 * rng.random()
    phase = 2.0 * np.pi * rng.random()
    amp = 0.01 + 0.03 * rng.random()
    mod = 1.0 + amp * np.cos(np.pi * f * x * x / (2.0 * n) + phase)
    a_pert = a * mod
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    return a_pert


def targeted_perturb(a, rng=None, strength=0.05):
    """Reduce elements that contribute most to the peak, redistribute."""
    n = len(a)
    if rng is None:
        rng = np.random.default_rng(0)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[k_star - j_end:k_star - j_start + 1][::-1] * a[j_start:j_end + 1]
    # Find top contributors and bottom
    top_count = max(5, n // 50)
    sorted_idx = np.argsort(contrib)[::-1]
    top_idx = sorted_idx[:top_count]
    bottom_idx = sorted_idx[top_count:]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength * rng.uniform(0.5, 1.5)
        a_pert[j] = max(0.0, a_pert[j] - reduce_amt)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def spectral_perturb(a, amp=0.02, rng=None):
    """Perturb in spectral domain."""
    n = len(a)
    if rng is None:
        rng = np.random.default_rng(0)
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.fft(a, M_fft)
    noise = amp * (rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A)))
    A_noisy = A * (1.0 + noise)
    A_noisy[0] = A[0]  # Preserve DC (sum)
    a_pert = np.real(np.fft.ifft(A_noisy))[:n]
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    return a_pert


def coordinate_descent_step(a, k_star, n_steps=100):
    """Simple coordinate descent: adjust individual elements to reduce peak."""
    n = len(a)
    a = a.copy()
    c = fast_conv(a)
    S = a.sum()
    M = c.max()
    best_sc = 2.0 * n * M / (S * S)
    
    for _ in range(n_steps):
        c = fast_conv(a)
        S = a.sum()
        M = c.max()
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
        
        # Try adjusting a few coordinates
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        
        # Find the element that contributes most to the peak
        # and try reducing it slightly
        seg = a[j_start:j_end + 1]
        rev_seg = a[k_star - j_end:k_star - j_start + 1][::-1]
        prod = seg * rev_seg
        j_local = int(np.argmax(prod))
        j_global = j_start + j_local
        
        # Try reducing a[j_global]
        for frac in [0.01, 0.005, 0.002]:
            a_trial = a.copy()
            a_trial[j_global] *= (1.0 - frac)
            sc_trial = compute_score(a_trial)
            if sc_trial < best_sc:
                a = a_trial
                best_sc = sc_trial
                break
    
    return best_sc, a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimum peak convolution score sequence.
    
    Phase 1: Initialize from parent + diverse starting points (5% time)
    Phase 2: LSE-continuation L-BFGS with adaptive beta (45% time)  
    Phase 3: Diverse local search with targeted perturbations (30% time)
    Phase 4: Final high-beta polish + subgradient (20% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 20.0
    deadline = t_start + effective_budget
    
    best_score = np.inf
    best_a = None
    
    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    # ================================================================
    # Phase 1: Initialization (5% of budget)
    # ================================================================
    init_deadline = t_start + 0.05 * effective_budget
    print(f"--- Phase 1: Initialization (deadline +{init_deadline - t_start:.0f}s) ---")
    
    candidates = []
    
    # Try to load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    if parent is not None:
        n = parent.size
        print(f"  Parent loaded: n={n}, score={compute_score(parent):.8f}")
        candidates.append(('parent', parent.copy()))
        report(compute_score(parent), parent, "parent")
    else:
        n = 4096
        print(f"  No parent, using n={n}")
        candidates.append(('uniform', np.ones(n) * 0.5))
    
    # Generate a few diverse initializations at the same length
    N_fft = 1 << (2 * n - 1).bit_length()
    
    if time.monotonic() < init_deadline:
        # Spectral flattening from uniform
        sf1 = spectral_flatten(np.ones(n) * 0.5, N_fft, n_iters=150, lam=0.5)
        candidates.append(('sf_uniform', sf1))
        report(compute_score(sf1), sf1, "sf_uniform")
    
    if time.monotonic() < init_deadline:
        # Spectral flattening from random
        rand_init = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0)
        sf2 = spectral_flatten(rand_init, N_fft, n_iters=150, lam=0.5)
        candidates.append(('sf_random', sf2))
        report(compute_score(sf2), sf2, "sf_random")
    
    if time.monotonic() < init_deadline:
        # Chirp-based
        x = np.arange(n, dtype=np.float64)
        chirp = np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01
        sf3 = spectral_flatten(chirp, N_fft, n_iters=150, lam=0.4)
        candidates.append(('sf_chirp', sf3))
        report(compute_score(sf3), sf3, "sf_chirp")
    
    # Sort candidates by score
    candidates.sort(key=lambda x: compute_score(x[1]))
    print(f"  Best init: {candidates[0][0]} with score {compute_score(candidates[0][1]):.8f}")
    
    # ================================================================
    # Phase 2: LSE Continuation (45% of budget)
    # ================================================================
    ph2_deadline = t_start + 0.50 * effective_budget
    print(f"\n--- Phase 2: LSE Continuation (deadline +{ph2_deadline - t_start:.0f}s) ---")
    
    # Take the best candidate and run continuation
    a_cur = candidates[0][1].copy()
    sc_cur = compute_score(a_cur)
    report(sc_cur, a_cur, "phase2 start")
    
    beta_schedule = [20, 50, 100, 200, 400, 800, 1500]
    
    for beta in beta_schedule:
        if time.monotonic() > ph2_deadline:
            break
        remaining = ph2_deadline - time.monotonic()
        if remaining < 10:
            break
        
        t0 = time.monotonic()
        # Estimate iterations based on remaining time
        # Each L-BFGS iteration takes roughly 2 FFT evaluations
        # For n=7000, each FFT pair takes ~1ms, so ~2ms per iteration
        est_time_per_iter = 0.003 * (n / 4096)
        est_scipy_func_call = est_time_per_iter * 3  # scipy overhead
        maxiter = min(8000, int(remaining * 0.7 / max(est_scipy_func_call, 0.01)))
        maxfun = max(500, maxiter * 25)
        
        print(f"  Beta={beta}, maxiter={maxiter}, remaining={remaining:.0f}s")
        sc, a_opt, res = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
        a_cur = a_opt
        report(sc, a_opt, f"lbfgs beta={beta}")
        elapsed = time.monotonic() - t0
        print(f"    -> score={sc:.8f} ({elapsed:.1f}s, nfev={res.nfev})")
    
    print(f"  Phase 2 complete: best={best_score:.8f}")
    
    # ================================================================
    # Phase 3: Diverse Local Search (30% of budget)
    # ================================================================
    ph3_deadline = t_start + 0.80 * effective_budget
    print(f"\n--- Phase 3: Diverse Local Search (deadline +{ph3_deadline - t_start:.0f}s) ---")
    
    n_trials = 60
    strategies = ['mirror', 'random', 'multi', 'chirp', 'targeted', 'spectral']
    
    for trial in range(n_trials):
        if time.monotonic() > ph3_deadline:
            break
        remaining = ph3_deadline - time.monotonic()
        if remaining < 5:
            break
        
        strategy = strategies[trial % len(strategies)]
        strength = 0.01 + 0.03 * rng.random()
        
        if strategy == 'mirror':
            a_pert = mirror_perturb(a_cur, strength, rng)
        elif strategy == 'random':
            a_pert = random_perturb(a_cur, strength, rng)
        elif strategy == 'multi':
            a_pert = multiplicative_perturb(a_cur, strength, rng)
        elif strategy == 'chirp':
            a_pert = chirp_perturb(a_cur, rng)
        elif strategy == 'targeted':
            a_pert = targeted_perturb(a_cur, rng, strength=0.03 + 0.04 * rng.random())
        else:  # spectral
            a_pert = spectral_perturb(a_cur, 0.01 + 0.02 * rng.random(), rng)
        
        # Quick L-BFGS polish from perturbed point
        # Use medium beta for speed
        beta_local = 300 + 200 * rng.random()
        maxiter_local = min(3000, int(remaining * 0.4 / 0.01))
        maxfun_local = max(200, maxiter_local * 20)
        
        sc, a_opt, _ = run_lbfgs(a_pert, beta_local, maxiter=max(100, maxiter_local),
                                   maxfun=max(500, maxfun_local))
        improved = report(sc, a_opt, f"local #{trial} {strategy}")
        if improved:
            a_cur = a_opt.copy()
    
    print(f"  Phase 3 complete: best={best_score:.8f}")
    
    # ================================================================
    # Phase 4: Final Polish (20% of budget)
    # ================================================================
    ph4_deadline = deadline
    print(f"\n--- Phase 4: Final Polish (deadline +{deadline - t_start:.0f}s) ---")
    
    a_cur = best_a.copy()
    
    # High-beta LSE continuation
    for beta in [2000, 4000, 8000]:
        if time.monotonic() > ph4_deadline - 30:
            break
        remaining = ph4_deadline - time.monotonic()
        if remaining < 10:
            break
        maxiter = min(5000, int(remaining * 0.5 / 0.01))
        sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=max(200, maxiter),
                                   maxfun=max(1000, maxiter * 20))
        a_cur = a_opt
        report(sc, a_opt, f"polish beta={beta}")
    
    # Subgradient polish
    if time.monotonic() < ph4_deadline - 10:
        sc, a_sg = subgradient_polish(a_cur, total_iters=300)
        report(sc, a_sg, "subgrad")
        if sc < best_score:
            a_cur = a_sg.copy()
    
    # Final very high beta
    if time.monotonic() < ph4_deadline - 5:
        remaining = ph4_deadline - time.monotonic()
        sc, a_opt, _ = run_lbfgs(a_cur, 10000, maxiter=min(3000, int(remaining/0.01)),
                                   maxfun=60000)
        report(sc, a_opt, "final_lbfgs")
    
    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
