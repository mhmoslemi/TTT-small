Hybrid DE + L-BFGS + Peak-Reduction optimizer for minimum peak convolution.
Uses Differential Evolution for global exploration, L-BFGS with LSE continuation

import numpy as np
from scipy.optimize import minimize, differential_evolution

def fast_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective: 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """Compute LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -100, 100)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    # Gradient: d/d a[j] of sum_k p[k] * conv(a,a)[k]
    # conv(a,a)[k] = sum_i a[i]*a[k-i]
    # d/d a[j] conv(a,a)[k] = a[k-j] + a[j-k] (out-of-range = 0)
    # So grad component = sum_k p[k] * (a[k-j] + a[j-k])
    # = conv(p, a_rev)[n-1+j] + conv(a, p)[j]

    rev_a = a[::-1]
    M2_len = len(p) + n - 1
    M2_fft = 1 << (M2_len - 1).bit_length()

    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    A_f = np.fft.rfft(a, M2_fft)

    conv_p_ra = np.fft.irfft(P * RA, M2_fft)
    conv_a_p = np.fft.irfft(A_f * P, M2_fft)

    total = conv_p_ra[n - 1:2 * n - 1] + conv_a_p[:n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (total / S2 - 2.0 * L / S3)

    return score, grad

def spectral_flatten(a0, N, n_iters=200, lam=0.5, rng=None):
    """Iterative spectral flattening."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a


def peak_decomposition_move(a, rng, n_steps=200, strength=0.02):
    """
    Novel local move: identify peak lags, find contributing pairs,
    and redistribute mass from high-contributing pairs to low-impact positions.
    """
    n = len(a)
    a = a.copy()
    c = fast_conv(a)
    S = float(a.sum())

    for step in range(n_steps):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        k_star = int(np.argmax(c))
        peak_val = c[k_star]

        # Find contributing pairs for this lag
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break

        seg_a = a[j_start:j_end + 1]
        seg_b = a[j_start:j_end + 1][::-1]
        products = seg_a * seg_b
        total_prod = products.sum()
        if total_prod < 1e-12:
            break

        # Find positions with high product contribution
        n_reduce = max(2, len(products) // 20)
        top_indices = np.argsort(products)[::-1][:n_reduce]

        # Find positions that contribute least to peak (good places to add mass)
        bottom_indices = np.argsort(products)[:n_reduce]

        # Compute reduction
        total_reduce = 0.0
        for idx in top_indices:
            j = j_start + idx
            reduce_amt = a[j] * strength * (1.0 + 0.5 * rng.random())
            reduce_amt = min(reduce_amt, a[j] * 0.3)  # Don't reduce more than 30%
            a[j] -= reduce_amt
            a[j] = max(a[j], 0.0)
            total_reduce += reduce_amt

        # Redistribute to bottom positions
        if total_reduce > 0 and len(bottom_indices) > 0:
            per = total_reduce / len(bottom_indices)
            for idx in bottom_indices:
                j = j_start + idx
                a[j] += per

        a = np.clip(a, 0.0, 1000.0)

    return a

    """Simulated annealing local search."""
    n = len(a)

    sc_cur = compute_score(a_cur)
    a_best = a_cur.copy()
    sc_best = sc_cur

    for it in range(n_iters):

            mag[0] += noise[0]
            new_mag = np.maximum(mag + noise, 0)

        sc = compute_score(a_polished)
        report(sc, a_polished, f"lbfgs_cont{bi}")

    # Polish from DE results (if different from current best)
    if len(top_inits) > 1:
        for ii in range(1, min(4, len(top_inits))):
            if time.monotonic() > phase3_deadline:
                break
            remaining = phase3_deadline - time.monotonic()
            if remaining < 20:
                break
            maxiter = min(3000, max(200, int(remaining / 3)))
            a_pol = lse_continuation(top_inits[ii], [50, 200, 800], maxiter_per_beta=maxiter)
            sc = compute_score(a_pol)
            report(sc, a_pol, f"lbfgs_from_init{ii}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ================================================================
    # Phase 4: Peak Decomposition + Targeted Local Search
    # ================================================================
    phase4_deadline = t_start + 0.88 * effective_budget
    print(f"\n--- Phase 4: Peak Decomposition & Local Search ---")

    n_trials = 30
    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        strategy = trial % 6

        if strategy == 0:
            # Peak decomposition move
            a_pert = peak_decomposition_move(best_a, rng, n_steps=100, strength=0.01 + 0.03 * rng.random())
        elif strategy == 1:
            # Gaussian perturbation
            sc_pert = 0.01 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + sc_pert * rng.standard_normal(n_primary))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Frequency-domain perturbation
            N_f = 1 << (2 * n_primary - 1).bit_length()
            A = np.fft.fft(best_a, N_f)
            noise = 0.01 * rng.standard_normal(len(A)) + 0.01j * rng.standard_normal(len(A))
            A_noisy = A * (1.0 + noise)
            A_noisy[0] = A[0]
            a_pert = np.real(np.fft.ifft(A_noisy))[:n_primary]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Simulated annealing
            remaining = phase4_deadline - time.monotonic()
            n_sa = min(1000, max(200, int(remaining / 0.1)))
            sc_sa, a_sa = simulated_anneal(best_a, rng, n_iters=n_sa, T_start=0.005, T_end=0.0001)
            report(sc_sa, a_sa, f"SA_{trial}")
            continue
        elif strategy == 4:
            # Blend with spectral-flattened version
            sf_a = spectral_flatten(best_a, N_fft, n_iters=50, lam=0.3)
            t_blend = 0.1 + 0.3 * rng.random()
            a_pert = (1 - t_blend) * best_a + t_blend * sf_a
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Chirp-modulated perturbation
            x = np.arange(n_primary, dtype=np.float64)
            f = 1.0 + 5.0 * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + amp * np.cos(np.pi * f * x * x / (2.0 * n_primary)))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick L-BFGS polish from perturbed point
        remaining = phase4_deadline - time.monotonic()
        if remaining < 5:
            break
        maxiter = min(2000, max(100, int(remaining / 0.3)))
        sc, a_opt = run_lbfgs(a_pert, 500, maxiter=maxiter)
        report(sc, a_opt, f"local#{trial}_s{strategy}")

    # ================================================================
    # Phase 5: Final High-Beta Polishing
    # ================================================================
    print(f"\n--- Phase 5: Final Polishing ---")

    for beta in [1000, 3000, 8000]:
        if time.monotonic() > deadline - 10:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=3000, maxfun=60000)
        report(sc, a_opt, f"final_beta{beta}")

    # One more peak decomposition + polish cycle
    a_pert = peak_decomposition_move(best_a, rng, n_steps=200, strength=0.01)
    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(a_pert, 5000, maxiter=2000, maxfun=40000)
        report(sc, a_opt, "final_peak_decomp")

    # Final simulated annealing
    if time.monotonic() < deadline - 20:
        sc_sa, a_sa = simulated_anneal(best_a, rng, n_iters=1000, T_start=0.002, T_end=0.00005)
        report(sc_sa, a_sa, "final_SA")

    # ================================================================
    # Phase 6: Try alternative lengths (if time permits)
    # ================================================================
    if time.monotonic() < deadline - 60:
        print(f"\n--- Phase 6: Alternative Lengths ---")
        alt_lengths = [4096, 8192]
        # Remove parent length
        alt_lengths = [l for l in alt_lengths if l != n_primary]

        for alt_n in alt_lengths:
            if time.monotonic() > deadline - 30:
                break
            print(f"  Trying n={alt_n}")
            N_alt = 1 << (2 * alt_n - 1).bit_length()

            # Generate init for this length
            alt_init = spectral_flatten(np.ones(alt_n) * 0.5, N_alt, n_iters=150, lam=0.5)
            sc = compute_score(alt_init)
            if sc < best_score:
                # Polish it
                remaining = deadline - time.monotonic() - 30
                if remaining > 30:
                    a_pol = lse_continuation(alt_init, [50, 200, 800, 2000],
                                            maxiter_per_beta=min(2000, int(remaining / 4)))
                    sc = compute_score(a_pol)
                    report(sc, a_pol, f"alt_n{alt_n}")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
