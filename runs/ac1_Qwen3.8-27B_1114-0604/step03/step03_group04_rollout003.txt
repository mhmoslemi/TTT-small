import numpy as np
from scipy.optimize import minimize

def fast_conv(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv) / sum^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    exp_scaled = np.exp(scaled - scaled.max())
    log_sum = np.log(exp_scaled.sum()) + scaled.max()
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a[j]
    # dL/da[j] = sum_k p[k] * d(conv[k])/da[j]
    # d(conv[k])/da[j] = 2*a[k-j] (if valid) or a[j] (if k=2j)
    # Using FFT trick: conv(p, rev(a)) 
    rev_a = a[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:n + len(p) - 1]
    # The gradient of conv w.r.t. a is: for each j, sum_k p[k] * 2*a[k-j] (valid k-j)
    # This is 2 * conv(p, a) evaluated at specific positions
    # Actually: d(conv(a,a)[k])/da[j] = a[k-j] + a[j-k] (when valid)
    # = 2 * a[k-j] when k-j is valid and k != 2j
    # The sum over k of p[k] * 2*a[k-j] = 2 * (p * a_reversed)[j + something]
    # Let me use the direct convolution approach
    # grad_L[j] = sum_k p[k] * d(conv_k)/da[j]
    # d(conv_k)/da[j] = a[k-j] if 0<=k-j<n else 0, PLUS a[j-k] if 0<=j-k<n else 0
    # = a[k-j] (if valid) + a[j-k] (if valid)
    # Note: a[k-j] valid means 0<=k-j<n, i.e., j<=k<=j+n-1
    # a[j-k] valid means 0<=j-k<n, i.e., k<=j and k>=j-n+1, i.e., j-n+1<=k<=j
    # So for a given j:
    #   term1: a[k-j] for k in [j, j+n-1] (intersection with [0, 2n-2])
    #   term2: a[j-k] for k in [max(0,j-n+1), j] (intersection with [0, 2n-2])
    # This is: conv(p, a)[j] + conv(p, a)[2n-2-j]... hmm, let me just do it directly.
    
    # Actually, the simplest: d(conv(a,a)[k])/da[j] for the full convolution
    # conv(a,a)[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i]*a[k-i]
    # The terms involving a[j]:
    #   a[j]*a[k-j] (when k-j is in [0,n-1], i.e., j <= k <= j+n-1)
    #   a[k-j]*a[j] (when k-j is in [0,n-1] AND j is in [0,n-1], which is the same condition)
    # Wait, these are the same term! conv(a,a)[k] = sum_i a[i]*a[k-i], and the term with i=j is a[j]*a[k-j].
    # But also, when i = k-j, the term is a[k-j]*a[j], which is the same!
    # So if j != k-j (i.e., k != 2j), the term a[j]*a[k-j] appears twice (once as i=j and once as i=k-j).
    # If k = 2j, the term is a[j]*a[j] = a[j]^2, which appears only once.
    
    # So d(conv_k)/da[j] = 2*a[k-j] if 0<=k-j<n and k!=2j
    #                    = a[j] if k=2j (and 0<=j<n, which is always true)
    #                    = 0 otherwise
    
    # For the gradient: grad_j = sum_k p[k] * d(conv_k)/da[j]
    # = 2*sum_{k: 0<=k-j<n, k!=2j} p[k]*a[k-j] + p[2j]*a[j] (if 2j < 2n-1)
    
    # Let m = k-j, so k = m+j:
    # = 2*sum_{m: 0<=m<n, m+j!=2j} p[m+j]*a[m] + p[2j]*a[j] (if 2j < 2n-1)
    # = 2*sum_{m: 0<=m<n} p[m+j]*a[m] - 2*p[2j]*a[j] (if 2j valid) + p[2j]*a[j] (if 2j valid)
    # = 2*sum_{m: 0<=m<n} p[m+j]*a[m] - p[2j]*a[j] (if 2j < 2n-1)
    # = 2*sum_{m: 0<=m<n} p[m+j]*a[m] (if 2j >= 2n-1, i.e., j >= n-0.5, which is always true for j in [0,n-1] since 2j >= 2n-1 iff j >= n-0.5)
    
    # Wait, 2j < 2n-1 means j < n-0.5, so for j in [0, n-1], 2j < 2n-1 is always true (since 2(n-1) = 2n-2 < 2n-1).
    # So: grad_j = 2*sum_{m=0}^{n-1} p[m+j]*a[m] - p[2j]*a[j]
    
    # The sum sum_{m=0}^{n-1} p[m+j]*a[m] is the cross-correlation of p and a at lag j.
    # This is (p * rev_a)[n-1+j] where rev_a = a[::-1]... let me verify.
    # (p * rev_a)[k] = sum_i p[i] * rev_a[k-i] = sum_i p[i] * a[n-1-(k-i)] = sum_i p[i] * a[n-1-k+i]
    # For k = n-1+j: sum_i p[i] * a[n-1-(n-1+j)+i] = sum_i p[i] * a[i-j]
    # Hmm, that's not quite what I want.
    
    # Let me use a different approach. Define q = p (length 2n-1) and a (length n).
    # I want T[j] = sum_{m=0}^{n-1} p[m+j] * a[m] for j = 0, ..., n-1.
    # This is the cross-correlation <p_shifted, a> where p is shifted by j.
    # T[j] = sum_m a[m] * p[m+j]
    # = (a * rev_p)[n-1+j]... no.
    # Actually: conv(a, p)[j + n - 1]... let me just compute it via FFT.
    
    # T[j] = sum_{m=0}^{n-1} a[m] * p[m+j]
    # This is like a cross-correlation. Let's define it as:
    # T = cross_corr(a, p) where cross_corr(a,p)[j] = sum_m a[m]*p[m+j]
    # Using FFT: T[j] = IFFT(FFT(a) * conj(FFT(p)))[j]... but the lengths differ.
    
    # Simpler: pad a to length 2n-1, compute conv(rev_a, p), extract.
    # conv(rev_a, p)[k] = sum_i rev_a[i] * p[k-i] = sum_i a[n-1-i] * p[k-i]
    # Let m = n-1-i: = sum_m a[m] * p[k-n+1+m]
    # For k = n-1+j: = sum_m a[m] * p[j+m] = T[j] ✓
    
    # So T[j] = conv(rev_a, p)[n-1+j] for j = 0, ..., n-1.
    
    # But I already computed conv_res = conv(rev_a, p) above! It has length n + (2n-1) - 1 = 3n-2.
    # T[j] = conv_res[n-1+j] for j = 0, ..., n-1.
    
    # Wait, I computed conv_res using a different M2. Let me redo this properly.
    
    # Actually, let me just recompute the gradient more carefully.
    # I'll use a direct FFT approach.
    
    # The gradient of the LSE loss L = c_max + (1/beta) * log(sum exp(beta*(c[k] - c_max)))
    # with respect to a[j] is:
    # dL/da[j] = sum_k p[k] * d(c[k])/da[j]
    # where p[k] = exp(beta*(c[k]-c_max)) / sum exp(beta*(c[k]-c_max))
    
    # I already have the formula:
    # grad_L[j] = 2*sum_{m=0}^{n-1} p[m+j]*a[m] - p[2j]*a[j]
    #           = 2*T[j] - p[2j]*a[j]
    
    # where T[j] = sum_m a[m]*p[m+j]
    
    # And the full score gradient:
    # score = 2n*L/S^2
    # d(score)/da[j] = 2n*(grad_L[j]/S^2 - 2*L*a[j]/S^3)
    
    # Let me compute T using FFT.
    # T[j] = conv(rev_a, p)[n-1+j]
    # rev_a has length n, p has length 2n-1.
    # conv has length n + 2n-1 - 1 = 3n-2.
    # We need indices n-1 to 2n-2 (i.e., n values).
    
    M3 = 3 * n - 2  # length of conv(rev_a, p)
    M3_fft = 1 << (M3 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M3_fft)
    P = np.fft.rfft(p, M3_fft)
    conv_res = np.fft.irfft(RA * P, M3_fft)[:M3]
    T = conv_res[n - 1:2 * n - 1]  # T[j] for j = 0,...,n-1
    
    # p has length 2n-1, so p[2j] is valid for j = 0,...,n-1 (since 2j <= 2n-2 = len(p)-1)
    p_2j = p[0:2*n-1:2]  # p[0], p[2], p[4], ..., p[2n-2]
    
    grad_L = 2.0 * T - p_2j * a
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L * a / S3)
    
    return score, grad


def spectral_water_fill(a0, n, n_iters=300, n_fft_pad=None, seed=0):
    """Aggressive spectral water-filling with annealing."""
    rng = np.random.default_rng(seed)
    a = a0.astype(np.float64).copy()
    if n_fft_pad is None:
        M = 2 * n - 1
        n_fft = 1 << (M - 1).bit_length()
    else:
        n_fft = n_fft_pad
    
    S0 = a.sum()
    
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        # Annealing: start with gentle flattening, become more aggressive
        # Then back off to avoid destroying the signal
        if progress < 0.7:
            lam = 0.3 + 0.7 * (progress / 0.7)  # 0.3 to 1.0
        else:
            lam = 1.0 - 0.5 * ((progress - 0.7) / 0.3)  # 1.0 to 0.5
        
        A = np.fft.rfft(a, n_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude at non-DC frequencies
        non_dc = mag[1:]
        target = float(np.median(non_dc))
        
        new_mag = mag.copy()
        for k in range(1, len(mag)):
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
        
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.irfft(new_A, n_fft))[:n]
        a = np.maximum(a_new, 0.0)
        
        # Rescale to maintain sum
        S = a.sum()
        if S > 1e-12:
            a *= S0 / S
    
    return a


def spectral_water_fill_noisy(a0, n, n_iters=200, noise_scale=0.05, seed=0):
    """Spectral water-filling with noise to escape local minima."""
    rng = np.random.default_rng(seed)
    a = a0.astype(np.float64).copy()
    M = 2 * n - 1
    n_fft = 1 << (M - 1).bit_length()
    S0 = a.sum()
    
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        lam = 0.5 + 0.5 * progress
        noise = noise_scale * (1.0 - progress)
        
        A = np.fft.rfft(a, n_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        
        non_dc = mag[1:]
        target = float(np.median(non_dc))
        
        new_mag = mag.copy()
        for k in range(1, len(mag)):
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            # Add noise
            new_mag[k] *= (1.0 + noise * rng.standard_normal())
            new_mag[k] = max(new_mag[k], 0.0)
        
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.irfft(new_A, n_fft))[:n]
        a = np.maximum(a_new, 0.0)
        
        S = a.sum()
        if S > 1e-12:
            a *= S0 / S
    
    return a


def peak_shaving(a, n_steps=50, n_top=20, step_size=0.02):
    """Directly reduce the top peaks of the convolution."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for step in range(n_steps):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top lags
        avg = float(c.mean())
        # Identify lags where c > avg
        above = c > avg
        if not above.any():
            break
        
        # For each element j, compute its "responsibility" for the above-avg lags
        # resp[j] = sum_{k: c[k]>avg} d(c[k])/da[j]
        # d(c[k])/da[j] = 2*a[k-j] (if 0<=k-j<n) approximately
        resp = np.zeros(n)
        for j in range(n):
            # k ranges over valid values where 0 <= k-j < n
            k_start = j
            k_end = min(j + n, len(c))
            if k_end > k_start:
                valid_k = slice(k_start, k_end)
                m = np.arange(k_start, k_end) - j  # m = k - j, ranges 0 to n-1
                resp[j] = 2.0 * float(np.dot(a[m], above[valid_k].astype(float) * c[valid_k]))
        
        # Normalize responsibility
        resp_max = resp.max()
        if resp_max < 1e-15:
            break
        resp = resp / resp_max
        
        # Reduce high-responsibility elements, increase low-responsibility
        # Use a smooth version
        reduce_mask = resp > 0.5
        increase_mask = resp < 0.3
        total_reduce = float(a[reduce_mask].sum()) * step_size
        if total_reduce < 1e-15:
            break
        n_inc = max(1, int(increase_mask.sum()))
        per_inc = total_reduce / n_inc
        a[reduce_mask] *= (1.0 - step_size)
        inc_idx = np.where(increase_mask)[0][:n_inc]
        a[inc_idx] += per_inc / max(1, len(inc_idx))
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a


def coordinate_descent_polish(a, n_iters=200, n_coord=20):
    """Coordinate descent: try adjusting the most impactful elements."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng(12345)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find the peak lag
        k_star = int(np.argmax(c))
        
        # Elements that contribute to this peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        # Pick a few elements to try adjusting
        if j_end >= j_start:
            contrib = a[j_start:j_end + 1] * a[k_star - j_end:k_star - j_start + 1][::-1]
            # Try adjusting the top contributors
            order = np.argsort(contrib)[::-1]
            for _ in range(min(n_coord, len(order))):
                j = j_start + order[_]
                # Try both increasing and decreasing
                for delta in [0.01, -0.01, 0.05, -0.05]:
                    a_test = a.copy()
                    a_test[j] += delta * a[j] + delta * 0.01
                    a_test = np.clip(a_test, 0.0, 1000.0)
                    sc_test = compute_score(a_test)
                    if sc_test < sc:
                        a = a_test
                        sc = sc_test
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a.copy()
                        break
    
    return best_sc, best_a


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def chirp_init(n, rng):
    """Generate chirp-like initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = []
    inits.append(np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01)
    freq = 1.0 + 3.0 * rng.random()
    inits.append(0.5 + 0.4 * np.cos(np.pi * freq * x * x / (2.0 * n)))
    f1 = 1.0 + 4.0 * rng.random()
    f2 = 1.0 + 4.0 * rng.random()
    inits.append(0.5 + 0.3 * np.cos(np.pi * f1 * x * x / (2.0 * n)) + 0.2 * np.cos(np.pi * f2 * x / n))
    for s in inits:
        inits[-1] = np.clip(s, 0.01, 1000.0)
    return inits

            print(f"    {key}: score={sc:.6f} ({time.monotonic()-t0:.1f}s)")
            
            # Noisy version
            sf_a2 = spectral_water_fill_noisy(binit, n, n_iters=150, noise_scale=0.08, seed=seed+100+ii)
            sc2 = compute_score(sf_a2)
            report(sc2, sf_a2, key + "_noisy")
            print(f"    {key}_noisy: score={sc2:.6f}")
        
        # If parent is this length, add it
        if parent is not None and parent.size == n:
            sc_p = compute_score(parent)
            sf_candidates[f"parent_n{n}"] = (sc_p, parent.copy())
            report(sc_p, parent, f"parent_n{n}")
            print(f"    parent_n{n}: score={sc_p:.6f}")
    
    # ================================================================
    # Phase 2: L-BFGS continuation
    # ================================================================
    lbfgs_deadline = t_start + 0.65 * (deadline - t_start)
    print(f"\n--- Phase 2: L-BFGS Continuation ---")
    
    # Sort candidates by score
    all_cands = sorted(sf_candidates.values(), key=lambda x: x[0])
    
    for n in lengths:
        if time.monotonic() > lbfgs_deadline:
            break
        print(f"\n  >> n = {n}")
        
        # Get candidates for this length
        cand_n = [(sc, a) for sc, a in all_cands if len(a) == n]
        if not cand_n:
            # Generate fresh
            N_fft = 1 << (2 * n - 1).bit_length()
            a_fresh = spectral_water_fill(np.ones(n) * 0.5, n, 100, N_fft, seed)
            cand_n = [(compute_score(a_fresh), a_fresh)]
        
        for ci, (sc_init, a_init) in enumerate(cand_n[:4]):
            if time.monotonic() > lbfgs_deadline:
                break
            t0 = time.monotonic()
            a_cur = a_init.copy()
            for beta in [50, 100, 300, 800, 2000]:
                if time.monotonic() > lbfgs_deadline:
                    break
                remaining = lbfgs_deadline - time.monotonic()
                if remaining < 10:
                    break
                maxiter = min(5000, int(remaining * 0.7 / 0.05))
                maxiter = max(200, maxiter)
                sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxiter * 20)
                a_cur = a_opt
                report(sc, a_opt, f"lbfgs n={n} ci={ci} b={beta}")
            
            sc_final = compute_score(a_cur)
            report(sc_final, a_cur, f"lbfgs_done n={n} ci={ci}")
            print(f"    n={n} ci={ci}: score={sc_final:.6f} ({time.monotonic()-t0:.1f}s)")
    
    print(f"\nAfter Phase 2: best = {best_score:.8f}")
    
    # ================================================================
    # Phase 3: Peak shaving
    # ================================================================
    ps_deadline = t_start + 0.80 * (deadline - t_start)
    print(f"\n--- Phase 3: Peak Shaving ---")
    
    sc, a_ps = peak_shaving(best_a, n_steps=100, n_top=20, step_size=0.03)
    report(sc, a_ps, "peak_shave")
    
    # L-BFGS from peak-shaved
    if time.monotonic() < ps_deadline - 10:
        for beta in [500, 1500]:
            if time.monotonic() > ps_deadline - 5:
                break
            sc, a_opt = run_lbfgs(a_ps, beta, maxiter=3000, maxfun=60000)
            report(sc, a_opt, f"ps+lbfgs b={beta}")
            a_ps = a_opt
    
    # Peak shave again
    sc, a_ps2 = peak_shaving(a_ps, n_steps=80, step_size=0.02)
    report(sc, a_ps2, "peak_shave2")
    
    # ================================================================
    # Phase 4: Coordinate descent + final L-BFGS
    # ================================================================
    cd_deadline = t_start + 0.92 * (deadline - t_start)
    print(f"\n--- Phase 4: Coordinate Descent + Final Polish ---")
    
    sc, a_cd = coordinate_descent_polish(best_a, n_iters=50, n_coord=10)
    report(sc, a_cd, "coord_descent")
    
    # Final L-BFGS with high beta
    if time.monotonic() < cd_deadline - 5:
        for beta in [2000, 5000]:
            if time.monotonic() > cd_deadline - 3:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
            report(sc, a_opt, f"final_lbfgs b={beta}")
            best_a = a_opt  # Keep updating for next iterations
    
    # ================================================================
    # Phase 5: Multi-start local search
    # ================================================================
    print(f"\n--- Phase 5: Multi-Start Local Search ---")
    ms_deadline = deadline
    n_trials = 30
    n_best = len(best_a)
    for trial in range(n_trials):
        if time.monotonic() > ms_deadline:
            break
        remaining = ms_deadline - time.monotonic()
        if remaining < 5:
            break
        
        # Generate perturbation
        strategy = trial % 4
        if strategy == 0:
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strategy == 1:
            # Targeted perturbation at peak contributors
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            if j_end >= j_start:
                seg = a_pert[j_start:j_end+1]
                seg[j_end-j_start//2] *= 0.8  # Reduce middle element
                seg[j_start] *= 1.2  # Increase edge
                a_pert[j_start:j_end+1] = seg
        elif strategy == 2:
            # Spectral perturbation
            M = 2 * n_best - 1
            N_fft = 1 << (M - 1).bit_length()
            A = np.fft.rfft(best_a, N_fft)
            noise = 0.01 * rng.standard_normal(len(A)) + 0.01j * rng.standard_normal(len(A))
            A_noisy = A * (1.0 + noise)
            A_noisy[0] = A[0]
            a_pert = np.real(np.fft.irfft(A_noisy, N_fft))[:n_best]
        else:
            # Random coordinate perturbation
            a_pert = best_a.copy()
            idx = rng.choice(n_best, size=max(5, n_best//100), replace=False)
            a_pert[idx] *= (1.0 + 0.05 * rng.standard_normal(len(idx)))
        
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        # Quick L-BFGS
        maxiter = min(2000, int(remaining * 0.8 / 0.03))
        maxiter = max(100, maxiter)
        sc, a_opt = run_lbfgs(a_pert, 500, maxiter=maxiter, maxfun=maxiter * 20)
        report(sc, a_opt, f"ms #{trial} s={strategy}")
    
    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
