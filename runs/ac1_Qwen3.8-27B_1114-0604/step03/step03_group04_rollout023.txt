    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(np.sum(exp_scaled))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def spectral_flatten(a0, N, n_iters=300, lam=0.5):
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam_eff = lam * (1.0 - 0.5 * progress)
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc = mag[1:N//2+1]
        target = np.median(non_dc)
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam_eff) * mag[k] + lam_eff * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a

def multi_peak_subgrad(a, total_iters=500, top_k=5, lr_decay=0.999):
    """Subgradient method focusing on top-k peaks in autocorrelation."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = 1.0
    
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        
        # Find top-k peaks
        c_sorted = np.sort(c)[::-1]
        c_max = c_sorted[0]
        sc = 2.0 * n * c_max / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Use top-k peaks for subgradient
        # For each peak at position x, the subgradient of (f*f)(x) w.r.t. a is:
        # d/d a_j [(f*f)(x)] = f(x-j) + f(x-j) = 2*f(x-j) (if x-j in [0,n-1])
        # Actually, (f*f)(x) = sum_j f(j)*f(x-j)
        # d/d a_j = f(x-j) if 0 <= x-j <= n-1, else 0
        # So subgradient of max is: for the argmax x*, sg_j = f(x*-j) if valid
        
        # For top-k, average the subgradients
        top_idx = np.argsort(c)[::-1][:top_k]
        # Weight by how much each peak exceeds the mean
        c_mean = np.mean(c)
        weights = np.maximum(c[top_idx] - c_mean, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(top_k)
        weights = weights / weights.sum()
        
        sg = np.zeros(n)
        for wi, x in zip(weights, top_idx):
            j_start = max(0, int(x) - n + 1)
            j_end = min(int(x), n - 1)
            if j_end >= j_start:
                sg[j_start:j_end+1] += wi * a[j_start:j_end+1][::-1]
        
        # Full gradient of the score (approximate)
        S2 = S * S
        grad = (2.0 * n / S2) * sg - (2.0 * n * 2.0 * c_max / S**3) * a
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Adaptive step size
        step = lr / (gn + 1e-10)
        step = min(step, 0.5)
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        
        # Check if it improved
        sc_new = compute_score(a_new)
        if sc_new < sc:
            a = a_new
            lr *= lr_decay  # slow decay
        else:
            lr *= 0.9
            # Try half step
            a_new = a - 0.5 * step * grad
            a_new = np.maximum(a_new, 0.0)
            sc_new = compute_score(a_new)
            if sc_new < sc:
                a = a_new
                lr *= 0.95
            else:
                lr *= 0.8
        
        if lr < 1e-15:
            break
    
    return best_sc, best_a

"""
Peak autocorrelation minimizer: spectral init + LP-based updates + high-beta LSE L-BFGS + multi-peak subgradient.
Combines the SOTA LP blending idea with aggressive L-BFGS continuation and targeted peak reduction.
"""

import numpy as np
from scipy.optimize import minimize
import time

# Global best tracking (no closures)
_BEST_SCORE = np.inf
_BEST_A = None


def reset_state():
    global _BEST_SCORE, _BEST_A
    _BEST_SCORE = np.inf
    _BEST_A = None


def report_result(sc, a):
    global _BEST_SCORE, _BEST_A
    if sc < _BEST_SCORE and sc > 0:
        _BEST_SCORE = sc
        _BEST_A = a.copy()
        return True
    return False


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_full(f, g):
    n_f = len(f)
    n_g = len(g)
    M = n_f + n_g - 1
    M_fft = 1 << (M - 1).bit_length()
    F = np.fft.rfft(f, M_fft)
    G = np.fft.rfft(g, M_fft)
    return np.fft.irfft(F * G, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(np.sum(exp_scaled))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def spectral_flatten(a0, N, n_iters=300, lam=0.5):
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam_eff = lam * (1.0 - 0.5 * progress) + 0.1
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc = mag[1:N // 2 + 1]
        target = np.median(non_dc)
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam_eff) * mag[k] + lam_eff * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a


def multi_peak_subgrad(a, total_iters=500, top_k=5):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = 1.0
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        top_idx = np.argsort(c)[::-1][:top_k]
        c_mean = float(np.mean(c))
        weights = np.maximum(c[top_idx].astype(np.float64) - c_mean, 0.0)
        if weights.sum() < 1e-15:
            weights = np.ones(top_k)
        weights = weights / weights.sum()
        sg = np.zeros(n)
        for wi, x in zip(weights, top_idx):
            x = int(x)
            j_start = max(0, x - n + 1)
            j_end = min(x, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += wi * 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n / S2) * sg - (2.0 * n * 2.0 * c_max / S3) * a
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        step = min(lr / (gn + 1e-10), 0.5)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        sc_new = compute_score(a_new)
        if sc_new < sc:
            a = a_new
            lr = min(lr * 1.001, 2.0)
        else:
            lr *= 0.85
            if lr < 1e-18:
                break
    return best_sc, best_a


def lp_blend_step(f, M, solver_name='GLOP'):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)(x) <= M for all x. Returns b or None."""
    n = len(f)
    # For large n, use reduced constraints
    if n > 5000:
        step = max(1, (2 * n - 1) // 6000)
        rows = list(range(0, 2 * n - 1, step))
        if 0 not in rows:
            rows.insert(0, 0)
        if n - 1 not in rows:
            rows.append(n - 1)
    else:
        rows = list(range(2 * n - 1))
    n_rows = len(rows)
    A_mat = np.zeros((n_rows, n))
    for i, x in enumerate(rows):
        k_start = max(0, x - n + 1)
        k_end = min(x, n - 1)
        if k_end >= k_start:
            A_mat[i, k_start:k_end + 1] = f[x - k_start:x - k_end + 1][::-1]
    try:
        import cvxpy as cp
        b_var = cp.Variable(n, nonneg=True)
        objective = cp.Maximize(cp.sum(b_var))
        constraints = [A_mat @ b_var <= M]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=solver_name, verbose=False)
        if b_var.value is not None:
            return np.maximum(b_var.value, 0.0)
    except Exception:
        try:
            import cvxpy as cp
            b_var = cp.Variable(n, nonneg=True)
            objective = cp.Maximize(cp.sum(b_var))
            constraints = [A_mat @ b_var <= M]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver='GLPK', verbose=False)
            if b_var.value is not None:
                return np.maximum(b_var.value, 0.0)
        except Exception:
            pass
    return None

            best_a = a_new.copy()

        else:

    return best_sc, best_a

            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
            a_cur = a_opt
            report_result(sc, a_opt)
            print(f"    beta={beta:>5d}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    print(f"\n  After L-BFGS: best={_BEST_SCORE:.8f}")

    # Phase 4: Multi-peak subgradient
    print("\n--- Phase 4: Multi-Peak Subgradient ---")
    subgrad_deadline = t_start + 0.78 * (deadline - t_start)
    for top_k in [3, 5, 10, 20, 50]:
        if time.monotonic() > subgrad_deadline:
            break
        remaining = subgrad_deadline - time.monotonic()
        n_iters = max(200, int(remaining / 0.05))
        t0 = time.monotonic()
        sc_sg, a_sg = multi_peak_subgrad(_BEST_A, total_iters=n_iters, top_k=top_k)
        report_result(sc_sg, a_sg)
        print(f"  top_k={top_k:>2d}: {sc_sg:.8f} ({time.monotonic()-t0:.1f}s)")

    # Phase 5: High-beta L-BFGS
    print("\n--- Phase 5: High-Beta L-BFGS ---")
    hi_beta_deadline = t_start + 0.88 * (deadline - t_start)
    for beta in [8000, 20000, 50000]:
        if time.monotonic() > hi_beta_deadline - 5:
            break
        remaining = hi_beta_deadline - time.monotonic()
        n = len(_BEST_A)
        est_eval = 0.002 * (n / 4096.0) + 0.001
        maxiter = max(500, min(8000, int(remaining * 0.6 / max(est_eval * 12, 0.01))))
        maxfun = max(2000, min(150000, maxiter * 20))
        t0 = time.monotonic()
        sc_hb, a_hb = run_lbfgs(_BEST_A, beta, maxiter=maxiter, maxfun=maxfun)
        report_result(sc_hb, a_hb)
        print(f"  beta={beta:>5d}: {sc_hb:.8f} ({time.monotonic()-t0:.1f}s)")

    # Phase 6: Local search
    print("\n--- Phase 6: Local Search ---")
    local_deadline = deadline - 30.0
    n_best = len(_BEST_A)
    n_trials = 0
    while time.monotonic() < local_deadline:
        n_trials += 1
        strategy = n_trials % 6
        if strategy == 0:
            scale = 0.01 + 0.02 * rng.random()
            a_pert = _BEST_A * (1.0 + scale * rng.standard_normal(n_best))
        elif strategy == 1:
            c = fast_conv(_BEST_A)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = _BEST_A.copy()
            seg = a_pert[j_start:j_end + 1][::-1]
            top_count = max(5, n_best // 200)
            top_local = np.argsort(seg)[::-1][:top_count]
            total_red = 0.0
            for j in top_local:
                amt = a_pert[j_start + j] * 0.05
                a_pert[j_start + j] -= amt
                a_pert[j_start + j] = max(a_pert[j_start + j], 0.0)
                total_red += amt
            bot_local = np.argsort(seg)[:top_count]
            if len(bot_local) > 0:
                per = total_red / len(bot_local)
                for j in bot_local:
                    a_pert[j_start + j] += per
        elif strategy == 2:
            N_fft = 1 << (2 * n_best - 1).bit_length()
            A = np.fft.fft(_BEST_A, N_fft)
            noise = 0.015 * (rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A)))
            A_noisy = A * (1.0 + noise)
            A_noisy[0] = A[0]
            a_pert = np.real(np.fft.ifft(A_noisy))[:n_best]
        elif strategy == 3:
            x = np.arange(n_best, dtype=np.float64)
            f = 1.0 + 5.0 * rng.random()
            pert = 0.015 * np.cos(np.pi * f * x * x / (2.0 * n_best))
            a_pert = _BEST_A * (1.0 + pert)
        elif strategy == 4:
            # Block perturbation
            nb = max(4, n_best // 500)
            per = max(1, n_best // nb)
            block_noise = rng.standard_normal(nb) * 0.02
            a_pert = _BEST_A.copy()
            for bi in range(nb):
                a_pert[bi * per:(bi + 1) * per] *= (1.0 + block_noise[bi])
        else:
            # Random sparse perturbation
            n_sparse = max(10, n_best // 50)
            idx = rng.choice(n_best, size=n_sparse, replace=False)
            a_pert = _BEST_A.copy()
            a_pert[idx] *= (1.0 + 0.1 * rng.standard_normal(n_sparse))
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        remaining = local_deadline - time.monotonic()
        if remaining < 5:
            break
        maxiter = max(200, min(3000, int(remaining * 0.2 / 0.01)))
        sc_ls, a_ls = run_lbfgs(a_pert, 5000, maxiter=maxiter, maxfun=max(1000, maxiter * 15))
        report_result(sc_ls, a_ls)
        if n_trials % 5 == 0:
            print(f"  trial {n_trials}: best={_BEST_SCORE:.8f}")

    # Final polish
    print("\n--- Final Polish ---")
    if time.monotonic() < deadline - 10:
        sc_f, a_f = multi_peak_subgrad(_BEST_A, total_iters=500, top_k=10)
        report_result(sc_f, a_f)
        if time.monotonic() < deadline - 5:
            sc_f2, a_f2 = run_lbfgs(_BEST_A, 30000, maxiter=3000, maxfun=60000)
            report_result(sc_f2, a_f2)

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in _BEST_A]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={_BEST_SCORE:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
