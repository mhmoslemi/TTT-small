**Spectral equalization:**

Given a (length n), we want to flatten the power spectrum |A[m]|^2 where A = FFT(a).

Step:
1. A = FFT(a)
2. target_power = (sum(a))^2 / n = S^2/n
3. current_power = |A|^2
4. Update magnitude: |A_new| = (1-eta)*|A| + eta*sqrt(target_power)
   (or: |A_new|^2 = (1-eta)*|A|^2 + eta*target_power)
5. A_new = |A_new| * exp(i*angle(A))
6. a_new = real(IFFT(A_new))
7. a_new = max(0, a_new)  // project to non-negative
8. Optionally renormalize to keep sum the same.

This is similar to the "flat spectrum" iteration used in PRBS design.

The parameter eta controls the aggressiveness. Small eta = gentle, large eta = aggressive.

def spectral_equalize(a, eta=0.1, n_iters=5):
    """Spectral equalization to flatten the power spectrum."""
    n = len(a)
    a = a.astype(np.float64).copy()

        S = float(a.sum())
        A = np.fft.rfft(a)
        power = np.abs(A)**2
        target = S*S/n
        # Update magnitude
        new_mag = np.sqrt(np.maximum((1-eta)*power + eta*target, 0))
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j*phase)
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(0, a_new)
        if a_new.sum() > 1e-12:
            a = a_new
        else:
            break
    return a

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    exp_scaled = np.exp(np.maximum(scaled, -500))
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum/beta
    score = 2.0*n*L/(S*S)
    p = exp_scaled/exp_scaled.sum()
    # Gradient computation
    rev_a = a[::-1]
    M2 = 3*n - 2
    M2_fft = 1 << (M2-1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA*P, M2_fft)[n-1:2*n-1]
    g = 2.0*conv_res
    S2 = S*S
    S3 = S2*S
    grad = (2.0*n)*(g/S2 - 2.0*L/S3)
    return score, grad
