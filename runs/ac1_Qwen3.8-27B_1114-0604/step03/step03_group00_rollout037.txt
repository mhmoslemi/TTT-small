    S = a.sum()

        return a

"""
Hybrid optimizer: Spectral Flattening (POCS-like) + LSE Optimization + Subgradient Polish.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Innovations:
1. Spectral flattening that iteratively equalizes |FFT(a)| for non-DC bins
   (minimizes total convolution energy, a proxy for peak)
2. LSE-smoothed L-BFGS optimization with adaptive beta schedule
3. Combined peak-directed subgradient and coordinate descent polish
4. Multi-scale length exploration
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def spectral_flatten(a, n_iters=2000, gamma=0.7, L=None, tol=1e-14):
    """
    Iteratively flatten the power spectrum to reduce autocorrelation peak.
    This is a POCS-like projection between:
      - The set of sequences with flat power spectrum (|A(k)| = const for k>=1)
      - The set of non-negative, sum-1 sequences
    """
    n = len(a)
    if L is None:
        L = 1 << (2 * n - 1).bit_length()
    
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    prev_sc = best_sc
    
    for it in range(n_iters):
        A = np.fft.fft(a, L)
        amp = np.abs(A)
        
        # Target amplitude for non-DC bins
        dc_power = float(np.abs(A[0])**2)  # Should be 1
        total_power = L * float(np.sum(a**2))
        off_dc_power = total_power - dc_power
        if off_dc_power <= 0:
            break
        target_amp = np.sqrt(off_dc_power / (L - 1))
        
        # Soft spectral update
        ratio = np.ones(L)
        # Avoid division by zero
        mask = amp > 1e-15
        ratio[1:] = np.ones(L - 1)
        ratio[1:][mask[1:]] = (target_amp / amp[mask[1:]]) ** gamma
        # For very small amplitudes, set to target
        ratio[1:][~mask[1:]] = target_amp / max(amp[~mask[1:]].max(), 1e-15) if (~mask[1:]).any() else 1.0
        
        A_new = A * ratio
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
        
        # Convergence check
        if abs(prev_sc - sc) < tol:
            if it > 100:
                break
        prev_sc = sc
        
        # Adaptive gamma: start aggressive, become conservative
        # Actually, let's use a schedule
        gamma_iter = 0.3 + 0.7 * (1.0 - np.exp(-it / 200.0))
    
    return best_a


def spectral_flatten_v2(a, n_iters=2000, L=None):
    """
    Improved spectral flattening with adaptive gamma and damping.
    Uses a smoother update rule.
    """
    n = len(a)
    if L is None:
        L = 1 << (2 * n - 1).bit_length()
    
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.fft(a, L)
        amp = np.abs(A).copy()
        phase = np.angle(A)
        
        # Compute target: equalize amplitudes
        off_dc_power = L * float(np.sum(a**2)) - 1.0
        if off_dc_power <= 1e-15:
            break
        target_amp = np.sqrt(off_dc_power / (L - 1))
        
        # Log-domain update (more stable)
        log_amp = np.where(amp > 1e-15, np.log(amp), -15.0)
        log_target = np.log(max(target_amp, 1e-15))
        
        # Adaptive step size
        step = 0.3 if it < 100 else (0.1 if it < 500 else 0.05)
        
        new_log_amp = log_amp.copy()
        new_log_amp[0] = 0.0  # Keep DC fixed (log(1) = 0)
        new_log_amp[1:] = log_amp[1:] + step * (log_target - log_amp[1:])
        
        new_amp = np.exp(new_log_amp)
        A_new = new_amp * np.exp(1j * phase)
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def error_reduction(a, n_iters=200, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]
        
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0
        
        damp = 0.5 + 0.5 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def subgradient_polish(a, total_iters=3000):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.1)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a


def coordinate_descent_peak(a, n_sweeps=5, n_top=100):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue
        
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        
        # Also find all peak lags (within 2% of max)
        c_max = float(c.max())
        peak_lags = np.where(c > 0.98 * c_max)[0]
        
        for j in top_global:
            a_old_j = a[j]
            # Try reducing this element
            for factor in [0.1, 0.3, 0.5, 0.7, 0.9]:
                cv = a_old_j * factor
                a[j] = cv
                S = a.sum()
                if S < 1e-12:
                    a[j] = a_old_j
                    break
                a = a / S
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                else:
                    a[j] = a_old_j

            S = a.sum()
            if S > 1e-12:

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    # Envelope: U-shaped
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    # Random blocks
    for nb in (16, 64):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    # Sobol-like (low-discrepancy)
    random_perm = rng.permutation(n)
    inits.append(0.3 + 0.7 * (random_perm / n))
    
    return inits

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def spectral_flatten_v2(a, n_iters=1500, L=None):
    """
    Log-domain spectral flattening (POCS-like).
    Iteratively equalizes |FFT(a)| across non-DC bins to flatten the convolution.
    """
    n = len(a)
    if L is None:
        L = 1 << (2 * n - 1).bit_length()

    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.fft(a, L)
        amp = np.abs(A).copy()
        phase = np.angle(A)

        # Target: equalize amplitudes for non-DC bins
        off_dc_power = L * float(np.sum(a ** 2)) - 1.0
        if off_dc_power <= 1e-15:
            break
        target_amp = np.sqrt(off_dc_power / (L - 1))

        # Log-domain update with adaptive step
        log_amp = np.where(amp > 1e-15, np.log(amp), -15.0)
        log_target = np.log(max(target_amp, 1e-15))

        # Adaptive step size: aggressive early, conservative late
        if it < 50:
            step = 0.5
        elif it < 200:
            step = 0.3
        elif it < 500:
            step = 0.15
        else:
            step = 0.08

        new_log_amp = log_amp.copy()
        new_log_amp[0] = 0.0
        new_log_amp[1:] = log_amp[1:] + step * (log_target - log_amp[1:])

        new_amp = np.exp(new_log_amp)
        A_new = new_amp * np.exp(1j * phase)

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def error_reduction(a, n_iters=200, alpha=0.75):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess

        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.4 + 0.6 * (it / max(1, n_iters // 3))
        A_new = A + damp * d

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def multi_lag_subgradient(a, total_iters=5000):
    """
    Subgradient descent using all near-peak lags for a richer subgradient.
    This gives a smoother effective gradient than single-peak subgradient.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find near-peak lags (within 5% of max)
        threshold = 0.95 * c_max
        peak_lags = np.where(c >= threshold)[0]

        # Compute subgradient contribution from all peak lags
        sg = np.zeros(n)
        for k in peak_lags:
            j_start = max(0, int(k) - n + 1)
            j_end = min(int(k), n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size with adaptation
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        # Additional damping
        step *= (1.0 + 0.5 * np.cos(it * 0.01))

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def coordinate_descent_multiplicative(a, n_sweeps=5, n_top=150):
    """
    Multiplicative coordinate descent on peak-contributing elements.
    Uses ratio updates (like EM) which are more stable than additive.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        c_max = float(c.max())

        # Find all significant peak lags
        peak_lags = np.where(c >= 0.97 * c_max)[0]

        # Accumulate contribution weights
        weights = np.zeros(n)
        for k in peak_lags:
            j_start = max(0, int(k) - n + 1)
            j_end = min(int(k), n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                weights[j_start:j_end + 1] += seg[::-1]

        # Focus on top contributors
        top_count = min(n_top, n)
        top_indices = np.argsort(weights)[::-1][:top_count]

        for j in top_indices:
            if a[j] < 1e-15:
                continue
            # Try multiplicative reductions
            best_local_sc = best_sc
            best_factor = 1.0
            current_sc = compute_score(a)
            for factor in [0.3, 0.5, 0.7, 0.85]:
                saved = a[j]
                a[j] = saved * factor
                S = a.sum()
                if S < 1e-12:
                    a[j] = saved
                    break
                a = a / S
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_factor = factor
                a[j] = saved

            if best_factor < 1.0:
                a[j] *= best_factor
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                if best_local_sc < best_sc:
                    best_sc = best_local_sc
                    best_a = a.copy()

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Check convergence
        if sweep > 0 and abs(best_sc - sc) < 1e-10:
            break

    return best_a

            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * (1.0 / n))
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(1.0/n + 0.1 * rng.standard_normal(n) / np.sqrt(n), 0.001, 100.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    # Random blocks
    for nb in (32, 128):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Strategy: Spectral flattening for global search, then LSE L-BFGS and
    multi-lag subgradient for local refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 15.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * (1.0 / 4096)

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    # Add nearby lengths for exploration
    for n in [4096, 6000, 8192]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:4]

    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes' if parent is not None else 'no'}")
    if parent is not None:
        print(f"  Parent len={len(parent)}, score={compute_score(parent):.8f}")

    # ============ Phase 1: Spectral Flattening + Error Reduction (25% time) ============
    phase1_deadline = t_start + 0.25 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 1: Spectral Flattening (by t={phase1_deadline - t_start:.0f}s) ---")

    sf_results = {}

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"  n={n}...", end=" ", flush=True)
        L_sf = 1 << (2 * n - 1).bit_length()

        inits = build_initializations(n, rng, parent)
        remaining = max(1, int((phase1_deadline - time.monotonic()) / 20.0))
        inits = inits[:max(1, remaining)]

        phase_best = None
        phase_best_sc = np.inf

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a0_clip = np.clip(a0, 0, 1000)
            S0 = a0_clip.sum()
            if S0 < 1e-12:
                continue
            a0_norm = a0_clip / S0

            # Spectral flattening
            a_sf = spectral_flatten_v2(a0_norm, n_iters=800, L=L_sf)
            sc = compute_score(a_sf)
            if sc < phase_best_sc:
                phase_best_sc = sc
                phase_best = a_sf
            report(sc, a_sf, f"SF n={n} init={i}")

        if phase_best is not None:
            sf_results[n] = phase_best
            # Error reduction from best SF result
            if time.monotonic() < phase1_deadline - 5:
                a_er = error_reduction(phase_best, n_iters=150, alpha=0.8)
                sc = compute_score(a_er)
                report(sc, a_er, f"ER n={n}")
                if sc < compute_score(phase_best):
                    sf_results[n] = a_er
        print(f"done (best={phase_best_sc:.8f})")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============ Phase 2: LSE L-BFGS Optimization (45% time) ============
    phase2_deadline = t_start + 0.70 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 2: LSE L-BFGS (by t={phase2_deadline - t_start:.0f}s) ---")

    # Build beta schedule: start moderate, go high
    beta_schedule = [30, 80, 200, 500, 1500, 5000, 15000]

    # First, optimize the best sequence found in Phase 1
    n_cur = len(best_a)
    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        remaining_time = phase2_deadline - time.monotonic()
        maxiter = min(8000, max(500, int(remaining_time * 3)))
        maxfun = min(150000, max(10000, maxiter * 15))
        print(f"  L-BFGS beta={beta} (n={n_cur})...", end=" ", flush=True)
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS b={beta}")
        print(f"score={sc:.8f}")

    # Try other lengths/starting points from Phase 1 results
    for n_alt, a_alt in sf_results.items():
        if time.monotonic() > phase2_deadline:
            break
        if n_alt == len(best_a):
            continue
        for beta in [200, 1000, 5000]:
            if time.monotonic() > phase2_deadline:
                break
            remaining_time = phase2_deadline - time.monotonic()
            maxiter = min(5000, max(300, int(remaining_time * 3)))
            maxfun = min(80000, max(5000, maxiter * 15))
            print(f"  Alt L-BFGS n={n_alt} beta={beta}...", end=" ", flush=True)
            sc, a_opt, _ = run_lbfgs(a_alt, beta, maxiter=maxiter, maxfun=maxfun)
            report(sc, a_opt, f"AltLBFGS n={n_alt} b={beta}")
            print(f"score={sc:.8f}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============ Phase 3: Local Search with Perturbations (15% time) ============
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 3: Local Search (by t={phase3_deadline - t_start:.0f}s) ---")

    n_best = len(best_a)

    for trial in range(25):
        if time.monotonic() > phase3_deadline:
            break
        strategy = trial % 5
        if strategy == 0:
            # Small Gaussian perturbation
            scale = 0.01 + 0.02 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Peak-lag targeted: reduce top contributors, add to bottom
            c = fast_conv(best_a)
            c_max = float(c.max())
            peak_lags = np.where(c >= 0.97 * c_max)[0]
            a_pert = best_a.copy()
            weights = np.zeros(n_best)
            for k in peak_lags:
                j_start = max(0, int(k) - n_best + 1)
                j_end = min(int(k), n_best - 1)
                if j_end >= j_start:
                    weights[j_start:j_end + 1] += a_pert[j_start:j_end + 1][::-1]
            top_count = max(10, n_best // 50)
            top_idx = np.argsort(weights)[::-1][:top_count]
            bot_idx = np.argsort(weights)[:top_count]
            total_red = 0.0
            for j in top_idx:
                red = a_pert[j] * 0.08
                a_pert[j] = max(a_pert[j] - red, 0.0)
                total_red += red
            if total_red > 0:
                per = total_red / len(bot_idx)
                for j in bot_idx:
                    a_pert[j] += per
        elif strategy == 2:
            # Sinusoidal modulation
            freq = 1.0 + 15.0 * rng.random()
            phase_shift = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase_shift))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Spectral perturbation
            L_sf = 1 << (2 * n_best - 1).bit_length()
            A = np.fft.fft(best_a, L_sf)
            phase_pert = 0.03 * rng.standard_normal(L_sf)
            A_pert = A * np.exp(1j * phase_pert)
            a_pert = np.fft.ifft(A_pert).real[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Block scaling
            a_pert = best_a.copy()
            n_blocks = 8 + int(rng.integers(0, 12))
            block_size = max(1, n_best // n_blocks)
            for b in range(min(n_blocks, n_best // block_size)):
                s = b * block_size
                e = min(s + block_size, n_best)
                factor = 0.6 + 0.8 * rng.random()
                a_pert[s:e] *= factor

        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S

        # Quick LSE optimization from perturbed start
        if time.monotonic() < phase3_deadline - 10:
            remaining_time = phase3_deadline - time.monotonic()
            maxiter = min(3000, max(500, int(remaining_time * 2)))
            sc, a_opt, _ = run_lbfgs(a_pert, 1000, maxiter=maxiter, maxfun=max(5000, maxiter * 10))
            report(sc, a_opt, f"local #{trial} s={strategy}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============ Phase 4: Final Polish (15% time) ============
    print(f"\n--- Phase 4: Final Polish ---")

    # Multi-lag subgradient
    if time.monotonic() < deadline - 90:
        a_sg = multi_lag_subgradient(best_a, total_iters=3000)
        sc = compute_score(a_sg)
        report(sc, a_sg, "multi-lag subgrad")

    # Coordinate descent
    if time.monotonic() < deadline - 60:
        a_cd = coordinate_descent_multiplicative(best_a, n_sweeps=5, n_top=100)
        sc = compute_score(a_cd)
        report(sc, a_cd, "coord descent")

    # Very high beta LSE
    if time.monotonic() < deadline - 40:
        sc, a_opt, _ = run_lbfgs(best_a, 20000, maxiter=5000, maxfun=80000)
        report(sc, a_opt, "final LBFGS b=20000")

    # Short spectral flatten + LSE
    if time.monotonic() < deadline - 20:
        L_sf = 1 << (2 * len(best_a) - 1).bit_length()
        a_sf = spectral_flatten_v2(best_a, n_iters=200, L=L_sf)
        if time.monotonic() < deadline - 10:
            sc, a_opt, _ = run_lbfgs(a_sf, 10000, maxiter=3000, maxfun=50000)
            report(sc, a_opt, "SF+LBFGS final")

    # One more subgradient pass
    if time.monotonic() < deadline - 5:
        a_sg2 = multi_lag_subgradient(best_a, total_iters=2000)
        sc = compute_score(a_sg2)
        report(sc, a_sg2, "final subgrad")

    # ============ Final Output ============
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
