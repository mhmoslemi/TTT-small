"""
Optimized peak-autocorrelation minimizer for non-negative sequences.
Uses corrected LogSumExp gradient, spectral flattening initialization,
L-BFGS continuation with adaptive beta, and peak-targeted local search.
Key improvements: correct gradient derivation, adaptive beta scheduling,
peak-specific perturbation, and multi-length exploration.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """FFT-based linear convolution a*a."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the exact objective: 2n * max(a*a) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    Compute the LogSumExp-smoothed score and its gradient.
    
    f(a) = 2n * L / S^2 where L = LSE_beta(a*a), S = sum(a)
    grad_j = 4n * h_j / S^2 - 4n * L / S^3
    where h_j = sum_m a[m] * p[j+m], p = softmax(beta * (v - v_max))
    """
    n = len(a)
    v = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    v_max = float(np.max(v))
    scaled = beta * (v - v_max)
    # Numerical stability
    exp_scaled = np.exp(scaled)
    log_sum = np.log(np.sum(exp_scaled))
    L = v_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / np.sum(exp_scaled)
    
    # Compute h_j = sum_{m=0}^{n-1} a[m] * p[j+m] for j = 0,...,n-1
    # Using FFT: h = ifft(fft(a, L) * conj(fft(p, L)))[:n]
    L_fft = 1 << (3 * n - 2).bit_length()
    A_ft = np.fft.fft(a, L_fft)
    P_ft = np.fft.fft(p, L_fft)
    h_full = np.fft.ifft(A_ft * np.conj(P_ft))[:n].real
    
    S2 = S * S
    S3 = S2 * S
    grad = (4.0 * n / S2) * h_full - (4.0 * n * L / S3)
    
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B optimization with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def spectral_flatten(a0, N, n_iters=200, lam=0.5):
    """Gerchberg-Saxton spectral flattening for non-negative sequences."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    S_target = a.sum()
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        half = N // 2
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            else:
                new_mag[k] = (1.0 - lam) * mag[k] + lam * target
                new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (S_target / S)
    return a


def spectral_flatten_targeted(a0, N, n_iters=200, lam=0.5, target_peak_lag=None):
    """
    Spectral flattening with targeted peak reduction.
    Modifies the target spectrum to specifically reduce the peak at a given lag.
    """
    a = a0.astype(np.float64).copy()
    n = len(a)
    S_target = a.sum()
    
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compute current peak
        v = fast_conv(a)
        if target_peak_lag is None:
            k_peak = int(np.argmax(v))
        else:
            k_peak = target_peak_lag
        
        # Target: flat spectrum, but with reduced contribution at the peak frequency
        # The peak at lag k corresponds to frequency component m where m*k mod N = 0
        # Actually, (a*a)[k] = (1/N) * sum_m |A_m|^2 * exp(2*pi*i*m*k/N)

        new_mag = mag.copy()

            cos_val = np.cos(2.0 * np.pi * k * k_peak / N)
            # Reduce where cos > 0 (contributes positively to peak)

            else:
                new_mag[k] = (1.0 - lam) * mag[k] + lam * new_target
                new_mag[N - k] = new_mag[k]
        
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (S_target / S)
    return a


def subgrad_polish(a, total_iters=500):
    """Subgradient method directly on the max-of-convolution objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(np.max(c))
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of M = (a*a)[k_star] w.r.t. a:
        # dM/da_j = 2 * a[k_star - j] if 0 <= k_star - j < n, else 0
        sg_M = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        # Gradient of 2n*M/S^2:
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step size
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.5 / (1.0 + 0.01 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        
        # Small exploration step
        a -= (0.0005 / (1.0 + 0.005 * it)) * grad / (gn + 1e-30)
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a


def peak_perturb(a, rng, strength=0.03):
    """Targeted perturbation: reduce contributions to the peak lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    
    # Compute contribution of each a_j to the peak
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    
    a_pert = a.copy()
    # Reduce top contributors
    top_count = max(5, n // 20)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    
    # Redistribute to bottom contributors
    bottom_idx = np.argsort(contrib)[:top_count]
    if total_reduced > 0 and len(bottom_idx) > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    
    return np.clip(a_pert, 0.0, 1000.0)

        scored.sort(key=lambda x: x[0])
        scored = scored[:min(5, len(scored))]
        
        # Run continuation on best candidate(s)
        for ci, (sc_init, a_init) in enumerate(scored[:3]):
            if time.monotonic() > lbfgs_deadline:
                break
            t0 = time.monotonic()
            a_cur = a_init.copy()
            print(f"    Candidate {ci}: init_score={sc_init:.6f}")
            
            for bi, beta in enumerate(beta_schedule):
                if time.monotonic() > lbfgs_deadline:
                    break
                remaining = lbfgs_deadline - time.monotonic()
                if remaining < 10:
                    break
                
                # Scale iterations by n and remaining time
                scale_factor = max(1, n // 2048)
                maxiter = min(3000, max(200, int(remaining * 0.6 / (0.01 * scale_factor))))
                maxfun = min(50000, maxiter * 25)
                
                sc, a_opt, res = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
                a_cur = a_opt
                improved = report(sc, a_opt, f"n={n} c{ci} b={beta}")
                msg = f"    beta={beta}: {sc:.8f}" + (" *" if improved else "")
                print(msg)
            
            sc_final = compute_score(a_cur)
            report(sc_final, a_cur, f"n={n} c{ci} final")

    print(f"\nAfter Phase 2: best = {best_score:.8f} (n={len(best_a)})")

    # ================================================================
    # Phase 3: Peak-targeted local search
    # ================================================================
    local_deadline = t_start + 0.88 * effective_budget
    print(f"\n--- Phase 3: Peak-Targeted Local Search (budget: {local_deadline - t_start:.0f}s) ---")

    n_best = len(best_a)
    n_trials = 60
    for trial in range(n_trials):
        if time.monotonic() > local_deadline:
            break
        
        strategy = trial % 6
        if strategy == 0:
            # Peak-targeted perturbation
            a_pert = peak_perturb(best_a, rng, strength=0.02 + 0.03 * rng.random())
        elif strategy == 1:
            # Gaussian noise
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Spectral perturbation
            N_fft = 1 << (2 * n_best - 1).bit_length()
            A = np.fft.fft(best_a, N_fft)
            noise_mag = 0.01 + 0.03 * rng.random()
            phase_noise = noise_mag * rng.standard_normal(len(A))
            A_noisy = A * np.exp(1j * phase_noise)
            A_noisy[0] = A[0]  # Preserve DC
            a_pert = np.real(np.fft.ifft(A_noisy))[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Chirp modulation
            x = np.arange(n_best, dtype=np.float64)
            f = 0.5 + 4.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + amp * np.cos(np.pi * f * x / n_best + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 4:
            # Block perturbation: modify a random contiguous block
            block_len = max(10, n_best // 20)
            start = int(rng.integers(0, max(1, n_best - block_len)))
            a_pert = best_a.copy()
            a_pert[start:start+block_len] *= (0.7 + 0.6 * rng.random())
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Spectral flattening from current best (short)
            N_fft = 1 << (2 * n_best - 1).bit_length()
            a_pert = spectral_flatten(best_a, N_fft, n_iters=30, lam=0.2)
        
        # Quick L-BFGS polish
        remaining = local_deadline - time.monotonic()
        if remaining < 3:
            break
        scale_factor = max(1, n_best // 2048)
        maxiter = min(2000, max(100, int(remaining * 0.7 / (0.02 * scale_factor))))
        maxfun = min(30000, maxiter * 20)
        sc, a_opt, _ = run_lbfgs(a_pert, 500, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"local #{trial} s={strategy}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ================================================================
    # Phase 4: Subgradient polish + final L-BFGS
    # ================================================================
    print(f"\n--- Phase 4: Final Polish ---")
    
    sc, a_pol = subgrad_polish(best_a, total_iters=1000)
    report(sc, a_pol, "subgrad1")
    
    # L-BFGS from polished point
    if time.monotonic() < deadline - 10:
        for beta in [3000, 6000, 12000]:
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            scale_factor = max(1, len(best_a) // 2048)
            maxiter = min(5000, max(500, int(remaining * 0.5 / (0.01 * scale_factor))))
            sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 20)
            report(sc, a_opt, f"final lbfgs beta={beta}")
    
    # Another subgradient pass
    sc, a_pol2 = subgrad_polish(best_a, total_iters=500)
    report(sc, a_pol2, "subgrad2")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
