"""
Optimizes non-negative sequence to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses: LSE-LBFGS polishing, LP cutting-plane direction finding with geometric blending,
peak-shaving local search, and multi-start exploration.
Key innovation over prior: LP cutting-plane (avoids full matrix build), geometric blending,
and adaptive peak-shaving based on KKT-inspired responsibility weighting.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """Linear cross-convolution a*b via FFT."""
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a via FFT
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B minimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

    using constraint generation. Returns (t_opt, b_opt) or (None, None).
    """
    n = len(a)
    S = float(a.sum())
    c_conv = fast_conv(a)

    # Start with top peaks
    top_k = np.argsort(c_conv)[::-1][:3]
    active_k = [int(k) for k in top_k]

    t_opt = None
    b_opt = None

    for iteration in range(max_cuts):
        m = len(active_k)
        # Build constraint matrix: m x (n+1)
        # Constraint i: sum_j a[k_i - j] * b_j - t <= 0
        A_ub = np.zeros((m, n + 1))
        for i, k in enumerate(active_k):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            js = np.arange(j_start, j_end + 1)
            A_ub[i, js] = a[k - js]
            A_ub[i, n] = -1.0

        b_ub = np.zeros(m)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])

        c_vec = np.zeros(n + 1)
        c_vec[n] = 1.0

        # Initial point
        x0 = np.concatenate([np.full(n, S / n), [float(c_conv.max())]])

        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0, None)] * n + [(None, None)],
                      method='highs', options={'presolve': True})

        if not res.success:
            # Try with a simpler constraint set
            if t_opt is not None:
                return t_opt, b_opt
            return None, None

        b_new = res.x[:n]
        t_new = res.x[n]

        if t_new < (t_opt if t_opt is not None else np.inf):
            t_opt = t_new
            b_opt = b_new.copy()

        # Check for violated constraints
        cross_conv = fast_conv_cross(a, b_new)
        violations = cross_conv - t_new
        max_viol = float(violations.max())

        if max_viol < tol * max(1.0, abs(t_new)):
            return t_opt, b_opt

        # Add most violated constraint
        worst_k = int(np.argmax(violations))
        if worst_k not in active_k:
            active_k.append(worst_k)
        elif len(active_k) < 5:
            # Add next most violated
            violations[worst_k] = -np.inf
            worst_k2 = int(np.argmax(violations))
            if violations[worst_k2] > tol and worst_k2 not in active_k:
                active_k.append(worst_k2)
        else:
            break

    return t_opt, b_opt


def geometric_blend(a, b, tau, S_target):
    """Geometric blend: a_new ∝ a^(1-tau) * b^tau, normalized to sum S_target."""
    eps = 1e-8
    a_f = np.maximum(a, eps)
    b_f = np.maximum(b, eps)
    a_new = a_f ** (1.0 - tau) * b_f ** tau
    s = a_new.sum()
    if s < 1e-12:
        return a.copy()
    a_new = a_new * (S_target / s)
    return np.clip(a_new, 0.0, 1000.0)


def linear_blend(a, b, tau, S_target):
    """Linear blend: a_new = (1-tau)*a + tau*b, normalized to sum S_target."""
    a_new = (1.0 - tau) * a + tau * b
    s = a_new.sum()
    if s < 1e-12:
        return a.copy()
    a_new = a_new * (S_target / s)
    return np.clip(a_new, 0.0, 1000.0)


def peak_shave(a, n_rounds=50, step_size=0.05):
    """
    Peak shaving: iteratively reduce elements responsible for the peak
    and redistribute mass to elements not in the peak's support.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    S = float(a.sum())

    for rnd in range(n_rounds):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        M = float(c[k_star])

        # Responsibility: for peak at k_star, a_j contributes a_j * a[k_star - j]
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        resp = np.zeros(n)
        js = np.arange(j_start, j_end + 1)
        resp[js] = a[js] * a[k_star - js]

        # Find top contributors and bottom contributors
        top_count = max(3, n // 100)
        top_idx = np.argsort(resp)[::-1][:top_count]

        # Elements NOT in the peak's active window
        in_window = np.zeros(n, dtype=bool)
        in_window[j_start:j_end + 1] = True
        out_idx = np.where(~in_window)[0]
        if len(out_idx) == 0:
            out_idx = np.where(resp == 0)[0]
        if len(out_idx) == 0:
            break

        # Reduce top contributors, add to random outside elements
        total_reduce = 0.0
        for j in top_idx:
            reduce_amt = a[j] * step_size
            a[j] -= reduce_amt
            a[j] = max(a[j], 0.0)
            total_reduce += reduce_amt

        if total_reduce > 0:
            n_add = min(len(out_idx), max(5, n // 50))
            add_idx = np.random.default_rng(rnd).choice(out_idx, size=n_add, replace=False)
            per_add = total_reduce / n_add
            for j in add_idx:
                a[j] += per_add

        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-12:
            a = best_a.copy()
            break
        a = a * (S / s)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            a = best_a.copy()
            # Decrease step size if no improvement
            step_size *= 0.7
            if step_size < 1e-4:
                break

    return best_sc, best_a


def kkt_polish(a, n_iters=200):
    """
    KKT-inspired polish: at optimum, the weighted anti-diagonal sum should be
    constant on the support. Use this to guide updates.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    S = float(a.sum())

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        # Soft weights on top convolutions
        margin = c_max * 0.05
        w = np.clip(1.0 - (c_max - c) / margin, 0, 1)
        w_sum = w.sum()
        if w_sum < 1e-12:
            break
        w = w / w_sum

        # Compute the "potential" phi_j = sum_k w_k * (a_{k-j} + a_{j-k})
        rev_a = a[::-1]
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        W = np.fft.rfft(w, M2_fft)
        conv_res = np.fft.irfft(RA * W, M2_fft)[n - 1:2 * n - 1]
        phi = 2.0 * conv_res

        # Update: reduce elements where phi > mean(phi on support)
        support = a > 1e-8
        if support.sum() < 10:
            break
        phi_support = phi[support]
        phi_mean = float(phi_support.mean())
        phi_std = float(phi_support.std()) + 1e-12

        # Multiplicative update: a_j *= exp(-eta * (phi_j - phi_mean) / phi_std)
        eta = 0.1
        log_update = -eta * (phi - phi_mean) / phi_std
        log_update = np.clip(log_update, -2.0, 2.0)
        a = a * np.exp(log_update)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-12:
            a = best_a.copy()
            break
        a = a * (S / s)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            a = best_a.copy()
            eta *= 0.8
            if eta < 1e-4:
                break

    return best_sc, best_a


def subgrad_polish(a, total_iters=300):
    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    S = float(a.sum())
    sc_lb = 1.0

    for it in range(total_iters):
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        js = np.arange(j_start, j_end + 1)
        sg[js] = 2.0 * a[k_star - js]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = min((sc - sc_lb) / (gn * gn + 1e-30), 0.1)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-12:
            a = best_a.copy()
            break
        a = a * (S / s)

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                print(f"Loaded parent: length={p.size}, score={compute_score(parent):.8f}")
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    if parent is not None:
        sc_p = compute_score(parent)
        report(sc_p, parent, "parent")
        a_work = parent.copy()
    else:
        a_work = np.ones(4096) * 0.5

    n_work = len(a_work)
    S_work = float(a_work.sum())
    print(f"Starting: n={n_work}, score={best_score:.8f}")

    # =========================================================================
    # Phase 1: Quick LSE-LBFGS polish (3 min)
    # =========================================================================
    phase1_dead = t_start + 180.0
    print(f"\n{'='*60}\nPhase 1: LSE-LBFGS polish\n{'='*60}")

    for beta in [50, 200, 800]:
        if time.monotonic() > phase1_dead:
            break
        sc, a_opt, _ = run_lbfgs(a_work, beta, maxiter=5000, maxfun=50000)
        report(sc, a_opt, f"LBFGS beta={beta}")
        a_work = a_opt

    print(f"After Phase 1: {best_score:.8f}")

    # =========================================================================
    # Phase 2: LP cutting-plane iteration with blending (10 min)
    # =========================================================================
    phase2_dead = t_start + 780.0
    print(f"\n{'='*60}\nPhase 2: LP cutting-plane + blend\n{'='*60}")

    a_lp = a_work.copy()
    S_lp = float(a_lp.sum())

    for lp_iter in range(15):
        if time.monotonic() > phase2_dead:
            break
        t_iter_start = time.monotonic()

        # Solve LP
        t_opt, b_opt = solve_lp_cuts(a_lp, max_cuts=200)
        if t_opt is None or b_opt is None:
            print(f"  LP failed at iter {lp_iter}, skipping")
            break

        t_elapsed = time.monotonic() - t_iter_start
        print(f"  LP iter {lp_iter}: t_opt={t_opt:.8f}, time={t_elapsed:.1f}s")

        # Try different blending strategies and step sizes
        best_this_round = compute_score(a_lp)
        best_a_round = a_lp.copy()

        for blend_type in ['linear', 'geometric']:
            for tau in [0.1, 0.2, 0.3, 0.5, 0.7]:
                if time.monotonic() > phase2_dead:
                    break
                if blend_type == 'linear':
                    a_cand = linear_blend(a_lp, b_opt, tau, S_lp)
                else:
                    a_cand = geometric_blend(a_lp, b_opt, tau, S_lp)

                # Quick polish
                sc, a_pol, _ = run_lbfgs(a_cand, 500, maxiter=2000, maxfun=20000)
                if sc < best_this_round:
                    best_this_round = sc
                    best_a_round = a_pol.copy()

        report(best_this_round, best_a_round, f"LP round {lp_iter}")
        a_lp = best_a_round
        S_lp = float(a_lp.sum())

    print(f"After Phase 2: {best_score:.8f}")

    # =========================================================================
    # Phase 3: KKT polish + peak shaving (4 min)
    # =========================================================================
    phase3_dead = t_start + 1020.0
    print(f"\n{'='*60}\nPhase 3: KKT polish + peak shaving\n{'='*60}")

    a_cur = best_a.copy()
    for it in range(10):
        if time.monotonic() > phase3_dead:
            break
        # KKT polish
        sc, a_kkt = kkt_polish(a_cur, n_iters=100)
        report(sc, a_kkt, f"KKT it={it}")
        a_cur = a_kkt

        # Peak shaving
        sc, a_ps = peak_shave(a_cur, n_rounds=30, step_size=0.04)
        report(sc, a_ps, f"peak_shave it={it}")
        a_cur = a_ps

        # Quick LSE polish
        if time.monotonic() > phase3_dead:
            break
        sc, a_opt, _ = run_lbfgs(a_cur, 1000, maxiter=3000, maxfun=30000)
        report(sc, a_opt, f"polish it={it}")
        a_cur = a_opt

    print(f"After Phase 3: {best_score:.8f}")

    # =========================================================================
    # Phase 4: Multi-start exploration (4 min)
    # =========================================================================
    phase4_dead = t_start + 1260.0
    print(f"\n{'='*60}\nPhase 4: Multi-start exploration\n{'='*60}")

    n_best = len(best_a)
    trial = 0
    while time.monotonic() < phase4_dead and trial < 20:
        strategy = trial % 5
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.05 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strategy == 1:
            # Frequency modulation perturbation
            x = np.arange(n_best, dtype=np.float64) / n_best
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.03 + 0.07 * rng.random()
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
        elif strategy == 2:
            # Sparse large perturbation
            n_sparse = max(20, n_best // 30)
            a_pert = best_a.copy()
            idx = rng.integers(0, n_best, size=n_sparse)
            a_pert[idx] *= (1.0 + 0.5 * rng.standard_normal(n_sparse))
        elif strategy == 3:
            # LP-based direction from perturbed start
            a_pert = best_a * (1.0 + 0.1 * rng.standard_normal(n_best))
            t_opt, b_opt = solve_lp_cuts(np.clip(a_pert, 0, 1000), max_cuts=50)
            if b_opt is not None:
                a_pert = linear_blend(best_a, b_opt, 0.4, float(best_a.sum()))
        else:
            # Mix with structured sequence
            x = np.arange(n_best, dtype=np.float64) / n_best
            f1 = 2.0 + 6.0 * rng.random()
            f2 = 5.0 + 10.0 * rng.random()
            a_struct = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
            a_struct = np.clip(a_struct, 0.01, None)
            a_struct = a_struct * (float(best_a.sum()) / a_struct.sum())
            a_pert = 0.7 * best_a + 0.3 * a_struct

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        sc, a_opt, _ = run_lbfgs(a_pert, 800, maxiter=5000, maxfun=50000)
        report(sc, a_opt, f"multi-start #{trial} (strat={strategy})")
        trial += 1

    print(f"After Phase 4: {best_score:.8f}")

    # =========================================================================
    # Phase 5: Another round of LP + polish on best (3 min)
    # =========================================================================
    phase5_dead = t_start + 1440.0
    print(f"\n{'='*60}\nPhase 5: Final LP + polish\n{'='*60}")

    a_final = best_a.copy()
    S_final = float(a_final.sum())
    for it in range(8):
        if time.monotonic() > phase5_dead:
            break
        t_opt, b_opt = solve_lp_cuts(a_final, max_cuts=100)
        if b_opt is None:
            break
        # Geometric blend with small tau
        for tau in [0.1, 0.2, 0.3]:
            a_cand = geometric_blend(a_final, b_opt, tau, S_final)
            sc, a_pol, _ = run_lbfgs(a_cand, 1500, maxiter=5000, maxfun=50000)
            report(sc, a_pol, f"final LP {it} tau={tau}")
            if sc < compute_score(a_final):
                a_final = a_pol
                S_final = float(a_final.sum())

    # =========================================================================
    # Phase 6: Subgradient + final high-beta polish (remaining time)
    # =========================================================================
    print(f"\n{'='*60}\nPhase 6: Final refinement\n{'='*60}")

    # Subgradient
    if time.monotonic() < deadline - 60.0:
        sc, a_sg = subgrad_polish(best_a, total_iters=500)
        report(sc, a_sg, "subgrad final")

    # High beta LSE
    for beta in [3000, 8000]:
        if time.monotonic() > deadline - 30.0:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=10000, maxfun=100000)
        report(sc, a_opt, f"final LBFGS beta={beta}")

    # One more KKT pass
    if time.monotonic() < deadline - 20.0:
        sc, a_kkt = kkt_polish(best_a, n_iters=200)
        report(sc, a_kkt, "final KKT")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
