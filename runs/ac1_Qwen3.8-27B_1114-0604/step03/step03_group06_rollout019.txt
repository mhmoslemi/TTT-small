def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    Mf = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, Mf)

def compute_score(a):

    n = len(a)

    S = float(a.sum())

def _grad_from_weights(x, w):
    """Compute gradient contribution from weighted convolution. w has length 2n-1."""
    n = len(x)
    ra = x[::-1]
    M2 = 3 * n - 2
    M2f = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(ra, M2f)
    P = np.fft.rfft(w, M2f)
    cr = np.fft.irfft(RA * P, M2f)[n - 1:2 * n - 1]
    return 2.0 * cr


def _obj_lse(x):
    """LSE-smoothed objective and gradient (uses global _g_beta)."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    scaled = _g_beta * (c - cm)
    es = np.exp(scaled)
    es_sum = es.sum()

    score = 2.0 * n * L / (S * S)
    p = es / es_sum
    g = _grad_from_weights(x, p)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)

def _compute_topk_weights(c, k):
    """Compute soft top-k weights from convolution values."""
    c_sorted = np.sort(c)[::-1]
    if k < len(c_sorted):
        threshold = c_sorted[k - 1]
        idx_below = min(k + 3, len(c_sorted) - 1)
        gap = max(float(c_sorted[k - 1] - c_sorted[idx_below]), 1e-8)
        width = gap * 0.5 + 1e-8
        w = 1.0 / (1.0 + np.exp(-10.0 * (c - threshold) / width))
    else:
        w = np.ones(len(c))
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.ones(len(c)) / len(c)
    else:
        w = w / w_sum
    return w

def run_lbfgs(a0, obj_func, bounds, maxiter, maxfun):

def peak_targeting_step(a, n_steps=20, step_scale=0.05):
    """
    Peak-targeting: identify contributing elements at the peak lag,
    reduce high contributors and increase low contributors.
    """
    n = len(a)
    a_cur = a.copy()
    best_sc = compute_score(a_cur)
    best_a = a_cur.copy()

    for _ in range(n_steps):
        c = fast_conv(a_cur)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break

        # Contribution of each element to the peak
        contrib = np.zeros(n)
        seg = a_cur[j_start:j_end + 1]
        contrib[j_start:j_end + 1] = 2.0 * seg[::-1]

        # Find top and bottom contributors
        top_n = max(2, (j_end - j_start + 1) // 10)
        bottom_n = max(5, n // 20)

        active_idx = np.arange(j_start, j_end + 1)
        top_idx = active_idx[np.argsort(contrib[active_idx])[::-1][:top_n]]
        # Bottom: elements outside the peak range
        outside_mask = np.zeros(n, dtype=bool)
        outside_mask[:j_start] = True
        outside_mask[j_end + 1:] = True
        outside_idx = np.where(outside_mask)[0]
        if len(outside_idx) > 0:
            bottom_idx = outside_idx[np.argsort(a_cur[outside_idx])[:bottom_n]]
        else:
            bottom_idx = active_idx[np.argsort(contrib[active_idx])[:bottom_n]]

        # Move mass from top to bottom
        total_move = 0.0
        for j in top_idx:
            move = a_cur[j] * step_scale
            a_cur[j] -= move
            a_cur[j] = max(a_cur[j], 0.0)
            total_move += move

        if len(bottom_idx) > 0 and total_move > 0:
            per = total_move / len(bottom_idx)
            for j in bottom_idx:
                a_cur[j] += per

        a_cur = np.clip(a_cur, 0.0, 1000.0)
        if a_cur.sum() < 0.01:
            a_cur = best_a.copy()
            break

        sc = compute_score(a_cur)
        if sc < best_sc:
            best_sc = sc
            best_a = a_cur.copy()
        else:
            # Revert and try smaller step
            a_cur = best_a.copy()
            step_scale *= 0.7
            if step_scale < 0.001:
                break

    return best_sc, best_a


def subgradient_polish(a, n_iters=200):
    """Exact subgradient descent on max-conv objective."""
    n = len(a)
    a_cur = a.astype(np.float64).copy()
    best_a = a_cur.copy()
    best_sc = compute_score(a_cur)
    sc_lb = 1.0

    for it in range(n_iters):
        c = fast_conv(a_cur)
        S = float(a_cur.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a_cur.copy()

        # Subgradient of max(c) w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a_cur[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.2)
        a_cur -= step * grad
        a_cur = np.clip(a_cur, 0.0, 1000.0)

    return best_sc, best_a


def solve_lp_direction(a, max_wall_s=25.0):
    """Solve LP to find best direction keeping convolution bounded."""
    if not HAVE_CVXPY:
        return None
    n = len(a)
    t0 = time.monotonic()
    if n > 8000:
        return None
    M = float(np.max(fast_conv(a)))
    if M < 1e-12:
        return a.copy()

    # Build constraint matrix (banded)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_mat[k_start:k_end, j] = a[:k_end - k_start]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=cp.PDLP, max_time=max_wall_s)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_time=max_wall_s)
        except Exception:
            return None

    if time.monotonic() - t0 > max_wall_s or b_var.value is None:
        return None

    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None
    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    return b_opt

            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine primary length
    if parent is not None:
        primary_len = parent.size
    else:
        primary_len = 8192

    print(f"CVXPY: {HAVE_CVXPY}, Parent len: {parent.size if parent is not None else 0}")
    print(f"Primary length: {primary_len}")

    # =========================================================================
    # Phase 1: Aggressive LSE continuation from parent (50% of time)
    # =========================================================================
    phase1_deadline = t_start + 0.50 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: LSE Continuation (deadline #{phase1_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    n_primary = primary_len
    bounds = [(0.0, 1000.0)] * n_primary

    # Start from parent at primary length
    if parent is not None:
        a_cur = parent.astype(np.float64).copy()
        if len(a_cur) != n_primary:
            idx = np.linspace(0, len(a_cur) - 1, n_primary).astype(int)
            a_cur = a_cur[idx].copy()
    else:
        a_cur = np.ones(n_primary) * 0.5

    sc0 = compute_score(a_cur)
    print(f"Starting score: {sc0:.10f}")
    report(sc0, a_cur, "init")

    # Beta continuation schedule - much more aggressive than before
    beta_schedule = [50, 100, 200, 400, 800, 1500, 3000, 5000, 8000, 15000, 25000, 40000, 60000]

    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 5.0:
            break

        # Scale iterations with remaining time
        maxiter = min(20000, max(500, int(remaining / 2.0)))
        maxfun = max(maxiter * 3, 1000)

        _g_beta = float(beta)
        print(f"  beta={beta}, remaining={remaining:.0f}s, maxiter={maxiter}")

        a_opt, res = run_lbfgs(a_cur, _obj_lse, bounds, maxiter, maxfun)
        sc = compute_score(a_opt)
        report(sc, a_opt, f"beta={beta}")
        a_cur = a_opt

        # Check convergence
        if res.success and res.status in [1, 2, 3]:
            # Try one more level of beta for extra precision
            pass

    print(f"After Phase 1: best = {best_score:.10f}")

    # =========================================================================
    # Phase 2: Multi-peak and top-k objective (15% of time)
    # =========================================================================
    phase2_deadline = t_start + 0.65 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: Multi-peak / Top-k (deadline #{phase2_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    a_mp = best_a.copy()
    for beta in [5000, 10000, 20000]:
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 5.0:
            break
        maxiter = min(10000, max(200, int(remaining / 3.0)))
        _g_beta = float(beta)
        a_opt, _ = run_lbfgs(a_mp, _obj_lse_multi, bounds, maxiter, maxiter * 3)
        sc = compute_score(a_opt)
        report(sc, a_opt, f"multi beta={beta}")
        a_mp = a_opt

    # Top-k objective
    for k in [3, 5, 8]:
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 5.0:
            break
        _g_topk_k = k
        maxiter = min(5000, max(200, int(remaining / 3.0)))
        a_opt, _ = run_lbfgs(a_mp, _obj_topk, bounds, maxiter, maxiter * 2)
        sc = compute_score(a_opt)
        report(sc, a_opt, f"topk={k}")
        a_mp = a_opt

    print(f"After Phase 2: best = {best_score:.10f}")

    # =========================================================================
    # Phase 3: LP direction + peak targeting (20% of time)
    # =========================================================================
    phase3_deadline = t_start + 0.85 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: LP + Peak Targeting (deadline #{phase3_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    a_lp = best_a.copy()
    n_lp = len(a_lp)
    bounds_lp = [(0.0, 1000.0)] * n_lp

    if HAVE_CVXPY and n_lp <= 8000:
        for it_lp in range(20):
            if time.monotonic() > phase3_deadline:
                break
            remaining = phase3_deadline - time.monotonic()
            if remaining < 10.0:
                break

            max_wall = min(20.0, remaining * 0.4)
            b_opt = solve_lp_direction(a_lp, max_wall_s=max_wall)
            if b_opt is not None:
                # Try line search
                best_local = a_lp
                best_local_sc = compute_score(a_lp)
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                    a_try = (1.0 - t) * a_lp + t * b_opt
                    a_try = np.clip(a_try, 0.0, 1000.0)
                    if a_try.sum() < 0.01:
                        continue
                    sc_try = compute_score(a_try)
                    if sc_try < best_local_sc:
                        best_local_sc = sc_try
                        best_local = a_try
                report(best_local_sc, best_local, f"LP {it_lp}")
                a_lp = best_local

            # Polish after LP step
            if time.monotonic() > phase3_deadline:
                break
            _g_beta = 10000.0
            maxiter = min(3000, max(100, int((phase3_deadline - time.monotonic()) / 5.0)))
            a_opt, _ = run_lbfgs(a_lp, _obj_lse, bounds_lp, maxiter, maxiter * 2)
            sc = compute_score(a_opt)
            report(sc, a_opt, f"LP+polish {it_lp}")
            a_lp = a_opt

    # Peak targeting steps
    for it_pt in range(10):
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 5.0:
            break
        sc_pt, a_pt = peak_targeting_step(a_lp, n_steps=30, step_scale=0.03)
        report(sc_pt, a_pt, f"peak_target {it_pt}")
        # Polish
        _g_beta = 5000.0
        maxiter = min(2000, max(100, int(remaining / 5.0)))
        a_opt, _ = run_lbfgs(a_pt, _obj_lse, bounds_lp, maxiter, maxiter * 2)
        sc = compute_score(a_opt)
        report(sc, a_opt, f"peak+polish {it_pt}")
        a_lp = a_opt

    print(f"After Phase 3: best = {best_score:.10f}")

    # =========================================================================
    # Phase 4: Iterated local search with perturbations (10% of time)
    # =========================================================================
    phase4_deadline = deadline
    print(f"\n{'='*60}")
    print(f"Phase 4: Iterated Local Search (deadline #{phase4_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    bounds_best = [(0.0, 1000.0)] * n_best
    trial = 0
    while time.monotonic() < phase4_deadline and trial < 30:
        remaining = phase4_deadline - time.monotonic()
        if remaining < 10.0:
            break
        strategy = trial % 5

        if strategy == 0:
            # Gaussian perturbation
            scale = 0.01 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Peak-targeted perturbation
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            a_pert[j_start:j_end+1] *= (1.0 - 0.05 * rng.random(j_end - j_start + 1))
            # Redistribute
            total_red = best_a[j_start:j_end+1].sum() - a_pert[j_start:j_end+1].sum()
            outside = np.concatenate([np.arange(j_start), np.arange(j_end+1, n_best)])
            if len(outside) > 0:
                a_pert[outside] += total_red / len(outside)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Frequency perturbation
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Sparse perturbation
            n_sparse = max(10, n_best // 30)
            a_pert = best_a.copy()
            idx = rng.integers(0, n_best, size=n_sparse)
            a_pert[idx] *= (1.0 + 0.2 * rng.standard_normal(n_sparse))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Mix with structured sequence
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_struct = 0.5 + 0.3 * np.cos(2.0 * np.pi * rng.random() * 5 * x + rng.random() * 6.28)
            a_struct = np.clip(a_struct, 0.01, 1000.0)
            a_struct = a_struct * (best_a.sum() / a_struct.sum())
            a_pert = 0.8 * best_a + 0.2 * a_struct
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        if a_pert.sum() < 0.01:
            a_pert = best_a.copy()

        # Polish with high-beta LSE
        _g_beta = 10000.0
        maxiter = min(5000, max(200, int(remaining / 4.0)))
        a_opt, _ = run_lbfgs(a_pert, _obj_lse, bounds_best, maxiter, maxiter * 2)
        sc = compute_score(a_opt)
        report(sc, a_opt, f"ILS #{trial} (s={strategy})")
        trial += 1

    print(f"After Phase 4: best = {best_score:.10f}")

    # =========================================================================
    # Phase 5: Final ultra-high-beta polish
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Polish")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        for beta in [30000, 50000, 80000]:
            if time.monotonic() > deadline - 3.0:
                break
            _g_beta = float(beta)
            maxiter = min(10000, max(100, int((deadline - time.monotonic()) / 3.0)))
            a_opt, _ = run_lbfgs(best_a, _obj_lse, bounds_best, maxiter, maxiter * 2)
            sc = compute_score(a_opt)
            report(sc, a_opt, f"final beta={beta}")
            best_a = a_opt

    # Subgradient final touch
    if time.monotonic() < deadline - 2.0:
        sc_sg, a_sg = subgradient_polish(best_a, n_iters=100)
        report(sc_sg, a_sg, "subgrad final")

    # One last L-BFGS
    if time.monotonic() < deadline - 1.0:
        _g_beta = 100000.0
        a_opt, _ = run_lbfgs(best_a, _obj_lse, bounds_best, 5000, 10000)
        sc = compute_score(a_opt)
        report(sc, a_opt, "absolute final")

    # =========================================================================
    # Return
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
