def evaluate_sequence(sequence):
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = list(np.convolve(sequence, sequence, mode="full"))
    max_b = max(b_sequence) if b_sequence else 0
    sum_a = sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * float(max_b) / (sum_a ** 2)

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M.bit_length() - 1)
    if M_fft < M:
        M_fft = M
    A = np.zeros(M_fft, dtype=np.float64)
    np.put(A, np.arange(len(a)), a)
    RA = np.fft.rfft(A, M_fft)
    convolve_fft = np.fft.irfft(np.conj(RA) * RA, M_fft)
    convolve_fft = convolve_fft[:M]
    return convolve_fft

def compute_score(a):
    n = len(a)
    if n == 0:
        return np.inf
    c = fast_conv(a)
    S = float(c.sum())
    if S < 1e-12:
        return np.inf
    max_c = float(c.max())
    score = 2.0 * n * max_c / (S ** 2)
    return score

def lse_score_grad(a, beta):
    n = len(a)
    if n == 0:
        return np.inf
    c = fast_conv(a)
    S = c.sum()
    if S < 1e-12:
        return np.inf
    c_max = c.max()
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S ** 2)
    # Compute gradient: LSE-smoothed convolution + log-sum-exponential term
    p = exp_scaled / exp_scaled.sum()
    # Create reversed sequence without cycle shift penalties
    p_rev = np.zeros(n, dtype=np.float64)
    for i in range(n):
        for j in range(i, n):
            p_rev[i + j] += p[j] * p[i]
    # Gradient is 2 * conv(p_rev, p) then multiply by beta correction
    cross_conv = np.zeros(n, dtype=np.float64)
    for i in range(n):
        cross_conv[i] = p_rev[i] * p[i] * beta * beta
    grad = cross_conv[:-1] / (S * S) * float(a.sum()) * float(a.sum())
    return score, grad

def lse_score(a):
    n = len(a)
    if n == 0:
        return np.inf
    score, grad = lse_score_grad(a, beta=10.0)  # Default beta for surrogate
    if score < 0 or not np.isfinite(score):
        score = 10.0**6
    return score
