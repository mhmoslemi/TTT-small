import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)

def topk_gradient(a, c, top_k=20, beta=50):
    """
    Compute gradient of the top-K LSE smoothed objective.
    Returns (smoothed_score, gradient).
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    # Top-K lags
    k_idx = np.argsort(c)[::-1][:top_k]
    c_top = c[k_idx]
    c_max = c_top[0]
    
    # LSE weights
    scaled = beta * (c_top - c_max)
    scaled -= scaled.max()  # numerical stability
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    L = c_max + np.log(exp_s.sum()) / beta
    
    # h[j] = sum_{k in topk} p_k * 2 * a[k-j] (off-diagonal part)
    h = np.zeros(n)
    for idx in range(top_k):
        k = int(k_idx[idx])
        w = 2.0 * p[idx]
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            h[js:je+1] += w * a[k-je:k-js+1][::-1]
    
    # Diagonal correction: for even k, the term at j=k/2 should be a[j] not 2*a[j]
    for idx in range(top_k):
        k = int(k_idx[idx])
        if k % 2 == 0:
            m = k // 2
            if 0 <= m < n:
                h[m] -= p[idx] * a[m]
    
    # Full gradient of (2n/S^2) * L
    # d/da_j [(2n/S^2) * L] = (2n/S^2) * dL/da_j - (2n * 2L * S_j) / S^3
    # where S_j = 1 (derivative of sum w.r.t. a_j)
    # dL/da_j = h[j] (from above, h already includes the factor of 2 for off-diagonal)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (h / S2 - 2.0 * L / S3)
    
    score = (2.0 * n * L) / S2
    return score, grad


def adam_topk(a_init, lr=0.003, n_steps=100000, beta=50, top_k=20,
              time_limit=None, t_start=None, normalize=True,
              verbose_interval=50000):

    n = len(a_init)
    a = a_init.astype(np.float64).copy()

def mirror_descent(a_init, lr=0.01, n_steps=50000, beta=50, top_k=20,

    """

def block_coord_descent(a, block_size=32, n_sweeps=50, time_limit=None, t_start=None):

    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    
    deadline = np.inf
    if time_limit is not None and t_start is not None:
        deadline = t_start + time_limit
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        for start in range(0, n, block_size):
            if time.monotonic() > deadline:
                break
            
            end = min(start + block_size, n)
            a_block = a[start:end].copy()
            
            # Try a few perturbations of this block
            best_block = a_block.copy()
            best_sc = best_score
            
            # Gradient direction
            c = fast_conv(a)
            S = a.sum()
            k_star = c.argmax()
            
            # Compute gradient for this block
            js = max(0, k_star - n + 1)
            je = min(k_star, n - 1)
            
            # Try reducing elements that contribute to the peak
            for ratio in [0.5, 0.7, 0.85]:
                trial = a.copy()
                for j in range(start, end):
                    if js <= j <= je:
                        trial[j] *= ratio
                sc = compute_score(trial)
                if sc < best_sc:
                    best_sc = sc
                    best_block = trial[start:end].copy()
            
            # Try increasing non-contributing elements
            for ratio in [1.2, 1.5, 2.0]:
                trial = a.copy()
                for j in range(start, end):
                    if not (js <= j <= je):
                        trial[j] *= ratio
                # Renormalize
                if trial.sum() > 1e-12:
                    trial = trial / trial.sum() * S
                sc = compute_score(trial)
                if sc < best_sc:
                    best_sc = sc
                    best_block = trial[start:end].copy()
            
            a[start:end] = best_block
            if best_sc < best_score:
                best_score = best_sc
    
    return best_score, a


def lse_lbfgs(a0, beta, maxiter=5000, maxfun=100000, top_k=30):
    """L-BFGS-B with top-K LSE smoothed gradient."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    
    def obj_grad(a):
        c = fast_conv(a)
        return topk_gradient(a, c, top_k=top_k, beta=beta)
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        obj_grad, a0, jac=True,
        method='L-BFGS-B', bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def spectral_init(n, mode='cosine'):
    """Generate initial sequence using spectral methods."""
    if mode == 'cosine':
        # Cosine envelope (smooth, peaks in middle)
        x = np.arange(n) / n
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif mode == 'sinc':
        # Sinc-like (concentrated in middle)
        x = np.arange(n) / n - 0.5
        a = np.sinc(3 * x) ** 2
        a = np.clip(a, 0, None)
    elif mode == 'gaussian':
        x = np.arange(n) / n - 0.5
        a = np.exp(-20 * x**2)
    elif mode == 'bimodal':
        x = np.arange(n) / n
        a = np.exp(-30*(x-0.25)**2) + np.exp(-30*(x-0.75)**2)
    else:
        a = np.ones(n) / n
    
    a = np.clip(a, 0, None)
    s = a.sum()
    if s > 0:
        a = a / s
    return a
