```python
import numpy as np
import cvxpy as cp
import time

def lp_iteration(f, M, solver=cp.CLARABEL):
    """
    Given f (non-negative, sum > 0) and M > 0, solve:
    maximize sum(b)
    subject to: b >= 0
                (f * b)_k <= M for all k = 0, ..., 2n-2
    where (f*b) is the linear convolution.
    """
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    
    # Build convolution constraints
    # (f * b)_k = sum_{i=max(0,k-n+1)}^{min(k,n-1)} f[i] * b[k-i]
    # We need this <= M for all k
    
    # For efficiency, build as a sparse matrix
    # Row k: entries at column (k-i) for valid i
    # Actually, row k of the constraint matrix has:
    # entry at column j = f[k-j] if 0 <= k-j < n
    
    # Let's build it as a list of (row, col, value) tuples
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i  # column index for b
            rows.append(k)
            cols.append(j)
            vals.append(f[i])
    
    A = cp.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver)
    
    if b.value is None:
        return None
    return np.clip(b.value, 0, None)
```

def build_conv_matrix(f):
    n = len(f)
    # A has shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n
    rows = []

    data_list = []

    for i, k in enumerate(row_indices):
        j_min = max(0, int(k) - n + 1)
        j_max = min(int(k), n - 1)
        if j_max <= j_min:
            continue
        js = np.arange(j_min, j_max)
        row_list.extend([i] * len(js))
        col_list.extend(js.tolist())
        data_list.extend(f[int(k) - js].tolist())

    if len(row_list) == 0:
        return csc_matrix((n_active, n))

    rows_arr = np.array(row_list, dtype=np.int32)
    cols_arr = np.array(col_list, dtype=np.int32)
    data_arr = np.array(data_list, dtype=np.float64)
    return csc_matrix((data_arr, (rows_arr, cols_arr)), shape=(n_active, n))


def lp_solve(f, M, n_active=150, solver_str='CLARABEL'):
    """
    Solve LP: maximize sum(b) s.t. b >= 0, (f*b)_k <= M for active k.
    Returns b or None on failure.
    """
    n = len(f)
    conv_ff = fast_conv(f)

    # Select active constraints
    active_idx = np.argsort(conv_ff)[::-1][:n_active]

    b_var = cp.Variable(n, nonneg=True)

    A = build_conv_matrix(f, active_idx)
    rhs = np.full(len(active_idx), M)

    constraints = [A @ b_var <= rhs]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=solver_str)
    except Exception:
        try:
            prob.solve(solver='ECOS')
        except Exception:
            return None

    if b_var.value is None:
        return None

    b_val = np.clip(np.asarray(b_var.value, dtype=np.float64), 0, None)

    # Check for violated constraints and re-solve if needed
    conv_fb = conv_ab(f, b_val)
    violations = conv_fb - M
    if violations.max() > 1e-6 * M:
        violated_idx = np.where(violations > 1e-6 * M)[0]
        all_active = np.unique(np.concatenate([active_idx, violated_idx]))
        if len(all_active) < 5 * n:  # Safety bound
            A2 = build_conv_matrix(f, all_active)
            rhs2 = np.full(len(all_active), M)
            constraints2 = [A2 @ b_var <= rhs2]
            prob2 = cp.Problem(objective, constraints2)
            try:
                prob2.solve(solver=solver_str)
            except Exception:
                pass
            if b_var.value is not None:
                b_val = np.clip(np.asarray(b_var.value, dtype=np.float64), 0, None)

    return b_val if b_val.sum() > 1e-10 else None


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    # Gradient: d/da_j of 2n*L/S^2
    # = 2n * (dL/da_j / S^2 - 2L * da_j_sum / S^3)
    # dL/da_j = sum_k p_k * d(conv_k)/da_j
    # d(conv_k)/da_j = a[k-j] + a[j-k] (considering both terms in the convolution)
    # Actually for self-conv: conv_k = sum_i a[i]*a[k-i]
    # d(conv_k)/da_j = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    # = a[k-j] + a[j-k] but one of these might be out of range
    # More carefully: d(conv_k)/da_j = a[k-j] if 0<=k-j<n, plus a[j-k] if 0<=j-k<n
    # But since conv_k = sum_i a_i * a_{k-i}, d/da_j gives:
    #   a_{k-j} (from the term where i=j) + a_{j-k} (from the term where k-i=j, i.e. i=k-j)
    # Wait, that's not right. conv_k = sum_i a[i] * a[k-i]
    # d/da_j: appears when i=j (gives a[k-j]) and when k-i=j i.e. i=k-j (gives a[j])
    # So d(conv_k)/da_j = a[k-j] (if 0<=k-j<n) + a[j] * delta_{i=k-j} ... no
    # Let me redo: conv_k = sum_i a[i] * a[k-i]
    # The term involving a[j] is: a[j]*a[k-j] (when i=j) + a[k-j]*a[j] (when i=k-j)
    # If j != k-j (i.e., 2j != k), these are two separate terms, so d(conv_k)/da_j = a[k-j] + a[k-j] = 2*a[k-j]
    # If j == k-j (i.e., 2j == k), it's a[j]^2, so d(conv_k)/da_j = 2*a[j]
    # In both cases: d(conv_k)/da_j = 2*a[k-j] if 0 <= k-j < n, else (a[j] if 2j==k... no)
    # Actually: d(conv_k)/da_j = a[k-j] (from first factor) + a[j] (from second factor, when i=k-j)
    # Wait no. conv_k = sum_i a[i] * a[k-i]. The derivative w.r.t. a[j]:
    # - When i = j: derivative of a[j]*a[k-j] w.r.t. a[j] = a[k-j]
    # - When k-i = j, i.e., i = k-j: derivative of a[k-j]*a[j] w.r.t. a[j] = a[k-j]
    # If j != k-j: total = 2*a[k-j]
    # If j == k-j: total = 2*a[j] (since a[k-j] = a[j])
    # So in all cases: d(conv_k)/da_j = 2*a[k-j] if 0 <= k-j < n
    # But wait, we also need 0 <= i < n for the term to exist.
    # The term i=j exists if 0 <= j < n (always true) AND 0 <= k-j < n.
    # The term i=k-j exists if 0 <= k-j < n AND 0 <= j < n (always true).
    # So: d(conv_k)/da_j = 2*a[k-j] if 0 <= k-j < n, else 0.
    #
    # Therefore: dL/da_j = 2 * sum_k p[k] * a[k-j] (where 0 <= k-j < n)
    # = 2 * (p * rev_a)[j] ... let me think
    # sum_k p[k] * a[k-j] where k-j ranges from 0 to n-1, so k ranges from j to j+n-1
    # = sum_{m=0}^{n-1} p[j+m] * a[m]
    # This is a cross-correlation of p and a!
    # = (p * a_rev)[n-1-j] where a_rev = a[::-1]
    # Or equivalently: it's the (linear) convolution of p (reversed) with a, evaluated at appropriate index.
    
    # Let me just compute it directly using the formula:
    # grad_L[j] = 2 * sum_{k: 0<=k-j<n} p[k] * a[k-j]
    # = 2 * sum_{m=0}^{n-1} p[j+m] * a[m] (where j+m < 2n-1)
    
    # Using FFT: this is a cross-correlation
    # cross_corr[j] = sum_m p[j+m] * a[m] = sum_m a[m] * p[j+m]
    # This is the (n-1-j)th element of conv(a, p) ... let me verify
    # conv(a,p)[k] = sum_i a[i] * p[k-i]
    # We want sum_m a[m] * p[j+m] = conv(a,p)[j + 2m]... no that's not right
    # Let me use: sum_m a[m] * p[j+m] = sum_i a[i] * p[j+i]
    # conv(a_reversed, p)[j] = sum_i a_reversed[i] * p[j-i] = sum_i a[n-1-i] * p[j-i]
    # Let m = n-1-i: = sum_m a[m] * p[j-n+1+m]
    # So sum_m a[m] * p[j+m] = conv(a_reversed, p)[j + n - 1]
    
    # So dL/da_j = 2 * conv(a_rev, p)[j + n - 1] for j = 0, ..., n-1
    # That gives indices n-1 to 2n-2 of conv(a_rev, p), which has length 2n-1. ✓
    
    S2 = S * S
    S3 = S2 * S
    # conv_res already computed above is conv(rev_a, p) which is conv(a[::-1], p)
    # dL/da_j = 2 * conv_res[j + n - 1]
    dL = 2.0 * conv_res[n - 1:2 * n - 1]
    grad = (2.0 * n) * (dL / S2 - 2.0 * L / S3)
    grad += (2.0 * n) * (-2.0 * L / S3)  # Wait, let me redo
    # score = 2n*L/S^2
    # d(score)/da_j = 2n * (dL/da_j / S^2 - 2*L * dS/da_j / S^3)
    # dS/da_j = 1
    # So grad_j = 2n * (dL_j / S^2 - 2L / S^3)
    grad = (2.0 * n) * (dL / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=80000):
    """L-BFGS-B in a-space with bounds and LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def selective_coord_descent(a, top_k=50, n_sweeps=20, time_limit_s=None, t_start=None):
    """Coordinate descent on top-K coordinates by gradient magnitude."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        k_star = int(c.argmax())
        score = 2.0 * n * float(c[k_star]) / (S * S)
        if score < best_score:
            best_score = score

        # Subgradient at the peak lag
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        grad = np.zeros(n)
        idx = k_star - np.arange(js, je + 1)
        grad[js:je + 1] = 2.0 * a[idx]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad / S2 - 2.0 * c[k_star] / S3)

        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-15:
                continue
            if grad[j] > 1e-12:
                ratios = np.array([0.3, 0.5, 0.7, 0.8, 0.9, 1.0, 0.95])
            else:
                ratios = np.array([1.0, 1.2, 1.5, 2.0, 3.0])
                if a_old < 1e-10:
                    ratios = np.array([1e-10, 0.01, 0.1, 1.0])
            candidates = a_old * ratios
            candidates = np.clip(candidates, 0.0, 1000.0)

            best_x = a_old
            best_sc = score
            for x in candidates:
                if x == a_old:
                    continue
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: LP active-set iterations + L-BFGS polish + coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 40.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        parent_score = compute_score(parent)
        print(f"Parent: n={n}, score={parent_score:.8f}")
    else:
        n = 4096
        parent = np.ones(n) / n
        parent_score = compute_score(parent)
        print(f"No parent, uniform n={n}, score={parent_score:.8f}")

    best_score = parent_score
    best_a = parent.copy()

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.10f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ================================================================
    # Phase 1: LP active-set iterations (~60% of time)
    # ================================================================
    p1 = effective_budget * 0.60
    print(f"\n{'='*55}")
    print(f"Phase 1: LP Active-Set (budget={p1:.0f}s)")
    print(f"{'='*55}")

    # Normalize to sum = sqrt(2n)
    f = best_a.copy()
    target_sum = np.sqrt(2.0 * n)
    f = f * (target_sum / f.sum())
    M = float(fast_conv(f).max())
    print(f"  Initial: M={M:.10f}, score={compute_score(f):.10f}")

    # Try different solver
    solver_choice = 'CLARABEL'
    try:
        _ = lp_solve(f, M, n_active=50, solver_str='CLARABEL')
    except Exception:
        solver_choice = 'ECOS'
        print("  Using ECOS solver")

    iter_count = 0
    lp_deadline = t_start + p1
    prev_M = M

    while time.monotonic() < lp_deadline:
        iter_count += 1
        t_iter = time.monotonic()

        # Adaptive active set size
        if iter_count < 20:
            n_act = 100
        elif iter_count < 80:
            n_act = 200
        else:
            n_act = 400

        b = lp_solve(f, M, n_active=n_act, solver_str=solver_choice)
        t_lp = time.monotonic() - t_iter

        if b is None:
            print(f"  Iter {iter_count}: LP failed")
            # Try with smaller active set
            b = lp_solve(f, M, n_active=50, solver_str=solver_choice)
            if b is None:
                break

        # Rescale b to match target sum
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        b = b * (target_sum / b_sum)

        # Find best t via line search
        best_t = 0.1
        best_new_M = M
        for t in [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
            f_trial = (1.0 - t) * f + t * b
            new_M = float(fast_conv(f_trial).max())
            if new_M < best_new_M:
                best_new_M = new_M
                best_t = t

        if best_new_M >= M - 1e-14:
            # No improvement, try smaller active set or break
            if iter_count > 30 and best_new_M >= M - 1e-12:
                print(f"  Iter {iter_count}: Stuck at M={M:.10f}, LP_time={t_lp:.1f}s")
                break

        f = (1.0 - best_t) * f + best_t * b
        M = best_new_M

        sc = compute_score(f)
        if sc < best_score:
            try_update(sc, f, f"LP iter {iter_count}")
        elif iter_count % 20 == 0:
            print(f"  Iter {iter_count}: M={M:.10f}, score={sc:.10f}, t={best_t:.3f}, lp={t_lp:.1f}s")

        # Check convergence
        if M < prev_M * (1 - 1e-10) and iter_count > 100:
            pass  # Still making progress
        elif M >= prev_M - 1e-12 and iter_count > 50:
            # Plateau, try perturbation
            if iter_count % 50 == 0:
                f_pert = f * (1.0 + 0.02 * rng.standard_normal(n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert *= target_sum / f_pert.sum()
                M_pert = float(fast_conv(f_pert).max())
                if M_pert < M:
                    f = f_pert
                    M = M_pert
                    print(f"  Perturbation helped: M={M:.10f}")
        prev_M = M

    print(f"  LP phase done: {iter_count} iterations, M={M:.10f}")

    # ================================================================
    # Phase 2: L-BFGS polish (~25% of time)
    # ================================================================
    p2 = effective_budget * 0.25
    print(f"\n{'='*55}")
    print(f"Phase 2: L-BFGS Polish (budget={p2:.0f}s)")
    print(f"{'='*55}")

    lbfgs_deadline = t_start + p1 + p2
    for beta in [300, 1000, 3000, 8000]:
        if time.monotonic() > lbfgs_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=80000)
        try_update(sc, a_opt, f"L-BFGS beta={beta}")
        print(f"  L-BFGS beta={beta}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # ================================================================
    # Phase 3: Coordinate descent (~10% of time)
    # ================================================================
    p3 = effective_budget * 0.10
    print(f"\n{'='*55}")
    print(f"Phase 3: Coordinate Descent (budget={p3:.0f}s)")
    print(f"{'='*55}")

    t0 = time.monotonic()
    sc, a_opt = selective_coord_descent(best_a, top_k=80, n_sweeps=50,
                                         time_limit_s=p3, t_start=t_start)
    try_update(sc, a_opt, "Coord descent")
    print(f"  Coord descent: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # ================================================================
    # Phase 4: Restarts + final polish (~5% of time)
    # ================================================================
    print(f"\n{'='*55}")
    print(f"Phase 4: Restarts + Final Polish")
    print(f"{'='*55}")

    final_deadline = t_start + effective_budget - 10.0
    n_best = len(best_a)

    # A few targeted perturbations + quick LBFGS
    for r in range(8):
        if time.monotonic() > final_deadline:
            break

        strategy = r % 4
        if strategy == 0:
            a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(n_best))
        elif strategy == 1:
            # Reduce peak contributors
            c = fast_conv(best_a)
            k_star = int(c.argmax())
            js = max(0, k_star - n_best + 1)
            je = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            a_pert[js:je+1] *= 0.7
            removed = best_a[js:je+1].sum() - a_pert[js:je+1].sum()
            mask = (np.arange(n_best) < js) | (np.arange(n_best) > je)
            if mask.sum() > 0:
                a_pert[mask] += removed / mask.sum()
        elif strategy == 2:
            # Add small sinusoidal perturbation
            freq = 1.0 + 4.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + 0.015 * np.cos(2*np.pi*freq*x + phase))
        else:
            # Random block modification
            center = int(rng.integers(n_best//4, 3*n_best//4))
            width = max(10, n_best // 30)
            a_pert = best_a.copy()
            a_pert[max(0,center-width):min(n_best,center+width)] *= rng.uniform(0.5, 1.5)

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick LBFGS
        time_left = final_deadline - time.monotonic()
        if time_left > 5:
            sc, a_r = run_lbfgs(a_pert, 2000, maxiter=3000, maxfun=40000)
            try_update(sc, a_r, f"restart#{r} s{strategy}")

    # Final high-precision L-BFGS
    if time.monotonic() < final_deadline:
        sc, a_opt = run_lbfgs(best_a, 15000, maxiter=8000, maxfun=120000)
        try_update(sc, a_opt, "final LBFGS high-beta")

    # ================================================================
    # Final verification and return
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
