"""
Hybrid optimizer combining frequency-domain spectral flattening, mirror descent,
L-BFGS with log-sum-exp smoothing, and coordinate descent.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a, b=None):
    """Linear convolution via FFT. If b is None, computes self-convolution."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    M = 1 << (n + m - 1).bit_length()
    A = np.fft.rfft(a, M)
    B = np.fft.rfft(b, M)
    return np.fft.irfft(A * B, M)[:n + m - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clamp for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    # p[k] = exp(beta*(c[k]-c_max)) / sum
    p = exp_scaled / exp_sum
    
    # dL/da[j] = (2/beta) * sum_k p[k] * a[k-j]
    # = (2/beta) * r[j] where r[j] = sum_i a[i] * p[i+j]
    # r[j] = conv(a, p_flipped)[2n-2-j]
    p_flipped = p[::-1]
    conv_ap = fast_conv(a, p_flipped)
    # conv_ap has length n + (2n-1) - 1 = 3n-2
    # r[j] = conv_ap[2n-2-j] for j=0,...,n-1
    r = conv_ap[2*n-2 : 2*n-2-n : -1]  # or conv_ap[n-1:2*n-1][::-1]
    
    dL_da = (2.0 / beta) * r
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    
    return score, grad


def lse_objective(a, beta):
    return lse_score_and_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    S = a0.sum()
    a0 = a0 / S  # normalize
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0, None
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-12:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def spectral_flatten(a, n_iters=200, alpha_start=0.3, alpha_end=0.9, 
                     gamma=0.5, M_fft=None):
    """
    Frequency-domain iterative flattening.
    Modifies the power spectrum to make the convolution flatter,
    then projects back to non-negative time domain.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    if M_fft is None:
        M_fft = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        # Interpolate alpha
        t = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * t
        
        # Compute current convolution
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        C = np.abs(A) ** 2  # = FFT of circular conv = FFT of linear conv (padded)
        c = np.fft.irfft(C, M_fft)[:2*n-1]
        
        # Modify to flatten
        c_mean = float(c.mean())
        c_max = float(c.max())
        c_std = float(c.std())
        
        # Soft threshold: reduce values above threshold
        threshold = c_mean + (1.0 - gamma) * (c_max - c_mean)
        excess = np.maximum(c - threshold, 0.0)
        
        c_mod = c.copy()
        c_mod[:2*n-1] = c - alpha * excess
        
        # Ensure non-negative and has correct total sum
        c_mod = np.maximum(c_mod, 0)
        # Preserve total sum (which equals (sum a)^2 = 1)
        c_mod_sum = float(c_mod[:2*n-1].sum())
        if c_mod_sum < 1e-12:
            break
        c_mod[:2*n-1] = c_mod[:2*n-1] * (1.0 / c_mod_sum)
        
        C_target = np.fft.rfft(c_mod)[:M_fft//2+1]
        
        # Update spectrum: preserve phase, set new amplitude
        C_current = np.abs(A) ** 2
        ratio = np.sqrt(np.maximum(C_target, 0) / (C_current + 1e-15))
        A_new = A * ratio
        
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = float(a_new.sum())
        if a_new_sum < 1e-12:
            break
        a_new = a_new / a_new_sum
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def spectral_flatten_power(a, n_iters=200, beta_start=0.7, beta_end=0.3, M_fft=None):
    """
    Alternative: power compression in frequency domain.
    |A_new| = |A|^beta * |A|^(1-beta) = ... compressed amplitude.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n

    if M_fft is None:

    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        beta = beta_start + (beta_end - beta_start) * t
        
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        
        # Power compression: |A_new| = |A| * (|A| / |A|_rms)^(beta-1)
        # This compresses the dynamic range of the amplitude
        amp = np.abs(A)
        amp_rms = float(np.sqrt(np.mean(amp**2)))
        if amp_rms < 1e-15:
            break
        # Compress: raise to power < 1 to reduce peaks
        amp_new = amp * (amp / amp_rms) ** (beta - 1.0)
        # This makes large amplitudes smaller and small ones larger
        
        phase = A / (amp + 1e-30)
        A_new = amp_new * phase
        
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = float(a_new.sum())
        if a_new_sum < 1e-12:
            break
        a_new = a_new / a_new_sum
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def mirror_descent(a, n_iters=5000, eta=0.01):
    """
    Multiplicative mirror descent (Weisstein updates) for minimizing max convolution.
    Natural for the non-negative simplex.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound for score
    
    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        c_max = float(c[k_star])
        S = float(a.sum())
        sc = 2.0 * n * c_max / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient: g[j] = 2*a[k_star - j] if k_star - j in [0, n-1]
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        g = np.zeros(n)
        if j_end >= j_start:
            g[j_start:j_end+1] = 2.0 * a[k_star - j_end : k_star - j_start : -1]
        
        # Adaptive step size (Polyak-like)
        gap = sc - lower_bound
        g_norm = float(np.sum(a * g))  # for multiplicative updates, this is the relevant norm
        if g_norm < 1e-15:
            break
        eta_adapt = min(eta, max(0.0001, gap / (g_norm + 1e-10) * 0.1))
        
        # Multiplicative update
        log_update = -eta_adapt * g
        log_update = np.clip(log_update, -50, 50)  # numerical safety
        a_new = a * np.exp(log_update)
        a_new_sum = float(a_new.sum())
        if a_new_sum < 1e-12:
            break
        a = a_new / a_new_sum
    
    return best_a


def mirror_descent_lse(a, n_iters=3000, beta=100.0):
    """Mirror descent using LSE-smoothed gradient for better conditioning."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        sc_lse, grad = lse_score_and_grad(a, beta)
        sc_exact = compute_score(a)
        if sc_exact < best_sc:
            best_sc = sc_exact
            best_a = a.copy()
        
        S = float(a.sum())
        S2 = S * S
        # The gradient of the LSE-smoothed objective
        # For mirror descent, we use the gradient directly
        g_norm = float(np.sum(a * np.abs(grad)))
        if g_norm < 1e-15:
            break
        
        gap = sc_exact - lower_bound
        eta = min(0.05, max(0.0001, 0.5 * gap / (g_norm + 1e-10)))
        
        # Multiplicative update
        log_update = -eta * grad
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        a_new_sum = float(a_new.sum())
        if a_new_sum < 1e-12:
            break
        a = a_new / a_new_sum
    
    return best_a


def coordinate_descent(a, n_sweeps=5, n_top=100, n_evals_per_coord=7):
    """Coordinate descent on peak-contributing elements with golden section search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue
        
        seg = a[j_start:j_end+1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            # Golden section search in [0, 2*a_old_j]
            lo = 0.0
            hi = 2.0 * a_old_j
            gr = (np.sqrt(5) - 1) / 2
            best_val = a_old_j
            best_local_sc = compute_score(a)
            
            for _ in range(n_evals_per_coord):
                x1 = hi - gr * (hi - lo)
                x2 = lo + gr * (hi - lo)
                a[j] = x1
                sc1 = compute_score(a)
                a[j] = x2
                sc2 = compute_score(a)
                if sc1 < sc2:
                    hi = x2
                else:
                    lo = x1
                a[j] = best_val
            
            # Final evaluation
            optimal = (lo + hi) / 2
            a[j] = optimal
            sc_opt = compute_score(a)
            if sc_opt < best_local_sc:
                best_local_sc = sc_opt
                best_val = optimal
            a[j] = best_val
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.linspace(0, 1, n)
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            inits.append(np.interp(x, x_old, parent).copy())
    
    # Various structured inits
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2*np.pi*x))
    inits.append(0.5 + 0.3 * np.sin(2*np.pi*x))
    inits.append(0.3 + 0.5 * np.cos(2*np.pi*x)**2)
    inits.append(0.4 + 0.3 * np.cos(4*np.pi*x) * np.cos(2*np.pi*x))
    inits.append(np.clip(0.5 + 0.2*rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.clip(0.5 + 0.4*rng.standard_normal(n), 0.01, 10.0))
    # U-shape
    inits.append(0.2 + 0.8*(2*x-1)**2)
    # Cap shape
    inits.append(0.2 + 0.8*np.cos(np.pi*(2*x-1))**2)
    # Random blocks
    for nb in (8, 16, 32):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n-len(a)]])
        inits.append(a[:n].copy())
    
    # All non-negative
    inits = [np.maximum(a, 0.01) for a in inits]
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """

            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start

            return True
        return False

        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:4]
    
    print(f"Lengths to try: {lengths}")
    print(f"Parent: {parent is not None}, len={0 if parent is None else len(parent)}")
    
    # ============================================================
    # Phase 1: Spectral flattening (global search) - 35% time
    # ============================================================
    p1_deadline = t_start + 0.35 * (float(budget_s) - 20.0)

    for n in lengths:
        if time.monotonic() > p1_deadline:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)
        remaining = max(1, p1_deadline - time.monotonic())
        max_inits = max(2, int(remaining / 8.0))
        inits = inits[:min(len(inits), max_inits)]
        
        for i, a0 in enumerate(inits):
            if time.monotonic() > p1_deadline:
                break
            
            # Spectral flattening (soft threshold)
            a_sf = spectral_flatten(a0, n_iters=150, alpha_start=0.3, alpha_end=0.8, gamma=0.5)
            sc = compute_score(a_sf)
            report(sc, a_sf, f"SF n={n} i={i}")
            
            # Power compression
            if time.monotonic() < p1_deadline:
                a_pc = spectral_flatten_power(a0, n_iters=100, beta_start=0.6, beta_end=0.3)
                sc = compute_score(a_pc)
                report(sc, a_pc, f"PC n={n} i={i}")
            
            # Spectral flatten from the power-compressed result
            if time.monotonic() < p1_deadline:
                a_sf2 = spectral_flatten(a_pc, n_iters=100, alpha_start=0.5, alpha_end=0.9, gamma=0.3)
                sc = compute_score(a_sf2)
                report(sc, a_sf2, f"SF2 n={n} i={i}")

    # Phase 2: Mirror descent (medium-scale) - 20% time
    # ============================================================
    p2_deadline = t_start + 0.55 * (float(budget_s) - 20.0)

    a_md = mirror_descent(best_a, n_iters=5000, eta=0.02)

    if time.monotonic() < p2_deadline:

        sc = compute_score(a_md2)
        report(sc, a_md2, "MD lse beta=200")

        sc = compute_score(a_sf3)

        a_pert = np.clip(a_pert, 0, 1000)
        a_pert = a_pert / a_pert.sum()
        sc, a_opt, _ = run_lbfgs(a_pert, 300, maxiter=3000, maxfun=30000)
        report(sc, a_opt, "LBFGS pert")
    
    if time.monotonic() < p3_deadline:
        # From spectral flatten result
        a_sf_p = spectral_flatten(best_a, n_iters=50, alpha_start=0.7, alpha_end=0.9, gamma=0.3)
        sc, a_opt, _ = run_lbfgs(a_sf_p, 500, maxiter=3000, maxfun=30000)
        report(sc, a_opt, "LBFGS from SF")
    
    print(f"After Phase 3: best = {best_score:.8f}")
    
    # ============================================================
    # Phase 4: Coordinate descent + final polish (15% time)
    # ============================================================
    p4_deadline = deadline
    print(f"\n--- Phase 4: Polish (deadline {p4_deadline - t_start:.0f}s) ---")
    
    n_best = len(best_a)
    
    # Coordinate descent
    if time.monotonic() < p4_deadline:
        a_cd = coordinate_descent(best_a, n_sweeps=5, n_top=100, n_evals_per_coord=6)
        sc = compute_score(a_cd)
        report(sc, a_cd, "CD")
    
    # Final L-BFGS from CD
    if time.monotonic() < p4_deadline - 30:
        sc, a_opt, _ = run_lbfgs(best_a, 1000, maxiter=5000, maxfun=50000)
        report(sc, a_opt, "LBFGS final b=1000")

    if time.monotonic() < p4_deadline - 10:
        a_cd2 = coordinate_descent(best_a, n_sweeps=3, n_top=200, n_evals_per_coord=8)
        sc = compute_score(a_cd2)
        report(sc, a_cd2, "CD2")

    if time.monotonic() < p4_deadline - 20:
        a_final_sf = spectral_flatten(best_a, n_iters=80, alpha_start=0.8, alpha_end=0.95, gamma=0.2)
        sc = compute_score(a_final_sf)
        report(sc, a_final_sf, "SF final")
        
        if sc < best_score:
            best_score = sc
            best_a = a_final_sf.copy()
        
        if time.monotonic() < p4_deadline - 5:
            sc2, a_opt2, _ = run_lbfgs(a_final_sf, 2000, maxiter=3000, maxfun=30000)
            report(sc2, a_opt2, "LBFGS from SF final")
    
    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
