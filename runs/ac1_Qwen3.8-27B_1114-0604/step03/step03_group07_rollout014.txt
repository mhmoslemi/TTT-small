def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed max-conv score and gradient w.r.t. a.
    Uses numerical safeguards against overflow/underflow.
    Includes diagonal correction for exact self-convolution gradient.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.maximum(shifted, -60.0)
    exp_s = np.exp(shifted)
    exp_sum = float(exp_s.sum())
    if exp_sum < 1e-300:
        exp_sum = 1e-300
    L = c_max + np.log(exp_sum) / beta
    p = exp_s / exp_sum
    score = 2.0 * n * L / (S * S)

    # Gradient of L w.r.t. a:
    # dL/da[j] = (2/beta) * sum_k p[k] * a[k-j] (where 0<=k-j<n)
    #           - (1/beta) * a[j] * p[2j] (diagonal correction)
    # The first term is a cross-correlation.
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_pad = np.zeros(M2_fft)
    A_pad[:n] = a
    P_pad = np.zeros(M2_fft)
    P_pad[:2 * n - 1] = p
    xc = np.fft.irfft(np.conj(np.fft.rfft(A_pad)) * np.fft.rfft(P_pad), M2_fft)
    r = xc[:n]

    dL_da = (2.0 / beta) * r
    # Diagonal correction: when k=2j, the term a[j]*a[j] is counted once not twice
    diag_idx = np.arange(0, 2 * n - 1, 2)
    diag_idx = diag_idx[diag_idx < 2 * n - 1]
    diag_valid = diag_idx // 2
    dL_da[diag_valid] -= (1.0 / beta) * a[diag_valid] * p[diag_idx]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)

    if a0.sum() < 1e-6:

            obj_wrapper, a0, method='L-BFGS-B', jac=True, bounds=bounds,

        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgrad_step(a, top_k=5, beta_weight=50.0):
    """
    Compute a weighted subgradient using top-K peaks.
    Returns (gradient, score, k_star, M).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    M = float(c.max())
    sc = 2.0 * n * M / (S * S)

    # Find top K peaks
    k_all = np.argsort(c)[::-1][:top_k]
    # Weight them with softmax
    c_top = c[k_all]
    w = np.exp(beta_weight * (c_top - c_top[0]))
    w /= w.sum()

    # Weighted subgradient
    sg = np.zeros(n)
    for idx, k in enumerate(k_all):
        if w[idx] < 1e-10:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        seg = a[j_start:j_end + 1][::-1]
        sg[j_start:j_end + 1] += 2.0 * w[idx] * seg

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return grad, sc, int(k_all[0]), M


def nesterov_subgrad(a, n_iters, top_k=5, beta_weight=50.0):
    """
    Nesterov-accelerated subgradient method with Polyak step size.
    Returns best score and best array found.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    t = 1

    for it in range(n_iters):
        # Nesterov momentum
        momentum = (t - 1.0) / (t + 2.0)
        y = a + momentum * (a - a_prev)
        y = np.clip(y, 0.0, 1000.0)
        a_prev = a.copy()

        grad, sc_y, k_star, M = subgrad_step(y, top_k=top_k, beta_weight=beta_weight)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break

        # Polyak step size
        step = (sc_y - sc_lb) / gn2
        step = min(step, 2.0 / (np.sqrt(2.0 * t) * (1.0 + 0.1 * t)))
        step = min(step, 0.5)

        a_new = y - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)

        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            print(f"    Nesterov iter {it}: {sc_new:.8f}")

        a = a_new
        t += 1

    return best_sc, best_a


def multiplicative_updates(a, n_iters, top_k=5, beta_weight=50.0, eta_init=0.1):
    """
    Multiplicative (exponential gradient) updates with top-K peak tracking.
    a_new[j] = a[j] * exp(-eta * grad[j])
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S_target = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        grad, sc_cur, k_star, M = subgrad_step(a, top_k=top_k, beta_weight=beta_weight)
        if sc_cur < best_sc:
            best_sc = sc_cur
            best_a = a.copy()

        # Adaptive step size (Polyak-like for multiplicative)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        eta = min(eta_init * (best_sc - sc_lb) / (best_sc + 1e-10), 0.3)
        eta = max(eta, 1e-6)

        # Multiplicative update
        log_step = -eta * grad
        log_step = np.clip(log_step, -5.0, 5.0)
        a = a * np.exp(log_step)
        a = np.clip(a, 1e-10, 1000.0)
        # Renormalize to maintain sum
        a *= S_target / a.sum()

    return best_sc, best_a


def coordinate_descent(a, n_iters, top_k=3):

        M = float(c.max())

        if sc < best_sc:

            best_a = a.copy()

        # Find the contributor to the peak that has the most "excess"

        contribs = np.zeros(n)
        for j in range(j_start, j_end + 1):

        peak_region = np.where(contribs > 0)[0]
        if len(peak_region) == 0:
            break

        j_perturb = peak_region[np.argmax(a[peak_region])]

        a_lo = 0.0
        a_hi = a[j_perturb] * 2.0
        a_old = a[j_perturb]

            sc_test = compute_score(a)

            a[j_perturb] = a_old  # restore

            a[j_perturb] = a_old * (1.0 - frac)

            sc_test = compute_score(a)
            if sc_test < best_local_sc:
                best_local_sc = sc_test

        # Transfer mass to a random non-peak index

    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_budget:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 5:
            break

        strat = iteration % 6
        if strat < 2:
            # Multiplicative updates
            n_mu = max(50, int(remaining / max(0.01, (fast_conv(a).shape[0]) / 1e7) * 0.05))
            n_mu = min(n_mu, 500)
            sc, a_opt = multiplicative_updates(a, n_mu, top_k=top_k, beta_weight=beta_weight)
            if sc < best_sc:
                best_sc = sc
                best_a = a_opt.copy()
                a = a_opt
                print(f"    Hybrid iter {iteration} (MU): {sc:.8f}")
        elif strat < 4:
            # Nesterov subgradient
            n_ns = max(20, int(remaining / 10))
            n_ns = min(n_ns, 200)
            sc, a_opt = nesterov_subgrad(a, n_ns, top_k=top_k, beta_weight=beta_weight)
            if sc < best_sc:
                best_sc = sc
                best_a = a_opt.copy()
                a = a_opt
                print(f"    Hybrid iter {iteration} (NS): {sc:.8f}")
        else:
            # L-BFGS refinement with high beta
            if remaining > 30:
                beta = 5000.0
                maxit = max(200, min(2000, int(remaining / 0.05)))
                sc, a_opt = run_lbfgs(a, beta, maxiter=maxit, maxfun=maxit * 10)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_opt.copy()
                    a = a_opt
                    print(f"    Hybrid iter {iteration} (LBFGS): {sc:.8f}")
                else:
                    # If L-BFGS didn't help, try with the best
                    sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxit // 2, maxfun=maxit * 5)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_opt.copy()
                        a = a_opt
                        print(f"    Hybrid iter {iteration} (LBFGS best): {sc:.8f}")

        # Adaptive top_k and beta_weight: increase as we get closer to optimum
        gap = best_sc - sc_lb
        if gap < 0.1:
            top_k = 8
            beta_weight = 100.0
        elif gap < 0.3:
            top_k = 5
            beta_weight = 50.0

        iteration += 1

    return best_sc, best_a


def build_diverse_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / max(n - 1, 1)

    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())

    # Power-law ramps (anti-palindromic)
    for p in [0.5, 1.0, 2.0, 3.0]:
        inits.append((xn + 0.05) ** p)
        inits.append((1.0 - xn + 0.05) ** p)

    # Exponential ramps
    for alpha in [1.0, 2.0, 4.0]:
        inits.append(np.exp(alpha * xn))
        inits.append(np.exp(alpha * (1.0 - xn)))

    # Sinusoidal anti-palindromic
    for freq in [1, 2, 3]:
        inits.append(np.clip(0.3 + 0.7 * xn + 0.2 * np.sin(np.pi * freq * xn), 0.01, 1000.0))

    # Cosine-based
    inits.append(np.clip(0.5 + 0.3 * np.cos(np.pi * xn), 0.01, 1000.0))
    inits.append(np.clip(0.5 - 0.3 * np.cos(np.pi * xn), 0.01, 1000.0))

    # Random smoothed
    a_rand = np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 1000.0)
    ksize = max(3, n // 200)
    kernel = np.ones(ksize) / ksize
    a_smooth = np.convolve(a_rand, kernel, mode='same')
    inits.append(np.clip(a_smooth, 0.01, 1000.0))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Multi-phase optimization:
    Phase 1: LSE-continuation L-BFGS from diverse inits (30% of time)
    Phase 2: Hybrid subgradient/multiplicative/L-BFGS (50% of time)
    Phase 3: Intensive polish on best solution (15% of time)
    Phase 4: Final coordinate descent and verification (5% of time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    n_target = 6962
    if parent is not None:
        n_target = parent.size

    print(f"n={n_target}, parent={'yes' if parent is not None else 'no'}")
    if parent is not None:
        print(f"Parent score: {compute_score(parent):.8f}")

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 1: LSE-continuation L-BFGS (30% of time)
    # ============================================================
    phase1_end = t_start + 0.30 * total_budget
    print(f"\n=== Phase 1: LSE Continuation L-BFGS (until t={phase1_end-t_start:.0f}s) ===")

    inits = build_diverse_inits(n_target, rng, parent)
    print(f"  {len(inits)} initializations")

    # Quick score all inits
    init_scores = []
    for i, a_init in enumerate(inits):
        sc = compute_score(a_init)
        init_scores.append((sc, i))
        print(f"  Init {i}: {sc:.8f}")

    init_scores.sort()

    # Try top initializations with LSE continuation
    beta_schedule = [10, 30, 100, 300, 1000]
    n_to_try = min(len(inits), max(3, int((phase1_end - time.monotonic()) / 60)))

    for rank in range(n_to_try):
        if time.monotonic() > phase1_end:
            break
        sc_init, idx = init_scores[rank]
        a_cur = inits[idx].copy()

        for beta in beta_schedule:
            if time.monotonic() > phase1_end:
                break
            remaining = phase1_end - time.monotonic()
            if remaining < 10:
                break
            est_time = 0.001 * n_target
            maxit = max(200, min(3000, int(remaining / max(est_time, 0.1))))
            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxit * 15)
            a_cur = a_opt
            report(sc, a_opt, f"P1 r={rank} b={beta}")

    print(f"  After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Hybrid optimization (50% of time)
    # ============================================================
    phase2_end = t_start + 0.80 * total_budget
    print(f"\n=== Phase 2: Hybrid Optimization (until t={phase2_end-t_start:.0f}s) ===")

    if best_a is None:
        best_a = np.ones(n_target) * 0.5

    # First, do a moderate LSE refinement to get a good starting point
    if time.monotonic() < phase2_end - 60:
        sc, a_opt = run_lbfgs(best_a, 1000, maxiter=3000, maxfun=50000)
        report(sc, a_opt, "P2 pre-polish")

    # Run hybrid phase
    time_left = phase2_end - time.monotonic()
    if time_left > 30:
        sc, a_opt = hybrid_phase(best_a, time_budget=time_left * 0.9, top_k=5, beta_weight=50.0)
        report(sc, a_opt, "P2 hybrid")

    print(f"  After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Intensive polish (12% of time)
    # ============================================================
    phase3_end = t_start + 0.92 * total_budget
    print(f"\n=== Phase 3: Intensive Polish ===")

    if time.monotonic() < phase3_end and best_a is not None:
        # Multiple rounds of hybrid with increasing precision
        for round_idx in range(5):
            if time.monotonic() > phase3_end:
                break
            remaining = phase3_end - time.monotonic()
            if remaining < 20:
                break
            t_round = remaining * 0.3
            sc, a_opt = hybrid_phase(best_a, time_budget=t_round, top_k=8, beta_weight=100.0)
            report(sc, a_opt, f"P3 r{round_idx}")

        # Nesterov subgradient with small top_k for precision
        if time.monotonic() < phase3_end - 10:
            n_iters = max(100, int((phase3_end - time.monotonic()) * 5))
            sc, a_opt = nesterov_subgrad(best_a, n_iters, top_k=3, beta_weight=200.0)
            report(sc, a_opt, "P3 nesterov")

    print(f"  After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Final polish and perturbation search (8% of time)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")

    if best_a is not None and time.monotonic() < deadline:
        n_best = len(best_a)

        # Try small perturbations + L-BFGS
        trial = 0
        while time.monotonic() < deadline - 30 and trial < 15:
            trial += 1
            strat = trial % 5

            if strat == 0:
                # Gaussian perturbation
                scale = 0.02 + 0.03 * rng.random()
                a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            elif strat == 1:
                # Targeted: reduce top contributors to peak
                c = fast_conv(best_a)
                k_star = int(np.argmax(c))
                j_start = max(0, k_star - n_best + 1)
                j_end = min(k_star, n_best - 1)
                a_pert = best_a.copy()
                for j in range(j_start, j_end + 1):
                    a_pert[j] *= (1.0 - 0.05 * rng.random())
                # Add mass elsewhere
                non_peak = np.setdiff1d(np.arange(n_best), np.arange(j_start, j_end + 1))
                if len(non_peak) > 0:
                    add_idx = non_peak[rng.integers(len(non_peak))]
                    a_pert[add_idx] += 0.1 * a_pert[j_start]
            elif strat == 2:
                # Frequency domain smoothing
                M_fft = 1 << (2 * n_best - 1).bit_length()
                A = np.fft.rfft(a_pert if 'a_pert' in dir() else best_a, M_fft)
                freqs = np.arange(len(A))
                att = 1.0 - 0.05 * (freqs / len(A)) ** 2
                A *= att
                a_pert = np.real(np.fft.irfft(A, M_fft))[:n_best]
            elif strat == 3:
                # Anti-palindromic nudge
                a_pert = best_a.copy()
                for j in range(n_best // 2):
                    d = (j - n_best / 2.0) / (n_best / 10.0)
                    a_pert[j] *= (1.0 + 0.03 * np.exp(-d * d))
                    a_pert[n_best - 1 - j] *= (1.0 - 0.03 * np.exp(-d * d))
            else:
                # Random coordinate swap
                a_pert = best_a.copy()
                i1 = rng.integers(0, n_best)
                i2 = rng.integers(0, n_best)
                amt = 0.05 * best_a[i1]
                a_pert[i1] = max(0.0, best_a[i1] - amt)
                a_pert[i2] = min(1000.0, best_a[i2] + amt)

            a_pert = np.clip(a_pert, 0.0, 1000.0)
            if a_pert.sum() < 1e-6:
                a_pert = best_a.copy()

            sc, a_opt = run_lbfgs(a_pert, 5000, maxiter=2000, maxfun=30000)
            report(sc, a_opt, f"P4 #{trial} s={strat}")

        # Final Nesterov
        if time.monotonic() < deadline - 10:
            n_iters = max(200, int((deadline - time.monotonic() - 5) * 10))
            sc, a_opt = nesterov_subgrad(best_a, n_iters, top_k=3, beta_weight=150.0)
            report(sc, a_opt, "P4 final nesterov")

    # ============================================================
    # Output
    # ============================================================
    if best_a is None:
        best_a = np.ones(64) * 0.5

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    if verified < best_score:
        print("NOTE: verified score is better than internal (possible numerical difference)")
        best_score = verified

    return out
