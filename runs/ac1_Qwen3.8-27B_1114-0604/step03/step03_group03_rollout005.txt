Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key improvements over previous:
- Custom fast 1D solver (no scipy minimize_scalar overhead in inner loop)

import numpy as np
from scipy.optimize import minimize, minimize_scalar

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=8000, maxfun=100000):

        lse_objective, a0, args=(beta),

                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def solve_1d_coord(base, alpha, diag_idx, outside_max, S_rest, n, t_max):

def peak_ordered_sweep(a):

def targeted_peak_shave(a, rng, strength=0.03):
    """
    Shave peaks by reducing contributions at the convolution peak lags.
    """
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    
    # Find contributing coordinates
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end < j_start:
        return a.copy()
    
    contrib = a[j_start:j_end + 1][::-1].copy()
    total = contrib.sum()
    if total < 1e-12:
        return a.copy()
    
    # Proportionally reduce top contributors
    a_new = a.copy()
    n_reduce = max(3, len(contrib) // 5)
    top_idx_local = np.argsort(contrib)[::-1][:n_reduce]
    
    reduce_total = 0.0
    for idx in top_idx_local:
        actual_idx = j_end - int(idx)
        reduce_amt = a_new[actual_idx] * strength
        a_new[actual_idx] = max(a_new[actual_idx] - reduce_amt, 0.0)
        reduce_total += reduce_amt
    
    # Redistribute to non-contributing coordinates
    if reduce_total > 0:
        non_contrib = np.setdiff1d(np.arange(n), range(j_start, j_end + 1))
        if len(non_contrib) > 0:
            per = reduce_total / len(non_contrib)
            a_new[non_contrib] += per
    
    return np.clip(a_new, 0.0, 1000.0)

        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    if parent is not None:
        p_score = compute_score(parent)
        print(f"Parent score: {p_score:.8f}, length: {len(parent)}")
        best_score = p_score
        best_a = parent.copy()
    else:
        print("No parent available, using ones init.")
        best_score = compute_score(best_a)
        print(f"Initial score: {best_score:.8f}")

    n_work = len(best_a)
    print(f"Working length: {n_work}")

    # ============================================================
    # Phase 1: Fast coordinate descent (45% of time)
    # ============================================================
    phase1_deadline = t_start + 0.45 * total_budget
    print(f"\n=== Phase 1: Fast CD ({phase1_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    no_improve = 0
    sweep = 0

    while time.monotonic() < phase1_deadline:
        sweep += 1
        if sweep % 3 == 0:
            sc, a_cur = peak_ordered_sweep(a_cur)
        else:
            order = rng.permutation(n_work)
            sc, a_cur = coord_descent_sweep(a_cur, order=order)

        if sc < best_score:
            old = best_score
            best_score = sc
            best_a = a_cur.copy()
            no_improve = 0
            if sweep <= 10 or sweep % 20 == 0:
                print(f"  Sweep {sweep}: {sc:.8f} (NEW BEST, delta={old-sc:.2e})")
        else:
            no_improve += 1

        if sweep % 10 == 0:
            print(f"  Sweep {sweep}: {sc:.8f} (no_improve={no_improve})")

        if no_improve >= 50:
            print(f"  Stuck at sweep {sweep}, perturbing...")
            a_cur = targeted_peak_shave(a_cur, rng, strength=0.02 + 0.03 * rng.random())
            no_improve = 0

    print(f"Phase 1 done: {sweep} sweeps, best={best_score:.8f}")

    # ============================================================
    # Phase 2: LSE refinement (25% of time)
    # ============================================================
    phase2_deadline = t_start + 0.70 * total_budget
    print(f"\n=== Phase 2: LSE Refinement ===")

    lsq_original = best_score
    for beta in [100, 300, 800, 2000, 5000, 12000]:
        if time.monotonic() > phase2_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=10000, maxfun=200000)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  LSE beta={beta}: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            print(f"  LSE beta={beta}: {sc:.8f} ({elapsed:.1f}s)")

    # CD polish after LSE
    a_cur = best_a.copy()
    for _ in range(20):
        if time.monotonic() > phase2_deadline:
            break
        sc, a_cur = coord_descent_sweep(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

    print(f"Phase 2 done: best={best_score:.8f} (improved from {lsq_original:.8f})")

    # ============================================================
    # Phase 3: Peak shaving + CD cycles (20% of time)
    # ============================================================
    phase3_deadline = t_start + 0.90 * total_budget
    print(f"\n=== Phase 3: Peak Shaving Cycles ===")

    for trial in range(50):
        if time.monotonic() > phase3_deadline - 5.0:
            break

        strategy = trial % 6
        n_cur = len(best_a)

        if strategy == 0:
            # Peak shaving
            a_pert = targeted_peak_shave(best_a, rng, strength=0.01 + 0.02 * rng.random())
        elif strategy == 1:
            # Gaussian perturbation
            scale = 0.01 + 0.02 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_cur))
        elif strategy == 2:
            # Mirror reflection of a random segment
            seg_len = max(10, n_cur // 50)
            s = int(rng.integers(0, max(1, n_cur - seg_len)))
            a_pert = best_a.copy()
            a_pert[s:s + seg_len] = a_pert[s:s + seg_len][::-1]
        elif strategy == 3:
            # Smooth then sharpen (gaussian blur + inverse)
            from scipy.ndimage import gaussian_filter1d
            sigma = max(1.0, n_cur / 200.0)
            a_pert = gaussian_filter1d(best_a, sigma=sigma)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 4:
            # Reduce top peak contributors, boost bottom
            c = fast_conv(best_a)
            k_peak = int(np.argmax(c))
            j_start = max(0, k_peak - n_cur + 1)
            j_end = min(k_peak, n_cur - 1)
            a_pert = best_a.copy()
            if j_end >= j_start:
                contrib = a_pert[j_start:j_end + 1][::-1].copy()
                n_top = max(5, len(contrib) // 3)
                top_local = np.argsort(contrib)[::-1][:n_top]
                total_red = 0.0
                for idx in top_local:
                    ai = j_end - int(idx)
                    red = a_pert[ai] * 0.05
                    a_pert[ai] -= red
                    total_red += red
                bottom_local = np.argsort(contrib)[:(len(contrib) - n_top)]
                if len(bottom_local) > 0:
                    per = total_red / len(bottom_local)
                    for idx in bottom_local:
                        ai = j_end - int(idx)
                        a_pert[ai] += per
        else:
            # Sinusoidal modulation
            freq = 1.0 + 5.0 * rng.random()
            amp = 0.005 + 0.015 * rng.random()
            x = np.arange(n_cur, dtype=np.float64) / n_cur
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + 2.0 * np.pi * rng.random()))

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick CD from perturbed point
        for sw in range(15):
            if time.monotonic() > phase3_deadline - 5.0:
                break
            if sw % 4 == 0:
                sc_pert, a_pert = peak_ordered_sweep(a_pert)
            else:
                sc_pert, a_pert = coord_descent_sweep(a_pert, order=rng.permutation(n_cur))

        sc_pert = compute_score(a_pert)
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  P3 trial {trial} (s={strategy}): {sc_pert:.8f} (NEW BEST)")
        elif trial % 10 == 0:
            print(f"  P3 trial {trial} (s={strategy}): {sc_pert:.8f}")

    # ============================================================
    # Phase 4: Final refinement (remaining time)
    # ============================================================
    print(f"\n=== Phase 4: Final Refinement ===")

    # More coordinate descent
    a_cur = best_a.copy()
    cd_sweeps = 0
    while time.monotonic() < deadline - 30.0:
        cd_sweeps += 1
        sc, a_cur = coord_descent_sweep(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            if cd_sweeps % 10 == 0:
                print(f"  Final CD sweep {cd_sweeps}: {sc:.8f} (NEW BEST)")

    # One more LSE if time permits
    if time.monotonic() < deadline - 20.0:
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs(best_a, 3000, maxiter=5000, maxfun=80000)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final LSE: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        # Final CD
        a_cur = best_a.copy()
        for _ in range(5):
            if time.monotonic() > deadline - 5.0:
                break
            sc, a_cur = coord_descent_sweep(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
