"""
Optimized step-function height sequence: L-BFGS/LSE continuation, 
improved subgradient polish with Polyak step, and efficient coordinate descent.
Targets the minimum peak self-convolution ratio for non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    log_sum = max_sh + np.log(exp_sh.sum())
    L = c_max + log_sum / beta
    p = exp_sh / exp_sh.sum()
    # Compute grad_L[j] = sum_m p[j+m] * x[m] via FFT correlation
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    Xr = np.fft.rfft(x[::-1], M2_fft)
    conv_result = np.fft.irfft(P * Xr, M2_fft)
    grad_L = conv_result[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n / S2) * grad_L - (4.0 * n * L / S3)
    score = 2.0 * n * L / S2
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun, ftol=1e-14, gtol=1e-12):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        val, g = lse_obj_grad(x, beta)
        return val, g

    res = minimize(
        objective, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': ftol, 'gtol': gtol, 'maxiter': maxiter,
                 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out

    n = len(a0)
    a = a0.astype(np.float64).copy()
    best_sc = compute_score(a)

def coordinate_descent(a0, n_sweeps=20, time_limit=60.0):
    """Efficient coordinate descent using analytical convolution updates."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = float(a.sum())
    C = fast_conv(a)
    best_sc = 2.0 * n * float(C.max()) / (S * S)
    best_a = a.copy()
    t0 = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved_sweep = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val <= 1e-10:
                # Try small positive values
                candidates = [0.001, 0.01, S / (10.0 * n)]
            else:
                candidates = [0.0, old_val * 0.3, old_val * 0.6, old_val * 1.3, old_val * 1.6, S / n]
            best_local_sc = best_sc
            best_local_val = old_val
            for cand in candidates:
                if cand < -1e-10 or cand > 1000.0:
                    continue
                cand = max(0.0, cand)
                if abs(cand - old_val) < 1e-12:
                    continue
                delta = cand - old_val
                # Update C: C[k] += delta * a[k-j] for k in [j, j+n-1]
                k_start = j
                k_end = min(j + n - 1, 2 * n - 2)
                # The contribution: for k in [k_start, k_end], C[k] += delta * a[k-j]
                # a[k-j] for k in [j, j+n-1] is a[0], a[1], ..., a[n-1] (clamped)
                segment_len = k_end - k_start + 1
                a_seg = a[:segment_len]
                C_new = C.copy()
                C_new[k_start:k_end + 1] += delta * a_seg
                max_C_new = float(C_new.max())
                S_new = S + delta
                if S_new < 1e-12:
                    continue
                sc_new = 2.0 * n * max_C_new / (S_new * S_new)
                if sc_new < best_local_sc:
                    best_local_sc = sc_new
                    best_local_val = cand
                    best_local_C = C_new
                    best_local_S = S_new
            if best_local_val != old_val:
                a[j] = best_local_val
                C = best_local_C
                S = best_local_S
                improved_sweep = True
                if best_local_sc < best_sc:
                    best_sc = best_local_sc
                    best_a = a.copy()
        if not improved_sweep:
            break
    return best_sc, best_a


def spectral_flatten(a0, N_fft, n_iters, lam):
    """Iterative spectral flattening."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    orig_sum = a.sum()
    for i in range(n_iters):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc = mag[1:]
        target = np.median(non_dc)
        new_mag = mag.copy()
        half = N_fft // 2
        for k in range(1, half + 1):
            if N_fft % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N_fft - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (orig_sum / S)
    return a

            est_time_per_fun = 0.002 * (n_use / 2048)
            maxfun = int(min(50000, remaining * 0.7 / max(est_time_per_fun, 0.001)))
            maxfun = max(500, maxfun)
            maxiter = max(100, maxfun // 10)
            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
            a_cur = a_opt
            try_update(sc, a_opt, f"lbfgs ci={ci} beta={beta}")

        sc_final = compute_score(a_cur)
        try_update(sc_final, a_cur, f"lbfgs_final ci={ci}")
        print(f"    Final: score={sc_final:.6f}")

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: Subgradient polish (20% budget)
    # ================================================================
    sg_deadline = t_start + 0.78 * effective_budget
    print(f"\n--- Phase 3: Subgradient Polish ---")
    sg_time = sg_deadline - time.monotonic()
    if sg_time > 5:
        sc, a_sg = subgradient_polish(best_a, max_iters=50000,
                                       f_lb=1.0, time_limit=min(sg_time * 0.8, 180.0))
        try_update(sc, a_sg, "subgrad")
        print(f"    Subgrad result: {sc:.6f}")

    # ================================================================
    # Phase 4: High-beta L-BFGS from subgradient result (10% budget)
    # ================================================================
    hlbfgs_deadline = t_start + 0.88 * effective_budget
    print(f"\n--- Phase 4: High-Beta L-BFGS ---")
    for beta in [2000, 5000, 10000]:
        if time.monotonic() > hlbfgs_deadline:
            break
        remaining = hlbfgs_deadline - time.monotonic()
        if remaining < 5:
            break
        est_time_per_fun = 0.002 * (n_use / 2048)
        maxfun = int(min(30000, remaining * 0.8 / max(est_time_per_fun, 0.001)))
        maxfun = max(200, maxfun)
        maxiter = max(50, maxfun // 10)
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        try_update(sc, a_opt, f"hlbfgs beta={beta}")

    # ================================================================
    # Phase 5: Coordinate descent (5% budget)
    # ================================================================
    cd_deadline = t_start + 0.94 * effective_budget
    print(f"\n--- Phase 5: Coordinate Descent ---")
    cd_time = cd_deadline - time.monotonic()
    if cd_time > 5:
        sc, a_cd = coordinate_descent(best_a, n_sweeps=20, time_limit=min(cd_time * 0.9, 60.0))
        try_update(sc, a_cd, "coord_desc")
        print(f"    Coord descent result: {sc:.6f}")

    # ================================================================
    # Phase 6: Final subgradient + L-BFGS polish
    # ================================================================
    final_deadline = deadline
    print(f"\n--- Phase 6: Final Polish ---")
    remaining = final_deadline - time.monotonic()
    if remaining > 10:
        sc, a_final = subgradient_polish(best_a, max_iters=100000,
                                          f_lb=1.0, time_limit=min(remaining * 0.4, 60.0))
        try_update(sc, a_final, "final_subgrad")

    remaining = final_deadline - time.monotonic()
    if remaining > 5:
        sc, a_final2 = run_lbfgs(best_a, 5000, maxiter=2000,
                                  maxfun=min(10000, int(remaining * 20)))
        try_update(sc, a_final2, "final_lbfgs")

    # One more subgradient pass
    remaining = final_deadline - time.monotonic()
    if remaining > 5:
        sc, a_final3 = subgradient_polish(best_a, max_iters=50000,
                                           f_lb=1.0, time_limit=min(remaining * 0.8, 30.0))
        try_update(sc, a_final3, "last_subgrad")

    # ================================================================
    # Final verification and return
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
