```python
"""
Step function optimization via LSE-smoothed L-BFGS continuation,
soft dual ascent (LP-inspired), and targeted peak shaving.
Improves upon previous 1.509207 bound.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def _lse_fun(x, beta):
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    options = {'ftol': 1e-15, 'gtol': 1e-14, 'maxiter': maxiter,
               'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(200, int(remaining * 300)))
        options['maxiter'] = min(maxiter, max(200, int(remaining * 80)))
    try:
        res = minimize(_lse_fun, a0, args=(beta,), method='L-BFGS-B',
                       jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def dual_ascent(a, n_iters=100, temp=0.1, alpha=0.5, deadline=None):
    """
    LP-dual-inspired soft multiplicative update.
    At optimum, 2*sum_k lambda_k * a_{k-j} = const for all j with a_j > 0.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (3 * n - 2).bit_length()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        if M < 1e-15:
            break
        # Soft lambda: weight peaks more heavily
        scaled = -temp * (M - c) / M  # 0 at peak, negative elsewhere
        lambda_v = np.exp(np.clip(scaled, -50.0, 0.0))
        lambda_sum = lambda_v.sum()
        if lambda_sum < 1e-15:
            break
        lambda_v = lambda_v / lambda_sum
        
        # h[j] = 2 * sum_k lambda_v[k] * a[k-j]
        # This is 2 * cross-correlation of lambda_v and a
        L = np.fft.rfft(lambda_v, M_fft)
        A = np.fft.rfft(a, M_fft)
        cross = np.fft.irfft(L * np.conj(A), M_fft)
        h = 2.0 * cross[:n]
        h = np.maximum(h, 1e-15)
        h_target = float(np.mean(h[a > 1e-10])) if np.any(a > 1e-10) else float(np.mean(h))
        
        # Multiplicative update
        factor = np.ones(n)
        mask = a > 1e-10
        factor[mask] = 1.0 - alpha + alpha * (h_target / h[mask])
        factor = np.clip(factor, 0.01, 100.0)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)
        
        # Renormalize
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a = a * (S / S_new)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a


def peak_shave(a, n_rounds=20, strength=0.05, deadline=None):
    """
    Targeted peak reduction: identify top contributing elements to the
    maximum convolution lag and reduce them, redistributing mass.
```

```python
"""
Step function optimization via LSE-smoothed L-BFGS continuation,
soft-dual-ascent equalization, and targeted peak shaving.
Targets improvement below 1.5027 from the 1.509207 parent bound.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1: 2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def _lse_fun(x, beta):
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    options = {'ftol': 1e-15, 'gtol': 1e-14, 'maxiter': maxiter,
               'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(200, int(remaining * 350)))
        options['maxiter'] = min(maxiter, max(200, int(remaining * 90)))
    try:
        res = minimize(_lse_fun, a0, args=(beta,), method='L-BFGS-B',
                       jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def dual_ascent(a, n_iters=80, temp=0.05, alpha_init=0.6, alpha_decay=0.02,
                deadline=None):
    """Soft-dual-ascent multiplicative update with adaptive alpha."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (3 * n - 2).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        if M < 1e-15:
            break
        scaled = -temp * (M - c) / M
        lambda_v = np.exp(np.clip(scaled, -50.0, 0.0))
        ls = lambda_v.sum()
        if ls < 1e-15:
            break
        lambda_v = lambda_v / ls

        L = np.fft.rfft(lambda_v, M_fft)
        A = np.fft.rfft(a, M_fft)
        cross = np.fft.irfft(L * np.conj(A), M_fft)
        h = 2.0 * cross[:n]
        h = np.maximum(h, 1e-15)

        support = a > 1e-10
        if np.any(support):
            h_target = float(np.mean(h[support]))
        else:
            h_target = float(np.mean(h))

        alpha = alpha_init / (1.0 + alpha_decay * it)
        factor = np.ones(n)
        factor[support] = 1.0 - alpha + alpha * (h_target / h[support])
        factor = np.clip(factor, 0.01, 50.0)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)

        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a = a * (S / S_new)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a

            nb = len(a_cur) // bs
            if nb > 2:
                perm = rng.permutation(nb)
                for b in range(nb):
                    src = perm[b] * bs
                    dst = b * bs
                    a_pert[dst:dst + bs] = a_cur[src:src + bs]

        # Short L-BFGS polish
        sc, a_opt = run_lbfgs(a_pert, 500, maxiter=3000, maxfun=60000,
                              deadline=phase3_end)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            a_cur = a_opt.copy()
            print(f"  [trial {trial} s={strategy}] {sc:.8f}")
        else:
            a_cur = a_opt if sc < compute_score(a_cur) else a_cur

    # Phase 4: Final high-beta polish on best solution
    print(f"\n{'='*60}\nPhase 4: Final Polish")
    a_cur = best_a.copy()
    for beta in [2000, 5000, 10000]:
        if time.monotonic() > deadline - 30.0:
            break
        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=6000, maxfun=150000,
                              deadline=deadline - 30.0)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            a_cur = a_opt.copy()
            print(f"  final beta={beta}: {sc:.8f}")
        else:
            a_cur = a_opt

    # Final dual ascent
    if time.monotonic() < deadline - 20.0:
        sc_da, a_da = dual_ascent(best_a, n_iters=50, temp=0.02,
                                  alpha_init=0.3, deadline=deadline - 15.0)
        if sc_da < best_score:
            best_score = sc_da
            best_a = a_da.copy()
            print(f"  final dual: {sc_da:.8f}")

    # One more LSE
    if time.monotonic() < deadline - 10.0:
        sc, a_opt = run_lbfgs(best_a, 5000, maxiter=4000, maxfun=80000,
                              deadline=deadline - 5.0)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  final lbfgs: {sc:.8f}")

    # Verify
    out = [float(max(0.0, min(1000.0, v))) for v in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128
    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified
    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length={len(out)}, Time={time.monotonic()-t_start:.1f}s/{budget_s}s")
    return out
```
