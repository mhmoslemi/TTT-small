```python
"""

import numpy as np
from scipy.special import comb
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2*n*max(conv)/(sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient w.r.t. a.
    Uses the identity: d(conv[k])/da[j] = 2*a[k-j] if 0<=k-j<n, else 0.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        score = 2.0 * n * c_max / (S * S)
        # Fallback: use subgradient at the max position
        k_star = int(np.argmax(c))
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            g[js:je + 1] = 2.0 * a[idx]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (g / S2 - 2.0 * c_max / S3)
        return score, grad

    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum

    # Compute g[j] = sum_{i=0}^{n-1} a[i] * p[j+i] via cross-correlation
    N_fft = 1 << (3 * n - 2 - 1).bit_length()
    A_fft = np.fft.rfft(a, N_fft)
    P_fft = np.fft.rfft(p, N_fft)
    cross = np.fft.irfft(np.conj(A_fft) * P_fft, N_fft)
    g = 2.0 * cross[:n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_score(a):
    """Exact score and subgradient at the argmax of conv."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c_max = float(c.max())
    k_star = int(np.argmax(c))
    score = 2.0 * n * c_max / (S * S)
    g = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        g[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * c_max / S3)
    return score, grad


def init_half_power(n):
    """
    Initialize with coefficients of (1-x)^{-1/2} truncated to length n.
    c_k = C(2k, k) / 4^k
    Normalized to sum=1.
    """
    k = np.arange(n, dtype=np.float64)
    # Use recurrence: c_0 = 1, c_k = c_{k-1} * (2k-1)/(2k)
    c = np.ones(n)
    for i in range(1, n):
        c[i] = c[i-1] * (2.0*i - 1.0) / (2.0*i)
    S = c.sum()
    return c / S


def init_exponential(n, alpha):
    """Exponential decay initialization, normalized to sum=1."""
    k = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * k)
    return a / a.sum()


def adam_lse(a_init, beta_schedule, lr, n_steps, time_limit, t_start,
             normalize=True, verbose_every=5000):
    """
    Adam with LSE-smoothed gradient and beta continuation.
    beta_schedule: list of (beta, fraction_of_steps) pairs.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.85, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    # Determine steps per beta phase
    total_steps = n_steps
    step_counts = []
    remaining = total_steps
    for i, (beta, frac) in enumerate(beta_schedule):
        if i == len(beta_schedule) - 1:
            cnt = remaining
        else:
            cnt = int(total_steps * frac)
            remaining -= cnt
        step_counts.append(cnt)

    step = 0
    for beta, cnt in zip([b for b, _ in beta_schedule], step_counts):
        for t in range(cnt):
            step += 1
            if time.monotonic() - t_start > time_limit:
                return best_score, best_a

            lr_t = lr / (1.0 + 1e-5 * step)
            score, grad = lse_score_grad(a, beta)

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** (step + 1))
            v_hat = v / (1.0 - b2 ** (step + 1))
            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.clip(a_new, 0.0, 1e10)

            if normalize:
                s = a_new.sum()
                if s > 1e-12:
                    a_new = a_new / s

            a = a_new

            if (step + 1) % verbose_every == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def coord_descent_peak_shave(a, n_sweeps=20, time_limit=None, t_start=None,
                              top_k=50):
    """
    Coordinate descent that specifically targets peak reduction.
    For each coordinate, tries to find the value that minimizes the score.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit if (time_limit is not None and t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        score = 2.0 * n * c_max / (S * S)
        if score < best_score:
            best_score = score

        # Select coordinates: those with highest contribution to peak
        # Peak positions
        peak_threshold = c_max * 0.9
        peak_positions = np.where(c >= peak_threshold)[0]

        # Find which coordinates contribute to peak positions
        contrib = np.zeros(n)
        for k in peak_positions[:min(len(peak_positions), 200)]:
            js = max(0, int(k) - n + 1)
            je = min(int(k), n - 1)
            if je >= js:
                idx = int(k) - np.arange(js, je + 1)
                contrib[js:je + 1] += c[int(k)] * a[idx]

        # Also include coordinates with high gradient magnitude
        S2 = S * S
        S3 = S2 * S
        k_star = int(c.argmax())
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            g[js:je + 1] = 2.0 * a[idx]
        grad = (2.0 * n) * (g / S2 - 2.0 * c_max / S3)

        # Select top-k by combination of contribution and gradient
        selection_score = contrib + 0.01 * np.abs(grad) * S2 / (2.0 * n)
        order = np.argsort(selection_score)[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break

            a_old = a[j]
            if a_old < 1e-15:
                # Try setting to small positive values
                candidates = np.array([0.0, a.sum() * 1e-4, a.sum() * 1e-3, a.sum() * 1e-2])
            else:
                # Try reducing (if it contributes to peak) or increasing
                ratios = [0.0, 0.1, 0.3, 0.5, 0.7, 0.85, 0.95, 1.0, 1.05, 1.15, 1.3, 1.5, 2.0]
                # Also try the "optimal" value where conv[j*2] equals the peak
                # conv[2j] = a[j]^2 + C_{2j} where C_{2j} is the rest
                # We want a[j]^2 + C_{2j} = c_max * (S_new/S_old)^2 (approx)
                # This is hard to compute exactly, so just try empirically
                candidates = np.array(ratios) * a_old
                # Add a few absolute values based on the scale
                scale = S / n
                candidates = np.concatenate([candidates, [scale * 0.1, scale * 0.5, scale * 2.0]])
                candidates = np.clip(candidates, 0.0, 1000.0)

            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x

        # After each sweep, check if we improved
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc

    return best_score, a


def lbfgs_polish(a, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B polish with LSE smoothing."""
    n = len(a)
    a0 = np.clip(a.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def multi_start_optimize(a_init, time_budget, t_start, label=""):
    """
    Full optimization pipeline from a single initialization.
    Returns (best_score, best_a).
    """
    n = len(a_init)
    best_score = compute_score(a_init)
    best_a = a_init.copy()
    elapsed_0 = time.monotonic() - t_start

    # Phase 1: Adam with beta continuation
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 10:
        adam_time = remaining * 0.45
        beta_sched = [(20, 0.2), (50, 0.2), (100, 0.2), (300, 0.2), (1000, 0.2)]
        sc, a_opt = adam_lse(a_init, beta_sched, lr=0.005, n_steps=50000,
                            time_limit=adam_time, t_start=t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_opt
            print(f"    [Adam] {sc:.8f} ({label})")

    # Phase 2: L-BFGS polish
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 5:
        for beta in [500, 2000]:
            if time.monotonic() - t_start > time_budget:
                break
            t0 = time.monotonic()
            sc, a_opt = lbfgs_polish(best_a, beta, maxiter=3000, maxfun=50000)
            if sc < best_score:
                best_score = sc
                best_a = a_opt
                print(f"    [L-BFGS b={beta}] {sc:.8f} ({label}) [{time.monotonic()-t0:.1f}s]")

    # Phase 3: Coordinate descent peak shaving
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 5:
        cd_time = min(remaining, 60.0)
        sc, a_opt = coord_descent_peak_shave(best_a, n_sweeps=15,
                                             time_limit=cd_time, t_start=t_start,
                                             top_k=60)
        if sc < best_score:
            best_score = sc
            best_a = a_opt
            print(f"    [CoordDesc] {sc:.8f} ({label})")

    # Phase 4: Final L-BFGS with high beta
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 5:
        sc, a_opt = lbfgs_polish(best_a, 5000, maxiter=5000, maxfun=100000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt
            print(f"    [Final L-BFGS] {sc:.8f} ({label})")

    return best_score, best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Uses (1-x)^{-1/2} truncation as primary initialization,
    then refines via multi-phase optimization. Tries multiple lengths.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 30.0

    # Try to load parent sequence
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64 and p.sum() > 0.01:
                parent = np.clip(p, 0.0, 1000.0)
                parent = parent / parent.sum()
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ================================================================
    # Strategy: Try several lengths with (1-x)^{-1/2} initialization
    # The theoretical score for this family is ~pi/2 regardless of n,
    # but optimization can squeeze it lower.
    # ================================================================

    # Define candidate lengths
    # Larger n gives more degrees of freedom but slower per-iteration
    candidate_ns = []

    if parent is not None:
        parent_n = len(parent)
        candidate_ns.append(parent_n)
        # Also try nearby lengths
        for mult in [0.7, 1.0, 1.3, 1.7]:
            nl = int(parent_n * mult)
            if 256 <= nl <= 20000 and nl not in candidate_ns:
                candidate_ns.append(nl)

    # Always include some standard lengths
    for nl in [512, 1024, 2048, 4096, 8192]:
        if nl not in candidate_ns and 256 <= nl <= 20000:
            candidate_ns.append(nl)

    candidate_ns = sorted(set(candidate_ns))
    print(f"Parent: {'yes (n=' + str(len(parent)) + ')' if parent is not None else 'no'}")
    print(f"Candidate lengths: {candidate_ns}")

    # ================================================================
    # Phase A: Quick scan to find the best length (30% of budget)
    # ================================================================
    phase_a_budget = effective_budget * 0.25
    print(f"\n{'='*60}")
    print(f"Phase A: Length scan (budget={phase_a_budget:.0f}s)")
    print(f"{'='*60}")

    length_results = []  # (score, a, n)

    for nl in candidate_ns:
        if time.monotonic() - t_start > t_start + phase_a_budget:
            break

        # Initialize from (1-x)^{-1/2}
        a_init = init_half_power(nl)
        a_init_score = compute_score(a_init)
        print(f"  n={nl}: init score={a_init_score:.6f}")

        # Quick Adam (short)
        t0 = time.monotonic()
        adam_time = min(30.0, (t_start + phase_a_budget - time.monotonic()))
        if adam_time < 5:
            break
        beta_sched = [(50, 0.3), (200, 0.4), (1000, 0.3)]
        sc, a_opt = adam_lse(a_init, beta_sched, lr=0.004, n_steps=20000,
                           time_limit=adam_time, t_start=t_start)
        print(f"  n={nl}: after Adam score={sc:.6f} ({time.monotonic()-t0:.1f}s)")
        length_results.append((sc, a_opt, nl))

    if length_results:
        length_results.sort(key=lambda x: x[0])
        print(f"  Best from scan: n={length_results[0][2]}, score={length_results[0][0]:.6f}")

    # ================================================================
    # Phase B: Deep optimization on top-3 lengths (45% of budget)
    # ================================================================
    phase_b_budget = effective_budget * 0.45
    print(f"\n{'='*60}")
    print(f"Phase B: Deep optimization (budget={phase_b_budget:.0f}s)")
    print(f"{'='*60}")

    top_ns = [nr[2] for nr in length_results[:3]] if length_results else [1024]
    # Also include parent length if not already there
    if parent is not None and len(parent) not in top_ns:
        top_ns.insert(0, len(parent))

    per_length_time = phase_b_budget / max(len(top_ns), 1)

    for nl in top_ns:
        if time.monotonic() - t_start > t_start + phase_a_budget + phase_b_budget:
            break

        # Use the best result from Phase A for this length if available
        a_init = None
        for sc, a, n in length_results:
            if n == nl:
                a_init = a.copy()
                break
        if a_init is None:
            a_init = init_half_power(nl)

        # If we have a parent, also try resampling it
        if parent is not None:
            a_init = resample(parent, nl)
            a_init = np.clip(a_init, 0, 1000)
            if a_init.sum() < 0.01:
                a_init = init_half_power(nl)
            else:
                a_init = a_init / a_init.sum()

        t_budget = min(per_length_time,
                       t_start + phase_a_budget + phase_b_budget - time.monotonic())
        if t_budget < 10:
            break

        sc, a_opt = multi_start_optimize(a_init, t_budget, t_start, label=f"n={nl}")
        try_update(sc, a_opt, f"PhaseB n={nl}")

    # ================================================================
    # Phase C: Peak-shaving coordinate descent on best (15% of budget)
    # ================================================================
    phase_c_budget = effective_budget * 0.15
    print(f"\n{'='*60}")
    print(f"Phase C: Intensive coordinate descent (budget={phase_c_budget:.0f}s)")
    print(f"{'='*60}")

    if best_a is not None:
        n_best = len(best_a)
        # Multiple rounds of CD with increasing top_k
        cd_time = min(phase_c_budget,
                      t_start + effective_budget - time.monotonic())
        if cd_time > 5:
            sc, a_opt = coord_descent_peak_shave(best_a, n_sweeps=50,
                                                 time_limit=cd_time, t_start=t_start,
                                                 top_k=80)
            try_update(sc, a_opt, "PhaseC CD")

            # L-BFGS after CD
            if time.monotonic() - t_start < t_start + effective_budget - 10:
                sc2, a_opt2 = lbfgs_polish(a_opt, 3000, maxiter=5000, maxfun=80000)
                try_update(sc2, a_opt2, "PhaseC CD+LBFGS")

    # ================================================================
    # Phase D: Alternative initializations and perturbations (10%)
    # ================================================================
    phase_d_budget = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase D: Alternative starts (budget={phase_d_budget:.0f}s)")
    print(f"{'='*60}")

    if best_a is not None and time.monotonic() - t_start < t_start + effective_budget - 30:
        n_best = len(best_a)

        # Try perturbed versions of the best
        strategies = [
            # Gaussian perturbation
            lambda: best_a * (1.0 + 0.05 * rng.standard_normal(n_best)),
            # Peak-reduction perturbation
            lambda: _peak_reduce_perturb(best_a),
            # Exponential blend
            lambda: 0.7 * best_a + 0.3 * init_exponential(n_best, 1.0),
            # Half-power blend
            lambda: 0.8 * best_a + 0.2 * init_half_power(n_best),
        ]

        per_strategy_time = phase_d_budget / len(strategies)

        for i, strat_fn in enumerate(strategies):
            if time.monotonic() - t_start > t_start + effective_budget - 30:
                break
            try:
                a_pert = np.clip(strat_fn(), 0.0, 1000.0)
                if a_pert.sum() < 0.01:
                    continue
                a_pert = a_pert / a_pert.sum()

                # Quick optimization
                t_budget = min(per_strategy_time,
                               t_start + effective_budget - 30 - time.monotonic())
                if t_budget < 5:
                    break
                sc, a_opt = multi_start_optimize(a_pert, t_budget, t_start,
                                                 label=f"perturb{i}")
                try_update(sc, a_opt, f"PhaseD perturb{i}")
            except Exception as e:
                print(f"  Perturb {i} failed: {e}")

    # ================================================================
    # Phase E: Final polishing (10% of remaining)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase E: Final polishing")
    print(f"{'='*60}")

    if best_a is not None:
        final_deadline = t_start + effective_budget - 5.0

        # High-beta L-BFGS
        if time.monotonic() < final_deadline - 5:
            sc, a_opt = lbfgs_polish(best_a, 10000, maxiter=10000, maxfun=200000)
            try_update(sc, a_opt, "Final LBFGS high-beta")

        # Final CD
        if time.monotonic() < final_deadline - 3:
            sc, a_opt = coord_descent_peak_shave(best_a, n_sweeps=20,
                                                 time_limit=min(30.0, final_deadline - time.monotonic() - 2),
                                                 t_start=t_start, top_k=100)
            try_update(sc, a_opt, "Final CD")

        # One more L-BFGS
        if time.monotonic() < final_deadline - 2:
            sc, a_opt = lbfgs_polish(best_a, 5000, maxiter=5000, maxfun=100000)
            try_update(sc, a_opt, "Final LBFGS 2")

    # ================================================================
    # Final verification
    # ================================================================
    if best_a is None:
        # Fallback: use (1-x)^{-1/2} of length 1024
        best_a = init_half_power(1024)
        best_score = compute_score(best_a)

    # Ensure valid output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 1024

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out


def _peak_reduce_perturb(a):
    """Perturb by reducing elements that contribute most to the peak."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    a_pert = a.copy()
    if je >= js:
        # Reduce peak contributors
        scale = 0.7
        a_pert[js:je+1] *= scale
        # Redistribute to non-peak contributors
        removed = float((1 - scale) * a[js:je+1].sum())
        mask = np.arange(n) < js
        mask |= np.arange(n) > je
        if mask.sum() > 0:
            a_pert[mask] += removed / mask.sum()
    return a_pert
```
