**Peak shaving algorithm:**
1. Compute R = conv(a, a)
2. Find k* = argmax(R)
3. The gradient of R(k*) w.r.t. a is: g[j] = a[j + k*] + a[j - k*] (with appropriate boundary handling)
   More precisely, R(k*) = sum_j a[j] * a[j+k*], so dR(k*)/da[j] = a[j+k*] (if j+k* < n) + a[j-k*] (if j-k* >= 0)
   Wait, R(k) = sum_j a[j] * a[j+k] for k >= 0. So dR(k)/da[j] = a[j+k] (contribution from a[j] in the first factor) + a[j-k] (contribution from a[j] in the second factor, when the index in the first factor is j-k).
   
   Actually, let me be more careful. R(k) = sum_{j=0}^{n-1-k} a[j] * a[j+k].
   dR(k)/da[m] = a[m+k] (if m+k < n, i.e., m < n-k) + a[m-k] (if m-k >= 0, i.e., m >= k).
   
   So the subgradient vector has:
   - For m in [0, n-k-1]: contribution a[m+k]
   - For m in [k, n-1]: contribution a[m-k]
   - For m in [k, n-k-1]: both contributions, total a[m+k] + a[m-k]

4. The full gradient of the score F(a) = 2n*R(k*)/S^2 is:
   dF/da = (2n/S^2) * g - (4n*R(k*)/S^3) * 1 (the 1 vector, from differentiating S^2 = (sum a)^2)

5. Step: a_new = a - alpha * dF/da, then clip to [0, 1000].

The step size alpha can be chosen by:

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Improvements over previous:
1. Adaptive LSE-continuation with progress-based beta scheduling
2. Multi-length exploration (6962, 8192, 10000)
3. Momentum-based subgradient with backtracking
4. Spectral-domain ADMM for spectral shaping
5. Diversified restarts with parametric families
6. Active-set-aware local search targeting top-K correlation peaks
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_value_grad(a, beta):
    """
    LSE-smoothed max of autocorrelation and its gradient.
    Returns (lse_value, gradient_wrt_a, c_array, S).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n), c, S
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    # Gradient of L w.r.t. a: sum_k p[k] * dR[k]/da
    # dR[k]/da[j] = a[j+k] (if j+k<n) + a[j-k] (if j-k>=0)
    # = cross-correlation of a with p (reversed), evaluated at positions n-1..2n-2
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad_L = 2.0 * conv_res
    return L, grad_L, c, S


def score_grad(a, beta):
    """Full score and gradient: F(a) = 2n * L(a) / S(a)^2."""
    n = len(a)
    L, grad_L, c, S = lse_value_grad(a, beta)
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def lbfgs_run(a0, beta, maxiter=3000, time_limit=30.0):
    """Run L-BFGS-B on LSE-smoothed score. Returns (score, array)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    t0 = time.monotonic()

    def objective(x):
        if time.monotonic() - t0 > time_limit:
            raise TimeoutError()
        sc, g = score_grad(x, beta)
        return sc, g

    try:
        res = minimize(objective, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-18, 'gtol': 1e-15,
                                'maxiter': maxiter, 'maxfun': maxiter * 30, 'maxls': 30})
        a_out = np.clip(res.x, 0.0, 1000.0)
    except (TimeoutError, Exception):
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())
        inits.append(parent.astype(np.float64) * (1.0 + 0.1 * rng.standard_normal(n)))
        inits.append(parent[::-1].astype(np.float64).copy())

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Anti-palindromic ramps
    inits.append(np.linspace(0.05, 1.0, n))
    inits.append(np.linspace(1.0, 0.05, n))

    # Quadratic
    inits.append(0.05 + 0.95 * x ** 2)
    inits.append(0.05 + 0.95 * (1 - x) ** 2)
    inits.append(0.1 + 0.8 * x * (1 - x))

    # Sinusoidal anti-palindromic
    for freq in [1, 2, 3, 5, 7]:
        inits.append(np.clip(0.5 + 0.4 * np.sin(np.pi * freq * x), 0.01, 2.0))

    # Cosine modulation
    for freq in [3, 5, 7, 11]:
        inits.append(np.clip(0.5 + 0.3 * np.cos(2 * np.pi * freq * x), 0.01, 2.0))

    # Combined structures
    inits.append(np.clip(0.3 + 0.5 * x + 0.1 * np.sin(2 * np.pi * 13 * x), 0.01, 2.0))
    inits.append(np.clip(0.3 + 0.5 * (1 - x) + 0.1 * np.sin(2 * np.pi * 7 * x), 0.01, 2.0))

    # Smooth random
    a_rand = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 2.0)
    ksize = max(3, n // 100)
    kernel = np.ones(ksize) / ksize
    inits.append(np.clip(np.convolve(a_rand, kernel, mode='same'), 0.01, 2.0))

    # Pure random (mild)
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 2.0))

    # Two-block
    half = n // 2
    inits.append(np.concatenate([np.ones(half) * 0.7, np.ones(n - half) * 0.3]))
    inits.append(np.concatenate([np.ones(half) * 0.3, np.ones(n - half) * 0.7]))

    # Smooth exponential decay
    inits.append(np.exp(-3.0 * x))
    inits.append(np.exp(-3.0 * (1 - x)))

    # Spline-like: peak off-center
    for center in [0.3, 0.4, 0.7, 0.6]:
        inits.append(np.clip(0.1 + 0.9 * np.exp(-((x - center) * 4) ** 2), 0.01, 2.0))

    return inits


def subgradient_momentum(a, max_iters=5000, time_limit=30.0):
    """Momentum-based subgradient with adaptive step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    velocity = np.zeros(n)
    mu = 0.9  # momentum
    t0 = time.monotonic()
    step_scale = 1.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max(R) at the active lag
        k_star = int(np.argmax(c))
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break

        # Adaptive step with momentum
        step = (sc - sc_lb) / (gn2 + 1e-15)
        step = min(step, step_scale / (1.0 + 0.01 * it))
        step_scale = max(0.5 * step_scale, step)

        velocity = mu * velocity - step * grad
        a_new = a + velocity
        a_new = np.clip(a_new, 0.0, 1000.0)

        # Check if step improved; if not, shrink
        sc_new = compute_score(a_new)
        if sc_new > sc * 1.01:
            step_scale *= 0.5
            velocity = mu * velocity
            a = a + velocity
            a = np.clip(a, 0.0, 1000.0)
        else:
            a = a_new

    return best_sc, best_a


def multi_peak_subgradient(a, K=5, max_iters=3000, time_limit=20.0):
    """Subgradient using top-K peaks for smoother descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    velocity = np.zeros(n)
    mu = 0.85

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Use top-K lags
        sorted_idx = np.argsort(c)[::-1][:K]
        weights = np.exp(-0.5 * (c[M - c[sorted_idx]] / max(M * 0.01, 1e-10)))
        weights /= weights.sum()

        sg = np.zeros(n)
        for w, k in zip(weights, sorted_idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += w * 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break

        step = (sc - sc_lb) / (gn2 + 1e-15)
        step = min(step, 0.5 / (1.0 + 0.005 * it))
        velocity = mu * velocity - step * grad
        a = np.clip(a + velocity, 0.0, 1000.0)

    return best_sc, best_a


def spectral_admm(a, max_iters=200, time_limit=20.0, mu=100.0):
    """
    ADMM in the spectral domain to flatten the power spectrum while maintaining a >= 0.
    Variables: x (spectrum), A (time-domain constraint variable)
    Minimize: max(R(a)) + indicator{a >= 0}
    """
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()  # FFT size for linear conv
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    A_fft = np.fft.rfft(a, N)
    lam = np.zeros_like(A_fft)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # x-update: minimize LSE(R(x)) + (mu/2)||x - A_fft + lam||^2
        # gradient of LSE part: 2 * IFFT(N) of (p_conj * x) ... complex
        # Simplified: do a few gradient steps in time domain
        a_time = np.fft.irfft(A_fft + lam, N)[:n]
        a_time = np.clip(a_time, 0.0, 1000.0)
        # LSE step in time domain
        for _ in range(3):
            L, grad_L, c, S = lse_value_grad(a_time, 500.0)
            S2 = S * S
            S3 = S2 * S
            score = 2.0 * n * L / S2
            grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
            step = 0.1 / (np.linalg.norm(grad) + 1e-10)
            a_time = np.clip(a_time - step * grad, 0.0, 1000.0)

        x = np.fft.rfft(a_time, N)

        # A-update: project onto {A : IFFT(A)[:n] >= 0}
        b = np.fft.irfft(x + lam, N)[:n]
        b = np.clip(b, 0.0, 1000.0)
        A_fft = np.fft.rfft(b, N)

        # lambda update
        lam = lam + x - A_fft

        a_new = np.fft.irfft(A_fft, N)[:n]
        a_new = np.clip(a_new, 0.0, 1000.0)
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        a = a_new

    return best_sc, best_a


def parametric_search(n, rng, time_limit=30.0):
    """
    Search over a parametric family: a[j] = exp(polynomial of j/n).
    Uses L-BFGS in the low-dimensional coefficient space.
    """
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    best_sc = np.inf
    best_a = np.ones(n) * 0.5

    # Polynomial degrees to try
    for degree in [2, 3, 4, 5, 7]:
        if time.monotonic() - t0 > time_limit:
            break

        def param_to_a(coeffs):
            poly_val = np.zeros(n)
            for i, c in enumerate(coeffs):
                poly_val += c * x ** i
            return np.exp(np.clip(poly_val, -10, 10))

        def param_obj(coeffs):
            a = param_to_a(coeffs)
            S = float(a.sum())
            if S < 1e-12:
                return 1e10, np.zeros(degree + 1)
            c_conv = fast_conv(a)
            c_max = float(c_conv.max())
            beta = 300.0
            scaled = beta * (c_conv - c_max)
            exp_s = np.exp(scaled)
            L = c_max + np.log(exp_s.sum()) / beta
            score = 2.0 * n * L / (S * S)
            # Gradient via chain rule
            p = exp_s / exp_s.sum()
            rev_a = a[::-1]
            M2 = 3 * n - 2
            M2_fft = 1 << (M2 - 1).bit_length()
            RA = np.fft.rfft(rev_a, M2_fft)
            P = np.fft.rfft(p, M2_fft)
            conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
            grad_L = 2.0 * conv_res
            S2 = S * S
            S3 = S2 * S
            grad_a = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
            # d a[k] / d coeffs[i] = a[k] * k^i
            grad_c = np.zeros(degree + 1)
            for i in range(degree + 1):
                grad_c[i] = float((a * grad_a * x ** i).sum())
            return score, grad_c

        # Try multiple random inits
        for attempt in range(3):
            if time.monotonic() - t0 > time_limit:
                break
            c0 = rng.standard_normal(degree + 1) * 0.5
            bounds = [(-3, 3)] * (degree + 1)
            try:
                res = minimize(param_obj, c0, method='L-BFGS-B', jac=True,
                               bounds=bounds, options={'maxiter': 1000, 'maxfun': 30000})
                a_opt = param_to_a(res.x)
                sc = compute_score(a_opt)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_opt.copy()
            except Exception:
                pass

    return best_sc, best_a


def tikhonov_refine(a, time_limit=15.0):
    """Add small Tikhonov regularization to find smoother local optima."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    for lamb in [0.001, 0.01, 0.1]:
        if time.monotonic() - t0 > time_limit:
            break

        def obj_with_reg(x):
            if time.monotonic() - t0 > time_limit:
                raise TimeoutError()
            sc, g = score_grad(x, 2000.0)
            # Tikhonov: encourage smoothness
            if len(x) > 2:
                d = np.diff(x)
                g_reg = np.zeros_like(x)
                g_reg[:-1] += -lamb * d
                g_reg[1:] += lamb * d
                reg_val = lamb * float(d @ d)
            else:
                g_reg = np.zeros_like(x)
                reg_val = 0.0
            return sc + reg_val, g + g_reg

        try:

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Improvements over previous:
1. Adaptive LSE-continuation with progress-based beta scheduling
2. Multi-length exploration (6962, 8192, 9000, 10240)
3. Momentum-based subgradient with backtracking
4. Spectral-domain ADMM for spectral shaping
5. Diversified restarts with parametric families
6. Active-set-aware local search targeting top-K correlation peaks
7. Tikhonov-regularized refinement for smoother local optima
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_value_grad(a, beta):
    """LSE-smoothed max of autocorrelation and its gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n), c, S
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad_L = 2.0 * conv_res
    return L, grad_L, c, S


def score_grad(a, beta):
    """Full score and gradient: F(a) = 2n * LSE(a) / S(a)^2."""
    n = len(a)
    L, grad_L, c, S = lse_value_grad(a, beta)
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def lbfgs_run(a0, beta, maxiter=3000, time_limit=30.0):
    """Run L-BFGS-B on LSE-smoothed score. Returns (score, array)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    t0 = time.monotonic()
    timed_out = False

    def objective(x):
        nonlocal timed_out
        if time.monotonic() - t0 > time_limit:
            timed_out = True
            return 1e10, np.zeros(len(x))
        sc, g = score_grad(x, beta)
        return sc, g

    try:
        res = minimize(objective, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-18, 'gtol': 1e-15,
                                'maxiter': maxiter, 'maxfun': maxiter * 30, 'maxls': 30})
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    if parent is not None and len(parent) == n:

    inits.append(0.05 + 0.95 * (1 - x) ** 2)

    a_rand = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 2.0)

    inits.append(np.clip(np.convolve(a_rand, kernel, mode='same'), 0.01, 2.0))

def subgradient_momentum(a, max_iters=5000, time_limit=30.0):
    """Momentum-based subgradient with adaptive step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    velocity = np.zeros(n)
    mu = 0.9
    t0 = time.monotonic()
    step_scale = 1.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        k_star = int(np.argmax(c))
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break

        step = (sc - sc_lb) / (gn2 + 1e-15)
        step = min(step, step_scale / (1.0 + 0.01 * it))
        step_scale = max(0.5 * step_scale, step)

        velocity = mu * velocity - step * grad
        a_new = np.clip(a + velocity, 0.0, 1000.0)

        sc_new = compute_score(a_new)
        if sc_new > sc * 1.02:
            step_scale *= 0.5
            velocity = mu * velocity
            a = np.clip(a + velocity, 0.0, 1000.0)
        else:
            a = a_new

    return best_sc, best_a


def multi_peak_subgradient(a, K=5, max_iters=3000, time_limit=20.0):
    """Subgradient using top-K peaks for smoother descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    velocity = np.zeros(n)
    mu = 0.85

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        sorted_idx = np.argsort(c)[::-1][:K]
        diff = M - c[sorted_idx]
        weights = np.exp(-diff / max(M * 0.05, 1e-10))
        weights /= weights.sum()

        sg = np.zeros(n)
        for w, k in zip(weights, sorted_idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += w * 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break

        step = (sc - sc_lb) / (gn2 + 1e-15)
        step = min(step, 0.5 / (1.0 + 0.005 * it))
        velocity = mu * velocity - step * grad
        a = np.clip(a + velocity, 0.0, 1000.0)

    return best_sc, best_a


def spectral_admm(a, max_iters=200, time_limit=20.0, mu=50.0):
    """ADMM in spectral domain to flatten power spectrum while maintaining a >= 0."""
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    A_fft = np.fft.rfft(a, N)
    lam = np.zeros_like(A_fft)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        a_time = np.fft.irfft(A_fft + lam, N)[:n]
        a_time = np.clip(a_time, 0.0, 1000.0)

        for _ in range(3):
            L, grad_L, c, S = lse_value_grad(a_time, 500.0)
            S2 = S * S
            S3 = S2 * S
            grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = 0.1 / (gnorm + 1e-10)
            a_time = np.clip(a_time - step * grad, 0.0, 1000.0)

        x = np.fft.rfft(a_time, N)
        b = np.fft.irfft(x + lam, N)[:n]
        b = np.clip(b, 0.0, 1000.0)
        A_fft = np.fft.rfft(b, N)
        lam = lam + x - A_fft

        a_new = np.fft.irfft(A_fft, N)[:n]
        a_new = np.clip(a_new, 0.0, 1000.0)
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        a = a_new

    return best_sc, best_a
