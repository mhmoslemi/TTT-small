"""
Optimized step function search minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses multi-phase approach: LSE continuation, top-k peak subgradient, 
and coordinate-wise local search with exact 1D optimization.
"""

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta, top_k=None):
    """
    LSE-smoothed max-conv score and gradient.
    If top_k is given, only consider the top_k peaks in the LSE.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    if top_k is not None and top_k < len(c):
        # Get indices of top_k peaks
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[idx]
        c_max_top = c_top.max()
        # Create a masked version
        c_masked = np.full_like(c, -np.inf)
        c_masked[idx] = c_top
        c_eff = c_masked
    else:
        c_eff = c
        c_max_top = c.max()
    
    c_max = float(np.max(c_eff))
    scaled = beta * (c_eff - c_max)
    # For -inf entries, exp will be 0, which is fine
    exp_s = np.exp(scaled)
    exp_s[~np.isfinite(exp_s)] = 0.0
    total = exp_s.sum()
    if total < 1e-300:
        return 1e10, np.zeros(n)
    L = c_max + np.log(total) / beta
    p = exp_s / total
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapper(a, beta, top_k=None):
    return lse_score_and_grad(a, beta, top_k)


def run_lbfgs_timed(a0, beta, top_k=None, maxiter=5000, maxfun=100000, time_limit=None):
    """Run L-BFGS-B with time limit."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    t0 = time.monotonic()
    try:
        res = minimize(
            lse_wrapper, a0, args=(beta, top_k),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
    except Exception:
        a_out = a0
        return compute_score(a_out), a_out, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.success


def topk_subgrad_polish(a, max_iters=5000, k=5, sc_lb=None):
    """Subgradient descent targeting top-k peaks simultaneously."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    if sc_lb is None:
        sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top-k peaks
        k_eff = min(k, 2*n-1)
        top_idx = np.argpartition(c, -k_eff)[-k_eff:]
        
        # Weighted subgradient: weight each peak by (c[i] - M + epsilon)
        # This gives more weight to peaks close to the max
        eps = 0.1 * M + 1e-10
        weights = np.zeros(2*n-1)
        weights[top_idx] = (c[top_idx] + eps) / (c[top_idx] + eps).sum()
        
        # Compute combined subgradient
        sg = np.zeros(n)
        S2 = S * S
        S3 = S2 * S
        
        for k_star in top_idx:
            w = weights[k_star]
            j_start = max(0, int(k_star) - n + 1)
            j_end = min(int(k_star), n - 1)
            if j_end >= j_start:
                sg[j_start:j_end+1] += w * a[j_start:j_end+1][::-1]
        
        # Scale by 2 (from the derivative of the product)
        sg *= 2.0
        
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - sc_lb) / gn2
        step = min(step, 1.0 / (1.0 + 0.01 * it))
        a = np.clip(a - step * grad, 0.0, 1000.0)
    
    return best_sc, best_a

conv(a,a)[k] = sum_j a[j] * a[k-j]
The terms involving a[i] are:
- j = i: a[i] * a[k-i] (if 0 <= k-i < n)
- k-j = i, i.e., j = k-i: a[k-i] * a[i] (if 0 <= k-i < n)

- Momentum-based subgradient for faster convergence
- Adaptive multi-restart with diverse perturbation strategies
- Efficient time budgeting focused on high-beta refinement

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = c - c_max
    scaled = beta * shifted
    np.clip(scaled, -500, 500, out=scaled)
    exp_s = np.exp(scaled)
    total = exp_s.sum()
    if total < 1e-300:
        return 1e10, np.zeros(n)
    L = c_max + np.log(total) / beta
    p = exp_s / total
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj_wrapper(a, beta):
    return lse_score_grad(a, beta)

def topk_subgrad_momentum(a, max_iters=5000, k=8, momentum=0.85):
    """Subgradient descent with momentum, targeting top-k peaks."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    v = np.zeros(n)
    best_streak = 0

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a.copy()
            best_streak = 0
        else:
            best_streak += 1

        k_eff = min(k, 2 * n - 1)
        top_idx = np.argpartition(c, -k_eff)[-k_eff:]

        # Weights: sharp exponential based on proximity to max
        ratios = c[top_idx] / (M + 1e-10)
        weights = ratios ** 20.0
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights /= wsum

        sg = np.zeros(n)
        for ii in range(len(top_idx)):
            k_star = int(top_idx[ii])
            w = weights[ii]
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += w * a[j_start:j_end + 1][::-1]
        sg *= 2.0

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - sc_lb) / gn2
        max_step = 2.0 / (1.0 + 0.02 * it)
        step = min(step, max_step)

        v = momentum * v - step * grad
        a = np.clip(a + v, 0.0, 1000.0)

        if best_streak > 2000:
            break

    return best_sc, best_a

    # ============================================================
    phase3_end = t_start + 1900.0
    print(f"\n=== Phase 3: Multi-Restart (until {phase3_end - t_start:.0f}s) ===")

    n_restarts = 12
    for r in range(n_restarts):
        if time.monotonic() > phase3_end:
            break
        mode = r % 6
        a_pert = perturb_diverse(best_a, rng, mode)

        t0 = time.monotonic()
        # Run LSE at moderate-to-high beta from perturbed start
        sc, a_opt, _ = lbfgs_run(a_pert, 10000, maxiter=3000)
        if sc < best_score:
            report(sc, a_opt, f"P3 r={r} mode={mode} b=10000")
            # If improved, refine at higher beta
            if time.monotonic() < phase3_end - 60:
                sc2, a_opt2, _ = lbfgs_run(a_opt, 50000, maxiter=2000)
                report(sc2, a_opt2, f"P3 r={r} b=50000")
        else:
            print(f"    r={r} mode={mode}: no improvement ({sc:.8f} vs {best_score:.8f}), t={time.monotonic()-t0:.1f}s")

    # ============================================================
    # Phase 4: Top-k subgradient with momentum
    # ============================================================
    phase4_end = t_start + 2200.0
    print(f"\n=== Phase 4: Top-k Subgradient (until {phase4_end - t_start:.0f}s) ===")

    if time.monotonic() < phase4_end:
        n_iters = max(1000, int((phase4_end - time.monotonic()) * 50))
        t0 = time.monotonic()
        sc, a_opt = topk_subgrad_momentum(best_a, max_iters=n_iters, k=8, momentum=0.85)
        report(sc, a_opt, "P4 topk-subgrad")
        print(f"    Top-k subgrad: {sc:.8f}, t={time.monotonic()-t0:.1f}s")

        # Also try single-peak subgradient
        if time.monotonic() < phase4_end - 30:
            n_iters2 = max(500, int((phase4_end - time.monotonic()) * 80))
            t0 = time.monotonic()
            sc2, a_opt2 = single_peak_subgrad(best_a, max_iters=n_iters2)
            report(sc2, a_opt2, "P4 single-subgrad")
            print(f"    Single subgrad: {sc2:.8f}, t={time.monotonic()-t0:.1f}s")

    # ============================================================
    # Phase 5: Final high-beta LSE polish
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")

    for beta in [100000, 200000]:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        maxit = max(500, min(5000, int(remaining * 0.8 / 0.01)))
        t0 = time.monotonic()
        sc, a_opt, _ = lbfgs_run(best_a, beta, maxiter=maxit)
        report(sc, a_opt, f"P5 b={beta}")
        print(f"    Final b={beta}: {sc:.8f}, t={time.monotonic()-t0:.1f}s")

    # Final subgradient micro-polish
    if time.monotonic() < deadline - 5:
        sc, a_opt = single_peak_subgrad(best_a, max_iters=2000)
        report(sc, a_opt, "final-subgrad")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    # Verify with the actual evaluator
    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    if verified != best_score and abs(verified - best_score) > 1e-6:
        print(f"WARNING: mismatch between internal and verified score!")
        # Use the verified score's sequence (they should be the same)
        # But if there's a numerical discrepancy, trust the verified one
        if verified < best_score:
            best_score = verified

    return out
