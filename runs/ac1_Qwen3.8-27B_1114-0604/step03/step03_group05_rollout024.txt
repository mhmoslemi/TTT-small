    # This is the cross-correlation of a and c_pm1
    # corr = ifft(fft(a, L2) * conj(fft(c_pm1, L2)))
    # But we need to be careful about the indexing
    
    L2 = 1 << (n + len(c) - 1).bit_length()
    A_f = np.fft.fft(a, L2)
    C_f = np.fft.fft(c_pm1, L2)
    corr = np.fft.ifft(A_f * np.conj(C_f)).real
    # corr[i] = sum_m a[m] * c_pm1[(m+i) mod L2]
    # For the linear (non-circular) correlation, we need to handle the boundaries
    # But since c_pm1 is zero outside [0, 2n-2] and a is zero outside [0, n-1],
    # the circular correlation at lags 0..n-1 should give us what we want
    # Actually, corr[i] for i in 0..n-1 gives sum_m a[m] * c_pm1[(m+i) mod L2]
    # which equals sum_{m: 0<=m<n, 0<=m+i<2n-1} a[m] * c_pm1[m+i]
    # = sum_{m: max(0,-i)<=m<n, m+i<2n-1} a[m] * c_pm1[m+i]
    # Since i >= 0, this is sum_{m: 0<=m<=min(n-1, 2n-2-i)} a[m] * c_pm1[m+i]
    
    # But what we want is: sum_k c_k^(p-1) * a_{k-i} for valid k and k-i in [0, n-1]
    # = sum_k c_k^(p-1) * a_{k-i} where k in [0, 2n-2] and k-i in [0, n-1]
    # = sum_{k: i<=k<=min(2n-2, n-1+i)} c_k^(p-1) * a_{k-i}
    # Let j = k-i: sum_{j: 0<=j<=min(2n-2-i, n-1)} c_{j+i}^(p-1) * a_j
    # = sum_{j=0}^{n-1} c_{j+i}^(p-1) * a_j (since 2n-2-i >= n-1 for i <= n-1)
    # This is corr[i] as computed above! ✓
    
    # So d(sum_k c_k^p)/da_i = 2p * corr[i]
    # And dL/da_i = (sum_cp)^(1/p - 1) * 2p * corr[i] / p = 2 * (sum_cp)^(1/p - 1) * corr[i]
    # Wait: L = (sum_cp)^(1/p), dL/d(sum_cp) = (1/p) * (sum_cp)^(1/p - 1)
    # d(sum_cp)/da_i = p * sum_k c_k^(p-1) * db_k/da_i = p * sum_k c_k^(p-1) * 2 * a_{k-i}
    # = 2p * corr[i]
    # So dL/da_i = (1/p) * (sum_cp)^(1/p-1) * 2p * corr[i] = 2 * (sum_cp)^(1/p-1) * corr[i]
    
    factor = 2.0 * (sum_cp) ** (1.0 / p - 1.0)
    dL_da = factor * corr[:n]
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def pnorm_obj(a, p):
    return pnorm_score_grad(a, p)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient: dL/da_i = sum_k p_k * db_k/da_i = sum_k p_k * 2*a_{k-i}
    # = 2 * (a * p_rev)[i] ... or more carefully:
    # sum_k p_k * 2 * a_{k-i} for valid k-i
    # = 2 * sum_k p_k * a_{k-i}
    # Let j = k-i: 2 * sum_j p_{j+i} * a_j
    # This is 2 * corr_a_p[i] where corr is the cross-correlation
    
    L2 = 1 << (n + len(c) - 1).bit_length()
    A_f = np.fft.fft(a, L2)
    P_f = np.fft.fft(p, L2)
    corr = np.fft.ifft(A_f * np.conj(P_f)).real
    dL_da = 2.0 * corr[:n]
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lse_obj(a, beta):
    return lse_score_grad(a, beta)

def subgradient_with_linesearch(a, max_iter=5000):
    """Subgradient method with line search for the max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()

    c = fast_conv(a)

        sc = 2.0 * n * M / (S * S)

            best_sc = sc
            best_a = a.copy()
        
        # Subgradient
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step size with line search
        # Try several step sizes and pick the best
        best_step_sc = sc
        best_a_try = a.copy()
        for factor in [1.0, 0.5, 0.1, 0.01, 0.001]:
            step = factor * (sc - 1.0) / (gn * gn + 1e-20)
            step = min(step, 2.0 / (S + 1e-10))
            step = max(step, 0.0)
            a_try = np.clip(a - step * grad, 0.0, 1000.0)
            if a_try.sum() < 1e-6:
                continue
            sc_try = compute_score(a_try)
            if sc_try < best_step_sc:
                best_step_sc = sc_try
                best_a_try = a_try.copy()
        
        if best_step_sc < sc:
            a = best_a_try
        else:
            # No improvement; try a smaller step
            step = 0.01 / (gn * S + 1e-20)
            a_try = np.clip(a - step * grad, 0.0, 1000.0)

                if sc_try < sc:

                else:
                    # Stuck; break
                    if it > 100:
                        break
        
        # Decay step size
        if it % 500 == 0 and it > 0:

    return best_sc, best_a

def spectral_flatten(a, n_iter=200):
    """
    Flatten the spectrum |A(k)|^2 to reduce autocorrelation peaks.
    Iteratively: compute periodogram, find peaks, adjust a to reduce them.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    L = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iter):
        A = np.fft.rfft(a, L)
        P = np.abs(A) ** 2
        # The DC bin P[0] = S^2 is fixed (since sum(a) is fixed... well, not quite)
        # We want to reduce the peaks in P (excluding DC)
        # Equivalent to reducing peaks in the circular autocorrelation
        
        # Compute circular autocorrelation
        r = np.fft.irfft(P, L)
        # r[0] is the energy, r[1..] are the autocorrelation lags
        # We want to minimize max(r[1:]) (approximately)
        
        # Find the peak lag (excluding 0)
        r_no_dc = r.copy()
        r_no_dc[0] = -np.inf
        k_peak = int(np.argmax(r_no_dc))
        
        if r[k_peak] < 0:
            break
        
        # The subgradient of r[k_peak] w.r.t. a:
        # r[k_peak] = sum_j a_j * a_{(j+k_peak) mod L}
        # dr[k_peak]/da_i = a_{(i+k_peak) mod L} + a_{(i-k_peak) mod L}
        sg = np.zeros(n)
        for i in range(n):
            j1 = (i + k_peak) % L
            j2 = (i - k_peak) % L
            val = 0.0
            if j1 < n:
                val += a[j1]
            if j2 < n:
                val += a[j2]
            sg[i] = val
        
        # Also account for the normalization (sum constraint)
        S = float(a.sum())
        # The objective is 2n * max(r[1:]) / S^2
        # Subgradient of 2n * r[k_peak] / S^2:
        # = 2n * (sg / S^2 - 2 * r[k_peak] * (sum_j delta_{j,i}) / S^3)
        # = 2n * (sg / S^2 - 2 * r[k_peak] / S^3)
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * r[k_peak] / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Take a small step
        step = 0.5 / (gn * S + 1e-10)
        a_new = np.clip(a - step * grad, 0.0, 1000.0)
        
        if a_new.sum() < 1e-6:
            break
        
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_sc, best_a

def multi_scale_perturb(a, rng):
    """Perturb at multiple frequency scales simultaneously."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    A = np.fft.fft(a, L)
    A_mag = np.abs(A)
    A_phase = np.angle(A)
    
    # Perturb at different scales
    for scale in [0.01, 0.03, 0.08]:
        n_freqs = max(2, n // (100 // max(1, int(1/scale))))
        # Randomly select some frequencies to perturb
        freqs = rng.choice(L, size=min(n_freqs, L), replace=False)
        for f in freqs:
            A_mag[f] *= (1.0 + scale * rng.standard_normal())
            if f < L - 1:
                A_mag[L - f] = A_mag[f]  # maintain Hermitian symmetry
    
    A_mag = np.clip(A_mag, 0, None)
    A_new = A_mag * np.exp(1j * A_phase)
    a_new = np.fft.ifft(A_new).real[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new

def diverse_init(n, rng, method='parent'):
    """Create diverse initializations."""
    if method == 'uniform':
        return np.ones(n) * 0.5
    elif method == 'raised_cosine':
        x = np.arange(n) / n
        return 0.5 * (1 + np.cos(2 * np.pi * x))
    elif method == 'hamming':
        w = np.hamming(n)
        return w / w.sum() * n * 0.5
    elif method == 'random_smooth':
        # Generate smooth random signal
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.1, 0.5, size=5)
        x = np.arange(n)
        sig = np.ones(n)
        for f, p, a_val in zip(freqs, phases, amps):
            sig += a_val * np.cos(2 * np.pi * f * x / n + p)
        return np.clip(sig, 0.1, None)
    elif method == 'bimodal':
        # Two peaks
        center1 = n // 3
        center2 = 2 * n // 3
        sigma = n // 10
        sig = np.exp(-0.5 * ((np.arange(n) - center1) / sigma) ** 2) + \
              np.exp(-0.5 * ((np.arange(n) - center2) / sigma) ** 2)
        return sig
    elif method == 'comb':
        # Comb filter
        period = max(2, n // rng.integers(3, 8))
        sig = np.ones(n)
        sig[::period] *= 2.0
        return sig
    else:
        return np.ones(n) * 0.5

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient.
    Objective: 2n * LSE(beta; b) / S^2 where LSE = max(b) + log(sum(exp(beta*(b-max(b)))))/beta
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # dL/da_i = 2 * sum_j p_{j+i} * a_j  (cross-correlation of a and p at lag i)
    L2 = 1 << (n + len(c) - 1).bit_length()
    A_f = np.fft.fft(a, L2)

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, func, args, maxiter=2000, maxfun=50000):
    """Run L-BFGS-B optimization with bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:

    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-20, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0

    n = len(a)

            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

            break

        # Subgradient of 2n*M/S^2 at the argmax

        j_end = min(k_star, n - 1)

        S2 = S * S

        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)

        if gn < 1e-15:
            break

        # Line search: try multiple step sizes

        best_try_a = a.copy()

        for f in candidates:
            step = f * base_step
            a_try = np.clip(a - step * grad, 0.0, 1000.0)

            sc_try = compute_score(a_try)
            if sc_try < best_try_sc:
                best_try_sc = sc_try
                best_try_a = a_try.copy()

        if best_try_sc < best_sc:
            a = best_try_a
        elif best_try_sc < sc:
            a = best_try_a

            a_try = np.clip(a - step * grad, 0.0, 1000.0)

def multi_scale_perturb(a, rng, strength=0.05):
    """Perturb at multiple frequency scales to escape local minima."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, L)
    A_mag = np.abs(A).copy()
    A_phase = np.angle(A).copy()

    # Low-frequency perturbation
    n_low = max(2, n // 200)
    low_freqs = rng.integers(1, n_low, size=n_low)
    for f in low_freqs:
        d = strength * 2.0 * rng.standard_normal()
        A_mag[f] = max(0, A_mag[f] * (1.0 + d))
        if f < L // 2 and L - f < len(A_mag):
            A_mag[L - f] = A_mag[f]

    # Mid-frequency perturbation
    n_mid = max(2, n // 50)
    mid_freqs = rng.integers(n_low, L // 4, size=min(n_mid, L // 4 - n_low))
    for f in mid_freqs:
        d = strength * rng.standard_normal()
        A_mag[f] = max(0, A_mag[f] * (1.0 + d))
        if L - f < len(A_mag):
            A_mag[L - f] = A_mag[f]

    A_new = A_mag * np.exp(1j * A_phase)
    a_new = np.fft.irfft(A_new, L)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 1e-6:
        a_new = a.copy()
    return a_new


def targeted_rebalance(a, rng, strength=0.08):
    """Reduce peak-contributing elements, boost others. Sum-preserving."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)

    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

    top_k = max(10, n // 30)
    top_idx = np.argsort(contrib)[::-1][:top_k]
    bottom_k = top_k * 2
    bottom_idx = np.argsort(contrib)[:bottom_k]

    total_reduce = 0.0
    for idx in top_idx:
        amt = a_new[idx] * strength
        a_new[idx] -= amt
        total_reduce += amt

    per = total_reduce / bottom_k
    a_new[bottom_idx] += per
    return np.clip(a_new, 0.0, 1000.0)


def diverse_init(n, rng, method='uniform'):
    """Create diverse initial sequences for exploration."""
    if method == 'uniform':
        return np.ones(n) * 0.5
    elif method == 'raised_cosine':
        x = np.arange(n, dtype=np.float64) / n
        return 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif method == 'bimodal':
        c1, c2 = n // 3, 2 * n // 3
        sig = np.exp(-0.5 * ((np.arange(n) - c1) / (n // 8)) ** 2) + \
              np.exp(-0.5 * ((np.arange(n) - c2) / (n // 8)) ** 2)
        return sig
    elif method == 'smooth_random':
        x = np.arange(n, dtype=np.float64)
        sig = np.ones(n) * 0.8
        for _ in range(3):
            f = rng.uniform(1, 20)
            p = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.05, 0.3)
            sig += amp * np.cos(2 * np.pi * f * x / n + p)
        return np.clip(sig, 0.01, None)
    elif method == 'tapered':
        # Tapered ends, flat middle
        w = np.ones(n)
        taper = n // 20
        w[:taper] = np.linspace(0.3, 1.0, taper)
        w[-taper:] = np.linspace(1.0, 0.3, taper)
        return w
    else:
        return np.ones(n) * 0.5

    print("\n=== Phase 2: p-Norm Continuation ===")

    a_cur = best_a.copy()
    p_schedule = [2, 4, 8, 16, 32, 64]

    for p in p_schedule:
        if time.monotonic() > phase2_end:
            break
        t0 = time.monotonic()
        a_cur = run_lbfgs(a_cur, pnorm_obj, (float(p),), maxiter=800, maxfun=20000)
        sc = compute_score(a_cur)
        report(sc, a_cur, f"pNorm p={p}")
        print(f"  p={p}: score={sc:.8f}, t={time.monotonic()-t0:.0f}s")

    # ===================================================================
    # Phase 3: Subgradient Polishing - 13% time
    # ===================================================================
    phase3_end = t_start + 0.65 * (total_budget - 30.0)
    print("\n=== Phase 3: Subgradient Polishing ===")

    if time.monotonic() < phase3_end:
        t0 = time.monotonic()
        n_iters = int(min(5000, (phase3_end - time.monotonic()) * 300))
        sc, a_sg = subgradient_linesearch(best_a, max_iter=max(500, n_iters))
        report(sc, a_sg, "subgrad LS")
        print(f"  Subgrad: score={sc:.8f}, t={time.monotonic()-t0:.0f}s")

    # ===================================================================
    # Phase 4: Multi-Scale Basin Hopping - 28% time
    # ===================================================================
    phase4_end = deadline
    print("\n=== Phase 4: Basin Hopping ===")

    hop_count = 0
    max_hops = 40
    while time.monotonic() < phase4_end and hop_count < max_hops:
        t0 = time.monotonic()
        time_per_hop = (phase4_end - t0) / max(1, max_hops - hop_count)
        strategy = hop_count % 5

        if strategy == 0:
            # Multi-scale spectral perturbation
            a_pert = multi_scale_perturb(best_a, rng, strength=0.03 + 0.05 * rng.random())
        elif strategy == 1:
            # Targeted rebalance
            a_pert = targeted_rebalance(best_a, rng, strength=0.05 + 0.1 * rng.random())
        elif strategy == 2:
            # Blend with diverse init
            methods = ['raised_cosine', 'bimodal', 'smooth_random', 'tapered']
            a_div = diverse_init(n, rng, method=methods[hop_count % len(methods)])
            alpha = 0.05 + 0.2 * rng.random()
            a_pert = (1.0 - alpha) * best_a + alpha * a_div
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Large random restart (nuclear option)
            if rng.random() < 0.5:
                a_pert = diverse_init(n, rng, method='smooth_random')
            else:
                # Scatter: zero out random 20% and redistribute
                a_pert = best_a.copy()
                n_zero = max(10, n // 5)
                zero_idx = rng.choice(n, size=n_zero, replace=False)
                total = a_pert[zero_idx].sum()
                a_pert[zero_idx] = 0
                fill_idx = np.setdiff1d(np.arange(n), zero_idx)
                a_pert[fill_idx] += total / len(fill_idx)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Phase perturbation in frequency domain
            L = 1 << (2 * n - 1).bit_length()
            A = np.fft.rfft(best_a, L)
            A_mag = np.abs(A)
            A_phase = np.angle(A)
            d_phase = 0.15 * rng.standard_normal(len(A_phase))
            d_phase[0] = 0.0
            A_new = A_mag * np.exp(1j * (A_phase + d_phase))
            a_pert = np.fft.irfft(A_new, L)[:n]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            if a_pert.sum() < 1e-6:
                a_pert = best_a * (1.0 + 0.1 * rng.standard_normal(n))
                a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick LSE optimization from perturbed start
        maxiter = max(300, int(time_per_hop * 50))
        a_opt = run_lbfgs(a_pert, lse_obj, (2000,), maxiter=min(maxiter, 1000),
                         maxfun=min(maxiter * 25, 25000))
        sc = compute_score(a_opt)
        report(sc, a_opt, f"hop {hop_count} s{strategy}")

        # If promising, do additional refinement
        if sc < best_score + 0.01:
            a_opt = run_lbfgs(a_opt, lse_obj, (8000,), maxiter=500, maxfun=15000)
            sc = compute_score(a_opt)
            report(sc, a_opt, f"hop {hop_count} refine")
            # Quick subgradient
            sc2, a_sg2 = subgradient_linesearch(a_opt, max_iter=500)
            report(sc2, a_sg2, f"hop {hop_count} sg")

        hop_count += 1

    # ===================================================================
    # Phase 5: Final High-Precision Polish
    # ===================================================================
    print("\n=== Phase 5: Final Polish ===")
    if time.monotonic() < deadline:
        # Very high beta LSE
        a_fin = run_lbfgs(best_a, lse_obj, (25000,), maxiter=3000, maxfun=75000)
        sc = compute_score(a_fin)
        report(sc, a_fin, "final LSE")

        # Subgradient final
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_iters = int(min(5000, remaining * 200))
            sc, a_pol = subgradient_linesearch(best_a, max_iter=max(200, n_iters))
            report(sc, a_pol, "final subgrad")

    # Final output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
