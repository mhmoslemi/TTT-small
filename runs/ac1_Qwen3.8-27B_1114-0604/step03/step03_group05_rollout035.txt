```python
def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_corr_p_x(p, x):
    """
    Compute cross-correlation (p star x)_j = sum_i p[i+j] * x[i] for j=0..n-1.
    p has length 2n-1, x has length n.
    """
    n = len(x)
    L = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, L)
    X = np.fft.rfft(x, L)
    result = np.fft.irfft(P * np.conj(X), L)[:n]
    return result


def lse_value_grad(x, beta):
    """
    Compute LSE-smoothed max autocorrelation and its gradient.
    x is on the simplex (sum=1, x>=0).
    Returns (lse_value, gradient_of_length_n).
    """
    n = len(x)
    c = fast_conv(x)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    lse_val = (c_max + np.log(exp_sum)) / beta
    grad = (2.0 / beta) * cross_corr_p_x(p, x)
    return lse_val, grad


def exact_subgradient(x, k_star=None):
    """
    Compute exact subgradient of max_k (x*x)_k at x.
    Returns (M, k_star, gradient).
    """
    n = len(x)
    c = fast_conv(x)
    M = float(c.max())
    if k_star is None:
        k_star = int(np.argmax(c))
    grad = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        grad[j_start:j_end + 1] = 2.0 * x[j_start:j_end + 1][::-1]
    return M, k_star, grad


def proj_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_score_simplex(x):
    """Score for x on the simplex: 2n * max(x*x)_k."""
    n = len(x)
    c = fast_conv(x)
    return 2.0 * n * float(np.max(c))


def fista_lse(x0, beta, max_iter, verbose=False):
    """
    FISTA on LSE-smoothed objective over the simplex.
    Returns optimized x (on simplex).
    """
    n = len(x0)
    x = x0.copy()
    y = x0.copy()
    t_mom = 1.0
    # Step size: use 1/L where L is Lipschitz constant estimate
    # L ~ 4/beta for the LSE gradient (conservative)
    eta = 8.0 / beta
    best_val = np.inf
    best_x = x.copy()
    func_prev = None
    for it in range(max_iter):
        lse_val, grad = lse_value_grad(y, beta)
        if lse_val < best_val:
            best_val = lse_val
            best_x = y.copy()
        # Tentative step
        x_new = proj_simplex(y - eta * grad)
        # Check for decrease; if not, reduce step
        lse_val_new, _ = lse_value_grad(x_new, beta)
        if lse_val_new > lse_val + 1e-10:
            eta = eta * 0.5
            x_new = proj_simplex(y - eta * grad)
        else:
            eta = min(eta * 1.05, 20.0 / beta)
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        y = np.clip(y, 0.0, None)
        y_sum = y.sum()
        if y_sum > 0:
            y = y / y_sum
        else:
            y = np.ones(n) / n
        x = x_new
        t_mom = t_new
        if verbose and it % 1000 == 0:
            print(f"    FISTA iter {it}: LSE={lse_val:.8f}, eta={eta:.6f}")
    return best_x, best_val


def fista_lse_adaptive(x0, beta, max_iter, verbose=False):
    """
    FISTA with adaptive step size and periodic restart for momentum.
    """
    n = len(x0)
    x = x0.copy()
    best_x = x.copy()
    best_val = np.inf
    best_exact = compute_score_simplex(x)
    eta = 8.0 / beta
    restart_period = 500
    for epoch in range(max_iter // restart_period + 1):
        x = best_x.copy()
        y = x.copy()
        t_mom = 1.0
        for it in range(restart_period):
            lse_val, grad = lse_value_grad(y, beta)
            if lse_val < best_val:
                best_val = lse_val
                best_exampt = y.copy()
            x_new = proj_simplex(y - eta * grad)
            lse_val_new, _ = lse_value_grad(x_new, beta)
            if lse_val_new > lse_val + eta * 0.1 * float(grad @ grad):
                eta = max(eta * 0.5, 0.1 / beta)
                x_new = proj_simplex(y - eta * grad)
            else:
                eta = min(eta * 1.1, 30.0 / beta)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
            y = np.maximum(y, 0.0)
            s = y.sum()
            if s > 0:
                y = y / s
            else:
                y = np.ones(n) / n
            x = x_new
            t_mom = t_new
            # Check exact score periodically
            if it % 100 == 0:
                sc = compute_score_simplex(best_x if lse_val < best_val else y)
                if sc < best_exact:
                    best_exact = sc
                    best_x = (y if lse_val < best_val else y).copy()
        # Epoch restart: update best
        sc = compute_score_simplex(x)
        if sc < best_exact:
            best_exact = sc
            best_x = x.copy()
        if verbose:
            print(f"    Epoch {epoch}: best_exact={best_exact:.8f}")
    return best_x


def peak_shave_simplex(x, max_iter, verbose=False):
    """
    Exact subgradient descent on the convex simplex problem.
    Minimizes max_k (x*x)_k over the simplex.
    Uses Polyak-type step size.
    """
    n = len(x)
    x = x.copy()
    best_x = x.copy()
    best_M = compute_score_simplex(x) / (2.0 * n)  # just max_k (x*x)_k
    M_lb = 1.0 / (2 * n - 1)  # theoretical lower bound
    for it in range(max_iter):
        c = fast_conv(x)
        M = float(c.max())
        if M < best_M:
            best_M = M
            best_x = x.copy()
        k_star = int(np.argmax(c))
        grad_raw = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            grad_raw[j_start:j_end + 1] = 2.0 * x[j_start:j_end + 1][::-1]
        # Project gradient onto the tangent space of the simplex
        mean_grad = grad_raw.mean()
        grad = grad_raw - mean_grad
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        # Polyak step
        step = (M - M_lb) / (gn * gn + 1e-20)
        step = min(step, 0.5)
        x_new = proj_simplex(x - step * grad)
        x = x_new
        if verbose and it % 100 == 0:
            print(f"    Shave iter {it}: M={M:.10f}, step={step:.6f}")
    return best_x, best_M


def lbfgs_lse_simplex(x0, beta, maxiter=2000):
    """
    L-BFGS on LSE-smoothed objective with simplex constraint.
    Uses SciPy minimize with bounds and a penalty for the sum constraint.
    """
    n = len(x0)

    def obj_and_grad(x_raw):
        x = np.maximum(x_raw, 0.0)
        s = x.sum()
        if s < 1e-12:
            return 1e10, np.zeros(n)
        x = x / s
        lse_val, grad = lse_value_grad(x, beta)
        # Chain rule for the normalization x = max(x_raw,0)/sum(max(x_raw,0))
        # For simplicity, use implicit gradient (works well in practice)
        return lse_val, grad / s

    bounds = [(0.0, None)] * n
    try:
        res = minimize(
            obj_and_grad, x0, method='L-BFGS-B',
            jac=True, bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': maxiter * 20}
        )
        x_opt = np.maximum(res.x, 0.0)
        s = x_opt.sum()
        if s > 0:
            x_opt = x_opt / s
        else:
            x_opt = np.ones(n) / n
        return x_opt
    except Exception:
        return x0.copy()


def sinusoidal_init(n, rng):
    """Create a sinusoidal initialization on the simplex."""
    x = np.arange(n, dtype=np.float64) / n
    freq = 1.0 + 3.0 * rng.random()
    phase = 2.0 * np.pi * rng.random()
    amp = 0.3 + 0.4 * rng.random()
    v = 1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-10:
        v = np.ones(n)
        s = n
    return v / s


def gaussian_init(n, rng):
    """Create a Gaussian-shaped initialization on the simplex."""
    center = n * (0.3 + 0.4 * rng.random())
    sigma = n * (0.1 + 0.2 * rng.random())
    x = np.arange(n, dtype=np.float64)
    v = np.exp(-((x - center) ** 2) / (2.0 * sigma * sigma))
    s = v.sum()
    return v / s


def peak_shave_full(a, max_iter=3000):
    """Peak shaving on the original (unnormalized) sequence."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    M_lb = 1.0 / (2 * n - 1)
    for it in range(max_iter):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        # Project onto sum-preserving subspace (since scaling doesn't matter)
        mean_g = grad.mean()
        grad = grad - mean_g
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        sc_lb = 1.0
        step = (sc - sc_lb) / (gn * gn + 1e-20)
        step = min(step, 2.0 / (S * gn + 1e-20))
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses convex simplex reformulation with LSE-FISTA and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        sc = compute_score(parent)
        report(sc, parent, "parent")
        print(f"Parent: score={sc:.8f}, length={len(parent)}")

    # === Strategy: try multiple lengths ===
    # Use parent length and a couple of alternatives
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    # Add some alternatives
    for L in [8000, 10000, 12000]:
        if L not in lengths:
            lengths.append(L)

    # Sort by priority: parent first, then by distance from parent
    if parent is not None:
        lengths.sort(key=lambda L: abs(L - len(parent)))
    else:
        lengths.sort()

    # Time allocation: more time for parent length
    # Give 50% to first (parent), 30% to second, 20% to third (if any)
    n_lengths = min(3, len(lengths))
    time_alloc = [0.5, 0.3, 0.2][:n_lengths]
    # Scale to fit deadline
    total_frac = sum(time_alloc)
    time_alloc = [f * (deadline - t_start) / total_frac for f in time_alloc]

    for li in range(n_lengths):
        n = lengths[li]
        t_phase_start = time.monotonic()
        t_phase_end = t_phase_start + time_alloc[li]
        print(f"\n{'='*60}")
        print(f"Processing length n = {n} (time budget: {time_alloc[li]:.0f}s)")
        print(f"{'='*60}")

        # Initialize
        if li == 0 and parent is not None and len(parent) == n:
            x = parent / parent.sum()
            print(f"  Starting from parent")
        elif parent is not None and len(parent) < n:
            # Upsample parent
            x_new = np.zeros(n)
            for i, val in enumerate(parent):
                x_new[i * n // len(parent)] += val
            x = x_new / x_new.sum()
            print(f"  Upsampled parent")
        elif parent is not None:
            # Downsample parent
            x_new = np.zeros(n)
            for i, val in enumerate(parent):
                idx = min(int(i * n / len(parent)), n - 1)
                x_new[idx] += val
            x = x_new / x_new.sum()
            print(f"  Downsampled parent")
        else:
            x = sinusoidal_init(n, rng)
            print(f"  Sinusoidal init")

        sc = compute_score_simplex(x)
        report(sc, x * (np.sqrt(2 * n)), f"init n={n}")
        # Note: for simplex x, the actual sequence can be x (or c*x for any c>0)
        # Score is the same. We'll report x as the sequence.

        # === Phase 1: LSE-FISTA continuation ===
        print(f"\n  --- LSE-FISTA Continuation ---")
        beta_schedule = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
        for beta in beta_schedule:
            if time.monotonic() > t_phase_end - 60:
                break
            t0 = time.monotonic()
            iters = 3000
            # Adjust iterations based on time
            est_time_per_iter = 0.5  # ms per iteration (rough estimate)
            max_possible = int((t_phase_end - time.monotonic() - 30) * 1000 / max(est_time_per_iter, 0.1))
            iters = max(500, min(iters, max_possible))

            x, lse_val = fista_lse(x, beta, max_iter=iters)
            sc = compute_score_simplex(x)
            report(sc, x, f"FISTA beta={beta} n={n}")

            # Check time
            elapsed = time.monotonic() - t0
            print(f"    beta={beta}: score={sc:.8f}, time={elapsed:.1f}s")

        # === Phase 2: L-BFGS refinement at high beta ===
        print(f"\n  --- L-BFGS Refinement ---")
        if time.monotonic() < t_phase_end - 30:
            for beta in [5000, 15000]:
                if time.monotonic() > t_phase_end - 15:
                    break
                t0 = time.monotonic()
                x = lbfgs_lse_simplex(x, beta, maxiter=1000)
                sc = compute_score_simplex(x)
                report(sc, x, f"LBFGS beta={beta} n={n}")
                print(f"    LBFGS beta={beta}: score={sc:.8f}, time={time.monotonic()-t0:.1f}s")

        # === Phase 3: Peak shaving (exact subgradient) ===
        print(f"\n  --- Peak Shaving ---")
        if time.monotonic() < t_phase_end - 10:
            t0 = time.monotonic()
            max_shave = int((t_phase_end - time.monotonic() - 5) * 50)
            max_shave = max(500, min(max_shave, 10000))
            x_best, M_best = peak_shave_simplex(x, max_iter=max_shave)
            sc = 2.0 * n * M_best
            report(sc, x_best, f"shave n={n}")
            print(f"    Shave: score={sc:.8f}, time={time.monotonic()-t0:.1f}s")
            x = x_best

        # === Phase 4: Final LSE polish ===
        if time.monotonic() < deadline - 10:
            for beta in [20000, 50000]:
                if time.monotonic() > deadline - 10:
                    break
                t0 = time.monotonic()
                x, _ = fista_lse(x, beta, max_iter=5000)
                sc = compute_score_simplex(x)
                report(sc, x, f"polish beta={beta} n={n}")
                print(f"    Polish beta={beta}: score={sc:.8f}, time={time.monotonic()-t0:.1f}s")

    # === Final: full-sequence peak shaving on best solution ===
    print(f"\n{'='*60}")
    print(f"Final polishing on best solution (n={len(best_a)})")
    print(f"{'='*60}")
    if best_a is not None and time.monotonic() < deadline:
        a = best_a.copy()
        S = a.sum()
        if S > 0:
            a = a / S  # normalize to simplex for consistent scoring
        max_final = int((deadline - time.monotonic()) * 30)
        max_final = max(1000, min(max_final, 20000))
        sc, a_polished = peak_shave_full(a, max_iter=max_final)
        report(sc, a_polished, "final polish")

    # === Output ===
    if best_a is None:
        out = [1.0] * 100
    else:
        out = [float(max(0.0, min(1000.0, val))) for val in best_a]
        if sum(out) < 0.01:
            out = [1.0] * 100

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
