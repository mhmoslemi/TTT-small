Uses LSE continuation, power approximation, Nesterov-accelerated subgradient,
and coordinate descent. Starts from parent sequence for continuity.

import time
from scipy.optimize import minimize, minimize_scalar


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def cross_corr_weighted(w, a):
    """Compute corr[i] = sum_j a[j] * w[j+i] for i=0..n-1 using FFT.
    w has length 2n-1, a has length n."""
    n = len(a)
    w_rev = w[::-1]
    N2 = 1 << (3 * n - 2 - 1).bit_length()
    A2 = np.fft.rfft(a, N2)
    WR = np.fft.rfft(w_rev, N2)
    conv_aw = np.fft.irfft(A2 * WR, N2)[:3 * n - 2]
    return conv_aw[n - 1:2 * n - 1][::-1]


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    exp_sum = float(exp_s.sum())

    grad_L = cross_corr_weighted(p, a)
    S2 = S * S
    S3 = S2 * S

    y = x.copy()
    x_prev = x.copy()
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    best_sc = compute_score(x)
    best_x = x.copy()
    
    for t in range(1, max_iters + 1):
        c = fast_conv(y)
        S = float(y.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_x = x.copy()
        
        # Subgradient of max at k_star
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * y[j_start:j_end + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gnorm2 = float(grad @ grad)
        if gnorm2 < 1e-30:
            break
        
        # Polyak-like step size
        step = step_scale * (sc - sc_lb) / (gnorm2 + 1e-30)
        step = min(step, 2.0 / np.sqrt(t + 1.0))
        step = min(step, 0.1)
        
        x_new = y - step * grad
        x_new = np.clip(x_new, 0.0, 1000.0)
        
        # Nesterov momentum
        mom = (t - 1.0) / 0.5 * (1.0 - (2.0 * t + 1.0) / (1.0 + np.sqrt(1.0 + 4.0 * t * t)))
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mom = (t - 1.0) / t_next
        
        y = x_new + mom * (x_new - x_prev)
        y = np.clip(y, 0.0, 1000.0)
        x_prev = x_new
        x = x_new
    
    return best_sc, best_x


def plain_subgrad(a, max_iters, step_scale=1.0):
    """Plain subgradient with Polyak step and restarting."""
    n = len(a)
    x = a.astype(np.float64).copy()
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    best_sc = compute_score(x)
    best_x = x.copy()
    
    for t in range(1, max_iters + 1):
        c = fast_conv(x)
        S = float(x.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_x = x.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * x[j_start:j_end + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gnorm2 = float(grad @ grad)
        if gnorm2 < 1e-30:
            break
        
        step = step_scale * (sc - sc_lb) / (gnorm2 + 1e-30)
        step = min(step, 1.0 / (np.sqrt(t + 1.0) * 2.0))
        step = min(step, 0.05)
        
        x = x - step * grad
        x = np.clip(x, 0.0, 1000.0)
        
        # Restart every 1000 iters
        if t % 1000 == 0:
            sc_check = compute_score(x)
            if sc_check > best_sc * 1.01:
                x = best_x.copy()
    
    return best_sc, best_x


def coordinate_descent(a, n_iter, rng):
    """Random coordinate descent with 1D optimization."""
    n = len(a)
    x = a.astype(np.float64).copy()
    best_sc = compute_score(x)
    best_x = x.copy()
    
    for it in range(n_iter):
        i = rng.integers(0, n)
        old_val = x[i]
        
        # Try a range of values around current
        candidates = [
            old_val * 0.5,
            old_val * 0.75,
            old_val * 1.25,
            old_val * 1.5,
            max(0.0, old_val - 0.1 * x.mean()),
            old_val + 0.1 * x.mean(),
            0.0,
            x.mean(),
        ]
        
        best_local_val = old_val
        best_local_sc = best_sc
        
        # Use minimize_scalar for more precise 1D search
        def obj_1d(val):
            x[i] = max(0.0, min(1000.0, val))
            return compute_score(x)
        
        # Evaluate a few candidates quickly
        for cv in candidates:
            cv = max(0.0, min(1000.0, cv))
            x[i] = cv
            sc = compute_score(x)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local_val = cv
        
        x[i] = best_local_val
        if best_local_sc < best_sc:
            best_sc = best_local_sc
            best_x = x.copy()
    
    # Restore
    for i in range(n):
        pass  # not needed
    x = best_x.copy()
    return best_sc, x


def coordinate_descent_fast(a, n_iter, rng, batch_size=32):
    """Faster coordinate descent: evaluate multiple coordinates at once per conv."""
    n = len(a)
    x = a.astype(np.float64).copy()
    best_sc = compute_score(x)
    best_x = x.copy()
    
    for it in range(n_iter):
        # Pick a batch of coordinates
        idx = rng.integers(0, n, size=min(batch_size, n))
        improved = False
        for i in idx:
            old_val = x[i]
            if old_val < 1e-10:
                # Try increasing from zero
                x[i] = 0.01 * x.mean()
                sc = compute_score(x)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x.copy()
                    improved = True
                else:
                    x[i] = 0.0
                continue
            
            # Try multiplicative perturbations
            for factor in [0.7, 0.85, 1.15, 1.3]:
                x[i] = old_val * factor
                sc = compute_score(x)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x.copy()
                    improved = True
                    break
            x[i] = old_val
        
        if not improved and it > 100:
            break
    
    return best_sc, best_x


def run_lbfgs_score(args, a0, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B using the provided scorer function."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            args['fun'], a0, args=args.get('args', None),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.success


def lse_wrapper(a, beta):
    return lse_score_grad(a, beta)


def power_wrapper(a, p_power):
    return power_score_grad(a, p_power)


def build_inits_from_parent(parent, rng, n):
    """Generate diverse initializations based on parent."""
    inits = []
    if parent is not None:
        # Try to resize parent to n
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            # Interpolate to new length
            from numpy import interp
            xp = np.linspace(0, 1, len(parent))
            fp = np.linspace(0, 1, n)
            inits.append(np.interp(fp, xp, parent))
    
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    
    # Ramp
    inits.append(np.clip(0.02 + 0.98 * x, 0.01, 1000.0))
    inits.append(np.clip(0.02 + 0.98 * (1 - x), 0.01, 1000.0))
    
    # Sine windows
    inits.append(np.clip(0.3 + 0.5 * np.sin(np.pi * x), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.3 * np.sin(np.pi * 2 * x), 0.01, 1000.0))
    
    # Random smooth
    a_rand = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0)
    ksize = max(1, n // 100)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(a_rand, kernel, mode='same'))
    
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization: LSE continuation -> power approx -> Nesterov subgrad -> coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                print(f"Loaded parent: n={p.size}, score={compute_score(parent):.8f}")
    except Exception as e:
        print(f"Parent load error: {e}")

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.1f}s)")
            return True
        return False

    # Determine primary length
    n_primary = 6962
    if parent is not None:
        n_primary = len(parent)

    print(f"Primary length: {n_primary}")
    print(f"Time budget: {total_budget}s")

    # ============================================================
    # Phase 1: LSE continuation from parent (35% time)
    # ============================================================
    phase1_end = t_start + 0.30 * total_budget
    print(f"\n=== Phase 1: LSE Continuation (until t={phase1_end - t_start:.0f}s) ===")

    if parent is not None:
        a_cur = parent.astype(np.float64).copy()
        sc_cur = compute_score(a_cur)
        report(sc_cur, a_cur, "parent initial")
    else:
        a_cur = np.linspace(0.02, 1.0, n_primary)
        report(compute_score(a_cur), a_cur, "ramp init")

    beta_schedule = [20, 50, 100, 200, 500, 1000, 2000, 4000, 8000]

    for beta in beta_schedule:
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        remaining = phase1_end - time.monotonic()
        est_per_eval = 0.002 * n_primary
        maxit = max(200, min(5000, int(remaining / (est_per_eval * 20))))
        
        args = {'fun': lse_wrapper, 'args': (beta,)}
        sc, a_opt, success = run_lbfgs_score(args, a_cur, maxiter=maxit, maxfun=maxit * 30)
        elapsed = time.monotonic() - t0
        print(f"  beta={beta}: score={sc:.8f}, iters={maxit}, time={elapsed:.1f}s, success={success}")
        
        if sc < best_score:
            report(sc, a_opt, f"LSE beta={beta}")
            a_cur = a_opt
        else:
            a_cur = a_opt  # still use it as starting point for next beta

    # Also try from ramp initialization
    a_ramp = np.linspace(0.02, 1.0, n_primary)
    a_ramp = np.clip(a_ramp * (parent.sum() / a_ramp.sum() if parent is not None else 1.0), 0.01, 1000.0)
    for beta in [100, 500, 2000]:
        if time.monotonic() > phase1_end:
            break
        args = {'fun': lse_wrapper, 'args': (beta,)}
        sc, a_opt, _ = run_lbfgs_score(args, a_ramp, maxiter=2000, maxfun=60000)
        report(sc, a_opt, f"LSE ramp beta={beta}")

    print(f"  After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Power approximation refinement (15% time)
    # ============================================================
    phase2_end = t_start + 0.48 * total_budget
    print(f"\n=== Phase 2: Power Approximation (until t={phase2_end - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    for p_power in [4, 8, 16, 32, 64, 128, 256]:
        if time.monotonic() > phase2_end:
            break
        t0 = time.monotonic()
        remaining = phase2_end - time.monotonic()
        maxit = max(200, min(3000, int(remaining / 0.05)))
        
        args = {'fun': power_wrapper, 'args': (p_power,)}
        sc, a_opt, _ = run_lbfgs_score(args, a_cur, maxiter=maxit, maxfun=maxit * 30)
        elapsed = time.monotonic() - t0
        print(f"  p={p_power}: score={sc:.8f}, time={elapsed:.1f}s")
        
        if sc < best_score:
            report(sc, a_opt, f"Power p={p_power}")
            a_cur = a_opt
        else:
            a_cur = a_opt

    print(f"  After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Nesterov-accelerated subgradient (25% time)
    # ============================================================
    phase3_end = t_start + 0.73 * total_budget
    print(f"\n=== Phase 3: Nesterov Subgradient (until t={phase3_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    # Estimate iterations possible
    c_test = fast_conv(best_a)
    t0 = time.monotonic()
    for _ in range(10):
        j_start = max(0, int(np.argmax(c_test)) - n_best + 1)
        j_end = min(int(np.argmax(c_test)), n_best - 1)
    per_iter = (time.monotonic() - t0) / 10.0
    print(f"  Estimated per-iter time: {per_iter*1000:.2f}ms")

    n_iters_est = int((phase3_end - time.monotonic()) / max(per_iter, 0.001))
    n_iters_est = max(1000, min(200000, n_iters_est))
    
    sc, a_opt = nesterov_subgrad(best_a, max_iters=n_iters_est, step_scale=1.0)
    report(sc, a_opt, f"Nesterov ({n_iters_est} iters)")

    # Also try plain subgradient with restarts
    if time.monotonic() < phase3_end:
        remaining_iters = max(1000, int((phase3_end - time.monotonic()) / max(per_iter, 0.001)))
        sc2, a_opt2 = plain_subgrad(best_a, max_iters=remaining_iters, step_scale=0.5)
        report(sc2, a_opt2, "Plain subgrad")

    print(f"  After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Coordinate descent (15% time)
    # ============================================================
    phase4_end = t_start + 0.88 * total_budget
    print(f"\n=== Phase 4: Coordinate Descent (until t={phase4_end - t_start:.0f}s) ===")

    # Estimate iterations
    t0 = time.monotonic()
    cd_sc, cd_a = coordinate_descent_fast(best_a, 10, rng, batch_size=16)
    per_cd_iter = (time.monotonic() - t0) / 10.0
    n_cd_iters = max(100, int((phase4_end - time.monotonic()) / max(per_cd_iter, 0.001)))
    print(f"  CD per-batch time: {per_cd_iter*1000:.1f}ms, est iters: {n_cd_iters}")

    sc, a_opt = coordinate_descent_fast(best_a, n_cd_iters, rng, batch_size=32)
    report(sc, a_opt, f"CoordDescent ({n_cd_iters})")

    # Targeted perturbation + re-optimize
    if time.monotonic() < phase4_end:
        c = fast_conv(best_a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n_best + 1)
        j_end = min(k_star, n_best - 1)
        
        # Reduce top contributors
        contrib = np.zeros(n_best)
        if j_end >= j_start:
            for j in range(j_start, j_end + 1):
                contrib[j] = best_a[j] * best_a[k_star - j]
        
        top_n = max(10, n_best // 200)
        top_idx = np.argsort(contrib)[::-1][:top_n]
        bot_idx = np.argsort(contrib)[:top_n * 2]
        
        a_pert = best_a.copy()
        total_red = 0
        for j in top_idx:
            amt = a_pert[j] * 0.1
            a_pert[j] -= amt
            a_pert[j] = max(0.0, a_pert[j])
            total_red += amt
        if len(bot_idx) > 0 and total_red > 0:
            per = total_red / len(bot_idx)
            a_pert[bot_idx] += per
        
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() > 0.01:
            args = {'fun': lse_wrapper, 'args': (5000,)}
            sc, a_opt, _ = run_lbfgs_score(args, a_pert, maxiter=3000, maxfun=60000)
            report(sc, a_opt, "Targeted perturb")

    print(f"  After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final polish (12% time)
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")
    
    # More Nesterov on best
    if time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        n_it = max(500, int(remaining / max(per_iter, 0.001)))
        sc, a_opt = nesterov_subgrad(best_a, max_iters=n_it, step_scale=0.5)
        report(sc, a_opt, "Final Nesterov")

    # Try a high-beta LSE one more time
    if time.monotonic() < deadline - 20:
        args = {'fun': lse_wrapper, 'args': (10000,)}
        sc, a_opt, _ = run_lbfgs_score(args, best_a, maxiter=2000, maxfun=40000)
        report(sc, a_opt, "Final LSE beta=10000")

    # Try different length if time permits
    if time.monotonic() < deadline - 120:
        for n_try in [4096, 8192]:
            if n_try == n_primary:
                continue
            if time.monotonic() > deadline - 120:
                break
            print(f"  Trying n={n_try}...")
            a_init = np.linspace(0.02, 1.0, n_try)
            for beta in [100, 500, 2000]:
                if time.monotonic() > deadline - 60:
                    break
                args = {'fun': lse_wrapper, 'args': (beta,)}
                sc, a_opt, _ = run_lbfgs_score(args, a_init, maxiter=2000, maxfun=40000)
                report(sc, a_opt, f"Alt len n={n_try} beta={beta}")
                a_init = a_opt

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    # Final verification
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    
    return out
