"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses multi-phase strategy: Lp-smoothed Adam pre-optimization, LSE Adam refinement,
basin hopping with structured perturbations, and final subgradient polish.
Key improvements: Adam optimizer (faster per-iteration), power-sum objectives for
smoother landscape, multi-scale warm starts, and adaptive basin hopping.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def grad_conv_max_proxy(a, weights):
    """
    Compute gradient of sum_k weights[k] * (a*a)[k] w.r.t. a.
    weights has length 2n-1. Returns gradient of length n.
    """
    n = len(a)
    n2 = 2 * n - 1
    M2 = 1 << (n2 - 1).bit_length()
    rev_a = a[::-1]
    RA = np.fft.rfft(rev_a, M2)
    W = np.fft.rfft(weights, M2)
    conv_res = np.fft.irfft(RA * W, M2)[:n2]
    return 2.0 * conv_res[n-1:2*n-1]

def lp_objective_grad(a, p_power):
    """
    Objective: (sum c_k^p)^(1/p) with p > 1. Gradient computed via chain rule.
    Returns (objective_value, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_clipped = np.maximum(c, 1e-30)
    cp = c_clipped ** p_power
    total = cp.sum()
    # L_p = total^(1/p)
    Lp = total ** (1.0 / p_power)
    # d Lp / d c_k = (1/p) * total^(1/p - 1) * p * c_k^(p-1) = total^(1/p-1) * c_k^(p-1)
    alpha = total ** (1.0 / p_power - 1.0)
    weights = alpha * c_clipped ** (p_power - 1.0)
    grad_Lp = grad_conv_max_proxy(a, weights)
    # Full gradient: d/d a [2n * Lp / S^2]
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * Lp / S2
    grad = (2.0 * n) * (grad_Lp / S2 - 2.0 * Lp / S3)
    return score, grad

def lse_objective_grad(a, beta):
    """
    LSE-smoothed max objective. Returns (score, gradient).
    score = 2n * LSE(beta*(c - c_max)) / S^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clamp for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    Z = exp_scaled.sum()
    L = c_max + np.log(Z) / beta
    p = exp_scaled / Z
    grad_L = grad_conv_max_proxy(a, p)
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad

def adam_optimize(a0, beta, power, n_iters, lr, lr_decay, betas, eps,
                  max_wall_s, verbose=False, label=""):
    """
    Adam optimizer on the LSE (if power=0) or Lp (if power>1) objective.
    """
    a = a0.astype(np.float64).copy()
    n = len(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > max_wall_s:
            break
        t += 1
        # Compute objective and gradient
        if power > 1:
            sc, grad = lp_objective_grad(a, power)
        else:
            sc, grad = lse_objective_grad(a, beta)
        
        if not np.isfinite(sc):
            break
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if verbose and it % 500 == 0:
            print(f"    [{label}] it={it}, sc={sc:.8f}, best={best_sc:.8f}, t={time.monotonic()-t0:.1f}s")
        
        # Adam update
        lr_t = lr * (1.0 / (1.0 + it * lr_decay))
        m = betas[0] * m + (1 - betas[0]) * grad
        v = betas[1] * v + (1 - betas[1]) * grad * grad
        m_hat = m / (1 - betas[0] ** t)
        v_hat = v / (1 - betas[1] ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)
        # Ensure sum is reasonable
        if a.sum() < 0.01:
            a = best_a.copy()
    
    return best_sc, best_a

def lbfgs_optimize(a0, beta, maxiter=3000, maxfun=50000, use_lse=True, power=4):
    """L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    bounds = [(0.0, 1000.0)] * n
    
    if use_lse:
        def fun_and_grad(x):
            return lse_objective_grad(x, beta)
    else:
        def fun_and_grad(x):
            return lp_objective_grad(x, power)
    
    res = minimize(
        fun_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out

def basin_hop(a, rng, n_hops, max_wall_s, beta, power, base_lr, label=""):

    Basin hopping: make random perturbations, then optimize, keep the best.
    """
    n = len(a)

    t0 = time.monotonic()

        if time.monotonic() - t0 > max_wall_s:

        strategy = hop % 5
        a_pert = best_a.copy()
        
        if strategy == 0:
            # Gaussian noise
            scale = 0.05 + 0.1 * (1.0 - hop / max(n_hops, 1))
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Multi-frequency sinusoidal perturbation
            x = np.arange(n, dtype=np.float64) / n

            amp = 0.05 + 0.1 * rng.random()

            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Sparse large perturbation
            n_sparse = max(10, n // 30)
            idx = rng.integers(0, n, size=n_sparse)
            a_pert[idx] *= (1.0 + 0.5 * rng.standard_normal(n_sparse))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Block shuffle
            n_blocks = 8 + 16 * rng.integers(0, 3)

                    end_i = min(start_i + block_size, n)

                    end_j = min(start_j + block_size, n)
                    tmp = a_pert[start_i:end_i].copy()
                    a_pert[start_i:end_i] = a_pert[start_j:end_j]

            best_sc = sc
            best_a = a_opt.copy()
            print(f"    [{label}] hop={hop}: NEW BEST {sc:.8f} (t={time.monotonic()-t0:.0f}s)")
    
    return best_sc, best_a

def subgradient_polish(a, n_iters=500, max_wall_s=60.0):
    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > max_wall_s:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient at the argmax
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Polyak step size
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.1)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a

        report(sc, a_opt, f"LBFGS beta={beta}")
        best_a = a_opt if sc < best_score else best_a
    
    # Lp refinement
    for pwr in [8, 16]:
        if time.monotonic() > phase3_deadline - 30:
            break
        maxiter = min(3000, int((phase3_deadline - time.monotonic()) / 5.0))
        sc, a_opt = lbfgs_optimize(best_a, beta=0, maxiter=maxiter, maxfun=maxiter * 2,
                                   use_lse=False, power=pwr)
        report(sc, a_opt, f"LBFGS Lp={pwr}")
        if sc < best_score:
            best_a = a_opt
    
    print(f"\nAfter Phase 3: best = {best_score:.8f}")
    
    # =========================================================================
    # Phase 4: Subgradient polish and final refinement (10% of time)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Polish")
    print(f"{'='*60}")
    
    # Subgradient on exact objective
    if time.monotonic() < deadline - 30:
        sc, a_pol = subgradient_polish(best_a, n_iters=2000, max_wall_s=80.0)
        report(sc, a_pol, "subgrad")
        if sc < best_score:
            best_a = a_pol
    
    # Final LSE Adam polish
    if time.monotonic() < deadline - 20:
        sc, a_opt = adam_optimize(
            best_a, beta=5000, power=0,
            n_iters=10000, lr=base_lr * 0.1, lr_decay=1e-5,
            betas=(0.9, 0.999), eps=1e-8,
            max_wall_s=60.0, label="final_adam"
        )
        report(sc, a_opt, "final_adam")
    
    # One more subgradient
    if time.monotonic() < deadline - 10:
        sc, a_pol = subgradient_polish(best_a, n_iters=1000, max_wall_s=30.0)
        report(sc, a_pol, "subgrad_final")
        if sc < best_score:
            best_a = a_pol
    
    # =========================================================================
    # Final verification and return
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
