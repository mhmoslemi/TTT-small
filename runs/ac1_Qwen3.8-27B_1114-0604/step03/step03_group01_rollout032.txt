"""
Multi-phase optimizer for minimizing 2n*max(self-conv)/(sum)^2.
Uses LSE-continuation Adam, Nesterov subgradient, and exact 1D selective
coordinate descent. Key innovation: exact per-coordinate 1D optimization
exploiting the structure b_k(x) = base_k + 2*x*adj_k + quad_k*x^2.
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -700, 700)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # v[j] = sum_k p[k] * a[k-j]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A = np.fft.rfft(a, M2_fft)
    P_rev = np.fft.rfft(p[::-1], M2_fft)
    conv_ap = np.fft.irfft(A * P_rev, M2_fft)
    v = conv_ap[n - 1:2 * n - 1][::-1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (2.0 * v / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_score(a):
    """Exact score and subgradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(int(k_star), n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def multi_subgrad_score(a, top_k=20):
    """Averaged subgradient over top-k lags for smoother direction."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c_max = float(c.max())
    score = 2.0 * n * c_max / (S * S)
    sorted_idx = np.argsort(c)[::-1][:top_k]
    gaps = c_max - c[sorted_idx]
    weights = np.exp(-gaps / (0.05 * c_max + 1e-30))
    weights = weights / weights.sum()
    sg = np.zeros(n)
    for pos in range(len(sorted_idx)):
        k = int(sorted_idx[pos])
        w = weights[pos]
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            idx = k - np.arange(js, je + 1)
            sg[js:je + 1] += 2.0 * w * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
    return score, grad


def adam_direct(a_init, lr=0.005, n_steps=50000, beta_lse=100,
                time_limit_s=None, t_start=None, normalize=True):
    """Adam optimizer in a-space with LSE-smoothed gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad(a, beta_lse)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a / s
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def nesterov_subgrad(a_init, lr_init=0.01, n_steps=20000,
                     time_limit_s=None, t_start=None, top_k=10):
    """Nesterov-accelerated multi-subgradient descent."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        momentum = 2.0 / (t + 2.0)
        lr_t = lr_init * momentum
        if t % 50 == 0:
            score, grad = multi_subgrad_score(y, top_k=top_k)
        else:
            score, grad = subgrad_score(y)
        a_new = y - lr_t * grad
        a_new = np.clip(a_new, 0.0, 1e10)
        y = a_new + (momentum - 1.0) * (a_new - a)
        y = np.clip(y, 0.0, 1e10)
        a = a_new
        if t % 200 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def exact_coord_1d(a, j, c, S, max_iter_golden=60):
    """
    Exact 1D optimization for coordinate j.
    Given current convolution c and sum S, find optimal a[j].
    """
    n = len(a)
    x_old = a[j]
    S_prime = S - x_old  # sum of all other elements

    # Build adj vector: adj[k] = a[k-j] for valid k, else 0
    adj = np.zeros(2 * n - 1)
    adj[j:j + n] = a

    two_j = 2 * j
    # base[k] = c[k] - 2*x_old*adj[k] for k != 2j, c[k] - x_old^2 for k=2j
    base = c - 2.0 * x_old * adj
    if two_j < 2 * n - 1:
        base[two_j] = c[two_j] - x_old * x_old

    # f(x) = max(base + 2*x*adj + (x^2 - 2*x*adj[j]) * (k==2j)) / (S_prime + x)^2
    # For k != 2j: base[k] + 2*x*adj[k]
    # For k = 2j: base[2j] + x^2

    def f_of_x(x):
        vals = base + 2.0 * x * adj
        if two_j < 2 * n - 1:
            vals[two_j] = base[two_j] + x * x
        denom = S_prime + x
        if denom < 1e-15:
            return 1e10
        return float(vals.max()) / (denom * denom)

    # Golden section search on [0, x_max]
    # Reasonable upper bound: x_old * 3 or some cap
    x_lo = 0.0
    x_hi = max(x_old * 3.0, 0.1)
    x_hi = min(x_hi, 1000.0)

    # Ensure the current point is in the interval
    if x_old > x_hi:
        x_hi = x_old * 1.5
    if x_old < x_lo:
        x_lo = 0.0

    # Golden ratio
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # ~0.618

    c1 = x_hi - gr * (x_hi - x_lo)
    c2 = x_lo + gr * (x_hi - x_lo)
    f1 = f_of_x(c1)
    f2 = f_of_x(c2)

    for _ in range(max_iter_golden):
        if f1 < f2:
            x_hi = c2
            c2 = c1
            f2 = f1
            c1 = x_hi - gr * (x_hi - x_lo)
            f1 = f_of_x(c1)
        else:
            x_lo = c1
            c1 = c2
            f1 = f2
            c2 = x_lo + gr * (x_hi - x_lo)
            f2 = f_of_x(c2)

    x_opt = (x_lo + x_hi) / 2.0
    f_opt = f_of_x(x_opt)

    # Also check x_old and boundaries
    f_old = f_of_x(x_old)
    f_0 = f_of_x(0.0)
    if f_opt > f_old:
        x_opt = x_old
    if f_0 < f_of_x(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def selective_exact_coord_descent(a, top_k=50, n_sweeps=100,
                                  time_limit_s=None, t_start=None):
    """
    Selective coordinate descent with exact 1D optimization.
    Cycle through top-K coordinates by subgradient magnitude.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf

    no_improve_count = 0
    patience = 15

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        score = 2.0 * n * float(c.max()) / (S * S)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count >= patience:
            break

        # Compute subgradient for coordinate selection
        _, grad = subgrad_score(a)
        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break
            x_new = exact_coord_1d(a, j, c, S)
            if abs(x_new - a[j]) > 1e-12:
                a[j] = x_new
        # After sweep, recompute
        if sweep % 5 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            no_improve_count = 0

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    return best_score, best_a

def peak_equalization(a, n_rounds=20, time_limit_s=None, t_start=None):
    """
    Explicitly try to equalize the top convolution peaks by
    adjusting contributing coordinates.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        # Find top peaks
        threshold = c_max * 0.995
        active_lags = np.where(c >= threshold)[0]

        for k in active_lags:
            if time.monotonic() > deadline:
                break
            k = int(k)
            # Find contributing coordinates: j where a[j]*a[k-j] is significant
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je < js:
                continue
            products = a[js:je + 1] * a[k - np.arange(js, je + 1)]
            # Reduce the largest products
            top_prod = np.argsort(products)[::-1][:5]
            for tp in top_prod:
                j = js + int(tp)
                # Try reducing a[j] slightly
                for frac in [0.7, 0.8, 0.9]:
                    a_test = a.copy()
                    a_test[j] *= frac
                    sc = compute_score(a_test)
                    if sc < best_score:
                        # Accept and do 1D opt
                        a = a_test
                        c = fast_conv(a)
                        S = float(a.sum())
                        x_new = exact_coord_1d(a, j, c, S)
                        a[j] = x_new
                        best_score = sc
                        best_a = a.copy()
                        break

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    return best_score, best_a


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)

    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
    else:
        parent = np.ones(4096) * 0.5
        parent_score = compute_score(parent)
        print(f"No parent, using uniform n=4096, score={parent_score:.8f}")

    best_score = parent_score
    best_a = parent.copy()

    def try_update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ==================================================================
    # Phase 1: LSE-continuation Adam (25% time)
    # ==================================================================
    p1 = effective_budget * 0.25
    print(f"\n{'='*50}")
    print(f"Phase 1: LSE-continuation Adam (budget={p1:.0f}s)")
    print(f"{'='*50}")

    a_work = best_a / best_a.sum()

    for beta, lr in [(20, 0.01), (50, 0.008), (200, 0.004), (500, 0.003), (1500, 0.002)]:
        remaining = p1 - (time.monotonic() - t_start)
        if remaining < 10:
            break
        time_alloc = min(remaining * 0.35, 60.0)
        sc, a_opt = adam_direct(a_work, lr=lr, n_steps=100000, beta_lse=beta,
                               time_limit_s=time_alloc, t_start=t_start)
        try_update(sc, a_opt, f"Adam beta={beta}")
        a_work = a_opt

    # ==================================================================
    # Phase 2: Nesterov multi-subgradient (15% time)
    # ==================================================================
    p2 = effective_budget * 0.15
    print(f"\n{'='*50}")
    print(f"Phase 2: Nesterov subgradient (budget={p2:.0f}s)")
    print(f"{'='*50}")

    for tk in [5, 10, 20]:
        remaining = p2 - (time.monotonic() - t_start)
        if remaining < 10:
            break
        time_alloc = remaining * 0.3
        sc, a_opt = nesterov_subgrad(best_a, lr_init=0.005, n_steps=50000,
                                     time_limit_s=time_alloc, t_start=t_start,
                                     top_k=tk)
        try_update(sc, a_opt, f"Nesterov tk={tk}")

    # ==================================================================
    # Phase 3: Exact selective coordinate descent (40% time)
    # ==================================================================
    p3 = effective_budget * 0.40
    print(f"\n{'='*50}")
    print(f"Phase 3: Exact selective coord descent (budget={p3:.0f}s)")
    print(f"{'='*50}")

    # Progressive top_k: start wide, then narrow
    top_k_sequence = [100, 60, 40, 30, 20]
    for tk in top_k_sequence:
        remaining = p3 - (time.monotonic() - t_start)
        if remaining < 20:
            break
        time_alloc = min(remaining * 0.25, 80.0)
        sc, a_opt = selective_exact_coord_descent(best_a, top_k=tk, n_sweeps=200,
                                                  time_limit_s=time_alloc, t_start=t_start)
        try_update(sc, a_opt, f"SelectCoord tk={tk}")

    # ==================================================================
    # Phase 4: Peak equalization (10% time)
    # ==================================================================
    p4 = effective_budget * 0.10
    print(f"\n{'='*50}")
    print(f"Phase 4: Peak equalization (budget={p4:.0f}s)")
    print(f"{'='*50}")

    remaining = p4 - (time.monotonic() - t_start)
    if remaining > 5:
        sc, a_opt = peak_equalization(best_a, n_rounds=30,
                                     time_limit_s=remaining * 0.8, t_start=t_start)
        try_update(sc, a_opt, "Peak equalization")

    # ==================================================================
    # Phase 5: Final polish (10% time)
    # ==================================================================
    p5 = effective_budget * 0.10
    print(f"\n{'='*50}")
    print(f"Phase 5: Final polish (budget={p5:.0f}s)")
    print(f"{'='*50}")

    # L-BFGS polish
    for beta in [1000, 3000]:
        remaining = p5 - (time.monotonic() - t_start)
        if remaining < 10:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=3000, maxfun=50000)
        try_update(sc, a_opt, f"LBFGS beta={beta}")
        print(f"    L-BFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # More exact coordinate descent
    remaining = p5 - (time.monotonic() - t_start)
    if remaining > 10:
        sc, a_opt = selective_exact_coord_descent(best_a, top_k=30, n_sweeps=100,
                                                  time_limit_s=remaining * 0.7, t_start=t_start)
        try_update(sc, a_opt, "Final coord descent")

    # ==================================================================
    # Phase 6: Try different lengths (if time remains)
    # ==================================================================
    remaining_total = t_start + effective_budget - time.monotonic()
    if remaining_total > 60:
        print(f"\n{'='*50}")
        print(f"Phase 6: Different lengths (remaining={remaining_total:.0f}s)")
        print(f"{'='*50}")
        n_parent = len(best_a)
        for scale in [1.2, 0.85, 1.5]:
            new_n = int(n_parent * scale)
            if new_n < 1024 or new_n > 30000:
                continue
            time_left = t_start + effective_budget - time.monotonic()
            if time_left < 30:
                break
            a_res = resample_sequence(best_a, new_n)
            # Quick optimization
            sc, a_opt = selective_exact_coord_descent(a_res, top_k=40, n_sweeps=50,
                                                      time_limit_s=min(25.0, time_left * 0.5),
                                                      t_start=t_start)
            try_update(sc, a_opt, f"n={new_n}")

    # ==================================================================
    # Final: one more intensive coord descent if time allows
    # ==================================================================
    time_left = t_start + effective_budget - 5.0 - time.monotonic()
    if time_left > 15:
        print(f"\nFinal intensive coord descent ({time_left:.0f}s)...")
        sc, a_opt = selective_exact_coord_descent(best_a, top_k=25, n_sweeps=500,
                                                  time_limit_s=time_left, t_start=t_start)
        try_update(sc, a_opt, "Final intensive")

    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
