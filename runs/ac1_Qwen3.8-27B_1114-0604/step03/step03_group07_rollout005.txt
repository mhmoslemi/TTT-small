import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj_wrapper(a, beta):
    """Scipy-compatible objective."""
    return lse_score_grad(a, beta)


def run_lbfgs_bounded(a0, beta, maxiter=3000, time_limit_s=None):
    """Run L-BFGS-B with bounds on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_obj_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxiter * 30, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def run_lbfgs_logparam(x0, beta, maxiter=2000):
    """L-BFGS-B in log-parameterization: a_i = exp(x_i) / sum(exp(x))."""
    n = len(x0)
    x0 = x0.copy()

    def log_obj(x):
        a = np.exp(x - x.max())
        S = a.sum()
        a = a / S
        sc, grad_a = lse_score_grad(a, beta)
        # grad_a is gradient of score w.r.t. a (unnormalized)
        # We need grad w.r.t. x: a_i = exp(x_i)/S
        # d(a_i)/d(x_j) = delta_ij * a_i - a_i * a_j / S * S = a_i*(delta_ij - a_j)
        # Actually: a_i = exp(x_i)/sum(exp(x)), so da_i/dx_j = a_i*(delta_ij - a_j)
        # grad_x[j] = sum_i grad_a[i] * da_i/dx_j = sum_i grad_a[i]*a_i*(delta_ij - a_j)
        #           = grad_a[j]*a_j - a_j*sum_i grad_a[i]*a_i
        ga_dot_a = float(grad_a @ a)
        grad_x = a * (grad_a - ga_dot_a)
        return float(sc), grad_x

    try:
        res = minimize(log_obj, x0, method='L-BFGS-B', jac=True,
                       options={'ftol': 1e-18, 'gtol': 1e-15,
                                'maxiter': maxiter, 'maxfun': maxiter * 20, 'maxls': 30})
    except Exception:
        a = np.exp(x0 - x0.max())
        a = a / a.sum()
        return compute_score(a), a

    a = np.exp(res.x - res.x.max())
    a = a / a.sum()
    return compute_score(a), a


def peak_shave(a, strength=0.05, n_reduce=None):
    """Reduce elements contributing most to peak, redistribute to non-contributors."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    a_new = a.copy()
    if j_end < j_start:
        return a_new
    # Compute contribution of each a_j to c_{k_star}
    contrib = np.zeros(n)
    for j in range(j_start, j_end + 1):
        contrib[j] = a[j] * a[k_star - j]
    if n_reduce is None:
        n_reduce = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:n_reduce]
    bottom_mask = np.ones(n, dtype=bool)
    bottom_mask[top_idx] = False
    bottom_idx = np.where(bottom_mask)[0]
    if len(bottom_idx) == 0:
        return a_new
    total_red = 0.0
    for j in top_idx:
        amt = a_new[j] * strength
        a_new[j] -= amt
        a_new[j] = max(a_new[j], 0.0)
        total_red += amt
    if total_red > 0:
        per = total_red / len(bottom_idx)
        a_new[bottom_idx] += per
    return np.clip(a_new, 0.0, 1000.0)


def subgrad_polish(a, max_iters=5000):
    """Exact subgradient descent with adaptive step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient of max(c) at a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        # Polyak-like step with decay
        sc_lb = 2.0 * n / (2.0 * n - 1.0)
        step = (sc - sc_lb) / gn2
        step = min(step, 1.0 / (1.0 + 0.01 * it))
        step = min(step, 0.1)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 1e-6:
            break
    return best_sc, best_a


def build_spec_inits(n, rng, parent=None):
    """Generate diverse initializations using spectral windows and structured profiles."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else x

    if parent is not None and len(parent) == n:
        p = np.clip(parent.astype(np.float64), 0.0, 1000.0)
        if p.sum() > 1e-6:
            inits.append(p.copy())

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Hann window
    inits.append(0.5 * (1.0 - np.cos(2.0 * np.pi * xn)))

    # Blackman window
    inits.append(0.42 - 0.5 * np.cos(2.0 * np.pi * xn) + 0.08 * np.cos(4.0 * np.pi * xn))
    inits.append(np.clip(inits[-1], 0.001, 1000.0))

    # Raised cosine ( smoother transition)
    alpha = 0.3
    rc = np.ones(n) * 0.5
    m = int(alpha * n)
    if m > 0:
        for i in range(m):
            rc[i] = 0.5 * (1.0 + np.cos(np.pi * (i / m - 1.0)))
            rc[n - 1 - i] = rc[i]
    inits.append(rc)

    # Anti-palindromic ramp (strongest asymmetry)
    inits.append(np.linspace(0.01, 1.0, n))
    inits.append(np.linspace(1.0, 0.01, n))

    # Quadratic anti-palindromic
    inits.append(0.01 + 0.99 * xn ** 2)
    inits.append(0.01 + 0.99 * (1.0 - xn) ** 2)

    # Exponential decay
    inits.append(np.exp(-2.0 * xn))
    inits.append(np.exp(-2.0 * (1.0 - xn)))

    # Sinusoidal odd harmonics (anti-palindromic)
    for freq in [1, 2, 3]:
        inits.append(np.clip(0.5 + 0.4 * np.sin(np.pi * freq * xn), 0.001, 1000.0))

    # Combined: ramp + texture
    inits.append(np.clip(0.2 + 0.6 * xn + 0.1 * np.sin(2.0 * np.pi * 13 * xn), 0.001, 1000.0))
    inits.append(np.clip(0.2 + 0.6 * (1.0 - xn) + 0.1 * np.cos(2.0 * np.pi * 7 * xn), 0.001, 1000.0))

    # Random smoothed
    ksize = max(3, n // 100)
    kernel = np.ones(ksize) / ksize
    a_rand = 0.5 + 0.4 * rng.standard_normal(n)
    a_smooth = np.convolve(a_rand, kernel, mode='same')
    inits.append(np.clip(a_smooth, 0.001, 1000.0))

    # Pure random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.001, 1000.0))

    # Two-block asymmetric
    half = n // 2
    inits.append(np.concatenate([np.ones(half) * 0.7, np.ones(n - half) * 0.3]))
    inits.append(np.concatenate([np.ones(half) * 0.3, np.ones(n - half) * 0.7]))

    # Three-block
    third = n // 3
    inits.append(np.concatenate([np.ones(third) * 0.2, np.ones(n - 2 * third) * 0.9, np.ones(third) * 0.2]))

    # Frequency-flat attempt: inverse FFT of flat magnitude with random phase
    M_fft = 1 << (2 * n - 1 - 1).bit_length()
    A = np.ones(M_fft // 2 + 1) * 0.1
    A[0] = 0.5
    phase = rng.standard_normal(M_fft // 2 + 1)
    A_complex = A * np.exp(1j * phase)
    A_complex[0] = np.abs(A_complex[0])
    try:
        a_flat = np.fft.irfft(A_complex, M_fft)[:n]
        a_flat = np.clip(a_flat, 0.001, 1000.0)
        if a_flat.sum() > 1e-6:
            inits.append(a_flat)
    except Exception:
        pass

    return inits


def multi_beta_refine(a0, betas, time_per_beta=30.0):
    """Run LSE-continuation through a schedule of betas."""
    t0 = time.monotonic()
    a_cur = a0.copy()
    best_sc = compute_score(a_cur)
    best_a = a_cur.copy()
    for beta in betas:
        if time.monotonic() - t0 > len(betas) * time_per_beta:
            break
        t_beta = time.monotonic()
        remaining = time_per_beta - (t_beta - t0 if t_beta > t0 else 0)
        maxit = max(500, int(remaining * 20))
        sc, a_opt = run_lbfgs_bounded(a_cur, beta, maxiter=maxit)
        a_cur = a_opt
        if sc < best_sc:
            best_sc = sc
            best_a = a_cur.copy()
            print(f"    beta={beta}: {sc:.8f}")
    return best_sc, best_a


def log_param_refine(a0, betas, time_per_beta=20.0):
    """Run LSE-continuation in log-parameterization."""
    t0 = time.monotonic()
    a_cur = a0.copy()
    best_sc = compute_score(a_cur)
    best_a = a_cur.copy()
    for beta in betas:
        if time.monotonic() - t0 > len(betas) * time_per_beta:
            break
        x0 = np.log(a_cur + 1e-10)
        maxit = max(300, int(time_per_beta * 10))
        sc, a_opt = run_lbfgs_logparam(x0, beta, maxiter=maxit)
        a_cur = a_opt
        if sc < best_sc:
            best_sc = sc
            best_a = a_cur.copy()
            print(f"    logparam beta={beta}: {sc:.8f}")
    return best_sc, best_a


def peak_shave_cycle(a, n_cycles=5, strength=0.03):
    """Iterative peak shaving with mini LSE refinement."""
    best_sc = compute_score(a)
    best_a = a.copy()
    for cycle in range(n_cycles):
        a = peak_shave(a, strength=strength * (1.0 + 0.5 * cycle))
        sc = compute_score(a)
        # Mini refinement
        sc2, a2 = run_lbfgs_bounded(a, 2000, maxiter=500)
        if sc2 < sc:
            a, sc = a2, sc2
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"    shave cycle {cycle}: {sc:.8f}")
    return best_sc, best_a


def diverse_perturb(a, rng, method='auto'):
    """Generate a perturbed version of a using various strategies."""
    n = len(a)
    if method == 'auto':
        method = rng.choice(['gauss', 'peak', 'antisym', 'freq', 'block'])
    if method == 'gauss':
        scale = 0.02 + 0.06 * rng.random()
        return np.clip(a * (1.0 + scale * rng.standard_normal(n)), 0.0, 1000.0)
    elif method == 'peak':
        return peak_shave(a, strength=0.02 + 0.05 * rng.random())
    elif method == 'antisym':
        a_p = a.copy()
        center = n // 2
        width = max(1, n // 12)
        x = np.arange(n, dtype=float)
        d = (x - center) / width
        sign = np.where(x < center, 1.0, -1.0)
        a_p *= (1.0 + 0.1 * sign * np.exp(-d * d))
        return np.clip(a_p, 0.0, 1000.0)
    elif method == 'freq':
        M_fft = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(a, M_fft)
        freqs = np.arange(len(A), dtype=float)
        att = 1.0 - 0.15 * (freqs / len(A)) ** 2
        A *= att
        a_p = np.real(np.fft.irfft(A, M_fft))[:n]
        return np.clip(a_p, 0.0, 1000.0)
    else:  # block
        a_p = a.copy()
        block_size = max(1, n // 20)
        for _ in range(3):
            start = rng.integers(0, n - block_size)
            a_p[start:start + block_size] *= (0.7 + 0.6 * rng.random())
        return np.clip(a_p, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multi-phase optimization with aggressive LSE, peak shaving,
    log-parameterization, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0

    # Try to load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 128:
                parent = np.clip(p, 0.0, 1000.0)
                print(f"Loaded parent: n={p.size}, score={compute_score(parent):.8f}")
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = [8192]
    if parent is not None:
        lengths.insert(0, parent.size)
    if 6962 not in lengths:
        lengths.append(6962)
    lengths = sorted(set(lengths))[:3]

    print(f"Lengths to try: {lengths}")

    # Phase 1: Quick exploration with diverse inits (bounded LSE)
    phase1_end = t_start + 0.25 * total_budget
    print(f"\n=== Phase 1: Quick LSE Exploration (until {phase1_end - t_start:.0f}s) ===")

    beta_quick = [50, 200, 800, 3000]

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"\n  n={n}")
        inits = build_spec_inits(n, rng, parent)
        time_left = phase1_end - time.monotonic()
        n_inits = min(len(inits), max(3, int(time_left / 25)))

        for i in range(n_inits):
            if time.monotonic() > phase1_end:
                break
            a_cur = inits[i].copy()
            sc, a_opt = multi_beta_refine(a_cur, beta_quick, time_per_beta=6.0)
            report(sc, a_opt, f"P1 n={n} i={i}")

    print(f"\n  After Phase 1: best = {best_score:.8f}")

    # Phase 2: Aggressive LSE continuation on best
    phase2_end = t_start + 0.55 * total_budget
    print(f"\n=== Phase 2: Aggressive LSE (until {phase2_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    beta_aggressive = [500, 2000, 6000, 15000, 40000, 100000]

    a_cur = best_a.copy()
    for beta in beta_aggressive:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        maxit = max(1000, int(remaining / 2))
        sc, a_opt = run_lbfgs_bounded(a_cur, beta, maxiter=maxit)
        a_cur = a_opt
        report(sc, a_opt, f"P2 beta={beta}")

    # Peak shave between high betas
    if time.monotonic() < phase2_end:
        sc, a_shaved = peak_shave_cycle(a_cur, n_cycles=3, strength=0.02)
        report(sc, a_shaved, "P2 shave")
        a_cur = a_shaved

    print(f"\n  After Phase 2: best = {best_score:.8f}")

    # Phase 3: Log-parameterization refinement
    phase3_end = t_start + 0.70 * total_budget
    print(f"\n=== Phase 3: Log-Parameterized Refinement ===")

    if time.monotonic() < phase3_end:
        beta_log = [1000, 5000, 20000]
        a_cur = best_a.copy()
        for beta in beta_log:
            if time.monotonic() > phase3_end:
                break
            sc, a_opt = log_param_refine(a_cur, [beta], time_per_beta=15.0)
            a_cur = a_opt
            report(sc, a_opt, f"P3 log b={beta}")

    print(f"\n  After Phase 3: best = {best_score:.8f}")

    # Phase 4: Subgradient polish
    phase4_end = t_start + 0.82 * total_budget
    print(f"\n=== Phase 4: Subgradient Polish ===")

    if time.monotonic() < phase4_end:
        n_iters = max(1000, int((phase4_end - time.monotonic()) * 80))
        sc, a_opt = subgrad_polish(best_a, max_iters=n_iters)
        report(sc, a_opt, "P4 subgrad")

    print(f"\n  After Phase 4: best = {best_score:.8f}")

    # Phase 5: Diverse local search with restarts
    phase5_end = deadline
    print(f"\n=== Phase 5: Diverse Local Search ===")

    n_best = len(best_a)
    trial = 0
    stagnation_count = 0
    prev_best = best_score

    while time.monotonic() < phase5_end - 10 and trial < 40:
        trial += 1
        a_pert = diverse_perturb(best_a, rng)
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            a_pert = best_a * 0.9

        # Quick LSE refinement
        sc, a_opt = run_lbfgs_bounded(a_pert, 5000, maxiter=2000)
        improved = report(sc, a_opt, f"P5 #{trial}")

        if improved:
            stagnation_count = 0
            prev_best = best_score
        else:
            stagnation_count += 1
            # If stuck, try stronger perturbation
            if stagnation_count >= 5:
                a_pert = diverse_perturb(best_a, rng, method='gauss')
                scale = 0.1 + 0.1 * rng.random()
                a_pert = np.clip(a_pert * (1.0 + scale * rng.standard_normal(n_best)), 0.0, 1000.0)
                sc, a_opt = run_lbfgs_bounded(a_pert, 2000, maxiter=1500)
                report(sc, a_opt, f"P5 #{trial} restart")
                stagnation_count = 0

    # Final polish
    if time.monotonic() < deadline - 5:
        sc, a_opt = subgrad_polish(best_a, max_iters=3000)
        report(sc, a_opt, "final subgrad")

        # One more LSE at very high beta
        if time.monotonic() < deadline - 3:
            sc2, a_opt2 = run_lbfgs_bounded(best_a, 50000, maxiter=3000)
            report(sc2, a_opt2, "final LSE")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
