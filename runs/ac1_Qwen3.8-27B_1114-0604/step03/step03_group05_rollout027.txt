"""
Step function optimization using LP-based iterative refinement.
Key idea: At each step, solve an LP to find the "best" direction g that
keeps the convolution peak bounded, then update towards g.
Combined with LSE initialization and subgradient polishing.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lse_obj(a, beta):
    return lse_score_grad(a, beta)

def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0

def peak_shave(a, iters=3000):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - 1.0) / (gn * gn + 1e-20)
        step = min(step, 1.0 / S)
        step = max(step, 1e-6 / (gn * S))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a

def lp_iteration(a, M_override=None):
    """
    Solve the LP: max sum(g) s.t. g >= 0, (a*g)[k] <= M for all k.
    Returns the optimal g (or None if infeasible/unbounded).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    S = float(a.sum())
    c = fast_conv(a)
    M = float(c.max()) if M_override is None else M_override
    
    if M < 1e-15:
        return None
    
    # Build constraint matrix: A_cons[g] = (a*g) for each k
    # (a*g)[k] = sum_i a[i] * g[k-i] for valid i
    # Row k: A_cons[k, i] = a[i] if 0 <= k-i < n, else 0
    # i.e., A_cons[k, i] = a[i] if max(0, k-n+1) <= i <= min(k, n-1)
    
    # Build sparse matrix
    # Row k has non-zeros at columns i where max(0,k-n+1) <= i <= min(k,n-1)
    total_nnz = 0
    row_starts = []
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        cnt = i_end - i_start + 1 if i_end >= i_start else 0
        row_starts.append((k, i_start, i_end, cnt))
        total_nnz += cnt
    
    # Build COO format
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    
    idx = 0
    for k, i_start, i_end, cnt in row_starts:
        if cnt > 0:
            rows[idx:idx+cnt] = k
            cols[idx:idx+cnt] = range(i_start, i_end+1)
            vals[idx:idx+cnt] = a[i_start:i_end+1]
            idx += cnt
    
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: minimize -sum(g) => c_obj = -ones
    c_obj = -np.ones(n)
    
    # Constraints: A_sparse @ g <= M * ones
    b_rhs = np.full(2*n-1, M)
    
    try:
        res = linprog(
            c_obj,
            A_ub=A_sparse,
            b_ub=b_rhs,
            bounds=[(0, None)] * n,
            method='highs'
        )
        if res.success:
            return res.x
        else:
            return None
    except Exception as e:
        print(f"  LP error: {e}")
        return None

def lp_based_optimize(a, max_iters=200, t_base=0.1, deadline=None):
    """
    LP-based iterative optimization.
    At each step: solve LP to get direction g, then update a = (1-t)*a + t*g.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    # Normalize
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0/n
        S = 1.0
    a = a / S  # sum = 1
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = float(c.max())
        sc = 2.0 * n * M  # since sum(a) = 1
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Solve LP
        g = lp_iteration(a)
        if g is None:
            break
        
        g_sum = float(g.sum())
        if g_sum < 1e-12:
            break
        
        # Normalize g to sum = 1
        g = g / g_sum
        
        # Try different step sizes
        best_local_a = a.copy()
        best_local_sc = sc
        
        for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_trial = (1-t)*a + t*g
            a_trial = np.clip(a_trial, 0, None)
            a_trial_sum = float(a_trial.sum())
            if a_trial_sum < 1e-12:
                continue
            a_trial = a_trial / a_trial_sum
            sc_trial = compute_score(a_trial)
            if sc_trial < best_local_sc:
                best_local_sc = sc_trial
                best_local_a = a_trial.copy()
        
        if best_local_sc >= sc - 1e-15:
            # No improvement, try smaller step or break
            a_trial = (1-0.005)*a + 0.005*g
            a_trial = np.clip(a_trial, 0, None)
            a_trial_sum = float(a_trial.sum())
            if a_trial_sum > 1e-12:
                a_trial = a_trial / a_trial_sum
                sc_trial = compute_score(a_trial)
                if sc_trial < best_local_sc:
                    best_local_sc = sc_trial
                    best_local_a = a_trial.copy()
        
        a = best_local_a
        if best_sc - best_local_sc < 1e-10:
            # Converged
            break
    
    best_sc = compute_score(best_a)
    # Scale back to original magnitude (doesn't affect score)
    return best_sc, best_a

2. Compute weights w[k] = 1/C[k] (or some function that upweights small C[k] and downweights large C[k])
3. Solve a weighted least squares problem: minimize sum_k w[k] * (C[k] - C_target)^2
4. Update a

This is similar to "iteratively reweighted" methods used in robust statistics.

"""
Step function optimization: LP-guided iterative refinement with LSE continuation,
frequency-domain initialization, and aggressive peak shaving.
Key new idea: solve reduced LPs to find optimal update directions that keep
the convolution peak bounded, combined with Adam-accelerated LSE and Polyak

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(a, beta):
    """Objective wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000, deadline=None):
    """Run L-BFGS-B optimization with deadline check."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def func_with_deadline(x, *a_args):
        if deadline is not None and time.monotonic() > deadline:
            raise StopIteration
        return func(x, *a_args)

    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-17, 'gtol': 1e-13,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except (StopIteration, Exception):
        return a0


def adam_lse(a0, beta, max_iter=10000, lr=0.005, deadline=None):
    """Adam optimization for LSE objective with adaptive learning rate."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_a = a.copy()
    best_sc = compute_score(a)

    for i in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        t += 1
        sc, grad = lse_score_grad(a, beta)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = np.clip(a, 0.0, 1000.0)

        if t % 500 == 0:
            sc_check = compute_score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()

    return best_sc, best_a


def variance_score_grad(a):
    """Score and gradient for variance flattening objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c2 = float(np.sum(c * c))
    score = 2.0 * n * c2 / (S ** 4)
    L = 1 << (2 * n - 2).bit_length()
    A = np.fft.fft(a, L)
    E = np.abs(A) ** 2 * np.conj(A)
    E_ifft = np.fft.ifft(E).real
    idx = np.concatenate([[0], np.arange(L - 1, L - n, -1)])
    dc2_da = 4.0 * E_ifft[idx]
    S4 = S ** 4
    S5 = S4 * S
    grad = 2.0 * n * (dc2_da / S4 - 4.0 * c2 / S5)
    return score, grad


def variance_obj(a):
    """Objective wrapper for variance minimization."""
    return variance_score_grad(a)


def reduced_lp_direction(a, K=100, M_override=None):
    """
    Solve reduced LP: max sum(g) s.t. g >= 0, (a*g)[k] <= M for top K peaks.
    Returns direction g (sum-normalized) or None.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    c = fast_conv(a)
    M = float(c.max()) if M_override is None else M_override
    if M < 1e-15:
        return None

    # Select top K constraint positions
    K = min(K, 2 * n - 1)
    top_k = np.argsort(c)[::-1][:K]

    # Build sparse constraint matrix (K rows, n cols)
    row_list = []
    col_list = []
    val_list = []
    for row_idx, k_val in enumerate(top_k):
        k_int = int(k_val)
        i_start = max(0, k_int - n + 1)
        i_end = min(k_int, n - 1)
        if i_end >= i_start:
            for i in range(i_start, i_end + 1):
                row_list.append(row_idx)
                col_list.append(i)
                val_list.append(a[i])

    if not row_list:
        return None

    A_sparse = csr_matrix(
        (np.array(val_list), (np.array(row_list, dtype=np.int32), np.array(col_list, dtype=np.int32))),
        shape=(K, n)
    )

    c_obj = -np.ones(n)
    b_rhs = np.full(K, M)

    try:
        res = linprog(
            c_obj, A_ub=A_sparse, b_ub=b_rhs,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': 30}
        )
        if res.success and res.x is not None:
            g = np.clip(res.x, 0, None)
            g_sum = float(g.sum())
            if g_sum > 1e-12:
                return g / g_sum
    except Exception:
        pass
    return None


def lp_refinement(a, max_iters=100, K=80, deadline=None):
    """
    LP-guided iterative refinement.
    At each step: solve reduced LP for direction, line search for step size.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S  # normalize to sum=1

    best_a = a.copy()
    best_sc = compute_score(a)
    stagnant = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 15:
                break

        # Solve reduced LP
        g = reduced_lp_direction(a, K=K)
        if g is None:
            break

        # Ensure g doesn't violate all constraints badly
        ag = np.convolve(a, g, mode='full')
        ag_max = float(ag.max())
        a_max = float(fast_conv(a).max())
        if ag_max > 2.0 * a_max:
            # g is too aggressive, scale it back
            g = 0.5 * a / float(a.sum()) + 0.5 * g
            g = np.clip(g, 0, None)
            g = g / float(g.sum())

        # Line search over step sizes
        best_local_a = a.copy()
        best_local_sc = sc
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.clip(a_trial, 0, None)
            st = float(a_trial.sum())
            if st < 1e-12:
                continue
            a_trial = a_trial / st
            sc_trial = compute_score(a_trial)
            if sc_trial < best_local_sc:
                best_local_sc = sc_trial
                best_local_a = a_trial.copy()

        if best_local_sc >= sc * (1 - 1e-10):
            # Tiny step
            a_trial = (1 - 0.01) * a + 0.01 * g
            a_trial = np.clip(a_trial, 0, None)
            st = float(a_trial.sum())
            if st > 1e-12:
                a_trial = a_trial / st
                sc_trial = compute_score(a_trial)
                if sc_trial < best_local_sc:
                    best_local_sc = sc_trial
                    best_local_a = a_trial.copy()

        a = best_local_a
        # Increase K as we progress (tighter constraints)
        if it % 20 == 0 and it > 0:
            K = min(K + 20, 300)

    best_sc = compute_score(best_a)
    return best_sc, best_a


def peak_shave(a, iters=5000, deadline=None):
    """Polyak subgradient descent on the max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    S = float(a.sum())
    a = a / S

    for it in range(iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size with target score
        target = 1.5027
        step = (sc - target) / (gn * gn + 1e-20)
        step = min(step, 0.5 / (S + 1e-12))
        step = max(step, 1e-7 / (gn * S + 1e-20))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

        # Periodic renormalization
        if it % 100 == 0:
            S = float(a.sum())
            if S > 1e-12:
                a = a / S

    best_sc = compute_score(best_a)
    return best_sc, best_a

def freq_init(n, rng):
    """
    Initialize via flat spectrum: find a >= 0 whose power spectrum is close to flat.
    Uses iterative clipping in frequency domain.
    """
    M_fft = 1 << (2 * n - 1).bit_length()
    # Start with random phase, flat magnitude
    A = np.zeros(M_fft, dtype=np.complex128)
    A[0] = np.sqrt(n)  # DC component
    for k in range(1, M_fft // 2 + 1):
        A[k] = np.exp(1j * 2 * np.pi * rng.random())
        A[M_fft - k] = np.conj(A[k])
    a = np.fft.ifft(A).real[:n]
    a = np.clip(a, 0, None)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    return a / a.sum()

    phase1_end = t_start + 0.45 * total_budget
    print("\n=== Phase 1: LSE Continuation ===")
    a_cur = best_a.copy()
    beta_schedule = [10, 20, 40, 80, 160, 320, 640, 1280, 2560, 5120]

    for beta in beta_schedule:
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        # Two rounds at each beta
        for r in range(2):
            if time.monotonic() > phase1_end:
                break
            maxit = 800
            a_cur = run_lbfgs(a_cur, lse_obj, (beta,), maxiter=maxit, maxfun=maxit * 30, deadline=phase1_end)
            sc = compute_score(a_cur)
            report(sc, a_cur, f"LSE b={beta} r{r}")
        # Also try Adam at this beta
        if time.monotonic() < phase1_end - 30:
            sc_adam, a_adam = adam_lse(a_cur, beta, max_iter=3000, lr=0.003, deadline=phase1_end)
            report(sc_adam, a_adam, f"Adam b={beta}")
            if sc_adam < compute_score(a_cur):
                a_cur = a_adam
        print(f"  beta={beta}: score={compute_score(a_cur):.8f}, {time.monotonic()-t0:.1f}s")

    # === Phase 2: High-beta LSE refinement ===
    phase2_end = t_start + 0.60 * total_budget
    print("\n=== Phase 2: High-Beta Refinement ===")
    a_cur = best_a.copy()
    for beta in [5000, 10000, 20000]:
        if time.monotonic() > phase2_end:
            break
        t0 = time.monotonic()
        a_cur = run_lbfgs(a_cur, lse_obj, (beta,), maxiter=1500, maxfun=30000, deadline=phase2_end)
        sc = compute_score(a_cur)
        report(sc, a_cur, f"refine b={beta}")
        # Adam refinement
        if time.monotonic() < phase2_end - 20:
            sc_a, a_a = adam_lse(a_cur, beta, max_iter=5000, lr=0.001, deadline=phase2_end)
            report(sc_a, a_a, f"Adam refine b={beta}")
            if sc_a < compute_score(a_cur):
                a_cur = a_a
        print(f"  beta={beta}: {compute_score(a_cur):.8f}, {time.monotonic()-t0:.1f}s")

    # === Phase 3: LP-guided refinement (KEY NOVEL PHASE) ===
    phase3_end = t_start + 0.78 * total_budget
    print("\n=== Phase 3: LP-Guided Refinement ===")
    if time.monotonic() < phase3_end:
        t0 = time.monotonic()
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S
        sc_lp, a_lp = lp_refinement(a_norm, max_iters=80, K=60, deadline=phase3_end)
        report(sc_lp, a_lp, "LP refine")
        print(f"  LP: {sc_lp:.8f}, {time.monotonic()-t0:.1f}s")

        # Second round of LP with tighter K
        if time.monotonic() < phase3_end - 30:
            t0 = time.monotonic()
            sc_lp2, a_lp2 = lp_refinement(a_lp, max_iters=50, K=150, deadline=phase3_end)
            report(sc_lp2, a_lp2, "LP refine 2")
            print(f"  LP2: {sc_lp2:.8f}, {time.monotonic()-t0:.1f}s")

    # === Phase 4: Aggressive peak shaving ===
    phase4_end = t_start + 0.88 * total_budget
    print("\n=== Phase 4: Peak Shaving ===")
    if time.monotonic() < phase4_end:
        t0 = time.monotonic()
        sc_ps, a_ps = peak_shave(best_a, iters=3000, deadline=phase4_end)
        report(sc_ps, a_ps, "peak shave")
        print(f"  Peak shave: {sc_ps:.8f}, {time.monotonic()-t0:.1f}s")

    # === Phase 5: Multi-peak shaving ===
    phase5_end = t_start + 0.93 * total_budget
    print("\n=== Phase 5: Multi-Peak Shave ===")
    if time.monotonic() < phase5_end:
        t0 = time.monotonic()
        sc_mp, a_mp = multi_peak_shave(best_a, iters=1500, deadline=phase5_end)
        report(sc_mp, a_mp, "multi-peak")
        print(f"  Multi-peak: {sc_mp:.8f}, {time.monotonic()-t0:.1f}s")

    # === Phase 6: Perturbation + quick re-optimization ===
    phase6_end = deadline
    print("\n=== Phase 6: Restart Perturbations ===")
    n_best = len(best_a)
    restart_count = 0
    while time.monotonic() < phase6_end and restart_count < 10:
        t0 = time.monotonic()
        strategy = restart_count % 4
        if strategy == 0:
            scale = 0.01 + 0.02 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted: reduce peak contributors
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            contrib = a_pert[j_start:j_end+1][::-1]
            top_n = max(10, n_best // 50)
            top_idx_in = np.argsort(contrib)[::-1][:top_n]
            reduce_amt = 0.03 * contrib
            a_pert[j_start + top_idx_in] -= reduce_amt
            a_pert = np.clip(a_pert, 0, 1000)
            a_pert += (best_a.sum() - a_pert.sum()) / n_best
        elif strategy == 2:
            # Frequency domain perturbation
            L = 1 << (2 * n_best - 2).bit_length()
            A = np.fft.fft(best_a, L)
            A_mag = np.abs(A) * (1.0 + 0.015 * rng.standard_normal(L))
            A_mag = np.clip(A_mag, 0, None)
            A_new = A_mag * np.exp(1j * np.angle(A))
            a_pert = np.fft.ifft(A_new).real[:n_best]
            a_pert = np.clip(a_pert, 0, 1000)
        else:
            # Swap high and low elements
            a_pert = best_a.copy()
            med = np.median(a_pert)
            hi = a_pert > med
            lo = ~hi
            if hi.sum() > 0 and lo.sum() > 0:
                transfer = 0.02 * a_pert[hi].mean()
                a_pert[hi] -= transfer
                a_pert[lo] += transfer * hi.sum() / lo.sum()
            a_pert = np.clip(a_pert, 0, 1000)

        # Quick optimization from perturbed start
        a_opt = run_lbfgs(a_pert, lse_obj, (5000,), maxiter=400, maxfun=8000, deadline=phase6_end)
        sc = compute_score(a_opt)
        report(sc, a_opt, f"restart {restart_count}")
        restart_count += 1

    # === Phase 7: Final polish ===
    print("\n=== Phase 7: Final Polish ===")
    if time.monotonic() < deadline - 5:
        # High beta LSE
        a_final = run_lbfgs(best_a, lse_obj, (15000,), maxiter=800, maxfun=15000, deadline=deadline - 5)
        report(compute_score(a_final), a_final, "final LSE")
        # Final subgradient
        sc_f, a_f = peak_shave(best_a, iters=1000, deadline=deadline - 3)
        report(sc_f, a_f, "final subgrad")
        # LP on best
        if time.monotonic() < deadline - 3:
            a_n = best_a / float(best_a.sum())
            sc_l, a_l = lp_refinement(a_n, max_iters=30, K=200, deadline=deadline - 2)
            report(sc_l, a_l, "final LP")

    # === Output ===
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
