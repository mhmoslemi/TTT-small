```python
"""
Step function optimization: minimize 2n*max(conv(a,a))/(sum(a))^2 for a >= 0.

Algorithm:
  1. Start from parent solution
  2. Adaptive LSE-smoothed L-BFGS with continuation in beta
  3. Active-set gradient descent with backtracking
  4. Direct flattening heuristic (reduce top contributors to max conv)
  5. Final polish with exact subgradient + structured perturbations
"""

import numpy as np
from scipy.optimize import minimize
import time


def fft_self_conv(a):
    """Linear self-convolution a*a via FFT. Length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def grad_conv_weighted(a, w):
    """
    Compute 2 * result[m] where result[m] = sum_j a[j]*w[j+m].
    This is the gradient of sum_k w[k]*(a*a)[k] with respect to a.
    w has length 2n-1.
    """
    n = len(a)
    w_rev = w[::-1]
    M_fft = 1 << (3 * n - 3).bit_length()
    A = np.fft.rfft(a, M_fft)
    WR = np.fft.rfft(w_rev, M_fft)
    conv = np.fft.irfft(A * WR, M_fft)
    # result[m] = conv[2n-2-m] for m=0..n-1
    # So result = conv[n-1 : 2n-1][::-1]
    res = conv[n-1 : 2*n-1][::-1]
    return 2.0 * res


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fft_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta, n):
    """LSE-smoothed score and gradient."""
    c = fft_self_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = c.max()
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    # Gradient of L w.r.t. a: grad_L = 2 * cross_corr(p, a)
    # where cross_corr(p, a)[m] = sum_j a[j]*p[j+m]
    grad_L = grad_conv_weighted(x, p)
    # Total gradient
    grad = (2.0 * n) * (grad_L / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, deadline, max_iterations=3000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0
    bounds = [(0.0, 1000.0)] * n
    
    def fun(x):
        sc, _ = lse_obj_grad(x, beta, n)
        return sc
    
    def jac(x):
        _, g = lse_obj_grad(x, beta, n)
        return g
    
    # Time-based limits
    remaining = deadline - time.monotonic()
    maxfun = min(50000, max(100, int(remaining * 150)))
    maxiter = min(max_iterations, max(50, int(remaining * 30)))
    
    options = {'ftol': 1e-14, 'gtol': 1e-12, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    
    try:
        res = minimize(fun, a0, jac=jac, method='L-BFGS-B', bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    
    return a_out


def active_set_gradient_descent(a, n_iters=2000, deadline=None, verbose=False):
    """
    Gradient descent with adaptive active-set weighting and backtracking line search.
    Uses the exact max (via high-density softmax on conv values).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fft_self_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Adaptive beta: start high (sharp), decrease if stuck
        # Use a softmax-like weighting
        delta = 0.01 * M  # threshold
        active = c >= (M - delta)
        if active.sum() == 0:
            active[np.argmax(c)] = True
        
        # Weight active lags proportional to (c - (M-delta))^2
        w = np.zeros(2*n-1)
        diff = np.maximum(c - (M - delta), 0)
        w = diff ** 2
        w_sum = w.sum()
        if w_sum < 1e-30:
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient of score
        # d(score)/d(a[m]) = (2n/S^2) * 2*sum_k w[k]*a[k-m] - (4n*M/S^3)
        # = (4n/S^2) * cross_corr(w, a)[m] - (4n*M/S^3)
        cross = grad_conv_weighted(a, w)  # This gives 2*sum_k w[k]*a[k-m]
        grad = (2.0 * n / (S * S)) * cross - (4.0 * n * M) / (S * S * S)
        
        # Backtracking line search
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Initial step size: proportional to (sc - lower_bound) / ||grad||^2
        lower_bound = 1.0  # theoretical lower bound
        step = (sc - lower_bound) / (gn * gn + 1e-30)
        step = min(step, 0.1)
        
        # Try the step with backtracking
        improved = False
        for backtrack in range(10):
            a_trial = a - step * grad
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            sc_trial = compute_score(a_trial)
            if sc_trial < sc:
                a = a_trial
                improved = True
                break
            step *= 0.5
        
        if not improved and verbose and it % 100 == 0:
            print(f"    AS-GD iter {it}: sc={sc:.8f}, step exhausted")
    
    return best_sc, best_a


def direct_flatten(a, n_rounds=50, deadline=None):
    """
    Direct flattening: identify pairs that contribute most to the max conv value
    and reduce them, redistributing to less-contributing positions.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fft_self_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find the max lag
        k_star = int(np.argmax(c))
        
        # The contribution to conv[k_star] from a[i] is a[i]*a[k_star-i]
        # (for valid i). We want to reduce the products at the peak.
        i_start = max(0, k_star - n + 1)
        i_end = min(k_star, n - 1)
        
        if i_end < i_start:
            break
        
        # Compute contributions
        i_range = np.arange(i_start, i_end + 1)
        j_range = k_star - i_range
        contribs = a[i_range] * a[j_range]
        
        # Reduce the top contributors
        top_frac = 0.3  # Reduce top 30% of contributors
        n_top = max(1, int(top_frac * len(i_range)))
        top_idx_local = np.argsort(contribs)[::-1][:n_top]
        
        # Amount to reduce: proportional to the excess over the mean
        mean_contrib = contribs.mean()
        reductions = np.zeros(len(i_range))
        for idx in top_idx_local:
            excess = max(0, contribs[idx] - mean_contrib)
            reductions[idx] = excess * 0.1  # 10% of the excess
        
        # Apply reductions: reduce a[i] where i is in the top
        a_trial = a.copy()
        for idx in top_idx_local:
            i_val = i_range[idx]
            j_val = j_range[idx]
            # Reduce both a[i] and a[j] proportionally
            reduce_amt = reductions[idx]
            # The product a[i]*a[j] should decrease by ~reduce_amt
            # If we reduce a[i] by dt, the product decreases by dt*a[j]
            dt = reduce_amt / max(a[j_val], 1e-10)
            dt = min(dt, a[i_val] * 0.1)  # Don't reduce by more than 10%
            a_trial[i_val] -= dt
            a_trial[j_val] -= dt
        
        a_trial = np.clip(a_trial, 0.0, 1000.0)
        
        # Renormalize to preserve sum
        S_trial = a_trial.sum()
        if S_trial > 1e-12:
            a_trial = a_trial * (S / S_trial)
        
        sc_trial = compute_score(a_trial)
        if sc_trial < sc:
            a = a_trial
    
    return best_sc, best_a


def lse_continuation(a, deadline, betas=None):
    """
    LSE continuation: progressively increase beta from low to high.
    """
    if betas is None:
        betas = [10, 30, 100, 300, 1000, 3000, 10000]
    
    best_sc = compute_score(a)
    best_a = a.copy()
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        a_out = run_lbfgs(a, beta, deadline)
        sc = compute_score(a_out)
        if sc < best_sc:
            best_sc = sc
            best_a = a_out
        else:
            # Even if L-BFGS didn't improve the exact score, the LSE solution
            # might be in a better basin for higher beta
            a = a_out  # Continue from this point
        a = a_out
    
    return best_sc, best_a


def structured_perturb(a, rng, n_trials=20, deadline=None):
    """Try structured perturbations and keep improvements."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_sc = compute_score(a)
    best_a = a.copy()
    
    for trial in range(n_trials):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a_pert = a.copy()
        
        if trial % 5 == 0:
            # Gaussian perturbation
            scale = 0.01 + 0.03 * rng.random()
            a_pert = a_pert * (1.0 + scale * rng.standard_normal(n))
        elif trial % 5 == 1:
            # Sinusoidal perturbation
            freq = 1.0 + 4.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            x = np.arange(n) / n
            a_pert = a_pert * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
        elif trial % 5 == 2:
            # Block perturbation
            block = max(16, n // 64)
            nb = n // block
            if nb > 2:
                perm = rng.permutation(nb)
                for b in range(nb):
                    src = perm[b] * block
                    dst = b * block
                    a_pert[dst:dst+block] = a[src:src+block]
        elif trial % 5 == 3:
            # Targeted: reduce peak contributors
            c = fft_self_conv(a)
            k_star = int(np.argmax(c))
            i_start = max(0, k_star - n + 1)
            i_end = min(k_star, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            contribs = a[i_range] * a[k_star - i_range]
            top_n = max(5, n // 20)
            top_idx = np.argsort(contribs)[::-1][:top_n]
            for idx in top_idx:
                a_pert[i_range[idx]] *= (1.0 - 0.05 * rng.random())
            # Add back to random positions
            bottom_n = top_n
            bottom_idx = rng.choice(n, size=bottom_n, replace=False)
            total = a.sum() * 0.02
            a_pert[bottom_idx] += total / bottom_n
        else:
            # Mirror perturbation
            half = n // 2
            a_pert[:half] = a[:half] * (1.0 + 0.02 * rng.standard_normal(half))
            a_pert[half:] = a[half:] * (1.0 + 0.02 * rng.standard_normal(n - half))
        
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        # Quick LSE polish
        if time.monotonic() < deadline - 5:
            a_pert = run_lbfgs(a_pert, 500, deadline)
        
        sc = compute_score(a_pert)
        if sc < best_sc:
            best_sc = sc
            best_a = a_pert.copy()
            a = a_pert
    
    return best_sc, best_a

```python
"""
Step function optimization: minimize 2n*max(conv(a,a))/(sum(a))^2 for a >= 0.

Key mathematical insight:
  d(a*a)[k]/da[m] = 2*a[k-m] (with zero-padding)
  This gives O(n log n) gradient via FFT cross-correlation.

Strategy:
  1. Load parent solution, resample if needed
  2. Adaptive LSE continuation (broad-to-narrow active set via beta)
  3. Active-set projected gradient descent with backtracking
  4. Direct flattening heuristic targeting peak convolution contributions
  5. Multiple structured perturbations with LSE polish
  6. Final high-beta L-BFGS refinement
"""

import numpy as np
from scipy.optimize import minimize
import time


def fft_self_conv(a):
    """Linear self-convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def grad_conv_weighted(a, w):
    """
    Compute 2 * result[m] where result[m] = sum_j a[j]*w[j+m].
    This equals the gradient of sum_k w[k]*(a*a)[k] w.r.t. a.
    a has length n, w has length 2n-1.
    Result has length n.
    """
    n = len(a)
    w_rev = w[::-1]
    # Linear conv of a (len n) and w_rev (len 2n-1) has length 3n-2
    M_fft = 1 << (3 * n - 3).bit_length()
    A = np.fft.rfft(a, M_fft)
    WR = np.fft.rfft(w_rev, M_fft)
    conv = np.fft.irfft(A * WR, M_fft)
    # result[m] = conv[2n-2-m] for m=0..n-1
    res = conv[n - 1:2 * n - 1][::-1]
    return 2.0 * res


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fft_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and its gradient."""
    n = len(x)
    c = fft_self_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_s = np.exp(scaled)
    exp_sum = exp_s.sum()
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    p = exp_s / exp_sum
    score = 2.0 * n * L / (S * S)
    # grad_L[m] = 2 * sum_k p[k]*a[k-m] = grad_conv_weighted(x, p)[m]
    grad_L = grad_conv_weighted(x, p)
    # d(score)/da[m] = (2n/S^2)*grad_L[m] - 4n*L/S^3
    grad = (2.0 * n / (S * S)) * grad_L - 4.0 * n * L / (S * S * S)
    return score, grad


def run_lbfgs(a0, beta, deadline, maxfun_cap=50000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0
    bounds = [(0.0, 1000.0)] * n

    def fun(x):
        sc, _ = lse_obj_grad(x, beta)
        return sc

    def jac(x):
        _, g = lse_obj_grad(x, beta)
        return g

    remaining = deadline - time.monotonic()
    if remaining < 3.0:
        return a0
    maxfun = min(maxfun_cap, max(50, int(remaining * 120)))
    maxiter = max(20, int(remaining * 25))
    options = {'ftol': 1e-14, 'gtol': 1e-12, 'maxiter': maxiter,
               'maxfun': maxfun, 'maxls': 40}

    try:
        res = minimize(fun, a0, jac=jac, method='L-BFGS-B',
                       bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    return a_out


def active_set_gradient_descent(a, n_iters=3000, deadline=None,
                                delta_frac=0.02, p_power=2.0,
                                max_step=0.05, verbose=False):
    """
    Active-set projected gradient descent with backtracking line search.
    Uses smooth weighting of lags near the max to compute gradient.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve_count = 0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fft_self_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Adaptive delta: widen active set if stuck
        delta = delta_frac * M
        if no_improve_count > 200:
            delta = min(delta * 2.0, 0.1 * M)

        # Weight: (c[k] - (M - delta))^p for active lags
        threshold = M - delta
        diff = np.maximum(c - (M - delta), 0.0)
        w = diff ** p_power
        w_sum = w.sum()
        if w_sum < 1e-30:
            w[int(np.argmax(c))] = 1.0
        else:
