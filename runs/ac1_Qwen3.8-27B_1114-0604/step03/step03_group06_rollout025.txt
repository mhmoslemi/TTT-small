from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_smooth_max(c, beta):
    """Compute LSE approximation of max(c) and the probability weights."""
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    exp_scaled = np.exp(scaled - scaled.max())
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta
    return L, p

def score_and_grad_lse(a, beta):
    """Compute LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    L, p = lse_smooth_max(c, beta)
    score = 2.0 * n * L / (S * S)
    
    # Gradient of L w.r.t. a
    # dL/da[j] = sum_k p[k] * (a[k-j] + a[j-k]) for valid indices
    # This is a cross-correlation: (rev_a * p) evaluated at specific indices
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad_L = 2.0 * conv_res
    
    # Full gradient of score
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3 * a)
    return score, grad

def project_simplex(v, target_sum=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Scale to work with sum=1, then rescale
    v = v / target_sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    proj = np.maximum(v - theta, 0)
    return proj * target_sum

    score = 2.0 * n * L / (S * S)

    # Gradient of L w.r.t. a via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad_L = 2.0 * conv_res

    # Full score gradient
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3 * a)
    return score, grad


def project_simplex(v):
    """Project v onto standard simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho_candidates = np.where(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def run_lbfgs_smooth(a0, beta, maxiter, maxfun, deadline):
    """Run L-BFGS-B with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 60}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out


def nesterov_projected_grad(a_init, beta, max_iter, deadline):
    """
    Nesterov-accelerated projected gradient on the simplex.
    Works with normalized (sum=1) sequences.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, a_init
    a = a / S  # Normalize to simplex

    best_a = a.copy()
    best_score = compute_score(a)

    t = 1.0
    y = a.copy()
    a_prev = a.copy()

    for k in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0.0)
        y_sum = float(y.sum())
        if y_sum < 1e-12:
            y = a.copy()
        else:
            y = y / y_sum

        # Compute gradient at y (on the simplex, score = 2n * LSE since S=1)
        c = fast_conv(y)
        L, p = lse_max_weights(c, beta)
        score_y = 2.0 * n * L  # Since sum=1

        # Gradient of L w.r.t. y
        rev_y = y[::-1]
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_y, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
        grad_L = 2.0 * conv_res
        grad = 2.0 * n * grad_L

        # Step size: use 1/(L*k) with L estimated from beta and problem size
        # L ~ 2*n*beta*max(y) roughly
        L_lip = 2.0 * n * beta * max(0.01, float(y.max()))
        step = 1.0 / (L_lip * (1.0 + k * 0.01))
        step = min(step, 0.5 / max(1.0, float(np.linalg.norm(grad))))

        # Projected update
        a_new = y - step * grad
        a_new = project_simplex(a_new)

        # Check
        sc_new = compute_score(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a_prev = a
        a = a_new
        t = t_new

        # Periodic check with exact score
        if k % 100 == 0:
            sc_exact = compute_score(a)
            if sc_exact < best_score:
                best_score = sc_exact
                best_a = a.copy()

    return best_score, best_a


def topk_subgrad_descent(a_init, K, max_iter, deadline):
    """
    Subgradient descent using top-K active lags with Polyak step size.
    Works on normalized (sum=1) sequences.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, a_init
    a = a / S

    best_a = a.copy()
    best_score = compute_score(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)  # Theoretical lower bound

    for k in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = float(c.max())
        score = 2.0 * n * M  # Since sum=1

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Get top-K lags
        top_k_idx = np.argsort(c)[::-1][:K]
        c_topk = c[top_k_idx]
        # Weight: emphasize lags closer to max
        c_max = c_topk[0]
        c_min = c_topk[-1]
        if c_max - c_min > 1e-12:
            weights = (c_topk - c_min) / (c_max - c_min)
        else:
            weights = np.ones(K) / K
        weights = weights ** 2  # Emphasize peaks more
        weights = weights / weights.sum()

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i, kk in enumerate(top_k_idx):
            w = weights[i]
            j_start = max(0, kk - n + 1)
            j_end = min(kk, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1][::-1]
                sg[j_start:j_end + 1] += w * seg

        # Polyak step size
        gap = score - lower_bound
        if gap < 1e-10:
            break
        sg_norm_sq = float(np.dot(sg, sg))
        if sg_norm_sq < 1e-30:
            break
        step = gap / (sg_norm_sq + 1e-30)
        step = min(step, 0.1 / max(1.0, float(np.linalg.norm(sg))))

        a_new = a - step * sg
        a_new = project_simplex(a_new)
        a = a_new

    return best_score, best_a


def lp_direction_step(a, deadline, max_wall_s=15.0):
    """
    Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k.
    Returns rescaled b or None.
    """
    if not HAVE_CVXPY:
        return None
    n = len(a)
    if n > 4000:
        return None

    t0 = time.monotonic()
    if time.monotonic() > deadline:
        return None

    c = fast_conv(a)
    M = float(c.max())
    if M < 1e-12:
        return a.copy()

    # Build constraint matrix (banded)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_mat[k_start:k_end, j] = a[:k_end - k_start]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)

    try:
        remaining = max(1.0, max_wall_s - (time.monotonic() - t0))
        prob.solve(solver=cp.PDLP, max_time=remaining)
    except Exception:
        try:
            remaining = max(1.0, max_wall_s - (time.monotonic() - t0))
            prob.solve(solver=cp.ECOS, max_time=remaining)
        except Exception:
            return None

    if time.monotonic() - t0 > max_wall_s or b_var.value is None:
        return None

    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None

    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    return b_opt


def generate_inits(n, rng, parent):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].astype(np.float64).copy())

    # Structured inits
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.4 * np.cos(np.pi * x) ** 2)
    inits.append(np.clip(0.2 + 0.8 * x, 0.01, 1000.0))
    inits.append(np.clip(0.2 + 0.8 * (1.0 - x), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.3 * np.sin(4.0 * np.pi * x), 0.01, 1000.0))
    inits.append(np.clip(0.1 + 0.9 * np.exp(-4.0 * (x - 0.5) ** 2), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))

    # Asymmetric bump
    inits.append(np.clip(0.3 + 0.7 * np.exp(-6.0 * (x - 0.3) ** 2), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2), 0.01, 1000.0))

    return inits


def refine_active_set(a, deadline, max_iter=20000):
    """
    Refine using exact subgradient on adaptively-chosen active set.
    Uses decreasing step sizes and tracks the best solution.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, a
    a = a / S

    best_a = a.copy()
    best_score = compute_score(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    K = max(5, n // 500)  # Start with moderate K, increase over time

    for k in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = float(c.max())
        score = 2.0 * n * M

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Adapt K: start small, grow
        K_cur = min(K + k // 5000, n // 2)
        K_cur = max(3, K_cur)

        top_k_idx = np.argsort(c)[::-1][:K_cur]
        c_topk = c[top_k_idx]
        c_max = c_topk[0]
        c_min = c_topk[-1]

        if c_max - c_min > 1e-14:
            weights = (c_topk - c_min) / (c_max - c_min)
        else:
            weights = np.ones(K_cur) / K_cur
        weights = weights ** 3  # Strong emphasis on peaks
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, kk in enumerate(top_k_idx):
            w = weights[i]
            j_start = max(0, kk - n + 1)
            j_end = min(kk, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1][::-1]
                sg[j_start:j_end + 1] += w * seg

        gap = score - lower_bound
        if gap < 1e-12:
            break
        sg_norm_sq = float(np.dot(sg, sg))
        if sg_norm_sq < 1e-30:
            break

        # Polyak step with damping
        step = gap / (sg_norm_sq + 1e-30)
        step = min(step, 0.05 / max(1.0, np.sqrt(sg_norm_sq)))
        # Additional damping for stability
        step *= 0.5

        a_new = a - step * sg
        a_new = project_simplex(a_new)
        a = a_new

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Combines LSE-smoothed L-BFGS-B, Nesterov projected gradient,
    active-set subgradient refinement, and LP direction steps.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 25.0

    # Load parent solution
    parent = None
    try:
        hsq = globals().get('height_sequence_1', None)
        if hsq is not None:
            p = np.asarray(hsq, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        pa = len(parent)
        lengths.append(pa)
        for dl in [-512, -256, 256, 512]:
            cand = pa + dl
            if cand >= 1024:
                lengths.append(cand)
    if not lengths:
        lengths = [4096, 6144, 8192]
    lengths = sorted(set(lengths))
    # Limit to at most 4 lengths, prioritize parent length
    if len(lengths) > 4:
        if parent is not None:
            lengths = [len(parent)] + [l for l in lengths if l != len(parent)][:3]
        else:
            lengths = lengths[:4]

    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}, len={len(parent) if parent is not None else 0}")
    print(f"CVXPY available: {HAVE_CVXPY}")
    print(f"Total budget: {budget_s}s, deadline at t={deadline - t_start:.0f}s")

    # =================================================================
    # Phase 1: LSE-smoothed L-BFGS-B with continuation (50% of time)
    # =================================================================
    phase1_deadline = t_start + 0.50 * (total_budget - 25.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: L-BFGS-B with LSE continuation")
    print(f"  Deadline: {phase1_deadline - t_start:.0f}s")
    print(f"{'='*60}")

    beta_schedule = [20, 50, 100, 200, 400, 800, 1600, 3200, 6000]

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n  --- n = {n} ---")
        inits = generate_inits(n, rng, parent)

        # Limit inits based on time
        remaining = phase1_deadline - time.monotonic()
        max_inits = max(2, min(len(inits), int(remaining / 80.0)))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_cur = a0.copy()
            for bi, beta in enumerate(beta_schedule):
                if time.monotonic() > phase1_deadline:
                    break
                remaining_bi = phase1_deadline - time.monotonic()
                if remaining_bi < 5:
                    break
                maxiter = min(3000, max(200, int(remaining_bi / 10.0)))
                maxfun = maxiter * 3
                sc, a_opt = run_lbfgs_smooth(a_cur, beta, maxiter, maxfun,
                                             time.monotonic() + remaining_bi * 0.8)
                a_cur = a_opt
                report(sc, a_opt, f"n={n}, init={i}, beta={beta}")

    print(f"\n  After Phase 1: best = {best_score:.8f}")

    # =================================================================
    # Phase 2: Nesterov projected gradient on simplex (20% of time)
    # =================================================================
    phase2_deadline = t_start + 0.70 * (total_budget - 25.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: Nesterov projected gradient")
    print(f"  Deadline: {phase2_deadline - t_start:.0f}s")
    print(f"{'='*60}")

    if best_a is not None:
        for beta in [50, 100, 200, 500]:
            if time.monotonic() > phase2_deadline:
                break
            remaining = phase2_deadline - time.monotonic()
            max_iter = max(1000, int(remaining / 20.0) * 200)
            sc, a = nesterov_projected_grad(best_a, beta, max_iter,
                                           time.monotonic() + (phase2_deadline - time.monotonic()) * 0.7)
            report(sc, a, f"Nesterov beta={beta}")

    print(f"\n  After Phase 2: best = {best_score:.8f}")

    # =================================================================
    # Phase 3: Active-set subgradient refinement (15% of time)
    # =================================================================
    phase3_deadline = t_start + 0.85 * (total_budget - 25.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Active-set subgradient refinement")
    print(f"  Deadline: {phase3_deadline - t_start:.0f}s")
    print(f"{'='*60}")

    if best_a is not None:
        remaining = phase3_deadline - time.monotonic()
        max_iter = max(5000, int(remaining * 500))
        sc, a = refine_active_set(best_a, time.monotonic() + remaining * 0.8, max_iter)
        report(sc, a, "active-set refine")

        # Second round with the refined solution
        if time.monotonic() < phase3_deadline:
            remaining2 = phase3_deadline - time.monotonic()
            sc, a = refine_active_set(a, time.monotonic() + remaining2 * 0.8,
                                      max(2000, int(remaining2 * 300)))
            report(sc, a, "active-set refine 2")

    print(f"\n  After Phase 3: best = {best_score:.8f}")

    # =================================================================
    # Phase 4: LP-based improvement + final polish (15% of time)
    # =================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: LP + Final Polish")
    print(f"{'='*60}")

    if best_a is not None:
        # LP-based steps
        if HAVE_CVXPY and len(best_a) <= 5000:
            a_lp = best_a.copy()
            for it in range(10):
                if time.monotonic() > deadline:
                    break
                b_opt = lp_direction_step(a_lp, deadline, max_wall_s=12.0)
                if b_opt is not None:
                    for t_mix in [0.2, 0.4, 0.6]:
                        a_try = (1 - t_mix) * a_lp + t_mix * b_opt
                        a_try = np.clip(a_try, 0.0, 1000.0)
                        if a_try.sum() < 0.01:
                            continue
                        sc = compute_score(a_try)
                        if sc < best_score:
                            report(sc, a_try, f"LP step {it}, t={t_mix}")
                            a_lp = a_try
                            break
                    else:
                        a_lp = (1 - 0.3) * a_lp + 0.3 * b_opt
                        a_lp = np.clip(a_lp, 0.0, 1000.0)

        # Final L-BFGS polish with very high beta
        if time.monotonic() < deadline - 10:
            for beta_fin in [5000, 10000]:
                if time.monotonic() > deadline - 5:
                    break
                remaining = deadline - time.monotonic() - 5
                if remaining < 3:
                    break
                sc, a = run_lbfgs_smooth(best_a, beta_fin,
                                        max(500, int(remaining / 5.0)),
                                        max(2000, int(remaining / 2.0)),
                                        time.monotonic() + remaining * 0.7)
                report(sc, a, f"final LBFGS beta={beta_fin}")

        # Final subgradient polish
        if time.monotonic() < deadline - 3:
            sc, a = refine_active_set(best_a, deadline, max_iter=50000)
            report(sc, a, "final subgrad")

    # =================================================================
    # Final verification
    # =================================================================
    if best_a is None:
        best_a = np.ones(1024) * 0.5

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")

    return out
