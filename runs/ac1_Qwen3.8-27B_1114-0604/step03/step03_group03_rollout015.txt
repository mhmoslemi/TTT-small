Improved step function optimizer using hybrid CD + reduced LP approach.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
- Start from parent sequence (n=6962)
- Phase 1: Quick CD warmup
- Phase 2: Alternating CD + reduced LP iterations (top-K active constraints)
- Phase 3: Final CD polish

import numpy as np
from scipy.optimize import minimize_scalar, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def conv_cross(a, b):
    """Cross-convolution a * b via FFT."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def coordinate_descent_sweep(a, rng=None, n_coords=None):
    """One full sweep of coordinate descent with incremental conv updates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    if n_coords is None:
        n_coords = n

    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)

    for idx in range(min(n_coords, n)):
        i = int(order[idx])
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()

        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S / n * 3.0)
        t_upper = min(t_upper, 1000.0)

        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-8}
        )
        t_opt = float(res.x)

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a

def reduced_lp_update(a, K=100):
    """
    Solve a reduced LP using top-K binding constraints.
    Returns updated sequence or None if LP fails.
    """
    n = len(a)
    conv = fast_conv(a)
    M = float(conv.max())
    S = float(a.sum())
    S_target = np.sqrt(2.0 * n)
    
    # Get top K indices
    top_k_idx = np.argsort(conv)[::-1][:K]
    
    # Build constraint matrix: A[k_idx, j] = a[k_idx - j] for valid j
    # (a*b)[k] = sum_j a[k-j] * b[j]
    rows_list = []
    cols_list = []
    data_list = []
    
    for idx, k in enumerate(top_k_idx):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_hi >= j_lo:
            js = np.arange(j_lo, j_hi + 1)
            # A[idx, j] = a[k - j]
            vals = a[k - j_hi : k - j_lo + 1][::-1]
            rows_list.append(np.full(len(js), idx))
            cols_list.append(js)
            data_list.append(vals)
    
    if len(data_list) == 0:
        return None
    
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    data = np.concatenate(data_list)
    
    A_sparse = csr_matrix((data, (rows, cols)), shape=(K, n))
    
    # LP: maximize sum(b) subject to A_sp @ b <= M, b >= 0
    # scipy linprog minimizes, so negate: minimize -sum(b)
    c = -np.ones(n)
    res = linprog(
        c,
        A_ub=A_sparse,
        b_ub=np.full(K, M),
        bounds=[(0, None)] * n,
        method='highs'
    )
    
    if res.status != 0:
        return None
    
    b = np.clip(res.x, 0, None)
    b_sum = float(b.sum())
    if b_sum < 1e-12:
        return None
    
    # Rescale to match target sum
    b = b * S_target / b_sum
    return b


def line_search_blend(a, b, S_target, n):
    """Find best t in [0,1] to minimize objective of (1-t)*a + t*b."""
    def obj_at_t(t):
        c = (1 - t) * a + t * b
        Sc = float(c.sum())
        if Sc < 1e-12:
            return 1e10
        conv_c = fast_conv(c)
        return 2.0 * n * float(conv_c.max()) / (Sc * Sc)
    
    # Evaluate at several points and pick the best, then refine
    ts = np.linspace(0, 1, 20)
    scores = np.array([obj_at_t(t) for t in ts])
    best_idx = int(np.argmin(scores))
    
    # Refine around best
    t_lo = max(0, ts[max(0, best_idx - 1)])
    t_hi = min(1, ts[min(len(ts) - 1, best_idx + 1)])
    
    res = minimize_scalar(obj_at_t, bounds=(t_lo, t_hi), method='bounded',
                          options={'xatol': 1e-6})
    return float(res.x)
