def adam_lse(a_init, lr=0.01, n_steps=10000, beta=200,

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8

    best_score = compute_score(a)

        if time_limit_s is not None and t_start is not None:

                break
        
        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad(a, beta)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)

                best_a = a.copy()

    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    score = compute_score(a_out)
