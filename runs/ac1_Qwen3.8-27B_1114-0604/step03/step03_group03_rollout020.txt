```python
"""
Hybrid multiplicative-update + coordinate descent optimizer.
Uses FFT-based cross-correlation for fast global moves,
combined with targeted coordinate descent for local polish.
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def compute_pressure(w, a):
    """Compute p_j = sum_k w[k] * a[k-j] using FFT cross-correlation."""
    n = len(a)
    N = 1 << (len(w) + n - 2).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(W * np.conj(A), N)
    return R[:n].copy()

def multiplicative_update(a, R, beta, gamma, eps=1e-10):
    """One step of the multiplicative update."""
    n = len(a)
    # Compute weights
    R_clipped = np.maximum(R, 0)
    log_w = beta * R_clipped
    log_w -= log_w.max()  # for numerical stability
    w = np.exp(log_w)
    w /= w.sum()
    
    # Compute pressure
    p = compute_pressure(w, a)
    p = p / (p.mean() + eps)  # normalize to mean 1
    
    # Multiplicative update
    a_new = a * np.power(p + eps, -gamma)
    a_new /= a_new.sum()  # renormalize
    return a_new

def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)

def cd_sweep_fast(a, n_sweeps=1, hot_only=False, hot_frac=0.3):
    """Fast coordinate descent sweep."""
    n = len(a)
    a = a.copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

```python
"""
Hybrid multiplicative-FFT + coordinate descent optimizer for minimum peak autocorrelation.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovation over previous approach:
- Uses FFT-based cross-correlation for O(n log n) "global" updates
  that simultaneously adjust all coordinates based on their contribution
  to the peak autocorrelation lags.
- Annealing schedule: progressively focuses on the highest-peak lags.
- Combined with targeted coordinate descent for local polish.
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_pressure(w, a):
    """
    Compute p_j = sum_k w[k] * a[k-j] for j=0..n-1 using FFT cross-correlation.
    w has length 2n-1 (the convolution lags).
    """
    n = len(a)
    N = 1 << (len(w) + n - 2).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(W * np.conj(A), N)
    return R[:n].copy()


def multiplicative_step(a, R, beta, gamma, eps=1e-12):
    """
    One step of the multiplicative update.
    a: current sequence (will be normalized to sum 1 internally)
    R: current autocorrelation (conv(a,a))
    beta: sharpness of weight (higher = more focused on peaks)
    gamma: step size of multiplicative update
    Returns new a (normalized to same sum as input).
    """
    S = float(a.sum())
    a_norm = a / S  # work with normalized sequence
    
    # Compute weights
    R_pos = np.maximum(R, 0.0)
    # Scale R for numerical stability
    R_max = R_pos.max()
    if R_max < 1e-15:
        return a.copy()
    scaled = beta * R_pos / R_max  # scale so max is beta
    log_w = scaled - scaled.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-15:
        return a.copy()
    w /= w_sum
    
    # Compute pressure
    p = compute_pressure(w, a_norm)
    p_mean = p.mean()
    if p_mean < 1e-15:
        return a.copy()
    p_norm = p / (p_mean + eps)
    
    # Multiplicative update (always positive)
    a_new = a_norm * np.power(1.0 + p_norm, -gamma)
    a_new_sum = a_new.sum()
    if a_new_sum < 1e-15:
        return a.copy()
    a_new /= a_new_sum
    
    # Scale back to original sum
    a_new *= S
    return a_new


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def cd_single_sweep(a, order=None):
    """Single CD sweep. Returns (score, a)."""
    n = len(a)
    a = a.copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1
    
    if order is None:
        order = np.arange(n)
    
    for i in order:
        i = int(i)
        a_i = a[i]
        
        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)
        
        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0
        
        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()
        
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = np.where(is_2i, a_i * a_i, 2.0 * a_i * a_vals)
        base_inside -= old_contrib
        
        alpha = np.where(is_2i, 0.0, 2.0 * a_vals)
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0
        
        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)
        
        if a_i > 1e-12:
            res = minimize_scalar(
                coord_1d_obj,
                args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
                bounds=(0.0, t_upper),
                method='bounded',
                options={'xatol': 1e-6}
            )
            t_opt = float(res.x)
        else:
            # For zero coordinates, just check if positive value helps
            t_opt = 0.0
        
        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef
    
    final_score = compute_score(a)
    return final_score, a

    """CD sweep on the 'hottest' coordinates (those contributing to the peak)."""
    n = len(a)
    conv = fast_conv(a)
    k_star = int(np.argmax(conv))

    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    
    n_hot = max(50, int(n * frac))
    order = np.argsort(contrib)[::-1][:n_hot]
    
    return cd_single_sweep(a, order=order)


def targeted_perturb(a, rng, strength=0.03):
    """Perturb elements contributing most to the peak."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(10, n // 30)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)
