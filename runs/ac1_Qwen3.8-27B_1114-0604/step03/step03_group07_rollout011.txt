"""
Optimized step function search using multi-phase LSE-continuation L-BFGS with
adaptive beta scheduling, peak-tracking subgradient polish, and multi-scale
initialization. Improves upon previous 1.509482 bound by using more aggressive
beta values, better initialization from parent, periodic restarts, and a
novel peak-elimination local search phase.

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    # Gradient: dL/da[k] involves cross-correlation of a with p
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapper(a, beta):
    """Scipy-compatible objective returning (value, gradient)."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on LSE-smoothed objective. Returns (exact_score, array, success)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
    except Exception:
        return compute_score(a0), a0, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.success


def build_initializations(n, rng, parent):
    """Generate diverse initializations for the optimizer."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else x

    if parent is not None and len(parent) == n:
        inits.append((parent.astype(np.float64).copy(), "parent"))

    # Uniform
    inits.append((np.ones(n) * 0.5, "uniform"))

    # Anti-palindromic ramps (strongest structural bias)
    inits.append((np.linspace(0.01, 1.0, n), "ramp_up"))
    inits.append((np.linspace(1.0, 0.01, n), "ramp_down"))

    # Quadratic anti-palindromic
    inits.append((0.01 + 0.99 * xn ** 2, "quad_up"))
    inits.append((0.01 + 0.99 * (1 - xn) ** 2, "quad_down"))

    # Raised cosine taper (smooth anti-palindromic)
    taper = 0.5 - 0.5 * np.cos(2 * np.pi * xn)
    inits.append((0.05 + 0.95 * taper, "raised_cos"))
    inits.append((0.05 + 0.95 * (1 - taper), "raised_cos_rev"))

    # Sinusoidal anti-palindromic (odd half-wave)
    for freq in [1, 2, 3]:
        s = 0.5 + 0.45 * np.sin(np.pi * freq * xn)
        inits.append((np.clip(s, 0.01, 10.0), f"sin_{freq}"))

    # Combination: ramp + oscillation
    inits.append((np.clip(0.2 + 0.5 * xn + 0.2 * np.sin(2 * np.pi * 5 * xn), 0.01, 10.0), "ramp_osc"))
    inits.append((np.clip(0.2 + 0.5 * (1 - xn) + 0.2 * np.cos(2 * np.pi * 3 * xn), 0.01, 10.0), "ramp_osc2"))

    # Gaussian centered (reduces edge contributions)
    for sigma_frac in [0.3, 0.4, 0.5]:
        center = n * 0.5
        g = np.exp(-0.5 * ((x - center) / (n * sigma_frac)) ** 2)
        inits.append((0.01 + g, f"gauss_{sigma_frac}"))

    # Smoothed random
    a_rand = np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 10.0)
    ksize = max(3, n // 100)
    kernel = np.exp(-0.5 * (np.arange(-ksize, ksize + 1) / (ksize / 3)) ** 2)
    kernel /= kernel.sum()
    a_smooth = np.convolve(a_rand, kernel, mode='same')
    inits.append((np.clip(a_smooth, 0.01, 10.0), "smooth_rand"))

    # Pure random
    inits.append((np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0), "rand"))

    return inits


def subgradient_polish(a, max_iters=5000, verbose=False):
    """Exact subgradient descent with Polyak step size for the max-conv problem."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return compute_score(a), a
    # Normalize to sum=1 for numerical stability
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(best_a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(max_iters):
        c = fast_conv(a)
        S_cur = float(a.sum())
        if S_cur < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S_cur * S_cur)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max(conv) w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S_cur * S_cur
        S3 = S2 * S_cur
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - sc_lb) / gn2
        step = min(step, 2.0 / (1.0 + 0.01 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 1e-6:
            break

    return best_sc, best_a


def peak_elimination_search(a, rng, n_iters=200, verbose=False):
    """
    Local search that specifically targets the largest peaks in the convolution.
    For each peak, computes the exact gradient and takes a directional step
    that reduces it while increasing the sum (to maintain normalization).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(best_a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        S = float(a.sum())
        if S < 1e-12:
            break
        c = fast_conv(a)
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peaks
        n_peaks = min(5, 2 * n - 1)
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[peak_idx]

        # Aggregate gradient from top peaks with weighting
        grad = np.zeros(n)
        total_w = 0.0
        for i, k in enumerate(peak_idx):
            w = peak_vals[i] ** 2  # weight by peak height squared
            j_start = max(0, int(k) - n + 1)
            j_end = min(int(k), n - 1)
            if j_end >= j_start:
                grad[j_start:j_end + 1] += w * 2.0 * a[j_start:j_end + 1][::-1]
            total_w += w

        if total_w < 1e-30:
            break

        # Add the "sum" component gradient: -2*M/S^3 * (2n)
        S2 = S * S
        S3 = S2 * S
        grad -= (2.0 * n * total_w * M) / S3

        # Normalize gradient
        gn = np.linalg.norm(grad)
        if gn < 1e-30:
            break

        # Adaptive step size
        step = (sc - sc_lb) / (gn * gn) * 0.5
        step = min(step, 0.1 / (1.0 + 0.005 * it))

        # Try the step
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        if a_new.sum() < 1e-6:
            a_new = a * 0.95
        sc_new = compute_score(a_new)
        if sc_new < sc:
            a = a_new
        else:
            # Try smaller step
            step *= 0.3
            a_new = a - step * grad
            a_new = np.clip(a_new, 0.0, 1000.0)
            if a_new.sum() < 1e-6:
                a_new = a
            sc_new = compute_score(a_new)
            if sc_new < sc:
                a = a_new
            else:
                # Random kick to escape
                a += 0.01 * rng.standard_normal(n)
                a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def frequency_domain_refine(a, rng, n_iters=50):
    """
    Perturb the sequence in the frequency domain to explore different
    spectral shapes. Multiply the FFT by a learned correction factor.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(best_a)
    M_fft = 1 << (2 * n - 2).bit_length()
    freqs = np.arange(len(np.fft.rfft(a, M_fft)))
    freq_norm = freqs / len(freqs)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)

        # Apply a smooth spectral correction
        alpha = rng.uniform(0.05, 0.3)
        correction = 1.0 - alpha * freq_norm ** rng.uniform(1, 3)
        A_mod = A * correction
        a_new = np.real(np.fft.irfft(A_mod, M_fft))[:n]
        a_new = np.clip(a_new, 0.0, 1000.0)

        if a_new.sum() < 1e-6:
            continue
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Add small noise in time domain
            a += 0.02 * rng.standard_normal(n)
            a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def multi_scale_optimize(n_target, rng, budget_s, parent=None):
    """
    Multi-scale optimization: start from a coarse version, optimize,
    then upsample and refine at full resolution.
    """
    t0 = time.monotonic()

    # Build a multi-scale ladder
    scales = []
    s = max(16, n_target // 64)
    while s < n_target:
        scales.append(s)
        s = min(s * 2, n_target)
    if scales[-1] != n_target:
        scales.append(n_target)

    # Start from the coarsest scale
    current = None
    if parent is not None:
        # Downsample parent to coarsest scale
        ratio = len(parent) // scales[0]
        if ratio >= 1:
            current = parent[::ratio][:scales[0]].copy()
            if len(current) < scales[0]:
                current = np.resize(current, scales[0])

    if current is None:
        current = np.ones(scales[0]) * 0.5

    for scale_n in scales:
        if time.monotonic() - t0 > budget_s * 0.6:
            break

        if current is not None and len(current) != scale_n:
            # Upsample via interpolation
            old_n = len(current)
            new_n = scale_n
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, new_n)
            current = np.interp(x_new, x_old, current)

        # Optimize at this scale with a few beta levels
        if scale_n < n_target:
            beta_list = [50, 200, 1000]
            max_iter = 2000
        else:
            beta_list = [100, 500, 2000, 8000]
            max_iter = 4000

        for beta in beta_list:
            if time.monotonic() - t0 > budget_s * 0.7:
                break
            sc, a_opt, _ = run_lbfgs(current, beta, maxiter=max_iter, maxfun=max_iter * 15)
            current = a_opt

    return current

        if time.monotonic() > phase1_end:
            break
        print(f"  Initializing for n={n}")

        inits = build_initializations(n, rng, parent)
        # Quick screen with low-beta LSE
        for init_a, label in inits:
            if time.monotonic() > phase1_end:
                break
            sc, a_opt, _ = run_lbfgs(init_a, beta=50, maxiter=1000, maxfun=15000)
            candidates.append((sc, a_opt, f"n={n} {label}"))
            report(sc, a_opt, f"P1 {label}")

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"  Top 3 candidates after Phase 1:")
    for sc, a, lab in candidates[:3]:
        print(f"    {sc:.8f} ({lab})")

    # ============================================================
    # Phase 2: LSE Continuation (45% of budget)
    # ============================================================
    phase2_end = t_start + 0.60 * budget_s
    print(f"\n=== Phase 2: LSE Continuation (until t={phase2_end - t_start:.0f}s) ===")

    beta_schedule = [100, 300, 1000, 3000, 10000, 30000, 80000]

    # Take top candidates and refine
    n_top = min(5, len(candidates))
    for ci in range(n_top):
        if time.monotonic() > phase2_end:
            break
        sc_init, a_init, label = candidates[ci]
        n = len(a_init)
        time_left = phase2_end - time.monotonic()
        est_per_beta = 10.0 + 0.002 * n
        n_betas = min(len(beta_schedule), max(2, int(time_left / est_per_beta)))

        print(f"\n  Candidate {ci+1}/{n_top} ({label}, init={sc_init:.6f}):")
        a_cur = a_init.copy()

        for bi, beta in enumerate(beta_schedule[:n_betas]):
            if time.monotonic() > phase2_end:
                break
            remaining = phase2_end - time.monotonic()
            maxit = max(500, min(5000, int(remaining / (est_per_beta + 1))))
            sc, a_opt, success = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxit * 20)
            a_cur = a_opt
            improved = report(sc, a_opt, f"P2 c{ci} b={beta}")
            if not improved and sc > sc_init * 1.05:
                # Diverged, restart from original
                a_cur = a_init.copy()
                continue

        # Add to candidates for later phases
        candidates.append((compute_score(a_cur), a_cur, f"refined_{ci}"))

    # ============================================================
    # Phase 3: Peak elimination + frequency refinement (20%)
    # ============================================================
    phase3_end = t_start + 0.80 * budget_s
    print(f"\n=== Phase 3: Peak Elimination & Frequency Refinement ===")
    print(f"  Current best: {best_score:.8f}")

    n_best = len(best_a)
    a_work = best_a.copy()

    # Peak elimination
    if time.monotonic() < phase3_end - 60:
        n_iters = max(100, int((phase3_end - time.monotonic()) / max(1, 2 * n_best / 1000)))
        sc, a_opt = peak_elimination_search(a_work, rng, n_iters=n_iters)
        report(sc, a_opt, "peak_elim")
        a_work = a_opt

    # Frequency domain refinement
    if time.monotonic() < phase3_end - 30:
        time_left = phase3_end - time.monotonic()
        n_iters = max(20, int(time_left / max(1, n_best / 500)))
        sc, a_opt = frequency_domain_refine(a_work, rng, n_iters=n_iters)
        report(sc, a_opt, "freq_refine")
        a_work = a_opt

    # Another round of high-beta LSE
    if time.monotonic() < phase3_end:
        for beta in [50000, 150000]:
            if time.monotonic() > phase3_end:
                break
            sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=3000, maxfun=50000)
            report(sc, a_opt, f"P3 LSE b={beta}")

    # ============================================================
    # Phase 4: Final polish (until deadline)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    # Subgradient polish
    if time.monotonic() < deadline - 60:
        time_left = deadline - time.monotonic()
        n_iters = max(500, int(time_left * 50))
        sc, a_opt = subgradient_polish(best_a, max_iters=n_iters)
        report(sc, a_opt, "subgrad_polish")

    # One more LSE pass with very high beta
    if time.monotonic() < deadline - 30:
        sc, a_opt, _ = run_lbfgs(best_a, beta=200000, maxiter=2000, maxfun=30000)
        report(sc, a_opt, "final_LSE")

    # Final subgradient
    if time.monotonic() < deadline - 10:
        sc, a_opt = subgradient_polish(best_a, max_iters=3000)
        report(sc, a_opt, "final_subgrad")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified
        print(f"  Verified score better than internal: {verified:.8f}")

    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
