"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses top-k peak penalty, frequency-domain initialization, multi-resolution
extension, and aggressive LSE-continuation with peak-shaving refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_topk_score_grad(a, beta, k_top=5):
    """
    Score using LSE over top-k peaks.
    This focuses the penalty on the most problematic peaks.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    # Get top-k peaks
    c_max = float(c.max())
    
    # LSE over all peaks (standard)
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    rev_a = a[::-1].copy()
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements over previous:
1. Frequency-domain structured initialization with anti-palindromic constraints
2. Aggressive LSE-continuation with adaptive beta schedule
3. Iterative peak-shaving via top-lag targeted adjustments
4. Proximal gradient with simplex projection for final polish
5. Multi-resolution extension from parent to longer sequences
6. Diverse restart strategy including spectral and combinatorial inits

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_grad(x, beta)

    try:
        res = minimize(
            obj_and_grad, a0, jac=True,
            method='L-BFGS-B', bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0, False

    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.success


def project_onto_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgrad_polish(a, max_iters=5000, sc_lb=1.5):
    """Exact subgradient descent with Polyak step and simplex projection."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S  # Normalize to sum=1
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(max_iters):
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M  # sum=1

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        # Gradient of 2n*M (with sum=1 constraint handled by projection)
        grad = 2.0 * n * sg
        # Subtract mean to project onto tangent of sum=1
        grad = grad - np.mean(grad)

        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = max(0.0, (sc - sc_lb)) / gn2
        step = min(step, 0.1 / (1.0 + 0.002 * it))

        a_new = a - step * grad
        a_new = project_onto_simplex(a_new, 1.0)
        a = a_new

    return best_sc, best_a * (best_a.sum() * 0.5 + 0.5)  # Rescale for safety


def peak_shaving(a, n_rounds=50, strength=0.02):
    """Iteratively reduce the top peaks in the autocorrelation."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find all lags within 10% of max
        threshold = M * 0.9
        active_lags = np.where(c >= threshold)[0]

        for k in active_lags[:20]:  # Limit to top 20 active lags
            j_start = max(0, int(k) - n + 1)
            j_end = min(int(k), n - 1)
            if j_end < j_start:
                continue
            # Contributors: a[j] * a[k-j] for j in [j_start, j_end]
            js = np.arange(j_start, j_end + 1)
            pairs = a[js] * a[k - js]
            # Reduce the larger contributor at each pair
            for idx in range(len(js)):
                j = js[idx]
                j2 = k - j
                if j == j2:
                    a[j] *= (1.0 - strength)
                else:
                    if a[j] > a[j2]:
                        amt = a[j] * strength * pairs[idx] / max(M, 1e-20)
                        a[j] = max(0, a[j] - amt)
                        a[j2] += amt
                    else:
                        amt = a[j2] * strength * pairs[idx] / max(M, 1e-20)
                        a[j2] = max(0, a[j2] - amt)
                        a[j] += amt

        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def build_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / max(n - 1, 1)

    if parent is not None and len(parent) > 0:
        # Interpolate parent to length n
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            p = np.asarray(parent, dtype=np.float64)
            p_norm = p / p.sum() * n
            from scipy.interpolate import interp1d
            xp = np.arange(len(p_norm))
            fp = interp1d(xp, p_norm, kind='cubic', fill_value='extrapolate')
            a_interp = np.maximum(fp(np.linspace(0, len(p_norm) - 1, n)), 0)
            inits.append(a_interp)
            # Add small perturbation
            inits.append(np.clip(a_interp * (1 + 0.1 * rng.standard_normal(n)), 0.01, 1000.0))

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Strong anti-palindromic (ramp)
    inits.append(np.linspace(0.01, 1.0, n))
    inits.append(np.linspace(1.0, 0.01, n))

    # Anti-palindromic with smooth transition
    inits.append(0.05 + 0.95 * xn)
    inits.append(0.05 + 0.95 * (1.0 - xn))

    # Frequency-domain initialization: flat spectrum with anti-palindromic phase
    A = np.ones(n // 2 + 1) * 0.5
    # Add anti-palindromic phase: phi[k] = pi*k (linear phase = time shift)
    # For anti-palindromic: phi[k] = pi*(k-n/2)
    freqs = np.arange(n // 2 + 1)
    phase = np.pi * freqs  # This gives a shifted delta
    A_complex = A * np.exp(1j * phase)
    a_spec = np.fft.irfft(A_complex, n)
    inits.append(np.clip(a_spec, 0.01, 1000.0))

    # Random with anti-palindromic bias
    a_rand = 0.5 + 0.3 * rng.standard_normal(n)
    a_anti = a_rand * (0.3 + 0.7 * xn)
    inits.append(np.clip(a_anti, 0.01, 1000.0))
    inits.append(np.clip(a_rand * (0.3 + 0.7 * (1 - xn)), 0.01, 1000.0))

    # Sinusoidal anti-palindromic
    for freq in [1, 2, 3]:
        inits.append(np.clip(0.4 + 0.4 * xn + 0.2 * np.sin(np.pi * freq * xn), 0.01, 1000.0))

    # Two-block
    half = n // 2
    inits.append(np.concatenate([np.ones(half) * 0.9, np.ones(n - half) * 0.1]))
    inits.append(np.concatenate([np.ones(half) * 0.1, np.ones(n - half) * 0.9]))

    # Gradual ramp with oscillation
    inits.append(np.clip(0.1 + 0.8 * xn + 0.1 * np.sin(2 * np.pi * 5 * xn), 0.01, 1000.0))
    inits.append(np.clip(0.1 + 0.8 * (1 - xn) + 0.1 * np.cos(2 * np.pi * 3 * xn), 0.01, 1000.0))

    # Pure random
    inits.append(np.clip(0.5 * rng.random(n) + 0.01, 0.01, 1000.0))

    return inits

    print(f"\n  After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Subgradient / proximal polish (10% time)
    # ============================================================
    phase4_end = t_start + 0.92 * budget_s
    print(f"\n=== Phase 4: Subgradient Polish (until t={phase4_end - t_start:.0f}s) ===")

    time_left = phase4_end - time.monotonic()
    n_iters = max(2000, int(time_left * 50))
    sc, a_opt = subgrad_polish(best_a, max_iters=n_iters, sc_lb=1.45)
    report(sc, a_opt, "subgrad")

    # Final LSE from subgradient result
    if time.monotonic() < phase4_end:
        time_left = phase4_end - time.monotonic()
        maxit = max(500, int(time_left / 20))
        sc, a_opt = run_lbfgs(a_opt, 100000, maxiter=maxit, maxfun=maxit * 10)
        report(sc, a_opt, "final_lbfgs_from_subgrad")

    print(f"\n  After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final diverse restarts (8% time)
    # ============================================================
    print(f"\n=== Phase 5: Final Diverse Restarts ===")

    n_best = len(best_a)
    trial = 0
    while time.monotonic() < deadline and trial < 15:
        trial += 1
        strat = trial % 5

        if strat == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strat == 1:
            # Targeted: reduce elements at peak contributors
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            if j_end >= j_start:
                contrib = np.zeros(n_best)
                js = np.arange(j_start, j_end + 1)
                contrib[js] = best_a[js] * best_a[k_star - js]
                top_n = max(5, n_best // 200)
                top_idx = np.argsort(contrib)[::-1][:top_n]
                a_pert[top_idx] *= 0.95
                bottom_idx = np.argsort(contrib)[:-top_n]
                total_red = (best_a[top_idx] - a_pert[top_idx]).sum()
                if len(bottom_idx) > 0 and total_red > 0:
                    a_pert[bottom_idx] += total_red / len(bottom_idx)
        elif strat == 2:
            # Anti-palindromic nudge
            a_pert = best_a.copy()
            xn = np.arange(n_best) / max(n_best - 1, 1)
            a_pert *= (1.0 + 0.05 * (xn - 0.5))  # Slight ramp
        elif strat == 3:
            # Frequency-domain smoothing
            M_fft = 1 << (2 * n_best - 2).bit_length()
            A = np.fft.rfft(a_pert if 'a_pert' in dir() else best_a, M_fft)
            freqs = np.arange(len(A))
            att = 1.0 - 0.05 * (freqs / len(A)) ** 3
            A *= att
            a_pert = np.real(np.fft.irfft(A, M_fft))[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Large random restart but biased toward best
            a_pert = 0.7 * best_a + 0.3 * np.clip(rng.standard_normal(n_best), 0, 2) * best_a.mean()

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            a_pert = best_a * 0.9

        time_left = deadline - time.monotonic()
        if time_left < 30:
            break
        maxit = max(500, min(3000, int(time_left / 30)))
        sc, a_opt = run_lbfgs(a_pert, 30000, maxiter=maxit, maxfun=maxit * 15)
        report(sc, a_opt, f"P5 #{trial} s={strat}")

    # ============================================================
    # Final subgradient polish
    # ============================================================
    if time.monotonic() < deadline - 3:
        time_left = deadline - time.monotonic()
        n_iters = max(1000, int(time_left * 30))
        sc, a_opt = subgrad_polish(best_a, max_iters=n_iters, sc_lb=1.45)
        report(sc, a_opt, "final_subgrad")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
