"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines L-BFGS-B with LSE smoothing, multiplicative exponential gradient descent,
Nesterov-accelerated LSE, exact subgradient polish, LP-based direction finding,
and multi-length/multi-start exploration to escape local minima.
Key improvements over previous: multiplicative descent, Nesterov acceleration,
larger sequences (up to 16384), adaptive continuation, peak-targeted perturbation.
"""

import numpy as np
from scipy.optimize import minimize
import time
try:
    import cvxpy as cp
    HAVE_CVXPY = True
except ImportError:
    HAVE_CVXPY = False


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_score_only(a, beta):
    """LSE-smoothed score only (no gradient)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    L = c_max + np.log(exp_scaled.sum()) / beta
    return 2.0 * n * L / (S * S)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B minimization with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lambda x: lse_score_grad(x, beta), a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def multiplicative_descent(a0, beta, max_iters=5000, max_wall_s=60.0):
    """
    Multiplicative (exponential) gradient descent for LSE-smoothed objective.
    Maintains positivity automatically. Different trajectory than L-BFGS-B.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    if a.sum() < 1e-6:
        a[:] = 0.5
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()
    eta = 0.5 / (2.0 * n)
    stall_count = 0
    last_best_sc = best_sc

    for it in range(max_iters):
        if time.monotonic() - t0 > max_wall_s:
            break

        sc, grad = lse_score_grad(a, beta)
        gnorm_inf = float(np.max(np.abs(grad)))
        if gnorm_inf < 1e-14:
            break

        # Adaptive step size
        step = min(eta, 1.0 / (gnorm_inf + 1e-10))
        step = min(step, 0.3)

        a_new = a * np.exp(-step * grad)
        a_new = np.clip(a_new, 1e-10, 1000.0)

        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            # Slowly decrease eta for fine-tuning
            eta *= 0.995
            stall_count = 0
        else:
            a = a_new
            eta *= 0.8
            stall_count += 1
            if stall_count > 200:
                # Reset with larger step
                eta = 1.0 / (2.0 * n)
                stall_count = 0

        # Periodic exact score check
        if it % 100 == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()

    return best_sc, best_a


def nesterov_lse(a0, beta, max_iters=3000, max_wall_s=60.0):
    """
    Nesterov accelerated gradient descent for LSE-smoothed objective.
    Maintains non-negativity via projection.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    if a.sum() < 1e-6:
        a[:] = 0.5
    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    t_mom = 1.0
    # Initial step size based on Lipschitz estimate
    sc_y, gamma = lse_score_grad(y, beta)
    gn = float(np.linalg.norm(gamma))
    if gn < 1e-14:
        return best_sc, best_a
    # Rough Lipschitz: L ~ 2n * beta * 2 * max(c) / S^2
    c_y = fast_conv(y)
    S_y = float(y.sum())
    L_ lip = 2.0 * n * beta * 2.0 * float(np.max(c_y)) / (S_y * S_y + 1e-10)
    L_ lip = max(L_ lip, 1e-6)
    step = 1.0 / L_ lip
    step = min(step, 0.5 / n)

    for it in range(max_iters):
        if time.monotonic() - t0 > max_wall_s:
            break

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        alpha = (t_mom - 1.0) / t_new
        t_mom = t_new

        y = a + alpha * (a - a_prev)
        y = np.clip(y, 0.0, 1000.0)

        sc_y, gamma = lse_score_grad(y, beta)
        if float(np.linalg.norm(gamma)) < 1e-14:
            break

        a_prev = a.copy()
        a = y - step * gamma
        a = np.clip(a, 0.0, 1000.0)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Backtracking: if function increased too much, shrink step
        if it > 0 and sc > best_sc * 1.5:
            step *= 0.5

    return best_sc, best_a


def exact_subgrad_polish(a, total_iters=2000, max_wall_s=120.0):
    """Projected subgradient descent on exact max-conv objective with Polyak step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0  # Theoretical lower bound (actually 2n/(2n-1))
    sc_lb = max(1.0, 2.0 * n / (2.0 * n - 1.0) - 0.01)
    t0 = time.monotonic()
    gamma = 0.5  # Step scaling

    for it in range(total_iters):
        if time.monotonic() - t0 > max_wall_s:
            break

        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Polyak step size
        step = gamma * (sc - sc_lb) / (sc - sc_lb + 0.01)

        # Subgradient of max(c) w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, 0.5)
        a = a - step * grad / (gn + 1e-15) * gn
        a = np.clip(a, 0.0, 1000.0)

        # Occasionally try a larger step for exploration
        if it % 50 == 0:
            a_try = a - 0.01 * grad / (gn + 1e-15)
            a_try = np.clip(a_try, 0.0, 1000.0)
            sc_try = compute_score(a_try)
            if sc_try < sc:
                a = a_try

    return best_sc, best_a


def solve_lp_direction(a, max_wall_s=30.0):
    """
    Solve the LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k.
    Uses FFT-based constraint evaluation for speed.
    """
    if not HAVE_CVXPY:
        return None
    n = len(a)
    t0 = time.monotonic()
    if n > 8000:
        return None

    c = fast_conv(a)
    M = float(np.max(c))
    if M < 1e-12:
        return a.copy()

    # Build constraint matrix: A_mat[k, j] = a[k-j] if 0<=k-j<n else 0
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_mat[k_start:k_end, j] = a[:k_end - k_start]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)

    try:
        remaining = max_wall_s - (time.monotonic() - t0)
        if remaining < 1.0:
            return None
        prob.solve(solver=cp.PDLP, max_time=remaining)
    except Exception:
        try:
            remaining = max_wall_s - (time.monotonic() - t0)
            if remaining < 1.0:
                return None
            prob.solve(solver=cp.ECOS, max_time=remaining)
        except Exception:
            return None

    if time.monotonic() - t0 > max_wall_s:
        return None
    if b_var.value is None:
        return None

    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None
    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    return b_opt

def targeted_perturb(a, rng, strength):
    """Perturb elements that contribute most to the peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 30)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_count = max(5, n // 20)
    bottom_idx = np.argsort(contrib)[:bottom_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def peak_shave(a, rng, n_shaves=10):
    """Directly shave the peak by reducing contributing elements."""
    n = len(a)
    a = a.copy()
    for _ in range(n_shaves):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        # Get contributions
        contrib = a[j_start:j_end + 1] * a[j_start:j_end + 1][::-1]
        S = float(a.sum())
        # Reduce the top contributors
        top_k = max(3, (j_end - j_start + 1) // 5)
        top_local = np.argsort(contrib)[::-1][:top_k]
        total_reduce = float(contrib[top_local].sum()) * 0.05
        for idx in top_local:
            j = j_start + idx
            a[j] = max(0.0, a[j] - total_reduce / (2.0 * top_k))
        # Add to bottom contributors
        bot_k = max(3, (j_end - j_start + 1) // 5)
        bot_local = np.argsort(contrib)[:bot_k]
        for idx in bot_local:
            j = j_start + idx
            a[j] += total_reduce / (2.0 * bot_k)
        a = np.clip(a, 0.0, 1000.0)
    return a


def resample_sequence(parent, n):
    """Resample a parent sequence to length n using linear interpolation."""
    if parent is None or len(parent) == 0:
        return np.ones(n) * 0.5
    if len(parent) == n:
        return parent.copy()
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, parent)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent solution
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        pa = parent.size
        lengths.append(pa)
        for dl in [1000, 2000, 4000, 6000]:
            cand = pa + dl
            if cand >= 1024 and cand <= 20000:
                lengths.append(cand)
    for n_try in [8192, 10240, 12288, 16384]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}, len={parent.size if parent is not None else 0}")
    print(f"CVXPY available: {HAVE_CVXPY}")
    print(f"Total budget: {total_budget}s, deadline at {deadline - t_start:.0f}s")

    # =====================================================================
    # Phase 1: L-BFGS-B with LSE continuation on primary lengths (45% time)
    # =====================================================================
    phase1_deadline = t_start + 0.45 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: L-BFGS-B + LSE continuation (deadline at +{phase1_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    beta_schedule = [10, 30, 80, 200, 500, 1000, 2000]

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n--- L-BFGS-B: n = {n} ---")
        inits = build_inits(n, rng, parent)

        # Prioritize parent-based init if available
        remaining = phase1_deadline - time.monotonic()
        max_inits = max(2, min(len(inits), int(remaining / 60.0) + 1))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_cur = a0.copy()
            sc_cur = compute_score(a_cur)
            report(sc_cur, a_cur, f"n={n}, init={i} (start)")

            for bi, beta in enumerate(beta_schedule):
                if time.monotonic() > phase1_deadline:
                    break
                remaining_bi = phase1_deadline - time.monotonic()
                if remaining_bi < 10.0:
                    break
                # Scale iterations available
                maxiter = min(8000, max(500, int(remaining_bi / 0.05)))
                maxfun = min(50000, max(2000, int(remaining_bi / 0.02)))
                sc, a_opt, res = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
                a_cur = a_opt
                report(sc, a_opt, f"n={n}, init={i}, beta={beta}")

    print(f"\nAfter Phase 1: best = {best_score:.8f}")

    # =====================================================================
    # Phase 2: Multiplicative descent + Nesterov on best solutions (20% time)
    # =====================================================================
    phase2_deadline = t_start + 0.65 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: Multiplicative + Nesterov (deadline at +{phase2_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    # Try multiplicative descent with different betas
    for beta in [200, 500, 1000, 2000]:
        if time.monotonic() > phase2_deadline:
            break
        sc, a_opt = multiplicative_descent(best_a, beta, max_iters=3000,
                                           max_wall_s=min(60.0, phase2_deadline - time.monotonic()))
        report(sc, a_opt, f"mult_desc beta={beta}")

    # Nesterov
    for beta in [500, 1500]:
        if time.monotonic() > phase2_deadline:
            break
        sc, a_opt = nesterov_lse(best_a, beta, max_iters=3000,
                                 max_wall_s=min(45.0, phase2_deadline - time.monotonic()))
        report(sc, a_opt, f"nesterov beta={beta}")

    # LP-based direction if available
    if HAVE_CVXPY and len(best_a) <= 8000:
        if time.monotonic() < phase2_deadline - 30.0:
            a_lp = best_a.copy()
            for it_lp in range(10):
                if time.monotonic() > phase2_deadline:
                    break
                b_opt = solve_lp_direction(a_lp, max_wall_s=min(20.0, phase2_deadline - time.monotonic()))
                if b_opt is None:
                    break
                best_sc_lp = np.inf
                best_a_lp = a_lp
                for t_step in [0.1, 0.2, 0.3, 0.5]:
                    a_new = (1.0 - t_step) * a_lp + t_step * b_opt
                    a_new = np.clip(a_new, 0.0, 1000.0)
                    sc = compute_score(a_new)
                    if sc < best_sc_lp:
                        best_sc_lp = sc
                        best_a_lp = a_new
                report(best_sc_lp, best_a_lp, f"LP step {it_lp}")
                a_lp = best_a_lp
                # Quick L-BFGS polish
                if time.monotonic() < phase2_deadline - 5.0:
                    sc, a_opt, _ = run_lbfgs(a_lp, 500, maxiter=2000, maxfun=5000)
                    report(sc, a_opt, f"LP+LBFGS {it_lp}")
                    a_lp = a_opt

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # =====================================================================
    # Phase 3: Multi-start perturbation + local optimization (20% time)
    # =====================================================================
    phase3_deadline = t_start + 0.85 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-start local search (deadline at +{phase3_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    trial = 0
    n_best = len(best_a)
    while time.monotonic() < phase3_deadline and trial < 40:
        strategy = trial % 8
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.03 + 0.07 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted perturbation
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.07 * rng.random())
        elif strategy == 2:
            # Frequency modulation
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.03 + 0.08 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Block swap
            block_sz = max(10, n_best // 15)
            b1 = int(rng.integers(0, max(1, n_best - 2 * block_sz)))
            b2 = int(rng.integers(0, max(1, n_best - 2 * block_sz)))
            a_pert = best_a.copy()
            a_pert[b1:b1 + block_sz], a_pert[b2:b2 + block_sz] = \
                a_pert[b2:b2 + block_sz].copy(), a_pert[b1:b1 + block_sz].copy()
        elif strategy == 4:
            # Sparse large perturbation
            n_sparse = max(10, n_best // 30)
            a_pert = best_a.copy()
            idx = rng.integers(0, n_best, size=n_sparse)
            a_pert[idx] *= (1.0 + 0.5 * rng.standard_normal(n_sparse))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 5:
            # Mix with structured sequence
            x = np.arange(n_best, dtype=np.float64) / n_best
            f1 = 1.0 + 5.0 * rng.random()
            f2 = 3.0 + 10.0 * rng.random()
            a_struct = 0.5 + 0.3 * np.cos(2.0 * np.pi * f1 * x) + 0.2 * np.cos(2.0 * np.pi * f2 * x)
            a_struct = np.clip(a_struct, 0.01, None)
            a_struct = a_struct * (best_a.sum() / a_struct.sum())
            mix = 0.2 + 0.4 * rng.random()
            a_pert = (1 - mix) * best_a + mix * a_struct
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 6:
            # Peak shaving + optimize
            a_pert = peak_shave(best_a, rng, n_shaves=5)
        else:
            # Wild perturbation: random rescaling of blocks
            n_blocks = int(rng.integers(5, 30))
            block_sz = max(5, n_best // n_blocks)
            a_pert = best_a.copy()
            for b in range(0, n_best, block_sz):
                end = min(b + block_sz, n_best)
                factor = 0.5 + rng.random()
                a_pert[b:end] *= factor
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Optimize the perturbed point
        remaining_p3 = phase3_deadline - time.monotonic()
        if remaining_p3 < 5.0:
            break

        # Use L-BFGS-B with high beta for quick local opt
        sc, a_opt, _ = run_lbfgs(a_pert, 1000, maxiter=3000, maxfun=6000)
        report(sc, a_opt, f"local #{trial} (strat={strategy})")

        # Also try multiplicative descent from this point
        if remaining_p3 > 20.0:
            sc2, a_opt2 = multiplicative_descent(a_opt, 500, max_iters=2000,
                                                 max_wall_s=min(30.0, remaining_p3 - 10.0))
            report(sc2, a_opt2, f"local #{trial} mult")

        trial += 1

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # =====================================================================
    # Phase 4: Try different lengths from best solution (10% time)
    # =====================================================================
    phase4_deadline = deadline
    print(f"\n{'='*60}")
    print(f"Phase 4: Length exploration + final refinement")
    print(f"{'='*60}")

    # Try extending to a larger length
    for n_ext in [len(best_a) + 2000, len(best_a) + 4000, 16384]:
        if time.monotonic() > phase4_deadline - 30.0:
            break
        if n_ext <= len(best_a) or n_ext > 20000:
            continue
        a_ext = resample_sequence(best_a, n_ext)
        sc_ext = compute_score(a_ext)
        print(f"  Resampled to n={n_ext}, score={sc_ext:.6f}")
        report(sc_ext, a_ext, f"ext n={n_ext}")

        # Optimize the extended version
        for beta in [200, 800, 2000]:
            if time.monotonic() > phase4_deadline - 10.0:
                break
            remaining_p4 = phase4_deadline - time.monotonic()
            sc, a_opt, _ = run_lbfgs(a_ext, beta, maxiter=5000, maxfun=10000)
            report(sc, a_opt, f"ext n={n_ext} beta={beta}")
            if sc < best_score:
                a_ext = a_opt

    # =====================================================================
    # Phase 5: Final polish on the best solution
    # =====================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Polish")
    print(f"{'='*60}")

    # High-beta LSE
    for beta in [2000, 5000, 10000]:
        if time.monotonic() > deadline - 5.0:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5.0:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=min(10000, int(remaining / 0.03)),
                                  maxfun=min(50000, int(remaining / 0.01)))
        report(sc, a_opt, f"final LBFGS beta={beta}")

    # Exact subgradient
    if time.monotonic() < deadline - 5.0:
        remaining = deadline - time.monotonic()
        sc, a_pol = exact_subgrad_polish(best_a, total_iters=min(5000, int(remaining * 50)),
                                         max_wall_s=remaining - 3.0)
        report(sc, a_pol, "subgrad polish")

    # One more multiplicative descent with very high beta
    if time.monotonic() < deadline - 3.0:
        remaining = deadline - time.monotonic()
        sc, a_opt = multiplicative_descent(best_a, 5000, max_iters=3000,
                                           max_wall_s=remaining - 2.0)
        report(sc, a_opt, "final mult descent")

    # Final L-BFGS
    if time.monotonic() < deadline - 1.0:
        sc, a_opt, _ = run_lbfgs(best_a, 8000, maxiter=5000, maxfun=20000)
        report(sc, a_opt, "very final LBFGS")

    # =====================================================================
    # Final verification and return
    # =====================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
