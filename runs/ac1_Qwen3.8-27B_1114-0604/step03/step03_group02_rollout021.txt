import numpy as np

import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient: grad[j] = 2n * (2 * (p cross_corr a)[j] / S^2 - 2*L / S^3)
    # (p cross_corr a)[j] = sum_k p[k] * a[k-j]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (2.0 * conv_res / S2 - 2.0 * L / S3)
    return score, grad

        S = n * 0.5
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8

        g = grad / S  # Normalize by sum to handle scale invariance

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(a_rev, M2_fft)
    conv_res = np.fft.irfft(P * RA, M2_fft)[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (2.0 * conv_res / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs_phase(a0, beta, deadline, maxiter=3000):
    """Run L-BFGS-B on LSE objective with time budget."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        sc, g = lse_score_grad(x, beta)
        return sc, g

    remaining = deadline - time.monotonic()
    if remaining < 3.0:
        return compute_score(a0), a0
    maxfun = min(50000, max(500, int(remaining * 150)))
    options = {'ftol': 1e-18, 'gtol': 1e-15, 'maxiter': min(maxiter, max(100, int(remaining * 30))), 'maxfun': maxfun, 'maxls': 30}

    try:
        res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out


def adam_optimize(a0, beta, deadline, n_steps=500, lr=0.5):
    """Adam optimizer on LSE-smoothed objective."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-6:
        a[:] = 0.5
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_sc = compute_score(a)
    best_a = a.copy()

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        sc, grad = lse_score_grad(a, beta)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

        S = a.sum()
        g = grad / max(S, 1e-10)

        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * (g * g)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        lr_t = lr * (0.5 ** (t // 100))
        step = -lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = a + step
        a = np.clip(a, 0.0, 1000.0)

    return compute_score(a), a, best_sc, best_a


def lse_continuation_lbfgs(a0, deadline, betas=None):
    """LSE continuation with L-BFGS at each level."""
    if betas is None:
        betas = [20, 50, 100, 200, 400, 800, 1600, 3200]
    n = len(a0)
    a = a0.copy()
    best_sc = compute_score(a)
    best_a = a.copy()
    time_per_level = (deadline - time.monotonic()) / len(betas)

    for beta in betas:
        level_deadline = min(deadline, time.monotonic() + time_per_level)
        if level_deadline - time.monotonic() < 2.0:
            break
        sc, a = run_lbfgs_phase(a, beta, level_deadline, maxiter=2000)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def multiplicative_equalize_v2(a, n_iters=300, deadline=None):
    """
    Adaptive multiplicative equalization.
    Uses LSE-based dual with adaptive step size.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    alpha = 0.3

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break
        S = float(a.sum())

        # Adaptive beta for LSE dual
        beta_eff = max(20.0, 30.0 / (c_max / S + 1e-10))
        scaled = beta_eff * (c - c_max)
        exp_s = np.exp(scaled)
        p = exp_s / exp_s.sum()

        # Compute h[j] = sum_k p[k] * a[k-j]
        a_rev = a[::-1]
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        P = np.fft.rfft(p, M2_fft)
        RA = np.fft.rfft(a_rev, M2_fft)
        h = np.fft.irfft(P * RA, M2_fft)[n - 1:2 * n - 1]

        h = np.maximum(h, 1e-15)
        active = h > 1e-10
        if active.sum() == 0:
            break
        h_mean = h[active].mean()

        # Adaptive alpha: reduce if oscillating
        alpha_cur = alpha * (1.0 / (1.0 + 0.05 * it))
        factor = 1.0 - alpha_cur + alpha_cur * (h_mean / h)
        a_new = a * factor
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new * (S / S_new)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert and reduce alpha
            alpha *= 0.7
            if alpha < 0.01:
                break

    return best_sc, best_a


def exact_subgrad_polish(a, n_iters=500, deadline=None):
    """
    Exact subgradient method with averaging over near-peak lags.
    Uses Polyak step size and diminishing step.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lower = 1.0  # theoretical lower bound

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find near-peak lags
        threshold = M * 0.995
        active = np.where(c >= threshold)[0]

        # Compute subgradient averaged over active lags
        sg = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
        sg /= len(active)

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak + diminishing step
        step_polyak = (sc - sc_lower) / (gn * gn + 1e-30)
        step_dim = 0.1 / (1.0 + 0.05 * it)
        step = min(step_polyak, step_dim, 1.0)

        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def targeted_refinement(a, deadline, n_rounds=20):
    """
    Targeted refinement: identify peak-contributing coordinates and
    do 1D optimization on them.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = int(np.argmax(c))

        # Find coordinates contributing to peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break

        # Contribution of each coordinate to the peak
        contrib = np.zeros(n)
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

        # Top contributing coordinates
        top_k = max(5, min(50, n // 100))
        top_idx = np.argsort(contrib)[::-1][:top_k]

        for j in top_idx:
            if deadline is not None and time.monotonic() > deadline:
                break
            a_orig = a[j]
            if a_orig < 1e-12:
                continue

            # Try multiplicative perturbations
            best_val = a_orig
            best_local_sc = best_sc
            for mult in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 0.5, 1.5]:
                a[j] = a_orig * mult
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = a[j]

            a[j] = best_val

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
        else:
            # No improvement, try smaller perturbations
            for j in top_idx[:10]:
                if deadline is not None and time.monotonic() > deadline:
                    break
                a_orig = a[j]
                if a_orig < 1e-12:
                    continue
                best_val = a_orig
                best_local_sc = best_sc
                for mult in [0.95, 0.97, 1.03, 1.05]:
                    a[j] = a_orig * mult
                    sc = compute_score(a)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_val = a[j]
                a[j] = best_val

    return best_sc, a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        p = parent.astype(np.float64)
        if len(p) != n:
            p = np.interp(np.linspace(0, len(p) - 1, n), np.arange(len(p)), p)
        inits.append(np.clip(p, 0.01, 1000.0).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.5 * np.sin(np.pi * x))
    inits.append(0.1 + 0.9 * np.sin(np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))

    # Gaussian-like
    center = n / 2
    sigma = n / 4
    inits.append(np.exp(-0.5 * ((x * n - center) / sigma) ** 2))

    # Two-hump
    inits.append(0.3 + 0.4 * np.cos(4.0 * np.pi * (x - 0.25)) ** 2 + 0.4 * np.cos(4.0 * np.pi * (x - 0.75)) ** 2)

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Multi-phase optimization with LSE continuation,
    Adam, multiplicative equalization, and exact subgradient polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def _report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = []
    if parent is not None:
        pa = parent.size
        if pa <= 32768:
            lengths.append(pa)
    for n in [8192, 16384]:
        if n not in lengths:
            lengths.append(n)
    if parent is not None and parent.size != 8192 and parent.size != 16384:
        if parent.size <= 32768:
            lengths.insert(0, parent.size)
    lengths = sorted(set(lengths))[:3]

    print(f"Lengths: {lengths}, Parent: {parent is not None}")

    # ============================================================
    # Phase 1: LSE continuation with L-BFGS (50% of time)
    # ============================================================
    phase1_dl = t_start + 0.50 * total_budget
    print(f"\n{'='*60}\nPhase 1: LSE Continuation (t={phase1_dl-t_start:.0f}s)\n{'='*60}")

    for n in lengths:
        if time.monotonic() > phase1_dl:
            break
        print(f"\n  --- n={n} ---")
        inits = build_inits(n, rng, parent)

        # Use parent as primary init if available
        if parent is not None and len(parent) == n:
            inits = [inits[0]] + inits[1:]

        n_to_try = min(len(inits), 4 if n <= 8192 else 2)
        for i in range(n_to_try):
            if time.monotonic() > phase1_dl:
                break
            t0 = time.monotonic()
            sc, a_opt = lse_continuation_lbfgs(inits[i], phase1_dl,
                                                betas=[20, 50, 100, 300, 800])
            _report(sc, a_opt, f"P1 n={n} i={i}")
            elapsed = time.monotonic() - t0
            print(f"    init {i}: score={sc:.6f}, time={elapsed:.1f}s")

    print(f"\n  After Phase 1: best={best_score:.8f}, n={len(best_a)}")

    # ============================================================
    # Phase 2: Adam + Multiplicative Equalization (30% of time)
    # ============================================================
    phase2_dl = t_start + 0.80 * total_budget
    print(f"\n{'='*60}\nPhase 2: Adam + Equalization (t={phase2_dl-t_start:.0f}s)\n{'='*60}")

    a_cur = best_a.copy()
    n_cur = len(a_cur)

    for rnd in range(12):
        if time.monotonic() > phase2_dl:
            break
        print(f"\n  Round {rnd+1}:")

        # Multiplicative equalization
        sc, a_eq = multiplicative_equalize_v2(a_cur, n_iters=200, deadline=phase2_dl)
        _report(sc, a_eq, f"P2 r{rnd} eq")
        a_cur = a_eq

        # Adam with high beta
        if time.monotonic() < phase2_dl - 15:
            beta = 500 + 200 * rnd
            sc, a_adam, _, best_adam = adam_optimize(a_cur, beta, phase2_dl,
                                                      n_steps=300, lr=0.3)
            _report(best_adam, best_adam, f"P2 r{rnd} adam b={beta}")
            a_cur = best_adam

    print(f"\n  After Phase 2: best={best_score:.8f}")

    # ============================================================
    # Phase 3: Exact subgradient + L-BFGS high-beta (12% of time)
    # ============================================================
    phase3_dl = t_start + 0.92 * total_budget
    print(f"\n{'='*60}\nPhase 3: Exact Polish (t={phase3_dl-t_start:.0f}s)\n{'='*60}")

    a_cur = best_a.copy()

    # Low-beta L-BFGS from best
    for beta in [2000, 5000]:
        if time.monotonic() > phase3_dl:
            break
        sc, a_opt = run_lbfgs_phase(a_cur, beta, phase3_dl, maxiter=2000)
        _report(sc, a_opt, f"P3 lbfgs b={beta}")
        a_cur = a_opt

    # Exact subgradient
    if time.monotonic() < phase3_dl - 5:
        sc, a_sg = exact_subgrad_polish(best_a, n_iters=500, deadline=phase3_dl)
        _report(sc, a_sg, "P3 subgrad")

    # Targeted refinement
    if time.monotonic() < phase3_dl - 3:
        sc, a_tr = targeted_refinement(best_a, deadline=phase3_dl, n_rounds=10)
        _report(sc, a_tr, "P3 targeted")

    print(f"\n  After Phase 3: best={best_score:.8f}")

    # ============================================================
    # Phase 4: Final intensive polish (8% of time)
    # ============================================================
    print(f"\n{'='*60}\nPhase 4: Final Polish\n{'='*60}")

    a_final = best_a.copy()
    remaining = deadline - 10 - time.monotonic()
    if remaining > 5:
        # One more round of equalization + subgrad
        sc, a_eq = multiplicative_equalize_v2(a_final, n_iters=100, deadline=deadline - 10)
        _report(sc, a_eq, "P4 eq")
        a_final = a_eq

        if time.monotonic() < deadline - 10:
            sc, a_sg = exact_subgrad_polish(a_final, n_iters=300, deadline=deadline - 5)
            _report(sc, a_sg, "P4 subgrad")
            a_final = a_sg

        # Final L-BFGS
        if time.monotonic() < deadline - 5:
            sc, a_opt = run_lbfgs_phase(a_final, 10000, deadline - 3, maxiter=1000)
            _report(sc, a_opt, "P4 final lbfgs")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
