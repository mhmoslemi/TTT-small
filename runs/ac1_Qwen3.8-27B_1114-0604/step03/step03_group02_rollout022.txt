Step function optimization via exact coordinate descent with multiplicative equalization.

The key insight is that for each coordinate a_j, the objective as a function of x = a_j
(with all other coordinates fixed) is a univariate function that can be minimized exactly.
This allows guaranteed monotone decrease per coordinate update, leading to robust convergence.

import numpy as np
from scipy.optimize import minimize, brent
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def golden_section_min(f, a, b, tol=1e-10, maxiter=60):
    """Minimize f on [a, b] using golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(maxiter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    x = (a + b) / 2.0
    return x, f(x)


def coordinate_descent_step(a, j, c, S, n):
    """
    Find optimal value of a[j] given all other coordinates fixed.
    
    Parameters:
    - a: current sequence
    - j: coordinate to optimize
    - c: current convolution (length 2n-1)
    - S: current sum of a
    - n: length of a
    
    Returns: optimal value x for a[j]
    """
    aj = a[j]
    S0 = S - aj
    if S0 < 1e-15:
        return 0.0
    
    # Compute d_k = tilde_a_{k-j} for k in [0, 2n-2]
    # tilde_a has tilde_a_j = 0, tilde_a_i = a_i for i != j
    # d_k = a_{k-j} if 0 <= k-j <= n-1 and k-j != j, else 0
    #      = a_{k-j} if 0 <= k-j <= n-1 and k != 2j, else 0
    
    d = np.zeros(2 * n - 1)
    # a_{k-j} is valid for k-j in [0, n-1], i.e., k in [j, j+n-1]
    k_start = j
    k_end = j + n - 1
    d[k_start:k_end+1] = a[:]  # a[0], a[1], ..., a[n-1] placed at positions j, j+1, ..., j+n-1
    # Zero out position k = 2j (where tilde_a_j = 0)
    if 2 * j < 2 * n - 1:
        d[2 * j] = 0.0
    
    # Now v_k(x) = c_k + 2*x*d_k + x^2 * delta_{k,2j}
    # M(x) = max_k v_k(x)
    # f(x) = 2n * M(x) / (S0 + x)^2
    
    def f_of_x(x):
        v = c + 2.0 * x * d
        if 2 * j < len(v):
            v[2 * j] += x * x  # Wait, v[2j] = c[2j] + 2*x*d[2j] + x^2 = c[2j] + 0 + x^2
        # Actually d[2j] = 0, so v[2j] = c[2j] + x^2
        return 2.0 * n * float(np.max(v)) / ((S0 + x) * (S0 + x))
    
    # Try x = 0, x = aj, and golden section
    f0 = f_of_x(0.0)
    f_cur = f_of_x(aj)
    
    # Upper bound for x: f(x) -> 2n as x -> inf, so x doesn't need to be very large
    # f(x) = 2n * max(c + 2x*d + x^2*e) / (S0+x)^2
    # For x > S0: (S0+x)^2 > 4*S0^2, and max(c+2x*d+x^2*e) <= max(c) + 2x*max(d) + x^2
    # So f(x) < 2n*(max(c) + 2x*max(d) + x^2)/(S0+x)^2 -> 2n as x->inf
    # A reasonable upper bound: x_max = 2*S0 (beyond this, f is decreasing)
    x_lo = 0.0
    x_hi = max(2.0 * S0, 2.0 * aj + S0, 1.0)
    
    # Golden section on [x_lo, x_hi]
    x_gs, f_gs = golden_section_min(f_of_x, x_lo, x_hi, tol=1e-12 * max(x_hi, 1.0), maxiter=50)
    
    # Take the best of the candidates
    candidates = [(0.0, f0), (aj, f_cur), (x_gs, f_gs)]
    best_x = min(candidates, key=lambda t: t[1])
    return best_x[0]
