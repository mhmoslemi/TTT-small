"""
LP-guided optimization for minimum peak autocorrelation sequences.
Core innovation: iterative LP-based "best response" to globally restructure
the sequence, followed by spectral error reduction and L-BFGS polish.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.linalg import toeplitz
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix T where (T @ g)[k] = conv(f,g)[k]."""
    n = len(f)
    col = np.zeros(2 * n - 1)
    col[:n] = f
    row = np.zeros(n)
    row[0] = f[0]
    return toeplitz(col, row)


def solve_lp_response(f, M, method='highs'):
    """
    Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Returns optimal g.
    """
    n = len(f)
    T = build_conv_matrix(f)
    c_obj = -np.ones(n)  # minimize -sum(g)
    A_ub = T
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
    if res.success:
        return res.x
    return f.copy()  # fallback


def lp_iteration(a, t=0.5, n_lp_iters=1, method='highs'):
    """
    Perform LP-based best response iterations.
    a: current sequence (will be normalized internally)
    t: blending parameter
    Returns improved sequence.
    """
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_lp_iters):
        c = fast_conv(a)
        M = float(c.max())
        g = solve_lp_response(a, M, method=method)
        g = np.maximum(g, 0.0)
        Sg = float(g.sum())
        if Sg < 1e-12:
            break
        g = g / Sg
        
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        Sc = float(a_new.sum())
        if Sc < 1e-12:
            break
        a_new = a_new / Sc
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def error_reduction(a, n_iters=200, alpha=0.7):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]
        
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0
        
        damp = 0.5 + 0.5 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def coordinate_descent_peak(a, n_sweeps=3, n_top=50):
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue
        
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old_j = a[j]
            candidates = [a_old_j * 0.5, a_old_j * 0.7, a_old_j * 0.8,
                         a_old_j * 0.9, a_old_j * 1.1, a_old_j * 1.3, 0.0]
            best_val = a_old_j
            best_local_sc = compute_score(a)
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-10:
                    continue
                a[j] = cv
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def subgradient_polish(a, total_iters=2000):
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = (sc - sc_lb) / (gn * gn + 1e-30)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a

"""
LP-guided optimization for minimum peak autocorrelation sequences.
Core innovation: iterative LP-based "best response" method to globally restructure
the sequence, combined with spectral error reduction and L-BFGS polish.
The LP at each step finds the optimal "response" sequence g that maximizes sum(g)
subject to cross-correlation constraints with the current sequence, providing a
globally-informed update direction that escapes local minima.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.linalg import toeplitz
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_conv_matrix(f):
    """Build the (2n-1) x n matrix T where (T @ g)[k] = conv(f, g)[k]."""
    n = len(f)
    col = np.zeros(2 * n - 1)
    col[:n] = f
    row = np.zeros(n)
    row[0] = f[0]
    return toeplitz(col, row)


def solve_lp_response(f, M):
    """
    Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Returns optimal g or None on failure.
    """
    n = len(f)
    T = build_conv_matrix(f)
    c_obj = -np.ones(n)
    A_ub = T
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception as e:
        print(f"    LP solve failed: {e}")
    return None


def lp_best_response(a, t_blend=0.3):
    """
    One LP-based best response iteration.
    a: current non-negative sequence
    Returns (improved_sequence, score)
    """
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a, np.inf
    a = a / S

    c = fast_conv(a)
    M = float(c.max())
    g = solve_lp_response(a, M)
    if g is None:
        return a, compute_score(a)

    Sg = float(g.sum())
    if Sg < 1e-12:
        return a, compute_score(a)
    g = g / Sg

    a_new = (1.0 - t_blend) * a + t_blend * g
    a_new = np.maximum(a_new, 0.0)
    Sn = float(a_new.sum())
    if Sn < 1e-12:
        return a, compute_score(a)
    a_new = a_new / Sn

    return a_new, compute_score(a_new)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for the normalized sequence."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def error_reduction(a, n_iters=200, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.5 + 0.5 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def coordinate_descent_peak(a, n_sweeps=3, n_top=50):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            candidates = [a_old_j * 0.5, a_old_j * 0.7, a_old_j * 0.8,
                         a_old_j * 0.9, a_old_j * 1.1, a_old_j * 1.3, 0.0]
            best_val = a_old_j
            best_local_sc = compute_score(a)
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-10:
                    continue
                a[j] = cv
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_polish(a, total_iters=2000):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.1)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def interp_to_length(a, target_n):
    """Interpolate sequence to target length."""
    if len(a) == target_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, target_n)
    return np.interp(x_new, x_old, a).copy()

    deadline = t_start + total_budget - 15.0

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    # Get parent sequence
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n0 = len(parent) if parent is not None else 4096
    start_a = parent.copy() if parent is not None else np.ones(n0) * 0.5
    print(f"Starting: n={n0}, parent={'yes' if parent is not None else 'no'}")

    # ============================================================
    # Phase 1: LP Best Response (~45% of time)
    # ============================================================
    phase1_deadline = t_start + 0.45 * total_budget
    print(f"\n{'='*50}")
    print(f"Phase 1: LP Best Response (deadline {phase1_deadline - t_start:.0f}s)")
    print(f"{'='*50}")

    # Try LP at multiple lengths
    lp_configs = [2048, 4096]
    if n0 >= 8192:
        lp_configs.append(8192)

    for n_lp in lp_configs:
        if time.monotonic() > phase1_deadline:
            break

        a_work = interp_to_length(start_a, n_lp)
        a_work = np.maximum(a_work, 0.0)
        S = a_work.sum()
        if S > 1e-12:
            a_work = a_work / S

        sc = compute_score(a_work)
        if sc < best_score:
            best_score = sc
            best_a = a_work.copy()
            print(f"  [LP init n={n_lp}] score={sc:.8f}")

        print(f"  LP at n={n_lp}: initial score={sc:.8f}")
        t0 = time.monotonic()
        g = solve_lp_response(a_work, float(fast_conv(a_work).max()))
        t1 = time.monotonic()
        if g is not None:
            print(f"    First LP solve: {t1-t0:.1f}s, sum(g)={g.sum():.2f}")
        else:
            print(f"    First LP solve FAILED")
            continue

        # Run multiple LP iterations with decreasing blend
        n_lp_iters = 30
        for it in range(n_lp_iters):
            if time.monotonic() > phase1_deadline:
                break
            t_blend = max(0.05, 0.4 * (1.0 - 0.7 * it / n_lp_iters))
            t0 = time.monotonic()
            a_new, sc = lp_best_response(a_work, t_blend)
            dt = time.monotonic() - t0

            if sc < best_score:
                best_score = sc
                best_a = np.clip(a_new.copy(), 0.0, 1000.0)
                print(f"    [LP n={n_lp} it={it} t={t_blend:.3f}] NEW BEST: {sc:.8f} ({dt:.1f}s)")
            elif it % 5 == 0:
                print(f"    it={it}: score={sc:.8f} (no improvement, {dt:.1f}s)")

            a_work = a_new

    print(f"After Phase 1: best_score = {best_score:.8f} (n={len(best_a)})")

    # ============================================================
    # Phase 2: Error Reduction (~20% of time)
    # ============================================================
    phase2_deadline = t_start + 0.65 * total_budget
    print(f"\n{'='*50}")
    print(f"Phase 2: Error Reduction (deadline {phase2_deadline - t_start:.0f}s)")
    print(f"{'='*50}")

    n_work = len(best_a)
    for alpha in [0.6, 0.7, 0.8, 0.85, 0.9]:
        if time.monotonic() > phase2_deadline:
            break
        a_er = error_reduction(best_a, n_iters=200, alpha=alpha)
        sc = compute_score(a_er)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_er.copy(), 0.0, 1000.0)
            print(f"  [ER alpha={alpha}] NEW BEST: {sc:.8f}")
        else:
            print(f"  [ER alpha={alpha}] score={sc:.8f}")

    # Try error reduction at different lengths
    for n_target in [n_work, max(2048, n_work // 2)]:
        if time.monotonic() > phase2_deadline:
            break
        a_trunc = interp_to_length(best_a, n_target)
        a_er = error_reduction(a_trunc, n_iters=200, alpha=0.8)
        sc = compute_score(a_er)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_er.copy(), 0.0, 1000.0)
            print(f"  [ER n={n_target}] NEW BEST: {sc:.8f}")

    print(f"After Phase 2: best_score = {best_score:.8f} (n={len(best_a)})")

    # ============================================================
    # Phase 3: L-BFGS Polish (~25% of time)
    # ============================================================
    phase3_deadline = t_start + 0.90 * total_budget
    print(f"\n{'='*50}")
    print(f"Phase 3: L-BFGS Polish (deadline {phase3_deadline - t_start:.0f}s)")
    print(f"{'='*50}")

    # Main L-BFGS with increasing beta
    for beta in [50, 100, 200, 500, 1000]:
        if time.monotonic() > phase3_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
        dt = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_opt.copy(), 0.0, 1000.0)
            print(f"  [LBFGS beta={beta}] NEW BEST: {sc:.8f} ({dt:.1f}s)")
        else:
            print(f"  [LBFGS beta={beta}] score={sc:.8f} ({dt:.1f}s)")

    # L-BFGS from perturbed starts
    n_best = len(best_a)
    n_trials = 6
    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline:
            break
        strategy = trial % 3
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Sinusoidal modulation
            freq = 1.0 + 6.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Block rescaling
            a_pert = best_a.copy()
            n_blocks = 5 + int(rng.integers(0, 8))
            block_size = max(1, n_best // n_blocks)
            for b in range(min(n_blocks, n_best // block_size)):
                s = b * block_size
                e = min(s + block_size, n_best)
                factor = 0.5 + rng.random()
                a_pert[s:e] *= factor
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S

        sc, a_opt = run_lbfgs(a_pert, 300, maxiter=3000, maxfun=50000)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_opt.copy(), 0.0, 1000.0)
            print(f"  [LBFGS pert#{trial} s={strategy}] NEW BEST: {sc:.8f}")
        else:
            print(f"  [LBFGS pert#{trial} s={strategy}] score={sc:.8f}")

    print(f"After Phase 3: best_score = {best_score:.8f} (n={len(best_a)})")

    # ============================================================
    # Phase 4: Final Polish (~5% of time)
    # ============================================================
    print(f"\n{'='*50}")
    print(f"Phase 4: Final Polish")
    print(f"{'='*50}")

    # Coordinate descent
    if time.monotonic() < deadline - 60:
        a_cd = coordinate_descent_peak(best_a, n_sweeps=3, n_top=100)
        sc = compute_score(a_cd)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_cd.copy(), 0.0, 1000.0)
            print(f"  [coord descent] NEW BEST: {sc:.8f}")

    # Subgradient polish
    if time.monotonic() < deadline - 60:
        a_sp = subgradient_polish(best_a, total_iters=3000)
        sc = compute_score(a_sp)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_sp.copy(), 0.0, 1000.0)
            print(f"  [subgrad] NEW BEST: {sc:.8f}")

    # Final high-beta L-BFGS
    if time.monotonic() < deadline - 30:
        sc, a_opt = run_lbfgs(best_a, 2000, maxiter=3000, maxfun=50000)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_opt.copy(), 0.0, 1000.0)
            print(f"  [final LBFGS beta=2000] NEW BEST: {sc:.8f}")

    # One more error reduction from best
    if time.monotonic() < deadline - 20:
        a_er = error_reduction(best_a, n_iters=100, alpha=0.85)
        sc = compute_score(a_er)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_er.copy(), 0.0, 1000.0)
            print(f"  [final ER] NEW BEST: {sc:.8f}")

    # Final coordinate descent
    if time.monotonic() < deadline - 15:
        a_cd = coordinate_descent_peak(best_a, n_sweeps=2, n_top=150)
        sc = compute_score(a_cd)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_cd.copy(), 0.0, 1000.0)
            print(f"  [final CD] NEW BEST: {sc:.8f}")

    # Build output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
