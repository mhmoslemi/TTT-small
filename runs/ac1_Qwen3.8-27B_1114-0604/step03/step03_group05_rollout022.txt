def fista_lse(a0, beta, maxiter=100000):
    """FISTA for LSE-smoothed objective with non-negativity."""
    n = len(a0)
    a = a0.copy()
    z = a0.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Lipschitz constant estimate (rough)
    # The gradient involves convolutions, so L ~ O(n * max(a))
    L = n * np.max(a0) + 1.0
    
    for it in range(maxiter):
        # Compute gradient at z
        _, grad = lse_score_grad(z, beta)
        
        # Step size: 1/L
        alpha = 1.0 / L
        
        # Update: y = z - alpha * grad
        y = z - alpha * grad
        
        # Proximal step (soft threshold for non-negativity)
        a_new = np.maximum(y, 0.0)
        
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        z = a_new + ((t - 1) / t_new) * (a_new - a)
        t = t_new
        a = a_new
        
        # Track best
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

"""
Step function optimization: FISTA-based LSE continuation with peak equalization,
adaptive smoothing, and multi-length exploration.
Target: beat 1.5027.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def fista_lse(a0, beta, maxiter=50000, verbose=False):
    """FISTA with backtracking for LSE-smoothed objective."""
    n = len(a0)
    a = a0.copy()
    z = a0.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Initial step size
    _, grad0 = lse_score_grad(z, beta)
    gnorm = np.linalg.norm(grad0)
    if gnorm < 1e-15:
        return best_sc, best_a
    L = gnorm / max(np.max(np.abs(a0)) * 0.01, 1e-8)
    L = max(L, 0.1)
    
    for it in range(maxiter):
        # Compute gradient at z
        _, grad = lse_score_grad(z, beta)
        
        # Backtracking line search
        alpha = 1.0 / L
        y = z - alpha * grad
        y = np.maximum(y, 0.0)
        
        # Check if step is good (sufficient decrease)
        fz_lse, _ = lse_score_grad(z, beta)
        fy_lse, _ = lse_score_grad(y, beta)
        while fy_lse > fz_lse - 0.5 * alpha * np.dot(grad, y - z):
            alpha *= 0.5
            y = z - alpha * grad
            y = np.maximum(y, 0.0)
            fy_lse, _ = lse_score_grad(y, beta)
            if alpha < 1e-15:
                break
        L = 1.0 / alpha if alpha > 0 else L
        
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        a_new = y
        z = a_new + ((t - 1) / t_new) * (a_new - a)

The maximum number of pairs is n (at position n-1). So the expected max is approximately p^2 * n.

S = p * n (approximately, for large n).
score ≈ 2n * p^2 * n / (p*n)^2 = 2n^2 * p^2 / (p^2 * n^2) = 2.

So random binary sequences give score ≈ 2, same as uniform. We need structure to do better.

The key is to have cancellation in the convolution. With non-negative values, there's no cancellation, but we can have "sparsity" that reduces the peak.

Actually, I realize the problem is related to the "minimum peak convolution" problem, which is different from the "minimum peak autocorrelation" problem.

For the convolution of a sequence with itself, having a flat convolution means the sequence is a "convolutional idempotent" or something similar. This is related to "tiling" the integers.

A tiling of the integers by a set A means that every integer can be written uniquely as a + b where a ∈ A and b ∈ B (for some set B). In this case, the convolution of the indicator functions of A and B is the constant 1.

If A = B, then conv(1_A, 1_A) = 1 means A tiles the integers with itself. This is possible for certain sets (e.g., A = {0, 1, 3, 4, 9, 10, 12, 13, ...} which tiles with B = {0, 3, 9, 27, ...}).

But we want conv(1_A, 1_A) to be as flat as possible, not exactly 1. The flatness of conv(1_A, 1_A) is related to the "packing" properties of A.

Let me also add a new technique: **direct peak reduction via projection**.

Idea: If the convolution has a peak M at position k, we want to reduce it. The rate of change of c_k with respect to a_i is:
dc_k/da_i = a[k-i] (if 0 <= k-i < n) + a[k-i+n] (if k-i+n < n... this doesn't make sense)

Wait, let me be precise:
c_k = sum_j a[j] * a[k-j] where 0 <= j < n and 0 <= k-j < n.

dc_k/da_i: the terms where a_i appears are:
- j = i: contributes a[k-i] (if 0 <= k-i < n)
- k-j = i, i.e., j = k-i: contributes a[k-i] (if 0 <= k-i < n)
- If i = k-i (i.e., 2i = k): only one term, contributes 2*a_i

So dc_k/da_i = 2*a[k-i] if i ≠ k/2 and 0 <= k-i < n
              = 2*a_i if 2i = k
              = 0 otherwise

In all cases: dc_k/da_i = 2*a[max(0, min(n-1, k-i))] if 0 <= k-i <= n-1, else 0.
Actually: dc_k/da_i = 2*a[k-i] if 0 <= k-i < n, else 0. (The factor of 2 accounts for both j=i and j=k-i cases, and when they coincide, it's still 2*a_i.)

Wait, when 2i = k, j=i and j=k-i=i are the same, so the term is a_i*a_i = a_i^2, and its derivative is 2*a_i. So the formula dc_k/da_i = 2*a[k-i] works in all cases (with a[j] = 0 for j outside [0,n-1]).

Good. So the subgradient of the max-conv objective at the peak position k* is:
sg[i] = 2 * a[k*-i] for 0 <= k*-i < n, else 0.

And the gradient of the full score = 2n * (sg/S^2 - 2*M/S^3).

        sg = np.zeros(n)

        j_end = min(k, n - 1)

        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)

        best_new_sc = sc
        for alpha in [0.1, 0.5, 1.0, 2.0, 5.0, 10.0] / (np.linalg.norm(grad) * S + 1e-10):
            a_new = a - alpha * grad
            a_new = np.clip(a_new, 0, 1000)

                best_new_sc = new_sc
                best_step = alpha

            # Try very small step
            alpha = 0.01 / (np.linalg.norm(grad) * S + 1e-10)
            a = np.clip(a - alpha * grad, 0, 1000)

            a_new = np.clip(a_new, 0, 1000)

"""
Step function optimization v3: Aggressive LSE + line-search subgradient + structured restarts.
Key improvements: aggressive line search in peak shaving, longer high-beta phase,
more diverse restarts, and adaptive step sizes.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L_val = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L_val / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(func, a0, args=args, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter,
                                'maxfun': maxfun, 'maxls': 50})
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def peak_shave_line_search(a, max_iters=5000):
    """Peak shaving with line search for aggressive peak reduction."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    
    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 2000:
            break
        
        # Compute subgradient
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end+1] = 2.0 * a[j_start:j_end+1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Line search: try a range of step sizes
        base = 1.0 / (gn * S)
        candidates = [0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0, 10.0]
        best_new_sc = sc
        best_step = 0
        
        for mult in candidates:
            alpha = mult * base
            a_new = a - alpha * grad
            a_new = np.clip(a_new, 0.0, 1000.0)
            new_sc = compute_score(a_new)
            if new_sc < best_new_sc:
                best_new_sc = new_sc
                best_step = alpha
        
        if best_step > 0:
            a = a - best_step * grad
            a = np.clip(a, 0.0, 1000.0)
        else:
            # Tiny step
            a = a - 0.001 * base * grad
            a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a

    # Perturb magnitudes
    A_mag = np.abs(A)
    A_phase = np.angle(A)
    # Reduce peak magnitudes, boost low ones
    med = np.median(A_mag)
    ratio = med / (A_mag + 1e-10)
    # Blend toward flat spectrum
    A_mag_new = A_mag + strength * med * (1.0 - A_mag / (med + 1e-10))
    A_mag_new = np.clip(A_mag_new, 0, None)
    A_new = A_mag_new * np.exp(1j * A_phase)
    a_pert = np.fft.ifft(A_new).real[:n]
    return np.clip(a_pert, 0.0, 1000.0)
