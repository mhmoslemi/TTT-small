```python
"""
Adaptive Minimax Optimizer with Spectral Preconditioning.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovations:
1. Adaptive spectral flattening with progressive alpha
2. Minimax-aware coordinate descent tracking multiple near-peak lags
3. Golden section line search for precise 1D optimization
4. Momentum-based subgradient with Nesterov acceleration
5. Population-based diversity with crossover
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_smoothed_objective(a, beta):
    """LSE-smoothed score and gradient for scipy minimize."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 0)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient computation
    # d(c[k])/d(a[m]) = sum of a[k-m] terms
    # Using FFT: gradient of convolution
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def spectral_flatten(a, n_iters=100, alpha_schedule=None):
    """
    Iterative spectral flattening.
    At each step, compute the power spectrum, apply a soft compression,
    and reconstruct the time-domain signal.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / np.sqrt(n)
    a = a / S  # normalize to sum = 1
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M = 1 << (2 * n - 1).bit_length()
    
    if alpha_schedule is None:
        alpha_schedule = 0.5 + 0.5 * (np.arange(n_iters) / max(1, n_iters - 1))
    
    for it in range(n_iters):
        alpha = alpha_schedule[it] if it < len(alpha_schedule) else 0.99
        
        # Compute power spectrum
        A = np.fft.fft(a.astype(np.float64), M)
        P = np.abs(A) ** 2
        
        # Target: compress peaks
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Soft compression: P_new = P_mean * (P / P_mean)^(1-alpha)
        ratio = P / P_mean
        P_new = P_mean * np.power(ratio, 1.0 - alpha)
        
        # Reconstruct: maintain phase, modify amplitude
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:

def spectral_flatten_v2(a, n_iters=100):
    """
    Alternative: direct error reduction on the autocorrelation.
    More aggressive: directly modify the autocorrelation to be flatter,
    then solve for the time-domain signal.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / np.sqrt(n)
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        # Compute autocorrelation
        A = np.fft.fft(a.astype(np.float64), M)
        P = np.abs(A) ** 2
        c = np.fft.ifft(P).real[:2*n-1]
        
        c_mean = c.mean()
        c_max = c.max()
        
        if c_max - c_mean < 1e-15:
            break
        
        # Progressive: start gentle, become more aggressive
        progress = it / max(1, n_iters - 1)
        strength = 0.3 + 0.6 * progress
        
        # Compress: c_target = c_mean + (c - c_mean) * (1 - strength * max(0, (c-c_mean)/(c_max-c_mean)))
        excess = np.maximum(c - c_mean, 0.0)
        c_range = c_max - c_mean
        if c_range < 1e-15:
            break
        # Soft threshold
        c_target = c - strength * excess * (excess / c_range)
        
        # Convert back to spectrum
        c_pad = np.zeros(M, dtype=np.complex128)
        c_pad[:2*n-1] = c_target
        P_new = np.fft.fft(c_pad).real
        
        # Ensure non-negative spectrum
        P_new = np.maximum(P_new, 0.0)
        
        # Reconstruct
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def minimax_coord_descent(a, n_sweeps=5, n_top=100, lags_to_track=5):
    """
    Coordinate descent that tracks multiple near-peak lags.
    For each coordinate, use golden section search to find the optimal value.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Golden section search
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        c_max = c.max()
        
        # Find top lags
        top_lags = np.argsort(c)[::-1][:lags_to_track]
        
        # For each top lag, find which coordinates contribute most
        contrib_matrix = np.zeros(n)
        for k in top_lags:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                other = k - j
                contrib_matrix[j] += a[other]
        
        # Sort by contribution (most important first)
        order = np.argsort(contrib_matrix)[::-1][:n_top]
        
        for j in order:
            a_orig = a[j]
            # Golden section search in [0, 3*a_orig]
            lo = 0.0
            hi = 3.0 * a_orig if a_orig > 1e-15 else 1e-3
            
            if hi <= lo:
                hi = lo + 0.1
            
            c1 = hi - phi * (hi - lo)
            c2 = lo + phi * (hi - lo)
            a[c1_idx] if False else None  # placeholder
            
            # Evaluate at c1 and c2
            a[j] = c1
            S = a.sum()
            if S < 1e-12:
                a[j] = a_orig
                continue
            a_temp = a / S
            f1 = compute_score(a_temp)
            
            a[j] = c2
            S = a.sum()
            if S < 1e-12:
                a[j] = a_orig
                continue
            a_temp = a / S
            f2 = compute_score(a_temp)
            
            for _ in range(10):  # 10 iterations of golden section
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - phi * (hi - lo)
                    a[j] = c1
                    S = a.sum()
                    if S < 1e-12:
                        break
                    a_temp = a / S
                    f1 = compute_score(a_temp)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + phi * (hi - lo)
                    a[j] = c2
                    S = a.sum()
                    if S < 1e-12:
                        break
                    a_temp = a / S
                    f2 = compute_score(a_temp)
            
            # Set to best found
            mid = (lo + hi) / 2
            a[j] = mid
            S = a.sum()
            if S > 1e-12:
                a = a / S
            sc = compute_score(a)
            if sc > best_sc * 1.1:
                # Reverted too much, restore
                a[j] = a_orig
                S = a.sum()
                if S > 1e-12:
                    a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def nesterov_subgradient(a, n_iters=3000, lr_init=0.01):
    """
    Nesterov accelerated subgradient method for the min-max objective.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    v = np.zeros(n)  # momentum
    y = a.copy()  # Nesterov lookahead
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        # Nesterov step
        t1 = 1.0 + it
        t2 = 1.0 + it + 1
        theta = (1.0 - t1) / t2
        y = a + theta * (a - y)
        y = np.maximum(y, 0.0)
        S_y = y.sum()
        if S_y < 1e-12:
            y = a
            S_y = S
        y = y / S_y
        
        # Compute subgradient at y
        c = fast_conv(y)
        c_max = c.max()
        k_star = int(np.argmax(c))
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        seg = y[j_start:j_end + 1]
        sg[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S_y2 = S_y * S_y
        S_y3 = S_y2 * S_y
        grad = (2.0 * n) * (sg / S_y2 - 2.0 * c_max / S_y3)
        
        lr = lr_init / (1.0 + 0.01 * it)
        
        v = 0.9 * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new = a
        else:
            a_new = a_new / S_new
        
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def build_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xm = x / n  # normalized to [0,1)
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    # Various parametric families
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2 * np.pi * xm))

    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 10.0))
    
    # Block structure
    for nb in (4, 8, 16, 32):
        per = max(1, n // nb)
        vals = 0.1 + 0.9 * rng.random(nb)
        a = np.tile(vals, n // nb + 1)[:n]
        inits.append(a.copy())
    
    return inits
```

Specifically, if we define the objective as f(a) = 2n * max_k c[k] / S^2, then at the optimum:
- For each j where a[j] > 0: ∂f/∂a[j] = lambda (Lagrange multiplier for the sum constraint)
- For each j where a[j] = 0: ∂f/∂a[j] >= lambda

The gradient of max_k c[k] at the optimum is a convex combination of the gradients of the active constraints. If the active set is {k1, k2, ..., km}, then:
∂max/∂a[j] = sum_{l} w_l * ∂c[k_l]/∂a[j] where w_l >= 0 and sum w_l = 1.

At the optimum, this should equal a constant (for all j with a[j] > 0).

This gives us a system of equations to solve! If we know the active set K* and the weights w, we can solve for the optimal a.

But we don't know the active set a priori. However, we can:
1. Start with a good approximation
2. Identify the active set (lags near the max)
3. Solve the KKT system
4. Check if the solution satisfies the active set assumption
5. If not, update the active set and repeat

This is the "active set method" and it's very powerful for problems with a small number of active constraints.

Let me implement this:

```python
def active_set_solve(a, active_lags, n_inner=100):
    """
    Solve the KKT system for a given active set.
    
    The KKT conditions at optimum:
    - For j with a[j] > 0: sum_l w_l * (2*a[k_l - j]) / S^2 - 2*max_c/S^3 = mu
      where mu is the Lagrange multiplier for sum constraint
    
    This is equivalent to: sum_l w_l * a[k_l - j] = constant (for all j with a[j] > 0)
    
    We can solve this by iteratively updating a to satisfy this condition.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    
    # Compute the "target gradient"
    # At optimum: sum_l w_l * a[k_l - j] should be constant for all j with a[j] > 0
    
    for it in range(n_inner):
        # Compute the current gradient contributions from each active lag
        target = np.zeros(n)
        for k in active_lags:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            seg = a[j_start:j_end + 1]
            target[j_start:j_end + 1] += seg[::-1]
        
        # The target should be constant where a > 0
        mask = a > 1e-10
        if mask.sum() == 0:
            break
        
        target_mean = target[mask].mean()
        target_std = target[mask].std()
        
        if target_std < 1e-12:
            break  # Already at optimum
        
        # Update: reduce a where target > mean, increase where target < mean
        # This is a gradient step on the "variance of target"
        delta = (target - target_mean) / (target_std + 1e-15)
        step = 0.1
        a = a - step * delta * a  # multiplicative update
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return a
```

import numpy as np

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = np.clip(beta * (c - c_max), -500, 0)
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2f = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2f)
    P = np.fft.rfft(p, M2f)
    cr = np.fft.irfft(RA * P, M2f)[n-1:2*n-1]
    g = 2.0 * cr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g / S2 - 2.0 * L / S3)
    return score, grad


def spectral_flatten(a, n_iters=100):
    """Adaptive spectral flattening with progressive compression."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / np.sqrt(n)
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M = 1 << (2*n-1).bit_length()
    
    for it in range(n_iters):
        A = np.fft.fft(a, M)
        P = np.abs(A)**2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        # Progressive: alpha goes from 0.3 to 0.95
        alpha = 0.3 + 0.65 * (it / max(1, n_iters-1))
        ratio = P / P_mean
        P_new = P_mean * np.power(ratio, 1.0 - alpha)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    return best_a


def conv_error_reduction(a, n_iters=100):
    """Direct convolution error reduction (flattens the autocorrelation)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
