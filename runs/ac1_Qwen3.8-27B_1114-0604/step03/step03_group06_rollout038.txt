"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses simplex-projected LSE gradient descent with continuation, IRL with FFT-based
QP solves, and multi-scale exploration. Key improvements: efficient FFT-based
cross-correlation for gradients, simplex projection (faster than L-BFGS bounds),
and iteratively reweighted QP that directly targets peak reduction.
"""

import numpy as np
import time
import math


def fft_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fft_cross_corr(a, w):
    """
    Compute cross-correlation (a *corr* w)[i] = sum_j a[j] * w[j+i] for i=0..n-1.
    a has length n, w has length 2n-1.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    M = len(w)  # 2n-1
    L = 1 << (n + M - 1 - 1).bit_length()
    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    result = np.fft.irfft(np.conj(A) * W, L)
    return result[:n]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fft_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and gradient.
    f(a) = 2n * LSE_beta(R) / S^2 where R = conv(a,a), S = sum(a).
    Returns (score, grad).
    """
    n = len(a)
    R = fft_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)

    R_max = float(np.max(R))
    scaled = beta * (R - R_max)
    exp_scaled = np.exp(scaled)
    sum_exp = float(np.sum(exp_scaled))
    LSE = R_max + np.log(sum_exp) / beta

    # p_k = exp(beta*R_k) / sum exp(beta*R_j)
    p = exp_scaled / sum_exp

    # dLSE/da_i = 2 * (a *corr* p)[i]
    cross = fft_cross_corr(a, p)

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * LSE / S2
    grad = 2.0 * n * (2.0 * cross / S2 - 2.0 * LSE / S3)

    return score, grad


def proj_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def exact_subgradient_step(a, use_nesterov=False):
    """
    Compute exact subgradient of f(a) = 2n*max(R)/S^2 at a.
    Returns (score, subgrad).
    """
    n = len(a)
    R = fft_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)

    R_max = float(np.max(R))
    score = 2.0 * n * R_max / (S * S)

    # Find all peaks (within numerical tolerance)
    tol = 1e-10 * R_max + 1e-15
    peak_indices = np.where(R >= R_max - tol)[0]

    # Subgradient: average over peak indices
    sg = np.zeros(n)
    for k in peak_indices:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            # dR[k]/da[i] = 2*a[k-i] for i in [j_start, j_end]
            seg = a[j_start:j_end + 1][::-1]
            sg[j_start:j_end + 1] += 2.0 * seg
    sg /= len(peak_indices)

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * R_max / S3)

    return score, grad

    best_sc = compute_score(a)
    lr = lr_init

        for step in range(max_steps_per_beta):
            if time.monotonic() - t_start > time_budget:

def fft_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fft_cross_corr(a, w):
    """
    Compute (a *corr* w)[i] = sum_j a[j] * w[j+i] for i=0..n-1.
    a has length n, w has length >= n (typically 2n-1).
    """
    n = len(a)
    M = len(w)
    L = 1 << (n + M - 2).bit_length()
    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    result = np.fft.irfft(np.conj(A) * W, L)
    return result[:n]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fft_conv(a)

    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient. Returns (score, grad)."""
    n = len(a)
    R = fft_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)
    R_max = float(np.max(R))
    scaled = beta * (R - R_max)
    exp_scaled = np.exp(scaled)
    sum_exp = float(np.sum(exp_scaled))
    LSE = R_max + np.log(sum_exp) / beta
    p = exp_scaled / sum_exp
    cross = fft_cross_corr(a, p)
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * LSE / S2
    grad = 2.0 * n * (2.0 * cross / S2 - 2.0 * LSE / S3)
    return score, grad


def proj_simplex(v):
    """Project onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def exact_subgrad(a):
    """Exact subgradient of 2n*max(R)/S^2. Returns (score, grad)."""
    n = len(a)
    R = fft_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)
    R_max = float(np.max(R))
    score = 2.0 * n * R_max / (S * S)
    tol = 1e-8 * R_max + 1e-15
    peaks = np.where(R >= R_max - tol)[0]
    sg = np.zeros(n)
    for k in peaks:
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]
    sg /= len(peaks)
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * R_max / S3)
    return score, grad
