        'maxiter': min(maxiter, max(100, int(remaining * 30))),
        'maxfun': min(maxfun, max(200, int(remaining * 50))),
        'maxls': 100
    }

    try:
        res = minimize(
            obj_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options=options
        )
        a_out = np.clip(res.x, 0.0, 1e6)
    except Exception:
        a_out = a0

    sc = compute_score(a_out)
    return sc, a_out


def subgradient_polish(a, n_iters, deadline):
    """
    Projected subgradient method on the exact max-conv objective.
    Normalized (sum=1) formulation: minimize 2n*max(conv(p,p)), p>=0, sum p=1.
    """
    n = len(a)
    p = a.astype(np.float64).copy()
    S = p.sum()
    if S < 1e-12:
        p[:] = 1.0 / n
    else:
        p = p / S
    best_p = p.copy()
    best_sc = compute_score_norm(p)
    sc_lb = 1.0  # Theoretical lower bound (2n/(2n-1) ~ 1)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(p)
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M
        if sc < best_sc:
            best_sc = sc
            best_p = p.copy()

        # Subgradient of max C[k] at k = k_star
        # dC[k_star]/dp_j = 2*p[k_star - j] (if valid) or p[j] if k_star=2j
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            for j in range(j_start, j_end + 1):
                m = k_star - j
                if m == j:
                    sg[j] += p[m]
                else:
                    sg[j] += 2.0 * p[m]

        # Normalize: project onto sum=1, p>=0
        gn = float(np.linalg.norm(sg))
        if gn < 1e-15:
            break
        # Step size: (sc - sc_lb) / ||g||^2
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.1 / (1.0 + 0.005 * it))
        p -= step * sg
        p = np.clip(p, 0.0, 1e6)
        S_new = p.sum()
        if S_new < 1e-12:
            p[:] = 1.0 / n
        else:
            p = p / S_new

    return best_sc, best_p


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to the peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if total_reduced > 0:
        per = total_reduced / max(1, len(bottom_idx))
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1e6)


def build_diverse_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    # From parent
    if parent is not None:
        p = parent.astype(np.float64)
        if len(p) != n:
            p = np.interp(np.linspace(0, len(p) - 1, n), np.arange(len(p)), p)
        p = np.clip(p, 0.0, 1e6)
        if p.sum() > 1e-6:
            inits.append(p / p.sum())

    x = np.arange(n, dtype=np.float64) / n
    # Uniform
    inits.append(np.ones(n) / n)
    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    # Reverse ramp
    inits.append(ramp[::-1] / ramp.sum())
    # Quadratic
    quad = (x ** 2 + 0.1)
    inits.append(quad / quad.sum())
    # Cosine
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            cos = 1.0 + amp * np.cos(2.0 * np.pi * freq * x)
            cos = np.clip(cos, 0.01, None)
            inits.append(cos / cos.sum())
    # Exponential
    for rate in [1.0, 2.0, 3.0]:
        exp_a = np.exp(-rate * x)
        inits.append(exp_a / exp_a.sum())
    # Two-bump
    bump1 = np.exp(-50.0 * (x - 0.3) ** 2)
    bump2 = np.exp(-50.0 * (x - 0.7) ** 2)
    inits.append((bump1 + bump2) / (bump1 + bump2).sum())
    # Random
    for _ in range(3):
        r = np.clip(1.0 + 0.5 * rng.standard_normal(n), 0.01, None)
        inits.append(r / r.sum())
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0  # 20s buffer

    N = 8192  # Target length

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size > 0:
                p = np.clip(p, 0.0, 1e6)
                if p.size != N:
                    p = np.interp(np.linspace(0, p.size - 1, N), np.arange(p.size), p)
                parent = p
                print(f"Parent loaded: length={p.size}, sum={p.sum():.2f}, score={compute_score(p):.8f}")
    except Exception as e:
        print(f"Parent loading failed: {e}")

    best_score = np.inf
    best_a = np.ones(N) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # ==================================================================
    # Phase 1: Multiplicative equalization from parent + diverse inits
    # ==================================================================
    phase1_deadline = t_start + 0.30 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 1: Multiplicative Equalization (t < {phase1_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    # Start from parent if available
    candidates = []
    if parent is not None:
        p_norm = parent / parent.sum()
        candidates.append(p_norm.copy())

    # Add a few diverse inits
    inits = build_diverse_inits(N, rng, parent)
    # Take first few diverse ones
    for a_init in inits[:5]:
        candidates.append(a_init.copy())

    # Run multiplicative equalization with various parameters
    me_configs = [
        (200, 0.3, 0.5),   # 200 iters, alpha=0.3, active_frac=50%
        (200, 0.5, 0.3),   # 200 iters, alpha=0.5, active_frac=30%
        (200, 0.3, 0.2),   # 200 iters, alpha=0.3, active_frac=20%
        (150, 0.7, 0.1),   # 150 iters, alpha=0.7, active_frac=10%
        (150, 0.5, 0.5),   # 150 iters, alpha=0.5, active_frac=50%
    ]

    for ci, cand in enumerate(candidates):
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n  Candidate {ci} (score={compute_score(cand):.8f}):")
        a_cur = cand.copy()
        for n_it, alpha, af in me_configs:
            if time.monotonic() > phase1_deadline:
                break
            sc, a_eq = multiplicative_equalize_v2(a_cur, n_it, alpha, af, phase1_deadline)
            report(sc, a_eq, f"ME c={ci} it={n_it} a={alpha} af={af}")
            a_cur = a_eq

    print(f"\nAfter Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LSE-smoothed L-BFGS continuation (high beta)
    # ==================================================================
    phase2_deadline = t_start + 0.65 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: LSE L-BFGS (t < {phase2_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    beta_schedule = [500, 1000, 2000, 5000, 10000, 20000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 10:
            break
        maxit = min(5000, int(remaining * 20))
        maxf = min(50000, int(remaining * 40))
        sc, a_opt = run_lbfgs_norm(a_cur, beta, maxit, maxf, phase2_deadline)
        report(sc, a_opt, f"LSE-LBFGS beta={beta}")
        a_cur = a_opt

    # Try from a few other candidates too
    if parent is not None:
        for beta in [2000, 5000]:
            if time.monotonic() > phase2_deadline:
                break
            remaining = phase2_deadline - time.monotonic()
            if remaining < 10:
                break
            a_try = parent / parent.sum()
            # Quick ME first
            sc_me, a_me = multiplicative_equalize_v2(a_try, 100, 0.4, 0.3, phase2_deadline)
            maxit = min(3000, int(remaining * 15))
            sc, a_opt = run_lbfgs_norm(a_me, beta, maxit, maxit * 10, phase2_deadline)
            report(sc, a_opt, f"parent-LSE beta={beta}")

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Local search with perturbations
    # ==================================================================
    phase3_deadline = t_start + 0.85 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 3: Local Search (t < {phase3_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    n_local = 30
    for trial in range(n_local):
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 15:
            break
        a_base = best_a.copy()
        strat = trial % 6

        if strat == 0:
            # Gaussian
            scale = 0.01 + 0.03 * rng.random()
            a_pert = a_base * (1.0 + scale * rng.standard_normal(N))
        elif strat == 1:
            # Targeted
            a_pert = targeted_perturb(a_base, rng, strength=0.02 + 0.04 * rng.random())
        elif strat == 2:
            # Sinusoidal
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            x = np.arange(N, dtype=np.float64) / N
            a_pert = a_base * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
        elif strat == 3:
            # Quick ME + noise
            sc_tmp, a_eq = multiplicative_equalize_v2(a_base, 30, 0.6, 0.2, phase3_deadline)
            a_pert = a_eq * (1.0 + 0.02 * rng.standard_normal(N))
        elif strat == 4:
            # Asymmetry boost: make more asymmetric
            rev = a_base[::-1].copy()
            asym = a_base - rev
            a_pert = a_base + 0.05 * asym / (np.abs(asym).max() + 1e-10)
        else:
            # Block swap
            a_pert = a_base.copy()
            bs = max(16, N // 64)
            nb = N // bs
            if nb > 2:
                i = rng.integers(0, nb - 1)
                j = rng.integers(0, nb - 1)
                if i != j:
                    tmp = a_pert[i*bs:(i+1)*bs].copy()
                    a_pert[i*bs:(i+1)*bs] = a_pert[j*bs:(j+1)*bs]
                    a_pert[j*bs:(j+1)*bs] = tmp

        a_pert = np.clip(a_pert, 0.0, 1e6)

        # Quick L-BFGS polish
        beta_ls = 5000
        maxit = min(2000, int(remaining * 2))
        sc, a_opt = run_lbfgs_norm(a_pert, beta_ls, maxit, maxit * 10, phase3_deadline)
        improved = report(sc, a_opt, f"local #{trial} s={strat}")
        if improved:
            # Follow up with higher beta
            if time.monotonic() < phase3_deadline - 10:
                sc2, a_opt2 = run_lbfgs_norm(a_opt, 15000, 2000, 20000, phase3_deadline)
                report(sc2, a_opt2, f"local #{trial} polish")

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Final ultra-high-beta polish
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Polish")
    print(f"{'='*60}")

    for beta in [30000, 50000, 100000]:
        if time.monotonic() > deadline - 15:
            break
        sc, a_opt = run_lbfgs_norm(best_a, beta, 3000, 30000, deadline - 15)
        report(sc, a_opt, f"final beta={beta}")

    # Subgradient polish
    if time.monotonic() < deadline - 10:
        sc, a_pol = subgradient_polish(best_a, 500, deadline - 5)
        report(sc, a_pol, "subgrad")

        # L-BFGS from subgradient result
        if time.monotonic() < deadline - 5:
            sc, a_opt = run_lbfgs_norm(a_pol, 50000, 2000, 20000, deadline - 2)
            report(sc, a_opt, "post-subgrad LSE")

    # One final attempt
    if time.monotonic() < deadline - 3:
        sc, a_opt = run_lbfgs_norm(best_a, 20000, 1000, 10000, deadline - 1)
        report(sc, a_opt, "absolute final")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * N

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
