"""
Search for a non-negative sequence minimizing
    2*n*max(conv(a,a))/(sum(a)^2)

Unlike the previous attempt, this implementation:
  - works on the simplex (sum=1) through softmax/log parameters, removing scale drift;
  - uses LSE-smoothed max-convolution with beta scaled by n (normalized convolution scale);
  - explores a low-dimensional anti-palindromic Fourier subspace for global warm starts;
  - refines with high-dimensional L-BFGS and mirror-descent / multiplicative active-set polishing.
"""

import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def convolution_fft(a):
    """Linear self-convolution of a real sequence by zero-padded FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    M = 2 * n - 1
    N = next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)
    return out[:M]


def correlation_p_a(p, a):
    """
    h_i = sum_k p_k a_{k-i}, zero-padded a.
    This is the convolution-gradient term for c = a * a.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    p = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    M = 3 * n - 2
    N = next_pow2(M)
    ar = a[::-1]
    AR = np.fft.rfft(ar, N)
    P = np.fft.rfft(p, N)
    out = np.fft.irfft(AR * P, N)
    return out[n - 1:2 * n - 1]


def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = convolution_fft(a)
    if not np.all(np.isfinite(c)):
        return np.inf
    M = float(np.max(c))
    if M <= 0.0:
        return np.inf
    return float(2.0 * n * M / (s * s))


def resample_sequence(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if a.size == n:
        return normalize_sum1(a)
    old = np.linspace(0.0, a.size - 1, a.size)
    new = np.linspace(0.0, n - 1, n)
    a2 = np.interp(new, old, a)
    return normalize_sum1(a2)


def candidate_sort_key(item):
    return item[0]


def fourier_basis(n, R):
    """
    Anti-palindromic cosine basis on half-grid:
      B[:,j] = cos(pi * m * x_i), m = 1,3,5,...
    This subspace is useful because score < 2 requires non-palindromic structure.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    B = np.empty((n, R), dtype=np.float64)
    for j in range(R):
        m = 2 * j + 1
        B[:, j] = np.cos(np.pi * m * x)
    return B


def fourier_sequence(n, r, B):
    r = np.asarray(r, dtype=np.float64)
    z = B.dot(r)
    z = z - float(z.max())
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def fit_fourier(a, B):
    a = normalize_sum1(a)
    la = np.log(np.maximum(a, 1e-12 / max(1, a.size)))
    la = la - float(la.mean())
    try:
        r, _res, _rank, _sv = np.linalg.lstsq(B, la, rcond=None)
    except Exception:
        r = np.zeros(B.shape[1], dtype=np.float64)
    r = np.asarray(r, dtype=np.float64)
    return np.clip(r, -2.0, 2.0)


def make_fourier_inits(n, R, B, parent, rng):
    inits = []
    z = np.zeros(R, dtype=np.float64)

    # Strong anti-palindromic first mode.
    for s in (0.40, 0.60, 0.80, 1.10):
        r = z.copy()
        r[0] = s
        inits.append(r)

    # Smooth random anti-palindromic shapes.
    for amp in (0.65, 1.05, 1.45):
        r = amp * rng.standard_normal(R)
        # Keep at least some first mode, but allow signs.
        r[0] = abs(r[0]) + 0.25
        decay = 1.0 / np.sqrt(np.arange(1, R + 1, dtype=np.float64) + 1.0)
        r = r * decay
        inits.append(np.clip(r, -2.0, 2.0))

    # Fit parent, if available.
    if parent is not None:
        a = resample_sequence(parent, n)
        inits.append(fit_fourier(a, B))

    # Exponential ramps, fit to anti-cosine basis.
    x = np.arange(n, dtype=np.float64) / float(n)
    for lam in (1.0, 2.5, 4.0):
        for sign in (1.0, -1.0):
            a = np.exp(sign * lam * (x - 0.5))
            a = a / float(a.sum())
            inits.append(fit_fourier(a, B))

    # Keep a small, diverse set.
    return inits[:8]


def fourier_objective(x, args):
    n, beta, B, scale = args
    x = np.asarray(x, dtype=np.float64)
    x = np.clip(x, -2.5, 2.5)

    z = B.dot(x)
    z = z - float(z.max())
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return 1e100, np.zeros(len(x), dtype=np.float64)

    a = e / s
    c = convolution_fft(a)
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return 1e100, np.zeros(len(x), dtype=np.float64)

    zc = beta * (c - cmax)
    zc = zc - float(zc.max())
    ez = np.exp(zc)
    sz = float(ez.sum())
    if not np.isfinite(sz) or sz <= 0.0:
        return 1e100, np.zeros(len(x), dtype=np.float64)

    L = cmax + float(np.log(sz)) / beta
    score = 2.0 * n * L

    p = ez / sz
    h = correlation_p_a(p, a)
    E = float(np.dot(a, h))
    base = a * (h - E)
    grad = (4.0 * n) * (B.T.dot(base))

    return scale * score, scale * grad


def run_fourier_phase(n, R, deadline, parent, rng, beta_factors,
                      maxfun_per_run, max_inits):
    B = fourier_basis(n, R)
    inits = make_fourier_inits(n, R, B, parent, rng)
    inits = inits[:max_inits]

    bounds = [(-2.0, 2.0)] * R
    candidates = []
    best_sc = np.inf
    best_a = None
    scale = 10000.0

    for r0 in inits:
        if time.monotonic() > deadline:
            break
        r_cur = np.clip(np.asarray(r0, dtype=np.float64).copy(), -2.0, 2.0)

        for f in beta_factors:
            t0 = time.monotonic()
            remaining = deadline - t0
            if remaining < 8.0:
                break

            beta = float(f) * float(n)
            per_eval = 0.0012 if n <= 4096 else (0.0025 if n <= 8192 else 0.006)
            time_budget = min(25.0, max(6.0, 0.35 * remaining))
            maxfun = int(min(maxfun_per_run, max(80, time_budget / per_eval)))
            maxiter = max(200, maxfun)

            options = {
                "maxiter": maxiter,
                "maxfun": maxfun,
                "maxls": 18,
                "ftol": 1e-14,
                "gtol": 1e-8
            }

            try:
                res = minimize(
                    fourier_objective,
                    r_cur,
                    bounds=bounds,
                    args=(n, beta, B, scale),
                    method="L-BFGS-B",
                    options=options
                )
                if res is not None and np.all(np.isfinite(res.x)):
                    r_cur = np.clip(np.asarray(res.x, dtype=np.float64), -2.0, 2.0)
            except Exception:
                pass

            a = fourier_sequence(n, r_cur, B)
            sc = compute_score(a)
            if np.isfinite(sc):
                candidates.append((sc, a.copy()))
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()

    # Keep only a few strong candidates.
    candidates.sort(key=candidate_sort_key)
    return best_sc, best_a, candidates[:5]


def high_initial_y(a, bound):
    a = normalize_sum1(a)
    n = len(a)
    tiny = 1e-12 / max(1, n)
    la = np.log(np.maximum(a, tiny))
    y = la - la[0]
    rng = float(y.max() - y.min())
    if rng > 0.9 * (2.0 * bound):
        y = y * (0.9 * (2.0 * bound) / rng)
    return np.clip(y, -bound, bound)


def high_reconstruct(y, n):
    u = np.empty(n, dtype=np.float64)
    u[0] = 0.0
    u[1:] = np.asarray(y, dtype=np.float64)
    umax = float(u.max())
    e = np.exp(u - umax)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def high_objective(x, args):
    n, beta, bound, scale = args
    x = np.asarray(x, dtype=np.float64)
    x = np.clip(x, -bound, bound)

    u = np.empty(n, dtype=np.float64)
    u[0] = 0.0
    u[1:] = x

    umax = float(u.max())
    e = np.exp(u - umax)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return 1e100, np.zeros(n - 1, dtype=np.float64)

    a = e / s
    c = convolution_fft(a)
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return 1e100, np.zeros(n - 1, dtype=np.float64)

    zc = beta * (c - cmax)
    zc = zc - float(zc.max())
    ez = np.exp(zc)
    sz = float(ez.sum())
    if not np.isfinite(sz) or sz <= 0.0:
        return 1e100, np.zeros(n - 1, dtype=np.float64)

    L = cmax + float(np.log(sz)) / beta
    score = 2.0 * n * L

    p = ez / sz
    h = correlation_p_a(p, a)
    E = float(np.dot(a, h))
    gu = (4.0 * n) * (a * (h - E))
    gy = gu[1:] - gu[0]

    return scale * score, scale * gy

def make_p_from_c(c, beta):
    M = float(np.max(c))
    if M <= 0.0 or not np.isfinite(M):
        return np.ones(len(c), dtype=np.float64) / float(len(c))

    if np.isinf(beta):
        tol = max(1e-12, 1e-8 * M)
        mask = c >= (M - tol)
        s = int(np.count_nonzero(mask))
        if s <= 0:
            p = np.zeros(len(c), dtype=np.float64)
            p[int(np.argmax(c))] = 1.0
            return p
        p = mask.astype(np.float64)
        p /= float(p.sum())
        return p

    z = beta * (c - M)
    z = z - float(z.max())
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(len(c), dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p
    return e / s


def multiplicative_update(a, p, eta):
    a = np.asarray(a, dtype=np.float64).ravel()
    h = correlation_p_a(p, a)
    h = np.maximum(h, 1e-300)
    E = float(np.dot(a, h))
    n = a.size
    if not np.isfinite(E) or E <= 1e-300:
        return a

    factor = (E / h) ** eta
    factor = np.clip(factor, 1e-10, 1e10)
    a2 = a * factor
    floor = 1e-12 / max(1, n)
    a2 = np.maximum(a2, floor)
    s = float(a2.sum())
    if not np.isfinite(s) or s <= 0.0:
        return a
    return a2 / s


def multiplicative_refine(a, beta_factors, iters_per, deadline):
    a_cur = normalize_sum1(a)
    n = len(a_cur)
    c = convolution_fft(a_cur)
    M = float(np.max(c))
    best_sc = 2.0 * n * M
    best_a = a_cur.copy()
    no_improve = 0

    for f in beta_factors:
        if f < 0:
            eta = 0.12
            beta = np.inf
        else:
            beta = float(f) * float(n)
            if f <= 10:
                eta = 0.45
            elif f <= 100:
                eta = 0.28
            else:
                eta = 0.15

        for _it in range(int(iters_per)):
            if time.monotonic() > deadline:
                break

            c = convolution_fft(a_cur)
            M = float(np.max(c))
            sc = 2.0 * n * M
            if np.isfinite(sc):
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a_cur.copy()
                    no_improve = 0
                else:
                    no_improve += 1

                if no_improve >= 50:
                    if eta < 0.03:
                        break
                    eta *= 0.55
                    no_improve = 0
                if no_improve >= 200:
                    break
            else:
                break

            if _it == int(iters_per) - 1:
                continue

            p = make_p_from_c(c, beta)
            a_cur = multiplicative_update(a_cur, p, eta)

    c = convolution_fft(a_cur)
    M = float(np.max(c))
    sc = 2.0 * n * M
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_a = a_cur.copy()

    return best_sc, best_a


def random_log_perturb(a, rng, scale):
    a = normalize_sum1(a)
    n = len(a)
    la = np.log(np.maximum(a, 1e-12 / max(1, n)))
    la = la - float(la.mean())
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    delta = np.zeros(n, dtype=np.float64)
    for m in (1, 3, 5, 7):
        amp = scale / (0.5 * float(m) + 1.0) * rng.uniform(0.25, 1.0)
        delta += amp * np.cos(np.pi * m * x)
    la += delta
    e = np.exp(la - float(la.max()))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return a
    return e / s

        if len(seen_lengths) <= 3 or nlen == 8192 or nlen == 16384:
            seed_deadline = min(phase2_deadline, time.monotonic() + max(120.0, 0.18 * total))
            sc, a = run_high_phase(a_seed, [20, 100, 400, 1500], hmax, seed_deadline)
            if np.isfinite(sc):
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"  high polish n={nlen}: NEW BEST {best_score:.10f} at {time.monotonic()-t_start:.0f}s")
                candidate_pool.append((sc, a))
        else:
            sc, a = run_high_phase(a_seed, [40, 200, 800], hmax // 2,
                                   min(phase2_deadline, time.monotonic() + 90.0))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"  high polish small n={nlen}: NEW BEST {best_score:.10f}")

    # ------------------------------------------------------------------
    # Phase 3: multiplicative active-set polishing.
    # ------------------------------------------------------------------
    print("\nPhase 3: multiplicative active-set refinement")
    if best_a is not None:
        iters = 130 if len(best_a) <= 8192 else 75
        sc, a = multiplicative_refine(best_a, [20, 100, 500, 2000, -1], iters, phase3_deadline)
        if np.isfinite(sc):
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"  multiplicative: NEW BEST {best_score:.10f}")

    # A few smooth random perturbations around best, then multiplicative polish.
    if best_a is not None and time.monotonic() < phase3_deadline:
        for _trial in range(3):
            if time.monotonic() > phase3_deadline:
                break
            a_pert = random_log_perturb(best_a, rng, scale=0.18)
            iters = 70 if len(a_pert) <= 8192 else 45
            sc, a = multiplicative_refine(a_pert, [80, 400, 2000, -1], iters, phase3_deadline)
            if np.isfinite(sc):
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"  random perturb polish: NEW BEST {best_score:.10f}")

    # ------------------------------------------------------------------
    # Final phase: high-beta L-BFGS + exact-active multiplicative.
    # ------------------------------------------------------------------
    print("\nFinal polish")
    if best_a is not None:
        nlen = len(best_a)
        if nlen <= 4096:
            fmax = 3000
        elif nlen <= 8192:
            fmax = 1800
        else:
            fmax = 900

        sc, a = run_high_phase(best_a, [500, 2000, 5000], fmax, final_deadline)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  final high: NEW BEST {best_score:.10f}")

        if time.monotonic() < final_deadline:
            sc, a = multiplicative_refine(best_a, [400, 2000, 8000, -1], 90 if nlen <= 8192 else 50,
                                          final_deadline)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"  final multiplicative: NEW BEST {best_score:.10f}")

        if time.monotonic() < final_deadline - 30:
            sc, a = run_high_phase(best_a, [2000, 8000], max(400, fmax // 2), final_deadline)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"  final second high: NEW BEST {best_score:.10f}")

    # Fallback if something failed.
    if best_a is None or len(best_a) == 0 or not np.all(np.isfinite(best_a)):
        n = 8192
        B = fourier_basis(n, 1)
        r = np.zeros(1, dtype=np.float64)
        r[0] = 0.8
        best_a = fourier_sequence(n, r, B)
        best_score = compute_score(best_a)
        print("FALLBACK used")

    # Final validation-safe output.
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    out = [float(max(0.0, min(1000.0, v))) for v in best_a]
    if len(out) < 1 or sum(out) < 0.01:
        out = [0.5] * 64

    verified = best_score
    try:
        ev = evaluate_sequence(out)
        if np.isfinite(ev):
            verified = float(ev)
            if verified < best_score:
                best_score = verified
    except Exception:
        pass

    print("\n" + "=" * 70)
    print(f"FINAL internal={best_score:.10f}, verified={verified:.10f}, n={len(out)}, "
          f"elapsed={time.monotonic()-t_start:.1f}s")
    print("=" * 70)

    return out
