Let me now think about the overall algorithm more carefully.

**LP-based iteration algorithm:**

```
Initialize a (normalized: sum = 1)
Repeat:
    M = max(conv(a, a))
    Solve LP: max sum(b) s.t. conv(a, b) <= M, b >= 0
    Get b*, S* = sum(b*)
    q = b* / S*  (so sum(q) = 1, and max(conv(a, q)) = M/S*)
    # Update
    a = (1-t)*a + t*q
    Normalize: a = a / sum(a)
```

The step size t is important. Too large and we might oscillate; too small and convergence is slow.

By the theory of the "bottleneck" method (or "relaxed projection"), we expect convergence as t → 0, but in practice a fixed t = 0.1-0.5 works well.

Actually, I recall that this is related to the "game solving" approach. The LP gives us the best response of "player 2" (the sequence designer) to the current "peak profile" determined by a. The update a ← (1-t)a + t*q is like a "follow the leader" strategy.

Let me think about convergence rate. In the worst case, this is O(1/t) iterations to get within 1/eps of optimal. For t=0.3, that's about 3/eps iterations. If eps = 0.0001, that's 30000 iterations. But each LP takes ~1 second, so that's 30000 seconds. Way too slow!

The LP max{sum(b) : Fb <= M, b >= 0} is equivalent (by the von Neumann minimax) to finding the value of the zero-sum game where:
- Row player chooses a peak index k (from 0 to 2n-2)
- Column player chooses a coordinate j (from 0 to n-1)
- Payoff matrix: A[k, j] = F[k, j] / M = f[k-j] / M (if valid, else 0)

The value of this game is 1/S*, where S* is the LP optimal value.

We can solve this game using **fictitious play** or **MWU**:

Fictitious Play:
1. Initialize p (row player's strategy) uniformly over k
2. for t = 1, ..., T:
   a. Column player's best response: q_t = e_j where j = argmin_j sum_k p[k] * A[k,j]
   b. Row player's best response: r_t = e_k where k = argmax_k sum_j q_t[j] * A[k,j]
   c. Update p = (p + r_t) / (t+1)
3. Output: q = (1/T) sum_t q_t (average column strategy)

This converges at rate O(sqrt(log(N)/T)) where N is the number of pure strategies. Here N_row = 2n-1 and N_col = n. For n=4096, that's sqrt(log(8191)/T) ≈ 3/sqrt(T). To get error 1e-4, we need T ≈ 10^8. Too slow.

"""
LP-Iterative optimizer for minimum peak convolution.
Uses the zero-sum game / LP relaxation approach for global optimization,
then L-BFGS with LSE smoothing for local refinement.
"""

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    """Cross-convolution of a and b (same length n)."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def build_conv_matrix(f):
    """Build matrix F such that F @ b = convolve(f, b)."""
    n = len(f)
    F = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            F[k, js:je+1] = f[k - np.arange(js, je+1)]
    return F

def solve_response_lp(f, M):
    """Solve: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    F = build_conv_matrix(f)
    b = cp.Variable(n)
    objective = cp.Maximize(cp.sum(b))
    constraints = [F @ b <= M, b >= 0]
    prob = cp.Problem(objective, constraints)
    # Try different solvers
    for solver in [cp.ECOS, cp.SCS, cp.OSQP]:
        try:
            prob.solve(solver=solver)
            if b.value is not None:
                return np.maximum(b.value, 0)
        except:
            continue
    return np.ones(n) / n  # fallback

def lp_iteration(a, max_iters=100, t=0.3, label=""):
    """Run LP-based iteration."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S  # normalize to sum = 1
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(max_iters):
        c = fast_conv(a)
        M = float(c.max())
        
        b = solve_response_lp(a, M)
        S_b = b.sum()
        if S_b < 1e-12:
            break
        q = b / S_b  # normalized response
        
        # Update
        a_new = (1 - t) * a + t * q
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # Adaptive step: if score worsened, reduce t
        if sc > best_score * (1 + 0.01):
            t = max(0.05, t * 0.7)
        elif sc < best_score * (1 - 0.001):
            t = min(0.8, t * 1.1)
        
        a = a_new
        print(f"    {label} iter {it}: score={sc:.8f}, t={t:.3f}")
    
    return best_a

"""
LP-Iterative optimizer for minimum peak self-convolution ratio.
Core idea: solve the zero-sum game between peak selection and sequence design
via successive LP relaxations, then polish with L-BFGS on LSE-smoothed objective.
"""

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution of a and b (both length n) via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_conv_matrix(f):
    """Build matrix F (size (2n-1) x n) such that F @ b = convolve(f, b)."""
    n = len(f)
    F = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            F[k, js:je + 1] = f[k - np.arange(js, je + 1)]
    return F


def solve_response_lp(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0. Returns b."""
    n = len(f)
    F = build_conv_matrix(f)
    b = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [F @ b <= M, b >= 0])
    
    # Try solvers in order of preference
    last_err = None
    for solver in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, verbose=False)
            if b.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
            last_err = f"solver={solver}, status={prob.status}"
        except Exception as e:
            last_err = str(e)
    # Fallback: return uniform
    print(f"    LP failed: {last_err}. Using uniform fallback.")
    return np.ones(n) / n

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_score = compute_score(a)
    t = t_init

    for it in range(max_iters):
        c = fast_conv(a)
        M = float(np.max(c))
        
        b = solve_response_lp(a, M)
        S_b = float(np.sum(b))
        if S_b < 1e-12:
            break
        q = b / S_b
        
        # Update with step t
        a_new = (1.0 - t) * a + t * q
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        improved = sc < best_score
        if improved:
            best_score = sc
            best_a = a_new.copy()
            t = min(0.8, t * 1.05)
        else:
            t = max(0.05, t * 0.75)
        
        a = a_new
        if it % 5 == 0 or improved:
            print(f"    {label} iter={it}: score={sc:.8f}, t={t:.3f}, best={best_score:.8f}")
    
    return best_a

    L = c_max + np.log(exp_sum) / beta
    p = exp_s / exp_sum
    
    score = 2.0 * n * L / (S * S)

        maxfun = maxiter * 20
    
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0 = np.ones(n) * 0.5
    
    def obj(x):
        s, g = lse_score_and_grad(x, beta)
        return s, np.maximum(g, -1e6)  # clip extreme gradients
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coordinate_descent(a, n_sweeps=5, n_top=100):
    """Coordinate descent targeting peak contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        # Find top contributors to c[k_star]
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old = a[j]
            if a_old < 1e-15:
                continue
            # Try reducing a[j] and redistributing
            for factor in [0.7, 0.8, 0.9]:
                delta = a_old * (1.0 - factor)
                a[j] = a_old * factor
                # Redistribute to bottom contributors
                bottom_local = np.argsort(contrib)[:max(1, top_count // 2)]
                per = delta / len(bottom_local)
                for li in bottom_local:
                    a[j_start + li] += per
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                else:
                    # Revert
                    a = best_a.copy() if sc > compute_score(a) + 1e-10 else a
                    a[j] = a_old * factor  # keep the change if it's not worse
                    break
            # Normalize
            S = a.sum()
            if S > 1e-12:
                a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def upseq(a, target_n):
    """Upscale sequence to target length via interpolation."""
    n = len(a)
    if n == target_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, target_n)
    return np.interp(x_new, x_old, a)


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            inits.append(upseq(parent, n).copy())
    
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2 * np.pi * x) ** 2)
    inits.append(0.2 + 0.8 * (2 * x - 1) ** 2)  # U-shaped
    inits.append(0.5 + 0.3 * np.cos(4 * np.pi * x) * np.cos(2 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, None))
    
    # Envelope: high at boundaries
    inits.append(0.1 + 0.9 * np.abs(2 * x - 1))
    inits.append(0.1 + 0.9 * (2 * x - 1) ** 4)
    
    return inits
