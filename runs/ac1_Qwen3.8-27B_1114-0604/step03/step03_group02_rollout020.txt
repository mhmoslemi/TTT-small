"""
Step function optimization via soft multiplicative equalization + adaptive L-BFGS.

Key improvements:
1. Soft active-set weighting with geometric continuation in softness parameter.
2. Line-search guaranteed monotone multiplicative updates.
3. Tight alternation between multiplicative equalization and L-BFGS refinement.
4. Deflation-based targeted local search on peak lags.
5. Multi-length exploration with diverse restarts and mutations.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def pressure_vector(a, w):
    """
    Compute p[j] = 2 * sum_k w[k] * a[k-j] * 1(valid)
    which is the gradient of sum_k w[k]*(a*a)[k] w.r.t. a[j].
    Uses FFT cross-correlation: p = 2 * (a ⋆ w)[0:n]
    """
    n = len(a)
    M_fft = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, M_fft)
    W = np.fft.rfft(w, M_fft)
    # (a ⋆ w)[j] = sum_m a[m]*w[m+j], FFT = conj(FFT(a)) * FFT(w)
    R = np.fft.irfft(np.conj(A) * W, M_fft)
    return 2.0 * R[:n]


def lse_score_grad(a, beta):
    """LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: 2 * (a ⋆ p)[0:n]
    M_fft = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, M_fft)
    P = np.fft.rfft(p, M_fft)
    R = np.fft.irfft(np.conj(A) * P, M_fft)
    g_L = 2.0 * R[:n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """Run L-BFGS-B to minimize the LSE-smoothed score."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_wrapper(x, b):
        return lse_score_grad(x, b)

    opts = {'ftol': 1e-16, 'gtol': 1e-14, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 80}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 3.0:
            return compute_score(a0), a0
        opts['maxfun'] = min(maxfun, max(200, int(remaining * 300)))
        opts['maxiter'] = min(maxiter, max(200, int(remaining * 80)))

    try:
        res = minimize(obj_wrapper, a0, args=(beta,),
                       method='L-BFGS-B', jac=True, bounds=bounds, options=opts)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out


def soft_multiplicative_step(a, beta_soft, alpha):
    """
    One soft multiplicative equalization step.
    1. Compute c = conv(a,a), M = max(c)
    2. Soft weights: w[k] = exp(beta_soft*(c[k]-M))
    3. Pressure: p[j] = 2*(a ⋆ w)[j]
    4. Update: a[j] <- a[j] * (mean(p)/p[j])^alpha (for a[j]>0)
    5. Renormalize to preserve sum.
    Returns updated a and the score.
    """
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return a.copy(), np.inf
    c = fast_conv(a)
    M = float(c.max())
    if M < 1e-15:
        return a.copy(), np.inf
    scaled = beta_soft * (c - M)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = (c >= 0.999 * M).astype(np.float64)
    else:
        w /= w_sum
    p = pressure_vector(a, w / w_sum)
    # For coordinates with a[j] > 0, update multiplicatively
    mask = a > 1e-15
    if mask.sum() == 0:
        return a.copy(), np.inf
    p_pos = p[mask]
    p_mean = p_pos.mean()
    if p_mean < 1e-30:
        return a.copy(), np.inf
    ratio = p_mean / np.maximum(p_pos, 1e-30)
    a_new = a.copy()
    a_new[mask] = a_new[mask] * ratio ** alpha
    a_new = np.clip(a_new, 0.0, 1000.0)
    S_new = a_new.sum()
    if S_new < 1e-12:
        return a.copy(), np.inf
    a_new = a_new * (S / S_new)
    return a_new, compute_score(a_new)

    n_beta_steps = 12
    betas = np.geomspace(30, 2000, n_beta_steps)
    alphas = np.linspace(0.6, 0.08, n_beta_steps)

    for bi in range(n_beta_steps):
        beta_s = betas[bi]
        alpha_base = alphas[bi]

        for sub_it in range(max_iters // n_beta_steps):
            it += 1
            if deadline is not None and time.monotonic() > deadline:
                break
            # Try update with current alpha
            a_trial, sc_trial = soft_multiplicative_step(a, beta_s, alpha_base)
            if sc_trial < sc - 1e-15:
                a = a_trial
                sc = sc_trial
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    if int((bi * (max_iters // n_beta_steps) + sub_it) % 5000) == 0:
                        print(f"    [mult] it={it}, beta_s={beta_s:.1f}, sc={sc:.8f}")
            else:
                # Backtrack: try smaller alpha
                alpha_trial = alpha_base * 0.5
                for _ in range(4):
                    if alpha_trial < 0.01:
                        break
                    a_trial, sc_trial = soft_multiplicative_step(a, beta_s, alpha_trial)
                    if sc_trial < sc - 1e-15:
                        a = a_trial
                        sc = sc_trial
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a.copy()
                        break
                    alpha_trial *= 0.5
                # If no improvement after backtracking, try even smaller effective step
                # by reducing beta_soft slightly (soften the active set)
                if time.monotonic() > deadline:
                    break

    return best_sc, best_a


def deflation_search(a, n_rounds=50, deadline=None):
    """
    Deflation-based local search: identify peak lags, reduce contributing
    coefficients, redistribute mass to low-convolution regions.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc = best_sc

    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))

        # Find top contributing indices for the peak lag
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            continue
        # Contribution of a[j] to (a*a)[k_star] is 2*a[k_star-j] (from our gradient formula)
        contrib = np.zeros(n)
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

        # Reduce top contributors, add to bottom
        top_k = max(2, n // 200)
        top_idx = np.argsort(contrib)[::-1][:top_k]
        bottom_mask = np.zeros(n, dtype=bool)
        # Bottom: low values that don't contribute much to peak
        all_contrib = contrib.copy()
        all_contrib[top_idx] = 0
        bottom_idx = np.argsort(all_contrib)[:top_k]

        reduce_frac = 0.02 + 0.03 * (rnd / max(1, n_rounds))
        total_moved = 0.0
        for j in top_idx:
            amt = a[j] * reduce_frac
            a[j] -= amt
            a[j] = max(a[j], 0.0)
            total_moved += amt
        if bottom_idx.size > 0 and total_moved > 0:
            for j in bottom_idx:
                a[j] += total_moved / len(bottom_idx)
        a = np.clip(a, 0.0, 1000.0)
        S_new = a.sum()
        if S_new < 1e-12:
            break
        a = a * (S / S_new)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        elif sc > best_sc * 1.05:
            # Revert if it got too bad
            a = best_a.copy()

    return best_sc, best_a


def coordinate_local_search(a, n_trials=200, deadline=None):
    """
    Random coordinate perturbation + mini L-BFGS polish.
    """
    n = len(a)
    rng = np.random.default_rng(12345)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc = best_sc

    for trial in range(n_trials):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Pick a random block of coordinates to perturb
        block_size = int(n ** 0.4) + 1  # ~40 for n=8192
        start = rng.integers(0, n - block_size)
        idx = np.arange(start, start + block_size)

        a_trial = a.copy()
        scale = 0.05 + 0.1 * rng.random()
        a_trial[idx] *= (1.0 + scale * rng.standard_normal(len(idx)))
        a_trial = np.clip(a_trial, 0.0, 1000.0)

        # Quick score check
        sc_trial = compute_score(a_trial)
        if sc_trial < best_sc:
            a = a_trial
            best_sc = sc_trial
            best_a = a_trial.copy()
            sc = sc_trial
            # Do a quick L-BFGS polish
            if deadline is not None and time.monotonic() < deadline - 5:
                sc_pol, a_pol = run_lbfgs(a_trial, 500, maxiter=1000, maxfun=20000,
                                           deadline=deadline if deadline else None)
                if sc_pol < best_sc:
                    best_sc = sc_pol
                    best_a = a_pol.copy()
                    a = a_pol.copy()
                    sc = sc_pol

    return best_sc, best_a


def subgradient_polish(a, total_iters=500, deadline=None):
    """Projected subgradient with Polyak step size and averaging."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0  # theoretical lower bound (approximately)
    avg_a = np.zeros(n)
    n_avg = 0

    for it in range(total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max at k_star
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            # grad component = 2*a[k_star - j] for j in [j_start, j_end]
            sg[j_start:j_end + 1] = 2.0 * a[k_star - j_start:k_star - j_start + (j_end - j_start + 1)][::-1]
        # Full gradient of the score
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

        # Polyak step
        gap = max(sc - sc_lb, 1e-10)
        gn2 = float(np.dot(grad, grad))
        step = gap / (gn2 + 1e-30)
        step = min(step, 1.0)
        # Damped step for stability
        step *= 0.5 / (1.0 + 0.05 * it)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        # Preserve sum
        S_new = a.sum()
        if S_new > 1e-12:
            a = a * (S / S_new)

        # Running average (helps with convergence)
        avg_a += a
        n_avg += 1

    # Also check the average
    if n_avg > 0:
        avg_a /= n_avg
        S_avg = avg_a.sum()
        if S_avg > 1e-12:
            sc_avg = compute_score(avg_a)
            if sc_avg < best_sc:
                best_sc = sc_avg
                best_a = avg_a.copy()

    return best_sc, best_a


def build_diverse_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        p = parent.astype(np.float64)
        if len(p) != n:
            p = np.interp(np.linspace(0, len(p) - 1, n), np.arange(len(p)), p)
        inits.append(np.clip(p, 0.01, 1000.0).copy())
        # Mutated parent
        a_m = p * (1.0 + 0.1 * rng.standard_normal(n))
        inits.append(np.clip(a_m, 0.01, 1000.0))
        # More aggressive mutation
        a_m2 = p * (1.0 + 0.3 * rng.standard_normal(n))
        inits.append(np.clip(a_m2, 0.01, 1000.0))
        # Sparse parent (zero out some)
        a_sp = p.copy()
        mask = rng.random(n) > 0.3
        a_sp[~mask] = 0.0
        if a_sp.sum() > 0.01:
            inits.append(np.clip(a_sp, 0.0, 1000.0))

    # Standard inits
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.5 * np.cos(4.0 * np.pi * x) ** 2)
    inits.append(0.2 + 0.6 * np.abs(np.sin(3.0 * np.pi * x)))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))

    # Binary-like
    a_bin = rng.random(n)
    a_bin[a_bin < 0.5] = 0.1
    a_bin[a_bin >= 0.5] = 1.0
    inits.append(a_bin.copy())

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Finds a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0  # 20s buffer

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def _try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = [8192]
    if parent is not None and parent.size != 8192:
        if parent.size <= 16384:
            lengths.insert(0, parent.size)
    lengths = sorted(set(lengths))
    # Add one smaller and one larger
    if 4096 not in lengths:
        lengths.append(4096)
    if 16384 not in lengths:
        lengths.append(16384)
    lengths.sort()

    print(f"Lengths: {lengths}, Parent: {parent is not None}")
    print(f"Budget: {total_budget}s, Deadline at t+{deadline-t_start:.0f}s")

    # ================================================================
    # Phase 1: Quick initialization - find a good starting point
    # ================================================================
    p1_deadline = t_start + 0.10 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 1: Initialization (deadline t+{p1_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    candidates = []  # list of (score, array)
    for n in lengths:
        if time.monotonic() > p1_deadline:
            break
        inits = build_diverse_inits(n, rng, parent if (parent is not None and len(parent) == n) else
                                    (parent if parent is not None else None))
        for i, a0 in enumerate(inits[:5]):
            if time.monotonic() > p1_deadline:
                break
            sc = compute_score(a0)
            candidates.append((sc, a0))
            _try_update(sc, a0, f"init n={n},i={i}")

    # Quick L-BFGS on top candidates
    candidates.sort(key=lambda x: x[0])
    for sc, a0 in candidates[:3]:
        if time.monotonic() > p1_deadline:
            break
        sc2, a2 = run_lbfgs(a0, 50, maxiter=2000, maxfun=30000, deadline=p1_deadline)
        _try_update(sc2, a2, f"init lbfgs n={len(a0)}")

    print(f"  After Phase 1: best={best_score:.8f}, n={len(best_a)}")

    # ================================================================
    # Phase 2: Main optimization - multiplicative equalization
    # ================================================================
    p2_deadline = t_start + 0.50 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: Multiplicative Equalization (deadline t+{p2_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    sc, a_eq = multiplicative_phase(a_cur, deadline=p2_deadline, max_iters=300000)
    _try_update(sc, a_eq, "mult phase")
    a_cur = a_eq if sc < best_score else best_a.copy()

    # ================================================================
    # Phase 3: L-BFGS refinement with increasing beta
    # ================================================================
    p3_deadline = t_start + 0.75 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 3: L-BFGS Refinement (deadline t+{p3_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    beta_list = [100, 300, 1000, 3000]
    for bi, beta in enumerate(beta_list):
        if time.monotonic() > p3_deadline:
            break
        print(f"  L-BFGS beta={beta}, t={time.monotonic()-t_start:.0f}s")
        sc2, a2 = run_lbfgs(a_cur, beta, maxiter=8000, maxfun=150000, deadline=p3_deadline)
        _try_update(sc2, a2, f"lbfgs beta={beta}")
        a_cur = a2 if sc2 < best_score else best_a.copy()

        # Interleave with multiplicative
        if time.monotonic() < p3_deadline - 30:
            sc3, a3 = multiplicative_phase(a_cur, deadline=min(p3_deadline - 10, time.monotonic() + 60),
                                            max_iters=50000)
            _try_update(sc3, a3, f"mult interleave after beta={beta}")
            a_cur = a3 if sc3 < best_score else best_a.copy()

    # ================================================================
    # Phase 4: Local search - deflation + coordinate + subgradient
    # ================================================================
    p4_deadline = t_start + 0.90 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 4: Local Search (deadline t+{p4_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = best_a.copy()

    # 4a: Deflation search
    if time.monotonic() < p4_deadline:
        sc, a_d = deflation_search(a_cur, n_rounds=100, deadline=min(p4_deadline, time.monotonic() + 120))
        _try_update(sc, a_d, "deflation")
        if sc < best_score:
            a_cur = a_d

    # 4b: Coordinate local search
    if time.monotonic() < p4_deadline:
        sc, a_c = coordinate_local_search(a_cur, n_trials=300,
                                           deadline=min(p4_deadline, time.monotonic() + 120))
        _try_update(sc, a_c, "coord search")
        if sc < best_score:
            a_cur = a_c

    # 4c: Subgradient polish
    if time.monotonic() < p4_deadline:
        sc, a_sg = subgradient_polish(a_cur, total_iters=500,
                                       deadline=min(p4_deadline, time.monotonic() + 60))
        _try_update(sc, a_sg, "subgrad")
        if sc < best_score:
            a_cur = a_sg

    # 4d: Another multiplicative pass
    if time.monotonic() < p4_deadline:
        sc, a_m2 = multiplicative_phase(a_cur, deadline=min(p4_deadline, time.monotonic() + 60),
                                         max_iters=100000)
        _try_update(sc, a_m2, "mult pass 2")
        if sc < best_score:
            a_cur = a_m2

    # ================================================================
    # Phase 5: Try other lengths from the best solution
    # ================================================================
    p5_deadline = deadline - 60
    print(f"\n{'='*60}")
    print(f"Phase 5: Cross-length exploration (deadline t+{p5_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    for n_try in lengths:
        if time.monotonic() > p5_deadline:
            break
        if n_try == len(best_a):
            continue
        # Resample best to new length
        a_resamp = np.interp(np.linspace(0, len(best_a) - 1, n_try),
                             np.arange(len(best_a)), best_a)
        a_resamp = np.clip(a_resamp, 0.0, 1000.0)
        sc = compute_score(a_resamp)
        _try_update(sc, a_resamp, f"resample to n={n_try}")

        if time.monotonic() < p5_deadline - 30:
            sc2, a2 = multiplicative_phase(a_resamp, deadline=min(p5_deadline, time.monotonic() + 120),
                                            max_iters=100000)
            _try_update(sc2, a2, f"mult n={n_try}")
            if sc2 < best_score:
                sc3, a3 = run_lbfgs(a2, 1000, maxiter=5000, maxfun=80000, deadline=p5_deadline)
                _try_update(sc3, a3, f"lbfgs n={n_try}")

    # ================================================================
    # Phase 6: Final polish
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final Polish")
    print(f"{'='*60}")

    for beta in [2000, 5000]:
        if time.monotonic() > deadline - 20:
            break
        sc2, a2 = run_lbfgs(best_a, beta, maxiter=5000, maxfun=80000, deadline=deadline - 10)
        _try_update(sc2, a2, f"final lbfgs beta={beta}")

    # Final subgradient
    if time.monotonic() < deadline - 10:
        sc2, a2 = subgradient_polish(best_a, total_iters=200, deadline=deadline - 5)
        _try_update(sc2, a2, "final subgrad")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
