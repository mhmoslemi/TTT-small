import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    # Gradient: d/da[j] of sum_k p[k]*c[k] where c = a*a
    # c[k] = sum_j a[j]*a[k-j]
    # d(c[k])/da[j] = a[k-j] (for valid k-j)
    # So grad[j] = sum_k p[k] * a[k-j] = (p * a_rev)[j] cross-correlation
    # Specifically: grad[j] = sum_k p[k] * a[k-j] where k-j in [0,n-1]
    # This is the cross-correlation of p and a, evaluated at specific lags
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj_func(a, beta):
    """Scipy-compatible objective."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_obj_func, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out, res.success


def fista_optimize(a0, beta, n_iter=5000, L=None):
    """FISTA (accelerated proximal gradient) for LSE-smoothed objective with a>=0 constraint."""
    n = len(a0)
    x = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    y = x.copy()
    t = 1.0
    
    if L is None:
        # Estimate Lipschitz constant
        # The Hessian of the LSE objective is bounded by ~ 2n * beta / S^2 * ||a||^2 roughly
        # Use a practical estimate
        test = x * (1 + 0.1 * np.random.standard_normal(n))
        test = np.clip(test, 0, 1000)
        s1, g1 = lse_score_grad(x, beta)
        s2, _ = lse_score_grad(test, beta)
        diff = np.linalg.norm(test - x)
        if diff < 1e-10:
            L = 10.0
        else:
            L = min(1e6, 2.0 * (s2 - s1) / (diff * diff) + 1.0)
    
    step = 1.0 / L
    best_sc = compute_score(x)
    best_x = x.copy()
    
    for it in range(n_iter):
        s, g = lse_score_grad(y, beta)
        # Gradient step
        x_new = y - step * g
        # Proximal step (project to a >= 0)
        x_new = np.clip(x_new, 0.0, 1000.0)
        # Backtracking if needed
        s_new = lse_score_grad(x_new, beta)[0]
        # Note: we skip backtracking for speed; the L estimate should be good enough
        y_new = x_new + ((t - 1) / (t + 1)) * (x_new - x)

        y = y_new
        t = 0.5 * (1 + np.sqrt(1 + 4 * t * t))

def subgrad_polish(a, max_iters=5000):
    """Nesterov-accelerated subgradient with adaptive step size."""
    n = len(a)
    x = a.astype(np.float64).copy()
    y = x.copy()
    t = 1.0
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    best_sc = compute_score(x)
    best_x = x.copy()
    
    # Initial step size estimate
    S = float(x.sum())
    c = fast_conv(x)
    M = float(c.max())
    sc = 2.0 * n * M / (S * S)
    step_init = (sc - sc_lb) / (n * M * 4.0 / (S * S))
    step_init = min(step_init, 1.0 / (10.0 * n * M / (S * S)))
    
    for it in range(max_iters):
        c = fast_conv(y)
        S = float(y.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        
        # Subgradient
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * y[j_start:j_end + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        
        # Polyak step size
        gap = max(sc - sc_lb, 1e-10)
        step = gap / gn2
        step = min(step, step_init / (1.0 + 0.01 * it))
        
        x_new = y - step * grad
        x_new = np.clip(x_new, 0.0, 1000.0)
        
        # Nesterov update
        t_new = 0.5 * (1 + np.sqrt(1 + 4 * t * t))
        y = x_new + ((t - 1) / t_new) * (x_new - x)
        y = np.clip(y, 0.0, 1000.0)
        x = x_new
        t = t_new
        
        # Track best
        if it % 50 == 0:
            sc_x = 2.0 * n * float(fast_conv(x).max()) / (float(x.sum()) ** 2)
            if sc_x < best_sc:
                best_sc = sc_x
                best_x = x.copy()
    
    final_sc = compute_score(x)
    if final_sc < best_sc:
        best_sc = final_sc
        best_x = x.copy()
    return best_sc, best_x

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LSE-continuation L-BFGS + FISTA + Nesterov-accelerated subgradient.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj_func(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_obj_func, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out, res.success


def fista_optimize(a0, beta, n_iter=5000):
    n = len(a0)
    x = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    y = x.copy()
    t = 1.0
    best_sc = compute_score(x)
    best_x = x.copy()
    
    # Estimate Lipschitz constant heuristically
    L = 2.0 * n * beta * float(np.max(a0) ** 2) / (float(a0.sum()) ** 2) + 10.0
    L = max(L, 1.0)
    step = 1.0 / L
    
    for it in range(n_iter):
        s, g = lse_score_grad(y, beta)
        x_new = y - step * g
        x_new = np.clip(x_new, 0.0, 1000.0)
        # Backtracking (occasionally)
        if it % 50 == 0 and it > 0:
            s_new = lse_score_grad(x_new, beta)[0]
            if s_new > s + 0.5 * L * float(np.linalg.norm(x_new - y) ** 2) + 1e-10:
                # Inaccurate, increase L
                L *= 2
                step = 1.0 / L
                x_new = y - step * g
                x_new = np.clip(x_new, 0.0, 1000.0)
        t_new = 0.5 * (1 + np.sqrt(1 + 4 * t * t))
