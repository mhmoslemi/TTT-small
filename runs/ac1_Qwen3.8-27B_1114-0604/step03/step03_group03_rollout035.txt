"""
Hybrid optimizer for minimum peak autocorrelation of non-negative sequences.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2.

Strategy:
  - Phase 1: Coordinate descent with exact 1D optimization (fast local refinement)
  - Phase 2: LP-based global moves via cutting-plane dual (escapes local minima)
  - Phase 3: LSE-smoothed L-BFGS refinement
  - Phase 4: Multi-strategy perturbation + CD cycles
  - Phase 5: Final polish

Key improvements over previous:
  - LP-based moves using cutting-plane on dual (avoids large sparse LP)
  - Multi-peak aware coordinate descent
  - Better perturbation strategies targeting peak structure
  - Adaptive time allocation
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar, linprog
from scipy import sparse
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def cross_correlation(f, y):
    """Compute (f star y)(i) = sum_j f[j]*y[i+j] for i=0..n-1.
    f has length n, y has length 2n-1."""
    n = len(f)
    fr = f[::-1]
    M = n + len(y) - 1
    M_fft = 1 << (M - 1).bit_length()
    FR = np.fft.rfft(fr, M_fft)
    Y = np.fft.rfft(y, M_fft)
    conv = np.fft.irfft(FR * Y, M_fft)
    return conv[n - 1:2 * n - 1]


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha
    if beta_coef is not None:
        conv_in = conv_in + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def coordinate_descent_sweep(a, rng=None, use_hot=False, n_sweeps=1):
    """
    Coordinate descent with incremental conv updates.
    use_hot: prioritize coordinates contributing to the peak.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    for _ in range(n_sweeps):
        if use_hot:
            k_star = int(np.argmax(conv))
            contrib = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
            order = np.argsort(contrib)[::-1]
        elif rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            i = int(i)
            a_i = a[i]

            k_lo = i
            k_hi = min(M_conv - 1, i + n - 1)

            # Outside max
            if k_lo > 0 and k_hi < M_conv - 1:
                outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
            elif k_lo > 0:
                outside_max = float(np.max(conv[:k_lo]))
            elif k_hi < M_conv - 1:
                outside_max = float(np.max(conv[k_hi + 1:]))
            else:
                outside_max = 0.0

            inside_len = k_hi - k_lo + 1
            base_inside = conv[k_lo:k_hi + 1].copy()

            ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
            a_vals = a[ks - i]
            is_2i = (ks == 2 * i)
            old_contrib = 2.0 * a_i * a_vals
            old_contrib[is_2i] = a_i * a_i
            base_inside -= old_contrib

            alpha = 2.0 * a_vals.copy()
            alpha[is_2i] = 0.0

            # beta_coef has exactly one non-zero at the 2i position
            t2_idx = np.where(is_2i)[0]
            t2_pos = int(t2_idx[0]) if len(t2_idx) > 0 else -1

            S_rest = S - a_i
            t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
            t_upper = min(t_upper, 1000.0)

            # Custom 1D optimization with the t^2 shortcut
            best_t = a_i
            best_val = coord_1d_obj(a_i, base_inside, alpha, None, outside_max, S_rest, n)
            if t2_pos >= 0:
                best_val = min(best_val, 2.0 * n * max(float(np.max(base_inside + a_i * alpha)),
                                                       base_inside[t2_pos] + a_i * alpha[t2_pos] + a_i * a_i,
                                                       outside_max) / (S_rest + a_i) ** 2)

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo, hi = 0.0, t_upper
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = _eval_1d(c_pt, base_inside, alpha, t2_pos, outside_max, S_rest, n)
            fd = _eval_1d(d_pt, base_inside, alpha, t2_pos, outside_max, S_rest, n)

            for _ in range(30):
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = _eval_1d(c_pt, base_inside, alpha, t2_pos, outside_max, S_rest, n)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = _eval_1d(d_pt, base_inside, alpha, t2_pos, outside_max, S_rest, n)

            t_opt = (lo + hi) / 2.0
            # Refine with a few more around the center
            for dt in [-0.5, -0.1, 0.0, 0.1, 0.5]:
                t_try = t_opt + dt
                if 0 <= t_try <= t_upper:
                    v = _eval_1d(t_try, base_inside, alpha, t2_pos, outside_max, S_rest, n)
                    if v < best_val:
                        best_val = v
                        t_opt = t_try

            a[i] = t_opt
            S = S_rest + t_opt
            conv_new_section = base_inside + t_opt * alpha
            if t2_pos >= 0:
                conv_new_section[t2_pos] += t_opt * t_opt
            conv[k_lo:k_hi + 1] = conv_new_section

    final_score = compute_score(a)
    return final_score, a


def _eval_1d(t, base_inside, alpha, t2_pos, outside_max, S_rest, n):
    """Fast 1D eval with t^2 shortcut."""
    conv_in = base_inside + t * alpha
    im = float(np.max(conv_in))
    if t2_pos >= 0:
        val_t2 = base_inside[t2_pos] + t * alpha[t2_pos] + t * t
        if val_t2 > im:
            im = val_t2
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_wrapper(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(

            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-14, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:

    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lp_global_move(f, M, max_cp_iter=80, time_limit=25):
    """
    Find g maximizing sum(g) subject to f*g <= M and g >= 0.
    Uses cutting-plane on the dual.
    Returns g or None if failed.
    """
    n = len(f)
    y_len = 2 * n - 1
    t_start = time.monotonic()

    # Initialize constraint set
    step = max(1, n // 40)
    S = list(range(0, n, step))

    y_opt = None
    for cp_iter in range(max_cp_iter):
        if time.monotonic() - t_start > time_limit:
            break

        # Build dual constraint matrix
        # Dual constraint i: sum_k f[k-i]*y[k] >= 1 (for valid k)
        # Standard form: -sum_k f[k-i]*y[k] <= -1
        n_c = len(S)
        rows = np.empty(n_c * n, dtype=np.int32)
        cols = np.empty(n_c * n, dtype=np.int32)
        vals = np.empty(n_c * n, dtype=np.float64)
        ptr = 0
        for idx, i in enumerate(S):
            k_start = i
            k_end = min(i + n, y_len)
            cnt = k_end - k_start
            rows[ptr:ptr + cnt] = idx
            cols[ptr:ptr + cnt] = np.arange(k_start, k_end, dtype=np.int32)
            vals[ptr:ptr + cnt] = -f[:cnt]
            ptr += cnt

        A_cp = sparse.csc_matrix((vals[:ptr], (rows[:ptr], cols[:ptr])), shape=(n_c, y_len))
        b_cp = -np.ones(n_c)
        c_obj = np.full(y_len, M)
        bounds = [(0, None)] * y_len

        try:
            res = linprog(c_obj, A_ub=A_cp, b_ub=b_cp, bounds=bounds,
                          method='highs', options={'time_limit': time_limit * 0.8,
                                                    'dual_feasibility_tolerance': 1e-7,
                                                    'primal_feasibility_tolerance': 1e-7})
        except Exception:
            break

        if res.status != 0:
            break

        y_opt = res.x

        # Check all constraints
        cc = cross_correlation(f, y_opt)
        violations = 1.0 - cc
        max_viol = float(np.max(violations))

        if max_viol < 1e-5:
            break

        # Add most violated
        i_new = int(np.argmax(violations))
        if i_new not in S:
            S.append(i_new)

        # Add a few random for diversity
        if len(S) < 300:
            for _ in range(5):
                i_r = int(np.random.randint(0, n))
                if i_r not in S:
                    S.append(i_r)

    if y_opt is None or np.max(y_opt) < 1e-10:
        return None

    # Find support from dual: g_i > 0 where cc(i) is close to max
    cc = cross_correlation(f, y_opt)
    cc_max = float(np.max(cc))
    threshold = 0.999 * cc_max
    support = np.where(cc >= threshold)[0]

    if len(support) == 0:
        support = np.arange(n)
    elif len(support) > n // 2:
        # Too many in support, use top half
        support = np.where(cc >= 0.9 * cc_max)[0]
        if len(support) > n // 2:
            support = np.arange(n)

    # Solve primal restricted to support
    s = len(support)
    c_p = np.ones(s)
    rows_p = np.empty(s * n, dtype=np.int32)
    cols_p = np.empty(s * n, dtype=np.int32)
    vals_p = np.empty(s * n, dtype=np.float64)
    ptr = 0
    for idx, i in enumerate(support):
        k_start = int(i)
        k_end = min(int(i) + n, 2 * n - 1)
        cnt = k_end - k_start
        rows_p[ptr:ptr + cnt] = np.arange(k_start, k_end, dtype=np.int32)
        cols_p[ptr:ptr + cnt] = idx
        vals_p[ptr:ptr + cnt] = f[:cnt]
        ptr += cnt

    A_p = sparse.csc_matrix((vals_p[:ptr], (rows_p[:ptr], cols_p[:ptr])), shape=(2 * n - 1, s))
    b_p = np.full(2 * n - 1, M)

    try:
        res_p = linprog(-c_p, A_ub=A_p, b_ub=b_p, bounds=[(0, None)] * s,
                        method='highs', options={'time_limit': time_limit * 0.5})
    except Exception:
        return None

    if res_p.status != 0:
        return None

    g_support = res_p.x
    g = np.zeros(n)
    g[support] = g_support

    if g.sum() < 1e-6:
        return None

    return g


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def anti_phase_perturb(a, rng, strength=0.03):
    """Create a perturbation that has anti-phase with the peak-contributing pattern."""
    n = len(a)
    c = fast_conv(a)
    top_k = min(10, (2 * n - 1) // 4)
    peak_lags = np.argsort(c)[::-1][:top_k]

    # Build anti-phase pattern
    pattern = np.zeros(n)
    for k in peak_lags:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            # Anti-phase: reduce where a[j]*a[k-j] is large
            pattern[j] -= a[j] * a[k - j]

    # Normalize
    pnorm = np.linalg.norm(pattern)
    if pnorm < 1e-10:
        return a.copy()
    pattern = pattern / pnorm * strength * np.mean(a)

    a_pert = a + pattern
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    return a_pert

    # ============================================================
    phase1_deadline = t_start + 0.25 * total_budget
    print(f"\n=== Phase 1: Coordinate Descent ({phase1_deadline - t_start:.0f}s) ===")

    a_cur = a_work.copy()
    sweep_count = 0
    no_improve = 0

    while time.monotonic() < phase1_deadline:
        sweep_count += 1
        if sweep_count % 3 == 0:
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng, use_hot=True)
        else:
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)

        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_cur.copy()
            no_improve = 0
            if sweep_count <= 5 or sweep_count % 10 == 0:
                print(f"  P1 sweep {sweep_count}: {sc:.8f} (NEW BEST)")
        else:
            no_improve += 1

        if sweep_count % 10 == 0:
            print(f"  P1 sweep {sweep_count}: {sc:.8f} (no_improve={no_improve})")

        if no_improve >= 30:
            a_cur = targeted_perturb(a_cur, rng, strength=0.03)
            no_improve = 0

    print(f"Phase 1 done: {sweep_count} sweeps, best={best_score:.8f}")

    # ============================================================
    # Phase 2: LP-based global moves (40% of time)
    # ============================================================
    phase2_deadline = t_start + 0.65 * total_budget
    print(f"\n=== Phase 2: LP Global Moves ({phase2_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    if a_cur.sum() > 0:
        a_cur = a_cur / a_cur.sum() * np.sqrt(2 * n_work)

    lp_count = 0
    lp_success = 0

    while time.monotonic() < phase2_deadline - 10.0:
        # Do a few CD sweeps first to get to local min
        sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng, use_hot=True)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_cur.copy()
            a_cur = a_cur / a_cur.sum() * np.sqrt(2 * n_work) if a_cur.sum() > 0 else a_cur

        # Try LP move
        M_cur = float(np.max(fast_conv(a_cur)))
        t_lp_start = time.monotonic()
        g = lp_global_move(a_cur, M_cur, max_cp_iter=60, time_limit=20)
        t_lp_elapsed = time.monotonic() - t_lp_start

        if g is not None and g.sum() > 0:
            lp_success += 1
            # Rescale g to same sum as a_cur
            g = g / g.sum() * a_cur.sum()

            # Line search on (1-t)*a_cur + t*g
            def line_search_obj(t):
                mix = (1 - t) * a_cur + t * g
                return compute_score(mix)

            res_ls = minimize_scalar(line_search_obj, bounds=(0.0, 1.0),
                                     method='bounded', options={'xatol': 1e-5})
            t_opt = float(res_ls.x)
            a_new = (1 - t_opt) * a_cur + t_opt * g
            sc_new = compute_score(a_new)

            lp_count += 1
            if lp_count <= 5 or lp_count % 10 == 0:
                print(f"  LP {lp_count} ({t_lp_elapsed:.1f}s, t={t_opt:.3f}): {sc_new:.8f}"
                      f"{' (NEW BEST)' if sc_new < best_score else ''}")

            if sc_new < best_score - 1e-10:
                best_score = sc_new
                best_a = a_new.copy()
                a_cur = a_new / a_new.sum() * np.sqrt(2 * n_work)
            else:
                # Try smaller step
                a_new2 = (1 - 0.3 * t_opt) * a_cur + 0.3 * t_opt * g
                sc_new2 = compute_score(a_new2)
                if sc_new2 < best_score - 1e-10:
                    best_score = sc_new2
                    best_a = a_new2.copy()
                    a_cur = a_new2 / a_new2.sum() * np.sqrt(2 * n_work)
        else:
            # LP failed, use perturbation instead
            strategy = lp_count % 4
            if strategy == 0:
                a_cur = targeted_perturb(a_cur, rng, strength=0.02 + 0.02 * rng.random())
            elif strategy == 1:
                a_cur = anti_phase_perturb(a_cur, rng, strength=0.02)
            elif strategy == 2:
                # Block redistribution
                block = max(5, n_work // 80)
                s = int(rng.integers(0, n_work - block))
                saved = a_cur[s:s + block].sum()
                a_cur = a_cur.copy()
                a_cur[s:s + block] *= 0.3
                rest = np.ones(n_work, dtype=bool)
                rest[s:s + block] = False
                a_cur[rest] += (saved * 0.7) / rest.sum()
            else:
                # Gaussian perturbation
                scale = 0.01 + 0.02 * rng.random()
                a_cur = a_cur * (1.0 + scale * rng.standard_normal(n_work))
                a_cur = np.clip(a_cur, 0.0, 1000.0)

            lp_count += 1

    print(f"Phase 2 done: {lp_count} moves ({lp_success} LP successes), best={best_score:.8f}")

    # ============================================================
    # Phase 3: LSE refinement (20% of time)
    # ============================================================
    phase3_deadline = t_start + 0.85 * total_budget
    print(f"\n=== Phase 3: LSE Refinement ({phase3_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    if a_cur.sum() > 0:
        a_cur = a_cur / a_cur.sum() * np.sqrt(2 * n_work)

    for beta in [100, 300, 800, 2000, 5000, 10000]:
        if time.monotonic() > phase3_deadline - 5.0:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=3000, maxfun=50000)
        elapsed = time.monotonic() - t0
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_opt.copy()
            a_cur = a_opt / a_opt.sum() * np.sqrt(2 * n_work) if a_opt.sum() > 0 else a_opt
            print(f"  LSE beta={beta}: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            print(f"  LSE beta={beta}: {sc:.8f} ({elapsed:.1f}s)")
            a_cur = a_opt  # still use for next iteration even if not better

    # ============================================================
    # Phase 4: Perturbation + CD cycles (10% of time)
    # ============================================================
    phase4_deadline = deadline
    print(f"\n=== Phase 4: Perturbation Cycles ({phase4_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    if a_cur.sum() > 0:
        a_cur = a_cur / a_cur.sum() * np.sqrt(2 * n_work)

    for trial in range(50):
        if time.monotonic() > phase4_deadline - 5.0:
            break

        strategy = trial % 6
        n_cur = len(a_cur)

        if strategy == 0:
            a_pert = targeted_perturb(a_cur, rng, strength=0.015)
        elif strategy == 1:
            a_pert = anti_phase_perturb(a_cur, rng, strength=0.015)
        elif strategy == 2:
            # Segment swap
            seg = max(10, n_cur // 50)
            s1 = int(rng.integers(0, max(1, n_cur - seg)))
            s2 = int(rng.integers(0, max(1, n_cur - seg)))
            if abs(s1 - s2) > seg:
                a_pert = a_cur.copy()
                a_pert[s1:s1 + seg], a_pert[s2:s2 + seg] = a_cur[s2:s2 + seg].copy(), a_cur[s1:s1 + seg].copy()
            else:
                a_pert = a_cur * (1 + 0.02 * rng.standard_normal(n_cur))
        elif strategy == 3:
            # Sinusoidal
            freq = 3.0 + 15.0 * rng.random()
            amp = 0.01 + 0.02 * rng.random()
            x = np.arange(n_cur, dtype=np.float64) / n_cur
            a_pert = a_cur * (1 + amp * np.cos(2 * np.pi * freq * x + 2 * np.pi * rng.random()))
        elif strategy == 4:
            # Mirror flip of a segment
            seg = max(20, n_cur // 20)
            s = int(rng.integers(0, max(1, n_cur - seg)))
            a_pert = a_cur.copy()
            a_pert[s:s + seg] = a_cur[s:s + seg][::-1]
        else:
            # Try LP again
            M_cur = float(np.max(fast_conv(a_cur)))
            g = lp_global_move(a_cur, M_cur, max_cp_iter=40, time_limit=10)
            if g is not None and g.sum() > 0:
                g = g / g.sum() * a_cur.sum()
                a_pert = 0.5 * a_cur + 0.5 * g
            else:
                a_pert = a_cur * (1 + 0.02 * rng.standard_normal(n_cur))

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick CD from perturbed point
        for sw in range(5):
            if time.monotonic() > phase4_deadline - 5.0:
                break
            sc_pert, a_pert = coordinate_descent_sweep(a_pert, rng=rng,
                                                       use_hot=(sw == 0))

        sc_pert = compute_score(a_pert)
        if sc_pert < best_score - 1e-10:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  P4 trial {trial} (s={strategy}): {sc_pert:.8f} (NEW BEST)")
        elif trial % 10 == 0:
            print(f"  P4 trial {trial} (s={strategy}): {sc_pert:.8f}")

    # ============================================================
    # Final polish
    # ============================================================
    print(f"\n=== Final Polish ===")
    a_cur = best_a.copy()
    if a_cur.sum() > 0:
        a_cur = a_cur / a_cur.sum() * np.sqrt(2 * n_work)

    for _ in range(30):
        if time.monotonic() > deadline - 3.0:
            break
        sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_cur.copy()

    # One more LSE if time permits
    if time.monotonic() < deadline - 15.0:
        a_tmp = best_a.copy()
        if a_tmp.sum() > 0:
            a_tmp = a_tmp / a_tmp.sum() * np.sqrt(2 * n_work)
        sc, a_opt = run_lbfgs(a_tmp, 5000, maxiter=3000, maxfun=40000)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final LSE: {best_score:.8f} (NEW BEST)")
        # Final CD
        a_cur = best_a.copy()
        if a_cur.sum() > 0:
            a_cur = a_cur / a_cur.sum() * np.sqrt(2 * n_work)
        sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  Final CD: {best_score:.8f} (NEW BEST)")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
