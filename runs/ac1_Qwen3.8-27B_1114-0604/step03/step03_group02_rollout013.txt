Step function optimization: LSE-weighted multiplicative equalization + L-BFGS continuation.

Key improvements over previous:
1. Soft (LSE-weighted) active set in multiplicative update instead of hard threshold
2. Adaptive step size with geometric decay
3. Interleaved L-BFGS refinement after every batch of multiplicative steps
4. Quasi-periodic initializations for diversity
5. Scale invariance exploitation (fix sum(a) = sqrt(2n))

import numpy as np
from scipy.optimize import minimize
import time

# Top-level helper functions (no closures)

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_objective(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    return 2.0 * n * L / (S * S)

def lse_gradient(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    g = 2.0 * conv_res
    return (2.0 * n) * (g / (S * S) - 2.0 * L / (S ** 3))

def lse_score_and_grad(a, beta):
    return lse_objective(a, beta), lse_gradient(a, beta)

    """Wrapper for scipy minimize: returns (f, g)."""
    return lse_score_and_grad(x, beta)

def multiplicative_lse_update(a, beta, n_iters, alpha_start, alpha_end, deadline):
    """
    LSE-weighted multiplicative update.
    Uses soft weights from LSE instead of hard active set.
    Adaptive alpha: geometric from alpha_start to alpha_end.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M_fft = 1 << (3 * n - 2).bit_length()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Adaptive alpha: geometric decay
        if n_iters > 1:
            frac = it / (n_iters - 1)
            alpha = alpha_start * (alpha_end / alpha_start) ** frac
        else:
            alpha = alpha_start
        
        c = fast_conv(a)
        c_max = float(c.max())
        
        # LSE weights
        scaled = beta * (c - c_max)
        exp_s = np.exp(scaled)
        p = exp_s / exp_s.sum()
        
        # Compute h[j] = 2 * sum_k p[k] * a[k-j] * 1[0<=k-j<n]
        # = 2 * (a_rev * p)[n-1 + j]
        rev_a = a[::-1]
        RA = np.fft.rfft(rev_a, M_fft)
        P = np.fft.rfft(p, M_fft)
        cross = np.fft.irfft(RA * P, M_fft)
        h = 2.0 * cross[n-1:2*n-1]
        
        # Ensure positive
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()
        
        # Multiplicative update
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)
        
        # Renormalize
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = S / n
            break
        a = a * (S / S_new)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

def run_lbfgs_lse(a0, beta, maxiter, maxfun, deadline):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)

        a_out = np.clip(res.x, 0.0, 1000.0)

        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

```python
"""
Optimized step function search for n=8192.
Uses LSE-weighted multiplicative equalization (soft active set) interleaved with L-BFGS-B.
Key innovation: the multiplicative update uses the full LSE distribution over lags
rather than a hard threshold, leading to smoother and more effective gradient steps.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution of a via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the exact evaluation score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_and_grad(x, beta):
    """
    LSE-smoothed objective and gradient for scipy minimize.
    Returns (f, g) tuple.
    """
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    ps = float(exp_s.sum())
    L = c_max + np.log(ps) / beta
    f = 2.0 * n * L / (S * S)
    p = exp_s / ps
    # Gradient of L w.r.t. x_j = 2 * sum_{m=0}^{n-1} x[m] * p[m+j]
    # = 2 * (x_rev * p)[n-1 + j]
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * P, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    # Full gradient: d(f)/dx_j = 2n * (g_L_j / S^2 - 2L / S^3)
    S2 = S * S
    g = (2.0 * n) * (g_L / S2 - 2.0 * L / (S2 * S))
    return f, g


def run_lbfgs(a0, beta, maxiter, maxfun, deadline):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_obj_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-14, 'gtol': 1e-10, 'maxiter': maxiter,
                     'maxfun': maxfun, 'maxls': 40}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out


def mul_lse_update(a, beta, n_iters, alpha_start, alpha_end, deadline):
    """
    LSE-weighted multiplicative update with adaptive step size.
    At each step, compute LSE weights over convolution lags,
    compute the 'pressure' on each element, and multiplicatively update.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    best_a = a.copy()
    best_sc = 2.0 * n * float(fast_conv(a).max()) / (S * S)
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    M2_fft = 1 << (3 * n - 2 - 1).bit_length()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        frac = it / max(1, n_iters - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        # Self-conv
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        c_max = float(c.max())
        
        # LSE weights (with numerical safety)
        scaled = beta * (c - c_max)
        # Clip to avoid underflow issues (not strictly needed but safe)
        np.clip(scaled, -700, 0, out=scaled)
        exp_s = np.exp(scaled)
        ps = float(exp_s.sum())
        if ps < 1e-300:
            p = np.zeros(M)
            p[int(np.argmax(c))] = 1.0
        else:
            p = exp_s / ps
        
        # Pressure: h[j] = 2 * sum_k p[k] * a[k-j] * 1[0<=k-j<n]
        rev_a = a[::-1]
        RA = np.fft.rfft(rev_a, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        cross = np.fft.irfft(RA * P, M2_fft)
        h = 2.0 * cross[n - 1:2 * n - 1]
        
        # Safety
        h = np.maximum(h, 1e-15)
        h_mean = float(h.mean())
        
        # Multiplicative update (damped)
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        np.clip(a, 0.0, 1000.0, out=a)
