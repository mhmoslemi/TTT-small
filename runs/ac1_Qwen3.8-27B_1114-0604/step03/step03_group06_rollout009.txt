        if gn < 1e-15:
            break
        
        # Nesterov: v = mu*v + grad; step = (sc - sc_lb) / gn^2
        v = mu * v + grad
        step = min((sc - sc_lb) / (float(np.dot(v, v)) + 1e-30), 0.5)
        a_new = a - step * v
        a_new = np.clip(a_new, 0.0, 1000.0)
        if a_new.sum() < 0.01:
            a_new = a.copy()
        a = a_new
    
    return best_sc, best_a


def exact_subgrad_plain(a, n_iters, t0_wall):
    """Plain projected subgradient with Polyak step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.4
    
    for it in range(n_iters):
        if time.monotonic() > t0_wall:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = min((sc - sc_lb) / (gn * gn + 1e-30), 0.3)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            a = a.copy()
    
    return best_sc, best_a


def topk_lse_obj_grad(a, k, beta_extra):
    """Top-k weighted LSE objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_sorted = np.sort(c)[::-1]
    if k < len(c_sorted):
        threshold = c_sorted[k - 1]
        gap = max(c_sorted[k - 1] - c_sorted[min(k + 2, len(c_sorted) - 1)], 1e-8)
        width = gap * 0.5 + 1e-8
        w = 1.0 / (1.0 + np.exp(-8.0 * (c - threshold) / width))
    else:
        w = np.ones(len(c))
    w = w / w.sum()
    L = float(np.dot(w, c))
    score = 2.0 * n * L / (S * S)
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PW = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(RA * PW, M2_fft)[:M2]
    CC = conv_res[n - 1:2 * n - 1]
    grad_L = 2.0 * CC
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs_topk(a0, k, maxiter, maxfun, maxls):
    """Run L-BFGS-B with top-k objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        topk_lse_obj_grad, a0, args=(k,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': maxls}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out
