def fast_conv(a):

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)

    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(k, n-1)
        row = np.zeros(n)
        idx = k - np.arange(j_start, j_end+1)

        A_rows.append(row)

"""
Hybrid optimizer: Alternating Linear Programming + L-BFGS + Evolutionary Polish.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovation: Uses alternating LP (each subproblem is a standard LP) as the
primary global search method, which is fundamentally different from the
gradient-based approaches in the previous state of the art. Each LP subproblem
finds the optimal sequence given the other "copy" is fixed.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def build_A_matrix(x, n):
    """Build (2n-1) x n matrix where A[k,j] = x[k-j] if 0<=k-j<n else 0."""
    k = np.arange(2*n-1)[:, None].astype(np.int64)
    j = np.arange(n)[None, :].astype(np.int64)
    diff = k - j
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2*n-1, n))
    valid_idx = diff[mask]
    A[mask] = x[valid_idx]
    return A

def solve_one_sided_lp(fixed_x, n):
    """
    Solve: min t s.t. (fixed_x * a)_k <= t for all k, sum(a)=1, a>=0, t>=0
    Returns (a_opt, t_opt) or (None, inf) if infeasible.
    """
    A = build_A_matrix(fixed_x, n)  # (2n-1) x n
    
    # Variables: [a_0, ..., a_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: A @ a - t <= 0
    # A_ub = [A | -1]
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n+1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.status != 0:
        return None, np.inf
    
    a_opt = np.clip(res.x[:n], 0, None)
    s = a_opt.sum()
    if s < 1e-12:
        return None, np.inf
    a_opt = a_opt / s
    t_opt = float(res.x[-1])
    return a_opt, t_opt

def alternating_lp(a_init, n_max_iters=30, rng=None, perturb_prob=0.1, perturb_scale=0.01):
    """
    Alternating LP method.
    minimizes max_k (x*x)_k subject to sum(x)=1, x>=0.
    """
    n = len(a_init)
    x = np.clip(a_init.astype(np.float64), 0, None).copy()
    x = x / x.sum()
    y = x.copy()
    
    best_x = x.copy()
    best_score = compute_score(x)
    
    if rng is None:
        rng = np.random.default_rng(0)
    
    for it in range(n_max_iters):
        # Update x: fix y, solve LP
        x_new, t_x = solve_one_sided_lp(y, n)
        if x_new is None:
            break
        x = x_new
        
        # Perturbation for diversity
        if rng.random() < perturb_prob:
            x = x + perturb_scale * rng.standard_normal(n)
            x = np.clip(x, 0, None)
            x = x / x.sum()
        
        # Update y: fix x, solve LP
        y_new, t_y = solve_one_sided_lp(x, n)
        if y_new is None:
            break
        y = y_new
        
        # Track best
        # The actual solution is the "merge" of x and y
        merged = (x + y) / 2
        merged = merged / merged.sum()
        sc = compute_score(merged)
        if sc < best_score:
            best_score = sc
            best_x = merged.copy()
        
        # Also check x and y individually
        sc_x = compute_score(x)
        if sc_x < best_score:
            best_score = sc_x
            best_x = x.copy()
        
        sc_y = compute_score(y)
        if sc_y < best_score:
            best_score = sc_y
            best_x = y.copy()
        
        # Convergence check
        if abs(compute_score(merged) - best_score) < 1e-10:
            break
    
    return best_x, best_score

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: d/d a_j of L = (2/n) * sum over peaks... 
    # Actually: dL/da_j = 2 * sum_k p_k * a_{k-j} (for valid k-j)
    # Let's compute this via FFT
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res  # dL/da_j
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad

def lse_objective(a, beta):
    return lse_score_grad(a, beta)

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    S = a0.sum()
    a0 = a0 / S  # normalize
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S > 1e-12:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def build_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64).copy(), 0.01, None))
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.clip(np.interp(x_new, x_old, parent), 0.01, None))
    
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, None))
    # U-shaped (counteracts triangular convolution)
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    # Bell-shaped

    # Multi-peak
    inits.append(0.3 + 0.7 * np.abs(np.cos(3.0 * np.pi * x)))
    return inits

def subgradient_polish(a, total_iters=3000):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0/n
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step size
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a

def coordinate_descent(a, n_sweeps=5, n_top=100):
    """Coordinate descent targeting peak contributors."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue
        
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old_j = a[j]
            # Try to reduce a[j]
            best_val = a_old_j
            best_local_sc = compute_score(a)
            for factor in [0.3, 0.5, 0.7, 0.1, 0.9]:
                cv = a_old_j * factor
                a[j] = cv
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

```python
"""
Hybrid optimizer: Alternating Linear Programming + Multi-Scale LP + L-BFGS + Local Search.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovation over previous SOTA: Uses alternating LP as primary global search.
Each LP subproblem (min t s.t. B@a <= t, sum(a)=1, a>=0) is solved exactly,
providing provably optimal "half-steps" that are fundamentally different from
gradient-based approaches. Multi-scale LP (small n → large n) provides good
initialization for the expensive large-n solves.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_A_matrix(x, n):
    """Build (2n-1) x n matrix where A[k,j] = x[k-j] if 0<=k-j<n else 0."""
    k = np.arange(2 * n - 1)[:, None].astype(np.int64)
    j = np.arange(n)[None, :].astype(np.int64)
    diff = k - j
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = x[diff[mask]]
    return A


def solve_one_sided_lp(fixed_x, n, time_limit=30):
    """
    Solve: min t s.t. (fixed_x * a)_k <= t for all k, sum(a)=1, a>=0, t>=0
    Returns (a_opt, t_opt) or (None, inf).
    """
    A = build_A_matrix(fixed_x, n)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None, np.inf

    if res.status != 0:
        return None, np.inf

    a_opt = np.clip(res.x[:n], 0, None)
    s = a_opt.sum()
    if s < 1e-12:
        return None, np.inf
    a_opt = a_opt / s
    t_opt = float(res.x[-1])
    return a_opt, t_opt

    """

    x = np.clip(a_init.astype(np.float64), 0, None).copy()
    s = x.sum()
    if s < 1e-12:
        x = np.ones(n) / n
    else:
        x = x / s
    y = x.copy()

    best_x = x.copy()
    best_score = compute_score(x)

    if rng is None:
        rng = np.random.default_rng(0)

    for it in range(n_max_iters):
        # Update x: fix y
        x_new, t_x = solve_one_sided_lp(y, n)
        if x_new is None:
            break
        x = x_new

        if rng.random() < perturb_prob:
            x = x + perturb_scale * rng.standard_normal(n)
            x = np.clip(x, 0, None)
            ss = x.sum()
            if ss < 1e-12:
                x = x_new.copy()
            else:
                x = x / ss

        # Update y: fix x
        y_new, t_y = solve_one_sided_lp(x, n)
        if y_new is None:
            break
        y = y_new

        # Check merged and individual
        merged = (x + y) / 2.0
        merged = merged / merged.sum()
        sc = compute_score(merged)
        if sc < best_score:
            best_score = sc
            best_x = merged.copy()

        sc_x = compute_score(x)
        if sc_x < best_score:
            best_score = sc_x
            best_x = x.copy()

        sc_y = compute_score(y)
        if sc_y < best_score:
            best_score = sc_y
            best_x = y.copy()

    return best_x, best_score


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for the exact objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S > 1e-12:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64).copy(), 0.01, None))
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.clip(np.interp(x_new, x_old, parent), 0.01, None))

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.clip(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01, None))
    inits.append(np.clip(0.5 + 0.3 * np.sin(2.0 * np.pi * x), 0.01, None))
    inits.append(np.clip(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2, 0.01, None))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, None))
    inits.append(np.clip(0.2 + 0.8 * (2.0 * x - 1.0) ** 2, 0.01, None))
    inits.append(np.clip(0.2 + 0.8 * np.exp(-4.0 * (2.0 * x - 1.0) ** 2), 0.01, None))
    inits.append(np.clip(0.3 + 0.7 * np.abs(np.cos(3.0 * np.pi * x)), 0.01, None))
    return inits


def subgradient_polish(a, total_iters=3000):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def coordinate_descent(a, n_sweeps=5, n_top=100):
    """Coordinate descent targeting peak contributors."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            best_val = a_old_j
            best_local_sc = compute_score(a)
            for factor in [0.3, 0.5, 0.7, 0.1, 0.85]:
                cv = a_old_j * factor
                a[j] = cv
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def error_reduction(a, n_iters=100, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        A_new = A + 0.7 * d
        a = np.fft.ifft(A_new).real[:n]
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12:
            break
        a = a / a.sum()

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_time = max(30.0, float(budget_s))
    deadline = t_start + total_time - 10.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f}  (n={len(a)}, t={time.monotonic()-t_start:.1f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}")
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"  Parent length: {len(parent)}, score: {sc_p:.10f}")
        report(sc_p, parent, "parent")

    # ==================================================================
    # Phase 1: Multi-Scale Alternating LP (45% of time)
    # ==================================================================
    phase1_deadline = t_start + 0.45 * total_time
    print(f"\n{'='*60}")
    print(f"Phase 1: Multi-Scale Alternating LP")
    print(f"{'='*60}")

    # Scale 1: small n for fast exploration
    n_small = 512
    print(f"\n  --- Scale 1: n={n_small} ---")
    inits_small = build_initializations(n_small, rng, parent)
    best_small = None
    best_small_sc = np.inf
    for i, a0 in enumerate(inits_small[:4]):
        if time.monotonic() > phase1_deadline:
            break
        a_lp, sc_lp = alternating_lp(a0, n_max_iters=25, rng=rng, time_per_lp=0.5)
        report(sc_lp, a_lp, f"LP small n={n_small} init={i}")
        if sc_lp < best_small_sc:
            best_small_sc = sc_lp
            best_small = a_lp.copy()

    # Scale 2: medium n
    n_med = 2048
    print(f"\n  --- Scale 2: n={n_med} ---")
    inits_med = build_initializations(n_med, rng, parent)
    # Add best from small scale interpolated
    if best_small is not None:
        x_old = np.linspace(0, 1, n_small)
        x_new = np.linspace(0, 1, n_med)
        inits_med.insert(0, np.clip(np.interp(x_new, x_old, best_small), 0.01, None))

    best_med = None
    best_med_sc = np.inf
    for i, a0 in enumerate(inits_med[:4]):
        if time.monotonic() > phase1_deadline:
            break
        a_lp, sc_lp = alternating_lp(a0, n_max_iters=20, rng=rng, time_per_lp=1.0)
        report(sc_lp, a_lp, f"LP med n={n_med} init={i}")
        if sc_lp < best_med_sc:
            best_med_sc = sc_lp
            best_med = a_lp.copy()

    # Scale 3: large n (most expensive)
    n_large = 4096
    print(f"\n  --- Scale 3: n={n_large} ---")
    inits_large = build_initializations(n_large, rng, parent)
    if best_med is not None:
        x_old = np.linspace(0, 1, n_med)
        x_new = np.linspace(0, 1, n_large)
        inits_large.insert(0, np.clip(np.interp(x_new, x_old, best_med), 0.01, None))
    if best_small is not None:
        x_old = np.linspace(0, 1, n_small)
        x_new = np.linspace(0, 1, n_large)
        inits_large.insert(1, np.clip(np.interp(x_new, x_old, best_small), 0.01, None))

    best_large = None
    best_large_sc = np.inf
    for i, a0 in enumerate(inits_large[:3]):
        if time.monotonic() > phase1_deadline:
            break
        a_lp, sc_lp = alternating_lp(a0, n_max_iters=15, rng=rng, time_per_lp=3.0,
                                     perturb_prob=0.2, perturb_scale=0.005)
        report(sc_lp, a_lp, f"LP large n={n_large} init={i}")
        if sc_lp < best_large_sc:
            best_large_sc = sc_lp
            best_large = a_lp.copy()

    print(f"\n  After Phase 1: best = {best_score:.10f}")

    # ==================================================================
    # Phase 2: L-BFGS Refinement (30% of time)
    # ==================================================================
    phase2_deadline = t_start + 0.75 * total_time
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS Refinement")
    print(f"{'='*60}")

    n_best = len(best_a)
    refine_lengths = sorted(set([n_best]))
    for nl in [4096, 5120, 6962, 8192, 10240]:
        if nl not in refine_lengths:
            refine_lengths.append(nl)
    if parent is not None:
        refine_lengths.append(len(parent))
        refine_lengths = sorted(set(refine_lengths))
    refine_lengths = refine_lengths[:5]

    for n_ref in refine_lengths:
        if time.monotonic() > phase2_deadline:
            break

        if len(best_a) == n_ref:
            a_start = best_a.copy()
        else:
            x_old = np.linspace(0, 1, len(best_a))
            x_new = np.linspace(0, 1, n_ref)
            a_start = np.clip(np.interp(x_new, x_old, best_a), 0.01, None)

        if parent is not None:
            if len(parent) == n_ref:
                a_parent = parent.copy()
            else:
                x_old = np.linspace(0, 1, len(parent))
                x_new = np.linspace(0, 1, n_ref)
                a_parent = np.clip(np.interp(x_new, x_old, parent), 0.01, None)
        else:
            a_parent = a_start

        for a_src, src_name in [(a_start, "best"), (a_parent, "parent")]:
            if time.monotonic() > phase2_deadline:
                break
            for beta in [100, 300, 1000]:
                if time.monotonic() > phase2_deadline:
                    break
                sc, a_opt, _ = run_lbfgs(a_src, beta, maxiter=5000, maxfun=100000)
                report(sc, a_opt, f"LBFGS n={n_ref} {src_name} b={beta}")

    print(f"\n  After Phase 2: best = {best_score:.10f}")

    # ==================================================================
    # Phase 3: Local Search with Diverse Perturbations (12% of time)
    # ==================================================================
    phase3_deadline = t_start + 0.87 * total_time
    print(f"\n{'='*60}")
    print(f"Phase 3: Local Search")
    print(f"{'='*60}")

    n_best = len(best_a)
    for trial in range(25):
        if time.monotonic() > phase3_deadline:
            break
        strategy = trial % 5
        if strategy == 0:
            scale = 0.02 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.01, None)
        elif strategy == 1:
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            seg = a_pert[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = max(10, n_best // 50)
            top_local = np.argsort(contrib)[::-1][:top_count]
            for li in top_local:
                gi = j_start + li
                a_pert[gi] *= (0.6 + 0.2 * rng.random())
            a_pert = np.clip(a_pert, 0.01, None)
        elif strategy == 2:
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.03 + 0.05 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.01, None)
        elif strategy == 3:
            a_pert = best_a.copy()
            n_blocks = 4 + int(rng.integers(0, 8))
            block_size = max(1, n_best // n_blocks)
            for b in range(min(n_blocks, n_best // block_size)):
                s = b * block_size
                e = min(s + block_size, n_best)
                a_pert[s:e] *= (0.5 + 0.8 * rng.random())
            a_pert = np.clip(a_pert, 0.01, None)
        else:
            x = np.arange(n_best, dtype=np.float64) / n_best
            f1 = 1 + 4 * rng.random()
            f2 = 2 + 8 * rng.random()
            a_pert = best_a * (1.0 + 0.05 * np.cos(2*np.pi*f1*x) + 0.03 * np.sin(2*np.pi*f2*x))
            a_pert = np.clip(a_pert, 0.01, None)

        a_pert = a_pert / a_pert.sum()
        sc, a_opt, _ = run_lbfgs(a_pert, 500, maxiter=3000, maxfun=50000)
        report(sc, a_opt, f"local #{trial} s={strategy}")

    print(f"\n  After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Final Polish (8% of time)
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Polish")
    print(f"{'='*60}")

    # Error reduction
    if time.monotonic() < deadline - 60:
        a_er = error_reduction(best_a, n_iters=8
