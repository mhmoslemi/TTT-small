def run_lbfgs_custom(a0, beta, maxiter=3000, maxfun=50000, time_limit=None, t_start=None):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def objective_with_time_check(x):

            if time.monotonic() - t_start > time_limit:
                # Raise to stop
                raise TimeoutError
        return lse_score_and_grad(x, beta)
    
    try:
        res = minimize(
            objective_with_time_check, a0, args=(),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-14, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40,
                     'disp': False}
        )
    except TimeoutError:
        # Return the best we had
        return compute_score(a0), a0, None
    except Exception:
        return compute_score(a0), a0, None
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def subgradient_polish(a, total_iters=1000, tol=1e-10):
    """Subgradient method with Backtracking line search for the exact max objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Get initial step size
    c = fast_conv(a)
    S = float(a.sum())
    M = float(c.max())
    sc = 2.0 * n * M / (S * S)
    step = 1.0 / (n + 1)
    
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max at k_star
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Step with decay
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 1.0 / (n * 0.1))
        step = max(step, 1e-10)
        
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        # Check improvement
        sc_new = compute_score(a_new)
        if sc_new < sc:
            a = a_new
        else:
            # Try smaller step
            a_trial = a - step * 0.1 * grad
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            sc_trial = compute_score(a_trial)
            if sc_trial < sc:
                a = a_trial
            else:
                step *= 0.5
                if step < 1e-12:
                    break
    
    return best_sc, best_a


def lag_targeted_perturb(a, rng, strength=0.05):
    """Perturb elements that contribute most to the peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    
    a_pert = a.copy()
    # Reduce top contributors, increase bottom ones (preserving sum approximately)
    top_count = max(5, n // 20)
    sorted_idx = np.argsort(contrib)[::-1]
    top_idx = sorted_idx[:top_count]
    bot_idx = sorted_idx[-top_count:]
    
    total_reduce = 0.0
    for j in top_idx:
        amt = a_pert[j] * strength
        a_pert[j] -= amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduce += amt
    
    if total_reduce > 0 and len(bot_idx) > 0:
        per = total_reduce / len(bot_idx)
        for j in bot_idx:
            a_pert[j] += per
    
    return np.clip(a_pert, 0.0, 1000.0)


def spectral_perturb(a, rng, strength=0.03):
    """Perturb in the frequency domain."""
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.fft(a, M_fft)
    # Add multiplicative noise to magnitude (preserve DC)
    noise_mag = 1.0 + strength * rng.standard_normal(len(A))
    noise_mag = np.clip(noise_mag, 0.5, 1.5)
    noise_mag[0] = 1.0
    A_new = A * noise_mag
    # Conjugate symmetry for real output
    A_new[n//2] = A_new[n//2].real if n % 2 == 0 else A_new[n//2]
    for k in range(1, n//2 + 1):
        A_new[n-k] = np.conj(A_new[k])
    a_pert = np.real(np.fft.ifft(A_new))[:n]
    return np.clip(a_pert, 0.0, 1000.0)


def multi_freq_perturb(a, rng, strength=0.04):
    """Perturb using random multi-frequency modulation."""
    n = len(a)
    x = np.arange(n, dtype=np.float64)
    n_freqs = rng.integers(2, 6)
    mod = np.zeros(n)
    for _ in range(n_freqs):
        f = rng.uniform(0.5, 10.0)
        phase = rng.uniform(0, 2 * np.pi)
        mod += np.cos(2 * np.pi * f * x / n + phase)
    mod /= n_freqs
    a_pert = a * (1.0 + strength * mod)
    return np.clip(a_pert, 0.0, 1000.0)


def block_perturb(a, rng, strength=0.1):
    """Perturb blocks of the sequence."""
    n = len(a)
    n_blocks = rng.integers(4, 16)
    block_size = n // n_blocks
    a_pert = a.copy()
    for _ in range(rng.integers(1, 4)):
        which = rng.integers(0, n_blocks)
        start = which * block_size
        end = min(start + block_size, n)
        a_pert[start:end] *= (1.0 + strength * rng.standard_normal())
    return np.clip(a_pert, 0.0, 1000.0)


def upsample(a, target_n):
    """Upsample a sequence to target length using linear interpolation."""
    n = len(a)
    if n == target_n:
        return a.copy()
    x_orig = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, target_n)
    a_up = np.interp(x_new, x_orig, a)
    return np.clip(a_up, 0.0, 1000.0)


def spectral_flatten_quick(a0, N, n_iters=100, lam=0.6):
    """Fast spectral flattening for initialization."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a

                new_val = 0.0
            a[idx] = new_val
            c_new = fast_conv(a)
            S_new = float(a.sum())
            M_new = float(c_new.max())
            if S_new < 1e-12:
                a[idx] = orig
                break
            sc_new = 2.0 * n * M_new / (S_new * S_new)
            if sc_new < best_sc:
                best_sc = sc_new
            else:
                a[idx] = orig
                break
    
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses LSE-continuation L-BFGS with aggressive beta values, multiple
    perturbation strategies, subgradient polishing, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0
    deadline = t_start + effective_budget

    best_score = np.inf
    best_a = np.ones(512) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.1f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                print(f"Loaded parent: n={p.size}, score={compute_score(parent):.8f}")
    except Exception:
        pass

    # Determine which lengths to try
    lengths_to_try = []
    if parent is not None:
        lengths_to_try.append(parent.size)
    # Always try a power of 2 near the parent size
    if parent is not None:
        p2 = 1 << (parent.size - 1).bit_length()
        if p2 != parent.size:
            lengths_to_try.append(p2)
    else:
        lengths_to_try.extend([2048, 4096, 8192])
    
    # Try one smaller and one larger
    if lengths_to_try:
        base_n = max(lengths_to_try)
        if base_n // 2 >= 512:
            lengths_to_try.append(base_n // 2)
        if base_n * 2 <= 16384:
            lengths_to_try.append(base_n * 2)
    
    lengths_to_try = sorted(set(lengths_to_try))[:4]
    print(f"Lengths to explore: {lengths_to_try}")

    # ================================================================
    # Phase 1: Quick LSE optimization from parent and spectral inits
    # ================================================================
    print(f"\n=== Phase 1: Initial LSE Optimization (t_budget={effective_budget:.0f}s) ===")
    
    phase1_deadline = t_start + 0.40 * effective_budget
    
    # Build initial candidates
    candidates = []
    if parent is not None:
        candidates.append(("parent", parent.copy()))
    
    for n in lengths_to_try:
        if time.monotonic() > phase1_deadline:
            break
        N_fft = 1 << (2 * n - 1).bit_length()
        
        # Spectral flattening from uniform
        sf1 = spectral_flatten_quick(np.ones(n) * 0.5, N_fft, n_iters=150, lam=0.6)
        candidates.append((f"sf_uni_{n}", sf1))
        
        # Spectral flattening from chirp
        x = np.arange(n, dtype=np.float64)
        chirp = np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01
        sf2 = spectral_flatten_quick(chirp, N_fft, n_iters=150, lam=0.5)
        candidates.append((f"sf_chirp_{n}", sf2))
        
        # Random init with spectral flattening
        rand_init = np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 5.0)
        sf3 = spectral_flatten_quick(rand_init, N_fft, n_iters=150, lam=0.5)
        candidates.append((f"sf_rand_{n}", sf3))
    
    # Sort candidates by score
    scored_candidates = []
    for label, a in candidates:
        sc = compute_score(a)
        scored_candidates.append((sc, label, a))
    scored_candidates.sort(key=lambda x: x[0])
    
    print(f"  {len(scored_candidates)} initial candidates, best: {scored_candidates[0][1]} = {scored_candidates[0][0]:.6f}")
    
    # Run LSE-LBFGS on top candidates with medium beta
    for sc_init, label, a_init in scored_candidates[:8]:
        if time.monotonic() > phase1_deadline:
            break
        t0 = time.monotonic()
        # Medium beta LSE
        sc, a_opt, _ = run_lbfgs_no_timeout(a_init, beta=200, maxiter=2000, maxfun=30000)
        report(sc, a_opt, f"P1 {label} beta=200")
        # Higher beta
        if time.monotonic() < phase1_deadline:
            sc, a_opt, _ = run_lbfgs_no_timeout(a_opt, beta=1000, maxiter=2000, maxfun=30000)
            report(sc, a_opt, f"P1 {label} beta=1000")
        elapsed = time.monotonic() - t0
        print(f"  {label}: {sc:.8f} ({elapsed:.1f}s)")

    print(f"  After Phase 1: best = {best_score:.8f}")

    # ================================================================
    # Phase 2: Aggressive high-beta LSE continuation
    # ================================================================
    print(f"\n=== Phase 2: High-Beta Continuation ===")
    
    phase2_deadline = t_start + 0.65 * effective_budget
    
    beta_schedule = [2000, 5000, 10000, 20000, 50000]
    
    a_cur = best_a.copy()
    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 10:
            break
        # Scale iterations with remaining time
        maxiter = min(5000, int(remaining * 0.7 / max(0.01, 0.005 * len(a_cur) / 1024)))
        maxfun = min(80000, maxiter * 25)
        sc, a_opt, res = run_lbfgs_no_timeout(a_cur, beta=beta, maxiter=max(500, maxiter), maxfun=max(2000, maxfun))
        report(sc, a_opt, f"P2 beta={beta}")
        a_cur = a_opt
        print(f"  beta={beta}: {sc:.8f}")
    
    print(f"  After Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: Diverse restarts with perturbation + re-optimization
    # ================================================================
    print(f"\n=== Phase 3: Diverse Restarts ===")
    
    phase3_deadline = t_start + 0.85 * effective_budget
    
    n_restarts = 12
    for i in range(n_restarts):
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 20:
            break
        
        # Choose perturbation strategy
        strategy = i % 5
        if strategy == 0:
            # Lag-targeted
            a_pert = lag_targeted_perturb(best_a, rng, strength=0.03 + 0.03 * rng.random())
        elif strategy == 1:
            # Spectral
            a_pert = spectral_perturb(best_a, rng, strength=0.02 + 0.03 * rng.random())
        elif strategy == 2:
            # Multi-frequency
            a_pert = multi_freq_perturb(best_a, rng, strength=0.03 + 0.03 * rng.random())
        elif strategy == 3:
            # Block
            a_pert = block_perturb(best_a, rng, strength=0.05 + 0.1 * rng.random())
        else:
            # Gaussian
            a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(len(best_a)))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        # Quick LSE optimization from perturbed point
        maxiter = min(3000, int(remaining * 0.4 / max(0.01, 0.005 * len(a_pert) / 1024)))
        sc, a_opt, _ = run_lbfgs_no_timeout(a_pert, beta=5000, maxiter=max(300, maxiter), maxfun=max(1000, maxiter * 20))
        report(sc, a_opt, f"P3 r{i} s{strategy}")
        
        # If this is good, do a higher-beta polish
        if sc < best_score + 0.001 and time.monotonic() < phase3_deadline - 10:
            sc2, a_opt2, _ = run_lbfgs_no_timeout(a_opt, beta=20000, maxiter=1000, maxfun=20000)
            report(sc2, a_opt2, f"P3 r{i} s{strategy} polish")

    print(f"  After Phase 3: best = {best_score:.8f}")

    # ================================================================
    # Phase 4: Subgradient polishing + final LSE
    # ================================================================
    print(f"\n=== Phase 4: Final Polish ===")
    
    # Subgradient
    sc_sg, a_sg = subgradient_polish(best_a, total_iters=500)
    report(sc_sg, a_sg, "subgrad")
    
    # LSE from subgradient result
    if time.monotonic() < deadline - 30:
        sc, a_opt, _ = run_lbfgs_no_timeout(a_sg, beta=50000, maxiter=3000, maxfun=50000)
        report(sc, a_opt, "post-sg LSE")
    
    # Another subgradient pass
    if time.monotonic() < deadline - 20:
        sc_sg2, a_sg2 = subgradient_polish(best_a, total_iters=300)
        report(sc_sg2, a_sg2, "subgrad2")
    
    # Coordinate descent
    if time.monotonic() < deadline - 30:
        a_cd = coordinate_descent_step(best_a, n_coords=200, max_inner=30)
        sc_cd = compute_score(a_cd)
        report(sc_cd, a_cd, "coord_desc")
        if sc_cd < best_score:
            # Polish the coordinate descent result
            sc, a_opt, _ = run_lbfgs_no_timeout(a_cd, beta=20000, maxiter=2000, maxfun=30000)
            report(sc, a_opt, "cd_polish")

    # ================================================================
    # Phase 5: Try upscaling from best smaller solution
    # ================================================================
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 5: Upscale Exploration ===")
        n_cur = len(best_a)
        n_target = n_cur * 2
        if n_target <= 16384:
            a_up = upsample(best_a, n_target)
            sc_up = compute_score(a_up)
            report(sc_up, a_up, f"upscale_{n_target}")
            if sc_up < 2.0:
                # Optimize the upscaled version
                maxiter = min(3000, int((deadline - time.monotonic() - 30) / 0.01))
                sc, a_opt, _ = run_lbfgs_no_timeout(a_up, beta=5000, maxiter=max(500, maxiter), maxfun=max(2000, maxiter * 20))
                report(sc, a_opt, f"upscale_opt_{n_target}")

    # ================================================================
    # Final verification and return
    # ================================================================
    # One final high-beta LSE
    if time.monotonic() < deadline - 10:
        sc_final, a_final, _ = run_lbfgs_no_timeout(best_a, beta=100000, maxiter=2000, maxfun=30000)
        report(sc_final, a_final, "final_LSE")
    
    # One final subgradient
    if time.monotonic() < deadline - 5:
        sc_sg3, a_sg3 = subgradient_polish(best_a, total_iters=200)
        report(sc_sg3, a_sg3, "final_sg")

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
