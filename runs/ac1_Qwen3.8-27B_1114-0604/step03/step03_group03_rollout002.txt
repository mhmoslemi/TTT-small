"""
Improved optimizer using basin hopping with peak-aware perturbations,
gradient flow, and enhanced coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def cd_single_coord(a, conv, i, S, n):
    """Optimize coordinate i, return new value and updated conv slice."""
    M_conv = 2 * n - 1
    a_i = a[i]
    k_lo = i
    k_hi = min(M_conv - 1, i + n - 1)
    
    if k_lo > 0 and k_hi < M_conv - 1:
        outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
    elif k_lo > 0:
        outside_max = float(np.max(conv[:k_lo]))
    elif k_hi < M_conv - 1:
        outside_max = float(np.max(conv[k_hi + 1:]))
    else:
        outside_max = 0.0
    
    inside_len = k_hi - k_lo + 1
    base_inside = conv[k_lo:k_hi + 1].copy()
    
    ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
    a_vals = a[ks - i]
    is_2i = (ks == 2 * i)
    old_contrib = 2.0 * a_i * a_vals
    old_contrib[is_2i] = a_i * a_i
    base_inside -= old_contrib
    
    alpha = 2.0 * a_vals.copy()
    alpha[is_2i] = 0.0
    beta_coef = np.zeros(inside_len)
    beta_coef[is_2i] = 1.0
    
    S_rest = S - a_i
    t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
    t_upper = min(t_upper, 1000.0)
    
    res = minimize_scalar(
        coord_1d_obj,
        args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
        bounds=(0.0, t_upper),
        method='bounded',
        options={'xatol': 1e-8}
    )
    t_opt = float(res.x)
    return t_opt, base_inside, alpha, beta_coef


def coordinate_descent_sweep(a, rng, n_sweeps=1):
    """Multiple sweeps of coordinate descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            i = int(i)
            t_opt, base_inside, alpha, beta_coef = cd_single_coord(a, conv, i, S, n)
            old_val = a[i]
            a[i] = t_opt
            S = S - old_val + t_opt
            conv[i:min(M_conv, i+n)] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef
    
    final_score = compute_score(a)
    return final_score, a


def peak_aware_perturb(a, rng, strength=0.03):
    """Perturb by reducing peak-contributing coords and increasing others."""
    n = len(a)
    c = fast_conv(a)
    # Find top few peak lags
    top_lags = np.argsort(c)[::-1][:5]
    
    # Compute total contribution of each a[i] to top peaks
    contrib = np.zeros(n)
    for k in top_lags:
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            contrib[i] += a[k - i]
    
    # Normalize contribution
    max_contrib = contrib.max()
    if max_contrib < 1e-10:
        return a.copy()
    
    # Reduce high-contribution, increase low-contribution
    a_pert = a.copy()
    threshold = max_contrib * 0.5
    high_mask = contrib > threshold
    low_mask = contrib < max_contrib * 0.2
    
    total_reduced = 0.0
    for i in np.where(high_mask)[0]:
        amt = a_pert[i] * strength * rng.random()
        a_pert[i] -= amt
        total_reduced += amt
    
    low_indices = np.where(low_mask)[0]
    if len(low_indices) > 0 and total_reduced > 0:
        per = total_reduced / len(low_indices)
        a_pert[low_indices] += per * rng.random(len(low_indices))
    
    return np.clip(a_pert, 0.0, 1000.0)


def segment_swap_perturb(a, rng):

    n = len(a)
    seg_len = int(rng.integers(max(5, n // 200), max(10, n // 20)))
    s1 = int(rng.integers(0, n - seg_len))
    s2 = int(rng.integers(0, n - seg_len))
    if abs(s1 - s2) < seg_len:
        s2 = (s2 + seg_len) % max(1, n - seg_len)
    a_pert = a.copy()
    tmp = a_pert[s1:s1+seg_len].copy()
    a_pert[s1:s1+seg_len] = a_pert[s2:s2+seg_len]
    a_pert[s2:s2+seg_len] = tmp
    return a_pert

def hole_punch_perturb(a, rng):
    """Zero out a random segment and redistribute to others."""
    n = len(a)
    seg_len = int(rng.integers(max(10, n // 200), max(20, n // 50)))
    s = int(rng.integers(0, n - seg_len))
    a_pert = a.copy()
    saved = a_pert[s:s+seg_len].sum()
    a_pert[s:s+seg_len] = 0.0

    a_pert[mask] += saved / max(1, int(mask.sum())) * rng.random(int(mask.sum()))
    return np.clip(a_pert, 0.0, 1000.0)
