2. Compute the power spectrum $|P_m|^2$.
3. Compute the "target" power spectrum $|P_m^{target}|^2 = \frac{L}{2n-1} \cdot \left|\frac{\sin(\pi m (2n-2)/L)}{\sin(\pi m /L)}\right|^2$ (this gives a flat autocorrelation).
4. Blend: $|P_m^{new}| = |P_m|^{\alpha} \cdot |P_m^{target}|^{1-\alpha}$ for some $\alpha \in (0,1)$.
5. Reconstruct $p$ from the modified spectrum: $p_j = \frac{1}{L}\sum_m |P_m^{new}| e^{i\theta_m} e^{2\pi i mj/L}$ where $\theta_m$ is the phase of $P_m$.
6. Project onto the probability simplex: $p_j = \max(0, p_j)$, then normalize.
7. Repeat.

This is essentially a "Gerchberg-Saxton" style iteration in the spectral domain. The phase is preserved (from the current $p$), and the amplitude is shaped toward the target.

The key parameter is $\alpha$ (blend factor). Small $\alpha$ means strong shaping toward the target (fast convergence to flat autocorrelation but might create negative values in $p$). Large $\alpha$ means slow but safe.

"""
Step function optimization: spectral shaping + LSE + accelerated subgradient.

import numpy as np
from scipy.optimize import minimize
import time

    S = float(a.sum())

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0

def nesterov_subgrad(a, iters=5000, f_star=1.0, verbose=False):

        if Sy < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (Sy * Sy)
        
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        # Subgradient of max_k c_k / S^2 * 2n
        # d/dy_i [2n * M / S^2] where M = c_{k*}

        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            for j in range(j_start, j_end + 1):
                if 2 * j == k_star:
                    sg[j] = y[j]
                else:
                    sg[j] = 2.0 * y[k_star - j]

        # d/dy_i [2n * M / S^2] = 2n * (dg_i / S^2 - 2*M / S^3 * y_i)

        S3 = S2 * Sy

        gn = float(np.linalg.norm(grad))

            break

            print(f"    nesterov it={it}, sc={sc_cur:.8f}, best={best_sc:.8f}")
