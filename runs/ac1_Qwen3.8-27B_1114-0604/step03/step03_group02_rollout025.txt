    T_imag = np.zeros(M_fft // 2 + 1)
    
    # T[k] = C * exp(-πi(2n-2)k/M_fft) * sin((2n-1)πk/M_fft) / sin(πk/M_fft)
    # = C * [cos(phase)*ratio - i*sin(phase)*ratio]
    # where phase = π(2n-2)k/M_fft, ratio = sin((2n-1)πk/M_fft)/sin(πk/M_fft)
    
    phase = np.pi * (2*n - 2) * k / M_fft
    ratio = np.zeros(M_fft // 2 + 1)
    mask = np.abs(den) > 1e-15
    ratio[mask] = num[mask] / den[mask]
    ratio[0] = 2*n - 1  # Limit
    
    T_real = np.cos(phase) * ratio
    T_imag = -np.sin(phase) * ratio
    
    # T[0] should be C * (2n-1) (real and positive)
    T_real[0] = 2*n - 1
    T_imag[0] = 0
    
    # Target X = sqrt(T)
    # |X| = sqrt(|T|), arg(X) = arg(T)/2
    T_mag = np.sqrt(T_real**2 + T_imag**2)
    T_phase = np.arctan2(T_imag, T_real)
    
    X_mag = np.sqrt(T_mag + 1e-30)
    X_phase = T_phase / 2.0
    
    X_real = X_mag * np.cos(X_phase)
    X_imag = X_mag * np.sin(X_phase)
    
    return X_real, X_imag  # Real and imaginary parts of target X[k] for k=0..M_fft//2


def spectral_newton(a, n_iters=2000, deadline=None, verbose=False):
    """
    Newton method in the frequency domain to achieve flat self-convolution.
    """
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0/n
        S = 1.0
    
    # Precompute target
    Xr_t, Xi_t = target_spectrum(n, M_fft)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Current spectrum
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        X = np.fft.rfft(a_pad)
        
        # Newton step: we want X^2 = T, so X_new = X + (T - X^2)/(2X)
        # = X + T/(2X) - X/2 = (X + T/X)/2
        # This is the Halley's method / Newton for sqrt
        X2 = X * X
        T = Xr_t + 1j * Xi_t
        X_new = (X + T / (X + 1e-15 * np.sign(X))) / 2.0
        
        # Convert back to time domain
        # Ensure conjugate symmetry
        X_full = np.zeros(M_fft, dtype=np.complex128)
        X_full[:M_fft//2+1] = X_new
        X_full[M_fft//2+1:] = np.conj(X_new[M_fft//2:0:-1])
        # Handle Nyquist if M_fft is even
        if M_fft % 2 == 0:
            X_full[M_fft//2] = X_new[M_fft//2].real + 0j
        
        a_full = np.fft.ifft(X_full)
        a_new = np.real(a_full[:n])
        a_new = np.maximum(a_new, 0.0)
        
        # Normalize
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new * (S / S_new)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if verbose and it % 100 == 0:
                print(f"    Spectral Newton it={it}: score={sc:.8f}")
    
    return best_sc, best_a


def spectral_amplitude_match(a, n_iters=5000, deadline=None, damping=0.5, verbose=False):
    """
    Simpler spectral method: match amplitude to target, update phase iteratively.
    """
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0/n
        S = 1.0
    
    # Target magnitude
    Xr_t, Xi_t = target_spectrum(n, M_fft)
    target_mag = np.sqrt(Xr_t**2 + Xi_t**2)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        X = np.fft.rfft(a_pad)
        
        # Match amplitude, keep phase
        X_mag = np.abs(X)
        scale = (target_mag / (X_mag + 1e-30)) ** damping
        X_new = X * scale
        
        # IFFT (ensuring real output)
        a_full = np.fft.irfft(X_new, M_fft)
        a_new = a_full[:n]
        a_new = np.maximum(a_new, 0.0)
        
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new * (S / S_new)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if verbose and it % 500 == 0:
                print(f"    Spectral AM it={it}: score={sc:.8f}")
    
    return best_sc, best_a


def multiplicative_equalize(a, n_iters=200, alpha=0.3, deadline=None, threshold_frac=0.99):
    """
    LP-dual-inspired multiplicative update.
    At optimum, the "pressure" from active convolution lags should be equalized.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0/n
        S = 1.0
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M_fft = 1 << (3 * n - 2).bit_length()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M_val = float(c.max())
        if M_val < 1e-15:
            break
        
        # Active lags
        threshold = M_val * threshold_frac
        lambda_v = (c >= threshold).astype(np.float64)
        if lambda_v.sum() == 0:
            lambda_v[int(np.argmax(c))] = 1.0
        
        # Compute h[j] = 2 * sum_{m} lambda[m] * a[m-j] (cross-correlation)
        # h[j] = 2 * sum_{m valid for j} lambda[m] * a[m-j]
        # = 2 * (lambda * a)[j] in cross-correlation sense
        # cross_corr[j] = sum_m lambda[m] * a[m-j] = sum_k lambda[k+j] * a[k]
        # So h[j] = 2 * sum_k lambda[k+j] * a[k]
        
        L = np.fft.rfft(lambda_v, M_fft)
        A = np.fft.rfft(a, M_fft)
        # We need sum_k a[k] * lambda[k+j] for j = 0, ..., n-1
        # This is the cross-correlation of a and lambda at lag j
        # = IFFT(conj(A) * L)[j] ... let me think
        # Actually: sum_k a[k] * lambda[k+j] = (a * lambda)[j] in cross-correlation
        # In FFT: cross_correlation[j] = sum_k a[k] * lambda[k+j]
        # = sum_k a[k] * flip(lambda)[len-j-k] ... this is getting confusing
        # Let me just do: cross[j] = sum_{k=0}^{n-1-j} a[k] * lambda[k+j]
        # This is the inner product of a[0:n-j] and lambda[j:2n-1]
        # In FFT terms: it's the cross-correlation at lag j
        # numpy: np.correlate(lambda, a, mode='full')[n-1:n-1+n] gives cross[j] for j=0..n-1
        
        # Using FFT: cross_corr = ifft(conj(A) * L)
        # cross_corr[j] = sum_k a[k] * conj(lambda[k-j]) ... hmm
        # Let me just use direct computation for correctness
        
        # Actually, let me use the formula:
        # h[j] = 2 * sum_{m=max(j,0)}^{min(j+n-1, 2n-2)} lambda[m] * a[m-j]
        # where m-j ranges from 0 to n-1, so m ranges from j to j+n-1
        # and m must be in [0, 2n-2]
        
        # For efficiency, use FFT:
        # h[j]/2 = sum_{k=0}^{n-1} a[k] * lambda[j+k] (where lambda is zero-padded to [0, 2n-2])
        # This is the cross-correlation of a and (reversed) lambda... 
        # Actually it's: sum_k a[k] * lambda[j+k]
        # = (a * lambda)[j] where * denotes cross-correlation with lambda "forward"
        
        # Let's use: conv(a, lambda) gives sum_k a[k]*lambda[m-k]
        # We want sum_k a[k]*lambda[j+k] = sum_k a[k]*flip(lambda)[len-1-(j+k)]
        # This is getting too complicated. Let me just use direct or a simpler FFT trick.
        
        # Simple approach: flip lambda and do convolution
        # sum_k a[k] * lambda[j+k] for k=0..n-1 (where j+k valid)
        # = conv(a, lambda_rev)[(2n-2) - j] ... let me verify
        # conv(a, lambda_rev)[m] = sum_k a[k] * lambda_rev[m-k] = sum_k a[k] * lambda[(2n-2)-(m-k)]
        # = sum_k a[k] * lambda[2n-2-m+k]
        # We want this to equal sum_k a[k] * lambda[j+k]
        # So: 2n-2-m+k = j+k => m = 2n-2-j
        # So: sum_k a[k] * lambda[j+k] = conv(a, lambda_rev)[2n-2-j]
        
        lambda_rev = lambda_v[::-1]
        M2 = 3*n - 2  # length of conv(a, lambda_rev) = n + (2n-1) - 1 = 3n-2
        M2_fft = 1 << (M2 - 1).bit_length()
        A_fft = np.fft.rfft(a, M2_fft)
        L_fft = np.fft.rfft(lambda_rev, M2_fft)
        conv_result = np.fft.irfft(A_fft * L_fft, M2_fft)[:M2]
        
        # h[j]/2 = conv_result[2n-2-j] for j = 0, ..., n-1
        h_vals = conv_result[2*n-2 : 2*n-2-(n-1) : -1]
        h = 2.0 * h_vals
        h = np.maximum(h, 1e-15)
        
        h_mean = h.mean()
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)
        
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 1.0/n
            break
        a = a * (S / S_new)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient of L w.r.t. a[j] = 2 * sum_m p[m] * a[m+j] (for valid indices)
    # = 2 * cross_correlation(a, p)[j] where cross_correlation[j] = sum_{m-valid} p[m] * a[m-j]
    # Actually: dL/da[j] = 2 * sum_m p[m] * (da[m]/da[j]) = 2 * sum_m p[m] * a[m-j] (if m-j valid, i.e., j <= m < j+n)
    # = 2 * sum_{m=j}^{j+n-1} p[m] * a[m-j]
    
    # This is 2 * (p * a)[j] in the "cross-correlation with forward shift" sense
    # = 2 * sum_k p[j+k] * a[k] for k=0..n-1 (where j+k < 2n-1)
    
    # Using the same trick as before:
    # sum_k a[k] * p[j+k] = conv(a, p_rev)[2n-2-j]
    p_rev = p[::-1]
    M2 = 3*n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    P_fft = np.fft.rfft(p_rev, M2_fft)
    conv_res = np.fft.irfft(A_fft * P_fft, M2_fft)[:M2]
    cross = conv_res[2*n-2 : 2*n-2-(n-1) : -1]
    
    g = 2.0 * cross
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """Run L-BFGS-B on the LSE-smoothed score."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj_wrapper(x, b):
        return lse_score_grad(x, b)
    
    options = {'ftol': 1e-15, 'gtol': 1e-13, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = max(0, deadline - time.monotonic())
        if remaining < 2.0:
            sc, _ = lse_score_grad(a0, beta)
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(100, int(remaining * 300)))
        options['maxiter'] = min(maxiter, max(100, int(remaining * 80)))
    
    try:
        res = minimize(
            obj_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options=options
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out


def subgradient_polish(a, total_iters=1000, deadline=None):
    """
    Projected subgradient descent with Nesterov acceleration.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0/n
        S = 1.0
    a = a / S  # Normalize to sum = 1
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Lower bound on max(conv) when sum=1: 1/(2n-1)
    lb = 1.0 / (2*n - 1)
    sc_lb = 2.0 * n * lb  # = 2n/(2n-1)
    
    # Nesterov acceleration
    a_prev = a.copy()
    momentum = 0.0
    
    for it in range(total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val  # sum = 1
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end+1] = 2.0 * a[k_star - j_end:k_star - j_start + 1][::-1]
        
        # Wait, let me redo this.
        # conv[k_star] = sum_j a[j] * a[k_star - j]
        # d(conv[k_star])/da[j] = a[k_star - j] (if 0 <= k_star - j < n)
        # So the subgradient of max_conv w.r.t. a[j] is:
        #   a[k_star - j] if 0 <= k_star - j < n, else 0
        
        # Actually it's 2*a[k_star-j] because conv[k] = sum_j a[j]*a[k-j]
        # d/da[j] = a[k-j] + a[j] * (d/da[j])(a[k-j]) ... no
        # conv[k] = sum_j a[j] * a[k-j]
        # If we increase a[j] by eps, conv[k] increases by a[k-j] (from the j term)
        # and if k-j != j, also by a[j] (from the k-j term)
        # If k-j == j (i.e., k = 2j), then it increases by 2*a[j]
        
        # More carefully: conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
        # d(conv[k])/da[j] = a[k-j] (if 0<=k-j<n) + a[j] (if 0<=j<n and j != k-j, i.e., k != 2j)
        # Wait no. Let me index properly.
        # conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/da[j] for a specific j:
        #   Term where first index is j: a[k-j] (if 0<=k-j<n)
        #   Term where second index is j, i.e., k-j'=j means j'=k-j: a[k-j] (if 0<=k-j<n and k-j != j)
        #   If k-j == j (k=2j): a[j] (only counted once, but derivative is 2*a[j]... no)
        #   
        # Actually, conv[k] = sum_j a[j]*a[k-j]. This is a bilinear form in a.
        # d(conv[k])/da[j] = (coefficient of a[j] in conv[k])
        # a[j] appears in the term where first index = j: contributes a[k-j]
        # a[j] appears in the term where second index = j, i.e., j' + (k-j') = k and j' = k-j... 
        # Wait: the term a[j']*a[k-j'] has second index j when k-j' = j, i.e., j' = k-j.
        # So a[j] appears as the second factor in the term with j' = k-j: contributes a[k-j]
        # Unless j = k-j (i.e., k = 2j), in which case it's a[j]*a[j] = a[j]^2, and d/da[j] = 2a[j].
        
        # So: d(conv[k])/da[j] = a[k-j] (if 0<=k-j<n and k-j != j)
        #                        + a[k-j] (if 0<=k-j<n and k-j != j) [from the other term]
        # Hmm, this is getting confusing. Let me just think of it as:
        # conv[k] = sum_{j=0}^{n-1} a[j] * a[k-j] (where a[m] = 0 for m<0 or m>=n)
        # d(conv[k])/da[j] = sum_{j'} [d(a[j'])/da[j]] * a[k-j'] + a[j'] * (d(a[k-j'])/da[j])
        # = a[k-j] + (if k-j < n and k-j >= 0) a[k-j] ... 
        # No:
        # = sum_{j'} delta(j',j) * a[k-j'] + sum_{j'} a[j'] * delta(k-j', j)
        # = a[k-j] + sum_{j'} a[j'] * delta(j', k-j)
        # = a[k-j] + a[k-j] (if 0 <= k-j < n)
        # = 2*a[k-j] (if 0 <= k-j < n)
        # Wait, that's only if k-j is a valid index (0 <= k-j < n).
        # And the first term a[k-j] is valid when 0 <= k-j < n.
        # The second term: sum_{j'} a[j'] * delta(j', k-j) = a[k-j] if 0 <= k-j < n.
        # So d(conv[k])/da[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
        # But wait, if k-j = j (i.e., k = 2j), then both terms contribute a[j], giving 2*a[j].
        # And if k-j != j, the first term gives a[k-j] and the second gives a[k-j], total 2*a[k-j].
        # So in all cases: d(conv[k])/da[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
        
        # Great! So the subgradient is:
        sg = np.zeros(n)
        # For each j, if 0 <= k_star - j < n: sg[j] = 2*a[k_star - j]
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        if j_max >= j_min:
            idx = np.arange(j_min, j_max + 1)
            sg[idx] = 2.0 * a[k_star - idx]
        
        # Step size (Polyak): step = (sc - sc_lb) / ||sg||^2
        gn = float(np.linalg.norm(sg))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.1)  # Cap the step
        
        a_new = a - step * sg
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        # Nesterov: momentum
        momentum = (it + 1) / (it + 1 + 2.0)  # ~ 1/2 for large it
        a = a_new + momentum * (a_new - a_prev)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S > 0:
            a = a / S
        a_prev = a_new.copy()
    
    return best_sc, best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        p = parent.astype(np.float64)
        if len(p) != n:
            p = np.interp(np.linspace(0, len(p)-1, n), np.arange(len(p)), p)
        inits.append(np.clip(p, 0.01, 1000.0).copy())
    
    x = np.arange(n, dtype=np.float64) / n
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Target shape (square root of Dirichlet kernel, shifted)
    # This is the shape of the optimal solution!
    M_fft = 1 << (2*n - 1).bit_length()
    Xr_t, Xi_t = target_spectrum(n, M_fft)
    Xt = Xr_t + 1j * Xi_t
    a_target = np.fft.irfft(Xt, M_fft)[:n]
    a_target = np.maximum(a_target, 0.0)
    if a_target.sum() > 1e-12:
        inits.append(a_target * (0.5 / a_target.sum() * n))
    
    # Cosine modulations
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.3 * np.sin(np.pi * x) ** 2)
    
    # Random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n),
