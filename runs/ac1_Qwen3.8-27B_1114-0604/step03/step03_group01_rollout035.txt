def lp_direction(f, M):
    """Solve the LP to find the best direction g."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix
    # (f*g)(k) <= M for k = 0,...,2n-2
    # (f*g)(k) = sum_i f[i] * g[k-i] (valid indices)
    
    # Build sparse matrix F where F[k,i] = f[k-i] if 0 <= k-i < n
    # Use the structure: for each i, the non-zero entries are at positions k = i, i+1, ..., i+n-1
    # with values f[0], f[1], ..., f[n-1]
    
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for k in range(i, i+n):
            rows.append(k)
            cols.append(i)
            vals.append(f[k-i])
    
    F = csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [F @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, max_iters=10000)
    
    if g.value is None:
        return None
    return np.clip(g.value, 0, 1000)
