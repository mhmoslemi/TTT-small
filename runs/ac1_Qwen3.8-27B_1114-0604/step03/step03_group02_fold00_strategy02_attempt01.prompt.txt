def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and its gradient.
    Returns (score, grad).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: grad_L[j] = 2 * sum_m p[m] * a[m+j] (cross-correlation)
    # = 2 * conv(a_rev, p)[n-1 : 2n-1]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def multiplicative_equalize(a, n_iters=100, alpha=0.3, deadline=None):
    """
    LP-dual-inspired multiplicative update.
    At optimum, h[j] = 2*sum_k lambda[k]*a[k-j] should be constant.
    This function iteratively equalizes h using a damped multiplicative update.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    # Precompute FFT size for cross-correlation
    M_fft = 1 << (3 * n - 2).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        if M < 1e-15:
            break
        # Active lags (those close to the max)
        threshold = M * 0.99
        lambda_v = (c >= threshold).astype(np.float64)
        if lambda_v.sum() == 0:
            lambda_v[int(np.argmax(c))] = 1.0

        # Compute h[j] = 2 * sum_{m=0}^{n-1} lambda[j+m] * a[m]
        # This is 2 * (lambda * a) cross-correlation at lags 0..n-1
        L = np.fft.rfft(lambda_v, M_fft)
        A = np.fft.rfft(a, M_fft)
        cross = np.fft.irfft(L * np.conj(A), M_fft)
        h = 2.0 * cross[:n]

        # Ensure h is positive (it should be since a >= 0 and lambda >= 0)
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()

        # Damped multiplicative update: a[j] <- a[j] * (1 - alpha + alpha * h_mean/h[j])
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)

        # Renormalize to preserve sum
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a = a * (S / S_new)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements that contribute most to the peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:-top_count] if n > 2 * top_count else np.argsort(contrib)[:top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def exact_subgrad_polish(a, total_iters=500, deadline=None):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0

    for it in range(total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        step_decay = 1.0 / (1.0 + 0.01 * it)
        a -= step_decay * 0.001 * grad / gn
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a
