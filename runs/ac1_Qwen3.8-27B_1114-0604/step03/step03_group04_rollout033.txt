    # conv(a, b)[k] = sum_j a[j] * b[k-j]
    # We want g[i] = sum_j a[j] * p[i+j] = sum_j a[j] * b[2n-2-(i+j)] = sum_j a[j] * b[(2n-2-i)-j]
    # = conv(a, b)[2n-2-i]
    
    # So g[i] = conv(a, reverse(p))[2n-2-i]
    # g = conv(a, reverse(p))[2n-2 : n-2 : -1]  (reversed slice)
    
    # Length of conv(a, b) = n + (2n-1) - 1 = 3n-2
    # We want indices 2n-2, 2n-3, ..., n-2 (that's n values)
    # conv_result[2n-2 : n-2 : -1] gives us [2n-2, 2n-3, ..., n-1]... 
    
    # Hmm, let me just use numpy directly for small checks and FFT for efficiency.
    
    # For efficiency, let's use the FFT approach:
    # g[i] = sum_j a[j] * p[i+j] for i in [0, n-1], j in [0, n-1]
    
    # This is a special case of convolution. Let's define:
    # h = [a[0], a[1], ..., a[n-1], 0, ..., 0] (length N)
    # q = [0, ..., 0, p[0], p[1], ..., p[2n-2], 0, ..., 0] (length N, p starts at position n-1)
    # Then (h * q)[i + n - 1] = sum_j h[j] * q[i+n-1-j]
    # = sum_{j=0}^{n-1} a[j] * p[i+n-1-j-(n-1)] = sum_j a[j] * p[i-j]
    # That's cross-correlation, not what we want.
    
    # Let me try: q = [0, ..., 0, p[2n-2], p[2n-3], ..., p[0], 0, ..., 0] (p reversed, starting at n-1)
    # (h * q)[i + 2n - 2] ... this is getting complicated.
    
    # Let me just use a direct FFT-based correlation:
    # g[i] = sum_j a[j] * p[i+j]
    # This equals the cross-correlation of p and a at lag -i... 
    # Actually, let me define it as:
    # g = np.correlate(p, a, mode='valid')  # This gives sum_j p[i+j] * a[j] for i in [0, len(p)-len(a)] = [0, n-1]
    # Wait, np.correlate(a, v, 'valid') returns the correlation of a with v for valid positions
    # np.correlate(p, a, 'valid')[i] = sum_j p[i+j] * conj(a[j])
    # Since everything is real, = sum_j p[i+j] * a[j]
    # Length of result = len(p) - len(a) + 1 = (2n-1) - n + 1 = n
    # Perfect!
    
    # So g = np.correlate(p, a, mode='valid') gives exactly what we want!
    # But for large n, np.correlate might be slow (O(n^2)). Let's use FFT.
    
    # FFT approach for g[i] = sum_j a[j] * p[i+j]:
    # Pad a to length N >= n + (2n-1) - 1 = 3n - 2
    # Pad p to length N
    # g = ifft(fft(p, N) * conj(fft(a, N)))
    # This gives circular cross-correlation. For linear, we need N >= 3n-2.
    # The result at index i gives sum_j p[i+j mod N] * a[j]... 
    
    # Actually, the standard formula:
    # circular_cross_corr(p, a)[i] = sum_j p[(i+j) mod N] * conj(a[j])
    # For linear cross-correlation (what we want), we need N large enough so there's no wrap-around.
    # We need (i+j) < N for all valid i, j. Max i+j = (n-1) + (n-1) = 2n-2.
    # So N > 2n-2, i.e., N >= 2n-1 is sufficient.
    
    # But the standard FFT cross-correlation formula gives:
    # corr[i] = sum_j p[(i+j) mod N] * conj(a[j])
    # For N >= 2n-1, and i in [0, n-1], j in [0, n-1]:
    # (i+j) mod N = i+j (since i+j <= 2n-2 < N)
    # So corr[i] = sum_j p[i+j] * a[j] = g[i]. 
    
    N_fft = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, N_fft)
    A = np.fft.rfft(a, N_fft)
    g_full = np.fft.irfft(P * np.conj(A), N_fft)[:n]
    
    dL_da = 2.0 * g_full
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        return lse_objective_and_grad(x, beta)
    
    res = minimize(
        obj_and_grad, a0, jac=True, bounds=bounds,
        method='L-BFGS-B',
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def peak_shave(a, n_peaks=5, strength=0.1, n_iters=100):
    """
    Iteratively identify the top autocorrelation peaks and reduce the
    contributions to those peaks by adjusting the affected coefficients.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        sc = 2.0 * n * float(c.max()) / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top peaks (excluding lag 0 which is c[0] = sum(a^2))
        # Actually, for non-negative sequences, c[0] is always the max...
        # But let's work with the full convolution just in case.
        
        # For the full convolution of length 2n-1, the "peaks" we care about
        # are the ones that are close to c[0]. Let's find the top n_peaks.
        
        # Get the indices of the top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # For each peak at lag k (in the full conv sense), the contributing pairs are:
        # c[k] = sum_j a[j]*a[k-j] (for 0 <= j <= n-1, 0 <= k-j <= n-1)
        # To reduce c[k], we should reduce a[j] and/or a[k-j] for the contributing j's.
        
        # Simple strategy: for each top peak, find the coefficient that contributes
        # most to it and reduce it.
        
        modified = False
        for k in top_idx:
            # Find contributing pairs
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            if j_end < j_start:
                continue
            
            # Contributions from each j
            contribs = np.zeros(n)
            for j in range(j_start, j_end + 1):
                contribs[j] = a[j] * a[k - j]
            
            # Find the coefficient that, if reduced, would most reduce c[k]
            # Reducing a[i] by delta reduces c[k] by delta * (a[k-i] + a[k-i]) = 2*delta*a[k-i]
            # (approximately, for small delta)
            # So we want to reduce the a[i] that has the largest a[k-i] (the partner)
            # while a[i] itself is also significant.
            
            # Simple approach: reduce the maximum coefficient that contributes to this peak
            max_contrib_j = j_start + int(np.argmax(a[j_start:j_end+1] * a[k-j_start:k-j_start+1][::-1]))
            
            # Reduce it
            reduce_amt = a[max_contrib_j] * strength * (1.0 / (1.0 + 0.01 * it))
            a[max_contrib_j] -= reduce_amt
            a[max_contrib_j] = max(a[max_contrib_j], 0.0)
            
            # Compensate by increasing a low-contributing coefficient
            # (to maintain the sum)
            if reduce_amt > 0:
                # Find a coefficient that doesn't contribute much to any peak
                low_contrib = np.argsort(np.minimum(a, 1.0))[0]
                a[low_contrib] += reduce_amt
            modified = True
        
        if not modified:
            break
    
    return best_sc, best_a


def spectral_anneal(a, N_fft, n_iters=200, start_lam=0.9, end_lam=0.3):
    """Spectral annealing: gradually flatten the spectrum while maintaining non-negativity."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S_orig = float(a.sum())
    
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam = start_lam * (1 - progress) + end_lam * progress
        
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat spectrum (CAZAC-like)
        non_dc = mag[1:]
        target = np.mean(non_dc)
        
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N_fft // 2
        if N_fft % 2 == 0:
            new_mag[half] = mag[half]
        
        for k in range(1, half + 1):
            if N_fft % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N_fft - k] = new_mag[k]
        
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        
        S = float(a.sum())
        if S > 1e-12:
            a = a * (S_orig / S)
    
    return a


def multi_scale_optimize(a, deadline):
    """Multi-scale optimization: optimize in frequency domain at different resolutions."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    N_fft = 1 << (2 * n - 1).bit_length()
    
    for lam in [0.9, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
        if time.monotonic() > deadline:
            break
        
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc = mag[1:]
        target = np.mean(non_dc)
        
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N_fft // 2
        if N_fft % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if N_fft % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N_fft - k] = new_mag[k]
        
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a_new = np.maximum(a_new, 0.0)
        S = float(a_new.sum())
        if S > 1e-12:
            a_new = a_new * (float(a.sum()) / S)
        
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

dL/dc_k = exp(beta * (c_k - c_max)) / sum_j exp(beta * (c_j - c_max)) = p_k

High-beta LSE + targeted perturbation optimizer for minimum peak self-convolution.
Uses progressively higher log-sum-exp smoothing with L-BFGS-B, combined with
diverse spectral and time-domain perturbations to escape local optima.
Exploits the scale invariance of the objective and FFT-based gradient computation.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Compute full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the exact objective: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    Compute LSE-smoothed objective and its gradient.
    Returns (score, grad) tuple for use with scipy minimize.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted_max = shifted.max()
    exp_shifted = np.exp(shifted - shifted_max)
    sum_exp = exp_shifted.sum()
    log_sum = shifted_max + np.log(sum_exp)
    L = c_max + log_sum / beta

    score = 2.0 * n * L / (S * S)

    # Softmax weights p_k = exp(shifted_k) / sum_exp
    p = exp_shifted / sum_exp  # length 2n-1

    # Gradient: dL/da_i = 2 * sum_j a[j] * p[i+j] for i in [0, n-1]
    # Computed via FFT cross-correlation
    N_fft = 1 << (2 * n - 1 - 1).bit_length()
    P = np.fft.rfft(p, N_fft)
    A = np.fft.rfft(a, N_fft)
    g = np.fft.irfft(P * np.conj(A), N_fft)[:n]

    dL_da = 2.0 * g

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)

    return score, grad


def lse_wrapped(x, beta):
    """Wrapper for scipy minimize that passes beta."""
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter, maxfun):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_wrapped, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def spectral_flatten_step(a, N_fft, lam):
    """One step of spectral flattening toward flat power spectrum."""
    n = len(a)
    A = np.fft.fft(a, N_fft)
    mag = np.abs(A)
    phase = np.angle(A)
    non_dc = mag[1:]
    target = np.mean(non_dc)

    new_mag = mag.copy()
    new_mag[0] = mag[0]
    half = N_fft // 2
    if N_fft % 2 == 0:
        new_mag[half] = mag[half]
    for k in range(1, half + 1):
        if N_fft % 2 == 0 and k == half:
            continue
        new_mag[k] = (1.0 - lam) * mag[k] + lam * target
        new_mag[N_fft - k] = new_mag[k]

    new_A = new_mag * np.exp(1j * phase)
    a_new = np.real(np.fft.ifft(new_A))[:n]
    return np.maximum(a_new, 0.0)


def spectral_anneal(a, N_fft, n_iters, start_lam, end_lam):
    """Iterative spectral annealing with decreasing strength."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S_orig = float(a.sum())
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam = start_lam * (1.0 - progress) + end_lam * progress
        a = spectral_flatten_step(a, N_fft, lam)
        S = float(a.sum())
        if S > 1e-12:
            a = a * (S_orig / S)
    return a


def gradient_descent_polish(a, n_iters=500, lr_init=0.01):
    """
    Fixed-point contraction: move a toward the "flat convolution" target.
    Uses the subgradient of the max (directed at the worst peak).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max(c) w.r.t. a
        k_star = int(np.argmax(c))
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        lr = lr_init / (1.0 + 0.005 * it)
        a -= lr * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a
