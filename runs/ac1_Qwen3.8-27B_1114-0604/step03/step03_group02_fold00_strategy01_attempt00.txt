def optimize_sequence(parent=None, n_len=0, beta_min=1, beta_max=2400, alpha=0.2):
    if parent is not None and n_len == 0:
        n_len = len(parent)
        
    sequence = np.ones(n_len) * 0.5 if parent is None else parent.astype(np.float64)
    sequence = np.array(sequence, dtype=np.float64)
    
    def get_smoothed_gradient(seq, beta):
        convolved = np.convolve(seq, seq, mode='full')
        max_val = np.max(convolved)
        if max_val < 1e-3:
            return np.zeros(n_len)
            
        # Invert gradient components on active lags
        weights = convolved[(n_len-1)//2: (3*n_len-1)//2] / max_val
        
        # Exponential smoothing on weights
        weights = np.log(np.exp(weights / beta).sum())
        weights = weights / beta
        
        # Calculate gradient using analytical decomposed form
        numerator = beta * (weights - np.min(weights))
        numerator = numerator.clip(0, 1)
        exp_numer = np.exp(numerator)
        
        p = exp_numer / exp_numer.sum()
        p = p.clip(1e-15, 1)
        
        # Calculate cross-correlation gradient
        grad_correlation = np.zeros(n_len)
        for lag in range(n_len):
            idx_start = max(0, lag)
            idx_end = min(n_len, lag + n_len)
            if idx_end > idx_start:
                grad_correlation[lag] += p[lag + n_len - 1] * (1 - p[lag + n_len - 1])
                
        grad = grad_correlation / (alpha + np.mean(grad_correlation))
        grad = grad.clip(1e-10, 100)
        return grad

    def modified_score(seq):
        total = seq.sum()
        if total < 0.01:
            return np.inf
            
        convolved = np.convolve(seq, seq, mode='full')
        max_val = np.max(convolved)
        return float(2 * len(seq) * max_val / (total ** 2))

    # Cooling schedule transformations
    for temperature in range(beta_max, beta_min, -100):

        if grad is None or np.abs(grad).sum() < 1e-15:

        # Gradient descent step optimization

        if not success:
            # Fallback to iterative cooling
            continue
            
        # Update best sequence
        sequence = new_sequence
        best_score = modified_score(sequence)
        yield sequence, best_score
