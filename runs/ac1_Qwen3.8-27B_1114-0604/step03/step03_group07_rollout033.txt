"""
Spectral-optimized step function for minimum autocorrelation peak.
Uses frequency-domain initialization, LSE-gradient optimization,
and iterative spectrum-flattening with non-negativity projection.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    # Gradient
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapper(a, beta):
    return lse_score_and_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=3000, time_limit=60):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    t0 = time.monotonic()
    
    # Custom callback to stop if time exceeded
    def callback(pk):
        return time.monotonic() - t0 > time_limit
    
    try:
        res = minimize(
            lse_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-10,
                     'maxiter': maxiter, 'maxfun': maxiter * 20, 'maxls': 50},
            callback=callback
        )
    except Exception:
        return compute_score(a0), a0
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def spectral_init(n, rng):
    """Initialize from frequency domain: random phase, flat magnitude, clip to positive."""
    N = 1 << (2 * n - 1).bit_length()
    # Flat magnitude with random phase
    A = np.ones(N // 2 + 1) * (n ** 0.5 / N ** 0.5)
    phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    # DC component must be real and positive
    A[0] = np.abs(A[0])
    # Make Hermitian
    A_complex = np.empty(N, dtype=np.complex128)
    A_complex[:N // 2 + 1] = A * np.exp(1j * phase)
    A_complex[N // 2 + 1:] = np.conj(A_complex[1:N // 2][::-1])
    # Ensure Hermitian symmetry
    a_full = np.fft.irfft(A_complex, N)
    a = a_full[:n]
    # Clip to non-negative
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    return a


def anti_palindromic_init(n, rng, strength=0.5):
    """Create anti-palindromic initialization."""
    base = np.ones(n) * 0.5
    # Add anti-palindromic perturbation
    half = n // 2
    d = rng.standard_normal(half) * strength
    pert = np.zeros(n)
    pert[:half] = d
    pert[half:] = -d[::-1] if n % 2 == 0 else np.concatenate([-d[::-1], [0.0]])
    if n % 2 == 1:
        pert = np.zeros(n)
        pert[:half] = d
        pert[half+1:] = -d[::-1]
    a = base + pert
    return np.clip(a, 0.01, 100.0)


def block_anti_palindromic(n, rng, n_blocks=8):
    """Create piecewise-constant anti-palindromic sequence."""
    block_size = n // n_blocks
    remainders = n - block_size * n_blocks
    # Create block values (anti-palindromic)
    half_blocks = n_blocks // 2
    vals = rng.uniform(0.1, 1.0, half_blocks)
    # Anti-palindromic: block i and block (n_blocks-1-i) sum to constant
    c = rng.uniform(1.0, 2.0)
    block_vals = np.zeros(n_blocks)
    for i in range(half_blocks):
        block_vals[i] = vals[i]
        block_vals[n_blocks - 1 - i] = c - vals[i]
    if n_blocks % 2 == 1:
        block_vals[half_blocks] = c / 2
    
    # Build sequence
    a = np.repeat(block_vals, block_size)
    if remainders > 0:
        extra = block_vals[-remainders:]
        a = np.concatenate([a, extra])
    return np.clip(a, 0.01, 100.0)


def freq_flat_init(n, rng, n_trials=20):
    """Try multiple frequency-domain initializations, keep best."""
    best_a = None
    best_sc = np.inf
    for _ in range(n_trials):
        a = spectral_init(n, rng)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a if best_a is not None else np.ones(n) * 0.5


def subgrad_polish(a, max_iters=2000, time_limit=60):
    """Subgradient descent with Polyak step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max conv at k_star
        sg = np.zeros(n)
        # d(conv[k_star])/d(a[m]) = 2*a[k_star - m] if 0 <= k_star - m < n
        m_range = max(0, k_star - n + 1)
        m_end = min(k_star, n - 1)
        if m_end >= m_range:
            sg[m_range:m_end + 1] = 2.0 * a[k_star - m_end:k_star - m_range + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - 1.0) / gn2  # lower bound is ~1
        step = min(step, 1.0 / (1.0 + 0.01 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses spectral-domain initialization, Adam projected gradient, LSE-continuation
L-BFGS, multi-length exploration, and subgradient polishing.
Key idea: flatten the autocorrelation profile by exploiting anti-palindromic
structure and frequency-domain randomization, then refine with adaptive
gradient methods.

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed max-convolution score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    # Gradient via cross-correlation of a^R with p
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapper(a, beta):
    """Scipy-compatible objective returning (value, gradient)."""
    return lse_score_and_grad(a, beta)


def est_time_per_iter(n):
    """Estimate time per LSE evaluation in seconds."""
    if n <= 1024:
        return 0.003
    elif n <= 4096:
        return 0.015
    elif n <= 8192:
        return 0.04
    elif n <= 16384:
        return 0.1
    else:
        return 0.25


def run_lbfgs(a0, beta, time_limit=30.0):
    """Run L-BFGS-B on LSE-smoothed objective with time limit."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    tpi = est_time_per_iter(n)
    maxiter = max(100, min(10000, int(time_limit / (tpi * 3))))
    maxfun = maxiter * 20
    t0 = time.monotonic()
    try:
        res = minimize(
            lse_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-10,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def run_adam(a0, beta, time_limit=30.0, lr=0.5):
    """Adam projected gradient on LSE-smoothed objective."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a[:] = 0.5
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    t0 = time.monotonic()
    best_sc = compute_score(a)
    best_a = a.copy()
    t_pi = est_time_per_iter(n)
    max_steps = int(time_limit / (t_pi * 2))
    
    for t in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        sc, grad = lse_score_and_grad(a, beta)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Adaptive learning rate
        scale = lr / (1.0 + 0.1 * t)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** (t + 1))
        v_hat = v / (1.0 - b2 ** (t + 1))
        a = a - scale * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def spectral_init(n, rng):
    """Frequency-domain random phase, flat magnitude, clip to non-negative."""
    N = 1 << (2 * n - 1).bit_length()
    mag = np.ones(N // 2 + 1)
    phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    A = np.empty(N, dtype=np.complex128)
    A[:N // 2 + 1] = mag * np.exp(1j * phase)
    A[N // 2 + 1:] = np.conj(A[1:N // 2][::-1])
    a_full = np.fft.irfft(A, N)
    a = a_full[:n]
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    return a


def spectral_init_tapered(n, rng, taper_power=1.0):
    """Frequency-domain init with tapered magnitude (less energy at edges)."""
    N = 1 << (2 * n - 1).bit_length()
    freqs = np.arange(N // 2 + 1) / (N // 2)
    mag = 1.0 - taper_power * freqs ** 2
    mag = np.clip(mag, 0.01, None)
    phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    A = np.empty(N, dtype=np.complex128)
    A[:N // 2 + 1] = mag * np.exp(1j * phase)
    A[N // 2 + 1:] = np.conj(A[1:N // 2][::-1])
    a_full = np.fft.irfft(A, N)
    a = a_full[:n]
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    return a


def anti_palindromic_init(n, rng, strength=0.5):
    """Anti-palindromic random initialization."""
    base = np.ones(n) * 0.5
    half = n // 2
    d = rng.standard_normal(half) * strength
    a = base.copy()
    a[:half] += d
    if n % 2 == 0:
        a[half:] -= d[::-1]
    else:
        a[half + 1:] -= d[::-1]
    return np.clip(a, 0.01, 100.0)


def block_anti_palindromic(n, rng, n_blocks=8):
    """Piecewise-constant anti-palindromic sequence."""
    block_size = max(1, n // n_blocks)
    actual_blocks = n // block_size
    half_b = actual_blocks // 2
    c = rng.uniform(1.0, 2.0)
    vals = np.zeros(actual_blocks)
    for i in range(half_b):
        v = rng.uniform(0.05, c - 0.05)
        vals[i] = v
        vals[actual_blocks - 1 - i] = c - v
    if actual_blocks % 2 == 1:
        vals[half_b] = c / 2
    a = np.repeat(vals, block_size)[:n]
    return np.clip(a, 0.01, 100.0)


def ramp_init(n, rng, direction=1):
    """Linear ramp with noise."""
    x = np.linspace(0, 1, n)
    if direction == 1:
        a = 0.05 + 0.95 * x
    else:
        a = 0.05 + 0.95 * (1.0 - x)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, 100.0)


def cosine_mod_init(n, rng, freq=3, amp=0.3):
    """Cosine-modulated initialization."""
    x = np.arange(n) / n
    a = 0.5 + amp * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    return np.clip(a, 0.01, 100.0)


def two_bump_init(n, rng):
    """Two-bump anti-palindromic: high at 1/3, low at 2/3."""
    x = np.arange(n) / n
    a = 0.2 + 0.8 * np.exp(-((x - 0.25) / 0.15) ** 2)
    a2 = 0.2 + 0.8 * np.exp(-((x - 0.75) / 0.15) ** 2)
    a = np.where(x < 0.5, a, a2)
    return np.clip(a, 0.01, 100.0)


def subgrad_polish(a, max_iters=3000, time_limit=45.0):
    """Subgradient descent with Polyak step for exact max."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()
    lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient
        sg = np.zeros(n)
        m_start = max(0, k_star - n + 1)
        m_end = min(k_star, n - 1)
        if m_end >= m_start:
            indices = np.arange(m_start, m_end + 1)
            others = k_star - indices
            sg[indices] = 2.0 * a[others]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(np.dot(grad, grad))
        if gn2 < 1e-30:
            break
        step = (sc - lb) / gn2
        step = min(step, 1.0 / (1.0 + 0.005 * it))
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def targeted_perturb(a, rng, strength=0.05):
    """Reduce elements contributing most to peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    a_pert = a.copy()
    m_start = max(0, k_star - n + 1)
    m_end = min(k_star, n - 1)
    if m_end < m_start:
        return a_pert
    indices = np.arange(m_start, m_end + 1)
    others = k_star - indices
    contribs = a[indices] * a[others]
    top_count = max(3, n // 100)
    order = np.argsort(contribs)
    top_local = order[-top_count:]
    bot_local = order[:-top_count]
    total_red = 0.0
    for idx in top_local:
        amt = a_pert[indices[idx]] * strength
        a_pert[indices[idx]] = max(a_pert[indices[idx]] - amt, 0.0)
        total_red += amt
    if len(bot_local) > 0 and total_red > 0:
        per = total_red / len(bot_local)
        a_pert[indices[bot_local]] += per
    return np.clip(a_pert, 0.0, 1000.0)

            break
        tl = min(40.0, time_left / 3.0)
        sc, a_opt = run_lbfgs(best_a, beta, time_limit=tl)
        report(sc, a_opt, f"P3 b={beta}")

    # Adam at medium beta
    if time.monotonic() < p3_end:
        tl = min(30.0, p3_end - time.monotonic())
        sc, a_opt = run_adam(best_a, 500, time_limit=tl, lr=0.2)
        report(sc, a_opt, "P3 adam500")

    # Subgradient
    if time.monotonic() < p3_end:
        tl = min(30.0, p3_end - time.monotonic())
        sc, a_opt = subgrad_polish(best_a, max_iters=5000, time_limit=tl)
        report(sc, a_opt, "P3 subgrad")

    # ========== Phase 4: Local Search (13% time) ==========
    p4_end = deadline
    print(f"\n=== Phase 4: Local Search (until t={p4_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    trial = 0
    while time.monotonic() < p4_end - 8 and trial < 40:
        trial += 1
        time_left = p4_end - 8 - time.monotonic()
        if time_left < 8:
            break

        strat = trial % 5
        if strat == 0:
            # Targeted perturbation
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.04 * rng.random())
        elif strat == 1:
            # Gaussian mutation
            scale = 0.02 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strat == 2:
            # Anti-palindromic push
            a_pert = best_a.copy()
            half = n_best // 2
            for j in range(min(half, 200)):
                diff = a_pert[j] - a_pert[n_best - 1 - j]
                a_pert[j] -= 0.015 * diff
                a_pert[n_best - 1 - j] += 0.015 * diff
        elif strat == 3:
            # Frequency smoothing
            M_fft = 1 << (2 * n_best - 2).bit_length()
            A = np.fft.rfft(best_a, M_fft)
            freqs = np.arange(len(A))
            att = 1.0 - 0.08 * (freqs / len(A)) ** 2
            A *= att
            a_pert = np.real(np.fft.irfft(A, M_fft))[:n_best]
        else:
            # Multi-peak reduction: reduce top 5 lags
            a_pert = best_a.copy()
            c = fast_conv(a_pert)
            c_max = c.max()
            threshold = 0.97 * c_max
            near_max_lags = np.where(c >= threshold)[0][:5]
            for k in near_max_lags:
                m_start = max(0, k - n_best + 1)
                m_end = min(k, n_best - 1)
                for m in range(m_start, m_end + 1):
                    other = k - m
                    if a_pert[m] > 0.01 and a_pert[other] > 0.01:
                        a_pert[m] *= 0.995
                        a_pert[other] *= 0.995

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 0.01:
            a_pert = best_a * 0.9

        tl = min(25.0, time_left / 2.0)
        sc, a_opt = run_lbfgs(a_pert, 2000, time_limit=tl)
        report(sc, a_opt, f"P4 #{trial} s={strat}")

    # ========== Final Polish ==========
    if time.monotonic() < deadline - 3:
        tl = min(25.0, deadline - 3 - time.monotonic())
        sc, a_opt = subgrad_polish(best_a, max_iters=5000, time_limit=tl)
        report(sc, a_opt, "final subgrad")

    # One more L-BFGS at very high beta
    if time.monotonic() < deadline - 5:
        tl = min(20.0, deadline - 5 - time.monotonic())
        sc, a_opt = run_lbfgs(best_a, 30000, time_limit=tl)
        report(sc, a_opt, "final lbfgs hi-beta")

    # ========== Output ==========
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
