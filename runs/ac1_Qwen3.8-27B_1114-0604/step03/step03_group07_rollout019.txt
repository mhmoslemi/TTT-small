def subgrad_polish(a, max_iters=5000, lb=1.0):

    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_indices = np.where(c >= M * (1 - 1e-10))[0]
        k_star = int(k_indices[0])
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient at the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - sc_lb) / gn2
        step = min(step, 1.0 / (1.0 + 0.01 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a

    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    cross = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]

    # P_arr[m] = sum_{k=m}^{min(m+n-1, 2n-2)} p[k]
    cumsum_p = np.concatenate([[0.0], np.cumsum(p)])
    P_arr = np.zeros(n)
    lo_idx = np.arange(n)
    hi_idx = np.minimum(lo_idx + n, 2 * n - 1)
    P_arr = cumsum_p[hi_idx] - cumsum_p[lo_idx]

    dL_da = cross + a * P_arr
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)

    return score, grad


def lse_grad_log(la, beta):
    """LSE-smoothed score and gradient w.r.t. log(a) parametrization."""
    n = len(la)
    a = np.exp(np.clip(la, -30, 10))
    sc, grad_a = lse_grad_direct(a, beta)
    grad_la = a * grad_a
    return sc, grad_la


def run_lbfgs_direct(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on LSE-smoothed objective (direct parametrization)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_grad_direct, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out, res.success


def run_lbfgs_log(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on LSE-smoothed objective (log parametrization)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()
    la0 = np.log(a0)
    try:
        res = minimize(
            lse_grad_log, la0, args=(beta,),
            method='L-BFGS', jac=True,
            options={'ftol': 1e-18, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0, False
    a_out = np.exp(np.clip(res.x, -30, 10))
    a_out = np.clip(a_out, 0.0, 1000.0)
    return compute_score(a_out), a_out, res.success


def subgrad_polish_multi(a, max_iters=5000, top_k=3):
    """Subgradient descent focusing on top-k peaks simultaneously."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top-k active peaks
        sorted_idx = np.argsort(c)[::-1]
        active = [int(sorted_idx[i]) for i in range(min(top_k, len(c)))
                  if c[sorted_idx[i]] > M * 0.999]

        # Accumulate subgradient from all active peaks
        sg = np.zeros(n)
        for k_star in active:
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        # Normalize subgradient (divide by number of active peaks)
        if len(active) > 0:
            sg /= len(active)

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - sc_lb) / gn2
        step = min(step, 0.5 / (1.0 + 0.005 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def subgrad_polish_single(a, max_iters=5000):
    """Subgradient descent on single peak (original)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - sc_lb) / gn2
        step = min(step, 0.5 / (1.0 + 0.005 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def build_inits(n, rng, parent=None):
    """Generate diverse initializations emphasizing anti-palindromic structure."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else x

    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Anti-palindromic ramps (strongest single-mode)
    inits.append(np.linspace(0.01, 1.0, n))
    inits.append(np.linspace(1.0, 0.01, n))

    # Power-law ramps
    inits.append(0.01 + 0.99 * xn ** 0.5)
    inits.append(0.01 + 0.99 * xn ** 2.0)
    inits.append(0.01 + 0.99 * (1 - xn) ** 0.5)
    inits.append(0.01 + 0.99 * (1 - xn) ** 2.0)

    # Anti-palindromic sinusoidal (odd harmonics)
    for freq in [1, 2, 3, 5, 7]:
        inits.append(np.clip(0.5 + 0.4 * np.sin(np.pi * freq * xn), 0.01, 1.0))

    # Combined: ramp + oscillation
    inits.append(np.clip(0.1 + 0.7 * xn + 0.1 * np.sin(2 * np.pi * 11 * xn), 0.01, 1.0))
    inits.append(np.clip(0.1 + 0.7 * xn + 0.1 * np.cos(2 * np.pi * 17 * xn), 0.01, 1.0))

    # W-shaped (high at edges) - may help reduce central peak
    inits.append(np.clip(0.5 + 0.4 * np.cos(np.pi * xn), 0.01, 1.0))

    # Chirp-like
    inits.append(np.clip(0.5 + 0.3 * np.sin(2 * np.pi * 3 * xn ** 2), 0.01, 1.0))

    # Random smoothed
    a_rand = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1.0)
    ksize = max(1, n // 200)
    kernel = np.ones(ksize) / ksize
    inits.append(np.clip(np.convolve(a_rand, kernel, mode='same'), 0.01, 1.0))

    # Pure random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1.0))

    # Two-block
    half = n // 2
    inits.append(np.concatenate([np.ones(half) * 0.7, np.ones(n - half) * 0.3]))

    return inits


def targeted_perturb(a, rng, strength=0.04):
    """Reduce elements contributing most to the peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    a_pert = a.copy()
    if j_end >= j_start:
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[j] * a[k_star - j]
        top_count = max(5, n // 80)
        top_idx = np.argsort(contrib)[::-1][:top_count]
        bottom_idx = np.argsort(contrib)[:n // 2]
        total_red = 0.0
        for j in top_idx:
            amt = a_pert[j] * strength
            a_pert[j] -= amt
            a_pert[j] = max(a_pert[j], 0.0)
            total_red += amt
        if len(bottom_idx) > 0 and total_red > 0:
            per = total_red / len(bottom_idx)
            a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: Quick screen of lengths and inits (low beta LSE).
    Phase 2: Deep LSE continuation (increasing beta) on best candidate.
    Phase 3: Log-parametrization refinement.
    Phase 4: Multi-peak subgradient polish.
    Phase 5: Diversified local search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for n_try in [8192, 12288, 16384]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))
    lengths = lengths[:4]

    print(f"Lengths to try: {lengths}")
    print(f"Parent: n={parent.size if parent is not None else 'none'}")

    # ============================================================
    # Phase 1: Quick screen
    # ============================================================
    phase1_end = t_start + 0.12 * budget_s
    print(f"\n=== Phase 1: Quick Screen (until t={phase1_end - t_start:.0f}s) ===")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        inits = build_inits(n, rng, parent)
        n_try_init = min(6, len(inits))
        for i in range(n_try_init):
            if time.monotonic() > phase1_end:
                break
            t0 = time.monotonic()
            sc, a_opt, _ = run_lbfgs_direct(inits[i], beta=50, maxiter=800, maxfun=15000)
            elapsed = time.monotonic() - t0
            report(sc, a_opt, f"P1 n={n} i={i} ({elapsed:.1f}s)")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ============================================================
    # Phase 2: Deep LSE continuation on best
    # ============================================================
    phase2_end = t_start + 0.55 * budget_s
    print(f"\n=== Phase 2: LSE Continuation (until t={phase2_end - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    beta_schedule = [100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 30:
            break
        maxit = max(500, min(15000, int(remaining / 3)))
        t0 = time.monotonic()
        sc, a_opt, success = run_lbfgs_direct(a_cur, beta, maxiter=maxit, maxfun=maxit * 10)
        elapsed = time.monotonic() - t0
        a_cur = a_opt
        report(sc, a_opt, f"P2 beta={beta} ({elapsed:.1f}s)")
        # Periodic re-normalization for numerical stability
        S = a_cur.sum()
        if S > 0:
            a_cur = a_cur * (50.0 / S)
            a_cur = np.clip(a_cur, 0.0, 1000.0)

    print(f"After Phase 2: best = {best_score:.10f}")

    # ============================================================
    # Phase 3: Log-parametrization refinement
    # ============================================================
    phase3_end = t_start + 0.72 * budget_s
    print(f"\n=== Phase 3: Log-Param Refinement ===")

    if time.monotonic() < phase3_end:
        a_cur = best_a.copy()
        for beta in [5000, 10000, 20000, 50000]:
            if time.monotonic() > phase3_end:
                break
            remaining = phase3_end - time.monotonic()
            if remaining < 20:
                break
            maxit = max(300, min(8000, int(remaining / 3)))
            t0 = time.monotonic()
            sc, a_opt, _ = run_lbfgs_log(a_cur, beta, maxiter=maxit, maxfun=maxit * 10)
            elapsed = time.monotonic() - t0
            a_cur = a_opt
            report(sc, a_opt, f"P3 log beta={beta} ({elapsed:.1f}s)")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ============================================================
    # Phase 4: Multi-peak subgradient polish
    # ============================================================
    phase4_end = t_start + 0.88 * budget_s
    print(f"\n=== Phase 4: Multi-Peak Subgradient ===")

    if time.monotonic() < phase4_end:
        n_iters = max(2000, int((phase4_end - time.monotonic()) * 30))
        t0 = time.monotonic()
        sc, a_opt = subgrad_polish_multi(best_a, max_iters=n_iters, top_k=3)
        report(sc, a_opt, f"P4 multi-subgrad ({time.monotonic()-t0:.1f}s)")

        # Also try single-peak subgradient
        if time.monotonic() < phase4_end:
            n_iters2 = max(1000, int((phase4_end - time.monotonic()) * 30))
            sc, a_opt = subgrad_polish_single(best_a, max_iters=n_iters2)
            report(sc, a_opt, "P4 single-subgrad")

    # ============================================================
    # Phase 5: Diversified local search
    # ============================================================
    print(f"\n=== Phase 5: Local Search ===")
    n_best = len(best_a)
    trial = 0
    while time.monotonic() < deadline - 30 and trial < 25:
        trial += 1
        strat = trial % 5
        if strat == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strat == 1:
            # Targeted perturbation
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.03 * rng.random())
        elif strat == 2:
            # Anti-palindromic bump at center
            a_pert = best_a.copy()
            center = n_best // 2
            width = max(1, n_best // 10)
            for j in range(n_best):
                d = (j - center) / width
                sign = 1.0 if j < center else -1.0
                a_pert[j] *= (1.0 + 0.06 * sign * np.exp(-0.5 * d * d))
        elif strat == 3:
            # Frequency-domain smoothing
            a_pert = best_a.copy()
            M_fft = 1 << (2 * n_best - 1 - 1).bit_length()
            A = np.fft.rfft(a_pert, M_fft)
            freqs = np.arange(len(A))
            att = 1.0 - 0.08 * (freqs / len(A)) ** 2
            A *= att
            a_pert = np.real(np.fft.irfft(A, M_fft))[:n_best]
        else:
            # Random block swap
            a_pert = best_a.copy()
            blk = max(10, n_best // 100)
            i1 = rng.integers(0, n_best - 2 * blk)
            i2 = rng.integers(0, n_best - 2 * blk)
            if abs(i1 - i2) > blk:
                tmp = a_pert[i1:i1 + blk].copy()
                a_pert[i1:i1 + blk] = a_pert[i2:i2 + blk]
                a_pert[i2:i2 + blk] = tmp

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            a_pert = best_a * 0.9

        sc, a_opt, _ = run_lbfgs_direct(a_pert, beta=10000, maxiter=3000, maxfun=30000)
        report(sc, a_opt, f"P5 #{trial} s={strat}")

    # ============================================================
    # Final polish
    # ============================================================
    if time.monotonic() < deadline - 10:
        sc, a_opt = subgrad_polish_multi(best_a, max_iters=3000, top_k=3)
        report(sc, a_opt, "final-subgrad")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
