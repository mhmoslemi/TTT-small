def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def entropy_score_grad(a, beta):
    """
    Entropy-smoothed max: L = (1/beta) * log(sum(exp(beta * c)))
    This is the log-sum-exp, equivalent to LSE.
    But we add a floor to prevent numerical issues.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip to prevent overflow
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: sum_k p[k] * d(conv[k])/da
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def topk_entropy_score_grad(a, k, gamma):
    """
    Weighted top-k: weight each c[j] by exp(-gamma * rank(j)) and normalize.
    This gives more weight to the top elements but still smooth.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = c.max()
    # Soft weighting: w[j] = exp(-gamma * (c_max - c[j]))
    # This gives weight ~1 to elements near the max, decaying for smaller ones
    diff = c_max - c
    w = np.exp(-gamma * diff)
    w = w / w.sum()
    L = float(np.dot(w, c))
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    W = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(RA * W, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def solve_lp_direction(a, M_override=None, max_wall_s=30.0):
    """
    Solve LP: max sum(b) s.t. b>=0, conv(a,b)[k] <= M for all k.
    """
    if not HAVE_CVXPY:
        return None
    n = len(a)
    t0 = time.monotonic()
    if n > 8000:
        return None
    if M_override is None:
        M = float(np.max(fast_conv(a)))
    else:
        M = M_override
    if M < 1e-12:
        return a.copy()
    
    # Build constraint matrix
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_mat[k_start:k_end, j] = a[:k_end - k_start]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    
    try:
        remaining = max_wall_s - (time.monotonic() - t0)
        if remaining <= 0:
            return None
        prob.solve(solver=cp.PDLP, max_time=remaining)
    except Exception:
        try:
            remaining = max_wall_s - (time.monotonic() - t0)
            if remaining <= 0:
                return None
            prob.solve(solver=cp.ECOS, max_time=remaining)
        except Exception:
            return None
    
    if b_var.value is None:
        return None
    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None
    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    return b_opt


def lp_step_multi_m(a, max_wall_s=20.0):
    """
    Try multiple M values (slightly above and below current max) to find best direction.
    """
    c = fast_conv(a)
    M_cur = float(c.max())
    S = float(a.sum())
    old_sc = 2.0 * len(a) * M_cur / (S * S)
    
    # Try M values: current, slightly above, slightly below
    M_values = [M_cur, M_cur * 1.01, M_cur * 0.99, M_cur * 1.05, M_cur * 0.95]
    
    best_sc = old_sc
    best_a = a
    
    for M_val in M_values:
        b_opt = solve_lp_direction(a, M_override=M_val, max_wall_s=max_wall_s)
        if b_opt is None:
            continue
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_new = (1.0 - t) * a + t * b_opt
            a_new = np.clip(a_new, 0.0, 1000.0)
            if a_new.sum() < 0.01:
                continue
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new
    
    return best_sc, best_a

def targeted_perturb_v2(a, rng, strength=0.05):
    """
    Perturb elements that contribute most to the peak convolution lag,
    and redistribute to elements that contribute least.
    """

    c = fast_conv(a)
    
    # Find all positions near the maximum
    c_max = c.max()
    threshold = c_max * 0.95
    active_positions = np.where(c >= threshold)[0]
    
    # Compute contribution of each element to the active positions
    contrib = np.zeros(n)
    for k in active_positions:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]
    
    # Reduce top contributors, increase bottom contributors
    top_count = max(5, n // 40)
    bottom_count = max(10, n // 20)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:bottom_count]
    
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    
    return np.clip(a_pert, 0.0, 1000.0)


def build_inits_v2(n, rng, parent):

        else:

            inits.append(parent[idx].astype(np.float64).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))

    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, None))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, None))

    for nb in (8, 16, 32, 64):

        a = np.repeat(vals, per)

"""
Improved optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements over previous: (1) combined LSE + L2-spread objective for better flattening,
(2) multi-M LP exploitation with dual variable recovery, (3) Nesterov-accelerated subgradient
with adaptive top-k, (4) adaptive-beta continuation, (5) extended runtime allocation to LP.
"""

import numpy as np
from scipy.optimize import minimize
import time

    import cvxpy as cp
    HAVE_CVXPY = True

    HAVE_CVXPY = False


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """Linear cross-convolution via FFT."""
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta, alpha_l2):
    """
    LSE-smoothed score + L2 spread penalty.
    Objective: 2n * (LSE(c) + alpha_l2 * R) / (sum(a))^2
    where R = sqrt(mean(c^2)) (RMS of convolution).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    
    # L2 spread term
    R = float(np.sqrt(np.mean(c * c)))
    
    L_total = L + alpha_l2 * R
    score = 2.0 * n * L_total / (S * S)
    
    # Gradient of LSE part
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_lse = 2.0 * conv_res
    
    # Gradient of R = sqrt(mean(c^2)) = sqrt(sum(c^2)/(2n-1))
    # dR/dc[k] = c[k] / ((2n-1) * R)
    # dR/da[i] = sum_k (c[k] / ((2n-1)*R)) * dc[k]/da[i]
    if R > 1e-12:
        w_r = c / ((2 * n - 1) * R)
        Wr = np.fft.rfft(w_r, M2_fft)
        conv_res_r = np.fft.irfft(RA * Wr, M2_fft)[n - 1:2 * n - 1]
        g_r = 2.0 * conv_res_r
    else:
        g_r = np.zeros(n)
    
    g_total = g_lse + alpha_l2 * g_r
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_total / S2 - 2.0 * L_total / S3)
    return score, grad


def lse_pure_score_grad(a, beta):
    """Pure LSE-smoothed score (no L2 penalty)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def make_lbfgs_obj_combined(beta, alpha_l2):
    """Create objective for L-BFGS with combined loss."""
    def obj(x):
        return lse_score_grad(x, beta, alpha_l2)
    return obj


def make_lbfgs_obj_pure(beta):
    """Create objective for L-BFGS with pure LSE."""
    def obj(x):
        return lse_pure_score_grad(x, beta)
    return obj


def run_lbfgs_combined(a0, beta, alpha_l2, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B with combined LSE + L2 objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    obj = make_lbfgs_obj_combined(beta, alpha_l2)
    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out

def solve_lp_dir(a, M_override=None, max_wall_s=30.0):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)[k] <= M."""
    if not HAVE_CVXPY:
        return None
    n = len(a)
    t0 = time.monotonic()
    if n > 10000:
        return None
    if M_override is None:
        M = float(np.max(fast_conv(a)))
    else:
        M = M_override
    if M < 1e-12:
        return a.copy()
    
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_mat[k_start:k_end, j] = a[:k_end - k_start]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    
    try:
        remaining = max(max_wall_s - (time.monotonic() - t0), 1.0)
        prob.solve(solver=cp.PDLP, max_time=remaining)
    except Exception:
        try:
            remaining = max(max_wall_s - (time.monotonic() - t0), 1.0)
            prob.solve(solver=cp.ECOS, max_time=remaining)
        except Exception:
            return None
    
    if b_var.value is None:
        return None
    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None
    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    return b_opt


def lp_step_multi(a, max_wall_s=20.0):
    """Try multiple M values for LP direction."""
    c = fast_conv(a)
    M_cur = float(c.max())
    S = float(a.sum())
    old_sc = 2.0 * len(a) * M_cur / (S * S)
    
    M_values = [M_cur, M_cur * 1.005, M_cur * 1.02, M_cur * 1.05,
                M_cur * 0.98, M_cur * 0.95]
    
    best_sc = old_sc
    best_a = a
    
    for M_val in M_values:
        if time.monotonic() - t0_global > 100:
            pass
        b_opt = solve_lp_dir(a, M_override=M_val, max_wall_s=max(max_wall_s / 3, 5.0))
        if b_opt is None:
            continue
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_new = (1.0 - t) * a + t * b_opt
            a_new = np.clip(a_new, 0.0, 1000.0)
            if a_new.sum() < 0.01:
                continue
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
    
    return best_sc, best_a


def nesterov_subgrad(a, n_iters=2000, max_wall_s=60.0):
    """Nesterov-accelerated projected subgradient with top-k averaging."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    
    y = a.copy()
    x_prev = a.copy()
    momentum = 0.0
    
    t0 = time.monotonic()
    for it in range(n_iters):
        if time.monotonic() - t0 > max_wall_s:
            break
        
        c = fast_conv(y)
        S = float(y.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        
        a_check = np.clip(y, 0.0, 1000.0)
        if a_check.sum() >= 0.01:
            sc_check = compute_score(a_check)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a_check.copy()
        
        # Adaptive top-k: use more elements if they're close together
        c_sorted = np.sort(c)[::-1]
        k = max(3, min(20, n // 200))
        if k < len(c_sorted):
            gap = max(c_sorted[0] - c_sorted[min(k, len(c_sorted)-1)], 1e-10)
            k_adapt = max(3, min(50, int(k * c_sorted[0] / gap) + 1))
        else:
            k_adapt = k
        
        c_sorted_idx = np.argsort(c)[::-1][:k_adapt]
        top_vals = c[c_sorted_idx]
        weights = np.exp(-0.3 * (top_vals[0] - top_vals))
        weights = weights / weights.sum()
        
        sg = np.zeros(n)
        for idx, w in zip(c_sorted_idx, weights):
            k_star = int(idx)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                seg = y[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        L = float(np.dot(weights, top_vals))
        grad = (2.0 * n) * (sg / S2 - 2.0 * L / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        
        # Nesterov: beta_k = 2/(k+2)
        beta_k = 2.0 / (it + 2)
        x_new = y - step * grad
        x_new = np.clip(x_new, 0.0, 1000.0)
        y = x_new + (beta_k - 1.0) / beta_k * (x_new - x_prev)
        y = np.clip(y, 0.0, 1000.0)
        x_prev = x_new
    
    return best_sc, best_a


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing to peak convolution."""
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    threshold = c_max * 0.95
    active_positions = np.where(c >= threshold)[0]
    
    contrib = np.zeros(n)
    for k in active_positions:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]
    
    top_count = max(5, n // 40)
    bottom_count = max(10, n // 20)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:bottom_count]
    
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    
    return np.clip(a_pert, 0.0, 1000.0)


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].astype(np.float64).copy())
    
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(np.clip(0.5 * (1.0 + np.cos(2.0 * np.pi * x)), 0.01, None))
    inits.append(np.clip(0.3 + 0.7 * np.cos(np.pi * x) ** 2, 0.01, None))
    inits.append(np.clip(0.5 + 0.5 * np.cos(np.pi * (2.0 * x - 1.0)), 0.01, None))
    inits.append(np.clip(0.5 * (1.0 + np.cos(2.0 * np.pi * x)) + 0.1 * np.sin(4.0 * np.pi * x), 0.01, None))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, None))
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, None))
    inits.append(np.clip(0.3 + 0.5 * rng.random(n), 0.01, None))
    
    for nb in (8, 16, 32):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    
    a_alt = np.ones(n)
    a_alt[1::2] = 0.3
    inits.append(a_alt * 0.5)
    
    inits.append(np.clip(0.5 + 0.3 * np.sinc(2.0 * (x - 0.5)), 0.01, None))
    
    return inits
