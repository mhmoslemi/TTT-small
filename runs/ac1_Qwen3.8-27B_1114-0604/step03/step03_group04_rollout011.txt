"""
Coordinate descent optimizer for minimizing peak autocorrelation of non-negative sequences.
Uses exact 1D subproblem solving via golden section search for each coordinate,
combined with peak-targeted perturbation and LSE-smoothed L-BFGS polishing.
Key insight: the convolution update for a single coordinate is O(n) with a fixed
weight vector (2*a), enabling fast sweeps even for n~7000+.
"""

import numpy as np
from scipy.optimize import minimize
import time


def compute_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a, conv=None):
    n = len(a)
    if conv is None:
        conv = compute_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(conv)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    conv = compute_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(conv.max())
    scaled = beta * (conv - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500, 500))
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def coord_desc_sweep(a, c, n, S, order, max_gs_iters=30):
    """
    One full coordinate descent sweep.
    a: sequence (modified in place)
    c: convolution (modified in place)
    n: length
    S: sum (returned updated)
    order: permutation of indices
    Returns updated S.
    """
    two_a = 2.0 * a
    a_i_old = a[0]  # to detect changes

    for i in order:
        a_i = a[i]
        S_others = S - a_i

        # c_slice: c[i:i+n] (view)
        c_slice = c[i:i + n]

        # c_inv_max: max over inactive lags
        c_inv_max = -np.inf
        if i > 0:
            c_inv_max = float(np.max(c[:i]))
        if i + n < 2 * n - 1:
            right = float(np.max(c[i + n:]))
            c_inv_max = max(c_inv_max, right)
        if c_inv_max == -np.inf:
            c_inv_max = 0.0

        diag_base = c[2 * i]  # c[2i]

        # Define objective as function of a_new
        def obj_fn(a_new):
            delta = a_new - a_i
            delta_sq = a_new * a_new - a_i * a_i
            # max over active: c[i+j] + delta*2*a[j] for all j
            max_active = float(np.max(c_slice + delta * two_a))
            diag = diag_base + delta_sq
            total_max = max(max_active, diag, c_inv_max)
            S_new = S_others + a_new
            if S_new < 1e-12:
                return 1e15
            return 2.0 * n * total_max / (S_new * S_new)

        # Initial bracket
        upper = min(1000.0, max(3.0 * a_i, 3.0 * S / n, 1.0))
        if a_i < 1e-10:
            upper = max(upper, 1.0)

        # Try several starting points
        candidates = [0.0, a_i * 0.5, a_i, a_i * 1.5, upper * 0.5, upper]
        candidates = [max(0.0, min(1000.0, float(x))) for x in candidates]
        # Remove duplicates
        seen = set()
        unique_candidates = []
        for x in candidates:
            if x not in seen:
                seen.add(x)
                unique_candidates.append(x)
        candidates = unique_candidates

        scores = [obj_fn(x) for x in candidates]
        best_idx = int(np.argmin(scores))
        best_val = candidates[best_idx]

        # Golden section refinement
        if best_idx > 0 and best_idx < len(candidates) - 1:
            lo = candidates[best_idx - 1]
            hi = candidates[best_idx + 1]
        elif best_idx == 0:
            lo = 0.0
            hi = candidates[1] if len(candidates) > 1 else 1.0
        else:
            lo = candidates[-2]
            hi = candidates[-1]

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(max_gs_iters):
            m1 = hi - gr * (hi - lo)
            m2 = lo + gr * (hi - lo)
            f1 = obj_fn(m1)
            f2 = obj_fn(m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
            if abs(hi - lo) < 1e-12 * max(1.0, a_i):
                break

        new_val = (lo + hi) / 2.0
        new_val = max(0.0, min(1000.0, new_val))

        # Compare with current
        if obj_fn(new_val) < obj_fn(a_i):
            delta = new_val - a_i
            delta_sq = new_val * new_val - a_i * a_i
            # Update c
            c[i:i + n] += delta * two_a[i:i + n]  # WRONG - two_a changed? No, two_a is 2*a before update
            # Actually need to be careful: for position i in the active range (j=i, k=2i):
            # The update should be delta_sq at k=2i, not delta*2*a[i]
            # Let me redo:
            # c[k] += delta * 2*a[k-i] for k in [i, i+n-1], k != 2i
            # c[2i] += delta_sq
            # Since two_a = 2*a (old values), for k != 2i: c[k] += delta * two_a[k-i]
            # For k = 2i: c[2i] should get delta_sq, but the formula gives delta * two_a[i] = delta * 2*a[i]
            # delta_sq = delta*(a_new + a_i) = delta*(a_i + delta + a_i) = delta*2*a_i + delta^2
            # So c[2i] += delta*2*a_i + delta^2 = delta*two_a[i] + delta^2
            # The general formula c[i:i+n] += delta * two_a gives delta*2*a[i] at position i
            # We need to add an extra delta^2 at position i (k=2i)
            c[i:i + n] += delta * two_a  # this updates all active lags
            # But wait: two_a has length n, and c[i:i+n] has length n. two_a[j] = 2*a[j].
            # c[i+j] += delta * 2*a[j] for j=0..n-1. This is correct for j != i.
            # For j = i (k = 2i): we added delta*2*a[i], but we need delta_sq = delta*2*a[i] + delta^2
            # So add extra delta^2:
            if 2 * i - i < n:  # always true
                c[2 * i] += delta * delta
            # Update a and S
            a[i] = new_val
            two_a[i] = 2.0 * new_val
            S += delta

    return S


def spectral_flatten_simple(a0, N_fft, n_iters=100, lam=0.5):
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        if len(non_dc_mag) == 0:
            break
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        half = N_fft // 2
        for k in range(1, half + 1):
            if N_fft % 2 == 0 and k == half:
                new_mag[k] = (1.0 - lam) * mag[k] + lam * target
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N_fft - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a


def targeted_peak_reduction(a, c, n, rng, strength=0.03):
    """Reduce the peak lag by redistributing mass from high-contributing coords."""
    k_star = int(np.argmax(c))
    # Coordinates contributing to c[k_star]: a[j] * a[k_star - j]
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1] * a[k_star - j_end:k_star - j_start + 1][::-1]

    top_count = max(5, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]

    a_new = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_new[j] * strength
        a_new[j] -= reduce_amt
        a_new[j] = max(a_new[j], 0.0)
        total_reduced += reduce_amt

    if total_reduced > 0:
        bottom_idx = np.argsort(contrib)[:top_count]
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_new[j] += per

    return np.clip(a_new, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function. Uses coordinate descent as the primary optimizer,
    with peak-targeted perturbations and LSE-L-BFGS polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0
    deadline = t_start + effective_budget

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Build initializations
    print("Building initializations...")
    inits = []
    lengths_to_try = []

    if parent is not None:
        inits.append(parent.copy())
        lengths_to_try.append(parent.size)
        print(f"  Parent loaded: n={parent.size}, score={compute_score(parent):.6f}")
        sc_parent = compute_score(parent)
        report(sc_parent, parent, "parent")

    # Generate additional initializations at various lengths
    for n in [4096, 8192, 12288, 16384]:
        if n in lengths_to_try:
            continue
        if time.monotonic() > deadline:
            break
        N_fft = 1 << (2 * n - 1).bit_length()
        # Uniform
        a = np.ones(n) * (n / (2 * n))  # sum = n/2
        a = np.ones(n) * 0.5
        sc = compute_score(a)
        inits.append(a)
        lengths_to_try.append(n)
        print(f"  Uniform n={n}: score={sc:.6f}")
        # Spectral flatten
        sf_a = spectral_flatten_simple(np.ones(n) * 0.5, N_fft, n_iters=80, lam=0.6)
        sc = compute_score(sf_a)
        inits.append(sf_a)
        print(f"  SpectralFlatten n={n}: score={sc:.6f}")
        # Chirp
        x = np.arange(n, dtype=np.float64)
        chirp = np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01
        sf_chirp = spectral_flatten_simple(chirp, N_fft, n_iters=60, lam=0.5)
        sc = compute_score(sf_chirp)
        inits.append(sf_chirp)
        print(f"  ChirpSF n={n}: score={sc:.6f}")

    # Sort initializations by score
    scored_inits = [(compute_score(a), i, a) for i, a in enumerate(inits)]
    scored_inits.sort(key=lambda x: x[0])

    print(f"\nTop initializations:")
    for sc, idx, a in scored_inits[:5]:
        print(f"  n={len(a)}, score={sc:.6f}")

    # Phase 1: Coordinate descent from top initializations
    print(f"\n{'='*60}")
    print("Phase 1: Coordinate Descent")
    print(f"{'='*60}")

    cd_deadline = t_start + 0.65 * effective_budget

    for sc_init, idx, a_init in scored_inits[:min(6, len(scored_inits))]:
        if time.monotonic() > cd_deadline:
            break
        n = len(a_init)
        a = a_init.astype(np.float64).copy()
        S = float(a.sum())
        if S < 1e-12:
            continue
        c = compute_conv(a)
        sc = compute_score(a, c)
        report(sc, a, f"CD start n={n}")

        n_sweeps_done = 0
        prev_sc = sc
        stagnation = 0

        while time.monotonic() < cd_deadline:
            # Determine sweep order
            order = np.arange(n)
            r = rng.random()
            if r < 0.5:
                rng.shuffle(order)
            elif r < 0.75:
                order = order[::-1]

            S = coord_desc_sweep(a, c, n, S, order, max_gs_iters=25)
            sc = compute_score(a, c)
            n_sweeps_done += 1

            if sc < best_score:
                report(sc, a, f"CD n={n} sweep={n_sweeps_done}")
            elif sc < prev_sc - 1e-8:
                pass
            else:
                stagnation += 1

            if sc < prev_sc - 1e-10:
                stagnation = 0
            prev_sc = sc

            # Periodic perturbation to escape local minima
            if stagnation >= 15:
                stagnation = 0
                strategy = rng.integers(0, 4)
                if strategy == 0:
                    a = targeted_peak_reduction(a, c, n, rng, strength=0.02 + 0.04 * rng.random())
                    c = compute_conv(a)
                    S = float(a.sum())
                elif strategy == 1:
                    # Random coordinate perturbation
                    n_perturb = max(5, n // 100)
                    pert_idx = rng.choice(n, size=n_perturb, replace=False)
                    for j in pert_idx:
                        a[j] = max(0.0, a[j] * (1.0 + 0.3 * rng.standard_normal()))
                    a = np.clip(a, 0.0, 1000.0)
                    c = compute_conv(a)
                    S = float(a.sum())
                elif strategy == 2:
                    # Scaled perturbation
                    scale = 1.0 + 0.05 * rng.standard_normal(n)
                    a = np.clip(a * scale, 0.0, 1000.0)
                    c = compute_conv(a)
                    S = float(a.sum())
                else:
                    # Swap-based perturbation: move mass from peak contributors to others
                    a = targeted_peak_reduction(a, c, n, rng, strength=0.01)
                    c = compute_conv(a)
                    S = float(a.sum())

                sc = compute_score(a, c)
                sc = min(sc, compute_score(a))  # verify
            elif stagnation >= 5:
                # Mild perturbation
                n_perturb = max(3, n // 200)
                pert_idx = rng.choice(n, size=n_perturb, replace=False)
                for j in pert_idx:
                    a[j] = max(0.0, a[j] * (1.0 + 0.1 * rng.standard_normal()))
                a = np.clip(a, 0.0, 1000.0)
                c = compute_conv(a)
                S = float(a.sum())
                sc = compute_score(a, c)

            # Progress print
            if n_sweeps_done % 50 == 0:
                elapsed = time.monotonic() - t_start
                remaining = deadline - time.monotonic()
                print(f"    n={n}: sweep={n_sweeps_done}, sc={sc:.8f}, "
                      f"best={best_score:.8f}, t={elapsed:.0f}s, rem={remaining:.0f}s")

        # Final score for this initialization
        sc_final = compute_score(a)
        report(sc_final, a, f"CD final n={n} idx={idx}")

    print(f"\nAfter Phase 1: best = {best_score:.8f} (n={len(best_a) if best_a is not None else 0})")

    # Phase 2: LSE-L-BFGS polishing
    print(f"\n{'='*60}")
    print("Phase 2: LSE-L-BFGS Polish")
    print(f"{'='*60}")

    lbfgs_deadline = t_start + 0.85 * effective_budget
    a_cur = best_a.copy()
    n_cur = len(a_cur)

    for beta in [100, 300, 800, 2000, 5000]:
        if time.monotonic() > lbfgs_deadline:
            break
        remaining = lbfgs_deadline - time.monotonic()
        if remaining < 10:
            break
        maxiter = min(5000, int(remaining * 0.7 / 0.05))
        sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=max(200, maxiter), maxfun=max(1000, maxiter * 20))
        report(sc, a_opt, f"L-BFGS beta={beta}")
        a_cur = a_opt

    # Phase 3: More coordinate descent from polished result
    print(f"\n{'='*60}")
    print("Phase 3: Post-Polish Coordinate Descent")
    print(f"{'='*60}")

    final_deadline = t_start + 0.95 * effective_budget
    a = a_cur.astype(np.float64).copy()
    n = len(a)
    S = float(a.sum())
    c = compute_conv(a)
    sc = compute_score(a, c)
    report(sc, a, "Pre-Phase3")

    prev_sc = sc
    stagnation = 0
    sweep_count = 0
    while time.monotonic() < final_deadline:
        order = np.arange(n)
        r = rng.random()
        if r < 0.5:
            rng.shuffle(order)
        elif r < 0.8:
            order = order[::-1]

        S = coord_desc_sweep(a, c, n, S, order, max_gs_iters=30)
        sc = compute_score(a, c)
        sweep_count += 1

        if sc < best_score:
            report(sc, a, f"P3 CD sweep={sweep_count}")
        elif sc < prev_sc - 1e-10:
            stagnation = 0
        else:
            stagnation += 1
        prev_sc = sc

        if stagnation >= 20:
            stagnation = 0
            a = targeted_peak_reduction(a, c, n, rng, strength=0.015 + 0.03 * rng.random())
            c = compute_conv(a)
            S = float(a.sum())
            sc = compute_score(a, c)
        elif stagnation >= 8:
            n_p = max(3, n // 150)
            pert_idx = rng.choice(n, size=n_p, replace=False)
            for j in pert_idx:
                a[j] = max(0.0, a[j] * (1.0 + 0.08 * rng.standard_normal()))
            a = np.clip(a, 0.0, 1000.0)
            c = compute_conv(a)
            S = float(a.sum())
            sc = compute_score(a, c)

        if sweep_count % 30 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    P3: sweep={sweep_count}, sc={sc:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")

    # Phase 4: Final L-BFGS + coordinate descent alternation
    print(f"\n{'='*60}")
    print("Phase 4: Final Alternation")
    print(f"{'='*60}")

    abs_deadline = t_start + effective_budget
    a = best_a.astype(np.float64).copy()
    n = len(a)

    for round_i in range(10):
        if time.monotonic() > abs_deadline - 10:
            break
        # L-BFGS step
        remaining = abs_deadline - time.monotonic() - 30
        if remaining > 20:
            beta = 1000 + 500 * round_i
            maxiter = min(2000, int(remaining * 0.5 / 0.05))
            sc, a_opt, _ = run_lbfgs(a, beta, maxiter=max(100, maxiter), maxfun=max(500, maxiter * 15))
            report(sc, a_opt, f"Final LBFGS r={round_i}")
            a = a_opt

        # Coordinate descent step
        if time.monotonic() > abs_deadline - 15:
            break
        n_cd = len(a)
        S = float(a.sum())
        c = compute_conv(a)
        for _ in range(min(5, max(1, int((abs_deadline - time.monotonic() - 5) / 5)))):
            if time.monotonic() > abs_deadline - 5:
                break
            order = np.arange(n_cd)
            rng.shuffle(order)
            S = coord_desc_sweep(a, c, n_cd, S, order, max_gs_iters=35)
            sc = compute_score(a, c)
            report(sc, a, f"Final CD round={round_i}")

    # Final verification
    print(f"\n{'='*60}")
    print("Final Verification")
    print(f"{'='*60}")
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    internal = compute_score(np.array(out))
    elapsed_total = time.monotonic() - t_start
    print(f"FINAL: internal={internal:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
