"""
Optimized search for minimum peak autocorrelation non-negative sequences.
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using multi-phase LSE-continuation
gradient descent with subgradient polishing and targeted local search.
Key improvements: (1) verified gradient via FFT cross-correlation,
(2) normalized optimization for numerical stability, (3) aggressive beta
schedule focused on refining parent, (4) adaptive subgradient step sizes,
(5) peak-lag-targeted perturbations.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and exact gradient.
    score = 2n * LSE_beta(conv(a,a)) / S^2
    Gradient computed via FFT cross-correlation.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()

    score = 2.0 * n * L / (S * S)

    # Gradient of LSE w.r.t. a[j] = 2 * sum_m a[m]*p[j+m]
    # Computed as: 2 * (a * p_rev)[2n-2-j] for j=0..n-1
    p_rev = p[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    PR_fft = np.fft.rfft(p_rev, M2_fft)
    conv_ap = np.fft.irfft(A_fft * PR_fft, M2_fft)[:M2]
    # Need indices n-1 to 2n-2, reversed
    conv_res = conv_ap[n - 1:2 * n - 1][::-1]
    grad_LSE = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_LSE / S2 - 2.0 * L / S3)

    return score, grad


def lse_wrapper(a, beta):
    """Scipy-compatible objective with gradient."""
    return lse_score_grad(a, beta)


def norm_subgrad_polish(a, max_iters=5000, target_lb=1.50):
    """
    Subgradient polish with Polyak step size and adaptive damping.
    Works on the normalized sequence (sum=1) for stability.
    """
    n = len(a)
    S0 = float(a.sum())
    if S0 < 1e-12:
        return compute_score(a), a.copy()
    # Normalize to sum = 1
    a_norm = a.astype(np.float64) / S0
    sc0 = compute_score(a_norm)
    best_sc = sc0
    best_a = a_norm.copy()
    
    sc_lb = target_lb  # Use target as optimistic lower bound

    step_scale = 1.0
    for it in range(max_iters):
        c = fast_conv(a_norm)
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M  # since sum=1, S^2=1

        if sc < best_sc:
            best_sc = sc
            best_a = a_norm.copy()

        # Subgradient of max(conv) at peak lag k_star:
        # db[k_star]/da[j] = 2*a[k_star - j] if 0 <= k_star - j < n
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            # sg[j] = 2 * a[k_star - j] for valid j
            indices = np.arange(j_start, j_end + 1)
            sg[indices] = 2.0 * a_norm[k_star - indices]

        # Gradient of score = 2n*M (with S=1): grad = 2n * sg (since S=1, the S-derivative term vanishes)
        # Actually: score = 2n*M/S^2, with S=1: score = 2n*M
        # d(score)/d(a[j]) = 2n * dM/da[j] (since S is fixed at 1 via normalization)
        # But we're not actually enforcing S=1 during the step... 
        # Let me use the full gradient: score = 2n*M/S^2
        S = float(a_norm.sum())
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break

        # Polyak step with damping
        gap = sc - sc_lb
        if gap <= 0:
            break
        step = gap / gn2
        step = min(step, step_scale / (1.0 + 0.001 * it))
        step = min(step, 1.0)  # Cap step size

        a_norm = a_norm - step * grad
        a_norm = np.clip(a_norm, 0.0, None)
        # Re-normalize to sum=1
        S_new = float(a_norm.sum())
        if S_new < 1e-12:
            break
        a_norm = a_norm / S_new

        # Adaptive: if score increased, reduce step scale
        if it > 20:
            cur_sc = 2.0 * n * float(fast_conv(a_norm).max())
            if cur_sc > best_sc * 1.001:
                step_scale *= 0.8

    return best_sc, best_a * S0  # Scale back


def lse_subgrad_hybrid(a, max_iters=8000, beta_start=500, beta_end=50000, target_lb=1.50):
    """
    Hybrid: alternate between LSE-smoothed gradient and exact subgradient.
    Gradually increase beta for sharper peak tracking.
    """
    n = len(a)
    S0 = float(a.sum())
    if S0 < 1e-12:
        return compute_score(a), a.copy()
    a_cur = a.astype(np.float64) / S0
    best_sc = compute_score(a_cur)
    best_a = a_cur.copy()
    sc_lb = target_lb

    n_beta_steps = min(max_iters, 8)
    betas = np.geomspace(beta_start, beta_end, n_beta_steps)
    iters_per_beta = max(50, max_iters // n_beta_steps)
    it = 0

    for beta in betas:
        # LSE phase
        for _ in range(iters_per_beta):
            if it >= max_iters:
                break
            it += 1
            score, grad = lse_score_grad(a_cur, beta)
            S = float(a_cur.sum())
            if S < 1e-12:
                break
            exact_sc = 2.0 * n * float(fast_conv(a_cur).max())
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a_cur.copy()

            gn2 = float(grad @ grad)
            if gn2 < 1e-30:
                break
            gap = exact_sc - sc_lb
            if gap <= 0:
                return best_sc, best_a * S0
            step = gap / gn2
            step = min(step, 0.5 / (1.0 + 0.002 * it))
            a_cur = a_cur - step * grad
            a_cur = np.clip(a_cur, 0.0, None)
            S_new = float(a_cur.sum())
            if S_new < 1e-12:
                break
            a_cur = a_cur / S_new

        # Subgradient phase (refine the exact peak)
        n_sg = max(50, iters_per_beta // 3)
        for _ in range(n_sg):
            if it >= max_iters:
                break
            it += 1
            c = fast_conv(a_cur)
            M = float(c.max())
            k_star = int(np.argmax(c))
            sc = 2.0 * n * M

            if sc < best_sc:
                best_sc = sc
                best_a = a_cur.copy()

            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            sg = np.zeros(n)
            if j_end >= j_start:
                indices = np.arange(j_start, j_end + 1)
                sg[indices] = 2.0 * a_cur[k_star - indices]

            S = float(a_cur.sum())
            S2 = S * S
            S3 = S2 * S
            grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
            gn2 = float(grad @ grad)
            if gn2 < 1e-30:
                break
            gap = sc - sc_lb
            if gap <= 0:
                break
            step = gap / gn2
            step = min(step, 0.3 / (1.0 + 0.001 * it))
            a_cur = a_cur - step * grad
            a_cur = np.clip(a_cur, 0.0, None)
            S_new = float(a_cur.sum())
            if S_new < 1e-12:
                break
            a_cur = a_cur / S_new

    return best_sc, best_a * S0


def targeted_perturbation(a, rng, strength=0.04):
    """
    Create a perturbation that specifically reduces the peak lag contribution.
    Reduces elements that pair at the peak lag, redistributes to non-peak lags.
    """
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    
    a_pert = a.copy()
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    
    if j_end >= j_start:
        # Compute contribution of each pair at the peak lag
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            partner = k_star - j
            if 0 <= partner < n:
                contrib[j] = a[j] * a[partner]
        
        # Reduce top contributors
        n_top = max(5, n // 150)
        top_idx = np.argsort(contrib)[::-1][:n_top]
        bottom_idx = np.argsort(contrib)[:-n_top] if len(contrib) > n_top else np.array([], dtype=int)
        
        total_red = 0.0
        for j in top_idx:
            amt = a_pert[j] * strength
            a_pert[j] -= amt
            a_pert[j] = max(a_pert[j], 0.0)
            total_red += amt
        
        if len(bottom_idx) > 0 and total_red > 0:
            per = total_red / len(bottom_idx)
            a_pert[bottom_idx] += per
    
    return np.clip(a_pert, 0.0, 1000.0)


def antipalindromic_perturbation(a, rng, strength=0.06):
    """Strengthen anti-palindromic structure to reduce central peak."""
    n = len(a)
    a_pert = a.copy()
    C = float(a.sum()) / n * 2.0  # Target anti-palindromic constant
    for i in range(n // 2):
        j = n - 1 - i
        current_sum = a_pert[i] + a_pert[j]
        if current_sum > 1e-12:
            # Move toward anti-palindromic: a[i] + a[j] = C
            target_i = min(1000.0, max(0.0, (C + (a[i] - a[j])) / 2.0))
            target_j = min(1000.0, max(0.0, C - target_i))
            a_pert[i] += strength * (target_i - a_pert[i])
            a_pert[j] += strength * (target_j - a_pert[j])
    return np.clip(a_pert, 0.0, 1000.0)


def spectral_smooth_perturbation(a, rng, amount=0.05):
    """Apply mild spectral smoothing to reduce oscillatory components."""
    n = len(a)
    M_fft = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M_fft)
    freqs = np.arange(len(A))
    # Attenuate high frequencies
    att = 1.0 - amount * (freqs / len(A)) ** 1.5
    A *= att
    a_pert = np.real(np.fft.irfft(A, M_fft))[:n]
    return np.clip(a_pert, 0.0, 1000.0)
