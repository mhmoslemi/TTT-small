def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def spectral_flatten_v2(a0, N, n_iters=300, lam=0.5):
    """Improved spectral flattening with better convergence."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    S0 = a.sum()
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        if N % 2 == 0:
            non_dc_mag = np.concatenate([mag[1:N//2], mag[N//2+1:]])
        else:
            non_dc_mag = mag[1:(N+1)//2]
        target = float(np.median(non_dc_mag))
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (S0 / S)
    return a


def spectral_flatten_aggressive(a0, N, n_iters=500):

def peak_reduction_search(a, n_rounds=100, step_scale=0.1):
    """
    Targeted peak reduction: identify autocorrelation peaks and reduce them
    by adjusting the contributing elements.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for rnd in range(n_rounds):
        c = fft_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top peaks
        sorted_idx = np.argsort(c)[::-1]
        n_peaks = min(5, len(c))
        
        # Subgradient of max: average of subgradients at active peaks
        sg = np.zeros(n)
        for k in sorted_idx[:n_peaks]:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # dc[k]/da[j] = 2*a[k-j] for j in [j_start, j_end]
                m_range = np.arange(j_start, j_end + 1)
                sg[m_range] += 2.0 * a[k - m_range]
        sg /= n_peaks
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step size
        step = step_scale * S / (gn + 1e-10)
        step = min(step, 0.5)
        
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        # Accept if improvement
        sc_new = compute_score(a_new)
        if sc_new < sc:
            a = a_new
        else:
            # Try smaller step
            a_new = a - 0.1 * step * grad
            a_new = np.clip(a_new, 0.0, 1000.0)
            sc_new2 = compute_score(a_new)
            if sc_new2 < sc:
                a = a_new
            else:
                # Give up on this round
                break
    
    return best_sc, best_a

"""
Spectral peak-flattening optimizer with adaptive LSE-LBFGS and targeted autocorrelation
peak reduction for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses aggressive spectral flattening with geometric-mean targets, careful LSE gradient
descent, and multi-stage peak-reduction local search.
"""

import numpy as np
from scipy.optimize import minimize
import time
from collections import defaultdict


def fft_conv(a):
    """Compute full convolution using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective: 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = fft_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed objective and gradient with respect to a."""
    n = len(a)
    c = fft_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: dL/da[j] = 2 * sum_m a[m] * p[j+m] = 2 * (a_rev * p)[j+n-1]
    a_rev = a[::-1].copy()
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:3 * n - 2]
    dL_da = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B optimization with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def spectral_flatten_agg(a0, N, n_iters=400, use_geom=True):
    """Aggressive spectral flattening with geometric or median target."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    S0 = a.sum()
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam = 0.85 * (1.0 - progress ** 2) + 0.1
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        half = N // 2
        if N % 2 == 0:
            non_dc = np.concatenate([mag[1:half], mag[half + 1:]])
        else:
            non_dc = mag[1:half + 1]
        if use_geom:
            target = float(np.exp(np.mean(np.log(non_dc + 1e-30))))
        else:
            target = float(np.median(non_dc))
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        if N % 2 == 0:
            new_mag[half] = mag[half]
            upper = range(1, half)
        else:
            upper = range(1, half + 1)
        for k in upper:
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (S0 / S)
    return a


def spectral_flatten_simple(a0, N, n_iters=300, lam=0.5):
    """Simple spectral flattening with median target."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    S0 = a.sum()
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        half = N // 2
        if N % 2 == 0:
            non_dc = np.concatenate([mag[1:half], mag[half + 1:]])
        else:
            non_dc = mag[1:half + 1]
        target = float(np.median(non_dc))
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        if N % 2 == 0:
            new_mag[half] = mag[half]
            upper = range(1, half)
        else:
            upper = range(1, half + 1)
        for k in upper:
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (S0 / S)
    return a


def peak_reduction(a, n_rounds=200, step_scale=0.1, n_top=3):
    """
    Targeted peak reduction using subgradient of top autocorrelation peaks.
    Returns (best_score, best_a).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fft_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Use top-K peaks for subgradient averaging
        sorted_idx = np.argsort(c)[::-1]
        k_peaks = sorted_idx[:n_top]

        sg = np.zeros(n)
        for k in k_peaks:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                m_range = np.arange(j_start, j_end + 1)
                sg[m_range] += 2.0 * a[k - m_range]
        sg /= float(n_top)

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step with backtracking
        step = step_scale * S / (gn + 1e-10)
        step = min(step, 0.5)
        accepted = False
        for shrink in [1.0, 0.5, 0.25, 0.1]:
            a_new = a - shrink * step * grad
            a_new = np.clip(a_new, 0.0, 1000.0)
            sc_new = compute_score(a_new)
            if sc_new < sc * (1.0 - 1e-8):
                a = a_new
                accepted = True
                break
        if not accepted:
            break

    return best_sc, best_a


def diverse_init(n, rng, N_fft):
    """Generate diverse initializations for sequence of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Quadratic chirp (CAZAC-inspired)
    inits.append(np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01)

    # Linear chirp
    inits.append(0.5 + 0.3 * np.cos(np.pi * 2.0 * x * x / (2.0 * n)))

    # Cosine modes
    for freq in [2, 3, 5]:
        inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * freq * x / n))

    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.clip(0.3 + 0.5 * rng.random(n), 0.01, 10.0))

    # Triangle / parabolic
    t = x / (n - 1.0)
    inits.append(0.5 + 0.4 * (1.0 - 4.0 * (t - 0.5) ** 2))

    # Golay-like: sum of two shifted chirps
    a1 = np.cos(np.pi * x * x / (2.0 * n)) ** 2
    shift = int(n * rng.uniform(0.1, 0.4))
    a2 = np.roll(a1, shift)
    inits.append(0.5 * (a1 + a2) + 0.01)

    # Spectral flatten some of these
    for bi in [0, 2, 4]:
        if bi < len(inits):
            sf = spectral_flatten_agg(inits[bi], N_fft, n_iters=200)
            inits.append(sf)

    # Simple spectral flattening
    sf_simple = spectral_flatten_simple(np.ones(n) * 0.5, N_fft, n_iters=150)
    inits.append(sf_simple)

    return inits
