"""
Optimized step function search using adaptive L-BFGS with LSE smoothing,
sensitivity-based flow diversification, and multi-scale exploration.
Focuses on improving upon the parent solution and exploring larger lengths.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: dL/da[j] = 2 * sum_k p[k] * a[k-j] (cross-correlation of p and a)
    L_fft = 1 << (2 * n - 1).bit_length()
    PF = np.fft.rfft(p, L_fft)
    AF = np.fft.rfft(a, L_fft)
    # cross[j] = sum_k p[k] * a[k-j]
    # Using FFT: ifft(FFT(p) * conj(FFT(a)))[:n] gives this
    # But with rfft, we need to be careful
    # Actually let's use the full FFT for the cross-correlation
    PF_full = np.fft.fft(p, L_fft)
    AF_full = np.fft.fft(a, L_fft)
    cross = np.fft.ifft(PF_full * np.conj(AF_full)).real[:n]
    grad_c = 2.0 * cross
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def spectral_flatten(a0, N, n_iters=200, lam=0.5):
    """Iterative spectral flattening to generate flat-autocorrelation-like sequences."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    target_sum = a.sum()
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (target_sum / S)
    return a


def u_shaped_init(n, alpha=1.0, rng=None):
    """Generate a U-shaped initialization (high at edges, low in middle)."""
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = 1.0 + alpha * (2.0 * x - 1.0) ** 2
    if rng is not None:
        a = a + 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, 1000.0)


def v_shaped_init(n, alpha=1.0, rng=None):
    """Generate a V-shaped initialization (low at edges, high in middle)."""
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = 1.0 + alpha * np.cos(np.pi * (2.0 * x - 1.0))
    if rng is not None:
        a = a + 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, 1000.0)


def chirp_init(n, rng):
    """CAZAC-inspired chirp initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = []
    inits.append(np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01)

    inits.append(0.5 + 0.3 * np.cos(np.pi * f1 * x * x / (2.0 * n)) + 0.2 * np.cos(2.0 * np.pi * f2 * x / n))
    for s in inits:
        inits[-1] = np.clip(s, 0.01, 1000.0)
    return inits


def sensitivity_flow(a, n_steps=100, strength=0.05, beta=50):
    """Sensitivity-based flow: move mass from high-sensitivity to low-sensitivity positions."""
    a = a.astype(np.float64).copy()
    n = len(a)
    target_sum = a.sum()
    for step in range(n_steps):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        exp_scaled = np.exp(scaled)
        p = exp_scaled / exp_scaled.sum()
        # Sensitivity: how much does adding mass to position j increase the soft objective?
        L_fft = 1 << (2 * n - 1).bit_length()
        PF = np.fft.fft(p, L_fft)
        AF = np.fft.fft(a, L_fft)
        sens = np.fft.ifft(PF * np.conj(AF)).real[:n]
        # Normalize sensitivity
        sens = (sens - sens.mean()) / (sens.std() + 1e-15)
        # Move mass from high-sens to low-sens
        delta = strength * sens
        a = a + delta
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (target_sum / S)
    return a


def targeted_reduction(a, strength=0.03):
    """Reduce the peak by decreasing the most contributing elements."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_count = top_count
    bottom_idx = np.argsort(contrib)[:bottom_count]
    a_new = a.copy()
    total_moved = 0.0
    for j in top_idx:
        amt = a_new[j] * strength
        a_new[j] -= amt
        a_new[j] = max(a_new[j], 0.0)
        total_moved += amt
    if total_moved > 0 and len(bottom_idx) > 0:
        per = total_moved / len(bottom_idx)
        for j in bottom_idx:
            a_new[j] += per
    return np.clip(a_new, 0.0, 1000.0)

"""
Multi-strategy optimizer for minimum peak convolution score.
Uses: (1) adaptive L-BFGS with LSE smoothing, (2) sensitivity-based flow diversification,
(3) targeted peak reduction, (4) multi-length exploration with U-shaped and spectral inits.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: dL/da[j] = 2 * sum_k p[k] * a[k-j] (cross-correlation)
    L_fft = 1 << (2 * n - 1).bit_length()
    PF = np.fft.fft(p, L_fft)
    AF = np.fft.fft(a, L_fft)
    cross = np.fft.ifft(PF * np.conj(AF)).real[:n]
    grad_c = 2.0 * cross
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def spectral_flatten(a0, N, n_iters=200, lam=0.6):
    a = a0.astype(np.float64).copy()
    n = len(a)
    target_sum = a.sum()
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (target_sum / S)
    return a


def u_shaped_init(n, alpha=1.0, rng=None):
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = 1.0 + alpha * (2.0 * x - 1.0) ** 2
    if rng is not None:
        a = a + 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, 1000.0)


def chirp_init(n, rng):
    x = np.arange(n, dtype=np.float64)
    inits = []
    inits.append(np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01)
    freq = 1.0 + 5.0 * rng.random()
    inits.append(0.5 + 0.4 * np.cos(np.pi * freq * x * x / (2.0 * n)))
    f1 = 1.0 + 6.0 * rng.random()
    f2 = 1.0 + 6.0 * rng.random()
    inits.append(0.5 + 0.3 * np.cos(np.pi * f1 * x * x / (2.0 * n)) + 0.2 * np.cos(2.0 * np.pi * f2 * x / n))
    for i in range(len(inits)):
        inits[i] = np.clip(inits[i], 0.01, 1000.0)
    return inits


def sensitivity_flow(a, n_steps=80, strength=0.04, beta=50):
    a = a.astype(np.float64).copy()
    n = len(a)
    target_sum = a.sum()
    L_fft = 1 << (2 * n - 1).bit_length()
    for step in range(n_steps):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        exp_scaled = np.exp(scaled)
        p = exp_scaled / exp_scaled.sum()
        PF = np.fft.fft(p, L_fft)
        AF = np.fft.fft(a, L_fft)
        sens = np.fft.ifft(PF * np.conj(AF)).real[:n]
        sens_norm = (sens - sens.mean()) / (sens.std() + 1e-15)
        a = a + strength * sens_norm
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (target_sum / S)
    return a


def targeted_reduction(a, strength=0.03):
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:top_count]
    a_new = a.copy()
    total_moved = 0.0
    for j in top_idx:
        amt = a_new[j] * strength
        a_new[j] -= amt
        a_new[j] = max(a_new[j], 0.0)
        total_moved += amt
    if total_moved > 0 and len(bottom_idx) > 0:
        per = total_moved / len(bottom_idx)
        for j in bottom_idx:
            a_new[j] += per
    return np.clip(a_new, 0.0, 1000.0)


def subgrad_polish(a, total_iters=500):
    a = a.astype(np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:

        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.2)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a

        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                print(f"Parent loaded: n={parent.size}, score={compute_score(parent):.8f}")
    except Exception:
        pass

    # ================================================================
    # Phase 1: Aggressively polish the parent (30% budget)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Parent polishing")
    print(f"{'='*60}")

    phase1_deadline = t_start + 0.30 * effective_budget

    if parent is not None:
        p_score = compute_score(parent)
        report(p_score, parent, "parent")
        a_cur = parent.copy()
        n_p = len(a_cur)

        # Progressive beta refinement
        for beta in [30, 80, 200, 500, 1000, 2000, 4000]:
            if time.monotonic() > phase1_deadline:
                break
            remaining = phase1_deadline - time.monotonic()
            if remaining < 5:
                break
            est_per_iter = 0.004 * (n_p / 1024)
            maxiter = min(8000, int(remaining * 0.7 / max(0.05, est_per_iter * 15)))
            maxiter = max(200, maxiter)
            sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxiter * 25)
            report(sc, a_opt, f"parent b={beta}")
            a_cur = a_opt

        # Subgradient polish
        if time.monotonic() < phase1_deadline - 5:
            sc, a_sg = subgrad_polish(a_cur, total_iters=600)
            report(sc, a_sg, "parent subgrad")
            if a_sg is not None:
                a_cur_copy = a_sg.copy()
                sc2, a_opt2, _ = run_lbfgs(a_cur_copy, 1000, maxiter=3000, maxfun=75000)
                report(sc2, a_opt2, "parent subgrad+lbfgs")

    # ================================================================
    # Phase 2: Explore other lengths (35% budget)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Length exploration")
    print(f"{'='*60}")

    phase2_deadline = t_start + 0.65 * effective_budget

    # Candidate lengths
    lengths = []
    if best_a is not None:
        lengths.append(len(best_a))
    # Try sizes around parent and larger
    if best_a is not None:
        pn = len(best_a)
        for delta in [-500, 0, 500, 1000, 2000]:
            candidate = pn + delta
            if candidate >= 2048 and candidate not in lengths:
                lengths.append(candidate)
    for n in [8192, 10000, 12288, 16384, 20000]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(l for l in lengths if l >= 2048))[:5]

    print(f"  Lengths to try: {lengths}")
    time_per_len = (phase2_deadline - t_start) / len(lengths) if lengths else 0

    for n in lengths:
        if time.monotonic() > phase2_deadline:
            break
        t_len_end = min(phase2_deadline, time.monotonic() + time_per_len)
        print(f"\n  >> n = {n} (budget {time_per_len:.0f}s)")

        # Skip if same as best and we already polished it
        if best_a is not None and len(best_a) == n and compute_score(best_a) < best_score + 1e-9:
            # Only do a quick check
            if time.monotonic() > t_len_end - 60:
                continue

        N_fft = 1 << (2 * n - 1).bit_length()
        inits = []

        # Diverse initializations
        inits.append(u_shaped_init(n, alpha=0.5, rng=rng))
        inits.append(u_shaped_init(n, alpha=1.0, rng=rng))
        inits.append(u_shaped_init(n, alpha=2.0, rng=rng))
        inits.append(np.ones(n) * 0.5)
        inits.append(0.5 + 0.4 * np.cos(np.pi * np.arange(n, dtype=np.float64) / n))
        inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
        inits.extend(chirp_init(n, rng))

        # Spectral flattening from best inits
        for bi in [0, 3, 5]:
            if time.monotonic() > t_len_end - 40:
                break
            sf_a = spectral_flatten(inits[bi].copy(), N_fft, n_iters=120, lam=0.6)
            inits.append(sf_a)

        # Score and rank
        scored = [(compute_score(a), i, a) for i, a in enumerate(inits)]
        scored.sort(key=lambda x: x[0])

        # Optimize top candidates
        for rank, (sc_init, idx, a_init) in enumerate(scored[:3]):
            if time.monotonic() > t_len_end:
                break
            report(sc_init, a_init, f"init n={n} r={rank}")

            a_cur = a_init.copy()
            for beta in [30, 100, 300, 1000, 2000]:
                if time.monotonic() > t_len_end:
                    break
                remaining = t_len_end - time.monotonic()
                if remaining < 5:
                    break
                est_per_iter = 0.004 * (n / 1024)
                maxiter = min(5000, int(remaining * 0.6 / max(0.05, est_per_iter * 12)))
                maxiter = max(150, maxiter)
                sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxiter * 25)
                a_cur = a_opt
                report(sc, a_opt, f"n={n} r={rank} b={beta}")

            report(compute_score(a_cur), a_cur, f"n={n} r={rank} end")

        print(f"    Best after n={n}: {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # ================================================================
    # Phase 3: Diversification and local search (25% budget)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Diversification & local search")
    print(f"{'='*60}")

    phase3_deadline = t_start + 0.90 * effective_budget
    n_best = len(best_a)

    n_trials = 80
    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline:
            break

        strategy = trial % 7
        if strategy == 0:
            scale = 0.01 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            a_pert = targeted_reduction(best_a, strength=0.02 + 0.03 * rng.random())
        elif strategy == 2:
            freq = 1.0 + 12.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            a_pert = sensitivity_flow(best_a.copy(), n_steps=40, strength=0.03, beta=50)
        elif strategy == 4:
            x = np.arange(n_best, dtype=np.float64)
            f = 1.0 + 8.0 * rng.random()
            pert = 0.02 * np.cos(np.pi * f * x * x / (2.0 * n_best))
            a_pert = best_a * (1.0 + pert)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 5:
            # Block perturbation
            n_blocks = int(4 + 16 * rng.random())
            block_size = max(1, n_best // n_blocks)
            block_offsets = (rng.random(n_blocks) - 0.5) * 0.08
            a_pert = best_a.copy()
            for b in range(n_blocks):
                start = b * block_size
                end = min(start + block_size, n_best)
                a_pert[start:end] *= (1.0 + block_offsets[b])
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Asymmetric shift: dampen one half, boost the other
            half = n_best // 2
            a_pert = best_a.copy()
            damp = 0.9 + 0.05 * rng.random()
            boost = 1.0 + 0.05 * rng.random()
            a_pert[:half] *= damp
            a_pert[half:] *= boost
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick score check before expensive optimization
        sc_pert = compute_score(a_pert)
        if sc_pert > best_score * 1.5:
            continue

        # L-BFGS polish
        remaining = phase3_deadline - time.monotonic()
        if remaining < 5:
            break
        est_per_iter = 0.004 * (n_best / 1024)
        maxiter = min(3000, int(remaining * 0.08 / max(0.05, est_per_iter * 10)))
        maxiter = max(100, maxiter)
        sc, a_opt, _ = run_lbfgs(a_pert, 500, maxiter=maxiter, maxfun=maxiter * 25)
        report(sc, a_opt, f"local #{trial} s={strategy}")

    # ================================================================
    # Phase 4: Final refinement (10% budget)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final refinement")
    print(f"{'='*60}")

    for beta in [1000, 3000, 6000, 10000]:
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        est_per_iter = 0.004 * (len(best_a) / 1024)
        maxiter = min(10000, int(remaining * 0.25 / max(0.05, est_per_iter * 10)))
        maxiter = max(500, maxiter)
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 20)
        report(sc, a_opt, f"final b={beta}")

    # Subgradient final touch
    if time.monotonic() < deadline - 10:
        sc, a_sg = subgrad_polish(best_a, total_iters=400)
        report(sc, a_sg, "final subgrad")

    # One more L-BFGS from subgrad result
    if time.monotonic() < deadline - 5:
        sc, a_opt, _ = run_lbfgs(a_sg, 5000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final subgrad+lbfgs")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
