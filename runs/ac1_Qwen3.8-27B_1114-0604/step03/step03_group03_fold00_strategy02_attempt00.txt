        for i in range(mid_point):
            if i < n_val // 2:
                a_full[i] = a_full[n_val - 1 - i] = a_sym[i]
            else:
                # Middle element explicitly set for odd length
                a_full[i] = a_sym[i]
        return a_full, 2.0 * n_val * np.max(fourier_convolve(a_full)) / (np.sum(a_full) ** 2)

    # Phase 1: Initial optimization with high-constraint regime
    a = init_arr.copy()
    original_n = n
    best_n = 0
    while best_n < 50 and time.time() - t_start < budget_s * 0.4:
        best_n *= 2
        # Extend sequence to give more optimization space
        a_extended = np.pad(a, best_n - len(a), constant_values=0).astype(np.float32)
        # Optimize with aggressive constraints
        score, grad = lse_convolve(a_extended, 10000)
        if score < best_score:
            best_score = score
            best_sequence = a_extended
            print(f"Initial high-beta smooth score: {best_score:.6f}, length: {len(a_extended)}")

    # Phase 2: Main iterations with time-bounded optimization
    t0 = time.time()
    counter = 0
    while time.time() - t0 < budget_s * 0.75 and counter < 100:
        try:
            # Calculate the next sequence elements to explore the curvature
            new_a = np.clip(a, 0, 10000)  # Increased valid range intentionally
            new_a[new_a < 0.01] = 0.01   # Protection against numerical instability

            # Use gradient information to direct the search
            score, grad = lse_convolve(new_a, 500)  # Lower beta than phase 1 for steady refinement
            
            # Decision to accept or reject based on gradient change amount
            if score < best_score:
                best_score = score
                best_sequence = new_a
                print(f"Stage iteration {counter}: new score {best_score:.6f}")
            
            # Adaptive beta adjustment based on progress
            if abs(score - next_score) < 0.00001:
                beta_adjust("increase")
            else:
                beta_adjust("decrease")
            
            counter += 1
        except np.inf or np.isnan:
            # Curve it out
            new_a = np.clip(new_a, 0, 1000.0)
            # Check if sum is too low
            if np.sum(new_a) < 0.01:
                beta_adjust("increase")  # Less smoothing
            else:
                # Otherwise, target the hot spots and rebuild
                hot_spot = np.argsort(np.abs(grad))[-int(len(new_a)*0.05):]
                new_a[hot_spot] = 100.0  # Give more weight to hot areas
                score, grad = lse_convolve(new_a, 500)
                if score < best_score:
                    best_score = score
                    best_sequence = new_a

    # Post-optimization validation and fallback procedure if needed
    try:
        # Re-validate with original evaluator
        final_score = evaluate_sequence(best_sequence.tolist())
        if final_score != np.inf and final_score < best_score:
            best_score = final_score
        print(f"Optimize_escalation completed in {time.time()-t0:.2f}s")
    except:
        print("Final validation failed!")
        # Return best_sequence with fallback if necessary
        # Use finite-length sequence checking
        # Check length of best_sequence must be original_n or a multiple
        best_sequence = [0.0] * original_n if best_sequence is None else best_sequence[:original_n].tolist()

    # Final polishing: even after budgets, ensure lowest possible score
    final_score = evaluate_sequence(best_sequence.tolist())
    if final_score < best_score:
        best_score = final_score

    # Output clean-up and fallback if time is insufficient
    if time.time() - t_start > budget_s * 0.9:
        # Cap the magnitude for safety during return
        a_final = [min(x, 30.0) for x in best_sequence[:original_n].tolist()]
    else:
        a_final = best_sequence.tolist()

    formatted = [round(x, 5) for x in a_final[:len(init_arr)]
    print(f"Optimization score: {best_score:.6f} (time={time.time()-t_start:.1f}/2400s)")
    return formatted
