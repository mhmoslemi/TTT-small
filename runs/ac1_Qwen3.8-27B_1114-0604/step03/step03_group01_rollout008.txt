"""
Multi-phase optimizer for minimizing 2n*max(self-conv)/(sum)^2.
Uses LSE-Adam with continuation, LP-based direction finding,
and multi-restart with diverse initializations.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    exp_sum = np.exp(scaled).sum()
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_sum
    # dL/d(a[j]) = 2 * sum_m p[j+m] * a[m]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    cross = np.fft.irfft(A_rev * P, M2_fft)[n-1:2*n-1]
    # cross[j] = sum_m a[m] * p[j+m] = dL/d(a[j]) / 2
    dL = 2.0 * cross
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL / S2 - 2.0 * L / S3)
    return score, grad

def subgrad_score(a):
    """Exact score and subgradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
    return score, grad

def adam_lse(a_init, beta, lr=0.005, time_limit=100.0, t_start=None, normalize=True):
    """Adam with LSE-smoothed gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    deadline = t_start + time_limit if t_start is not None else np.inf
    
    t = 0
    while t < 1000000:
        t += 1
        if time.monotonic() > deadline:
            break
        
        lr_t = lr / np.sqrt(1.0 + 1e-5 * t)
        
        score, grad = lse_score_grad(a, beta)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        
        step = lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = np.clip(a, 0.0, 1e10)
        
        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a / s
        
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_sc, best_a

def lp_direction(a, M, n_max_lp=5000):
    """
    Solve LP: max sum(b) s.t. b>=0, conv(a,b)[k] <= M for k=0..n-1.
    Returns the optimal b, normalized to sum=1.
    """
    n = len(a)
    if n > n_max_lp:
        return None
    
    # By symmetry, conv(a,b)[k] = conv(a,b)[2n-2-k]
    # So we only need constraints for k=0..n-1
    # conv(a,b)[k] = sum_j a[j]*b[k-j] (valid indices only)
    
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog
    
    # Build constraint matrix A_ub (n x n)
    # Row k: A_ub[k, m] = a[k-m] if 0 <= k-m < n, else 0
    rows = []
    cols = []
    data = []
    for k in range(n):
        for m in range(max(0, k - n + 1), min(k, n - 1) + 1):
            rows.append(k)
            cols.append(m)
            data.append(a[k - m])
    
    A = csr_matrix((data, (rows, cols)), shape=(n, n))
    b_ub = np.full(n, M)
    
    c = -np.ones(n)  # minimize -sum(b)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs')
    
    if res.success:
        b = res.x
        b_sum = b.sum()
        if b_sum > 1e-12:
            return b / b_sum
        else:
            return np.ones(n) / n
    return None

def lp_refine(a, n_iter=50, time_limit=300.0, t_start=None):
    """Iteratively use LP direction to improve the sequence."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    deadline = t_start + time_limit if t_start is not None else np.inf
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = float(c.max())
        
        b_dir = lp_direction(a, M, n_max_lp=max(5000, n))
        if b_dir is None:
            break
        
        # Line search: a_new = (1-t)*a + t*b_dir for t in (0,1]
        best_t = 0.0
        best_line_sc = best_sc
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * b_dir
            a_try = np.clip(a_try, 0.0, None)
            s = a_try.sum()
            if s < 1e-12:
                continue
            sc = compute_score(a_try / s)
            if sc < best_line_sc:
                best_line_sc = sc
                best_t = t
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.05, 0.1, 0.15]:
                a_try = (1 - t) * a + t * b_dir
                a_try = np.clip(a_try, 0.0, None)
                s = a_try.sum()
                if s < 1e-12:
                    continue
                sc = compute_score(a_try / s)
                if sc < best_line_sc:
                    best_line_sc = sc
                    best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * b_dir
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s > 1e-12:
                a = a / s
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"    LP iter {it}: score={sc:.8f}, t={best_t:.3f}")
    
    return best_sc, best_a

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)

def make_chirp(n, alpha=0.5):
    """Create a chirp-based initialization."""
    x = np.arange(n, dtype=np.float64) / n
    a = 1.0 + alpha * np.cos(2 * np.pi * alpha * x * x * n)
    a = np.clip(a, 0.1, None)
    a = a / a.sum()
    return a

def make_raised_cosine(n):
    """Raised cosine window."""
    x = np.arange(n, dtype=np.float64) / n
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.clip(a, 0.01, None)
    a = a / a.sum()
    return a

def make_eval_polynomial(n, deg=3):
    """Based on Chebyshev-like distribution."""
    # Use arcsine distribution: density proportional to 1/sqrt(x*(1-x))
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = 1.0 / np.sqrt(x * (1 - x) + 1e-10)
    a = a / a.sum()
    return a

def make_flat_spectrum(n, rng):
    """Try to create a sequence with flat power spectrum."""
    # Random phase in frequency domain
    freqs = np.fft.rfftfreq(n)
    k_max = n // 2
    phases = rng.uniform(0, 2 * np.pi, k_max + 1)
    A = np.exp(1j * phases)
    a = np.fft.irfft(A * np.sqrt(n), n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        return np.ones(n) / n
    return a / s
