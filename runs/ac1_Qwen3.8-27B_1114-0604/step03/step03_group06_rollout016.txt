def peak_shave_step(a, step_size=0.01):
    """
    One step of peak shaving: find the peak of the convolution,
    reduce elements that contribute to it, redistribute to others.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    k_star = int(np.argmax(c))
    
    # Elements contributing to c[k_star]:
    # c[k_star] = sum_j a[j] * a[k_star - j] for valid j
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    
    if j_end < j_start:
        return a.copy()
    
    # Contribution of a[j] to c[k_star] is a[k_star - j]
    contrib = np.zeros(n)
    for j in range(j_start, j_end + 1):
        contrib[j] = a[k_star - j]
    
    # Reduce high-contributing elements, add to low-contributing ones
    total_mass = 0.0
    # Scale: how much to reduce
    peak_val = c[k_star]
    second_peak = np.sort(c)[::-1][1] if len(c) > 1 else 0
    gap = peak_val - second_peak
    
    if gap < 1e-10:
        return a.copy()
    
    # Reduce proportional to contribution
    max_contrib = contrib.max()
    if max_contrib < 1e-15:
        return a.copy()
    
    # How much total to move
    move_fraction = min(step_size, 0.05)
    total_to_move = move_fraction * peak_val / (2.0 * max_contrib + 1e-15)
    total_to_move = min(total_to_move, 0.1 * S)
    
    a_new = a.copy()
    moved = 0.0
    for j in range(j_start, j_end + 1):
        reduce_amt = total_to_move * contrib[j] / (contrib[j_start:j_end+1].sum() + 1e-15)
        reduce_amt = min(reduce_amt, a_new[j] * 0.5)  # Don't reduce below 50%
        a_new[j] -= reduce_amt
        moved += reduce_amt
    
    # Redistribute to low-contributing elements
    low_mask = contrib < 0.3 * max_contrib
    low_indices = np.where(low_mask)[0]
    if len(low_indices) > 0 and moved > 0:
        per = moved / len(low_indices)
        for idx in low_indices:
            a_new[idx] += per
    
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new

def projected_gradient_step(a, grad, step_size):
    """Take a gradient step and project onto {a >= 0}."""
    a_new = a - step_size * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new

def coordinate_descent(a, n_iters=100, rng=None):
    """
    Coordinate descent: for each coordinate, find the best value by 1D search.
    Uses the fact that for fixed other coordinates, the objective as a function of a[j]
    is a unimodal function (ratio of a convex piecewise-linear to a quadratic).
    """
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a)
    a = a.astype(np.float64).copy()
    best_sc = compute_score(a)
    
    # Precompute: c[k] = sum_i a[i]*a[k-i]
    # d(c[k])/da[j] = a[k-j] + a[j-k] (for valid indices)
    # So if we change a[j] to a[j]+delta:
    #   c[k] changes by delta * (a[k-j] + a[j-k])
    #   (a[j])^2 changes by 2*a[j]*delta + delta^2
    # The peak is the max of these linear (or quadratic for k=2j) functions.
    
    # For efficiency, use random coordinate selection
    order = rng.permutation(n)
    
    for it in range(n_iters):
        for j in order[max(0, n-50):]:  # Only optimize last 50 per pass for speed
            # Actually, let's do a few random coordinates per iteration
            pass
    
    # Simpler: just do a few random coordinate updates
    for it in range(n_iters):
        j = rng.integers(0, n)
        # Try several values for a[j]
        best_val = a[j]
        best_local_sc = best_sc
        current = a[j]
        for delta_frac in [-0.5, -0.3, -0.1, 0.1, 0.3, 0.5, 1.0]:
            new_val = max(0.0, current * (1.0 + delta_frac))
            if new_val == current:
                continue
            a[j] = new_val
            sc = compute_score(a)
            a[j] = current
            if sc < best_local_sc:
                best_local_sc = sc
                best_val = new_val
        if best_val != current:
            a[j] = best_val
            best_sc = best_local_sc
    
    return best_sc, a

def exact_subgrad_polish(a, n_iters=200, rng=None):
    """
    Projected subgradient descent on the exact max-conv objective.
    Normalizes sum to 1 internally for the step computation.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of the max term
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end+1] = 2.0 * a[j_start:j_end+1][::-1]
        
        # Gradient of full score
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        # Step size: Polyak-like
        sc_lb = 1.0  # theoretical lower bound
        step = (sc - sc_lb) / (np.dot(grad, grad) + 1e-30)
        step = min(step, 0.5)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        
        # Occasionally add small random perturbation to escape
        if it % 50 == 49:
            a += 0.001 * rng.standard_normal(n)
            a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a

def multi_avg_subgrad(a, n_iters=500):
    """
    Multi-average subgradient method for better convergence.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Store iterates for averaging
    iters = [a.copy()]
    weights = [1.0]
    
    for it in range(1, n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end+1] = 2.0 * a[j_start:j_end+1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        # Step size: 1/sqrt(t)
        step = 0.1 / np.sqrt(it + 1)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        
        if it % 10 == 0:
            iters.append(a.copy())
            weights.append(1.0 / (it + 1))
    
    # Multi-average
    total_w = sum(weights)
    avg_a = sum(w * a_i for w, a_i in zip(weights, iters)) / total_w

"""
Optimized solver for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses LSE-smoothed L-BFGS-B with high-beta continuation, accelerated subgradient,
peak-shaving with line search, and targeted local search. Focused refinement from parent.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: dL/da[j] = 2 * (rev(a) * p)[n-1+j]
    rev_a = a[::-1]
    conv_len = len(rev_a) + len(p) - 1
    conv_fft = 1 << (conv_len - 1).bit_length()
    RA = np.fft.rfft(rev_a, conv_fft)
    P = np.fft.rfft(p, conv_fft)
    conv_res = np.fft.irfft(RA * P, conv_fft)[:conv_len]
    dL_da = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    """Run L-BFGS-B with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-11,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out


def accelerated_subgradient(a, n_iters=500):
    """
    Nesterov-accelerated projected subgradient on exact max-conv objective.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a_aux = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(1, n_iters + 1):
        # Evaluate at auxiliary point
        c = fast_conv(a_aux)
        S = float(a_aux.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a_aux.copy()
        
        # Subgradient at a_aux
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a_aux[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        # Nesterov momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / 2.0
        t_km1 = (1.0 + np.sqrt(1.0 + 4.0 * (it - 2) ** 2)) / 2.0 if it > 1 else 1.0
        momentum = (t_km1 - 1.0) / t_k
        
        # Step size
        step = 0.5 / np.sqrt(it + 1)
        a_new = a_aux - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        # Update auxiliary
        a_aux = a_new + momentum * (a_new - a)
        a_aux = np.clip(a_aux, 0.0, 1000.0)
        a = a_new
    
    return best_sc, best_a


def peak_shave(a, rng, n_trials=20):
    """
    Peak shaving with line search: identify peak lag, reduce contributing elements,
    redistribute to low-contribution elements, with multiple trial step sizes.
    """
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    
    if j_end < j_start:
        return compute_score(a), a.copy()
    
    # Contribution of each a[j] to c[k_star]
    contrib = np.zeros(n)
    contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    contrib_sum = contrib.sum()
    if contrib_sum < 1e-15:
        return compute_score(a), a.copy()
    
    # Find low-contribution elements for redistribution
    threshold = 0.2 * contrib.max()
    low_mask = contrib < threshold
    low_indices = np.where(low_mask)[0]
    
    best_sc = compute_score(a)
    best_a = a.copy()
    
    # Try various step sizes and redistribution strategies
    for trial in range(n_trials):
        a_try = a.copy()
        
        if trial < 8:
            # Strategy 1: Reduce high-contrib elements proportional to contribution
            frac = 0.01 * (trial + 1)
            total_moved = 0.0
            for j in range(j_start, j_end + 1):
                reduce_amt = a_try[j] * frac * (contrib[j] / contrib.max() if contrib.max() > 0 else 0)
                a_try[j] -= reduce_amt
                a_try[j] = max(a_try[j], 0.0)
                total_moved += reduce_amt
            if len(low_indices) > 0 and total_moved > 0:
                per = total_moved / len(low_indices)
                for idx in low_indices:
                    a_try[idx] += per
        elif trial < 14:
            # Strategy 2: Reduce top few elements, add to bottom few
            k = max(5, n // 100)
            top_idx = np.argsort(contrib)[::-1][:k]
            bot_idx = np.argsort(contrib)[:k]
            total_moved = 0.0
            for j in top_idx:
                reduce_amt = a_try[j] * 0.05
                a_try[j] -= reduce_amt
                a_try[j] = max(a_try[j], 0.0)
                total_moved += reduce_amt

                per = total_moved / len(bot_idx)
                for j in bot_idx:
                    a_try[j] += per
        else:

            a_try[low_indices] += scale * 0.5

            continue
        sc = compute_score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_a = a_try.copy()

def block_refine(a, n_blocks=16, max_wall_s=30.0, rng=None):

    Block coordinate refinement: perturb each block and keep improvement.
    """

        rng = np.random.default_rng(0)

    best_sc = compute_score(a)

            if time.monotonic() - t0 > max_wall_s:
                break
            s = b * block_sz

                scale = 0.02 + 0.04 * rng.random()
                a_try = a.copy()
                a_try[s:e] = a_try[s:e] * (1.0 + scale * rng.standard_normal(e - s))
                a_try = np.clip(a_try, 0.0, 1000.0)
                if a_try.sum() < 0.01:
                    continue
                sc = compute_score(a_try)
                if sc < best_sc:
                    best_sc = sc
                    a = a_try
    
    return best_sc, a


def diverse_perturb(a, rng, strategy):
    """Generate a perturbed version using the given strategy."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    
    if strategy == 0:
        scale = 0.01 + 0.04 * rng.random()
        return np.clip(a * (1.0 + scale * rng.standard_normal(n)), 0.0, 1000.0)
    elif strategy == 1:
        # Frequency-modulated
        freq = 1.0 + 8.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = 0.01 + 0.03 * rng.random()
        return np.clip(a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase)), 0.0, 1000.0)
    elif strategy == 2:
        # Sparse large perturbation
        n_sp = max(10, n // 40)
        a_p = a.copy()
        idx = rng.integers(0, n, size=n_sp)
        a_p[idx] *= (1.0 + 0.1 * rng.standard_normal(n_sp))
        return np.clip(a_p, 0.0, 1000.0)
    elif strategy == 3:
        # Block swap
        bs = max(10, n // 16)
        b1 = int(rng.integers(0, n - 2 * bs))
        b2 = int(rng.integers(0, n - 2 * bs))
        a_p = a.copy()
        a_p[b1:b1+bs], a_p[b2:b2+bs] = a_p[b2:b2+bs].copy(), a_p[b1:b1+bs].copy()
        return a_p
    elif strategy == 4:
        # Mix with structured
        struct = 0.5 + 0.3 * np.cos(2.0 * np.pi * rng.random() * 4 * x)
        t = 0.1 + 0.2 * rng.random()
        a_mix = (1 - t) * a + t * struct * (a.sum() / struct.sum())
        return np.clip(a_mix, 0.0, 1000.0)
    else:
        # Targeted: reduce peak contributors, boost others
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        a_p = a.copy()
        if je >= js:
            contrib = np.zeros(n)
            contrib[js:je+1] = a[js:je+1][::-1]
            top = np.argsort(contrib)[::-1][:max(5, n//60)]
            bot = np.argsort(contrib)[:max(5, n//60)]
            a_p[top] *= 0.95
            a_p[bot] *= 1.05
        return np.clip(a_p, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for optimal non-negative sequence minimizing 2n*max(conv)/(sum)^2.
    Uses LSE L-BFGS, accelerated subgradient, peak shaving, block refinement,
    and diverse restarts with targeted local search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 100:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = np.ones(256) * 0.5
    
    if parent is not None:
        best_a = parent.copy()
        best_score = compute_score(parent)
        print(f"Started from parent: score={best_score:.8f}, n={len(parent)}")
    
    # Determine lengths
    lengths = []
    if parent is not None:
        pa = parent.size
        lengths.append(pa)
        for dl in [-256, 256, -512, 512, -1024, 1024, 2048]:
            cand = pa + dl
            if 2048 <= cand <= 20000:
                lengths.append(cand)
    for n in [4096, 6962, 8192, 10240, 12288]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths: {lengths}")
    
    # =========================================================================
    # Phase 1: High-beta LSE L-BFGS (55% of time)
    # =========================================================================
    phase1_end = t_start + 0.55 * (budget_s - 20)
    print(f"\n=== Phase 1: L-BFGS LSE (until t={phase1_end-t_start:.0f}s) ===")
    
    # Start with high beta since parent is already near-optimal
    if parent is not None:
        beta_schedule = [500, 1000, 2000, 4000, 8000, 15000]
    else:
        beta_schedule = [50, 100, 300, 600, 1200, 2500, 5000]
    
    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"\n  n = {n}")
        
        # Build inits
        inits = []
        if parent is not None:
            if len(parent) == n:
                inits.append(parent.copy())
            else:
                idx = np.linspace(0, len(parent) - 1, n).astype(int)
                inits.append(parent[idx].copy())
        x = np.arange(n, dtype=np.float64) / n
        inits.append(np.clip(0.5 + 0.35 * np.sinc(3.5 * (x - 0.5)), 0.01, 1000.0))
        inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
        
        remaining = phase1_end - time.monotonic()
        max_inits = max(1, min(len(inits), int(remaining / 80)))
        inits = inits[:max_inits]
        
        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            a_cur = a0.copy()
            for bi, beta in enumerate(beta_schedule):
                if time.monotonic() > phase1_end:
                    break
                mi = min(5000, 800 * (bi + 1))
                sc, a_opt = run_lbfgs(a_cur, beta, maxiter=mi, maxfun=mi * 2)
                a_cur = a_opt
                if sc < best_score:
                    best_score = sc
                    best_a = a_opt.copy()
                    print(f"    [{n},i={i},b={beta}] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
    
    print(f"\nAfter Phase 1: best = {best_score:.8f}")
    
    # =========================================================================
    # Phase 2: Accelerated subgradient + peak shaving (20% of time)
    # =========================================================================
    phase2_end = t_start + 0.75 * (budget_s - 20)
    print(f"\n=== Phase 2: Subgrad + Peak Shave (until t={phase2_end-t_start:.0f}s) ===")
    
    a_work = best_a.copy()
    
    while time.monotonic() < phase2_end:
        # Accelerated subgradient
        sc, a_sub = accelerated_subgradient(a_work, n_iters=300)
        if sc < best_score:
            best_score = sc
            best_a = a_sub.copy()
            print(f"  [subgrad] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
            a_work = a_sub
        
        # Peak shaving
        if time.monotonic() < phase2_end:
            sc, a_sh = peak_shave(a_work, rng)
            if sc < best_score:
                best_score = sc
                best_a = a_sh.copy()
                print(f"  [peak_shave] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
                a_work = a_sh
            else:
                a_work = a_sub
    
    print(f"\nAfter Phase 2: best = {best_score:.8f}")
    
    # =========================================================================
    # Phase 3: Block refinement + diverse restarts (15% of time)
    # =========================================================================
    phase3_end = t_start + 0.90 * (budget_s - 20)
    print(f"\n=== Phase 3: Block + Restarts (until t={phase3_end-t_start:.0f}s) ===")
    
    # Block refinement on best
    sc, a_blk = block_refine(best_a, n_blocks=32, max_wall_s=60.0, rng=rng)
    if sc < best_score:
        best_score = sc
        best_a = a_blk.copy()
        print(f"  [block] NEW BEST: {sc:.8f}")
    
    # Diverse restarts
    n_best = len(best_a)
    trial = 0
    while time.monotonic() < phase3_end and trial < 25:
        strat = trial % 6
        a_pert = diverse_perturb(best_a, rng, strat)
        sc, a_opt = run_lbfgs(a_pert, 2000, maxiter=2000, maxfun=4000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [restart {trial},s={strat}] NEW BEST: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        # If improvement, also try block refine
        if sc < best_score + 0.001 and time.monotonic() < phase3_end:
            sc2, a_b2 = block_refine(a_opt, n_blocks=16, max_wall_s=15.0, rng=rng)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_b2.copy()
                print(f"  [restart+block {trial}] NEW BEST: {sc2:.8f}")
        trial += 1
    
    print(f"\nAfter Phase 3: best = {best_score:.8f}")
    
    # =========================================================================
    # Phase 4: Ultra-high-beta final polish (10% of time)
    # =========================================================================
    print(f"\n=== Phase 4: Final Polish ===")
    
    for beta in [20000, 50000]:
        if time.monotonic() > deadline - 10:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [final b={beta}] NEW BEST: {sc:.8f}")
    
    # Final subgradient
    if time.monotonic() < deadline - 5:
        sc, a_sub = accelerated_subgradient(best_a, n_iters=300)
        if sc < best_score:
            best_score = sc
            best_a = a_sub.copy()
            print(f"  [final subgrad] NEW BEST: {sc:.8f}")
    
    # One last L-BFGS
    if time.monotonic() < deadline - 3:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=3000, maxfun=60000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [FINAL lbfgs] NEW BEST: {sc:.8f}")
    
    # =========================================================================
    # Output
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
