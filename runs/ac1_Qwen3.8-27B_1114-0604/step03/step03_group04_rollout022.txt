def proj_simplex(v, S):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng = np.arange(1, n + 1)
    cond = u * rng > (cssv - S)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.full(n, S / n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgrad_polish(a, total_iters=500):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    S = float(a.sum())
    for it in range(total_iters):
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient of max at k_star
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            idx = k_star - np.arange(j_start, j_end + 1)
            sg[j_start:j_end + 1] = 2.0 * a[idx]
        # Also subtract the "uniform" part to get the gradient of the normalized objective
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        # Step size: use a diminishing step
        step = 1.0 / (gn * (1.0 + 0.1 * it))
        step = min(step, 0.5 / gn)
        a_new = a - step * grad
        a_new = proj_simplex(a_new, S)
        # Check improvement
        sc_new = compute_score(a_new)
        if sc_new < sc or it < 10:
            a = a_new

def freq_shape(a, N, n_iters=100, target_flatness=0.3):
    """Shape the frequency domain to reduce peaks."""
    a_work = a.astype(np.float64).copy()

        # Reduce them toward the median

        new_mag[mask] = threshold + (mag[mask] - threshold) * 0.5
        new_A = new_mag * np.exp(1j * phase)

"""
Multi-phase optimizer for minimum peak autocorrelation of non-negative sequences.
Combines: (1) spectral flattening initialization, (2) L-BFGS with top-K LSE
continuation (decreasing K for sharper focus), (3) projected subgradient steps,
(4) frequency-domain annealing, and (5) adaptive multi-length exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Fast full auto-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_full_func(a, beta):
    """Full LSE smoothed objective with gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def lse_topk_func(a, beta, K):
    """Top-K LSE smoothed objective with gradient (sharper focus on peaks)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    K_eff = min(K, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:K_eff]
    c_top = c[top_idx]
    c_max = c_top[0]
    scaled = beta * (c_top - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    p_full = np.zeros(2 * n - 1)
    p_full[top_idx] = p
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p_full, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def proj_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng_arr = np.arange(1, n + 1)
    cond = u * rng_arr > (cssv - S)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.full(n, S / n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgrad_lean(a, total_iters, target_score):
    """Projected subgradient with adaptive step size, targeting a score."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    S = float(a.sum())
    n_pegged = 0
    for it in range(total_iters):
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if best_sc <= target_score:
            break
        # Subgradient
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            idx = k_star - np.arange(j_start, j_end + 1)
            sg[j_start:j_end + 1] = 2.0 * a[idx]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        # Adaptive step: larger when far from target
        gap = max(sc - target_score, 1e-6)
        step = gap / (gn * gn)
        step = min(step, 1.0 / (gn + 1e-10))
        # Diminishing
        step = step * (1.0 / (1.0 + 0.005 * it))
        a_new = a - step * grad
        a_new = proj_simplex(a_new, S)
        sc_new = compute_score(a_new)
        if sc_new <= sc * 1.001:
            a = a_new
        else:
            n_pegged += 1
            if n_pegged > 200:
                break
    return best_sc, best_a


def spectral_flatten(a0, N, n_iters, lam):
    """Iterative spectral flattening (POCS: flat spectrum + non-negativity)."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc = mag[1:]
        valid = non_dc[non_dc > 1e-12]
        if len(valid) == 0:
            target = 1e-10
        else:
            target = np.median(valid)
        new_mag = mag.copy()
        half = N // 2
        for k in range(1, N // 2 + 1):
            if N % 2 == 0 and k == N // 2:
                new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            else:
                new_mag[k] = (1.0 - lam) * mag[k] + lam * target
                new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a


def spectral_anneal(a0, N, n_iters, lam_start, lam_end):
    """Spectral flattening with annealed lambda (strong early, weak late)."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam = lam_start + (lam_end - lam_start) * progress
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc = mag[1:]
        valid = non_dc[non_dc > 1e-12]
        target = np.median(valid) if len(valid) > 0 else 1e-10
        new_mag = mag.copy()
        for k in range(1, N // 2 + 1):
            if N % 2 == 0 and k == N // 2:
                new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            else:
                new_mag[k] = (1.0 - lam) * mag[k] + lam * target
                new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a

def peak_shave(a, n_shaves, S):
    """Targeted peak shaving: reduce contributions to the top few lags."""
    n = len(a)
    a = a.copy()
    for _ in range(n_shaves):
        c = fast_conv(a)
        top_k = min(5, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        # For each top lag, find the element with the largest individual contribution
        # and shave it slightly
        for k in top_idx:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            idx = k - np.arange(j_start, j_end + 1)
            contribs = a[j_start:j_end + 1] * a[idx]
            max_local = np.argmax(contribs)
            j = j_start + max_local
            # Shave: reduce a[j] and redistribute
            reduce_amt = a[j] * 0.01
            a[j] -= reduce_amt
            a[j] = max(a[j], 0.0)
            # Redistribute to a random low-contributing element
            other_idx = np.random.randint(n)
            a[other_idx] += reduce_amt
        # Re-normalize
        S_cur = a.sum()
        if S_cur > 1e-12:
            a = a * (S / S_cur)
    a = np.clip(a, 0.0, 1000.0)
    return a

            sf = spectral_flatten(a_struct, N_fft, 100, 0.4)
            inits_by_len[n].append(sf)

    # Score inits
    print("  Scoring initializations...")
    for n in lengths:
        if n not in inits_by_len:
            continue
        for a in inits_by_len[n]:
            sc = compute_score(a)
            try_report(sc, a, f"init n={n}")
        if inits_by_len[n]:
            best_init_sc = min(compute_score(a) for a in inits_by_len[n])
            print(f"  n={n}: {len(inits_by_len[n])} inits, best={best_init_sc:.6f}")

    # =========================================================
    # Phase 2: L-BFGS with Top-K LSE Continuation (50% of budget)
    # =========================================================
    # Key innovation: decrease K over the continuation schedule to focus gradient
    p2_deadline = t_start + 0.65 * effective_budget
    print(f"\n--- Phase 2: L-BFGS Continuation (deadline +{p2_deadline - t_start:.0f}s) ---")

    # Continuation schedule: (beta, K) pairs
    # Start smooth and broad, end sharp and focused
    schedule = [
        (15, 200),
        (30, 100),
        (60, 50),
        (100, 25),
        (200, 15),
        (400, 10),
        (800, 5),
    ]

    # Sort lengths by parent priority
    sorted_lengths = sorted(lengths, key=lambda n: (0 if (parent is not None and parent.size == n) else 1, n))

    for n in sorted_lengths:
        if time.monotonic() > p2_deadline:
            break
        print(f"\n  >> Optimizing n = {n}")

        # Select best inits for this length
        cands = inits_by_len.get(n, [])
        if not cands:
            N_fft = 1 << (2 * n - 1).bit_length()
            cands = [spectral_flatten(np.ones(n) * 0.5, N_fft, 100, 0.5)]
            cands.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 10.0))

        # Score and sort
        scored = [(compute_score(c), c) for c in cands]
        scored.sort(key=lambda x: x[0])
        top_cands = scored[:min(3, len(scored))]

        for ci, (sc_init, a_init) in enumerate(top_cands):
            if time.monotonic() > p2_deadline:
                break
            t0 = time.monotonic()
            a_cur = a_init.copy()
            scale = max(1, n // 2048)

            for si, (beta, K) in enumerate(schedule):
                if time.monotonic() > p2_deadline:
                    break
                remaining = p2_deadline - time.monotonic()
                if remaining < 8:
                    break
                # Estimate time per function eval
                est_per_eval = 0.004 * scale
                maxiter = min(5000, max(100, int(remaining * 0.25 / est_per_eval)))
                maxfun = max(500, maxiter * 20)
                use_topk = (K < 2 * n - 1)
                sc, a_opt = run_lbfgs(a_cur, beta, K, maxiter, maxfun, use_topk)
                a_cur = a_opt
                try_report(sc, a_opt, f"L n={n} c={ci} s={si} b={beta} K={K}")

            sc_final = compute_score(a_cur)
            try_report(sc_final, a_cur, f"lbfgs_end n={n} c={ci}")
            elapsed = time.monotonic() - t0
            print(f"    n={n} c={ci}: score={sc_final:.6f} ({elapsed:.1f}s)")

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # =========================================================
    # Phase 3: Subgradient + Peak Shaving (20% of budget)
    # =========================================================
    p3_deadline = t_start + 0.85 * effective_budget
    print(f"\n--- Phase 3: Subgradient & Peak Shaving ---")

    n_best = len(best_a)
    S_best = float(best_a.sum())

    for round_i in range(4):
        if time.monotonic() > p3_deadline:
            break

        # Subgradient polish
        remaining = p3_deadline - time.monotonic()
        iters = min(800, max(200, int(remaining * 0.2 / 0.005 * (2048 / max(1, n_best)))))
        target = best_score * 0.999
        sc, a_pol = subgrad_lean(best_a, iters, target)
        try_report(sc, a_pol, f"subgrad r{round_i}")

        # Peak shaving
        if time.monotonic() < p3_deadline - 5:
            a_shaved = peak_shave(a_pol, n_shaves=20, S=S_best)
            sc_sh = compute_score(a_shaved)
            try_report(sc_sh, a_shaved, f"shave r{round_i}")

        # Quick L-BFGS from best
        if time.monotonic() < p3_deadline - 10:
            remaining2 = p3_deadline - time.monotonic()
            maxiter = min(3000, max(200, int(remaining2 * 0.5 / (0.004 * max(1, n_best // 2048)))))
            sc2, a2 = run_lbfgs(best_a, 200, 10, maxiter, maxiter * 15, True)
            try_report(sc2, a2, f"lbfgs_s3 r{round_i}")

    # =========================================================
    # Phase 4: Final High-Precision Polish (10% of budget)
    # =========================================================
    print(f"\n--- Phase 4: Final Polish ---")

    for beta, K in [(1000, 5), (2000, 3), (5000, 2)]:
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        maxiter = min(5000, max(500, int(remaining * 0.7 / (0.004 * max(1, n_best // 2048)))))
        sc, a_opt = run_lbfgs(best_a, beta, K, maxiter, maxiter * 15, True)
        try_report(sc, a_opt, f"final b={beta} K={K}")

    # Final subgradient
    if time.monotonic() < deadline - 5:
        sc, a_pol = subgrad_lean(best_a, 300, best_score * 0.9995)
        try_report(sc, a_pol, "final_subgrad")

    # One more L-BFGS from absolute best
    if time.monotonic() < deadline - 5:
        sc, a_opt = run_lbfgs(best_a, 5000, 2, 2000, 30000, True)
        try_report(sc, a_opt, "last_ditch")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
