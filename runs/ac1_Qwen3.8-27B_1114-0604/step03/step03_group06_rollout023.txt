```python
"""
Improved optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements over previous:
1. Finer LSE continuation schedule with more aggressive final polish
2. Log-parameterization (softmax) to remove bound constraints
3. Nesterov-momentum subgradient for exact objective
4. Peak shaving heuristic directly targeting convolution peaks
5. Multiple restart strategies and adaptive step sizes
6. Top-k smooth objective as alternative to LSE
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient. Returns (score, grad)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj_with_args(x, beta):
    """Objective wrapper for scipy minimize with args."""
    return lse_score_grad(x, beta)


def topk_score_grad(a, k):
    """Top-k smooth convolution score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_sorted = np.sort(c)[::-1]
    kk = min(k, len(c_sorted) - 1)
    threshold = c_sorted[kk]
    next_val = c_sorted[min(kk + 1, len(c_sorted) - 1)]
    gap = max(threshold - next_val, 1e-8)
    width = gap * 0.5 + 1e-8
    w = 1.0 / (1.0 + np.exp(-10.0 * (c - threshold) / width))
    w = w / (w.sum() + 1e-30)
    L = float(np.dot(w, c))
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def topk_obj_with_args(x, k):
    """Top-k objective wrapper."""
    return topk_score_grad(x, k)


def softmax(theta):
    """Numerically stable softmax."""
    t = theta - theta.max()
    e = np.exp(t)
    return e / (e.sum() + 1e-30)


def logparam_lse_obj_grad(theta, beta):
    """LSE score in log-parameterization. theta -> a = softmax(theta)."""
    a = softmax(theta)
    score, grad_a = lse_score_grad(a, beta)
    dot_ga = float(np.dot(grad_a, a))
    grad_theta = a * (grad_a - dot_ga)
    return score, grad_theta


def nesterov_subgrad(a, n_iters, lr_init, momentum, time_limit):
    """Nesterov-accelerated subgradient for exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        m = momentum * (1.0 + it * 0.001)
        m = min(m, 0.99)
        lr = lr_init / (1.0 + 0.01 * it)

        a_look = a - m * v
        a_look = np.clip(a_look, 0.0, 1000.0)

        c = fast_conv(a_look)
        S = float(a_look.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))

        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a_look[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        v = m * v + lr * grad / gn
        a = a_look - v
        a = np.clip(a, 0.0, 1000.0)

        if it % 100 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_sc, best_a


def peak_shave(a, n_iters, lr, time_limit):
    """Iteratively shave the peak convolution value."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        S = float(a.sum())
        sc = 2.0 * n * c[k_star] / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        contrib = np.zeros(n)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1] * a[j_start:j_end + 1]

        total_contrib = float(contrib.sum())
        if total_contrib < 1e-15:
            break

        reduce_mask = contrib > 0.1 * contrib.max()
        n_reduce = int(reduce_mask.sum())
        if n_reduce == 0:
            break

        reduce_amt = lr * contrib[reduce_mask] / max(total_contrib, 1e-15)
        a[reduce_mask] -= reduce_amt
        a = np.clip(a, 0.0, 1000.0)

        non_reduce = ~reduce_mask
        n_nr = int(non_reduce.sum())
        if n_nr > 0:
            mass = float(reduce_amt.sum())
            a[non_reduce] += mass / n_nr
            a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def exact_subgrad_polish(a, total_iters, time_limit):
    """Plain subgradient descent on exact max-conv with Polyak step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    prev_sc = best_sc
    stall_count = 0

    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        gap = max(sc - sc_lb, 1e-6)
        step = gap / (gn * gn + 1e-30)
        step = min(step, 0.1)
        if stall_count > 200:
            step *= 0.5

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].astype(np.float64).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    inits.append(0.1 + 0.9 * (1.0 - (2.0 * x - 1.0) ** 4))
    inits.append(np.clip(0.5 + 0.3 * np.sinc(3.0 * (x - 0.5)), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.4 * rng.random(n), 0.01, 1000.0))

    for nb in (8, 16, 32):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())

    a_bimodal = 0.2 + 0.8 * (np.cos(np.pi * x) ** 2)
    inits.append(a_bimodal)

    a_edges = np.ones(n) * 0.1
    edge_w = max(1, n // 10)
    a_edges[:edge_w] = 0.8
    a_edges[-edge_w:] = 0.8
    inits.append(a_edges)

    return inits


def run_lbfgs_lse(a0, beta, maxiter=3000, maxfun=60000):
    """Run L-BFGS-B with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_with_args, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out

def targeted_perturb(a, rng, strength):
    """Perturb elements contributing most to peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:max(top_count, n // 4)]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)
