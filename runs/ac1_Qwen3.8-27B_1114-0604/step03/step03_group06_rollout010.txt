def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def topk_obj_grad(a, k, temperature):
    """Soft top-k convolution score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_sorted = np.sort(c)[::-1]
    if k < len(c_sorted):
        # Assign higher weights to top-k elements
        top_k_vals = c_sorted[:k]
        threshold = c_sorted[k - 1]
        # Smooth step function
        w = 1.0 / (1.0 + np.exp(-temperature * (c - threshold)))
        w = w / w.sum()
    else:
        w = np.ones(len(c)) / len(c)
    
    L = float(np.dot(w, c))
    score = 2.0 * n * L / (S * S)
    
    N = 1 << (3 * n - 2).bit_length()
    RA = np.fft.rfft(a[::-1], N)
    WA = np.fft.rfft(w, N)

    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun, use_topk=False, topk_k=5, topk_temp=10.0):
    """Run L-BFGS-B with LSE or top-k objective."""
    n = len(a0)

            return topk_obj_grad(x, topk_k, topk_temp)
    else:

            return lse_obj_grad(x, beta)
    
    res = minimize(
        obj, a0,

def subgrad_step(a, step_size):
    """One exact subgradient step on the max-conv objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    M = float(c.max())

    S3 = S2 * S

    a_new = a - step_size * grad
    return np.clip(a_new, 0.0, 1000.0)


def nesterov_subgrad(a, total_iters, initial_lr, final_lr):
    """Nesterov accelerated subgradient descent on exact max-conv."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    y = a.copy()
    
    for it in range(total_iters):
        t = 1.0 + it
        lr = initial_lr * (final_lr / initial_lr) ** (it / max(total_iters - 1, 1))
        
        c = fast_conv(y)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        
        # Polyak step size
        sc_lb = 1.0
        sc_cur = 2.0 * n * M / (S * S)
        polyak_lr = (sc_cur - sc_lb) / (gn * gn + 1e-30)
        step = min(lr, polyak_lr, 0.5)
        
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        # Nesterov momentum
        alpha = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = (1.0 + alpha) * a_new - alpha * a
        y = np.clip(y, 0.0, 1000.0)
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

import numpy as np

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient for the convolution objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient: g[i] = sum_k p[k] * 2*a[k-i] for 0<=k-i<n
    # Computed as 2 * convolve(rev(a), p)[n-1 : 2n-1]
    N = 1 << (3 * n - 2).bit_length()
    RA = np.fft.rfft(a[::-1], N)
    PA = np.fft.rfft(p, N)
    conv_res = np.fft.irfft(RA * PA, N)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def topk_obj_grad(a, k, temperature):
    """Soft top-k convolution score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_sorted = np.sort(c)[::-1]
    if k < len(c_sorted):
        threshold = c_sorted[min(k - 1, len(c_sorted) - 1)]
        w = 1.0 / (1.0 + np.exp(-temperature * (c - threshold)))
        w = w / float(w.sum())
    else:
        w = np.ones(len(c)) / len(c)

    L = float(np.dot(w, c))
    score = 2.0 * n * L / (S * S)

    N = 1 << (3 * n - 2).bit_length()
    RA = np.fft.rfft(a[::-1], N)
    WA = np.fft.rfft(w, N)
    conv_res = np.fft.irfft(RA * WA, N)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def nesterov_subgrad(a, total_iters, initial_lr, decay_factor, max_wall_s):
    """Nesterov accelerated subgradient on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    y = a.copy()
    t0 = time.monotonic()
    sc_lb = 1.0

    for it in range(total_iters):
        if time.monotonic() - t0 > max_wall_s:
            break
        t = 1.0 + it
        lr = initial_lr * (decay_factor ** (it / max(total_iters - 1, 1)))

        c = fast_conv(y)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        sc_cur = 2.0 * n * M / (S * S)
        polyak_lr = max(0.0, (sc_cur - sc_lb) / (gn * gn + 1e-30))
        step = min(lr, polyak_lr, 0.5)
        if step < 1e-15:
            break

        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)

        alpha = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = (1.0 + alpha) * a_new - alpha * a
        y = np.clip(y, 0.0, 1000.0)
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def multi_peak_subgrad(a, total_iters, top_k, initial_lr, decay_factor, max_wall_s):
    """Subgradient using weighted average of top-k peaks."""
    n = len(a)

        c = fast_conv(a)

        if S < 1e-12:

        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        # Exponential weighting
        w = np.exp(20.0 * (top_vals - top_vals[0]))
        w = w / float(w.sum())

        S2 = S * S
        S3 = S2 * S
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            j_start = max(0, int(ki) - n + 1)
            j_end = min(int(ki), n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                grad[j_start:j_end + 1] += wi * 2.0 * seg[::-1]

        L_eff = float(np.dot(w, c[top_idx]))

        gn = float(np.linalg.norm(grad))

        if step < 1e-15:

        a = a_new

        sc = compute_score(a)
        if sc < best_sc:

            best_a = a.copy()

        max_inits = max(2, min(len(inits), int(remaining / 70.0)))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            a_cur = a0.astype(np.float64).copy()
            for bi, beta in enumerate(beta_schedule):
                if time.monotonic() > phase1_end:
                    break
                remaining_bi = phase1_end - time.monotonic()
                if remaining_bi < 30:
                    break
                # Scale maxiter with remaining time
                maxiter = min(8000, max(200, int(remaining_bi / 10.0)))
                maxfun = maxiter * 3
                obj_fn = make_lse_fn(beta)
                sc, a_opt = run_lbfgs(a_cur, obj_fn, maxiter, maxfun)
                a_cur = a_opt
                add_score_state(state, sc, a_opt, f"n={n},i={i},b={beta}")

    print(f"\nAfter Phase 1: best = {state['best_score']:.10f}")

    # =========================================================================
    # Phase 2: Top-k focused optimization (15% of budget)
    # =========================================================================
    phase2_end = t_start + 0.65 * (budget_s - 25.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: Top-k optimization (deadline {phase2_end - t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = state['best_a'].copy()
    for topk_k in [3, 5, 8]:
        if time.monotonic() > phase2_end:
            break
        for temp in [5.0, 10.0, 25.0, 50.0, 100.0]:
            if time.monotonic() > phase2_end:
                break
            remaining = phase2_end - time.monotonic()
            if remaining < 20:
                break
            maxiter = min(5000, max(500, int(remaining / 15.0)))
            obj_fn = make_topk_fn(topk_k, temp)
            sc, a_opt = run_lbfgs(a_cur, obj_fn, maxiter, maxiter * 3)
            add_score_state(state, sc, a_opt, f"topk={topk_k},T={temp}")
            a_cur = a_opt

    print(f"\nAfter Phase 2: best = {state['best_score']:.10f}")

    # =========================================================================
    # Phase 3: Nesterov + multi-peak subgradient (15% of budget)
    # =========================================================================
    phase3_end = t_start + 0.80 * (budget_s - 25.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Subgradient methods (deadline {phase3_end - t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = state['best_a'].copy()
    n_best = len(a_cur)
    base_lr = 0.3 / (float(np.linalg.norm(a_cur)) + 1e-6)

    for it in range(15):
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 30:
            break

        # Nesterov subgradient
        n_iters = min(500, int(remaining * 2))
        sc, a_nest = nesterov_subgrad(a_cur, n_iters, base_lr, 0.02, max_wall_s=remaining * 0.4)
        add_score_state(state, sc, a_nest, f"nest#{it}")
        a_cur = a_nest

        # Multi-peak subgradient
        if time.monotonic() > phase3_end - 20:
            break
        sc, a_mp = multi_peak_subgrad(a_cur, 300, top_k=5,
                                      initial_lr=base_lr * 0.5,
                                      decay_factor=0.05,
                                      max_wall_s=remaining * 0.3)
        add_score_state(state, sc, a_mp, f"multi#{it}")
        a_cur = a_mp

        # Polish with L-BFGS
        if time.monotonic() > phase3_end - 10:
            break
        obj_fn = make_lse_fn(2000)
        sc, a_pol = run_lbfgs(a_cur, obj_fn, 2000, 6000)
        add_score_state(state, sc, a_pol, f"polish#{it}")
        a_cur = a_pol

    print(f"\nAfter Phase 3: best = {state['best_score']:.10f}")

    # =========================================================================
    # Phase 4: Population diversification (10% of budget)
    # =========================================================================
    phase4_end = t_start + 0.90 * (budget_s - 25.0)
    print(f"\n{'='*60}")
    print(f"Phase 4: Diversification (deadline {phase4_end - t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(state['best_a'])
    x = np.arange(n_best, dtype=np.float64) / n_best

    populations = []

    # Gaussian perturbation
    a_p = state['best_a'] * (1.0 + 0.04 * rng.standard_normal(n_best))
    populations.append(np.clip(a_p, 0.01, 1000.0))

    # Frequency perturbation
    freq = 1.0 + 12.0 * rng.random()
    a_p = state['best_a'] * (1.0 + 0.03 * np.cos(2.0 * np.pi * freq * x))
    populations.append(np.clip(a_p, 0.01, 1000.0))

    # Sparse strong perturbation
    a_p = state['best_a'].copy()
    n_sparse = max(30, n_best // 40)
    idx = rng.integers(0, n_best, size=n_sparse)
    a_p[idx] *= (1.0 + 0.4 * rng.standard_normal(n_sparse))
    populations.append(np.clip(a_p, 0.01, 1000.0))

    # Mix with structured
    a_struct = 0.5 + 0.2 * np.cos(2.0 * np.pi * 5 * x) + 0.1 * np.cos(2.0 * np.pi * 11 * x)
    a_p = 0.8 * state['best_a'] + 0.2 * a_struct * (state['best_a'].sum() / a_struct.sum())
    populations.append(np.clip(a_p, 0.01, 1000.0))

    # Block shuffle
    a_p = state['best_a'].copy()
    block_sz = max(32, n_best // 24)
    n_blocks = n_best // block_sz
    if n_blocks > 2:
        perm = rng.permutation(n_blocks)
        for i in range(n_blocks):
            a_p[i * block_sz:(i + 1) * block_sz] = state['best_a'][perm[i] * block_sz:(perm[i] + 1) * block_sz]
    populations.append(a_p)

    # Targeted perturbation on peak contributors
    c = fast_conv(state['best_a'])
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n_best + 1)
    j_end = min(k_star, n_best - 1)
    a_p = state['best_a'].copy()
    if j_end >= j_start:
        # Reduce elements that contribute to the peak
        reduce_idx = np.arange(j_start, j_end + 1)
        a_p[reduce_idx] *= 0.95
        # Redistribute to non-peak elements
        non_peak = np.setdiff1d(np.arange(n_best), reduce_idx)
        if len(non_peak) > 0:
            deficit = float(state['best_a'][reduce_idx].sum() - a_p[reduce_idx].sum())
            a_p[non_peak] += deficit / len(non_peak)
    populations.append(np.clip(a_p, 0.01, 1000.0))

    for idx_pop, pop in enumerate(populations):
        if time.monotonic() > phase4_end:
            break
        # L-BFGS polish
        remaining = phase4_end - time.monotonic()
        if remaining < 30:
            break
        maxiter = min(5000, max(500, int(remaining / 20.0)))
        obj_fn = make_lse_fn(1000)
        sc, a_opt = run_lbfgs(pop, obj_fn, maxiter, maxiter * 2)
        add_score_state(state, sc, a_opt, f"pop#{idx_pop}")

        # Second round with higher beta
        if time.monotonic() > phase4_end - 10:
            break
        obj_fn2 = make_lse_fn(5000)
        sc, a_opt2 = run_lbfgs(a_opt, obj_fn2, 3000, 9000)
        add_score_state(state, sc, a_opt2, f"pop#{idx_pop}+")

    print(f"\nAfter Phase 4: best = {state['best_score']:.10f}")

    # =========================================================================
    # Phase 5: Final refinement (5% of budget)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final refinement (deadline {deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 10.0:
        a_final = state['best_a'].copy()
        # Very high beta LSE
        obj_fn = make_lse_fn(10000)
        maxiter = min(15000, int((deadline - time.monotonic() - 10.0) / 5.0))
        maxiter = max(1000, maxiter)
        sc, a_opt = run_lbfgs(a_final, obj_fn, maxiter, maxiter * 2)
        add_score_state(state, sc, a_opt, "final_high_beta")

    # Nesterov on final best
    if time.monotonic() < deadline - 5.0:
        a_final = state['best_a'].copy()
        base_lr = 0.2 / (float(np.linalg.norm(a_final)) + 1e-6)
        remaining = deadline - time.monotonic() - 5.0
        sc, a_nest = nesterov_subgrad(a_final, min(1000, int(remaining * 3)),
                                      base_lr, 0.01, max_wall_s=remaining)
        add_score_state(state, sc, a_nest, "final_nesterov")

    # One last L-BFGS
    if time.monotonic() < deadline - 2.0:
        obj_fn = make_lse_fn(5000)
        sc, a_opt = run_lbfgs(state['best_a'], obj_fn, 3000, 10000)
        add_score_state(state, sc, a_opt, "absolute_final")

    # =========================================================================
    # Output
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in state['best_a']]
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={state['best_score']:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
