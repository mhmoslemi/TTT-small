    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter,
                 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def phase_retrieval_init(n, rng, n_iters=300):
    """Fienup phase retrieval targeting sinc-squared power spectrum."""
    M = 2 * n - 1
    M_use = 1 << (M - 1).bit_length()
    k = np.arange(M_use)
    L = 2 * n - 1
    with np.errstate(divide='ignore', invalid='ignore'):
        target = np.abs(np.sin(np.pi * L * k / M_use) / np.sin(np.pi * k / M_use))
    target[0] = L
    target = target / np.sqrt(float(target.sum()))
    a = np.zeros(M_use)
    a[:n] = 0.5 + 0.3 * rng.random(n)
    best_a = a[:n].copy()
    best_sc = compute_score(a[:n])
    for it in range(n_iters):
        A = np.fft.fft(a)
        phase = np.angle(A)
        blend = 0.3 + 0.5 * it / n_iters
        cur_mag = np.abs(A)
        new_mag = (1.0 - blend) * cur_mag + blend * target
        a_new = np.fft.ifft(new_mag * np.exp(1j * phase)).real
        a_new[:n] = np.maximum(a_new[:n], 0.0)
        a_new[n:] = 0.0
        a = a_new
        sc = compute_score(a[:n])
        if sc < best_sc:
            best_sc = sc
            best_a = a[:n].copy()
    S = best_a.sum()
    if S > 1e-12:
        best_a = best_a / S
    return best_a


def multiplicative_subgrad(a, n_iters=5000, f_lb=1.0):
    """Multiplicative weight subgradient descent (preserves non-negativity)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg[idx] = 2.0 * a[k_star - idx]
        norm = float(np.max(sg))
        if norm < 1e-15:
            break
        eta_cur = 0.5 * (n_iters / (n_iters + it)) ** 0.5 / (1.0 + 0.1 * it)
        a = a * np.exp(-eta_cur * sg / (norm + 1e-30))
        a = np.clip(a, 1e-15, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    return best_a


def multi_peak_subgrad(a, n_iters=3000, k_peaks=3, f_lb=1.0):
    """Subgradient using top-k peaks with EMA variance reduction."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    g_avg = np.zeros(n)
    for it in range(n_iters):
        c = fast_conv(a)
        top_k_idx = np.argsort(c)[::-1][:k_peaks]
        grad = np.zeros(n)
        for kp in top_k_idx:
            j_start = max(0, kp - n + 1)
            j_end = min(kp, n - 1)
            if j_end >= j_start:
                idx = np.arange(j_start, j_end + 1)
                grad[idx] += 2.0 * a[kp - idx]
        norm_grad = float(np.linalg.norm(grad))
        if norm_grad < 1e-15:
            break
        Mak = float(c.max())
        sc = 2.0 * n * Mak
        step = (sc - 1.0) / (norm_grad + 1e-30)
        step = min(step, 0.05)
        alpha = 0.9
        g_avg = alpha * g_avg + (1.0 - alpha) * (grad / (norm_grad + 1e-30))
        a = a - step * g_avg
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def nesterov_subgrad(a, n_iters=5000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    return best_a


def targeted_peak_reduction(a, n_sweeps=5, n_top=50):
    """Coordinate descent: reduce top contributors to peak, redistribute."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            best_val = a_old_j
            best_local_sc = compute_score(a)
            for frac in [0.3, 0.5, 0.7, 0.85]:
                new_val = a_old_j * frac
                a[j] = new_val
                reduction = a_old_j - new_val
                if reduction > 1e-15:
                    per = reduction / len(bottom_local)
                    for li in bottom_local:
                        a[j_start + li] += per
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = new_val
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version for iterated local search."""
    n = len(a)
    strategy = rng.integers(0, 5)
    x = np.arange(n, dtype=np.float64) / n
    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.1 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bottom_local)
            for li in bottom_local:
                a_p[j_start + li] += per
    else:
        n_sel = max(1, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))
    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses multi-strategy approach: phase retrieval init, LSE L-BFGS,
    multiplicative/multi-peak subgradient, targeted peak reduction, ILS,
    and length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    budget = float(budget_s)
    deadline = t_start + budget - 30.0

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Get parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        pn = parent / parent.sum()
        sc = compute_score(pn)
        print(f"Parent: n={len(parent)}, score={sc:.10f}")
        report(sc, pn, "parent")

    n_work = len(parent) if parent is not None else 8192
    print(f"Working length: {n_work}")

    # Build working copy
    if len(best_a) != n_work:
        a_cur = np.interp(np.linspace(0, 1, n_work),
                          np.linspace(0, 1, len(best_a)), best_a)
    else:
        a_cur = best_a.copy()
    a_cur = np.clip(a_cur, 0.0, 1000.0)
    S = a_cur.sum()
    if S > 1e-12:
        a_cur = a_cur / S

    # ============================================================
    # Phase 1: Quick LSE L-BFGS on parent (10% time)
    # ============================================================
    phase1_end = t_start + 0.10 * (budget - 30.0)
    print(f"\n--- Phase 1: Quick LSE L-BFGS ---")

    for beta in [300, 1000]:
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        mi = int(min(20000, max(3000, remaining * 40)))
        mf = int(min(100000, max(10000, remaining * 150)))
        sc, a_opt = run_lbfgs(a_cur, beta, mi, mf)
        report(sc, a_opt, f"P1 beta={beta}")
        a_cur = best_a.copy()

    # ============================================================
    # Phase 2: Frequency domain phase retrieval (10% time)
    # ============================================================
    phase2_end = t_start + 0.20 * (budget - 30.0)
    print(f"\n--- Phase 2: Phase Retrieval Init ---")

    n_pr = n_work
    if time.monotonic() < phase2_end:
        a_pr = phase_retrieval_init(n_pr, rng, n_iters=200)
        sc = compute_score(a_pr)
        report(sc, a_pr, "phase_retrieval")
        if sc < best_score:
            a_cur = a_pr.copy()
        # Refine the phase retrieval result with L-BFGS
        if time.monotonic() < phase2_end:
            sc, a_opt = run_lbfgs(a_pr, 500, 10000, 50000)
            report(sc, a_opt, "PR+LBFGS")

    # ============================================================
    # Phase 3: Multi-peak subgradient + multiplicative (10% time)
    # ============================================================
    phase3_end = t_start + 0.30 * (budget - 30.0)
    print(f"\n--- Phase 3: Multi-peak & Multiplicative Subgrad ---")

    if time.monotonic() < phase3_end:
        a_mp = multi_peak_subgrad(a_cur, n_iters=3000, k_peaks=3)
        sc = compute_score(a_mp)
        report(sc, a_mp, "multi_peak_sg")

    if time.monotonic() < phase3_end:
        a_ms = multiplicative_subgrad(a_cur, n_iters=3000)
        sc = compute_score(a_ms)
        report(sc, a_ms, "multiplicative_sg")

    if time.monotonic() < phase3_end:
        a_ns = nesterov_subgrad(a_cur, n_iters=3000)
        sc = compute_score(a_ns)
        report(sc, a_ns, "nesterov")

    # ============================================================
    # Phase 4: Main LSE L-BFGS optimization (30% time)
    # ============================================================
    phase4_end = t_start + 0.60 * (budget - 30.0)
    print(f"\n--- Phase 4: Main LSE L-BFGS ---")

    beta_schedule = [500, 1000, 3000, 5000, 10000, 30000, 50000, 100000]
    for beta in beta_schedule:
        if time.monotonic() > phase4_end:
            break
        remaining = phase4_end - time.monotonic()
        if remaining < 20:
            break
        mi = int(min(50000, max(5000, remaining * 30)))
        mf = int(min(200000, max(20000, remaining * 120)))
        sc, a_opt = run_lbfgs(best_a, beta, mi, mf)
        report(sc, a_opt, f"P4 beta={beta}")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ============================================================
    # Phase 5: Targeted peak reduction (5% time)
    # ============================================================
    phase5_end = t_start + 0.65 * (budget - 30.0)
    print(f"\n--- Phase 5: Targeted Peak Reduction ---")

    if time.monotonic() < phase5_end:
        a_tpr = targeted_peak_reduction(best_a, n_sweeps=3, n_top=50)
        sc = compute_score(a_tpr)
        report(sc, a_tpr, "targeted_peak")
        # Follow up with L-BFGS
        if time.monotonic() < phase5_end:
            sc, a_opt = run_lbfgs(a_tpr, 5000, 10000, 50000)
            report(sc, a_opt, "targeted+lbfgs")

    # ============================================================
    # Phase 6: ILS with perturbation (20% time)
    # ============================================================
    phase6_end = t_start + 0.85 * (budget - 30.0)
    print(f"\n--- Phase 6: Iterated Local Search ---")

    for trial in range(30):
        if time.monotonic() > phase6_end:
            break
        remaining = phase6_end - time.monotonic()
        if remaining < 30:
            break
        scale = 0.015 + 0.025 * rng.random()
        a_pert = perturbed_restart(best_a, rng, scale=scale)
        beta = 2000 + int(8000 * rng.random())
        mi = int(min(30000, max(3000, remaining * 15)))
        mf = int(min(100000, max(10000, remaining * 60)))
        sc, a_opt = run_lbfgs(a_pert, beta, mi, mf)
        report(sc, a_opt, f"ILS #{trial} b={beta}")

    print(f"After Phase 6: best = {best_score:.10f}")

    # ============================================================
    # Phase 7: Length exploration (5% time)
    # ============================================================
    phase7_end = t_start + 0.90 * (budget - 30.0)
    print(f"\n--- Phase 7: Length Exploration ---")

    if parent is not None:
        n_parent = len(parent)
        for n_try in [int(n_parent * 0.75), int(n_parent * 1.25),
                      int(n_parent * 1.5), int(n_parent * 0.5)]:
            if time.monotonic() > phase7_end:
                break
            if n_try < 256 or n_try > 50000:
                continue
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                  np.linspace(0, 1, len(best_a)), best_a)
            a_interp = np.clip(a_interp, 0.0, 1000.0)
            S = a_interp.sum()
            if S > 0:
                a_interp = a_interp / S
            sc, a_opt = run_lbfgs(a_interp, 1000, 15000, 60000)
            report(sc, a_opt, f"len={n_try}")

    # ============================================================
    # Phase 8: Final polish (5% time)
    # ============================================================
    print(f"\n--- Phase 8: Final Polish ---")

    if time.monotonic() < deadline:
        for beta in [50000, 200000]:
            if time.monotonic() > deadline - 15:
                break
            sc, a_opt = run_lbfgs(best_a, beta, 30000, 100000)
            report(sc, a_opt, f"final b={beta}")

    if time.monotonic() < deadline - 15:
        a_ns2 = nesterov_subgrad(best_a, n_iters=2000)
        sc = compute_score(a_ns2)
        report(sc, a_ns2, "final_nesterov")

    if time.monotonic() < deadline - 10:
        a_mps = multi_peak_subgrad(best_a, n_iters=2000, k_peaks=2)
        sc = compute_score(a_mps)
        report(sc, a_mps, "final_multi_peak")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
