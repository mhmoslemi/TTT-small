def spectral_init(n, iterations=100):
    M = 2 * n  # FFT size (power of 2)
    # Start with some phases
    A = np.zeros(M, dtype=complex)
    # Uniform magnitude, random phases (conjugate symmetric for real signal)
    phases = np.random.uniform(0, 2*np.pi, M)
    phases[0] = 0
    phases[M//2] = 0
    phases[M//2+1:] = -phases[1:M//2][::-1]  # conjugate symmetry
    A = np.exp(1j * phases)
    # Iterate
    for _ in range(iterations):
        a = np.real(np.fft.ifft(A))[:n]
        a = np.clip(a, 0, None)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        # Recompute spectrum
        A_full = np.fft.fft(a, M)
        # Drive magnitude toward flat
        target_mag = np.ones(M)
        target_mag[0] = 0  # DC bin determines the sum, keep it
        # Blend
        mag = np.abs(A_full)
        mag[0] = A_full[0].real
        new_mag = 0.5 * mag + 0.5 * target_mag
        phase = np.angle(A_full)
        A_full = new_mag * np.exp(1j * phase)
        A = A_full
    return a

def lse_score_and_grad_simple(a, beta):
    """Simpler LSE gradient using full convolution (no top-k). More stable."""
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    scaled = np.maximum(scaled, -50.0)
    exp_scaled = np.exp(scaled)
    Z = exp_scaled.sum()
    p = exp_scaled / Z
    L = c_max + np.log(Z) / beta
    score = 2.0 * n * L / (S * S)

    M2 = 1
    while M2 < 2 * n - 1:
        M2 *= 2

    P = np.fft.rfft(p, M2)
    A_fft = np.fft.rfft(a, M2)

    conv_pa = np.fft.irfft(P * A_fft, M2)
    P_conj = P.copy()
    P_conj.imag = -P_conj.imag
    xcorr_pa = np.fft.irfft(P_conj * A_fft, M2)

    g = xcorr_pa[:n] + conv_pa[:n]

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g / S2 - 2.0 * L / S3)

    return float(score), grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=100000, top_k=None, time_limit=60.0):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    t0 = time.monotonic()

    if top_k is not None:
        def func(x):
            return lse_score_and_grad(x, beta, top_k=top_k)
    else:
        def func(x):
            return lse_score_and_grad_simple(x, beta)

    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    elapsed = time.monotonic() - t0
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res, elapsed


def peak_shave_coordinate(a, n_sweeps=3, n_top_peaks=5, n_top_contrib=50):

    from high contributors to low contributors at the peak position.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:

    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Find top peak positions
        peak_indices = np.argsort(c)[::-1][:n_top_peaks]
        
        # For each peak, find the top contributors
        all_high = set()
        all_low = set()
        for k in peak_indices:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)

            bot_idx = np.argsort(contrib)[min(n_top_contrib, len(contrib)-1):]
            all_high.update(j_start + top_idx)
            all_low.update(j_start + bot_idx)
        
        all_high = list(all_high)
        all_low = list(all_low)
        
        if not all_high or not all_low:
            break
        
        # Try transferring mass from high to low contributors
        for j_high in all_high:
            if a[j_high] < 1e-12:
                continue
            for j_low in all_low:
                if j_low == j_high:
                    continue
                old_val = a[j_high]
                # Try several transfer amounts
                best_transfer = 0.0
                best_local_sc = best_sc
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    transfer = old_val * frac
                    a[j_high] = old_val - transfer
                    a[j_low] += transfer
                    sc = compute_score(a)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_transfer = transfer
                    # Restore
                    a[j_high] = old_val
                    a[j_low] -= transfer
                
                if best_transfer > 0:
                    a[j_high] = old_val - best_transfer
                    a[j_low] += best_transfer
                    S = a.sum()
                    a = a / S
                    sc = compute_score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
    
    return best_a


def structured_perturbation(a, rng, scale=0.02):
    """Generate a structured perturbation of a."""
    n = len(a)
    strategy = rng.integers(0, 6)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        # Gaussian
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        # Sinusoidal with random frequency
        freq = 1.0 + 20.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        a_p = a * (1.0 + scale * 2.0 * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        # Block perturbation
        n_blocks = 2 + int(rng.integers(0, 6))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 3.0 * (rng.random() - 0.5))
    elif strategy == 3:
        # Peak-targeted: reduce top contributors at the main peak
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = np.zeros(len(seg))
        for i, j in enumerate(range(j_start, j_end + 1)):
            contrib[i] = seg[i] * a_p[k_star - j]
        top_count = max(5, n // 100)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.15 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bot_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bot_local) if len(bot_local) > 0 else 0
            for li in bot_local:
                a_p[j_start + li] += per
    elif strategy == 4:
        # Frequency-domain perturbation
        M = 1
        while M < 2 * n:
            M *= 2
        A = np.fft.rfft(a, M)
        # Perturb phases slightly
        phase_pert = scale * 0.5 * rng.standard_normal(len(A))
        A_pert = A * np.exp(1j * phase_pert)
        a_p = np.real(np.fft.irfft(A_pert, M))[:n]
    else:
        # Random subset rescale
        n_sel = max(10, n // 5)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 4.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def spectral_refine(a, n_iters=50):
    """
    Spectral-domain refinement: iterate between time-domain (clip to non-negative)
    and frequency-domain (flatten the power spectrum).
    """
    n = len(a)
    M = 1
    while M < 2 * n - 1:
        M *= 2
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        # Compute power spectrum
        A = np.fft.rfft(a, M)
        P = A * np.conj(A)  # Power spectrum
        
        # Get autocorrelation
        r = np.fft.irfft(P, M)[:2 * n - 1]
        
        # In frequency domain, we want |A|^2 to be as flat as possible
        # (excluding DC bin which determines the sum)
        # Strategy: blend current power spectrum with a flatter target
        target = np.ones(len(P))
        target[0] = P[0]  # Keep DC
        # Smooth the target slightly
        alpha = 0.1 + 0.1 * (it / n_iters)
        new_P = (1 - alpha) * P + alpha * target * (P.sum() / target.sum())
        new_P[0] = P[0]  # Keep DC exactly
        
        # New magnitude, keep old phase
        new_mag = np.sqrt(np.maximum(new_P.real, 0))
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        
        # Back to time domain
        a_new = np.real(np.fft.irfft(A_new, M))[:n]
        a_new = np.clip(a_new, 0, None)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert if it got worse
            break
    
    return best_a

        report(sc_parent, parent_norm, "parent")
