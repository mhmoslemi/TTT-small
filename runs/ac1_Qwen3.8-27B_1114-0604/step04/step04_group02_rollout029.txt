```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(np.maximum(scaled, -500))
    p = exp_s / exp_s.sum()
    L = c_max + np.log(exp_s.sum()) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient of L with respect to a
    # dL/da_i = sum_k p_k * d(c_k)/da_i = sum_k p_k * 2 * a(k-i) (for valid k-i)
    # = 2 * (p_rev * a) [cross-correlation]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    g_L = 2.0 * conv_res
    
    # Gradient of score = 2n * L / S^2
    # d(score)/da_i = 2n * (g_L_i * S^2 - L * 2 * S) / S^4
    #                = 2n * (g_L_i / S^2 - 2L / S^3)
    grad = 2.0 * n * (g_L / (S*S) - 2.0 * L / (S*S*S))
    return score, grad

def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0"""
    n = len(f)
    # Build constraint matrix (sparse would be ideal, but let's try dense first)
    # For n=7000, (2n-1) x n = 14000 x 7000 = 98M entries... too large for dense
    # Need sparse
    from scipy import sparse
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)  # minimize -sum(g)
    
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
4. Iterated local search with diverse perturbations

import numpy as np
from scipy.optimize import minimize
import time
from scipy import sparse


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta, top_k=None):
    """LSE-smoothed score with gradient. Optional top-k restriction."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    work = c.copy()
    
    if top_k is not None and top_k < len(c):
        # Proper top-k: zero out non-top-k elements
        idx = np.argpartition(c, -top_k)[-top_k:]
        mask = np.zeros(len(c), dtype=bool)
        mask[idx] = True
        work = np.where(mask, c, -np.inf)
    
    scaled = beta * (work - c_max)
    exp_s = np.exp(np.maximum(scaled, -500))
    
    # Handle case where all top-k values are -inf
    if exp_s.sum() == 0:
        return 2.0 * n * c_max / (S * S), np.zeros(n)
    
    p = exp_s / exp_s.sum()
    L = c_max + float(np.log(exp_s.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient computation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    g_L = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=100000, top_k=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    func = lambda x: lse_score_grad(x, beta, top_k)
    
    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def nesterov_subgrad(a, n_iters=3000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]
        
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.5)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    
    return best_a
```

Actually, for the LP, I realize I should think about it differently. The LP gives us a direction g that is "compatible" with the current f. The idea is:

1. f is current, M = max(f*f)
2. g maximizes sum(g) subject to f*g <= M, g >= 0
3. Since f itself is feasible (f*f <= M), we have sum(g) >= sum(f)
4. Rescale: g' = g * sum(f) / sum(g), so sum(g') = sum(f)
5. f*g' <= M * sum(f)/sum(g) <= M
6. New f = (1-t)*f + t*g'

For the new f, the self-convolution is:
f_new * f_new = (1-t)^2 f*f + 2(1-t)t f*g' + t^2 g'*g'

We know f*f <= M and f*g' <= M. But g'*g' could be anything. However, for small t, the t^2 term is negligible, so:
f_new * f_new <= (1-t)^2 * M + 2(1-t)t * M + t^2 * max(g'*g')
= M * (1-t)^2 + 2M*t*(1-t) + t^2 * max(g'*g')
= M * (1 - 2t + t^2) + 2Mt - 2Mt^2 + t^2 * max(g'*g')
= M - 2Mt + Mt^2 + 2Mt - 2Mt^2 + t^2 * max(g'*g')
= M + t^2 * (max(g'*g') - M)

So if max(g'*g') > M, the peak increases! The LP doesn't prevent this.

For the iteration to decrease the peak, we need max(g'*g') <= M, which is not guaranteed.

However, in practice, the iteration often works because:
- g tends to be "flatter" than f (since it maximizes sum subject to constraints)
- max(g*g) tends to be smaller than max(f*f) for flatter sequences
- The convex combination then has a smaller peak

So it's a heuristic, not a guaranteed descent method. But it can still be effective.
