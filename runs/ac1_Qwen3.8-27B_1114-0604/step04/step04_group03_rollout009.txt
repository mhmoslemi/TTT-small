    # From the previous code:
    # rev_a = a[::-1]
    # r = ifft(fft(rev_a) * fft(p))[n-1:2n-1]
    # Let me verify: conv(rev_a, p)[m] = sum_j rev_a[j] * p[m-j] = sum_j a[n-1-j] * p[m-j]
    # Let k = n-1-j: = sum_k a[k] * p[m-n+1+k]
    # We want r[i] = sum_k a[k] * p[k+i], so m-n+1 = i, m = i+n-1
    # r[i] = conv(rev_a, p)[i+n-1]
    
    rev_a = a[::-1]
    RA = np.fft.rfft(rev_a, M2_fft)
    R = np.fft.irfft(RA * P_f, M2_fft)[n-1:2*n-1]
    dL_da = 2.0 * R
    
    # Full gradient of objective = 2n * L / S^2
    # d(2nL/S^2)/da = 2n * (dL_da * S^2 - L * 2*S * 1) / S^4

    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))

def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000, timeout=None, t_start=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    call_count = 0
    should_stop = False
    
    def objective(x):
        nonlocal call_count, should_stop
        call_count += 1
        if timeout is not None and time.monotonic() - t_start > timeout:
            should_stop = True
        s, g = lse_objective_and_gradient(x, beta)
        return s, g
    
    def callback(xk):
        nonlocal should_stop
        if timeout is not None and time.monotonic() - t_start > timeout:
            should_stop = True
        return should_stop
    
    try:
        res = minimize(
            objective, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            callback=callback,
            options={'ftol': 1e-16, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        a_out = a0
    
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def mirror_descent_momentum(a, n_iters=50000, momentum=0.9, eta0=None, sc_lb=1.0):
    """
    Mirror descent with Nesterov momentum on the simplex.
    Maintains sum(a) = 1 and a >= 0.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    v = np.zeros(n)  # momentum velocity
    x = a.copy()  # Nesterov lookahead
    
    if eta0 is None:
        eta0 = 0.5
    
    for it in range(n_iters):
        # Nesterov step
        x = a + momentum * v
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        
        c = fast_conv(x)
        S = float(x.sum())
        M = float(c.max())
        k_star = int(np.argmax(c))
        
        # Subgradient of 2n*M/S^2 at sum=1:
        # g[i] = 4n*(a_source[k*-i] - M) for active i
        # g[i] = -4n*M for inactive i
        # where a_source is x (the Nesterov point)
        
        g = np.full(n, -4.0 * n * M)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = x[j_start:j_end + 1]
            g[j_start:j_end + 1] = 4.0 * n * (seg[::-1] - M)
        
        # Adaptive step size
        sc_now = 2.0 * n * M / (S * S)
        gap = max(sc_now - sc_lb, 1e-8)
        g_norm_sq = float(np.sum(g * g))
        if g_norm_sq < 1e-30:
            eta = eta0
        else:
            eta = gap / g_norm_sq
            eta = min(eta, eta0)
            eta = max(eta, 1e-8)
        
        # Mirror descent update (multiplicative)
        log_a = np.log(a) - eta * g
        log_a -= log_a.max()  # stability
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        
        # Update momentum
        v = momentum * v + (a_new - a)
        a = a_new
        
        # Track best
        if (it + 1) % 100 == 0:
            sc = 2.0 * n * M / (S * S)
            # Recompute with actual a
            c_a = fast_conv(a)
            S_a = float(a.sum())
            sc_a = 2.0 * n * float(c_a.max()) / (S_a * S_a)
            if sc_a < best_sc:
                best_sc = sc_a
                best_a = a.copy()
    
    # Final check
    c_a = fast_conv(a)
    S_a = float(a.sum())
    sc_a = 2.0 * n * float(c_a.max()) / (S_a * S_a)
    if sc_a < best_sc:
        best_sc = sc_a
        best_a = a.copy()
    
    return best_a, best_sc


def adagrad_descent(a, n_iters=100000, lr=0.1, eps=1e-8):
    """
    AdaGrad on the simplex with projective updates.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    G = np.zeros(n)  # accumulated gradient squares
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        k_star = int(np.argmax(c))
        
        # Subgradient
        g = np.full(n, -4.0 * n * M)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            g[j_start:j_end + 1] = 4.0 * n * (seg[::-1] - M)
        
        G += g * g
        eta_t = lr / np.sqrt(G + eps)
        
        a -= eta_t * g
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        if (it + 1) % 200 == 0:
            c_a = fast_conv(a)
            S_a = float(a.sum())
            sc_a = 2.0 * n * float(c_a.max()) / (S_a * S_a)
            if sc_a < best_sc:
                best_sc = sc_a
                best_a = a.copy()
    
    c_a = fast_conv(a)
    S_a = float(a.sum())
    sc_a = 2.0 * n * float(c_a.max()) / (S_a * S_a)
    if sc_a < best_sc:
        best_sc = sc_a
        best_a = a.copy()
    
    return best_a, best_sc


def frequent_peak_reduction(a, n_iters=20000, top_k=5):
    """
    Reduce the top-k peaks simultaneously using a water-filling approach.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top-k peaks
        top_peaks = np.argsort(c)[::-1][:top_k]
        
        # For each peak, compute the subgradient contribution
        g = np.zeros(n)
        for k_star in top_peaks:
            k_star = int(k_star)
            weight = c[k_star]  # weight by peak height
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                g[j_start:j_end + 1] += weight * seg[::-1]
                g[j_start:j_end + 1] -= weight  # subtract mean contribution
        
        # Actually, let's use a simpler approach:
        # For the dominant peak, do a multiplicative update
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        if j_end >= j_start:
            seg = a[j_start:j_end + 1].copy()
            # Products contribute to the peak
            products = seg * seg[::-1]
            total = products.sum()
            if total > 1e-15:
                # Reduce high-product pairs, increase low-product pairs
                # Multiplicative update
                alpha = 0.01 + 0.02 * (it / n_iters)
                # a[i] *= (1 - alpha * contribution[i] / total) for active
                # This reduces the peak
                for i in range(j_start, j_end + 1):
                    j = i - j_start
                    red = alpha * products[j] / total
                    a[i] = max(a[i] * (1.0 - red), 0.0)
        
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a, best_sc


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    x = np.arange(n, dtype=np.float64) / n
    
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x) * np.cos(4.0 * np.pi * x))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.3 + 0.7 * np.abs(2.0 * x - 1.0))
    inits.append(0.3 + 0.7 * (1.0 - (2.0 * x - 1.0) ** 2))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    
    # Block structure
    for nb in (4, 8, 16):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    
    return inits

"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: long-running mirror descent with momentum, LSE-smoothed L-BFGS polish,
multi-length exploration, and adaptive peak targeting.
"""

import numpy as np
from scipy.optimize import minimize
import time


class TimeLimitExceeded(Exception):
    pass


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_gradient(a, beta):
    """LSE-smoothed objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    exp_sum = float(exp_s.sum())
    p = exp_s / exp_sum
    
    L = c_max + float(np.log(exp_sum)) / beta
    objective = 2.0 * n * L / (S * S)
    
    # Gradient computation via FFT
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    
    rev_a = a[::-1]
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    R = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * R
    
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    
    return objective, grad


class LseProblem:
    """Callable class for L-BFGS that computes LSE objective and gradient."""
    def __init__(self, beta):
        self.beta = beta
        self.deadline = None
    
    def __call__(self, x):
        if self.deadline is not None and time.monotonic() > self.deadline:
            raise TimeLimitExceeded()
        s, g = lse_objective_and_gradient(x, self.beta)
        return s, g


class TimeCallback:
    """Callback for L-BFGS that stops when deadline is reached."""
    def __init__(self, deadline):
        self.deadline = deadline
    
    def __call__(self, xk):
        return time.monotonic() > self.deadline


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000, deadline=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    problem = LseProblem(beta)
    if deadline is not None:
        problem.deadline = deadline
        callback = TimeCallback(deadline)
    else:
        callback = None
    
    try:
        kwargs = {
            'ftol': 1e-16, 'gtol': 1e-12,
            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50
        }
        if callback is not None:
            kwargs['callback'] = callback
        res = minimize(
            problem, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            **kwargs
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except TimeLimitExceeded:
        a_out = a0  # Return the input if we ran out of time
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        a_out = a0
    
    if a_out.sum() < 1e-12:
        a_out = a0
    
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def mirror_descent_loop(a, n_iters, momentum, eta0, sc_lb, check_interval):
    """
    Mirror descent with Nesterov momentum.
    Returns (best_a, best_score, final_a).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    v = np.zeros(n)
    x = a.copy()
    
    for it in range(n_iters):
        # Nesterov lookahead
        x = a + momentum * v
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        
        c = fast_conv(x)
        S = float(x.sum())
        M = float(c.max())
        k_star = int(np.argmax(c))
        
        # Subgradient of 2n*M/S^2 (with S=1):
        g = np.full(n, -4.0 * n * M)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = x[j_start:j_end + 1]
            g[j_start:j_end + 1] = 4.0 * n * (seg[::-1] - M)
        
        # Adaptive step
        sc_now = 2.0 * n * M / (S * S)
        gap = max(sc_now - sc_lb, 1e-10)
        g_norm_sq = float(np.sum(g * g))
        if g_norm_sq < 1e-30:
            eta = eta0
        else:
            eta = gap / g_norm_sq
            eta = min(eta, eta0)
            eta = max(eta, 1e-10)
        
        # Multiplicative update
        log_a = np.log(a) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        
        # Momentum update
        v = momentum * v + (a_new - a)
        a = a_new
        
        # Periodic check
        if (it + 1) % check_interval == 0:
            c_a = fast_conv(a)
            S_a = float(a.sum())
            sc_a = 2.0 * n * float(c_a.max()) / (S_a * S_a)
            if sc_a < best_sc:
                best_sc = sc_a
                best_a = a.copy()
    
    # Final check
    c_a = fast_conv(a)
    S_a = float(a.sum())
    sc_a = 2.0 * n * float(c_a.max()) / (S_a * S_a)
    if sc_a < best_sc:
        best_sc = sc_a
        best_a = a.copy()
    
    return best_a, best_sc, a
