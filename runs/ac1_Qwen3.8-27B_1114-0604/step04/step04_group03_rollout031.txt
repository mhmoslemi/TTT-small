        # Actually, let's just use the direct formula with FFT

        A_rev = np.fft.rfft(a[::-1], M)

        # cross[i] = sum_j c[i+j] * a[j]... let me verify
        
        # Actually, (c * a_rev)[k] = sum_j c[j] * a_rev[k-j] = sum_j c[j] * a[n-1-(k-j)]
        # This isn't quite what we want. Let me use a different approach.
        
        # We want: g[i] = sum_{j=0}^{n-1} c[i+j] * a[j] (where c[i+j] is defined for i+j < 2n-1)
        # This is the cross-correlation of c and a at "negative lag" i.
        # In FFT terms: g = ifft(FFT(c) * conj(FFT(a))) ... but c and a have different lengths.
        
        # Simpler: just pad and use the formula
        # g[i] = sum_j c[i+j] * a[j]
        # = (c * a_reversed_padded)[?]
        
        # Let me just do it directly with a loop-free numpy approach:
        # For n ~= 7000, this is a 7000x7000 matrix multiply... that's 49M, feasible but slow.
        # Better to use FFT.
        
        # g = correlation(c, a) where correlation[i] = sum_j c[i+j] * a[j]
        # = ifft(fft(c, M) * conj(fft(a, M))) ... no, that gives sum_j c[j] * a[i-j]
        
        # Actually: sum_j c[i+j] * a[j] = sum_j c[m] * a[m-i] where m = i+j
        # = (c * a_flip)[i + n - 1] where a_flip = a[::-1]
        # No wait: (c * a_flip)[k] = sum_m c[m] * a_flip[k-m] = sum_m c[m] * a[n-1-k+m]
        # Setting n-1-k+m = j => m = k - n + 1 + j
        # So (c * a_flip)[k] = sum_j c[k-n+1+j] * a[j]
        # Setting k-n+1 = i: k = i + n - 1
        # (c * a_flip)[i+n-1] = sum_j c[i+j] * a[j] = g[i] ✓

        C = np.fft.rfft(c, M)

        g = cross_full[n-1:2*n-1]  # g[i] = sum_j c[i+j]*a[j]
        
        grad = 4.0 * g
        # Normalize
        grad = grad / grad.max()
        
        a -= eta * grad
        a = np.maximum(a, 0)
        a = a / a.sum()
    
    return a

**Key improvement idea: Use the structure of the "active set" of peaks.**

At the optimum, typically 2-3 lags have the maximum convolution value (they're "active"). The condition is that at these active lags, the gradient of c_k with respect to any free a_i should be zero (or the subgradient condition should hold).

For a free coordinate i (a_i > 0), the condition for lag k to be active is:
d c_k / d a_i = 2 * a[k-i] = 0 (if k-i is in [0,n-1])

This means a[k-i] = 0. So at the optimum, for each active lag k, we need a[k-i] = 0 for all i where a_i > 0. This seems very restrictive and suggests that the optimum has many zeros.

Wait, that's not quite right. The condition is that the subgradient of max(c) contains 0 in the normal cone of the feasible set. More precisely:

At the optimum, there exist dual variables lambda_k >= 0 (sum lambda_k = 1) such that:
sum_k lambda_k * d c_k / d a_i + mu = 0 for all i with a_i > 0
sum_k lambda_k * d c_k / d a_i + mu <= 0 for all i with a_i = 0

where mu is the Lagrange multiplier for the sum constraint.

This means: sum_k lambda_k * 2*a[k-i] = -mu for all i with a_i > 0.

So the weighted sum of a[k-i] over active lags k is constant for all "free" coordinates i.

This gives us a system of equations that the optimal solution must satisfy. We can use this to design a better optimization algorithm.

**New idea: Active set method**

1. Identify the active lags (those with c_k close to the max).
2. For these lags, solve the linear system: sum_k lambda_k * a[k-i] = const.
3. This gives us the "target" structure of the optimal solution.
4. Use this as a guide for optimization.

    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    c = fast_conv(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    return 2.0 * n * c.max() / (S * S)

"""
Optimizes step function sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
Strategy: Spectrum flattening -> Intensive coordinate descent -> LSE-LBFGS polish -> Final CD.
Key improvements: efficient incremental convolution updates in CD, adaptive candidate
factors, spectrum pre-flattening via L2 objective, and aggressive LSE beta schedule.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_score_from_conv(a, c):
    """Compute score given precomputed convolution."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def flatten_spectrum(a, n_iters=300, eta_init=0.08, eta_end=0.002):
    """Flatten the power spectrum via gradient descent on ||c||_2^2."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (3 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        # Gradient of ||c||_2^2: g[i] = 4 * sum_j c[i+j] * a[j]
        C = np.fft.rfft(c, M)
        A_rev = np.fft.rfft(a[::-1], M)
        cross = np.fft.irfft(C * A_rev, M)
        g = cross[n - 1:2 * n - 1]
        grad = 4.0 * g

        t_frac = it / max(1, n_iters)
        eta = eta_init * (1.0 - t_frac) ** 2 + eta_end

        grad_max = np.abs(grad).max()
        if grad_max < 1e-15:
            break

        a_new = a - eta * grad / grad_max
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def coord_descent(a, n_sweeps=200, verbose=False, deadline=None, rng=None):
    """
    Efficient coordinate descent with incremental convolution updates.
    Returns (best_a, best_score).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) * 0.5
        S = a.sum()
    c = fast_conv(a)
    best_score = 2.0 * n * float(np.max(c)) / (S * S)
    best_a = a.copy()

    # Candidate factors ordered by proximity to 1.0 for early termination
    factors = np.array([0.9, 1.1, 0.8, 1.2, 0.7, 1.3, 0.5, 1.5, 0.3, 2.0, 0.0])

    consecutive_no_improve = 0

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        sweep_improved = 0
        for i in range(n):
            if deadline is not None and (i % 500 == 0) and time.monotonic() > deadline:
                break

            old_val = a[i]
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            n_aff = k_hi - k_lo + 1

            old_c_seg = c[k_lo:k_hi + 1].copy()
            current_max = float(np.max(c))
            current_score = 2.0 * n * current_max / (S * S)
            best_local_score = current_score
            best_cand = old_val

            # We only need to check factors that differ from 1.0
            for f in factors:
                if old_val > 1e-14:
                    cv = old_val * f
                else:
                    cv = 0.01 if f > 0 else 0.0

                cv = max(0.0, cv)
                delta = cv - old_val
                if abs(delta) < 1e-13 * max(1.0, old_val):
                    continue

                # Compute new c in affected region
                if n_aff > 1:
                    idx = np.arange(k_lo, k_hi + 1) - i
                    new_seg = old_c_seg + 2.0 * delta * a[idx]
                else:
                    new_seg = old_c_seg.copy()

                # Self-term correction at k = 2i
                self_k = 2 * i
                if k_lo <= self_k <= k_hi:
                    new_seg[self_k - k_lo] = old_c_seg[self_k - k_lo] + (cv * cv - old_val * old_val)

                # Compute new max
                new_seg_max = float(np.max(new_seg))
                # Max of unaffected parts
                max_unaff = 0.0
                if k_lo > 0:
                    max_unaff = max(max_unaff, float(c[:k_lo].max()))
                if k_hi < 2 * n - 2:
                    max_unaff = max(max_unaff, float(c[k_hi + 1:].max()))
                new_max = max(max_unaff, new_seg_max)

                new_S = S + delta
                if new_S < 1e-12:
                    continue
                new_score = 2.0 * n * new_max / (new_S * new_S)

                if new_score < best_local_score - 1e-12:
                    best_local_score = new_score
                    best_cand = cv

            if best_cand != old_val:
                delta = best_cand - old_val
                S += delta
                idx = np.arange(k_lo, k_hi + 1) - i
                if n_aff > 1:
                    c[k_lo:k_hi + 1] += 2.0 * delta * a[idx]
                self_k = 2 * i
                if k_lo <= self_k <= k_hi:
                    c[self_k] += delta * delta
                a[i] = best_cand
                sweep_improved += 1

        current_score = 2.0 * n * float(np.max(c)) / (S * S)
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if verbose and sweep % 20 == 0:
            print(f"    CD sweep {sweep}: current={current_score:.10f}, best={best_score:.10f}, improved={sweep_improved}")

        if consecutive_no_improve >= 15:
            if verbose:
                print(f"    CD stalled at sweep {sweep}, best={best_score:.10f}")
            break

    return best_a, best_score


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    exp_sum = exp_s.sum()
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    p = exp_s / exp_sum
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient: dL/da[i] = 2 * (conv(p, rev_a))[n-1+i]
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def pairwise_cd(a, n_sweeps=20, verbose=False, deadline=None):
    """
    Pairwise coordinate descent: change two coordinates simultaneously
    to allow sum-preserving moves.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a, np.inf
    c = fast_conv(a)
    best_score = 2.0 * n * float(np.max(c)) / (S * S)
    best_a = a.copy()

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        sweep_improved = 0
        # Try pairs (i, j) where i < j
        step = max(1, n // 200)
        for i in range(0, n - 1, step):
            if deadline is not None and time.monotonic() > deadline:
                break
            for j in range(i + step, min(i + 10 * step, n), max(1, step // 2)):
                old_vi = a[i]
                old_vj = a[j]
                if old_vi < 1e-14 and old_vj < 1e-14:
                    continue

                # Try transferring mass from one to the other, or scaling both
                for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                    # i gives mass to j
                    transfer = old_vi * frac
                    if transfer < 1e-14:
                        continue
                    new_vi = old_vi - transfer
                    new_vj = old_vj + transfer
                    delta_i = new_vi - old_vi
                    delta_j = new_vj - old_vj

                    # Compute new c (affected by both i and j)
                    c_new = c.copy()
                    # Update for i
                    ki_lo = max(0, i)
                    ki_hi = min(2 * n - 2, i + n - 1)
                    idx_i = np.arange(ki_lo, ki_hi + 1) - i
                    c_new[ki_lo:ki_hi + 1] += 2.0 * delta_i * a[idx_i]
                    if ki_lo <= 2 * i <= ki_hi:
                        c_new[2 * i] += delta_i * delta_i
                    # Update for j
                    kj_lo = max(0, j)
                    kj_hi = min(2 * n - 2, j + n - 1)
                    idx_j = np.arange(kj_lo, kj_hi + 1) - j
                    c_new[kj_lo:kj_hi + 1] += 2.0 * delta_j * a[idx_j]
                    if kj_lo <= 2 * j <= kj_hi:
                        c_new[2 * j] += delta_j * delta_j

                    new_max = float(np.max(c_new))
                    new_score = 2.0 * n * new_max / (S * S)  # S unchanged

                    if new_score < best_score - 1e-12:
                        a[i] = new_vi
                        a[j] = new_vj
                        c = c_new
                        best_score = new_score
                        best_a = a.copy()
                        sweep_improved += 1
                        old_vi = a[i]
                        old_vj = a[j]
                        break

        if verbose and sweep % 5 == 0:
            print(f"    PairCD sweep {sweep}: best={best_score:.10f}, improved={sweep_improved}")

    return best_a, best_score


def mirror_descent_polish(a, n_iters=500, eta0=0.05):
    """Mirror descent polish for final improvements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))

        grad_max = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            grad_max[j_start:j_end + 1] = 2.0 * seg[::-1]

        eta = eta0 / (1.0 + 0.05 * it)
        log_a = np.log(a) - eta * grad_max
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_sc


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            inits.append(np.maximum(parent.astype(np.float64), 0.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.maximum(np.interp(x_new, x_old, parent), 0.0).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.3 + 0.7 * np.abs(2.0 * x - 1.0))
    inits.append(0.3 + 0.7 * (1.0 - (2.0 * x - 1.0) ** 2))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    # Block structure
    for nb in (4, 8, 16, 32):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a_init = np.repeat(vals, per)
        if len(a_init) < n:
            a_init = np.concatenate([a_init, vals[:n - len(a_init)]])
        inits.append(a_init[:n].copy())
    return inits

    total_cd_sweeps = 0
    while time.monotonic() < cd_deadline:
        remaining_sweeps = min(chunk_sweeps, max(1, int((cd_deadline - time.monotonic()) / max(0.1, (n_cd * 2.0) / 5000.0))))
        a_cd, sc_cd = coord_descent(best_a, n_sweeps=remaining_sweeps,
                                    verbose=True, deadline=cd_deadline, rng=rng)
        total_cd_sweeps += remaining_sweeps
        improved = report(sc_cd, a_cd, f"CD total_sweeps={total_cd_sweeps}")
        if not improved:
            # Try a perturbation to escape local optimum
            n_pert = min(50, n_cd // 20)
            indices = rng.choice(n_cd, size=n_pert, replace=False)
            a_pert = best_a.copy()
            a_pert[indices] *= 0.5 + rng.random(n_pert)
            a_pert = np.maximum(a_pert, 0.0)
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S
            a_cd2, sc_cd2 = coord_descent(a_pert, n_sweeps=min(50, remaining_sweeps // 2),
                                          deadline=cd_deadline)
            report(sc_cd2, a_cd2, f"CD restart")

    print(f"After Phase 2: best = {best_score:.10f} (CD sweeps: {total_cd_sweeps})")

    # ==================================================================
    # Phase 3: LSE-LBFGS polishing (~25% time)
    # ==================================================================
    phase3_deadline = t_start + 0.85 * effective_budget
    print(f"\n--- Phase 3: LSE-LBFGS Polish ---")

    n_lbfgs = len(best_a)
    beta_schedule = [100, 300, 1000, 3000, 8000]

    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline:
            break
        if n_lbfgs > 10000:
            maxiter = 2000
            maxfun = 40000
        elif n_lbfgs > 5000:
            maxiter = 5000
            maxfun = 80000
        else:
            maxiter = 10000
            maxfun = 200000

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS beta={beta}")
        n_lbfgs = len(best_a)

    # Try from a slightly perturbed start
    if time.monotonic() < phase3_deadline:
        a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(len(best_a)))
        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S > 1e-12:
            a_pert = a_pert / S
        sc, a_opt = run_lbfgs(a_pert, 2000, maxiter=5000, maxfun=80000)
        report(sc, a_opt, "LBFGS perturbed")

    # Pairwise CD after LSE
    if time.monotonic() < phase3_deadline:
        a_pw, sc_pw = pairwise_cd(best_a, n_sweeps=15, deadline=phase3_deadline)
        report(sc_pw, a_pw, "PairwiseCD")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Final polish with remaining time
    # ==================================================================
    print(f"\n--- Phase 4: Final Polish ---")

    # More coordinate descent
    if time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        n_sweeps_final = int(min(100, remaining / max(0.5, len(best_a) * 2.0 / 5000.0)))
        a_final, sc_final = coord_descent(best_a, n_sweeps=n_sweeps_final,
                                          verbose=True, deadline=deadline)
        report(sc_final, a_final, f"Final CD ({n_sweeps_final} sweeps)")

    # Mirror descent
    if time.monotonic() < deadline - 10:
        a_md, sc_md = mirror_descent_polish(best_a, n_iters=300, eta0=0.03)
        report(sc_md, a_md, "MirrorDesc")

    # One more LSE with very high beta
    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(best_a, 10000, maxiter=3000, maxfun=50000)
        report(sc, a_opt, "LBFGS beta=10000 final")

    # Final short CD
    if time.monotonic() < deadline - 5:
        a_f, sc_f = coord_descent(best_a, n_sweeps=30, deadline=deadline)
        report(sc_f, a_f, "Final short CD")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
