import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_and_grad(a, beta):
    """LSE-smoothed objective and gradient for the exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500, 0))
    p = exp_scaled / exp_scaled.sum()

    score = 2.0 * n * L / (S * S)

    p_rev = p[::-1]
    M2 = 3 * n - 2

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)

def run_lbfgs(a0, beta, maxiter=50000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_obj_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def nesterov_exact(a, n_iters=5000, f_lb=1.0):
    """Nesterov-accelerated subgradient on the exact max objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.5)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)

        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    sc_final = compute_score(best_a)
    if sc_final < best_sc:
        best_sc = sc_final
    return best_a


def subgrad_momentum(a, n_iters=3000, f_lb=1.0):
    """Subgradient with heavy-ball momentum."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    step0 = 0.05
    decay = 0.9995

    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * a[k_star - idx]

        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = step0 * decay ** it
        step = min(step, 2.0 * (sc - f_lb) / (gnorm + 1e-30))

        v = 0.95 * v - step * grad
        a_new = a + v
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa

    return best_a


def mass_transfer_search(a, n_rounds=200, n_transfer=50, verbose=False):
    """
    Novel mass-transfer coordinate descent.
    For each peak, identify top contributors and transfer mass to
    low-impact positions using 1D line search.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        S_a = float(a.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_n = min(n_transfer, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_n]
        bottom_local = np.argsort(contrib)[:-top_n] if len(contrib) > top_n else np.argsort(contrib)[:max(1, len(contrib) // 4)]

        improved = False
        for ti in range(top_n):
            j = j_start + top_local[ti]
            a_old = a[j]
            if a_old < 1e-12:
                continue

            for bi in bottom_local:
                j_dest = j_start + bi
                if j_dest == j:
                    continue

            # Try transferring from j to a randomly chosen bottom position
            if len(bottom_local) == 0:
                continue
            j_dest = j_start + bottom_local[rnd % len(bottom_local)]
            if j_dest == j:
                continue

            # 1D line search: a[j] -= t, a[j_dest] += t
            # Find optimal t in [0, a_old]
            best_t = 0.0
            best_local_sc = sc

            # Bisection on t
            t_lo = 0.0
            t_hi = a_old * 0.5  # Don't transfer more than half

            if t_hi < 1e-15:
                continue

            for _ in range(12):
                t_mid = (t_lo + t_hi) / 2.0
                a[j] = a_old - t_mid
                a[j_dest] = a[j_dest] + t_mid
                S_tmp = a.sum()
                a[j] = (a_old - t_mid) / S_tmp * S_a  # maintain sum
                a[j_dest] = (a[j_dest] + t_mid) / S_tmp * S_a
                c_tmp = fast_conv(a)
                S_tmp2 = a.sum()
                sc_tmp = 2.0 * n * float(c_tmp.max()) / (S_tmp2 * S_tmp2)

                if sc_tmp < best_local_sc:
                    best_local_sc = sc_tmp
                    best_t = t_mid
                    t_hi = t_mid
                else:
                    t_lo = t_mid

                # Restore
                a[j] = a_old
                Sa_restore = a.sum()
                a = a / Sa_restore * S_a

            # Apply best transfer
            if best_t > 1e-15 and best_local_sc < sc:
                a[j] -= best_t
                a[j_dest] += best_t
                S_a2 = a.sum()
                a = a / S_a2
                sc = compute_score(a)
                improved = True

        if not improved:
            # Try reducing peak contributors without transfer (just scale down)
            # and redistribute uniformly
            for ti in range(min(5, top_n)):
                j = j_start + top_local[ti]
                a_old = a[j]
                if a_old < 1e-12:
                    continue
                reduction = a_old * 0.3
                a[j] -= reduction
                S_def = reduction / n
                a += S_def
                a = np.clip(a, 0.0, 1000.0)
                Sc = a.sum()
                if Sc > 0:
                    a = a / Sc
                sc_check = compute_score(a)
                if sc_check > best_sc:
                    a = best_a.copy()

    return best_a


def simulated_annealing(a, rng, n_steps=5000, T0=0.1, T1=0.001):
    """Simulated annealing with structured coordinate moves."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    cur_sc = best_sc

    T = T0
    cooling = (T1 / T0) ** (1.0 / max(1, n_steps))

    for step in range(n_steps):
        T *= cooling

        # Choose move type
        move_type = rng.integers(0, 4)

        if move_type == 0:
            # Single coordinate perturbation
            j = rng.integers(n)
            delta = 0.05 * a[j] * rng.standard_normal()
            a[j] = max(0.0, a[j] + delta)
        elif move_type == 1:
            # Pair transfer
            j1 = rng.integers(n)
            j2 = rng.integers(n)
            if j1 == j2:
                continue
            delta = 0.03 * a[j1]
            a[j1] = max(0.0, a[j1] - delta)
            a[j2] += delta
        elif move_type == 2:
            # Block scaling
            block_size = max(1, n // 100)
            start = rng.integers(0, max(1, n - block_size))
            scale = 1.0 + 0.1 * rng.standard_normal()
            a[start:start + block_size] *= max(0.01, scale)
        else:
            # Peak-targeted: reduce top contributor, increase random
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_j = j_start + int(np.argmax(contrib))
            j_dest = rng.integers(n)
            if j_dest != top_j:
                delta = 0.05 * a[top_j]
                a[top_j] = max(0.0, a[top_j] - delta)
                a[j_dest] += delta

        a = np.clip(a, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            a = best_a.copy()
            continue
        a = a / Sa

        sc_new = compute_score(a)
        delta_sc = sc_new - cur_sc

        if delta_sc < 0 or rng.random() < np.exp(-delta_sc / max(T, 1e-10)):
            cur_sc = sc_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()
        else:
            a = best_a.copy() if rng.random() < 0.01 else a  # occasional reset
            # Revert
            a = np.clip(a, 0.0, 1000.0)
            Sa = a.sum()
            if Sa < 1e-12:
                a = best_a.copy()
            else:
                a = a / Sa

    return best_a


def perturb(a, rng, scale=0.03):
    """Generate a perturbed version of a."""
    n = len(a)
    strategy = rng.integers(0, 6)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 15.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.1 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            n_bot = max(1, len(contrib) - top_count)
            bottom_local = np.argsort(contrib)[:n_bot]
            per = deficit / n_bot
            for li in bottom_local:
                a_p[j_start + li] += per
    elif strategy == 4:
        n_sel = max(1, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))
    else:
        # Gaussian window perturbation
        center = rng.random() * n
        width = n * (0.05 + 0.2 * rng.random())
        a_p = a * (1.0 - scale * 2 * np.exp(-0.5 * ((x * n - center) / width) ** 2))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses mass-transfer descent, high-beta LSE L-BFGS, Nesterov subgradient,
    simulated annealing, and iterated local search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 25.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Initialize from parent
    if parent is not None:
        print(f"Parent length: {len(parent)}")
        parent_norm = parent / parent.sum()
        sc_parent = compute_score(parent_norm)
        print(f"Parent score: {sc_parent:.10f}")
        report(sc_parent, parent_norm, "parent")
        n_work = len(parent)
    else:
        n_work = 8192
        print(f"No parent, using n={n_work}")

    print(f"Working length: {n_work}")
    a_cur = best_a.copy()
    if len(a_cur) != n_work:
        a_cur = np.interp(np.linspace(0, 1, n_work), np.linspace(0, 1, len(a_cur)), a_cur)
        a_cur = np.clip(a_cur, 0.0, 1000.0)
        S = a_cur.sum()
        if S > 0:
            a_cur = a_cur / S

    # ============================================================
    # Phase 1: Nesterov subgradient (10% time)
    # ============================================================
    phase1_end = t_start + 0.10 * (total_budget - 25.0)
    print(f"\n--- Phase 1: Nesterov Subgradient ---")

    a_ns = nesterov_exact(a_cur, n_iters=5000, f_lb=1.0)
    sc = compute_score(a_ns)
    report(sc, a_ns, "nesterov")

    if time.monotonic() < phase1_end:
        a_hb = subgrad_momentum(best_a, n_iters=3000, f_lb=1.0)
        sc = compute_score(a_hb)
        report(sc, a_hb, "heavyball")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ============================================================
    # Phase 2: Mass transfer search (15% time)
    # ============================================================
    phase2_end = t_start + 0.25 * (total_budget - 25.0)
    print(f"\n--- Phase 2: Mass Transfer Search ---")

    a_mt = mass_transfer_search(best_a, n_rounds=100, n_transfer=30)
    sc = compute_score(a_mt)
    report(sc, a_mt, "mass_transfer")

    if time.monotonic() < phase2_end:
        a_mt2 = mass_transfer_search(best_a, n_rounds=50, n_transfer=15)
        sc = compute_score(a_mt2)
        report(sc, a_mt2, "mass_transfer_fine")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ============================================================
    # Phase 3: L-BFGS with increasing beta (40% time)
    # ============================================================
    phase3_end = t_start + 0.65 * (total_budget - 25.0)
    print(f"\n--- Phase 3: L-BFGS Refinement ---")

    beta_schedule = [200, 500, 1000, 5000, 20000, 100000, 500000]

    for beta in beta_schedule:
        if time.monotonic() > phase3_end:
            break
        remaining_s = phase3_end - time.monotonic()
        if remaining_s < 30:
            break
        maxiter = int(min(80000, max(5000, remaining_s * 80)))
        maxfun = int(min(400000, max(20000, remaining_s * 300)))

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS b={beta}")

        # If significant improvement, continue with higher beta
        if time.monotonic() > phase3_end:
            break

    print(f"After Phase 3: best = {best_score:.10f}")

    # ============================================================
    # Phase 4: Iterated local search (20% time)
    # ============================================================
    phase4_end = t_start + 0.85 * (total_budget - 25.0)
    print(f"\n--- Phase 4: Iterated Local Search ---")

    n_trials = 30
    for trial in range(n_trials):
        if time.monotonic() > phase4_end:
            break
        remaining_s = phase4_end - time.monotonic()
        if remaining_s < 60:
            break

        scale = 0.01 + 0.04 * rng.random()
        a_pert = perturb(best_a, rng, scale=scale)

        # Quick L-BFGS on perturbation
        beta = 1000 + int(5000 * rng.random())
        maxiter = int(min(40000, max(5000, remaining_s * 30)))
        maxfun = int(min(100000, max(10000, remaining_s * 100)))
        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"ILS#{trial}")

        # Mass transfer on the result
        if time.monotonic() < phase4_end - 30:
            a_mt = mass_transfer_search(a_opt, n_rounds=30, n_transfer=20)
            sc = compute_score(a_mt)
            report(sc, a_mt, f"ILS#{trial}+MT")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ============================================================
    # Phase 5: Simulated annealing (5% time)
    # ============================================================
    phase5_end = t_start + 0.90 * (total_budget - 25.0)
    print(f"\n--- Phase 5: Simulated Annealing ---")

    if time.monotonic() < phase5_end:
        remaining_s = phase5_end - time.monotonic()
        n_sa = int(min(5000, max(500, remaining_s * 10)))
        a_sa = simulated_annealing(best_a, rng, n_steps=n_sa, T0=0.05, T1=0.001)
        sc = compute_score(a_sa)
        report(sc, a_sa, "sim_anneal")

    # ============================================================
    # Phase 6: Final high-precision refinement (5% time)
    # ============================================================
    print(f"\n--- Phase 6: Final Refinement ---")

    if time.monotonic() < deadline - 30:
        for beta in [1000000, 5000000]:
            if time.monotonic() > deadline - 20:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=50000, maxfun=200000)
            report(sc, a_opt, f"final b={beta}")

    # Final Nesterov pass
    if time.monotonic() < deadline - 20:
        a_ns2 = nesterov_exact(best_a, n_iters=2000, f_lb=1.0)
        sc = compute_score(a_ns2)
        report(sc, a_ns2, "final_nesterov")

    # Final mass transfer
    if time.monotonic() < deadline - 15:
        a_mt_final = mass_transfer_search(best_a, n_rounds=50, n_transfer=20)
        sc = compute_score(a_mt_final)
        report(sc, a_mt_final, "final_MT")

    # Final L-BFGS polish
    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(best_a, 2000000, maxiter=30000, maxfun=100000)
        report(sc, a_opt, "final_polish")

    # ============================================================
    # Try different lengths if time permits
    # ============================================================
    if time.monotonic() < deadline - 60 and parent is not None:
        print(f"\n--- Phase 7: Length Exploration ---")
        for n_try in [len(parent) // 2, int(len(parent) * 1.5), len(parent) * 2]:
            if time.monotonic() > deadline - 30:
                break
            if n_try < 256 or n_try > 30000:
                continue
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                  np.linspace(0, 1, len(best_a)), best_a)
            a_interp = np.clip(a_interp, 0.0, 1000.0)
            S = a_interp.sum()
            if S > 0:
                a_interp = a_interp / S
            sc, a_opt = run_lbfgs(a_interp, 1000, maxiter=20000, maxfun=80000)
            report(sc, a_opt, f"len={n_try}")
            if time.monotonic() < deadline - 30:
                sc2, a_opt2 = run_lbfgs(a_opt, 10000, maxiter=20000, maxfun=60000)
                report(sc2, a_opt2, f"len={n_try} polish")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
