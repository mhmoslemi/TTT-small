"""
Multi-method optimizer for step function peak minimization.
Combines: (1) multi-scale warmup, (2) incremental coordinate descent with
1D subproblem solving, (3) active-set subgradient with multiple peaks,
(4) L-BFGS with adaptive LSE smoothing, (5) iterated local search.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta, top_k=None):
    """LSE-smoothed score with gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_work = c.copy()
    if top_k is not None and top_k < len(c):
        threshold = np.partition(c, len(c) - top_k)[len(c) - top_k]
        c_work = np.minimum(c_work, threshold)

    c_max = float(c_work.max())
    scaled = beta * (c_work - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=100000, top_k=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    if top_k is not None:
        func = lambda x: lse_score_grad(x, beta, top_k)
    else:
        func = lambda x: lse_score_grad(x, beta)

    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coordinate_descent(a, n_sweeps=100, anneal_temp=0.0, rng=None):
    """
    Incremental coordinate descent with 1D subproblem solving.
    Uses the formula t* = S + a_i - c[k*]/a[k*-i] for the active peak.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    c_full = fast_conv(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        order = np.random.permutation(n) if rng is None else rng.permutation(n)
        for i in order:
            ai = a[i]
            if ai < 1e-15:
                # Try to activate this coordinate
                # Find best k for this coordinate
                valid_k = np.arange(max(0, i - n + 1), min(2 * n - 1, i + n - 1) + 1)
                if len(valid_k) == 0:
                    continue
                # c_k with a_i = 0 is c_full_k (since ai=0, contribution is 0)
                k_star = int(np.argmax(c_full[valid_k])) + valid_k[0]
                # t* = S - c_full[k_star] / a[k_star - i] (if valid)
                idx = k_star - i
                if 0 <= idx < n and a[idx] > 1e-15:
                    t_star = S - c_full[k_star] / a[idx]
                    t_star = max(0.0, min(t_star, 1000.0))
                    # Check if this improves
                    a[i] = t_star
                    S_new = S + t_star
                    if S_new > 1e-12:
                        # Quick check: compute new peak
                        # c_new[k] = c_full[k] + 2*t_star*a[k-i]
                        delta = 2.0 * t_star
                        c_new_k = c_full[k_star] + delta * a[idx]
                        S2_new = S_new * S_new
                        sc_est = 2.0 * n * c_new_k / S2_new
                        if sc_est < best_sc * 1.01:  # Allow slight degradation
                            S = S_new
                            # Update c_full incrementally
                            update_k = np.arange(max(0, i - n + 1), min(2 * n - 1, i + n - 1) + 1)
                            c_full[update_k] += delta * a[update_k - i]
                            c_full[2 * i] += t_star * t_star  # diagonal correction
                        else:
                            a[i] = 0.0
                continue

            # Normal case: ai > 0
            # Find the peak that this coordinate most affects
            # The contribution to c_k is 2*ai*a[k-i]
            # We want to find k* = argmax c_full
            k_star = int(np.argmax(c_full))
            idx = k_star - i
            if 0 <= idx < n and a[idx] > 1e-15:
                # t* = S + ai - c_full[k_star] / a[idx]
                t_star = S + ai - c_full[k_star] / a[idx]
            else:
                # Fallback: use current peak contribution
                t_star = ai * 0.5

            t_star = max(0.0, min(t_star, 1000.0))

            # Annealing: sometimes try a random value
            if anneal_temp > 0 and rng is not None:
                if rng.random() < 0.05:
                    t_star = max(0.0, ai + anneal_temp * rng.standard_normal())

            if abs(t_star - ai) < 1e-15:
                continue

            # Update: a[i] from ai to t_star, delta = t_star - ai
            delta = t_star - ai
            S_new = S + delta

            # Check if this helps (estimate new score)
            if delta > 0:
                # Convolution increases at positions k where a[k-i] > 0
                # New max is at most c_full[k_star] + 2*delta*a[idx]
                c_est = c_full[k_star] + 2.0 * delta * a[idx]
            else:
                # Convolution decreases
                c_est = c_full[k_star] + 2.0 * delta * a[idx]
                if c_est < 0:
                    c_est = 0.0

            S2_new = S_new * S_new if S_new > 1e-12 else 1e12
            sc_est = 2.0 * n * max(c_est, 1e-15) / S2_new

            # Accept if score improves or with small probability (annealing)
            accept = sc_est < best_sc * (1.0 + anneal_temp * 0.1)
            if accept or (anneal_temp > 0 and rng is not None and rng.random() < anneal_temp * 0.05):
                a[i] = t_star
                S = S_new
                # Update c_full incrementally
                k_min = max(0, i - n + 1)
                k_max = min(2 * n - 2, i + n - 1)
                if k_max >= k_min:
                    ks = np.arange(k_min, k_max + 1)
                    c_full[ks] += 2.0 * delta * a[ks - i]
                if 2 * i < 2 * n - 1:
                    c_full[2 * i] += delta * (ai + t_star)

            # Track best
            if sweep % 10 == 0:
                sc_actual = compute_score(a)
                if sc_actual < best_sc:
                    best_sc = sc_actual
                    best_a = a.copy()

    # Final check
    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def active_set_subgrad(a, n_iters=5000, f_lb=1.0, n_active=5):
    """
    Subgradient method using multiple active peaks.
    The subgradient is a weighted combination of gradients from the top peaks.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        if Sy < 1e-12:
            break
        Mak = float(np.max(c))
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        # Find top active peaks
        top_idx = np.argsort(c)[::-1][:n_active]
        # Weight by proximity to max
        weights = np.maximum(0, c[top_idx] - Mak * 0.9)
        if weights.sum() < 1e-15:
            weights = np.ones(n_active) / n_active
        else:
            weights = weights / weights.sum()

        # Compute combined subgradient
        sg = np.zeros(n)
        for w, k in zip(weights, top_idx):
            j_start = max(0, int(k) - n + 1)
            j_end = min(int(k), n - 1)
            if j_end >= j_start:
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += w * 2.0 * y[int(k) - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.5)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        theta = (t_mom - 1.0) / t_new
        t_mom = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)

        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def build_multiscale_init(parent, target_n, rng):
    """Build a good initialization via multi-scale optimization."""
    if parent is None:
        a = np.ones(64) * 0.5
    else:
        a = parent.copy()

    scales = []
    # Build up from small to target_n
    current_n = len(a)
    if current_n < target_n:
        # Plan scales
        s = current_n
        while s < target_n:
            s2 = min(s * 2, target_n)
            if s2 == s:
                s2 = target_n
            scales.append(s2)
            s = s2
        if not scales or scales[-1] != target_n:
            scales.append(target_n)

        for target_n_scale in scales:
            # Upsample
            x_old = np.linspace(0, 1, len(a))
            x_new = np.linspace(0, 1, target_n_scale)
            a = np.interp(x_new, x_old, a)
            a = np.clip(a, 0.0, 1000.0)
            S = a.sum()
            if S > 0:
                a = a / S

            # Quick L-BFGS refinement
            if target_n_scale <= 4096:
                sc, a_opt = run_lbfgs(a, beta=200, maxiter=2000, maxfun=10000)
                if sc < compute_score(a):
                    a = a_opt

    return a


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version."""
    n = len(a)
    strategy = int(rng.integers(0, 5))
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.15 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bottom_local) if len(bottom_local) > 0 else 0
            for li in bottom_local:
                a_p[j_start + li] += per
    else:
        n_sel = max(1, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/(sum)^2.
    Uses multi-scale init, coordinate descent, active-set subgradient,
    and L-BFGS polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
        print(f"Parent length: {n_work}, score: {compute_score(parent/parent.sum()):.10f}")
        parent_norm = parent / parent.sum()
        report(compute_score(parent_norm), parent_norm, "parent")
    else:
        n_work = 8192

    print(f"Working length: {n_work}")

    # ==================================================================
    # Phase 0: Multi-scale warmup (5% time)
    # ==================================================================
    phase0_dl = t_start + 0.05 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 0: Multi-scale Warmup ---")

    if parent is not None:
        a_cur = parent / parent.sum()
        a_cur = a_cur[:n_work] if len(a_cur) >= n_work else np.interp(
            np.linspace(0, 1, n_work), np.linspace(0, 1, len(a_cur)), a_cur)
    else:
        a_cur = np.ones(n_work) * 0.5

    a_cur = np.clip(a_cur, 0, 1000.0)
    S = a_cur.sum()
    if S > 0:
        a_cur = a_cur / S

    # Quick L-BFGS at low beta
    sc, a_opt = run_lbfgs(a_cur, beta=50, maxiter=3000, maxfun=15000)
    report(sc, a_opt, "warmup_lbfgs")

    if time.monotonic() < phase0_dl:
        sc, a_opt = run_lbfgs(a_cur, beta=200, maxiter=3000, maxfun=15000)
        report(sc, a_opt, "warmup_lbfgs2")

    # ==================================================================
    # Phase 1: Coordinate Descent (35% time)
    # ==================================================================
    phase1_dl = t_start + 0.40 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 1: Coordinate Descent (n={n_work}) ---")

    iter_count = 0
    stall_count = 0
    prev_best = best_score

    while time.monotonic() < phase1_dl:
        # Adaptive annealing temperature
        frac_done = (time.monotonic() - t_start - (phase0_dl - t_start)) / max(
            phase1_dl - phase0_dl, 1)
        anneal = 0.05 * (1.0 - frac_done)

        n_sweeps = 50
        a_cd, sc_cd = coordinate_descent(best_a, n_sweeps=n_sweeps,
                                         anneal_temp=anneal, rng=rng)
        iter_count += 1
        report(sc_cd, a_cd, f"CD #{iter_count}")

        # Check for stall
        if best_score < prev_best * (1 - 1e-5):
            stall_count = 0
            prev_best = best_score
        else:
            stall_count += 1

        # If stalled, try a perturbation
        if stall_count >= 3:
            a_pert = perturbed_restart(best_a, rng, scale=0.05)
            sc_p, a_p_opt = run_lbfgs(a_pert, beta=500, maxiter=5000, maxfun=20000)
            report(sc_p, a_p_opt, f"CD restart #{stall_count}")
            stall_count = 0
            prev_best = best_score

    print(f"After Phase 1: best = {best_score:.10f}")

    # ==================================================================
    # Phase 2: Active-set subgradient (20% time)
    # ==================================================================
    phase2_dl = t_start + 0.60 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 2: Active-Set Subgradient ---")

    # Multiple active-set runs with different parameters
    for n_act in [3, 5, 10]:
        if time.monotonic() > phase2_dl:
            break
        a_as = active_set_subgrad(best_a, n_iters=3000, f_lb=1.0, n_active=n_act)
        sc = compute_score(a_as)
        report(sc, a_as, f"active_set n={n_act}")

    # Follow up with coordinate descent
    if time.monotonic() < phase2_dl:
        a_cd, sc_cd = coordinate_descent(best_a, n_sweeps=50, anneal_temp=0.02, rng=rng)
        report(sc_cd, a_cd, "CD after AS")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ==================================================================
    # Phase 3: L-BFGS refinement with increasing beta (25% time)
    # ==================================================================
    phase3_dl = t_start + 0.85 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 3: L-BFGS Refinement ---")

    beta_schedule = [200, 500, 1000, 3000, 10000, 30000, 100000]

    for beta in beta_schedule:
        if time.monotonic() > phase3_dl:
            break
        remaining_s = phase3_dl - time.monotonic()
        if remaining_s < 20:
            break
        maxiter = int(min(30000, max(3000, remaining_s * 30)))
        maxfun = int(min(100000, max(10000, remaining_s * 100)))

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS beta={beta}")

        # Top-k variant
        if time.monotonic() < phase3_dl - 30:
            sc_k, a_k = run_lbfgs(best_a, beta, maxiter=maxiter // 2,
                                  maxfun=maxfun // 2, top_k=5)
            report(sc_k, a_k, f"LBFGS beta={beta} topk=5")

        # Coordinate descent between L-BFGS runs
        if time.monotonic() < phase3_dl - 20:
            a_cd, sc_cd = coordinate_descent(best_a, n_sweeps=30,
                                             anneal_temp=0.01, rng=rng)
            report(sc_cd, a_cd, f"CD after beta={beta}")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Iterated Local Search (10% time)
    # ==================================================================
    phase4_dl = deadline
    print(f"\n--- Phase 4: Iterated Local Search ---")

    n_trials = 10
    for trial in range(n_trials):
        if time.monotonic() > phase4_dl - 30:
            break
        remaining = phase4_dl - time.monotonic()
        if remaining < 30:
            break

        scale = 0.02 + 0.04 * rng.random()
        a_pert = perturbed_restart(best_a, rng, scale=scale)

        # Optimize perturbation
        beta = 500 + int(2000 * rng.random())
        maxiter = int(min(20000, max(2000, remaining * 15)))
        maxfun = int(min(50000, max(5000, remaining * 50)))

        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxfun)
        improved = report(sc, a_opt, f"ILS #{trial}")

        # Coordinate descent on the result
        if time.monotonic() < phase4_dl - 20:
            a_cd, sc_cd = coordinate_descent(
                best_a if not improved else a_opt,
                n_sweeps=20, anneal_temp=0.01, rng=rng)
            report(sc_cd, a_cd, f"ILS #{trial} CD")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ==================================================================
    # Final polish
    # ==================================================================
    if time.monotonic() < deadline - 15:
        print(f"\n--- Final Polish ---")
        for beta in [50000, 200000]:
            if time.monotonic() > deadline - 10:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=20000, maxfun=80000)
            report(sc, a_opt, f"final beta={beta}")

    # One more coordinate descent pass
    if time.monotonic() < deadline - 10:
        a_cd, sc_cd = coordinate_descent(best_a, n_sweeps=30,
                                         anneal_temp=0.0, rng=rng)
        report(sc_cd, a_cd, "final CD")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
