"""
Multi-strategy optimizer for flat-autocorrelation non-negative sequences.
Combines frequency-domain flattening, progressive LSE L-BFGS, Adam optimization,
and targeted peak shaving to minimize 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def lse_objective_wrapper(a, beta):
    """Wrapper for scipy.minimize (returns value and gradient)."""
    return lse_score_and_grad(a, beta)


def freq_flatten(a, n_iters=200, alpha=0.5):
    """Frequency-domain flattening of the self-convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    N = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        power = np.abs(A) ** 2
        conv = np.fft.irfft(A * A, N)[:2 * n - 1]
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-14:
            break

        # Compute target magnitude spectrum
        # We want |A_new|^2 such that IFFT(|A_new|^2) is flatter
        scale = (conv_mean / (np.abs(conv) + 1e-15)) ** alpha
        # Apply in frequency domain
        A_new = A * np.sqrt(scale)[np.r_[0, (np.arange(1, N // 2 + 1)) % (N // 2)]]
        # Simpler: just scale by sqrt of ratio
        mag_ratio = np.sqrt(conv_mean / (power + 1e-15)) ** alpha
        A_new = A * mag_ratio

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        A = np.fft.rfft(a_new, N)
        # Mix back to avoid getting stuck
        A = 0.7 * A + 0.3 * np.fft.rfft(best_a, N)

    return best_a


def adam_optimize(a0, beta_schedule, steps_per_phase=500, lr=0.005, momentum1=0.9, momentum2=0.999):
    """Adam optimization with progressive LSE beta."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_sc = compute_score(a)

    for beta in beta_schedule:
        for _ in range(steps_per_phase):
            t += 1
            sc, grad = lse_score_and_grad(a, beta)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

            # Project gradient onto sum-preserving subspace
            grad_centered = grad - np.mean(grad)

            m = momentum1 * m + (1 - momentum1) * grad_centered
            v = momentum2 * v + (1 - momentum2) * grad_centered ** 2
            m_hat = m / (1 - momentum1 ** t)
            v_hat = v / (1 - momentum2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            a = np.maximum(a, 0.0)

            # Re-normalize to sum=1
            S = float(a.sum())
            if S < 1e-12:
                a = np.ones(n) / n
            else:
                a = a / S

    return best_a, best_sc


def lbfgs_polish(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B polish on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = float(a0.sum())
    if S < 1e-12:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = float(a_out.sum())
    if S > 1e-12:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_shave(a, n_iters=500, lr=0.01):
    """Targeted peak shaving: reduce elements contributing most to the max convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = self_conv(a)
        c_max = float(c.max())
        # Find top-k peaks
        k_threshold = c_max - 0.005 * (c_max - c.min() + 1e-15)
        peak_mask = c > k_threshold

        # Compute responsibility of each a[j] for peaks
        # a[j] contributes to c[k] via a[j]*a[k-j]
        resp = np.zeros(n)
        for k in range(n - 1, 2 * n - 1):
            if peak_mask[k]:
                w = c[k] - k_threshold
                j_start = max(0, k - n + 1)
                j_end = min(k, n - 1)
                # c[k] = sum_{j=j_start}^{j_end} a[j]*a[k-j]
                # dc[k]/da[j] = a[k-j]
                resp[j_start:j_end + 1] += w * a[k - j_end:j_start - 1 if j_start > 0 else None][::-1] if j_start > 0 else w * a[k - j_end:-1 if k < n else None]

        # Simpler: use the gradient of max
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        contrib = a[j_end:j_start - 1 if j_start > 0 else None][::-1]
        if j_start == 0:
            contrib = a[k_star - j_end:-1 if k_star < n - 1 else None]
            if contrib is None or len(contrib) == 0:
                contrib = a[0:k_star + 1][::-1]
        # Actually let me just compute it directly
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[k_star - j]

        # Reduce high-contribution elements, increase low-contribution
        mean_contrib = float(contrib.mean())
        a_new = a.copy()
        for j in range(n):
            if contrib[j] > mean_contrib * 1.5:
                a_new[j] *= (1.0 - lr * 0.1)
            elif contrib[j] < mean_contrib * 0.5:
                a_new[j] *= (1.0 + lr * 0.1)

        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

    return best_a


def subgradient_steps(a, n_iters=2000):
    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = self_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            for j in range(j_start, j_end + 1):
                sg[j] = a[k_star - j]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        # Subtract mean to preserve sum
        grad = grad - np.mean(grad)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = 0.5 * (sc - 1.0) / (gn * gn + 1e-10)
        step = min(step, 0.02)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

    return best_a


def build_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.5 + 0.3 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(6.0 * np.pi * x))
    # Triangle wave
    inits.append(0.5 + 0.3 * (2.0 * np.abs(2.0 * x - np.floor(2.0 * x + 0.5)) - 1.0))
    # Exponential decay from center
    center = n / 2
    inits.append(0.3 + 0.5 * np.exp(-((x - 0.5) ** 2) / 0.05))

    for init in inits:
        init = np.maximum(init, 0.0)
        S = float(init.sum())
        if S > 1e-12:
            init = init / S
    return inits
