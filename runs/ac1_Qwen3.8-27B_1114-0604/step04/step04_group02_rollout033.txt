def solve_lp_primal_from_dual(f, peak_indices, M):
    """Solve the primal LP using the dual solution."""
    n = len(f)
    m = len(peak_indices)
    
    # Build A matrix (m x n): A[i,j] = f(peak_indices[i] - j) if valid
    A = np.zeros((m, n))
    for i, k in enumerate(peak_indices):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[i, j] = f[idx]
    
    b = np.full(m, M)
    
    # Solve dual: min M * sum(y) s.t. A^T y >= 1, y >= 0
    c_dual = M * np.ones(m)
    A_ub = -A.T  # shape (n, m)
    b_ub = -np.ones(n)
    
    res = linprog(c_dual, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*m, method='highs')
    
    if not res.success:
        return None
    
    y = res.x
    optimal = res.fun  # = M * sum(y) = max sum(g)
    
    # Find active dual constraints: (A^T y)_j ≈ 1
    At_y = A.T @ y  # length n
    tol = 1e-8
    active_J = np.where(np.abs(At_y - 1.0) < tol)[0]
    
    # Find active primal constraints: y_i > 0
    active_I = np.where(y > 1e-10)[0]
    
    if len(active_J) == 0 or len(active_I) == 0:
        return None
    
    # Solve A[active_I, active_J] @ g[active_J] = b[active_I]
    if len(active_I) != len(active_J):
        # Degenerate case - use least squares
        sub_A = A[np.ix_(active_I, active_J)]
        sub_b = b[active_I]
        g_J, _, _, _ = np.linalg.lstsq(sub_A, sub_b, rcond=None)
        g_J = np.maximum(g_J, 0)
    else:
        sub_A = A[np.ix_(active_I, active_J)]
        sub_b = b[active_I]
        g_J = np.linalg.solve(sub_A, sub_b)
        g_J = np.maximum(g_J, 0)
    
    g = np.zeros(n)
    g[active_J] = g_J
    
    # Verify: sum(g) should ≈ optimal
    # Normalize if needed
    S = g.sum()
    if S > 1e-12:
        g = g * (optimal / S)  # scale to match optimal value

LP Active Set + Adaptive LSE optimizer for step function peak minimization.
Novel approach: uses the LP-based direction finding from the problem statement,
combined with adaptive LSE smoothing and iterated local search.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta, top_k):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_work = c.copy()
    if top_k is not None and top_k < len(c_work):
        idx = np.argpartition(c_work, -top_k)[-top_k:]
        threshold = c_work[idx].min()
        c_work = np.minimum(c_work, threshold)
    
    c_max = float(c_work.max())
    scaled = beta * (c_work - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def build_conv_matrix(f, peak_indices):
    n = len(f)
    m = len(peak_indices)
    A = np.zeros((m, n))
    for i in range(m):
        k = int(peak_indices[i])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            j_range = np.arange(j_start, j_end + 1)
            idx = k - j_range
            A[i, j_range] = f[idx]
    return A


def solve_lp_direction(f, peak_indices, M):
    n = len(f)
    m = len(peak_indices)
    A = build_conv_matrix(f, peak_indices)
    b = np.full(m, M)
    c_obj = -np.ones(n)
    
    # Use highs solver
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                  method='highs', options={'time_limit': 30})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def lp_active_set_iteration(f, K_size=50, max_cutting=20, blend_factor=0.1):
    """One LP active set step with cutting plane."""
    n = len(f)
    c = fast_conv(f)
    M = float(c.max())
    S_f = float(f.sum())
    
    # Start with top K peaks
    peak_indices = np.argsort(c)[::-1][:K_size]
    
    for cut_iter in range(max_cutting):
        g = solve_lp_direction(f, peak_indices, M)
        if g is None:
            return f
        
        # Check for violated constraints
        fg = np.convolve(f, g)  # or use fast_conv-style
        # Actually need f * g (cross-correlation, not auto-conv)
        # (f*g)(k) = sum_j f(k-j) * g(j)
        # This is just regular convolution
        fg = np.convolve(f, g, mode='full')
        
        violations = fg > M + 1e-10
        if not np.any(violations):
            break
        
        # Add violated peaks
        violated_idx = np.where(violations)[0]
        # Add the most violated
        v_vals = fg[violations]
        order = violated_idx[np.argsort(v_vals)[::-1][:K_size]]
        peak_indices = np.unique(np.concatenate([peak_indices, order]))
        peak_indices = peak_indices[np.argsort(fg[peak_indices])[::-1][:K_size*2]]
    
    # Rescale g to match sum of f
    S_g = g.sum()
    if S_g < 1e-12:
        return f
    g = g * (S_f / S_g)
    
    # Blend
    f_new = (1.0 - blend_factor) * f + blend_factor * g
    f_new = np.maximum(f_new, 0.0)
    S_new = f_new.sum()
    if S_new > 1e-12:
        f_new = f_new / S_new
    return f_new

def nesterov_subgrad(a, n_iters=3000, f_lb=1.0):
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]
        
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    
    return best_a


def perturb_sequence(a, rng, scale):
    n = len(a)
    strategy = rng.integers(0, 4)
    x = np.arange(n, dtype=np.float64) / n
    
    if strategy == 0:

    elif strategy == 2:

            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    else:
        c = fast_conv(a)

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.1 * rng.random())

    S = a_p.sum()
    if S > 1e-12:

"""
LP Active Set + Adaptive LSE optimizer for step function peak minimization.
Novel approach: Uses cutting-plane LP-based direction finding (from the problem's
theoretical framework) combined with adaptive LSE-smoothed L-BFGS refinement,
multi-peak subgradient methods, and iterated local search with diverse perturbations.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from functools import partial
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution a * b via FFT. Returns array of length len(a)+len(b)-1."""
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta, top_k):
    """LSE-smoothed score and gradient with optional top-k restriction."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_work = c.copy()
    if top_k is not None and 0 < top_k < len(c_work):
        idx = np.argpartition(c_work, -top_k)[-top_k:]
        threshold = float(c_work[idx].min())
        c_work = np.minimum(c_work, threshold)

    c_max = float(c_work.max())
    scaled = beta * (c_work - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapper(x, beta, top_k):
    """Wrapper for minimize: takes x and returns (f, grad)."""
    return lse_score_and_grad(x, beta, top_k)


def build_conv_matrix(f, peak_indices):
    """Build constraint matrix A where A[i,j] = f(peak_indices[i] - j) if valid."""
    n = len(f)
    m = len(peak_indices)
    A = np.zeros((m, n))
    for i in range(m):
        k = int(peak_indices[i])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            j_range = np.arange(j_start, j_end + 1)
            idx = k - j_range
            A[i, j_range] = f[idx]
    return A


def solve_lp_direction(f, peak_indices, M, time_limit=30):
    """Solve the LP: max sum(g) s.t. (f*g)(k) <= M for k in peak_indices, g >= 0."""
    n = len(f)
    m = len(peak_indices)
    A = build_conv_matrix(f, peak_indices)
    b = np.full(m, M)
    c_obj = -np.ones(n)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0), float(-res.fun)
    except Exception:
        pass
    return None, 0.0


def lp_active_set_step(f, K_size, max_cutting, blend_factor, time_limit=30):
    """One LP active set iteration with cutting plane. Returns (new_f, improved)."""
    n = len(f)
    c = fast_conv(f)
    M = float(c.max())
    S_f = float(f.sum())
    if S_f < 1e-12:
        return f, False

    peak_indices = np.argsort(c)[::-1][:K_size]

    g, lp_val = solve_lp_direction(f, peak_indices, M, time_limit)
    if g is None:
        return f, False

    # Cutting plane: check for violated constraints
    for cut_iter in range(max_cutting):
        fg = cross_conv(f, g)
        violations = fg > M + 1e-10 * M
        if not np.any(violations):
            break
        violated_idx = np.where(violations)[0]
        v_vals = fg[violations]
        order = violated_idx[np.argsort(v_vals)[::-1][:K_size]]
        peak_indices = np.unique(np.concatenate([peak_indices, order]))
        # Keep only top 2*K_size most important
        fg_peaks = fg[peak_indices]
        keep = peak_indices[np.argsort(fg_peaks)[::-1][:2 * K_size]]
        peak_indices = keep

        g, lp_val = solve_lp_direction(f, peak_indices, M, time_limit)
        if g is None:
            break

    S_g = float(g.sum())
    if S_g < 1e-12:
        return f, False

    # Rescale g to match sum of f
    g = g * (S_f / S_g)

    # Blend with line search on t
    best_f = f.copy()
    best_sc = compute_score(f)
    for t in [blend_factor, blend_factor * 2, blend_factor * 0.5, blend_factor * 4]:
        if t > 1.0:
            continue
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        S_new = f_new.sum()
        if S_new < 1e-12:
            continue
        f_new = f_new / S_new
        sc = compute_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new

    improved = best_sc < compute_score(f) - 1e-12
    return best_f, improved


def nesterov_subgrad(a, n_iters, f_lb):
    """Nesterov-accelerated subgradient descent on the exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def multi_peak_subgrad(a, n_iters, n_peaks, f_lb):
    """Subgradient using average of top n_peaks subgradients."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.05

    for it in range(n_iters):
        c = fast_conv(a)
        Sa = float(a.sum())
        top_idx = np.argsort(c)[::-1][:n_peaks]
        Mak = float(c.max())
        sc = 2.0 * n * Mak / (Sa * Sa)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        S2 = Sa * Sa
        S3 = S2 * Sa
        grad = np.zeros(n)
        for k_star in top_idx:
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                idx = np.arange(j_start, j_end + 1)
                grad[idx] += 2.0 * a[k_star - idx]
        grad /= len(top_idx)
        grad = (2.0 * n) * (grad / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = min(step * 0.9995, (sc - f_lb) / (gnorm + 1e-30))
        a = np.clip(a - step * grad, 0.0, 1000.0)
        Sa2 = a.sum()
        if Sa2 < 1e-12:
            break
        a = a / Sa2

    return best_a


def run_lbfgs(a0, beta, maxiter, maxfun, top_k):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5

    obj = partial(lse_wrapper, beta=beta, top_k=top_k)
    bounds = [(0.0, 1000.0)] * n

    try:
        res = minimize(
            obj, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-20, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0

    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def perturb_sequence(a, rng, scale):
    """Generate a perturbed version of a."""
    n = len(a)
    strategy = rng.integers(0, 5)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        a_p = a * (1.0 + scale * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        a_p = a.copy()
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)

        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.1 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S

            per = deficit / len(bottom_local)

    else:
        n_sel = max(1, n // 6)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)

def coordinate_peak_shave(a, n_sweeps, n_top):
    """Targeted coordinate descent on peak contributors."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            best_val = a_old_j
            best_local_sc = best_sc
            for frac in [0.0, 0.3, 0.5, 0.7, 0.85]:
                new_val = a_old_j * frac
                a[j] = new_val
                reduction = a_old_j - new_val
                if reduction > 1e-15:
                    bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
                    per = reduction / len(bottom_local)
                    for li in bottom_local:
                        a[j_start + li] += per
                S2 = a.sum()
                if S2 > 1e-12:
                    a = a / S2
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = new_val
            a[j] = best_val
            S3 = a.sum()
            if S3 > 1e-12:
                a = a / S3

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

            blend = min(0.25, blend * 1.05)
        else:
            no_improve_count += 1
            if no_improve_count >= 10:
                # Reduce blend and reset
                blend = max(0.05, blend * 0.5)
                no_improve_count = 0

        lp_iter += 1
        if lp_iter % 50 == 0:
            print(f"  LP iter {lp_iter}: score={best_score:.10f}, K={K}, blend={blend:.3f}, t={time.monotonic()-t_start:.0f}s")

    print(f"After Phase 2: best = {best_score:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # ============================================================
    # Phase 3: Multi-peak subgradient + LSE refinement (25% time)
    # ============================================================
    phase3_deadline = t_start + 0.75 * (total_budget - 30.0)
    print(f"\n--- Phase 3: Multi-peak SG + LSE Refinement ---")

    # Multi-peak subgradient
    for n_peaks in [2, 3, 5]:
        if time.monotonic() > phase3_deadline:
            break
        a_mp = multi_peak_subgrad(best_a, n_iters=2000, n_peaks=n_peaks, f_lb=1.0)
        sc = compute_score(a_mp)
        report(sc, a_mp, f"multi-peak SG n={n_peaks}")

    # LSE refinement with adaptive beta
    beta_schedule = [500, 2000, 5000, 10000, 20000, 50000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline - 60:
            break
        remaining = phase3_deadline - time.monotonic()
        maxiter = int(min(30000, max(5000, remaining * 30)))
        maxfun = int(min(150000, max(10000, remaining * 150)))

        sc, a_opt = run_lbfgs(best_a, beta=beta, maxiter=maxiter,
                              maxfun=maxfun, top_k=None)
        report(sc, a_opt, f"LSE beta={beta}")

        # Also try top-k variants
        if time.monotonic() < phase3_deadline - 30:
            for top_k in [3, 5]:
                sc_k, a_k = run_lbfgs(best_a, beta=beta, maxiter=maxiter // 2,
                                      maxfun=maxfun // 2, top_k=top_k)
                report(sc_k, a_k, f"LSE beta={beta} topk={top_k}")

    print(f"After Phase 3: best = {best_score:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # ============================================================
    # Phase 4: Iterated Local Search (15% time)
    # ============================================================
    phase4_deadline = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n--- Phase 4: Iterated Local Search ---")

    n_trials = 30
    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        remaining = phase4_deadline - time.monotonic()
        if remaining < 30:
            break

        scale = 0.02 + 0.04 * rng.random()
        a_pert = perturb_sequence(best_a, rng, scale=scale)

        # Quick LP repair on the perturbation
        a_rep, _ = lp_active_set_step(a_pert, K_size=30, max_cutting=5,
                                       blend_factor=0.1, time_limit=10)

        # LSE polish
        beta = 2000 + int(5000 * rng.random())
        maxiter = int(min(20000, max(3000, remaining * 15)))
        maxfun = int(min(80000, max(5000, remaining * 60)))
        sc, a_opt = run_lbfgs(a_rep, beta=beta, maxiter=maxiter,
                              maxfun=maxfun, top_k=None)
        report(sc, a_opt, f"ILS #{trial} b={beta}")

        # Small perturbation variant
        if time.monotonic() < phase4_deadline - 20:
            a_pert2 = perturb_sequence(best_a, rng, scale=scale * 0.3)
            sc2, a_opt2 = run_lbfgs(a_pert2, beta=10000,
                                    maxiter=maxiter // 2, maxfun=maxfun // 2,
                                    top_k=None)
            report(sc2, a_opt2, f"ILS #{trial} small")

    print(f"After Phase 4: best = {best_score:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # ============================================================
    # Phase 5: Final Polish (10% time)
    # ============================================================
    print(f"\n--- Phase 5: Final Polish ---")

    # Coordinate peak shaving
    a_shave = coordinate_peak_shave(best_a, n_sweeps=5, n_top=50)
    sc = compute_score(a_shave)
    report(sc, a_shave, "peak shave")

    # Very high beta LSE
    if time.monotonic() < deadline - 30:
        for beta in [50000, 100000, 200000]:
            if time.monotonic() > deadline - 15:
                break
            sc, a_opt = run_lbfgs(best_a, beta=beta, maxiter=20000,
                                  maxfun=100000, top_k=None)
            report(sc, a_opt, f"polish b={beta}")

    # Final Nesterov pass
    if time.monotonic() < deadline - 15:
        a_ns = nesterov_subgrad(best_a, n_iters=2000, f_lb=1.0)
        sc = compute_score(a_ns)
        report(sc, a_ns, "final nesterov")

    # Final multi-peak
    if time.monotonic() < deadline - 10:
        for nh in [2, 3]:
            a_mp = multi_peak_subgrad(best_a, n_iters=1000, n_peaks=nh, f_lb=1.0)
            sc = compute_score(a_mp)
            report(sc, a_mp, f"final mp n={nh}")

    # One more LP pass with high K
    if time.monotonic() < deadline - 10:
        a_lp, _ = lp_active_set_step(best_a, K_size=100, max_cutting=15,
                                      blend_factor=0.05, time_limit=15)
        sc = compute_score(a_lp)
        report(sc, a_lp, "final LP K=100")

    # ============================================================
    # Try a slightly different length if time permits
    # ============================================================
    if time.monotonic() < deadline - 60:
        print(f"\n--- Phase 6: Length Exploration ---")
        for n_try in [int(n_work * 1.2), int(n_work * 0.8)]:
            if time.monotonic() > deadline - 30:
                break
            if n_try < 512 or n_try > 20000:
                continue
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                 np.linspace(0, 1, len(best_a)), best_a)
            a_interp = np.clip(a_interp, 0.0, 1000.0)
            S = a_interp.sum()
            if S > 0:
                a_interp = a_interp / S
            sc, a_opt = run_lbfgs(a_interp, beta=500, maxiter=10000,
                                  maxfun=50000, top_k=None)
            report(sc, a_opt, f"len={n_try}")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
