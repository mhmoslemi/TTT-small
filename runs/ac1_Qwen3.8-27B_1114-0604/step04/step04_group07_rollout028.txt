Wait, I need to double-check the gradient formula. The score is:
`score = 2n * L / S^2` where `S = sum(a)` and `L = LSE_smoothed_max_conv`

`d(score)/da[j] = 2n * (dL/da[j] * S^2 - L * 2*S * da-sum/da[j]) / S^4`
`= 2n * (dL/da[j] * S^2 - 2*L*S*1) / S^4`
`= 2n * dL/da[j] / S^2 - 4n * L / S^3`

But wait, `dL/da[j]` already accounts for the full derivative of L with respect to a[j]. And `dS/da[j] = 1`.

So: `d(score)/da[j] = 2n * dL/da[j] / S^2 - 4n * L / S^3`

This is what I have. Good.

But for the mirror descent method, I'm normalizing to `S=1` at each step, so effectively I'm minimizing `L` subject to `sum(a) = 1`. In that case, the gradient I need is just `dL/da[j]` (the Lagrangian multiplier for the sum constraint is handled by the normalization).

Actually, in mirror descent on the simplex, we minimize the smooth approximation `L(a)` over the simplex. The update is:
`a_new = argmin_{x in simplex} <grad_L, x> + (1/eta) * KL(x || a)`

The solution is `x[j] = a[j] * exp(-eta * grad_L[j]) / Z`.

So I should use `grad_L` (not `grad_score`) for the mirror descent update.

Let me simplify: since we normalize to sum=1 at each step, the effective objective is `2n * max(c)` (or its LSE approximation `2n * L`). The factor `2n` is constant, so we can just minimize `L`. The gradient is `grad_L = term1 + term2`.

- Too small eta: slow progress
- Too large eta: oscillation or convergence to a boundary point (all mass on one element)

A good adaptive strategy: start with a moderate eta, and if the score doesn't improve, reduce eta. If it improves consistently, we can keep it or slightly increase.

"""
Mirror descent (multiplicative weights) optimizer for minimizing
2n * max(self-conv) / (sum)^2 over non-negative sequences.
Uses LSE smoothing with continuation, multi-length exploration,
and L-BFGS polishing.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_grad_L(a, beta):
    """
    Compute LSE-smoothed max-conv value and its gradient w.r.t. a.
    Returns (L, grad_L) where L = LSE_approx(max(c)) and grad_L = dL/da.
    """
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    Z = exp_scaled.sum()
    p = exp_scaled / Z
    L = c_max + np.log(Z) / beta
    
    # grad_L[j] = sum_k p[k]*(a[k-j] + a[j-k])
    # = term1[j] + term2[j]
    # term1[j] = sum_i a[i]*p[i+j] = (a_rev * p)[n-1+j]
    # term2[j] = sum_i a[i]*p[j-i] = (a * p_rev)[n-1+j]
    
    M = 3 * n - 2
    M_fft = 1 << (M - 1).bit_length()
    
    A = np.fft.rfft(a, M_fft)
    P = np.fft.rfft(p, M_fft)
    A_rev = np.fft.rfft(a[::-1], M_fft)
    P_rev = np.fft.rfft(p[::-1], M_fft)
    
    conv_ar_p = np.fft.irfft(A_rev * P, M_fft)[:M]
    conv_a_pr = np.fft.irfft(A * P_rev, M_fft)[:M]
    
    term1 = conv_ar_p[n-1 : 2*n-1]
    term2 = conv_a_pr[n-1 : 2*n-1]
    
    grad_L = term1 + term2
    return L, grad_L


def mirror_descent(a_init, beta_schedule, eta_base, t_start, time_limit, verbose=False):
    """
    Mirror descent with LSE smoothing and continuation.
    a is maintained on the simplex (sum=1, a>=0).
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_score = compute_score(a)
    global_step = 0
    
    for beta in beta_schedule:
        if time.monotonic() - t_start > time_limit:
            break
        
        n_inner = 0
        while time.monotonic() - t_start < time_limit:
            L, grad_L = lse_grad_L(a, beta)
            max_grad = float(np.abs(grad_L).max())
            if max_grad < 1e-15:
                break
            
            eta = eta_base / max_grad
            
            # Mirror descent update
            log_update = -eta * grad_L
            # Clip for numerical stability
            log_update = np.clip(log_update, -50.0, 50.0)
            a_new = a * np.exp(log_update)
            a_new = np.maximum(a_new, 1e-30)
            S = a_new.sum()
            if S < 1e-12:
                break
            a_new = a_new / S
            
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            
            a = a_new
            n_inner += 1
            global_step += 1
        
        if verbose:
            print(f"    beta={beta}: {n_inner} steps, score={best_score:.8f}")
    
    return best_score, best_a


def lbfgs_optimize(a_init, beta, maxiter=5000, maxfun=50000):
    """L-BFGS-B optimization with LSE smoothing."""
    n = len(a_init)
    a0 = np.clip(a_init.astype(np.float64), 0.0, 1000.0)
    S = a0.sum()
    if S < 1e-12:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        x_c = np.clip(x, 1e-30, 1e10)
        L, grad_L = lse_grad_L(x_c, beta)
        S = x_c.sum()
        S2 = S * S
        score = 2.0 * n * L / S2
        grad = 2.0 * n * (grad_L / S2 - 2.0 * L * x_c / S2 / S)
        return score, grad
    
    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)

"""
Mirror descent (multiplicative weights) optimizer for minimizing
2n * max(self-conv) / (sum)^2 over non-negative sequences.
Combines LSE continuation, top-K peak targeting, L-BFGS polishing,
multi-length exploration, and diverse restarts.
Key difference from previous: uses mirror descent (KL metric) instead of
Adam (Euclidean metric), which is more natural for the simplex constraint.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_grad_L(a, beta):
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    Z = exp_scaled.sum()
    p = exp_scaled / Z
    L = c_max + np.log(Z) / beta
    M = 3 * n - 2
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    P = np.fft.rfft(p, M_fft)
    A_rev = np.fft.rfft(a[::-1], M_fft)
    P_rev = np.fft.rfft(p[::-1], M_fft)
    conv_ar_p = np.fft.irfft(A_rev * P, M_fft)[:M]
    conv_a_pr = np.fft.irfft(A * P_rev, M_fft)[:M]
    term1 = conv_ar_p[n - 1:2 * n - 1]
    term2 = conv_a_pr[n - 1:2 * n - 1]
    grad_L = term1 + term2
    return L, grad_L


def topk_grad(a, top_k, temperature):
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    if temperature is None or temperature <= 0:
        top_indices = np.argsort(c)[-top_k:]
        w = np.zeros(len(c))
        w[top_indices] = 1.0 / top_k
    else:
        scaled = temperature * (c - c_max)
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()
    M = 3 * n - 2
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    W = np.fft.rfft(w, M_fft)
    A_rev = np.fft.rfft(a[::-1], M_fft)
    W_rev = np.fft.rfft(w[::-1], M_fft)
    conv_ar_w = np.fft.irfft(A_rev * W, M_fft)[:M]
    conv_a_wr = np.fft.irfft(A * W_rev, M_fft)[:M]
    term1 = conv_ar_w[n - 1:2 * n - 1]
    term2 = conv_a_wr[n - 1:2 * n - 1]
    return term1 + term2


def mirror_descent(a_init, beta_schedule, eta_base, t_start, time_limit,
                   use_topk=False, top_k=10):
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S
    best_a = a.copy()
    best_score = compute_score(a)
    for beta in beta_schedule:
        if time.monotonic() - t_start > time_limit:
            break
        n_inner = 0
        stall = 0
        while time.monotonic() - t_start < time_limit:
            if use_topk:
                grad = topk_grad(a, top_k, beta if beta > 5 else None)
            else:
                L, grad = lse_grad_L(a, beta)
            max_grad = float(np.abs(grad).max())
            if max_grad < 1e-15:
                break
            eta = eta_base / max_grad
            log_update = np.clip(-eta * grad, -40.0, 40.0)
            a_new = a * np.exp(log_update)
            a_new = np.maximum(a_new, 1e-30)
            S = a_new.sum()
            if S < 1e-12:
                break
            a_new = a_new / S
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
            a = a_new
            n_inner += 1
            if stall > 5000:
                break
        if n_inner > 0:
            pass
    return best_score, best_a


def lbfgs_optimize(a_init, beta, maxiter=3000, maxfun=50000):
    n = len(a_init)
    a0 = np.clip(a_init.astype(np.float64), 1e-10, 1000.0)
    S = a0.sum()
    if S < 1e-12:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        x_c = np.clip(x, 1e-10, 1e10)
        L, grad_L = lse_grad_L(x_c, beta)
        S = x_c.sum()
        S2 = S * S
        score = 2.0 * n * L / S2
        grad = 2.0 * n * (grad_L / S2 - 2.0 * L * x_c / (S2 * S))
        return score, grad

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def resample_seq(a, new_n):

    if n == new_n:

    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def peak_coordinate_descent(a, n_sweeps=50, t_start=None, time_limit=None):
    n = len(a)

        je = min(k_star, n - 1)
        contrib_mask = np.zeros(n, dtype=bool)
        contrib_mask[js:je + 1] = True
        non_contrib_mask = ~contrib_mask
        nc_count = non_contrib_mask.sum()
        if nc_count == 0:

        contrib_idx = np.where(contrib_mask)[0]
        contrib_vals = a[contrib_idx] * a[k_star - contrib_idx]

                best_score = current_score
                best_a = a.copy()
        else:

            break

                                   use_topk=True, top_k=top_k)
        if sc < best_score:
            try_update(sc, a_opt, f"TopK MD k={top_k}")
            a_current = a_opt.copy()
        else:
            a_blend = 0.5 * a_current + 0.5 * a_opt
            a_blend = np.clip(a_blend, 0, 1000)
            a_blend = a_blend / a_blend.sum()
            sc_b = compute_score(a_blend)
            if sc_b < best_score:
                try_update(sc_b, a_blend, f"TopK blend k={top_k}")
                a_current = a_blend.copy()
            else:
                a_current = a_opt.copy()
        print(f"  TopK k={top_k}: best={best_score:.8f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 3: L-BFGS polishing (25% time)
    p3_deadline = p2_deadline + total_budget * 0.25
    print(f"\nPhase 3: L-BFGS polishing (deadline={p3_deadline - t_start:.0f}s)")
    for beta in [2000, 5000, 10000, 20000, 50000]:
        if time.monotonic() > p3_deadline - 5:
            break
        remaining = p3_deadline - 5 - time.monotonic()
        if remaining < 10:
            break
        t0 = time.monotonic()
        sc, a_opt = lbfgs_optimize(best_a, beta, maxiter=3000, maxfun=40000)
        elapsed = time.monotonic() - t0
        try_update(sc, a_opt, f"LBFGS beta={beta} ({elapsed:.1f}s)")
    
    # L-BFGS with perturbation
    if time.monotonic() < p3_deadline - 10:
        n_best = len(best_a)
        for pert_scale in [0.05, 0.1, 0.2]:
            if time.monotonic() > p3_deadline - 5:
                break
            a_pert = best_a * (1.0 + pert_scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0, 1000)
            a_pert = a_pert / a_pert.sum()
            sc, a_opt = lbfgs_optimize(a_pert, 10000, maxiter=2000, maxfun=30000)
            try_update(sc, a_opt, f"LBFGS pert={pert_scale}")

    # Phase 4: Peak coordinate descent (10% time)
    p4_deadline = p3_deadline + total_budget * 0.10
    print(f"\nPhase 4: Peak coordinate descent (deadline={p4_deadline - t_start:.0f}s)")
    if time.monotonic() < p4_deadline - 5:
        t_alloc = min(p4_deadline - 5 - time.monotonic(), total_budget * 0.08)
        sc, a_opt = peak_coordinate_descent(best_a, n_sweeps=200,
                                            t_start=time.monotonic(), time_limit=t_alloc)
        try_update(sc, a_opt, "peak coord descent")

    # Phase 5: Multi-length exploration (5% time)
    p5_deadline = p4_deadline + total_budget * 0.05
    print(f"\nPhase 5: Multi-length exploration (deadline={p5_deadline - t_start:.0f}s)")
    n_best = len(best_a)
    for factor in [0.5, 1.5, 2.0]:
        if time.monotonic() > p5_deadline - 10:
            break
        n_try = int(n_best * factor)
        n_try = max(512, min(16384, n_try))
        n_try = 1 << (n_try - 1).bit_length()
        if n_try == n_best:
            continue
        a_res = resample_seq(best_a, n_try)
        a_res = a_res / a_res.sum()
        t_alloc = min(p5_deadline - 10 - time.monotonic(), 20.0)
        if t_alloc < 5:
            break
        sc, a_opt = mirror_descent(a_res, [500, 2000, 5000], 1.0,
                                   t_start=time.monotonic(), time_limit=t_alloc)
        remaining = p5_deadline - 5 - time.monotonic()
        if remaining > 5:
            sc2, a_opt2 = lbfgs_optimize(a_opt, 5000, maxiter=1500, maxfun=20000)
            if sc2 < sc:
                sc, a_opt = sc2, a_opt2
        try_update(sc, a_opt, f"multi-n={n_try}")

    # Phase 6: Diverse restarts (5% time)
    p6_deadline = min(t_start + total_budget - 5, p5_deadline + total_budget * 0.05)
    print(f"\nPhase 6: Diverse restarts (deadline={p6_deadline - t_start:.0f}s)")
    n_best = len(best_a)
    n_restarts = 4
    for r in range(n_restarts):
        if time.monotonic() > p6_deadline - 10:
            break
        strategy = r % 4
        if strategy == 0:
            log_a = np.log(np.clip(best_a, 1e-10, 1e10))
            a_pert = np.exp(log_a + 0.08 * rng.standard_normal(n_best))
        elif strategy == 1:
            c = fast_conv(best_a)
            k_star = int(c.argmax())
            js = max(0, k_star - n_best + 1)
            je = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            a_pert[js:je + 1] *= 0.4
            nc = np.ones(n_best, dtype=bool)
            nc[js:je + 1] = False
            a_pert[nc] *= 2.0
        elif strategy == 2:
            x = np.arange(n_best) / n_best
            f1 = 3 + 7 * rng.random()
            f2 = 8 + 12 * rng.random()
            p1 = 2 * np.pi * rng.random()
            p2 = 2 * np.pi * rng.random()
            mod = 1.0 + 0.15 * np.cos(2 * np.pi * f1 * x + p1) + 0.1 * np.cos(2 * np.pi * f2 * x + p2)
            a_pert = best_a * np.maximum(mod, 0.05)
        else:
            n_blocks = 8
            a_pert = best_a.copy()
            bs = n_best // n_blocks
            for b in range(n_blocks):
                s = b * bs
                e = min(s + bs, n_best)
                a_pert[s:e] *= rng.uniform(0.5, 1.5)
        a_pert = np.clip(a_pert, 0, 1000)
        a_pert = a_pert / a_pert.sum()
        t_alloc = min(p6_deadline - 10 - time.monotonic(), 15.0)
        if t_alloc < 3:
            break
        sc, a_opt = mirror_descent(a_pert, [500, 2000], 1.0,
                                   t_start=time.monotonic(), time_limit=t_alloc)
        remaining = p6_deadline - 5 - time.monotonic()
        if remaining > 5:
            sc2, a_opt2 = lbfgs_optimize(a_opt, 10000, maxiter=1500, maxfun=20000)
            if sc2 < sc:
                sc, a_opt = sc2, a_opt2
        try_update(sc, a_opt, f"restart#{r}")

    # Final polish
    print(f"\nFinal polish...")
    final_deadline = t_start + total_budget - 3
    if time.monotonic() < final_deadline - 10:
        a_cur = best_a / best_a.sum()
        t_alloc = min(final_deadline - 10 - time.monotonic(), 15.0)
        sc, a_opt = mirror_descent(a_cur, [20000, 50000], 0.5,
                                   t_start=time.monotonic(), time_limit=t_alloc)
        try_update(sc, a_opt, "final MD")
    if time.monotonic() < final_deadline - 5:
        sc, a_opt = lbfgs_optimize(best_a, 30000, maxiter=5000, maxfun=80000)
        try_update(sc, a_opt, "final LBFGS")

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
