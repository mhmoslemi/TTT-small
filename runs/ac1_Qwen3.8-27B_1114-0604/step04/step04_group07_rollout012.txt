```python
def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    
    # dL/da = sum_k p_k * dc_k/da
    # dc_k/da_i = a[k-i]*1(0<=k-i<n) + a[i-k]*1(0<=i-k<n)
    # Sum over k: dL_da[i] = sum_k p_k*a[k-i]*1(0<=k-i<n) + sum_k p_k*a[i-k]*1(0<=i-k<n)
    # First term: sum_m a[m]*p[m+i] (m=k-i, k=m+i)
    # Second term: sum_m a[m]*p[i-m] (m=i-k, k=i-m)
    # = R1[i] + R2[i] where R1[i] = sum_m a[m]p[m+i] and R2[i] = sum_m a[m]p[i-m]
    # R2[i] = (a * p)[i] (convolution) for i in [0, n-1] (valid range)
    # R1[i] = sum_m a[m]p[m+i] = cross-correlation
    
    # Compute R1 via FFT:
    # R1[i] = conv(a, p_rev)[2n-2-i] where p_rev = p[::-1]
    # conv has length 3n-2
    M_fft = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, M_fft)
    PR = np.fft.rfft(p[::-1], M_fft)
    conv_ApR = np.fft.irfft(A * PR, M_fft)[:3*n-2]
    R1 = conv_ApR[n-1:2*n-1][::-1]  # R1[i] = conv_ApR[2n-2-i] for i=0..n-1
    
    # Compute R2 = (a * p)[0:n] (first n elements of the convolution)
    P = np.fft.rfft(p, M_fft)
    conv_Ap = np.fft.irfft(A * P, M_fft)[:3*n-2]
    R2 = conv_Ap[:n]  # R2[i] = (a*p)[i] for i=0..n-1
    
    dL_da = R1 + R2
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3 * a)
    return score, grad
```
