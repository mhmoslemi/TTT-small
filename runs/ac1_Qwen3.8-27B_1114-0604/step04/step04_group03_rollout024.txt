"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses improved LSE annealing, frequency-domain equalization,
momentum subgradient, and targeted peak reduction.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + (log_sum - scaled.max()) / beta
    
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    
    rev_a = a[::-1]
    M2 = 3 * n
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    r = np.fft.irfft(P * RA, M2_fft)[n-1:2*n-1]
    
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        return lse_obj_grad(x, beta)
    
    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def freq_equalize(a, n_iters=300, lr_start=1.0, lr_end=0.1):
    """
    Frequency-domain equalization: iteratively modify the spectrum
    to flatten the convolution.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M_fft = 1 << (2 * n - 2).bit_length()
    
    for it in range(n_iters):
        lr = lr_start + (lr_end - lr_start) * (it / n_iters)
        
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        power = A * np.conj(A)
        conv = np.fft.irfft(power, M_fft)[:2*n-1]
        
        conv_max = float(conv.max())
        conv_mean = float(conv.mean())
        if conv_max - conv_mean < 1e-15:
            break
        
        # Target: make conv more uniform
        # Use a soft threshold: reduce values above threshold
        thresh = conv_mean + 0.5 * (conv_max - conv_mean)
        target = np.minimum(conv, thresh)
        err = target - conv  # negative where we want to reduce
        
        # In frequency domain: modulate the power spectrum
        err_pad = np.zeros(M_fft)
        err_pad[:2*n-1] = err
        err_spec = np.fft.rfft(err_pad)
        
        # Linearized update: delta_power = err_spec / (2 * Re(A))
        # This gives us delta in the power spectrum
        # Then modify A accordingly
        with np.errstate(divide='ignore', invalid='ignore'):
            # We want to change |A|^2 by err_spec (approximately)
            # |A_new|^2 = |A|^2 + err_spec
            # |A_new| = sqrt(|A|^2 + err_spec)
            new_power = power + lr * err_spec
            new_power = np.maximum(new_power, 0)  # ensure non-negative power
            new_mag = np.sqrt(new_power)
            # Preserve phase
            phase = np.angle(A)
            A_new = new_mag * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def momentum_subgrad(a, n_iters=5000, momentum=0.9):
    """Subgradient method with heavy ball momentum and Polyak averaging."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    v = np.zeros(n)
    avg_a = a.copy()
    sc_lb = 1.0  # theoretical lower bound
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max(conv) wrt a
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end+1] = 2.0 * a[j_start:j_end+1][::-1]
        
        # Full gradient of the objective (including the 1/S^2 term)
        grad = (2.0 * n) * (sg / (S * S) - 2.0 * M / (S * S * S))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        
        # Polyak step size
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        # Polyak averaging
        avg_a = avg_a + (a_new - avg_a) / (it + 1)
    
    # Check averaged solution too
    sc_avg = compute_score(avg_a)
    if sc_avg < best_sc:
        best_a = avg_a.copy()
        best_sc = sc_avg
    
    return best_a


def peak_targeted_cd(a, n_sweeps=3, top_k=5):
    """Coordinate descent targeting top peaks with larger step sizes."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        peak_idx = np.argsort(c)[::-1][:top_k]
        
        for k_star in peak_idx:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue
            
            seg = a[j_start:j_end+1]
            contrib = seg[::-1].copy()
            
            # Sort by contribution and try reducing top contributors
            order = np.argsort(contrib)[::-1]
            for j in order[:20]:
                j_global = j_start + j
                old_val = a[j_global]
                if old_val < 1e-10:
                    continue
                
                # Try several reduction factors
                best_local = best_sc
                best_val = old_val
                for factor in [0.1, 0.3, 0.5, 0.7, 0.8]:
                    new_val = old_val * factor
                    a[j_global] = new_val
                    sc = compute_score(a)
                    if sc < best_local:
                        best_local = sc
                        best_val = new_val
                
                a[j_global] = best_val
                S = a.sum()
                if S > 1e-12:
                    a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LSE annealing with L-BFGS, improved frequency-domain equalization,
momentum subgradient with Polyak averaging, harmonic parameterization,
and targeted coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + (log_sum - scaled.max()) / beta

    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()

    # dL/da[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    rev_a = a[::-1]
    M2 = 3 * n
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    r = np.fft.irfft(P * RA, M2_fft)[n - 1:2 * n - 1]

    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        return lse_obj_grad(x, beta)

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def freq_equalize_v2(a, n_iters=300):
    """
    Improved frequency-domain equalization.
    Uses iterative power spectrum shaping to flatten convolution.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    M_fft = 1 << (2 * n - 2).bit_length()

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        lr = 0.8 * (1.0 - 0.5 * progress)

        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        power = np.real(A * np.conj(A))
        conv = np.fft.irfft(power, M_fft)[:2 * n - 1]

        conv_max = float(conv.max())
        conv_mean = float(conv.mean())
        if conv_max - conv_mean < 1e-14:
            break

        # Aggressive flattening: push conv toward its mean
        # Use a soft knee: values above mean + delta get pushed down
        delta = 0.8 * (conv_max - conv_mean)
        target = np.where(conv > conv_mean + delta,
                          conv_mean + delta * np.exp(1 - (conv - conv_mean) / delta),
                          conv)
        err = target - conv

        err_pad = np.zeros(M_fft)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad)

        # Update power spectrum: new_power = power + lr * err_spec_real
        new_power = power + lr * np.real(err_spec)
        new_power = np.maximum(new_power, 0.0)
        new_mag = np.sqrt(new_power)
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def momentum_subgrad(a, n_iters=5000, momentum=0.85):
    """Subgradient with momentum and Polyak step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    v = np.zeros(n)
    avg_a = a.copy()
    sc_lb = 1.0

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        grad = (2.0 * n) * (sg / (S * S) - 2.0 * M / (S * S * S))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        step = max(step, 1e-8)

        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        avg_a += (a_new - avg_a) / (it + 1)

    sc_avg = compute_score(avg_a)
    if sc_avg < best_sc:
        best_a = avg_a.copy()
        best_sc = sc_avg

    return best_a


def harmonic_optimize(n, n_harmonics=10, n_iters=500, seed=42):
    """
    Optimize over a cosine series parameterization.
    a[i] = max(0, c0 + sum_{k=1}^{K} c_k * cos(2*pi*k*i/n + phi_k))
    This reduces dimensionality and may find good structures.
    """
    rng = np.random.default_rng(seed)

    def eval_coeffs(coeffs):
        c0 = coeffs[0]
        c_rest = coeffs[1:2 * n_harmonics + 1]
        amps = c_rest[0::2]
        phases = c_rest[1::2]
        x = np.arange(n) / n
        a = np.full(n, c0)
        for k in range(n_harmonics):
            a += amps[k] * np.cos(2 * np.pi * (k + 1) * x + phases[k])
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12:
            return 1e10, np.zeros(2 * n_harmonics + 1)
        a = a / a.sum()
        sc = compute_score(a)
        return sc, a

    # Initialize
    coeffs = np.zeros(2 * n_harmonics + 1)
    coeffs[0] = 1.0
    for k in range(n_harmonics):
        coeffs[2 * k + 1] = 0.1 * np.cos(rng.random() * 2 * np.pi) / (k + 1)
        coeffs[2 * k + 2] = rng.random() * 2 * np.pi

    best_sc, best_a = eval_coeffs(coeffs)

    # Coordinate descent on coefficients
    for it in range(n_iters):
        improved = False
        for d in range(len(coeffs)):
            old = coeffs[d]
            best_local = best_sc
            best_val = old
            for delta in [-0.1, -0.05, -0.01, 0.01, 0.05, 0.1, 0.2, -0.2]:
                coeffs[d] = old + delta
                sc, a = eval_coeffs(coeffs)
                if sc < best_local:
                    best_local = sc
                    best_val = coeffs[d]
            coeffs[d] = best_val
            if best_val != old:
                improved = True
            sc, a = eval_coeffs(coeffs)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        if not improved:
            break

    return best_sc, best_a


def coord_descent_cd(a, n_sweeps=3):
    """Simple coordinate descent with random ordering."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng(123)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            old = a[i]
            if old < 1e-12:
                continue
            best_local = best_sc
            best_val = old
            for factor in [0.3, 0.5, 0.7, 1.3, 1.5]:
                a[i] = old * factor
                sc = compute_score(a)
                if sc < best_local:
                    best_local = sc
                    best_val = a[i]
            a[i] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent))

    x = np.arange(n, dtype=np.float64) / n

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2 * np.pi * x))
    inits.append(0.5 + 0.3 * np.cos(4 * np.pi * x))
    inits.append(0.3 + 0.7 * (2 * x - 1) ** 2)
    inits.append(0.3 + 0.7 * (1 - (2 * x - 1) ** 2))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, None))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, None))

    # Multi-frequency
    inits.append(0.5 + 0.2 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x + 0.5)
                 + 0.1 * np.cos(6 * np.pi * x + 1.0))
    inits.append(0.5 + 0.3 * np.cos(3 * np.pi * x) + 0.2 * np.cos(5 * np.pi * x))

    # Exponential decay from center
    center = (n - 1) / 2.0
    inits.append(0.2 + 0.8 * np.exp(-((np.arange(n) - center) ** 2) / (2 * (n / 4) ** 2)))

    return inits
