"""
Improved optimization for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key improvements over previous: top-k LSE smoothing, convolution-aware perturbations,
adaptive beta scheduling, block coordinate descent, and multi-scale length exploration.
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv2(a, b):
    """Linear convolution of a and b via real FFT."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta, topk=1):
    """
    LSE-smoothed score and gradient. Uses top-k LSE for smoother optimization.
    topk=1 is standard LSE on the max. topk>1 uses a sorted-LSE approach.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    if topk <= 1:
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -500.0, 0.0)
        exp_scaled = np.exp(scaled)
        log_sum = np.log(exp_scaled.sum())
        L = c_max + log_sum / beta
        p = exp_scaled / exp_scaled.sum()
    else:
        # Top-k LSE: use the k largest values in LSE
        # Approximate by: LSE of all values with a "threshold" at the k-th largest
        sorted_c = np.sort(c)[::-1]
        threshold = sorted_c[min(topk, len(c)) - 1]
        # LSE on values above threshold, plus threshold for the rest
        mask = c >= threshold
        if mask.sum() == 0:
            L = float(c.max())
            p = np.zeros_like(c)
            p[int(c.argmax())] = 1.0
        else:
            c_above = c[mask]
            c_max_above = float(c_above.max())
            scaled = beta * (c_above - c_max_above)
            scaled = np.clip(scaled, -500.0, 0.0)
            exp_scaled = np.exp(scaled)
            log_sum = np.log(exp_scaled.sum())
            L_above = c_max_above + log_sum / beta
            # Blend: L = L_above (since values below threshold contribute little)
            L = L_above
            p = np.zeros_like(c)
            p_above = exp_scaled / exp_scaled.sum()
            p[mask] = p_above

    score = 2.0 * n * L / (S * S)

    # Gradient: dL/da_j = 2 * sum_m a[m] * p[j+m] (cross-correlation)
    M3 = 3 * n - 2
    M3_fft = 1 << (M3 - 1).bit_length()
    A3 = np.fft.rfft(a, M3_fft)
    P3 = np.fft.rfft(p, M3_fft)
    cross = np.fft.irfft(np.conj(A3) * P3, M3_fft)[:n]

    dL_da = 2.0 * cross
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)

    return score, grad


def run_lbfgs(a0, beta, topk=1, maxiter=3000, maxfun=80000, time_budget=None, t_start=None):
    """L-BFGS-B optimization with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    objective_and_grad = lambda x: lse_score_grad(x, beta, topk)

    if time_budget is not None and t_start is not None:
        deadline = t_start + time_budget
        def timed_obj(x):
            if time.monotonic() > deadline:
                raise TimeoutError
            return lse_score_grad(x, beta, topk)
        try:
            res = minimize(
                timed_obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                options={'ftol': 1e-15, 'gtol': 1e-12,
                         'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
            )
        except (TimeoutError, Exception):
            a_out = a0
            exact_sc = compute_score(a_out)
            return exact_sc, a_out
    else:
        res = minimize(
            objective_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )

    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def golden_section_1d(a, j, n_evals=20):
    """Golden-section search for optimal value of a[j]."""
    a_old = a[j]
    max_val = a.max()

    if a_old < 1e-12:
        lo = 0.0
        hi = max_val * 1.5
    else:
        lo = 0.0
        hi = max(a_old * 5.0, max_val * 0.5)

    if hi < 1e-10:
        hi = max_val

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

    a[j] = c1
    f1 = compute_score(a)
    a[j] = c2
    f2 = compute_score(a)

    for _ in range(n_evals - 2):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            a[j] = c1
            f1 = compute_score(a)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            a[j] = c2
            f2 = compute_score(a)

    best_val = c1 if f1 < f2 else c2
    a[j] = best_val
    return compute_score(a)


def coord_descent(a, n_sweeps=5, top_k=40, time_budget=60.0, t_start=None):
    """Coordinate descent with golden-section 1D search on top-k coordinates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score

        # Subgradient of peak
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = a[idx]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break
            sc = golden_section_1d(a, int(j), n_evals=12)
            if sc < best_score:
                best_score = sc

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def adam_optimize(a_init, lr=0.003, n_steps=100000, beta_lse=200, topk=1,
                  time_budget=None, t_start=None, lr_decay=1e-5):
    """Adam optimizer with LSE-smoothed gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        lr_t = lr / (1.0 + lr_decay * t)
        score, grad = lse_score_grad(a, beta_lse, topk)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def conv_aware_perturbation(a, rng, n_peaks=8, reduction=0.15):
    """Perturb a by reducing contributions to current convolution peaks."""
    n = len(a)
    c = fast_conv(a)
    top_positions = np.argsort(c)[::-1][:n_peaks]
    a_new = a.copy()
    S0 = a.sum()

    for k in top_positions:
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je < js:
            continue
        contrib = a[js:je + 1] * a[k - js:k - je]
        if contrib.sum() < 1e-15:
            continue
        # Reduce the top contributors
        n_reduce = max(3, len(contrib) // 5)
        top_idx = np.argsort(contrib)[::-1][:n_reduce]
        for idx in top_idx:
            j = js + idx
            partner = k - j
            if 0 <= j < n:
                a_new[j] *= (1.0 - reduction)
            if 0 <= partner < n and partner != j:
                a_new[partner] *= (1.0 - reduction * 0.5)

    a_new = np.clip(a_new, 0.0, None)
    if a_new.sum() > 1e-10:
        a_new = a_new * (S0 / a_new.sum())
    return a_new


def spectral_perturbation(a, rng, strength=0.15):
    """Perturb in the DFT domain."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)

    amp_factors = 1.0 + strength * rng.standard_normal(len(A))
    amp_factors = np.clip(amp_factors, 0.2, 3.0)
    phase_shifts = strength * 0.5 * rng.standard_normal(len(A))

    A_new = A * amp_factors * np.exp(1j * phase_shifts)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)

    if a_new.sum() < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def anti_palin_perturbation(a, rng, strength=0.12):
    """Add anti-palindromic component to reduce central peak."""
    n = len(a)
    a_new = a.copy()
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    tilt = (x - center) / max(center, 1.0)
    freq = float(rng.uniform(1, 4))
    tilt += 0.5 * np.sin(2.0 * np.pi * freq * x / n)
    tilt_anti = 0.5 * (tilt - tilt[::-1])
    a_new += strength * tilt_anti * a.max()
    a_new = np.clip(a_new, 0.0, None)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def gaussian_perturbation(a, rng, n_bumps=4):
    """Add Gaussian bumps at random locations."""
    n = len(a)
    a_new = a.copy()
    S0 = a.sum()
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(n // 40, n // 8))
        amplitude = float(rng.uniform(-0.1, 0.1)) * S0 / n
        x = np.arange(n, dtype=np.float64)
        bump = amplitude * np.exp(-0.5 * ((x - center) / width) ** 2)
        a_new += bump
    a_new = np.clip(a_new, 0.0, None)
    if a_new.sum() > 1e-10:
        a_new = a_new * (S0 / a_new.sum())
    return a_new


def block_swap_perturbation(a, rng, n_blocks=2):
    """Swap random blocks."""
    n = len(a)
    a_new = a.copy()
    for _ in range(n_blocks):
        block_len = int(rng.integers(max(2, n // 30), n // 6))
        i = int(rng.integers(0, n - block_len))
        j = int(rng.integers(0, n - block_len))
        if abs(i - j) > block_len:
            a_new[i:i + block_len], a_new[j:j + block_len] = \
                a_new[j:j + block_len].copy(), a_new[i:i + block_len].copy()
    return a_new


def chirp_perturbation(a, rng):
    """Add a chirp component."""
    n = len(a)
    a_new = a.copy()
    x = np.arange(n, dtype=np.float64) / n
    f0 = float(rng.uniform(1, 3))
    f1 = float(rng.uniform(5, 15))
    phase = float(rng.uniform(0, 2 * np.pi))
    chirp = np.sin(2.0 * np.pi * (f0 * x + (f1 - f0) * x * x / 2.0 + phase))
    amplitude = float(rng.uniform(0.02, 0.06)) * a.max()
    a_new += amplitude * chirp
    a_new = np.clip(a_new, 0.0, None)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def adaptive_continuation(a0, time_budget, t_start):
    """
    Adaptive continuation: increase LSE beta based on convergence.
    Stops early if beta is high enough that LSE ≈ true max.
    """
    a = a0.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    # Start with moderate beta, increase adaptively
    betas = [30, 80, 200, 500, 1200, 3000, 8000, 20000]
    topk_schedule = [3, 3, 2, 2, 1, 1, 1, 1]

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        # Allocate time: more for lower beta (easier landscape)
        time_alloc = min(remaining * 0.35, 45.0)
        if i >= len(betas) - 2:
            time_alloc = min(remaining * 0.4, 60.0)

        topk = topk_schedule[min(i, len(topk_schedule) - 1)]
        sc, a_out = run_lbfgs(a, beta, topk=topk, maxiter=3000, maxfun=60000,
                              time_budget=time_alloc, t_start=t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            print(f"    Cont beta={beta}, topk={topk}: score={sc:.8f}")
        a = a_out

        # Check if LSE ≈ true max (converged)
        c = fast_conv(a)
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -500.0, 0.0)
        exp_scaled = np.exp(scaled)
        LSE_val = c_max + np.log(exp_scaled.sum()) / beta
        gap = (LSE_val - c_max) / max(c_max, 1e-15)
        if gap < 1e-6 and beta >= 2000:
            break

    return best_score, best_a


def multi_scale_try(a_best, time_budget, t_start):
    """Try different sequence lengths by resampling."""
    n_best = len(a_best)
    best_score = compute_score(a_best)
    best_a = a_best.copy()
    deadline = t_start + time_budget

    factors = [0.6, 0.75, 0.9, 1.1, 1.3, 1.5, 1.8]
    for factor in factors:
        if time.monotonic() > deadline - 30:
            break
        tl = int(n_best * factor)
        if tl < 256 or tl > 40000:
            continue

        x_old = np.arange(n_best, dtype=np.float64)
        x_new = np.linspace(0, n_best - 1, tl)
        a_res = np.interp(x_new, x_old, a_best)
        a_res = np.clip(a_res, 0.0, None)

        time_alloc = min(deadline - time.monotonic() - 30, 35.0)
        if time_alloc < 15:
            break

        # Quick Adam + L-BFGS
        sc, a_opt = adam_optimize(a_res, lr=0.005, n_steps=30000, beta_lse=200,
                                  time_budget=time_alloc * 0.4, t_start=t_start)
        if time.monotonic() < deadline - 30:
            sc2, a_opt2 = run_lbfgs(a_opt, 1500, topk=1, maxiter=2000, maxfun=30000,
                                    time_budget=min(deadline - time.monotonic() - 30, 15.0),
                                    t_start=t_start)
            if sc2 < sc:
                sc = sc2
                a_opt = a_opt2

        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Multi-scale n={tl}: score={sc:.8f}")

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: adaptive continuation L-BFGS + convolution-aware perturbations
    + multi-scale exploration + intensive final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 25.0

    # Get parent sequence
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={parent_score:.8f}")
    else:
        parent = np.ones(4096)
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"No parent, using uniform n={n_parent}, score={parent_score:.8f}")

    best_score = parent_score
    best_a = parent.copy()

    def try_update(sc, a_cand, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_cand, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a_cand)} | t={elapsed:.0f}s")
            return True
        return False

    # ==================================================================
    # Phase 1: Initial optimization from parent (25% budget)
    # ==================================================================
    p1_budget = effective_budget * 0.25
    print(f"\n{'='*60}")
    print(f"Phase 1: Initial optimization (budget={p1_budget:.0f}s)")
    print(f"{'='*60}")

    # Quick Adam to escape bad local structure
    t0 = time.monotonic()
    adam_time = min(25.0, p1_budget * 0.3)
    sc, a_quick = adam_optimize(parent, lr=0.004, n_steps=80000, beta_lse=100, topk=2,
                                time_budget=adam_time, t_start=t_start)
    try_update(sc, a_quick, "Quick Adam")
    print(f"  Adam: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Adaptive continuation
    remaining = p1_budget - (time.monotonic() - t_start)
    if remaining > 15:
        t0 = time.monotonic()
        sc, a_cont = adaptive_continuation(best_a, remaining, t_start)
        try_update(sc, a_cont, "Adaptive Continuation")
        print(f"  Continuation: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Coordinate descent polish
    remaining = p1_budget - (time.monotonic() - t_start)
    if remaining > 10:
        t0 = time.monotonic()
        sc, a_cd = coord_descent(best_a, n_sweeps=4, top_k=35,
                                time_budget=min(remaining, 30.0), t_start=t_start)
        try_update(sc, a_cd, "Coord Descent")
        print(f"  CoordDesc: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==================================================================
    # Phase 2: Diverse restarts with convolution-aware perturbations (40% budget)
    # ==================================================================
    p2_budget = effective_budget * 0.40
    print(f"\n{'='*60}")
    print(f"Phase 2: Diverse restarts (budget={p2_budget:.0f}s)")
    print(f"{'='*60}")

    p2_deadline = t_start + p1_budget + p2_budget
    n_restarts = 12
    per_restart_time = p2_budget / (n_restarts + 2)

    for ri in range(n_restarts):
        if time.monotonic() > p2_deadline - 15:
            break

        t_remain = p2_deadline - time.monotonic()
        time_alloc = min(t_remain * 0.85, per_restart_time * 1.8)
        if time_alloc < 15:
            break

        # Generate diverse perturbation
        strategy = ri % 6
        if strategy == 0:
            a_pert = conv_aware_perturbation(best_a, rng, n_peaks=10, reduction=0.2)
            strat_name = "conv-aware"
        elif strategy == 1:
            a_pert = spectral_perturbation(best_a, rng, strength=0.2)
            strat_name = "spectral"
        elif strategy == 2:
            a_pert = anti_palin_perturbation(best_a, rng, strength=0.15)
            strat_name = "anti-palin"
        elif strategy == 3:
            a_pert = gaussian_perturbation(best_a, rng, n_bumps=5)
            a_pert = conv_aware_perturbation(a_pert, rng, n_peaks=5, reduction=0.1)
            strat_name = "gauss+conv"
        elif strategy == 4:
            a_pert = block_swap_perturbation(best_a, rng, n_blocks=2)
            a_pert = spectral_perturbation(a_pert, rng, strength=0.1)
            strat_name = "swap+spectral"
        else:
            a_pert = chirp_perturbation(best_a, rng)
            a_pert = conv_aware_perturbation(a_pert, rng, n_peaks=5, reduction=0.15)
            strat_name = "chirp+conv"

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick optimization of perturbation
        t0 = time.monotonic()
        # Adam
        adam_t = time_alloc * 0.35
        sc, a_opt = adam_optimize(a_pert, lr=0.004, n_steps=50000, beta_lse=150, topk=2,
                                  time_budget=adam_t, t_start=t_start)
        # L-BFGS
        if time.monotonic() < p2_deadline - 15:
            lb_t = min(time_alloc - (time.monotonic() - t0) - 5, 25.0)
            if lb_t > 5:
                sc2, a_opt2 = run_lbfgs(a_opt, 2000, topk=1, maxiter=2000, maxfun=40000,
                                        time_budget=lb_t, t_start=t_start)
                if sc2 < sc:
                    sc = sc2
                    a_opt = a_opt2

        try_update(sc, a_opt, f"R{ri}:{strat_name}")

    # ==================================================================
    # Phase 3: Intensive optimization of best (20% budget)
    # ==================================================================
    p3_budget = effective_budget * 0.20
    print(f"\n{'='*60}")
    print(f"Phase 3: Intensive optimization (budget={p3_budget:.0f}s)")
    print(f"{'='*60}")

    p3_deadline = p2_deadline + p3_budget

    # High-beta L-BFGS with multiple betas
    for beta in [3000, 10000, 30000, 80000]:
        if time.monotonic() > p3_deadline - 20:
            break
        t0 = time.monotonic()
        remaining = p3_deadline - time.monotonic() - 20
        if remaining < 10:
            break
        time_alloc = min(remaining * 0.3, 35.0)
        sc, a_opt = run_lbfgs(best_a, beta, topk=1, maxiter=5000, maxfun=80000,
                              time_budget=time_alloc, t_start=t_start)
        try_update(sc, a_opt, f"Intensive LBFGS beta={beta}")
        print(f"    beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Coordinate descent with more sweeps
    if time.monotonic() < p3_deadline - 10:
        t0 = time.monotonic()
        remaining = p3_deadline - time.monotonic() - 10
        sc, a_opt = coord_descent(best_a, n_sweeps=8, top_k=50,
                                  time_budget=min(remaining, 40.0), t_start=t_start)
        try_update(sc, a_opt, "Intensive CoordDesc")
        print(f"    CoordDesc: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==================================================================
    # Phase 4: Multi-scale exploration (10% budget)
    # ==================================================================
    p4_budget = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-scale exploration (budget={p4_budget:.0f}s)")
    print(f"{'='*60}")

    if time.monotonic() < t_start + effective_budget - 30:
        p4_deadline = min(t_start + effective_budget - 30, p3_deadline + p4_budget)
        sc, a_opt = multi_scale_try(best_a, p4_deadline - time.monotonic(), t_start)
        try_update(sc, a_opt, "Multi-scale")

    # ==================================================================
    # Phase 5: Final polish (remaining time)
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    final_deadline = t_start + effective_budget - 8.0

    # Very high beta L-BFGS
    if time.monotonic() < final_deadline - 15:
        t0 = time.monotonic()
        remaining = final_deadline - time.monotonic() - 15
        if remaining > 10:
            sc, a_opt = run_lbfgs(best_a, 100000, topk=1, maxiter=8000, maxfun=100000,
                                  time_budget=min(remaining, 45.0), t_start=t_start)
            try_update(sc, a_opt, "Final LBFGS beta=100k")
            print(f"    Final LBFGS: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Final coordinate descent
    if time.monotonic() < final_deadline - 5:
        remaining = final_deadline - time.monotonic() - 5
        if remaining > 5:
            t0 = time.monotonic()
            sc, a_opt = coord_descent(best_a, n_sweeps=6, top_k=60,
                                      time_budget=min(remaining, 25.0), t_start=t_start)
            try_update(sc, a_opt, "Final CoordDesc")
            print(f"    Final CoordDesc: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
