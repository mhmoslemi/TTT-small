```python
    return np.fft.irfft(A * A, M_fft)[:M]

    S = float(a.sum())

    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum / beta
    return 2.0 * n * L / (S * S), L, c

    """Gradient of LSE-smoothed score."""

    p = exp_s / exp_s.sum()
```

```python
"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Incremental O(n) convolution updates for fast coordinate descent (vs O(n log n) FFT)
2. Vectorized peak-targeted mass transfer with adaptive step sizing
3. Geometric beta annealing with L-BFGS and periodic true-objective checking
4. Multi-length exploration with smart parent interpolation
5. "Joint peak" handling: simultaneously reduce top-2/3 convolution peaks
6. Reheating strategy to escape local minima
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_and_grad(a, beta):
    """LSE-smoothed score and gradient. Returns (score, grad)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum_exp = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum_exp
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # dL/da[i] = 2 * (rev_a correlate p)[i] using FFT
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped_obj(x):
        s, g = lse_obj_and_grad(x, beta)
        return s, g

    res = minimize(
        wrapped_obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def conv_update_vec(c, i, delta, a, n):
    """
    Vectorized update of c after a[i] += delta.
    a must have the NEW value at position i already set.
    old_a_i = a[i] - delta.
    """
    old_a_i = a[i] - delta
    k_lo = max(0, i)
    k_hi = min(2 * n - 1, i + n)
    if k_hi <= k_lo:
        return
    j_vals = np.arange(k_lo, k_hi) - i
    partners = a[j_vals]
    updates = 2.0 * delta * partners
    # Special case: k = 2*i (self-term)
    if 2 * i >= k_lo and 2 * i < k_hi:
        idx = 2 * i - k_lo
        updates[idx] = 2.0 * old_a_i * delta + delta * delta
    c[k_lo:k_hi] += updates


def coord_descent_incremental(a, max_iters=5000, top_peaks=3, patience=200):
    """
    Fast coordinate descent with incremental convolution updates.
    Transfers mass within the peak lag segment from top to bottom contributors.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    a /= S
    c = fast_conv(a)
    best_a = a.copy()
    best_max_c = float(c.max())
    best_score = 2.0 * n * best_max_c  # since sum=1

    no_improve = 0
    for it in range(max_iters):
        # Find top peak lags
        peak_idx = np.argsort(c)[::-1][:top_peaks]
        c_max = float(c.max())
        
        if c_max < best_max_c - 1e-14:
            best_max_c = c_max
            best_a = a.copy()
            best_score = 2.0 * n * c_max
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        improved = False
        for k_star in peak_idx:
            if improved:
                # Re-find peaks after improvement
                break
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end < j_start:
                continue
            seg_len = j_end - j_start + 1
            
            # Contributions at this lag
            seg = a[j_start:j_end + 1]
            partner_pos = k_star - np.arange(j_start, j_end + 1)
            partner = a[partner_pos]
            contrib = seg * partner
            
            # Sort by contribution
            order = np.argsort(contrib)
            
            # Try multiple transfer amounts
            for frac in [0.4, 0.2, 0.1, 0.05, 0.02]:
                success = False
                # Transfer from top contributor to bottom
                n_top = max(1, seg_len // 10)
                n_bot = max(1, seg_len // 10)
                
                for t_pos in range(-1, -n_top - 1, -1):
                    src = j_start + order[t_pos]
                    src_val = a[src]
                    if src_val < 1e-10:
                        continue
                    give = src_val * frac
                    if give < 1e-12:
                        continue
                    
                    # Choose destination: bottom contributor
                    for b_pos in range(0, min(n_bot, 5)):
                        dst = j_start + order[b_pos]
                        if src == dst:
                            continue
                        
                        # Perform transfer
                        old_src = a[src]
                        old_dst = a[dst]
                        
                        # Step 1: reduce src
                        a[src] = old_src - give
                        conv_update_vec(c, src, -give, a, n)
                        
                        # Step 2: increase dst
                        a[dst] = old_dst + give
                        conv_update_vec(c, dst, give, a, n)
                        
                        new_max = float(c.max())
                        if new_max < best_max_c - 1e-14:
                            best_max_c = new_max
                            best_a = a.copy()
                            best_score = 2.0 * n * new_max
                            no_improve = 0
                            improved = True
                            success = True
                            break
                        else:
                            # Revert
                            a[dst] = old_dst
                            conv_update_vec(c, dst, -give, a, n)
                            a[src] = old_src
                            conv_update_vec(c, src, give, a, n)
                    
                    if success:
                        break
                if improved:
                    break
        # End peak loop
    return best_a

                        give_i = a[i] * 0.1
                        a[i] -= give_i
                        conv_update_vec(c, i, -give_i, a, n)
                    for i in low_indices:
                        a[i] += per_receive
                        conv_update_vec(c, i, per_receive, a, n)
                    
                    if c.max() < best_max_c - 1e-14:
                        best_max_c = c.max()
                        best_a = a.copy()
                    else:
                        a = a_old
                        c = c_old
        else:
            # Peaks are far apart - just reduce the top one
            pass
    
    return best_a


def error_reduction_iter(a, n_iters=100):
    """
    Frequency-domain error reduction: iteratively flatten the convolution
    by modifying the spectrum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        v = A * A
        conv = np.fft.irfft(v, M)[:2 * n - 1]
        
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        
        # Create target: flatten the excess
        alpha = 0.3 + 0.6 * (it / n_iters)
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        
        conv_mod = conv - excess
        err = conv_mod - conv
        
        err_pad = np.zeros(M, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0
        
        A_new = A + d
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def build_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.3 + 0.7 * (1.0 - (2.0 * x - 1.0) ** 2))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    # Chirp
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * (x ** 2 * 5.0)))
    # Multi-frequency
    inits.append(0.5 + 0.2 * np.cos(2.0 * np.pi * 3.0 * x) + 0.1 * np.cos(2.0 * np.pi * 7.0 * x))
    return inits


def annealed_lbfgs(a0, betas, total_time):
    """
    Run L-BFGS with geometrically increasing beta.
    Allocates time proportionally to each beta level.
    """
    n = len(a0)
    best_sc = compute_score(a0)
    best_a = a0.copy()
    n_levels = len(betas)
    time_per_level = total_time / n_levels
    
    for i, beta in enumerate(betas):
        t_start = time.monotonic()
        t_avail = time_per_level - (time.monotonic() - t_start)
        maxiter = max(100, int(t_avail * 200))
        maxfun = max(200, int(t_avail * 500))
        maxiter = min(maxiter, 20000)
        maxfun = min(maxfun, 100000)
        
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    beta={beta}: score={sc:.10f}")
        else:
            print(f"    beta={beta}: no improvement (stayed at {best_sc:.10f})")
    
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses incremental coordinate descent + adaptive LSE annealing + peak targeting.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 15.0

    # Get parent if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(2048) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine which lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [4096, 6962, 8192, 10240, 12288]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}")

    # ================================================================
    # Phase 1: Quick exploration (12% time)
    # ================================================================
    phase1_end = t_start + 0.12 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 1: Quick Exploration ---")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        inits = build_inits(n, rng, parent)
        max_inits = min(len(inits), 6)
        for i in range(max_inits):
            if time.monotonic() > phase1_end:
                break
            a0 = inits[i]
            a0 = np.maximum(a0, 0)
            S = a0.sum()
            if S < 1e-12:
                continue
            a0 = a0 / S
            # Quick LSE at low beta
            sc, a_opt = run_lbfgs(a0, beta=30, maxiter=2000, maxfun=50000)
            report(sc, a_opt, f"P1 n={n} i={i}")

    print(f"  After Phase 1: best={best_score:.10f}")

    # ================================================================
    # Phase 2: Adaptive LSE Annealing (38% time)
    # ================================================================
    phase2_end = t_start + 0.50 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 2: LSE Annealing ---")

    n_best = len(best_a)
    a_work = best_a.copy()
    S = a_work.sum()
    a_work = a_work / S

    # Beta schedule: geometric from 20 to 5000
    betas = [20, 40, 80, 160, 320, 640, 1280, 2560, 5120]
    t_avail = phase2_end - time.monotonic()
    sc, a_opt = annealed_lbfgs(a_work, betas, t_avail)
    report(sc, a_opt, "P2 annealed")

    # Try from parent directly if it was good
    if parent is not None and time.monotonic() < phase2_end:
        n_p = len(parent)
        a_p = parent.copy()
        S_p = a_p.sum()
        a_p = a_p / S_p
        betas_p = [50, 100, 200, 500, 1000]
        t_avail_p = min(phase2_end - time.monotonic(), 120.0)
        if t_avail_p > 30:
            sc_p, a_opt_p = annealed_lbfgs(a_p, betas_p, t_avail_p)
            report(sc_p, a_opt_p, "P2 parent")

    print(f"  After Phase 2: best={best_score:.10f}")

    # ================================================================
    # Phase 3: Incremental Coordinate Descent (35% time)
    # ================================================================
    phase3_end = t_start + 0.85 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 3: Coordinate Descent ---")

    n_best = len(best_a)
    a_cd = best_a.copy()
    S = a_cd.sum()
    a_cd = a_cd / S

    # Run multiple rounds of coordinate descent with increasing patience
    rounds = [
        (3000, 3, 300),   # (max_iters, top_peaks, patience)
        (5000, 3, 500),
        (8000, 4, 500),
        (10000, 3, 1000),
    ]
    
    for r, (max_it, tp, pat) in enumerate(rounds):
        if time.monotonic() > phase3_end:
            break
        t_start_cd = time.monotonic()
        a_cd = coord_descent_incremental(a_cd, max_iters=max_it, top_peaks=tp, patience=pat)
        S = a_cd.sum()
        a_cd = a_cd / S
        sc = compute_score(a_cd)
        report(sc, a_cd, f"P3 CD round{r+1}")
        
        # Reheat: small perturbation to escape local minima
        if r < len(rounds) - 1 and time.monotonic() < phase3_end:
            scale = 0.005 + 0.01 * (r + 1)
            a_pert = a_cd * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.maximum(a_pert, 0)
            S_p = a_pert.sum()
            if S_p > 1e-12:
                a_pert = a_pert / S_p
                sc_pert = compute_score(a_pert)
                if sc_pert < best_score:
                    a_cd = a_pert
                    report(sc_pert, a_pert, f"P3 reheat{r+1}")

    # Joint peak reduction
    if time.monotonic() < phase3_end:
        a_jp = joint_peak_reduction(best_a, max_iters=3000)
        sc = compute_score(a_jp)
        report(sc, a_jp, "P3 joint")

    print(f"  After Phase 3: best={best_score:.10f}")

    # ================================================================
    # Phase 4: Final Polish (15% time)
    # ================================================================
    print(f"\n--- Phase 4: Final Polish ---")

    n_best = len(best_a)
    a_polish = best_a.copy()
    S = a_polish.sum()
    a_polish = a_polish / S

    remaining = deadline - time.monotonic()
    
    # High-beta LSE
    if remaining > 60:
        t_alloc = min(remaining * 0.3, 120.0)
        betas_final = [2000, 5000, 10000]
        sc, a_opt = annealed_lbfgs(a_polish, betas_final, t_alloc)
        report(sc, a_opt, "P4 high-beta LSE")
        a_polish = a_opt
        S = a_polish.sum()
        a_polish = a_polish / S

    # Final coordinate descent
    if remaining > 30:
        a_cd2 = coord_descent_incremental(a_polish, max_iters=10000, top_peaks=3, patience=500)
        sc = compute_score(a_cd2)
        report(sc, a_cd2, "P4 final CD")

    # Error reduction one more time
    if remaining > 20:
        a_er = error_reduction_iter(a_polish, n_iters=100)
        sc = compute_score(a_er)
        report(sc, a_er, "P4 ER")
        if sc < best_score:
            # Polish the ER result with CD
            a_er_cd = coord_descent_incremental(a_er, max_iters=5000, top_peaks=3, patience=300)
            sc2 = compute_score(a_er_cd)
            report(sc2, a_er_cd, "P4 ER+CD")

    # Try a different length if time permits
    if remaining > 120 and len(best_a) < 12000:
        n_alt = 12288
        print(f"  Trying alternate length n={n_alt}")
        a_alt = np.interp(
            np.linspace(0, 1, n_alt),
            np.linspace(0, 1, len(best_a)),
            best_a
        )
        a_alt = np.maximum(a_alt, 0)
        a_alt = a_alt / a_alt.sum()
        betas_alt = [50, 100, 200, 500, 1000]
        t_alloc_alt = min(remaining * 0.4, 180.0)
        sc_alt, a_opt_alt = annealed_lbfgs(a_alt, betas_alt, t_alloc_alt)
        report(sc_alt, a_opt_alt, f"P4 alt n={n_alt}")
        
        if sc_alt < best_score and remaining > 30:
            a_cd_alt = coord_descent_incremental(a_opt_alt, max_iters=5000, top_peaks=3, patience=300)
            sc_alt2 = compute_score(a_cd_alt)
            report(sc_alt2, a_cd_alt, f"P4 alt CD n={n_alt}")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 256

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
