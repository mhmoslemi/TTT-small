"""
Continuation-method optimizer for step function peak minimization.
Uses a geometric beta schedule on the LSE-smoothed objective with L-BFGS,
interleaved subgradient steps, and structured perturbations for ILS.
Focuses on polishing the parent solution to squeeze below 1.5027.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta, top_k=None):
    """LSE-smoothed score gradient with optional top-k restriction."""
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)

    if top_k is not None and top_k < len(c):
        idx = np.argpartition(c, -(top_k))[-(top_k):]
        threshold = c[idx].min()
        c_use = c.copy()
        c_use[c_use < threshold] = threshold

    c_max = float(np.max(c_use))
    scaled = beta * (c_use - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100.0))
    exp_sum = float(np.sum(exp_scaled))
    p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1].copy()
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return float(score), grad


def run_lbfgs(a0, beta, maxiter, maxfun, top_k=None, deadline=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    if top_k is not None:
        func = lambda x: lse_score_grad(x, beta, top_k)
    else:
        func = lambda x: lse_score_grad(x, beta)

    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S > 1e-12:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def nesterov_subgrad(a, n_iters=2000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent on the exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(np.sum(y))
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.2)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def structured_perturbation(a, rng, scale=0.02):
    """Structured perturbation targeting the peak lag contributors."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)

    a_p = a.copy()
    seg = a_p[j_start:j_end + 1]
    contrib = seg[::-1].copy()
    nn = len(contrib)

    strategy = int(rng.integers(0, 4))

    if strategy == 0:
        # Reduce top contributors to peak, boost bottom
        top_count = max(5, nn // 30)
        bot_count = max(5, nn // 30)
        top_local = np.argsort(contrib)[::-1][:top_count]
        bot_local = np.argsort(contrib)[:bot_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - scale * rng.random())
        deficit = a.sum() - a_p.sum()
        if deficit > 1e-10:
            per = deficit / len(bot_local)
            for li in bot_local:
                a_p[j_start + li] += per

    elif strategy == 1:
        # Gaussian noise
        a_p = a * (1.0 + scale * rng.standard_normal(n))

    elif strategy == 2:
        # Sinusoidal modulation
        freq = 1.0 + 5.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        x = np.arange(n, dtype=np.float64) / n
        a_p = a * (1.0 + scale * np.cos(2.0 * np.pi * freq * x + phase))

    else:
        # Block perturb
        n_blocks = 4 + int(rng.integers(0, 6))
        block_size = max(1, n // n_blocks)
        n_acts = min(n_blocks, n // block_size)
        for b in range(n_acts):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def continuation_run(a_init, beta_schedule, deadline, top_k_high=None):
    """
    Run continuation method: L-BFGS at each beta level,
    with occasional subgradient steps between levels.
    Returns best score and best a found.
    """
    best_sc = compute_score(a_init)
    best_a = a_init.copy()
    a_cur = a_init.copy()

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break

        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        # Determine maxiter based on remaining time and beta
        # Higher beta = harder = more iterations needed
        base_iter = 3000
        beta_factor = min(10.0, 10000.0 / max(beta, 100.0))
        maxiter = int(base_iter * max(1.0, beta_factor) * min(2.0, remaining / 300.0))
        maxiter = max(1000, min(50000, maxiter))
        maxfun = maxiter * 8

        # Choose top_k for high beta
        tk = None
        if top_k_high is not None and beta >= 10000:
            tk = top_k_high

        sc, a_opt, res = run_lbfgs(a_cur, beta, maxiter, maxfun, top_k=tk, deadline=deadline)

        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            a_cur = a_opt.copy()
        else:
            # No improvement; try a small subgradient nudge then retry
            a_nudged = nesterov_subgrad(a_cur, n_iters=200, f_lb=1.0)
            sc2 = compute_score(a_nudged)
            if sc2 < best_sc:

            elif time.monotonic() < deadline - 30:

                if sc3 < best_sc:
                    best_sc = sc3
                    best_a = a3.copy()
                    a_cur = a3.copy()

    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline - 20:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 20:
            break

        # Generate structured perturbation
        scale = 0.01 + 0.02 * rng.random()
        a_pert = structured_perturbation(best_a, rng, scale=scale)

        # Quick continuation from perturbation
        quick_betas = [1000, 5000, 20000, 80000]
        q_deadline = min(phase3_deadline - 15, time.monotonic() + max(10, remaining * 0.25))
        sc, a_opt = continuation_run(a_pert, quick_betas, q_deadline, top_k_high=5)
        report(sc, a_opt, f"ILS #{trial}")

        # If improved, do one more refinement
        if sc < best_score and time.monotonic() < phase3_deadline - 30:
            maxiter = int(min(30000, max(2000, (phase3_deadline - time.monotonic()) * 20)))
            maxfun = maxiter * 5
            sc2, a2, _ = run_lbfgs(a_opt, 200000, maxiter, maxfun, top_k=3)
            report(sc2, a2, f"ILS #{trial} polish")

    print(f"After Phase 3: best = {best_score:.10f}")

    # =================================================================
    # Phase 4: Final polish (5%)
    # =================================================================
    print(f"\n--- Phase 4: Final Polish ---")

    if time.monotonic() < deadline - 10:
        # Ultra-high beta L-BFGS
        maxiter = int(min(60000, max(5000, (deadline - time.monotonic()) * 30)))
        maxfun = maxiter * 5
        sc, a_opt, _ = run_lbfgs(best_a, 1000000, maxiter, maxfun, top_k=2)
        report(sc, a_opt, "final polish")

    # One last Nesterov
    if time.monotonic() < deadline - 5:
        a_ns = nesterov_subgrad(best_a, n_iters=500, f_lb=1.0)
        sc = compute_score(a_ns)
        report(sc, a_ns, "final nesterov")

    # =================================================================
    # Final output
    # =================================================================
    if best_a is None:
        out = [0.5] * 128
    else:
        out = [float(max(0.0, min(1000.0, x))) for x in best_a]
        if sum(out) < 0.01:
            out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
