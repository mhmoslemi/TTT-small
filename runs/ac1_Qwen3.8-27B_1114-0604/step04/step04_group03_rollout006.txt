def freq_peak_shave(a, n_iters=300, strength=0.5):
    """
    Frequency-domain peak shaving: directly reduce peaks in the convolution
    by modifying the spectrum, then project back to non-negative.
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return a.copy()
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        
        # Current convolution
        conv = np.fft.irfft(A * A, M_fft)[:2*n-1]
        
        # Find excess above median
        median = np.median(conv)
        mean = conv.mean()
        
        # Create target: reduce peaks above threshold
        threshold = mean + 0.3 * (conv.max() - mean)
        excess = np.maximum(conv - threshold, 0)
        
        if excess.max() < 1e-15:
            break
        
        # Modify spectrum: reduce energy at frequencies that contribute to peaks
        # The error signal in frequency domain
        err = -strength * excess / (2 * n - 1)
        err_pad = np.zeros(M_fft)
        err_pad[:2*n-1] = err
        err_F = np.fft.rfft(err_pad)
        
        # Update: A_new = A + delta, where delta is chosen to reduce the error
        # We want IFFT((A+delta)^2) ≈ IFFT(A^2) + IFFT(2*A*delta) = conv + IFFT(err_F)
        # So 2*A*delta ≈ err_F, i.e., delta ≈ err_F / (2*A)
        with np.errstate(divide='ignore', invalid='ignore'):
            delta = err_F / (2.0 * A)
        delta[~np.isfinite(delta)] = 0
        
        # Damping
        damp = 0.3 + 0.5 * min(1.0, it / 50.0)
        A_new = A + damp * delta
        
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def structured_perturbation(a, rng, mode=0):
    """Generate structured perturbations of a."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    
    if mode == 0:  # Gaussian noise
        scale = 0.01 + 0.05 * rng.random()
        return np.clip(a * (1 + scale * rng.standard_normal(n)), 0, 1000)
    elif mode == 1:  # Low-frequency sinusoidal
        freq = 1 + 5 * rng.random()
        phase = 2 * np.pi * rng.random()
        amp = 0.02 + 0.05 * rng.random()
        return np.clip(a * (1 + amp * np.cos(2 * np.pi * freq * x + phase)), 0, 1000)
    elif mode == 2:  # Block perturbation
        n_blocks = 2 + int(rng.integers(0, 10))
        a_p = a.copy()
        for _ in range(n_blocks):
            s = int(rng.integers(0, n))
            e = min(s + int(n * rng.random() * 0.1) + 1, n)
            a_p[s:e] *= 0.5 + rng.random()
        S = a_p.sum()
        return a_p / S if S > 1e-12 else a.copy()
    elif mode == 3:  # Peak-targeted reduction
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        a_p = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end > j_start:
            seg = a_p[j_start:j_end+1]
            # Reduce the top contributors
            top_k = max(3, n // 20)
            contrib = seg[::-1]
            top_idx = np.argsort(contrib)[-top_k:]
            for idx in top_idx:
                a_p[j_start + idx] *= 0.7
            # Add to bottom contributors
            bot_idx = np.argsort(contrib)[:top_k]
            total_red = a.sum() - a_p.sum()
            a_p[j_start + bot_idx] += total_red / len(bot_idx)
            S = a_p.sum()
            if S > 1e-12:
                a_p = a_p / S
        return a_p
    elif mode == 4:  # Symmetric perturbation
        a_p = a.copy()
        half = n // 2
        noise = rng.standard_normal(half) * 0.03
        a_p[:half] *= (1 + noise)
        a_p[half:n-half] *= (1 + noise[::-1])
        if n % 2 == 0:
            a_p[half:n-half] = a_p[half:n-half]  # symmetric middle
        a_p = np.clip(a_p, 0, 1000)
        S = a_p.sum()
        return a_p / S if S > 1e-12 else a.copy()
    else:  # Random block swap
        a_p = a.copy()
        s = int(rng.integers(0, n - 10))
        e = min(s + int(n * (0.05 + 0.1 * rng.random())), n)
        a_p[s:e] = a[s:e][::-1]
        S = a_p.sum()
        return a_p / S if S > 1e-12 else a.copy()


def blackman_harris_init(n):
    """Blackman-Harris window as initialization (known good sidelobe properties)."""
    x = np.arange(n, dtype=np.float64)
    a = (0.35875
         - 0.48829 * np.cos(2 * np.pi * x / (n - 1))
         + 0.14128 * np.cos(4 * np.pi * x / (n - 1))
         - 0.01168 * np.cos(6 * np.pi * x / (n - 1)))
    return np.maximum(a, 0)


def kaiser_init(n, beta_k=5.0):
    """Kaiser window initialization."""
    x = np.arange(n, dtype=np.float64)
    x = (x - (n - 1) / 2) / ((n - 1) / 2)
    x = np.abs(x)
    # Bessel function approximation (use scipy if available)
    from scipy.special import i0
    a = i0(beta_k * np.sqrt(np.maximum(1 - x**2, 0))) / i0(beta_k)
    return np.maximum(a, 0)


def nuttall_init(n):
    """Nuttall window initialization."""
    x = np.arange(n, dtype=np.float64)
    a = (0.3635819
         - 0.4891775 * np.cos(2 * np.pi * x / (n - 1))
         + 0.1365995 * np.cos(4 * np.pi * x / (n - 1))
         - 0.0106411 * np.cos(6 * np.pi * x / (n - 1)))
    return np.maximum(a, 0)


def coarse_to_fine(parent, target_n, rng):
    """Coarse-to-fine: optimize at small scale, upsample to target scale."""
    small_n = min(64, target_n)
    # Interpolate parent to small size
    if len(parent) != small_n:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, small_n)
        a_small = np.interp(x_new, x_old, parent)
    else:
        a_small = parent.copy()
    a_small = np.maximum(a_small, 0)
    
    # Optimize at small scale
    sc, a_small_opt = multi_beta_lbfgs(a_small, betas=[50, 200, 1000], total_time=60)
    
    # Upsample to target size
    x_old = np.linspace(0, 1, small_n)
    x_new = np.linspace(0, 1, target_n)
    a_target = np.interp(x_new, x_old, a_small_opt)
    
    # Add fine-scale details
    noise = 0.02 * rng.standard_normal(target_n)
    a_target = np.clip(a_target * (1 + noise), 0, 1000)
    S = a_target.sum()
    if S > 1e-12:
        a_target = a_target / S
    
    return a_target

"""
Focused optimizer for 2n*max(conv(a,a))/(sum(a))^2.
Strategy: hierarchical LSE smoothing, frequency-domain peak shaving,
and aggressive multi-restart local search starting from previous best.

import numpy as np

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_and_grad(a, beta):
    """Returns (LSE_score, gradient) for the scaled objective 2n*L/S^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    L = c_max + (scaled.max() + np.log(exp_s.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_s / exp_s.sum()
    
    # dL/da[i] = 2 * sum_j p[i+j] * a[j]
    af = a[::-1]
    full_len = 3 * n - 2
    M2 = 1 << (full_len - 1).bit_length()
    AF = np.fft.rfft(af, M2)
    PF = np.fft.rfft(p, M2)
    r = np.fft.irfft(AF * PF, M2)[:full_len]
    dL_da = 2.0 * r[n-1:2*n-1]
    
    grad = 2.0 * n * (dL_da / S**2 - 2.0 * L / S**3)
    return score, grad


def lbfgs_step(a0, beta, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0 = np.ones(n) * 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        s, g = lse_obj_and_grad(x, beta)
        return s, g
    
    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-16, 'gtol': 1e-12, 'maxiter': 5000, 'maxfun': maxfun})
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def freq_shave(a, n_iters=150, strength=0.5):
    """Reduce convolution peaks in frequency domain."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return a.copy()
    a = a / S
    M_fft = 1 << (2*n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        conv = np.fft.irfft(A*A, M_fft)[:2*n-1]
        
        mean_c = conv.mean()
        thresh = mean_c + 0.3*(conv.max() - mean_c)
        excess = np.maximum(conv - thresh, 0)
        if excess.max() < 1e-15:
            break
        
        err = -strength * excess
        err_pad = np.zeros(M_fft)
        err_pad[:2*n-1] = err
        err_F = np.fft.rfft(err_pad)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            delta = err_F / (2.0 * A)
        delta[~np.isfinite(delta)] = 0
        
        damp = 0.3 + 0.5*min(1.0, it/30.0)
        A_new = A + damp * delta
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a
