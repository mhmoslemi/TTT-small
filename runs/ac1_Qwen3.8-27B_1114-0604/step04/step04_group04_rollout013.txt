    x = cp.Variable(n)

LP-alternating optimizer with exact LP sub-solver (cvxpy/SCS) for 
moderate-length sequences, combined with FISTA-based approximate LP for

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import diags, csr_matrix
import time
import cvxpy as cp


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def build_conv_matrix(a, n):
    diagonal_data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = list(range(0, -(n), -1))
    return csr_matrix((diagonal_data, offsets), shape=(2 * n - 1, n))


def solve_lp_scs(a, M, n, max_iters=20000, eps=1e-8):
    C = build_conv_matrix(a, n)
    x = cp.Variable(n)
    constraints = [C @ x <= M, x >= 0]
    objective = cp.Maximize(cp.sum(x))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=max_iters, eps=eps)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return np.ones(n) / n
    if x.value is None:
        return np.ones(n) / n
    return np.maximum(np.asarray(x.value, dtype=np.float64), 0.0)

def fista_lp(a, M, n, n_iters=5000):
    """FISTA-based approximate LP solver."""
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    for it in range(n_iters):
        beta = 5.0 + 45.0 * min(1.0, it / (n_iters * 0.3))
        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L

        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def lp_alternating_exact(a, solver_func, max_iters=50, t_start=0.5, t_min=0.02, time_limit=None):
    """LP alternating with exact LP solver."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(max_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        t0 = time.monotonic()
        c = self_conv(a)
        M = float(c.max())
        x = solver_func(a, M, n)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        progress = it / max(1, max_iters)
        t = max(t_min, t_start * (1.0 - progress) ** 0.7)

        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new
        if it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      iter {it}: score={best_sc:.8f}, t={t:.3f}, time/iter={elapsed:.1f}s")

    return best_a, best_sc


def lse_score_and_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def error_reduction(a, n_iters=100, alpha=0.7):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(1 << (2 * n - 1).bit_length(), dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros_like(a_pad)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.5 + 0.5 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def make_initialization(n, init_type, rng):
    x = np.arange(n, dtype=np.float64) / n
    if init_type == 0:
        return np.ones(n) * 0.5
    elif init_type == 1:
        return np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01)
    elif init_type == 2:
        return np.maximum(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2, 0.01)
    elif init_type == 3:
        return np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0)
    elif init_type == 4:
        return 0.2 + 0.8 * (2.0 * x - 1.0) ** 2
    elif init_type == 5:
        return np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x), 0.01)
    elif init_type == 6:
        # U-shaped
        return 0.3 + 0.7 * (2.0 * x - 1.0) ** 2
    else:
        return np.ones(n) * 0.5


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, len={0 if parent is None else len(parent)}")

    # =================================================================
    # Phase 1: Quick initialization with error reduction (10% time)
    # =================================================================
    phase1_deadline = t_start + 0.10 * (budget_s - 30.0)
    print(f"\n--- Phase 1: Initialization ---")

    n_init = 2048
    for init_type in range(7):
        if time.monotonic() > phase1_deadline:
            break
        a0 = make_initialization(n_init, init_type, rng)
        a_er = error_reduction(a0, n_iters=80, alpha=0.7)
        sc = compute_score(a_er)
        report(sc, a_er, f"ER init={init_type}")

    # Try parent if available
    if parent is not None and time.monotonic() < phase1_deadline:
        n_p = min(len(parent), 4096)
        a_p = parent[:n_p] if len(parent) >= n_p else np.interp(
            np.linspace(0, 1, n_p), np.linspace(0, 1, len(parent)), parent)
        a_p = np.maximum(a_p, 0.0)
        sc = compute_score(a_p)
        report(sc, a_p, "parent")
