def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

"""
Improved step function optimizer using:
1. FFT-accelerated LSE gradient computation
2. Anti-palindromic structural bias to reduce center peak
3. Continuation method with adaptive beta schedule
4. Spectral shaping for diverse initializations
5. Peak shaving post-processing
6. FISTA-style accelerated gradient as alternative to L-BFGS

import numpy as np
from scipy.optimize import minimize
import time
import math

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_and_grad(a, beta):
    """
    LSE-smoothed score and gradient, computed efficiently via FFT.
    Returns (score, grad).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta

    score = 2.0 * n * L / (S * S)

    # Gradient of L w.r.t. a:
    # dL/da[j] = sum_k p[k] * dc[k]/da[j]
    # dc[k]/da[j] = a[k-j] + a[j] when 0 <= k-j < n, else appropriate partial
    # = T1[j] + a[j] * Sp[j]
    # where T1[j] = sum_k p[k]*a[k-j]*1{0<=k-j<n} (cross-correlation)
    # and Sp[j] = sum_k p[k]*1{0<=k-j<n} = sum of p over valid range

    # Sp via cumulative sum
    p_cs = np.zeros(2 * n)
    p_cs[:2*n-1] = p
    p_cs_cum = np.concatenate([[0.0], np.cumsum(p_cs)])
    Sp = np.zeros(n)
    for j in range(n):
        lo = j
        hi = j + n
        hi = min(hi, 2 * n - 1)
        Sp[j] = p_cs_cum[hi] - p_cs_cum[lo]

    # T1[j] = sum_i a[i] * p[i+j] for valid i, i+j
    # = (reversed_a * p)[n-1+j] (linear convolution)
    a_rev = a[::-1]
    M_t1 = n + (2 * n - 1) - 1
    M_t1_fft = 1 << (M_t1 - 1).bit_length()
    t1_full = np.fft.irfft(np.fft.rfft(a_rev, M_t1_fft) * np.fft.rfft(p, M_t1_fft), M_t1_fft)
    T1 = t1_full[n - 1: n - 1 + n]

    grad_L = T1 + a * Sp

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L * a / S3)

    return score, grad

def project_simplex(v, S):
    """Project v onto {x >= 0, sum x = S}."""
    n = len(v)
    if n == 0:
        return v
    # Bisection on tau: x_j = max(v_j - tau, 0), sum x_j = S
    lo = float(v.min()) - S
    hi = float(v.max())
    for _ in range(100):
        tau = (lo + hi) / 2.0
        x = np.maximum(v - tau, 0.0)
        s = x.sum()
        if s > S:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    x = np.maximum(v - tau, 0.0)
    # Correct for floating point
    diff = S - x.sum()
    if abs(diff) > 1e-12 and x.sum() > 0:
        x += diff / n
        x = np.maximum(x, 0.0)
    return x

def fista_lse(a0, beta, time_budget, t_start, max_iter=500000):
    """
    FISTA (accelerated proximal gradient) for LSE objective with simplex constraint.
    a is constrained to {a >= 0, sum a = S} where S = sum(a0).
    """
    n = len(a0)
    S = float(a0.sum())
    a = a0.copy()
    y = a.copy()
    t_k = 1.0
    best_score = compute_score(a)
    best_a = a.copy()

    # Estimate Lipschitz constant via line search
    L_ip = None
    deadline = t_start + time_budget

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        sc_y, grad_y = lse_score_and_grad(y, beta)

        # Adaptive step size (backtracking)
        if L_ip is None:
            L_ip = max(1e-6, 2.0 * n * beta * np.linalg.grad_norm if hasattr(np, 'grad_norm') else np.linalg.norm(grad_y) * 10.0 / max(1, n))
            L_ip = max(1.0, np.linalg.norm(grad_y) * 20.0 / max(1.0, np.linalg.norm(a)))

        step = 1.0 / L_ip

        # Gradient step
        z = y - step * grad_y

        # Project onto simplex {a >= 0, sum a = S}
        a_new = project_simplex(z, S)

        # FISTA momentum
        t_next = (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + (t_k - 1.0) / t_next * (a_new - a)
        y = np.maximum(y, 0.0)
        # Renormalize y to maintain sum
        Sy = y.sum()
        if Sy > 1e-12:
            y = y * S / Sy
        t_k = t_next
        a = a_new

        if it % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a

def run_lbfgs(a0, beta, time_budget, t_start, maxiter=5000, maxfun=50000):
    """L-BFGS-B with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    deadline = t_start + time_budget

    def obj_and_grad(x):
        if time.monotonic() > deadline:
            raise TimeoutError
        return lse_score_and_grad(x, beta)

    try:
        res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True,
                       bounds=bounds,
                       options={'ftol': 1e-15, 'gtol': 1e-12,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40})
    except (TimeoutError, Exception):
        return compute_score(a0), a0

    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out

def peak_shave(a, n_iterations=200, strength=0.02):
    """
    Iteratively reduce the peak of self-convolution by adjusting elements
    that contribute most to the peak.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    best_score = compute_score(a)
    best_a = a.copy()

    for it in range(n_iterations):
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = c[k_star]

        # Find pairs contributing to c[k_star]
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je < js:
            break

        # Products a[j] * a[k_star - j]
        idx1 = np.arange(js, je + 1)
        idx2 = k_star - idx1
        products = a[idx1] * a[idx2]

        # Find the top contributors
        top_n = min(20, len(idx1))
        top_idx = np.argsort(products)[::-1][:top_n]

        # Reduce elements in heavy pairs (reduce the larger one)
        for ti in top_idx:
            j1 = int(idx1[ti])
            j2 = int(idx2[ti])
            if a[j1] >= a[j2]:
                a[j1] *= (1.0 - strength)
            else:
                a[j2] *= (1.0 - strength)

        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new > 1e-12:
            a = a * (S / S_new)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        else:
            # Revert if it got worse
            a = best_a.copy()

    return best_score, best_a

def anti_palin_init(n, rng, base='chirp'):
    """
    Generate an anti-palindromic initial sequence of length n.
    a[j] = mu + d[j] for j < n/2, a[j] = mu - d[n-1-j] for j >= n/2
    """
    m = n // 2
    d = np.zeros(m)

    if base == 'chirp':
        x = np.arange(m, dtype=np.float64)
        freq = rng.uniform(2.0, 8.0)
        d = np.sin(2.0 * np.pi * freq * x / n) * 0.3
    elif base == 'linear':
        x = np.arange(m, dtype=np.float64)
        d = (x / m - 0.5) * 0.8
    elif base == 'random':
        d = rng.standard_normal(m) * 0.2
    elif base == 'sinusoidal':
        x = np.arange(m, dtype=np.float64)
        freq = rng.uniform(1.0, 5.0)
        d = np.sin(2.0 * np.pi * freq * x / n + rng.uniform(0, 2*np.pi)) * 0.3
    else:
        d = rng.standard_normal(m) * 0.15

    # Ensure |d[j]| < mu for non-negativity
    mu = 1.0 / n
    scale = 0.9 * mu / max(np.max(np.abs(d)), 1e-10)
    d = d * scale
    mu = 1.0 / n

    a = np.zeros(n)
    a[:m] = mu + d
    if n % 2 == 0:
        a[m:] = mu - d[::-1]
    else:
        a[m] = mu
        a[m+1:] = mu - d[::-1]

    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 0:
        a = a * (math.sqrt(2 * n) / S)
    return a

def spectral_init(n, rng, target='flat'):
    """
    Generate a sequence with controlled spectral properties.
    """
    if target == 'flat':
        # Random phases with flat magnitude spectrum
        M_fft = 1 << (2 * n - 2).bit_length()
        A_mag = np.ones(M_fft // 2 + 1)
        A_phase = rng.standard_normal(M_fft // 2 + 1) * np.pi / 2
        A = A_mag * np.exp(1j * A_phase)
        A[0] = np.abs(A[0])  # Keep DC real and positive
        a = np.fft.irfft(A, M_fft)[:n]
    elif target == 'lowpass':
        # Concentrate energy at low frequencies
        M_fft = 1 << (2 * n - 2).bit_length()
        freqs = np.fft.rfftfreq(M_fft)
        A_mag = np.exp(-n * freqs * rng.uniform(1.0, 5.0))
        A_phase = rng.standard_normal(M_fft // 2 + 1) * np.pi / 2
        A = A_mag * np.exp(1j * A_phase)
        A[0] = np.abs(A[0])
        a = np.fft.irfft(A, M_fft)[:n]
    elif target == 'bandpass':
        # Energy in a middle frequency band
        M_fft = 1 << (2 * n - 2).bit_length()
        freqs = np.fft.rfftfreq(M_fft)
        f_center = rng.uniform(0.05, 0.3)
        f_width = rng.uniform(0.05, 0.15)
        A_mag = np.exp(-((freqs - f_center) / f_width) ** 2)
        A_phase = rng.standard_normal(M_fft // 2 + 1) * np.pi / 2
        A = A_mag * np.exp(1j * A_phase)
        A[0] = max(A[0].real, 0)
        a = np.fft.irfft(A, M_fft)[:n]
    else:
        a = rng.standard_normal(n) + 1.0

    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 1e-12:
        a = a * (math.sqrt(2 * n) / S)
    else:
        a = np.ones(n) * math.sqrt(2.0 / n)
    return a

def quadratic_residue_init(n, rng):
    """
    Initialize using quadratic residues mod p (difference set structure).
    """
    # Find a prime p close to n with p ≡ 3 (mod 4)
    p = n
    while p > 10:
        if p % 4 == 3:
            # Check if p is prime (simple trial division)
            is_prime = True
            for i in range(2, int(math.isqrt(p)) + 1):
                if p % i == 0:
                    is_prime = False
                    break
            if is_prime:
                break
        p -= 1

    if p <= 10:
        p = 7

    # Quadratic residues mod p
    residues = set()
    for i in range(1, p):
        residues.add((i * i) % p)

    # Create sequence: 1 at residue positions, 0 elsewhere, then extend to length n
    a_p = np.zeros(p)
    for r in residues:
        a_p[r] = 1.0
    a_p[0] = 1.0  # Include 0

    # Extend to length n by interpolation
    if n > p:
        x_old = np.arange(p, dtype=np.float64)
        x_new = np.linspace(0, p - 1, n)
        a = np.interp(x_new, x_old, a_p)
    else:
        a = a_p[:n].copy()

    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 1e-12:
        a = a * (math.sqrt(2 * n) / S)
    return a

def refinement_pipeline(a_init, time_budget, t_start):

    a_pert = np.maximum(a_pert, 0.0)
    S = a_pert.sum()
    a_pert = a_pert * (best_a.sum() / S)
    init_strategies.append(("parent-perturb", a_pert))

    for si, (name, a_init) in enumerate(init_strategies):
        if time.monotonic() > p2_deadline - 15:
            break
        t_remain = p2_deadline - time.monotonic()
        t_alloc = min(t_remain * 0.9, per_start_time * 1.3)
        if t_alloc < 10:
            break

        print(f"  Start {si+1}/{len(init_strategies)}: {name} (alloc={t_alloc:.0f}s)")
        sc_init = compute_score(a_init)
        print(f"    Initial score: {sc_init:.6f}")

        sc, a_opt = refinement_pipeline(a_init, t_alloc, t_start)
        try_update(sc, a_opt, f"init: {name}")
        print(f"    After refinement: {best_score:.8f}")

    # ================================================================
    # Phase 3: Intensive optimization of best (15% budget)
    # ================================================================
    p3_budget = effective_budget * 0.15
    print(f"\n{'='*60}")
    print(f"Phase 3: Intensive optimization (budget={p3_budget:.0f}s)")
    print(f"{'='*60}")

    p3_deadline = t_start + (time.monotonic() - t_start) + p3_budget

    # High-beta L-BFGS
    for beta in [5000, 20000, 50000]:
        if time.monotonic() > p3_deadline - 10:
            break
        t_alloc = min(p3_deadline - time.monotonic() - 5, 25.0)
        if t_alloc < 5:
            break
        sc, a_opt = run_lbfgs(best_a, beta, time_budget=t_alloc, t_start=t_start,
                              maxiter=5000, maxfun=30000)
        try_update(sc, a_opt, f"Intensive LBFGS beta={beta}")

    # Peak shaving with more iterations
    if time.monotonic() < p3_deadline - 5:
        t_alloc = min(p3_deadline - time.monotonic() - 2, 30.0)
        n_shaves = int(max(100, t_alloc * 8))
        sc, a_opt = peak_shave(best_a, n_iterations=n_shaves, strength=0.01)
        try_update(sc, a_opt, "Intensive peak shave")

        # L-BFGS after peak shaving
        if time.monotonic() < p3_deadline - 5:
            sc, a_opt = run_lbfgs(best_a, 50000, time_budget=min(15.0, p3_deadline - time.monotonic() - 3),
                                  t_start=t_start, maxiter=3000, maxfun=20000)
            try_update(sc, a_opt, "LBFGS after shave")

    # ================================================================
    # Phase 4: Try different lengths (10% budget)
    # ================================================================
    p4_budget = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 4: Different lengths (budget={p4_budget:.0f}s)")
    print(f"{'='*60}")

    p4_deadline = t_start + (time.monotonic() - t_start) + p4_budget
    resample_targets = []
    for factor in [0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
        tl = int(n_best * factor)
        if 512 <= tl <= 40000:
            resample_targets.append(tl)

    for tl in resample_targets[:4]:
        if time.monotonic() > p4_deadline - 15:
            break
        t_left = p4_deadline - time.monotonic()
        if t_left < 15:
            break

        x_old = np.arange(n_best, dtype=np.float64)
        x_new = np.linspace(0, n_best - 1, tl)
        a_res = np.interp(x_new, x_old, best_a)
        a_res = np.clip(a_res, 0.0, 1000.0)
        S = a_res.sum()
        if S > 0:
            a_res = a_res * (best_a.sum() / S)

        t_alloc = min(t_left * 0.8, 30.0)
        sc, a_opt = refinement_pipeline(a_res, t_alloc, t_start)
        try_update(sc, a_opt, f"resample n={tl}")

    # ================================================================
    # Phase 5: Final polish (remaining time)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    final_deadline = t_start + effective_budget - 5.0

    # Final high-precision L-BFGS
    if time.monotonic() < final_deadline - 10:
        t_alloc = min(final_deadline - time.monotonic() - 5, 20.0)
        sc, a_opt = run_lbfgs(best_a, 100000, time_budget=t_alloc, t_start=t_start,
                              maxiter=10000, maxfun=50000)
        try_update(sc, a_opt, "Final LBFGS")

    # Final peak shaving
    if time.monotonic() < final_deadline - 3:
        t_alloc = min(final_deadline - time.monotonic() - 2, 15.0)
        n_shaves = int(max(200, t_alloc * 10))
        sc, a_opt = peak_shave(best_a, n_iterations=n_shaves, strength=0.008)
        try_update(sc, a_opt, "Final peak shave")

    # One more L-BFGS at very high beta
    if time.monotonic() < final_deadline - 3:
        t_alloc = min(final_deadline - time.monotonic() - 2, 10.0)
        if t_alloc > 3:
            sc, a_opt = run_lbfgs(best_a, 200000, time_budget=t_alloc, t_start=t_start,
                                  maxiter=5000, maxfun=20000)
            try_update(sc, a_opt, "Ultra-final LBFGS")

    # ================================================================
    # Final output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")

    return out
