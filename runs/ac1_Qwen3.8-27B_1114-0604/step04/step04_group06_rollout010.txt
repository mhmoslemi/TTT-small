    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(c.argmax())
        
        sg = subgrad_step(a, k_star)
        eta = eta0 / (1.0 + 0.01 * np.log1p(t))
        
        v = momentum * v - eta * sg
        a_new = a + v
        a = project_simplex(a_new)
        
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_score, best_a


def lse_objective_and_grad(a, beta):
    """LSE-smoothed score and exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum() + 1e-300)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / (exp_s.sum() + 1e-300)
    
    # cross[m] = sum_k p[k] * a[k-m] for valid indices
    # = cross-correlation of p and a
    flip_p = p[::-1]
    M3 = 3 * n - 2
    M3_fft = 1 << (M3 - 1).bit_length()
    A = np.fft.rfft(a, M3_fft)
    FP = np.fft.rfft(flip_p, M3_fft)
    conv_ap = np.fft.irfft(A * FP, M3_fft)[:3 * n - 2]
    cross = conv_ap[n - 1: 2 * n - 1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (4.0 * n * cross) / S2 - (4.0 * n * L) / S3
    return score, grad


def ssq_objective_and_grad(a, target=None):
    """Sum-of-squares of (conv - target) plus a max penalty."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    a_norm = a / S
    c_norm = c / (S * S)
    if target is None:
        target = 1.0 / (2 * n - 1)
    c_diff = c_norm - target
    obj = float(np.sum(c_diff * c_diff)) + 10.0 * float(np.maximum(c_diff, 0).max())
    
    # Gradient of sum(c_diff^2) w.r.t. a:
    # d/dc_k [sum(c_diff^2)] = 2 * c_diff[k]
    # dc_k/da_m = 2*a_norm[k-m] (for normalized a)
    # But we need d/da (not da_norm)
    # c_norm[k] = c[k] / S^2
    # d(c_norm[k])/da_m = 2*a[m-related] / S^2 - 2*S_m * c[k] / S^3
    # This gets complicated. Let me just use the score gradient approach.
    
    # Actually, let me simplify: just use the LSE gradient with a large beta
    # which approximates the max, and add a small sum-of-squares term.
    # For now, return a placeholder.
    grad = np.zeros(n)
    return obj, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """L-BFGS-B optimization with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, None)
        s = x.sum()
        if s < 1e-12:
            return 1e10, np.zeros(n)
        sc, g = lse_objective_and_grad(x, beta)
        return sc, g
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        obj_and_grad, a0, jac=True,
        method='L-BFGS-B', bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coord_descent(a, top_k=50, n_sweeps=20, time_limit_s=None, t_start=None):
    """Coordinate descent on the exact (non-smooth) objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if time_limit_s else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = a.sum()
        score = 2.0 * n * c.max() / (S * S)
        
        # Find coordinates most likely to improve
        k_star = int(c.argmax())
        sg = subgrad_step(a, k_star)
        order = np.argsort(np.abs(sg))[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-15:
                continue
            # Try a few multipliers
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3]:
                a[j] = a_old * mult
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    # Keep this value
                else:
                    a[j] = a_old
                    break
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a
