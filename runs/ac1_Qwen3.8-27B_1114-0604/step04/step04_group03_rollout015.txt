"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: adaptive LSE with very high beta, frequency-domain peak targeting,
momentum-based optimization, multi-length exploration, and structured
perturbation search. Key innovations: peak-targeting gradient, 
momentum L-BFGS, and adaptive beta annealing.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and its exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()  # numerical stability
    exp_s = np.exp(scaled)
    exp_sum = exp_s.sum()
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_sum
    # dL/da[i] = 2 * sum_j p[i+j] * a[j] (with appropriate range)
    # conv_k = sum_j a[j] * a[k-j], so dconv_k/da[i] = a[k-i] (if 0<=k-i<n)
    # dL/da[i] = sum_k p_k * a[k-i] (summed over valid k-i)
    # This is the cross-correlation of p and a, evaluated at specific lags.
    # dL/da[i] = sum_k p_k * a[k-i] where 0 <= k-i < n
    # = sum_{j=0}^{n-1} a[j] * p[i+j]  (substituting j = k-i)
    # So dL/da = cross-correlation of a and p, specifically (a ⋆ p)[i] = sum_j a[j]*p[i+j]
    # This equals the convolution of reverse(a) and p, shifted.
    rev_a = a[::-1]
    M2 = 2 * n  # enough for the cross-correlation
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)
    # r[k] = sum_j rev_a[j] * p[k-j] = sum_j a[n-1-j] * p[k-j]
    # We want dL/da[i] = sum_j a[j] * p[i+j]
    # Let's verify: r[n-1+i] = sum_j a[n-1-j] * p[n-1+i-j]
    # Let m = n-1-j, so j = n-1-m:
    # = sum_m a[m] * p[i+m]  ✓
    dL_da = 2.0 * r[n - 1:2 * n - 1]
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=500000):

    n = len(a0)

    if a0.sum() < 1e-6:

        return s, g

    res = minimize(

    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

Actually, one more important thought: **the relationship between the sequence and its convolution in the frequency domain.**

If `A = fft(a, M)` (zero-padded to length M >= 2n-1), then `conv = ifft(|A|^2)[:2n-1]`.

The peak of `conv` is the maximum of `ifft(|A|^2)`. We want this to be as flat as possible.

Now, `|A|^2` is the power spectrum. If we could make `|A|^2` a constant (i.e., `a` is white noise), then `conv` would be a delta function (only the DC component would be non-zero). But that's not possible with `a >= 0`.

With `a >= 0`, the spectrum `A(f)` has the property that it's the Fourier transform of a non-negative function. By the Bochner theorem, `|A(f)|^2` (the power spectrum) must be a positive definite function.

The constraint that `a >= 0` translates to: `ifft(A) >= 0` at all sample points. This is a set of linear constraints on `A` (since ifft is linear).

So the problem is: minimize `max(ifft(|A|^2))` subject to `ifft(A) >= 0` and `A[0] = sum(a) = 1` (DC constraint).

This is a non-convex problem because of the quadratic `|A|^2` term. But it's a structured non-convex problem.

One approach: **alternating minimization**
- Fix the "shape" of `|A|^2` (amplitudes), optimize the phases of `A` to satisfy `ifft(A) >= 0`
- Fix the phases, optimize the amplitudes to flatten `ifft(|A|^2)`

This is similar to the Gerchberg-Saxton algorithm in optics!

**Gerchberg-Saxton style approach:**
1. Start with some `a >= 0`, compute `A = fft(a)`
2. In the "object plane" (time domain): enforce `a >= 0` (set negative values to 0) and `sum(a) = 1`
3. In the "far field" (frequency domain): adjust `|A|^2` to be more flat (while preserving the phase of `A`)
   - New `|A|^2`: flatten toward the mean, e.g., `|A_new|^2 = |A|^2 * (mean / |A|^2)^alpha` for some alpha in (0,1)
   - Or: `|A_new|^2 = |A|^2 + beta * (mean - |A|^2)` (move toward constant)
4. Combine: `A_new = new_amplitude * exp(i * phase(A))`
5. Go to step 2

def gs_optimize(a0, n_iters=500, alpha=0.5):
    """Gerchberg-Saxton style optimization."""
    n = len(a0)
    M = 1 << (2*n - 2).bit_length()
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        # Forward: time -> freq
        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.fft(a_pad, M)
        
        # In frequency domain: flatten |A|^2
        amp2 = np.abs(A)**2
        amp2_mean = amp2.mean()
        # Move amplitudes toward constant
        new_amp2 = amp2 * (amp2_mean / (amp2 + 1e-30))**alpha
        # Or simpler: new_amp2 = amp2 + 0.1*(mean - amp2)
        new_amp = np.sqrt(new_amp2)
        
        # Preserve phase, get new amplitudes
        phase = np.angle(A)
        A_new = new_amp * np.exp(1j * phase)
        
        # Backward: freq -> time
        a_new = np.fft.ifft(A_new, M).real[:n]
        
        # Enforce non-negativity and normalize
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:

        a_new = a_new / S
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

"""
Optimized search for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Strategy: Progressive L-BFGS with very high beta LSE smoothing,
Gerchberg-Saxton frequency-domain flattening, peak-shaving coordinate
descent, and diverse multi-length exploration. Key innovations over
previous approach: GS-style amplitude flattening, adaptive beta schedule,
peak-shaving local search, and smarter time allocation.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and its exact gradient. Score = 2n*L/S^2 where L is smoothed max of conv."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    exp_sum = exp_s.sum()
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_sum
    rev_a = a[::-1]
    M2 = 2 * n + 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)
    dL_da = 2.0 * r[n - 1:2 * n - 1]
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L * dL_da / (S * S * S) - 2.0 * n * L * 2.0 / (S * S * S) * np.ones(n) * 0.0)
    # Correct gradient: d(2n*L/S^2)/da_i = 2n*(dL_da_i/S^2 - 2*L/S^3)
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def error_reduction(a, n_iters=200):
    """Frequency-domain error reduction: iteratively flatten the convolution spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        v = A * A
        conv = np.fft.irfft(v, M)[:2 * n - 1]
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-16:
            break
        progress = it / max(1, n_iters)
        alpha = 0.3 + 0.5 * progress
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-16:

        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv

        err_spec = np.fft.rfft(err_pad, M)
        denom = 2.0 * A + 1e-15
        with np.errstate(divide='ignore', invalid='ignore'):

        A_new = A + damp * d

        a_new = a_new / a_new.sum()
        sc = compute_score(a_new)

    n = len(a0)
    M = 1 << (2 * n - 1).bit_length()

def peak_shave(a, n_sweeps=10):
    """Coordinate descent: for each coordinate, try reducing it to lower the peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        # Determine which coordinates contribute to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        # Try reducing top contributors
        top_count = min(20, len(contrib))
        top_idx = np.argsort(contrib)[::-1][:top_count]
        improved = False
        for li in top_idx:
            gi = j_start + li
            old_val = a[gi]
            if old_val < 1e-15:
                continue
            # Try reducing by 5%
            test_val = old_val * 0.95
            a[gi] = test_val
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                improved = True
            else:
                a[gi] = old_val
        # If we improved, redistribute to low-contributor positions
        if improved:
            bottom_count = min(10, len(contrib) - top_count)
            if bottom_count > 0:
                bottom_idx = np.argsort(contrib)[:bottom_count]
                for li in bottom_idx:
                    gi = j_start + li
                    a[gi] += a[gi] * 0.02 + 1e-8
        S = a.sum()
        if S > 1e-12:
            a = a / S
    return best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, np.maximum(parent, 0)).copy())
    x = np.arange(n, dtype=np.float64) / n
    # Hann window
    inits.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    # Blackman
    inits.append(0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))
    # Raised cosine
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * x), 0.01))
    # Dual peak
    inits.append(0.3 + 0.7 * np.cos(4 * np.pi * x) ** 2)
    # Inverted (large in middle)
    inits.append(0.2 + 0.8 * np.sin(np.pi * x) ** 2)
    # Quasi-random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    # Block structure
    nb = 8
    per = max(1, n // nb)
    vals = 0.3 + 0.7 * rng.random(nb)
    a = np.repeat(vals, per)
    if len(a) < n:
        a = np.concatenate([a, vals[:n - len(a)]])
    inits.append(a[:n].copy())
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 20.0

    # Get parent if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 32:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n in [4096, 8192, 12288, 16384]:
        if n not in lengths and n <= 16384:
            lengths.append(n)
    # Remove very large ones if parent is small
    lengths = sorted(set(lengths))
    if len(lengths) > 4:
        lengths = lengths[:4]

    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}")

    # Phase 1: Quick exploration to find best length (15% time)
    phase1_end = t_start + 0.15 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 1: Exploration (t=0 to {phase1_end - t_start:.0f}s) ===")

    best_per_length = {}
    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  n={n}")
        inits = build_inits(n, rng, parent)
        # Try at most 3 inits per length in this phase
        inits = inits[:3]
        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-6:
                a0 = np.ones(n) * 0.5
            a0 = a0 / a0.sum()
            # Quick L-BFGS
            sc, a_opt = run_lbfgs(a0, beta=200, maxiter=3000, maxfun=50000)
            if n not in best_per_length or sc < best_per_length[n][0]:
                best_per_length[n] = (sc, a_opt)
            report(sc, a_opt, f"P1 n={n} i={i}")

    if not best_per_length:
        # Fallback
        a0 = np.ones(4096) * 0.5
        sc, a_opt = run_lbfgs(a0, beta=200, maxiter=5000, maxfun=100000)
        best_per_length[4096] = (sc, a_opt)
        report(sc, a_opt, "fallback")

    # Pick the best length
    best_len = min(best_per_length.keys(), key=lambda k: best_per_length[k][0])
    best_start = best_per_length[best_len][1]
    report(best_per_length[best_len][0], best_start, f"best_len={best_len}")
    n_best = len(best_a)
    print(f"Selected length: {n_best}, current best: {best_score:.8f}")

    # Phase 2: Progressive L-BFGS on best length (50% time)
    phase2_end = t_start + 0.65 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 2: Progressive L-BFGS (t=15% to 65%) ===")

    beta_schedule = [500, 1000, 2000, 5000, 10000, 20000, 50000]
    for beta in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        remaining_frac = (phase2_end - time.monotonic()) / max(1e-6, phase2_end - t_start)
        maxiter = int(15000 * remaining_frac)
        maxfun = int(200000 * remaining_frac)
        maxiter = max(maxiter, 2000)
        maxfun = max(maxfun, 20000)
        print(f"  L-BFGS beta={beta}, maxiter={maxiter}")
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"L-BFGS b={beta}")
        #穿插 a quick peak shave after each L-BFGS
        a_shaved = peak_shave(a_opt, n_sweeps=5)
        sc_shave = compute_score(a_shaved)
        report(sc_shave, a_shaved, f"shave after b={beta}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # Phase 3: GS optimization + error reduction (15% time)
    phase3_end = t_start + 0.80 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 3: GS + Error Reduction ===")

    if time.monotonic() < phase3_end:
        a_gs = gs_optimize(best_a, n_iters=300, alpha=0.2)
        sc = compute_score(a_gs)
        report(sc, a_gs, "GS")
        if time.monotonic() < phase3_end:
            sc, a_opt = run_lbfgs(a_gs, beta=5000, maxiter=5000, maxfun=100000)
            report(sc, a_opt, "GS+LBFGS")

    if time.monotonic() < phase3_end:
        a_er = error_reduction(best_a, n_iters=200)
        sc = compute_score(a_er)
        report(sc, a_er, "ER")
        if time.monotonic() < phase3_end:
            sc, a_opt = run_lbfgs(a_er, beta=5000, maxiter=5000, maxfun=100000)
            report(sc, a_opt, "ER+LBFGS")

    # Phase 4: Local search with perturbations (15% time)
    phase4_end = t_start + 0.95 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 4: Local Search ===")

    n_best = len(best_a)
    n_trials = 15
    for trial in range(n_trials):
        if time.monotonic() > phase4_end:
            break
        strategy = trial % 5
        if strategy == 0:
            scale = 0.005 + 0.02 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted peak reduction
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            a_pert = best_a.copy()
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            seg = a_pert[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = max(5, n_best // 100)
            top_local = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a_pert[gi] * 0.02
                a_pert[gi] = max(a_pert[gi] - red, 0.0)
                total_red += red
            if total_red > 0:
                bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
                per = total_red / len(bottom_local)
                for li in bottom_local:
                    a_pert[j_start + li] += per
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S
        elif strategy == 2:
            # Sinusoidal perturbation at peak-related frequency
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            freq = k_star / n_best
            phase = 2.0 * np.pi * rng.random()
            amp = 0.005 + 0.015 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * n_best * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # GS perturbation
            a_gs_pert = best_a * (1.0 + 0.02 * rng.standard_normal(n_best))
            a_gs_pert = np.maximum(a_gs_pert, 0.0)
            a_pert = gs_optimize(a_gs_pert, n_iters=100, alpha=0.3)
        else:
            # Random block scaling
            a_pert = best_a.copy()
            n_blocks = 4 + int(rng.integers(0, 8))
            block_size = max(1, n_best // n_blocks)
            for b in range(min(n_blocks, n_best // block_size)):
                s = b * block_size
                e = min(s + block_size, n_best)
                factor = 0.8 + 0.4 * rng.random()
                a_pert[s:e] *= factor
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S

        # Short L-BFGS from perturbed point
        sc, a_opt = run_lbfgs(a_pert, beta=3000, maxiter=3000, maxfun=50000)
        report(sc, a_opt, f"local#{trial} s={strategy}")

    # Phase 5: Final polish (5% time)
    print(f"\n=== Phase 5: Final Polish ===")
    n_best = len(best_a)

    if time.monotonic() < deadline - 10:
        print(f"  Final L-BFGS beta=50000")
        sc, a_opt = run_lbfgs(best_a, beta=50000, maxiter=10000, maxfun=150000)
        report(sc, a_opt, "final LBFGS b=50000")

    if time.monotonic() < deadline - 10:
        print(f"  Final L-BFGS beta=100000")
        sc, a_opt = run_lbfgs(best_a, beta=100000, maxiter=5000, maxfun=80000)
        report(sc, a_opt, "final LBFGS b=100000")

    if time.monotonic() < deadline - 5:
        a_shaved = peak_shave(best_a, n_sweeps=20)
        sc = compute_score(a_shaved)
        report(sc, a_shaved, "final shave")

    if time.monotonic() < deadline - 5:
        a_er = error_reduction(best_a, n_iters=100)
        sc = compute_score(a_er)
        report(sc, a_er, "final ER")
        if time.monotonic() < deadline - 3:
            sc, a_opt = run_lbfgs(a_er, beta=50000, maxiter=3000, maxfun=50000)
            report(sc, a_opt, "ER+final LBFGS")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 1024

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
