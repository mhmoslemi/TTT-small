    n = len(f)
    b = f.copy()  # initial feasible point
    S = sum(b)
    b = b / S  # normalize
    # Actually, we want to maximize sum(b) s.t. C@b <= M, b >= 0
    # b = f is feasible since C@f = conv(f,f) <= M
    
    active = set()  # indices j where (f*b)_j = M
    # Find initial active set
    conv_fb = fft_convolve(f, b)
    active = set(np.where(conv_fb >= M * 0.999)[0])
    
    for it in range(max_iters):
        # Solve QP: max sum(b) s.t. (C_A b)_j <= M for j in A, b >= 0
        # This is: max 1^T b s.t. sum_{i} f[j-i] b[i] <= M for j in A, b >= 0
        # Can be solved as a QP or LP
        
        # Use scipy.optimize.linprog for the LP (since objective is linear)
        # Variables: b_0, ..., b_{n-1}
        # Constraints: for each j in A: sum_i C[j,i] b_i <= M
        #             b_i >= 0
        
        # Build the constraint matrix for active set
        A_mat = np.zeros((len(active), n))
        for idx, j in enumerate(active):
            for i in range(max(0, j-n+1), min(j+1, n)):
                A_mat[idx, i] = f[j-i]
        
        # Solve LP
        from scipy.optimize import linprog
        res = linprog(-np.ones(n), A_ub=A_mat, b_ub=np.ones(len(active))*M, 
                      bounds=[(0, None)]*n, method='highs')
        if res.success:
            b_new = res.x
        else:
            break
        
        # Check if all constraints are satisfied
        conv_new = fft_convolve(f, b_new)
        violated = np.where(conv_new > M * 1.001)[0]
        
        if len(violated) == 0:
            break  # Found optimal
        
        # Update active set
        for j in violated:
            active.add(int(j))
        # Remove constraints that are not tight
        for j in list(active):
            if conv_new[j] < M * 0.99:
                active.discard(j)
    
    return b

PEN_SUM = 1e-3


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_conv(a, b):
    """Compute linear convolution of a and b using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n, m = int(a.size), int(b.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    out = np.fft.irfft(A * B, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))

        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def lse_score_grad_tuple(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, grad
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, grad
    z = beta * (d - dmax)
    z = np.clip(z, -500, 500)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, grad
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) * (S - 1.0)
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad

def build_conv_constraint_matrix(f, lags, n):
    """Build sparse matrix where row i corresponds to lag lags[i]."""
    rows = []
    cols = []
    vals = []
    for idx, j in enumerate(lags):
        j = int(j)
        i_start = max(0, j - n + 1)
        i_end = min(j + 1, n)
        for i in range(i_start, i_end):
            val = f[j - i]
            if val > 1e-14:
                rows.append(idx)
                cols.append(i)
                vals.append(val)
    if not rows:
        return sparse.csr_matrix((0, n))
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(len(lags), n))
    return A


def lp_direction(f, M, top_k=20, deadline=None):
    """
    Solve LP: maximize sum(b) s.t. (f*b)_j <= M for j in active set, b >= 0.
    Returns b or None if LP fails.
    """
    n = len(f)
    if n < 4:
        return None
    conv_f = cross_conv(f, f)
    if len(conv_f) == 0:
        return None
    conv_max = float(np.max(conv_f))
    if conv_max <= 0:
        return None
    # Use lags where convolution is at or near the maximum
    thresh = conv_max * 0.999
    active = np.where(conv_f >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv_f))])
    if len(active) > top_k:
        # Keep the top_k largest
        order = np.argsort(conv_f[active])[::-1][:top_k]
        active = active[order]
    
    A_mat = build_conv_constraint_matrix(f, active, n)
    if A_mat.nnz == 0:
        return None
    
    c_vec = -np.ones(n)
    b_ub = np.full(len(active), M, dtype=np.float64)
    
    try:
        if deadline is not None and time.monotonic() > deadline:
            return None
        res = linprog(c_vec, A_ub=A_mat, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': 30})
    except Exception:
        return None
    
    if res.success and res.x is not None:
        b = np.maximum(res.x, 0.0)
        S = float(b.sum())
        if S > 1e-10:
            return b / S
    return None

        for t in (0.4, 0.25, 0.15, 0.1, 0.05, 0.02):
            a_new = (1.0 - t) * a + t * b
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_a(a_new)
            sc_new = exact_score_a(a_new)
            if np.isfinite(sc_new) and sc_new < cur_sc * (1.0 - 1e-6):
                a = a_new
                improved = True
                break
        
        if not improved:
            stuck_count += 1
            if stuck_count >= max_stuck:
                break
            # Try a different direction: perturb and retry
            a = normalize_a(a * (1.0 + 0.05 * np.random.standard_normal(n)))
        else:
            stuck_count = 0
    
    final_sc = exact_score_a(a)
    if final_sc < best_sc:
        best_sc = final_sc
        best_a = a.copy()
    
    return float(best_sc), best_a


def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.10, 0.50))
    mode = int(rng.integers(0, 7))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.10, 0.40))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.08 * rng.standard_normal(n))
    elif mode == 3:
        d, S = d_from_a(p)
        w = active_weights(d, 6)
        h = corr_pair(p, w)
        hmax = float(np.max(h))
        if hmax < 1e-12:
            q = p.copy()
        else:
            q = p * np.exp(-strength * h / hmax)
    elif mode == 4:
        logp = np.log(p + 1e-300)
        # Smooth noise
        noise = rng.standard_normal(n)
        k = min(31, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        if k > n:
            k = max(3, n if n % 2 == 1 else n - 1)
        if k >= 3:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode="same")
        noise = noise - noise.mean()
        logp = logp + smooth_strength = strength * 0.4 * noise
        q = np.exp(logp - logp.max())
    elif mode == 5:
        # Block perturbation
        nb = int(rng.integers(4, min(32, n // 4) + 1))
        per = max(1, n // nb)
        blocks = 1.0 + 0.3 * rng.standard_normal(nb)
        blocks = np.clip(blocks, 0.3, 2.0)
        mask = np.repeat(blocks, per)[:n]
        if len(mask) < n:
            mask = np.concatenate([mask, np.ones(n - len(mask))])
        q = p * mask
    else:
        # Local random perturbation
        idx = rng.integers(0, n, size=max(1, n // 10))
        q = p.copy()
        q[idx] *= np.exp(strength * rng.standard_normal(len(idx)))
    
    return normalize_a(q)


def generate_inits(n, rng, parent, cap=20):
    inits = []
    n = int(n)
    x = np.linspace(0.0, 1.0, n)
    
    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.nan_to_num(np.asarray(arr, dtype=np.float64), nan=0.0, posinf=1e4, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        s = arr.sum()
        if s < 1e-12:
            arr = np.ones(n) / n
        else:
            arr = arr / s
        inits.append(arr)
    
    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
    
    add(np.ones(n))
    
    for alpha in (0.3, 0.5, 0.7, 0.9):
        add(1.0 + alpha * (2.0 * x - 1.0))
        add(1.0 - alpha * (2.0 * x - 1.0))
    
    # Sinusoidal
    for r in range(6):
        vals = np.ones(n)
        K = int(rng.integers(2, 7))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 6)))
            amp = float(rng.uniform(0.1, 0.5))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)
    
    # Spectrum-based
    Nf = 1 << (n - 1).bit_length()
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf) * Nf
        alpha = float(rng.uniform(1.5, 4.0))
        power = 1.0 / ((np.abs(freqs) + 1.0) ** alpha)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.mean()
        add(1.0 + 0.7 * noise)
    
    # Sparse/binary-like
    for frac in (0.4, 0.6, 0.8):
        mask = rng.random(n) < frac
        vals = np.where(mask, 0.2 + 0.8 * rng.random(n), 0.0)
        add(vals)
    
    # Gaussian bumps
    for _ in range(4):
        nc = int(rng.integers(1, 5))
        vals = np.ones(n) * 0.3
        for _ in range(nc):
            center = float(rng.uniform(0.1, 0.9))
            width = float(rng.uniform(0.02, 0.15))
            amp = float(rng.uniform(0.5, 2.0))
            vals += amp * np.exp(-((x - center) ** 2) / (2.0 * width ** 2))
        add(vals)
    
    # Random
    for _ in range(4):
        add(0.1 + 0.9 * rng.random(n))
    
    return inits


def optimize_trajectory_lp(a_start, deadline, rng, label):
    """Main optimization trajectory combining LP, LSE, and subgradient methods."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    t0 = time.monotonic()
    
    print("[{}] start score={:.8f} n={}".format(label, best_sc, len(a)))
    
    # Phase 1: LSE continuation (get to a good basin)
    total_time = max(10.0, float(deadline - t0))
    lse_deadline = t0 + 0.40 * total_time
    
    betas = [30, 80, 200, 500, 1000, 2000, 4000]
    for beta in betas:
        if time.monotonic() > lse_deadline - 5:
            break
        rem = lse_deadline - time.monotonic()
        if rem < 10:
            break
        mf = estimate_maxfun(rem * 0.8, len(a))
        if mf < 64:
            break
        sc, a2 = run_lbfgs(a, float(beta), lse_deadline, mf)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            print("[{}] LSE beta={:.0f}: {:.8f}".format(label, beta, best_sc))
        a = normalize_a(a2)
    
    # Phase 2: LP-based iterative improvement
    lp_deadline = t0 + 0.75 * total_time
    if time.monotonic() < lp_deadline - 10:
        # Multiple LP rounds
        for round_i in range(5):
            if time.monotonic() > lp_deadline - 5:
                break
            sc, a2 = lp_iterative_step(a, lp_deadline, max_lp_iters=20)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] LP round{}: {:.8f}".format(label, round_i, best_sc))
            a = normalize_a(a2)
            
            # LSE polish between LP rounds
            if time.monotonic() < lp_deadline - 15:
                rem = lp_deadline - time.monotonic()
                if rem > 15:
                    mf = estimate_maxfun(rem * 0.5, len(a))
                    sc2, a3 = run_lbfgs(a, 3000.0, lp_deadline, mf)
                    if sc2 < best_sc:
                        best_sc = sc2
                        best_a = a3.copy()
                        print("[{}] LP+LSE round{}: {:.8f}".format(label, round_i, best_sc))
                    a = normalize_a(a3)
    
    # Phase 3: Final high-beta LSE polish
    if time.monotonic() < deadline - 5:
        for beta in (5000, 10000, 15000):
            if time.monotonic() > deadline - 3:
                break
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            mf = estimate_maxfun(rem * 0.8, len(a))
            sc, a2 = run_lbfgs(best_a, float(beta), deadline, mf)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("[{}] final LSE beta={:.0f}: {:.8f}".format(label, beta, best_sc))
    
    return float(best_sc), best_a


def quick_optimize(a, deadline):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    for beta in (200, 800, 3000, 8000):
        rem = float(deadline - time.monotonic())
        if rem < 10:
            break
        mf = estimate_maxfun(rem * 0.7, len(a))
        if mf < 64:
            break
        sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize_a(a2)
    
    # Subgradient polish
    if time.monotonic() < deadline - 5:
        sc, a2 = subgrad_polish(best_a, max_iters=10, deadline=deadline - 2)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
    
    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a low-score nonnegative sequence under the 2400s budget."""
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 12.0
    rng = np.random.default_rng(seed)
    
    # Get parent sequence
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        parent = None
    
    # Determine lengths to try
    parent_len = int(parent.size) if parent is not None else 0
    lengths = []
    if parent_len > 0:
        lengths.append(min(parent_len, 8192))
    if 8192 not in lengths:
        lengths.append(8192)
    if 4096 not in lengths:
        lengths.append(4096)
    
    # Filter to valid range
    lengths = [l for l in lengths if 1024 <= l <= 10240]
    if not lengths:
        lengths = [4096]
    lengths = lengths[:3]
    
    print("Parent: len={}".format(parent_len if parent is not None else 0))
    print("Lengths: {}".format(lengths))
    
    # Generate and evaluate all initializations
    all_seqs = []
    all_scores = []
    all_lens = []
    
    for n in lengths:
        inits = generate_inits(n, rng, parent, cap=16)
        for arr in inits:
            p = normalize_a(arr)
            sc = exact_score_a(p)
            if np.isfinite(sc):
                all_seqs.append(p)
                all_scores.append(float(sc))
                all_lens.append(int(n))
    
    if not all_seqs:
        n0 = lengths[0]
        all_seqs.append(np.ones(n0) / n0)
        all_scores.append(2.0)
        all_lens.append(n0)
    
    order = np.argsort(np.asarray(all_scores))
    print("Best init score: {:.8f}".format(all_scores[int(order[0])]))
    
    # Select top starting points (diverse lengths)
    starts = []
    used_lens = set()
    
    # First: parent if available
    if parent is not None:
        ps = normalize_a(resample_sequence(parent, min(parent_len, 8192)))
        starts.append(ps)
        used_lens.add(len(ps))
    
    # Fill with best inits from different lengths
    for idx in order:
        if len(starts) >= 3:
            break
        nidx = int(all_lens[int(idx)])
        if nidx not in used_lens:
            starts.append(all_seqs[int(idx)].copy())
            used_lens.add(nidx)
    
    while len(starts) < 2:
        starts.append(perturb_best(starts[0], rng))
    
    # Initial best
    best_seq = starts[0].copy()
    best_score = exact_score_a(best_seq)
    for i in range(1, len(starts)):
        sc = exact_score_a(starts[i])
        if sc < best_score:
            best_score = sc
            best_seq = starts[i].copy()
    
    print("Initial best: {:.8f} (len={})".format(best_score, len(best_seq)))
    
    # Run main trajectories
    n_traj = len(starts)
    if n_traj == 1:
        fracs = [1.0]
    elif n_traj == 2:
        fracs = [0.60, 0.40]
    else:
        fracs = [0.50, 0.30, 0.20]
    
    main_end = deadline - 180.0
    total_main = max(60.0, main_end - t_start)
    cum = 0.0
    
    for i, st in enumerate(starts):
        cum += fracs[i]
        sub_dead = min(t_start + cum * total_main, main_end)
        
        if time.monotonic() > sub_dead - 10:
            continue
        
        # For non-priority trajectories, use best as base
        sq = st
        if i > 0 and best_score < 2.0:
            base = best_seq
            if len(base) != len(sq):
                base = resample_sequence(base, len(sq))
            base = normalize_a(base)
            if i == 1:
                sq = base
            else:
                sq = perturb_best(base, rng)
        
        sc, seq = optimize_trajectory_lp(sq, sub_dead, rng, "T{}".format(i + 1))
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best (after T{}): {:.8f}".format(i + 1, best_score))
    
    best_seq = normalize_a(best_seq)
    
    # Phase: Diversified restarts with LP
    restart_end = deadline - 90.0
    n_restarts = 0
    while time.monotonic() < restart_end and n_restarts < 6:
        left = float(restart_end - time.monotonic())
        if left < 60:
            break
        chunk_dead = time.monotonic() + min(left / (6 - n_restarts), 120.0)
        
        # Create diversity
        mode = n_restarts % 3
        if mode == 0:
            p = perturb_best(best_seq, rng, strength=0.3)
        elif mode == 1:
            p = perturb_best(best_seq, rng, strength=0.15)
        else:
            # Try different length
            n_alt = int(len(best_seq) * float(rng.choice([0.7, 1.0, 1.4])))
            n_alt = max(1024, min(n_alt, 10240))
            p = resample_sequence(best_seq, n_alt)
            p = perturb_best(p, rng, strength=0.2)
            p = normalize_a(p)
        
        # Quick optimize + LP
        sc, seq = quick_optimize(p, chunk_dead - 5.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Restart {} (LSE): {:.8f}".format(n_restarts, best_score))
        
        # LP step on the result
        if time.monotonic() < chunk_dead - 3.0:
            sc2, seq2 = lp_iterative_step(seq, chunk_dead, max_lp_iters=10)
            if sc2 < best_score:
                best_score = sc2
                best_seq = seq2.copy()
                print("Restart {} (LP): {:.8f}".format(n_restarts, best_score))
        
        n_restarts += 1
    
    # Final polish: subgradient
    if time.monotonic() < deadline - 20:
        sc, seq = subgrad_polish(best_seq, max_iters=15, deadline=deadline - 10)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgrad: {:.8f}".format(best_score))
    
    # Final high-beta LSE
    if time.monotonic() < deadline - 8:
        rem = float(deadline - time.monotonic())
        if rem > 8:
            mf = estimate_maxfun(rem * 0.8, len(best_seq))
            sc, seq = run_lbfgs(best_seq, 12000.0, deadline, mf)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Final LSE: {:.8f}".format(best_score))
    
    # Output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
    
    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score
    
    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
