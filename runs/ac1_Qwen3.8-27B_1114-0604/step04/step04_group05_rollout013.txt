"""
Self-convolution peak minimization using adaptive LSE-smoothed Adam optimization,
L-BFGS continuation, analytical coordinate descent, and structured spectral restarts.
Key innovations: (1) analytical 1D coordinate optimization exploiting the linear
structure of conv_k* in a_j, (2) adaptive beta continuation with cosine schedule,
(3) multi-peak targeting via LSE soft-max, (4) peak-directed spectral perturbations.
"""

import numpy as np
from scipy.optimize import minimize
import time
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    sum_exp = float(exp_scaled.sum())
    p = exp_scaled / sum_exp
    L = c_max + math.log(sum_exp) / beta
    score = 2.0 * n * L / (S * S)
    corr_fft = 1 << (3 * n - 2).bit_length()
    P_ft = np.fft.rfft(p, corr_fft)
    A_ft = np.fft.rfft(a, corr_fft)
    T = np.fft.irfft(P_ft * np.conj(A_ft), corr_fft)[:n]
    dL_da = 2.0 * T
    grad = 2.0 * n * (dL_da * S - 2.0 * L) / (S * S * S)
    return score, grad


def subgrad_exact(a):
    """Exact subgradient of the (non-smooth) score at the max peak."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    M = float(c[k_star])
    S = float(a.sum())
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        sg[js:je+1] = a[k_star - np.arange(js, je+1)]
    grad = 4.0 * n * (sg - M / S) / (S * S)
    return c, k_star, M, S, grad


def adam_lse(a_init, beta_start, beta_end, n_steps=50000, lr=0.004,
             time_budget=60.0, t_start=None):
    """Adam with LSE-smoothed objective and cosine beta schedule."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    S0 = a.sum()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        frac = (t - 1) / max(1, n_steps - 1)
        beta_t = beta_start + (beta_end - beta_start) * (1 - math.cos(math.pi * frac)) / 2.0
        beta_t = max(10.0, beta_t)
        lr_t = lr * 0.5 * (1 + math.cos(math.pi * frac))
        lr_t = max(lr_t, lr * 0.01)
        score, grad = lse_score_grad(a, beta_t)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e6:
            grad = grad / grad_norm * 1e6
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s > 1e-12:
            a = a * (S0 / s)
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def run_lbfgs(a0, beta, maxiter=3000, maxfun=30000, time_budget=60.0, t_start=None):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    def obj_jac(x):
        if time.monotonic() > deadline:
            raise TimeoutError
        return lse_score_grad(x, beta)

    try:
        res = minimize(obj_jac, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-15, 'gtol': 1e-12,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    sc = compute_score(a_out)
    return sc, a_out


def analytical_coord_desc(a, n_sweeps=10, top_k=40, time_budget=30.0, t_start=None):
    """
    Coordinate descent using analytical 1D solution.
    For each coordinate j, the peak conv_k* is linear in a[j],
    giving a closed-form optimal value.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf
    golden_gr = (math.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c, k_star, M, S, grad = subgrad_exact(a)
        score = compute_score(a)
        if score < best_score:
            best_score = score
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            a_old = a[j]
            c_k = float(c[k_star])
            S_j = S - a_old
            if 0 <= k_star - j < n:
                c_j = a[k_star - j]
            else:
                c_j = 0.0
            if c_j < 1e-15:
                candidates = [0.0, 0.01 * a.max(), 0.05 * a.max()]
            else:
                d = c_k - a_old * c_j
                a_opt = S_j - 2.0 * d / c_j
                a_opt = max(0.0, a_opt)
                candidates = [a_opt, max(0.0, a_opt * 0.5), max(0.0, a_opt * 1.5), 0.0, a_old]
            best_sc = np.inf
            best_val = a_old
            for val in candidates:
                val = float(max(0.0, min(1000.0, val)))
                a[j] = val
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_val = val
            a[j] = best_val
            if best_sc < best_score:
                best_score = best_sc
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def golden_coord(a, j, n_evals=10):
    """Golden section 1D search for coordinate j."""
    a_old = a[j]
    a_max = a.max()
    if a_old < 1e-15:
        lo, hi = 0.0, max(0.1 * a_max, 1.0)
    else:
        lo, hi = 0.0, max(a_old * 3.0, 0.1 * a_max)
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    a[j] = c1
    f1 = compute_score(a)
    a[j] = c2
    f2 = compute_score(a)
    for _ in range(n_evals - 2):
        if f1 < f2:
            hi = c2
            c2, f2 = c1, f1
            c1 = hi - gr * (hi - lo)
            a[j] = c1
            f1 = compute_score(a)
        else:
            lo = c1
            c1, f1 = c2, f2
            c2 = lo + gr * (hi - lo)
            a[j] = c2
            f2 = compute_score(a)
    best_val = c1 if f1 < f2 else c2
    a[j] = best_val
    return compute_score(a)


def spectral_perturb(a, rng, strength=0.15):
    """Perturb in DFT domain for diversity."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    amp_f = 1.0 + strength * rng.standard_normal(len(A))
    amp_f = np.clip(amp_f, 0.3, 2.0)
    ph_f = strength * 0.5 * rng.standard_normal(len(A))
    A_new = A * amp_f * np.exp(1j * ph_f)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 1e-10:
        return a.copy()
    return a_new * (a.sum() / a_new.sum())


def anti_palin_perturb(a, rng, strength=0.12):
    """Anti-palindromic perturbation to reduce middle peak."""
    n = len(a)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    tilt = (x - center) / max(center, 1.0)
    freq = float(rng.uniform(1, 5))
    tilt += 0.5 * np.sin(2.0 * np.pi * freq * x / n)
    tilt_anti = 0.5 * (tilt - tilt[::-1])
    a_new = a + strength * tilt_anti * a.max()
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new
