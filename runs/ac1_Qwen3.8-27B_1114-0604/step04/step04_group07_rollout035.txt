Hybrid L2/L-infinity optimizer for minimizing 2n*max(self-conv)/(sum)^2.
Key innovations over previous approach:
1. FFT-based L2 (Parseval) objective with O(N log N) gradient for fast "flattening"
2. Alternating L2/LSE optimization to escape local minima

4. Population-based exploration with diverse perturbation strategies
5. Adaptive beta continuation with L-BFGS polishing
6. Equi-ripple condition exploitation via spectral domain

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def l2_obj_grad(a, N):
    """
    Scale-invariant L2 objective: sum(c^2) / (sum(a))^4.
    Uses Parseval's theorem for O(N log N) evaluation and gradient.
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    A = np.fft.fft(a, N)
    C = np.abs(A) ** 2
    sum_c2 = float(np.sum(C * C).real / N)
    B = C * np.conj(A)
    T = np.fft.fft(B)
    grad_raw = (4.0 / N) * np.real(T[:n])
    S2 = S * S
    S4 = S2 * S2
    S5 = S4 * S
    obj = sum_c2 / S4
    grad = grad_raw / S4 - 4.0 * sum_c2 / S5
    return obj, grad

    log_sum = np.log(exp_scaled.sum() + 1e-300)
    L = c_max + log_sum / beta

    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a[::-1], M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def l2_obj_grad_pal(half_a, N):
    """L2 objective and gradient for palindromic sequence (work in half-space)."""
    n_half = len(half_a)
    n = 2 * n_half
    a = np.concatenate([half_a, half_a[::-1]])
    obj, grad_full = l2_obj_grad(a, N)
    grad_half = grad_full[:n_half] + grad_full[n_half - 1::-1]
    return obj, grad_half

def adam_optimize(obj_grad_fn, x_init, lr, n_steps, time_limit_s, t_start,
                  normalize=True, clip_min=0.0, clip_max=1e10,
                  b1=0.9, b2=0.999, eps=1e-8, lr_decay=1e-5):
    """
    Generic Adam optimizer with time limit.
    obj_grad_fn(x) -> (objective, gradient)
    """
    x = x_init.astype(np.float64).copy()
    n = len(x)
    m = np.zeros(n)
    v = np.zeros(n)

    best_x = x.copy()
    best_obj = obj_grad_fn(x)[0]

    for t in range(n_steps):
        if time.monotonic() - t_start > time_limit_s:
            break
        lr_t = lr / (1.0 + lr_decay * t)
        obj, grad = obj_grad_fn(x)
        if not np.isfinite(obj):
            break
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        t_g = t + 1
        m_hat = m / (1.0 - b1 ** t_g)
        v_hat = v / (1.0 - b2 ** t_g)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        x = np.clip(x, clip_min, clip_max)
        if normalize:
            s = x.sum()
            if s > 1e-12:
                x = x / s
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best_x = x.copy()

    return best_obj, best_x

def mirror_descent_subgrad(a, n_steps=1000, eta=0.01, time_limit_s=30.0, t_start=None):
    """
    Mirror descent (multiplicative weight) on L-infinity subgradient.
    Updates: a <- a * exp(-eta * sg) / ||a * exp(-eta * sg)||
    """
    n = len(a)
    if t_start is None:
        t_start = time.monotonic()
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_score = compute_score(a)
    best_a = a.copy()

    for t in range(n_steps):
        if time.monotonic() - t_start > time_limit_s:
            break
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        M = c[k_star]

        # Subgradient of 2n*max(c) w.r.t. a (with sum(a)=1)
        sg = np.full(n, -4.0 * n * M)  # constant part from -2M*S term
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] += 2.0 * n * a[idx]

        # Multiplicative update
        a = a * np.exp(-eta * sg)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_score, best_a


def multiplicative_peak_reduce(a, n_iters=200, lr=0.3, time_limit_s=30.0, t_start=None):
    """
    Aggressively reduce peak by multiplicatively decreasing elements
    that contribute most to the current peak.
    """
    n = len(a)
    if t_start is None:
        t_start = time.monotonic()
    g = a.astype(np.float64).copy()
    g = g / g.sum()
    best_score = compute_score(g)
    best_g = g.copy()

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit_s:
            break
        c = fast_conv(g)
        k_star = int(np.argmax(c))
        M = c[k_star]

        # Contribution of each element to the peak
        contrib = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            contrib[js:je + 1] = g[j for j in range(js, je + 1)] * g[idx]

        norm = max(M, 1e-10)
        g = g * np.exp(-lr * contrib / norm)
        g = np.maximum(g, 1e-20)
        g = g / g.sum()

        sc = compute_score(g)
        if sc < best_score:
            best_score = sc
            best_g = g.copy()

    return best_score, best_g


def spectral_equalization(a, n_iters=100, lr=0.1, N=None, time_limit_s=30.0, t_start=None):
    """
    Iteratively equalize the magnitude spectrum |A(f)| to be as flat as possible.
    This drives the convolution towards being flat (equi-ripple).
    """
    n = len(a)
    if N is None:
        N = 1 << (2 * n - 1).bit_length()
    if t_start is None:
        t_start = time.monotonic()
    a_cur = a.astype(np.float64).copy()
    a_cur = a_cur / a_cur.sum()
    best_score = compute_score(a_cur)
    best_a = a_cur.copy()

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit_s:
            break
        A = np.fft.fft(a_cur, N)
        mag = np.abs(A)
        # Target: make |A| as flat as possible (on non-DC bins)
        # Use a "soft" target: geometric mean of |A|
        dc = mag[0]
        non_dc = mag[1:]
        target_mag = np.geometric_mean(non_dc[non_dc > 1e-10]) if np.any(non_dc > 1e-10) else 1.0
        target = np.ones(N) * target_mag
        target[0] = dc

        # Gradient of sum((|A| - target)^2) w.r.t. a
        # d/da_j |A_k| = 2*Re[conj(A_k) * exp(-2*pi*i*k*j/N)]
        # d/da_j (|A_k| - target_k)^2 = 2*(|A_k| - target_k) * 2*Re[conj(A_k)*exp(-2*pi*i*k*j/N)]
        # Sum over k: this is 4 * IFT((|A| - target) * conj(A))
        diff = mag - target
        G = np.fft.ifft(diff * np.conj(A))
        grad = 4.0 * np.real(G[:n])

        a_cur = a_cur - lr * grad
        a_cur = np.maximum(a_cur, 1e-20)
        a_cur = a_cur / a_cur.sum()

        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

    return best_score, best_a


def generate_diverse_inits(rng, n, n_inits=12):
    """Generate diverse initial sequences."""
    inits = []
    x = np.arange(n) / n

    # 1. Uniform
    inits.append(np.ones(n) / n)

    # 2. Gaussian
    sigma = 0.15 + 0.15 * rng.random()
    g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    inits.append(g / g.sum())

    # 3. Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())

    # 4. Triangular
    tri = np.maximum(0, 1 - np.abs(2 * x - 1))
    inits.append(tri / tri.sum())

    # 5. Sinusoidal
    freq = 2 + 6 * rng.random()
    phase = 2 * np.pi * rng.random()
    s = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x + phase)
    inits.append(np.maximum(s, 0.01) / max(s, 0.01).sum())

    # 6. Multi-frequency
    f1 = 1 + 3 * rng.random()
    f2 = 4 + 8 * rng.random()
    p1 = 2 * np.pi * rng.random()
    p2 = 2 * np.pi * rng.random()
    m = 1.0 + 0.2 * np.cos(2 * np.pi * f1 * x + p1) + 0.1 * np.cos(2 * np.pi * f2 * x + p2)
    inits.append(np.maximum(m, 0.01) / max(m, 0.01).sum())

    # 7. Two-bump
    c1 = 0.2 + 0.1 * rng.random()
    c2 = 0.6 + 0.1 * rng.random()
    s1 = 0.08 + 0.05 * rng.random()
    s2 = 0.08 + 0.05 * rng.random()
    b = np.exp(-0.5 * ((x - c1) / s1) ** 2) + np.exp(-0.5 * ((x - c2) / s2) ** 2)
    inits.append(b / b.sum())

    # 8. Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.3
    k = max(3, n // 40)
    r_s = np.convolve(r, np.ones(k) / k, mode='same')
    inits.append(r_s / r_s.sum())

    # 9. Palindromic random smooth
    half_n = n // 2
    h = np.abs(rng.standard_normal(half_n)) + 0.3
    h_s = np.convolve(h, np.ones(max(3, half_n // 40)) / max(3, half_n // 40), mode='same')
    pal = np.concatenate([h_s, h_s[::-1]])
    inits.append(pal / pal.sum())

    # 10. Flat-top (Kaiser-like)
    alpha = 3 + 5 * rng.random()
    y = np.abs(x - 0.5) / 0.5
    if alpha > 0:
        kt = np.zeros(n)
        m1 = y <= 1.0
        import math
        for i in range(n):
            if y[i] <= 1.0:
                if alpha <= 3.5:
                    kt[i] = 1 - (alpha**2 / (alpha**2 - 4)) * y[i]**2 + (alpha**2 / (2*(alpha**2-4))) * y[i]**4
                else:
                    kt[i] = _bessel_i0(alpha * np.sqrt(1 - y[i]**2)) / _bessel_i0(alpha)
        kt = np.maximum(kt, 0.01)
        inits.append(kt / kt.sum())

    # 11. Cosine window variant
    cw = 0.54 + 0.46 * np.cos(2 * np.pi * x)
    inits.append(cw / cw.sum())

    # 12. Power-law decay from center
    p = 0.5 + 2.0 * rng.random()
    d = np.abs(x - 0.5)
    pl = (1.0 - d + 1e-10) ** p
    inits.append(pl / pl.sum())

    return [np.clip(x, 1e-10, 1000.0) for x in inits[:n_inits]]


def _bessel_i0(x):
    """Simple Bessel I0 approximation."""
    # Series expansion for small x, asymptotic for large x
    if x < 20:
        s = 1.0
        term = 1.0
        for k in range(1, 30):
            term *= (x * x / 4.0) / (k * k)
            s += term
        return s
    else:
        return np.exp(x) / np.sqrt(2 * np.pi * x)


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def perturbed_candidate(a, rng, strategy):
    """Generate a perturbed version of a using different strategies."""
    n = len(a)
    a_copy = a.astype(np.float64).copy()

    if strategy == 0:
        # Gaussian perturbation in log-space
        log_a = np.log(np.clip(a_copy, 1e-10, 1e10))
        noise = 0.03 + 0.07 * rng.random()
        log_pert = log_a + noise * rng.standard_normal(n)
        a_copy = np.exp(log_pert - np.max(log_pert))
    elif strategy == 1:
        # Peak-avoidance modification
        c = fast_conv(a_copy)
        k_star = int(np.argmax(c))
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        a_copy[js:je + 1] *= (0.5 + 0.3 * rng.random())
        non_peak = np.ones(n, dtype=bool)
        non_peak[js:je + 1] = False
        if non_peak.sum() > 0:
            a_copy[non_peak] *= (1.0 + 0.1 * rng.random())
    elif strategy == 2:
        # Spectral perturbation
        N = 1 << (2 * n - 1).bit_length()
        A = np.fft.fft(a_copy, N)
        phase_noise = rng.standard_normal(N) * 0.05
        A_new = A * np.exp(1j * phase_noise)
        A_new[0] = A[0]  # preserve DC
        a_copy = np.real(np.fft.ifft(A_new)[:n])
        a_copy = np.maximum(a_copy, 0.0)
    elif strategy == 3:
        # Block scaling
        n_blocks = 3 + int(5 * rng.random())
        block_size = n // n_blocks
        for b in range(n_blocks):
            s = b * block_size
            e = min(s + block_size, n)
            a_copy[s:e] *= rng.uniform(0.5, 1.5)
    elif strategy == 4:
        # Swap halves (test non-palindromic direction)
        half = n // 2
        swap_frac = 0.1 + 0.2 * rng.random()
        a_new = a_copy.copy()
        a_new[:half] = (1 - swap_frac) * a_copy[:half] + swap_frac * a_copy[half:]
        a_new[half:] = (1 - swap_frac) * a_copy[half:] + swap_frac * a_copy[:half]
        a_copy = a_new
    else:
        # Random element scaling
        n_mod = max(1, n // 10)
        idx = rng.choice(n, size=n_mod, replace=False)
        a_copy[idx] *= rng.uniform(0.5, 1.5, size=n_mod)

    s = a_copy.sum()
    if s > 1e-12:
        a_copy = a_copy / s
    return np.clip(a_copy, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer combining L2 (Parseval), LSE, mirror descent, and spectral methods.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 25.0

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
            if parent.sum() > 1e-12:
                parent = parent / parent.sum()
                parent_score = compute_score(parent)
                print(f"Parent loaded: n={len(parent)}, score={parent_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n_work = len(parent)
        best_a = parent.copy()
        best_score = compute_score(parent)
        # Ensure even length
        if n_work % 2 != 0:
            n_work += 1
            best_a = resample_sequence(best_a, n_work)
            best_score = compute_score(best_a)
    else:
        n_work = 4096
        print("No parent, starting with n=4096")
        best_a = np.ones(n_work) / n_work
        best_score = compute_score(best_a)

    N = 1 << (2 * n_work - 1).bit_length()
    n_half = n_work // 2
    print(f"Working with n={n_work}, N={N}")

    # ==========================================
    # Phase 1: L2 optimization from diverse inits (15% time)
    # ==========================================
    p1_time = effective_budget * 0.15

    inits = generate_diverse_inits(rng, n_work, n_inits=10)

        # Quick LSE polish

        print(f"  Init {i}: L2={sc_l2:.6f}, final={sc:.8f} ({time.monotonic()-t0:.1f}s)")

    candidates.sort(key=lambda x: x[0])
    for sc, a_opt in candidates[:3]:
        try_update(sc, a_opt, "Phase1 screening")
    print(f"  Best after Phase 1: {best_score:.8f}")

    # ==========================================
    # Phase 2: LSE continuation (35% time)

    p2_time = effective_budget * 0.35
    print(f"\n{'=' * 60}")

    print(f"{'=' * 60}")

    # Palindrome half-space
    half_best = (best_a[:n_half] + best_a[::-1][:n_half]) / 2.0

            break
        t0 = time.monotonic()
        sc, h_opt = adam_lse_pal(half_best, beta, lr=lr, n_steps=100000,
                                  time_limit_s=min(t_per_beta * 1.5, remaining),
                                  t_start=t_start)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        try_update(sc, a_full, f"Adam-LSE-pal beta={beta}")
        half_best = h_opt
        print(f"  beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Also try non-palindrome
    remaining = t_start + p2_time - time.monotonic()
    if remaining > 10:
        for beta, lr in [(500, 0.003), (2000, 0.001)]:
            remaining = t_start + p2_time - time.monotonic()
            if remaining < 5:
                break
            t0 = time.monotonic()
            sc, a_opt = adam_lse_direct(best_a, beta, lr=lr, n_steps=50000,
                                         time_limit_s=min(remaining * 0.4, 20.0),
                                         t_start=t_start)
            try_update(sc, a_opt, f"Adam-LSE-direct beta={beta}")
            print(f"  Direct beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 3: L-BFGS with L2 (15% time)
    # ==========================================
    p3_time = effective_budget * 0.15
    print(f"\n{'=' * 60}")
    print(f"Phase 3: L-BFGS L2 polish (budget={p3_time:.0f}s)")
    print(f"{'=' * 60}")

    half_best = (best_a[:n_half] + best_a[::-1][:n_half]) / 2.0
    half_best = half_best / half_best.sum()

    # L-BFGS in palindrome L2 space
    t0 = time.monotonic()
    sc, h_opt = lbfgs_l2_pal(half_best, N, maxiter=3000, maxfun=50000)
    a_full = np.concatenate([h_opt, h_opt[::-1]])
    try_update(sc, a_full, "LBFGS-L2-pal")
    half_best = h_opt
    print(f"  LBFGS-L2-pal: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # L-BFGS in full space L2
    remaining = t_start + p3_time - time.monotonic()
    if remaining > 10:
        t0 = time.monotonic()
        sc, a_opt = lbfgs_l2(best_a, N, maxiter=3000, maxfun=50000)
        try_update(sc, a_opt, "LBFGS-L2-direct")
        print(f"  LBFGS-L2-direct: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 4: L-BFGS with high-beta LSE (15% time)
    # ==========================================
    p4_time = effective_budget * 0.15
    print(f"\n{'=' * 60}")
    print(f"Phase 4: L-BFGS LSE polish (budget={p4_time:.0f}s)")
    print(f"{'=' * 60}")

    half_best = (best_a[:n_half] + best_a[::-1][:n_half]) / 2.0
    half_best = half_best / half_best.sum()

    for beta in [1000, 5000, 15000]:
        remaining = t_start + p4_time - time.monotonic()
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, h_opt = lbfgs_lse_pal(half_best, beta, maxiter=5000, maxfun=80000)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        try_update(sc, a_full, f"LBFGS-LSE-pal beta={beta}")
        half_best = h_opt
        print(f"  LBFGS-LSE-pal beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 5: Mirror descent + spectral equalization (10% time)
    # ==========================================
    p5_time = effective_budget * 0.10
    print(f"\n{'=' * 60}")
    print(f"Phase 5: Mirror descent + spectral (budget={p5_time:.0f}s)")
    print(f"{'=' * 60}")

    # Mirror descent on subgradient
    t0 = time.monotonic()
    sc, a_opt = mirror_descent_subgrad(best_a, n_steps=5000, eta=0.005,
                                        time_limit_s=p5_time * 0.4, t_start=time.monotonic())
    try_update(sc, a_opt, "Mirror descent")
    print(f"  Mirror descent: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Spectral equalization
    remaining = t_start + p5_time - time.monotonic()
    if remaining > 5:
        t0 = time.monotonic()
        sc, a_opt = spectral_equalization(best_a, n_iters=200, lr=0.05, N=N,
                                           time_limit_s=min(remaining, p5_time * 0.4),
                                           t_start=time.monotonic())
        try_update(sc, a_opt, "Spectral eq")
        print(f"  Spectral equalization: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Multiplicative peak reduction
    remaining = t_start + p5_time - time.monotonic()
    if remaining > 5:
        t0 = time.monotonic()
        sc, a_opt = multiplicative_peak_reduce(best_a, n_iters=500, lr=0.2,
                                                time_limit_s=min(remaining, p5_time * 0.3),
                                                t_start=time.monotonic())
        try_update(sc, a_opt, "Mult peak reduce")
        print(f"  Multiplicative: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 6: Perturbation + re-optimization (10% time)
    # ==========================================
    p6_time = effective_budget * 0.10
    print(f"\n{'=' * 60}")
    print(f"Phase 6: Perturbation + refine (budget={p6_time:.0f}s)")
    print(f"{'=' * 60}")

    n_restarts = 10
    restart_budget = p6_time / n_restarts

    for r in range(n_restarts):
        if time.monotonic() > t_start + effective_budget - 20:
            break
        strategy = r % 5
        a_pert = perturbed_candidate(best_a, rng, strategy)

        # Quick L2 + LSE refinement
        time_left = t_start + effective_budget - 20 - time.monotonic()
        if time_left < 3:
            break
        t_l2 = min(restart_budget * 0.4, time_left * 0.4)
        t_lse = min(restart_budget * 0.5, time_left * 0.5)

        sc_l2, a_r = adam_l2_direct(a_pert, N, lr=0.01, n_steps=20000,
                                      time_limit_s=t_l2, t_start=t_start)
        sc_lse, a_r = adam_lse_direct(a_r, 500, lr=0.003, n_steps=10000,
                                       time_limit_s=t_lse, t_start=t_start)
        sc = compute_score(a_r)
        try_update(sc, a_r, f"restart#{r} s{strategy}")
        print(f"  Restart {r} (s{strategy}): {sc:.8f}")

    # ==========================================
    # Phase 7: Final intensive polish (5% time)
    # ==========================================
    p7_time = effective_budget * 0.05
    print(f"\n{'=' * 60}")
    print(f"Phase 7: Final polish (budget={p7_time:.0f}s)")
    print(f"{'=' * 60}")

    final_deadline = t_start + effective_budget - 5.0

    # High-beta L-BFGS in palindrome space
    n_half_curr = len(best_a) // 2
    if n_half_curr * 2 != len(best_a):
        n_half_curr = len(best_a) // 2
        best_a = resample_sequence(best_a, 2 * n_half_curr)
    half_best = (best_a[:n_half_curr] + best_a[::-1][:n_half_curr]) / 2.0
    half_best = half_best / half_best.sum()

    for beta in [5000, 20000]:
        if time.monotonic() > final_deadline:
            break
        remaining = final_deadline - time.monotonic()
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, h_opt = lbfgs_lse_pal(half_best, beta, maxiter=8000, maxfun=100000)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        try_update(sc, a_full, f"final LBFGS-pal beta={beta}")
        half_best = h_opt
        print(f"  Final LBFGS-pal beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Final Adam with very high beta
    if time.monotonic() < final_deadline - 3:
        remaining = final_deadline - time.monotonic() - 3
        t0 = time.monotonic()
        sc, a_opt = adam_lse_direct(best_a, 5000, lr=0.0003, n_steps=50000,
                                     time_limit_s=min(remaining, p7_time),
                                     t_start=t_start)
        try_update(sc, a_opt, "final Adam high-beta")
        print(f"  Final Adam: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # One more L2 L-BFGS pass
    if time.monotonic() < final_deadline - 2:
        remaining = final_deadline - time.monotonic() - 2
        t0 = time.monotonic()
        N_final = 1 << (2 * len(best_a) - 1).bit_length()
        sc, a_opt = lbfgs_l2(best_a, N_final, maxiter=2000, maxfun=30000)
        try_update(sc, a_opt, "final LBFGS-L2")
        print(f"  Final LBFGS-L2: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Final verification
    # ==========================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
