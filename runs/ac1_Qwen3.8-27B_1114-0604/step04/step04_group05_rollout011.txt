import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a. Uses FFT for efficiency."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    c_max = float(c.max())

    exp_scaled = np.exp(scaled)

    log_sum = np.log(exp_sum)

    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    S0 = a.sum()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)
    t = 0

    while t < n_steps and time.monotonic() < deadline:
        t += 1
        # Determine beta for current step
        if betas_schedule:
            frac = t / n_steps
            idx = min(int(frac * len(betas_schedule)), len(betas_schedule) - 1)
            beta = betas_schedule[idx]
        else:
            beta = 200.0

        lr_t = lr_init / (1.0 + lr_decay * t)
        score, grad = lse_score_grad(a, beta)

        m_buf = momentum * m_buf + (1.0 - momentum) * grad
        v_buf = decay * v_buf + (1.0 - decay) * grad * grad
        m_hat = m_buf / (1.0 - momentum ** t)
        v_hat = v_buf / (1.0 - decay ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)

        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a * (S0 / s)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def lbfgs_optimize(a0, beta, maxiter, maxfun, t_start, deadline):
    """L-BFGS-B optimization with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_grad(x, beta)

    try:
        res = minimize(
            obj_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter,
                     'maxfun': maxfun, 'maxls': 30}
        )
    except Exception:
        return compute_score(a0), a0

    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def coord_descent_grad(a, n_sweeps, top_k, t_start, deadline):
    """Gradient-based coordinate descent: line search along each coordinate."""
    n = len(a)
    a = a.copy()
    best_score = compute_score(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c, k_star = fast_conv(a), None
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score

        # Subgradient at peak
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        sg = np.zeros(n)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = a[idx]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

        # Select top-k by |grad|
        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            a_old = a[j]
            g = grad[j]
            if abs(g) < 1e-15:
                continue
            # Line search: try moving in -grad direction
            # Parametric: a[j] = a_old * (1 + alpha * g * scale)
            # Use a few trial steps
            scale = a_old / (abs(g) * n + 1e-10) if a_old > 0 else 1.0
            best_val = a_old
            best_sc = compute_score(a)
            for alpha in [0.5, 1.0, 2.0, 0.1, -0.3, -0.6]:
                new_val = max(0.0, a_old - alpha * abs(g) * scale)
                if new_val == a_old:
                    continue
                a[j] = new_val
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_val = new_val
            a[j] = best_val
            if best_sc < best_score:
                best_score = best_sc

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def init_modulated(n, rng, freq=None, alpha=None, phase=None):
    """Initialize with 1 + alpha*cos(2*pi*freq*j/n + phase)."""
    if freq is None:
        freq = float(rng.uniform(1, max(2, n // 100)))
    if alpha is None:
        alpha = float(rng.uniform(0.3, 0.95))
    if phase is None:
        phase = float(rng.uniform(0, 2 * np.pi))
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * x / n + phase)
    a = np.clip(a, 0.01, None)
    return a


def init_ramp(n, rng, direction=None):
    """Initialize with a ramp shape."""
    if direction is None:
        direction = rng.choice([1, -1])
    x = np.arange(n, dtype=np.float64) / (n - 1)
    if direction == 1:
        a = 0.1 + 0.9 * x
    else:
        a = 0.1 + 0.9 * (1.0 - x)
    # Add noise
    a += rng.standard_normal(n) * 0.05
    a = np.clip(a, 0.001, None)
    return a


def init_two_sided(n, rng):
    """High on one side, low on the other."""
    a = np.ones(n)
    center = n // 2
    spread = int(rng.uniform(n // 10, n // 3))
    left = max(0, center - spread)
    right = min(n, center + spread)
    if rng.random() > 0.5:
        a[left:right] *= 0.3
    else:
        a[:center] *= 0.5
        a[center:] *= 1.5
    a += rng.standard_normal(n) * 0.05
    a = np.clip(a, 0.001, None)
    return a


def init_spectral(n, rng):
    """Initialize by designing in frequency domain."""
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    # Create a random spectrum with positive real inverse
    A = np.zeros(M_fft // 2 + 1, dtype=np.complex128)
    A[0] = np.sqrt(n)  # DC component
    # Add low-frequency components
    n_freqs = int(n // 4)
    for k in range(1, min(n_freqs, M_fft // 2)):
        amp = float(rng.uniform(0, 1)) / k ** 0.5
        phase = float(rng.uniform(0, 2 * np.pi))
        A[k] = amp * np.exp(1j * phase)
    a = np.fft.irfft(A, M_fft)[:n]
    a = np.clip(a, 0.001, None)
    if a.sum() < 1e-6:
        a = np.ones(n)
    return a


def init_random(n, rng):
    """Random non-negative initialization."""
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, None)


def refine_candidate(a_init, t_start, time_budget):
    """Multi-stage refinement: Adam -> L-BFGS -> Coord descent."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    # Stage 1: Adam with increasing beta (50% of budget)
    adam_deadline = min(deadline, t_start + time_budget * 0.5)
    if time.monotonic() < adam_deadline:
        betas = [10, 50, 200, 500, 2000, 5000]
        sc, a_opt = adam_optimize(a, n_steps=200000, betas_schedule=betas,
                                   lr_init=0.01, t_start=t_start,
                                   deadline=adam_deadline)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()

    # Stage 2: L-BFGS continuation (35% of budget)
    lbfgs_deadline = min(deadline, t_start + time_budget * 0.85)
    if time.monotonic() < lbfgs_deadline - 5:
        for beta in [2000, 10000, 50000]:
            if time.monotonic() > lbfgs_deadline - 5:
                break
            alloc = min(20.0, (lbfgs_deadline - time.monotonic()) * 0.8)
            if alloc < 5:
                break
            sc, a_opt = lbfgs_optimize(best_a, beta, maxiter=3000, maxfun=20000,
                                        t_start=t_start, deadline=t_start + alloc)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()

    # Stage 3: Coordinate descent (15% of budget)
    if time.monotonic() < deadline - 2:
        coord_deadline = deadline - 2
        sc, a_opt = coord_descent_grad(best_a, n_sweeps=5, top_k=40,
                                        t_start=t_start, deadline=coord_deadline)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multi-length, multi-init optimization.
    Strategy: diverse inits at multiple lengths, then intensive refinement of best.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 30.0
    deadline = t_start + effective_budget

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None
    init_counter = 0

    def try_update(sc, a, label):
        nonlocal best_score, best_a, init_counter
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [BEST] {sc:.10f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # Determine lengths to try
    n_parent = len(parent) if parent is not None else 4096
    lengths = []
    for factor in [0.5, 0.7, 0.85, 1.0, 1.2, 1.5, 2.0]:
        tl = int(n_parent * factor)
        tl = max(512, min(tl, 32768))
        # Round to power of 2 for FFT efficiency
        tl_pow2 = 1 << int(round(math.log2(tl)))
        if tl_pow2 >= 512 and tl_pow2 <= 32768 and tl_pow2 not in lengths:
            lengths.append(tl_pow2)
    # Also add some fixed useful lengths
    for fixed in [1024, 2048, 4096, 8192, 16384]:
        if fixed not in lengths and 512 <= fixed <= 32768:
            lengths.append(fixed)
    lengths = sorted(set(lengths))
    print(f"Lengths to try: {lengths}")
    print(f"Parent: n={n_parent}, score={compute_score(parent) if parent is not None else 'N/A'}")

    # Phase 1: Diverse initialization and quick evaluation (30% budget)
    p1_deadline = t_start + effective_budget * 0.30
    print(f"\n--- Phase 1: Diverse init (deadline t={p1_deadline - t_start:.0f}s) ---")

    candidates = []
    for n in lengths:
        if time.monotonic() > p1_deadline - 5:
            break

        # Generate diverse inits for this length
        inits = []
        if parent is not None:
            inits.append(("parent", interpolate_sequence(parent, n)))
        inits.append(("mod1", init_modulated(n, rng, freq=3, alpha=0.7, phase=0.5)))
        inits.append(("mod2", init_modulated(n, rng, freq=7, alpha=0.5, phase=1.0)))
        inits.append(("ramp", init_ramp(n, rng)))
        inits.append(("two_sided", init_two_sided(n, rng)))
        inits.append(("random", init_random(n, rng)))

        for label, a_init in inits:
            if time.monotonic() > p1_deadline - 3:
                break
            sc = compute_score(a_init)
            candidates.append((sc, a_init.copy(), f"{label}_n{n}"))

    # Sort by score and keep top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, best init score: {candidates[0][0]:.6f}")
    for sc, _, label in candidates[:5]:
        print(f"    {label}: {sc:.6f}")

    # Quick Adam on top candidates
    n_quick = min(6, len(candidates))
    quick_time_each = (p1_deadline - time.monotonic()) / max(1, n_quick) - 1
    for i in range(n_quick):
        if time.monotonic() > p1_deadline - 2:
            break
        sc_init, a_init, label = candidates[i]
        alloc = max(5.0, quick_time_each)
        dl = min(p1_deadline - 1, time.monotonic() + alloc)
        betas = [50, 200, 800]
        sc, a_opt = adam_optimize(a_init, n_steps=100000, betas_schedule=betas,
                                   lr_init=0.01, t_start=t_start, deadline=dl)
        try_update(sc, a_opt, f"quick_{label}")

    # Phase 2: Intensive refinement of best candidates (45% budget)
    p2_deadline = t_start + effective_budget * 0.75
    print(f"\n--- Phase 2: Intensive refinement (deadline t={p2_deadline - t_start:.0f}s) ---")

    # Refine the current best
    if best_a is not None:
        ref_budget = min(120.0, (p2_deadline - time.monotonic()) * 0.4)
        if ref_budget > 10:
            sc, a_opt = refine_candidate(best_a, time.monotonic(), ref_budget)
            try_update(sc, a_opt, "refine_best")

    # Try a few more diverse candidates at the best length
    best_n = len(best_a) if best_a is not None else 4096
    diverse_count = 4
    div_time_each = (p2_deadline - time.monotonic()) / max(1, diverse_count) - 2
    for d in range(diverse_count):
        if time.monotonic() > p2_deadline - 10:
            break
        alloc = max(20.0, min(div_time_each, 80.0))
        if time.monotonic() + alloc > p2_deadline - 5:
            alloc = p2_deadline - 5 - time.monotonic()
        if alloc < 10:
            break

        # Create diverse perturbation of best
        if best_a is not None:
            a_pert = best_a.copy()
            # Random spectral perturbation
            M = 2 * best_n - 1
            M_fft = 1 << (M - 1).bit_length()
            A = np.fft.rfft(a_pert, M_fft)
            strength = 0.1 + 0.1 * d
            amp_mod = 1.0 + strength * rng.standard_normal(len(A))
            amp_mod = np.clip(amp_mod, 0.3, 2.0)
            phase_mod = strength * 0.3 * rng.standard_normal(len(A))
            A_new = A * amp_mod * np.exp(1j * phase_mod)
            a_pert = np.fft.irfft(A_new, M_fft)[:best_n]
            a_pert = np.clip(a_pert, 0.0, None)
            if a_pert.sum() > 1e-10:
                a_pert = a_pert * (best_a.sum() / a_pert.sum())
        else:
            a_pert = init_modulated(best_n, rng)

        sc, a_opt = refine_candidate(a_pert, time.monotonic(), alloc)
        try_update(sc, a_opt, f"diversify_{d}")

    # Phase 3: Try different lengths for the best shape (15% budget)
    p3_deadline = t_start + effective_budget * 0.90
    print(f"\n--- Phase 3: Length sweep (deadline t={p3_deadline - t_start:.0f}s) ---")

    if best_a is not None:
        current_n = len(best_a)
        for factor in [0.6, 0.75, 1.25, 1.5]:
            if time.monotonic() > p3_deadline - 10:
                break
            new_n = 1 << int(round(math.log2(current_n * factor)))
            new_n = max(512, min(new_n, 32768))
            if new_n == current_n:
                continue
            a_res = interpolate_sequence(best_a, new_n)
            alloc = min(40.0, p3_deadline - time.monotonic() - 5)
            if alloc < 10:
                break
            sc, a_opt = refine_candidate(a_res, time.monotonic(), alloc)
            try_update(sc, a_opt, f"n={new_n}")

    # Phase 4: Final intensive optimization (remaining ~10%)
    p4_deadline = deadline
    print(f"\n--- Phase 4: Final optimization (deadline t={p4_deadline - t_start:.0f}s) ---")

    if best_a is not None and time.monotonic() < p4_deadline - 10:
        # High-beta L-BFGS
        for beta in [50000, 200000]:
            if time.monotonic() > p4_deadline - 5:
                break
            alloc = min(30.0, p4_deadline - time.monotonic() - 3)
            if alloc < 5:
                break
            sc, a_opt = lbfgs_optimize(best_a, beta, maxiter=5000, maxfun=30000,
                                        t_start=t_start, deadline=t_start + alloc)
            try_update(sc, a_opt, f"final_lbfgs_b{beta}")

        # Final coordinate descent
        if time.monotonic() < p4_deadline - 3:
            alloc = min(30.0, p4_deadline - time.monotonic() - 2)
            if alloc > 5:
                sc, a_opt = coord_descent_grad(best_a, n_sweeps=8, top_k=50,
                                                t_start=t_start,
                                                deadline=time.monotonic() + alloc)
                try_update(sc, a_opt, "final_coord")

    # Final verification
    if best_a is None:
        best_a = np.ones(4096)

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
