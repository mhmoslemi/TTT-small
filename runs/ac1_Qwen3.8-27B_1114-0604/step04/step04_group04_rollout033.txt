def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def self_conv(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    exp_sum = float(exp_z.sum())
    if exp_sum < 1e-300:
        L_val = c_max
        p = np.ones(len(c)) / len(c)
    else:
        log_sum = z.max() + np.log(exp_sum)
        L_val = c_max + log_sum / beta
        p = exp_z / exp_sum
    score = 2.0 * n * L_val / (S * S)
    # Gradient of L w.r.t. a: grad_L[j] = sum_k p[k] * (a[k-j] + a[j-k])
    # Using FFT cross-correlation
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    A = np.fft.rfft(a, M2_fft)
    # cross_corr(p, a)[j] = sum_k p[k]*a[k-j] = ifft(FFT(p)*conj(FFT(a)))[j]
    xcorr1 = np.fft.irfft(PA * np.conj(A), M2_fft)[:2 * n - 1]
    # Second term: sum_k p[k]*a[j-k] = cross_corr(p, a_rev) reversed
    A_rev = np.fft.rfft(a[::-1], M2_fft)
    xcorr2 = np.fft.irfft(PA * np.conj(A_rev), M2_fft)[:2 * n - 1]
    grad_L = xcorr1 + xcorr2[::-1]
    grad_L = grad_L[:n] if len(grad_L) >= n else np.pad(grad_L, (0, n - len(grad_L)))
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):

def error_reduction(a, n_iters=200, alpha=0.7):

        if conv_max - conv_mean < 1e-15:

def nesterov_subgradient(a, n_iters=5000, start_lr=None):
    """Nesterov-accelerated projected subgradient on simplex."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:

        lr = start_lr / (np.sqrt(it + 1) + 1.0)

        c = self_conv(y)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        # Subgradient of h_k(a) = sum_i a[i]*a[k-i] at position j:
        # grad[j] = a[k-j] + a[j-k] (out of range = 0)
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= k_star - j <= n - 1:
                val += y[k_star - j]
            if 0 <= j - k_star <= n - 1:
                val += y[j - k_star]
            grad[j] = val
        grad *= 2.0  # gradient of the quadratic form

        # Update y (Nesterov momentum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        x_new = y - lr * grad
        # Project onto simplex {a >= 0, sum(a) = 1}
        x_new = project_simplex(x_new)
        y_new = x_new + (t - 1.0) / t_new * (x_new - y)
        y_new = project_simplex(y_new)

        t = t_new
        y = y_new

        sc = compute_score(y)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

    return best_a


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u * np.arange(1, n + 1) > cssv)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def build_freq_shaped_init(n, rng):
    """Build initialization from a shaped frequency spectrum."""
    # Create a spectrum with controlled magnitude profile
    M = 1 << (2 * n - 1).bit_length()
    freq = np.fft.rfftfreq(M) * 2  # Normalize to [0, 1]

    # Try different spectral shapes
    choice = rng.integers(5)
    if choice == 0:
        # Low-pass with roll-off
        mag = 1.0 / (1.0 + (freq * 5.0) ** 2)
    elif choice == 1:
        # Band-pass
        center = 0.1 + 0.3 * rng.random()
        width = 0.05 + 0.2 * rng.random()
        mag = np.exp(-((freq - center) ** 2) / (2 * width ** 2))
    elif choice == 2:
        # Multi-lobe
        mag = 1.0
        for _ in range(3):
            c = rng.random()
            w = 0.05 + 0.1 * rng.random()
            mag += np.exp(-((freq - c) ** 2) / (2 * w ** 2))
    elif choice == 3:
        # Cosine window in frequency
        mag = 0.5 + 0.5 * np.cos(np.pi * freq / 0.5)
        mag[freq > 0.5] = 0
    else:
        # Random smooth magnitude
        mag = np.exp(-freq * rng.uniform(1, 5))

    phase = rng.standard_normal(len(mag)) * np.pi
    spectrum = mag * np.exp(1j * phase)
    spectrum[0] = abs(spectrum[0])  # Keep DC positive
    a_full = np.fft.irfft(spectrum, M)
    a = a_full[:n]
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) * 0.5
    else:
        a = a / S
    return a

def targeted_peak_reduction(a, n_iters=200):

                a[idx] += per_inc

        a = np.clip(a, 0.0, 1000.0)

            break
        a = a / S

        sc = compute_score(a)

    # Phase 2: Adaptive L-BFGS (45% time)
    # ==================================================================
    phase2_deadline = t_start + 0.53 * total_time
    print(f"\n--- Phase 2: Adaptive L-BFGS (t_limit={phase2_deadline - t_start:.0f}s) ---")

    # Run on the best sequence so far
    n_main = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    sc, a_opt = adaptive_lbfgs(a_work, phase2_deadline - 10, max_total_iters=60000)
    report(sc, a_opt, f"adaptLBFGS n={n_main}")

    # Also try on parent if different length
    if parent is not None and len(parent) != n_main:
        if time.monotonic() < phase2_deadline - 120:
            n_p = len(parent)
            a_p = parent.astype(np.float64).copy()
            S_p = a_p.sum()
            if S_p > 0:
                a_p = a_p / S_p
            sc, a_opt = adaptive_lbfgs(a_p, phase2_deadline - 30, max_total_iters=30000)
            report(sc, a_opt, f"adaptLBFGS parent n={n_p}")

    print(f"After Phase 2: best = {best_score:.8f} (n={len(best_a)})")

    # ==================================================================
    # Phase 3: LP-alternating (30% time)
    # ==================================================================
    phase3_deadline = t_start + 0.83 * total_time
    print(f"\n--- Phase 3: LP-Alternating (t_limit={phase3_deadline - t_start:.0f}s) ---")

    n_best = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Multiple passes with decreasing step
    for pass_idx in range(5):
        if time.monotonic() > phase3_deadline - 30:
            break
        fista_it = 5000 + pass_idx * 1000
        t_s = max(0.05, 0.4 * (0.7 ** pass_idx))
        t_m = max(0.01, 0.05 * (0.7 ** pass_idx))

        remaining = phase3_deadline - time.monotonic()
        max_outer = max(5, int(remaining / (fista_it * n_best / 400000.0 + 0.5)))
        max_outer = min(max_outer, 80)
        print(f"  Pass {pass_idx}: fista={fista_it}, t=[{t_s:.3f},{t_m:.3f}], outer~{max_outer}")

        a_out, sc_out = lp_alternating(
            a_work, max_outer_iters=max_outer, fista_iters=fista_it,
            t_start=t_s, t_min=t_m, time_limit=phase3_deadline - 15
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP pass{pass_idx}")
        a_work = a_out.copy()

    print(f"After Phase 3: best = {best_score:.8f} (n={len(best_a)})")

    # ==================================================================
    # Phase 4: Targeted peak reduction + local search (12% time)
    # ==================================================================
    phase4_deadline = t_start + 0.95 * total_time
    print(f"\n--- Phase 4: Targeted Reduction + Local Search ---")

    n_best = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Targeted peak reduction
    a_red = targeted_peak_reduction(a_work, n_iters=300)
    sc = compute_score(a_red)
    report(sc, a_red, "peak_red")

    # Then L-BFGS polish on the result
    if time.monotonic() < phase4_deadline - 60:
        sc, a_opt, _ = run_lbfgs(a_red, 500, maxiter=8000, maxfun=150000)
        report(sc, a_opt, "LBFGS after peak_red")

    # Diversified local search
    for trial in range(6):
        if time.monotonic() > phase4_deadline - 30:
            break
        strategy = trial % 3
        a_base = best_a.copy()
        S_b = a_base.sum()
        if S_b > 0:
            a_base = a_base / S_b

        if strategy == 0:
            # Gaussian perturbation
            scale = 0.005 + 0.02 * rng.random()
            a_pert = a_base * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Frequency-domain perturbation
            M_fft = 1 << (2 * n_best - 1).bit_length()
            A = np.fft.rfft(a_base, M_fft)
            noise = rng.standard_normal(len(A)) * 0.01
            A += noise * np.abs(A)
            a_pert = np.fft.irfft(A, M_fft)[:n_best]
            a_pert = np.maximum(a_pert, 0.0)
        else:
            # Error reduction restart
            a_pert = error_reduction(
                a_base * (1.0 + 0.02 * rng.standard_normal(n_best)),
                n_iters=60, alpha=0.8)
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        S_p = float(a_pert.sum())
        if S_p > 1e-12:
            a_pert = a_pert / S_p
            sc, a_opt, _ = run_lbfgs(a_pert, 300, maxiter=5000, maxfun=100000)
            report(sc, a_opt, f"local #{trial}")

    print(f"After Phase 4: best = {best_score:.8f} (n={len(best_a)})")

    # ==================================================================
    # Phase 5: Final polish (5% time)
    # ==================================================================
    print(f"\n--- Phase 5: Final Polish ---")

    a_final = best_a.copy()
    S = float(a_final.sum())
    if S > 0:
        a_final = a_final / S

    # Short LP-alternating
    if time.monotonic() < deadline - 40:
        a_out, sc_out = lp_alternating(
            a_final, max_outer_iters=15, fista_iters=6000,
            t_start=0.1, t_min=0.02, time_limit=deadline - 20
        )
        sc = compute_score(a_out)
        report(sc, a_out, "final LP")

    # Very high beta L-BFGS
    if time.monotonic() < deadline - 15:
        a_final = best_a.copy()
        S = float(a_final.sum())
        if S > 0:
            a_final = a_final / S
        sc, a_opt, _ = run_lbfgs(a_final, 5000, maxiter=5000, maxfun=80000)
        report(sc, a_opt, "final LBFGS high-beta")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
