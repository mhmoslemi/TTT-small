"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses Adam with LSE smoothing (fixed gradient), multiple restarts,
adaptive beta schedule, L-BFGS polish, and coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and exact gradient.
    Score = 2n * LSE_beta(c) / S^2 where LSE approximates max(c).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient of L w.r.t. a[i] = (1/beta) * sum_k p[k] * dc[k]/da[i]
    # dc[k]/da[i] = a[k-i] (if valid) + a[i-k] (if valid)
    # 
    # g1[i] = sum_k p[k]*a[k-i] = (reversed_a * p)[n-1+i]
    # g2[i] = sum_k p[k]*a[i-k] = (a * p)[i]
    
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    
    A_fft = np.fft.rfft(a, M2_fft)
    RA_fft = np.fft.rfft(a[::-1], M2_fft)
    P_fft = np.fft.rfft(p, M2_fft)
    
    # g2 = (a * p)[0:n]
    conv_ap = np.fft.irfft(A_fft * P_fft, M2_fft)
    g2 = conv_ap[:n]
    
    # g1 = (reversed_a * p)[n-1 : 2n-1]
    conv_ra_p = np.fft.irfft(RA_fft * P_fft, M2_fft)
    g1 = conv_ra_p[n-1:2*n-1]
    
    dL_da = (g1 + g2) / beta
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    
    return score, grad


def subgrad_score(a):
    """Exact score and subgradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    
    # Subgradient at the max: dc[k*]/da[i] = a[k*-i] + a[i-k*]
    sg = np.zeros(n)
    # a[k*-i] is valid when 0 <= k*-i <= n-1, i.e., k*-(n-1) <= i <= k*
    # a[i-k*] is valid when 0 <= i-k* <= n-1, i.e., k* <= i <= k*+(n-1)
    
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] += a[idx]
    
    js2 = max(k_star, 0)
    je2 = min(k_star + n - 1, n - 1)
    if je2 >= js2:
        idx2 = np.arange(js2, je2 + 1) - k_star
        sg[js2:je2 + 1] += a[idx2]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def adam_optimize(a_init, time_budget, t_start, beta_schedule=None):
    """
    Adam optimizer with LSE smoothing and adaptive beta.
    Normalizes a to sum=1 at each step.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S  # Normalize to sum=1
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t = 0
    while True:
        elapsed = time.monotonic() - t_start
        if elapsed > time_budget:
            break
        
        t += 1
        # Adaptive beta: start low, increase over time
        frac = min(1.0, elapsed / time_budget)
        beta = 20.0 + 500.0 * frac * frac  # quadratic ramp from 20 to 520
        
        lr_t = 0.01 / (1.0 + 1e-4 * t)
        
        score, grad = lse_score_grad(a, beta)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        S = a.sum()
        if S > 1e-12:
            a = a / S
        
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"    Adam t={t}, beta={beta:.0f}, score={sc:.8f}")
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


def lbfgs_polish(a0, beta, maxiter=3000, maxfun=50000, t_start=None, time_budget=None):
    """L-BFGS-B polish with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
        S = 1.0
    a0 = a0 / S  # work with sum=1
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, None)
        S_x = x.sum()
        if S_x < 1e-12:
            return 1e10, np.zeros_like(x)
        return lse_score_grad(x, beta)
    
    bounds = [(0.0, None)] * n
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, None)
    S_out = a_out.sum()
    if S_out > 1e-12:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coord_descent(a, n_sweeps=5, t_start=None, time_budget=None):
    """
    Coordinate descent using exact score.
    For each coordinate, find optimal value via golden section search.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    
    deadline = t_start + time_budget if (time_budget is not None and t_start is not None) else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        score, grad = subgrad_score(a)
        # Process coordinates in order of |gradient| (frenzy)
        order = np.argsort(np.abs(grad))[::-1]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            
            a_old = a[j]
            # Try a range of candidate values
            # Based on gradient direction
            if grad[j] > 1e-10:
                # Want to decrease a[j]
                ratios = [0.0, 0.2, 0.4, 0.6, 0.8, 0.95, 1.0]
            elif grad[j] < -1e-10:
                # Want to increase a[j]
                ratios = [1.0, 1.1, 1.3, 1.5, 2.0, 3.0, 5.0]
            else:
                continue
            
            best_x = a_old
            best_sc = score
            for r in ratios:
                x = a_old * r if a_old > 0 else (1.0/n if r > 0 else 0.0)
                x = min(x, 1000.0)
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
        
        if time.monotonic() > deadline:
            break
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    
    return best_score, a
